Let me think about the line search more carefully:

For direction v (zero-sum), the feasible interval for alpha is [lo, hi] where:
- For v[i] > 0: lo = max(lo, -h[i]/v[i]), hi = min(hi, (1-h[i])/v[i])
- For v[i] < 0: lo = max(lo, (1-h[i])/v[i]), hi = min(hi, -h[i]/v[i])

Along this line, for each lag s:
- a_s = q_s(h) (current value)
- b_s = dx * sum_j [v[p+j]*(1-h[r+j]) - h[p+j]*v[r+j]]
- c_s = -dx * sum_j v[p+j]*v[r+j]

So q_s(h + alpha*v) = a_s + b_s*alpha + c_s*alpha^2

For exact line search (n ≤ 64):
- Collect all critical points: endpoints, zero, stationary points, pairwise intersections
- Evaluate F(alpha) = max_s(a_s + b_s*alpha + c_s*alpha^2) at each
- Pick the minimum

For the pairwise intersections of quadratic s and t:
(a_s - a_t) + (b_s - b_t)*alpha + (c_s - c_t)*alpha^2 = 0

This is a quadratic in alpha. Solve and keep roots in [lo, hi].

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None

    s = n / 2.0

def box_dir(h, n, rng):

        r_sel = recvs[:nr + len(overlap)]

            return None

    tr = np.sum(w_r)

def quad_roots(a, b, c, lo, hi):
    # solve a*x^2 + b*x + c = 0, return roots in [lo,hi]
    roots = []
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return roots
        x = -c / b
        if lo - tol <= x <= hi + tol:
            roots.append(max(lo, min(hi, x)))
        return roots
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return roots
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    q_val = -0.5 * (b + (1.0 if b >= 0 else -1.0) * sq)
    r1 = q_val / a
    r2 = c / q_val if abs(q_val) > 1e-30 else 0.0
    for x in (r1, r2):
        if math.isfinite(x) and lo - tol <= x <= hi + tol:
            roots.append(max(lo, min(hi, x)))
    return roots

def line_crit_alpha(h, v, n, lo, hi):
    # return list of critical alpha values
    dx = 2.0 / n
    L = 2 * n - 1
    q0 = lag_values(h, n)
    A = np.zeros(L)
    B = np.zeros(L)
    C = np.zeros(L)
    for s in range(-(n - 1), n):
        row = s + n - 1
        p0 = max(s, 0)
        r0 = max(-s, 0)
        m = n - abs(s)
        A[row] = q0[row]
        B[row] = dx * (np.dot(v[p0:p0+m], 1.0 - h[r0:r0+m]) - np.dot(h[p0:p0+m], v[r0:r0+m]))
        C[row] = -dx * np.dot(v[p0:p0+m], v[r0:r0+m])

        if abs(C[s]) > 1e-15:
            x = -B[s] / (2.0 * C[s])
            if lo <= x <= hi and math.isfinite(x):
                alphas.append(x)
    # pairwise intersections
    for i in range(L):
        for j in range(i + 1, L):
            da = A[i] - A[j]
            db = B[i] - B[j]
            dc = C[i] - C[j]
            roots = quad_roots(dc, db, da, lo, hi)
            alphas.extend(roots)
    # deduplicate
    alphas = sorted(set(round(a, 14) for a in alphas))
    # bucket dedup
    merged = [alphas[0]]
    for a in alphas[1:]:
        if a - merged[-1] > 1e-12:
            merged.append(a)
    return merged, A, B, C, False

def eval_envelope(A, B, C, alphas):

    results = np.zeros(n_al)

        results[st:ed] = np.max(vals, axis=1)
    return results

    return float(np.max(q))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def dct_dir(k, n):
    i = np.arange(n, dtype=np.float64)
    v = np.cos(np.pi * k * (i + 0.5) / n)
    v -= np.mean(v)
    v[-1] -= np.sum(v)
    mx = np.max(np.abs(v))
    if mx < 1e-15:
        return None
    return v / mx

def box_dir(h, n, rng):
    don = np.where(h > 1e-8)[0]
    rec = np.where(h < 1.0 - 1e-8)[0]
    if len(don) < 2 or len(rec) < 2:
        return None
    rng.shuffle(don)
    rng.shuffle(rec)
    nd = min(len(don), 5)
    nr = min(len(rec), 5)
    dsel = don[:nd]
    rset = set(dsel)
    rsel = [x for x in rec if x not in rset][:nr]
    if len(rsel) < 1:
        return None
    v = np.zeros(n)
    wd = rng.uniform(0.3, 1.0, len(dsel))
    wr = rng.uniform(0.3, 1.0, len(rsel))
    td = np.sum(wd)
    tr = np.sum(wr)
    for i in range(len(dsel)):
        v[dsel[i]] = -wd[i] / td
    for i in range(len(rsel)):
        v[rsel[i]] = wr[i] / tr
    mx = np.max(np.abs(v))
    if mx < 1e-15:
        return None
    return v / mx

def quad_roots(a, b, c, lo, hi):
    roots = []
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return roots
        x = -c / b
        if lo - tol <= x <= hi + tol:
            roots.append(max(lo, min(hi, x)))
        return roots
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return roots
    if disc < 0.0:
        disc = 0.0
    sq = math.sqrt(disc)
    qv = -0.5 * (b + (sq if b >= 0 else -sq))
    r1 = qv / a
    r2 = c / qv if abs(qv) > 1e-30 else 0.0
    for x in (r1, r2):
        if math.isfinite(x) and lo - tol <= x <= hi + tol:
            roots.append(max(lo, min(hi, x)))
    return roots

def line_coeffs(h, v, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q0 = lag_values(h, n)
    A = np.zeros(L)
    B = np.zeros(L)
    C = np.zeros(L)
    for s in range(-(n - 1), n):
        row = s + n - 1
        p0 = max(s, 0)
        r0 = max(-s, 0)
        m = n - abs(s)
        A[row] = q0[row]
        B[row] = dx * (np.dot(v[p0:p0+m], 1.0 - h[r0:r0+m]) - np.dot(h[p0:p0+m], v[r0:r0+m]))
        C[row] = -dx * np.dot(v[p0:p0+m], v[r0:r0+m])
    ok = np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))
    return A, B, C, ok

def eval_env(A, B, C, alphas):
    na = len(alphas)
    res = np.zeros(na)
    ch = 512
    for st in range(0, na, ch):
        ed = min(st + ch, na)
        aa = np.array(alphas[st:ed])
        vv = A[np.newaxis, :] + aa[:, np.newaxis] * B[np.newaxis, :] + (aa[:, np.newaxis] ** 2) * C[np.newaxis, :]
        res[st:ed] = np.max(vv, axis=1)
    return res

def feasible_interval(h, v, n):
    lo = -1e10
    hi = 1e10
    for i in range(n):
        if v[i] > 1e-15:
            lo = max(lo, -h[i] / v[i])
            hi = min(hi, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / v[i])
            hi = min(hi, -h[i] / v[i])
    lo = max(lo, -10.0)
    hi = min(hi, 10.0)
    if lo >= hi - 1e-12:
        return None
    return lo, hi

def one_line_exact(h, v, n, lo, hi):
    A, B, C, ok = line_coeffs(h, v, n)
    if not ok:
        return None, float('inf')
    alphas = [lo, hi, 0.0]
    alphas_arr = np.array(alphas)
    for s in range(len(A)):
        if abs(C[s]) > 1e-15:
            x = -B[s] / (2.0 * C[s])
            if lo <= x <= hi and math.isfinite(x):
                alphas.append(x)
    for i in range(len(A)):
        for j in range(i + 1, len(A)):
            rr = quad_roots(C[i]-C[j], B[i]-B[j], A[i]-A[j], lo, hi)
            alphas.extend(rr)
    alphas = sorted(set(round(x, 14) for x in alphas))
    if len(alphas) == 0:
        return None, float('inf')
    merged = [alphas[0]]
    for x in alphas[1:]:
        if x - merged[-1] > 1e-12:
            merged.append(x)
    results = eval_env(A, B, C, merged)
    idx = np.argsort(results)[:min(8, len(results))]
    best_a = merged[int(idx[0])]
    if abs(best_a) < 1e-14:
        return None, float('inf')
    ht = project_capped(h + best_a * v, n)
    if ht is None:
        return None, float('inf')
    return ht, exact_score(ht, n)

def one_line_grid(h, v, n, lo, hi):
    A, B, C, ok = line_coeffs(h, v, n)
    if not ok:
        return None, float('inf')
    np_pts = 33 if n <= 256 else 17
    merged = list(np.linspace(lo, hi, np_pts))
    merged.append(0.0)
    merged = sorted(set(round(x, 14) for x in merged))
    results = eval_env(A, B, C, merged)
    n_ref = 4 if n <= 256 else 3
    n_sub = 9 if n <= 256 else 7
    for _ in range(n_ref):
        ib = int(np.argmin(results))
        kl = merged[max(0, ib - 1)]
        kh = merged[min(len(merged) - 1, ib + 1)]
        if kh - kl < 1e-12:
            break
        ref = list(np.linspace(kl, kh, n_sub))
        rres = eval_env(A, B, C, ref)
        merged = merged[:ib] + ref + merged[ib + 1:]
        merged = sorted(set(round(x, 14) for x in merged))
        results = eval_env(A, B, C, merged)
    idx = np.argsort(results)[:min(4, len(results))]
    best_a = merged[int(idx[0])]
    if abs(best_a) < 1e-14:
        return None, float('inf')
    ht = project_capped(h + best_a * v, n)
    if ht is None:
        return None, float('inf')
    return ht, exact_score(ht, n)

def one_line(h, v, n):
    fi = feasible_interval(h, v, n)
    if fi is None:
        return None, float('inf')
    lo, hi = fi
    if n <= 64:
        return one_line_exact(h, v, n, lo, hi)
    return one_line_grid(h, v, n, lo, hi)

def get_freqs(n, rng):
    if n <= 48:
        fr = list(range(1, n))
    elif n <= 64:
        fr = sorted(set(int(x) for x in np.geomspace(1, n-1, 24)))
    elif n <= 256:
        fr = sorted(set(int(x) for x in np.geomspace(1, n-1, 12)))
    else:
        fr = sorted(set(int(x) for x in np.geomspace(1, n-1, 8)))
    rng.shuffle(fr)
    return fr

def one_sweep(h, n, rng, nbx):
    cur = h.copy()
    cs = exact_score(cur, n)
    for k in get_freqs(n, rng):
        v = dct_dir(k, n)
        if v is None:
            continue
        ht, st = one_line(cur, v, n)
        if ht is not None and st < cs - 1e-12:
            cur, cs = ht, st
    for _ in range(nbx):
        v = box_dir(cur, n, rng)
        if v is None:
            continue
        ht, st = one_line(cur, v, n)
        if ht is not None and st < cs - 1e-12:
            cur, cs = ht, st
    return cur, cs

def run(seed=42, **kwargs):
    n = 47
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:

                parent = p.copy()
    except Exception:
        pass
    if 'n_points' in kwargs:
        try:
            vn = kwargs['n_points']
            if not isinstance(vn, bool):
                vn = int(vn)
                if 1 <= vn <= 999:
                    n = vn
        except Exception:
            pass
    if n < 1 or n > 999:
        n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb = None
    if parent is not None:
        if parent.shape[0] == n:
            fb = project_capped(parent, n)
        else:
            op = (np.arange(parent.shape[0]) + 0.5) / parent.shape[0]
            np_ = (np.arange(n) + 0.5) / n
            fb = project_capped(np.interp(np_, op, parent), n)
    if fb is None:
        fb = np.full(n, 0.5)
    if n == 1:
        return fb, exact_score(fb, 1), 1
    starts = [fb.copy()]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(project_capped(blk, n))
    if not np.allclose(fb, 0.5):
        starts.append(project_capped(fb + rng.normal(0, 0.06, n), n))
    starts.append(project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * (np.arange(n) + 0.5) / n), n))
    ms = 3 if n <= 256 else 2
    fs_list = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for f in fs_list:
            if np.max(np.abs(s - f)) < 1e-8:
                dup = True
                break
        if not dup:
            fs_list.append(s)
        if len(fs_list) >= ms:
            break
    gb = fb.copy()
    gs = exact_score(fb, n)
    for s in fs_list:
        ss = exact_score(s, n)
        if ss < gs - 1e-12:
            gs, gb = ss, s.copy()
    nbx = 2 if n <= 256 else 1
    for st in fs_list:
        try:
            cur, cs = one_sweep(st, n, rng, nbx)
            if cs < gs - 1e-12:
                gs, gb = cs, cur.copy()
        except Exception:
            continue
    npol = 2 if n <= 64 else 1
    for _ in range(npol):
        try:
            cur, cs = one_sweep(gb, n, rng, nbx)
            if cs >= gs - 1e-12:
                break
            gs, gb = cs, cur.copy()
        except Exception:
            break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss_fb = exact_score(fb, n)
    if sf > ss_fb:
        final, sf = fb, ss_fb
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
