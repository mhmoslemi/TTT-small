def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return False
    return True

def exact_score(h, n):

def propose_exchange(u, n, qmax, rng):
    # Choose k from {2, 4, 8}
    sizes = [2, 4, 8]
    sizes = [s for s in sizes if s <= n]
    weights = [0.55, 0.30, 0.15]
    weights = [w for s, w in zip(sizes, weights)]
    # Renormalize
    total = sum(weights)
    weights = [w / total for w in weights]
    k = rng.choice(len(sizes), p=weights)
    k = sizes[k]
    # Choose k distinct indices
    idx = rng.choice(n, size=k, replace=False)
    # Choose k/2 to receive +q
    half = k // 2
    signs_idx = rng.choice(k, size=half, replace=False)
    signs = np.full(k, -1)
    signs[signs_idx] = 1
    q = rng.integers(1, qmax + 1)
    v = u.copy()
    for j in range(k):
        v[idx[j]] += signs[j] * q
    if np.any(v < 0) or np.any(v > M):
        return None
    if int(np.sum(v)) != M * n // 2:
        return None
    return v

def anneal_one(start, n, rng, steps_per_level):
    dx = 2.0 / n
    current = start.copy()
    current_h = units_to_h(current)
    current_score = exact_score(current_h, n)
    if current_score == float('inf'):
        return start, float('inf')
    best = current.copy()
    best_score = current_score
    
    # Pilot: calibrate temperature
    pilot_qmax = max(1, round(0.08 * M))
    deltas = []
    for _ in range(24):
        prop = propose_exchange(start, n, pilot_qmax, rng)
        if prop is None:
            continue
        ph = units_to_h(prop)
        ps = exact_score(ph, n)
        if ps != float('inf'):
            if ps < best_score - 1e-12:
                best_score = ps
                best = prop.copy()
            d = ps - current_score
            if d > 0:
                deltas.append(d)
    
    if deltas:
        T0 = float(np.median(deltas)) / math.log(2.0)
    else:
        T0 = 0.04 * dx
    T0 = max(0.002 * dx, min(0.25 * dx, T0))
    
    # Six stages
    fracs = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025]
    for ell in range(6):
        qmax = max(1, round(M * fracs[ell]))
        T = T0 * (0.4 ** ell)
        for _ in range(steps_per_level):
            prop = propose_exchange(current, n, qmax, rng)
            if prop is None:
                continue
            ph = units_to_h(prop)
            ps = exact_score(ph, n)
            if ps == float('inf'):
                continue
            if ps < best_score - 1e-12:
                best_score = ps
                best = prop.copy()
            delta = ps - current_score
            if delta <= 0:
                current = prop
                current_score = ps
            else:
                ratio = delta / T
                if ratio < 745.0:
                    if rng.random() < math.exp(-ratio):
                        current = prop
                        current_score = ps
    return best, best_score

    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def to_units(h, n):
    if not is_feasible(h, n):
        return None
    K = M * n // 2
    z = np.clip(h, 0.0, 1.0) * M
    base = np.floor(z).astype(np.int64)
    base = np.clip(base, 0, M)
    frac = z - base.astype(np.float64)
    R = K - int(np.sum(base))
    if R > 0:
        order = np.argsort(-frac)
        cnt = 0
        for i in order:
            if cnt >= R:
                break
            if base[i] < M:
                base[i] += 1
                cnt += 1
        if cnt < R:
            return None
    elif R < 0:
        order = np.argsort(frac)
        cnt = 0
        for i in order:
            if cnt >= -R:
                break
            if base[i] > 0:
                base[i] -= 1
                cnt += 1
        if cnt < -R:
            return None
    if np.any(base < 0) or np.any(base > M) or int(np.sum(base)) != K:
        return None
    return base
