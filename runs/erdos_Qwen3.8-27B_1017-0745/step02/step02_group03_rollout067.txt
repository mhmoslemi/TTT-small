import numpy as np

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        if hi - lo < 1e-16:
            break
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    if abs(r) > 1e-14:
        if r > 0:
            cap = 1.0 - h
            tc = float(np.sum(cap))
            if tc > 1e-15:
                h = h + r * cap / tc
            else:
                j = int(np.argmax(1.0 - h))
                h[j] = min(1.0, h[j] + r)
        else:

            th = float(np.sum(h))

                h = h - r2 * h / th
            else:
                j = int(np.argmax(h))
                h[j] = max(0.0, h[j] - r2)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return None
    if abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

    L = 2 * n - 1
    v = np.empty(L)

        lo = -d if d < 0 else 0
        hi = n - d if d > 0 else n
        v[d + n - 1] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def lse_phi_grad(h, n, theta):
    v = lag_vals(h, n)
    m = float(np.max(v))
    a = np.clip((v - m) / theta, -700.0, 0.0)
    e = np.exp(a)
    Z = float(np.sum(e))
    if Z < 1e-300 or not np.isfinite(Z):
        return None, None
    p = e / Z
    phi = m + theta * math.log(Z)
    g = np.zeros(n)

        if pk < 1e-14:
            continue

        lo = -d if d < 0 else 0
        hi = n - d if d > 0 else n
        g[lo:hi] += pk * (1.0 - h[lo + d:hi + d])
        g[lo + d:hi + d] -= pk * h[lo:hi]
    return phi, g
