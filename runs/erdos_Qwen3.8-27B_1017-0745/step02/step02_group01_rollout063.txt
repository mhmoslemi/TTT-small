        bh, bs = cur.copy(), cs
    used, stag, nd = 0, 0, len(dirs)
    while used < budget:
        imp = False
        for k in range(nd):
            if used >= budget:
                break
            used += 1
            try:
                nh, ns = do_line_search(cur, cs, dirs[k], n)
            except Exception:
                continue
            if ns < cs - 1e-12:
                cur, cs = nh, ns
                imp = True
                if cs < bs - 1e-12:
                    bh, bs = cur.copy(), cs
        stag = 0 if imp else stag + 1
        if stag >= 2:
            break
    return bh, bs

def add_start(pool, v, n, lim):
    if len(pool) >= lim:
        return
    p = project_capped(v, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(p)

def build_starts(fb, n, rng, lim=10):
    pool = [fb.copy()]
    add_start(pool, fb[::-1].copy(), n, lim)
    add_start(pool, (1.0 - fb[::-1]).copy(), n, lim)
    add_start(pool, (1.0 - fb).copy(), n, lim)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_start(pool, blk, n, lim)
    add_start(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n, lim)
    t = np.linspace(0, 1, n)
    add_start(pool, 0.5 + 0.35 * np.cos(2.0 * math.pi * t), n, lim)
    add_start(pool, 0.5 + 0.35 * np.cos(4.0 * math.pi * t), n, lim)
    add_start(pool, fb + rng.normal(0, 0.05, n), n, lim)
    add_start(pool, fb + rng.normal(0, 0.12, n), n, lim)
    return pool
