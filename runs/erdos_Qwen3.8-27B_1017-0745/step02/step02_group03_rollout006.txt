def exact_score(h, n):

        return float('inf')

def hit_and_run(h, n, rng):

            hi_v = min(hi_v, max(r0, r1))

def pair_proposal(h, n, a, rng, full):
    p = int(rng.integers(n))
    q = int(rng.integers(n - 1))
    if q >= p:
        q += 1
    L = max(-h[p], h[q] - 1.0)
    U = min(1.0 - h[p], h[q])
    if U - L <= 1e-14:
        return None

        t = rng.uniform(L, U)
    else:
        t = rng.uniform(-a, a)
        if t < L or t > U:
            return None

    hp[q] = h[q] - t

        return None
    if abs(np.sum(hp) - np.sum(h)) > 1e-8 * max(1, n):
        return None
    return hp
