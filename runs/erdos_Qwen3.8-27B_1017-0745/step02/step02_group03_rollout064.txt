        if w < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += w * (1.0 - h[lo + d:hi + d])
        g[lo + d:hi + d] -= w * h[lo:hi]
    return g

def obj_smoothed(h, n, theta):
    F = lag_values(h, n)
    m = float(np.max(F))
    ex = np.exp(np.clip((F - m) / theta, -700.0, 700.0))
    return m + theta * math.log(float(np.sum(ex)))

def continuation_search(h0, n, iters, best_ref, best_sc):
    h = h0.copy()
    taus = [0.015, 0.005, 0.0015, 0.0004, 0.0001, 0.00003, 0.000008]
    for tau in taus:
        theta = (n / 2.0) * tau
        alpha = 0.2
        for it in range(iters):
            g = grad_smoothed(h, n, theta)
            y = project_capped(h - alpha * g, n)
            if y is None:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
                continue
            d = y - h
            if float(np.max(np.abs(d))) < 1e-11:
                break
            gd = float(np.dot(g, d))
            if gd > -1e-14:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
                continue
            fy = obj_smoothed(y, n, theta)
            fh = obj_smoothed(h, n, theta)
            arm = 1e-4 * gd + 1e-13 * max(1.0, abs(fh))
            if fy <= fh + arm:
                h = y
                alpha = min(0.5, 1.15 * alpha)
                sc = exact_score(h, n)
                if np.isfinite(sc) and sc < best_sc[0] - 1e-12:
                    best_sc[0] = sc
                    best_ref[0] = h.copy()
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return h, best_ref, best_sc

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    if sd < 0:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = uniform.copy()
    else:
        fb = uniform.copy()
    fb_sc = exact_score(fb, n)
    if not np.isfinite(fb_sc):
        fb = uniform.copy()
        fb_sc = exact_score(fb, n)
    best = fb.copy()
    best_sc = fb_sc
    starts = [fb]
    if parent is not None:
        props = [parent[::-1], 1.0 - parent, np.roll(parent, n // 3),
                 parent + rng.normal(0, 0.03, n), parent + rng.normal(0, 0.10, n),
                 parent + rng.normal(0, 0.06, n)]
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        tt = np.arange(n)
        props = [uniform, blk, 0.5 + 0.45 * np.cos(2 * np.pi * tt / n),
                 0.5 + 0.45 * np.cos(4 * np.pi * tt / n), rng.uniform(0, 1, n)]
    for prop in props:
        hp = project_capped(prop, n)
        if hp is None:
            continue
        dup = False
        for ex in starts:
            if float(np.max(np.abs(hp - ex))) < 1e-8:
                dup = True
                break
        if not dup:
            starts.append(hp)
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = st.copy()
    if n <= 128:
        n_st, iters = min(len(starts), 6), 80
    elif n <= 512:
        n_st, iters = min(len(starts), 4), 30
    else:
        n_st, iters = min(len(starts), 2), 12
    for st in starts[:n_st]:
        br, bs = [best.copy()], [best_sc]
        _, br, bs = continuation_search(st, n, iters, br, bs)
        best_sc, best = bs[0], br[0]
    hops = [0.02, 0.05, 0.09] if n <= 128 else [0.04]
    for sds in hops:
        pert = project_capped(best + rng.normal(0, sds, n), n)
        if pert is None:
            continue
        br, bs = [best.copy()], [best_sc]
        _, br, bs = continuation_search(pert, n, iters, br, bs)
        best_sc, best = bs[0], br[0]
    # Final polish with low temperature only
    br, bs = [best.copy()], [best_sc]
    h_pol = best.copy()
    theta_fin = (n / 2.0) * 0.00003
    alpha = 0.1
    for it in range(40):
        g = grad_smoothed(h_pol, n, theta_fin)
        y = project_capped(h_pol - alpha * g, n)
        if y is None:
            alpha *= 0.5
            continue
        dd = y - h_pol
        if float(np.max(np.abs(dd))) < 1e-11:
            break
        gd = float(np.dot(g, dd))
        if gd > -1e-14:
            alpha *= 0.5
            continue
        fy = obj_smoothed(y, n, theta_fin)
        fh = obj_smoothed(h_pol, n, theta_fin)
        if fy <= fh + 1e-4 * gd + 1e-13:
            h_pol = y
            alpha = min(0.4, 1.1 * alpha)
            sc = exact_score(h_pol, n)
            if np.isfinite(sc) and sc < bs[0] - 1e-12:
                bs[0] = sc
                br[0] = h_pol.copy()
        else:
            alpha *= 0.5
    if bs[0] < best_sc - 1e-12:
        best_sc, best = bs[0], br[0]
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_sc:
        final, fs = fb.copy(), fb_sc
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
