            best_h = starts[i].copy()
    indexed = []
    for i in range(len(starts)):
        indexed.append((scores[i], i))
    indexed.sort()
    order = []
    for item in indexed:
        order.append(item[1])
    n_k = max(1, min(4, int(round(4.0 * math.sqrt(47.0 / n)))))
    n_chains = min(len(starts), n_k)
    selected = order[:n_chains]
    if 0 not in selected:
        selected[0] = 0
    max_iter = max(8, min(35, int(round(30.0 * math.sqrt(47.0 / n)))))
    for idx in selected:
        try:
            best_h, best_score = refine_chain(starts[idx], n, max_iter, 0.08, best_h, best_score)
        except Exception:
            pass
    try:
        best_h, best_score = refine_chain(best_h, n, max(6, max_iter // 2), 0.04, best_h, best_score)
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb_score = exact_score(fallback, n)
    if fs > fb_score + 1e-12 or not np.isfinite(fs):
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
