import numpy as np
import math
import cvxpy as cp

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n/2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    if len(c) != 2*n-1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0-h, mode='full') * dx
    if len(L) != 2*n-1 or not np.all(np.isfinite(L)):
        return None, None
    G = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(-(n-1), n):
        row = k + n - 1
        start = max(0, -k)
        stop = min(n, n-k)
        i = np.arange(start, stop)
        j = i + k
        G[row, j] += dx * (1.0 - h[i])
        G[row, i] -= dx * h[j]
    if not np.all(np.isfinite(G)):
        return None, None
    return L, G

def solve_mm_step(x, L, G, n, solvers):
    nlags = 2*n - 1
    y = cp.Variable(n)
    u = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == n/2.0]
    for k in range(nlags):
        cons.append(L[k] + G[k,:] @ (y - x) <= u)
    prob = cp.Problem(cp.Minimize(u), cons)
    for solver in solvers:
        try:
            kwargs = {}
            if solver in ('ECOS', 'CVXOPT'):
                kwargs = {'abstol': 1e-9, 'reltol': 1e-9, 'max_iters': 1000}
            elif solver == 'MOSEK':
                kwargs = {'MSK_DPAR_INTPNT_CO_TOL_REL_GAP': 1e-9, 'reductions': True}
            elif solver == 'GUROBI':
                kwargs = {'Threads': 1, 'BarIterLimit': 1000}
            prob.solve(solver=solver, warm_start=True, verbose=False, **kwargs)
        except Exception:
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
        if prob.status in ('optimal', 'optimal_inaccurate'):
            if y.value is not None and np.all(np.isfinite(y.value)):
                return np.asarray(y.value, dtype=np.float64)
    return None

def mm_trajectory(x0, n, max_iters, solvers):
    x = x0.copy()
    score = exact_score(x, n)
    if not np.isfinite(score):
        return x, score
    for it in range(max_iters):
        L, G = lag_data(x, n)
        if L is None:
            break
        y_new = solve_mm_step(x, L, G, n, solvers)
        if y_new is None:
            break
        best_x, best_s = x, score
        for alpha in [1.0, 0.5, 0.25, 0.125]:
            trial = x + alpha * (y_new - x)
            proj = project_capped(trial, n)
            if proj is None or not is_feasible(proj, n):
                continue
            s = exact_score(proj, n)
            if s < best_s - 1e-12:
                best_s, best_x = s, proj
        if best_s >= score - 1e-12:
            break
        x = best_x
        score = best_s
        if np.max(np.abs(y_new - x0)) < 1e-10:
            break
    return x, score

def build_starts(fallback, n, rng, count):
    pool = []
    def add(c):
        p = project_capped(c, n)
        if p is None:
            return
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-7:
                return
        pool.append(p)
    add(fallback)
    add(fallback.copy()[::-1])
    add(1.0 - fallback.copy()[::-1])
    add(1.0 - fallback.copy())
    add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n//2] = 1.0
    if n % 2:
        blk[n//2] = 0.5
    add(blk)
    add(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    for m in [1, 2, 3]:
        for ph in [0.0, math.pi/2]:
            add(np.clip(0.5 + 0.45*np.cos(2*math.pi*m*np.arange(n)/n + ph), 0, 1))
    for sd in [0.04, 0.12, 0.25]:
        add(fallback + rng.normal(0, sd, n))
    add(rng.uniform(0, 1, n))
    # Score and sort
    scored = []
    for p in pool:
        s = exact_score(p, n)
        scored.append((s, p))
    scored.sort(key=lambda x: x[0])
    return [p for _, p in scored[:count]]
