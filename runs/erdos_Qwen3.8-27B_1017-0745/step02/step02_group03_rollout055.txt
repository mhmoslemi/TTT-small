    z = np.clip(z.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = s - float(np.sum(h))
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - h
            tot = float(np.sum(cap))
            if tot > 1e-15:
                h += r * cap / tot
        else:
            tot = float(np.sum(h))
            if tot > 1e-15:
                h += r * h / tot
    h = np.clip(h, 0.0, 1.0)
    r2 = s - float(np.sum(h))
    if abs(r2) > 1e-14:
        if r2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(r2, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            h[idx] += max(r2, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals(h, n):
    L = 2 * n - 1
    vals = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        vals[row] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return vals

def smooth_obj(vals, theta):
    m = float(np.max(vals))
    if not np.isfinite(m):
        return float('inf'), None
    ex = np.exp(np.clip((vals - m) / theta, -700, 700))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        return float('inf'), None
    p = ex / s
    phi = m + theta * math.log(s)
    return phi, p

def smooth_grad(h, n, p):
    grad = np.zeros(n)
    L = 2 * n - 1
    for k in range(L):
        d = k - (n - 1)
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += pk * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= pk * h[lo:hi]
    return grad

def cont_opt(h0, n, mip):
    taus = [0.02, 0.006, 0.0018, 0.0005, 0.00015, 0.00004, 0.00001]
    h = h0.copy()
    for tau in taus:
        theta = (n / 2.0) * tau
        alpha = 0.2
        vals = lag_vals(h, n)
        phi, p = smooth_obj(vals, theta)
        if not np.isfinite(phi):
            phi, p = 0.0, np.ones(2 * n - 1) / (2 * n - 1)
        for _ in range(mip):
            g = smooth_grad(h, n, p)
            if not np.all(np.isfinite(g)):
                break
            y = project_capped(h - alpha * g, n)
            if y is None:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
                continue
            dvec = y - h
            if float(np.max(np.abs(dvec))) < 1e-10:
                break
            gd = float(np.dot(g, dvec))
            if gd >= -1e-14:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
                continue
            vy = lag_vals(y, n)
            py, pw = smooth_obj(vy, theta)
            if not np.isfinite(py):
                alpha *= 0.5
                if alpha < 1e-10:
                    break
                continue
            if py <= phi + 1e-4 * gd + 1e-12 * max(1.0, abs(phi)):
                h = y
                phi, p = py, pw
                alpha = min(0.5, 1.15 * alpha)
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    return h

def coord_polish(h, n, sweeps):
    best = h.copy()
    bs = exact_score(best, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [lo + 1e-12, hi - 1e-12]
                for f in (0.2, 0.5, 0.8):
                    cands.append(lo + f * (hi - lo))
                for st in (0.01, 0.05, 0.1, 0.2):
                    for sg in (1, -1):
                        dd = sg * st
                        if lo - 1e-14 <= dd <= hi + 1e-14:
                            cands.append(max(lo, min(hi, dd)))
                bd, bsc = 0.0, bs
                for dd in cands:
                    dd = max(lo, min(hi, dd))
                    tr = best.copy()
                    tr[i] += dd
                    tr[j] -= dd
                    sc = exact_score(tr, n)
                    if sc < bsc - 1e-13:
                        bd, bsc = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    bs = bsc
                    imp = True
        if not imp:
            break
    return best, bs

def add_start(starts, seen, vec, n):
    h2 = project_capped(vec, n)
    if h2 is None:
        return
    for ex in seen:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)
    seen.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    if sd < 0:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 128:
        ns, mip = 8, 55
    elif n <= 512:
        ns, mip = 4, 25
    else:
        ns, mip = 2, 10
    try:
        ns = max(1, min(int(kwargs.get('starts', ns)), 8))
    except Exception:
        pass
    try:
        mip = max(1, min(int(kwargs.get('max_iter', mip)), 120))
    except Exception:
        pass
    starts, seen = [], []
    add_start(starts, seen, fallback, n)
    if parent is not None:
        add_start(starts, seen, fallback[::-1].copy(), n)
        add_start(starts, seen, 1.0 - fallback, n)
        add_start(starts, seen, np.roll(fallback, max(1, n // 4)), n)
        add_start(starts, seen, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, seen, fallback + rng.normal(0, 0.10, n), n)
        tt = np.arange(n)
        add_start(starts, seen, fallback + 0.04 * np.sin(2.0 * np.pi * tt / n), n)
        add_start(starts, seen, fallback + 0.04 * np.cos(4.0 * np.pi * tt / n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, seen, blk, n)
        tt = np.arange(n)
        add_start(starts, seen, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, seen, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, seen, rng.uniform(0, 1, n), n)
    starts = starts[:ns]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    for st in starts:
        try:
            res = cont_opt(st, n, mip)
            rr = project_capped(res, n)
            if rr is not None:
                sc = exact_score(rr, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = rr.copy(), sc
        except Exception:
            continue
    nh = 3 if n <= 128 else 1
    for k in range(nh):
        std = [0.02, 0.06, 0.45][k] if k < 3 else 0.05
        pert = project_capped(best + rng.normal(0, std, n), n)
        if pert is None:
            continue
        try:
            res = cont_opt(pert, n, mip)
            rr = project_capped(res, n)
            if rr is not None:
                sc = exact_score(rr, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = rr.copy(), sc
        except Exception:
            continue
    if n <= 64:
        try:
            pol, ps = coord_polish(best, n, 2)
            fp = project_capped(pol, n)
            if fp is not None:
                fs = exact_score(fp, n)
                if np.isfinite(fs) and fs < best_score - 1e-12:
                    best, best_score = fp, fs
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
