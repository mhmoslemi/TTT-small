def pair_polish(h0, n, rng, sweeps, n_pairs):
    h = h0.copy()
    cur_s = exact_score(h, n)
    best_h = h0.copy()
    best_s = cur_s
    fractions = np.array([1.0/64, 1.0/16, 1.0/4, 1.0/2, 1.0])
    for sw in range(sweeps):
        improved = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            pairs = [(int(rng.integers(n)), int(rng.integers(n))) for _ in range(n_pairs)]
            pairs = [(a, b) for a, b in pairs if a != b]
        for i, j in pairs:
            if i == j:
                continue
            for direction in (0, 1):
                if direction == 0:
                    donor, recv = j, i
                else:
                    donor, recv = i, j
                c = min(1.0 - h[recv], h[donor])
                if c < 1e-12:
                    continue
                for fr in fractions:
                    t = c * fr
                    hp = h.copy()
                    hp[recv] += t
                    hp[donor] -= t
                    hp = project_capped(hp, n)
                    if hp is None:
                        continue
                    hs = exact_score(hp, n)
                    if np.isfinite(hs) and hs < cur_s - 1e-12:
                        h = hp
                        cur_s = hs
                        improved = True
                        if hs < best_s - 1e-12:
                            best_s = hs
                            best_h = hp.copy()
                        break
        if not improved:
            break
    return best_h, best_s

        q = gen / max(1, G - 1)
        T = T0 * (T1 / T0) ** q
        F = (0.5 + 0.4 * rng.uniform()) * (1 - 0.25 * q)
        CR = 0.85
        new_pop = [x.copy() for x in pop]
        new_scores = list(scores)
        for i in range(P):
            avail = list(range(i)) + list(range(i + 1, P))
            a, b, c = list(rng.choice(avail, size=3, replace=False))
            g = best_h
            if rng.uniform() < 0.3:
                v = pop[i] + F * (g - pop[i]) + F * (pop[b] - pop[c])
            else:
                v = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(n) < CR
            if not mask.any():
                mask[int(rng.integers(n))] = True
            raw = np.where(mask, v, pop[i])
            trial = project_capped(raw, n)
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if not np.isfinite(ts):
                continue
            delta = ts - scores[i]
            if delta < -1e-12:
                acc = True
            elif abs(delta) <= 1e-12:
                acc = rng.uniform() < 0.1
            else:
                acc = rng.uniform() < math.exp(-max(delta, 0) / max(T, 1e-15))
            if acc:
                new_pop[i] = trial
                new_scores[i] = ts
            if ts < best_s - 1e-12:
                best_s = ts
                best_h = trial.copy()
        if best_s < min(scores) - 1e-12:
            stall = 0
        else:
            stall += 1
        new_pop[0] = best_h.copy()
        new_scores[0] = best_s
        pop, scores = new_pop, new_scores
        if stall >= stall_lim and gen < G - 1:
            keep = max(2, P // 3)
            idxs = [0] + list(rng.choice(np.arange(1, P), size=min(keep - 1, P - 1), replace=False))
            pop = [best_h.copy()]
            scores = [best_s]
            for j in idxs:
                if j != 0:
                    pop.append(pop[j].copy() if j < len(pop) else best_h.copy())
                    scores.append(best_s)
            while len(pop) < P:
                if rng.uniform() < 0.5:
                    rc = project_capped(best_h + rng.normal(0, 0.02 + 0.08 * (1 - q), n), n)
                else:
                    rc = project_capped(rng.uniform(0, 1, n), n)
                if rc is None:
                    rc = best_h.copy()
                sc = exact_score(rc, n)
                pop.append(rc)
                scores.append(sc if np.isfinite(sc) else best_s)
                if scores[-1] < best_s - 1e-12:
                    best_s = scores[-1]
                    best_h = rc.copy()
            pop[0] = best_h.copy()
            scores[0] = best_s
            stall = 0
    return best_h, best_s

def pair_anneal(h0, n, rng, L, T0, T1):
    h = h0.copy()
    cur_s = exact_score(h, n)
    best_h, best_s = h0.copy(), cur_s
    for t in range(L):
        q = t / max(1, L - 1)
        T = T0 * (T1 / T0) ** q
        rho = 1.0 - 0.98 * q
        found = False
        i = j = 0
        c = 0.0
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            c = min(1.0 - h[i], h[j])
            if c > 1e-12:
                found = True
                break
        if not found:
            continue
        amt = c * rho * (0.05 + 0.95 * rng.uniform())
        hp = h.copy()
        hp[i] += amt
        hp[j] -= amt
        hp = project_capped(hp, n)
        if hp is None:
            continue
        hs = exact_score(hp, n)
        if not np.isfinite(hs):
            continue
        delta = hs - cur_s
        if delta < -1e-12:
            h, cur_s = hp, hs
            if hs < best_s - 1e-12:
                best_s, best_h = hs, hp.copy()
        elif rng.uniform() < math.exp(-max(delta, 0) / max(T, 1e-15)):
            h, cur_s = hp, hs
    return best_h, best_s

def pair_polish(h0, n, rng, sweeps, max_pairs):
    h = h0.copy()
    cur_s = exact_score(h, n)
    best_h, best_s = h0.copy(), cur_s
    fracs = np.array([1.0 / 64, 1.0 / 16, 1.0 / 4, 0.5, 1.0])
    for sw in range(sweeps):
        improved = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = []
            for _ in range(min(max_pairs, n * n)):
                a, b = int(rng.integers(n)), int(rng.integers(n))
                if a != b:
                    pairs.append((a, b))
        for i, j in pairs:
            for direction in (0, 1):
                recv, donor = (i, j) if direction == 0 else (j, i)
                c = min(1.0 - h[recv], h[donor])
                if c < 1e-12:
                    continue
                for fr in fracs:
                    t = c * fr
                    hp = h.copy()
                    hp[recv] += t
                    hp[donor] -= t
                    hp = project_capped(hp, n)
                    if hp is None:
                        continue
                    hs = exact_score(hp, n)
                    if np.isfinite(hs) and hs < cur_s - 1e-12:
                        h, cur_s = hp, hs
                        improved = True
                        if hs < best_s - 1e-12:
                            best_s, best_h = hs, hp.copy()
                        break
        if not improved:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(int(kwargs.get('n_points', 47)), 999))
        except Exception:
            n = 47
    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.25, 1
    rng = np.random.default_rng(int(seed))
    fb = np.full(n, 0.5)
    if parent is not None:
        fp = project_capped(parent, n)
        if fp is not None:
            fb = fp
    safe_h, safe_s = fb.copy(), exact_score(fb, n)
    if n <= 64:
        P, G, L, SL = 12, 120, 1600, 20
    elif n <= 256:
        P, G, L, SL = 8, 70, 500, 14
    else:
        P, G, L, SL = 6, 25, 80, 7
    pop = build_pop(fb, n, rng, P)
    scores = [exact_score(x, n) for x in pop]
    best_h, best_s = safe_h.copy(), safe_s
    for x, sc in zip(pop, scores):
        if sc < best_s - 1e-12:
            best_s, best_h = sc, x.copy()
    try:
        dh, ds = de_evolve(pop, scores, n, rng, G, P, SL)
        if ds < best_s - 1e-12:
            best_s, best_h = ds, dh
    except Exception:
        pass
    try:
        ah, as_ = pair_anneal(best_h, n, rng, L, 5e-4, 1e-7)
        if as_ < best_s - 1e-12:
            best_s, best_h = as_, ah
    except Exception:
        pass
    n_pairs = min(n * (n - 1) // 2, max(200, 4 * n * n))
    try:
        ph, ps = pair_polish(best_h, n, rng, 2, n_pairs)
        if ps < best_s - 1e-12:
            best_s, best_h = ps, ph
    except Exception:
        pass
    try:
        ah2, as2 = pair_anneal(best_h, n, rng, L // 2, 2e-4, 1e-8)
        if as2 < best_s - 1e-12:
            best_s, best_h = as2, ah2
    except Exception:
        pass
    try:
        ph2, ps2 = pair_polish(best_h, n, rng, 2, n_pairs)
        if ps2 < best_s - 1e-12:
            best_s, best_h = ps2, ph2
    except Exception:
        pass
    final = project_capped(best_h, n)
    fs = exact_score(final, n) if final is not None else float('inf')
    if final is None or not np.isfinite(fs) or fs > safe_s + 1e-12:
        final, fs = safe_h, safe_s
    return final.astype(np.float64), float(fs), int(n)
