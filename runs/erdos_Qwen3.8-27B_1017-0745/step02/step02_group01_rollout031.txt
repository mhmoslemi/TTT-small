        scores.append(exact_score(st, n))
    best_h = fallback.copy()
    best_score = exact_score(fallback, n)
    for i, st in enumerate(starts):
        if scores[i] < best_score - 1e-12:
            best_score = scores[i]
            best_h = st.copy()
    scale = math.sqrt(47.0 / max(n, 1))
    n_chains = min(len(starts), max(1, min(4, round(4.0 * scale))))
    iter_per = max(8, min(35, round(32.0 * scale)))
    order = sorted(range(len(starts)), key=lambda i: scores[i])
    if 0 not in order[:n_chains]:
        order = [0] + [x for x in order if x != 0]
    order = order[:n_chains]
    for oi in order:
        try:
            best_h, best_score = refine_chain(starts[oi], n, iter_per, 0.08, best_h, best_score)
        except Exception:
            pass
    try:
        n_polish = max(5, iter_per // 2)
        best_h, best_score = refine_chain(best_h, n, n_polish, 0.03, best_h, best_score)
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb = exact_score(fallback, n)
    if not np.isfinite(fs) or fs > fb + 1e-12:
        final, fs = fallback, fb
    return final, float(fs), int(n)
