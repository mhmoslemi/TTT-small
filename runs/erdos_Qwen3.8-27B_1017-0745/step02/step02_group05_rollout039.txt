def exact_score(h, n):

        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12):

    if not np.all(np.isfinite(corr)):

    return float(np.max(corr))

def helmert_fwd(x, n):
    d = n - 1
    if d <= 0:
        return np.zeros(n)
    u = np.array([x[k]/math.sqrt((k+1)*(k+2)) for k in range(d)])
    sfx = np.zeros(d+1)
    for k in range(d-1, -1, -1):
        sfx[k] = u[k] + sfx[k+1]
    y = np.zeros(n)
    y[0] = sfx[0]
    for i in range(1, n):
        y[i] = sfx[i] - (i/math.sqrt(i*(i+1))) * x[i-1]
    return y

def helmert_trans(y, n):
    d = n - 1
    if d <= 0:
        return np.zeros(0)
    x = np.zeros(d)
    prefix = 0.0
    for k in range(d):
        prefix += y[k]
        x[k] = (prefix - (k+1)*y[k+1]) / math.sqrt((k+1)*(k+2))
    return x

        return None
    return project_capped(0.5 + helmert_fwd(x, n), n)

            C = (evv*ev)@evv.T
            A = evv*np.sqrt(ev)

            Cn = (1 - c1 - cmu) * C + c1 * (np.outer(pcn, pcn) + (1 - hs) * cc * (2 - cc) * C) + cmu * rm
        else:
            Cn = np.zeros(d)
            for i in range(d):
                yi = (Xo[:, i] - mo[i]) / sigma
                Cn[i] = (1 - c1 - cmu) * C[i] + c1 * (pcn[i] ** 2 + (1 - hs) * cc * (2 - cc) * C[i]) + cmu * np.dot(w, yi * yi)
        ea = np.clip((cs / ds) * (np.linalg.norm(psn) / chi - 1.0), -50, 50)
        sigma = np.clip(sigma * math.exp(ea), 1e-4, 0.5)
        if not (np.all(np.isfinite(mn)) and np.all(np.isfinite(psn)) and np.all(np.isfinite(pcn)) and np.all(np.isfinite(Cn))):
            break
        m, ps, pc, C = mn, psn, pcn, Cn
        if sigma <= 1e-4 + 1e-15:
            min_cnt += 1
            if min_cnt >= 5:
                break
        else:
            min_cnt = 0
    hm = latent_to_h(m, n)
    if hm is not None:
        sm = exact_score(hm, n)
        if sm < best_s - 1e-12:
            best_s, best_h = sm, hm
    return best_h, best_s

def local_refine(h, n, rng, ntr, sig0):
    h = h.copy()
    s = exact_score(h, n)
    sig = sig0
    for _ in range(ntr):
        sig *= 0.97
        if sig < 1e-8:
            break
        z = h + rng.normal(0, sig, n)
        zp = project_capped(z, n)
        if zp is None or not is_feasible(zp, n):
            continue
        sz = exact_score(zp, n)
        if sz < s - 1e-13:
            h, s = zp, sz
    return h, s

def build_starts(fb, n, rng, cnt):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb, np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.05, 0.12, 0.25):
        cands.append(fb + rng.normal(0, sd, n))
    for fq in (2.0, 4.0, 6.0):
        cands.append(0.5 + 0.3 * np.sin(fq * np.pi * np.arange(n) / n + rng.uniform(0, 6.28)))
    cands.append(rng.uniform(0, 1, n))
    cands.append(np.roll(fb, n // 4))
    cands.append(np.roll(fb, -n // 4))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        if any(np.max(np.abs(p - s)) < 1e-8 for s in pool):
            continue
        pool.append(p)
        if len(pool) >= cnt:
            break
    return pool[:cnt]

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - p.size / 2.0) <= 1e-6 * max(1, p.size):
                    parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if not (1 <= n <= 999):
            n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    rng = np.random.default_rng(int(seed)) if isinstance(seed, (int, np.integer)) else np.random.default_rng(42)
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    gs, gb = exact_score(fb, n), fb.copy()
    if n <= 64:
        nst, P, G, ful = 5, max(8, min(24, 4 + int(3 * math.log(max(2, n - 1))))), 120, True
    elif n <= 256:
        nst, P, G, ful = 4, max(8, min(20, 4 + int(3 * math.log(max(2, n - 1))))), 40, False
    else:
        nst, P, G, ful = 3, max(8, min(12, 4 + int(3 * math.log(max(2, n - 1))))), 12, False
    starts = build_starts(fb, n, rng, nst)
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for idx, st in enumerate(starts):
        sig0 = 0.08 if idx == 0 else (0.14 if n <= 64 else 0.10)
        try:
            bh, bs = cmaes_run(st, n, sig0, P, G, rng, ful)
            if is_feasible(bh, n) and bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            pass
    try:
        gb, gs = local_refine(gb, n, rng, 300, 0.04)
    except Exception:
        pass
    try:
        bh, bs = cmaes_run(gb, n, 0.03, max(8, P // 2), G // 2, rng, ful)
        if is_feasible(bh, n) and bs < gs - 1e-12:
            gs, gb = bs, bh
    except Exception:
        pass
    try:
        gb, gs = local_refine(gb, n, rng, 250, 0.015)
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    fsb = exact_score(fb, n)
    if fs > fsb + 1e-12:
        final, fs = fb.copy(), fsb
    return final.astype(np.float64), float(fs), int(n)
