            u0, z0, h0s, s0 = u.copy(), z.copy(), h.copy(), score
            for j in perm:
                col = Phi[:, j]
                nh, ns, nt = scan_coeff(z, u[j], col, h, score, n, R, ng, nr, nrd)
                if ns < score - 1e-12:
                    z = z + (nt - u[j]) * col
                    u[j] = nt
                    h, score = nh, ns
                    nacc += 1
                    if score < is_ - 1e-12:
                        is_, ih = score, h.copy()
            if nacc == 0:
                sweep_ok = False
            elif score > s0 + 1e-12:
                u, z, h, score = u0, z0, h0s, s0
    return ih, is_

def build_starts(base, n, rng, count):
    pool = []
    cands = [base, np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1: blk[n//2] = 0.5
    cands.append(blk)
    cands.append(base + rng.normal(0, 0.06, n))
    cands.append(base + rng.normal(0, 0.16, n))
    cands.append(1.0 - base)
    cands.append(base[::-1].copy())
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    for k, ph in [(1, 0.0), (2, 1.5)]:
        cands.append(0.5 + 0.3 * np.sin(2.0*np.pi*k*np.arange(n)/n + ph))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        try:
            p = project_capped(c, n)
        except Exception:
            continue
        if p is None: continue
        dup = any(np.max(np.abs(p - s)) < 1e-8 for s in pool)
        if not dup:
            pool.append(p)
            if len(pool) >= count: break
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if n is None: n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5]); return h1, 0.5, 1
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fs_core = np.full(n, 0.5)
    baselines = [(fb.copy(), exact_score(fb, n)), (fs_core.copy(), exact_score(fs_core, n))]
    safe_h, safe_s = baselines[0]
    if baselines[1][1] < safe_s: safe_h, safe_s = baselines[1]
    gh, gs = safe_h.copy(), safe_s
    if n <= 64: nc, nsw, ng, nr, nrd = 6, 2, 9, 2, 3
    elif n <= 256: nc, nsw, ng, nr, nrd = 3, 2, 7, 2, 2
    else: nc, nsw, ng, nr, nrd = 2, 1, 5, 1, 2
    F = (n - 1) // 2
    if n <= 64:
        cutoffs = sorted(set([c for c in [1,2,4,8,16,F] if 1 <= c <= F]) + [F])
    elif n <= 256:
        cutoffs = sorted(set([c for c in [1,2,4,8,min(F,12)] if 1 <= c <= F]) + [F])
    else:
        cutoffs = sorted(set([c for c in [1,2,4,8,min(F,8)] if 1 <= c <= F]) + [F])
    if not cutoffs: cutoffs = [F]
    Phi, labels = build_fourier(n)
    base = fb.copy()
    starts = build_starts(base, n, rng, nc)
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12: gs, gh = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, Phi, labels, n, rng, cutoffs, nsw, ng, nr, nrd)
            if xo is not None and is_feasible(xo, n):
                sx = exact_score(xo, n)
                if sx < gs - 1e-12: gs, gh = sx, xo.copy()
        except Exception:
            pass
    final = project_capped(gh, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    ff = exact_score(final, n)
    if ff > safe_s + 1e-12:
        final, ff = safe_h.copy(), exact_score(safe_h, n)
    return final.astype(np.float64), float(ff), int(n)
