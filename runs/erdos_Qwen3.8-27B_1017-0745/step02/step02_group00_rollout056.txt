                hi_d = min(1.0 - h[i], h[j])
                if hi_d <= lo_d + 1e-12:
                    continue
                best_d, best_sc = 0.0, cur_s
                all_d = list(deltas) + list(-x for x in deltas) + [lo_d + 1e-6 if lo_d > 0 else lo_d, hi_d - 1e-6 if hi_d < 1.0 else hi_d]
                for dv in all_d:
                    if dv == 0.0:
                        continue
                    dv = max(lo_d, min(hi_d, dv))
                    h[i] += dv
                    h[j] -= dv
                    sc = exact_score(h, n)
                    h[i] -= dv
                    h[j] += dv
                    if np.isfinite(sc) and sc < best_sc - 1e-13:
                        best_sc, best_d = sc, dv
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    cur_s = best_sc
                    improved = True
        if not improved:
            break
    return h, cur_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if iv.size > 0 and iv.size <= 999:
            n = iv.size
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = np.clip(iv, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**32) if sd else 42)
    uniform = np.full(n, 0.5)
    best_h, best_s = uniform.copy(), exact_score(uniform, n)
    if parent is not None:
        p_proj = project_capped(parent, n)
        if p_proj is not None:
            ps = exact_score(p_proj, n)
            if np.isfinite(ps) and ps < best_s:
                best_h, best_s = p_proj, ps
    starts = []
    if parent is not None:
        p_proj = project_capped(parent, n)
        if p_proj is not None:
            starts.append(p_proj)
        for s2 in [0.015, 0.04, 0.08, 0.15]:
            p = project_capped(parent + rng.normal(0, s2, n), n)
            if p is not None:
                starts.append(p)
        for raw in [parent[::-1], 1.0 - parent, 1.0 - parent[::-1]]:
            p = project_capped(raw, n)
            if p is not None:
                starts.append(p)
    starts.append(best_h.copy())
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        p = project_capped(0.5 + 0.42 * np.cos(2.0 * np.pi * freq * idx / n), n)
        if p is not None:
            starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    if n <= 64:
        max_iter_tau = int(kwargs.get('max_iter', 45))
        max_starts = min(int(kwargs.get('starts', 10)), len(starts))
        max_hops = int(kwargs.get('shake_rounds', 5))
        polish_iters = int(kwargs.get('polish_iters', 3))
    elif n <= 128:
        max_iter_tau = int(kwargs.get('max_iter', 30))
        max_starts = min(int(kwargs.get('starts', 6)), len(starts))
        max_hops = int(kwargs.get('shake_rounds', 3))
        polish_iters = int(kwargs.get('polish_iters', 2))
    else:
        max_iter_tau = int(kwargs.get('max_iter', 15))
        max_starts = min(int(kwargs.get('starts', 3)), len(starts))
        max_hops = 1
        polish_iters = 1
    max_iter_tau = max(1, min(max_iter_tau, 100))
    for st in starts[:max_starts]:
        bh, bs = local_optimize(st, n, best_h, best_s, max_iter_tau)
        best_h, best_s = bh, bs
    sigma = 0.03
    for _ in range(max_hops):
        pert = project_capped(best_h + sigma * rng.normal(0, 1, n), n)
        if pert is None:
            sigma = min(0.35, 1.4 * sigma)
            continue
        bh, bs = local_optimize(pert, n, best_h, best_s, max_iter_tau)
        if bs < best_s - 1e-12:
            best_h, best_s = bh, bs
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.35, 1.4 * sigma)
    if n <= 64 and polish_iters > 0:
        for _ in range(polish_iters):
            polished, ps = pair_polish(best_h, n, polish_iters)
            polished = project_capped(polished, n)
            if polished is not None:
                psc = exact_score(polished, n)
                if np.isfinite(psc) and psc < best_s - 1e-13:
                    best_h, best_s = polished, psc
    final = project_capped(best_h, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-12:
        final = best_h.copy()
        fs = best_s
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(uniform, n)
    return final, float(fs), int(n)
