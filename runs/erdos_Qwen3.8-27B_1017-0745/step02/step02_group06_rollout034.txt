def projected_descent(x0, n, iters_per_tau, best_h, best_s):

    p = project_capped(x, n)
    if p is None:
        return x0, best_h, best_s
    x = p
    temps = [1e-2, 3e-3, 1e-3, 3e-4, 1e-4, 3e-5]
    for tau in temps:
        alpha = 1.0
        phi, g = smooth_value_grad(x, n, tau)
        if phi is None:
            break
        for it in range(iters_per_tau):
            y = project_capped(x - alpha * g, n)
            if y is None:
                break
            d = y - x
            if np.max(np.abs(d)) / alpha <= max(1e-7, 1e-3 * tau):
                break
            # Backtracking
            accepted = False
            for bt in range(25):
                phi_y, _ = smooth_value_grad(y, n, tau)
                if phi_y is None:
                    break
                dd = np.dot(d, d)
                if phi_y <= phi - 1e-4 * dd / alpha + 1e-14 * max(1.0, abs(phi)):
                    accepted = True
                    break
                alpha *= 0.5
                y = project_capped(x - alpha * g, n)
                if y is None:
                    break
                d = y - x
            if not accepted:
                break
            # Update
            x = y
            phi = phi_y
            # Update alpha
            phi_new, g_new = smooth_value_grad(x, n, tau)
            if phi_new is None:
                break
            phi = phi_new
            s = y - (x - d)  # this is just d
            v = g_new - g
            sv = np.dot(d, v)
            ss = np.dot(d, d)
            if sv > 1e-14 and np.isfinite(sv / ss):
                alpha = float(np.clip(ss / sv, 1e-4, 10.0))
            else:
                alpha = float(np.clip(alpha, 1e-4, 10.0))
            g = g_new
            # Check exact score
            s_exact = exact_score(x, n)
            if s_exact < best_s - 1e-12:
                best_s = s_exact
                best_h = x.copy()
    return x, best_h, best_s

            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(r) > 1e-9 or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or c.size != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

def smooth_value_grad(h, n, tau):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(L)):
        return None, None
    a = float(np.max(L))
    e = np.exp(np.maximum((L - a) / tau, -745.0))
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return None, None
    w = e / se
    phi = a + tau * (math.log(se) - math.log(2 * n - 1))
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(2 * n - 1):
        wi = float(w[idx])
        if wi < 1e-15:
            continue
        d = idx - (n - 1)
        q0, q1 = max(d, 0), min(n - 1 + d, n - 1)
        if q0 <= q1:
            grad[q0:q1 + 1] += dx * wi * (1.0 - h[q0 - d:q1 - d + 1])
        q2, q3 = max(0, -d), min(n - 1, n - 1 - d)
        if q2 <= q3:
            grad[q2:q3 + 1] -= dx * wi * h[q2 + d:q3 + d + 1]
    if not np.all(np.isfinite(grad)):
        return None, None
    return phi, grad

def projected_descent(x0, n, iters, best_h, best_s):
    p = project_capped(x0, n)
    if p is None:
        return x0, best_h, best_s
    x = p.copy()
    temps = [1e-2, 3e-3, 1e-3, 3e-4, 1e-4, 3e-5]
    for tau in temps:
        phi, g = smooth_value_grad(x, n, tau)
        if phi is None:
            break
        alpha = 1.0
        for _ in range(iters):
            accepted = False
            d, phi_y = None, None
            for _bt in range(25):
                y = project_capped(x - alpha * g, n)
                if y is None:
                    break
                d = y - x
                dd = float(np.dot(d, d))
                if dd < 1e-30:
                    break
                phi_y, _ = smooth_value_grad(y, n, tau)
                if phi_y is None:
                    break
                if phi_y <= phi - 1e-4 * dd / alpha + 1e-14 * max(1.0, abs(phi)):
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            x = y
            phi_new, g_new = smooth_value_grad(x, n, tau)
            if phi_new is None:
                break
            phi = phi_new
            sv = float(np.dot(d, g_new - g))
            ss = float(np.dot(d, d))
            if sv > 1e-14 and math.isfinite(sv / max(ss, 1e-30)):
                alpha = float(np.clip(ss / sv, 1e-4, 10.0))
            else:
                alpha = float(np.clip(alpha, 1e-4, 10.0))
            g = g_new
            s_ex = exact_score(x, n)
            if s_ex < best_s - 1e-12:
                best_s = s_ex
                best_h = x.copy()
    return x, best_h, best_s

def add_to_pool(pool, c, n):
    p = project_capped(c, n)
    if p is None:
        return
    for s in pool:
        if float(np.max(np.abs(p - s))) < 1e-6:
            return
    pool.append(p)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(float(np.sum(p)) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
    if n <= 1:
        h1 = np.array([0.5]); return h1, float(exact_score(h1, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    ub = np.full(n, 0.5)
    fb = parent.copy() if parent is not None else ub.copy()
    if parent is not None:
        pr = project_capped(parent, n)
        if pr is not None:
            fb = pr
    if n <= 64:
        n_starts, max_it, n_restarts = 8, 120, 3
    elif n <= 128:
        n_starts, max_it, n_restarts = 6, 80, 2
    elif n <= 256:
        n_starts, max_it, n_restarts = 4, 50, 1
    else:
        n_starts, max_it, n_restarts = 3, 30, 1
    pool = []
    add_to_pool(pool, fb, n)
    add_to_pool(pool, ub, n)
    base = fb if parent is not None else ub
    add_to_pool(pool, base[::-1].copy(), n)
    add_to_pool(pool, 1.0 - base[::-1].copy(), n)
    add_to_pool(pool, 1.0 - base.copy(), n)
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_to_pool(pool, blk, n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_to_pool(pool, alt, n)
    xs = np.linspace(0, 2 * np.pi * (1 - 1.0 / n), n)
    for m in [1, 2, 3]:
        for ph in [0.0, np.pi / 2]:
            add_to_pool(pool, 0.5 + 0.45 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph), n)
    for sd in [0.04, 0.12, 0.25]:
        add_to_pool(pool, base + rng.normal(0.0, sd, n), n)
    add_to_pool(pool, rng.uniform(0.0, 1.0, n), n)
    while len(pool) < n_starts:
        add_to_pool(pool, base + rng.normal(0.0, 0.15, n), n)
    scored = [(exact_score(s, n), s) for s in pool[:min(len(pool), n_starts * 2)]]
    scored.sort(key=lambda t: t[0])
    starts = [s for _, s in scored[:n_starts]]
    best_h, best_s = fb.copy(), exact_score(fb, n)
    for st in starts:
        ss = exact_score(st, n)
        if ss < best_s - 1e-12:
            best_s, best_h = ss, st.copy()
        _, best_h, best_s = projected_descent(st, n, max_it, best_h, best_s)
    sigma = 0.08
    for _r in range(n_restarts):
        saved = best_s
        ok = False
        for _a in range(3):
            prop = best_h + sigma * rng.standard_normal(n)
            pp = project_capped(prop, n)
            if pp is not None and float(np.max(np.abs(pp - best_h))) > 1e-5:
                ok = True
                break
        if not ok:
            break
        pp_s = exact_score(pp, n)
        if pp_s < best_s - 1e-12:
            best_s, best_h = pp_s, pp.copy()
        _, best_h, best_s = projected_descent(pp, n, max_it, best_h, best_s)
        if best_s < saved - 1e-12:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
    fs = exact_score(final, n)
    fb_s = exact_score(fb, n)
    ub_s = exact_score(ub, n)
    if fs > min(fb_s, ub_s) + 1e-12:
        final = fb if fb_s <= ub_s else ub
        fs = min(fb_s, ub_s)
    return final, float(fs), int(n)
