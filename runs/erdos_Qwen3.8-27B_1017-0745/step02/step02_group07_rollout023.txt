def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)

    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')

def lag_values(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    Q = np.zeros(m)
    Q[0] = dx * np.dot(h, 1.0 - h)
    for r in range(1, n):
        mm = n - r
        Q[r + r - 1] = dx * np.dot(h[:mm], 1.0 - h[r:])
        Q[r + r] = dx * np.dot(h[r:], 1.0 - h[:mm])
    return Q

def lag_jacobian(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    Q = np.zeros(m)
    G = np.zeros((m, n))
    Q[0] = dx * np.dot(h, 1.0 - h)
    G[0, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        mm = n - r
        Q[r + r - 1] = dx * np.dot(h[:mm], 1.0 - h[r:])
        Q[r + r] = dx * np.dot(h[r:], 1.0 - h[:mm])
        G[r + r - 1, :mm] += dx * (1.0 - h[r:])
        G[r + r - 1, r:] -= dx * h[:mm]
        G[r + r, :mm] -= dx * h[r:]
        G[r + r, r:] += dx * (1.0 - h[:mm])
    return Q, G

def slsqp_obj(z, n):
    return z[-1]

def slsqp_obj_jac(z, n):
    j = np.zeros(n + 1)

def slsqp_ineq(z, n):
    h = z[:n]
    t = z[-1]
    Q, _ = lag_jacobian(h, n)
    return t - Q

def slsqp_ineq_jac(z, n):
    h = z[:n]
    _, G = lag_jacobian(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

def smooth_val(h, n, tau):
    Q = lag_values(h, n)
    qmax = np.max(Q)
    e = np.exp((Q - qmax) / tau)
    return qmax + tau * np.log(np.sum(e))

        g[r:] += p[r + r] * dx * (1.0 - h[:mm])
