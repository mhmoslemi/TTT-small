            continue
        lag = m - (n - 1)
        q0 = max(0, lag); q1 = min(n, lag + n)
        if q1 > q0:
            grad[q0:q1] += dx * wm * (1.0 - h[q0 - lag:q1 - lag])
        q0b = max(0, -lag); q1b = min(n, n - lag)
        if q1b > q0b:
            grad[q0b:q1b] -= dx * wm * h[q0b + lag:q1b + lag]
    if not np.all(np.isfinite(grad)):
        return None, None
    return val, grad

def smooth_descent(h0, n, max_it, best_h, best_s):
    h = h0.copy().astype(np.float64)
    temps = [1e-2, 3e-3, 1e-3, 3e-4, 1e-4, 3e-5]
    for tau in temps:
        alpha = 1.0
        val, grad = smooth_val_grad(h, n, tau)
        if val is None or grad is None:
            break
        for _ in range(max_it):
            accepted = False
            for _bt in range(20):
                y = project_capped(h - alpha * grad, n)
                if y is None:
                    break
                d = y - h
                if np.max(np.abs(d)) <= 1e-14:
                    break
                vc, _ = smooth_val_grad(y, n, tau)
                if vc is None:
                    break
                sl = float(np.dot(grad, d))
                if sl < 0 and vc <= val + 1e-4 * sl + 1e-14:
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            g_old = grad.copy()
            step = y - h
            h = y
            val, grad = smooth_val_grad(h, n, tau)
            if val is None or grad is None:
                break
            sv = float(np.dot(step, grad - g_old))
            ss = float(np.dot(step, step))
            if sv > 1e-14 and np.isfinite(sv):
                alpha = max(1e-4, min(10.0, ss / sv))
            s = exact_score(h, n)
            if s < best_s:
                best_s = s
                best_h = h.copy()
    s = exact_score(h, n)
    if s < best_s:
        best_s = s
        best_h = h.copy()
    return h, best_h, best_s

def build_starts(parent, n, rng, count):
    pool = []
    def try_add(c):
        p = project_capped(c, n)
        if p is None:
            return
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-6:
                return
        pool.append(p)
    try_add(np.full(n, 0.5))
    if parent is not None:
        try_add(parent)
        try_add(parent[::-1].copy())
        try_add(1.0 - parent[::-1])
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(blk)
    try_add(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    base = parent if parent is not None else np.full(n, 0.5)
    for m in [1, 2, 3]:
        for ph in [0.0, math.pi / 2]:
            wv = 0.5 + 0.45 * np.cos(2.0 * math.pi * m * np.arange(n) / n + ph)
            try_add(wv)
    for sd in [0.04, 0.12, 0.25]:
        try_add(base + rng.normal(0, sd, n))
    try_add(rng.uniform(0, 1, n))
    while len(pool) < count:
        try_add(base + rng.normal(0, 0.1, n))
        if len(pool) >= count:
            break
    scored = []
    for p in pool:
        scored.append((exact_score(p, n), p))
    scored.sort(key=lambda x: x[0])
    return [p for _, p in scored[:count]]
