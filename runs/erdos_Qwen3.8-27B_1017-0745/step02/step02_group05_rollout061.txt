                    t = cap * f
                    h2 = best_h.copy()
                    h2[di] += t
                    h2[dj] -= t
                    h2 = project_capped(h2, n)
                    if h2 is None:
                        continue
                    s2 = exact_score(h2, n)
                    if np.isfinite(s2) and s2 < best_s - 1e-13:
                        best_s = s2
                        best_h = h2
            if best_s < s_cur - 1e-13:
                h = best_h
                s_cur = best_s
                improved = True
        if not improved:
            break
    return h, s_cur

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if n < 1 or n > 999:
            n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    const_vec = np.full(n, 0.5)
    safe_h = const_vec
    safe_s = exact_score(const_vec, n)
    base_vec = const_vec.copy()
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and is_feasible(pp, n):
            ps = exact_score(pp, n)
            base_vec = pp
            if ps < safe_s:
                safe_h = pp.copy()
                safe_s = ps
    if n <= 64:
        P, G_total, L, stall, n_sweeps = 12, 120, 1600, 20, 2
    elif n <= 256:
        P, G_total, L, stall, n_sweeps = 8, 70, 500, 14, 2
    else:
        P, G_total, L, stall, n_sweeps = 6, 25, 80, 7, 1
    pop = make_starts(base_vec, n, rng, P)
    scores = [exact_score(p, n) for p in pop]
    best_h = pop[int(np.argmin(scores))].copy()
    best_s = min(scores)
    pop[0] = best_h.copy()
    scores[0] = best_s
    prev_best = best_s
    stall_cnt = 0
    for gen in range(G_total):
    try:
        pop, scores = de_generation(pop, scores, n, rng, gen, G_total)
    except Exception:
        break
    gm = min(scores)
    if gm < prev_best - 1e-12:
        prev_best = gm
        stall_cnt = 0
    else:
        stall_cnt += 1
    if gm < best_s - 1e-12:
        best_s = gm
        best_h = pop[int(np.argmin(scores))].copy()
    if stall_cnt >= stall and gen < G_total - 1:
        keep = max(2, P // 3)
        idx_sorted = sorted(range(P), key=lambda x: scores[x])[:keep]
        np2 = [pop[i].copy() for i in idx_sorted]
        q = gen / max(1, G_total - 1)
        sig = 0.025 + 0.075 * (1.0 - q)
        for _ in range(P - len(np2)):
            c = best_h + sig * rng.normal(0, 1, n) if rng.random() < 0.5 else rng.uniform(0, 1, n)
            p2 = project_capped(c, n)
            np2.append(p2 if p2 is not None else best_h.copy())
        pop = np2[:P]
        scores = [exact_score(p, n) for p in pop]
        gm2 = min(scores)
        if gm2 < best_s - 1e-12:
            best_s = gm2
            best_h = pop[int(np.argmin(scores))].copy()
        pop[0] = best_h.copy()
        scores[0] = best_s
        stall_cnt = 0
        prev_best = best_s
    try:
        ah, as_ = pair_anneal(best_h, n, L, rng)
        if as_ < best_s - 1e-12:
            best_h, best_s = ah, as_
    except Exception:
        pass
    try:
        ph, ps_ = pair_polish(best_h, n, n_sweeps, rng)
        if ps_ < best_s - 1e-12:
            best_h, best_s = ph, ps_
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        fs = exact_score(safe_h, n)
        final = safe_h.copy()
    return final.astype(np.float64), float(fs), int(n)
