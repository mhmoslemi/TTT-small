import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def helmert_forward(x, n):
    d = n - 1
    u = np.empty(d)
    for k in range(d):
        u[k] = x[k] / np.sqrt((k + 1) * (k + 2))
    suffix = np.zeros(d + 1)
    for i in range(d - 1, -1, -1):
        suffix[i] = suffix[i + 1] + u[i]
    y = np.empty(n)
    y[0] = suffix[0]
    for i in range(1, n):
        y[i] = suffix[i] - i * x[i - 1] / np.sqrt(i * (i + 1))
    return y

def helmert_transpose(y, n):
    d = n - 1
    x = np.empty(d)
    for k in range(d):
        x[k] = (np.sum(y[:k + 1]) - (k + 1) * y[k + 1]) / np.sqrt((k + 1) * (k + 2))
    return x

def latent_to_h(x, n):
    y = helmert_forward(x, n)
    if not np.all(np.isfinite(y)):
        return None
    return project_capped(0.5 + y, n)

    mu = P // 2
    if mu < 1:
        return start, exact_score(start, n)
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    w = w / w.sum()
    mu_eff = 1.0 / np.sum(w * w)
    c_sigma = (mu_eff + 2) / (d + mu_eff + 5)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (d + 1)) - 1) + c_sigma
    c_c = (4 + mu_eff / d) / (d + 4 + 2 * mu_eff / d)
    c_1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = max(0, min(1 - c_1, 2 * (mu_eff - 2 + 1 / mu_eff) / ((d + 2) ** 2 + mu_eff)))
    chi_d = np.sqrt(d) * (1 - 1 / (4 * d) + 1 / (21 * d * d))
    m = helmert_transpose(start - 0.5, n)
    if not np.all(np.isfinite(m)):
        return start, exact_score(start, n)
    sigma = sigma0
    p_sigma = np.zeros(d)
    p_c = np.zeros(d)
    if full_cov:
        C = np.eye(d)
    else:
        C = np.ones(d)

    best_s = exact_score(start, n)
    stall = 0
    for t in range(1, G + 1):
        if full_cov:
            C = 0.5 * (C + C.T)
            eig, U = np.linalg.eigh(C)
            eig = np.clip(eig, 1e-10, 1e4)
            sqrt_eig = np.sqrt(eig)
            A = U * sqrt_eig  # U @ diag(sqrt_eig)
            Ainv = U / sqrt_eig  # U @ diag(1/sqrt_eig)
        else:

        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * y_w

            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if n is None:
        n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    safe_h = np.full(n, 0.5)
    safe_s = exact_score(safe_h, n)
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and is_feasible(pp, n):
            sp = exact_score(pp, n)
            if sp < safe_s:
                safe_s, safe_h = sp, pp
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_st, P, G, fc = 5, 15, 120, True
        s0, so = 0.06, 0.12
    elif n <= 256:
        n_st, P, G, fc = 4, 18, 40, False
        s0, so = 0.05, 0.10
    else:
        n_st, P, G, fc = 3, 12, 12, False
        s0, so = 0.05, 0.08
    base = safe_h.copy()
    starts = build_starts(base, n, rng, n_st)
    gb, gs = safe_h.copy(), safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for i_st, st in enumerate(starts):
        sig = s0 if i_st == 0 else so
        try:
            ho, so2 = cma_es(st, n, sig, P, G, fc, rng)
            if ho is not None and is_feasible(ho, n) and so2 < gs - 1e-12:
                gs, gb = so2, ho.copy()
        except Exception:
            pass
    try:
        gb, gs = local_refine(gb, n, rng, 150, 0.03)
    except Exception:
        pass
    try:
        gb2, gs2 = cma_es(gb, n, 0.02, max(8, P // 2), 40, fc, rng)
        if is_feasible(gb2, n) and gs2 < gs - 1e-12:
            gs, gb = gs2, gb2.copy()
    except Exception:
        pass
    try:
        gb, gs = local_refine(gb, n, rng, 100, 0.01)
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs), int(n)
