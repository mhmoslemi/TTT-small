def project_capped(z, n):

    if z.shape != (n,) or not np.all(np.isfinite(z)):

        else:

def exact_score(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def pairwise_polish(h, n, sweeps):

            for j in range(i + 1, n):

                hi = min(1.0 - best[i], best[j])

                    continue

                    best[j] -= best_d
                    best_s = best_sc
                    improved = True
