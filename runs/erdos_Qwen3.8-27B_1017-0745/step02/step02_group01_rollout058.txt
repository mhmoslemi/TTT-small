def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,):
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i])
            h[i] += c
            r -= c
        else:
            c = min(-r, h[i])
            h[i] -= c
            r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def pair_bounds(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    return lo, hi

def make_proposal(h, n, rng, step):
    if rng.random() < 0.8:
        for _ in range(4):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            lo, hi = pair_bounds(h, i, j)
            if hi - lo < 1e-12:
                continue
            if rng.random() < 0.75:
                a = max(lo, -step)
                b = min(hi, step)
                if b - a < 1e-12:
                    continue
                t = rng.uniform(a, b)
            else:
                t = lo if rng.random() < 0.5 else hi
                if abs(t) < 1e-12:
                    t = hi if abs(lo) < 1e-12 else lo
            if abs(t) < 1e-12:
                continue
            z = h.copy()
            z[i] += t
            z[j] -= t
            return project_capped(z, n)
    z = h + step * rng.standard_normal(n)
    return project_capped(z, n)

    current = start.copy()
    cur_score = exact_score(current, n)
    if cur_score < best_score - 1e-12:
        best_score = cur_score
        best_h = current.copy()
    step = 0.20
    T0 = 0.002
    Tmin = 1e-6
    epoch_len = max(1, (steps + 2) // 3)
    accepted = 0
    attempted = 0
    for t in range(steps):
        phase = t % epoch_len
        T = T0 * (Tmin / T0) ** (phase / max(1, epoch_len - 1))
        if t > 0 and t % epoch_len == 0:
            sigma = 0.08 + 0.12 * rng.random()
            current = project_capped(best_h + sigma * rng.standard_normal(n), n)
            if current is None:
                current = best_h.copy()
            cur_score = exact_score(current, n)
            step = 0.20
            accepted = 0
            attempted = 0
        prop = make_proposal(current, n, rng, step)
        if prop is None:
            continue
        attempted += 1
        s_new = exact_score(prop, n)
        if s_new <= cur_score:
            current = prop
            cur_score = s_new
            accepted += 1
        else:
            exp_arg = max(-(s_new - cur_score) / T, -745.0)
            if rng.random() < math.exp(exp_arg):
                current = prop
                cur_score = s_new
                accepted += 1
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_h = current.copy()
        if attempted > 0 and attempted % 100 == 0:
            ratio = accepted / attempted
            if ratio > 0.45:
                step = min(step * 1.20, 0.6)
            elif ratio < 0.15:
                step = max(step * 0.70, 1e-4)
            accepted = 0
            attempted = 0
    return current, best_h, best_score

    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
