def slsqp_refine(h0, n, maxiter):

        g = np.zeros(N + 1)
        g[N] = 1.0
        return g

        hh = np.clip(z[:N], 0.0, 1.0)
        return z[N] - lag_values(hh, N)

    def ineq_jac(z):
        hh = np.clip(z[:N], 0.0, 1.0)
        dx = 2.0 / N

        J = np.zeros((L, N + 1))

            r = -s if s < 0 else 0
            m = N - abs(s)
            J[k, p:p+m] -= dx * (1.0 - hh[r:r+m])
            J[k, r:r+m] += dx * hh[p:p+m]
            J[k, N] = 1.0
        return J

        return np.sum(z[:N]) - N / 2.0

    def eq_jac(z):
        J = np.zeros(N + 1)
        J[:N] = 1.0
        return J

    starts = build_starts(fb, n, rng, n_sc)
    scored = [(exact_score(s, n), s) for s in starts]
    scored.sort(key=lambda x: x[0])

    for _, st in scored:
        for _ in range(n_pass):
            try:
                best_c, best_cs = cont_descent(st, n, taus, maxit_c)
                if best_cs < gs - 1e-12:
                    gs, gb = best_cs, best_c
            except Exception:
                continue

    # SLSQP refinement on best
    if n <= 64:
        for _ in range(n_pass):
            try:
                found = slsqp_refine(gb, n, 200)
                if found is None:
                    break
                y = project_capped(found, n)
                if y is None:
                    break
                bz, bs = gb, gs
                for a in (1.0, 0.5, 0.25, 0.125, 0.0625):
                    tr = project_capped(gb + a * (y - gb), n)
                    if tr is None:
                        continue
                    sz = exact_score(tr, n)
                    if sz < bs - 1e-12:
                        bs, bz = sz, tr
                if bs < gs - 1e-12:
                    gs, gb = bs, bz.copy()
            except Exception:
                break

    # Perturbation restarts
    sds = [0.04, 0.10, 0.02] if gs < fb_s - 1e-5 else [0.08, 0.18, 0.04]
    for sd in sds[:3]:
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            best_c, best_cs = cont_descent(pert, n, [0.005, 0.001, 0.0003, 0.0001], maxit_c // 2)
            if n <= 64:
                found = slsqp_refine(best_c, n, 150)
                if found is not None:
                    y = project_capped(found, n)
                    if y is not None:
                        sz = exact_score(y, n)
                        if sz < best_cs:
                            best_c, best_cs = y, sz
            if best_cs < gs - 1e-12:
                gs, gb = best_cs, best_c
        except Exception:
            continue

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5).astype(np.float64)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
