        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    sd = sd % (2**32)
    rng = np.random.default_rng(sd)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    B = min(16000, max(120, 30000000 // (n * n)))
    try:
        B = max(120, min(int(kwargs.get('eval_budget', B)), 20000, 16000))
    except Exception:
        pass
    B = min(B, max(120, 30000000 // (n * n)))
    n_rep = 6
    temps = np.array([0.0005, 0.002, 0.008, 0.032, 0.128, 0.512])[:n_rep]
    scales = np.array([0.002, 0.006, 0.018, 0.054, 0.162, 0.486])[:n_rep]
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback + rng.normal(0, 0.02, n), n)
        add_start(starts, fallback + rng.normal(0, 0.08, n), n)
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    att = 0
    while len(starts) < n_rep and att < 16:
        add_start(starts, rng.uniform(0, 1, n), n)
        att += 1
    while len(starts) < n_rep:
        starts.append(starts[len(starts) % max(1, len(starts))].copy())
    starts = starts[:n_rep]
    replicas, rep_scores = [], []
    for i in range(n_rep):
        h_r = 0.99 * starts[i] + 0.01 * 0.5
        h_r = project_capped(h_r, n)
        if h_r is None:
            h_r = starts[i].copy()
        s_r = exact_score(h_r, n)
        replicas.append(h_r)
        rep_scores.append(s_r)
    eval_count = n_rep
    best, best_score = fallback.copy(), fb_score
    for i in range(n_rep):
        if np.isfinite(rep_scores[i]) and rep_scores[i] < best_score - 1e-12:
            best, best_score = replicas[i].copy(), rep_scores[i]
    stagnation = 0
    st_thresh = max(30, min(200, B // (n_rep * 12)))
    n_restarts = 0
    prev_best = best_score
    max_sweeps = B + 1
    for sweep in range(max_sweeps):
        for i in range(n_rep):
            if eval_count >= B:
                break
            u = rng.random()
            if u < 0.35:
                h_prop = hit_run(replicas[i], n, rng)
            else:
                h_prop = pair_proposal(replicas[i], n, scales[i], rng)
            if h_prop is None:
                continue
            s_prop = exact_score(h_prop, n)
            eval_count += 1
            if not np.isfinite(s_prop):
                continue
            delta = s_prop - rep_scores[i]
            if delta <= 0 or math.log(max(rng.random(), 1e-300)) < -delta / temps[i]:
                replicas[i] = h_prop
                rep_scores[i] = s_prop
            if np.isfinite(rep_scores[i]) and rep_scores[i] < best_score - 1e-12:
                best, best_score = replicas[i].copy(), rep_scores[i]
        if eval_count >= B:
            break
        if sweep % 2 == 0:
            pairs = [(0, 1), (2, 3), (4, 5)]
        else:
            pairs = [(1, 2), (3, 4)]
        for (a, b) in pairs:
            if a >= n_rep or b >= n_rep:
                continue
            log_ratio = (1.0 / temps[a] - 1.0 / temps[b]) * (rep_scores[a] - rep_scores[b])
            if log_ratio < 0 and math.log(max(rng.random(), 1e-300)) < log_ratio:
                replicas[a], replicas[b] = replicas[b], replicas[a]
                rep_scores[a], rep_scores[b] = rep_scores[b], rep_scores[a]
        if best_score > prev_best - 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        prev_best = best_score
        if stagnation >= st_thresh and n_restarts < 2:
            h_new = 0.5 * best + 0.5 * rng.uniform(0, 1, n) if n_restarts == 0 \
                else project_capped(rng.beta(0.7, 0.7, n), n)
            if h_new is None:
                h_new = rng.uniform(0, 1, n)
            h_new = project_capped(0.99 * h_new + 0.01 * 0.5, n)
            if h_new is not None and eval_count < B:
                s_new = exact_score(h_new, n)
                eval_count += 1
                replicas[-1], rep_scores[-1] = h_new, s_new
                if np.isfinite(s_new) and s_new < best_score - 1e-12:
                    best, best_score = h_new.copy(), s_new
            n_restarts += 1
            stagnation = 0
    if n <= 128:
        bounds = [(0.0, 1.0)] * (n + 1)
        con_eq = {'fun': _eq_fun, 'jac': _eq_jac, 'type': 'eq', 'args': (n,)}
        con_ineq = {'fun': _ineq_fun, 'jac': _ineq_jac, 'type': 'ineq', 'args': (n,)}
        for init in [best, fallback]:
            t0 = min(1.0, exact_score(init, n) + 2e-4)
            z0 = np.concatenate([init, [t0]])
            try:
                res = minimize(_obj, z0, args=(n,), jac=_obj_jac, bounds=bounds,
                               constraints=[con_eq, con_ineq], method='SLSQP',
                               options={'ftol': 1e-11, 'maxiter': 200})
            except Exception:
                continue
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x[:n])):
                hc = project_capped(res.x[:n], n)
                if hc is not None:
                    sc = exact_score(hc, n)
                    if np.isfinite(sc) and sc < best_score - 1e-12:
                        best, best_score = hc.copy(), sc
    if n <= 64:
        pp, pp_s = pairwise_polish(best, n, 1)
        fp = project_capped(pp, n)
        if fp is not None:
            fs = exact_score(fp, n)
            if np.isfinite(fs) and fs < best_score - 1e-12:
                best, best_score = fp, fs
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    fb2 = exact_score(fallback, n)
    if not np.isfinite(fs) or fs > fb2:
        final, fs = fallback.copy(), fb2
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
