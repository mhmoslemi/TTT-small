def lag_values(h, n):

def quad_roots(a, b, c):
    """Solve a*t^2 + b*t + c = 0, return list of real roots."""
    roots = []
    if abs(a) < 1e-16:
        if abs(b) < 1e-16:
            return roots
        roots.append(-c / b)
    else:
        disc = b * b - 4.0 * a * c
        tol = 1e-14 * (b * b + abs(4.0 * a * c) + 1e-300)
        if disc < -tol:
            return roots
        if disc < 0:
            disc = 0.0
        sd = np.sqrt(disc)
        if abs(b) > 1e-300:
            q = -0.5 * (b + np.copysign(sd, b))
            roots.append(q / a)
            roots.append(c / q) if abs(q) > 1e-300 else roots.append((sd - b) / (2.0 * a))
        else:
            roots.append((-b + sd) / (2.0 * a))
            roots.append((-b - sd) / (2.0 * a))
    return [r for r in roots if np.isfinite(r)]
