def polish_pairwise(h0, n, rng):
    cur = h0.copy()
    cs = exact_score(cur, n)
    budget = min(2500, max(300, int(2500 * (47.0 / n) ** 2)))
    rem = budget
    meshes = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
    npairs = min(100, max(20, 2 * n))
    for delta in meshes:
        if rem <= 0:
            break
        improved = True
        while improved and rem > 0:
            improved = False
            base = cur.copy()
            pairs = rng.integers(0, n, size=(npairs, 2))
            for k in range(npairs):
                if rem <= 0:
                    break
                i, j = int(pairs[k, 0]), int(pairs[k, 1])
                if i == j:
                    continue
                lo = max(-base[i], base[j] - 1.0)
                hi = min(1.0 - base[i], base[j])
                for dt in (-delta, -0.5 * delta, 0.5 * delta, delta):
                    if rem <= 0:
                        break
                    t = max(lo, min(hi, dt))
                    if abs(t) < 1e-14:
                        rem -= 1
                        continue
                    z = base.copy(); z[i] += t; z[j] -= t
                    y = project_capped(z, n)
                    rem -= 1
                    if y is None:
                        continue
                    ys = exact_score(y, n)
                    if ys < cs - 1e-12:
                        cs, cur = ys, y.copy()
                        improved = True
    return cur, cs

def try_add(pool, v, n, limit):
    if len(pool) >= limit or v is None:
        return
    v = project_capped(v, n)
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v)
