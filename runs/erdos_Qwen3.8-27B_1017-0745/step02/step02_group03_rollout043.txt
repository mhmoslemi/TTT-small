    m_s = m[order]

    unique_m = []
    unique_c = []

            unique_c.append(c_s[i])

            if c_s[i] > unique_c[-1]:
                unique_c[-1] = c_s[i]

    unique_c = np.array(unique_c, dtype=np.float64)

        return np.array([]), np.array([]), np.array([])

def pair_sweep(h, n, pair_order, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(corr)) * (2.0 / n)
    improved = False
    for (p, q) in pair_order:
        L = max(-h[p], h[q] - 1.0)
        U = min(1.0 - h[p], h[q])
        if U - L < 1e-14:
            continue
        t = chord_min_t(corr, h, n, p, q, L, U)
        if abs(t) < 1e-13:
            continue
        # Apply move
        h_new = h.copy()
        h_new[p] += t
        h_new[q] -= t
        h_new[p] = np.clip(h_new[p], 0, 1)
        h_new[q] = np.clip(h_new[q], 0, 1)
        # Verify feasibility
        if abs(np.sum(h_new) - n/2.0) > 1e-10:
            continue
        sc_new = float(np.max(np.correlate(h_new, 1.0-h_new, mode='full'))) * (2.0 / n)
        if sc_new < score - 1e-12:
            h = h_new
            score = sc_new
            corr = np.correlate(h, 1.0 - h, mode='full')
            improved = True
    return h, score, improved
