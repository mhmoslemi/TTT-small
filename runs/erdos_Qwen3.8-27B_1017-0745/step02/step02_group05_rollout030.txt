def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:

        else:

def is_feasible(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

        return None

    A = np.correlate(h, 1.0 - h, mode='full') * dx

    return A

    return float(np.max(A))

def compute_B(h, n, i, j):
    """Compute B_d for all lags d for the pair direction v = e_i - e_j."""
    dx = 2.0 / n
    B = np.zeros(2 * n - 1, dtype=np.float64)
    for d in range(-(n - 1), n):
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        val = 0.0
        # i on a-side
        if a <= i < a + m:
            val += (1.0 - h[b + (i - a)])
        # j on a-side
        if a <= j < a + m:
            val -= (1.0 - h[b + (j - a)])
        # i on b-side
        if b <= i < b + m:
            val -= h[a + (i - b)]
        # j on b-side
        if b <= j < b + m:
            val += h[a + (j - b)]
        B[d + n - 1] = dx * val
    return B

def solve_quad_pairs(cands, polys, lo, hi):
    """Add roots of pairwise differences of polys (each a list [a,b,c] for a*t^2+b*t+c) to cands."""
    npolys = len(polys)
    for p in range(npolys):
        for q in range(p + 1, npolys):
            aa = polys[p][0] - polys[q][0]
            bb = polys[p][1] - polys[q][1]
            cc = polys[p][2] - polys[q][2]
            if abs(aa) < 1e-15:
                if abs(bb) > 1e-15:
                    t = -cc / bb
                    if lo - 1e-12 <= t <= hi + 1e-12:
                        cands.append(max(lo, min(hi, t)))

                tol = 1e-14 * (bb * bb + abs(4.0 * aa * cc) + 1e-300)

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):

    ct = ct[np.isfinite(ct)]
    if ct.size == 0:
        return 0.0, float(np.max(A))
    ct = np.clip(ct, lo, hi)
    ct.sort()
    mask = np.concatenate(([True], np.diff(ct) > 1e-12))
    ct = ct[mask]
    t2 = ct * ct
    scores = A[None, :] + B[None, :] * ct[:, None] + D[None, :] * t2[:, None]
    ms = np.max(scores, axis=1)
    bi = int(np.argmin(ms))
    return float(ct[bi]), float(ms[bi])

def sweep_pairs(h, n, rng, n_sweeps, all_pairs):
    h = h.copy()
    A = lag_values(h, n)
    if A is None:
        return h, exact_score(h, n)
    cur = float(np.max(A))
    pairs = list(all_pairs)
    stalled = 0
    for sw in range(n_sweeps):
        rng.shuffle(pairs)
        start_score = cur
        n_acc = 0
        for (i, j) in pairs:
            t_star, pred = pair_line_min(h, A, n, i, j)
            if not isfinite(t_star) or pred >= cur - 1e-12:
                continue
            h[i] += t_star
            h[j] -= t_star
            if (not np.all(np.isfinite(h)
                  and h >= -1e-12 and h <= 1.0 + 1e-12
                  and abs(np.sum(h) - n / 2.0) <= 1e-10 * n)):
                h[i] -= t_star
                h[j] += t_star
                continue
            A = A + B * t_star + D * t_star * t_star
            cur = float(np.max(A))
            n_acc += 1
            if n_acc % max(1, n) == 0:
                A = lag_values(h, n)
                if A is None:
                    break
                cur = float(np.max(A))
        A = lag_values(h, n)
        if A is not None:
            cur = float(np.max(A))
        if cur < start_score - 1e-12:
            stalled = 0
        else:
            stalled += 1
            if stalled >= 2:
                break
    return h, cur

def build_starts(base, n, rng, count):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base,
             np.roll(base, n // 4), np.roll(base, -n // 4)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.extend([blk, blk[::-1].copy()])
    for sd in (0.05, 0.15):
        cands.append(base + rng.normal(0, sd, n))
    for fr in (2.0 * np.pi, 4.0 * np.pi):
        ph = rng.uniform(0, 2 * np.pi)
        cands.append(0.5 + 0.35 * np.sin(fr * np.arange(n) / n + ph))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = any(np.max(np.abs(p - s)) < 1e-8 for s in pool)
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    return pool[:count]

def nm_polish(h, n, rng):
    def obj(z):
        hp = project_capped(z, n)
        return exact_score(hp, n) if hp is not None else 10.0
    from scipy.optimize import minimize
    try:
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': 400, 'xatol': 1e-10, 'fatol': 1e-12})
        hp = project_capped(res.x, n)
        if hp is not None and is_feasible(hp, n):
            return hp, exact_score(hp, n)
    except Exception:
        pass
    return h, exact_score(h, n)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(0.0), 1
    fb = project_capped(parent, n) if parent is not None else None
    const = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = const.copy()
    safe_h = fb.copy()
    safe_s = exact_score(fb, n)
    s_const = exact_score(const, n)
    if s_const < safe_s:
        safe_h, safe_s = const.copy(), s_const
    best_h, best_s = safe_h.copy(), safe_s
    if n <= 64:
        n_starts, n_sweeps, use_all = 6, 14, True
    elif n <= 256:
        n_starts, n_sweeps, use_all = 4, 8, False
    else:
        n_starts, n_sweeps, use_all = 2, 5, False
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    starts = build_starts(fb, n, rng, n_starts)
    for st in starts:
        ss = exact_score(st, n)
        if ss < best_s - 1e-12:
            best_s, best_h = ss, st.copy()
    for st in starts:
        try:
            if use_all:
                xo, so = sweep_pairs(st, n, rng, n_sweeps, all_pairs)
            else:
                K = min(len(all_pairs), 6 * n)
                shuffle_idx = rng.choice(len(all_pairs), size=K, replace=False)
                xp = [all_pairs[k] for k in shuffle_idx]
                xo, so = sweep_pairs(st, n, rng, n_sweeps, xp)
            if is_feasible(xo, n):
                so = exact_score(xo, n)
                if so < best_s - 1e-12:
                    best_s, best_h = so, xo.copy()
        except Exception:
            pass
    try:
        ph, ps = nm_polish(best_h, n, rng)
        if is_feasible(ph, n):
            ps = exact_score(ph, n)
            if ps < best_s - 1e-12:
                best_s, best_h = ps, ph.copy()
    except Exception:
        pass
    try:
        xo2, so2 = sweep_pairs(best_h, n, rng, 6, all_pairs[:min(len(all_pairs), 4 * n)])
        if is_feasible(xo2, n):
            so2 = exact_score(xo2, n)
            if so2 < best_s - 1e-12:
                best_s, best_h = so2, xo2.copy()
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs), int(n)
