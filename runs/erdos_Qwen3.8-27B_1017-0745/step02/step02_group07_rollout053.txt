            if y is None:
                break
            d = y - h
            if np.max(np.abs(d)) < 1e-9:
                break
            yQ = lag_values(y, n)
            Fy = tau * np.log(np.sum(np.exp((yQ - np.max(yQ)) / tau))) + np.max(yQ)
            armijo = F + 1e-4 * np.dot(g, d) + 1e-14
            if Fy <= armijo:
                es = exact_score(y, n)
                if es < best_s - 1e-13:
                    best_s, best_h = es, y.copy()
                h = y
                Q = yQ
                F = Fy
                alpha = min(1.5 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_s = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    best_h, best_s = fb.copy(), fb_s
    starts = build_starts(fb, n, rng)
    mx_it = 220 if n <= 64 else (140 if n <= 128 else 80)
    for h0 in starts:
        if n <= 128:
            hr = slsqp_solve(h0, n, mx_it)
            for a in (0.0, 0.25, 0.5, 0.75, 1.0):
                if a == 0.0:
                    cand = h0
                elif a == 1.0 and hr is not None:
                    cand = hr
                elif hr is not None:
                    cand = (1 - a) * h0 + a * hr
                else:
                    continue
                pc = project_capped(cand, n)
                if pc is not None:
                    sc = exact_score(pc, n)
                    if sc < best_s - 1e-12:
                        best_s, best_h = sc, pc.copy()
        iters = 15 if n <= 128 else (25 if n <= 256 else 15)
        bh, bs = smooth_refine(h0, n, rng, iters)
        if bs < best_s - 1e-12:
            best_s, best_h = bs, bh
    bh, bs = smooth_refine(best_h, n, rng, 20)
    if bs < best_s - 1e-12:
        best_s, best_h = bs, bh
    if not is_feasible(best_h, n) or not np.isfinite(best_s):
        best_h, best_s = fb.copy(), fb_s
    best_h = best_h.astype(np.float64)
    final_s = exact_score(best_h, n)
    if not np.isfinite(final_s):
        best_h, final_s = fb.copy(), fb_s
    return best_h, float(final_s), int(n)
