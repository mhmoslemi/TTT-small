        model_val = float(np.max(q[act] + G @ e))
        pred2 = f - model_val
        rho = actual / max(pred2, 1e-12) if pred2 > 1e-12 else 0.0
        h, f = best_t, best_tf
        if f < bf - 1e-12:
            bf, bh = f, h.copy()
        if rho < 0.1:
            Delta = max(Delta * 0.5, Dmin)
        elif rho > 0.75 and np.max(np.abs(e)) >= 0.8 * Delta:
            Delta = min(1.5 * Delta, Dmax)
    return bh, bf

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n, dtype=np.float64)
    cands.append(np.clip(0.5 + 0.45 * np.cos(2 * np.pi * (j + 0.5) / n), 0, 1))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count: break
        p = project_capped(c, n)
        if p is None: continue
        if any(np.max(np.abs(p - s)) < 1e-8 for s in pool): continue
        pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
            n = int(p.size); parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None and parent.size != n:
        try:
            old_n = parent.size
            old_c = 0.5 + np.arange(old_n) / old_n
            new_c = 0.5 + np.arange(n) / n
            parent = np.interp(new_c, old_c, parent, left=parent[0], right=parent[-1])
        except Exception:
            parent = None
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None: fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    if n <= 64: n_it, n_st, n_rs, D0 = 80, 7, 3, 0.25
    elif n <= 128: n_it, n_st, n_rs, D0 = 45, 4, 2, 0.25
    elif n <= 256: n_it, n_st, n_rs, D0 = 25, 3, 1, 0.2
    elif n <= 512: n_it, n_st, n_rs, D0 = 12, 2, 1, 0.15
    else: n_it, n_st, n_rs, D0 = 6, 2, 0, 0.15
    starts = build_starts(fb, n, rng, n_st)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12: gs, gb = s, st.copy()
        try:
            bh, bf = local_search(st, n, n_it, D0)
            if bf < gs - 1e-12: gs, gb = bf, bh.copy()
        except Exception:
            pass
    sds = [0.12, 0.22, 0.08] if n <= 64 else ([0.10, 0.18] if n <= 256 else [0.08])
    for i in range(min(n_rs, len(sds))):
        try:
            pert = project_capped(gb + rng.normal(0, sds[i], n), n)
            if pert is None: continue
            bh, bf = local_search(pert, n, n_it, max(0.1, D0 * 0.6))
            if bf < gs - 1e-12: gs, gb = bf, bh.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss or sf == float('inf'):
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5); sf = exact_score(final, n)
    return final, float(sf), int(n)
