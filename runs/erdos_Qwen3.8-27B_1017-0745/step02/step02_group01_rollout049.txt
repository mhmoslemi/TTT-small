    if z.shape != (n,): return None
    if not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1e12,1e12)
    target=0.5*n
    lo=float(np.min(z))-1.0
    hi=float(np.max(z))+1.0
    for _ in range(100):
       mid=0.5*(lo+hi)

       if s>target: lo=mid

    h=np.clip(z-0.5*(lo+hi),0.0,1.0)

                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    return h if ok(h, n) else None

def score(h, n):
    if not ok(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))

def smooth_obj(z, beta):
    if not np.all(np.isfinite(z)):
        return 1e50
    n = z.size
    q = np.correlate(z, 1.0 - z, mode='full') * (2.0 / float(n))
    mx = float(np.max(q))
    ex = np.exp(beta * (q - mx))
    s = float(np.sum(ex))
    if not math.isfinite(s) or s <= 0.0:
        return mx
    return float(mx + math.log(s) / beta)

def smooth_jac(z, beta):
    if not np.all(np.isfinite(z)):
        return np.zeros(z.size)
    n = z.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    q = np.correlate(z, 1.0 - z, mode='full') * dx
    mx = float(np.max(q))
    ex = np.exp(beta * (q - mx))
    s = float(np.sum(ex))
    if not math.isfinite(s) or s <= 0.0:
        w = np.zeros(L)
        w[int(np.argmax(q))] = 1.0
    else:
        w = ex / s
    G = np.zeros(n)
    for d in range(1 - n, n):
        wt = float(w[d + n - 1])
        if wt <= 1e-10:
            continue
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        G[p0:p0 + m] += wt * dx * (1.0 - z[q0:q0 + m])
        G[q0:q0 + m] -= wt * dx * z[p0:p0 + m]
    return G

def mass_eq(z):
    return float(np.sum(z) - 0.5 * z.size)

def mass_grad(z):
    return np.ones(z.size)

def line_search(cur, n, scq, prop):
    best = None
    bs = scq
    for k in range(8):
        a = 0.5 ** k
        z = prop if k == 0 else project(cur + a * (prop - cur), n)
        if z is None:
            continue
        s = score(z, n)
        if s < bs - 1e-13:
            bs = s
            best = z.copy()
    return best, bs

def smooth_refine(h, n, betas=(10.0, 35.0, 110.0), max_it=140):
    cur = h.copy()
    scq = score(cur, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_grad}]
    for b in betas:
        try:
            res = minimize(smooth_obj, cur, args=(b,), jac=smooth_jac, bounds=bounds,
                           constraints=cons, method='SLSQP',
                           options={'maxiter': max_it, 'ftol': 1e-12})
        except Exception:
            break
        if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
            break
        y = project(res.x, n)
        if y is None:
            break
        z, ns = line_search(cur, n, scq, y)
        if z is not None:
            cur, scq = z, ns
    return cur, scq

def add_feasible(lst, h, n):
    p = project(h, n)
    if p is not None:
        lst.append(p)

def sinusoids(n):
    out = []
    if n < 4:
        return out
    i = np.arange(n, dtype=np.float64)
    tp = 2.0 * math.pi
    if n > 500:
        for amp in (0.42, 0.48):
            for ph in (0.0, 0.25, 0.5, 0.75):
                p = project(0.5 + amp * np.cos(tp * i / n + tp * ph), n)
                if p is not None:
                    out.append(p)
        return out
    if n <= 160:
        amps = (0.36, 0.42, 0.48, 0.50)
        freqs = (1.0, 2.0)
        nph = min(31, max(11, n))
    else:
        amps = (0.42, 0.48)
        freqs = (1.0,)
        nph = 13
    phases = np.linspace(0.0, 1.0, nph, endpoint=False)
    for f in freqs:
        for amp in amps:
            for ph in phases:
                add_feasible(out, 0.5 + amp * np.cos(tp * f * i / n + tp * ph), n)
        add_feasible(out, 0.5 + 0.44 * np.cos(tp * f * i / n) + 0.06 * np.cos(2.0 * tp * f * i / n), n)
        add_feasible(out, 0.5 + 0.44 * np.cos(tp * f * i / n) + 0.06 * np.cos(2.0 * tp * f * i / n + tp * 0.5), n)
    return out

def unique(h, pop):
    for s in pop:
        if np.max(np.abs(h - s)) < 1e-8:
            return False
    return True

def build_pop(scored, best, n, rng, P):
    pop = []
    sc = []
    p = project(best, n)
    if p is not None:
        pop.append(p)
        sc.append(score(p, n))
    for s, h in scored:
        if len(pop) >= P:
            break
        if unique(h, pop):
            p = project(h, n)
            if p is not None:
                pop.append(p)
                sc.append(score(p, n))
    while len(pop) < P:
        p = project(best + rng.normal(0.0, 0.15, n), n)
        if p is None or not unique(p, pop):
            p = project(rng.uniform(0.0, 1.0, n), n)
        if p is None:
            p = np.full(n, 0.5, dtype=np.float64)
        pop.append(p)
        sc.append(score(p, n))
    return pop, sc

def choice_excl(rng, P, excl):
    if P <= 1:
        return 0
    for _ in range(50):
        x = int(rng.integers(0, P))
        if x not in excl:
            return x
    return int(rng.integers(0, P))

def de_search(pop, sc, best_h, best_s, n, rng, budget):
    P = len(pop)
    if P < 5 or budget <= 0:
        return best_h, best_s
    arr = np.asarray(sc, dtype=float)
    order = np.argsort(arr)
    pop = [pop[int(i)].copy() for i in order]
    sc = [float(arr[int(i)]) for i in order]
    bi = 0
    best_s = min(best_s, sc[0])
    if sc[0] < best_s + 1e-15:
        best_h = pop[0].copy()
    attempts = 0
    gen = 0
    stag = 0
    resets = 0
    maxgen = max(1, budget // P + 15)
    while attempts < budget and gen < maxgen:
        gen += 1
        imp = False
        for i in range(P):
            if attempts >= budget:
                break
            attempts += 1
            f = 0.5 + 0.7 * float(rng.random())
            try:
                if rng.random() < 0.2:
                    r1 = choice_excl(rng, P, (i, bi))
                    r2 = choice_excl(rng, P, (i, bi, r1))
                    v = pop[i] + f * (pop[bi] - pop[i]) + f * (pop[r1] - pop[r2])
                else:
                    r1 = choice_excl(rng, P, (i, bi))
                    r2 = choice_excl(rng, P, (i, bi, r1))
                    r3 = choice_excl(rng, P, (i, bi, r1, r2))
                    v = pop[r1] + f * (pop[r2] - pop[r3])
            except Exception:
                continue
            if not np.all(np.isfinite(v)):
                continue
            x = pop[i].copy()
            mask = rng.random(n) < 0.7
            j = int(rng.integers(0, n))
            mask[j] = True
            x[mask] = v[mask]
            z = project(x, n)
            if z is None:
                continue
            s = score(z, n)
            if s < sc[i] - 1e-13:
                pop[i] = z
                sc[i] = s
            if s < best_s - 1e-13:
                best_h = z.copy()
                best_s = s
                bi = i
                imp = True
        if imp:
            stag = 0
        else:
            stag += 1
            if stag >= 12 and resets < 3:
                stag = 0
                resets += 1
                order2 = np.argsort(np.asarray(sc, dtype=float))[::-1]
                for idx in order2[max(2, P // 2):]:
                    if attempts >= budget:
                        break
                    attempts += 1
                    p = project(best_h + rng.normal(0.0, 0.2, n), n)
                    if p is None:
                        p = project(rng.uniform(0.0, 1.0, n), n)
                    if p is None:
                        p = np.full(n, 0.5, dtype=np.float64)
                    s = score(p, n)
                    idx = int(idx)
                    pop[idx] = p
                    sc[idx] = s
                    if s < best_s - 1e-13:
                        best_h = p.copy()
                        best_s = s
                        bi = idx
                        imp = True
            if stag > 30:
                break
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)):
                if np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                    parent = np.clip(p, 0.0, 1.0)
                else:
                    parent = project(p, n)
    except Exception:
        pass
    if n is None or n < 1 or n > 999:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if n == 1:
        h = np.array([0.5], dtype=np.float64)
        return h, score(h, 1), 1
    if parent is not None and parent.shape == (n,):
        fallback = project(parent, n)
        if fallback is None:
            fallback = np.full(n, 0.5, dtype=np.float64)
    else:
        fallback = np.full(n, 0.5, dtype=np.float64)
    if not ok(fallback, n):
        fallback = np.full(n, 0.5, dtype=np.float64)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    pool = []
    add_feasible(pool, fallback, n)
    add_feasible(pool, fallback[::-1], n)
    add_feasible(pool, 1.0 - fallback, n)
    add_feasible(pool, 1.0 - fallback[::-1], n)
    add_feasible(pool, np.full(n, 0.5, dtype=np.float64), n)
    b = np.zeros(n, dtype=np.float64)
    b[:n // 2] = 1.0
    if n % 2:
        b[n // 2] = 0.5
    add_feasible(pool, b, n)
    add_feasible(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64), n)
    for sd in (0.04, 0.10, 0.20):
        add_feasible(pool, fallback + rng.normal(0.0, sd, n), n)
        add_feasible(pool, 1.0 - fallback + rng.normal(0.0, sd, n), n)
    for hsi in sinusoids(n):
        if len(pool) < 1200:
            pool.append(hsi)

    scored = []
    for h in pool:
        s = score(h, n)
        if math.isfinite(s):
            scored.append((s, h))
    if not scored:
        fb = project(fallback, n)
        if fb is None:
            fb = np.full(n, 0.5, dtype=np.float64)
        scored = [(score(fb, n), fb)]
    scored.sort(key=key)
    best_h = scored[0][1].copy()
    best_s = float(scored[0][0])

    if n <= 160:
        if n <= 80:
            top_k = 4
            betas = (10.0, 35.0, 110.0)
            max_it = 140
        else:
            top_k = 3
            betas = (8.0, 25.0)
            max_it = 100
        top = [(best_s, best_h)]
        for s, h in scored:
            if len(top) >= top_k:
                break
            if unique(h, [best_h]):
                top.append((s, h))
        top.sort(key=key)
        for s, h in top:
            rh, rs = smooth_refine(h, n, betas=betas, max_it=max_it)
            if rs < best_s - 1e-12:
                best_h, best_s = rh, rs

    if n <= 80:
        P = 18
        budget = 3500
    elif n <= 160:
        P = 12
        budget = 1200
    elif n <= 300:
        P = 8
        budget = 350
    else:
        P = 8
        budget = 0

    if P >= 5 and budget > 0:
        pop, psc = build_pop(scored, best_h, n, rng, P)
        best_h, best_s = de_search(pop, psc, best_h, best_s, n, rng, budget)

    if n <= 160:
        if n <= 80:
            rh, rs = smooth_refine(best_h, n, betas=(20.0, 70.0), max_it=110)
        else:
            rh, rs = smooth_refine(best_h, n, betas=(15.0, 45.0), max_it=80)
        if rs < best_s - 1e-12:
            best_h, best_s = rh, rs

    fb = project(fallback, n)
    if fb is None or not ok(fb, n):
        fb = np.full(n, 0.5, dtype=np.float64)
    fs = score(fb, n)
    bh = project(best_h, n)
    if bh is None or not ok(bh, n):
        final = fb
        val = fs
    else:
        bs = score(bh, n)
        if bs < fs - 1e-12:
            final = bh
            val = bs
        else:
            final = fb
            val = fs
    if not ok(final, n):
        final = np.full(n, 0.5, dtype=np.float64)
        val = score(final, n)
    return final.astype(np.float64), float(val), int(n)
