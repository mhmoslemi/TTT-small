def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    rmax = np.max(r)
    w = np.exp((r - rmax) / tau)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(r))] = 1.0
        wsum = 1.0
    w /= wsum
    F = rmax + tau * np.log(wsum)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        d = idx - (n - 1)
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        grad[p0:p0+m] += w[idx] * dx * (1.0 - h[q0:q0+m])
        grad[q0:q0+m] -= w[idx] * dx * h[p0:p0+m]
    if not np.all(np.isfinite(F)) or not np.all(np.isfinite(grad)):
        return None, None
    return F, grad

def smooth_optimize(x0, n, iters_per_temp):
    x = x0.copy()
    best_x = x0.copy()
    best_s = exact_score(x0, n)
    temps = [0.03, 0.008, 0.002, 0.0004, 0.00008, 0.000015]
    for tau in temps:
        alpha = min(1.0, 0.15 * tau * n)
        if alpha < 1e-6:
            alpha = 1e-6
        F_cur, g_cur = smooth_grad(x, n, tau)
        if F_cur is None:
            break
        for _ in range(iters_per_temp):
            trial = project_capped(x - alpha * g_cur, n)
            if trial is None:
                break
            v = trial - x
            mv = np.max(np.abs(v))
            if mv < 1e-13:
                break
            F_t, g_t = smooth_grad(trial, n, tau)
            if F_t is None:
                break
            lin = F_cur + np.dot(g_cur, v) + np.dot(v, v) / (2.0 * alpha) + 1e-12
            if F_t <= lin:
                x = trial
                F_cur, g_cur = F_t, g_t
                alpha = min(1.0, 1.5 * alpha)
                s = exact_score(x, n)
                if s < best_s - 1e-12:
                    best_s, best_x = s, x.copy()
            else:
                alpha *= 0.5
                if alpha < 1e-15:
                    break
    return best_x, best_s

def pairwise_refine(h0, n, rng):
    h = h0.copy()
    s = exact_score(h, n)
    if n <= 64:
        pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
        meshes = [0.05, 0.02, 0.005, 0.001, 0.00025]
    else:
        budget = 128 if n <= 128 else (64 if n <= 256 else 16)
        half = min(budget // 2, n - 1)
        pair_list = [(i, i + 1) for i in range(half)]
        seen = set(pair_list)
        while len(pair_list) < budget:
            i = int(rng.integers(n)); j = int(rng.integers(n))
            if i == j:
                continue
            if i > j:
                i, j = j, i
            if (i, j) not in seen:
                seen.add((i, j))
                pair_list.append((i, j))
        meshes = [0.05, 0.015, 0.004] if n > 128 else [0.04, 0.01, 0.002]
    for dm in meshes:
        bh, bs = h.copy(), s
        for (i, j) in pair_list:
            c1 = min(dm, 1.0 - h[i], h[j])
            if c1 > 1e-12:
                t = h.copy(); t[i] += c1; t[j] -= c1
                sc = exact_score(t, n)
                if sc < bs - 1e-14:
                    bs, bh = sc, t.copy()
            c2 = min(dm, h[i], 1.0 - h[j])
            if c2 > 1e-12:
                t = h.copy(); t[i] -= c2; t[j] += c2
                sc = exact_score(t, n)
                if sc < bs - 1e-14:
                    bs, bh = sc, t.copy()
        if bs < s - 1e-12:
            h, s = bh, bs
    return h, s

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    P = max(2, n // 3)
    cands = [
        fallback[::-1].copy(),
        1.0 - fallback,
        (1.0 - fallback)[::-1].copy(),
        fallback + rng.normal(0, 0.05, n),
        fallback + rng.normal(0, 0.15, n),
        fallback + rng.normal(0, 0.30, n),
        0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P),
    ]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(np.asarray(c, dtype=np.float64), n)
        if p is None:
            continue
        dup = any(np.max(np.abs(p - s)) < 1e-8 for s in pool)
        if not dup:
            pool.append(p)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    fb = exact_score(fallback, n)
    if n == 1:
        return fallback, fb, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, iters_pt = 9, 80
    elif n <= 128:
        n_starts, iters_pt = 7, 50
    elif n <= 256:
        n_starts, iters_pt = 5, 30
    else:
        n_starts, iters_pt = 3, 15
    starts = build_starts(fallback, n, rng, n_starts)
    gb, gs = fallback.copy(), fb
    for st in starts:
        try:
            xo, so = smooth_optimize(st, n, iters_pt)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        pf, sf = pairwise_refine(fallback, n, rng)
        if sf < gs - 1e-12:
            gs, gb = sf, pf.copy()
    except Exception:
        pf = fallback
    if np.max(np.abs(gb - pf)) > 1e-8:
        try:
            pg, sg = pairwise_refine(gb, n, rng)
            if sg < gs - 1e-12:
                gs, gb = sg, pg.copy()
        except Exception:
            pass
    gf = project_capped(gb, n)
    if gf is None or not is_feasible(gf, n):
        gf = fallback.copy()
    fs = exact_score(gf, n)
    if fs == float('inf') or fs > fb + 1e-12:
        gf, fs = fallback.copy(), fb
    if fs == float('inf'):
        gf = np.full(n, 0.5)
        fs = exact_score(gf, n)
    return gf, float(fs), int(n)
