def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def build_fourier(n):
    F = (n - 1) // 2
    m = 2 * F + (1 if n % 2 == 0 else 0)
    Phi = np.zeros((n, m), dtype=np.float64)
    fr = np.zeros(m, dtype=int)
    ii = np.arange(n, dtype=np.float64)
    j = 0
    amp = math.sqrt(2.0 / n)
    for k in range(1, F + 1):
        w = 2.0 * math.pi * k * ii / n
        Phi[:, j] = amp * np.cos(w); fr[j] = k; j += 1
        Phi[:, j] = amp * np.sin(w); fr[j] = k; j += 1
    if n % 2 == 0:
        Phi[:, j] = ((ii % 2 == 0) * 2.0 - 1.0) * math.sqrt(1.0 / n)
        fr[j] = n // 2; j += 1
    return Phi, fr

def build_fourier(n):
    F = (n - 1) // 2; m = 2 * F + (1 if n % 2 == 0 else 0)
    Phi = np.zeros((n, m), dtype=np.float64); fr = np.zeros(m, dtype=int)
    ii = np.arange(n, dtype=np.float64); j = 0; amp = math.sqrt(2.0 / n)
    for k in range(1, F + 1):
        w = 2.0 * math.pi * k * ii / n
        Phi[:, j] = amp * np.cos(w); fr[j] = k; j += 1
        Phi[:, j] = amp * np.sin(w); fr[j] = k; j += 1
    if n % 2 == 0:
        Phi[:, j] = ((ii % 2 == 0) * 2.0 - 1.0) * math.sqrt(1.0 / n); fr[j] = n // 2

def direct_obj(z, n):
    return float(np.asarray(z, dtype=np.float64)[n])
def direct_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64); g[n] = 1.0; return g
def sum_h_eq(z, n):
    return float(np.sum(np.asarray(z, dtype=np.float64)[:n]) - 0.5 * n)
def sum_h_eq_jac(z, n):
    J = np.zeros(n + 1, dtype=np.float64); J[:n] = 1.0; return J
def direct_ineq(z, n):
    z = np.asarray(z, dtype=np.float64); h = np.clip(z[:n], 0.0, 1.0)
    return float(z[n]) - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
def direct_ineq_jac(z, n):
    z = np.asarray(z, dtype=np.float64); h = np.clip(z[:n], 0.0, 1.0)
    L = 2 * n - 1; J = np.zeros((L, n + 1), dtype=np.float64); J[:, n] = 1.0; dx = 2.0 / n
    for d in range(-(n - 1), n):

        J[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        J[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    return J

import numpy as np
from scipy.optimize import minimize
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    tol = 1e-14 * max(1, n)
    for _ in range(2):
        for i in range(n):
            if abs(r) <= tol:
                break
            if r > 0:
                d = min(r, 1.0 - h[i])
                if d > 0:
                    h[i] += d
                    r -= d
            else:
                d = min(-r, h[i])
                if d > 0:
                    h[i] -= d
                    r += d
    if not np.all(np.isfinite(h)) or np.any(h < 0.0) or np.any(h > 1.0):
        return None
    if abs(float(np.sum(h)) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def build_fourier(n):
    F = (n - 1) // 2
    m = 2 * F + (1 if n % 2 == 0 else 0)
    Phi = np.zeros((n, m), dtype=np.float64)
    fr = np.zeros(m, dtype=int)
    ii = np.arange(n, dtype=np.float64)
    j = 0
    amp = math.sqrt(2.0 / n)
    for k in range(1, F + 1):
        w = 2.0 * math.pi * k * ii / n
        Phi[:, j] = amp * np.cos(w)
        fr[j] = k
        j += 1
        Phi[:, j] = amp * np.sin(w)
        fr[j] = k
        j += 1
    if n % 2 == 0:
        Phi[:, j] = ((ii % 2 == 0) * 2.0 - 1.0) * math.sqrt(1.0 / n)
        fr[j] = n // 2
    return Phi, fr

def get_settings(n):
    maxlab = n // 2 if n % 2 == 0 else (n - 1) // 2
    cuts = []
    for k in ((1, 2, 4, 8, 16) if n <= 64 else (1, 2, 4, 8)):
        if 1 <= k <= maxlab and k not in cuts:
            cuts.append(k)
    if maxlab >= 1 and maxlab not in cuts:
        cuts.append(maxlab)
    if not cuts:
        cuts = [1]
    if n <= 64:
        return cuts, 3, 9, 2, 3
    if n <= 256:
        return cuts, 2, 7, 2, 2
    return cuts, 1, 5, 1, 2

def scan_coefficient(h, c, v, n, cur_s, grid, refn, rnds):

    ts = np.linspace(-R, R, grid, dtype=np.float64)
    if float(np.min(np.abs(ts - c))) > 1e-12:
        ts = np.sort(np.concatenate((ts, np.array([c]))))
    vals = []
    best_h = h
    best_s = cur_s if np.isfinite(cur_s) else exact_score(h, n)

        if ht is None:
            continue
        st = exact_score(ht, n)
        if not np.isfinite(st):
            continue
        vals.append((float(t), ht, float(st)))
        if st < best_s - 1e-12:

    if not vals:
        return best_h, best_s
    Lm = len(vals)
    mins = []
    for i in range(Lm):

        if i > 0 and si > vals[i - 1][2] + 1e-15:

        if i + 1 < Lm and si > vals[i + 1][2] + 1e-15:
            continue
        mins.append((si, i))
    if not mins:
        bi, bs = 0, vals[0][2]
        for i in range(1, Lm):
            if vals[i][2] < bs:
                bs, bi = vals[i][2], i
        mins = [(bs, bi)]
    mins.sort()
    refn = min(refn, len(mins))
    for q in range(refn):
        s0, idx = mins[q]
        t0, h0 = vals[idx][0], vals[idx][1]
        if idx == 0:
            low, high = -R, (vals[1][0] if Lm > 1 else R)
        elif idx == Lm - 1:

        br, bt, bh = s0, t0, h0

                high = pts[1]
            elif pos >= len(pts):
                low = pts[-2]
            else:
                low, high = pts[pos - 1], pts[pos]
    return best_h, best_s

def optimize_start(h0, phi, fr, n, rng, cuts, sweeps, grid, refn, rnds):
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    s = exact_score(h, n)
    inc_h, inc_s = h.copy(), s
    m = phi.shape[1]
    if m == 0:
        return inc_h, inc_s
    u = phi.T @ (h - 0.5)
    for K in cuts:
        active = [j for j in range(m) if fr[j] <= K]
        if not active:
            continue
        for _sw in range(sweeps):
            perm = rng.permutation(active)
            ok = False
            for j in perm:
                bh, bs = scan_coefficient(h, float(u[j]), phi[:, j], n, s, grid, refn, rnds)
                if bs < s - 1e-12:
                    h, s = bh, bs
                    u = phi.T @ (h - 0.5)
                    if s < inc_s - 1e-12:
                        inc_s, inc_h = s, h.copy()
                    ok = True
            if not ok:
                break
    return inc_h, inc_s

def make_starts(base, n, rng, count):
    if n == 1:
        return [np.array([0.5], dtype=np.float64)]
    st = []
    ii = np.arange(n, dtype=np.float64)
    cands = [base]
    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(1.0 - base)
    cands.append(base[::-1])
    r = max(1, n // 4)
    cands.append(np.roll(base, r))
    cands.append(np.roll(base, -r))
    for sd in (0.06, 0.18, 0.35):
        cands.append(base + sd * rng.normal(0.0, 1.0, n))
    for k in (1, 2, 3):
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        cands.append(0.5 + 0.3 * np.sin(2.0 * np.pi * k * ii / n + ph))
    cands.append(np.where(ii % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(rng.uniform(0.0, 1.0, n))
    for c in cands:
        if len(st) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for q in st:
            if float(np.max(np.abs(p - q))) < 1e-8:
                dup = True
                break
        if not dup:
            st.append(p)
    if not st:
        st.append(np.full(n, 0.5, dtype=np.float64))
    return st

def random_polish(h0, n, rng, trials, sigma):
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    s = exact_score(h, n)
    for it in range(trials):
        sig = sigma * (0.95 ** it)
        if sig < 1e-8:
            break
        hp = project_capped(h + sig * rng.normal(0.0, 1.0, n), n)
        if hp is None:
            continue
        sp = exact_score(hp, n)
        if sp < s - 1e-12:
            h, s = hp, sp
    return h, float(s)

def direct_obj(z, n):
    return float(np.asarray(z, dtype=np.float64)[n])

def direct_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64)
    g[n] = 1.0
    return g

def sum_h_eq(z, n):
    return float(np.sum(np.asarray(z, dtype=np.float64)[:n]) - 0.5 * n)

def sum_h_eq_jac(z, n):
    J = np.zeros(n + 1, dtype=np.float64)
    J[:n] = 1.0
    return J

def direct_ineq(z, n):
    z = np.asarray(z, dtype=np.float64)
    h = np.clip(z[:n], 0.0, 1.0)
    return float(z[n]) - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def direct_ineq_jac(z, n):
    z = np.asarray(z, dtype=np.float64)
    h = np.clip(z[:n], 0.0, 1.0)
    L = 2 * n - 1
    J = np.zeros((L, n + 1), dtype=np.float64)
    J[:, n] = 1.0
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        p0 = d if d > 0 else 0
        q0 = -d if d < 0 else 0
        m = n - abs(d)
        row = d + n - 1
        J[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        J[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    return J

def direct_polish(h0, n, maxiter):
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    s0 = exact_score(h, n)
    if not np.isfinite(s0):
        return h, float(s0)
    z0 = np.empty(n + 1, dtype=np.float64)
    z0[:n] = h
    z0[n] = s0 + 1e-7
    bounds = [(0.0, 1.0)] * n
    bounds.append((None, None))
    cons = [
        {'type': 'eq', 'fun': sum_h_eq, 'jac': sum_h_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': direct_ineq, 'jac': direct_ineq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(direct_obj, z0, jac=direct_jac, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    except Exception:
        return h, float(s0)
    if getattr(res, 'x', None) is None:
        return h, float(s0)
    xr = np.asarray(res.x, dtype=np.float64).ravel()
    if xr.shape != (n + 1,):
        return h, float(s0)
    hp = project_capped(xr[:n], n)
    if hp is None:
        return h, float(s0)
    sp = exact_score(hp, n)
    if np.isfinite(sp) and sp < s0 - 1e-12:
        return hp, float(sp)
    return h, float(s0)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-6) and np.all(p <= 1.0 + 1e-6):
                parent = project_capped(p, n)
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, exact_score(h1, 1), 1
    base = parent if parent is not None else np.full(n, 0.5, dtype=np.float64)
    safe_h = np.full(n, 0.5, dtype=np.float64)
    safe_s = exact_score(safe_h, n)
    if base is not None:
        bs = exact_score(base, n)
        if bs < safe_s - 1e-12:
            safe_h, safe_s = base, bs
    best_h, best_s = safe_h.copy(), float(safe_s)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    phi, fr = build_fourier(n)
    cuts, sweeps, grid, refn, rnds = get_settings(n)
    count = 4 if n <= 64 else (3 if n <= 256 else 2)
    starts = make_starts(base, n, rng, count)
    for idx, st in enumerate(starts):
        sw = sweeps + 2 if idx == 0 else sweeps
        try:
            xo, so = optimize_start(st, phi, fr, n, rng, cuts, sw, grid, refn, rnds)
        except Exception:
            xo, so = st, exact_score(st, n)
        if xo is None or not np.isfinite(so):
            continue
        if xo.shape != (n,):
            xo = project_capped(xo, n)
        if xo is not None and so < best_s - 1e-12:
            best_s, best_h = float(so), xo.copy()
    dmax = 80 if n <= 64 else 40
    try:
        dh, ds = direct_polish(best_h, n, dmax)
    except Exception:
        dh, ds = best_h, best_s
    if dh is not None and np.isfinite(ds) and ds < best_s - 1e-12:
        best_s, best_h = float(ds), dh.copy()
    try:
        rh, rs = random_polish(best_h, n, rng, 120, 0.05)
    except Exception:
        rh, rs = best_h, best_s
    if rh is not None and np.isfinite(rs) and rs < best_s - 1e-12:
        best_s, best_h = float(rs), rh.copy()
    try:
        xo, so = optimize_start(best_h, phi, fr, n, rng, cuts, max(2, sweeps), grid, refn, rnds)
    except Exception:
        xo, so = best_h, best_s
    if xo is not None and np.isfinite(so) and so < best_s - 1e-12:
        best_s, best_h = float(so), xo.copy()
    try:
        rh, rs = random_polish(best_h, n, rng, 80, 0.02)
    except Exception:
        rh, rs = best_h, best_s
    if rh is not None and np.isfinite(rs) and rs < best_s - 1e-12:
        best_s, best_h = float(rs), rh.copy()
    try:
        dh, ds = direct_polish(best_h, n, dmax)
    except Exception:
        dh, ds = best_h, best_s
    if dh is not None and np.isfinite(ds) and ds < best_s - 1e-12:
        best_s, best_h = float(ds), dh.copy()
    final = project_capped(best_h, n)
    if final is None:
        final = safe_h
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final = safe_h
        fs = exact_score(safe_h, n)
    return final.astype(np.float64), float(fs), int(n)
