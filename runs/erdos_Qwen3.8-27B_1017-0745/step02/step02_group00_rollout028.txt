def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape[0] != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_value_grad(h, tau, n):
    L = 2 * n - 1
    q = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        q[row] = (2.0 / n) * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    m = float(np.max(q))
    sh = (q - m) / tau
    ev = np.exp(np.clip(sh, -700.0, 0.0))
    sm = float(np.sum(ev))
    if sm < 1e-300:
        w = np.zeros(L)
        w[np.argmax(q)] = 1.0
        return m, np.zeros(n)
    w = ev / sm
    F = m + tau * np.log(sm)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        wd = (2.0 / n) * w[row]
        if abs(wd) < 1e-15:
            continue
        grad[lo:hi] += wd * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= wd * h[lo:hi]
    return float(F), grad

def pair_polish(h, n, sweeps):
    h = h.copy()
    deltas = [0.005, 0.015, 0.04, 0.1, 0.2, 0.35]
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d <= lo_d + 1e-12:
                    continue
                cur = exact_score(h, n)
                best_d, best_s = 0.0, cur
                cands = deltas + [-x for x in deltas] + [lo_d, hi_d]
                for dv in cands:
                    if dv == 0.0:
                        continue
                    dv2 = max(lo_d, min(hi_d, dv))
                    h[i] += dv2
                    h[j] -= dv2
                    sv = exact_score(h, n)
                    h[i] -= dv2
                    h[j] += dv2
                    if np.isfinite(sv) and sv < best_s - 1e-12:
                        best_s = sv
                        best_d = dv2
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        if not improved:
            break
    return h
