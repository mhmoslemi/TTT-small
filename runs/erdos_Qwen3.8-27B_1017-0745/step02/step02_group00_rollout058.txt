    v = np.asarray(v, dtype=np.float64).ravel()
    m = v.size
    if m == 0 or not np.all(np.isfinite(v)):
        return None

    rho = np.arange(1, m + 1)
    cond = u - (css - 1.0) / rho > 0
    j = int(np.max(np.where(cond)[0])) + 1

import numpy as np
from math import sqrt

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def dir_products(x, p, n):
    dx = 2.0 / n
    L = 2 * n - 1
    r = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        r[d + n - 1] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d]))
                              - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def weighted_grad(x, w, n):
    dx = 2.0 / n
    gbar = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        gbar[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        gbar[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return gbar

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    m = v.size
    if m == 0 or not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, m + 1, dtype=np.float64)
    cond = u - (css - 1.0) / rho > 0
    j = int(np.max(np.where(cond)[0])) + 1
    lam = (css[j - 1] - 1.0) / j
    out = np.maximum(v - lam, 0.0)
    r = 1.0 - float(np.sum(out))
    if abs(r) > 1e-14:
        idx = int(np.argmax(out))
        out[idx] += r
    if not np.all(np.isfinite(out)):
        return None
    return np.maximum(out, 0.0)

def frob_sq_G(x, n):
    dx = 2.0 / n
    total = 0.0
    for d in range(-(n - 1), n):
        g = np.zeros(n)
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += dx * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * x[lo:hi]
        total += float(np.dot(g, g))
    return total

def dual_solve(x, a, mu, w_init, n, max_iter):
    L = 2 * n - 1
    fsq = frob_sq_G(x, n)
    Ld = max(1.05 * fsq / mu, 1e-8)
    w = project_simplex(w_init)
    if w is None:
        w = np.zeros(L)
        w[int(np.argmax(a))] = 1.0
    z = w.copy()
    t = 1.0
    best_y, best_px = None, float('inf')
    no_change = 0
    for it in range(max_iter):
        gbar = weighted_grad(x, z, n)
        y = project_capped(x - gbar / mu, n)
        if y is None:
            y = x.copy()
        p = y - x
        r = dir_products(x, p, n)
        gradD = a + r
        p_val = float(np.max(a + r)) + 0.5 * mu * float(np.dot(p, p))
        d_val = float(np.dot(z, a)) + float(np.dot(gbar, p)) + 0.5 * mu * float(np.dot(p, p))
        gap = max(0.0, p_val - d_val)
        if np.isfinite(p_val) and p_val < best_px:

            break
        w_new = project_simplex(z + gradD / Ld)

            no_change = 0

        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        w, t = w_new, t_new
    return best_y, w

def polish_pairs(h, n, sweeps):
    h = h.copy()
    deltas = np.array([0.001, 0.005, 0.02, 0.05, 0.1, 0.2])
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d <= lo_d + 1e-12:
                    continue
                cur = exact_score(h, n)
                best_dv, best_s = 0.0, cur
                for dv in np.concatenate([deltas, -deltas, [lo_d, hi_d]]):
                    if dv == 0.0:
                        continue
                    dv = max(lo_d, min(hi_d, dv))
                    h[i] += dv
                    h[j] -= dv
                    sv = exact_score(h, n)
                    h[i] -= dv
                    h[j] += dv
                    if np.isfinite(sv) and sv < best_s - 1e-12:
                        best_s, best_dv = sv, dv
                if best_dv != 0.0:
                    h[i] += best_dv
                    h[j] -= best_dv
                    improved = True
        if not improved:
            break
    return h

def local_mm(x, n, outer_cap, dual_cap, mu_retries):
    x = x.copy()

    if not np.isfinite(cur_score):

    mu_min = 2.0 * dx
    mu = mu_min

    w[int(np.argmax(q))] = 1.0
    best_local, best_local_score = x.copy(), cur_score
    for _ in range(outer_cap):
        accepted = False
        for r in range(mu_retries):
            mu_t = mu * (2.0 ** r)
            y, w = dual_solve(x, q, mu_t, w, n, dual_cap)

                break

                    continue

                if np.isfinite(sv) and sv < best_local_score - 1e-12:
                    best_local, best_local_score = zt.copy(), sv

                    best_th = theta

    if cur_score < best_local_score - 1e-12:

    return best_local, best_local_score
