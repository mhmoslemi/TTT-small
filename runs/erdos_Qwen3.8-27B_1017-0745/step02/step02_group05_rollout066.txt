import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14 * max(1, n):
            break
        if r > 0:
            h[i] = min(h[i] + r, 1.0)
        else:
            h[i] = max(h[i] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def try_add(pool, cand, n):
    p = project_capped(cand, n)
    if p is None or not is_feasible(p, n):
        return
    for e in pool:
        if np.max(np.abs(p - e)) < 1e-8:
            return
    pool.append(p)

def build_population(base, n, rng, P):
    pool = []
    try_add(pool, base, n)
    try_add(pool, base[::-1].copy(), n)
    try_add(pool, 1.0 - base, n)

    try_add(pool, np.full(n, 0.5), n)

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)

def de_step(pop, scores, n, rng, gen, G, T0, CR, use_best_prob):
    P = len(pop)
    next_pop = [x.copy() for x in pop]
    next_sc = list(scores)

    for i in range(P):

        while b == a:
            b = int(cands[rng.integers(len(cands))])
