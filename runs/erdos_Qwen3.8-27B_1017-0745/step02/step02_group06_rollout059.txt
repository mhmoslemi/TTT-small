import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        lo1 = max(0, k); hi1 = min(n - 1, k + n - 1)
        if lo1 <= hi1:
            q1 = np.arange(lo1, hi1 + 1)
            J[idx, q1] += dx * (1.0 - h[q1 - k])
        lo2 = max(0, -k); hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            q2 = np.arange(lo2, hi2 + 1)
            J[idx, q2] -= dx * h[q2 + k]
    return J

def _obj(z):
    return z[-1]

def _obj_jac(z):
    g = np.zeros(z.shape[0])
    g[-1] = 1.0
    return g

def _ineq_fun(z, n):
    return z[-1] - lag_values(z[:-1], n)

def _ineq_jac(z, n):
    nv = z[:-1].shape[0]
    J = lag_jac(z[:-1], n)
    out = np.zeros((2 * n - 1, nv + 1))
    out[:, :nv] = -J
    out[:, -1] = 1.0
    return out

def _eq_fun(z, n):
    return np.sum(z[:-1]) - n / 2.0

def _eq_jac(z, n):
    g = np.ones(z.shape[0])
    g[-1] = 0.0
    return g

def solve_epigraph(h0, n, maxiter):
    z0 = np.append(h0.copy(), min(0.5, exact_score(h0, n) + 1e-9))
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None
    if hasattr(res, 'x') and res.x is not None and len(res.x) == n + 1:
        hc = project_capped(res.x[:-1], n)
        if hc is not None and is_feasible(hc, n):
            return hc
    return None

def build_pool(parent, n, rng):
    pool = []
    uni = np.full(n, 0.5)
    pool.append(uni)
    if parent is not None:
        pool.extend([parent.copy(), parent[::-1].copy(),

        if p is None:
            continue
        if any(np.max(np.abs(p - s)) < 1e-6 for s in seen):
            continue
        seen.append(p); out.append(p)
    return out
