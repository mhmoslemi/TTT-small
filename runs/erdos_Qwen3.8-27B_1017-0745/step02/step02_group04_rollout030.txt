            if ns == float('inf'):
                continue
            if ns < best_s - 1e-12:
                best_s = ns
                best_u = nu.copy()
            d = ns - cur_s
            if d <= 0:
                cur, cur_s = nu, ns
            elif T > 0 and d / T < 745:
                if rng.random() < math.exp(-d / T):
                    cur, cur_s = nu, ns
    return best_u, best_s

def build_starts(fallback, n, M, rng, count):
    pool = []
    seen = set()
    def add(hv):
        if hv is None or len(pool) >= count:
            return
        u = to_units(hv, n, M)
        if u is None:
            return
        key = u.tobytes()
        if key in seen:
            return
        seen.add(key)
        pool.append(u)
    add(fallback)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(project_capped(blk, n))
    P = max(2, n // 3)
    ix = np.arange(n, dtype=np.float64)
    add(project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * ix / P), n))
    add(project_capped(fallback + rng.normal(0, 0.08, n), n))
    add(project_capped(fallback + rng.normal(0, 0.18, n), n))
    add(project_capped(rng.uniform(0, 1, n), n))
    fb_u = to_units(fallback, n, M)
    if fb_u is not None:
        for vu in (fb_u[::-1].copy(), (M - fb_u).copy(), (M - fb_u)[::-1].copy()):
            key = vu.tobytes()
            if key not in seen:
                seen.add(key)
                pool.append(vu)
    const_u = np.full(n, M // 2, dtype=np.int64)
    key = const_u.tobytes()
    if key not in seen:
        seen.add(key)
        pool.append(const_u)
    add(project_capped(np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n))
    att = 0
    while len(pool) < count and att < 4 * count:
        add(project_capped(rng.uniform(0, 1, n), n))
        att += 1
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 0))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if n is None:
        n = 47
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, exact_score(h1, 1), 1
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    fb_score = exact_score(fallback, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        try:
            rng = np.random.default_rng(42)
        except Exception:
            return fallback, fb_score, n
    if n <= 64:
        ns, sp = 10, 700
    elif n <= 128:
        ns, sp = 6, 350
    elif n <= 256:
        ns, sp = 4, 150
    elif n <= 512:
        ns, sp = 2, 60
    else:
        ns, sp = 1, 16
    starts = build_starts(fallback, n, M_L, rng, ns)
    gb_s = fb_score
    gb_u = None
    for su in starts:
        ss = exact_score(units_to_h(su, M_L), n)
        if ss < gb_s - 1e-12:
            gb_s = ss
            gb_u = su.copy()
    for su in starts:
        try:
            bu, bs = anneal_one(su, n, M_L, rng, sp)
            if bs < gb_s - 1e-12:
                gb_s = bs
                gb_u = bu.copy()
        except Exception:
            pass
    if gb_u is not None:
        gh = units_to_h(gb_u, M_L)
        gs = exact_score(gh, n)
        if gs < fb_score:
            if is_feasible(gh, n):
                return gh, gs, n
            proj = project_capped(gh, n)
            if proj is not None:
                ps = exact_score(proj, n)
                if ps < fb_score:
                    return proj, ps, n
    return fallback, fb_score, n
