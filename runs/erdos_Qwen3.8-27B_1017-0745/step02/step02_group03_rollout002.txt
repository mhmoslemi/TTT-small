    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if 1.0 - h[idx] <= 1e-15:
                break
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * n:
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8 * n:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals(h, n):
    nv = 2 * n - 1
    v = np.zeros(nv)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        v[d + n - 1] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def phi_grad(h, n, theta):
    dx = 2.0 / n
    F = lag_vals(h, n)
    q = dx * F
    m = float(np.max(q))
    ex = np.exp(np.clip((q - m) / theta, -700, 0))
    sx = float(np.sum(ex))
    if sx < 1e-300:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        phi = m
    else:
        p = ex / sx
        phi = m + theta * float(np.log(sx))
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        w = p[d + n - 1] * dx
        if w < 1e-18:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += w * (1.0 - h[lo + d:hi + d])
        rl = max(0, lo + d)
        rh = min(n, hi + d)
        g[rl:rh] -= w * h[rl - d:rh - d]
    return phi, g

def cont_opt(h0, n, temps, mit):
    h = h0.copy()
    bh = h0.copy()
    bs = exact_score(h0, n)
    for tau in temps:
        alpha = 0.15
        for _ in range(mit):
            phi_h, grad = phi_grad(h, n, tau)
            if not np.all(np.isfinite(grad)) or not np.isfinite(phi_h):
                break
            accepted = False
            for _ in range(24):
                y = project_capped(h - alpha * grad, n)
                if y is None:
                    alpha *= 0.5
                    continue
                delta = y - h
                gd = float(np.dot(grad, delta))
                if gd > -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _ = phi_grad(y, n, tau)
                if phi_y <= phi_h + 1e-4 * gd + 1e-12 * (1.0 + abs(phi_h)):
                    h = y
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            alpha = min(0.35, alpha * 1.12)
            s = exact_score(h, n)
            if np.isfinite(s) and s < bs - 1e-13:
                bs = s
                bh = h.copy()
    return bh, bs

def add_start(lst, v, n, tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in lst:
        if float(np.max(np.abs(h2 - ex))) < tol:
            return
    lst.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        sd = abs(int(seed)) % (2 ** 31)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_s = exact_score(fallback, n)
    if not np.isfinite(fb_s):
        fallback = uniform.copy()
        fb_s = exact_score(fallback, n)
    temps = [0.025, 0.008, 0.0025, 0.0008, 0.00025, 0.00007, 0.00002, 0.000006]
    if n <= 128:
        mit, n_starts, n_hops = 90, 7, 4
    elif n <= 512:
        mit, n_starts, n_hops = 35, 4, 2
    else:
        mit, n_starts, n_hops = 12, 2, 1
    try:
        n_starts = max(1, min(9, int(kwargs.get('starts', n_starts))))
    except Exception:
        pass
    try:
        mit = max(1, min(140, int(kwargs.get('max_iter', mit))))
    except Exception:
        pass
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 4), n)
        add_start(starts, np.roll(fallback, -n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.035, n), n)
        add_start(starts, fallback + rng.normal(0, 0.12, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, 1.0 - fallback[::-1], n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best, best_s = fallback.copy(), fb_s
    for st in starts:
        s = exact_score(st, n)
        if np.isfinite(s) and s < best_s - 1e-13:
            best, best_s = st.copy(), s
    for st in starts:
        bh, bs = cont_opt(st, n, temps, mit)
        if bs < best_s - 1e-13:
            best, best_s = bh.copy(), bs
    hstds = [0.025, 0.07, 0.04, 0.1]
    for hs in hstds[:n_hops]:
        pert = project_capped(best + rng.normal(0, hs, n), n)
        if pert is None:
            continue
        ps = exact_score(pert, n)
        if np.isfinite(ps) and ps < best_s - 1e-13:
            best, best_s = pert.copy(), ps
        bh, bs = cont_opt(pert, n, temps, mit)
        if bs < best_s - 1e-13:
            best, best_s = bh.copy(), bs
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_s:
        final, fs = fallback.copy(), fb_s
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
