import numpy as np
from scipy.optimize import minimize, minimize_scalar

_LAG_CACHE = {}

def lag_cache(n):
    if n not in _LAG_CACHE:
        segs = [None] * (2 * n - 1)
        p = 0
        for d in range(-(n - 1), n):
            i0 = max(0, -d)
            i1 = min(n, n - d)
            if i0 < i1:
                i = np.arange(i0, i1, dtype=np.int64)
                segs[p] = (i, i + d)
            p += 1
        _LAG_CACHE[n] = segs
    return _LAG_CACHE[n]

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e8, 1e8)
    target = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(h))
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-11) or np.any(h > 1.0 + 1e-11):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-11) or np.any(h > 1.0 + 1e-11):
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-9 * max(1, n)

def score_fast(tmp, n):
    if tmp.shape != (n,) or not np.all(np.isfinite(tmp)):
        return float('inf')
    c = np.correlate(tmp, 1.0 - tmp, mode='full') * (2.0 / n)
    if c.size != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return score_fast(np.clip(h, 0.0, 1.0), n)

def lag_values(h, n):
    dx = 2.0 / n
    vals = np.zeros(2 * n - 1, dtype=np.float64)
    for p, seg in enumerate(lag_cache(n)):
        if seg is not None:
            i, j = seg
            vals[p] = dx * float(np.sum(h[j] * (1.0 - h[i])))
    return vals

def lag_jac(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n), dtype=np.float64)
    for p, seg in enumerate(lag_cache(n)):
        if seg is not None:
            i, j = seg
            J[p, j] += dx * (1.0 - h[i])
            J[p, i] += -dx * h[j]
    return J

def obj(z):
    return float(z[-1])

def obj_jac(z):
    g = np.zeros(z.shape, dtype=np.float64)
    g[-1] = 1.0
    return g

def ineq(z, n):
    return z[-1] - lag_values(z[:-1], n)

def ineq_jac(z, n):
    J = -lag_jac(z[:n], n)
    J[:, n] = 1.0
    return J

def eq(z, n):
    return float(np.sum(z[:n]) - 0.5 * n)

def eq_jac(z, n):
    J = np.empty(n + 1, dtype=np.float64)
    J[:n] = 1.0
    J[n] = 0.0
    return J

def solve_nlp(h0, n, maxiter):
    s = exact_score(h0, n)
    lv = lag_values(h0, n)
    t0 = min(0.55, max(s, float(np.max(lv)) if lv.size else 0.0) + 1e-7)
    z = np.empty(n + 1, dtype=np.float64)
    z[:n] = h0
    z[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.55)]
    cons = [
        {'type': 'eq', 'fun': eq, 'jac': eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(
            obj, z, method='SLSQP',
            jac=obj_jac, bounds=bounds, constraints=cons,
            options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False}
        )
    except Exception:
        return None
    if res.x is not None and len(res.x) == n + 1:
        hraw = np.asarray(res.x[:n], dtype=np.float64)
        if np.all(np.isfinite(hraw)):
            hp = project_capped(hraw, n)
            if hp is not None and is_feasible(hp, n):
                return hp
    return None

def process_candidate(best_h, best_s, cand, n):
    if cand is None or not is_feasible(cand, n):
        return best_h, best_s
    bs = exact_score(cand, n)
    ph = best_h
    ps = best_s
    if np.isfinite(bs) and bs < ps - 1e-12:
        ph = cand
        ps = bs

            ts = exact_score(tp, n)
            if np.isfinite(ts) and ts < ps - 1e-12:
                ph = tp
                ps = ts
    return ph, ps

def add_unique(pool, raw, n):
    if raw is None:
        return -1
    if not np.all(np.isfinite(raw)):
        return -1
    p = project_capped(raw, n)
    if p is None or not is_feasible(p, n):
        return -1

    if not np.isfinite(s) or s > 0.55:
        return -1
    for q in pool:
        if np.max(np.abs(p - q)) < 1e-6:
            return -1

    return len(pool) - 1

def pair_score(t, h, n, i, j):
    ht = h.copy()
    ht[i] = h[i] + t
    ht[j] = h[j] - t
    np.clip(ht, 0.0, 1.0, out=ht)
    return score_fast(ht, n)

def pair_best_move(h, n, i, j, base_s):
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-11:
        return 0.0, base_s
    best_t = 0.0
    best_s = base_s
    for q in range(13):
        t = L + (U - L) * q / 12.0
        s = pair_score(t, h, n, i, j)
        if s < best_s:
            best_s = s
            best_t = t
    if best_s < base_s - 1e-13:
        w = (U - L) / 6.0
        lo = max(L, best_t - w)
        hi = min(U, best_t + w)
        if hi - lo > 1e-10:
            try:
                r = minimize_scalar(
                    pair_score, args=(h, n, i, j),
                    bounds=(lo, hi), method='bounded',
                    options={'xatol': 1e-8}
                )
                if r.x is not None and np.isfinite(r.x):
                    t = float(r.x)
                    s = pair_score(t, h, n, i, j)
                    if s < best_s:
                        best_s = s
                        best_t = t
            except Exception:
                pass
    if best_s < base_s - 1e-13:
        return best_t, best_s
    return 0.0, base_s

def local_pair_polish(h, n, rng, moves, pairs_per_move):

    s = exact_score(h, n)
    if moves <= 0 or not np.isfinite(s):

                jj = jj[chosen]
            else:

            jj = rng.integers(0, n, pairs_per_move * 2)
            mask = ii < jj
            ii = ii[mask]
            jj = jj[mask]
        cnt = min(pairs_per_move, len(ii))

            i = int(ii[p])
            j = int(jj[p])
            t, ps = pair_best_move(h, n, i, j, s)
            if ps < best_s - 1e-13:
                best_s = ps
                best_t = t
                best_ij = (i, j)
        if best_ij is not None:
            i, j = best_ij
            ht = h.copy()
            ht[i] += best_t
            ht[j] -= best_t
            np.clip(ht, 0.0, 1.0, out=ht)
            hp = project_capped(ht, n)
            if hp is not None and is_feasible(hp, n):

                if np.isfinite(hs) and hs < s - 1e-13:
                    h = hp
                    s = hs
                    noimp = 0
                else:
                    noimp += 1
            else:
                noimp += 1
        else:
            noimp += 1
        if noimp >= 3:
            break
    return h, s
