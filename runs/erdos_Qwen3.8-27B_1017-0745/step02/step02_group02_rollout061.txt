```python
def lag_gradient(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        p = max(s, 0)
        r = max(-s, 0)
        m = n - abs(s)
        if w[k] == 0.0:
            continue
        grad[p:p+m] += dx * w[k] * (1.0 - h[r:r+m])
        grad[r:r+m] -= dx * w[k] * h[p:p+m]
    return grad
```

```python
        np_ = (np.arange(n) + 0.5) / n
        parent = np.interp(np_, op, parent, left=parent[0], right=parent[-1])
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n <= 64:
        nsc, nmi, npr = 5, 140, 2
    elif n <= 128:
        nsc, nmi, npr = 4, 90, 1
    elif n <= 256:
        nsc, nmi, npr = 3, 55, 1
    else:
        nsc, nmi, npr = 2, 25, 0
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    starts = build_starts(n, rng, fb)
    sc_list = []
    for i in range(len(starts)):
        sc_list.append((exact_score(starts[i], n), i))
    sc_list.sort(key=lambda x: x[0])
    for _, idx in sc_list[:nsc]:
        try:
            bh, bs = cont_search(starts[idx], n, taus, nmi)
            if bs < gs - 1e-12:
                gs, gb = bs, bh.copy()
        except Exception:
            pass
    if npr > 0:
        if n <= 64:
            sds = (0.08, 0.18) if gs > fs - 1e-5 else (0.03, 0.08)
        else:
            sds = (0.12,) if gs > fs - 1e-5 else (0.05,)
        for sd in sds[:npr]:
            try:
                pert = project_capped(gb + rng.normal(0, sd, n), n)
                if pert is None:
                    continue
                bh, bs = cont_search(pert, n, taus, nmi)
                if bs < gs - 1e-12:
                    gs, gb = bs, bh.copy()
            except Exception:
                continue
    fin = project_capped(gb, n)
    if fin is None or not is_feasible(fin, n):
        fin = fb
    sfin = exact_score(fin, n)
    if sfin > fs:
        fin, sfin = fb, fs
    if not is_feasible(fin, n):
        fin = np.full(n, 0.5)
        sfin = exact_score(fin, n)
    return fin, float(sfin), int(n)
```
