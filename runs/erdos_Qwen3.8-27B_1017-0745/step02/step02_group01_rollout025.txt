    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0); q0 = max(-d, 0); m = n - abs(d)

        G[row, q0:q0+m] -= dx * h[p0:p0+m]
    J = np.zeros((L, n + 1))
    J[:, :n] = -G; J[:, n] = 1.0
    return J

            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1 or n > 999:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    P = min(64, max(20, round(64 * math.sqrt(47.0 / n))))
    budget = max(2*P, min(6000, int(math.floor(6000 * (47.0/n)**2))))
    pop = build_pop(fallback, n, rng, P)
    scores = np.array([exact_score(pop[i], n) for i in range(P)])
    fb_score = float(exact_score(fallback, n))
    best_h = fallback.copy(); best_score = fb_score
    for i in range(P):
        if scores[i] < best_score - 1e-12:
            best_score = float(scores[i]); best_h = pop[i].copy()
    best_h, best_score = de_evolve(pop, scores, P, n, rng, budget, best_h, best_score)
    for _ in range(3):
        polished = slsqp_polish(best_h, n, max_iter=250)
        if polished is not None:
            ps = exact_score(polished, n)
            if ps < best_score - 1e-12:
                best_score = ps; best_h = polished
            else:
                break
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = float(exact_score(final, n))
    fb_s = float(exact_score(fallback, n))
    if fs > fb_s - 1e-12:
        final, fs = fallback, fb_s
    return final, float(fs), int(n)
