    _ctx['s'] = n / 2.0
    q0 = lag_q(x, n)
    t0 = float(np.max(q0)) + 1e-6
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    y = project_capped(res.x[:n], n)
    return y

def build_starts(fallback, n, rng, count):
    pool = []
    idx = np.arange(n)
    cands = []
    cands.append(fallback.copy())
    cands.append(fallback[::-1].copy())
    cands.append(1.0 - fallback)
    cands.append((1.0 - fallback)[::-1].copy())
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    alt = np.where(idx % 2 == 0, 1.0, 0.0).astype(np.float64)
    if n % 2 == 1:
        alt[n // 2] = 0.5
    cands.append(alt)
    cands.append(fallback + rng.normal(0, 0.04, n))
    cands.append(fallback + rng.normal(0, 0.10, n))
    cands.append(fallback + rng.normal(0, 0.20, n))
    cands.append(rng.uniform(0, 1, n))
    P1 = max(3, n // 3)
    cands.append(0.5 + 0.45 * np.cos(2.0 * np.pi * idx / P1))
    P2 = max(3, n // 5)
    cands.append(0.5 + 0.40 * np.cos(2.0 * np.pi * 2 * idx / P2))
    P3 = max(3, n // 7)
    cands.append(0.5 + 0.35 * np.sin(2.0 * np.pi * idx / P3))
    for c in cands:
        if len(pool) >= count:
            break
        cp = project_capped(np.asarray(c, dtype=np.float64), n)
        if cp is None:
            continue
        dup = False
        for s in pool:
            if float(np.max(np.abs(cp - s))) < 1e-6:
                dup = True
                break
        if not dup:
            pool.append(cp)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    rng = np.random.default_rng(int(seed))
    if n <= 64:
        n_starts, maxiter, n_hops = 12, 300, 14
    elif n <= 128:
        n_starts, maxiter, n_hops = 8, 200, 7
    else:
        n_starts, maxiter, n_hops = 4, 120, 3
    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-14:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            y = solve_slsqp(st, n, maxiter)
        except Exception:
            y = None
        if y is not None and is_feasible(y, n):
            sy = exact_score(y, n)
            if sy < gs - 1e-14:
                gs, gb = sy, y.copy()
    current = gb.copy()
    cur_score = gs
    sigmas = [0.03, 0.06, 0.10, 0.18, 0.28, 0.08, 0.14]
    for r in range(n_hops):
        sigma = sigmas[r % len(sigmas)]
        if r % 5 == 4:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            prop = project_capped(current + sigma * rng.standard_normal(n), n)
        if prop is None:
            continue
        try:
            y = solve_slsqp(prop, n, maxiter)
        except Exception:
            y = None
        if y is None or not is_feasible(y, n):
            continue
        sy = exact_score(y, n)
        if sy < gs - 1e-14:
            gs, gb = sy, y.copy()
        if sy <= cur_score + 1e-15:
            current, cur_score = y.copy(), sy
        else:
            T = max(1e-5, 2.0e-3 * (0.72 ** r))
            if rng.random() < math.exp(-(sy - cur_score) / T):
                current, cur_score = y.copy(), sy
    for _ in range(3):
        try:
            y = solve_slsqp(gb + rng.normal(0, 0.01, n), n, maxiter // 2)
        except Exception:
            y = None
        if y is not None and is_feasible(y, n):
            sy = exact_score(y, n)
            if sy < gs - 1e-14:
                gs, gb = sy, y.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = exact_score(gb, n)
    fb = exact_score(fallback, n)
    if fs > fb + 1e-14:
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = exact_score(gb, n)
    return gb, fs, int(n)
