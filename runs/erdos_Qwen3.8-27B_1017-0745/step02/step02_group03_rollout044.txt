        # m0_act*t + c0_act = -2t² + b0*t + c0
        # 2t² + (m0_act - b0)*t + (c0_act - c0) = 0
        roots = quad_roots_in(2.0, m0_act - b0, c0_act - c0, seg_L, seg_U)
        seg_cands.extend(roots)
        
        candidates.extend(seg_cands)
    
    # Evaluate all candidates
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        if t < L - 1e-12 or t > U + 1e-12:
            continue
        t = max(L, min(U, t))
        # Evaluate max of the three functions
        v_m2 = -2.0 * t * t + b0 * t + c0
        v_0 = active_val(hull0, brk0, t)
        v_1 = t * t + active_val(hull1, brk1, t)
        val = max(v_m2, v_0, v_1)
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
        elif abs(val - best_val) <= 1e-14 and abs(t) < abs(best_t):
            best_t = t
    
    return best_t

    if not hull:

def quad_roots_in(A, B, C, lo, hi):
    roots = []
    if abs(A) < 1e-15:
        if abs(B) < 1e-15:
            return roots
        r = -C / B
        if lo <= r <= hi:
            roots.append(r)
        return roots
    disc = B * B - 4.0 * A * C
    if disc < 0:

            disc = 0.0
        else:

        r2 = C / q

    for r in (r1, r2):
        if lo - 1e-12 <= r <= hi + 1e-12:
            roots.append(max(lo, min(hi, r)))
    return roots
