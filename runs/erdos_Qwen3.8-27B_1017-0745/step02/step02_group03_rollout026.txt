                    if not (lo <= trial_d <= hi): continue

                    s = exact_score(t, n)
                    if s < best_sc - 1e-12:

                if best_d != 0.0:
                    best[i] += best_d; best[j] -= best_d
                    bs = best_sc; imp = True
