        max_iter, n_starts, shake_rounds = 45, 5, 3
    elif n <= 512:
        max_iter, n_starts, shake_rounds = 20, 3, 1
    else:
        max_iter, n_starts, shake_rounds = 10, 2, 1
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', max_iter)), 100))
    except Exception:
        pass
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts)), 8))
    except Exception:
        pass
    try:
        shake_rounds = max(0, min(int(kwargs.get('shake_rounds', shake_rounds)), 6))
    except Exception:
        pass
    starts = starts[:n_starts]

    for st in starts:
        try:
            best, best_score = local_opt(st, n, max_iter, best, best_score)
        except Exception:
            pass

    sigma = 0.04
    for _ in range(shake_rounds):
        try:
            pert = project_capped(best + sigma * rng.normal(0, 1, n), n)
            if pert is None:
                sigma = min(0.25, 1.5 * sigma)
                continue
            nb, ns = local_opt(pert, n, max_iter, best, best_score)
            if ns < best_score - 1e-12:
                best, best_score = nb, ns
                sigma = max(0.01, 0.75 * sigma)
            else:
                sigma = min(0.3, 1.5 * sigma)
        except Exception:
            pass

    final = project_capped(best, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best.copy()
        fs = best_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(uniform, n)
    return final, float(fs), int(n)
