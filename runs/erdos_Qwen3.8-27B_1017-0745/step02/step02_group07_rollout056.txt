        G[idx + 1, :m] += dx * (-h[r:])
        G[idx + 1, r:] += dx * (1.0 - h[:m])
        idx += 2
    return G

                if Fy <= F + 1e-4 * np.dot(g, d) + 1e-14:
                    accepted = True
                    break
                alpha *= 0.5
                y = project_capped(h - alpha * g, n)
                if y is None:
                    break
                d = y - h
            if not accepted or y is None:
                break
            h = y
            alpha = min(1.5 * alpha, 10.0)
            s = exact_score(h, n)
            if s < best_s - 1e-12:
                best_s = s
                best_h = h.copy()
    return best_h, best_s

def try_add(pool, v, th=1e-9):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)
    v[p[:n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v

def build_starts(fb, n, rng, max_starts):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, project_capped(np.full(n, 0.5), n))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    try_add(pool, project_capped(make_binary(rng, n), n))
    try_add(pool, project_capped(fb + rng.normal(0, 0.12, n), n))
    try_add(pool, project_capped(fb + rng.normal(0, 0.25, n), n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    try_add(pool, project_capped(np.roll(fb, n // 2), n))
    i = np.arange(n, dtype=np.float64)
    for k in (2, 3, 4):
        for ph in (0.0, np.pi / 2):
            try_add(pool, project_capped(0.5 + 0.4 * np.cos(2.0 * np.pi * k * i / n + ph), n))
    while len(pool) < max_starts:
        try_add(pool, project_capped(rng.uniform(0, 1, n), n))
        if len(pool) < max_starts:
            pool.append(pool[0].copy())
    return pool[:max_starts]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_s = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), float(exact_score(np.array([0.5]), 1)), 1
    best_h = fb.copy()
    best_s = fb_s
    max_starts = 10 if n <= 64 else (7 if n <= 128 else 4)
    starts = build_starts(fb, n, rng, max_starts)
    start_scores = [exact_score(sv, n) for sv in starts]
    if n <= 128:
        maxiter = 220 if n <= 64 else 140
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        ineq_con = {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)}
        eq_con = {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
        for si in range(len(starts)):
            h0 = starts[si]
            t0 = max(start_scores[si], 0.01) + 1e-8
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(slsqp_obj, z0, method='SLSQP',
                               jac=slsqp_obj_grad, args=(n,),
                               bounds=bounds,
                               constraints=[ineq_con, eq_con],
                               options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
            except Exception:
                continue
            if res is None or not hasattr(res, 'x') or res.x is None:
                continue
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if len(x) != n + 1 or not np.all(np.isfinite(x)):
                continue
            hr = project_capped(x[:n], n)
            if hr is None:
                continue
            for a in (1.0, 0.75, 0.5, 0.25):
                mix = (1.0 - a) * h0 + a * hr
                cm = project_capped(mix, n)
                if cm is None:
                    continue
                cs = exact_score(cm, n)
                if cs < best_s - 1e-12:
                    best_s = cs
                    best_h = cm.copy()
    iters = 12 if n <= 128 else (30 if n <= 256 else 20)
    polish_list = [(best_h.copy(), best_s)]
    order = np.argsort(start_scores)
    for oi in order[:2]:
        if not np.allclose(starts[oi], best_h, atol=1e-10):
            polish_list.append((starts[oi].copy(), start_scores[oi]))
    for pi in range(len(polish_list)):
        rh, rs = smooth_refine(polish_list[pi][0], n, iters)
        if rs < best_s - 1e-12:
            best_s = rs
            best_h = rh.copy()
    if not is_feasible(best_h, n):
        best_h = fb.copy()
        best_s = fb_s
    final_s = exact_score(best_h, n)
    if not np.isfinite(final_s):
        best_h = fb.copy()
        final_s = fb_s
    return best_h.astype(np.float64), float(final_s), int(n)
