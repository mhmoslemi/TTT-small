    if h.shape != (n,) or not np.all(np.isfinite(h)):

        return False

        return False
    return True

        if np.sum(v) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        return None

            z[j] -= t
