        return None

    lo = np.min(z) - 1.0

        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(h)

            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

    return h

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_val_grad(h, n, tau):

        r = max(-s, 0)

        g[p:p + m] += dx * wk * (1.0 - h[r:r + m])
        g[r:r + m] -= dx * wk * h[p:p + m]
    return F, g
