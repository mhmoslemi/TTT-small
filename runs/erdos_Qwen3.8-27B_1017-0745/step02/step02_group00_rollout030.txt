def project_simplex(v):
    m = v.size
    if not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    cond = u[rho] - (css[rho] - 1.0) / (rho + 1.0)
    j = np.where(cond > 0.0)[0]

        return np.full(m, 1.0 / m)

    lam = (css[j] - 1.0) / (j + 1.0)

import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    if float(np.max(np.abs(z))) > 1e12:
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return v if np.isfinite(v) else float('inf')

def lag_values(x, n):
    m, dx = 2 * n - 1, 2.0 / n
    q = np.zeros(m)
    for r in range(m):
        d = r - (n - 1)
        lo, hi = max(0, -d), min(n, n - d)
        q[r] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def dir_products(x, p, n):
    m, dx = 2 * n - 1, 2.0 / n
    r = np.zeros(m)
    for idx in range(m):
        d = idx - (n - 1)
        lo, hi = max(0, -d), min(n, n - d)
        r[idx] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d]))
                       - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def wgrad(x, w, n):
    m, dx = 2 * n - 1, 2.0 / n
    g = np.zeros(n)
    for idx in range(m):
        wt = w[idx]
        if wt == 0.0:
            continue
        d = idx - (n - 1)
        lo, hi = max(0, -d), min(n, n - d)
        g[lo:hi] += dx * wt * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wt * x[lo:hi]
    return g

def project_simplex(v):
    m = v.size
    if not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u > 0.0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    cond = u[rho] - (css[rho] - 1.0) / (rho + 1.0)
    j = np.where(cond > 0.0)[0]
    if len(j) == 0:
        return np.full(m, 1.0 / m)
    j = j[-1]
    lam = (css[j] - 1.0) / (j + 1.0)
    out = np.maximum(v - lam, 0.0)
    res = 1.0 - float(np.sum(out))
    if abs(res) < 1e-10:
        pos = np.where(out > 0.0)[0]
        if len(pos) > 0:
            out[pos[0]] += res
    return out

def frob_sq(x, n):
    m, dx = 2 * n - 1, 2.0 / n
    total = 0.0
    for idx in range(m):
        d = idx - (n - 1)
        lo, hi = max(0, -d), min(n, n - d)
        g = np.zeros(n)
        g[lo:hi] += dx * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * x[lo:hi]
        total += float(np.dot(g, g))
    return total

def dual_solve(x, a, mu, w_init, n, K):
    m = 2 * n - 1
    try:
        LD = max(1.05 * frob_sq(x, n) / mu, 1e-8)
    except Exception:
        LD = max(8.0 / mu, 1e-8)
    w = project_simplex(np.asarray(w_init, dtype=np.float64).copy())
    if w is None or not np.all(np.isfinite(w)):
        w = np.zeros(m)
        w[int(np.argmax(a))] = 1.0
    z = w.copy()
    t = 1.0
    best_cand, best_pv = None, float('inf')
    for _ in range(K):
        gbar = wgrad(x, z, n)
        y = project_capped(x - gbar / mu, n)
        if y is None:
            y = x.copy()
        p = y - x
        rp = dir_products(x, p, n)
        gradD = a + rp
        pv = float(np.max(a + rp) + 0.5 * mu * float(np.dot(p, p)))
        if np.isfinite(pv) and pv < best_pv:
            best_pv = pv
            best_cand = y.copy()
        dv = float(np.dot(z, a) + float(np.dot(gbar, p)) + 0.5 * mu * float(np.dot(p, p)))
        gap = max(0.0, pv - dv) if np.isfinite(dv) else 1.0
        if gap <= 1e-8 * max(1.0, abs(pv)):
            break
        w_new = project_simplex(z + gradD / LD)
        if w_new is None or not np.all(np.isfinite(w_new)):
            break
        if np.max(np.abs(w_new - w)) <= 1e-11:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        w, t = w_new, t_new
    if best_cand is not None:
        return best_cand, w
    gbar = wgrad(x, w, n)
    y = project_capped(x - gbar / mu, n)
    return (y if y is not None else None), w

def mm_optimize(x0, n, max_outer, dual_iter, mu_retries):
    dx = 2.0 / n
    mu_min = 2.0 * dx
    h = x0.copy()
    score = exact_score(h, n)
    if not np.isfinite(score):
        return h, score
    a = lag_values(h, n)
    mu = mu_min
    w_prev = None
    for _ in range(max_outer):
        accepted = False
        for retry in range(mu_retries):
            mu_try = mu * (2.0 ** retry)
            if mu_try < mu_min:
                mu_try = mu_min
            wi = w_prev.copy() if w_prev is not None else np.zeros(2 * n - 1)
            if float(np.sum(wi)) < 1e-15:
                wi[int(np.argmax(a))] = 1.0
            y, w_new = dual_solve(h, a, mu_try, wi, n, dual_iter)
            if y is None or not np.all(np.isfinite(y)):
                continue
            if float(np.linalg.norm(y - h)) <= 1e-10 * max(1.0, np.sqrt(n)):
                break
            best_loc, best_th = score, 0.0
            for j in range(9):
                th = 2.0 ** (-j)
                zp = project_capped((1.0 - th) * h + th * y, n)
                if zp is None:
                    continue
                sc = exact_score(zp, n)
                if np.isfinite(sc) and sc < best_loc - 1e-12:
                    best_loc, best_th = sc, th
                    h, score = zp.copy(), sc
                    w_prev = w_new.copy()
                    accepted = True
            if best_th >= 1.0:
                mu = max(mu_min, 0.75 * mu_try)
            elif best_th > 0.0:
                mu = mu_try
            if best_th > 0.0:
                a = lag_values(h, n)
                break
        if not accepted:
            break
    return h, score
