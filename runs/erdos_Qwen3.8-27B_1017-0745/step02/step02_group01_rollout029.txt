import math

    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

def project_capped(z, n):

def exact_score(h, n):

def build_directions(n):

def line_interval(h, v):

        return None

def line_coefficients(h, v, n):
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))

def append_roots(A, B, C, lo, hi, out):
    scale = max(abs(A), abs(B), abs(C), 1e-300)
    tol = 1e-14 * scale
    if abs(A) <= tol:

            r = -C / B

        return
    disc = B * B - 4.0 * A * C
    thr = 1e-13 * (B * B + abs(4.0 * A * C) + 1e-300)
    if disc < -thr:
        return
    if disc < 0:
        disc = 0.0
    s = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(s, B))
    roots = []
    if abs(q) > 1e-300:
        roots = [q / A, C / q]
    else:
        r = -B / (2.0 * A)
        roots = [r]

        if np.isfinite(r) and lo - 1e-12 * max(1, abs(lo)) <= r <= hi + 1e-12 * max(1, abs(hi)):

def line_candidates(a, b, c, lo, hi):

    La = a.shape[0]
