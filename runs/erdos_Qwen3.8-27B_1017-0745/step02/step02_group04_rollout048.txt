import numpy as np
import math
from scipy.optimize import minimize

_ctx = {}
_sctx = {}
_M = 65536

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(h.sum())
    if r > 1e-14:
        for i in np.argsort(-(1.0 - h)):
            if r <= 1e-14:
                break
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
    elif r < -1e-14:
        for i in np.argsort(h):
            if r >= -1e-14:
                break
            a = min(-r, h[i])
            h[i] -= a
            r += a
    tol = 1e-9 * max(1, n)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(h.sum() - s) > tol:
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    return (h.shape == (n,) and bool(np.all(np.isfinite(h))) and
            bool(np.all(h >= -1e-10)) and bool(np.all(h <= 1.0 + 1e-10)) and
            abs(h.sum() - 0.5 * n) <= 1e-9 * max(1, n))

def raw_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))

def _eq_sum(z):
    return np.array([float(z.sum()) - _sctx["s"]])

def _eq_sum_jac(z):
    n = _sctx["n"]
    J = np.zeros((1, n))
    J[0, :] = 1.0
    return J

def _so_obj(z):
    n = _sctx["n"]
    b = _sctx["b"]
    c = np.correlate(z, 1.0 - z, mode="full") * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return 1e100
    mx = float(c.max())
    e = np.exp(b * (c - mx))
    return float(mx + np.log(float(e.sum())) / b)

def _so_jac(z):
    n = _sctx["n"]
    b = _sctx["b"]
    dx = 2.0 / n
    L = 2 * n - 1
    c = np.correlate(z, 1.0 - z, mode="full") * dx
    mx = float(c.max())
    e = np.exp(b * (c - mx))
    w = e / float(e.sum())
    h = z
    g = np.zeros(n)
    for idx in range(L):
        d = idx - n + 1
        wd = w[idx] * dx
        if d == 0:
            g += wd * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            g[:m] += wd * (1.0 - h[d:d + m])
            g[d:d + m] -= wd * h[:m]
        else:
            l = -d
            m = n - l
            g[l:l + m] += wd * (1.0 - h[:m])
            g[:m] -= wd * h[l:l + m]
    return g

def solve_smooth(h, n, beta, maxiter):
    if maxiter <= 0:
        return None
    h0 = project_capped(h, n)
    if h0 is None:
        return None
    _sctx["n"] = n
    _sctx["b"] = float(beta)
    _sctx["s"] = 0.5 * n
    cons = [{"type": "eq", "fun": _eq_sum, "jac": _eq_sum_jac}]
    try:
        r = minimize(_so_obj, h0.copy(), jac=_so_jac, method="SLSQP",
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
    except Exception:
        return None
    if r.x is None or not np.all(np.isfinite(r.x)):
        return None
    return project_capped(r.x, n)

def to_units(h, n):
    if not is_feasible(h, n):
        return None
    M = _M
    K = M * n // 2
    z = h * M
    base = np.floor(z).astype(np.int64)
    frac = z - base.astype(np.float64)
    R = K - int(base.sum())
    if R > 0:
        rem = R
        for i in np.argsort(frac)[::-1]:
            if rem <= 0:
                break
            if base[i] < M:
                base[i] += 1
                rem -= 1
        if rem:
            return None
    elif R < 0:
        rem = R
        for i in np.argsort(frac):
            if rem >= 0:
                break
            if base[i] > 0:
                base[i] -= 1
                rem += 1
        if rem:
            return None
    if np.any(base < 0) or np.any(base >= M) or int(base.sum()) != K:
        return None
    return base

def propose_units(u, n, rng, qmax):

        return None
    sizes, ws = [], []
    for s in (2, 4, 8):
        if s <= n:

            ws.append(0.55 if s == 2 else 0.30 if s == 4 else 0.15)
    if not sizes:
        return None
    w = np.array(ws, dtype=np.float64)
    w /= w.sum()
    k = int(rng.choice(sizes, p=w))
    idx = rng.choice(n, size=k, replace=False)
    half = k // 2
    pos = np.zeros(k, dtype=bool)
    pos[rng.choice(k, size=half, replace=False)] = True

        q = qmax
    v = u.copy()
    v[gains] += q
    v[losses] -= q
    if np.any(v < 0) or np.any(v >= _M):
        return None
    return v

def transfer_qs(mq):
    if mq <= 0:
        return []
    s = {1, 2, 4, 8, mq, mq // 2, mq // 4, (3 * mq) // 4}
    r = []
    for x in s:
        x = int(x)
        if 1 <= x <= mq and x not in r:
            r.append(x)
    return sorted(r)

def pair_polish(u0, n, passes):
    M = _M
    inv = 1.0 / float(M)
    u = u0.copy()
    h = units_to_h(u)
    best = raw_score(h, n)
    if not math.isfinite(best):
        return u, best
    for _ in range(passes):
        imp = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                for rev in (0, 1):
                    a, b = (i, j) if rev == 0 else (j, i)
                    mq = int(min(u[a], M - u[b]))
                    if mq <= 0:
                        continue
                    for q in transfer_qs(mq):
                        u[a] -= q; u[b] += q
                        h[a] -= q * inv; h[b] += q * inv

                            imp = True
                            break
                        u[a] += q; u[b] -= q
                        h[a] += q * inv; h[b] -= q * inv
                    if imp:
                        break
        if not imp:
            break
    return u, best

            if c2 < best_c - 1e-12:
                best_c = c2
                best_h = h2.copy()
                best_u = to_units(h2, n)

    ub = best_u if best_u is not None else to_units(best_h, n)
    if ub is not None:
        up, _ = pair_polish(ub, n, 2 if n <= 64 else 1)
        hp = units_to_h(up)
        cp = float(exact_score(hp, n))
        if cp < best_c - 1e-12:
            best_c = cp
            best_h = hp.copy()

    if not math.isfinite(best_c) or best_c > fb_c + 1e-12:
        best_h = fb.copy()
        best_c = fb_c
    fh = project_capped(best_h, n) if not is_feasible(best_h, n) else best_h
    if fh is None:
        fh = project_capped(fb, n)
    if fh is None:
        fh = fb.copy()
    fc = float(exact_score(fh, n))
    if not math.isfinite(fc):
        fh = fb.copy()
        fc = float(exact_score(fb, n))
    return fh, float(fc), int(n)
