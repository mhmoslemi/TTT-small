    scores_list = scores_list[:P_max]
    P = len(pop_list)
    if n == 1 or P < 3 or max_gens == 0:
        bi = int(np.argmin(scores_list))
        h_final = pop_list[bi].copy()
        return h_final, float(exact_score(h_final, n)), n
    pop = np.array(pop_list)
    scores = np.array(scores_list)
    pop, scores = de_optimize(pop, scores, n, rng, max_gens, stall_limit, max_restarts)
    bi = int(np.argmin(scores))
    best_h = pop[bi].copy()
    best_s = float(scores[bi])
    ps = local_polish(best_h, n)
    ps = project_capped(ps, n)
    if ps is not None:
        s_pol = exact_score(ps, n)
        if s_pol < best_s - 1e-12:
            best_h, best_s = ps, s_pol
    candidates = [best_h]
    if parent_proj is not None:
        candidates.append(parent_proj)
    candidates.append(uniform)
    fb_h, fb_s = None, float('inf')
    for c in candidates:
        sc = exact_score(c, n)
        if sc < fb_s:
            fb_s = sc; fb_h = c.copy()
    if fb_h is None or not np.isfinite(fb_s):
        fb_h = uniform.copy(); fb_s = exact_score(uniform, n)
    return fb_h, float(fb_s), n
