        rng = np.random.default_rng(42)

    if n <= 64:
        n_starts, maxiter, n_hops = 12, 400, 10
    elif n <= 128:
        n_starts, maxiter, n_hops = 8, 250, 5
    elif n <= 256:
        n_starts, maxiter, n_hops = 5, 150, 3
    else:
        n_starts, maxiter, n_hops = 3, 80, 1

    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))

    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()

    for st in starts:
        y = solve_epigraph(st, n, maxiter)
        if y is not None:
            ys = float(exact_score(y, n))
            if ys < gs - 1e-12:
                gs, gb = ys, y.copy()

    cur = gb.copy()
    cur_s = gs
    scales = [0.03, 0.06, 0.10, 0.04, 0.08, 0.12, 0.05, 0.07, 0.15, 0.09]
    for r in range(n_hops):
        sigma = scales[r % len(scales)]
        if r % 5 == 4:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            prop = project_capped(cur + sigma * rng.standard_normal(n), n)
        if prop is None:
            continue
        y = solve_epigraph(prop, n, maxiter)
        if y is None:
            continue
        ys = float(exact_score(y, n))
        if ys < gs - 1e-12:
            gs, gb = ys, y.copy()
            cur = y.copy()
            cur_s = ys
        else:
            T = max(1e-6, 2.5e-3 * (0.72 ** r))
            if ys < cur_s + 0.02:
                if rng.random() < np.exp(-(ys - cur_s) / T):
                    cur = y.copy()
                    cur_s = ys

    # Final polish with more iterations
    y = solve_epigraph(gb, n, max(600, maxiter * 2))
    if y is not None:
        ys = float(exact_score(y, n))
        if ys < gs - 1e-12:
            gs, gb = ys, y.copy()

    # Second polish from perturbed best
    y2 = solve_epigraph(project_capped(gb + 0.02 * rng.standard_normal(n), n), n, max(600, maxiter * 2))
    if y2 is not None:
        ys2 = float(exact_score(y2, n))
        if ys2 < gs - 1e-12:
            gs, gb = ys2, y2.copy()

    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12 or fs == float('inf'):
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
