    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    if n <= 64:
        max_starts, max_iter, n_hops = 6, 60, 4
    elif n <= 128:
        max_starts, max_iter, n_hops = 4, 45, 2
    elif n <= 512:
        max_starts, max_iter, n_hops = 3, 20, 1
    else:
        max_starts, max_iter, n_hops = 2, 10, 1
    try:
        max_starts = max(1, min(int(kwargs.get('starts', max_starts)), 10))
    except Exception:
        pass
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', max_iter)), 100))
    except Exception:
        pass
    try:
        n_hops = max(0, min(int(kwargs.get('shake_rounds', n_hops)), 5))
    except Exception:
        pass
    starts = starts[:max_starts]
    if max_iter > 0:
        for st in starts:
            h_res, best_score = local_optimize(st, n, max_iter, best_score)
            if h_res is not None:
                sc = exact_score(h_res, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best = h_res.copy()
    if n_hops > 0 and n <= 128:
        sigma = 0.04
        for _hop in range(n_hops):
            pert = best + sigma * rng.normal(0, 1, n)
            hp = project_capped(pert, n)
            if hp is None:
                sigma = min(0.25, 1.5 * sigma)
                continue
            h_res, bs2 = local_optimize(hp, n, max_iter, best_score)
            if h_res is not None:
                sc = exact_score(h_res, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best = h_res.copy()
                    sigma = max(0.01, 0.75 * sigma)
                else:
                    sigma = min(0.25, 1.5 * sigma)
            else:
                sigma = min(0.25, 1.5 * sigma)
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(uniform, n)
    return final, float(fs), int(n)
