        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n < 2:
        return best, best_score, n
    chain_idx = 0
    restart_noise = 0.04
    first_start = parent if parent is not None else uniform.copy()
    while budget_left > 0 and chain_idx < 12:
        if chain_idx == 0:
            x0 = first_start.copy()
        elif chain_idx < len(starts):
            x0 = starts[chain_idx].copy()
        else:
            x0 = project_capped(best + rng.normal(0, restart_noise, n), n)
            if x0 is None:
                x0 = uniform.copy()
            restart_noise = min(0.20, restart_noise * 1.5)
        chain_idx += 1
        x, score_x = x0.copy(), exact_score(x0, n)
        if not np.isfinite(score_x):
            x, score_x = uniform.copy(), uni_score
        if np.isfinite(score_x) and score_x < best_score - 1e-12:
            best, best_score = x.copy(), score_x
        T0 = calibrate_temp(x, score_x, n, rng, delta0, n_pilot)
        Tmin = max(1e-7, 1e-3 * T0)
        delta = delta0
        for ep in range(epochs):
            if budget_left <= 0:
                break
            T_e = T0 if epochs == 1 else T0 * (Tmin / T0) ** (ep / (epochs - 1))
            accepted, attempted = 0, 0
            for _ in range(moves):
                if budget_left <= 0:
                    break
                attempted += 1
                budget_left -= 1
                prop = make_exchange(x, n, rng, delta)
                if prop is None:
                    continue
                sp = exact_score(prop, n)
                if not np.isfinite(sp):
                    continue
                if sp < best_score - 1e-12:
                    best, best_score = prop.copy(), sp
                d = sp - score_x
                if d <= 0:
                    x, score_x, accepted = prop, sp, accepted + 1
                elif d / T_e > 500.0:
                    continue
                else:
                    if math.log(rng.random()) < -d / T_e:
                        x, score_x, accepted = prop, sp, accepted + 1
            rate = accepted / max(1, attempted)
            if rate > 0.40:
                delta = min(0.35, delta * 1.25)
            elif rate < 0.15:
                delta = max(0.002, delta * 0.75)
    if n <= 64:
        polished = polish_pairs(best, n, 2)
        pp = project_capped(polished, n)
        if pp is not None:
            ps = exact_score(pp, n)
            if np.isfinite(ps) and ps < best_score - 1e-13:
                best, best_score = pp, ps
    finalists = [(best, best_score), (fallback, fb_score), (uniform, uni_score)]
    if parent is not None:
        finalists.append((parent, exact_score(parent, n)))
    final, final_score = uniform, uni_score
    for fc, fs in finalists:
        if fc is None or not is_feasible(fc, n):
            fc2 = project_capped(fc, n) if fc is not None else None
            if fc2 is not None:
                fc, fs = fc2, exact_score(fc2, n)
            else:
                continue
        if np.isfinite(fs) and fs < final_score - 1e-15:
            final, final_score = fc, fs
    final_score = exact_score(final, n)
    if not np.isfinite(final_score):
        final, final_score = uniform, uni_score
    return final, float(final_score), int(n)
