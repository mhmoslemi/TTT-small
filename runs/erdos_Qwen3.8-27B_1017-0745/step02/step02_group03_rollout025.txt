import numpy as np
from scipy.optimize import linprog

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - mid, 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if 1.0 - h[idx] <= 1e-15:
                break
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    F = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        jdx = idx + d
        F[row] = float(np.sum(h[idx] * (1.0 - h[jdx])))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return F, G

def repair_step(h, delta, r, n):
    lo_b = np.maximum(-r, -h)
    hi_b = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, lo_b, hi_b)
    resid = -float(np.sum(delta))
    slack_hi = hi_b - delta
    slack_lo = delta - lo_b
    tot = float(np.sum(slack_hi) + np.sum(slack_lo))
    if tot < 1e-15 and abs(resid) > 1e-14:
        return None
    if abs(resid) > 1e-14 and tot > 1e-15:
        if resid > 0:
            mask = slack_hi > 1e-15
            w = slack_hi[mask] / max(float(np.sum(slack_hi[mask])), 1e-15)
            delta[mask] += resid * w
        else:
            mask = slack_lo > 1e-15
            w = slack_lo[mask] / max(float(np.sum(slack_lo[mask])), 1e-15)
            delta[mask] += resid * w
    delta = np.clip(delta, lo_b, hi_b)
    trial = h + delta
    proj = project_capped(trial, n)
    if proj is None:
        return None
    if float(np.max(np.abs(proj - trial))) > r + 1e-7:
        return None
    return proj

def lp_search(h0, n, max_iter, bh_init, bs_init):
    h = h0.copy()
    bh, bs = bh_init, bs_init
    cur_sc = exact_score(h, n)
    if np.isfinite(cur_sc) and cur_sc < bs - 1e-12:
        bh, bs = h.copy(), cur_sc
    r = 0.08
    r_min, r_max = 1e-5, 0.25
    for _ in range(max_iter):
        if r < r_min:
            break
        F, G = lag_vals_grads(h, n)
        L = 2 * n - 1
        A_ub = np.empty((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -F.copy()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        lo_b = np.maximum(-r, -h)
        hi_b = np.minimum(r, 1.0 - h)
        bounds = list(zip(lo_b.tolist(), hi_b.tolist()))
        bounds.append((None, None))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            r *= 0.5
            continue
        if res.status != 0 or res.x is None or len(res.x) != n + 1:
            r *= 0.5
            continue
        if not np.all(np.isfinite(res.x)):
            r *= 0.5
            continue
        delta = res.x[:n].copy()
        trial = repair_step(h, delta, r, n)
        if trial is None:
            r *= 0.5
            continue
        trial_sc = exact_score(trial, n)
        if not np.isfinite(trial_sc) or trial_sc >= cur_sc - 1e-12:
            r *= 0.5
            continue
        m_lin = float(np.max(F + G @ delta))
        p = float(np.max(F)) - m_lin
        rho = (cur_sc - trial_sc) / ((2.0 / n) * p) if p > 1e-14 else 0.0
        h = trial
        cur_sc = trial_sc
        if cur_sc < bs - 1e-12:
            bh, bs = h.copy(), cur_sc
        if float(np.max(np.abs(delta))) < 1e-8:
            break
        if p > 1e-14 and rho >= 0.75 and float(np.max(np.abs(delta))) >= 0.8 * r:
            r = min(r_max, 1.5 * r)
        elif p > 1e-14 and rho < 0.25:
            r = max(r_min, 0.5 * r)
    return bh, bs

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def coord_polish(h, n, sweeps=3):
    best = h.copy()
    best_s = exact_score(best, n)
    if not np.isfinite(best_s):
        return best, best_s
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [0.0]
                for frac in (0.1, 0.25, 0.5, 0.75, 0.9):
                    cands.append(lo + frac * (hi - lo))
                bd, bsc = 0.0, best_s
                for dd in cands:
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    sc = exact_score(trial, n)
                    if sc < bsc - 1e-13:
                        bd, bsc = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    best_s = bsc
                    improved = True
        if not improved:
            break
    return best, best_s
