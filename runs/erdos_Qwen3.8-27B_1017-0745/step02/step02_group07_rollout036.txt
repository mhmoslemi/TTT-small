import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s: lo = mid
        else: hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = s - np.sum(h)

        if r > 0:
            for i in range(n):
                if r <= 0: break

        else:
            for i in range(n):
                if r >= 0: break
                a2 = min(-r, h[i]); h[i] -= a2; r += a2
    if abs(np.sum(h)-s) > 1e-9 or np.any(h < -1e-12) or np.any(h > 1.0+1e-12):

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-9) and
            np.all(h <= 1.0+1e-9) and abs(np.sum(h)-n/2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n): return float('inf')
    try:
        v = float(np.max(np.correlate(h, 1.0-h, mode='full')*(2.0/n)))
        return v if np.isfinite(v) else float('inf')
    except: return float('inf')

def line_coeffs(h, d, n):
    dx = 2.0/n; Nl = 2*n-1
    a = np.zeros(Nl); b = np.zeros(Nl); c = np.zeros(Nl)
    a[0] = dx*np.sum(h*(1.0-h)); b[0] = dx*np.sum(d*(1.0-2.0*h)); c[0] = -dx*np.sum(d*d)
    idx = 1
    for r in range(1, n):
        m = n-r; h0,h1 = h[:m], h[r:]; d0,d1 = d[:m], d[r:]
        a[idx] = dx*np.sum(h1*(1.0-h0))

        a[idx] = dx*np.sum(h0*(1.0-h1))
        b[idx] = dx*(np.sum(d0*(1.0-h1))-np.sum(h0*d1)); c[idx] = -dx*np.sum(d0*d1); idx += 1
    return a, b, c

def feasible_interval(h, d, n):
    lo, hi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if di > 1e-15:
            lo = max(lo, -h[i]/di); hi = min(hi, (1.0-h[i])/di)
        elif di < -1e-15:
            lo = max(lo, (1.0-h[i])/di); hi = min(hi, -h[i]/di)
    if not (np.isfinite(lo) and np.isfinite(hi)) or hi-lo <= 1e-12: return None
    return lo, hi

def try_direction(h, d, n):
