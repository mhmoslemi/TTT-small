def project_capped(z, n):

    s = n / 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):

                    if r < 1e-14: break

                    if r > -1e-14: break

def is_feasible(h, n):

    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_data(h, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    q = np.zeros(nl); G = np.zeros((nl, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        q[idx] = dx * np.sum(h[:m] * (1.0 - h[r:r + m]))
        G[idx, :m] += dx * (1.0 - h[r:r + m])
        G[idx, r:r + m] -= dx * h[:m]
        idx += 1
        q[idx] = dx * np.sum(h[r:r + m] * (1.0 - h[:m]))

        G[idx, r:r + m] += dx * (1.0 - h[:m])
        idx += 1

        return None, None

def try_add(pool, v, th=1e-8):

def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)

def build_pool(fb, n, rng, cap=16):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, np.full(n, 0.5))

def mm_search(h0, n, max_iter, solvers):
    h = h0.copy()
    sc = exact_score(h, n)
    if not np.isfinite(sc):
        return h, float('inf')
    dx = 2.0 / n; L = 2.0 * dx
    nl = 2 * n - 1
    small_cnt = 0
    for it in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        d = cp.Variable(n); u = cp.Variable()
        obj = u + 0.5 * L * cp.sum_squares(d)
        cons = [G @ d + q <= u * np.ones(nl), cp.sum(d) == 0, d >= -h, d <= 1.0 - h]
        prob = cp.Problem(cp.Minimize(obj), cons)
        dv = None
        for sl in solvers:
            try:
                prob.solve(solver=sl)
                if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:
                    cands = np.array(d.value, dtype=np.float64).ravel()

                        dv = cands; break

                continue

            break
        best_h, best_s = None, sc

            break
