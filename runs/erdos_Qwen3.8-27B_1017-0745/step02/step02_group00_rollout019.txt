import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    v = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        v[d + n - 1] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def build_upper_hull(lines):
    # lines: list of (a, b) where y = a + b*t
    # Returns list of (a, b, t_start) for upper envelope
    if not lines:
        return []
    # Sort by slope, for same slope keep max intercept
    lines_sorted = sorted(lines, key=lambda x: (x[1], -x[0]))
    unique = []
    for l in lines_sorted:
        if unique and l[1] == unique[-1][1]:
            if l[0] > unique[-1][0]:
                unique[-1] = l
        else:
            unique.append(l)
    if len(unique) == 1:
        return [(unique[0][0], unique[0][1], float('-inf'))]
    hull = []
    hull.append((unique[0][0], unique[0][1], float('-inf')))
    for i in range(1, len(unique)):
        a_new, b_new = unique[i][0], unique[i][1]
        while hull:
            a_last, b_last, t_start = hull[-1]
            if b_new == b_last:
                break
            t_cross = (a_last - a_new) / (b_new - b_last)
            if t_cross <= t_start + 1e-15:
                hull.pop()
            else:
                hull.append((a_new, b_new, t_cross))
                break
        if not hull:
            hull.append((a_new, b_new, float('-inf')))
    return hull

def poly_eval(coeff, t):
    # coeff = (a, b, c) for a + b*t + c*t^2
    return coeff[0] + coeff[1] * t + coeff[2] * t * t

def quad_roots(a, b, c):
    # Returns real roots of a*t^2 + b*t + c = 0
    roots = []
    if abs(a) < 1e-14:
        if abs(b) > 1e-14:
            roots.append(-c / b)
    else:
        disc = b * b - 4.0 * a * c
        if disc >= -1e-12:
            disc = max(0.0, disc)
            sd = sqrt(disc)
            if abs(b) < 1e-14:
                # a*t^2 + c = 0
                roots.append(sd / abs(a))
                if sd > 1e-14:
                    roots.append(-sd / abs(a))
            else:
                q = -0.5 * (b + np.sign(b) * sd)
                roots.append(q / a)
                if abs(q) > 1e-14:
                    roots.append(c / q)
    return roots

def pair_search(h, q, n, i, j):
    dx = 2.0 / n
    r = j - i
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return 0.0, float('inf')
    Lw = 2 * n - 1
    # Compute b_d for all lags
    b = np.zeros(Lw)
    for d in range(-(n - 1), n):
        idx_d = d + n - 1
        # Term from k=i: (1-h[i+d]) if valid - h[i]*v[i+d]
        val = 0.0
        if 0 <= i + d < n:
            val += 1.0 - h[i + d]
        if d == 0:
            val -= h[i]
        elif d == r:
            val += h[i]  # -h[i]*v[i+d] = -h[i]*(-1) = h[i]
        # Term from k=j: -(1-h[j+d]) if valid - h[j]*v[j+d]
        if 0 <= j + d < n:
            val -= (1.0 - h[j + d])
        if d == 0:
            val += h[j]  # -h[j]*v[j+d] = -h[j]*(-1) = h[j]
        elif d == -r:
            val -= h[j]  # -h[j]*v[j+d] = -h[j]*(1) = -h[j]
        b[idx_d] = dx * val
    # c_d
    c = np.zeros(Lw)
    c[0 + n - 1] = -2.0 * dx
    c[r + n - 1] = dx
    c[-r + n - 1] = dx
    
    # Identify special lags: d=0, d=r, d=-r
    special = {0 + n - 1, r + n - 1, -r + n - 1}
    
    # Build affine lines for non-special lags
    lines = []
    for d_idx in range(Lw):
        if d_idx in special:
            continue
        lines.append((q[d_idx], b[d_idx]))
    
    hull = build_upper_hull(lines)
    
    best_t, best_val = 0.0, float(np.max(q))
    
    # Process each hull segment
    for seg in range(len(hull)):
        a_h, b_h, t_start = hull[seg]
        if seg + 1 < len(hull):
            a_next, b_next, _ = hull[seg + 1]
            t_end = (a_h - a_next) / (b_next - b_h)
        else:
            t_end = float('inf')
        seg_lo = max(L, t_start)
        seg_hi = min(U, t_end)
        if seg_lo >= seg_hi:
            continue
        # Polynomials on this segment
        polys = [(a_h, b_h, 0.0)]  # hull line
        # Add special lags
        for d_idx in sorted(special):
            polys.append((q[d_idx], b[d_idx], c[d_idx]))
        
        # Candidate t values
        cands = [seg_lo, seg_hi]
        for (a, bb, cc) in polys:
            if abs(cc) > 1e-14:
                sp = -bb / (2.0 * cc)
                if seg_lo - 1e-12 <= sp <= seg_hi + 1e-12:
                    cands.append(sp)
        for pi in range(len(polys)):
            for pj in range(pi + 1, len(polys)):
                da = polys[pi][2] - polys[pj][2]
                db = polys[pi][1] - polys[pj][1]
                dc = polys[pi][0] - polys[pj][0]
                roots = quad_roots(da, db, dc)
                for rt in roots:
                    if seg_lo - 1e-12 <= rt <= seg_hi + 1e-12:
                        cands.append(rt)

        for t in cands:
            t = max(seg_lo, min(seg_hi, t))
            if not np.isfinite(t):
                continue
            val = max(poly_eval(p, t) for p in polys)
            if val < best_val - 1e-13:
                best_val = val
                best_t = t
    
    return best_t, best_val

def coord_descent(h, n, max_sweeps, pair_budget, rng):

    q = lag_values(h, n)
    thr = np.max(q)
    improved_any = False
    
    for sweep in range(max_sweeps):
        improved = False
        # Build pair list
        if pair_budget >= n * (n - 1) // 2:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = [(i, i + 1) for i in range(n - 1)]
            remaining = pair_budget - len(pairs)
            if remaining > 0:
                all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
                all_pairs = [p for p in all_pairs if p[1] - p[0] > 1]
                idx = rng.choice(len(all_pairs), size=min(remaining, len(all_pairs)), replace=False)
                pairs.extend(all_pairs[k] for k in idx)
        rng.shuffle(pairs)
        
        for (i, j) in pairs:
            t, _ = pair_search(h, q, n, i, j)
            if abs(t) < 1e-14:
                continue
            # Apply
            new_i = h[i] + t
            new_j = h[j] - t
            if new_i < -1e-12 or new_i > 1 + 1e-12 or new_j < -1e-12 or new_j > 1 + 1e-12:
                continue
            # Update q
            r = j - i
            dx = 2.0 / n
            gi = np.zeros(2 * n - 1)
            gj = np.zeros(2 * n - 1)
            for d in range(-(n - 1), n):
                idx_d = d + n - 1
                gi[idx_d] = 0.0
                gj[idx_d] = 0.0
                if 0 <= i + d < n:
                    gi[idx_d] += (1.0 - h[i + d]) * dx
                if d == 0:
                    gi[idx_d] -= h[i] * dx
                elif d == r:
                    gi[idx_d] += h[i] * dx
                if 0 <= j + d < n:
                    gj[idx_d] -= (1.0 - h[j + d]) * dx
                if d == 0:
                    gj[idx_d] += h[j] * dx
                elif d == -r:
                    gj[idx_d] -= h[j] * dx
            q = q + gi * t + gj * t
            q[0 + n - 1] -= dx * (t * t + t * t)  # c_0 = -2dx, so subtract 2dx*t^2
            # Wait: q_new = q + b*t + c*t^2, and b is already in gi*t + gj*t
            # The c term: c_0 = -2dx, c_r = dx, c_-r = dx
            q[0 + n - 1] += (-2.0 * dx) * t * t
            q[r + n - 1] += dx * t * t
            q[-r + n - 1] += dx * t * t
            
            h[i] = new_i
            h[j] = new_j
            new_thr = np.max(q)
            if new_thr < thr - 1e-12:
                thr = new_thr
                improved = True
                improved_any = True
        
        if not improved:
            break
    
    return h, thr
