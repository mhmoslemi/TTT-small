            if is_feasible(bh, n):
                bs = exact_score(bh, n)
            if np.isfinite(bs) and bs < gbest_s - 1e-12:
                gbest_s = bs
                gbest_h = bh.copy()
        except Exception:
            pass
    if not is_feasible(gbest_h, n) or not np.isfinite(gbest_s):
        gbest_h = fb.copy()
        gbest_s = fb_score
    if gbest_s > fb_score + 1e-12:
        gbest_h = fb.copy()
        gbest_s = fb_score
    final_h = gbest_h
    final_s = exact_score(final_h, n)
    if not np.isfinite(final_s) or not is_feasible(final_h, n):
        final_h = fb.copy()
        final_s = exact_score(final_h, n)
    return final_h, final_s, n
