from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def ineq_f(z, n):
    h = z[:-1]
    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def ineq_g(z, n):
    h = z[:-1]
    nl, dx = 2 * n - 1, 2.0 / n
    J = np.zeros((nl, n + 1))
    J[:, n] = 1.0
    for k in range(nl):
        d = k - (n - 1)
        lo, hi = max(0, -d), min(n - 1, n - 1 - d)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        np.add.at(J[k, :n], js, -dx * (1.0 - h[js + d]))
        np.add.at(J[k, :n], js + d, dx * h[js])
    return J

def eq_f(z, n):
    return z[:-1].sum() - n / 2.0

def eq_g(z, n):
    J = np.ones(n + 1)
    J[-1] = 0.0
    return J

def obj_f(z, n):
    return z[-1]

def obj_g(z, n):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def solve_slsqp(h0, n, maxiter):
    sc = exact_score(h0, n)
    if not np.isfinite(sc):
        return None
    t0 = max(0.0, min(0.5, sc + 1e-9))
    z0 = np.append(h0.copy().astype(np.float64), t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_g, 'args': (n,)},
    ]
    try:
        res = minimize(obj_f, z0, jac=obj_g, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    if res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:-1], n)

def subgrad_polish(h, n, n_iter, step):
    h = h.copy().astype(np.float64)
    dx = 2.0 / n
    best_s = exact_score(h, n)
    for _ in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(c))
        d = k_star - (n - 1)
        lo, hi = max(0, -d), min(n - 1, n - 1 - d)
        if lo > hi:
            break
        js = np.arange(lo, hi + 1)
        g = np.zeros(n)
        np.add.at(g, js, dx * (1.0 - h[js + d]))
        np.add.at(g, js + d, -dx * h[js])
        h_new = project_capped(h - step * g, n)
        if h_new is None:
            break
        s_new = exact_score(h_new, n)
        if s_new < best_s - 1e-14:
            h = h_new
            best_s = s_new
        step *= 0.97
    return h

def try_add_start(pool, h, n):
    p = project_capped(h, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-6:
            return
    pool.append(p)

    attempt = 0
    while len(pool) < count and attempt < 20:
        try_add_start(pool, base + rng.normal(0, 0.15, n), n)
        attempt += 1
    return pool[:count]
