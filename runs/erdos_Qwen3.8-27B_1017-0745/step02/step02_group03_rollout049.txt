def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Correct residual using slack
    for _ in range(8):
        r = s - float(np.sum(h))
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            tot = float(np.sum(slack))
            if tot < 1e-15:
                break
            h += r * slack / tot
        else:
            tot = float(np.sum(h))
            if tot < 1e-15:
                break
            h += r * h / tot
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9*max(1, n):
        return None
    return h

import numpy as np
from math import sqrt, copysign

def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        r = s - float(np.sum(h))
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            tot = float(np.sum(slack))
            if tot < 1e-15:
                break
            h = h + r * slack / tot
        else:
            tot = float(np.sum(h))
            if tot < 1e-15:
                break
            h = h + r * h / tot
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')

def lag_vals(h, n):
    L = 2 * n - 1
    c = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        c[d + n - 1] = float(np.sum(h[lo:hi] * (1.0 - h[lo + d:hi + d])))
    return c

def grad_col(h, n, r):
    g = np.zeros(2 * n - 1)
    d1, d2 = -r, n - 1 - r
    if d1 <= d2:
        dr = np.arange(d1, d2 + 1)
        g[dr + n - 1] += 1.0 - h[r + dr]
    d3, d4 = r - n + 1, r
    if d3 <= d4:
        dr = np.arange(d3, d4 + 1)
        g[dr + n - 1] -= h[r - dr]
    return g

def upper_hull(bv, cv):
    if len(bv) == 0:
        return np.array([]), np.array([]), np.array([])
    order = np.argsort(bv)
    bv = bv[order].astype(float)
    cv = cv[order].astype(float)
    hs, hc, hst = [], [], []
    i, nl = 0, len(bv)
    while i < nl:
        j = i
        while j + 1 < nl and abs(bv[j + 1] - bv[i]) < 1e-15:
            j += 1
        best = i + int(np.argmax(cv[i:j + 1]))
        if not hs:
            hs.append(bv[best]); hc.append(cv[best]); hst.append(-1e18)
        else:
            x = (hc[-1] - cv[best]) / (bv[best] - hs[-1]) if abs(bv[best] - hs[-1]) > 1e-15 else -1e18
            while x <= hst[-1] + 1e-14:
                hs.pop(); hc.pop(); hst.pop()
                if not hs:
                    x = -1e18
                    break
                x = (hc[-1] - cv[best]) / (bv[best] - hs[-1]) if abs(bv[best] - hs[-1]) > 1e-15 else -1e18
            hs.append(bv[best]); hc.append(cv[best]); hst.append(x)
        i = j + 1
    return np.array(hs), np.array(hc), np.array(hst)

def rroots(A, B, C, lo, hi):
    out = []
    if abs(A) < 1e-15:
        if abs(B) > 1e-15:
            r = -C / B
            if lo - 1e-12 <= r <= hi + 1e-12 and np.isfinite(r):
                out.append(max(lo, min(hi, r)))
        return out
    D = B * B - 4.0 * A * C
    if D < -1e-8 * (B * B + abs(4.0 * A * C) + 1.0):
        return out
    D = max(0.0, D)
    sq = sqrt(D)
    q = -0.5 * (B + copysign(sq, B)) if abs(B) > 1e-15 else -0.5 * sq
    if abs(q) < 1e-30:
        r = -B / (2.0 * A)
        if lo - 1e-12 <= r <= hi + 1e-12 and np.isfinite(r):
            out.append(max(lo, min(hi, r)))
    else:
        for r in (q / A, C / q):
            if lo - 1e-12 <= r <= hi + 1e-12 and np.isfinite(r):
                out.append(max(lo, min(hi, r)))
    return out

def chord_min(h, p, q, n):
    Lb = max(-h[p], h[q] - 1.0)
    Ub = min(1.0 - h[p], h[q])
    if Lb >= Ub - 1e-14:
        return 0.0
    c = lag_vals(h, n)
    b = grad_col(h, n, p) - grad_col(h, n, q)
    a = np.zeros(2 * n - 1)
    off = q - p
    a[n - 1] = -2.0
    a[n - 1 + off] = 1.0
    a[n - 1 - off] = 1.0
    if not (np.all(np.isfinite(c)) and np.all(np.isfinite(b))):
        return 0.0
    excluded = {n - 1, n - 1 + off, n - 1 - off}
    n0 = np.array([i for i in range(2 * n - 1) if i not in excluded])
    hs, hc, hst = upper_hull(b[n0], c[n0])
    bps = [Lb, Ub, 0.0]
    for s in hst:
        if np.isfinite(s) and Lb < s < Ub:
            bps.append(float(s))
    if off > 0:
        d1, d2 = n - 1 + off, n - 1 - off
        dx = b[d1] - b[d2]
        if abs(dx) > 1e-15:
            xc = (c[d2] - c[d1]) / dx
            if Lb < xc < Ub and np.isfinite(xc):
                bps.append(xc)
    bps = sorted(set(round(x, 13) for x in bps))
    bps = [max(Lb, min(Ub, x)) for x in bps]
    bps_d = [bps[0]]
    for x in bps[1:]:
        if x - bps_d[-1] > 1e-12:
            bps_d.append(x)
    cands = [0.0, Lb, Ub]
    for i in range(len(bps_d) - 1):
        tl, th = bps_d[i], bps_d[i + 1]
        if th - tl < 1e-13:
            continue
        tm = 0.5 * (tl + th)
        quads = [(-2.0, b[n - 1], c[n - 1])]
        if len(hs) > 0:
            idx = max(0, min(int(np.searchsorted(hst, tm, side='right')) - 1, len(hs) - 1))
            quads.append((0.0, float(hs[idx]), float(hc[idx])))
        if off > 0:
            d1, d2 = n - 1 + off, n - 1 - off
            if b[d1] * tm + c[d1] >= b[d2] * tm + c[d2]:
                quads.append((1.0, b[d1], c[d1]))
            else:
                quads.append((1.0, b[d2], c[d2]))
        sub = [tl, th]
        for i2 in range(len(quads)):
            for j2 in range(i2 + 1, len(quads)):
                sub += rroots(quads[i2][0] - quads[j2][0], quads[i2][1] - quads[j2][1], quads[i2][2] - quads[j2][2], tl, th)
        sub = sorted(set(round(x, 13) for x in sub))
        sub_d = [sub[0]]
        for x in sub[1:]:
            if x - sub_d[-1] > 1e-12:
                sub_d.append(x)
        cands.extend(sub_d)
        for k in range(len(sub_d) - 1):
            cands.append(0.5 * (sub_d[k] + sub_d[k + 1]))
            vals = [qd[0] * tm * tm + qd[1] * tm + qd[2] for qd in quads]
            bi = int(np.argmax(vals))
            if quads[bi][0] > 0:
                tv = -quads[bi][1] / (2.0 * quads[bi][0])
                if tl < tv < th:
                    cands.append(tv)
    best_t, best_v = 0.0, float('inf')
    for t in cands:
        t = max(Lb, min(Ub, t))
        obj = float(np.max(c + b * t + a * t * t))
        if obj < best_v - 1e-14:
            best_v, best_t = obj, t
        elif abs(obj - best_v) < 1e-14 and abs(t) < abs(best_t):
            best_t = t
    return best_t

def build_pairs(n):
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    return pairs

def pair_search(h, n, budget, rng):
    best = h.copy()
    best_s = exact_score(best, n)
    if not np.isfinite(best_s):
        return best, best_s
    pairs = build_pairs(n)
    npairs = len(pairs)
    if npairs == 0:
        return best, best_s
    order = list(range(npairs))
    for att in range(budget):
        if att % npairs == 0 and att > 0:
            rng.shuffle(order)
        p, q = pairs[order[att % npairs]]
        old_p, old_q = best[p], best[q]
        t = chord_min(best, p, q, n)
        if abs(t) > 1e-13:
            sp = old_p + old_q
            new_p = max(0.0, min(1.0, old_p + t))
            new_q = sp - new_p
            best[p], best[q] = new_p, new_q
            sc = exact_score(best, n)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best_s = sc
            else:
                best[p], best[q] = old_p, old_q
    return best, best_s
