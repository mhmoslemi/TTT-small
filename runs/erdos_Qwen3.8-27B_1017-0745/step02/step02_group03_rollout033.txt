        add_start(starts, 0.5 + 0.45 * np.cos(4 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    while len(starts) < 6:
        add_start(starts, rng.uniform(0, 1, n), n)
        if len(starts) == 0:
            starts.append(fb.copy())
        if len(starts) >= 6:
            break
    for _ in range(10):
        if len(starts) >= 6:
            break
        add_start(starts, rng.uniform(0, 1, n), n)
    while len(starts) < 6:
        starts.append(starts[0].copy())

    temps = np.array([5e-4, 2e-3, 8e-3, 32e-3, 128e-3, 512e-3])
    scales = np.array([0.002, 0.006, 0.018, 0.054, 0.162, 0.486])
    NR = 6
    reps, rsc = [], []
    for i in range(NR):
        h_int = project_capped(0.99 * starts[i] + 0.01 * 0.5, n)
        if h_int is None:
            h_int = starts[i].copy()
        sc = exact_score(h_int, n)
        if not np.isfinite(sc):
            h_int, sc = fb.copy(), fb_score
        reps.append(h_int)
        rsc.append(sc)

    best, best_sc = fb.copy(), fb_score
    for i in range(NR):
        if rsc[i] < best_sc - 1e-12:
            best, best_sc = reps[i].copy(), rsc[i]

    B_def = min(16000, max(120, int(30000000 / (n * n))))
    try:
        B = int(kwargs.get('eval_budget', B_def))
    except Exception:
        B = B_def
    B = max(120, min(B, B_def, 16000))
    score_ct = 0
    stall, stall_lim = 0, max(30, min(200, B // (NR * 12)))
    n_restart = 0
    max_sw = B + 1

    for sw in range(max_sw):
        if score_ct >= B:
            break
        prev_bs = best_sc
        for i in range(NR):
            if score_ct >= B:
                break
            rv = rng.random()
            hp = None
            if rv < 0.35:
                hp = hit_and_run(reps[i], rng, n)
            elif rv < 0.65:
                hp = full_pair(reps[i], rng, n)
            else:
                hp = local_pair(reps[i], rng, scales[i], n)
            if hp is None:
                continue
            sp = exact_score(hp, n)
            score_ct += 1
            if not np.isfinite(sp):
                continue
            d = sp - rsc[i]
            if d <= 0:
                reps[i], rsc[i] = hp, sp
                if sp < best_sc - 1e-12:
                    best, best_sc = hp.copy(), sp
            else:
                la = -d / temps[i]
                if math.log(max(rng.random(), 1e-300)) < la:
                    reps[i], rsc[i] = hp, sp
                    if sp < best_sc - 1e-12:
                        best, best_sc = hp.copy(), sp
        if sw % 2 == 0:
            xch = [(0, 1), (2, 3), (4, 5)]
        else:
            xch = [(1, 2), (3, 4)]
        for (a, b) in xch:
            lr = min(0.0, (1.0 / temps[a] - 1.0 / temps[b]) * (rsc[a] - rsc[b]))
            if math.log(max(rng.random(), 1e-300)) < lr:
                reps[a], reps[b] = reps[b], reps[a]
                rsc[a], rsc[b] = rsc[b], rsc[a]
        if best_sc < prev_bs - 1e-12:
            stall = 0
        else:
            stall += 1
            if stall >= stall_lim and n_restart < 2:
                n_restart += 1
                stall = 0
                if n_restart == 1:
                    rv2 = 0.5 * rng.uniform(0, 1, n) + 0.5 * best
                else:
                    rv2 = rng.uniform(0, 1, n)
                pv = project_capped(rv2, n)
                if pv is not None:
                    pv2 = project_capped(0.99 * pv + 0.01 * 0.5, n)
                    if pv2 is None:
                        pv2 = pv
                    sc2 = exact_score(pv2, n)
                    score_ct += 1
                    if np.isfinite(sc2):
                        reps[5], rsc[5] = pv2, sc2
                        if sc2 < best_sc - 1e-12:
                            best, best_sc = pv2.copy(), sc2

    final = best
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fb.copy(), fb_score
    pr = project_capped(final, n)
    if pr is not None:
        fs2 = exact_score(pr, n)
        if np.isfinite(fs2) and (not np.isfinite(fs) or fs2 < fs):
            final, fs = pr, fs2
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
