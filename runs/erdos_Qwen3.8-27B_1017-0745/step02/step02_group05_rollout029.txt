import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    if r > 0:
        for i in range(n):
            if r <= 1e-14: break
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= -1e-14: break
            a = min(-r, h[i]); h[i] -= a; r += a
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14) or abs(np.sum(h) - s) > 1e-10 * n:
        return None
    return h

        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

    idx = 0

        Phi[:, idx] = np.sqrt(2.0 / n) * np.cos(a); labs[idx] = k; idx += 1
        Phi[:, idx] = np.sqrt(2.0 / n) * np.sin(a); labs[idx] = k; idx += 1
    if n % 2 == 0:

    return Phi, labs
