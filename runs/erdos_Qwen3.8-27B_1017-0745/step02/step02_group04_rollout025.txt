    for it in range(n_iters):
        scale = 0.06 * (0.85 ** (it // 20))
        trial = project_capped(best_h + scale * rng.normal(0, 1, n), n)
        if trial is None or not is_feasible(trial, n):
            continue
        ts = exact_score(trial, n)
        if ts < best_s - 1e-12:
            best_h = trial
            best_s = ts
    return best_h, best_s

def coord_polish(h, n, rng, n_sweeps):
    best_h = h.copy()
    best_s = exact_score(h, n)
    if best_s == float('inf'):
        return best_h, best_s
    for _ in range(n_sweeps):
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            for delta in [0.015, -0.015, 0.008, -0.008, 0.03, -0.03]:
                trial = best_h.copy()
                new_val = best_h[i] + delta
                if new_val < -1e-12 or new_val > 1.0 + 1e-12:
                    continue
                trial[i] = new_val
                rem = -delta
                cands = np.where((np.arange(n) != i))[0]
                if len(cands) == 0:
                    continue
                if rem > 0:
                    avail = best_h[cands]
                    ok = cands[avail > 1e-8]
                else:
                    avail = 1.0 - best_h[cands]
                    ok = cands[avail > 1e-8]
                if len(ok) == 0:
                    continue
                j = ok[int(rng.integers(len(ok)))]
                trial[j] += rem
                if not is_feasible(trial, n):
                    continue
                ts = exact_score(trial, n)
                if ts < best_s - 1e-12:
                    best_h = trial
                    best_s = ts
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = None
    if fallback is None or not is_feasible(fallback, n):
        fallback = project_capped(np.full(n, 0.5), n)
    if fallback is None:
        fallback = np.full(n, 0.5)
    if n == 1:
        return fallback, float(exact_score(fallback, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P, n_gen, pat, mx_r, npol, csweeps = 48, 220, 30, 2, 600, 4
    elif n <= 128:
        P, n_gen, pat, mx_r, npol, csweeps = 32, 150, 25, 2, 400, 3
    elif n <= 256:
        P, n_gen, pat, mx_r, npol, csweeps = 24, 100, 20, 1, 300, 2
    else:
        P, n_gen, pat, mx_r, npol, csweeps = 16, 60, 15, 1, 200, 2
    pop = build_population(fallback, n, rng, P)
    best_h, best_s = de_search(pop, n, rng, n_gen, pat, mx_r)
    best_h, best_s = local_polish(best_h, n, rng, npol)
    best_h, best_s = coord_polish(best_h, n, rng, csweeps)
    best_h, best_s = local_polish(best_h, n, rng, npol // 2)
    fb = fallback.copy()
    fs = exact_score(fb, n)
    if best_s > fs - 1e-12:
        best_h, best_s = fb, fs
    result = project_capped(best_h, n)
    if result is None or not is_feasible(result, n):
        result = fb
    final_s = exact_score(result, n)
    if final_s == float('inf'):
        result = np.full(n, 0.5)
        final_s = exact_score(result, n)
    return result, float(final_s), int(n)
