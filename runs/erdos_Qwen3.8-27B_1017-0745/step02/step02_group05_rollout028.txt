            v = pop[a] + F * (pop[b] - pop[c])
        mask = rng.random(n) < CR
        mask[int(rng.integers(n))] = True
        trial = np.where(mask, v, pop[i])
        tp = project_capped(trial, n)
        if tp is None or not is_feasible(tp, n):
            npop.append(pop[i].copy())
            nscs.append(scs[i])
            continue
        ts = exact_score(tp, n)
        if not np.isfinite(ts):
            npop.append(pop[i].copy())
            nscs.append(scs[i])
            continue
        d = ts - scs[i]
        acc = False
        if d < -1e-12:
            acc = True
        elif abs(d) <= 1e-12:
            acc = rng.random() < 0.1
        else:
            acc = rng.random() < safe_exp(-d / T) if T > 1e-30 else False
        if acc:
            npop.append(tp)
            nscs.append(ts)
        else:
            npop.append(pop[i].copy())
            nscs.append(scs[i])
    return npop, nscs

def pair_propose(h, n, rng, rho):
    for _ in range(20):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        cap = min(1.0 - h[i], h[j])
        if cap > 1e-12:
            t = cap * rho * (0.05 + 0.95 * rng.random())
            hn = h.copy()
            hn[i] += t
            hn[j] -= t
            return hn
    return None

def pair_polish(h, n, rng, sweeps, max_rand_pairs):
    h = h.copy()
    sc = exact_score(h, n)
    fracs = [1.0 / 64, 1.0 / 16, 1.0 / 4, 1.0 / 2, 1.0]
    for sw in range(sweeps):
        imp = False
        if n <= 64:
            for i in range(n):
                for j in range(i + 1, n):
                    for di, dj in [(i, j), (j, i)]:
                        cap = min(1.0 - h[di], h[dj])
                        if cap < 1e-12:
                            continue
                        for fr in fracs:
                            hn = h.copy()
                            hn[di] += cap * fr
                            hn[dj] -= cap * fr
                            hp = project_capped(hn, n)
                            if hp is None or not is_feasible(hp, n):
                                continue
                            hs = exact_score(hp, n)
                            if np.isfinite(hs) and hs < sc - 1e-12:
                                h, sc, imp = hp, hs, True
        else:
            for _ in range(max_rand_pairs):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                for di, dj in [(i, j), (j, i)]:
                    cap = min(1.0 - h[di], h[dj])
                    if cap < 1e-12:
                        continue
                    for fr in fracs:
                        hn = h.copy()
                        hn[di] += cap * fr
                        hn[dj] -= cap * fr
                        hp = project_capped(hn, n)
                        if hp is None or not is_feasible(hp, n):
                            continue
                        hs = exact_score(hp, n)
                        if np.isfinite(hs) and hs < sc - 1e-12:
                            h, sc, imp = hp, hs, True
        if not imp:
            break
    return h, sc

def anneal_pass(h, n, rng, L):
    lh = h.copy()
    ls = exact_score(lh, n)
    for t in range(L):
        qt = t / max(1, L - 1)
        rho = 1.0 - 0.98 * qt
        T = 5e-4 * (1e-7 / 5e-4) ** qt
        hn = pair_propose(lh, n, rng, rho)
        if hn is None:
            continue
        hp = project_capped(hn, n)
        if hp is None or not is_feasible(hp, n):
            continue
        hs = exact_score(hp, n)
        if not np.isfinite(hs):
            continue
        d = hs - ls
        if d < -1e-12:
            lh, ls = hp, hs
        else:
            p_acc = safe_exp(-d / T) if T > 1e-30 else 0.0
            if rng.random() < p_acc:
                lh, ls = hp, hs
    return lh, ls

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if n == 1:
        return np.array([0.5], dtype=np.float64), float(exact_score(np.array([0.5]), 1)), 1
    fb = None
    if parent is not None:
        fb = project_capped(parent, n)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    safe_h = fb.copy()
    safe_s = exact_score(safe_h, n)
    P, G, L, stall_lim, psw = 12, 120, 1600, 20, 2
    if 65 <= n <= 256:
        P, G, L, stall_lim, psw = 8, 70, 500, 14, 2
    elif n > 256:
        P, G, L, stall_lim, psw = 6, 25, 80, 7, 1
    pop, scs = build_pop(fb, n, rng, P)
    gi = int(np.argmin(scs))
    bs, bh = scs[gi], pop[gi].copy()
    pop[0], scs[0] = bh.copy(), bs
    stall = 0
    for gen in range(G):
        q = gen / max(1, G - 1)
        F = (0.5 + 0.4 * rng.random()) * (1.0 - 0.25 * q)
        T = 2e-3 * (1e-5 / 2e-3) ** q
        ob = bs
        pop, scs = de_gen(pop, scs, F, 0.85, T, rng)
        gi = int(np.argmin(scs))
        if scs[gi] < bs - 1e-12:
            bs, bh = scs[gi], pop[gi].copy()
        pop[0], scs[0] = bh.copy(), bs
        stall = 0 if bs < ob - 1e-12 else stall + 1
        if stall >= stall_lim and gen < G - 1:
            keep = max(2, P // 3)
            np_, ns_ = [pop[i].copy() for i in range(keep)], [scs[i] for i in range(keep)]
            sig = 0.025 + 0.075 * (1.0 - q)
            for _ in range(P - keep):
                if rng.random() < 0.5:
                    cand = bh + sig * rng.standard_normal(n)
                else:
                    cand = rng.uniform(0, 1, n)
                cp = project_capped(cand, n)
                if cp is not None and is_feasible(cp, n):
                    cs = exact_score(cp, n)
                    np_.append(cp)
                    ns_.append(cs if np.isfinite(cs) else bs)
                else:
                    np_.append(bh.copy())
                    ns_.append(bs)
            pop, scs = np_[:P], ns_[:P]
            gi = int(np.argmin(scs))
            if scs[gi] < bs - 1e-12:
                bs, bh = scs[gi], pop[gi].copy()
            pop[0], scs[0] = bh.copy(), bs
            stall = 0
    ah, as_ = anneal_pass(bh, n, rng, L)
    if as_ < bs - 1e-12:
        bs, bh = as_, ah.copy()
    pr = 160 if n > 64 else 0
    ph, ps = pair_polish(bh, n, rng, psw, pr)
    if ps < bs - 1e-12:
        bs, bh = ps, ph.copy()
    ah, as_ = anneal_pass(bh, n, rng, L // 2)
    if as_ < bs - 1e-12:
        bs, bh = as_, ah.copy()
    ph, ps = pair_polish(bh, n, rng, 1, pr)
    if ps < bs - 1e-12:
        bs, bh = ps, ph.copy()
    final = project_capped(bh, n)
    if final is None or not is_feasible(final, n):
        final = safe_h
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final, fs = safe_h, safe_s
    return final.astype(np.float64), float(fs), int(n)
