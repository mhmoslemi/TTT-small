    except Exception:
        rng = np.random.default_rng(42)
    ITP = 60 if n <= 64 else (40 if n <= 128 else (25 if n <= 256 else 12))
    starts = build_starts(fallback, n, rng)
    gb = fallback.copy()
    gs = exact_score(fallback, n)
    for st in starts:
        s0 = exact_score(st, n)
        if s0 < gs - 1e-12:
            gs, gb = s0, st.copy()
    for st in starts:
        try:
            xo, so = smooth_optimize(st, n, ITP)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xo, so = pairwise_refine(fallback, n, rng)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()
    except Exception:
        pass
    try:
        if not np.allclose(gb, fallback, atol=1e-10):
            xo, so = pairwise_refine(gb, n, rng)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, n)
    fb = exact_score(fallback, n)
    if fs > fb + 1e-12:
        final, fs = fallback.copy(), fb
    if fs == float('inf'):
        final = np.clip(np.full(n, 0.5), 0.0, 1.0)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
