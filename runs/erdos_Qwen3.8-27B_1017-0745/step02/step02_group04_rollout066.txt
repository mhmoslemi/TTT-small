            solved_pool.append(st)
    sigmas = [0.03, 0.06, 0.10, 0.18, 0.30]
    current = gb.copy()
    c_score = gs
    for r in range(n_hops):
        if r % 5 == 4:
            proposal = project_capped(rng.uniform(0, 1, n), n)
        else:
            sigma = sigmas[r % len(sigmas)]
            proposal = project_capped(current + sigma * rng.normal(0, 1, n), n)
        if proposal is None:
            continue
        h_new = slsqp_solve(proposal, n, maxiter)
        cand = h_new if h_new is not None else proposal
        s_cand = float(exact_score(cand, n))
        if s_cand < gs - 1e-12:
            gs, gb = s_cand, cand.copy()
        if s_cand <= c_score:
            current = cand.copy()
            c_score = s_cand
        else:
            T = max(1e-5, 2.0e-3 * (0.75 ** r))
            if rng.random() < np.exp(-(s_cand - c_score) / T):
                current = cand.copy()
                c_score = s_cand
    h_polish = slsqp_solve(gb, n, max(200, maxiter))
    if h_polish is not None:
        sp = float(exact_score(h_polish, n))
        if sp < gs - 1e-12:
            gs, gb = sp, h_polish.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb_score = float(exact_score(fallback, n))
    if fs > fb_score + 1e-12:
        gb, fs = fallback.copy(), fb_score
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
