    gb = fallback.copy() if fallback is not None else const_fb.copy()
    gbs = exact_score(gb, n)
    cfs = exact_score(const_fb, n)
    if cfs < gbs:
        gb, gbs = const_fb.copy(), cfs
    for i in range(P):
        if scores[i] < gbs:
            gb, gbs = pop[i].copy(), scores[i]
    best_stagnant = 0
    n_restarts_done = 0
    for gen in range(G):
        restart_slots = []
        if best_stagnant >= stag_len and n_restarts_done < n_restart:
            n_restarts_done += 1
            best_stagnant = 0
            order = np.argsort(scores)
            best_idx = -1
            for i in range(P - 1, -1, -1):
                if np.max(np.abs(pop[int(order[i])] - gb)) < 1e-14:
                    best_idx = int(order[i])
                    break
            to_rep = []
            for i in range(P - 1, -1, -1):
                idx = int(order[i])
                if idx != best_idx:
                    to_rep.append(idx)
                if len(to_rep) >= max(2, P // 4):
                    break
            to_rep = to_rep[:P - 1]
            sigma = 0.08 if n_restarts_done == 1 else 0.20
            for slot in to_rep:
                if rng.random() < 2.0 / 3.0:
                    trial = project_capped(gb + sigma * rng.standard_normal(n), n)
                else:
                    trial = project_capped(rng.uniform(0, 1, n), n)
                if trial is not None and is_feasible(trial, n):
                    pop[slot] = trial
                    scores[slot] = exact_score(trial, n)
                    restart_slots.append(slot)
        X = [h.copy() for h in pop]
        f = scores.copy()
        next_pop = [h.copy() for h in pop]
        next_f = scores.copy()
        gen_best = gbs
        for i in range(P):
            if i in restart_slots:
                continue
            others = list(range(i)) + list(range(i + 1, P))
            donors = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = int(donors[0]), int(donors[1]), int(donors[2])
            F = 0.45 + 0.45 * rng.random()
            v = X[r1] + F * (X[r2] - X[r3])
            mask = rng.random(n) < 0.85
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial_raw = np.where(mask, v, X[i])
            trial = project_capped(trial_raw, n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < f[i] - 1e-12:
                next_pop[i] = trial
                next_f[i] = ts
            if ts < gen_best:
                gen_best = ts
        if gen_best < gbs - 1e-12:
            gbs = gen_best
            for i in range(P):
                if abs(next_f[i] - gbs) < 1e-14:
                    gb = next_pop[i].copy()
                    break
            best_stagnant = 0
        else:
            best_stagnant += 1
        pop = next_pop
        scores = next_f
        found_elite = False
        for i in range(P):
            if np.max(np.abs(pop[i] - gb)) < 1e-14:
                found_elite = True
                break
        if not found_elite:
            worst_i = int(np.argmax(scores))
            pop[worst_i] = gb.copy()
            scores[worst_i] = gbs
        if best_stagnant >= stag_len and n_restarts_done >= n_restart:
            break
    final = None
    fs = float('inf')
    for cand in [gb, fallback, const_fb]:
        if cand is None:
            continue
        c = project_capped(cand, n)
        if c is None or not is_feasible(c, n):
            continue
        cs = exact_score(c, n)
        if cs < fs:
            fs = cs
            final = c.copy()
    if final is None:
        final = const_fb
        fs = exact_score(const_fb, n)
    return final, float(fs), int(n)
