        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * n

    if not is_feasible(h, n):
        return float('inf')
    val = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    return float(val) if np.isfinite(val) else float('inf')

def lag_values(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1, dtype=np.float64)
    Q[0] = dx * np.sum(h * (1.0 - h))
    idx = 1
    for r in range(1, n):
        m = n - r
        Q[idx] = dx * np.dot(h[:m], 1.0 - h[r:])
        Q[idx + 1] = dx * np.dot(h[r:], 1.0 - h[:m])
        idx += 2
    return Q

def lag_jacobian(h, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    G = np.zeros((nl, n), dtype=np.float64)
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        G[idx, :m] += dx * (1.0 - h[r:])
        G[idx, r:] -= dx * h[:m]
        G[idx + 1, r:] += dx * (1.0 - h[:m])
        G[idx + 1, :m] -= dx * h[r:]
        idx += 2
    return G

def obj(z, n):
    return z[n]

def obj_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64)
    g[n] = 1.0
    return g

def ineq_fun(z, n):
    h = z[:n]
    Q = lag_values(h, n)
    return z[n] - Q

def ineq_jac(z, n):
    h = z[:n]
    G = lag_jacobian(h, n)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def eq_fun(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def eq_jac(z, n):
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0
    return J

def smooth_val_grad(h, n, tau):
    Q = lag_values(h, n)
    qmax = np.max(Q)
    shifted = (Q - qmax) / tau
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    F = qmax + tau * (np.log(np.sum(exps)))
    g = np.zeros(n, dtype=np.float64)
    dx = 2.0 / n
    g += p[0] * dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        g[:m] += p[idx] * dx * (1.0 - h[r:])
        g[r:] -= p[idx] * dx * h[:m]
        g[r:] += p[idx+1] * dx * (1.0 - h[:m])
        g[:m] -= p[idx+1] * dx * h[r:]
        idx += 2
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def slsqp_solve(h0, n, maxiter):
    t0 = exact_score(h0, n) + 1e-6
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constr_ineq = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (n,)}
    constr_eq = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}
    try:
        res = minimize(obj, z0, jac=obj_jac, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=[constr_ineq, constr_eq],
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):

        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s: lo = mid
        else: hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14: break
        for i in range(n):
            if r > 0:
                d = min(r, 1.0 - h[i]); h[i] += d; r -= d
            else:
                d = min(-r, h[i]); h[i] -= d; r += d
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12): return None
    if abs(np.sum(h) - s) > 1e-9 * n: return None
    return h

def is_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12): return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * n

def exact_score(h, n):
    if not is_feasible(h, n): return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def lag_values(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1, dtype=np.float64)
    Q[0] = dx * np.dot(h, 1.0 - h)
    idx = 1
    for r in range(1, n):
        m = n - r
        Q[idx] = dx * np.dot(h[:m], 1.0 - h[r:])
        Q[idx + 1] = dx * np.dot(h[r:], 1.0 - h[:m])
        idx += 2
    return Q

def lag_jacobian(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        G[idx, :m] += dx * (1.0 - h[r:])
        G[idx, r:] -= dx * h[:m]
        G[idx + 1, r:] += dx * (1.0 - h[:m])
        G[idx + 1, :m] -= dx * h[r:]
        idx += 2
    return G

def _slsqp_obj(z, n):
    return z[n]

def _slsqp_obj_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64)
    g[n] = 1.0
    return g

def _slsqp_ineq(z, n):
    h = z[:n]
    return z[n] - lag_values(h, n)

def _slsqp_ineq_jac(z, n):
    h = z[:n]
    G = lag_jacobian(h, n)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _slsqp_eq(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def _slsqp_eq_jac(z, n):
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0
    return J

def slsqp_run(h0, n, maxiter):
    t0 = exact_score(h0, n) + 1e-6
    z0 = np.concatenate([h0.copy(), [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    ci = {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac, 'args': (n,)}
    ce = {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}
    try:
        res = minimize(_slsqp_obj, z0, jac=_slsqp_obj_jac, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=[ci, ce],
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:n], n)

def _smooth_val_grad(h, n, tau):
    Q = lag_values(h, n)
    qmax = np.max(Q)
    sh = (Q - qmax) / tau
    ex = np.exp(sh)
    p = ex / np.sum(ex)
    F = qmax + tau * np.log(np.sum(ex))
    dx = 2.0 / n
    g = p[0] * dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        g[:m] += p[idx] * dx * (1.0 - h[r:])
        g[r:] -= p[idx] * dx * h[:m]
        g[r:] += p[idx + 1] * dx * (1.0 - h[:m])
        g[:m] -= p[idx + 1] * dx * h[r:]
        idx += 2
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def smooth_refine(h0, n, iters):
    best_h, best_s = h0.copy(), exact_score(h0, n)
    h = h0.copy()
    for tau in (0.02, 0.005, 0.001, 0.0001):
        alpha = 1.0
        for _ in range(iters):
            F, g = _smooth_val_grad(h, n, tau)
            if F is None: break
            y = project_capped(h - alpha * g, n)
            if y is None: break
            d = y - h
            if np.max(np.abs(d)) <= 1e-9: break
            Fy, _ = _smooth_val_grad(y, n, tau)
            if Fy is None or Fy > F + 1e-4 * np.dot(g, d) + 1e-12:
                alpha *= 0.5
                if alpha < 1e-12: break
                continue
            h = y
            alpha = min(1.5 * alpha, 5.0)
            s = exact_score(h, n)
            if s < best_s - 1e-13:
                best_s = s; best_h = h.copy()
        h = project_capped(h, n)
        if h is None: break
    return best_h, best_s

def _add_start(pool, v, tol=1e-9):
    if v is None: return
    for s in pool:
        if np.max(np.abs(v - s)) < tol: return
    pool.append(v)

def build_starts(fb, n, rng, max_starts):
    pool = []
    _add_start(pool, project_capped(fb, n))
    _add_start(pool, project_capped(fb[::-1], n))
    _add_start(pool, project_capped(1.0 - fb, n))
    _add_start(pool, project_capped(1.0 - fb[::-1], n))
    _add_start(pool, project_capped(np.full(n, 0.5), n))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    _add_start(pool, project_capped(blk, n))
    _add_start(pool, project_capped(np.roll(fb, n // 3), n))
    _add_start(pool, project_capped(np.roll(fb, n // 2), n))
    for sd in (0.08, 0.22):
        _add_start(pool, project_capped(fb + rng.normal(0, sd, n), n))
    for k in (1, 2, 3):
        _add_start(pool, project_capped(0.5 + 0.4 * np.cos(2.0 * np.pi * k * np.arange(n) / n + 0.5), n))
    sc = max(0.3, min(1.0, math.sqrt(30.0 / n)))
    att = 0
    while len(pool) < max_starts and att < 60:
        att += 1
        base = pool[int(rng.integers(len(pool)))]
        sig = float(rng.choice([0.04, 0.1, 0.2])) * sc
        _add_start(pool, project_capped(base + rng.normal(0, sig, n), n))
    while len(pool) < max_starts:
        pool.append(pool[0].copy())
    return pool[:max_starts]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999: n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, 0.5 * 0.5, 1
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_s = exact_score(fb, n)
    best_h, best_s = fb.copy(), fb_s
    # Determine budget
    if n <= 64:
        ms, mi, sr = 10, 220, 12
    elif n <= 128:
        ms, mi, sr = 7, 140, 10
    elif n <= 256:
        ms, mi, sr = 5, 0, 25
    elif n <= 512:
        ms, mi, sr = 4, 0, 18
    else:
        ms, mi, sr = 3, 0, 12
    starts = build_starts(fb, n, rng, ms)
    for h0 in starts:
        s0 = exact_score(h0, n)
        if s0 < best_s - 1e-12:
            best_s = s0; best_h = h0.copy()
        if mi > 0:
            hp = slsqp_run(h0, n, mi)
            cands = [hp]
            if hp is not None:
                for a in (0.25, 0.5, 0.75):
                    cands.append(project_capped((1.0 - a) * h0 + a * hp, n))
            for c in cands:
                if c is not None and is_feasible(c, n):
                    sc = exact_score(c, n)
                    if sc < best_s - 1e-12:
                        best_s = sc; best_h = c.copy()
        # smooth polish on current best and this start
        for base in ([best_h] if best_s < s0 - 1e-10 else [h0]):
            bh, bs = smooth_refine(base.copy(), n, sr)
            if bs < best_s - 1e-12:
                best_s = bs; best_h = bh.copy()
    # Final polish on incumbent
    bh, bs = smooth_refine(best_h.copy(), n, sr)
    if bs < best_s - 1e-12:
        best_s = bs; best_h = bh.copy()
    best_h = project_capped(best_h, n)
    if best_h is None or not is_feasible(best_h, n):
        best_h, best_s = fb.copy(), fb_s
    else:
        fs = exact_score(best_h, n)
        if fs > fb_s + 1e-12:
            best_h, best_s = fb.copy(), fb_s
        else:
            best_s = fs
    return best_h.astype(np.float64), float(best_s), int(n)
