import numpy as np
import math
from scipy.optimize import minimize


def is_feasible(h, n):
    if h is None:
        return False
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-9 * max(1, n)


def project_capped(z, n):
    try:
        arr = np.asarray(z, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.shape != (n,) or not np.all(np.isfinite(arr)):
        return None
    u = np.clip(arr, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(u)) - 1.0
    hi = float(np.max(u)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(u - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            for i in range(n):
                if r <= 1e-14:
                    break
                add = min(r, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
                    r -= add
        else:
            need = -r
            for i in range(n):
                if need <= 1e-14:
                    break
                sub = min(need, h[i])
                if sub > 0.0:
                    h[i] -= sub
                    need -= sub
        r = s - float(np.sum(h))
    if not is_feasible(h, n):
        return None
    return h


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / float(n))
    except Exception:
        return float('inf')
    if not math.isfinite(val):
        return float('inf')
    return val


def lag_values(h, n, with_jac=False):
    dx = 2.0 / float(n)
    Q = np.empty(2 * n - 1, dtype=np.float64)
    Q[0] = np.dot(h, 1.0 - h)
    if not with_jac:
        idx = 1
        for r in range(1, n):
            m = n - r
            hp = h[:m]
            hr = h[r:]
            Q[idx] = np.dot(hp, 1.0 - hr)
            Q[idx + 1] = np.dot(hr, 1.0 - hp)
            idx += 2
        Q *= dx
        return Q

    G = np.empty((2 * n - 1, n), dtype=np.float64)
    G[0, :] = 1.0 - 2.0 * h
    idx = 1
    for r in range(1, n):
        m = n - r
        hp = h[:m]
        hr = h[r:]
        Q[idx] = np.dot(hp, 1.0 - hr)
        G[idx, :] = 0.0
        G[idx, :m] += 1.0 - hr
        G[idx, r:] -= hp
        idx += 1
        Q[idx] = np.dot(hr, 1.0 - hp)
        G[idx, :] = 0.0
        G[idx, :m] -= hr
        G[idx, r:] += 1.0 - hp
        idx += 2
    Q *= dx
    G *= dx
    return Q, G


def slsqp_obj(z, args):
    n = args[0]
    return float(z[n])


def slsqp_obj_jac(z, args):
    n = args[0]
    g = np.zeros(n + 1, dtype=np.float64)
    g[n] = 1.0
    return g


def slsqp_eq(z, args):
    n = args[0]
    return float(np.sum(z[:n]) - n / 2.0)


def slsqp_eq_jac(z, args):
    n = args[0]
    g = np.zeros(n + 1, dtype=np.float64)
    g[:n] = 1.0
    return g


def slsqp_ineq(z, args):
    n = args[0]
    Q = lag_values(z[:n], n, False)
    return z[n] - Q


def slsqp_ineq_jac(z, args):
    n = args[0]
    _, G = lag_values(z[:n], n, True)
    J = -G
    J[:, n] = 1.0
    return J


def make_slsqp_bounds(n):
    return [(0.0, 1.0)] * (n + 1)


def make_slsqp_cons(n):
    return [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},
    ]


def solve_epigraph(h0, n, maxiter):
    h0 = project_capped(h0, n)
    if h0 is None:
        return None
    t0 = exact_score(h0, n)
    if not math.isfinite(t0):
        return None
    z0 = np.empty(n + 1, dtype=np.float64)
    z0[:n] = h0
    z0[n] = min(1.0, max(0.0, t0 + 1e-7))
    try:
        res = minimize(
            slsqp_obj,
            z0,
            args=(n,),
            method='SLSQP',
            jac=slsqp_obj_jac,
            bounds=make_slsqp_bounds(n),
            constraints=make_slsqp_cons(n),
            options={'ftol': 1e-10, 'maxiter': int(maxiter), 'disp': False}
        )
    except Exception:
        return None
    x = getattr(res, 'x', None)
    if x is None or len(x) != n + 1 or not np.all(np.isfinite(x)):
        return None
    return project_capped(x[:n], n)


def update_incumbent(hb, sb, cand, n):
    if cand is None:
        return hb, sb
    sc = exact_score(cand, n)
    if math.isfinite(sc) and sc < sb - 1e-12:
        return cand, sc
    return hb, sb


def make_binary(n, rng):
    v = np.zeros(n, dtype=np.float64)
    p = rng.permutation(n)
    v[p[:n // 2]] = 1.0
    if n % 2 == 1:
        v[p[n // 2]] = 0.5
    return v


def make_sinusoid(n, k, phase):
    i = np.arange(n, dtype=np.float64)
    return 0.5 + 0.4 * np.cos(2.0 * math.pi * float(k) * i / float(n) + phase)


def add_start(pool, v, n, th=1e-8):
    if v is None:
        return
    try:
        v = np.asarray(v, dtype=np.float64).ravel()
    except Exception:
        return
    if v.size != n or not np.all(np.isfinite(v)):
        return
    v = project_capped(v, n)
    if v is None:
        return
    for s in pool:
        if float(np.max(np.abs(v - s))) < th:
            return
    pool.append(v)


def build_starts(fb, n, rng, max_starts):
    pool = []
    add_start(pool, fb, n)
    scale = max(0.5, min(1.2, math.sqrt(47.0 / float(n))))
    for sig in (0.04, 0.12, 0.28):
        add_start(pool, fb + rng.normal(0.0, sig * scale, n), n)
    add_start(pool, make_sinusoid(n, 1, 0.0), n)
    add_start(pool, make_binary(n, rng), n)
    add_start(pool, make_sinusoid(n, 2, 0.0), n)
    add_start(pool, make_sinusoid(n, 1, 0.5 * math.pi), n)
    add_start(pool, fb[::-1], n)
    add_start(pool, 1.0 - fb, n)
    add_start(pool, np.roll(fb, n // 3), n)
    add_start(pool, np.roll(fb, n // 2), n)
    add_start(pool, np.full(n, 0.5), n)
    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(pool, blk, n)
    add_start(pool, make_sinusoid(n, 2, 0.5 * math.pi), n)
    add_start(pool, make_binary(n, rng), n)
    i = 1
    while len(pool) < max_starts and i < 80:
        add_start(pool, fb + rng.normal(0.0, 0.05 * (1.0 + 0.2 * i) * scale, n), n)
        i += 1
    while len(pool) < max_starts:
        pool.append(pool[0].copy())
    return pool[:max_starts]


def smooth_fgrad(h, n, tau):
    Q = lag_values(h, n, False)
    if not np.all(np.isfinite(Q)):
        return None, None
    qmax = float(np.max(Q))
    if tau <= 0.0:
        tau = 1e-4
    ex = np.exp((Q - qmax) / tau)
    s = float(np.sum(ex))
    if s <= 0.0 or not math.isfinite(s):
        return None, None
    p = ex / s
    F = qmax + tau * math.log(s)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=np.float64)
    g += p[0] * dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        p1 = float(p[idx])
        p2 = float(p[idx + 1])
        if abs(p1) > 1e-15:
            g[:m] += p1 * dx * (1.0 - h[r:])
            g[r:] -= p1 * dx * h[:m]
        if abs(p2) > 1e-15:
            g[:m] -= p2 * dx * h[r:]
            g[r:] += p2 * dx * (1.0 - h[:m])
        idx += 2
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g


def smooth_refine(h0, n, tau_seq, nit):
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    sb = exact_score(h, n)
    hbest = h.copy()
    for tau in tau_seq:
        Fh, gh = smooth_fgrad(h, n, tau)
        if Fh is None:
            break
        alpha = 1.0
        for _ in range(nit):
            if alpha < 1e-12:
                break
            y = project_capped(h - alpha * gh, n)
            if y is None:
                alpha *= 0.5
                continue
            d = y - h
            if float(np.max(np.abs(d))) <= 1e-10:
                break
            Fy, gy = smooth_fgrad(y, n, tau)
            if Fy is None:
                alpha *= 0.5
                continue
            dot = float(np.dot(gh, d))
            tol = 1e-9 * max(1.0, abs(Fh))
            if dot < 0.0:
                target = Fh + 1e-4 * dot + tol
            else:
                target = Fh + tol
            accept = Fy <= target
            if not accept:
                scy = exact_score(y, n)
                if math.isfinite(scy) and scy < sb - 1e-12 and Fy <= Fh + 1e-4 * max(1.0, abs(Fh)):
                    accept = True
            if accept:
                h = y
                Fh = Fy
                gh = gy
                alpha = min(1.5 * alpha, 10.0)
                sc = exact_score(h, n)
                if math.isfinite(sc) and sc < sb - 1e-12:
                    sb = sc
                    hbest = h.copy()
            else:
                alpha *= 0.5
        sc = exact_score(h, n)
        if math.isfinite(sc) and sc < sb:
            sb = sc
            hbest = h.copy()
    sc = exact_score(h, n)
    if math.isfinite(sc) and sc < sb:
        sb = sc
        hbest = h.copy()
    return hbest, sb


def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass

    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            n = 47

    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = project_capped(np.full(n, 0.5, dtype=np.float64), n)

    if fb is None or not is_feasible(fb, n):
        fb = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if fb is None:
        fb = np.full(n, 0.5, dtype=np.float64)

    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    fb_score = exact_score(fb, n)
    hb = fb.copy()
    sb = fb_score

    if n <= 64:
        max_starts = 6
        maxit = 100
        slsqp_starts = 5
    elif n <= 128:
        max_starts = 4
        maxit = 70
        slsqp_starts = 3
    else:
        max_starts = 4
        maxit = 0
        slsqp_starts = 0

    pool = build_starts(fb, n, rng, max_starts)

    if n <= 128:
        for idx in range(min(slsqp_starts, len(pool))):
            pv = pool[idx]
            cand = solve_epigraph(pv, n, maxit)
            hb, sb = update_incumbent(hb, sb, cand, n)
            if cand is not None:
                for a in (0.25, 0.5, 0.75):
                    mix = (1.0 - a) * pv + a * cand
                    hb, sb = update_incumbent(hb, sb, project_capped(mix, n), n)

    if n <= 128:
        tau_seq = (0.05, 0.01, 0.003, 0.0008)
        nit = 14
    elif n <= 256:
        tau_seq = (0.08, 0.02, 0.005)
        nit = 12
    else:
        tau_seq = (0.1, 0.03, 0.008)
        nit = 10

    psc = np.array([exact_score(pv, n) for pv in pool], dtype=np.float64)
    order = np.argsort(psc)
    smooth_ids = [0]
    if len(order) > 1:
        smooth_ids.append(int(order[0]))
    if len(order) > 2:
        smooth_ids.append(int(order[1]))
    if len(order) > 3 and n <= 64:
        smooth_ids.append(int(order[2]))

    seen = set()
    for j in smooth_ids:
        if j in seen:
            continue
        seen.add(j)
        h2, s2 = smooth_refine(pool[j], n, tau_seq, nit)
        hb, sb = update_incumbent(hb, sb, h2, n)

    h2, s2 = smooth_refine(hb, n, tau_seq, nit)
    hb, sb = update_incumbent(hb, sb, h2, n)

    final = project_capped(hb, n)
    fallback = project_capped(fb, n)
    if fallback is None or not is_feasible(fallback, n):
        fallback = np.full(n, 0.5, dtype=np.float64)
    fb_score = exact_score(fallback, n)

    if final is None or not is_feasible(final, n):
        hb = fallback.copy()
        sb = fb_score
    else:
        sc = exact_score(final, n)
        if not math.isfinite(sc):
            hb = fallback.copy()
            sb = fb_score
        else:
            hb = final
            sb = sc
            if math.isfinite(fb_score) and fb_score < sb - 1e-12:
                hb = fallback.copy()
                sb = fb_score

    return hb.astype(np.float64), float(sb), int(n)
