import numpy as np

def project_capped(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 2e-15:
            break
        if r > 0:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                if r <= 0:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            idx = np.argsort(h)
            for i in idx:
                if r >= 0:
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c))
    except Exception:
        return float('inf')

def add_unique(pool, v, n):

        return

        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def de_generation(X, scores, n, rng, F_lo, F_hi, CR_lo, CR_hi):
    P = len(X)
    Xnew = [x.copy() for x in X]
    snew = scores.copy()
    for i in range(P):
        others = [j for j in range(P) if j != i]
        r1, r2, r3 = rng.choice(others, size=3, replace=False)
        F = rng.uniform(F_lo, F_hi)
        CR = rng.uniform(CR_lo, CR_hi)
        donor = X[r1] + F * (X[r2] - X[r3])
        donor_p = project_capped(donor, n)
        if donor_p is None:
            continue
        trial = np.where(rng.uniform(size=n) < CR, donor_p, X[i])
        j0 = int(rng.integers(n))
        trial[j0] = donor_p[j0]
        trial_p = project_capped(trial, n)
        if trial_p is None:
            continue
        if np.max(np.abs(trial_p - X[i])) < 1e-12:
            continue
        sn = exact_score(trial_p, n)
        if sn < snew[i] - 1e-12:
            Xnew[i] = trial_p
            snew[i] = sn
    return Xnew, snew
