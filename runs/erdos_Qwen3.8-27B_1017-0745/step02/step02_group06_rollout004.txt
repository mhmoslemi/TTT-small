def solve_mm_qp(x, L, G, n, tried):

    rho = dx * math.cos(math.pi / (n + 1))

    prob = cp.Problem(cp.Minimize(obj), cons)

        if sn in tried:
            continue

def mm_trajectory(x0, n, max_iters, tried):

    sc = exact_score(x, n)
    for _ in range(max_iters):
        L, G = lag_data(x, n)
        if L is None:
            break
        yv = solve_mm_qp(x, L, G, n, tried)
        if yv is None:
            break
        best_x, best_s = None, sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial = x + alpha * (yv - x)
            pt = project_capped(trial, n)
            if pt is None or not is_feasible(pt, n):
                continue
            s = exact_score(pt, n)
            if s < best_s - 1e-12:

        if best_x is None:
            break
        step = np.max(np.abs(best_x - x))
        if step < 1e-10:
            x = best_x; sc = best_s
            break
