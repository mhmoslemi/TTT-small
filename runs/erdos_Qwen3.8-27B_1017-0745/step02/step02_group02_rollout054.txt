        fine_tau = [0.000005, 0.000002]
        lh, ls = continuation_descent(gb, n, fine_tau, 60)
        if ls < gs - 1e-12:
            gs, gb = ls, lh.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
