        res = run_slsqp(st, n, all_idx, 250, 1e-11)
        if res is not None:
            rs = exact_score(res, n)
            if rs < best_s - 1e-12:
                best, best_s = res, rs
    best, best_s = active_set_refine(best, n, 12, 5)
    best, best_s = active_set_refine(best, n, 8, 4)
    best, best_s = active_set_refine(best, n, 6, 4)
    polished = best.copy()
    line_search_sweep(polished, n, rng, 3)
    ps = exact_score(polished, n)
    if ps < best_s - 1e-13:
        best, best_s = polished, ps
    T0 = 0.003
    T_min = 1e-7
    delta = 0.08
    for chain_i in range(4):
        if chain_i == 0:
            st = best.copy()
        elif chain_i == 1:
            st = fb.copy()
        else:
            st = project_capped(best + rng.normal(0, 0.04 * chain_i, n), n)
            if st is None:
                st = fb.copy()
        ac, ac_s = anneal_chain(st, n, rng, T0, T_min, 3000, delta)
        if ac_s < best_s - 1e-12:
            best, best_s = ac, ac_s
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-12:
        final = best.copy()
        fs = best_s
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(uniform, n)
    return final, float(fs), int(n)
