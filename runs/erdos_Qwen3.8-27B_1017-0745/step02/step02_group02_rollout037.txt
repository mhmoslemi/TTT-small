            h_cur, s_cur = threshold_search(pert, n, rng, q_per, stages, pool)
            if s_cur < gs - 1e-12:
                gs, gb = s_cur, h_cur.copy()
                pool.append(gb.copy())
    max_pairs = max(10, (B - A) // 35)
    gb, gs = pair_line_scan(gb, n, rng, max_pairs)
    # Second pass of line scan with more pairs
    gb, gs = pair_line_scan(gb, n, rng, max_pairs)
    final = project_capped(gb, n)
    if final is None:
        final = gb
    fs = exact_score(final, n)
    if fs < fb_s:
        fb, fb_s = final.copy(), fs
    if fs < con_s:
        const_fb, con_s = final.copy(), fs
    best = min(((final, fs), (fb, fb_s), (const_fb, con_s)), key=lambda x: x[1])
    final = project_capped(best[0], n)
    if final is None or not is_feasible(final, n):
        final = const_fb
    sf = exact_score(final, n)
    if not math.isfinite(sf):
        final = const_fb
        sf = exact_score(final, n)
    return final, float(sf), int(n)
