import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def lag_values(h, n):
    dx = 2.0 / n
    q = dx * np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None
    return q

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = lag_values(h, n)
    if q is None:
        return float('inf')
    return float(np.max(q))

def compute_interaction_matrix(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    M = np.zeros((L, n), dtype=np.float64)
    for d in range(-(n-1), n):
        ad = max(d, 0)
        bd = max(-d, 0)
        md = n - abs(d)
        row = d + n - 1
        for k in range(n):
            val = 0.0
            # k in first array: k = ad + r, r in [0, md)
            r1 = k - ad
            if 0 <= r1 < md:
                q_idx = bd + r1
                val += (1.0 - h[q_idx])
            # k in second array: k = bd + r, r in [0, md)
            r2 = k - bd
            if 0 <= r2 < md:
                p_idx = ad + r2
                val -= h[p_idx]
            M[row, k] = dx * val
    return M

    cands = {lo, hi, 0.0}
    # Quadratic vertices (D>0): indices n-1+r0 and n-1-r0
    for idx in (n - 1 + r0, n - 1 - r0):
        tv = -B[idx] / (2.0 * dx)
        if lo <= tv <= hi:
            cands.add(tv)
    # Pairwise intersections
    for s in range(3):
        i1 = special[s]
        for s2 in range(s + 1, 3):
            i2 = special[s2]
            cands.update(solve_quad_roots(D[i1] - D[i2], B[i1] - B[i2], A[i1] - A[i2], lo, hi))
        for k in range(L):
            if k == special[0] or k == special[1] or k == special[2]:
                continue
            cands.update(solve_quad_roots(D[i1], B[i1] - B[k], A[i1] - A[k], lo, hi))
    
    T = sorted(t for t in cands if np.isfinite(t))
    if not T:
        T = [0.0]
    best_t = 0.0
    best_v = float('inf')
    for t in T:
        t = max(lo, min(hi, t))
        v = float(np.max(A + B * t + D * t * t))
        if v < best_v - 1e-14:
            best_v = v
            best_t = t
        elif abs(v - best_v) <= 1e-14 and abs(t) < abs(best_t):
            best_t = t
    return best_t

def coords_search(h, n, n_sweeps, rng):
    dx = 2.0 / n
    h_save = h.copy()
    for sweep in range(n_sweeps):
        score = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
        start_score = score
        M = compute_M(h, n)
        A = np.correlate(h, 1.0 - h, mode='full') * dx
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        accepted = 0
        for (i, j) in pairs:
            t = pair_line_min(h, A, M, n, i, j)
            if not np.isfinite(t) or abs(t) < 1e-14:
                continue
            hn = h.copy()
            hn[i] += t
            hn[j] -= t
            if not np.all(np.isfinite(hn)) or np.any(hn < -1e-12) or np.any(hn > 1.0 + 1e-12):
                continue
            A_new = np.correlate(hn, 1.0 - hn, mode='full') * dx
            ns = float(np.max(A_new))
            if ns < score - 1e-12:
                h = hn
                A = A_new
                M = compute_M(h, n)
                score = ns
                accepted += 1
        A_f = np.correlate(h, 1.0 - h, mode='full') * dx
        if float(np.max(A_f)) > start_score + 1e-12:
            h = h_save
        if accepted == 0:
            break
        h_save = h.copy()
    return h, exact_score(h, n)

def polish(h, n, rng, n_trials, sigma0):
    dx = 2.0 / n
    score = exact_score(h, n)
    sigma = sigma0
    for _ in range(n_trials):
        sigma *= 0.97
        if sigma < 1e-8:
            break
        z = h + rng.normal(0, sigma, n)
        zp = project_capped(z, n)
        if zp is None or not is_feasible(zp, n):
            continue
        sz = exact_score(zp, n)
        if sz < score - 1e-12:
            h = zp
            score = sz
    return h, score
