        fp = project_capped(polished, n)
        if fp is not None:
            fs = exact_score(fp, n)
            if np.isfinite(fs) and fs < best_sc:
                best_vec, best_sc = fp, fs
    # Finalize
    final = project_capped(best_vec, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
