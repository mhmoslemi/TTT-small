    cons.append(dx * (cp.sum(v) - cp.trace(W)) <= t)
    for r in range(1, n):
        m = n - r
        tp = [W[i + r, i] for i in range(m)]
        tm = [W[i, i + r] for i in range(m)]
        cons.append(dx * (cp.sum(v[r:r + m]) - cp.sum(tp)) <= t)
        cons.append(dx * (cp.sum(v[:m]) - cp.sum(tm)) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in [cp.MOSEK, cp.CVXOPT]:
        try:
            prob.solve(solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if v.value is not None and W.value is not None:
                    vv = np.array(v.value, dtype=np.float64).ravel()
                    WW = np.array(W.value, dtype=np.float64)
                    if vv.shape == (n,) and WW.shape == (n, n) and np.all(np.isfinite(vv)) and np.all(np.isfinite(WW)):
                        return vv, (WW + WW.T) * 0.5
        except Exception:
            pass
    return None, None

def build_pool(fb, n, rng, pool_size):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n))
    try_add(pool, project_capped(make_binary(rng, n), n))
    for sd in (0.05, 0.12, 0.25):
        try_add(pool, project_capped(fb + sd * rng.standard_normal(n), n))
    for _ in range(8):
        try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    try_add(pool, project_capped(np.roll(fb, n // 2), n))
    sc = max(0.5, min(1.5, math.sqrt(47.0 / n)))
    attempts = 0
    while len(pool) < pool_size and attempts < 300:
        attempts += 1
        r = rng.random()
        if r < 0.5:
            base = pool[int(rng.integers(len(pool)))]
            sig = float(rng.choice([0.03, 0.08, 0.16, 0.32])) * sc
            v = project_capped(base + rng.normal(0, sig, n), n)
        elif r < 0.75:
            base = pool[int(rng.integers(len(pool)))]
            v = project_capped(base + rng.normal(0, 0.12 * sc, n), n)
        elif r < 0.9:
            v = project_capped(rng.uniform(0, 1, n), n)
        else:
            v = project_capped(make_binary(rng, n), n)
        try_add(pool, v)
    while len(pool) < pool_size:
        pool.append(pool[0].copy())
    return pool

def anneal_chain(h0, score0, F, n, rng, steps, best_h, best_score):
    h = h0.copy()
    sc = score0
    dx = 2.0 / n
    T0, Tf = 0.25 * dx, 0.01 * dx
    has_F = F is not None and len(F) > 0
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = T0 * (Tf / T0) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        if rng.random() < 0.80:
            i = int(rng.integers(n))
            j = int(rng.integers(n - 1))
            if j >= i:
                j += 1
            tlo = max(-h[i], h[j] - 1.0, -delta)
            thi = min(1.0 - h[i], h[j], delta)
            if thi <= tlo + 1e-12 or not np.isfinite(tlo) or not np.isfinite(thi):
                continue
            t = rng.uniform(tlo, thi)
            cand = h.copy()
            cand[i] += t
            cand[j] -= t
        else:
            if has_F:
                z = rng.standard_normal(F.shape[1])
                d = F @ z
            else:
                d = rng.standard_normal(n)
            d = d - np.mean(d)
            d[-1] = -np.sum(d[:-1])
            mx = np.max(np.abs(d))
            if mx < 1e-14 or not np.isfinite(mx):
                continue
            d = d / mx
            p = d > 1e-14
            q = d < -1e-14
            tlo, thi = -delta, delta
            if np.any(p):
                tlo = max(tlo, float(np.max(-h[p] / d[p])) + 1e-14)
                thi = min(thi, float(np.min((1.0 - h[p]) / d[p])) - 1e-14)
            if np.any(q):
                tlo = max(tlo, float(np.max((1.0 - h[q]) / d[q])) + 1e-14)
                thi = min(thi, float(np.min(-h[q] / d[q])) - 1e-14)
            if thi <= tlo + 1e-12:
                continue
            t = rng.uniform(tlo, thi)
            cand = h + t * d
        cand = project_capped(cand, n)
        if cand is None or np.max(np.abs(cand - h)) < 1e-12:
            continue
        ns = exact_score(cand, n)
        if ns < best_score[0] - 1e-12:
            best_score[0] = ns
            best_h[0] = cand.copy()
        if ns <= sc + 1e-12 or math.log(max(rng.random(), 1e-300)) < -(ns - sc) / T:
            h = cand
            sc = ns
    return h, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    if n <= 64:
        M, steps_per = 40, 600
    elif n <= 128:
        M, steps_per = 32, 300
    elif n <= 256:
        M, steps_per = 24, 140
    elif n <= 512:
        M, steps_per = 16, 60
    else:
        M, steps_per = 12, 24
    pool = build_pool(fb, n, rng, M)
    F = None
    if n <= 64:
        sv, SW = solve_moment_sdp(n)
        if sv is not None:
            center = project_capped(sv, n)
            try_add(pool, center)
            if SW is not None:
                Sigma = SW - np.outer(sv, sv)
                Sigma = (Sigma + Sigma.T) * 0.5
                rm = np.mean(Sigma, axis=1)
                gm = np.mean(Sigma)
                Sigma = Sigma - rm[:, None] - rm[None, :] + gm
                try:
                    eigv, eigvecs = np.linalg.eigh(Sigma)
                    if np.all(np.isfinite(eigv)):
                        mx_ev = max(1.0, np.max(np.abs(eigv)))
                        if eigv[0] > -1e-5 * mx_ev:
                            eigv = np.clip(eigv, 0, n / 4.0)
                            pos = eigv > 1e-10 * max(1.0, eigv[-1])
                            if np.any(pos):
                                F = eigvecs[:, pos] * np.sqrt(np.maximum(eigv[pos], 0))
                except Exception:
                    F = None
            if F is None:
                for _ in range(24):
                    try_add(pool, project_capped(fb + 0.15 * rng.standard_normal(n), n))
            else:
                for gamma in (0.25, 0.5, 1.0, 1.5):
                    for _ in range(16):
                        z = rng.standard_normal(F.shape[1])
                        try_add(pool, project_capped(center + gamma * (F @ z), n))
                        try_add(pool, project_capped(center - gamma * (F @ z), n))
    pool = pool[:192]
    scores = np.array([exact_score(pv, n) for pv in pool])
    best_h = [pool[int(np.argmin(scores))].copy()]
    best_score = [float(scores.min())]
    fb_s = exact_score(fb, n)
    if fb_s < best_score[0] - 1e-12:
        best_h[0] = fb.copy()
        best_score[0] = fb_s
    order = np.argsort(scores)
    ncap = 5 if n <= 64 else 4 if n <= 128 else 3 if n <= 256 else 2 if n <= 512 else 1
    starts = [pool[0].copy()]
    for idx in order:
        if len(starts) >= ncap:
            break
        diverse = True
        for s in starts:
            if np.max(np.abs(pool[idx] - s)) < 0.03:
                diverse = False
                break
        if diverse:
            starts.append(pool[idx].copy())
    for idx in order:
        if len(starts) >= ncap:
            break
        if not any(np.max(np.abs(pool[idx] - s)) < 0.03 for s in starts):
            starts.append(pool[idx].copy())
    starts = starts[:ncap]
    for start in starts:
        ss = exact_score(start, n)
        try:
            anneal_chain(start, ss, F, n, rng, steps_per, best_h, best_score)
        except Exception:
            pass
    bg = project_capped(best_h[0], n)
    if bg is None or not is_feasible(bg, n):
        bg, bs = fb.copy(), fb_s
    else:
        bs = exact_score(bg, n)
        if bs >= fb_s + 1e-12:
            bg, bs = fb.copy(), fb_s
    return bg.astype(np.float64), float(bs), int(n)
