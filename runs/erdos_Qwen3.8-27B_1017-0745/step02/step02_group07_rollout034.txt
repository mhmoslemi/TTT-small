                if sl == 'MOSEK':
                    kw = dict(mosek_params={'MSK_IPAR_INTPNT_MAX_ITER': 200})
                elif sl == 'CVXOPT':
                    kw = dict(CVXOPT_MAX_ITERATIONS=200)
                prob.solve(solver=sl, **kw)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    vv = np.asarray(v.value, dtype=np.float64).ravel()
                    WW = np.asarray(W.value, dtype=np.float64)
                    if vv.shape == (n,) and WW.shape == (n, n) and np.all(np.isfinite(vv)) and np.all(np.isfinite(WW)):
                        return vv, WW
            except Exception:
                continue
    except Exception:
        pass
    return None, None

def get_cov_factor(sdp_v, sdp_W, n):
    if sdp_W is None or sdp_v is None:
        return None
    try:
        Sig = sdp_W - np.outer(sdp_v, sdp_v)
        Sig = 0.5 * (Sig + Sig.T)
        rm = Sig.mean(axis=1)
        gm = Sig.mean()
        Sig = Sig - rm[:, None] - rm[None, :] + gm
        ev, evec = np.linalg.eigh(Sig)
        if not np.all(np.isfinite(ev)):
            return None
        mx = np.max(np.abs(ev))
        if ev[0] < -1e-5 * max(1.0, mx):
            return None
        ev = np.clip(ev, 0.0, n / 4.0)
        kp = ev > 1e-10 * max(np.max(ev), 1e-30)
        if kp.sum() == 0:
            return None
        return evec[:, kp] * np.sqrt(ev[kp])
    except Exception:
        return None

def build_pool(fb, n, rng, sdp_v, sdp_W, Fmat):
    pool, seen = [], []
    dedup_add(pool, seen, fb.copy())
    dedup_add(pool, seen, project_capped(fb[::-1], n))
    dedup_add(pool, seen, project_capped(1.0 - fb, n))
    dedup_add(pool, seen, project_capped(1.0 - fb[::-1], n))
    dedup_add(pool, seen, project_capped(np.roll(fb, n // 3), n))
    dedup_add(pool, seen, project_capped(np.roll(fb, n // 2), n))
    dedup_add(pool, seen, np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    dedup_add(pool, seen, project_capped(blk, n))
    alt = np.zeros(n)
    ev_idx = np.arange(0, n, 2)
    alt[ev_idx[:n // 2]] = 1.0
    if n % 2 and len(ev_idx) > n // 2: alt[ev_idx[n // 2]] = 0.5
    dedup_add(pool, seen, project_capped(alt, n))
    rb = np.zeros(n)
    pm = rng.permutation(n)
    rb[pm[:n // 2]] = 1.0
    if n % 2: rb[pm[n // 2]] = 0.5
    dedup_add(pool, seen, project_capped(rb, n))
    sc = max(0.5, min(1.5, math.sqrt(47.0 / n)))
    for sd in (0.05, 0.12, 0.25):
        dedup_add(pool, seen, project_capped(fb + rng.normal(0, sd * sc, n), n))
    for _ in range(8):
        dedup_add(pool, seen, project_capped(rng.uniform(0, 1, n), n))
    if sdp_v is not None:
        vc = project_capped(sdp_v, n)
        dedup_add(pool, seen, vc)
        if vc is not None and Fmat is not None:
            for gam in (0.25, 0.5, 1.0, 1.5):
                for _ in range(16):
                    z = rng.standard_normal(Fmat.shape[1])
                    dedup_add(pool, seen, project_capped(vc + gam * (Fmat @ z), n))
                    dedup_add(pool, seen, project_capped(vc - gam * (Fmat @ z), n))
    while len(pool) < 40:
        pool.append(fb.copy())
    return pool[:192]

def anneal_chain(h0, sc0, Fmat, n, rng, bh, bs, steps):
    dx = 2.0 / n
    cur, cs = h0.copy(), sc0
    hc = Fmat is not None
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = 0.25 * dx * (0.04) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        if rng.random() < 0.8:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j: continue
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            tlo = max(-cur[i], cur[j] - 1.0)
            thi = min(1.0 - cur[i], cur[j])
        else:
            if hc:
                d = Fmat @ rng.standard_normal(Fmat.shape[1])
            else:
                d = rng.standard_normal(n)
            d = d - np.mean(d)
            d[-1] = -np.sum(d[:-1])
            mx = np.max(np.abs(d))
            if mx < 1e-14 or not np.isfinite(mx): continue
            d /= mx
            tlo, thi = -delta, delta
            for idx in range(n):
                if abs(d[idx]) < 1e-14: continue
                if d[idx] > 0:
                    tlo = max(tlo, -cur[idx] / d[idx])
                    thi = min(thi, (1.0 - cur[idx]) / d[idx])
                else:
                    tlo = max(tlo, (1.0 - cur[idx]) / d[idx])
                    thi = min(thi, -cur[idx] / d[idx])
        tlo = max(tlo, -delta)
        thi = min(thi, delta)
        if not np.isfinite(tlo) or not np.isfinite(thi) or thi <= tlo + 1e-12:
            continue
        t = rng.uniform(tlo, thi)
        cand = project_capped(cur + t * d, n)
        if cand is None: continue
        if np.max(np.abs(cand - cur)) < 1e-10: continue
        cv = exact_score(cand, n)
        if cv < bs[0] - 1e-12:
            bs[0] = cv
            bh[0] = cand.copy()
        if cv <= cs + 1e-15:
            cur, cs = cand, cv
        elif math.log(max(rng.random(), 1e-300)) < -(cv - cs) / T:
            cur, cs = cand, cv
    return cur, cs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999: n = vk
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    sdp_v, sdp_W = (None, None)
    if n <= 64:
        sdp_v, sdp_W = solve_moment_sdp(n)
    Fmat = get_cov_factor(sdp_v, sdp_W, n)
    pool = build_pool(fb, n, rng, sdp_v, sdp_W, Fmat)
    scores = np.array([exact_score(pv, n) for pv in pool], dtype=np.float64)
    order = np.argsort(scores)
    mc = 5 if n <= 64 else (4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1)))
    starts = [pool[order[0]].copy()]
    for idx in order:
        if len(starts) >= mc: break
        px = pool[idx]
        if all(np.max(np.abs(px - sx)) >= 0.03 for sx in starts):
            starts.append(px.copy())
    while len(starts) < min(3, mc):
        starts.append(pool[order[0]].copy())
    steps = 600 if n <= 64 else (300 if n <= 128 else (140 if n <= 256 else 60))
    bi = int(np.argmin(scores))
    bh = [pool[bi].copy()]
    bs = [float(scores[bi])]
    for st in starts:
        anneal_chain(st, exact_score(st, n), Fmat, n, rng, bh, bs, steps)
    anneal_chain(bh[0], bs[0], Fmat, n, rng, bh, bs, steps * 2 // 3)
    bg = project_capped(bh[0], n)
    fb_s = exact_score(fb, n)
    if bg is None or not is_feasible(bg, n):
        bg, bs = fb.copy(), fb_s
    else:
        bs2 = exact_score(bg, n)
        bg, bs = (bg, bs2) if bs2 <= fb_s + 1e-12 else (fb.copy(), fb_s)
    if not np.isfinite(bs):
        bg, bs = fb.copy(), fb_s
    return bg.astype(np.float64), float(bs), int(n)
