            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except Exception:
            n = 47
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(np.max(np.correlate(h1, 1.0 - h1, mode='full') * 2.0)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    base = np.full(n, 0.5)
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None:
            base = pp
    safe_h = base.copy()
    safe_s = exact_score(safe_h, n)
    F = (n - 1) // 2
    if F == 0:
        cutoffs = [0]
    elif n <= 64:
        cutoffs = sorted(set([k for k in [1, 2, 4, 8, 16, F] if 1 <= k <= F] + [F]))
    elif n <= 256:
        cutoffs = sorted(set([k for k in [1, 2, 4, 8, 16, min(F, 20)] if 1 <= k <= F] + [F]))
    else:
        cutoffs = sorted(set([k for k in [1, 2, 4, 8, min(F, 12)] if 1 <= k <= F] + [F]))
    Phi, labels = build_fourier_basis(n)
    n_starts = 6 if n <= 64 else (4 if n <= 256 else 3)
    starts = build_starts(base, n, rng, n_starts)
    gb, gs = safe_h.copy(), safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, Phi, labels, n, rng, cutoffs)
            if xo is not None and so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        kl, ks = random_local_search(gb, n, rng, 300, 0.04)
        if ks < gs - 1e-12:
            gs, gb = ks, kl.copy()
    except Exception:
        pass
    for st in starts[:2]:
        try:
            x2, s2 = optimize_start(gb, Phi, labels, n, rng, cutoffs)
            if x2 is not None and s2 < gs - 1e-12:
                gs, gb = s2, x2.copy()
        except Exception:
            pass
    try:
        k2, s3 = random_local_search(gb, n, rng, 200, 0.015)
        if s3 < gs - 1e-12:
            gs, gb = s3, k2.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or exact_score(final, n) > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    else:
        fs = exact_score(final, n)
        if fs > safe_s + 1e-12:
            final, fs = safe_h.copy(), safe_s
    final = final.astype(np.float64)
    return final, float(fs), int(n)
