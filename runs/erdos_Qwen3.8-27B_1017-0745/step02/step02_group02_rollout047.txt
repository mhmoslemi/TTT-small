    r_sel = list(receivers[:nr])
    d_set = set(d_sel)
    r_sel = [x for x in r_sel if x not in d_set]
    if len(r_sel) < 1 or len(d_sel) < 1:
        return None
    r_sel = r_sel[:nr]
    v = np.zeros(n)
    wd = rng.uniform(0.5, 1.0, len(d_sel))
    wr = rng.uniform(0.5, 1.0, len(r_sel))
    for j in range(len(d_sel)):
        v[d_sel[j]] = -wd[j] / wd.sum()
    for j in range(len(r_sel)):
        v[r_sel[j]] = wr[j] / wr.sum()
    v -= np.mean(v)
    mx = np.max(np.abs(v))
    if mx < 1e-14:
        return None
    v /= mx
    return v

def quad_roots(a, b, c):
    if abs(a) < 1e-16:
        if abs(b) < 1e-16:
            return []
        return [-c / b]
    disc = b * b - 4.0 * a * c
    if disc < -1e-10:
        return []
    if disc < 0.0:
        disc = 0.0
    sq = math.sqrt(disc)
    qval = -0.5 * (b + math.copysign(sq, b))
    r1 = qval / a
    r2 = c / qval if abs(qval) > 1e-30 else r1
    return [r1, r2]

def line_search(h, n, v):
    if v is None or not np.all(np.isfinite(v)):
        return h, exact_score(h, n)
    lo_b = -1e10
    hi_b = 1e10
    for i in range(n):
        if v[i] > 1e-15:
            lo_b = max(lo_b, -h[i] / v[i])
            hi_b = min(hi_b, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            lo_b = max(lo_b, (1.0 - h[i]) / v[i])
            hi_b = min(hi_b, -h[i] / v[i])
    if not (np.isfinite(lo_b) and np.isfinite(hi_b)) or hi_b - lo_b < 1e-13:
        return h, exact_score(h, n)
    lo_b = max(lo_b, -1e6)
    hi_b = min(hi_b, 1e6)
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros(L)
    B = np.zeros(L)
    C = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        hp = h[p0:p0 + m]
        hq = h[q0:q0 + m]
        vp = v[p0:p0 + m]
        vq = v[q0:q0 + m]
        A[row] = dx * np.sum(hp * (1.0 - hq))
        B[row] = dx * np.sum(vp * (1.0 - hq) - hp * vq)
        C[row] = -dx * np.sum(vp * vq)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, exact_score(h, n)
    cands = [lo_b, hi_b, 0.0]
    if n <= 72:
        for s in range(L):
            if abs(C[s]) > 1e-15:
                sp = -B[s] / (2.0 * C[s])
                if lo_b - 1e-10 <= sp <= hi_b + 1e-10:
                    cands.append(max(lo_b, min(hi_b, sp)))
        for s in range(L):
            for t in range(s + 1, L):
                da = C[s] - C[t]
                db = B[s] - B[t]
                dc = A[s] - A[t]
                for rt in quad_roots(da, db, dc):
                    if lo_b - 1e-10 <= rt <= hi_b + 1e-10:
                        cands.append(max(lo_b, min(hi_b, rt)))
        if len(cands) > 9000:
            cands = cands[:9000]
    else:
        npts = 49 if n <= 256 else 25
        cands.extend(np.linspace(lo_b, hi_b, npts).tolist())
    cands = np.array(cands, dtype=np.float64)
    cands = cands[(cands >= lo_b - 1e-12) & (cands <= hi_b + 1e-12)]
    cands = np.clip(cands, lo_b, hi_b)
    cands = np.unique(np.round(cands, 13))
    best_f = float('inf')
    best_a = 0.0
    chunk = 512
    for ci in range(0, len(cands), chunk):
        al = cands[ci:ci + chunk]
        vals = A[None, :] + al[:, None] * B[None, :] + al[:, None] * C[None, :]
        f = np.max(vals, axis=1)
        im = np.argmin(f)
        if f[im] < best_f:
            best_f = float(f[im])
            best_a = float(al[im])
    s_cur = exact_score(h, n)
    try_alphas = [best_a, 0.0]
    if abs(best_a) > 1e-14:
        try_alphas.extend([best_a * 0.9, best_a * 1.1, best_a * 0.5])
    best_h = h
    best_s = s_cur
    for ta in try_alphas:
        ht = h + ta * v
        ht = project_capped(ht, n)
        if ht is None:
            continue
        st = exact_score(ht, n)
        if st < best_s - 1e-13:
            best_s = st
            best_h = ht
    return best_h, best_s

def build_starts(fb, n, rng, count):
    pool = []
    cands = []
    if fb is not None:
        cands.append(fb.copy())
        cands.append(fb[::-1].copy())
        cands.append(1.0 - fb[::-1])
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    i = np.arange(n, dtype=np.float64)
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (i + 0.5) / n), 0, 1))
    cands.append(rng.uniform(0, 1, n))
    if fb is not None:
        cands.append(fb + 0.1 * rng.standard_normal(n))
        cands.append(fb + 0.05 * rng.standard_normal(n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool

def do_sweep(h, n, rng, n_modes, n_box):
    cur = h.copy()
    cur_s = exact_score(cur, n)
    k_list = np.arange(1, n)
    if len(k_list) > n_modes:
        k_list = np.unique(np.round(np.logspace(0, math.log10(n - 1), n_modes)).astype(int))
        k_list = np.clip(k_list, 1, n - 1)
    rng.shuffle(k_list)
    for k in k_list:
        v = dct_direction(int(k), n)
        if v is None:
            continue
        new_h, new_s = line_search(cur, n, v)
        if new_s < cur_s - 1e-12:
            cur = new_h
            cur_s = new_s
    for _ in range(n_box):
        v = box_direction(cur, n, rng)
        if v is None:
            continue
        new_h, new_s = line_search(cur, n, v)
        if new_s < cur_s - 1e-12:
            cur = new_h
            cur_s = new_s
    return cur, cur_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                n = int(p.size)
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
    if 'n_points' in kwargs:
        try:
            vv = int(kwargs['n_points'])
            if 1 <= vv <= 999:
                n = vv
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, 1), 1
    if n <= 48:
        n_modes, n_box, n_start, n_sw = n - 1, 3, 4, 3
    elif n <= 64:
        n_modes, n_box, n_start, n_sw = 24, 3, 3, 2
    elif n <= 256:
        n_modes, n_box, n_start, n_sw = 12, 2, 3, 1
    else:
        n_modes, n_box, n_start, n_sw = 8, 1, 2, 1
    starts = build_starts(fb, n, rng, n_start)
    if not starts:
        starts = [fb]
    gb = fb.copy()
    gs = exact_score(fb, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            cur, cur_s = st.copy(), exact_score(st, n)
            for _ in range(n_sw):
                new_cur, new_s = do_sweep(cur, n, rng, n_modes, n_box)
                if new_s >= cur_s - 1e-12:
                    break
                cur, cur_s = new_cur, new_s
            if cur_s < gs - 1e-12:
                gs, gb = cur_s, cur.copy()
        except Exception:
            continue
    for _ in range(2):
        new_cur, new_s = do_sweep(gb, n, rng, n_modes, n_box)
        if new_s >= gs - 1e-12:
            break
        gs, gb = new_s, new_cur.copy()
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
