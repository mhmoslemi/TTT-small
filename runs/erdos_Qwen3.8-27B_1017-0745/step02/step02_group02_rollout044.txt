        c = (1.0-a)*h + a*tgt
    else:

        k = 1 if r < 0.60 else (2 if r < 0.90 else 4)

            j = int(ar[int(rng.integers(len(ar)))])
            if i == j: continue

            if cv <= 0: continue
            d = cv * (0.15 + 0.85*rng.random())
            c[i] -= d; c[j] += d

def threshold_chain(h0, n, rng, bgt, pool, gb, gs, stages, quota):
    h = h0.copy()
    bgt.count += 1
    s = exact_score(h, n)
    if s < gs - 1e-12: gb, gs = h.copy(), s
    ns = len(stages)
    per = max(1, quota // ns) if ns > 0 else 0
    for si in range(ns):
        nsteps = per if si < ns-1 else per + quota - per*(ns-1)
        thr, rad = stages[si]
        for _ in range(nsteps):
            if bgt.count >= bgt.limit: return h, s, gb, gs
            c = make_proposal(h, n, rng, rad, pool, gb)
            if c is None: continue
            bgt.count += 1
            nv = exact_score(c, n)
            if nv <= s + thr + 1e-14:
                h, s = c, nv
                if s < gs - 1e-12: gb, gs = h.copy(), s
    return h, s, gb, gs

def line_scan(h, n, rng, bgt, n_pairs):
    best, bs = h.copy(), exact_score(h, n)
    seen = set()
    for _ in range(n_pairs):
        if bgt.count >= bgt.limit: break
        i, j = int(rng.integers(n)), int(rng.integers(n))
        pk = (min(i,j), max(i,j))
        if i == j or pk in seen: continue
        seen.add(pk)
        lo = max(-h[i], h[j]-1.0)
        hi = min(1.0-h[i], h[j])
        if hi-lo < 1e-14: continue
        scored = {}
        dvals = []
        for r in range(7):
            d = lo + r*(hi-lo)/6.0
            dvals.append(d)
            if abs(d) < 1e-14:
                scored[0.0] = bs
            else:
                c = h.copy(); c[i] += d; c[j] -= d
                bgt.count += 1
                if bgt.count > bgt.limit: return best, bs
                scored[d] = exact_score(c, n)
        if 0.0 not in scored: scored[0.0] = bs
        bd = pick_best(scored)
        if bd is None or scored[bd] >= bs - 1e-12: continue
        for _ in range(2):
            if bgt.count >= bgt.limit - 3: break
            idx = dvals.index(bd) if bd in dvals else 3
            ilo = dvals[idx-1] if idx > 0 else lo
            ihi = dvals[idx+1] if idx < 6 else hi
            for f in (0.25, 0.5, 0.75):
                md = ilo + f*(ihi-ilo)
                c = h.copy(); c[i] += md; c[j] -= md
                bgt.count += 1
                if bgt.count > bgt.limit: break
                scored[md] = exact_score(c, n)
            bd2 = pick_best(scored)
            if bd2 is not None and scored[bd2] < scored[bd]:
                bd = bd2
        c = h.copy(); c[i] += bd; c[j] -= bd
        p = project_capped(c, n)
        if p is not None:
            h, bs, best = p, scored[bd], p.copy()
    return best, bs
