    cands_list.append(blk)
    cands_list.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    idx = np.arange(n)
    cands_list.append(0.5 + 0.35 * np.cos(2.0 * np.pi * idx / n))
    cands_list.append(0.5 + 0.35 * np.cos(4.0 * np.pi * idx / n))
    cands_list.append(fallback + rng.normal(0, 0.05, n))
    cands_list.append(fallback + rng.normal(0, 0.15, n))
    for cv in cands_list:
        pz = project_capped(cv, n)
        if pz is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(pz - s)) < 1e-8:
                dup = True
                break
        if not dup and len(pool) < 10:
            pool.append(pz)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if not (1 <= n <= 999):
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    fb_score = exact_score(fallback, n)
    if n > 64:
        return fallback, float(fb_score), int(n)
    starts = build_starts(fallback, n, rng)
    scored = [(exact_score(st, n), st) for st in starts]
    scored.sort(key=lambda x: x[0])
    selected = [scored[i][1] for i in range(min(3, len(scored)))]
    dirs = build_directions(n)
    B = min(276, max(n - 1, int(276.0 * (47.0 / n)**2)))
    chain_count = min(len(selected), 3, max(1, B // max(1, n - 1)))
    per_chain = B // chain_count
    best_h, best_s = fallback.copy(), fb_score
    for ci in range(chain_count):
        budget = per_chain + (1 if ci < B % chain_count else 0)
        cur_h, cur_s = selected[ci].copy(), exact_score(selected[ci], n)
        if cur_s < best_s - 1e-12:
            best_h, best_s = cur_h.copy(), cur_s
        lines_used = 0
        stagnant = 0
        fwd = True
        while lines_used < budget and stagnant < 2:
            improved = False
            dlist = dirs if fwd else dirs[::-1]
            for v in dlist:
                if lines_used >= budget:
                    break
                lines_used += 1
                try:
                    nh, ns = do_line_search(cur_h, cur_s, v, n)
                except Exception:
                    continue
                if ns < cur_s - 1e-12:
                    cur_h, cur_s = nh, ns
                    improved = True
                    if cur_s < best_s - 1e-12:
                        best_h, best_s = cur_h.copy(), cur_s
            fwd = not fwd
            if not improved:
                stagnant += 1
            else:
                stagnant = 0
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
