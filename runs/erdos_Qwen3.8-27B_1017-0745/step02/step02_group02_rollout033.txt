Let me be careful about the quadratic coefficients:
- For lag s: p_s = max(s,0), r_s = max(-s,0), m_s = n - |s|
- q_s(h) = dx * sum_j h[p+j]*(1-h[r+j])
- Along h + alpha*v:
  - a_s = q_s(h)
  - b_s = dx * sum_j [v[p+j]*(1-h[r+j]) - h[p+j]*v[r+j]]
  - c_s = -dx * sum_j v[p+j]*v[r+j]

So q_s(h+alpha*v) = a_s + b_s*alpha + c_s*alpha^2

For the exact line search (n<=64):
- Find lo, hi from box constraints
- For each pair (s,t), find roots of (a_s-a_t) + (b_s-b_t)*alpha + (c_s-c_t)*alpha^2 = 0
- Include stationary points -b_s/(2*c_s) where c_s != 0
- Include 0, lo, hi
- Evaluate F at all candidates
- Pick best

For the quadratic roots: A*alpha^2 + B*alpha + C = 0
- If |A| < eps: linear B*alpha + C = 0
- Else: discriminant = B^2 - 4*A*C, if negative (within tolerance) set to 0
- Roots: (-B ± sqrt(disc)) / (2*A)

    return A, B, C

def feasible_interval(h, n, v):
    lo, hi = -1e18, 1e18
    for i in range(n):
        if v[i] > 1e-14:
            lo = max(lo, -h[i] / v[i]); hi = min(hi, (1.0-h[i]) / v[i])
        elif v[i] < -1e-14:
            lo = max(lo, (1.0-h[i]) / v[i]); hi = min(hi, -h[i] / v[i])
    if not (np.isfinite(lo) and np.isfinite(hi)) or hi - lo < 1e-12:
        return None
    return lo, hi

def do_line_exact(h, n, v):
    iv = feasible_interval(h, n, v)
    if iv is None:
        return h, exact_score(h, n)
    lo, hi = iv
    A, B, C = line_coeffs(h, n, v)
    if not np.all(np.isfinite(A + B + C)):
        return h, exact_score(h, n)
    cands = [lo, hi, 0.0]
    for s in range(len(A)):
        if abs(C[s]) > 1e-14:
            ap = -B[s] / (2.0 * C[s])
            if lo - 1e-9 < ap < hi + 1e-9:
                cands.append(ap)
    for s in range(len(A)):
        for t in range(s + 1, len(A)):
            for r in quad_roots(C[s]-C[t], B[s]-B[t], A[s]-A[t]):
                if np.isfinite(r) and lo - 1e-9 < r < hi + 1e-9:
                    cands.append(r)
    cands = sorted(set(max(lo, min(hi, c)) for c in cands if np.isfinite(c)))
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    merged = [cands[0]]
    for c in cands[1:]:
        if c - merged[-1] > tol:
            merged.append(c)
    best_alpha, best_f = 0.0, float(np.max(A))
    for i in range(0, len(merged), 256):
        pts = np.array(merged[i:i+256])
        vals = A[None,:] + B[None,:]*pts[:,None] + C[None,:]*pts[:,None]**2
        fs = np.max(vals, axis=1)
        if fs[0] < best_f:
            best_f, best_alpha = float(fs[0]), float(pts[0])
    s_cur = exact_score(h, n)
    if best_f >= s_cur - 1e-12:
        return h, s_cur
    h_new = project_capped(h + best_alpha * v, n)
    if h_new is None:
        return h, s_cur
    s_new = exact_score(h_new, n)
    return (h_new, s_new) if s_new < s_cur - 1e-12 else (h, s_cur)

def do_line_grid(h, n, v):
    iv = feasible_interval(h, n, v)
    if iv is None:
        return h, exact_score(h, n)
    lo, hi = iv
    A, B, C = line_coeffs(h, n, v)
    if not np.all(np.isfinite(A + B + C)):
        return h, exact_score(h, n)
    ng = 33 if n <= 256 else 17
    nr = 4 if n <= 256 else 3
    def F_eval(pts):
        return np.max(A[None,:]+B[None,:]*pts[:,None]+C[None,:]*pts[:,None]**2, axis=1)
    best_alpha, best_f = 0.0, float(np.max(A))
    pts = np.linspace(lo, hi, ng)
    pts = np.insert(pts, int(np.argmin(np.abs(pts))), 0.0)
    fs = F_eval(pts)
    lb = int(np.argmin(fs))
    if fs[lb] < best_f:
        best_f, best_alpha = float(fs[lb]), float(pts[lb])
    for _ in range(nr):
        plo = pts[0] if lb == 0 else pts[lb-1]
        phi = pts[-1] if lb == len(pts)-1 else pts[lb+1]
        pts = np.linspace(plo, phi, 9)
        fs = F_eval(pts); lb = int(np.argmin(fs))
        if fs[lb] < best_f:
            best_f, best_alpha = float(fs[lb]), float(pts[lb])
    s_cur = exact_score(h, n)
    if best_f >= s_cur - 1e-12:
        return h, s_cur
    h_new = project_capped(h + best_alpha * v, n)
    if h_new is None:
        return h, s_cur
    s_new = exact_score(h_new, n)
    return (h_new, s_new) if s_new < s_cur - 1e-12 else (h, s_cur)

def do_sweep(h, n, rng, modes):
    h_c = h.copy(); sc = exact_score(h_c, n)
    order = list(modes); rng.shuffle(order)
    for k in order:
        v = dct_dir(k, n)
        if v is None:
            continue
        if n <= 64:
            h_c, sc = do_line_exact(h_c, n, v)
        else:
            h_c, sc = do_line_grid(h_c, n, v)
    n_bd = 2 if n <= 256 else 1
    for _ in range(n_bd):
        v = box_dir(h_c, n, rng)
        if v is None:
            continue
        if n <= 64:
            h_c, sc = do_line_exact(h_c, n, v)
        else:
            h_c, sc = do_line_grid(h_c, n, v)
    return h_c, sc

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
            n = int(p.size); parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None and len(parent) != n:
        old_n = len(parent)
        parent = np.interp((np.arange(n)+0.5)/n, (np.arange(old_n)+0.5)/old_n, parent)
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, 1), 1
    gs, gb = exact_score(fb, n), fb.copy()
    starts = [fb.copy()]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2: blk[n//2] = 0.5
    p2 = project_capped(blk, n)
    if p2 is not None: starts.append(p2)
    if parent is not None:
        p3 = project_capped(parent + rng.normal(0, 0.08, n), n)
        if p3 is not None: starts.append(p3)
    uniq = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-8 for u in uniq):
            uniq.append(s)
    starts = uniq[:3] if n <= 256 else uniq[:2]
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 48:
        mode_keys = list(range(1, n))
    elif n <= 64:
        mode_keys = sorted(set(int(round(x)) for x in np.geomspace(1, n-1, 24)))
    elif n <= 256:
        mode_keys = sorted(set(int(round(x)) for x in np.geomspace(1, n-1, 12)))
    else:
        mode_keys = sorted(set(int(round(x)) for x in np.geomspace(1, n-1, 8)))
    for st in starts:
        try:
            h_c, sc = do_sweep(st, n, rng, mode_keys)
            if sc < gs - 1e-12:
                gs, gb = sc, h_c.copy()
        except Exception:
            continue
    n_polish = 2 if n <= 64 else 1
    for _ in range(n_polish):
        improved = False
        try:
            h_c, sc = do_sweep(gb, n, rng, mode_keys)
            if sc < gs - 1e-12:
                gs, gb = sc, h_c.copy(); improved = True
        except Exception:
            break
        if not improved:
            break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5); sf = exact_score(final, n)
    return final, float(sf), int(n)
