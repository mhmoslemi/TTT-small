                break
            g = smooth_grad(h, n, p)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _ in range(24):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h
                mn = float(np.max(np.abs(d)))
                if mn < 1e-11:
                    break
                gd = float(np.dot(g, d))
                if gd >= -1e-14:
                    alpha *= 0.5
                    continue
                Fy = lag_values(y, n)
                obj_y, _ = softmax_weights(Fy, theta)
                if obj_y is None:
                    alpha *= 0.5
                    continue
                if obj_y <= obj_h + 1e-4 * gd + 1e-12 * max(1.0, abs(obj_h)):
                    accepted = True
                    h = y
                    break
                alpha *= 0.5
            if not accepted:
                break
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < bs - 1e-12:
                bs = sc
                bh = h.copy()
    return bh, bs

def add_start(starts, v, n, tol=1e-8):
    h2 = project_capped(np.asarray(v, dtype=np.float64), n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < tol:
            return
    starts.append(h2)

def coord_refine(h, n, iters):
    best = h.copy()
    bs = exact_score(best, n)
    for _ in range(iters):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [0.0, lo + 1e-12, hi - 1e-12]
                for sgn in (1.0, -1.0):
                    for step in (0.01, 0.05, 0.15):
                        dd = sgn * step
                        if lo <= dd <= hi:
                            cands.append(dd)
                bd, bsc = 0.0, bs
                for dd in cands:
                    dd = max(lo, min(hi, dd))
                    tri = best.copy()
                    tri[i] += dd
                    tri[j] -= dd
                    sc = exact_score(tri, n)
                    if sc < bsc - 1e-13:
                        bd, bsc = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    bs = bsc
                    improved = True
        if not improved:
            break
    return best, bs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        sd = abs(int(seed)) % (2**31)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 128:
        itp, nsm = 80, 7
    elif n <= 512:
        itp, nsm = 30, 4
    else:
        itp, nsm = 12, 2
    try:
        itp = max(5, min(120, int(kwargs.get('max_iter', itp))))
    except Exception:
        pass
    try:
        ns = max(1, min(8, int(kwargs.get('starts', nsm))))
    except Exception:
        ns = nsm
    tau_list = [0.02, 0.006, 0.0018, 0.0005, 0.00015, 0.00004, 0.00001]
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:ns]
    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    for st in starts:
        try:
            bh, bs = cont_optimize(st, n, itp, tau_list, 0.2, best, best_score)
            if bs < best_score - 1e-12:
                best, best_score = bh, bs
        except Exception:
            pass
    if n <= 128:
        for std in (0.025, 0.08):
            try:
                pert = project_capped(best + rng.normal(0, std, n), n)
                if pert is not None:
                    bh, bs = cont_optimize(pert, n, itp, tau_list[2:], 0.15, best, best_score)
                    if bs < best_score - 1e-12:
                        best, best_score = bh, bs
            except Exception:
                pass
    elif n <= 512:
        try:
            pert = project_capped(best + rng.normal(0, 0.05, n), n)
            if pert is not None:
                bh, bs = cont_optimize(pert, n, itp // 2, tau_list[3:], 0.15, best, best_score)
                if bs < best_score - 1e-12:
                    best, best_score = bh, bs
        except Exception:
            pass
    if n <= 64:
        try:
            pr, ps = coord_refine(best, n, 2)
            pf = project_capped(pr, n)
            if pf is not None:
                fs2 = exact_score(pf, n)
                if np.isfinite(fs2) and fs2 < best_score - 1e-12:
                    best, best_score = pf, fs2
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
