import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def project_simplex(v):
    m = len(v)
    if m <= 0 or not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    cum = np.cumsum(u)
    idx = np.arange(1, m + 1)
    mask = u - (cum - 1.0) / idx > 0
    if not np.any(mask):
        w = np.zeros(m); w[0] = 1.0; return w
    j = int(np.max(np.where(mask)))
    lam = (cum[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    pos = np.where(w > 0)[0]
    if len(pos) > 0:
        w[pos[0]] += 1.0 - float(np.sum(w)); w[pos[0]] = max(w[pos[0]], 0.0)
    w = np.clip(w, 0.0, None)
    return w if np.all(np.isfinite(w)) else None

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    for d in range(-(n - 1), n):
        r = d + n - 1
        lo = max(0, -d); hi = min(n, n - d)
        q[r] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def dir_products(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    r = np.zeros(m)
    for d in range(-(n - 1), n):
        ri = d + n - 1
        lo = max(0, -d); hi = min(n, n - d)
        r[ri] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d])) - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def weighted_grad(x, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        ri = d + n - 1
        wd = dx * w[ri]
        if abs(wd) < 1e-15:
            continue
        lo = max(0, -d); hi = min(n, n - d)
        g[lo:hi] += wd * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= wd * x[lo:hi]
    return g

def fro_norm_sq(x, n):
    dx = 2.0 / n
    total = 0.0
    for d in range(-(n - 1), n):
        lo = max(0, -d); hi = min(n, n - d)
        gd = np.zeros(n)
        gd[lo:hi] += dx * (1.0 - x[lo + d:hi + d])
        gd[lo + d:hi + d] -= dx * x[lo:hi]
        total += float(np.dot(gd, gd))
    return total
