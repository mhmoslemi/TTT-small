import numpy as np

M_LAT = 65536

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-12:
            break
        if r > 0:
            for i in range(n):
                if abs(r) < 1e-14: break
                add = min(1.0 - h[i], r)
                if add > 0: h[i] += add; r -= add
        else:
            for i in range(n):
                if abs(r) < 1e-14: break
                sub = min(h[i], -r)
                if sub > 0: h[i] -= sub; r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c))
    except Exception:
        return float('inf')

def to_units(h, n):
    if not is_feasible(h, n):
        return None
    K = M_LAT * n // 2
    z = np.clip(h, 0.0, 1.0) * M_LAT
    base = np.floor(z).astype(np.int64)
    base = np.clip(base, 0, M_LAT)
    frac = z - base.astype(np.float64)
    R = K - int(np.sum(base))
    if R > 0:
        order = np.argsort(-frac)
        cnt = 0
        for i in order:
            if cnt >= R: break
            if base[i] < M_LAT:
                base[i] += 1; cnt += 1
        if cnt != R: return None
    elif R < 0:
        order = np.argsort(frac)
        cnt = 0
        for i in order:
            if cnt >= -R: break
            if base[i] > 0:
                base[i] -= 1; cnt += 1
        if cnt != -R: return None
    if np.any(base < 0) or np.any(base > M_LAT) or int(np.sum(base)) != K:
        return None
    return base

def units_to_h(u):
    return u.astype(np.float64) / M_LAT

def propose_exchange(u, n, qmax, rng):
    sizes = [sz for sz in (2, 4, 8) if sz <= n]
    if not sizes:
        return None
    ws = np.array([0.55, 0.30, 0.15])
    idxs = [i for i, sz in enumerate((2, 4, 8)) if sz <= n]
    ww = ws[idxs] / ws[idxs].sum()
    k = sizes[int(rng.choice(len(sizes), p=ww))]
    perm = rng.permutation(n)
    chosen = perm[:k]
    q = int(rng.integers(1, qmax + 1))
    v = u.copy()
    for i in chosen[:k // 2]: v[i] += q
    for i in chosen[k // 2:]: v[i] -= q
    if np.any(v < 0) or np.any(v > M_LAT):
        return None
    return v

    if ph is None: return
