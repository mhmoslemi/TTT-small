def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:

            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:

            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):

    t0 = max(0.0, t0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_fun, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': n_max_iter, 'disp': False})
    except Exception:
        return None
    if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x)):
        return res.x
    return None

def build_starts(fallback, n, rng, count):
    pool = []
    cands = [fallback.copy(), np.full(n, 0.5)]
    if n > 1:
        cands.append(fallback[::-1].copy())
        cands.append(1.0 - fallback.copy())
        cands.append(1.0 - fallback[::-1].copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk.copy())
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for m in [1, 2, 3]:
        for ph in [0.0, np.pi / 2]:
            cands.append(0.5 + 0.45 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph))
    for sd in [0.04, 0.10, 0.20]:
        cands.append(fallback + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(np.asarray(c, dtype=np.float64), n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool
