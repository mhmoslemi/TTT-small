            others = [j for j in range(P) if j != i]
            rng.shuffle(others)
            a, b, c = others[0], others[1], others[2]
            if rng.random() < 0.25:
                v = frozen[i] + F * (frozen[gidx] - frozen[i]) + F * (frozen[b] - frozen[c])
            else:
                v = frozen[a] + F * (frozen[b] - frozen[c])
            mask = rng.random(n) < 0.85
            if not mask.any():
                mask[rng.integers(n)] = True
            trial = np.where(mask, v, frozen[i])
            tp = project_capped(trial, n)
            if tp is None:
                continue
            ts = exact_score(tp, n)
            if not np.isfinite(ts):
                continue
            delta = ts - fr_sc[i]
            acc = False
            if delta < -1e-12:
                acc = True
            elif abs(delta) <= 1e-12:
                acc = rng.random() < 0.1
            else:
                acc = (delta / T_gen) < 700.0 and rng.random() < exp(-delta / T_gen)
            if acc:
                pop[i] = tp
                scores[i] = ts
            if ts < gen_best - 1e-12:
                gen_best = ts
                best_s = ts
                best_h = tp.copy()
        gi = int(np.argmin(scores))
        pop[0] = best_h.copy()
        scores[0] = best_s
        if best_s >= gen_best + 1e-14 and best_s >= safe_s:
            stall += 1
        else:
            stall = 0
        if stall >= stall_lim and gen < G - 1:
            keep = max(2, P // 3)
            pop = [pop[0].copy()] + [pop[k].copy() for k in range(1, min(keep, P))]
            scores = [scores[0]] + [scores[k] for k in range(1, min(keep, P))]
            rep = P - len(pop)
            for rr in range(rep):
                if rr < rep // 2:
                    cand = best_h + (0.025 + 0.075 * (1.0 - q)) * rng.standard_normal(n)
                else:
                    cand = rng.uniform(0, 1, n)
                cp = project_capped(cand, n)
                if cp is not None:
                    cs = exact_score(cp, n)
                    if np.isfinite(cs):
                        if cs < best_s - 1e-12:
                            best_s = cs
                            best_h = cp.copy()
                        pop.append(cp)
                        scores.append(cs)
            while len(pop) < P:
                pop.append(best_h.copy())
                scores.append(best_s)
            pop[0] = best_h.copy()
            scores[0] = best_s
            stall = 0
    curr = best_h.copy()
    curr_s = best_s
    for t in range(L):
        qq = t / max(1, L - 1)
        T_loc = 5e-4 * (1e-7 / 5e-4) ** qq
        rho = 1.0 - 0.98 * qq
        moved = False
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if rng.random() < 0.5:
                rv, dn = i, j
            else:
                rv, dn = j, i
            cap = min(1.0 - curr[rv], curr[dn])
            if cap <= 0:
                continue
            amt = cap * rho * (0.05 + 0.95 * rng.random())
            nh = curr.copy()
            nh[rv] += amt
            nh[dn] -= amt
            hp = project_capped(nh, n)
            if hp is None:
                continue
            hs = exact_score(hp, n)
            if not np.isfinite(hs):
                continue
            d = hs - curr_s
            if d < -1e-12 or (d / T_loc) < 700.0 and rng.random() < exp(-d / T_loc):
                curr = hp
                curr_s = hs
                moved = True
            break
        if not moved:
            if curr_s < best_s - 1e-12:
                best_s = curr_s
                best_h = curr.copy()
            continue
        if curr_s < best_s - 1e-12:
            best_s = curr_s
            best_h = curr.copy()
    curr = best_h.copy()
    curr_s = best_s
    fracs = (1.0 / 64, 1.0 / 16, 1.0 / 4, 1.0 / 2, 1.0)
    for sweep in range(n_pol):
        imp = 0
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            npairs = max(40, 4000 // n)
            ps = set()
            for _ in range(npairs * 2):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii != jj:
                    ps.add((min(ii, jj), max(ii, jj)))
            pairs = list(ps)
        for (i, j) in pairs:
            for (rv, dn) in [(i, j), (j, i)]:
                cap = min(1.0 - curr[rv], curr[dn])
                if cap <= 0:
                    continue
                found = False
                for fr in fracs:
                    test = curr.copy()
                    test[rv] += cap * fr
                    test[dn] -= cap * fr
                    tp = project_capped(test, n)
                    if tp is None:
                        continue
                    ts = exact_score(tp, n)
                    if ts < curr_s - 1e-12:
                        curr = tp
                        curr_s = ts
                        imp += 1
                        found = True
                        break
                if found:
                    break
        if curr_s < best_s - 1e-12:
            best_s = curr_s
            best_h = curr.copy()
        if imp == 0:
            break
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final = safe_h.copy()
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
