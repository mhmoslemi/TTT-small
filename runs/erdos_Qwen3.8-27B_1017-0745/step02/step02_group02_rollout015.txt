    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 5e-12 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_value_and_grad(h, n, tau):
    q = lag_values(h, n)
    qmax = np.max(q)
    shifted = (q - qmax) / tau
    ex = np.exp(np.clip(shifted, -50.0, 0.0))
    s = np.sum(ex)
    if s < 1e-300 or not np.isfinite(s):
        return None, None
    w = ex / s
    val = qmax + tau * np.log(s)
    if not np.isfinite(val):
        return None, None
    dx = 2.0 / n
    L = 2 * n - 1
    g = np.zeros(n)
    for k in range(L):
        s_val = k - (n - 1)
        p0 = max(s_val, 0)
        r0 = max(-s_val, 0)
        m = n - abs(s_val)
        g[p0:p0+m] += dx * w[k] * (1.0 - h[r0:r0+m])
        g[r0:r0+m] -= dx * w[k] * h[p0:p0+m]
    if not np.all(np.isfinite(g)):
        return None, None
    return val, g

def continuation_run(h0, n, maxiter_per_tau):
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    h = h0.copy().astype(np.float64)
    dx = 2.0 / n
    best_h = h.copy()
    best_s = exact_score(h, n)
    eta_max = 2.0 / dx
    for tau in taus:
        eta = 0.25 / dx
        for _it in range(maxiter_per_tau):
            Fcur, g = smooth_value_and_grad(h, n, tau)
            if Fcur is None or g is None:
                break
            z = project_capped(h - eta * g, n)
            if z is None:
                break
            d = z - h
            if np.max(np.abs(d)) <= 1e-11:
                break
            gd = np.dot(g, d)
            if not np.isfinite(gd) or gd >= 0:
                break
            accepted = False
            for _bt in range(30):
                Ft, _ = smooth_value_and_grad(z, n, tau)
                if Ft is None:
                    break
                if Ft <= Fcur + 1e-4 * gd + 1e-14:
                    h = z.copy()
                    Fcur = Ft
                    accepted = True
                    break
                eta *= 0.5
                z = project_capped(h - eta * g, n)
                if z is None:
                    break
                d = z - h
                gd = np.dot(g, d)
                if not np.isfinite(gd) or gd >= 0:
                    break
            if not accepted:
                break
            eta = min(1.2 * eta, eta_max)
            cs = exact_score(h, n)
            if cs < best_s - 1e-12:
                best_s = cs
                best_h = h.copy()
    return best_h, best_s

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n//2] = 1.0
    if n % 2:
        blk[n//2] = 0.5
    cands.append(blk)
    j = np.arange(n)
    cands.append(np.clip(0.5 + 0.45*np.cos(2*np.pi*(j+0.5)/n), 0, 1))
    cands.append(np.clip(0.5 + 0.45*np.cos(2*np.pi*(j+0.5)/n + np.pi/2), 0, 1))
    cands.append(np.clip(0.5 + 0.35*np.sin(4*np.pi*(j+0.5)/n), 0, 1))
    cands.append(rng.uniform(0, 1, n))
    if n <= 64:
        cands.append(fb + 0.06*rng.normal(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size == n:
        fb = project_capped(parent, n)
    elif parent is not None and parent.size != n:
        old_n = parent.size
        old_pos = (np.arange(old_n) + 0.5) / old_n
        new_pos = (np.arange(n) + 0.5) / n
        fb_raw = np.interp(new_pos, old_pos, parent, left=parent[0], right=parent[-1])
        fb = project_capped(fb_raw, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    if n <= 64:
        n_sc, maxit = 5, 140
    elif n <= 128:
        n_sc, maxit = 4, 90
    elif n <= 256:
        n_sc, maxit = 3, 55
    else:
        n_sc, maxit = 2, 25
    starts = build_starts(fb, n, rng, n_sc)
    for st in starts:
        try:
            bh, bs = continuation_run(st, n, maxit)
            if bs < gs - 1e-12:
                gs, gb = bs, bh.copy()
        except Exception:
            continue
    if n <= 64:
        sds = [0.08, 0.18] if (gs < fs - 1e-5) else [0.03, 0.08]
    elif n <= 256:
        sds = [0.12] if (gs < fs - 1e-5) else [0.05]
    else:
        sds = []
    for sd in sds:
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            bh, bs = continuation_run(pert, n, maxit)
            if bs < gs - 1e-12:
                gs, gb = bs, bh.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    fs2 = exact_score(fb, n)
    if sf > fs2:
        final, sf = fb, fs2
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
