            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - S) > 5e-12 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and
            abs(np.sum(h) - n / 2.0) < 5e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_val_grad(h, n, tau):
    q = lag_values(h, n)
    L = len(q)
    qmax = np.max(q)
    ex = np.exp((q - qmax) / tau)
    w = ex / np.sum(ex)
    val = qmax + tau * np.log(np.sum(ex))
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(L):
        if w[k] < 1e-16:
            continue
        s = k - (n - 1)
        p = max(s, 0)
        qq = max(-s, 0)
        m = n - abs(s)
        g[p:p+m] += dx * w[k] * (1.0 - h[qq:qq+m])
        g[qq:qq+m] -= dx * w[k] * h[p:p+m]
    return val, g

def continuation_run(start, n, taus, max_iter_per_tau):
    h = start.copy().astype(np.float64)
    best_h = h.copy()
    best_s = exact_score(h, n)
    dx = 2.0 / n
    for idx, tau in enumerate(taus):
        eta = 0.25 / dx
        eta_max = 4.0 / dx
        for _ in range(max_iter_per_tau[min(idx, len(max_iter_per_tau)-1)]):
            val, g = smooth_val_grad(h, n, tau)
            if not np.isfinite(val) or not np.all(np.isfinite(g)):
                break
            gnorm = np.max(np.abs(g))
            if gnorm < 1e-13:
                break
            accepted = False
            for _bt in range(25):
                z = project_capped(h - eta * g, n)
                if z is None:
                    break
                d = z - h
                if np.max(np.abs(d)) < 1e-12:
                    break
                gd = np.dot(g, d)
                if not np.isfinite(gd) or gd >= 0:
                    break
                val_bt, _ = smooth_val_grad(z, n, tau)
                if np.isfinite(val_bt) and val_bt <= val + 1e-4 * gd + 1e-14:
                    h = z
                    sz = exact_score(h, n)
                    if np.isfinite(sz) and sz < best_s - 1e-12:
                        best_s = sz
                        best_h = h.copy()
                    accepted = True
                    eta = min(1.15 * eta, eta_max)
                    break
                eta *= 0.5
                if eta < 1e-10 / dx:
                    break
            if not accepted:
                break
    return best_h, best_s

def _so(z):
    return z[-1]

def _soj(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _soi(z):
    h = np.clip(z[:-1], 0.0, 1.0)
    n = len(z) - 1
    return z[-1] - lag_values(h, n)

def _soij(z):
    h = np.clip(z[:-1], 0.0, 1.0)
    n = len(z) - 1
    q = lag_values(h, n)
    L = len(q)
    dx = 2.0 / n
    G = np.zeros((L, n))
    for k in range(L):
        s = k - (n - 1)
        p = max(s, 0); qq = max(-s, 0); m = n - abs(s)
        G[k, p:p+m] -= dx * (1.0 - h[qq:qq+m])
        G[k, qq:qq+m] += dx * h[p:p+m]
    J = np.zeros((L, n + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _soe(z):
    n = len(z) - 1
    return np.sum(z[:-1]) - n / 2.0

def _soej(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:-1] = 1.0
    return J

def slsqp_polish(h0, n, maxiter):
    h0 = h0.copy().astype(np.float64)
    q = lag_values(h0, n)
    t0 = np.max(q) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    con = [{'type': 'ineq', 'fun': _soi, 'jac': _soij},
           {'type': 'eq', 'fun': _soe, 'jac': _soej}]
    try:
        res = minimize(_so, z0, method='SLSQP', jac=_soj, bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    j = np.arange(n)
    cands = [fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2: blk[n//2] = 0.5
    cands.append(blk)
    cands.append(np.clip(0.5 + 0.45*np.cos(2*np.pi*(j+0.5)/n), 0, 1).astype(np.float64))
    cands.append(np.clip(0.5 + 0.45*np.cos(2*np.pi*(j+0.5)/n + 1.5708), 0, 1).astype(np.float64))
    cands.append(rng.uniform(0, 1, n))
    cands.append(fb + rng.normal(0, 0.08, n))
    cands.append(fb + rng.normal(0, 0.15, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True; break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
            n = int(p.size); parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None and parent.size == n:
        fb = project_capped(parent, n)
    elif parent is not None and parent.size >= 2:
        old_n = parent.size
        ox = (np.arange(old_n) + 0.5) / old_n
        nx = (np.arange(n) + 0.5) / n
        fb = project_capped(np.interp(nx, ox, parent, left=parent[0], right=parent[-1]), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n <= 64:
        n_st, mit, n_rst = 5, 120, 3
    elif n <= 128:
        n_st, mit, n_rst = 4, 70, 1
    elif n <= 256:
        n_st, mit, n_rst = 3, 45, 1
    else:
        n_st, mit, n_rst = 2, 25, 0
    taus = [0.04, 0.01, 0.003, 0.0008, 0.0002, 0.00005]
    mil = [mit]*len(taus)
    starts = build_starts(fb, n, rng, n_st)
    for st in starts:
        try:
            bh, bs = continuation_run(st, n, taus, mil)
            if bs < gs - 1e-12:
                gs, gb = bs, bh.copy()
        except Exception:
            continue

    if n <= 128:
        for _ in range(2):
            try:
                pert = project_capped(gb + rng.normal(0, 0.04, n), n)
                if pert is None:
                    continue
                bh, bs = continuation_run(pert, n, taus, mil)
                if bs < gs - 1e-12:
                    gs, gb = bs, bh.copy()
            except Exception:
                continue

    if n <= 128:
        try:
            h_pol = slsqp_polish(gb, n, 300)
            if h_pol is not None:
                h_pol = project_capped(h_pol, n)
                if h_pol is not None:
                    sp = exact_score(h_pol, n)
                    if np.isfinite(sp) and sp < gs - 1e-12:
                        gs, gb = sp, h_pol.copy()
        except Exception:
            pass
        for _ in range(2):
            try:
                h_pert = slsqp_polish(gb + rng.normal(0, 0.02, n), n, 200)
                if h_pert is not None:
                    h_pert = project_capped(h_pert, n)
                    if h_pert is not None:
                        sp = exact_score(h_pert, n)
                        if np.isfinite(sp) and sp < gs - 1e-12:
                            gs, gb = sp, h_pert.copy()
            except Exception:
                continue

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
