from scipy.optimize import minimize
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i])
            h[i] += c
            r -= c
        else:
            c = min(-r, h[i])
            h[i] -= c
            r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - target) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def pair_bounds(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    return lo, hi

def make_proposal(h, n, rng, step):
    if rng.random() < 0.8:
        for _ in range(4):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            lo, hi = pair_bounds(h, i, j)
            if hi - lo < 1e-12:
                continue
            if rng.random() < 0.75:
                lo2 = max(lo, -step)
                hi2 = min(hi, step)
                if hi2 - lo2 < 1e-12:
                    continue
                t = rng.uniform(lo2, hi2)
            else:
                t = lo if rng.random() < 0.5 else hi
                if abs(t) < 1e-12:
                    t = hi if abs(lo) < 1e-12 else lo
            if abs(t) < 1e-12:
                continue
            z = h.copy()
            z[i] += t
            z[j] -= t
            p = project_capped(z, n)
            if p is not None:
                return p
    z = h + step * rng.standard_normal(n)
    return project_capped(z, n)

def anneal_chain(start, n, rng, steps, best_h, best_s):
    current = start.copy()
    cur_s = exact_score(current, n)
    step_size = 0.20
    T0 = 0.002
    Tmin = 1e-6
    epoch_len = max(1, (steps + 2) // 3)
    accept_cnt = 0
    try_cnt = 0
    cur_epoch = 0
    for t in range(steps):
        new_epoch = t // epoch_len
        if new_epoch != cur_epoch:
            cur_epoch = new_epoch
            sigma = 0.08 + 0.12 * rng.random()
            rc = project_capped(best_h + sigma * rng.standard_normal(n), n)
            if rc is not None and is_feasible(rc, n):
                current = rc
                cur_s = exact_score(current, n)
            step_size = 0.20
            accept_cnt = 0
            try_cnt = 0
        phase = t % epoch_len
        T = T0 * (Tmin / T0) ** (phase / max(1, epoch_len - 1))
        prop = make_proposal(current, n, rng, step_size)
        if prop is None or not is_feasible(prop, n):
            continue
        s_new = exact_score(prop, n)
        try_cnt += 1
        if s_new <= cur_s:
            accept = True
        else:
            delta = s_new - cur_s
            exp_arg = max(-745.0, -delta / max(T, 1e-15))
            accept = rng.random() < math.exp(exp_arg)
        if accept:
            current = prop
            cur_s = s_new
            accept_cnt += 1
        if s_new < best_s - 1e-12:
            best_h = prop.copy()
            best_s = s_new
        if try_cnt > 0 and try_cnt % 100 == 0:
            ratio = accept_cnt / max(1, try_cnt)
            if ratio > 0.45:
                step_size = min(0.6, step_size * 1.20)
            elif ratio < 0.15:
                step_size = max(1e-4, step_size * 0.70)
            accept_cnt = 0
            try_cnt = 0
    return current, best_h, best_s

def add_unique_start(pool, candidate, n, limit):
    if len(pool) >= limit:
        return
    p = project_capped(candidate, n)
    if p is None or not is_feasible(p, n):
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p.copy())
