    B = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        val = 0.0

            val -= h[a+r]
        if b <= j < b+m:

            val += h[a+r]

    return B

def quad_roots(a, b, c):
    """Solve a*t^2 + b*t + c = 0, return list of real roots"""
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return []
        return [-c/b]
    disc = b*b - 4.0*a*c
    if disc < -1e-12*(b*b + 4.0*abs(a*c) + 1e-300):
        return []
    if disc < 0:
        disc = 0.0
    sq = np.sqrt(disc)
    if abs(b) < 1e-15:
        return [-sq/(2.0*a), sq/(2.0*a)]
    q = -0.5*(b + np.copysign(sq, b))
    r1 = q/a
    r2 = c/q if abs(q) > 1e-300 else -b/a
    # Deduplicate
    if abs(r1 - r2) < 1e-12*max(1.0, abs(r1), abs(r2)):
        return [r1]
    return [r1, r2]
