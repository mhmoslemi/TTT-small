    return float(np.max(c))

def add_cand(pool, prop, label, n):
    p = prop.copy() if is_feasible(prop, n) else project_capped(prop, n)
    if p is None:
        return
    s = exact_score(p, n)
    if not np.isfinite(s):
        return
    for e in pool:
        if np.max(np.abs(p - e[0])) < 1e-8:
            return
    pool.append((p, s, label))

def build_pool(fb, n, rng, parent):
    pool = []
    add_cand(pool, np.full(n, 0.5), 'uniform', n)
    if parent is not None:
        add_cand(pool, parent, 'parent', n)
        add_cand(pool, parent[::-1].copy(), 'sym', n)
        add_cand(pool, 1.0 - parent[::-1].copy(), 'sym', n)
        add_cand(pool, 1.0 - parent.copy(), 'sym', n)
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    add_cand(pool, blk, 'block', n)
    add_cand(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), 'alt', n)
    perm = rng.permutation(n); bb = np.zeros(n); bb[perm[:n // 2]] = 1.0
    if n % 2: bb[perm[n // 2]] = 0.5
    add_cand(pool, bb, 'binary', n)
    for m in range(1, 4):
        for ph in [0.0, math.pi / 2]:
            add_cand(pool, 0.5 + 0.45 * np.cos(2.0 * math.pi * m * np.arange(n) / n + ph), 'wave', n)
    ctr = parent if parent is not None else np.full(n, 0.5)
    for sd in [0.025, 0.06, 0.14, 0.30]:
        add_cand(pool, ctr + rng.normal(0, sd, n), 'noise', n)
    for sd in [0.08, 0.20, 0.35]:
        add_cand(pool, np.full(n, 0.5) + rng.normal(0, sd, n), 'noise', n)
    add_cand(pool, rng.uniform(0, 1, n), 'rand', n)
    for sd in [0.10, 0.18, 0.28]:
        add_cand(pool, ctr + rng.normal(0, sd, n), 'noise', n)
    return pool

def select_pop(pool, n, P, rng):
    if not pool:
        return np.full((P, n), 0.5), np.array([exact_score(np.full(n, 0.5), n)] * P)
    idx = list(range(len(pool)))
    idx.sort(key=lambda i: (pool[i][1], i))
    sel = []
    for i in idx:
        if pool[i][2] in ('uniform', 'parent') and len(sel) < P:
            sel.append(i)
    for i in idx:
        if len(sel) >= P: break
        if i not in sel and pool[i][2] not in ('uniform', 'parent'):
            sel.append(i)
    while len(sel) < P and len(pool) > 0:
        best_j, best_d = -1, -1.0
        for j in range(len(pool)):
            if j in sel: continue
            md = min(np.sqrt(np.mean((pool[j][0] - pool[s][0]) ** 2)) for s in sel) if sel else 0.0
            if md > best_d: best_d, best_j = md, j
        sel.append(best_j if best_j >= 0 else sel[rng.integers(len(sel))])
    pop = np.zeros((P, n)); sc = np.zeros(P)
    for i, j in enumerate(sel): pop[i], sc[i] = pool[j][0], pool[j][1]
    return pop, sc

def de_search(pop, scores, n, max_gen, stall_lim, max_rst, pool, rng):
    P = len(pop); mu_F, mu_CR, c_r = 0.55, 0.85, 0.1
    stall, n_rst = 0, 0; archive = []
    bs = float(np.min(scores)); bv = pop[int(np.argmin(scores))].copy()
    for v, s, _ in pool:
        if s < bs - 1e-12: bs, bv = s, v.copy()
    fp = int(math.ceil(0.25 * max_gen)); pc = max(2, int(math.ceil(0.2 * P)))
    for gen in range(max_gen):
        op, os_ = pop.copy(), scores.copy()
        gsb = bs; fs, crs = [], []
        npop, nsc = pop.copy(), scores.copy(); nar = list(archive)
        oord = np.argsort(os_)[:pc]
        for i in range(P):
            Fi = None
            for _ in range(10):
                Fi = mu_F + 0.1 * rng.standard_cauchy()
                if Fi > 0 and np.isfinite(Fi): break
            if Fi is None or Fi <= 0: Fi = mu_F
            Fi = float(np.clip(Fi, 0.05, 1.0))
            Cri = float(np.clip(rng.normal(mu_CR, 0.1), 0.0, 1.0))
            if gen < fp:
                ot = np.array([j for j in range(P) if j != i])
                r1, r2, r3 = int(ot[0]), int(ot[1]), int(ot[2])
                if len(ot) >= 3: r1, r2, r3 = int(rng.choice(ot, 3, replace=False))
                v = op[r1] + Fi * (op[r2] - op[r3])
            else:
                pbc = [int(j) for j in oord if j != i]
                pb = int(pbc[rng.integers(len(pbc))]) if pbc else int(oord[0])
                ot = np.array([j for j in range(P) if j != i])
                r1 = int(ot[rng.integers(len(ot))])
                if nar and rng.random() < 0.4:
                    r2v = nar[rng.integers(len(nar))][0]
                else:
                    o2 = np.array([j for j in range(P) if j != i and j != r1])
                    r2v = op[int(o2[rng.integers(len(o2))])]
                v = op[i] + Fi * (op[pb] - op[i]) + Fi * (op[r1] - r2v)
            if not np.all(np.isfinite(v)): continue
            tr = op[i].copy(); mk = rng.random(n) < Cri
            tr[mk] = v[mk]; tr[int(rng.integers(n))] = v[int(rng.integers(n))]
            pj = project_capped(tr, n)
            if pj is None or not is_feasible(pj, n): continue
            st_ = exact_score(pj, n)
            if not np.isfinite(st_): continue
            if st_ < bs - 1e-12: bs, bv = st_, pj.copy()
            if st_ < os_[i] - 1e-12:
                npop[i], nsc[i] = pj, st_
                nar.append((op[i].copy(), os_[i]))
                fs.append(Fi); crs.append(Cri)
        pop, scores = npop, nsc
        if len(nar) > P:
            rm = np.sort(rng.choice(len(nar), len(nar) - P, replace=False))[::-1]
            for ri in rm: nar.pop(int(ri))
        archive = nar
        if fs:
            mf = np.mean([f * f for f in fs]) / max(1e-12, np.mean(fs))
            mu_F = float(np.clip((1 - c_r) * mu_F + c_r * mf, 0.05, 1.0))
            mu_CR = float(np.clip((1 - c_r) * mu_CR + c_r * np.mean(crs), 0.0, 1.0))
        stall = 0 if bs < gsb - 1e-12 else stall + 1
        if stall >= stall_lim:
            if n_rst >= max_rst: break
            n_rst += 1; stall = 0
            kn = int(math.ceil(P / 2)); ko = np.argsort(scores)[:kn]
            in_keep = any(np.max(np.abs(pop[ko[k]] - bv)) < 1e-10 for k in range(kn))
            rtn_p, rtn_s = np.zeros((P, n)), np.zeros(P)
            for k in range(kn):
                if not in_keep and k == kn - 1:
                    rtn_p[k], rtn_s[k] = bv, bs
                else:
                    rtn_p[k], rtn_s[k] = pop[ko[k]], scores[ko[k]]
            for k in range(kn, P):
                done = False
                for at in range(6):
                    sc_ = min([0.04, 0.12, 0.25][at % 3] * (1.5 ** (at // 3)), 0.5)
                    pj = project_capped(bv + rng.normal(0, sc_, n), n)
                    if pj is None or not is_feasible(pj, n): continue
                    if any(np.max(np.abs(pj - rtn_p[pk])) < 1e-8 for pk in range(kn + k)): continue
                    rtn_p[k], rtn_s[k] = pj, exact_score(pj, n); done = True; break
                if not done:
                    pi = int(rng.integers(len(pool))) if pool else 0
                    rtn_p[k] = pool[pi][0] if pool else bv
                    rtn_s[k] = pool[pi][1] if pool else bs
            pop, scores, archive = rtn_p, rtn_s, []
            mu_F, mu_CR = 0.55, 0.85
    return bv, bs

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 < p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
                if abs(parent.sum() - n / 2.0) > 1e-6:
                    parent = project_capped(parent, n)
    except Exception:
        pass
    if n is None: n = 47
    fb = parent.copy() if parent is not None else np.full(n, 0.5)
    if fb is None: fb = np.full(n, 0.5)
    if n == 1:
        return fb, float(exact_score(fb, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64: P, mg, sl, mr = 64, 100, 18, 2
    elif n <= 128: P, mg, sl, mr = 48, 70, 14, 2
    elif n <= 256: P, mg, sl, mr = 32, 40, 10, 1
    elif n <= 512: P, mg, sl, mr = 20, 20, 7, 1
    else: P, mg, sl, mr = 12, 12, 5, 1
    pool = build_pool(fb, n, rng, parent)
    pop, sc = select_pop(pool, n, P, rng)
    bs = exact_score(fb, n); bv = fb.copy()
    for v, s, _ in pool:
        if s < bs - 1e-12: bs, bv = s, v.copy()
    for i in range(P):
        if sc[i] < bs - 1e-12: bs, bv = sc[i], pop[i].copy()
    try:
        dv, ds = de_search(pop, sc, n, mg, sl, mr, pool, rng)
        if ds < bs - 1e-12: bs, bv = ds, dv
    except Exception:
        pass
    final = project_capped(bv, n)
    if final is None or not is_feasible(final, n): final = project_capped(fb, n)
    if final is None: final = np.full(n, 0.5)
    fs = exact_score(final, n)
    if not np.isfinite(fs): final = np.full(n, 0.5); fs = exact_score(final, n)
    pf = project_capped(fb, n)
    if pf is not None and is_feasible(pf, n):
        ps = exact_score(pf, n)
        if ps < fs - 1e-12: final, fs = pf, ps
    return final.astype(np.float64), float(fs), int(n)
