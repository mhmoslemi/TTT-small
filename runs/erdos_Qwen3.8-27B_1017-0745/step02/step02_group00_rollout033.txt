            best, best_sc = st.copy(), ss
    if n < 2 or budget < 10:
        fin = project_capped(best, n)
        if fin is None:
            fin = uniform.copy()
        return fin, float(exact_score(fin, n)), n
    # Annealing
    remaining = budget
    delta_cur = delta
    restart_noise = 0.04
    for ci in range(nc):
        if remaining < 10:
            break
        if ci < len(starts):
            x = starts[ci].copy()
        else:
            x = project_capped(best + rng.normal(0, restart_noise, n), n)
            restart_noise = min(restart_noise * 1.5, 0.20)
            if x is None:
                x = uniform.copy()
        sx = exact_score(x, n)
        if not np.isfinite(sx):
            x = uniform.copy(); sx = s_uni
        if np.isfinite(sx) and sx < best_sc - 1e-12:
            best, best_sc = x.copy(), sx
        # Pilot for T0
        ups = []
        for _ in range(16):
            p = propose_exchange(x, n, rng, delta_cur)
            if p is not None:
                sp = exact_score(p, n)
                if np.isfinite(sp) and sp > sx + 1e-12:
                    ups.append(sp - sx)
        if ups:
            T0 = max(1e-5, min(float(np.median(ups)) / 0.2877, 0.02))
        else:
            T0 = 0.002
        Tmin = max(1e-7, 0.001 * T0)
        for ep in range(ne):
            if remaining < 1:
                break
            frac = float(ep) / max(1, ne - 1)
            T = T0 * (Tmin / T0) ** frac
            att = min(mp, remaining)
            acc = 0
            for _ in range(att):
                p = propose_exchange(x, n, rng, delta_cur)
                if p is None:
                    continue
                sp = exact_score(p, n)
                if not np.isfinite(sp):
                    continue
                if sp < best_sc - 1e-12:
                    best, best_sc = p.copy(), sp
                dv = sp - sx
                if dv <= 1e-14:
                    x, sx = p, sp
                    acc += 1
                else:
                    if dv / T > 700.0:
                        continue
                    if math.log(1.0 - rng.random() + 1e-15) < -dv / T:
                        x, sx = p, sp
                        acc += 1
            remaining -= att
            if att > 0:
                ar = acc / att
                if ar > 0.40:
                    delta_cur = min(delta_cur * 1.25, 0.35)
                elif ar < 0.15:
                    delta_cur = max(delta_cur * 0.75, 0.002)
    # Final polish
    if n <= 64:
        pol = pairwise_polish(best, n, 2)
        pol = project_capped(pol, n)
        if pol is not None:
            ps = exact_score(pol, n)
            if np.isfinite(ps) and ps < best_sc - 1e-12:
                best, best_sc = pol, ps
    # Final validation
    finalists = [best, uniform]
    if parent is not None:
        finalists.append(parent)
    final = uniform.copy(); final_sc = s_uni
    for fc in finalists:
        pr = project_capped(fc, n)
        if pr is None:
            pr = fc
        fs = exact_score(pr, n)
        if np.isfinite(fs) and fs < final_sc:
            final, final_sc = pr.copy(), fs
    final_sc = exact_score(final, n)
    if not np.isfinite(final_sc):
        final = uniform.copy(); final_sc = s_uni
    return final, float(final_sc), n
