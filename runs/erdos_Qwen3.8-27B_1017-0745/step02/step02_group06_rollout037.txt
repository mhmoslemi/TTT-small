    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(s - np.sum(h)) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):

        return False

            G[row, i] -= dx * h[i+k]

    ii, jj = np.triu_indices(n, 1)

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(s - np.sum(h)) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)


def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))


def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full').astype(np.float64) * dx
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        a, b = max(0, -k), min(n, n - k)
        for i in range(a, b):
            G[row, i + k] += dx * (1.0 - h[i])
            G[row, i] -= dx * h[i + k]
    if not (np.all(np.isfinite(L)) and np.all(np.isfinite(G))):
        return None, None
    return L, G


def solve_mm_qp(x, L, G, n, solvers):
    if not HAS_CP:
        return None
    rho = 2.0 / n
    yv = cp.Variable(n, value=x)
    u = cp.Variable(value=float(np.max(L)))
    obj = cp.Minimize(u + rho * cp.sum_squares(yv - x))
    cons = [yv >= 0, yv <= 1, cp.sum(yv) == n / 2.0]
    cons.append(u >= L + G @ (yv - x))
    prob = cp.Problem(obj, cons)
    for sol in solvers:
        try:
            prob.solve(solver=sol, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                rv = yv.value
                if rv is not None and rv.shape == (n,) and np.all(np.isfinite(rv)):
                    return rv.astype(np.float64)
        except Exception:
            continue
    return None


def mm_trajectory(x, n, max_it, solvers):
    x = x.copy().astype(np.float64)
    score = exact_score(x, n)
    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625]
    for _ in range(max_it):
        L, G = lag_data(x, n)
        if L is None:

        if yq is None:

        yq = project_capped(yq, n)
        if yq is None or not is_feasible(yq, n):

        x, score = best_t, best_s
