import numpy as np
import math
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,):
        return None
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(200):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if np.any(np.isnan(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return v if np.isfinite(v) else float('inf')
    except Exception:
        return float('inf')

def lag_values(h, n, need_grad=False):
    dx = 2.0 / n
    m = 2 * n - 1
    Q = np.empty(m)
    G = np.empty((m, n)) if need_grad else None
    Q[0] = np.dot(h, 1.0 - h)
    if need_grad:
        G[0] = 1.0 - 2.0 * h
    for r in range(1, n):
        mm = n - r
        a = h[:mm] * (1.0 - h[r:])
        Q[r] = np.sum(a)
        qb = h[r:] * (1.0 - h[:mm])
        Q[r + mm] = np.sum(qb)
        if need_grad:
            g = np.zeros(n)
            g[:mm] += (1.0 - h[r:])
            g[r:] -= h[:mm]
            G[r] = g
            g = np.zeros(n)
            g[:mm] -= h[r:]
            g[r:] += (1.0 - h[:mm])
            G[r + mm] = g
    Q *= dx
    if need_grad:
        G *= dx
    return Q, G if need_grad else None

def slsqp_obj(z, n):
    return z[-1]

def slsqp_obj_grad(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def slsqp_ineq(z, n):
    h = z[:-1]
    Q, _ = lag_values(h, n)
    return z[-1] - Q

def slsqp_ineq_jac(z, n):
    h = z[:-1]
    _, G = lag_values(h, n, True)
    J = np.empty((G.shape[0], n + 1))
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

def slsqp_eq(z, n):
    return np.sum(z[:-1]) - n / 2.0

def slsqp_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:-1] = 1.0
    return j

def slsqp_solve(h0, n):
    t0 = exact_score(h0, n) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},
    ]
    mi = 250 if n <= 64 else 160
    try:
        res = minimize(slsqp_obj, z0, jac=slsqp_obj_grad, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'ftol': 1e-12, 'maxiter': mi, 'disp': False})
        return res.x if res.x is not None else None
    except Exception:
        return None

def smooth_val(h, n, tau):
    Q, _ = lag_values(h, n)
    qm = np.max(Q)
    e = np.exp((Q - qm) / tau)
    return qm + tau * math.log(np.sum(e)), e

def smooth_grad(h, n, tau):
    Q, _ = lag_values(h, n)
    qm = np.max(Q)
    e = np.exp((Q - qm) / tau)
    p = e / np.sum(e)
    dx = 2.0 / n
    g = np.zeros(n)
    g += p[0] * dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        mm = n - r
        g[:mm] += dx * p[r] * (1.0 - h[r:])
        g[r:] -= dx * p[r] * h[:mm]
        g[:mm] -= dx * p[r + mm] * h[r:]
        g[r:] += dx * p[r + mm] * (1.0 - h[:mm])
    return g

def smooth_refine(h, n, iters_per_tau):
    best_h, best_s = h.copy(), exact_score(h, n)
    for tau in [0.02, 0.005, 0.001, 0.0002]:
        h_cur = best_h.copy()
        F, _ = smooth_val(h_cur, n, tau)
        alpha = 1.0
        for _ in range(iters_per_tau):
            g = smooth_grad(h_cur, n, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_capped(h_cur - alpha * g, n)
            if y is None:
                break
            d = y - h_cur
            if np.max(np.abs(d)) <= 1e-9:
                break
            accepted = False
            a = alpha
            for _b in range(25):
                yb = project_capped(h_cur - a * g, n)
                if yb is None:
                    break
                Fb, _ = smooth_val(yb, n, tau)
                if np.isfinite(Fb) and Fb <= F + 1e-4 * np.dot(g, yb - h_cur) + 1e-14:
                    h_cur = yb
                    F = Fb
                    accepted = True
                    break
                a *= 0.5
            if not accepted:
                break
            alpha = min(1.5 * alpha, 10.0)
            scur = exact_score(h_cur, n)
            if scur < best_s - 1e-13:
                best_h, best_s = h_cur.copy(), scur
    return best_h, best_s

def build_starts(fb, n, rng):
    starts = []
    def try_add(v):
        for s in starts:
            if np.max(np.abs(v - s)) < 1e-9:
                return
        starts.append(v.copy())
    try_add(fb)
    try_add(project_capped(fb[::-1], n) or fb)
    try_add(project_capped(1.0 - fb, n) or fb)
    try_add(project_capped(1.0 - fb[::-1], n) or fb)
    try_add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(project_capped(blk, n) or fb)
    for k in range(1, 4):
        for ph in [0.0, math.pi / 2]:
            v = 0.5 + 0.4 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph)
            try_add(project_capped(v, n) or fb)
    for sd in [0.08, 0.22]:
        try_add(project_capped(fb + rng.normal(0, sd, n), n) or fb)
    try_add(project_capped(np.roll(fb, n // 3), n) or fb)
    try_add(project_capped(np.roll(fb, n // 2), n) or fb)
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(project_capped(bv, n) or fb)
    limit = 10 if n <= 64 else (7 if n <= 128 else 4)
    return starts[:limit]

    parent = None

    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except Exception:
            n = 47
    try:

    except Exception:

    if n == 1:
        return np.array([0.5]), 0.5, 1
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):

    fb_s = exact_score(fb, n)
    best_h, best_s = fb.copy(), fb_s
    starts = build_starts(fb, n, rng)

        s0 = exact_score(h0, n)
        if s0 < best_s - 1e-12:

        hr = slsqp_solve(h0, n)
