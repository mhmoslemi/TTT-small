```
def smooth_polish(h,n,tau_seq,its):
    h=project_capped(h,n)
    if h is None: return None, inf
    best=h.copy(); bs=exact_score(h,n)
    for tau in tau_seq:
       cur=h.copy(); alpha=0.5
       for _ in range(its):
          F,g=smooth_val_grad(cur,n,tau)
          if not finite: break
          ng=np.linalg.norm(g)
          if ng<1e-12: break
          prev=cur.copy(); ok=False
          for b in range(16):
             a=alpha/(ng+1e-12)
             cand=project_capped(cur-a*g,n)
             if cand is None: alpha*=0.5; continue
             Fc,_=smooth_val_grad(cand,n,tau)
             if not finite: alpha*=0.5; continue
             if Fc <= F + 1e-4*np.dot(g,cand-cur) + 1e-14:
                cur=cand; ok=True; alpha=min(6,1.6*alpha)
                sc=exact_score(cur,n)
                if finite and sc<bs-1e-12: bs=sc; best=cur.copy()
                break
             alpha*=0.5
          if not ok or np.max(np.abs(cur-prev))<1e-9: break
       sc=exact_score(cur,n); if finite and sc<bs-1e-12: best=cur.copy(); bs=sc
       h=cur
    return best,bs
```

```
def lag_values(h,n,need_jac=False):
    h=np.asarray(h,dtype=np.float64).ravel()
    dx=2.0/n; size=2*n-1
    Q=np.empty(size)
    G=np.empty((size,n)) if need_jac else None
    Q[0]=dx*float(np.dot(h,1.0-h))
    if need_jac: G[0]=dx*(1.0-2.0*h)
    idx=1
    for r in range(1,n):
        m=n-r; a=h[:m]; b=h[r:]
        Q[idx]=dx*float(np.dot(a,1.0-b))
        Q[idx+1]=dx*float(np.dot(b,1.0-a))
        if need_jac:
            gp=np.zeros(n); gp[:m]+=1.0-b; gp[r:]-=a; G[idx]=dx*gp
            gn=np.zeros(n); gn[:m]-=b; gn[r:]+=1.0-a; G[idx+1]=dx*gn
        idx+=2
    return (Q,G) if need_jac else Q
```

```
def smooth_val_grad(h,n,tau):
    h=np.asarray(h,dtype=np.float64).ravel()
    Q=lag_values(h,n,False)
    if not np.all(np.isfinite(Q)): return float('inf'),None
    qmax=float(np.max(Q)); sh=(Q-qmax)/tau
    ex=np.exp(sh); s=ex.sum()
    if not np.isfinite(s) or s<=0: return float('inf'),None
    p=ex/s; F=qmax+tau*np.log(s)
    g=np.zeros(n); dx=2.0/n
    g+=dx*p[0]*(1.0-2.0*h)
    idx=1
    for r in range(1,n):
        m=n-r; wp=p[idx]; wn=p[idx+1]
        g[:m]+=dx*wp*(1.0-h[r:]); g[r:]-=dx*wp*h[:m]
        g[:m]-=dx*wn*h[r:]; g[r:]+=dx*wn*(1.0-h[:m])
        idx+=2
    if not np.all(np.isfinite(g)): return float('inf'),None
    return F,g
```

```
def solve_epi(h0,n,mi):

    bounds=[(0.0,1.0)]*n+[(0.0,1.0)]
```

```python
import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(n + 3):
        if abs(r) < 1e-13:
            break
        if r > 0:
            i = int(np.argmax(1.0 - h))
            d = min(r, 1.0 - h[i])
            h[i] += d
            r -= d
        else:
            i = int(np.argmax(h))
            d = min(-r, h[i])
            h[i] -= d
            r += d
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(h.sum() - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return False
    return (not np.any(h < -1e-12)) and (not np.any(h > 1.0 + 1e-12)) and abs(h.sum() - 0.5 * n) <= 1e-9

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if not is_feasible(h, n):
        return float('inf')
    try:
        v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        m = float(np.max(v))
        return m if np.isfinite(m) else float('inf')
    except Exception:
        return float('inf')

def lag_values(h, n, need_jac=False):
    h = np.asarray(h, dtype=np.float64).ravel()
    dx = 2.0 / n
    size = 2 * n - 1
    Q = np.empty(size, dtype=np.float64)
    G = np.empty((size, n), dtype=np.float64) if need_jac else None
    Q[0] = dx * float(np.dot(h, 1.0 - h))
    gp = gn = None
    if need_jac:
        gp = np.zeros(n, dtype=np.float64)
        gn = np.zeros(n, dtype=np.float64)
        G[0, :] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        a = h[:m]
        b = h[r:]
        Q[idx] = dx * float(np.dot(a, 1.0 - b))
        Q[idx + 1] = dx * float(np.dot(b, 1.0 - a))
        if need_jac:
            gp.fill(0.0)
            gp[:m] += 1.0 - b
            gp[r:] -= a
            G[idx, :] = dx * gp
            gn.fill(0.0)
            gn[:m] -= b
            gn[r:] += 1.0 - a
            G[idx + 1, :] = dx * gn
        idx += 2
    return (Q, G) if need_jac else Q

def smooth_val_grad(h, n, tau):
    h = np.asarray(h, dtype=np.float64).ravel()
    Q = lag_values(h, n, False)
    if not np.all(np.isfinite(Q)):
        return float('inf'), None
    qmax = float(np.max(Q))
    ex = np.exp((Q - qmax) / tau)
    s = ex.sum()
    if not np.isfinite(s) or s <= 0.0:
        return float('inf'), None
    p = ex / s
    F = qmax + tau * float(np.log(s))
    g = np.zeros(n, dtype=np.float64)
    dx = 2.0 / n
    g += dx * p[0] * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        wp = p[idx]
        wn = p[idx + 1]
        g[:m] += dx * wp * (1.0 - h[r:])
        g[r:] -= dx * wp * h[:m]
        g[:m] -= dx * wn * h[r:]
        g[r:] += dx * wn * (1.0 - h[:m])
        idx += 2
    if not np.all(np.isfinite(g)):
        return float('inf'), None
    return float(F), g

def epi_obj(z):
    return float(z[-1])

def epi_obj_jac(z):
    J = np.zeros(len(z), dtype=np.float64)
    J[-1] = 1.0
    return J

def epi_ineq(z):
    n = int(len(z) - 1)
    Q = lag_values(z[:-1], n, False)
    return z[-1] - Q

def epi_ineq_jac(z):
    n = int(len(z) - 1)
    Q, G = lag_values(z[:-1], n, True)
    J = np.empty((Q.size, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

def epi_eq(z):
    n = int(len(z) - 1)
    return float(np.sum(z[:-1]) - 0.5 * n)

def epi_eq_jac(z):
    n = int(len(z) - 1)
    J = np.zeros(n + 1, dtype=np.float64)
    J[:-1] = 1.0
    return J

def solve_epi(h0, n, mi):
    h0 = project_capped(h0, n)
    if h0 is None:
        return None
    s = exact_score(h0, n)
    if not np.isfinite(s):
        return None
    t0 = min(0.999999, s + 1e-4)
    z0 = np.concatenate((h0, np.array([t0], dtype=np.float64)))
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac},
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac}
    ]
    try:
        res = minimize(epi_obj, z0, bounds=bounds, method='SLSQP', jac=epi_obj_jac,
                       constraints=cons, options={'maxiter': int(max(20, mi)), 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    x = np.asarray(res.x, dtype=np.float64).ravel()
    if x.size != n + 1 or not np.all(np.isfinite(x)):
        return None
    return project_capped(x[:n], n)

def consider(n, best_h, best_s, cand):
    if cand is None:
        return best_h, best_s
    c = project_capped(cand, n)
    if c is None:
        return best_h, best_s
    s = exact_score(c, n)
    if np.isfinite(s) and s < best_s - 1e-12:
        return c, s
    return best_h, best_s

def try_add(pool, v, n, limit):
    if len(pool) >= limit or v is None:
        return pool
    p = project_capped(v, n)
    if p is None:
        return pool
    for q in pool:
        if np.max(np.abs(p - q)) < 1e-8:
            return pool
    pool.append(p)
    return pool

def snap_binary(h, n):
    o = np.argsort(h)[::-1]
    v = np.zeros(n, dtype=np.float64)
    v[o[:n // 2]] = 1.0
    if n % 2:
        v[o[n // 2]] = 0.5
    return v

def build_starts(base, n, rng, limit):
    pool = []
    pool = try_add(pool, base, n, limit)
    for sd in (0.02, 0.08, 0.20):
        pool = try_add(pool, base + rng.normal(0.0, sd, n), n, limit)
    if len(pool) < limit:
        pool = try_add(pool, snap_binary(base, n), n, limit)
    if len(pool) < limit:
        pool = try_add(pool, snap_binary(1.0 - base, n), n, limit)
    for v in (base[::-1], 1.0 - base, (1.0 - base)[::-1], np.roll(base, n // 3)):
        if len(pool) >= limit:
            break
        pool = try_add(pool, v, n, limit)
    if len(pool) < limit:
        perm = rng.permutation(n)
        vb = np.zeros(n, dtype=np.float64)
        vb[perm[:n // 2]] = 1.0
        if n % 2:
            vb[perm[n // 2]] = 0.5
        pool = try_add(pool, vb, n, limit)
    if len(pool) < limit:
        i = np.arange(n)
        for k, amp in ((1, 0.5), (2, 0.5), (1, 0.25), (3, 0.5)):
            if len(pool) >= limit:
                break
            for ph in (0.0, 0.5 * np.pi):
                pool = try_add(pool, 0.5 + amp * np.cos(2.0 * np.pi * k * i / n + ph), n, limit)
                if len(pool) >= limit:
                    break
    if len(pool) < limit:
        i = np.arange(n)
        pool = try_add(pool, 0.5 + 0.5 * np.cos(2.0 * np.pi * ((i * i) % n) / n), n, limit)
    while len(pool) < limit:
        pool.append(pool[0].copy())
    return pool

def smooth_polish(h, n, tseq, its):
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    best_h = h.copy()
    best_s = exact_score(h, n)
    for tau in tseq:
        cur = h.copy()
        alpha = 0.6
        for _ in range(int(its)):
            F, g = smooth_val_grad(cur, n, tau)
            if not np.isfinite(F):
                break
            ng = float(np.linalg.norm(g))
            if ng < 1e-12:
                break
            prev = cur.copy()
            ok = False
            for _b in range(16):
                cand = project_capped(cur - (alpha / (ng + 1e-12)) * g, n)
                if cand is None:
                    alpha *= 0.5
                    continue
                Fc, _ = smooth_val_grad(cand, n, tau)
                if np.isfinite(Fc) and Fc <= F + 1e-4 * float(np.dot(g, cand - cur)) + 1e-14:
                    cur = cand
                    ok = True
                    alpha = min(6.0, 1.5 * alpha)
                    sc = exact_score(cur, n)
                    if np.isfinite(sc) and sc < best_s - 1e-12:
                        best_s = sc
                        best_h = cur.copy()
                    break
                alpha *= 0.5
            if not ok or np.max(np.abs(cur - prev)) < 1e-9:
                break
        sc = exact_score(cur, n)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = sc
            best_h = cur.copy()
        h = cur
    return best_h, best_s

def refine_point(h0, n, epi_iters):
    h0 = project_capped(h0, n)
    if h0 is None:
        return None, float('inf')
    best_h = h0.copy()
    best_s = exact_score(h0, n)
    its = 12 if n <= 128 else 7
    ph, ps = smooth_polish(h0, n, (0.02, 0.005), its)
    best_h, best_s = consider(n, best_h, best_s, ph)
    base = ph if (ph is not None and ps < best_s) else h0
    if epi_iters > 0:
        c = solve_epi(base, n, epi_iters)
        if c is not None:
            best_h, best_s = consider(n, best_h, best_s, c)
            for a in (0.2, 0.5, 0.8):
                best_h, best_s = consider(n, best_h, best_s, base + a * (c - base))
        ph, ps = smooth_polish(best_h, n, (0.004, 0.001), max(5, its // 2))
        best_h, best_s = consider(n, best_h, best_s, ph)
    ph, ps = smooth_polish(best_h, n, (0.002, 0.0006), its)
    best_h, best_s = consider(n, best_h, best_s, ph)
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1

    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = np.full(n, 0.5, dtype=np.float64)

    best_h = fb.copy()
    best_s = exact_score(fb, n)
    if not np.isfinite(best_s):
        best_s = 0.5

    if n <= 64:
        limit, mi_main, mi_side = 10, 160, 90
    elif n <= 128:
        limit, mi_main, mi_side = 8, 120, 70
    elif n <= 256:
        limit, mi_main, mi_side = 6, 80, 40
    else:
        limit, mi_main, mi_side = 4, 0, 0

    starts = build_starts(fb, n, rng, limit)
    scores = []
    for p in starts:
        s = exact_score(p, n)
        scores.append(s)
        if np.isfinite(s) and s < best_s - 1e-12:
            best_s = s
            best_h = p.copy()

    order = np.argsort(scores)
    refined = [0]
    rh, rs = refine_point(starts[0], n, mi_main)
    best_h, best_s = consider(n, best_h, best_s, rh)

    cnt = 0
    for idx in order:
        idx = int(idx)
        if idx in refined or cnt >= 2:
            break
        rh, rs = refine_point(starts[idx], n, mi_side)
        best_h, best_s = consider(n, best_h, best_s, rh)
        refined.append(idx)
        cnt += 1

    if np.isfinite(best_s) and best_s > 0.3808 and n <= 128:
        p0 = project_capped(best_h + rng.normal(0.0, 0.04, n), n)
        if p0 is not None:
            p0, _ = smooth_polish(p0, n, (0.01, 0.003), 8)
            best_h, best_s = consider(n, best_h, best_s, p0)
            c = solve_epi(p0, n, 100)
            if c is not None:
                best_h, best_s = consider(n, best_h, best_s, c)
                for a in (0.2, 0.5, 0.8):
                    best_h, best_s = consider(n, best_h, best_s, p0 + a * (c - p0))

    ph, ps = smooth_polish(best_h, n, (0.0015, 0.0005), 12 if n <= 128 else 6)
    best_h, best_s = consider(n, best_h, best_s, ph)

    best_h = project_capped(best_h, n)
    if best_h is None:
        best_h = fb.copy()
    best_s = exact_score(best_h, n)
    if not np.isfinite(best_s):
        best_h = fb.copy()
        best_s = exact_score(best_h, n)
    if not np.isfinite(best_s):
        best_s = 0.5
    return best_h.astype(np.float64), float(best_s), int(n)
```
