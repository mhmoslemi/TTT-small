def lag_values(h,n):
    dx=2.0/float(n)

def lag_jacobian(h,n):

import numpy as np, math
from scipy.optimize import minimize

def project_capped(z,n):
    z=np.asarray(z,dtype=np.float64).ravel()
    if z.size!=n or not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1e6,1e6); s=0.5*n
    lo=float(np.min(z))-1.0; hi=float(np.max(z))+1.0
    for _ in range(80):
        mid=0.5*(lo+hi)
        if np.sum(np.clip(z-mid,0.0,1.0))>s: lo=mid
        else: hi=mid
    u=np.clip(z-0.5*(lo+hi),0.0,1.0)
    for _ in range(5):
        r=s-float(np.sum(u));
        if abs(r)<1e-13: break
        if r>0:
            for i in range(n):
                if abs(r)<1e-13: break
                c=1.0-u[i]
                if c>0.0:
                    a=min(r,c); u[i]+=a; r-=a
        else:
            for i in range(n):
                if abs(r)<1e-13: break
                if u[i]>0.0:
                    a=min(-r,u[i]); u[i]-=a; r+=a
    if not np.all(np.isfinite(u)) or np.any(u<-1e-9) or np.any(u>1+1e-9): return None
    u=np.clip(u,0.0,1.0)
    if abs(float(np.sum(u))-s)>1e-8*max(1,n): return None
    return u
