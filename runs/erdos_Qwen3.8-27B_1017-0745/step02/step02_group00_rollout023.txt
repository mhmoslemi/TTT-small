Let me think about the dual more carefully:

For x ∈ P, define q_d(x) = dx * Σ_{i∈I_d} x[i](1-x[i+d])

The gradient g_d[k] = dx * (1_{0≤k+d<n}(1-x[k+d]) - 1_{0≤k-d<n} x[k-d])

For the MM model at x with curvature μ:
M_x(y) = max_d [q_d(x) + g_d^T (y-x)] + (μ/2)||y-x||²

The dual: minimize over w ∈ Δ_m: D(w) = w^T a + min_{y∈P} [gbar^T (y-x) + (μ/2)||y-x||²]

where gbar = G^T w, a = q(x), and the inner minimizer is y(w) = proj_P(x - gbar/μ).

The inner minimization gives: min_{p: x+p∈P} [gbar^T p + (μ/2)||p||²] = -||gbar||²/(2μ) + [gbar^T (proj_P(x-gbar/μ) - x)] + (μ/2)||proj_P(x-gbar/μ) - x||²

Actually, the dual gap approach: D(w) = w^T a + min_p [gbar^T p + (μ/2)||p||²]

The minimizer of gbar^T p + (μ/2)||p||² over p such that x+p ∈ P is p* = proj_P(x - gbar/μ) - x.

The minimum value is: gbar^T p* + (μ/2)||p*||²

So D(w) = w^T a + gbar^T p* + (μ/2)||p*||²

The dual gradient: ∇D(w) = a + G p*

where G p* means the vector with components (G p*)_d = g_d^T p* (directional derivative of q_d at x in direction p*).

For accelerated gradient: we maximize D(w) over w ∈ Δ_m.

Lipschitz constant of ∇D: ||G||²/μ. A safe bound: since each entry of G has magnitude ≤ dx, and G is m×n, ||G||_F² ≤ m*n*dx². Since dx = 2/n, this is (2n-1)*n*(4/n²) = 4(2n-1)/n < 8.

    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    val = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    return float(val) if np.isfinite(val) else float('inf')

def lag_values(x, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def lag_grad_dir(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    r = np.zeros(m)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        r[d + n - 1] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d]))
                           - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def weighted_gradient(x, w, n):
    dx = 2.0 / n
    gbar = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        gbar[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        gbar[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return gbar

def project_simplex(v):
    m = len(v)
    if m == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.arange(1, m + 1)
    cond = u - (cs - 1.0) / idx > 0
    j = int(np.max(np.where(cond)[0])) + 1
    lam = (cs[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    r = 1.0 - float(np.sum(w))
    if abs(r) > 1e-12:
        pos = np.where(w > 1e-12)[0]
        if len(pos) > 0:
            w[pos[0]] += r
    return np.maximum(w, 0.0)

def dual_solve(x, q, w_init, n, mu, K):
    m = 2 * n - 1
    L_D = 8.0 / mu
    w = project_simplex(w_init.copy())
    z = w.copy()
    t = 1.0
    best_y = None
    best_P = float('inf')
    no_imp = 0
    for _ in range(K):
        gbar_z = weighted_gradient(x, z, n)
        y_z = project_capped(x - gbar_z / mu, n)
        if y_z is None:
            y_z = x.copy()
        p_z = y_z - x
        gradD = q + lag_grad_dir(x, p_z, n)
        w_new = project_simplex(z + gradD / L_D)
        gbar_new = weighted_gradient(x, w_new, n)
        y_new = project_capped(x - gbar_new / mu, n)
        if y_new is None:
            y_new = x.copy()
        p_new = y_new - x
        r_new = lag_grad_dir(x, p_new, n)
        D_val = float(np.dot(w_new, q)) + float(np.dot(gbar_new, p_new)) + 0.5 * mu * float(np.dot(p_new, p_new))
        P_val = float(np.max(q + r_new)) + 0.5 * mu * float(np.dot(p_new, p_new))
        if np.isfinite(P_val) and P_val < best_P and np.all(np.isfinite(y_new)):
            best_P = P_val
            best_y = y_new.copy()
        gap = max(0.0, P_val - D_val) if np.isfinite(D_val) else 1.0
        if gap <= 1e-8 * max(1.0, abs(P_val)):
            break
        if float(np.max(np.abs(w_new - w))) <= 1e-11:
            no_imp += 1
            if no_imp >= 4:
                break
        else:
            no_imp = 0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        t = t_new
        w = w_new
    return best_y

def mm_optimize(x0, n, max_outer, dual_iter, mu_retries):
    dx = 2.0 / n
    mu_min = 2.0 * dx
    x = x0.copy()
    score = exact_score(x, n)
    if not np.isfinite(score):
        return x0, score
    q = lag_values(x, n)
    m = 2 * n - 1
    w_init = np.zeros(m)
    w_init[int(np.argmax(q))] = 1.0
    for _ in range(max_outer):
        improved = False
        for retry in range(mu_retries):
            mu_t = mu_min * (2.0 ** retry)
            y = dual_solve(x, q, w_init, n, mu_t, dual_iter)
            if y is None or float(np.linalg.norm(y - x)) <= 1e-10 * max(1.0, np.sqrt(n)):
                continue
            best_theta, best_s = 0.0, score
            for j in range(10):
                theta = (0.5) ** j
                zc = (1.0 - theta) * x + theta * y
                zp = project_capped(zc, n)
                if zp is None:
                    continue
                st = exact_score(zp, n)
                if np.isfinite(st) and st < best_s - 1e-14:
                    best_s, best_theta = st, theta
            if best_theta > 0.0:
                x = project_capped((1.0 - best_theta) * x + best_theta * y, n)
                if x is None:
                    break
                score = best_s
                q = lag_values(x, n)
                improved = True
                break
        if not improved:
            break
    return x, score

def polish_zoom(h, n, rng, npairs):
    h = h.copy()
    cur = exact_score(h, n)
    for _ in range(npairs):
        i, j = int(rng.integers(0, n - 1)), int(rng.integers(0, n - 1))
        if i == j:
            continue
        for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
            lo_d = max(-h[i], h[j] - 1.0)
            hi_d = min(1.0 - h[i], h[j])
            d = max(lo_d, min(hi_d, delta))
            if abs(d) < 1e-12:
                continue
            h[i] += d
            h[j] -= d
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < cur - 1e-14:
                cur = sc
                best_vec = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return h, cur

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if iv.size > 0 and iv.size <= 999:
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                n = iv.size
                parent = np.clip(iv, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**31) if sd != 0 else 42)
    uniform = project_capped(np.full(n, 0.5), n)
    u_sc = exact_score(uniform, n)
    best, best_score = uniform.copy(), u_sc
    if parent is not None:
        fp = project_capped(parent, n)
        if fp is not None:
            ps = exact_score(fp, n)
            if np.isfinite(ps) and ps < best_score:
                best, best_score = fp.copy(), ps
    if n <= 64:
        ms, mo, di, mr = 5, 16, 120, 3
    elif n <= 128:
        ms, mo, di, mr = 3, 10, 80, 3
    elif n <= 256:
        ms, mo, di, mr = 2, 6, 40, 2
    else:
        ms, mo, di, mr = 1, 3, 20, 2
    for key, var, lo, hi in [('starts', ms, 1, 8), ('max_outer', mo, 0, 30), ('dual_iter', di, 1, 300), ('mu_retries', mr, 1, 5)]:
        try:
            v = int(kwargs.get(key, var))
            var = max(lo, min(hi, v))
        except Exception:
            pass
    ms, mo, di, mr = var if key == 'mu_retries' else (ms, mo, di, mr)
    # Re-parse properly
    ms = ms; mo = mo; di = di; mr = mr
    try:
        ms = max(1, min(8, int(kwargs.get('starts', 5 if n <= 64 else 3))))
    except Exception:
        pass
    try:
        mo = max(0, min(30, int(kwargs.get('max_outer', 16 if n <= 64 else 10))))
    except Exception:
        pass
    try:
        di = max(1, min(300, int(kwargs.get('dual_iter', 120 if n <= 64 else 80))))
    except Exception:
        pass
    try:
        mr = max(1, min(5, int(kwargs.get('mu_retries', 3 if n <= 128 else 2))))
    except Exception:
        pass
    if n == 1 or mo == 0:
        return best, float(best_score), int(n)
    starts = []
    if parent is not None:
        starts.extend([parent.copy(), parent[::-1].copy(), (1.0 - parent).copy(),
                       (1.0 - parent)[::-1].copy(), parent + rng.normal(0, 0.04, n),
                       parent + rng.normal(0, 0.12, n)])
    else:
        starts.extend([uniform.copy(), uniform + rng.normal(0, 0.10, n)])
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        starts.append(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n))
    starts.append(rng.uniform(0, 1, n))
    proj_starts = []
    for s in starts:
        p = project_capped(s, n)
        if p is None:
            continue
        if all(float(np.max(np.abs(p - d))) >= 1e-8 for d in proj_starts):
            proj_starts.append(p)
    proj_starts = proj_starts[:ms]
    for st in proj_starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-14:
            best, best_score = st.copy(), ss
        x_opt, s_opt = mm_optimize(st, n, mo, di, mr)
        if np.isfinite(s_opt) and s_opt < best_score - 1e-14:
            best, best_score = x_opt.copy(), s_opt
    # Final zoom polish
    bp, bs = polish_zoom(best, n, rng, 200)
    bp = project_capped(bp, n)
    if bp is not None:
        sp = exact_score(bp, n)
        if np.isfinite(sp) and sp < best_score - 1e-14:
            best, best_score = bp, sp
    final = project_capped(best, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best.copy()
        fs = best_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), u_sc
    return final, float(fs), int(n)
