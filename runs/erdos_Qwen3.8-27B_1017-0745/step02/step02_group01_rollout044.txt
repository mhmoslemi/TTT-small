    B = b_co[iu] - b_co[ju]
    C = c_co[iu] - c_co[ju]
    sc_ = np.maximum(np.maximum(np.abs(A), np.abs(B)), np.abs(C))
    sc_ = np.maximum(sc_, 1e-300)
    tol_ = 1e-14 * sc_
    qm = np.abs(A) > tol_
    lm = (~qm) & (np.abs(B) > tol_)
    if qm.any():
        Aq, Bq, Cq = A[qm], B[qm], C[qm]
        disc = Bq**2 - 4.0 * Aq * Cq
        ok = disc >= -1e-13 * (Bq**2 + 4.0 * np.abs(Aq * Cq) + 1e-300)
        if ok.any():
            disc = np.maximum(0.0, disc[ok])
            Aq, Bq, Cq = Aq[ok], Bq[ok], Cq[ok]
            s2 = np.sqrt(disc)
            q2 = -0.5 * (Bq + np.where(Bq >= 0, s2, -s2))
            vq = np.abs(q2) > 1e-300
            r1 = np.where(vq, q2 / Aq, -Bq / (2.0 * Aq))
            r2 = np.where(vq, Cq / q2, -Bq / (2.0 * Aq))
            for r in (r1, r2):
                vr = np.isfinite(r) & (r >= lo) & (r <= hi)
                if vr.any():
                    cands.extend(r[vr].tolist())
    if lm.any():
        r = -C[lm] / B[lm]
        vr = np.isfinite(r) & (r >= lo) & (r <= hi)
        if vr.any():
            cands.extend(r[vr].tolist())
    cands = np.array(cands)
    cands = cands[np.isfinite(cands)]
    if cands.size < 2:
        return None
    cands = np.clip(cands, lo, hi)
    cands.sort()
    if cands.size > 1:
        d = np.abs(np.diff(cands))
        msk = np.concatenate(([True], (d > 1e-12) & (d > 1e-12 * np.abs(cands[1:]))))
        cands = cands[msk]
    # Evaluate envelope and pick top candidates
    pred = np.full(cands.size, np.inf)
    for cs in range(0, cands.size, 256):
        tc = cands[cs:cs+256]
        vals = a_co[None, :] * tc[:, None]**2 + b_co[None, :] * tc[:, None] + c_co[None, :]
        pred[cs:cs+256] = vals.max(axis=1)
    order = np.argsort(pred, kind='stable')
    trials = set(order[:10].tolist())
    for sp_t in (lo, hi, 0.0):
        trials.add(int(np.argmin(np.abs(cands - sp_t))))
    best_s, best_z = cur_sc, None
    for idx in sorted(trials):
        t = cands[idx]
        if pred[idx] >= cur_sc:
            continue
        z = project_capped(h + t * v, n)
        if z is None or not is_feasible(z, n):
            continue
        sc = exact_score(z, n)
        if sc < best_s - 1e-12:
            best_s, best_z = sc, z
    return (best_z, best_s) if best_z is not None else None
