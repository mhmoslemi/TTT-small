import numpy as np
from scipy.optimize import minimize
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2 * n + 4):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            delta = min(r, 1.0 - h[idx])
            h[idx] += delta
            r -= delta
        else:
            idx = int(np.argmax(h))
            delta = min(-r, h[idx])
            h[idx] -= delta
            r += delta
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def helmert_forward(x, n):
    d = n - 1
    if d <= 0:
        return np.zeros(1)
    u = np.zeros(d)
    for k in range(d):
        u[k] = x[k] / math.sqrt((k + 1) * (k + 2))
    suffix = np.zeros(d + 1)
    for i in range(d - 1, -1, -1):
        suffix[i] = suffix[i + 1] + u[i]
    y = np.zeros(n)
    y[0] = suffix[0]
    for i in range(1, n):
        y[i] = suffix[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
    return y

def helmert_transpose(y, n):
    d = n - 1
    if d <= 0:
        return np.zeros(0)
    x = np.zeros(d)
    prefix = 0.0
    for k in range(d):
        prefix += y[k]
        x[k] = (prefix - (k + 1) * y[k + 1]) / math.sqrt((k + 1) * (k + 2))
    return x

def latent_to_h(x, n):
    if x.shape[0] != n - 1 or not np.all(np.isfinite(x)):
        return None
    h = project_capped(0.5 + helmert_forward(x, n), n)
    if h is None or not np.all(np.isfinite(h)):
        return None
    return h

def cma_es_start(start_h, n, sigma0, P, G, full_cov, rng):
    d = n - 1
    if d <= 1:
        return start_h, exact_score(start_h, n)
    mu = max(2, d // 2)
    # Actually mu = floor(P/2)
    mu = max(1, P // 2)
    # Weights
    w = np.zeros(mu)
    for i in range(mu):
        w[i] = math.log(mu + 0.5) - math.log(i + 1)
    w /= w.sum()
    mu_eff = 1.0 / np.sum(w * w)
    
    c_sigma = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = max(0.0, min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff)))
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    
    m = helmert_transpose(start_h - 0.5, n)
    sigma = sigma0
    p_sigma = np.zeros(d)
    p_c = np.zeros(d)
    
    if full_cov:
        C = np.eye(d)
    else:
        C = np.ones(d)
    
    best_h = start_h.copy()
    best_s = exact_score(start_h, n)
    sigma_min_count = 0
    
    for t in range(1, G + 1):
        # Sample
        Z = rng.standard_normal((P, d))
        
        if full_cov:
            C_sym = 0.5 * (C + C.T)
            try:
                eig, U = np.linalg.eigh(C_sym)
            except Exception:
                return best_h, best_s
            eig_clipped = np.clip(eig, 1e-10, 1e4)
            if not np.allclose(eig, eig_clipped):
                C = U @ np.diag(eig_clipped) @ U.T
                eig = eig_clipped
            A = U @ np.diag(np.sqrt(eig))
            Y = Z @ A.T
        else:
            C = np.clip(C, 1e-10, 1e4)
            Y = Z * np.sqrt(C)
        
        X = m[None, :] + sigma * Y
        # Convert to phenotypes and score
        scores = np.full(P, float('inf'))
        phenos = [None] * P
        for k in range(P):
            hk = latent_to_h(X[k], n)
            if hk is not None:
                sk = exact_score(hk, n)
                if np.isfinite(sk):
                    scores[k] = sk
                    phenos[k] = hk
        
        # Update incumbent
        for k in range(P):
            if scores[k] < best_s - 1e-12:
                best_s = scores[k]
                best_h = phenos[k].copy()
        
        # Check if enough finite
        finite_mask = np.isfinite(scores)
        if finite_mask.sum() < mu:
            break
        
        # Sort
        order = np.argsort(scores, kind='stable')
        
        if full_cov:
            # Precompute C^(-1/2)
            Cinv_half = U @ np.diag(1.0 / np.sqrt(eig)) @ U.T
        else:
            Cinv_half = 1.0 / np.sqrt(C)
        
        # Mean update
        m_old = m.copy()
        m_new = np.zeros(d)
        for i in range(mu):
            m_new += w[i] * X[order[i]]
        y_w = (m_new - m_old) / sigma
        
        # Evolution paths
        if full_cov:
            p_sigma_new = (1.0 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * (Cinv_half @ y_w)
        else:
            p_sigma_new = (1.0 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * (Cinv_half * y_w)
        
        h_sigma = 1.0 if (np.linalg.norm(p_sigma_new) / math.sqrt(1.0 - (1.0 - c_sigma) ** (2 * t))) / chi_d < 1.4 + 2.0 / (d + 1.0) else 0.0
        
        p_c_new = (1.0 - c_c) * p_c + h_sigma * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w
        
        # Covariance update
        if full_cov:
            C_new = (1.0 - c1 - c_mu) * C
            C_new += c1 * (np.outer(p_c_new, p_c_new) + (1.0 - h_sigma) * c_c * (2.0 - c_c) * C)
            for i in range(mu):
                yi = (X[order[i]] - m_old) / sigma
                C_new += c_mu * w[i] * np.outer(yi, yi)
        else:
            C_new = (1.0 - c1 - c_mu) * C
            C_new += c1 * (p_c_new * p_c_new + (1.0 - h_sigma) * c_c * (2.0 - c_c) * C)
            for i in range(mu):
                yi = (X[order[i]] - m_old) / sigma
                C_new += c_mu * w[i] * yi * yi
        
        # Sigma update
        exp_arg = (c_sigma / d_sigma) * (np.linalg.norm(p_sigma_new) / chi_d - 1.0)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        sigma = sigma * math.exp(exp_arg)
        sigma = np.clip(sigma, 1e-4, 0.5)
        
        m = m_new
        p_sigma = p_sigma_new
        p_c = p_c_new
        C = C_new
        
        if not np.all(np.isfinite(m)) or not np.all(np.isfinite(p_sigma)) or not np.all(np.isfinite(p_c)):
            return best_h, best_s
        if full_cov and not np.all(np.isfinite(C)):
            return best_h, best_s
        if not full_cov and not np.all(np.isfinite(C)):
            return best_h, best_s
        
        # Early stop
        if sigma <= 1e-4 + 1e-15:
            sigma_min_count += 1
            if sigma_min_count >= 5:
                break
        else:
            sigma_min_count = 0
    
    # Score final mean
    hm = latent_to_h(m, n)
    if hm is not None:
        sm = exact_score(hm, n)
        if sm < best_s - 1e-12:
            best_s = sm
            best_h = hm.copy()
    
    return best_h, best_s

    add(1.0 - base)
