    j = max(1, int(np.sum(pos)))
    lam = (cum[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    r = 1.0 - float(np.sum(w))
    if abs(r) > 1e-14:
        idx = int(np.argmax(w)); w[idx] = max(0.0, w[idx] + r)
        s2 = float(np.sum(w))
        if s2 > 1e-14:
            w /= s2
        else:
            w = np.ones(m) / m
    return w
