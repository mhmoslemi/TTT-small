import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_jacobian(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        idx = d + n - 1
        if d >= 0:
            # c(d) = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # d/dh[q]: (1-h[q-d]) if d<=q<=n-1, -h[q+d] if 0<=q<=n-1-d

            q2 = np.arange(j, n)
            J[idx, q2] -= h[q2 - j] * dx

    return z[n]

def slsqp_obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

    h = z[:n]
    t = z[n]
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - c

    h = z[:n]
    J = lag_jacobian(h, n)
    n_lags = 2 * n - 1
    out = np.zeros((n_lags, n + 1))
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

    if z.shape != (n,):

            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d

def lag_jacobian(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        idx = d + n - 1
        if d >= 0:
            q1 = np.arange(d, n)
            J[idx, q1] += (1.0 - h[q1 - d]) * dx
            q2 = np.arange(0, n - d)
            J[idx, q2] -= h[q2 + d] * dx
        else:
            j = -d
            q1 = np.arange(0, n - j)
            J[idx, q1] += (1.0 - h[q1 + j]) * dx
            q2 = np.arange(j, n)
            J[idx, q2] -= h[q2 - j] * dx
    return J

    return z[n] - c
