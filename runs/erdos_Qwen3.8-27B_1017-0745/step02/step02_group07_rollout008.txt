                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    fb_s = exact_score(fb, n)
    best_h = fb.copy()
    best_s = fb_s
    U = build_cosine_basis(n)
    pool = build_pool(fb, n, rng, 16)
    pool_s = np.array([exact_score(v, n) for v in pool], dtype=np.float64)
    for i in range(len(pool)):
        if pool_s[i] < best_s - 1e-12:
            best_s = pool_s[i]
            best_h = pool[i].copy()
    if n <= 64:
        n_restarts, gens_pick, gens_pol = 4, 120, 40
    elif n <= 128:
        n_restarts, gens_pick, gens_pol = 3, 70, 25
    elif n <= 512:
        n_restarts, gens_pick, gens_pol = 2, 35, 15
    else:
        n_restarts, gens_pick, gens_pol = 2, 20, 10
    order = np.argsort(pool_s, kind='stable')
    bases = [pool[0].copy()]
    for ii in order:
        if len(bases) >= n_restarts:
            break
        if ii == 0:
            continue
        diverse = True
        for b in bases:
            if np.max(np.abs(pool[ii] - b)) < 0.05:
                diverse = False
                break
        if diverse:
            bases.append(pool[ii].copy())
    while len(bases) < n_restarts:
        bases.append(pool[0].copy())
    p_dim = U.shape[1]
    sigma0 = float(np.clip(0.10 * math.sqrt(n / max(p_dim, 1)), 0.04, 0.35))
    for b in bases:
        best_h, best_s = cmaes_search(b, U, sigma0, gens_pick, rng, best_h, best_s, n)
    best_h, best_s = cmaes_search(best_h, U, sigma0 * 0.5, gens_pol, rng, best_h, best_s, n)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_s + 1e-12:
        final = fb.copy()
        fs = fb_s
    return final.astype(np.float64), float(fs), int(n)
