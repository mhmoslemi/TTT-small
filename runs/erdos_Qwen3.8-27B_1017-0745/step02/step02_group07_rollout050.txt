def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(v)):
        return float('inf')
    return float(np.max(v))

def lag_values(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    Q[0] = np.sum(h * (1.0 - h))
    for r in range(1, n):
        m = n - r
        Q[2 * r - 1] = np.sum(h[:m] * (1.0 - h[r:]))
        Q[2 * r] = np.sum(h[r:] * (1.0 - h[:m]))
    return Q * dx

def lag_jacobian(h, n):
    dx = 2.0 / n
    Q, G = np.empty(2 * n - 1), np.zeros((2 * n - 1, n))
    Q[0] = np.sum(h * (1.0 - h))
    G[0] = (1.0 - 2.0 * h) * dx
    for r in range(1, n):
        m = n - r
        Q[2 * r - 1] = np.sum(h[:m] * (1.0 - h[r:]))
        G[2 * r - 1, :m] += (1.0 - h[r:]) * dx
        G[2 * r - 1, r:] -= h[:m] * dx
        Q[2 * r] = np.sum(h[r:] * (1.0 - h[:m]))
        G[2 * r, :m] -= h[r:] * dx
        G[2 * r, r:] += (1.0 - h[:m]) * dx
    return Q, G

def smooth_val_grad(h, n, tau):
    Q = lag_values(h, n)
    qmax = Q[0]
    for i in range(1, len(Q)):
        if Q[i] > qmax:
            qmax = Q[i]

    if not np.all(np.isfinite(g)):

    return F, g
