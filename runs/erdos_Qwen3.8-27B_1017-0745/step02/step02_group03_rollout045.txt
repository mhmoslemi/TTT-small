        ps = h[p] + h[q]

        new_p = max(max(0.0, ps - 1.0), min(1.0, ps))
        new_q = ps - new_p
        if new_p < -1e-12 or new_p > 1.0 + 1e-12 or new_q < -1e-12 or new_q > 1.0 + 1e-12:
            continue
        trial = h.copy()
        trial[p] = new_p
        trial[q] = new_q
        sc = exact_score(trial, n)
        if np.isfinite(sc) and sc < score - 1e-12:
            h = trial
            score = sc
    return h, score

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(2, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        sd = max(0, int(seed) % (2**32))
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 1:
        return fallback, fb_score, n
    max_k = int(kwargs.get('max_iter', 6000))
    B = min(max(1, max_k), max(1, int(6e7 / (2 * n * n))))
    ns = int(kwargs.get('starts', 4))
    ns = max(1, min(ns, 6))
    starts = [fallback.copy()]
    if parent is not None:
        add_start(starts, fallback + rng.normal(0, 0.04, n), n)
        add_start(starts, fallback + rng.normal(0, 0.12, n), n)
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, fallback + rng.normal(0, 0.20, n), n)
    else:
        blk = np.zeros(n); blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:ns]
    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    per_start = max(1, B // (len(starts) + 1))
    for st in starts:
        h_out, sc_out = pair_search(st, n, per_start, rng)
        if sc_out < best_score - 1e-12:
            best, best_score = h_out.copy(), sc_out
    remaining = B - len(starts) * per_start
    if remaining > 0:
        h_out, sc_out = pair_search(best, n, remaining, rng)
        if sc_out < best_score - 1e-12:
            best, best_score = h_out.copy(), sc_out
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
