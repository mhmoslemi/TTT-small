    u = cp.Variable()
    qp = cp.Parameter(2*n-1)
    Gp = cp.Parameter((2*n-1, n))
    hp = cp.Parameter(n)
    obj = cp.Minimize(u + (L/2.0)*cp.sum_squares(d))
    cons = [Gp @ d + qp <= u, cp.sum(d) == 0, d >= -hp, d <= 1.0 - hp]
    return cp.Problem(obj, cons), d, u, qp, Gp, hp

def get_solvers():
    sl = []
    for name, mod in [('ECOS','ecos'),('MOSEK','mosek'),('GUROBI','gurobi'),('CVXOPT','cvxopt')]:
        try:
            __import__(mod)
            sl.append(name)
        except Exception:
            pass
    return sl

def mm_optimize(h, h0s, prob, dv, uv, qp, Gp, hp, solvers, n):
    dx = 2.0 / n
    L = 2.0 * dx
    if n <= 64:
        max_it = 60
    elif n <= 128:
        max_it = 40
    elif n <= 256:
        max_it = 25
    else:
        max_it = 15
    h = h.copy()
    h0s = h0s
    for it in range(max_it):
        ld = lag_data(h, n)
        if ld is None:
            break
        qv, Gv = ld
        qp.value = qv
        Gp.value = Gv
        hp.value = h
        solved = False
        for sv in solvers:
            try:
                kw = {}
                if sv == 'ECOS':
                    kw = {'max_iters': 5000, 'abs_tol': 1e-9, 'rel_tol': 1e-9}
                elif sv == 'CVXOPT':
                    kw = {'maxiter': 500}
                elif sv == 'GUROBI':
                    kw = {'Threads': 1, 'OutputFlag': 0}
                prob.solve(solver=sv, **kw)
                if prob.status in ('optimal', 'optimal_inaccurate') and dv.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        step = np.array(dv.value, dtype=np.float64).ravel()
        if step.shape != (n,) or not np.all(np.isfinite(step)):
            break
        if np.max(np.abs(step)) <= 1e-9:
            break
        best_cand, best_sc = None, h0s
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = project_capped(h + alpha * step, n)
            if cand is None or not np.all(np.isfinite(cand)):
                continue
            sc = exact_score(cand, n)
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best_cand = cand
        if best_cand is None:
            break
        if np.max(np.abs(best_cand - h)) < 1e-11:
            break
        h = best_cand
        h0s = best_sc
    return h, h0s

def local_pair(h, n, rounds=30):
    s = exact_score(h, n)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                for delta in (0.02, -0.02, 0.05, -0.05):
                    h2 = h.copy()
                    a, b = h2[i], h2[j]
                    h2[i] = a + delta
                    h2[j] = b - delta
                    if h2[i] < -1e-12 or h2[i] > 1+1e-12 or h2[j] < -1e-12 or h2[j] > 1+1e-12:
                        continue
                    h2 = project_capped(h2, n)
                    if h2 is None:
                        continue
                    sc = exact_score(h2, n)
                    if sc < s - 1e-13:
                        h = h2
                        s = sc
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_s = exact_score(fb, n)
    best_val = fb_s
    best_vec = fb.copy()
    pool = [fb.copy()]
    def try_add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    try_add(project_capped(fb[::-1], n))
    try_add(project_capped(1.0 - fb, n))
    try_add(project_capped(1.0 - fb[::-1], n))
    try_add(np.full(n, 0.5))
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2: blk[n//2] = 0.5
    try_add(project_capped(blk, n))
    for kk in (1, 2, 3, 5):
        for ph in (0.0, math.pi/3, math.pi/2):
            try_add(project_capped(0.5 + 0.35*np.cos(2*np.pi*kk*np.arange(n)/n + ph), n))
    try_add(project_capped(fb + rng.normal(0, 0.1, n), n))
    try_add(project_capped(fb + rng.normal(0, 0.2, n), n))
    try_add(project_capped(rng.uniform(0, 1, n), n))
    try_add(project_capped(np.roll(fb, n//3), n))
    try_add(project_capped(make_bin(rng, n), n))
    pool_scores = np.array([exact_score(v, n) for v in pool])
    for i in range(len(pool)):
        if pool_scores[i] < best_val - 1e-12:
            best_val = pool_scores[i]
            best_vec = pool[i].copy()
    prob = None
    dv = uv = qp = Gp = hp = None
    solvers = get_solvers()
    try:
        prob, dv, uv, qp, Gp, hp = build_qp(n)
    except Exception:
        prob = None
        solvers = []
    if n <= 64:
        n_starts = min(6, len(pool))
    elif n <= 128:
        n_starts = min(4, len(pool))
    else:
        n_starts = min(3, len(pool))
    sel = [0]
    for idx in np.argsort(pool_scores):
        if len(sel) >= n_starts:
            break
        idx = int(idx)
        if idx in sel:
            continue
        div = True
        for si in sel:
            if np.max(np.abs(pool[idx] - pool[si])) < 0.04:
                div = False
                break
        if div:
            sel.append(idx)
    for idx in np.argsort(pool_scores):
        if len(sel) >= n_starts:
            break
        if int(idx) not in sel:
            sel.append(int(idx))
    if prob is not None and solvers:
        for si in sel:
            h_init = pool[si].copy()
            h_mm, s_mm = mm_optimize(h_init, pool_scores[si], prob, dv, uv, qp, Gp, hp, solvers, n)
            if s_mm < best_val - 1e-12:
                best_val = s_mm
                best_vec = h_mm.copy()
    else:
        for si in sel:
            h_init = pool[si].copy()
            h_pw, s_pw = local_pair(h_init, n, rounds=20)
            if s_pw < best_val - 1e-12:
                best_val = s_pw
                best_vec = h_pw.copy()
    h_pw, s_pw = local_pair(best_vec, n, rounds=40)
    if s_pw < best_val - 1e-12:
        best_val = s_pw
        best_vec = h_pw.copy()
    h_final = project_capped(best_vec, n)
    if h_final is None or not is_feasible(h_final, n):
        h_final = fb.copy()
    s_final = exact_score(h_final, n)
    if not np.isfinite(s_final) or s_final >= fb_s - 1e-12:
        h_final = fb.copy()
        s_final = exact_score(fb, n)
    if not np.isfinite(s_final):
        h_final = np.full(n, 0.5)
        s_final = exact_score(h_final, n)
    return h_final.astype(np.float64), float(s_final), int(n)

def make_bin(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)
    v[p[:n//2]] = 1.0
    if n % 2:
        v[p[n//2]] = 0.5
    return v
