import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')

def lag_abcs(h, d, n):
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    c = np.zeros(2 * n - 1)
    a[0] = dx * np.sum(h * (1.0 - h))
    b[0] = dx * np.sum(d * (1.0 - 2.0 * h))
    c[0] = -dx * np.sum(d * d)
    idx = 1
    for r in range(1, n):
        m = n - r
        h0, h1 = h[:m], h[r:]
        d0, d1 = d[:m], d[r:]
        a[idx] = dx * np.sum(h1 * (1.0 - h0))
        b[idx] = dx * np.sum(d1 * (1.0 - h0) - h1 * d0)
        c[idx] = -dx * np.sum(d1 * d0)
        idx += 1
        a[idx] = dx * np.sum(h0 * (1.0 - h1))
        b[idx] = dx * np.sum(d0 * (1.0 - h1) - h0 * d1)
        c[idx] = -dx * np.sum(d0 * d1)
        idx += 1
    return a, b, c

def feasible_interval(h, d, n):
    lo, hi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if di > 1e-15:
            lo = max(lo, -h[i] / di)
            hi = min(hi, (1.0 - h[i]) / di)
        elif di < -1e-15:
            lo = max(lo, (1.0 - h[i]) / di)
            hi = min(hi, -h[i] / di)
    lo = max(lo, -1e6)
    hi = min(hi, 1e6)
    if not np.isfinite(lo) or not np.isfinite(hi) or hi - lo <= 1e-12:
        return None
    return lo, hi

def envelope_vals(a, b, c, ts):
    # ts: array of parameters, return max over lags for each t
    nlags = len(a)
    out = np.full(len(ts), -1e30)
    chunk = 256
    for s0 in range(0, len(ts), chunk):
        s1 = min(s0 + chunk, len(ts))
        t = ts[s0:s1]
        vals = a[None, :] + b[None, :] * t[:, None] + c[None, :] * (t[:, None]) ** 2
        out[s0:s1] = np.max(vals, axis=1)
    return out

def line_candidates_small(lo, hi, a, b, c):
    candidates = [lo, hi, 0.0]
    nl = len(a)
    # Stationary points
    for j in range(nl):
        if abs(c[j]) > 1e-14:
            t = -b[j] / (2.0 * c[j])
            if lo - 1e-10 <= t <= hi + 1e-10:
                candidates.append(t)
    # Pairwise intersections
    for i in range(nl):
        for k in range(i + 1, nl):
            A = c[i] - c[k]
            B = b[i] - b[k]
            C = a[i] - a[k]
            sc = max(1.0, abs(A), abs(B), abs(C))
            if abs(A) <= 1e-12 * sc:
                if abs(B) > 1e-12 * sc:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        candidates.append(t)
            else:
                disc = B * B - 4.0 * A * C
                disc_tol = 1e-14 * (1.0 + B * B + abs(4.0 * A * C))
                if disc < -disc_tol:
                    continue
                if disc < 0:
                    disc = 0.0
                sq = math.sqrt(disc)
                q = -0.5 * (B + math.copysign(sq, B))
                if abs(q) > 1e-30:
                    r1 = q / A
                    r2 = C / q
                else:
                    r1 = -B / (2.0 * A)
                    r2 = r1
                for t in (r1, r2):
                    if np.isfinite(t) and lo - 1e-10 <= t <= hi + 1e-10:
                        candidates.append(t)
    # Deduplicate
    candidates.sort()
    dedup = [candidates[0]]
    for t in candidates[1:]:
        if t - dedup[-1] > 1e-9 * max(1.0, hi - lo):
            dedup.append(t)
    # Evaluate envelope
    cv = envelope_vals(a, b, c, np.array(dedup))
    # Return top 4 distinct t with lowest envelope
    order = np.argsort(cv)
    result = []
    for idx in order:
        t = dedup[idx]
        if not result or abs(t - result[-1]) > 1e-9:
            result.append(t)
        if len(result) >= 4:
            break
    return result

def lag_gradient(h, lag_idx, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag_idx == 0:
        return dx * (1.0 - 2.0 * h)
    # lag_idx odd: positive lag r=(lag_idx+1)//2, negative lag r=lag_idx//2
    if lag_idx % 2 == 1:
        r = (lag_idx + 1) // 2
        m = n - r
        g[:m] += dx * h[r:]  # wait, let me re-derive
        # Q_{r,+} = dx * sum_{i=0}^{m-1} h_{i+r}(1-h_i)
        # d/dh_i for i in [0,m): -dx * h_{i+r}
        # d/dh_{i+r} for i+r in [r, n): dx * (1-h_i) = dx*(1-h[i-r])
        g[:m] -= dx * h[r:]
        g[r:] += dx * (1.0 - h[:m])
    else:
        r = lag_idx // 2
        m = n - r
        # Q_{r,-} = dx * sum_{i=0}^{m-1} h_i(1-h_{i+r})
        g[:m] += dx * (1.0 - h[r:])
        g[r:] -= dx * h[:m]
    return g

def build_pool(fb, n, rng):
    pool = [fb.copy()]
    def add(v):
        pv = project_capped(v, n)
        if pv is not None:
            for s in pool:
                if np.max(np.abs(pv - s)) < 1e-9:
                    return
            pool.append(pv)
    add(fb[::-1].copy())
    add(1.0 - fb)
    add(1.0 - fb[::-1].copy())
    add(np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    add(blk)
    add(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    add(rng.uniform(0, 1, n))
    add(fb + rng.normal(0, 0.08, n))
    add(fb + rng.normal(0, 0.18, n))
    add(np.roll(fb, n // 3))
    for k in (1, 2, 3):
        for ph in (0.0, math.pi / 2):
            add(0.5 + 0.4 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph))
    return pool[:16]

def local_search(h0, n, rng, sweeps, global_best, global_bs):
    h = h0.copy()
    hs = exact_score(h, n)
    bi = global_best if global_best is not None else h.copy()
    bs = global_bs
    kicks = 0
    for _sw in range(sweeps):
        # Compute active lags
        lv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        nl = 2 * n - 1
        top_k = min(8, nl)
        active_idx = np.argsort(lv)[::-1][:top_k]
        # Gradients for active lags
        grads = [lag_gradient(h, int(idx), n) for idx in active_idx]
        # Pair directions
        if n <= 64:

            keep = 10
        else:

            keep = 5 if n <= 256 else 3

            if iv is None:
                continue

            slopes = [g[i] - g[j] for g in grads]
            key = min(slopes)  # most negative slope means most improvement
            scored_pairs.append((key, i, j))
        scored_pairs.sort()
        best_moves = []
        for (key, i, j) in scored_pairs[:keep]:
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            iv = feasible_interval(h, d, n)
            if iv is None:

            lo, hi = iv
            abc = lag_abcs(h, d, n)
            cand_ts = line_candidates_small(lo, hi, abc[0], abc[1], abc[2])
            for t in cand_ts:

                if trial is None or not is_feasible(trial, n):

                if np.max(np.abs(trial - h)) < 1e-10:

                ts = exact_score(trial, n)
                if ts < hs - 1e-12:
                    best_moves.append((ts, trial))
                if ts < bs - 1e-12:
                    bs = ts; bi = trial.copy()
        # Also add a few random directions
        for _ in range(3 if n <= 64 else 1):
            d = rng.normal(0, 1, n)
            d -= np.mean(d)
            d[-1] = -np.sum(d[:-1])
            mx = np.max(np.abs(d))
            if mx < 1e-14:
                continue
            d /= mx
            iv = feasible_interval(h, d, n)
            if iv is None:
                continue
            lo, hi = iv
            abc = lag_abcs(h, d, n)
            cand_ts = line_candidates_small(lo, hi, abc[0], abc[1], abc[2])
            for t in cand_ts:
                t = max(lo, min(hi, t))
                trial = project_capped(h + t * d, n)
                if trial is None or not is_feasible(trial, n):
                    continue
                if np.max(np.abs(trial - h)) < 1e-10:
                    continue
                ts = exact_score(trial, n)
                if ts < hs - 1e-12:
                    best_moves.append((ts, trial))
                if ts < bs - 1e-12:
                    bs = ts; bi = trial.copy()
        if best_moves:
            best_moves.sort(key=lambda x: x[0])
            h = best_moves[0][1].copy()
            hs = best_moves[0][0]
            kicks = 0  # reset on improvement? No, strategy says line improvement does NOT reset
        else:

                sigma = 0.08 if kicks == 0 else 0.20
                z = h + sigma * rng.normal(0, 1, n)
                trial = project_capped(z, n)
                if trial is not None and is_feasible(trial, n):
                    ts = exact_score(trial, n)
                    h = trial
                    hs = ts
                    if ts < bs - 1e-12:
                        bs = ts; bi = h.copy()

            else:

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):

            break
        if r > 0:

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')

def lag_abcs(h, d, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    a = np.zeros(nl); b = np.zeros(nl); c = np.zeros(nl)
    a[0] = dx * np.dot(h, 1.0 - h)
    b[0] = dx * np.dot(d, 1.0 - 2.0 * h)
    c[0] = -dx * np.dot(d, d)
    idx = 1
    for r in range(1, n):
        m = n - r

        a[idx] = dx * np.dot(h1, 1.0 - h0)

def lag_grad(h, li, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if li == 0:
        return dx * (1.0 - 2.0 * h)
    if li % 2 == 1:
        r = (li + 1) // 2; m = n - r
        g[:m] -= dx * h[r:]
        g[r:] += dx * (1.0 - h[:m])
    else:
        r = li // 2; m = n - r
        g[:m] += dx * (1.0 - h[r:])
        g[r:] -= dx * h[:m]
    return g

    add(fb[::-1].copy()); add(1.0 - fb); add(1.0 - fb[::-1].copy())
    add(np.full(n, 0.5))
