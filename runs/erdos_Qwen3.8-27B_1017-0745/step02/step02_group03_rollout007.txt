        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed) % (2 ** 32)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 64:
        n_starts_def, mi_def = 7, 30
    elif n <= 128:
        n_starts_def, mi_def = 5, 20
    elif n <= 256:
        n_starts_def, mi_def = 2, 8
    else:
        n_starts_def, mi_def = 2, 0
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts_def)), 8))
    except Exception:
        n_starts = n_starts_def
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mi_def)), 60))
    except Exception:
        max_iter = mi_def
    if n > 256:
        max_iter = 0
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.04, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best = [fallback.copy(), fb_score]
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best[1] - 1e-12:
            best[0] = st.copy()
            best[1] = ss
    if max_iter > 0:
        for st in starts:
            try:
                trust_region_search(st, n, max_iter, best)
            except Exception:
                pass
    n_hops, hop_sds = 0, []
    if n <= 128 and max_iter > 0:
        n_hops, hop_sds = 2, [0.025, 0.08]
    elif n <= 256 and max_iter > 0:
        n_hops, hop_sds = 1, [0.05]
    for i in range(n_hops):
        pert = best[0] + rng.normal(0, hop_sds[i % len(hop_sds)], n)
        hp = project_capped(pert, n)
        if hp is None:
            continue
        try:
            trust_region_search(hp, n, max_iter, best)
        except Exception:
            pass
    final = project_capped(best[0], n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = fallback.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
