def subgrad_polish(h, n, steps):
    dx = 2.0 / n
    y = h.copy()
    bs = exact_score(h, n)
    for k in range(steps):
        a_max, g_a = -np.inf, np.zeros(n)
        for d in range(-(n - 1), n):
            p0, q0 = max(d, 0), max(-d, 0)
            m = n - abs(d)
            xp, xq = y[p0:p0 + m], y[q0:q0 + m]
            a_d = dx * np.dot(xp, 1.0 - xq)
            if a_d > a_max:
                a_max = a_d
                g_a = np.zeros(n)
                g_a[p0:p0 + m] += dx * (1.0 - xq)
                g_a[q0:q0 + m] -= dx * xp
        nv = np.linalg.norm(g_a)
        if nv < 1e-14:
            break

        y_new = project_capped(y - eta * g_a, n)

    if n == 1:
        return fb, fb_s, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P, mg, pat, mr, sp = 48, 250, 35, 2, 300
    elif n <= 128:
        P, mg, pat, mr, sp = 36, 180, 28, 2, 150
    elif n <= 256:
        P, mg, pat, mr, sp = 24, 120, 22, 1, 80
    else:
        P, mg, pat, mr, sp = 16, 70, 16, 1, 40
    pop = build_population(fb, n, rng, P)
    sc = np.array([exact_score(pop[i], n) for i in range(P)])
    best_h, best_s = de_search(pop, sc, n, rng, mg, pat, mr)
    try:
        best_h, best_s = subgrad_polish(best_h, n, sp)
    except Exception:
        pass
    fin = project_capped(best_h, n)
    if fin is None or not is_feasible(fin, n):
        fin = fb.copy()
    fs = exact_score(fin, n)
    if fs == float('inf') or fs > fb_s + 1e-12:
        fin, fs = fb.copy(), fb_s
    return fin, fs, int(n)
