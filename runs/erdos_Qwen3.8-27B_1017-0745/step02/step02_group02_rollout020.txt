            all_pts.sort(key=_sort_key_sc_d)
            in_range = [(sc, d, t) for sc, d, t in all_pts if a - 1e-14 <= d <= b + 1e-14]
            if len(in_range) < 2:
                break
            in_range.sort(key=_sort_key_sc_d)
            ns2, nd2, nt2 = in_range[0]
            if ns2 < best_s - 1e-12:
                best_s, best_h = ns2, nt2.copy()
            tested = in_range
        # Update h from current global best for next pair
        h = best_h
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
                n = int(p.size)
    except Exception:
        pass
    if n is None:
        n = 47
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size == n:
        fb = project_capped(parent, n)
    elif parent is not None:
        old_x = (np.arange(parent.size) + 0.5) / parent.size
        new_x = (np.arange(n) + 0.5) / n
        fb = project_capped(np.interp(new_x, old_x, parent), n)
    else:
        fb = None
    if fb is None:
        fb = np.full(n, 0.5)
    const_fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, n), 1
    B = min(10000, max(120, int(22000000 / (n * n))))
    A = int(0.62 * B)
    if n <= 64:
        n_starts, n_restarts = 7, 2
    elif n <= 128:
        n_starts, n_restarts = 4, 1
    elif n <= 256:
        n_starts, n_restarts = 3, 1
    else:
        n_starts, n_restarts = 2, 0
    pool = build_starts(fb, n, rng, n_starts)
    counter = [0, B]
    best_h, best_s = fb.copy(), exact_score(fb, n)
    cs = exact_score(const_fb, n)
    if cs < best_s:
        best_s, best_h = cs, const_fb.copy()
    for st in pool:
        sc = exact_score(st, n)
        counter[0] += 1
        if sc < best_s - 1e-12:
            best_s, best_h = sc, st.copy()
    dx = 2.0 / n
    stages = [dx * v for v in (0.05, 0.02, 0.006, 0.0015, 0.0)]
    radii = [0.45, 0.25, 0.12, 0.05, 0.02]
    M = len(pool) + n_restarts
    per_chain = max(0, (A - len(pool) - n_restarts) // M) if M > 0 else 0
    for st in pool:
        if counter[0] >= counter[1]:
            break
        cur = st.copy()
        cur_s = exact_score(cur, n)
        counter[0] += 1
        cur, cur_s, best_h, best_s = run_chain(
            cur, cur_s, rng, n, stages, radii, per_chain, counter, best_h, best_s, pool)
    parent_score = exact_score(fb, n)
    gain = min(parent_score, cs) - best_s
    if n <= 64:
        rstds = [0.12, 0.25] if gain < 1e-5 else [0.05, 0.12]
    else:
        rstds = [0.12 if gain < 1e-5 else 0.06]
    for ri in range(min(n_restarts, len(rstds))):
        if counter[0] >= counter[1]:
            break
        pert = project_capped(best_h + rng.normal(0, rstds[ri], n), n)
        if pert is None:
            continue
        cur, cur_s = pert.copy(), exact_score(pert, n)
        counter[0] += 1
        cur, cur_s, best_h, best_s = run_chain(
            cur, cur_s, rng, n, stages, radii, per_chain, counter, best_h, best_s, pool)
    # Second pass of chain from best with tighter stages
    stages2 = [dx * v for v in (0.004, 0.001, 0.0)]
    radii2 = [0.08, 0.04, 0.015]
    pass2_quota = max(0, (A - counter[0]) // 3)
    if pass2_quota > 5 and counter[0] < counter[1]:
        cur, cur_s = best_h.copy(), best_s
        cur, cur_s, best_h, best_s = run_chain(
            cur, cur_s, rng, n, stages2, radii2, pass2_quota, counter, best_h, best_s, pool)
    # Pair line scan
    best_h, best_s = pair_line_scan(best_h, rng, n, counter)
    # Final polish: short chain from line-scan result
    if counter[0] < counter[1] - 10:
        cur, cur_s = best_h.copy(), best_s
        cur, cur_s, best_h, best_s = run_chain(
            cur, cur_s, rng, n, stages2, radii2, 30, counter, best_h, best_s, pool)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = const_fb.copy()
        sf = exact_score(final, n)
    return final, float(sf), int(n)
