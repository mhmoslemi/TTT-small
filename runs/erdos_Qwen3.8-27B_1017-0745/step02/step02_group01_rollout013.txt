                zc = project_capped(zc, n)
                if zc is None:
                    continue
                zs = exact_score(zc, n)
                if zs < best_score - 1e-12:
                    best_score = zs
                    best_h = zc.copy()
            vc, _ = soft_fg(current, n, tau)
            vy, _ = soft_fg(y, n, tau)
            if np.isfinite(vy) and vy <= vc + 1e-9:
                current = y
            else:
                current = project_capped(0.5 * (current + y), n)
                if current is None:
                    current = y
    for polish_tau in (1e-5, 3e-6, 1e-6):
        cur = project_capped(best_h, n)
        if cur is None:
            break
        try:
            res = minimize(soft_fg, cur, jac=True, args=(n, polish_tau),
                           bounds=bounds, constraints=cons,
                           method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        except Exception:
            break
        if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
            continue
        y = project_capped(res.x, n)
        if y is None:
            continue
        ys = exact_score(y, n)
        if ys < best_score - 1e-12:
            best_score = ys
            best_h = y.copy()
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb_s = exact_score(fallback, n)
    if fs > fb_s + 1e-12:
        final, fs = fallback, fb_s
    return final, float(fs), int(n)
