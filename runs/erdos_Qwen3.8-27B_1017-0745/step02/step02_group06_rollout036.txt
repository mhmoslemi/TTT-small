import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_forward(x, n):
    # x: length d = n-1, returns Qx of length n
    d = n - 1
    if d == 0:
        return np.zeros(n)
    D = np.sqrt(np.arange(1, d + 1).astype(np.float64) * np.arange(2, d + 2).astype(np.float64))
    a = x / D
    T = np.zeros(n)
    # T[i] = sum_{j=i}^{d-1} a[j], T[d] = 0
    # reverse cumsum
    T[:d] = np.zeros(d)
    running = 0.0
    for j in range(d - 1, -1, -1):
        running += a[j]
        T[j] = running
    # (Qx)[0] = T[0]
    # (Qx)[i] = T[i] - sqrt(i/(i+1)) * x[i-1] for i=1,...,n-1
    result = np.zeros(n)
    result[0] = T[0]
    for i in range(1, n):
        if i <= d - 1:
            result[i] = T[i] - math.sqrt(i / (i + 1)) * x[i - 1]
        else:
            # i = n-1 = d
            # T[i] = 0 (since i > d-1)
            result[i] = -math.sqrt(i / (i + 1)) * x[i - 1]
    return result

def helmert_backward(v, n):
    # v: length n, returns Q^T v of length d = n-1
    d = n - 1
    if d == 0:
        return np.zeros(0)
    # prefix sum
    ps = np.cumsum(v)
    result = np.zeros(d)
    for j in range(d):
        D = math.sqrt((j + 1) * (j + 2))
        result[j] = (ps[j] - (j + 1) * v[j + 1]) / D
    return result

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x, n):
    d = n - 1
    if d == 0:
        return np.zeros(n)
    jj = np.arange(1, d + 1, dtype=np.float64)
    D = np.sqrt(jj * (jj + 1.0))
    a = x / D
    T = np.zeros(n)
    running = 0.0
    for j in range(d - 1, -1, -1):
        running += a[j]
        T[j] = running
    result = np.empty(n)
    result[0] = T[0]
    for i in range(1, n):
        result[i] = T[i] - math.sqrt(i / (i + 1.0)) * x[i - 1]
    return result

def helmert_bwd(v, n):
    d = n - 1
    if d == 0:
        return np.zeros(0)
    ps = np.cumsum(v)
    jj = np.arange(1, d + 1, dtype=np.float64)
    D = np.sqrt(jj * (jj + 1.0))
    return (ps - jj * v[1:d + 1]) / D

    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-4 * n:
                    parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if parent is not None:
        parent = project_capped(parent, n)
    fallback = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is not None:
            fallback = fb
    rng = np.random.default_rng(int(seed))
    if n <= 1:
        return fallback, exact_score(fallback, n), n
    if n <= 64:
        n_starts, lam, max_gen, stall_lim = 5, 20, 55, 16
    elif n <= 128:
        n_starts, lam, max_gen, stall_lim = 4, 16, 35, 12
    elif n <= 256:
        n_starts, lam, max_gen, stall_lim = 3, 12, 22, 10
    else:
        n_starts, lam, max_gen, stall_lim = 2, 8, 12, 6
    full_cov = (n - 1) <= 64
    starts = build_starts(fallback, n, rng, n_starts)
    gb_vec = fallback.copy()
    gb_score = exact_score(fallback, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gb_score - 1e-12:
            gb_score = s
            gb_vec = st.copy()
    for st in starts:
        gb_vec, gb_score = cem_run(st, n, lam, max_gen, stall_lim, full_cov, rng, gb_vec.copy(), gb_score)
    gb_vec, gb_score = polish(gb_vec, n, rng, 15)
    final = project_capped(gb_vec, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    pf = exact_score(fallback, n)
    if pf < fs - 1e-12:
        final, fs = fallback, pf
    return final.astype(np.float64), float(fs), int(n)
