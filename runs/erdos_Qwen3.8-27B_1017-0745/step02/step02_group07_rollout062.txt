import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(np.asarray(z, float).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0, 1)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,)) and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and \
           np.all(h <= 1 + 1e-12) and abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def get_Q(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    Q[0] = dx * np.dot(h, 1.0 - h)
    for r in range(1, n):
        m = n - r
        a, b = h[:m], h[r:]
        Q[2*r-1] = dx * np.dot(a, 1.0 - b)
        Q[2*r] = dx * np.dot(b, 1.0 - a)
    return Q

def Q_grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n - 1, n))
    G[0] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[2*r-1, :m] += dx * (1.0 - h[r:])
        G[2*r-1, r:] -= dx * h[:m]
        G[2*r, :m] -= dx * h[r:]
        G[2*r, r:] += dx * (1.0 - h[:m])
    return G

def _epi_obj(z, n):
    return z[n]

def _epi_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _epi_ineq(z, n):
    return z[n] - get_Q(z[:n], n)

def _epi_ineq_jac(z, n):
    J = np.zeros((2*n - 1, n + 1))
    J[:, :n] = -Q_grads(z[:n], n)
    J[:, n] = 1.0
    return J

def _epi_eq(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def _epi_eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def solve_epi(h0, n):
    t0 = exact_score(h0, n) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_jac, 'args': (n,)}
    ]
    mx = 300 if n <= 64 else 160

        res = minimize(_epi_obj, z0, method='SLSQP', jac=_epi_jac, args=(n,),

    if not np.all(np.isfinite(res.x)):

def smooth_fg(h, n, tau):
    Q = get_Q(h, n)
    qmax = Q.max()
    e = np.exp((Q - qmax) / tau)
    p = e / e.sum()
    F = qmax + tau * np.log(e.sum())
    dx = 2.0 / n
    g = p[0] * dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        g[:m] += dx * (p[2*r-1] * (1.0 - h[r:]) - p[2*r] * h[r:])
        g[r:] += dx * (-p[2*r-1] * h[:m] + p[2*r] * (1.0 - h[:m]))
    return g, F

def try_add(pool, v, th=1e-9):
    if v is None:
        return
    for s in pool:

    pool.append(v)
