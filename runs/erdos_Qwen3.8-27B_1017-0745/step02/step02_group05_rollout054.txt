        ts = exact_score(tr, n)
        if not np.isfinite(ts): continue
        d = ts - s
        if d < -1e-12 or (rng.random() < np.exp(-d / T) if d / T > -700 else False):
            h, s = tr, ts
        if s < bs: bh, bs = h.copy(), s
    return bh, bs

def pair_polish(h, n, rng, sweeps, npr):
    h = h.copy(); s = exact_score(h, n)
    if n <= 64:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    else:
        pairs = []
        for _ in range(npr):
            a, b = int(rng.integers(n)), int(rng.integers(n))
            if a < b: pairs.append((a, b))
            elif b < a: pairs.append((b, a))
    for _ in range(sweeps):
        imp = False
        for (i, j) in pairs:
            for f in (1.0/64, 1.0/16, 1.0/4, 0.5, 1.0):
                for (src, dst) in ((i, j), (j, i)):
                    cap = min(1.0 - h[dst], h[src])
                    if cap <= 1e-14: continue
                    tr = h.copy(); tr[src] -= f * cap; tr[dst] += f * cap
                    tr = project_capped(tr, n)
                    if tr is None: continue
                    ts = exact_score(tr, n)
                    if ts < s - 1e-12: h, s = tr, ts; imp = True
        if not imp: break
    return h, s

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if n is None: n = 47
    if n == 1:
        h1 = np.array([0.5]); return h1, exact_score(h1, 1), 1
    rng = np.random.default_rng(int(seed))
    fb = project_capped(parent, n) if parent is not None else None
    const = np.full(n, 0.5)
    safe_h, safe_s = const.copy(), exact_score(const, n)
    if fb is not None and is_feasible(fb, n):
        fs = exact_score(fb, n)
        if fs < safe_s: safe_h, safe_s = fb.copy(), fs
        base = fb
    else:
        base = const.copy()
    if n <= 64:
        P, G, L, nsw, npr = 12, 130, 2000, 2, 0
    elif n <= 256:
        P, G, L, nsw, npr = 8, 70, 600, 2, 200
    else:
        P, G, L, nsw, npr = 6, 25, 100, 1, 50
    pop = build_population(base, n, rng, P)
    scores = [exact_score(x, n) for x in pop]
    bh, bs = safe_h.copy(), safe_s
    for i in range(P):
        if scores[i] < bs: bh, bs = pop[i].copy(), scores[i]
    try:
        dh, ds = differential_evolution(pop, scores, n, rng, G, P)
        if ds < bs: bh, bs = dh.copy(), ds
    except Exception:
        pass
    try:
        ah, as_ = pair_anneal(bh, n, rng, L)
        if as_ < bs: bh, bs = ah.copy(), as_
    except Exception:
        pass
    try:
        ph, ps_ = pair_polish(bh, n, rng, nsw, npr)
        if ps_ < bs: bh, bs = ph.copy(), ps_
    except Exception:
        pass
    try:
        ah2, as2 = pair_anneal(bh, n, rng, L // 2)
        if as2 < bs: bh, bs = ah2.copy(), as2
    except Exception:
        pass
    try:
        ph2, ps2 = pair_polish(bh, n, rng, nsw, npr)
        if ps2 < bs: bh, bs = ph2.copy(), ps2
    except Exception:
        pass
    final = project_capped(bh, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs_ = exact_score(final, n)
    if not np.isfinite(fs_) or fs_ > safe_s + 1e-12:
        final, fs_ = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs_), int(n)
