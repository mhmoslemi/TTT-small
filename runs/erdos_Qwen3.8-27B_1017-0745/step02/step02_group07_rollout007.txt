def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 4):
        r = s - np.sum(h)
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    v = np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))
    return float(v) if np.isfinite(v) else float('inf')

def make_cosine_basis(n, p):
    idx = np.arange(n, dtype=np.float64)
    U = np.zeros((n, p), dtype=np.float64)
    for k in range(p):
        freq = k + 1
        U[:, k] = np.sqrt(2.0 / n) * np.cos(np.pi * (idx + 0.5) * freq / n)
        U[:, k] -= np.mean(U[:, k])
        nn = np.linalg.norm(U[:, k])
        if nn > 1e-14:
            U[:, k] /= nn
    return U

def add_to_pool(pool, v, max_size=16):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    if len(pool) < max_size:

def build_pool(fb, n, rng):
    pool = [fb.copy()]
    add_to_pool(pool, project_capped(fb[::-1], n))
    add_to_pool(pool, project_capped(1.0 - fb, n))
