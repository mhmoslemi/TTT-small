    except Exception:
        sd = 42
    sd = max(0, sd) % (2**31)
    rng = np.random.default_rng(sd)
    if n == 1:
        return np.array([0.5], dtype=np.float64), 1.0, 1
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = np.full(n, 0.5)
    else:
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    starts = []
    add_start(starts, fb, n)
    if parent is not None:
        add_start(starts, fb[::-1].copy(), n)
        add_start(starts, 1.0 - fb, n)
        add_start(starts, fb + rng.normal(0, 0.02, n), n)
        add_start(starts, fb + rng.normal(0, 0.08, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    n_rep = 6
    while len(starts) < n_rep:
        add_start(starts, fb + rng.normal(0, 0.1, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_rep]
    temps = np.array([0.0005, 0.002, 0.008, 0.032, 0.128, 0.512], dtype=np.float64)[:n_rep]
    scales = np.array([0.002, 0.006, 0.018, 0.054, 0.162, 0.486], dtype=np.float64)[:n_rep]
    reps, rep_scores = [], []
    for st in starts:
        h_int = 0.99 * st + 0.005
        h_int = project_capped(h_int, n)
        if h_int is None:
            h_int = st.copy()
        sc = exact_score(h_int, n)
        if not np.isfinite(sc):
            h_int = st.copy()
            sc = exact_score(st, n)
        reps.append(h_int)
        rep_scores.append(sc)
    best, best_score = fb.copy(), fb_score
    for i in range(n_rep):
        if rep_scores[i] < best_score - 1e-12:
            best, best_score = reps[i].copy(), rep_scores[i]
    B_def = min(16000, max(120, int(30000000 / (n * n))))
    try:
        B = int(kwargs.get('eval_budget', B_def))
    except Exception:
        B = B_def
    B = max(120, min(B, B_def, 20000))
    score_count = n_rep
    stagn_limit = max(30, min(200, int(B / (n_rep * 12))))
    stagn_count, restart_count = 0, 0
    max_sweeps = B + 1
    for sweep in range(max_sweeps):
        if score_count >= B:
            break
        prev_best = best_score
        for i in range(n_rep):
            if score_count >= B:
                break
            u = rng.random()
            if u < 0.35:
                hp = hit_and_run(reps[i], n, rng)
            elif u < 0.65:
                hp = pair_move(reps[i], n, scales[i], 0, rng)
            else:
                hp = pair_move(reps[i], n, scales[i], 1, rng)
            if hp is None:
                continue
            if not (np.all(hp >= -1e-10) and np.all(hp <= 1.0 + 1e-10)):
                continue
            if abs(float(np.sum(hp)) - n / 2.0) > 1e-8 * n:
                continue
            sc_new = exact_score(hp, n)
            score_count += 1
            if not np.isfinite(sc_new):
                continue
            T = temps[i]
            delta = sc_new - rep_scores[i]
            accept = False
            if delta <= 0:
                accept = True
            else:
                rv = rng.random()
                if rv > 0 and log(rv) < -delta / T:
                    accept = True
            if accept:
                reps[i] = hp
                rep_scores[i] = sc_new
                if sc_new < best_score - 1e-12:
                    best, best_score = hp.copy(), sc_new
        if n_rep >= 2:
            if sweep % 2 == 0:
                ex_pairs = [(0, 1), (2, 3), (4, 5)]
            else:
                ex_pairs = [(1, 2), (3, 4)]
            for (ai, bi) in ex_pairs:
                if bi >= n_rep:
                    continue
                Ta, Tb = temps[ai], temps[bi]
                Ca, Cb = rep_scores[ai], rep_scores[bi]
                lr = (1.0 / Ta - 1.0 / Tb) * (Ca - Cb)
                do_swap = False
                if lr >= 0:
                    do_swap = True
                else:
                    rv = rng.random()
                    if rv > 0 and log(rv) < lr:
                        do_swap = True
                if do_swap:
                    reps[ai], reps[bi] = reps[bi].copy(), reps[ai].copy()
                    rep_scores[ai], rep_scores[bi] = Cb, Ca
        if best_score < prev_best - 1e-12:
            stagn_count = 0
        else:
            stagn_count += 1
        if stagn_count >= stagn_limit and restart_count < 2:
            restart_count += 1
            if restart_count == 1:
                v = 0.5 * rng.uniform(0, 1, n) + 0.5 * best
            else:
                v = rng.beta(0.7, 0.7, n)
            vr = project_capped(v, n)
            if vr is not None:
                v_int = project_capped(0.99 * vr + 0.005, n)
                if v_int is None:
                    v_int = vr
                sc_r = exact_score(v_int, n)
                score_count += 1
                if np.isfinite(sc_r):
                    reps[n_rep - 1] = v_int
                    rep_scores[n_rep - 1] = sc_r
                    if sc_r < best_score - 1e-12:
                        best, best_score = v_int.copy(), sc_r
            stagn_count = 0
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fb.copy(), fb_score
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
