    z0[:n] = h0
    z0[n] = float(np.max(q0)) + 1e-10
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_grad},
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_grad},
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:n], n)
