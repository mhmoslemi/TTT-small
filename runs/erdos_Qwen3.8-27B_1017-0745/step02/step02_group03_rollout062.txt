import numpy as np
import math

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(z.min()) - 1.0, float(z.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if hi - lo < 1e-16:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(h.sum())
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - h
            tc = float(cap.sum())
            if tc > 1e-14:
                h += r * cap / tc
        else:
            th = float(h.sum())
            if th > 1e-14:
                h += r * h / th
    h = np.clip(h, 0.0, 1.0)
    r2 = s - float(h.sum())
    if abs(r2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if r2 > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + r2, 0.0, 1.0))
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - s) > 1e-8 * n:
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(float(h.sum()) - n / 2.0) > 1e-8 * n:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def lag_vals(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    om = 1.0 - h
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        v[d + n - 1] = float(np.dot(h[lo:hi], om[lo + d:hi + d]))
    return v

def surrogate_grad(h, n, tau):
    F = lag_vals(h, n)
    m = float(F.max())
    sh = np.clip((F - m) / tau, -700.0, 700.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return float('inf'), None
    p = e / s
    phi = (2.0 / n) * (m + tau * math.log(s))
    g = np.zeros(n)
    om = 1.0 - h
    for d in range(-(n - 1), n):
        w = p[d + n - 1]
        if w < 1e-18:
            continue
        lo, hi = max(0, -d), min(n, n - d)
        g[lo:hi] += w * om[lo + d:hi + d]
        g[lo + d:hi + d] -= w * h[lo:hi]
    g *= 2.0 / n
    return phi, g

def cont_opt(h0, n, taus, niter):
    h = h0.copy()
    for tau in taus:
        alpha = 0.3
        for _ in range(niter):
            phi, g = surrogate_grad(h, n, tau)
            if not np.isfinite(phi) or g is None:
                break
            ok = False
            for _bt in range(28):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h

                    ok = True
                    break

                if not np.isfinite(phi_y):

def pairwise_polish(h, n, sweeps):
    best = h.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = np.array([0.0, lo + 1e-10, hi - 1e-10])
                fracs = np.array([0.2, 0.4, 0.6, 0.8])
                cands = np.concatenate([cands, lo + fracs * (hi - lo)])
                bd, bs = 0.0, best_s
                for dd in cands:
                    dd = float(np.clip(dd, lo, hi))
                    t = best.copy()
                    t[i] += dd
                    t[j] -= dd
                    sc = exact_score(t, n)
                    if sc < bs - 1e-14:
                        bd, bs = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    best_s = bs
                    imp = True
        if not imp:
            break
    return best, best_s
