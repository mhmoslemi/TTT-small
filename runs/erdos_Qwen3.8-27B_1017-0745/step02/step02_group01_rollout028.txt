            rho = max(rho_min, 0.5 * rho)
            solver_fail += 1; no_improve += 1
            continue
        q, A = model
        u = lp_step(curr, n, rho, q, A)
        if u is None:
            rho = max(rho_min, 0.5 * rho)
            solver_fail += 1; no_improve += 1
            continue
        solver_fail = 0
        best_trial, best_trial_s = None, curr_s
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            z = curr + alpha * u
            proj = project_capped(z, n)
            if proj is None or not is_feasible(proj, n):
                continue
            s = exact_score(proj, n)
            if not np.isfinite(s):
                continue
            if s < best_trial_s - 1e-12:
                best_trial_s = s
                best_trial = proj.copy()
            if s < best_score - 1e-12:
                best_score, best_h = s, proj.copy()
        if best_trial is None:
            rho = max(rho_min, 0.5 * rho)
            no_improve += 1
            continue
        delta = best_trial - curr
        actual = curr_s - best_trial_s
        Grad = A[:, :n]
        model_val = np.max(q + Grad @ delta)
        predicted = curr_s - model_val
        curr, curr_s = best_trial, best_trial_s
        no_improve = 0
        pred_ok = predicted > 1e-12 and np.isfinite(predicted) and np.isfinite(actual)
        if pred_ok:
            ratio = actual / predicted
            if ratio < 0.1:
                rho = max(rho_min, 0.5 * rho)
            elif ratio > 0.75 and np.max(np.abs(delta)) >= 0.8 * rho:
                rho = min(1.5 * rho, rho_max)
        else:
            rho = max(rho_min, 0.5 * rho)
    return best_h, best_score

def add_start(pool, z, n, limit):
    if z is None:
        return
    for s in pool:
        if np.max(np.abs(z - s)) < 1e-8:
            return
    if len(pool) < limit:
        pool.append(z.copy())

def build_starts(fallback, n, rng, limit):
    pool = []
    add_start(pool, project_capped(fallback.copy(), n), n, limit)
    add_start(pool, project_capped(fallback[::-1].copy(), n), n, limit)
    add_start(pool, project_capped((1.0 - fallback[::-1]).copy(), n), n, limit)
    add_start(pool, np.full(n, 0.5), n, limit)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_start(pool, project_capped(blk, n), n, limit)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_start(pool, project_capped(alt, n), n, limit)
    for sd in (0.05, 0.15):
        add_start(pool, project_capped(fallback + rng.normal(0, sd, n), n), n, limit)
    t = np.arange(n)
    cos_v = 0.5 + 0.3 * np.cos(2.0 * math.pi * t / n)
    add_start(pool, project_capped(cos_v, n), n, limit)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    if n < 1:
        n = 47
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        s = exact_score(fallback, n)
        return fallback, s, 1
    fb_score = exact_score(fallback, n)
    best_h, best_score = fallback.copy(), fb_score
    if n > 999:
        return fallback, fb_score, n
    starts = build_starts(fallback, n, rng, 8)
    scored = []
    for i in range(len(starts)):
        s = exact_score(starts[i], n)
        scored.append((s, i, starts[i]))
    scored.sort(key=lambda x: (x[0], x[1]))
    n_chains = max(1, min(len(starts), min(4, round(4.0 * math.sqrt(47.0 / n)))))
    selected = []
    for s, i, st in scored:
        if i == 0:
            if not any(np.allclose(x, st, atol=1e-8) for x in selected):
                selected.append(st)
        elif len(selected) < n_chains:
            selected.append(st)
    if not any(np.allclose(x, fallback, atol=1e-8) for x in selected):
        selected.append(fallback)
    iters = max(6, min(30, round(30.0 * math.sqrt(47.0 / n))))
    for st in selected[:n_chains]:
        try:
            best_h, best_score = refine_chain(st, n, iters, 0.08, best_h, best_score)
        except Exception:
            pass
    try:
        best_h, best_score = refine_chain(best_h, n, max(5, iters // 2), 0.03, best_h, best_score)
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
