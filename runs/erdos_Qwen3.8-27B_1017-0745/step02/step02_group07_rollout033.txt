def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)

        return None

    lo = np.min(z) - 1.0
    hi = np.max(z)

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

                    break

                    break

                h[i] -= sub
                r += sub

        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n))
    p = rng.permutation(n)
    vb = np.zeros(n)
    vb[p[:n // 2]] = 1.0
    if n % 2:
        vb[p[n // 2]] = 0.5
    try_add(pool, project_capped(vb, n))
    try_add(pool, project_capped(fb + rng.normal(0, 0.06, n), n))
    try_add(pool, project_capped(fb + rng.normal(0, 0.18, n), n))
    try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    for k in (1, 2, 3):
        for ph in (0.0, math.pi / 2):
            cs = 0.5 + 0.35 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph)
            try_add(pool, project_capped(cs, n))
    while len(pool) < 16:
        pool.append(fb.copy())
    return pool[:16]

def build_qp(n, dx):
    import cvxpy as cp
    L = 2.0 * dx
    d_v = cp.Variable(n)
    u_v = cp.Variable()
    q_p = cp.Parameter(2 * n - 1)
    G_p = cp.Parameter((2 * n - 1, n))
    h_p = cp.Parameter(n)
    obj = cp.Minimize(u_v + (L / 2.0) * cp.sum_squares(d_v))
    cons = [
        G_p @ d_v + q_p <= u_v,
        cp.sum(d_v) == 0,
        d_v >= -h_p,
        d_v <= 1.0 - h_p
    ]
    prob = cp.Problem(obj, cons)
    return prob, d_v, u_v, q_p, G_p, h_p

def pick_solvers():
    try:
        import cvxpy as cp
        avail = cp.installed_solvers
    except Exception:
        return []
    order = ['ECOS', 'MOSEK', 'GUROBI', 'CVXOPT']
    return [s for s in order if s in avail]

def solve_qp_iter(prob, d_v, u_v, q_p, G_p, h_p, q, G, h_val, n, solvers):
    q_p.value = q
    G_p.value = G
    h_p.value = h_val
    d_v.value = np.zeros(n)
    u_v.value = np.max(q)
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate', 'optimal_unknown'):
                if d_v.value is not None:
                    dv = np.array(d_v.value, dtype=np.float64).ravel()
                    if dv.shape == (n,) and np.all(np.isfinite(dv)):
                        return dv
        except Exception:
            continue
    return None

def mm_optimize(h0, n, max_iter, qp_data, solvers):
    if qp_data is None:
        return h0.copy(), exact_score(h0, n)
    prob, d_v, u_v, q_p, G_p, h_p = qp_data
    dx = 2.0 / n
    h = h0.copy()
    s = exact_score(h, n)
    alphas = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125)
    for it in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        dv = solve_qp_iter(prob, d_v, u_v, q_p, G_p, h_p, q, G, h, n, solvers)
        if dv is None:
            break
        if np.max(np.abs(dv)) <= 1e-8:
            break
        best_score = s
        best_h = h.copy()
        for alpha in alphas:
            cand = project_capped(h + alpha * dv, n)
            if cand is None:
                continue
            if np.max(np.abs(cand - h)) < 1e-10:
                continue
            cs = exact_score(cand, n)
            if cs < best_score - 1e-12:
                best_score = cs
                best_h = cand
        if best_score < s - 1e-12:
            h = best_h
            s = best_score
        else:
            break
    return h, s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if n <= 64:
        n_starts, max_iter = 5, 50
    elif n <= 128:
        n_starts, max_iter = 4, 32
    elif n <= 256:
        n_starts, max_iter = 3, 20
    elif n <= 512:
        n_starts, max_iter = 2, 12
    else:
        n_starts, max_iter = 1, 8
    dx = 2.0 / n
    qp_data = None
    solvers = pick_solvers()
    if solvers:
        try:
            qp_data = build_qp(n, dx)
        except Exception:
            qp_data = None
            solvers = []
    pool = build_pool(fb, n, rng)
    scores = np.array([exact_score(pv, n) for pv in pool])
    global_best = fb.copy()
    global_score = exact_score(fb, n)
    order = np.argsort(scores)
    selected = [pool[0].copy()]
    for idx in order:
        if len(selected) >= n_starts:
            break
        pv = pool[idx]
        if all(np.max(np.abs(pv - sv)) > 0.05 for sv in selected):
            selected.append(pv.copy())
    for idx in order:
        if len(selected) >= n_starts:
            break
        selected.append(pool[idx].copy())
    for start in selected:
        try:
            h_res, s_res = mm_optimize(start, n, max_iter, qp_data, solvers)
        except Exception:
            h_res, s_res = start, exact_score(start, n)
        if h_res is not None and is_feasible(h_res, n):
            s_chk = exact_score(h_res, n)
            if s_chk < global_score - 1e-12:
                global_score = s_chk
                global_best = h_res.copy()
    for idx in order[:3]:
        pv = pool[idx]
        ps = exact_score(pv, n)
        if ps < global_score - 1e-12:
            global_score = ps
            global_best = pv.copy()
    final = project_capped(global_best, n)
    if final is None or not is_feasible(final, n):
        final = project_capped(fb, n)
        if final is None or not is_feasible(final, n):
            final = np.full(n, 0.5)
    final_score = exact_score(final, n)
    fb_score = exact_score(fb, n)
    if not np.isfinite(final_score) or final_score > fb_score - 1e-12:
        final = fb.copy()
        final_score = fb_score
    final = project_capped(final, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
        final_score = exact_score(final, n)
    return final.astype(np.float64), float(final_score), int(n)
