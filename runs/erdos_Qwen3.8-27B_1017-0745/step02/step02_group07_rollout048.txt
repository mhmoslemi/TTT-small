```
def project_capped(z,n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(5):
        if abs(r) < 1e-12:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-14: break
                add = min(r, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
                    r -= add
        else:
            nr = -r
            for i in range(n):
                if nr <= 1e-14: break
                sub = min(nr, h[i])
                if sub > 0.0:
                    h[i] -= sub
                    nr -= sub
            r = s - float(np.sum(h))
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - s) > 1e-7:
        return None
    return h
```

```
def is_feasible(h,n):
    if h.size != n or not np.all(np.isfinite(h)):
        return False
    tol = max(1e-8, 1e-10*n)
    if float(np.sum(h)) < n/2.0 - tol or float(np.sum(h)) > n/2.0 + tol:
        return False
    return bool(np.all(h >= -1e-10) and np.all(h <= 1.0+1e-10))
```

```
def exact_score(h,n):
    if not is_feasible(h,n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    if not np.isfinite(val):
        return float('inf')
    return val
```

```
def lag_raw_values(h,n):
    vals = np.empty(2*n-1, dtype=np.float64)
    vals[0] = float(np.dot(h, 1.0 - h))
    for r in range(1, n):
        L = n - r
        vals[r] = float(np.dot(h[:L], 1.0 - h[r:]))
        vals[n + r - 1] = float(np.dot(h[r:], 1.0 - h[:L]))
    return vals
```

```
        a = float(w[r])
        if abs(a) > 1e-14:
```

```
        if n % d == 0:
```

```python
import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    zc = np.clip(z, 0.0, 1.0)
    if (abs(float(np.sum(zc)) - s) <= 1e-9 and
            float(np.max(z)) <= 1.0 + 1e-9 and
            float(np.min(z)) >= -1e-9):
        return zc
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(6):
        if abs(r) < 1e-12:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-14:
                    break
                add = min(r, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
                    r -= add
        else:
            nr = -r
            for i in range(n):
                if nr <= 1e-14:
                    break
                sub = min(nr, h[i])
                if sub > 0.0:
                    h[i] -= sub
                    nr -= sub
        r = s - float(np.sum(h))
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - s) > 1e-7:
        return None
    return h


def is_feasible(h, n):
    if h.size != n or not np.all(np.isfinite(h)):
        return False
    tol = max(1e-8, 1e-10 * n)
    sm = float(np.sum(h))
    if sm < n / 2.0 - tol or sm > n / 2.0 + tol:
        return False
    return bool(np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10))


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    if not np.isfinite(val):
        return float('inf')
    return val


def lag_raw_values(h, n):
    vals = np.empty(2 * n - 1, dtype=np.float64)
    vals[0] = float(np.dot(h, 1.0 - h))
    for r in range(1, n):
        L = n - r
        vals[r] = float(np.dot(h[:L], 1.0 - h[r:]))
        vals[n + r - 1] = float(np.dot(h[r:], 1.0 - h[:L]))
    return vals


def smooth_grad(h, n, tau):
    vals = lag_raw_values(h, n)
    mx = float(np.max(vals))
    if tau <= 1e-12:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[vals == mx] = 1.0
        wsum = 1.0
    else:
        w = np.exp((vals - mx) / tau)
        wsum = float(np.sum(w))
        if not np.isfinite(wsum) or wsum <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[vals == mx] = 1.0
            wsum = 1.0
        else:
            w = w / wsum
    g = np.zeros(n, dtype=np.float64)
    wp = float(w[0])
    if abs(wp) > 1e-14:
        g += wp * (1.0 - 2.0 * h)
    for r in range(1, n):
        L = n - r
        a = float(w[r])
        if abs(a) > 1e-14:
            g[:L] += a * (1.0 - h[r:])
            g[r:] -= a * h[:L]
        b = float(w[n + r - 1])
        if abs(b) > 1e-14:
            g[r:] += b * (1.0 - h[:L])
            g[:L] -= b * h[r:]
    return float((2.0 / n) * mx), g


def subgrad_polish(h, n, rng, iters):
    h = np.array(h, dtype=np.float64).copy()
    bs = exact_score(h, n)
    if not np.isfinite(bs):
        return h, bs
    step = 0.05
    fails = 0
    iters = int(max(iters, 1))
    for it in range(iters):
        tau = 0.12 + 0.9 * math.sqrt((it + 1) / float(iters))
        ms, g = smooth_grad(h, n, tau)
        if not np.isfinite(ms):
            break
        if ms < bs - 1e-12:
            bs = ms
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        d = -g / gn
        improved = False
        for t in (step, 0.5 * step, 0.25 * step, 0.125 * step):
            d2 = d + rng.normal(0.0, 0.02, n)

                improved = True

        if not improved:

    return h, bs


def balanced_binary(n, rng):
    v = np.zeros(n, dtype=np.float64)
    p = rng.permutation(n)
    v[p[:n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v


def is_prime(n):
    if n < 2:
        return False
    d = 2
    while d * d <= n:
        if n % d == 0:
            return False
        d += 1
    return True


def legendre_h(n):
    if n < 3 or not is_prime(n):
        return None
    s = np.zeros(n, dtype=np.float64)
    e = (n - 1) // 2
    for x in range(1, n):
        s[x] = 1.0 if pow(x, e, n) == 1 else -1.0
    h = 0.5 * (1.0 + s)
    return project_capped(h, n)


def add_candidate(pool, z, n, thr=1e-5):
    h = project_capped(z, n)
    if h is None:
        return
    for s in pool:
        if np.max(np.abs(h - s)) < thr:
            return
    pool.append(h)


def build_pool(fb, n, rng, maxsize):
    pool = []
    add_candidate(pool, fb, n)
    add_candidate(pool, fb[::-1], n)
    add_candidate(pool, 1.0 - fb, n)
    add_candidate(pool, (1.0 - fb)[::-1], n)
    add_candidate(pool, np.roll(fb, n // 3), n)
    add_candidate(pool, np.roll(fb, n // 2), n)
    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_candidate(pool, blk, n)
    add_candidate(pool, balanced_binary(n, rng), n)
    add_candidate(pool, rng.uniform(0.0, 1.0, n), n)
    add_candidate(pool, fb + rng.normal(0.0, 0.08, n), n)
    add_candidate(pool, fb + rng.normal(0.0, 0.20, n), n)
    leg = legendre_h(n)
    if leg is not None:
        add_candidate(pool, leg, n)
        add_candidate(pool, 1.0 - leg, n)
        add_candidate(pool, leg[::-1], n)
    attempts = 0
    while len(pool) < maxsize and attempts < 20 * maxsize:
        attempts += 1
        r = rng.random()
        if pool and r < 0.60:
            base = pool[int(rng.integers(len(pool)))]
            sig = float(rng.choice((0.05, 0.15, 0.30)))
            add_candidate(pool, base + rng.normal(0.0, sig, n), n)
        elif r < 0.80:
            add_candidate(pool, rng.uniform(0.0, 1.0, n), n)
        else:
            add_candidate(pool, balanced_binary(n, rng), n)
    if not pool:
        pool.append(project_capped(np.full(n, 0.5), n))
    while len(pool) < maxsize:
        pool.append(pool[0].copy())
    return pool


def pair_sweep(h, n, rng, rounds):
    h = np.array(h, dtype=np.float64).copy()
    bs = exact_score(h, n)
    if not np.isfinite(bs):
        return h, bs
    ts = (0.20, 0.10, 0.05, 0.025, 0.012)

        for _ in range(rounds * 1200):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            t = ts[int(rng.integers(len(ts)))]
            vi = h[i] + sgn * t

            if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                continue
            z = h.copy()
            z[i] = vi
            z[j] = vj
            sc = exact_score(z, n)
            if sc < bs - 1e-12:
                h = z
                bs = sc
        return h, bs
    for rd in range(rounds):
        improved = False
        order_i = rng.permutation(n) if rd > 0 else np.arange(n)
        for ii in order_i:
            i = int(ii)
            for j in range(i + 1, n):
                for sgn in (1.0, -1.0):
                    for t in ts:
                        vi = h[i] + sgn * t
                        vj = h[j] - sgn * t
                        if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                            continue
                        z = h.copy()
                        z[i] = vi
                        z[j] = vj
                        sc = exact_score(z, n)
                        if sc < bs - 1e-12:
                            h = z
                            bs = sc
                            improved = True
        if not improved:
            break
    return h, bs


def cosine_basis(n):
    p = min(n - 1, 48)
    if p <= 0:
        return np.empty((n, 0), dtype=np.float64)
    i = np.arange(n, dtype=np.float64)[:, None]
    k = np.arange(1, p + 1, dtype=np.float64)[None, :]
    return np.sqrt(2.0 / n) * np.cos(np.pi * (i + 0.5) * k / n)


def cma_run(base, U, n, rng, best_h, best_s, gens, sigma0):
    p = U.shape[1]
    if p <= 0:
        return best_h, best_s
    lam = max(6, int(4 + 3 * math.log(p))) if p > 1 else 6
    mu = max(1, lam // 2)
    j = np.arange(1, mu + 1, dtype=np.float64)
    w = np.log(mu + 0.5) - np.log(j)
    w = w / float(np.sum(w))
    meff = 1.0 / float(np.sum(w * w))
    cs = (meff + 2.0) / (p + meff + 5.0)
    ds = 1.0 + 2.0 * max(0.0, math.sqrt(max(0.0, (meff - 1.0) / (p + 1.0))) - 1.0) + cs
    cc = (4.0 + meff / p) / (p + 4.0 + 2.0 * meff / p)
    c1 = 2.0 / ((p + 1.3) * (p + 1.3) + meff)
    cmu = max(0.0, min(1.0 - c1,
                       2.0 * (meff - 2.0 + 1.0 / meff) / ((p + 2.0) * (p + 2.0) + meff)))
    chis = math.sqrt(p) * (1.0 - 1.0 / (4.0 * p) + 1.0 / (21.0 * p * p))
    m = np.zeros(p, dtype=np.float64)
    C = np.eye(p, dtype=np.float64)
    ps = np.zeros(p, dtype=np.float64)
    pc = np.zeros(p, dtype=np.float64)

            vals, V = np.linalg.eigh(Cs)

            vals = np.ones(p, dtype=np.float64)
            V = np.eye(p, dtype=np.float64)

            ps = np.zeros(p, dtype=np.float64)

            V = np.eye(p, dtype=np.float64)

        ys = []
        hs = []

        if len(fs) < mu:

        order = np.argsort(fs)

        scores = np.empty(len(pool), dtype=np.float64)
        for idx in range(len(pool)):
            scores[idx] = exact_score(pool[idx], n)
        bi = int(np.argmin(scores))
        if scores[bi] < best_s - 1e-12:
            best_h = pool[bi].copy()
            best_s = float(scores[bi])
            if n <= 200:
                best_h, best_s = pair_sweep(best_h, n, rng, 1)
            best_h, best_s = subgrad_polish(best_h, n, rng, 30 if n <= 128 else 15)

        if n <= 512:
            U = cosine_basis(n)
            if n <= 64:
                nr, gens, sgensi = 5, 110, 0.09
            elif n <= 128:
                nr, gens, sgensi = 3, 70, 0.08
            elif n <= 256:
                nr, gens, sgensi = 2, 35, 0.07
            else:
                nr, gens, sgensi = 1, 15, 0.05
            bases = select_bases(pool, scores, nr)
            if not bases:
                bases = [best_h.copy()]
            dup = False
            for b in bases:
                if np.max(np.abs(best_h - b)) < 1e-6:
                    dup = True
                    break
            if not dup:
                bases.append(best_h.copy())
            try:
                for base in bases:
                    tmp_h, tmp_s = cma_run(base, U, n, rng, best_h, best_s, gens, sgensi)
                    best_h, best_s = tmp_h, tmp_s
            except Exception:
                pass
            best_h, best_s = subgrad_polish(best_h, n, rng, 100 if n <= 128 else 30)
            if n <= 200:
                best_h, best_s = pair_sweep(best_h, n, rng, 2)
            best_h, best_s = subgrad_polish(best_h, n, rng, 30)
        else:
            best_h, best_s = subgrad_polish(best_h, n, rng, 20)
    except Exception:
        best_h = orig_fb.copy()
        best_s = float(orig_s)

    cand = project_capped(best_h, n)
    if cand is None:
        cand = orig_fb
    cand_s = exact_score(cand, n)
    if (not np.isfinite(cand_s)) or (np.isfinite(orig_s) and cand_s > orig_s + 1e-9):
        cand = orig_fb
        cand_s = exact_score(orig_fb, n)
    return np.asarray(cand, dtype=np.float64), float(cand_s), int(n)
```
