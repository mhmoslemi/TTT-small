def local_stag(h, n, rng, max_iters=40, target=0.3808):
    hx = np.array(h, dtype=np.float64).copy()

    alphas = np.array([0.003,0.007,0.016,0.035,0.075,0.16,0.35])

        if not (np.all(np.isfinite(q)) and np.all(np.isfinite(G))): break
        C = float(np.max(q))
        if C < target - 1e-8: break
        dirs = []
        order = np.argsort(q)[::-1]
        # active averages
        for tol in tol_levels:
            act = np.where(q >= C - tol)[0]
            if act.size:
                d = -G[act].mean(axis=0)
                dirs.append(d)
        # top singles
        for j in order[:5]:
            dirs.append(-G[j].copy())
        # random subsets
        top = order[:12]
        for _ in range(4):
            k = min(int(rng.integers(2, len(top)+1)), len(top))
            sel = rng.choice(top, k, replace=False)
            dirs.append(-G[sel].mean(axis=0))
        # dedupe directions
        used = []
        best_h = None; best_s = sx
        for d in dirs:
            mx = float(np.max(np.abs(d)))
            if mx < 1e-14 or not np.isfinite(mx): continue
            d = d / mx
            dup=False
            for u in used:
                if np.max(np.abs(d-u)) < 0.05:
                    dup=True; break
            if dup: continue
            used.append(d)
            # line search: maybe try alphas descending to accept first improvement? Could try all and pick best.

import numpy as np
import math
try:
    import cvxpy as cp

ALPHAS = (0.004, 0.01, 0.025, 0.06, 0.15, 0.35, 0.75, 1.25)

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1.0e6, 1.0e6)
    s = n * 0.5
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    if not (lo < hi):
        hi = lo + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = s - float(np.sum(h))
        if abs(r) < 1.0e-13:
            break
        if r > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if r > 0:
                d = min(r, max(0.0, 1.0 - h[idx]))
                if d <= 1.0e-16:
                    continue
                h[idx] += d
                r -= d
            else:
                d = min(-r, max(0.0, h[idx]))
                if d <= 1.0e-16:
                    continue
                h[idx] -= d
                r += d
            if abs(r) < 1.0e-13:
                break
    if (not np.all(np.isfinite(h)) or np.any(h < -1.0e-11) or
            np.any(h > 1.0 + 1.0e-11) or abs(float(np.sum(h)) - s) > 1.0e-9):
        return None
    return h


def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1.0e-10) or np.any(h > 1.0 + 1.0e-10):
        return False
    return abs(float(np.sum(h)) - n * 0.5) <= 1.0e-8


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    v = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    out = np.max(np.correlate(v, 1.0 - v, mode='full') * (2.0 / n))
    if np.isfinite(out):
        return float(out)
    return float('inf')


def lag_data(h, n):
    h = np.asarray(h, dtype=np.float64)
    dx = 2.0 / n
    q = np.zeros(2 * n - 1, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        h0 = h[:m]
        h1 = h[r:]
        q[idx] = dx * float(np.sum(h0 * (1.0 - h1)))
        q[idx + 1] = dx * float(np.sum(h1 * (1.0 - h0)))
        G[idx, :m] += dx * (1.0 - h1)
        G[idx, r:] -= dx * h0
        G[idx + 1, :m] -= dx * h1
        G[idx + 1, r:] += dx * (1.0 - h0)
        idx += 2
    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(G))):
        return None, None
    return q, G


def vec_from_order(n, order):
    order = np.asarray(order)
    v = np.zeros(n, dtype=np.float64)
    half = n // 2
    v[order[:half]] = 1.0
    if n % 2:
        v[order[half]] = 0.5
    return v


def add_pool(pool, n, v, th=1.0e-7):
    if v is None:
        return
    v = np.asarray(v, dtype=np.float64).ravel()
    if not is_feasible(v, n):
        return
    for w in pool:
        if np.max(np.abs(v - w)) < th:
            return
    pool.append(v)


def make_pool(fb, n, rng, cap):
    pool = []
    add_pool(pool, n, fb.copy())

    for sh in (n // 3, n // 2, n // 4, (3 * n) // 4):
        if 0 < sh < n:
            add_pool(pool, n, project_capped(np.roll(fb, sh), n), 1.0e-6)

    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_pool(pool, n, blk, 1.0e-6)

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_pool(pool, n, project_capped(alt, n), 1.0e-6)

    alphas = (0.113, 0.161, 0.207, 0.269, 0.333, 0.414, 0.508, 0.618)
    phases = (0.0, 0.137, 0.326, 0.571, 0.832)
    for a in alphas:
        for ph in phases:
            frac = (np.arange(n) * a + ph) % 1.0
            add_pool(pool, n, vec_from_order(n, np.argsort(frac)), 1.0e-6)
        if len(pool) >= cap:
            break

    for f in (1, 2, 3, 4):
        x = 2.0 * np.arange(n) / n
        for ph in (0.0, 0.35, 0.75):
            v = 0.5 + 0.95 * np.cos(2.0 * math.pi * f * x + math.pi * ph)
            add_pool(pool, n, project_capped(v, n), 1.0e-6)
        if len(pool) >= cap:
            break

    add_pool(pool, n, vec_from_order(n, rng.permutation(n)))
    add_pool(pool, n, project_capped(rng.uniform(0.0, 1.0, n), n))

    sc = max(0.5, min(1.5, math.sqrt(47.0 / max(1, n))))
    for sd in (0.04, 0.10, 0.22):
        add_pool(pool, n, project_capped(fb + rng.normal(0.0, sd, n), n), 1.0e-6)

    same = 0
    while len(pool) < cap and same < 5 * cap:
        r = rng.random()
        if len(pool) == 0:
            v = np.full(n, 0.5)
        elif r < 0.55:
            base = pool[int(rng.integers(len(pool)))]
            v = project_capped(base + rng.normal(0.0, 0.15 * sc, n), n)
        elif r < 0.8:
            v = project_capped(rng.uniform(0.0, 1.0, n), n)
        else:
            v = vec_from_order(n, rng.permutation(n))
        before = len(pool)
        add_pool(pool, n, v, 1.0e-7)
        same = 0 if len(pool) > before else same + 1

    if not pool:
        pool.append(np.full(n, 0.5, dtype=np.float64))
    return pool


def select_starts(pool, scores, base, max_count):

    n = base.size
    order = np.argsort(scores)
    for idx in order:

            v = pool[idx]
            ok = True
            for w in sel:

def accept_dir(hx, sx, d, n):
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size != n or not np.all(np.isfinite(d)):
        return None, None
    mx = float(np.max(np.abs(d)))
    if mx < 1.0e-13:
        return None, None
    d = d / mx
    besth = None
    best = sx
    for a in ALPHAS:
        cand = project_capped(hx + a * d, n)
        if cand is None:
            continue
        if np.max(np.abs(cand - hx)) < 1.0e-10:
            continue
        s = exact_score(cand, n)
        if np.isfinite(s) and s < best - 1.0e-12:
            best = s
            besth = cand
    return besth, best


def local_stag(h, n, rng, max_iters, target):
    hx = project_capped(h, n)
    if hx is None:
        return np.full(n, 0.5, dtype=np.float64), float('inf')
    sx = exact_score(hx, n)

    for _ in range(max_iters):
        q, G = lag_data(hx, n)
        if q is None:
            break
        C = float(np.max(q))
        if C <= target:
            break

        dirs = []
        order = np.argsort(q)[::-1]

        for beta in (3.0, 12.0, 45.0):
            w = np.exp(beta * (q - C))
            tw = float(np.sum(w))
            if tw > 0.0:
                w = w / tw
                dirs.append(-(G.T @ w))

        for tol in (2.0e-5, 1.0e-4):
            act = np.where(q >= C - tol)[0]
            if act.size:
                dirs.append(-G[act].mean(axis=0))

        for j in order[:3]:
            dirs.append(-G[j].copy())

        top = order[:12]
        if len(top) >= 2:
            for _ in range(2):
                k = int(rng.integers(2, min(5, len(top)) + 1))
                sel = rng.choice(top, k, replace=False)
                dirs.append(-G[sel].mean(axis=0))

        used = []
        besth = None
        best_s = sx

        for d in dirs:
            if d is None:
                continue
            d = np.asarray(d, dtype=np.float64).ravel()
            mx = float(np.max(np.abs(d)))
            if mx < 1.0e-14:
                continue
            d = d / mx
            dup = False
            for u in used:
                if np.max(np.abs(d - u)) < 0.12:
                    dup = True
                    break
            if dup:
                continue
            used.append(d)
            nh, ns = accept_dir(hx, sx, d, n)
            if nh is not None and ns < best_s - 1.0e-12:
                best_s = ns
                besth = nh

        if besth is None:
            break
        hx, sx = besth, best_s

    return hx, sx


def build_mm(n):
    if cp is None or n < 2:
        return None
    nl = 2 * n - 1
    L = 4.0 / n
    d = cp.Variable(n)
    u = cp.Variable()
    qp = cp.Parameter(nl)
    Gp = cp.Parameter((nl, n))
    hp = cp.Parameter(n)
    cons = [
        Gp @ d + qp <= u * cp.ones(nl),
        cp.sum(d) == 0,
        d >= -hp,
        d <= 1.0 - hp,
    ]
    obj = cp.Minimize(u + 0.5 * L * cp.sum_squares(d))
    try:
        prob = cp.Problem(obj, cons)
        return (prob, d, u, qp, Gp, hp)
    except Exception:
        return None


def qp_solve(data, h, n, solver, active_tol):
    if data is None:
        return None
    h = np.asarray(h, dtype=np.float64)
    q, G = lag_data(h, n)
    if q is None:
        return None
    C = float(np.max(q))
    mask = q >= (C - active_tol)
    if not np.any(mask):
        mask = np.zeros(q.shape, dtype=bool)
        mask[int(np.argmax(q))] = True
    qu = np.full(q.shape, -10.0, dtype=np.float64)
    qu[mask] = q[mask]
    Gu = np.zeros_like(G)
    Gu[mask] = G[mask]

    prob, dvar, uvar, qp, Gp, hp = data
    try:
        qp.value = qu
        Gp.value = Gu
        hp.value = h
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            val = dvar.value
            if val is not None:
                val = np.asarray(val, dtype=np.float64).ravel()
                if val.size == n and np.all(np.isfinite(val)):
                    return val
    except Exception:
        pass
    return None


def get_solver(data, h, n):
    if data is None or cp is None:
        return None, None
    for nm in ('ECOS', 'MOSEK', 'GUROBI', 'CVXOPT', 'SCS', 'SLSQP'):
        s = getattr(cp, nm, None)
        if s is None:
            continue
        try:
            dv = qp_solve(data, h, n, s, 1.0e-4)
            if dv is not None:
                return s, dv
        except Exception:
            pass
    return None, None


def mm_search(data, starts, n, solver, iters_per_tol, target):
    besth = None
    bests = float('inf')
    for h0 in starts:
        hx = project_capped(h0, n)
        if hx is None:
            continue
        sx = exact_score(hx, n)
        if sx < bests - 1.0e-12:
            bests = sx
            besth = hx
        if sx <= target:
            continue
        for tol in (4.0e-4, 1.2e-4, 4.0e-5):
            for _ in range(iters_per_tol):
                if sx <= target:
                    break
                dv = qp_solve(data, hx, n, solver, tol)
                if dv is None:
                    break
                before = sx
                nh, ns = accept_dir(hx, sx, dv, n)
                if nh is None:
                    break
                hx, sx = nh, ns
                if sx < bests - 1.0e-12:
                    bests = sx
                    besth = hx
                if before - sx < 1.0e-10:
                    break
    return besth, bests


def finalize(best, fb, n):
    b = project_capped(best, n)
    s = exact_score(b, n)
    f = project_capped(fb, n)
    fs = exact_score(f, n)
    if f is None:
        f = np.full(n, 0.5, dtype=np.float64)
        fs = exact_score(f, n)
    if b is not None and np.isfinite(s) and (not np.isfinite(fs) or s <= fs + 1.0e-12):
        return b, s, n
    if f is not None and np.isfinite(fs):
        return f, fs, n
    u = np.full(n, 0.5, dtype=np.float64)
    return u, exact_score(u, n), n


def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1.0e-8) and np.all(p <= 1.0 + 1.0e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass

    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1

    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5, dtype=np.float64)
    fb = project_capped(fb, n)
    if fb is None:
        fb = np.full(n, 0.5, dtype=np.float64)

    fb_s = exact_score(fb, n)
    if not np.isfinite(fb_s):
        fb = np.full(n, 0.5, dtype=np.float64)
        fb_s = exact_score(fb, n)

    best_h = fb.copy()
    best_s = fb_s

    if n <= 64:
        cap, max_top, max_iters = 18, 6, 18
        qp_starts, iters_per_tol = 2, 2
    elif n <= 128:
        cap, max_top, max_iters = 14, 5, 14
        qp_starts, iters_per_tol = 2, 2
    elif n <= 256:
        cap, max_top, max_iters = 12, 3, 10
        qp_starts, iters_per_tol = 1, 2
    else:
        cap, max_top, max_iters = 10, 2, 7
        qp_starts, iters_per_tol = 0, 0

    pool = make_pool(fb, n, rng, cap)
    scores = np.empty(len(pool), dtype=np.float64)
    for i, v in enumerate(pool):
        scores[i] = exact_score(v, n)
    for i, v in enumerate(pool):
        if scores[i] < best_s - 1.0e-12:
            best_h = v.copy()
            best_s = scores[i]

    starts = select_starts(pool, scores, fb, max_top)
    top_list = [best_h.copy()]
    for v in starts:
        if len(top_list) >= max_top:
            break
        if np.max(np.abs(v - best_h)) > 0.04:
            top_list.append(v.copy())
    j = 0
    while len(top_list) < max_top and j < len(starts):
        v = starts[j]
        j += 1
        dup = False
        for w in top_list:
            if np.max(np.abs(v - w)) < 1.0e-8:
                dup = True
                break
        if not dup:
            top_list.append(v.copy())

    for v in top_list:
        if best_s <= TARGET:
            break
        nv, ns = local_stag(v, n, rng, max_iters, TARGET)
        if ns < best_s - 1.0e-12:
            best_h, best_s = nv, ns

    if qp_starts > 0 and best_s > TARGET:
        data = build_mm(n)
        if data is not None:
            solver, test_dv = get_solver(data, best_h, n)
            if solver is not None:
                if test_dv is not None:
                    nh, ns = accept_dir(best_h, best_s, test_dv, n)
                    if nh is not None and ns < best_s - 1.0e-12:
                        best_h, best_s = nh, ns
                q_starts = [best_h.copy()]
                for v in starts:
                    if np.max(np.abs(v - best_h)) > 0.06:
                        q_starts.append(v.copy())
                        break
                jit = 0
                while len(q_starts) < qp_starts and jit < len(starts):
                    v = starts[jit]
                    jit += 1
                    dup = False
                    for qv in q_starts:
                        if np.max(np.abs(v - qv)) < 1.0e-6:
                            dup = True
                            break
                    if not dup:
                        q_starts.append(v.copy())
                hb, hs = mm_search(data, q_starts, n, solver, iters_per_tol, TARGET)
                if hb is not None and hs < best_s - 1.0e-12:
                    best_h, best_s = hb, hs

    if best_s > TARGET:
        if n <= 128:
            nv, ns = local_stag(best_h, n, rng, 10, TARGET)
            if ns < best_s - 1.0e-12:
                best_h, best_s = nv, ns
        elif n <= 256:
            nv, ns = local_stag(best_h, n, rng, 6, TARGET)
            if ns < best_s - 1.0e-12:
                best_h, best_s = nv, ns

    h_out, c5_out, n_out = finalize(best_h, fb, n)
    return h_out.astype(np.float64), float(c5_out), int(n_out)
