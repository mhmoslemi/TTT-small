                x[j] += cap_j * rho * (0.05 + 0.95 * rng.random())
                x[i] -= cap_j * rho * (0.05 + 0.95 * rng.random())
            done = True
            break
        if not done:
            continue
        xp = project_capped(x, n)
        if xp is None:
            continue
        ns = exact_score(xp, n)
        if not np.isfinite(ns):
            continue
        d = ns - score
        if d < -1e-12:
            acc = True
        elif abs(d) <= 1e-12:
            acc = rng.random() < 0.1
        else:
            acc = rng.random() < (math.exp(-d / T) if -d / T > -700 else 0.0)
        if acc:
            x, score = xp, ns
    return x, score

def pair_polish(h, n, rng, sweeps, n_pairs):
    x = h.copy()
    score = exact_score(x, n)
    fracs = (1.0 / 64, 1.0 / 16, 1.0 / 4, 1.0 / 2, 1.0)
    for sw in range(sweeps):
        imp = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = []
            for _ in range(n_pairs or 160):
                i, j = int(rng.integers(n)), int(rng.integers(n))
                if i != j:
                    pairs.append((i, j))
        for i, j in pairs:
            bx, bs = x, score
            for fr in fracs:
                for di, dj in ((1, -1), (-1, 1)):
                    cap = min(1.0 - x[i if di == 1 else j], x[j if di == 1 else i])
                    if cap <= 1e-14:
                        continue
                    trial = x.copy()
                    trial[i] += di * fr * cap
                    trial[j] += dj * fr * cap
                    xp = project_capped(trial, n)
                    if xp is not None:
                        ns = exact_score(xp, n)
                        if np.isfinite(ns) and ns < bs - 1e-12:
                            bs, bx = ns, xp
            if bs < score - 1e-12:
                x, score, imp = bx, bs, True
        if not imp:
            break
    return x, score

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    if n < 1:
        n = 47
    if n > 999:
        n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    base = project_capped(parent, n) if parent is not None else None
    if base is None or not is_feasible(base, n):
        base = np.full(n, 0.5)
    safe_h = base.copy()
    safe_score = exact_score(safe_h, n)
    cst = np.full(n, 0.5)
    if exact_score(cst, n) < safe_score:
        safe_h, safe_score = cst.copy(), exact_score(cst, n)
    if n <= 64:
        P, G, L, sl, svr, npr = 12, 120, 1600, 20, 2, None
    elif n <= 256:
        P, G, L, sl, svr, npr = 8, 70, 500, 14, 2, 160
    else:
        P, G, L, sl, svr, npr = 6, 25, 80, 7, 1, 40
    pop = build_population(base, n, rng, P)
    sc = np.array([exact_score(p, n) for p in pop])
    bi = int(np.argmin(sc))
    gb, gs = pop[bi].copy(), float(sc[bi])
    try:
        gb, gs = diff_evo(pop, sc, n, rng, G, sl)
    except Exception:
        pass
    for _ in range(2):
        try:
            xa, sa = pair_anneal(gb, n, rng, L)
            if sa < gs - 1e-12:
                gs, gb = sa, xa.copy()
        except Exception:
            pass
    try:
        xp, sp = pair_polish(gb, n, rng, svr, npr)
        if sp < gs - 1e-12:
            gs, gb = sp, xp.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_score + 1e-12:
        final, fs = safe_h.copy(), safe_score
    return final.astype(np.float64), float(fs), int(n)
