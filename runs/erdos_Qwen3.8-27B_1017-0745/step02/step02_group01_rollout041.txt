import numpy as np

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i]); h[i] += c; r -= c
        else:
            c = min(-r, h[i]); h[i] -= c; r += c
    if not np.all(np.isfinite(h)) or np.any(h < 0) or np.any(h > 1):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def make_proposal(h, n, rng, step):
    if rng.random() < 0.8:
        for _ in range(4):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            lo_t, hi_t = max(-h[i], h[j] - 1.0), min(1.0 - h[i], h[j])
            if hi_t - lo_t < 1e-12:
                continue
            if rng.random() < 0.75:
                lo_t, hi_t = max(lo_t, -step), min(hi_t, step)
                if hi_t - lo_t < 1e-12:
                    continue
                t = rng.uniform(lo_t, hi_t)
            else:
                t = lo_t if rng.random() < 0.5 else hi_t

                    continue

            z[i] += t
            z[j] -= t
            p = project_capped(z, n)
            if p is not None:
                return p
    return project_capped(h + step * rng.standard_normal(n), n)

def _epi_ineq(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[-1] - q

def _epi_ineq_grad(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        row = d + n - 1
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        G[row, p0:p0+m] += dx * (1.0 - h[q0:q0+m])
        G[row, q0:q0+m] -= dx * h[p0:p0+m]
    J = np.zeros((L, n+1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _epi_eq(z):
    n = z.shape[0] - 1
    return np.sum(z[:n]) - n / 2.0

def _epi_eq_grad(z):
    n = z.shape[0] - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def _epi_obj(z):
    return z[-1]

def _epi_obj_grad(z):
    g = np.zeros(z.shape[0])
    g[-1] = 1.0
    return g

Equivalently, output[k] = sum_i a[i + (n-1-k)] * v[i]

This is: sum over i of h shifted by d, times (1-h) at original position.

The quantity we want is: sum_i h[i] * (1-h[i+d])  (h at original, (1-h) shifted).

So sum_i h[i] * (1-h[i+d]) = output[(n-1) - d]
