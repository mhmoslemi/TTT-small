            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_vals_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n - 1), n):
        a, b = max(d, 0), max(-d, 0)
        m = n - abs(d)
        ha, hb = h[a:a+m], h[b:b+m]
        q[d + n - 1] = dx * np.dot(ha, 1.0 - hb)
        J[d + n - 1, a:a+m] += dx * (1.0 - hb)
        J[d + n - 1, b:b+m] -= dx * ha
    return q, J

def _obj(z):
    return z[_ctx['n']]

def _obj_jac(z):
    g = np.zeros(_ctx['n'] + 1)
    g[_ctx['n']] = 1.0
    return g

def _ineq(z):
    n = _ctx['n']
    q, _ = lag_vals_jac(z[:n], n)
    return z[n] - q

def _ineq_jac(z):
    n = _ctx['n']
    _, J = lag_vals_jac(z[:n], n)
    L = 2 * n - 1
    out = np.empty((L, n + 1))
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

def _eq(z):
    return np.array([np.sum(z[:_ctx['n']]) - _ctx['n'] / 2.0])

def _eq_jac(z):
    n = _ctx['n']
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def solve_epi(x, n, maxiter=300):
    _ctx['n'] = n
    q, _ = lag_vals_jac(x, n)
    t0 = float(np.max(q)) + 1e-6
    z0 = np.concatenate([x, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None:
        return None
    y = res.x[:n]
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    return project_capped(y, n)

def _try_add(pool, v, n):
    if v is None or v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    if np.any(v < -1e-6) or np.any(v > 1.0 + 1e-6):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fb, n, rng, count):
    pool = []
    _try_add(pool, fb, n)
    _try_add(pool, project_capped(fb[::-1].copy(), n), n)
    _try_add(pool, project_capped(1.0 - fb, n), n)
    _try_add(pool, project_capped((1.0 - fb)[::-1].copy(), n), n)
    _try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    _try_add(pool, project_capped(alt, n), n)
    i = np.arange(n)
    P = max(2, n // 3)
    _try_add(pool, project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * i / P), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.05, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.18, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.30, n), n), n)
    _try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    return pool[:count]
