def pair_anneal(h, n, rng, L):
    x = h.copy()
    xs = exact_score(x, n)
    gb = x.copy()
    gs = xs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        rho = 1 - 0.98 * q
        found = False
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if x[i] < x[j]:
                cap = min(1.0 - x[i], x[j])
            else:
                cap = min(1.0 - x[j], x[i])
                i, j = j, i
            if cap > 1e-12:
                found = True
                break
        if not found:
            break
        amt = cap * rho * (0.05 + 0.95 * rng.random())

            if ts < gs - 1e-12:
                gs = ts; gb = tr.copy(); imp = True
        npool[0] = gb.copy(); nscs[0] = gs
        pool, scs = npool, nscs
        if imp:
            sc = 0
        else:
            sc += 1
            if sc >= stall and gen < G - 1:
                kn = max(2, P // 3)
                ki = sorted(set(list(np.argsort(scs)[:kn]) + [0]))
                np2, ns2 = [], []
                for k in ki:
                    np2.append(pool[k].copy()); ns2.append(scs[k])
                sg = 0.025 + 0.075 * (1 - q)
                while len(np2) < P:
                    if rng.random() < 0.5:
                        z = project_capped(gb + sg * rng.normal(0, 1, n), n)
                    else:
                        z = project_capped(rng.uniform(0, 1, n), n)
                    if z is None:
                        z = gb.copy()
                    np2.append(z); ns2.append(score(z, n))
                np2[0] = gb.copy(); ns2[0] = gs
                pool, scs = np2, ns2; sc = 0
    return gb, gs

def sa_phase(h, n, rng, L):
    x, xs = h.copy(), score(h, n)
    gb, gs = x.copy(), xs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        rho = 1 - 0.98 * q
        found = False
        for _ in range(20):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            cap = min(1.0 - x[i], x[j])
            if cap <= 1e-12:
                cap = min(1.0 - x[j], x[i])
                i, j = j, i
            if cap > 1e-12:
                found = True; break
        if not found:
            break
        am = cap * rho * (0.05 + 0.95 * rng.random())
        xn = x.copy(); xn[i] += am; xn[j] -= am
        xp = project_capped(xn, n)
        if xp is None:
            continue
        sn = score(xp, n)
        d = sn - xs
        if d < -1e-12:
            acc = True
        elif abs(d) <= 1e-12:
            acc = rng.random() < 0.1
        else:
            lp = -d / max(T, 1e-15)
            acc = (lp > -745) and (rng.random() < np.exp(lp))
        if acc:
            x, xs = xp, sn
            if xs < gs - 1e-12:
                gs = xs; gb = x.copy()
    return gb, gs

def polish(h, n, rng, n_rand, sweeps):
    x, xs = h.copy(), score(h, n)
    gb, gs = x.copy(), xs
    fr = np.array([1 / 64, 1 / 16, 0.25, 0.5, 1.0])
    for sw in range(sweeps):
        imp = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = [(int(rng.integers(n)), int(rng.integers(n))) for _ in range(n_rand)]
        for i, j in pairs:
            if i == j:
                continue
            for di in (0, 1):
                if di == 0:
                    cap = min(1.0 - x[i], x[j]); ii, jj = i, j
                else:
                    cap = min(1.0 - x[j], x[i]); ii, jj = j, i
                if cap < 1e-12:
                    continue
                for f in fr:
                    xn = x.copy(); xn[ii] += cap * f; xn[jj] -= cap * f
                    xp = project_capped(xn, n)
                    if xp is None:
                        continue
                    sn = score(xp, n)
                    if sn < xs - 1e-12:
                        x, xs = xp, sn; imp = True
        if xs < gs - 1e-12:
            gs, gb = xs, x.copy()
        if not imp:
            break
    if xs < gs:
        gs, gb = xs, x.copy()
    return gb, gs

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
        if n is None:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(score(h1, 1)), 1
    fb = None
    if parent is not None:
        fb = project_capped(parent, n)
    if fb is None:
        fb = np.full(n, 0.5)
    fs = score(fb, n)
    c5 = score(np.full(n, 0.5), n)
    safe_h, safe_s = fb.copy(), fs
    if c5 < safe_s:
        safe_h, safe_s = np.full(n, 0.5), c5
    if n <= 64:
        P, G, L, stall, sw = 12, 120, 1600, 20, 3
    elif n <= 256:
        P, G, L, stall, sw = 8, 70, 500, 14, 2
    else:
        P, G, L, stall, sw = 6, 25, 80, 7, 1
    n_rand = max(40, 160 - n // 2)
    pool, scs = build_pool(fb, n, rng, P)
    gh, gs = de_phase(pool, scs, n, rng, G, stall)
    gh, gs = sa_phase(gh, n, rng, L)
    gh, gs = polish(gh, n, rng, n_rand, sw)
    pool2, scs2 = build_pool(gh, n, rng, max(4, P // 2))
    gh2, gs2 = de_phase(pool2, scs2, n, rng, max(20, G // 3), stall)
    if gs2 < gs:
        gh, gs = gh2, gs2
    gh, gs = sa_phase(gh, n, rng, max(100, L // 4))
    gh, gs = polish(gh, n, rng, n_rand, sw)
    if gs > safe_s + 1e-12:
        gh, gs = safe_h.copy(), safe_s
    final = project_capped(gh, n)
    if final is None or not ok(final, n):
        final = safe_h.copy()
    fs = score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), score(safe_h, n)
    return final.astype(np.float64), float(fs), int(n)
