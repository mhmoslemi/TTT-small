def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if hi - lo < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - h
            tot = float(np.sum(cap))
            if tot > 1e-15:
                h = np.clip(h + r * cap / tot, 0.0, 1.0)
        else:
            tot = float(np.sum(h))
            if tot > 1e-15:
                h = np.clip(h + r * h / tot, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def lag_values(h, n):

        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        v[row] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def smooth_obj_grad(h, n, theta):

    m = float(np.max(v))
    ex = np.exp(np.clip((v - m) / theta, -700.0, 0.0))
    s = float(np.sum(ex))
    if s < 1e-300:
        return m, np.zeros(n)
    p = ex / s
    phi = m + theta * math.log(s)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        pd = p[d + n - 1]

        hi = min(n, n - d)
        grad[lo:hi] += pd * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= pd * h[lo:hi]
    return phi, grad

def coord_refine(h, n, sweeps=3):
    best = h.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for dv in (-0.1, -0.05, 0.05, 0.1, -0.3, 0.3):
                trial = best.copy()
                new_v = best[i] + dv
                if new_v < 0.0 or new_v > 1.0:
                    continue
                # Redistribute: adjust another coordinate to keep sum
                rem = best[i] - new_v
                # Put rem on the coordinate with most slack
                slack = np.where(rem > 0, 1.0 - best, best).copy()
                slack[i] = -1.0 if rem > 0 else 2.0
                j = int(np.argmax(slack))

                    j = 1 - i if n > 1 else i
                if j == i:
                    continue

                if np.isfinite(sc) and sc < best_s - 1e-12:
                    best = trial
                    best_s = sc
                    improved = True
        if not improved:
            break
    return best, best_s
