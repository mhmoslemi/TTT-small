        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p)

def build_starts(fallback, n, rng):
    pool = []
    add_start(pool, fallback, n, 10)
    add_start(pool, fallback[::-1], n, 10)
    add_start(pool, 1.0 - fallback, n, 10)
    add_start(pool, np.full(n, 0.5), n, 10)
    blk = np.zeros(n)
    blk[:int(n // 2)] = 1.0
    if n % 2:
        blk[int(n // 2)] = 0.5
    add_start(pool, blk, n, 10)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_start(pool, alt, n, 10)
    add_start(pool, fallback + rng.normal(0, 0.07, n), n, 10)
    add_start(pool, fallback + rng.normal(0, 0.18, n), n, 10)
    t = np.linspace(0, 1, n)
    add_start(pool, 0.5 + 0.3 * np.cos(2.0 * math.pi * t), n, 10)
    add_start(pool, 0.5 + 0.4 * np.sin(math.pi * t), n, 10)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1:
        n = 47
    if parent is None:
        try:
            if 'n_points' in kwargs:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
        except Exception:
            pass
    if n < 1:
        n = 47
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    fb_score = exact_score(fallback, n)
    if n > 999:
        return fallback, fb_score, n
    best_h = fallback.copy()
    best_score = fb_score
    starts = build_starts(fallback, n, rng)
    scores = np.zeros(len(starts))
    for i in range(len(starts)):
        scores[i] = exact_score(starts[i], n)
    order = np.argsort(scores)
    scale = math.sqrt(47.0 / float(n))
    n_chains = min(len(order), max(1, min(5, int(round(5 * scale)))))
    n_iters = max(8, min(40, int(round(40 * scale))))
    used = set()
    for idx_pos in range(n_chains):
        idx = int(order[idx_pos])
        if idx in used:
            continue
        used.add(idx)
        try:
            best_h, best_score = refine_chain(starts[idx], n, n_iters, 0.09, best_h, best_score)
        except Exception:
            pass
    try:
        best_h, best_score = refine_chain(best_h, n, max(6, n_iters // 2), 0.03, best_h, best_score)
    except Exception:
        pass
    try:
        best_h, best_score = refine_chain(best_h, n, 12, 0.015, best_h, best_score)
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback
        fs = fb_score
    return final, float(fs), int(n)
