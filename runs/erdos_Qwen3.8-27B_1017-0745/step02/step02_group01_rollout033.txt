Let me analyze the problem and the strategy carefully.

The goal is to find a step function h: [0,2] → [0,1] minimizing C₅ = max_k ∫h(x)(1-h(x+k))dx, subject to h(x) ∈ [0,1] and ∫₀²h(x)dx = 1.

The key innovation in the strategy is to use a sequential linear programming trust-region approach instead of SLSQP or subgradient methods. At each step:
1. Compute the exact correlation at all lags
2. Form a linear model of each lag function
3. Solve a bounded LP to find a displacement that reduces the maximum lag
4. Accept only if exact score improves

                      bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 2000})
    except Exception:
        return None
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if abs(np.sum(u)) > 1e-5:
        return None
    for i in range(n):
        if u[i] < -rho - 1e-7 or u[i] > rho + 1e-7:
            return None
    return u

def refine_chain(start, n, max_iter, rho0, best_h, best_score):
    curr = project_capped(start, n)
    if curr is None:
        return best_h, best_score
    curr_score = exact_score(curr, n)
    if not np.isfinite(curr_score):
        return best_h, best_score
    if curr_score < best_score - 1e-12:
        best_h, best_score = curr.copy(), curr_score
    rho = rho0
    no_imp = 0
    no_solve = 0
    rho_min = 1e-5
    rho_max = 0.32
    for _ in range(max_iter):
        if no_solve >= 3 or no_imp >= 6 or rho < rho_min:
            break
        q, A = lag_model(curr, n)
        if q is None:
            rho = max(rho_min, 0.5 * rho)
            no_solve += 1
            no_imp += 1
            continue
        u = lp_step(curr, n, rho, q, A)
        if u is None:
            rho = max(rho_min, 0.5 * rho)
            no_solve += 1
            no_imp += 1
            continue
        no_solve = 0
        best_trial = None
        best_trial_score = curr_score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            z = curr + alpha * u
            y = project_capped(z, n)
            if y is None or not is_feasible(y, n):
                continue
            sy = exact_score(y, n)
            if not np.isfinite(sy) or sy >= best_trial_score - 1e-12:
                continue
            best_trial_score = sy
            best_trial = y.copy()
        if best_trial is None:
            rho = max(rho_min, 0.5 * rho)
            no_imp += 1
            continue
        delta = best_trial - curr
        actual = curr_score - best_trial_score
        model_val = np.max(q + A[:, :n] @ delta)
        predicted = curr_score - model_val
        curr = best_trial
        curr_score = best_trial_score
        no_imp = 0
        if curr_score < best_score - 1e-12:
            best_h, best_score = curr.copy(), curr_score
        if predicted > 1e-12 and np.isfinite(predicted) and np.isfinite(actual):
            ratio = actual / predicted
            if ratio < 0.1:
                rho = max(rho_min, 0.5 * rho)
            elif ratio > 0.75 and np.max(np.abs(delta)) >= 0.8 * rho:
                rho = min(1.5 * rho, rho_max)
        else:
            rho = max(rho_min, 0.5 * rho)
    return best_h, best_score

def add_start(pool, z, n):
    p = project_capped(z, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    if len(pool) < 8:
        pool.append(p)

def build_starts(fallback, n, rng):
    pool = [fallback.copy()]
    add_start(pool, fallback[::-1].copy(), n)
    add_start(pool, 1.0 - fallback[::-1].copy(), n)
    add_start(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_start(pool, blk, n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_start(pool, alt, n)
    add_start(pool, fallback + rng.normal(0, 0.05, n), n)
    add_start(pool, fallback + rng.normal(0, 0.15, n), n)
    t = np.arange(n)
    add_start(pool, 0.5 + 0.3 * np.cos(2.0 * np.pi * t / n), n)
    return pool

def polish(best_h, best_score, n, rng, n_tries=200):
    curr = best_h.copy()
    cs = best_score
    for _ in range(n_tries):
        perturb = rng.normal(0, 0.01, n)
        y = project_capped(curr + perturb, n)
        if y is None or not is_feasible(y, n):
            continue
        sy = exact_score(y, n)
        if np.isfinite(sy) and sy < cs - 1e-13:
            curr = y.copy()
            cs = sy
    return curr, cs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb_score = exact_score(fallback, n)
    if n == 1 or not np.isfinite(fb_score):
        return fallback, float(fb_score if np.isfinite(fb_score) else 0.5), n
    if n > 999:
        return fallback, float(fb_score), n
    starts = build_starts(fallback, n, rng)
    best_h, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best_h, best_score = st.copy(), ss
    scores = np.array([exact_score(st, n) for st in starts])
    scores[np.where(~np.isfinite(scores))[0]] = 1e10
    order = list(np.argsort(scores))
    n_chains = min(len(starts), max(1, min(5, round(5 * math.sqrt(47.0 / max(n, 1))))))
    selected = order[:n_chains]
    if 0 not in selected:
        selected[0] = 0
    iters = max(8, min(35, round(35 * math.sqrt(47.0 / max(n, 1)))))
    for idx in selected:
        try:
            best_h, best_score = refine_chain(starts[idx], n, iters, 0.08, best_h, best_score)
        except Exception:
            pass
    try:
        best_h, best_score = refine_chain(best_h.copy(), n, max(5, iters // 2), 0.03, best_h, best_score)
    except Exception:
        pass
    try:
        best_h, best_score = polish(best_h, best_score, n, rng, 300)
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    fs = exact_score(final, n)
    return final, float(fs), int(n)
