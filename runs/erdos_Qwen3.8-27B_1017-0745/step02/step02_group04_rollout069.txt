from scipy.optimize import minimize

_ctx = {}

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2 * n + 4):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_vals_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n - 1), n):
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        hp = h[a:a + m]
        hq = h[b:b + m]
        idx = d + n - 1
        q[idx] = dx * np.dot(hp, 1.0 - hq)
        J[idx, a:a + m] += dx * (1.0 - hq)
        J[idx, b:b + m] -= dx * hp
    return q, J

def slsqp_obj(z):
    return z[_ctx['n']]

def slsqp_obj_jac(z):
    g = np.zeros(_ctx['n'] + 1)
    g[_ctx['n']] = 1.0
    return g

def slsqp_ineq(z):
    h = z[:_ctx['n']]
    q, _ = lag_vals_jac(h, _ctx['n'])
    return z[_ctx['n']] - q

def slsqp_ineq_jac(z):
    h = z[:_ctx['n']]
    _, J = lag_vals_jac(h, _ctx['n'])
    n = _ctx['n']
    L = 2 * n - 1
    Gj = np.zeros((L, n + 1))
    Gj[:, :n] = -J
    Gj[:, n] = 1.0
    return Gj

def slsqp_eq(z):
    return np.array([np.sum(z[:_ctx['n']]) - _ctx['n'] / 2.0])

def slsqp_eq_jac(z):
    n = _ctx['n']
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def solve_slsqp(x, n, maxiter):
    _ctx['n'] = n
    q, _ = lag_vals_jac(x, n)
    t0 = float(np.max(q)) + 1e-8
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
    ]
    try:
        res = minimize(slsqp_obj, z0, jac=slsqp_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def _try_add(pool, v, n):
