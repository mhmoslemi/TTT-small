            break

            lo = mid
        else:
            hi = mid

def pair_proposal(h, n, a, rng, full_mode):
    if n < 2:

    p = int(rng.integers(n))
    q = int(rng.integers(n - 1))

        q += 1

        return None

        t = rng.uniform(L, U)
    else:
        t = rng.uniform(-a, a)
        if t < L or t > U:
            return None
    trial = h.copy()
    trial[p] = h[p] + t
    trial[q] = h[q] - t
    if not is_feasible(trial, n):
        return None
    return trial

def hit_proposal(h, n, rng):
    for _ in range(5):
        d = rng.standard_normal(n)
        d -= np.mean(d)
        nm = float(np.linalg.norm(d))
        if nm < 1e-14:
            continue

        trial = h + t * d
