    ex = np.clip((v - m) / theta, -700.0, 700.0)
    ex = np.exp(ex)
    se = float(np.sum(ex))
    if se <= 0.0 or not np.isfinite(se):
        return None, None
    p = ex / se
    phi = m + theta * math.log(se)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        w = p[d + n - 1]
        if w < 1e-16:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += w * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= w * h[lo:hi]
    return phi, grad

def continuation_opt(h, n, iters_per_temp, tau_list, best_holder):
    h = h.copy()
    for tau in tau_list:
        theta = (n / 2.0) * tau
        alpha = 0.2
        for _ in range(iters_per_temp):
            phi_h, grad = phi_and_grad(h, n, theta)
            if phi_h is None:
                break
            norm_d = 1.0
            accepted = False
            for _bt in range(24):
                y = project_capped(h - alpha * grad, n)
                if y is None:
                    alpha *= 0.5
                    continue
                dr = y - h
                norm_d = float(np.max(np.abs(dr)))
                if norm_d < 1e-10:
                    accepted = True
                    break
                gd = float(np.dot(grad, dr))
                if gd > -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _ = phi_and_grad(y, n, theta)
                if phi_y is None:
                    alpha *= 0.5
                    continue
                if phi_y <= phi_h + 1e-4 * gd + 1e-12 * max(1.0, abs(phi_h)):
                    h = y
                    alpha = min(0.5, 1.15 * alpha)
                    accepted = True
                    sc = exact_score(h, n)
                    if np.isfinite(sc) and sc < best_holder[0] - 1e-12:
                        best_holder[0] = sc
                        best_holder[1] = h.copy()
                    break
                alpha *= 0.5
            if accepted and norm_d < 1e-8:
                break
            if not accepted:
                break
    return h

def _add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def _coordinate_polish(h, n, n_sweeps):
    best = h.copy()
    best_sc = exact_score(best, n)
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            for delta in (-0.02, -0.005, 0.005, 0.02):
                cand = best[i] + delta
                if cand < 0.0 or cand > 1.0:
                    continue
                j = (i + n // 2) % n
                save = best.copy()
                best[i] = cand
                best[j] -= delta
                if best[j] < 0.0 or best[j] > 1.0:
                    best[:] = save
                    continue
                sc = exact_score(best, n)
                if np.isfinite(sc) and sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                else:
                    best[:] = save
        if not improved:
            break
    return best, best_sc

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    starts = []
    _add_start(starts, fallback, n)
    if parent is not None:
        _add_start(starts, fallback[::-1], n)
        _add_start(starts, 1.0 - fallback, n)
        _add_start(starts, np.roll(fallback, n // 3), n)
        _add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
        _add_start(starts, np.roll(fallback, 2 * n // 5), n)
    else:
        _add_start(starts, uniform, n)
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        _add_start(starts, alt, n)
        tt = np.arange(n)
        _add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
    if n <= 64:
        n_starts_def, iters_def = 6, 60
    elif n <= 128:
        n_starts_def, iters_def = 5, 50
    elif n <= 256:
        n_starts_def, iters_def = 3, 25
    else:
        n_starts_def, iters_def = 2, 10
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts_def)), 8))
    except Exception:
        n_starts = n_starts_def
    try:
        max_iter = max(1, min(int(kwargs.get('max_iter', iters_def)), 120))
    except Exception:
        max_iter = iters_def
    starts = starts[:n_starts]
    bh = [fb_score, fallback.copy()]
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < bh[0] - 1e-12:
            bh[0] = ss
            bh[1] = st.copy()
    tau_list = [0.02, 0.006, 0.0018, 0.0005, 0.00015, 0.00004, 0.00001]
    for st in starts:
        try:
            continuation_opt(st, n, max_iter, tau_list, bh)
        except Exception:
            continue
    hop_sds = [0.025, 0.08, 0.05] if n <= 128 else [0.05]
    for hs in hop_sds:
        pert = bh[1] + rng.normal(0, hs, n)
        pr = project_capped(pert, n)
        if pr is None:
            continue
        pr_sc = exact_score(pr, n)
        if np.isfinite(pr_sc) and pr_sc < bh[0] - 1e-12:
            bh[0] = pr_sc
            bh[1] = pr.copy()
        try:
            continuation_opt(pr, n, max_iter, tau_list, bh)
        except Exception:
            continue
    if n <= 96:
        pol, pol_sc = _coordinate_polish(bh[1], n, 3)
        fp = project_capped(pol, n)
        if fp is not None:
            fs = exact_score(fp, n)
            if np.isfinite(fs) and fs < bh[0] - 1e-12:
                bh[0] = fs
                bh[1] = fp.copy()
    final = project_capped(bh[1], n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
