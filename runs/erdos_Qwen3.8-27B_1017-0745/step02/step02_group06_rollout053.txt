def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or c.shape != (2 * n - 1,):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        nlo = max(0, -d); nhi = min(n, n - d)
        j = np.arange(nlo, nhi)
        i = j + d
        J[d + n - 1, i] += dx * (1.0 - h[j])
        J[d + n - 1, j] += -dx * h[i]
    return J

def obj_fun(z, n):
    return z[-1]

def obj_jac(z, n):
    g = np.zeros(z.shape)
    g[-1] = 1.0
    return g

def ineq_fun(z, n):
    h = z[:n]
    return z[-1] - lag_values(h, n)

def ineq_jac(z, n):
    h = z[:n]
    Lj = lag_jacobian(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -Lj
    J[:, n] = 1.0
    return J

def eq_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):

def solve_slswp(h0, t0, n, maxiter):
    z0 = np.append(h0.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (n,)}
