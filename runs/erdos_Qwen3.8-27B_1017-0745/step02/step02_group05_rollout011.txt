    if n_lines == 0:
        return [(0.0, 0.0, lo, hi)]  # degenerate
    
    order = np.argsort(slopes, kind='mergesort')
    # For equal slopes, keep only the one with max intercept
    filtered = []
    for k in order:
        if filtered and abs(slopes[k] - slopes[filtered[-1]]) < 1e-15:
            if intercepts[k] > intercepts[filtered[-1]]:
                filtered[-1] = k
            continue
        filtered.append(k)
    
    # Now build upper envelope
    hull = []  # list of (index, start_t)
    for k in filtered:
        m, c = slopes[k], intercepts[k]
        if not hull:
            hull.append((k, lo))  # active from lo (or -inf)
            continue
        # Find intersection with last hull line
        lm, lc = slopes[hull[-1][0]], intercepts[hull[-1][0]]
        if abs(m - lm) < 1e-15:
            # Parallel, keep the one with higher intercept (shouldn't happen after filtering)
            continue
        t_cross = (lc - c) / (m - lm)
        # Pop while t_cross <= start of last
        while len(hull) >= 1 and t_cross <= hull[-1][1] + 1e-15:
            hull.pop()
            if not hull:
                break
            lm, lc = slopes[hull[-1][0]], intercepts[hull[-1][0]]
            if abs(m - lm) < 1e-15:
                break
            t_cross = (lc - c) / (m - lm)
        if not hull:
            hull.append((k, lo))
        else:
            hull.append((k, t_cross))
    
    # Build segments
    segments = []
    for i in range(len(hull)):
        k, start = hull[i]
        end = hull[i + 1][1] if i + 1 < len(hull) else hi
        if end > start:
            segments.append((slopes[k], intercepts[k], max(lo, start), min(hi, end)))
    if not segments:
        segments = [(0.0, 0.0, lo, hi)]
    return segments

def pair_line_min(h, A, n, i, j):
    """Find t* minimizing max_d Q_d(h+t*(e_i-e_j)) over feasible interval."""
    dx = 2.0 / n
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0, A.max(), None, None
    
    # Compute B for each lag
    B = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        a_d = max(d, 0)
        b_d = max(-d, 0)
        m_d = n - abs(d)
        bd = 0.0
        # v[a_d+r]*(1-h[b_d+r]) term
        if a_d <= i < a_d + m_d:
            r = i - a_d
            bd += (1.0 - h[b_d + r])
        if a_d <= j < a_d + m_d:
            r = j - a_d
            bd -= (1.0 - h[b_d + r])
        # -h[a_d+r]*v[b_d+r] term
        if b_d <= i < b_d + m_d:
            r = i - b_d
            bd -= h[a_d + r]
        if b_d <= j < b_d + m_d:
            r = j - b_d
            bd += h[a_d + r]
        B[d + n - 1] = dx * bd
    if not np.all(np.isfinite(B)):
        return 0.0, A.max(), None, None
    
    D = np.zeros(2 * n - 1)
    D[n - 1] = -2.0 * dx
    r0 = abs(i - j)
    if r0 > 0:
        D[n - 1 + r0] = dx
        D[n - 1 - r0] = dx
    
    # Identify the 3 special indices
    special = [n - 1, n - 1 + r0, n - 1 - r0]
    if r0 == 0:
        special = [n - 1]
    # Remove duplicates
    special_set = set(special)
    
    # Collect affine lines (non-special lags)
    aff_idx = []
    for k in range(2 * n - 1):
        if k not in special_set:
            aff_idx.append(k)
    
    slopes = B[aff_idx]
    intercepts = A[aff_idx]
    
    segments = upper_envelope(slopes, intercepts, lo, hi)
    
    # Special quadratics: (A[k], B[k], D[k]) for k in special
    quad_data = [(A[k], B[k], D[k]) for k in special if k < 2*n-1 and k >= 0]
    
    best_t = 0.0
    best_val = float(np.max(A))  # current score (t=0)
    
    for (sm, sc, seg_lo, seg_hi) in segments:
        if seg_hi <= seg_lo:
            continue
        # Functions on this segment: line (sm*t+sc) and quadratics
        # Collect all functions as (a, b, c) where f(t) = a*t^2+b*t+c
        funcs = [(0.0, sm, sc)]
        for (a_q, b_q, c_q) in quad_data:
            funcs.append((a_q, b_q, c_q))
        
        # Gather candidate t values
        cands = [seg_lo, seg_hi]
        if seg_lo < 0.0 < seg_hi:
            cands.append(0.0)
        
        # Intersections of pairs
        for p in range(len(funcs)):
            for q in range(p + 1, len(funcs)):
                ap, bp, cp = funcs[p]
                aq, bq, cq = funcs[q]
                roots = poly_roots_in(ap, bp, cp, aq, bq, cq, seg_lo, seg_hi)
                cands.extend(roots)
        
        # Vertices of convex quadratics
        for (a_q, b_q, c_q) in quad_data:
            if a_q > 1e-15:
                tv = -b_q / (2.0 * a_q)
                if seg_lo - 1e-12 <= tv <= seg_hi + 1e-12:
                    cands.append(min(seg_hi, max(seg_lo, tv)))
        
        # Evaluate at all candidates
        for t in cands:
            if not np.isfinite(t):
                continue
            t = min(seg_hi, max(seg_lo, t))
            val = sm * t + sc
            for (a_q, b_q, c_q) in quad_data:
                fq = a_q * t * t + b_q * t + c_q
                if fq > val:
                    val = fq
            if np.isfinite(val) and val < best_val - 1e-14:
                best_val = val
                best_t = t
    
    # Also check: is best_val really ≤ t=0 value? (safety)
    if not np.isfinite(best_val):
        return 0.0, A.max(), None, None
    
    return best_t, best_val, B, D

def coordinate_sweep(h, n, rng, n_sweeps, n_pairs_per_sweep=None):
    h = h.copy()
    A = lag_values(h, n)
    score = float(np.max(A))
    total_pairs = n * (n - 1) // 2
    if n_pairs_per_sweep is None:
        n_pairs_per_sweep = total_pairs
    
    for sw in range(n_sweeps):
        start_h = h.copy()
        start_score = score
        # Build pair schedule
        pairs = []
        if n_pairs_per_sweep >= total_pairs:
            for i in range(n):
                for j in range(i + 1, n):
                    pairs.append((i, j))
            rng.shuffle(pairs)
        else:
            # Cycle neighbors + random
            for i in range(n):
                pairs.append((i, (i + 1) % n))
            while len(pairs) < n_pairs_per_sweep:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    pairs.append((min(i,j), max(i,j)))
            pairs = pairs[:n_pairs_per_sweep]
            rng.shuffle(pairs)
        
        n_accepted = 0
        for (i, j) in pairs:
            t_star, pred, B, D = pair_line_min(h, A, n, i, j)
            if B is None or not np.isfinite(t_star):
                continue
            if pred >= score - 1e-12:
                continue
            h[i] += t_star
            h[j] -= t_star
            A = A + B * t_star + D * t_star * t_star
            score = float(np.max(A))
            n_accepted += 1
            # Periodic refresh
            if n_accepted % max(1, n) == 0:
                h_pitch = project_capped(h, n)
                if h_pitch is None or not is_feasible(h_pitch, n):
                    h = start_h.copy()
                    A = lag_values(h, n)
                    score = start_score
                    break
                h = h_pitch
                A = lag_values(h, n)
                score = float(np.max(A))
        
        # End of sweep: refresh
        h_end = project_capped(h, n)
        if h_end is None or not is_feasible(h_end, n):
            h = start_h.copy()
            A = lag_values(h, n)
            score = start_score
            continue
        h = h_end
        A = lag_values(h, n)
        end_score = float(np.max(A))
        if end_score > start_score + 1e-12:
            h = start_h.copy()
            A = lag_values(h, n)
            score = start_score
            continue
        score = end_score
        if abs(end_score - start_score) < 1e-12:
            # stalled
            if sw > 0:
                break
    
    return h, score

def build_starts(base, n, rng, max_starts):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(blk[::-1].copy())
    cands.append(base + rng.normal(0, 0.05, n))
    cands.append(base + rng.normal(0, 0.15, n))
    for k in (1, 2):
        ph = rng.uniform(0, 6.28)
        cands.append(0.5 + 0.35 * np.sin(2 * np.pi * k * np.arange(n) / n + ph))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) >= max_starts:
            break
    return pool
