        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:max_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = st.copy(), sc
    for st in starts:
        try:
            best, best_score = continuation_opt(st, n, iters_ps, best, best_score, tau_list)
        except Exception:
            pass
    hop_sds = [0.025, 0.08, 0.05] if n <= 128 else [0.05]
    for sd_hop in hop_sds:
        try:
            pert = project_capped(best + rng.normal(0, sd_hop, n), n)
            if pert is None:
                continue
            ps = exact_score(pert, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = pert.copy(), ps
            best, best_score = continuation_opt(pert, n, iters_ps, best, best_score, tau_list)
        except Exception:
            pass
    if n <= 64:
        try:
            cr, cr_sc = coord_refine(best, n, best_score, sweeps=3)
            if np.isfinite(cr_sc) and cr_sc < best_score - 1e-12:
                best, best_score = cr, cr_sc
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
