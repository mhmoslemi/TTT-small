def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)

        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
        else:
            sub = min(-r, h[i])
            h[i] -= sub
            r += sub

    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False

        return False

        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def build_fourier_basis(n):
    if n <= 1:
        return np.zeros((0, 0)), np.array([], dtype=int)
    F = (n - 1) // 2
    ncols = n - 1
    Phi = np.zeros((n, ncols), dtype=np.float64)

        labels[idx] = k
        idx += 1
        Phi[:, idx] = np.sqrt(2.0 / n) * np.sin(col)
        labels[idx] = k
        idx += 1
