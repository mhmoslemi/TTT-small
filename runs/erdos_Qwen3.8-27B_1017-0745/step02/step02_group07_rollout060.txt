    best_h = h0.copy()
    best_s = exact_score(h0, n)
    h = h0.copy()

    for tau in taus:
        alpha = 1.0
        cur_F = None
        for it in range(iters_per_tau):
            F, g = smooth_value_grad(h, n, tau)
            if not np.all(np.isfinite(F)) or not np.all(np.isfinite(g)):
                break
            if cur_F is None:
                cur_F = F
            y = project_capped(h - alpha * g, n)
            if y is None:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
                continue
            d = y - h
            if np.max(np.abs(d)) <= 1e-8:
                break
            F_y, _ = smooth_value_grad(y, n, tau)
            # Armijo
            accepted = (F_y <= F + 1e-4 * np.dot(g, d) + 1e-14)
            tries = 0
            while not accepted and tries < 20:
                alpha *= 0.5
                y = project_capped(h - alpha * g, n)
                if y is None:
                    break
                d = y - h
                if np.max(np.abs(d)) <= 1e-10:

                F_y, _ = smooth_value_grad(y, n, tau)
                accepted = (F_y <= F + 1e-4 * np.dot(g, d) + 1e-14)
                tries += 1

                alpha = min(1.5 * alpha, 10.0)
                s = exact_score(h, n)
                if s < best_s - 1e-12:

                    best_h = h.copy()
            else:
                break

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-15:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_values(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    Q[0] = np.sum(h * (1.0 - h)) * dx
    for r in range(1, n):
        m = n - r
        Q[2 * r - 1] = np.sum(h[:m] * (1.0 - h[r:])) * dx
        Q[2 * r] = np.sum(h[r:] * (1.0 - h[:m])) * dx
    return Q

def lag_grad(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[0] = (1.0 - 2.0 * h) * dx
    for r in range(1, n):
        m = n - r
        G[2 * r - 1, :m] += (1.0 - h[r:]) * dx
        G[2 * r - 1, r:] -= h[:m] * dx
        G[2 * r, r:] += (1.0 - h[:m]) * dx
        G[2 * r, :m] -= h[r:] * dx
    return G

def dedup_add(pool, v, th=1e-9):
    if v is None or not np.all(np.isfinite(v)):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def build_starts(fb, n, rng, max_st):
    pool = []
    dedup_add(pool, fb.copy())
    dedup_add(pool, project_capped(fb[::-1], n))
    dedup_add(pool, project_capped(1.0 - fb, n))
    dedup_add(pool, project_capped(1.0 - fb[::-1], n))
    dedup_add(pool, np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    dedup_add(pool, project_capped(blk, n))
    dedup_add(pool, project_capped(np.roll(fb, n // 3), n))
    dedup_add(pool, project_capped(np.roll(fb, n // 2), n))
    for sd in (0.08, 0.2, 0.35):
        dedup_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    ii = np.arange(n)
    for k in (1, 2):
        dedup_add(pool, project_capped(0.5 + 0.4*np.cos(2*np.pi*k*ii/n), n))
        dedup_add(pool, project_capped(0.5 + 0.4*np.sin(2*np.pi*k*ii/n), n))
    while len(pool) < max_st:
        dedup_add(pool, project_capped(fb + rng.normal(0, 0.4, n), n))
    return pool[:max_st]

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n):
    return z[n] - lag_values(z[:n], n)

def slsqp_ineq_grad(z, n):
    G = lag_grad(z[:n], n)
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -G
    out[:, n] = 1.0
    return out

def slsqp_eq(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def slsqp_eq_grad(z, n):
    out = np.zeros((1, n + 1))
    out[0, :n] = 1.0
    return out

def solve_slsqp(h0, n, maxiter):
    t0 = exact_score(h0, n) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_grad, 'args': (n,)},
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_grad, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_grad,
                       bounds=bounds, constraints=cons, args=(n,),
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def smooth_value_grad(h, n, tau):
    Q = lag_values(h, n)
    qmax = float(np.max(Q))
    shifted = (Q - qmax) / tau
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    if not np.isfinite(total) or total <= 0:
        return None, None
    p = exps / total
    F = qmax + tau * np.log(total)
    dx = 2.0 / n
    g = p[0] * (1.0 - 2.0 * h) * dx
    for r in range(1, n):
        m = n - r
        pp, pm = p[2 * r - 1], p[2 * r]
        g[:m] += dx * (pp * (1.0 - h[r:]) - pm * h[r:])
        g[r:] += dx * (pm * (1.0 - h[:m]) - pp * h[:m])
    return F, g

def smooth_polish(h0, n, iters_per_tau):
    best_h, best_s = h0.copy(), exact_score(h0, n)
    h = h0.copy()
    for tau in (0.02, 0.005, 0.001, 0.0002):
        alpha = 1.0
        for it in range(iters_per_tau):
            F, g = smooth_value_grad(h, n, tau)
            if F is None or not np.all(np.isfinite(g)):
                break
            y, accepted, d = None, False, None
            for _bt in range(25):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    if alpha < 1e-7:
                        break
                    continue
                d = y - h
                if np.max(np.abs(d)) < 1e-9:
                    break
                Fy, _ = smooth_value_grad(y, n, tau)
                if Fy is not None and Fy <= F + 1e-4 * float(np.dot(g, d)) + 1e-14:
                    accepted = True
                    F = Fy
                    break
                alpha *= 0.5
                if alpha < 1e-7:
                    break
            if accepted and y is not None and np.max(np.abs(d)) > 1e-9:
                h = y
                alpha = min(1.5 * alpha, 8.0)
                s = exact_score(h, n)
                if np.isfinite(s) and s < best_s - 1e-12:
                    best_s, best_h = s, h.copy()
            else:
                break
    return best_h, best_s
