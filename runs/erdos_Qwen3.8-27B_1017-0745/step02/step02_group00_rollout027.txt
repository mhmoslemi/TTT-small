import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def svg(h, tau, n):
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    m = float(np.max(q))
    ex = np.exp(np.clip((q - m) / tau, -700.0, 700.0))
    tot = float(np.sum(ex))
    w = ex / tot
    F = m + tau * float(np.log(tot))
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        row = d + n - 1
        if w[row] < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += w[row] * dx * (1.0 - h[lo + d:hi + d])
        g[lo + d:hi + d] -= w[row] * dx * h[lo:hi]
    return F, g

def polish(h, n, sw):
    h = h.copy()
    cur = exact_score(h, n)
    if not np.isfinite(cur):
        return h
    dts = np.array([0.0005, 0.002, 0.005, 0.01, 0.03, 0.08, 0.2])

        imp = False

                lod = max(-h[i], h[j] - 1.0)
                hid = min(1.0 - h[i], h[j])
                if hid <= lod + 1e-12:
                    continue
                bd, bs = 0.0, cur
                for dv in np.concatenate([dts, -dts, [lod, hid]]):
