            qmax = np.max(Q)
            shifted = (Q - qmax) / tau
            exps = np.exp(shifted)
            ssum = np.sum(exps)
            if not np.isfinite(ssum) or ssum < 1e-300:
                break
            p = exps / ssum
            F = qmax + tau * math.log(ssum)
            dx = 2.0 / n
            g = np.zeros(n)
            g += p[0] * dx * (1.0 - 2.0 * h)
            for r in range(1, n):
                m = n - r
                g[:m] += p[2*r-1] * dx * (1.0 - h[r:])
                g[r:] += p[2*r-1] * dx * (-h[:m])
                g[:m] += p[2*r] * dx * (-h[r:])
                g[r:] += p[2*r] * dx * (1.0 - h[:m])
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _bt in range(20):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h
                Qy = lag_values(y, n)
                qy = np.max(Qy)
                ey = np.exp((Qy - qy) / tau)
                sy = np.sum(ey)
                Fy = qy + tau * math.log(sy) if sy > 1e-300 and np.isfinite(sy) else None
                if Fy is not None and Fy <= F + 1e-4 * np.dot(g, d) + 1e-14:
                    h = y
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            alpha = min(1.5 * alpha, 10.0)
            se = exact_score(h, n)
            if se < best_s - 1e-12:
                best_s = se
                best_h = h.copy()
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    starts = build_starts(parent if parent is not None else fb, n, rng)
    best_h = fb.copy()
    best_s = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < best_s - 1e-12:
            best_s = ss
            best_h = st.copy()
    if n <= 128:
        maxiter = 220 if n <= 64 else 140
        bounds = [(0, 1)] * n + [(0, 2)]
        cons = [
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
        ]
        for st in starts:
            ss = exact_score(st, n)
            t0 = ss + 1e-8
            z0 = np.concatenate([st, [t0]])
            try:
                res = minimize(slsqp_objective, z0, jac=slsqp_obj_jac, args=(n,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
                if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                    hr = project_capped(res.x[:n], n)
                    if hr is not None:
                        sr = exact_score(hr, n)
                        if sr < best_s - 1e-12:
                            best_s = sr
                            best_h = hr.copy()
                    for a in (0.25, 0.5, 0.75):
                        mix = project_capped((1.0 - a) * st + a * res.x[:n], n)
                        if mix is not None:
                            sm = exact_score(mix, n)
                            if sm < best_s - 1e-12:
                                best_s = sm
                                best_h = mix.copy()
            except Exception:
                pass
    if n <= 128:
        hi = 12
        h_ref, s_ref = smooth_refine(best_h, n, rng, hi)
        if s_ref < best_s - 1e-12:
            best_s = s_ref
            best_h = h_ref
        for st in starts[:3]:
            if np.max(np.abs(st - best_h)) > 1e-10:
                hr2, sr2 = smooth_refine(st, n, rng, hi)
                if sr2 < best_s - 1e-12:
                    best_s = sr2
                    best_h = hr2
    elif n <= 512:
        hi = 20 if n <= 256 else 12
        for st in starts:
            hr2, sr2 = smooth_refine(st, n, rng, hi)
            if sr2 < best_s - 1e-12:
                best_s = sr2
                best_h = hr2
    if not is_feasible(best_h, n):
        best_h = fb.copy()
        best_s = fb_score
    fin = exact_score(best_h, n)
    if not np.isfinite(fin):
        best_h = fb.copy()
        fin = fb_score
    else:
        best_s = fin
    return best_h.astype(np.float64), float(best_s), int(n)
