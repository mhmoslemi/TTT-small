        else:

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(4):
        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                a = min(r, 1.0 - h[i])
                h[i] += a; r -= a
        else:
            for i in range(n):
                if r >= 0: break
                a = min(-r, h[i])
                h[i] -= a; r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0+1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-8:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0+1e-10) and
            abs(np.sum(h) - n/2.0) <= 1e-8)

def exact_score(h, n):
    try:
        if not is_feasible(h, n):
            return float('inf')
        v = float(np.max(np.correlate(h, 1.0-h, mode='full') * (2.0/n)))
        return v if np.isfinite(v) else float('inf')
    except Exception:
        return float('inf')

def lag_data(h, n):
    dx = 2.0/n
    q = np.zeros(2*n-1)
    G = np.zeros((2*n-1, n))
    q[n-1] = dx * np.dot(h, 1.0-h)
    G[n-1] = dx*(1.0-2.0*h)
    for r in range(1, n):
        m = n - r
        ip = n-1-r
        q[ip] = dx * np.dot(h[:m], 1.0-h[r:])
        G[ip, :m] += dx*(1.0-h[r:r+m])
        G[ip, r:r+m] -= dx*h[:m]
        im = n-1+r
        q[im] = dx * np.dot(h[r:], 1.0-h[:m])
        G[im, r:r+m] += dx*(1.0-h[:m])
        G[im, :m] -= dx*h[r:r+m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None
    return q, G

def try_add(pool, v, thresh=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v-s)) < thresh:
            return
    pool.append(v)

def make_cosine(n, k, phase):
    t = np.arange(n)
    return 0.5 + 0.35*np.cos(2.0*math.pi*k*t/n + phase)

def build_pool(fb, n, rng, size):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0-fb, n))
    try_add(pool, project_capped(1.0-fb[::-1], n))
    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2: blk[n//2] = 0.5
    try_add(pool, project_capped(blk, n))
    alt = np.array([1.0 if i%2==0 else 0.0 for i in range(n)])
    try_add(pool, project_capped(alt, n))
    p = rng.permutation(n); vb = np.zeros(n); vb[p[:n//2]] = 1.0
    if n%2: vb[p[n//2]] = 0.5
    try_add(pool, project_capped(vb, n))
    for sd in (0.06, 0.18):
        try_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    try_add(pool, project_capped(rng.uniform(0,1,n), n))
    for k in (1,2,3):
        for ph in (0.0, math.pi/2):
            try_add(pool, project_capped(make_cosine(n, k, ph), n))
    try_add(pool, project_capped(np.roll(fb, n//3), n))
    try_add(pool, project_capped(np.roll(fb, n//2), n))
    while len(pool) < size:
        pool.append(fb.copy())
    return pool[:size]

def select_starts(pool, scores, max_starts):
    n = len(pool[0])
    order = np.argsort(scores)
    sel = [0]  # fallback always included
    sel_dists = [0.0]
    for idx in order:
        if len(sel) >= max_starts:
            break
        if idx == sel[0]:
            continue
        ok = True
        for j in sel:
            if np.max(np.abs(pool[idx]-pool[j])) < 0.05:
                ok = False; break
        if ok:
            sel.append(int(idx))
    for idx in order:
        if len(sel) >= max_starts:
            break
        if int(idx) not in sel:
            sel.append(int(idx))
    return sel

def solve_mm(h0, n, max_iters, solver_fn, solver_name):
    dx = 2.0/n
    L = 2.0*dx
    h = h0.copy()
    sc = exact_score(h, n)
    best_h = h.copy()
    best_sc = sc
    small_streak = 0
    for it in range(max_iters):
        res = lag_data(h, n)
        if res is None:
            break
        q, G = res
        d_val = solver_fn(q, G, h, L)
        if d_val is None:
            break
        d_val = np.asarray(d_val, dtype=np.float64)
        if d_val.shape != (n,) or not np.all(np.isfinite(d_val)):
            break
        dm = np.max(np.abs(d_val))
        if dm < 1e-8:
            break
        best_cand = None
        best_cand_sc = sc
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = project_capped(h + alpha*d_val, n)
            if cand is None:
                continue
            csc = exact_score(cand, n)
            if csc < best_cand_sc - 1e-12:
                best_cand = cand
                best_cand_sc = csc
        if best_cand is None:
            break
        red = sc - best_cand_sc
        h = best_cand
        sc = best_cand_sc
        if sc < best_sc - 1e-12:
            best_h = h.copy()
            best_sc = sc
        if red < 1e-10:
            small_streak += 1
            if small_streak >= 3:
                break
        else:
            small_streak = 0
        if np.max(np.abs(h - best_h)) < 1e-8 and red < 1e-10:
            break
    return best_h, best_sc
