import numpy as np
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    if z.shape != (n,):
        return None
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):

        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))

def build_directions(n):
    dirs = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        L = b - a
        if L < 2:
            continue
        L0 = L // 2
        L1 = L - L0
        if L1 == 0:
            L1 = 1
            L0 = L - 1
        v = np.zeros(n)
        v[a:a + L0] = L1
        v[a + L0:b] = -L0
        dirs.append(v)
        # Push right first so left is processed first (coarse-to-fine, left-to-right)
        stack.append((a + L0, b))
        stack.append((a, a + L0))
    return dirs

def line_interval(h, v):

    hi = 1e18
    pos = v > 0
    neg = v < 0
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / v[neg])))
        hi = min(hi, float(np.min(-h[neg] / v[neg])))
    if not (np.isfinite(lo) and np.isfinite(hi)):

    if lo > hi + 1e-14:

    # Ensure 0 is in [lo, hi]
    if lo > 1e-12:

    if hi < -1e-12:

def line_candidates(a, b, c, lo, hi):

                    t2 = C / q
                else:

                for t in (t1, t2):

    # Deduplicate

    deduped = []

    if 0.0 not in deduped:

def eval_envelope(a, b, c, t_vals):

    # t_vals: array of t values
    # Return array of G(t) for each t in t_vals
    # G(t) = max_k(a[k]*t^2 + b[k]*t + c[k])
    # Process in chunks
    results = np.zeros(len(t_vals))
    chunk = 256
    for start in range(0, len(t_vals), chunk):
        end = min(start + chunk, len(t_vals))
        tv = t_vals[start:end]
        # Shape: (chunk, m)
        vals = a[None, :] * (tv[:, None] ** 2) + b[None, :] * tv[:, None] + c[None, :]
        results[start:end] = np.max(vals, axis=1)
    return results

def line_search(h, current_score, v, n):
    interval = line_interval(h, v)
    if interval is None:
        return h.copy(), current_score
    lo, hi = interval
    dx = 2.0 / n
    c_arr = dx * np.correlate(h, 1.0 - h, mode='full')
    a_arr = -dx * np.correlate(v, v, mode='full')
    b_arr = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
    if not (np.all(np.isfinite(a_arr)) and np.all(np.isfinite(b_arr)) and np.all(np.isfinite(c_arr))):
        return h.copy(), current_score
    cands = line_candidates(a_arr, b_arr, c_arr, lo, hi)
    if len(cands) == 0:

    tv = np.array(cands, dtype=np.float64)
    pred = eval_envelope(a_arr, b_arr, c_arr, tv)
    if not np.all(np.isfinite(pred)):
        return h.copy(), current_score
    # Select top 8 by predicted score, plus forced lo, hi, 0
    order = np.argsort(pred, kind='stable')
    selected = set(order[:8].tolist())
    # Force lo, hi, 0 indices
    for target in (lo, hi, 0.0):
        idx = int(np.argmin(np.abs(tv - target)))
        selected.add(idx)
    best_h = h.copy()
    best_s = current_score
    for idx in sorted(selected):
        t = tv[idx]
        z = h + t * v
        proj = project_capped(z, n)
        if proj is None:
            continue
        sc = exact_score(proj, n)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = proj.copy()
    return best_h, best_s
