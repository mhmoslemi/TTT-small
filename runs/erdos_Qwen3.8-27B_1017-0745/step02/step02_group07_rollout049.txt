def is_feasible(h,n):
    h=np.asarray(h,dtype=np.float64)
    if h.shape != (n,): return False
    if not np.all(np.isfinite(h)): return False
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0+1e-8: return False
    return abs(float(np.sum(h)) - 0.5*float(n)) <= 1e-8

def exact_score(h,n):
    if not is_feasible(h,n): return float('inf')
    c=np.correlate(h,1.0-h,mode='full')*(2.0/float(n))
    if not np.all(np.isfinite(c)): return float('inf')
    m=float(np.max(c))
    return m if np.isfinite(m) else float('inf')

import numpy as np
import math
def is_feasible(h,n):
    h=np.asarray(h,dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0+1e-8: return False
    return abs(float(np.sum(h)) - 0.5*float(n)) <= 1e-8
def exact_score(h,n):

    h=np.asarray(h,dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0+1e-8: return False
    return abs(float(np.sum(h)) - 0.5*float(n)) <= 1e-8

def exact_score(h,n):
    if not is_feasible(h,n): return float('inf')
    c=np.correlate(h,1.0-h,mode='full')*(2.0/float(n))
    if not np.all(np.isfinite(c)): return float('inf')
    m=float(np.max(c)); return m if np.isfinite(m) else float('inf')

def project_capped(z,n):
    if n <= 0: return None
    z=np.asarray(z,dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1e6,1e6); s=0.5*float(n)
    lo=float(np.min(z))-1.0; hi=float(np.max(z)); span=hi-lo
    if not np.isfinite(span) or span < 1e-15: span=1.0
    iters=int(min(80,max(30,math.ceil(math.log2((span+1.0)/1e-12)))))
    for _ in range(iters):
        mid=0.5*(lo+hi); sm=float(np.sum(np.clip(z-mid,0.0,1.0)))
        if sm > s: lo=mid
        else: hi=mid
    u=np.clip(z-0.5*(lo+hi),0.0,1.0)
    for _ in range(n+5):
        r=s-float(np.sum(u))
        if abs(r) < 1e-12: break
        if r > 0.0:
            i=int(np.argmax(1.0-u)); u[i]=min(1.0,u[i]+r)
        else:
            i=int(np.argmax(u)); u[i]=max(0.0,u[i]+r)
    if not is_feasible(u,n):
        u=np.clip(u,0.0,1.0)
        for _ in range(n+5):
            r=s-float(np.sum(u))
            if abs(r) < 1e-12: break
            if r > 0.0:
                i=int(np.argmax(1.0-u)); u[i]=min(1.0,u[i]+r)
            else:
                i=int(np.argmax(u)); u[i]=max(0.0,u[i]+r)
    return u if is_feasible(u,n) else None

def cosine_basis(n):
    p=min(n-1,48)
    if p <= 0: return np.zeros((n,0),dtype=np.float64)
    i=np.arange(n,dtype=np.float64); k=np.arange(1,p+1,dtype=np.float64)
    return (math.sqrt(2.0/float(n))*np.cos(math.pi*(i[:,None]+0.5)*k[None,:]/float(n))).astype(np.float64,copy=False)

def decode_candidate(base,U,x,n):
    x=np.asarray(x,dtype=np.float64); p=U.shape[1]
    if x.shape != (p,) or not np.all(np.isfinite(x)): return None
    try: cand=base+U@x
    except Exception: return None
    if np.all(cand >= -1e-10) and np.all(cand <= 1.0+1e-10) and abs(float(np.sum(cand))-0.5*float(n)) <= 1e-8:
        return cand
    return project_capped(cand,n)

def add_pool(pool,v,n,thr=1e-8):
    if v is None: return
    for s in pool:
        if np.max(np.abs(v-s)) < thr: return
    pool.append(v)

def make_binary(rng,n):
    v=np.zeros(n,dtype=np.float64)
    if n==0: return v
    p=rng.permutation(n); v[p[:n//2]]=1.0
    if n%2: v[p[n//2]]=0.5
    return v

def make_block(n):
    v=np.zeros(n,dtype=np.float64); v[:n//2]=1.0
    if n%2: v[n//2]=0.5
    return v

def make_alt(n):
    idx=np.arange(n); return np.where(idx%2==0,1.0,0.0).astype(np.float64)

def make_cos(n,k,phase):
    x=np.arange(n,dtype=np.float64); return 0.5+0.35*np.cos(2.0*math.pi*k*x/float(n)+phase)

def legendre_candidate(n):
    if n < 3 or n % 2 == 0: return None
    r=int(math.isqrt(n))
    for d in range(3,r+1,2):
        if n % d == 0: return None
    v=np.zeros(n,dtype=np.float64); v[0]=0.5
    for i in range(1,n): v[i]=1.0 if pow(i,n-2,n)==1 else 0.0
    return v

def resample_h(h,m):
    h=np.asarray(h,dtype=np.float64).ravel(); n=h.size
    if n <= 0 or m <= 0: return None
    x_old=(np.arange(n,dtype=np.float64)+0.5)*2.0/float(n)
    x_new=(np.arange(m,dtype=np.float64)+0.5)*2.0/float(m)
    return project_capped(np.interp(x_new,x_old,h,left=float(h[0]),right=float(h[-1])),m)

def build_pool(fb,n,rng,cap=16):
    pool=[]
    leg=legendre_candidate(n)
    raws=[fb,fb[::-1],1.0-fb,1.0-fb[::-1],np.full(n,0.5),make_block(n),make_alt(n),make_binary(rng,n),np.roll(fb,n//3),np.roll(fb,n//2),make_cos(n,1,0.0),make_cos(n,2,0.0),leg]
    if leg is not None:
        raws.append(np.roll(leg,1)); raws.append(np.roll(leg,n//3))
    for v in raws:
        if v is None: continue
        hp=project_capped(v,n)
        if hp is not None: add_pool(pool,hp,n)
    attempts=0
    while len(pool)<cap and attempts<100:
        attempts+=1
        if pool: base=pool[int(rng.integers(len(pool)))]; scale=0.08+0.20*float(rng.random())
        else: base=np.full(n,0.5); scale=0.1
        hp=project_capped(base+rng.normal(0.0,scale,n),n)
        if hp is None: hp=project_capped(rng.uniform(0.0,1.0,n),n)
        add_pool(pool,hp,n)
    while len(pool)<cap: pool.append(pool[0].copy() if pool else np.full(n,0.5))
    return pool

def select_bases(pool,scores,R):
    if R <= 0: return []
    if len(pool) == 1: return [0]*R
    order=np.argsort(scores,kind='stable'); sel=[0]
    for idx in order:
        if len(sel) >= R: break
        k=int(idx)
        if k == 0: continue
        ok=True
        for b in sel:
            if np.max(np.abs(pool[k]-pool[b])) <= 0.05: ok=False; break
        if ok: sel.append(k)
    for idx in order:
        if len(sel) >= R: break
        k=int(idx)
        if k not in sel: sel.append(k)
    while len(sel) < R: sel.append(0)
    return sel

def cma_run(base,U,init_sig,gens,rng,best_h,best_s,n):
    p=U.shape[1]
    if p <= 0 or gens <= 0: return best_h,best_s
    lam=max(6,int(4+3*math.log(max(1,p)))); mu=max(1,lam//2)
    jj=np.arange(1,mu+1,dtype=np.float64); w=np.log(mu+0.5)-np.log(jj); sw=float(np.sum(w))
    if sw <= 0.0 or not np.isfinite(sw): w=np.ones(mu); sw=float(mu)
    w=w/sw; me=1.0/float(np.sum(w*w))
    cs=(me+2.0)/(p+me+5.0)
    ds=1.0+2.0*max(0.0,math.sqrt(max(0.0,(me-1.0)/(p+1.0)))-1.0)+cs
    cc=(4.0+me/p)/(p+4.0+2.0*me/p)
    c1=2.0/((p+1.3)**2+me)
    cmu=2.0*(me-2.0+1.0/me)/((p+2.0)**2+me); cmu=max(0.0,min(1.0-c1,cmu))
    chi=math.sqrt(float(p))*(1.0-1.0/(4.0*p)+1.0/(21.0*p*p))
    m=np.zeros(p); C=np.eye(p); ps=np.zeros(p); pc=np.zeros(p); sig=float(init_sig); t=0
    for _ in range(gens):
        C=0.5*(C+C.T)
        try:
            ev,evec=np.linalg.eigh(C)
            if not (np.all(np.isfinite(ev)) and np.all(np.isfinite(evec))): raise Exception
            ev=np.clip(ev,1e-10,1e4)
        except Exception:
            ev=np.ones(p); evec=np.eye(p)
        se=np.sqrt(ev); ys=[]; ss=[]
        for __ in range(lam):
            z=rng.standard_normal(p); y=evec@(se*z); x=m+sig*y
            h=decode_candidate(base,U,x,n)
            if h is None: sc=float('inf')
            else:
                sc=exact_score(h,n)
                if sc < best_s - 1e-12: best_s=sc; best_h=h.copy()
            ys.append(y); ss.append(sc)
        sa=np.array(ss,dtype=np.float64)
        if int(np.sum(np.isfinite(sa))) < mu: break
        idxs=np.argsort(sa,kind='stable'); selx=idxs[:mu]
        yw=np.zeros(p); rm=np.zeros((p,p))
        for a in range(mu):
            y=ys[int(selx[a])]; yw += float(w[a])*y; rm += float(w[a])*np.outer(y,y)
        m_new=m+sig*yw
        ps=(1-cs)*ps + math.sqrt(max(0.0,cs*(2-cs)*me))*(evec@((evec.T@yw)/se))
        t += 1
        denom=math.sqrt(max(1e-300,1.0-(1-cs)**(2*t)))
        hf=1.0 if float(np.linalg.norm(ps))/denom < (1.4+2.0/(p+1))*chi else 0.0
        pc=(1-cc)*pc + hf*math.sqrt(max(0.0,cc*(2-cc)*me))*yw
        C=(1-c1-cmu)*C + c1*(np.outer(pc,pc)+(1-hf)*cc*(2-cc)*C) + cmu*rm
        C=0.5*(C+C.T)
        eps=(cs/ds)*(float(np.linalg.norm(ps))/chi - 1.0); eps=max(-1.0,min(1.0,eps))
        sig=max(1e-5,min(1.0,sig*math.exp(eps))); m=m_new
        if not (np.all(np.isfinite(m)) and np.all(np.isfinite(C)) and np.all(np.isfinite(ps)) and np.all(np.isfinite(pc))):
            m=np.where(np.isfinite(m),m,0.0); C=np.eye(p); ps=np.zeros(p); pc=np.zeros(p); sig=max(1e-5,0.5*float(init_sig))
        if sig <= 1e-5: break
    return best_h,best_s

def threshold_mass(h,n):
    v=np.asarray(h,dtype=np.float64).ravel(); idx=np.argsort(v)[::-1]; w=np.zeros(n,dtype=np.float64)
    r=n//2
    if r > 0: w[idx[:r]]=1.0
    if n%2 and r < n: w[idx[r]]=0.5
    return w

def binary_polish(h,n,rng,iters):
    v=threshold_mass(h,n); s=exact_score(v,n)
    if not np.isfinite(s): return h,exact_score(h,n)
    best_v=v.copy(); best_s=s
    for it in range(iters):
        i=int(rng.integers(n)); j=int(rng.integers(n))
        if i == j or v[i] == v[j]: continue
        trial=v.copy(); a=trial[i]; b=trial[j]; trial[i]=b; trial[j]=a
        ts=exact_score(trial,n)
        if np.isfinite(ts) and ts < s - 1e-12:
            v=trial; s=ts
            if s < best_s - 1e-12: best_s=s; best_v=trial.copy()
        elif it % 250 == 249:
            cand=threshold_mass(best_v + rng.normal(0.0,0.25,n),n); cs=exact_score(cand,n)
            if np.isfinite(cs) and cs < s - 1e-12:
                v=cand; s=cs
                if cs < best_s - 1e-12: best_s=cs; best_v=cand.copy()
    return best_v,best_s

def optimize_n(n,start,rng,scale):
    try: n=int(n)
    except Exception: n=47
    if n <= 0 or n > 999: n=47
    if start is None: fb=np.full(n,0.5)
    else:
        st=np.asarray(start,dtype=np.float64).ravel()
        if st.size == n and is_feasible(st,n): fb=st.copy()
        else:
            fb=project_capped(st,n) if st.size == n else None
            if fb is None: fb=np.full(n,0.5)
    if n == 1: return fb, exact_score(fb,1)
    U=cosine_basis(n)
    pool=build_pool(fb,n,rng,16)
    scores=np.array([exact_score(v,n) for v in pool],dtype=np.float64)
    best_h=fb.copy(); best_s=exact_score(fb,n)
    bfactor=1.0 if n <= 128 else 0.35
    for i in range(len(pool)):
        if scores[i] < best_s - 1e-12: best_h=pool[i].copy(); best_s=float(scores[i])
    top_idx=int(np.argmin(scores))
    vt,bts=binary_polish(pool[top_idx],n,rng,max(80,int(300*scale*bfactor)))
    if np.isfinite(bts) and bts < best_s - 1e-12: best_h,best_s=vt,bts
    if U.shape[1] > 0 and scale > 0.0:
        if n <= 64: R,G,PG=5,160,70
        elif n <= 128: R,G,PG=4,90,40
        elif n <= 256: R,G,PG=3,45,20
        elif n <= 600: R,G,PG=2,20,10
        else: R,G,PG=1,10,5
        if scale < 1.0: R=max(2,R-1)
        G=max(5,int(G*scale)); PG=max(3,int(PG*scale))
        p=U.shape[1]
        sig0=0.10*math.sqrt(float(n)/float(p)); sig0=max(0.04,min(0.35,sig0))
        sel=select_bases(pool,scores,R)
        for bidx in sel: best_h,best_s=cma_run(pool[bidx],U,sig0,G,rng,best_h,best_s,n)
        best_h,best_s=cma_run(best_h,U,0.5*sig0,PG,rng,best_h,best_s,n)
    iters=max(200,int(1800*scale*bfactor))
    vb,sb=binary_polish(best_h,n,rng,iters)
    if np.isfinite(sb) and sb < best_s - 1e-12: best_h,best_s=vb,sb
    final=project_capped(best_h,n)
    if final is not None:
        fs=exact_score(final,n)
        if np.isfinite(fs) and fs <= best_s + 1e-12: best_h,best_s=final,fs
    if not is_feasible(best_h,n) or not np.isfinite(best_s):
        best_h=fb.copy(); best_s=exact_score(fb,n)
    return best_h.astype(np.float64), float(best_s)
