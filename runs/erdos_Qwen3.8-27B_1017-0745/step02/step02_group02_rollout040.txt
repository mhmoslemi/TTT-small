            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - S) > 5e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 5e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_grad(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    p = max(s, 0)
    r = max(-s, 0)
    m = n - abs(s)
    g[p:p + m] += dx * (1.0 - h[r:r + m])
    g[r:r + m] -= dx * h[p:p + m]
    return g

def solve_lp(h, n, q, active_lags, Delta):
    dx = 2.0 / n
    K = len(active_lags)
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, s in enumerate(active_lags):
        g = lag_grad(h, n, s)
        A_ub[i, :n] = g / dx
        A_ub[i, n] = -1.0
        b_ub[i] = -q[s + (n - 1)] / dx
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(max(-Delta, -h[i]), min(Delta, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success or res.x is None or res.x.shape != (n + 1,):
        return None
    d = res.x[:n]
    if not np.all(np.isfinite(d)) or abs(np.sum(d)) > 1e-7 * max(1, n):
        return None
    return d

def local_search(h0, n, max_iter, Delta0):
    h = h0.copy()
    f = exact_score(h, n)
    best_h, best_f = h.copy(), f
    Delta, Dmax, Dmin = Delta0, 0.5, 1e-6
    lp_fail, rej_fail = 0, 0
    K, gap_f = 12, 0.05
    for _ in range(max_iter):
        q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        f = float(np.max(q))
        gap = max(1e-10, gap_f * Delta)
        idx = np.where(q >= f - gap)[0]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(q))])
        if len(idx) > K:
            idx = idx[np.argsort(q[idx])[::-1]][:K]
        active = [int(k) - (n - 1) for k in idx]
        d = solve_lp(h, n, q, active, Delta)
        if d is None:
            Delta = max(0.5 * Delta, Dmin)
            lp_fail += 1
            if lp_fail >= 5 or Delta <= Dmin:
                break
            continue
        lp_fail = 0
        if np.max(np.abs(d)) <= 1e-10:
            Delta = max(0.5 * Delta, Dmin)
            if Delta <= Dmin:
                break
            continue
        pred_max = 0.0
        for s in active:
            g = lag_grad(h, n, s)
            pred_max = max(pred_max, q[s + (n - 1)] + float(g @ d))
        if f - pred_max <= 1e-12:
            Delta = max(0.5 * Delta, Dmin)
            if Delta <= Dmin:
                break
            continue
        best_trial, best_tf = None, f
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            trial = project_capped(h + alpha * d, n)
            if trial is None:
                continue
            tf = exact_score(trial, n)
            if tf < best_tf - 1e-12:
                best_trial, best_tf = trial, tf
        if best_trial is None:
            Delta = max(0.5 * Delta, Dmin)
            rej_fail += 1
            if rej_fail >= 8 or Delta <= Dmin:
                break
            continue
        rej_fail = 0
        e = best_trial - h
        actual = f - best_tf
        pred_p = 0.0
        for s in active:
            g = lag_grad(h, n, s)
            pred_p = max(pred_p, q[s + (n - 1)] + float(g @ e))
        pred_dec = f - pred_p
        rho = actual / pred_dec if pred_dec > 1e-12 else 0.0
        h, f = best_trial, best_tf
        if f < best_f - 1e-12:
            best_f, best_h = f, h.copy()
        if rho < 0.1:
            Delta = max(0.5 * Delta, Dmin)
        elif rho > 0.75 and np.max(np.abs(e)) >= 0.8 * Delta:
            Delta = min(1.5 * Delta, Dmax)
    return best_h, best_f

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1], np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n) + 0.5
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), 0.0, 1.0))
    cands.append(np.clip(0.5 + 0.35 * np.sin(4.0 * np.pi * j / n + 0.7), 0.0, 1.0))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        if any(np.max(np.abs(p - s)) < 1e-8 for s in pool):
            continue
        pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb_s = exact_score(fb, n)
    gb, gs = fb.copy(), fb_s
    const = np.full(n, 0.5)
    cs = exact_score(const, n)
    if cs < gs - 1e-12:
        gb, gs = const.copy(), cs
    if n == 1:
        return fb, fb_s, 1
    if n <= 64:
        n_st, n_it, n_rr = 7, 80, 3
    elif n <= 128:
        n_st, n_it, n_rr = 4, 45, 2
    elif n <= 256:
        n_st, n_it, n_rr = 3, 25, 1
    elif n <= 512:
        n_st, n_it, n_rr = 2, 12, 1
    else:
        n_st, n_it, n_rr = 2, 6, 0
    starts = build_starts(fb, n, rng, n_st)
    for st in starts:
        try:
            ss = exact_score(st, n)
            if ss < gs - 1e-12:
                gs, gb = ss, st.copy()
            bh, bf = local_search(st, n, n_it, 0.25)
            if bf < gs - 1e-12:
                gs, gb = bf, bh.copy()
        except Exception:
            continue
    if n_rr > 0:
        if n <= 64:
            sds = [0.12, 0.22, 0.08] if gs > fb_s - 1e-5 else [0.04, 0.10, 0.18]
        elif n <= 256:
            sds = [0.10, 0.18] if gs > fb_s - 1e-5 else [0.04, 0.08]
        else:
            sds = [0.08]
        for sd in sds[:n_rr]:
            try:
                pert = project_capped(gb + rng.normal(0, sd, n), n)
                if pert is None:
                    continue
                bh, bf = local_search(pert, n, n_it, 0.25)
                if bf < gs - 1e-12:
                    gs, gb = bf, bh.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    if sf > fb_s:
        final, sf = fb, fb_s
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
