    try_add(starts, to_units(project_capped(rng.uniform(0, 1, n), n), n), n)
    try_add(starts, np.full(n, M // 2, dtype=np.int64), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(starts, to_units(project_capped(alt, n), n), n)
    try_add(starts, to_units(project_capped(fallback + rng.normal(0, 0.20, n), n), n), n)
    try_add(starts, to_units(project_capped(fallback + rng.normal(0, 0.35, n), n), n), n)
    if n <= 64:
        n_starts, steps = 10, 700
    elif n <= 128:
        n_starts, steps = 6, 350
    elif n <= 256:
        n_starts, steps = 4, 150
    elif n <= 512:
        n_starts, steps = 2, 60
    else:
        n_starts, steps = 1, 16
    starts = starts[:n_starts]
    gb = fallback.copy()
    gs = fb_score
    for st in starts:
        ss = exact_score(units_to_h(st), n)
        if ss < gs - 1e-12:
            gs = ss
            gb = units_to_h(st).copy()
    for st in starts:
        try:
            bu, bs = anneal_one(st, n, rng, steps)
            if bs < float('inf') and bs < gs - 1e-12:
                bh = units_to_h(bu)
                if is_feasible(bh, n):
                    gs = bs
                    gb = bh.copy()
        except Exception:
            pass
    if not is_feasible(gb, n) or gs > fb_score + 1e-12:
        gb = fallback.copy()
        gs = fb_score
    final_h = np.clip(gb, 0.0, 1.0)
    residual = n / 2.0 - np.sum(final_h)
    if abs(residual) > 1e-10:
        for _ in range(n + 2):
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                idx = int(np.argmax(1.0 - final_h))
                final_h[idx] = min(final_h[idx] + residual, 1.0)
            else:
                idx = int(np.argmax(final_h))
                final_h[idx] = max(final_h[idx] + residual, 0.0)
            residual = n / 2.0 - np.sum(final_h)
    final_score = exact_score(final_h, n)
    if final_score == float('inf'):
        final_h = np.full(n, 0.5)
        final_score = exact_score(final_h, n)
    return final_h, final_score, int(n)
