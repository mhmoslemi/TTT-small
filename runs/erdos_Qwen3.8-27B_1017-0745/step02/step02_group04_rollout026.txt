    for s in pool:
        if np.max(np.abs(h - s)) < 1e-8:
            return
    pool.append(h.copy())

def evolve(mu, n, S, rng, lam, ngen, sigma0):
    mu = mu.copy()
    best = mu.copy()
    best_score = exact_score(best, n)
    sigma = sigma0
    P = np.eye(n) - np.ones((n, n)) / n
    if S is not None and n <= 64:
        evec, ev = S
        A = evec @ np.diag(ev) @ evec.T
        A = 0.5 * (A + A.T)
        tr = np.trace(A)
        if tr <= 1e-12:
            A = P.copy()
        else:
            A = A / tr * (n - 1)
        A = 0.85 * A + 0.15 * P
        tr = np.trace(A)
        if tr > 1e-12:
            A = A / tr * (n - 1)
        A = 0.5 * (A + A.T)
        try:
            ae, av = np.linalg.eigh(A)
            av = np.clip(av, 0, None)
        except Exception:
            ae, av = np.eye(n), np.ones(n)
    else:
        ae, av = None, np.ones(n)
    stall = 0
    for gen in range(ngen):
        pop, scores = [mu], [exact_score(mu, n)]
        if n <= 64 and ae is not None:
            base = ae * np.sqrt(np.maximum(av, 0))[None, :]
            noise = rng.standard_normal((lam, n)) @ base.T
        else:
            noise = rng.standard_normal((lam, n)) * np.sqrt(np.maximum(av, 0))[None, :]
        for i in range(lam):
            z = mu + sigma * noise[i]
            p = project_capped(z, n)
            if p is None:
                p = mu.copy()
            pop.append(p)
            scores.append(exact_score(p, n))
        order = np.argsort(scores)
        q = max(2, min(len(pop) // 4, len(pop)))
        wts = np.array([q - j for j in range(q)], dtype=np.float64)
        wts /= wts.sum()
        new_mu = np.zeros(n)
        for j in range(q):
            new_mu += wts[j] * pop[order[j]]
        new_mu = project_capped(new_mu, n)
        if new_mu is None:
            new_mu = mu.copy()
        for j in range(q):
            if scores[order[j]] < best_score - 1e-12:
                best_score = scores[order[j]]
                best = pop[order[j]].copy()
        if exact_score(new_mu, n) < best_score - 1e-12:
            best_score = exact_score(new_mu, n)
            best = new_mu.copy()
            stall = 0
        else:
            stall += 1
        if n <= 64 and ae is not None:
            E = np.zeros((n, n))
            for j in range(q):
                d = pop[order[j]] - new_mu
                E += wts[j] * np.outer(d, d)
            E = E / max(sigma * sigma, 1e-8)
            E = P @ E @ P
            trE = np.trace(E)
            if trE > 1e-10:
                E = E / trE * (n - 1)
                old_A = ae @ np.diag(av) @ ae.T
                A_new = 0.8 * old_A + 0.2 * E + 1e-4 * P
                try:
                    ae2, av2 = np.linalg.eigh(A_new)
                    av2 = np.clip(av2, 0, None)
                    tr2 = np.sum(av2)
                    if tr2 > 1e-12:
                        av2 = av2 * (n - 1) / tr2
                    ae, av = ae2, av2
                except Exception:
                    pass
        else:
            var = np.zeros(n)
            for j in range(q):
                d = (pop[order[j]] - new_mu) / max(sigma, 1e-4)
                var += wts[j] * d * d
            mn = np.mean(var)
            if mn > 1e-12:
                var = var / mn
            var = np.clip(var, 0.05, 20.0)
            var = var / max(np.mean(var), 1e-12)
            av = 0.8 * av + 0.2 * var
        n_suc = sum(1 for i in range(1, len(scores)) if scores[i] < scores[0] - 1e-12)
        p_suc = n_suc / max(lam, 1)
        if p_suc > 0.25:
            sigma = min(0.25, 1.12 * sigma)
        elif p_suc < 0.08:
            sigma = max(5e-4, 0.85 * sigma)
        mu = new_mu
        if stall >= 12 and sigma <= 0.002:
            break
    return best

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb_score = float(exact_score(fallback, n))
    S_sdp, sdp_mean = None, None
    if 2 <= n <= 64:
        sdp_mean, S_sdp = solve_sdp(n)
    pool = []
    _add(pool, fallback.copy(), n)
    _add(pool, project_capped(fallback[::-1].copy(), n), n)
    _add(pool, project_capped(1.0 - fallback, n), n)
    _add(pool, project_capped((1.0 - fallback)[::-1].copy(), n), n)
    _add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    _add(pool, project_capped(alt, n), n)
    P2 = max(2, n // 3)
    cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P2)
    _add(pool, project_capped(cos_v, n), n)
    _add(pool, project_capped(fallback + rng.normal(0, 0.04, n), n), n)
    _add(pool, project_capped(fallback + rng.normal(0, 0.12, n), n), n)
    _add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    if sdp_mean is not None:
        _add(pool, project_capped(sdp_mean, n), n)
        if S_sdp is not None:
            evec, ev = S_sdp
            base = evec * np.sqrt(np.maximum(ev, 0))[None, :]
            for gam in [0.5, 1.0, 1.5]:
                for _ in range(4):
                    xi = rng.standard_normal(n)
                    _add(pool, project_capped(sdp_mean + gam * (base @ xi), n), n)
    pool = pool[:32]
    best_h, best_s = fallback.copy(), fb_score
    for st in pool:
        sc = float(exact_score(st, n))
        if sc < best_s - 1e-12:
            best_s, best_h = sc, st.copy()
    if n <= 64:
        lam_ev, ngen_ev, n_restart = 20, 60, 4
    elif n <= 128:
        lam_ev, ngen_ev, n_restart = 16, 35, 3
    elif n <= 256:
        lam_ev, ngen_ev, n_restart = 12, 20, 2
    else:
        lam_ev, ngen_ev, n_restart = 8, 12, 1
    pool_scores = [(float(exact_score(st, n)), st) for st in pool]
    pool_scores.sort(key=lambda x: x[0])
    sigmas = [0.04, 0.09, 0.18, 0.07]
    selected = []
    if best_h is not None:
        selected.append(best_h.copy())
    for sc, st in pool_scores:
        if len(selected) >= n_restart:
            break
        far = all(np.max(np.abs(st - s2)) >= 1e-3 for s2 in selected)
        if far:
            selected.append(st.copy())
    for i in range(len(selected)):
        si = sigmas[i % len(sigmas)]
        try:
            res = evolve(selected[i], n, S_sdp, rng, lam_ev, ngen_ev, si)
            if res is not None:
                rs = float(exact_score(res, n))
                if rs < best_s - 1e-12:
                    best_s, best_h = rs, res.copy()
        except Exception:
            pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = float(exact_score(final, n))
    if fs > fb_score + 1e-12:
        final, fs = fallback.copy(), fb_score
    if fs == float('inf'):
        final = project_capped(np.full(n, 0.5), n)
        if final is None:
            final = np.full(n, 0.5)
        fs = float(exact_score(final, n))
    return final, fs, int(n)
