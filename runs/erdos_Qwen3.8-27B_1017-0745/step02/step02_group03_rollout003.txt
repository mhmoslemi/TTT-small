            break
    return best, bs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    if sd < 0:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_k(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)

    try:
        mi = max(1, min(int(kwargs.get('max_iter', 60)), 120))
    except Exception:
        mi = 60
    if n > 512:
        mi = min(mi, 12)
    elif n > 128:
        mi = min(mi, 30)

    tau_sched = [0.02, 0.008, 0.003, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.4 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, fallback + 0.05 * np.sin(4.0 * np.pi * tt / n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    nmax = 8 if n <= 64 else (6 if n <= 128 else 4)
    starts = starts[:nmax]

    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = st.copy(), sc
        try:
            lb, lbs = cont_opt(st, n, mi, tau_sched)
            if np.isfinite(lbs) and lbs < best_score - 1e-12:
                best, best_score = lb.copy(), lbs
        except Exception:
            pass

    # Basin hops
    hop_sdrs = (0.02, 0.06, 0.12)
    for sv in hop_sdrs:
        try:
            pert = project_k(best + rng.normal(0, sv, n), n)
            if pert is None:
                continue
            ps = exact_score(pert, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = pert.copy(), ps
            lb, lbs = cont_opt(pert, n, mi, tau_sched)
            if np.isfinite(lbs) and lbs < best_score - 1e-12:
                best, best_score = lb.copy(), lbs
        except Exception:
            pass

    # Pairwise polish for small n
    if n <= 80:
        try:
            pol, psc = pairwise_polish(best, n, 2)
            fp = project_k(pol, n)
            if fp is not None:
                fsc = exact_score(fp, n)
                if np.isfinite(fsc) and fsc < best_score - 1e-13:
                    best, best_score = fp, fsc
        except Exception:
            pass

    # Final
    final = project_k(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
