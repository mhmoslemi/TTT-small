            if np.isfinite(pps) and pps < best_score - 1e-12:
                best, best_score = ph, pps
    fs = exact_score(best, n)
    if not np.isfinite(fs):
        best = uniform.copy()
        fs = exact_score(uniform, n)
    return best, float(fs), int(n)
