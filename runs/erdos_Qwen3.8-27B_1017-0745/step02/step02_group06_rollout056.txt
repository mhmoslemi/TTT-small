    if h is None or not is_feasible(h, n):
        return None
    return h

def try_accept(h_sol, inc, sc, n):
    best, bs = inc.copy(), sc
    for alpha in [0.25, 0.5, 0.75, 1.0]:
        b = (1.0 - alpha) * inc + alpha * h_sol
        pb = project_capped(b, n)
        if pb is None:
            continue
        sb = exact_score(pb, n)
        if sb < bs - 1e-12:
            bs, best = sb, pb.copy()
    return best, bs

def run(seed=42, **kwargs):
    n, parent = 47, None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 < p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = project_capped(np.clip(p, 0, 1), n)
    except Exception:
        pass
    n = max(n, 2)
    fallback = parent if parent is not None else np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, n_restarts, maxiter = 8, 4, 220
    elif n <= 128:
        n_starts, n_restarts, maxiter = 5, 2, 150
    else:
        n_starts, n_restarts, maxiter = 3, 1, 80
    starts = build_starts(parent, n, rng, n_starts)
    if not any(np.allclose(s, fallback, atol=1e-8) for s in starts):
        starts.insert(0, fallback.copy())
    inc, inc_sc = fallback.copy(), fb_score
    for st in starts[:n_starts + 2]:
        h_sol = solve_ezh(st, n, maxiter)
        if h_sol is None:
            continue
        inc, inc_sc = try_accept(h_sol, inc, inc_sc, n)
    sigma = 0.04
    for _ in range(n_restarts):
        pp, found = None, False
        for _ in range(3):
            pert = inc + rng.normal(0, sigma, n)
            pp = project_capped(pert, n)
            if pp is None or exact_score(pp, n) > 0.5 + 1e-12:
                continue
            if np.allclose(pp, inc, atol=1e-6):
                continue
            found = True
            break
        if not found:
            sigma = min(0.20, 1.5 * sigma)
            continue
        h_sol = solve_ezh(pp, n, maxiter)
        improved = False
        if h_sol is not None:
            nb, ns = try_accept(h_sol, inc, inc_sc, n)
            if ns < inc_sc - 1e-12:
                inc, inc_sc, improved = nb, ns, True
        sigma = max(0.01, 0.7 * sigma) if improved else min(0.20, 1.5 * sigma)
    final = project_capped(inc, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
