def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):

    return float(np.max(c))

def project_capped(z, n):

    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            for i in range(n):
                if r >= 0: break
                a = min(-r, h[i]); h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def helmert_fwd(x, n):
    d = n - 1
    u = np.array([x[k] / math.sqrt((k+1)*(k+2)) for k in range(d)])
    suff = np.zeros(d + 1)
    for i in range(d-1, -1, -1):
        suff[i] = suff[i+1] + u[i]
    y = np.zeros(n)
    y[0] = suff[0]
    for i in range(1, n):
        y[i] = suff[i] - i * x[i-1] / math.sqrt(i * (i+1))

                rmu += w[i] * np.outer(yi, yi)
            else:
                rmu += w[i] * yi * yi
        if fcov:
            Cn = (1 - c1_ - cm_) * C + c1_ * (np.outer(pcn, pcn) + (1 - hs) * cc_ * (2 - cc_) * C) + cm_ * rmu
            Cn = 0.5 * (Cn + Cn.T)
        else:
            Cn = (1 - c1_ - cm_) * C + c1_ * (pcn * pcn + (1 - hs) * cc_ * (2 - cc_) * C) + cm_ * rmu
        ea = np.clip((cs_ / ds_) * (np.linalg.norm(psn) / chi - 1.0), -50, 50)
        sn = np.clip(sig * math.exp(ea), 1e-4, 0.5)
        if not np.all(np.isfinite(psn)) or not np.all(np.isfinite(pcn)):
            break
        if not np.all(np.isfinite(Cn)):
            break
        m, ps, pc, C, sig = mn, psn, pcn, Cn, sn
        ms = ms + 1 if sig <= 1e-4 + 1e-15 else 0
        if ms >= 5:
            break
    hm = latent_to_h(m, n)
    if hm is not None:
        sm = exact_score(hm, n)
        if sm < bs - 1e-12:
            bs, bh = sm, hm.copy()
    return bh, bs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if (np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8)
                    and abs(np.sum(p) - p.size / 2.0) <= 1e-6 * max(1, p.size)):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    ch = np.full(n, 0.5)
    cs_v = exact_score(ch, n)
    sh, ss = ch.copy(), cs_v
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and is_feasible(pp, n):
            ps = exact_score(pp, n)
            parent = pp
            if ps < ss:
                sh, ss = pp.copy(), ps
    if n <= 64:
        ns, Pm, Gm, sf, so, uf = 5, 24, 120, 0.08, 0.14, True
    elif n <= 256:
        ns, Pm, Gm, sf, so, uf = 4, 20, 40, 0.06, 0.10, False
    else:
        ns, Pm, Gm, sf, so, uf = 3, 12, 12, 0.06, 0.08, False
    P = max(8, min(Pm, 4 + int(math.log(max(2, n - 1)) * 3)))
    base = parent if parent is not None else ch
    raw = [base.copy(), base[::-1].copy(), 1.0 - base, np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    raw.append(blk)
    raw.append(base + rng.normal(0, 0.05, n))
    raw.append(base + rng.normal(0, 0.15, n))
    for k_exp in (1, 2):
        ph = rng.uniform(0, 2 * np.pi)
        raw.append(0.5 + 0.3 * np.sin(2 * np.pi * k_exp * np.arange(n) / n + ph))
    raw.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    raw.append(rng.uniform(0, 1, n))
    for sd in (0.30, 0.50):
        raw.append(base + rng.normal(0, sd, n))
    starts = []
    for r in raw:
        p = project_capped(r, n)
        if p is None or not is_feasible(p, n):
            continue
        dup = False
        for s in starts:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if dup:
            continue
        starts.append(p)
        if len(starts) >= ns:
            break
    gb, gs = sh.copy(), ss
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for si, st in enumerate(starts):
        try:
            ch2, cs2 = cma_es(st, n, sf if si == 0 else so, P, Gm, uf, rng)
            if ch2 is not None and cs2 < gs - 1e-12:
                gs, gb = cs2, ch2.copy()
        except Exception:
            pass
    fin = project_capped(gb, n)
    if fin is None or not is_feasible(fin, n):
        fin = sh.copy()
    fs = exact_score(fin, n)
    if not np.isfinite(fs) or fs > ss + 1e-12:
        fin, fs = sh.copy(), exact_score(sh, n)
    return fin.astype(np.float64), float(fs), int(n)
