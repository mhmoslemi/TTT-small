                if r >= 0:
                    break
                a = min(-r, h[i]); h[i] -= a; r += a
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9:
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    return (h.shape == (n,) and bool(np.all(np.isfinite(h))) and
            bool(np.all(h >= -1e-12)) and bool(np.all(h <= 1.0 + 1e-12)) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    v = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    return v if np.isfinite(v) else float('inf')

def try_add_candidate(pool, v, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def make_cosine_basis(n, max_p):
    p = min(n - 1, max_p)
    i = np.arange(n, dtype=np.float64)
    k = np.arange(1, p + 1, dtype=np.float64)
    return np.sqrt(2.0 / n) * np.cos(np.pi * (i[:, None] + 0.5) * k[None, :] / n)

def build_pool(fb, n, rng, max_sz):
    pool = []
    try_add_candidate(pool, fb.copy())
    try_add_candidate(pool, project_capped(fb[::-1], n))
    try_add_candidate(pool, project_capped(1.0 - fb, n))
    try_add_candidate(pool, project_capped(1.0 - fb[::-1], n))
    try_add_candidate(pool, project_capped(np.roll(fb, n // 3), n))
    try_add_candidate(pool, project_capped(np.roll(fb, n // 2), n))
    try_add_candidate(pool, np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add_candidate(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add_candidate(pool, project_capped(alt, n))
    perm = rng.permutation(n)
    bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
    if n % 2:
        bv[perm[n // 2]] = 0.5
    try_add_candidate(pool, project_capped(bv, n))
    try_add_candidate(pool, project_capped(fb + rng.normal(0, 0.06, n), n))
    try_add_candidate(pool, project_capped(fb + rng.normal(0, 0.18, n), n))
    ii = np.arange(n, dtype=np.float64)
    for kk in (1, 2):
        for ph in (0.0, math.pi / 2):
            try_add_candidate(pool, project_capped(0.5 + 0.35 * np.cos(2 * math.pi * kk * ii / n + ph), n))
    while len(pool) < max_sz:
        pool.append(pool[0].copy())
    return pool[:max_sz]

def run_cma_es(base, U, sig0, gens, rng, best_h, best_s, n):
    p = U.shape[1]
    if p < 1:
        return best_h, best_s
    lam = max(6, int(4 + 3 * math.log(p)))
    mu = max(2, lam // 2)
    w = np.array([math.log(mu + 0.5) - math.log(float(j)) for j in range(1, mu + 1)])
    w /= w.sum()
    mu_eff = 1.0 / float(np.sum(w ** 2))
    c_sig = (mu_eff + 2.0) / (p + mu_eff + 5.0)
    d_sig = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (p + 1.0)) - 1.0) + c_sig
    c_c = (4.0 + mu_eff / p) / (p + 4.0 + 2.0 * mu_eff / p)
    c_1 = 2.0 / ((p + 1.3) ** 2 + mu_eff)
    c_mu = max(0.0, min(1.0 - c_1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((p + 2.0) ** 2 + mu_eff)))
    chi_p = math.sqrt(p) * (1.0 - 1.0 / (4.0 * p) + 1.0 / (21.0 * p ** 2))
    m = np.zeros(p)
    C = np.eye(p)
    ev, V = np.linalg.eigh(C)
    sqrt_ev = np.sqrt(np.clip(ev, 1e-10, 1e4))
    p_sig = np.zeros(p)
    p_c = np.zeros(p)
    sig = sig0
    t = 0
    for g in range(gens):
        C = 0.5 * (C + C.T)
        try:
            ev, V = np.linalg.eigh(C)
        except np.linalg.LinAlgError:
            C = np.eye(p); ev = np.ones(p); V = np.eye(p)
            p_sig = np.zeros(p); p_c = np.zeros(p)
        ev = np.clip(ev, 1e-10, 1e4)
        sqrt_ev = np.sqrt(ev)
        xs = np.zeros((lam, p))
        ys = np.zeros((lam, p))
        hs = np.full(lam, float('inf'))
        hv = [None] * lam
        for i in range(lam):
            z = rng.standard_normal(p)
            y = V @ (sqrt_ev * z)
            x = m + sig * y
            xs[i] = x; ys[i] = y
            dec = base + U @ x
            h = project_capped(dec, n)
            if h is not None and is_feasible(h, n):
                hs[i] = exact_score(h, n)
                hv[i] = h
                if hs[i] < best_s - 1e-12:
                    best_s = hs[i]; best_h = h.copy()
        order = np.argsort(hs, kind='stable')
        n_fin = int(np.sum(np.isfinite(hs)))
        if n_fin < mu:
            break
        idx_m = order[:mu]
        y_w = np.zeros(p)
        for j in range(mu):
            y_w += w[j] * ys[idx_m[j]]
        m = m + sig * y_w
        y_w_Cinv = V @ ((V.T @ y_w) / sqrt_ev)
        p_sig = (1.0 - c_sig) * p_sig + math.sqrt(c_sig * (2.0 - c_sig) * mu_eff) * y_w_Cinv
        t += 1
        h_sig = 1.0 if np.linalg.norm(p_sig) / math.sqrt(max(1e-30, 1.0 - (1.0 - c_sig) ** (2 * t))) < (1.4 + 2.0 / (p + 1.0)) * chi_p else 0.0
        p_c = (1.0 - c_c) * p_c + (math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w if h_sig else np.zeros(p))
        rank_mu = np.zeros((p, p))
        for j in range(mu):
            yj = ys[idx_m[j]]
            rank_mu += w[j] * np.outer(yj, yj)
        C = (1.0 - c_1 - c_mu) * C + c_1 * (np.outer(p_c, p_c) + (1.0 - h_sig) * c_c * (2.0 - c_c) * C) + c_mu * rank_mu
        C = 0.5 * (C + C.T)
        sf = (c_sig / d_sig) * (np.linalg.norm(p_sig) / chi_p - 1.0)
        sig *= math.exp(np.clip(sf, -1.0, 1.0))
        sig = float(np.clip(sig, 1e-5, 1.0))
        if not (np.all(np.isfinite(m)) and np.all(np.isfinite(C)) and np.isfinite(sig)):
            C = np.eye(p); ev = np.ones(p); V = np.eye(p); sqrt_ev = np.ones(p)
            p_sig = np.zeros(p); p_c = np.zeros(p)
            sig = max(1e-5, sig0 * 0.5)
            if not np.all(np.isfinite(m)):
                m = np.zeros(p)
        if sig <= 1.0001e-5:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if n <= 64:
        n_rest, gp, pg = 4, 120, 40
    elif n <= 128:
        n_rest, gp, pg = 3, 70, 25
    elif n <= 512:
        n_rest, gp, pg = 2, 35, 15
    else:
        n_rest, gp, pg = 2, 20, 10
    U = make_cosine_basis(n, 48)
    p_dim = U.shape[1]
    pool = build_pool(fb, n, rng, 16)
    ps = np.array([exact_score(pool[i], n) for i in range(len(pool))])
    order = np.argsort(ps)
    bases = [pool[0].copy()]
    for idx in order:
        if len(bases) >= n_rest:
            break
        if idx == 0:
            continue
        diverse = True
        for b in bases:
            if np.max(np.abs(pool[idx] - b)) < 0.05:
                diverse = False
                break
        if diverse:
            bases.append(pool[idx].copy())
    while len(bases) < n_rest:
        bases.append(pool[min(len(bases), len(pool) - 1)].copy())
    sig0 = float(np.clip(0.10 * math.sqrt(n / max(1, p_dim)), 0.04, 0.35))
    best_h = fb.copy()
    best_s = exact_score(fb, n)
    for b in bases:
        bh2, bs2 = run_cma_es(b, U, sig0, gp, rng, best_h.copy(), best_s, n)
        if bs2 < best_s:
            best_h, best_s = bh2, bs2
    bh3, bs3 = run_cma_es(best_h.copy(), U, sig0 * 0.5, pg, rng, best_h.copy(), best_s, n)
    if bs3 < best_s:
        best_h, best_s = bh3, bs3
    final = project_capped(best_h, n)
    fs = exact_score(final, n) if final is not None else float('inf')
    fb_s = exact_score(fb, n)
    if final is None or not is_feasible(final, n) or fs > fb_s + 1e-12:
        best_h, best_s = fb.copy(), fb_s
    else:
        best_s = fs
    return best_h.astype(np.float64), float(best_s), int(n)
