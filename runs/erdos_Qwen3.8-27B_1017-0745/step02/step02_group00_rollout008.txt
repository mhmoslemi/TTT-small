                db = polys[ii][1] - polys[jj][1]
                dc = polys[ii][2] - polys[jj][2]
                if abs(dc) < 1e-15:
                    # Linear: db*t + da = 0

                        root = -da / db
                        if s_iv - 1e-12 <= root <= e_iv + 1e-12:
                            cands.append(min(max(root, s_iv), e_iv))
