        n_starts, maxiter, n_hops = 10, 350, 10
    elif n <= 128:
        n_starts, maxiter, n_hops = 7, 250, 6
    else:
        n_starts, maxiter, n_hops = 4, 120, 3

    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))

    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-14:
            gs, gb = s, st.copy()

    for st in starts:
        y = solve_epigraph(st, n, maxiter)
        if y is not None and is_feasible(y, n):
            sy = float(exact_score(y, n))
            if sy < gs - 1e-14:
                gs, gb = sy, y.copy()

    current = gb.copy()
    cur_s = gs
    sigma_cycle = [0.04, 0.08, 0.15, 0.25]
    T0, Tdecay = 2.0e-3, 0.7
    for r in range(n_hops):
        if (r + 1) % 5 == 0:
            proposal = project_capped(rng.uniform(0, 1, n), n)
        else:
            sigma = sigma_cycle[r % 4]
            proposal = project_capped(current + sigma * rng.standard_normal(n), n)
        if proposal is None or not is_feasible(proposal, n):
            continue
        y = solve_epigraph(proposal, n, maxiter)
        if y is not None and is_feasible(y, n):
            sy = float(exact_score(y, n))
            if sy < gs - 1e-14:
                gs, gb = sy, y.copy()
            T = max(1e-5, T0 * (Tdecay ** r))
            if sy <= cur_s + 1e-14:
                current, cur_s = y.copy(), sy
            elif rng.random() < np.exp(-(sy - cur_s) / T):
                current, cur_s = y.copy(), sy

    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-14:
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
