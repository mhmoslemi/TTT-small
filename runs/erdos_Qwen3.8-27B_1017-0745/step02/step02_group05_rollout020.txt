            for ds in (1, -1):
                cap = min(1.0 - (x[i] if ds == 1 else x[j]), x[j] if ds == 1 else x[i])
                if cap < 1e-14:
                    continue
                for f in fracs:
                    t = f * cap
                    xn = x.copy()
                    if ds == 1:
                        xn[i] += t
                        xn[j] -= t
                    else:
                        xn[j] += t
                        xn[i] -= t
                    xn = project_capped(xn, n)
                    if xn is None:
                        continue
                    xs = exact_score(xn, n)
                    if xs < score - 1e-12:
                        x = xn
                        score = xs
                        improved = True
        if not improved:
            break
    return x, score

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        for key in ('n_points', 'n'):
            if key in kwargs:
                try:
                    v = int(kwargs[key])
                    if 1 <= v <= 999:
                        n = v
                        break
                except Exception:
                    pass
    if n is None:
        n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    rng = np.random.default_rng(int(seed))
    base = project_capped(parent, n) if parent is not None else None
    if base is None or not is_feasible(base, n):
        base = np.full(n, 0.5)
    fb = base.copy()
    fb_score = exact_score(fb, n)
    if n <= 64:
        P, G, L_an, stall_lim, n_sweeps = 12, 130, 1800, 18, 3
    elif n <= 256:
        P, G, L_an, stall_lim, n_sweeps = 8, 70, 500, 14, 2
    else:
        P, G, L_an, stall_lim, n_sweeps = 6, 25, 80, 7, 1
    CR = 0.85
    pop = build_population(base, n, rng, P)
    scores = np.array([exact_score(p, n) for p in pop])
    bi = int(np.argmin(scores))
    pop[0], pop[bi] = pop[bi].copy(), pop[0].copy()
    scores[0], scores[bi] = scores[bi], scores[0]
    if scores[0] < fb_score - 1e-12:
        fb_score = scores[0]
        fb = pop[0].copy()
    stall = 0
    for gen in range(G):
        T0 = 2e-3 * (1e-5 / 2e-3) ** (gen / max(1, G - 1))
        pop, scores, improved = de_step(pop, scores, gen, n, rng, CR)
        if scores[0] < fb_score - 1e-12:
            fb_score = scores[0]
            fb = pop[0].copy()
        if not improved:
            stall += 1
        else:
            stall = 0
        if stall >= stall_lim and gen < G - 1:
            keep = max(2, P // 3)
            idxs = np.argsort(scores)[:keep]
            pop = [pop[i].copy() for i in idxs]
            scores = np.array([float(scores[i]) for i in idxs])
            for _ in range(P - keep):
                if rng.random() < 0.5:
                    c = pop[0] + (0.025 + 0.075 * (1 - gen / G)) * rng.normal(0, 1, n)
                else:
                    c = rng.uniform(0, 1, n)
                p2 = project_capped(c, n)
                if p2 is not None:
                    s2 = exact_score(p2, n)
                    if np.isfinite(s2):
                        pop.append(p2)
                        scores = np.append(scores, s2)
                    else:
                        pop.append(pop[0].copy())
                        scores = np.append(scores, scores[0])
                else:
                    pop.append(pop[0].copy())
                    scores = np.append(scores, scores[0])
            nbi = int(np.argmin(scores))
            pop[0], pop[nbi] = pop[nbi].copy(), pop[0].copy()
            scores[0], scores[nbi] = scores[nbi], scores[0]
            stall = 0
            if scores[0] < fb_score - 1e-12:
                fb_score = scores[0]
                fb = pop[0].copy()
    try:
        _, pa_best, pa_score = pair_anneal(pop[0], n, rng, L_an)
        if pa_score < fb_score - 1e-12:
            fb_score = pa_score
            fb = pa_best.copy()
    except Exception:
        pass
    try:
        xp, sp = pair_polish(fb, n, rng, n_sweeps)
        if sp < fb_score - 1e-12:
            fb_score = sp
            fb = xp.copy()
    except Exception:
        pass
    try:
        _, pa_best2, pa_score2 = pair_anneal(fb, n, rng, L_an // 3)
        if pa_score2 < fb_score - 1e-12:
            fb_score = pa_score2
            fb = pa_best2.copy()
    except Exception:
        pass
    try:
        xp2, sp2 = pair_polish(fb, n, rng, max(1, n_sweeps // 2))
        if sp2 < fb_score - 1e-12:
            fb_score = sp2
            fb = xp2.copy()
    except Exception:
        pass
    final = project_capped(fb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > exact_score(np.full(n, 0.5), n) + 1e-12:
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
