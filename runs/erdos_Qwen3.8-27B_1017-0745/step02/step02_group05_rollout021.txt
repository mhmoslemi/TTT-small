def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        for i in range(n):
            if r > 0 and h[i] < 1.0:
                a = min(r, 1.0 - h[i]); h[i] += a; r -= a
            elif r < 0 and h[i] > 0.0:
                a = min(-r, h[i]); h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def helmert_fwd(x, n):
    d = n - 1
    u = np.empty(d)
    for k in range(d):
        u[k] = x[k] / math.sqrt((k + 1) * (k + 2))
    suf = np.empty(n); suf[n - 1] = 0.0
    for i in range(n - 2, -1, -1):
        suf[i] = suf[i + 1] + u[i]
    y = np.empty(n); y[0] = suf[0]
    for i in range(1, n):
        y[i] = suf[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
    return y if np.all(np.isfinite(y)) else None

def helmert_t(y, n):
    d = n - 1
    x = np.empty(d); prefix = 0.0
    for k in range(d):
        prefix += y[k]
        x[k] = (prefix - (k + 1) * y[k + 1]) / math.sqrt((k + 1) * (k + 2))
    return x if np.all(np.isfinite(x)) else None

def latent_to_h(x, n):
    y = helmert_fwd(x, n)
    if y is None:
        return None
    return project_capped(0.5 + y, n)

def h_to_latent(h, n):
    return helmert_t(h - 0.5, n)

    mu = max(1, P // 2)

    w /= w.sum(); mu_eff = 1.0 / np.dot(w, w)
