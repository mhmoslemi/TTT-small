import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        start = max(0, -k)
        stop = min(n, n - k)
        ii = np.arange(start, stop, dtype=np.int64)
        jj = ii + k
        G[row, jj] += dx * (1.0 - h[ii])
        G[row, ii] -= dx * h[jj]
    if not np.all(np.isfinite(L)) or not np.all(np.isfinite(G)):
        return None, None
    return L, G

def solve_mm_qp(x, L, G, n, solvers):
    import cvxpy as cp
    dx = 2.0 / n
    rho = dx * math.cos(math.pi / (n + 1))
    if not np.isfinite(rho) or rho <= 0:
        rho = dx
    y = cp.Variable(n)
    u = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == n / 2]
    n_lags = 2 * n - 1
    for k in range(n_lags):
        cons.append(L[k] + G[k] @ (y - x) <= u)
    obj = cp.Minimize(u + rho * cp.sum_squares(y - x))
    prob = cp.Problem(obj, cons)
    for solver in list(solvers):
        try:
            if solver == 'ECOS':
                prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=5000)
            elif solver == 'CVXOPT':
                prob.solve(solver=cp.CVXOPT, verbose=False, abstol=1e-8, reltol=1e-8, feastol=1e-8)
            elif solver == 'MOSEK':
                prob.solve(solver=cp.MOSEK, verbose=False)
            elif solver == 'GUROBI':
                prob.solve(solver=cp.GUROBI, verbose=False, threads=1)
            else:
                continue
            if prob.status in ('optimal', 'optimal_inaccurate'):
                yv = np.array(y.value, dtype=np.float64).ravel()
                if yv.shape[0] == n and np.all(np.isfinite(yv)):
                    return yv
        except Exception:
            solvers.discard(solver)
    return None

def mm_trajectory(h0, n, solvers, max_iters):
    x = h0.copy().astype(np.float64)
    score = exact_score(x, n)
    for it in range(max_iters):
        L, G = lag_data(x, n)
        if L is None:
            break
        y = solve_mm_qp(x, L, G, n, solvers)
        if y is None:
            break
        best_x, best_s = x.copy(), score
        for alpha in [1.0, 0.7, 0.5, 0.25, 0.125, 0.0625]:
            trial = x + alpha * (y - x)
            pt = project_capped(trial, n)
            if pt is None or not is_feasible(pt, n):
                continue
            s = exact_score(pt, n)
            if s < best_s - 1e-13:
                best_s, best_x = s, pt
        if best_s < score - 1e-12:
            score = best_s; x = best_x
        else:
            break
        if np.max(np.abs(y - x)) < 1e-10:
            break
    return x, score

def pair_score(h, i, j, t, n):
    dx = 2.0 / n
    ht = h.copy()
    ht[i] += t; ht[j] -= t
    c = np.correlate(ht, 1.0 - ht, mode='full') * dx
    return float(np.max(c))

def pair_refine(h, n, rng, n_rounds, n_pairs_per_round):
    score = exact_score(h, n)
    for _ in range(n_rounds):
        improved = False
        for _ in range(n_pairs_per_round):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            Lb = max(-h[i], h[j] - 1.0)
            Ub = min(1.0 - h[i], h[j])
            if Ub - Lb < 1e-12:
                continue
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            a, b = Lb, Ub
            c = b - gr * (b - a); d = a + gr * (b - a)
            fc = pair_score(h, i, j, c, n); fd = pair_score(h, i, j, d, n)
            for _ in range(40):
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a); fc = pair_score(h, i, j, c, n)
                else:
                    a = c; c = d; fc = fd
                    d = a + gr * (b - a); fd = pair_score(h, i, j, d, n)
            t_best = 0.5 * (a + b)
            s_best = min(pair_score(h, i, j, t_best, n), pair_score(h, i, j, 0.0, n))
            if s_best < score - 1e-12:
                t_use = t_best if s_best == pair_score(h, i, j, t_best, n) else 0.0
                ht = project_capped(h + t_use * (np.where(np.arange(n) == i, 1, -np.where(np.arange(n) == j, 1, 0))), n)
                if ht is None:
                    continue
                s = exact_score(ht, n)
                if s < score - 1e-12:
                    h = ht; score = s; improved = True
        if not improved:
            break
    return h, score

def build_starts(fallback, n, rng, count):
    pool = []
    cands = [fallback.copy(), fallback[::-1].copy(), (1.0 - fallback[::-1]).copy(), np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.extend([blk, np.where(np.arange(n) % 2 == 0, 1.0, 0.0)])
    t = np.arange(n, dtype=np.float64)
    for m in [1, 2, 3]:
        for ph in [0.0, math.pi / 2]:
            cands.append(0.5 + 0.4 * math.cos(2.0 * math.pi * m * t / n + ph))
    cands.append(fallback + rng.normal(0, 0.06, n))
    cands.append(fallback + rng.normal(0, 0.15, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-7:
                dup = True; break
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    pool.sort(key=lambda x: exact_score(x, n))
    return pool[:count]

    if n is None:

    if n == 1:

        return hv, float(0.5 * 2.0), 1
