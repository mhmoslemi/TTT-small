    dx = 2.0 / n
    eta = 0.2 / dx
    failed = 0
    for _ in range(n_iter):
        if failed >= 12 or eta < 1e-10 / dx:
            break
        q = np.correlate(current, 1.0 - current, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p:p + m] += dx * (1.0 - current[r:r + m])
        g[r:r + m] -= dx * current[p:p + m]
        z = project_capped(current - eta * g, n)
        if z is None:
            eta *= 0.5
            failed += 1
            continue
        v = z - current
        if np.max(np.abs(v)) < 1e-12:
            eta *= 0.5
            failed += 1
            continue
        best_z, best_s = None, score
        for alpha in (1.0, 0.5, 0.25, 0.125):
            trial = z if alpha > 0.999 else project_capped(current + alpha * v, n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-12:
                best_s, best_z = ts, trial.copy()
        if best_z is None:
            eta *= 0.5
            failed += 1
        else:
            current, score = best_z, best_s
            failed = 0
            eta = min(1.1 * eta, 2.0 / dx)
    return current, score

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    fb_score = exact_score(fallback, n)
    starts = build_starts(fallback, n, rng)
    best_h = fallback.copy()
    best_s = fb_score
    for st in starts:
        s = exact_score(st, n)
        if s < best_s - 1e-12:
            best_s = s
            best_h = st.copy()
    scored = [(exact_score(st, n), i) for i, st in enumerate(starts)]
    scored.sort(key=lambda x: x[0])
    n_opt = 6 if n <= 64 else (3 if n <= 256 else 2)
    opt_idx = set(idx for _, idx in scored[:n_opt])
    opt_idx.add(0)
    opt_idx = sorted(opt_idx)[:n_opt + 1]
    taus = [0.02, 0.004, 0.0008, 0.00016, 0.000032, 0.0000064]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_jac, 'args': (n,)}]
    mx = max(30, min(150, int(round(150.0 * math.sqrt(47.0 / max(1, n))))))
    for oi in opt_idx:
        current = starts[oi].copy()
        for tau in taus:
            try:
                res = minimize(soft_obj, current, args=(n, tau),
                              jac=soft_grad, bounds=bounds, constraints=cons,
                              method='SLSQP',
                              options={'maxiter': mx, 'ftol': 1e-10, 'disp': False})
            except Exception:
                break
            if res.x is None or res.x.shape[0] != n or not np.all(np.isfinite(res.x)):
                break
            y = project_capped(res.x, n)
            if y is None:
                break
            ys = exact_score(y, n)
            if ys < best_s - 1e-12:
                best_s = ys
                best_h = y.copy()
            for alpha in (1.0, 0.5, 0.25, 0.125):
                trial = y if alpha > 0.999 else project_capped(current + alpha * (y - current), n)
                if trial is None:
                    continue
                ts = exact_score(trial, n)
                if ts < best_s - 1e-12:
                    best_s = ts
                    best_h = trial.copy()
            sv = soft_obj(y, n, tau)
            sc = soft_obj(current, n, tau)
            if np.isfinite(sv) and np.isfinite(sc) and sv <= sc + 1e-9:
                current = y
            else:
                break
    h_pol, s_pol = polish_subgrad(best_h, n, 80)
    if s_pol < best_s - 1e-12:
        best_s, best_h = s_pol, h_pol
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
