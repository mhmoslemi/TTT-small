            g_score = hs
        if not any_improve:
            break
    final = project_capped(g_inc, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_score + 1e-12:
        final = safe_h.copy()
        fs = safe_score
    return final.astype(np.float64), float(fs), int(n)
