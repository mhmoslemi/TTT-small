def cma_run(base, U, p, init_sigma, gens, rng, best_h, best_s, n):
    lam = max(6, int(4 + 3.0 * math.log(max(p, 2))))
    mu = max(2, lam // 2)
    w = np.zeros(mu, dtype=np.float64)
    for j in range(mu):
        w[j] = math.log(mu + 0.5) - math.log(j + 1)
    w /= w.sum()
    mu_eff = 1.0 / np.sum(w ** 2)
    c_sigma = (mu_eff + 2.0) / (p + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (p + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / p) / (p + 4.0 + 2.0 * mu_eff / p)
    c_1 = 2.0 / ((p + 1.3) ** 2 + mu_eff)
    c_mu = max(0.0, min(1.0 - c_1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((p + 2.0) ** 2 + mu_eff)))
    chi_p = math.sqrt(p) * (1.0 - 1.0 / (4.0 * p) + 1.0 / (21.0 * p * p))
    m = np.zeros(p, dtype=np.float64)
    C = np.eye(p, dtype=np.float64)
    p_sig = np.zeros(p, dtype=np.float64)
    p_cc = np.zeros(p, dtype=np.float64)
    sigma = init_sigma
    bh = best_h.copy() if best_h is not None else None
    bs = best_s
    for t in range(1, gens + 1):
        C = 0.5 * (C + C.T)
        try:
            ew, ev = np.linalg.eigh(C)
            if not np.all(np.isfinite(ew)):
                raise ValueError
            ew = np.clip(ew, 1e-10, 1e4)
            se = np.sqrt(ew)
        except Exception:
            C = np.eye(p, dtype=np.float64)
            p_sig[:] = 0.0
            p_cc[:] = 0.0
            ew = np.ones(p)
            ev = np.eye(p)
            se = np.ones(p)
        yw = np.zeros(p, dtype=np.float64)
        sc_list = []
        y_sel = []
        w_sum = np.zeros((p, p), dtype=np.float64)
        for ii in range(lam):
            z = rng.standard_normal(p)
            y = ev @ (se * z)
            x = m + sigma * y
            hc = project_capped(base + U @ x, n)
            if hc is None:
                sc_list.append(float('inf'))
            else:
                sc_list.append(exact_score(hc, n))
                if sc_list[-1] < bs - 1e-12:
                    bs = sc_list[-1]
                    bh = hc.copy()
            if ii < lam:
                y_sel.append(y)
        sc_arr = np.array(sc_list, dtype=np.float64)
        fin_count = int(np.sum(np.isfinite(sc_arr)))
        if fin_count < mu:
            break
        order = np.argsort(sc_arr).tolist()
        sel = order[:mu]
        yw = np.zeros(p, dtype=np.float64)
        rank_mu = np.zeros((p, p), dtype=np.float64)
        for j in range(mu):
            yj = y_sel[sel[j]]
            yw += w[j] * yj
            rank_mu += w[j] * np.outer(yj, yj)
        m_new = m + sigma * yw
        c_half_yw = ev @ ((ev.T @ yw) / se)
        p_sig = (1.0 - c_sigma) * p_sig + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * c_half_yw
        denom = math.sqrt(max(1e-30, 1.0 - (1.0 - c_sigma) ** (2 * t)))
        h_sig = 1.0 if (np.linalg.norm(p_sig) / denom) < (1.4 + 2.0 / (p + 1.0)) * chi_p else 0.0
        p_cc = (1.0 - c_c) * p_cc + h_sig * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * yw
        C = (1.0 - c_1 - c_mu) * C + c_1 * (np.outer(p_cc, p_cc) + (1.0 - h_sig) * c_c * (2.0 - c_c) * C) + c_mu * rank_mu
        C = 0.5 * (C + C.T)
        sigma *= math.exp(float(np.clip((c_sigma / d_sigma) * (np.linalg.norm(p_sig) / chi_p - 1.0), -1.0, 1.0)))
        sigma = max(1e-5, min(1.0, sigma))
        m = m_new
        if not (np.all(np.isfinite(m)) and np.all(np.isfinite(C)) and np.all(np.isfinite(p_sig))):
            C = np.eye(p, dtype=np.float64)
            p_sig[:] = 0.0
            p_cc[:] = 0.0
            sigma = max(1e-5, init_sigma * 0.5)
        if sigma < 1e-5:
            break
    return bh, bs

def local_poll(hb, sb, U, p, n, rng, rounds=12, ndirs=8):
    delta = 0.15
    hcur = hb
    scur = sb
    sc_scale = max(0.5, math.sqrt(47.0 / n))
    for _ in range(rounds):
        if delta < 1e-4:
            break
        improved = False
        for _ in range(ndirs):
            if rng.random() < 0.5:
                v = rng.standard_normal(p)
            else:
                v = rng.standard_normal(p) - rng.standard_normal(p)
            v -= np.mean(v)
            mx = np.max(np.abs(v))
            if mx < 1e-14:
                continue
            v /= mx
            for tt in (-delta, -0.5 * delta, 0.5 * delta, delta):
                cand = project_capped(hcur + (U @ (tt * v)), n)
                if cand is None:
                    continue
                cs = exact_score(cand, n)
                if cs < scur - 1e-12:
                    hcur = cand
                    scur = cs
                    improved = True
        if improved:
            delta = min(0.3, 1.1 * delta)
        else:
            delta *= 0.5
    return hcur, scur

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p_arr = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p_arr.size <= 999:
            n = int(p_arr.size)
            if np.all(np.isfinite(p_arr)) and np.all(p_arr >= -1e-8) and np.all(p_arr <= 1.0 + 1e-8):
                parent = np.clip(p_arr, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = fb.copy()
    U, p = build_cosine_basis(n)
    if p <= 0:
        return fb, float(exact_score(fb, n)), n
    pool = build_pool(fb, n, rng, 16)
    scs = np.array([exact_score(v, n) for v in pool], dtype=np.float64)
    order = np.argsort(scs).tolist()
    sel_bases = [0]
    for i in order:
        if len(sel_bases) >= 4:
            break
        if i == 0:
            continue
        ok = True
        for j in sel_bases:
            if np.max(np.abs(pool[i] - pool[j])) < 0.05:
                ok = False
                break
        if ok:
            sel_bases.append(i)
    for i in order:
        if len(sel_bases) >= 4:
            break
        if i not in sel_bases:
            sel_bases.append(i)
    if n <= 64:
        n_rest, gens_p, pol_gens = 4, 120, 40
    elif n <= 128:
        n_rest, gens_p, pol_gens = 3, 70, 25
    elif n <= 512:
        n_rest, gens_p, pol_gens = 2, 35, 15
    else:
        n_rest, gens_p, pol_gens = 2, 20, 10
    n_rest = min(n_rest, len(sel_bases))
    init_sig = max(0.04, min(0.35, 0.10 * math.sqrt(n / p)))
    best_h = fb.copy()
    best_s = exact_score(fb, n)
    for bi in sel_bases[:n_rest]:
        bh, bs = cma_run(pool[bi], U, p, init_sig, gens_p, rng, best_h, best_s, n)
        if bh is not None and bs < best_s - 1e-12:
            best_h, best_s = bh, bs
    ph = project_capped(best_h, n)
    if ph is not None:
        bh, bs = cma_run(ph, U, p, init_sig * 0.5, pol_gens, rng, best_h, best_s, n)
        if bh is not None and bs < best_s - 1e-12:
            best_h, best_s = bh, bs
    best_h, best_s = local_poll(best_h, best_s, U, p, n, rng, 10, 8)
    fv = project_capped(best_h, n)
    if fv is None or not is_feasible(fv, n):
        best_h = fb.copy()
    else:
        best_h = fv
    final_s = exact_score(best_h, n)
    fb_s = exact_score(fb, n)
    if final_s > fb_s:
        best_h = fb.copy()
        final_s = fb_s
    return best_h.astype(np.float64), float(final_s), int(n)
