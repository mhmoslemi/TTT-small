            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    fb_s = exact_score(fb, n)
    best_h = fb.copy()
    best_s = fb_s
    mx = 10 if n <= 64 else (7 if n <= 128 else 4)
    starts = build_starts(fb, n, rng, mx)
    for sv in starts:
        if sv is None:
            continue
        s = exact_score(sv, n)
        if np.isfinite(s) and s < best_s - 1e-12:
            best_s = s
            best_h = sv.copy()
    if n <= 128:
        for sv in starts:
            if sv is None:
                continue
            res = try_slsqp(sv, n)
            cands = []
            if res is not None:
                cands.append(res)
                for a in (0.25, 0.5, 0.75):
                    cands.append((1.0 - a) * sv + a * res)
            for cand in cands:
                pr = project_capped(cand, n)
                if pr is None:
                    continue
                s = exact_score(pr, n)
                if np.isfinite(s) and s < best_s - 1e-12:
                    best_s = s
                    best_h = pr.copy()
    itp = 14 if n <= 128 else (30 if n <= 256 else (20 if n <= 512 else 12))
    targets = [best_h.copy()]
    scs = []
    for sv in starts:
        if sv is not None and np.max(np.abs(sv - best_h)) > 1e-8:
            scs.append((exact_score(sv, n), sv))
    scs.sort()
    for _, sv in scs[:2]:
        targets.append(sv.copy())
    for pt in targets:
        ph, ps = smooth_polish(pt, n, itp)
        if np.isfinite(ps) and ps < best_s - 1e-12:
            best_s = ps
            best_h = ph.copy()
    fh = project_capped(best_h, n)
    if fh is not None and is_feasible(fh, n):
        best_h = fh
    final_s = exact_score(best_h, n)
    if not np.isfinite(final_s) or final_s > fb_s:
        best_h = fb.copy()
        final_s = fb_s
    return best_h.astype(np.float64), float(final_s), int(n)
