    nv = 2 * n - 1
    J = np.zeros((nv, n))
    for d in range(-(n-1), n):
        row = d + n - 1

        np.add.at(J, (r, jj), dx * (-h[ii]))
    return J

from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(s - np.sum(h)) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_vals(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        rw = d + n - 1
        lo = max(0, -d); hi = min(n, n - d)
        if lo >= hi:
            continue
        ii = np.arange(lo, hi); jj = ii + d
        np.add.at(J, (rw, ii), dx * (1.0 - h[jj]))
        np.add.at(J, (rw, jj), dx * (-h[ii]))
    return J

def of_f(z):
    return z[-1]

def of_g(z):
    g = np.zeros(len(z)); g[-1] = 1.0
    return g

def ineq_f(z, n):
    return z[n] - lag_vals(z[:n], n)

def ineq_j(z, n):
    J = -lag_jac(z[:n], n)
    return np.hstack([J, np.ones((2 * n - 1, 1))])

def eq_f(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_j(z, n):
    g = np.zeros(n + 1); g[:n] = 1.0
    return g

def solve_epi(h0, n, mit):
    s0 = exact_score(h0, n)
    if s0 == float('inf'):
        return None
    t0 = min(0.5, s0 + 1e-9)
    z0 = np.concatenate([h0, [max(0.0, t0)]])
    bds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cs = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_j, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j, 'args': (n,)}
    ]
    try:
        res = minimize(of_f, z0, jac=of_g, method='SLSQP',
                       bounds=bds, constraints=cs,
                       options={'ftol': 1e-10, 'maxiter': mit, 'disp': False})
    except Exception:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.shape[0] != n + 1 or not np.all(np.isfinite(x)):
        return None
    return project_capped(x[:n], n)
