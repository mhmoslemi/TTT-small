def project_capped(z, n):
    # project onto {h: 0<=h<=1, sum(h)=n/2}
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,):
        return None

        return None
    z = np.clip(z, -1e4, 1e4)
    s = n / 2.0

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid

        return False

        return False

        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):

            dmin = min(np.sqrt(np.mean((pool[i][0] - pool[j][0])**2)) for j in sel)
            if dmin > best_d:
                best_d, best_i = dmin, i
        if best_i < 0:
            break
        sel.append(best_i)
    return [pool[i][0].copy() for i in sel], [pool[i][1] for i in sel]

def run_de(pops, scs, n, rng, max_gen, stall_lim, n_restart, pool):
    P = len(pops)
    pop = np.array([pops[i].copy() for i in range(P)], dtype=np.float64)
    score = np.array(scs, dtype=np.float64)
    gbest = pops[int(np.argmin(score))].copy()
    gscore = float(score[int(np.argmin(score))])
    archive = []
    mu_F, mu_CR = 0.55, 0.85
    stall = 0
    restarts = 0
    rand_end = int(0.25 * max_gen)
    pb_frac = 0.2
    for gen in range(max_gen):
        npop = pop.copy()
        nsco = score.copy()
        F_s = []; CR_s = []
        narch = list(archive)
        for i in range(P):
            cands = np.delete(np.arange(P), i)
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            r1, r2, r3 = int(r1), int(r2), int(r3)
            F_i = mu_F + 0.1 * float(rng.standard_cauchy())
            for _ in range(10):
                if np.isfinite(F_i) and F_i > 0.05:
                    break
                F_i = mu_F + 0.1 * float(rng.standard_cauchy())
            F_i = float(np.clip(F_i, 0.05, 1.0))
            CR_i = float(np.clip(rng.normal(mu_CR, 0.1), 0.0, 1.0))
            if gen < rand_end:
                v = pop[r1] + F_i * (pop[r2] - pop[r3])
            else:
                p_cnt = max(2, int(np.ceil(pb_frac * P)))
                sidx = np.argsort(nsco)
                pb = int(sidx[int(rng.integers(0, p_cnt))])
                if pb == i and p_cnt > 1:
                    pb = int(sidx[1])
                v = pop[i] + F_i * (pop[pb] - pop[i]) + F_i * (pop[r1] - pop[r2])
            if not np.all(np.isfinite(v)):
                continue
            trial = pop[i].copy()
            mask = rng.random(n) < CR_i
            mask[int(rng.integers(0, n))] = True
            trial[mask] = v[mask]
            ph = project_capped(trial, n)
            if ph is None or not is_feasible(ph, n):
                continue
            s = exact_score(ph, n)
            if not np.isfinite(s):
                continue
            if s < nsco[i] - 1e-12:
                narch.append(pop[i].copy())
                npop[i] = ph
                nsco[i] = s
                F_s.append(F_i); CR_s.append(CR_i)
            if s < gscore - 1e-12:
                gscore = s; gbest = ph.copy()
        pop = npop; score = nsco
        if len(narch) > P:
            rm = rng.choice(len(narch), size=len(narch) - P, replace=False)
            for j in sorted(rm, reverse=True):
                del narch[j]
        archive = narch
        if F_s:
            F_a = np.array(F_s, dtype=np.float64)
            mu_F = float(np.clip(0.9 * mu_F + 0.1 * np.sum(F_a**2) / np.sum(F_a), 0.05, 1.0))
            mu_CR = float(np.clip(0.9 * mu_CR + 0.1 * np.mean(np.array(CR_s)), 0.0, 1.0))
        stall += 1
        if gscore < float(np.max(score)) + 1e-12:
            pass
        best_cur = float(np.min(score))
        if best_cur > gscore + 1e-12:
            stall += 1
        if stall >= stall_lim and restarts < n_restart:
            restarts += 1
            stall = 0
            sidx2 = np.argsort(score)
            keep = list(sidx2[:max(2, P // 2)])
            if i not in keep:
                pass
            if np.min([np.max(np.abs(gbest - pop[k])) for k in keep]) > 1e-8:
                keep[-1] = int(np.argmin([np.max(np.abs(gbest - pop[k])) for k in keep]))
            new_pops = [pop[k].copy() for k in keep]
            new_scs = [score[k] for k in keep]
            new_pops.append(gbest.copy()); new_scs.append(gscore)
            scales = [0.04, 0.12, 0.25]
            while len(new_pops) < P:
                added = False
                for attempt in range(5):
                    sd = scales[attempt % 3] * (1.5 ** (attempt // 3))
                    sd = min(sd, 0.5)
                    pv = project_capped(gbest + rng.normal(0, sd, n), n)
                    if pv is not None and is_feasible(pv, n):
                        ok = all(np.max(np.abs(pv - x)) >= 1e-8 for x in new_pops)
                        if ok:
                            ns = exact_score(pv, n)
                            new_pops.append(pv); new_scs.append(ns)
                            added = True; break
                if not added:
                    new_pops.append(new_pops[0].copy()); new_scs.append(new_scs[0])
            pop = np.array([new_pops[i].copy() for i in range(P)], dtype=np.float64)
            score = np.array(new_scs[:P], dtype=np.float64)
            archive = []
            mu_F, mu_CR = 0.55, 0.85
    return gbest, gscore

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0, 1)
                if abs(np.sum(parent) - n / 2.0) > 1e-6:
                    pp = project_capped(parent, n)
                    parent = pp if pp is not None else None
    except Exception:
        pass
    if n is None:
        n = 47
    h_unif = np.full(n, 0.5)
    parent_fallback = parent.copy() if parent is not None else h_unif.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        s = exact_score(h_unif, n)
        return h_unif, float(s), 1
    if n <= 64:
        P, max_g, sl, nr = 64, 100, 18, 2
    elif n <= 128:
        P, max_g, sl, nr = 48, 70, 14, 2
    elif n <= 256:
        P, max_g, sl, nr = 32, 40, 10, 1
    elif n <= 512:
        P, max_g, sl, nr = 20, 20, 7, 1
    else:
        P, max_g, sl, nr = 12, 12, 5, 1
    pool = build_pool(n, rng, parent)
    pops, scs = select_pop(pool, n, P)
    gb, gs = run_de(pops, scs, n, rng, max_g, sl, nr, pool)
    final = project_capped(gb, n)
    if final is None:
        final = gb.copy()
    fs = exact_score(final, n)
    ps = exact_score(parent_fallback, n)
    us = exact_score(h_unif, n)
    if fs >= ps:
        final = parent_fallback.copy(); fs = ps
    if fs >= us:
        final = h_unif.copy(); fs = us
    final = project_capped(final, n)
    if final is None:
        final = h_unif.copy(); fs = us
    else:
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
