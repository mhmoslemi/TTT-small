        scl = sum(1 for sc in scores[1:] if sc < mu_s - 1e-12) / max(1, lam)
        if scl > 0.25:
            sigma = min(0.25, 1.12 * sigma)
        elif scl < 0.08:
            sigma = max(5e-4, 0.85 * sigma)
        n_diff = q - 1
        if n_diff >= 1:
            E = np.zeros((n, n))
            for ii in range(q):
                d = cands[idx[ii]] - new_mu
                E += wts[ii] * np.outer(d, d)
            E *= 1.0 / max(sigma * sigma, 1e-8)
            Pm = np.eye(n) - np.ones((n, n)) / n
            E = Pm @ E @ Pm
            E = 0.5 * (E + E.T)
            trE = np.trace(E)
            if trE > 1e-10:
                E = E / trE * (n - 1)
                A = 0.8 * A + 0.2 * E + 1e-4 * Pm
                A = 0.5 * (A + A.T)
                trA = np.trace(A)
                if trA > 1e-12:
                    A = A / trA * (n - 1)
        if stale > 12 and sigma <= 0.002:
            break
    return best_x, best_s

def polish(h, n, rng, npass=3):
    x = h.copy()
    bs = exact_score(x, n)
    deltas = [0.05, 0.12, 0.3, 0.5]
    for _ in range(npass):
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            if x[i] <= 0 or x[i] >= 1:
                continue
            for j in order:
                if j == i:
                    continue
                for d in deltas:
                    z1 = x.copy(); z1[i] -= d; z1[j] += d
                    h1 = project_capped(z1, n)
                    if h1 is not None:
                        s1 = exact_score(h1, n)
                        if s1 < bs - 1e-12:
                            x, bs = h1.copy(), s1
                            improved = True
                    z2 = x.copy(); z2[i] += d; z2[j] -= d
                    h2 = project_capped(z2, n)
                    if h2 is not None:
                        s2 = exact_score(h2, n)
                        if s2 < bs - 1e-12:
                            x, bs = h2.copy(), s2
                            improved = True
        if not improved:
            break
    sorted_idx = np.argsort(x)
    k_ones = int(math.ceil(n / 2.0))
    for thr in [0.3, 0.4, 0.5, 0.6, 0.7]:
        q = x.copy()
        q[q < thr] = 0.0
        q[q >= 1.0 - thr] = 1.0
        q = project_capped(q, n)
        if q is not None:
            qs = exact_score(q, n)
            if qs < bs - 1e-12:
                x, bs = q.copy(), qs
    return x, bs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    fb_score = exact_score(fallback, n)
    if n == 1:
        return fallback, fb_score, int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    sdp_u, sdp_cov = solve_sdp(n)
    pool = build_pool(fallback, n, rng, sdp_u, sdp_cov, 20)
    best_x = fallback.copy()
    best_s = fb_score
    pool_scores = [exact_score(p, n) for p in pool]
    for i in range(len(pool)):
        if pool_scores[i] < best_s - 1e-12:
            best_s, best_x = pool_scores[i], pool[i].copy()
    if n <= 64:
        lam, ngen, nrestart = 20, 55, 4
    elif n <= 128:
        lam, ngen, nrestart = 16, 35, 3
    elif n <= 256:
        lam, ngen, nrestart = 12, 20, 2
    else:
        lam, ngen, nrestart = 8, 12, 1
    sigmas = [0.04, 0.12, 0.20, 0.07][:nrestart]
    pool_arr = np.argsort(pool_scores)
    centers = []
    cidx = pool_arr[best_s == min(pool_scores) * 0 + 0 if False else 0]
    centers.append(int(pool_arr[np.argmin(pool_scores)]))
    for ci in pool_arr:
        if len(centers) >= nrestart:
            break
        ci = int(ci)
        if ci in centers:
            continue
        ok = True
        for cj in centers:
            if np.max(np.abs(pool[ci] - pool[cj])) < 1e-3:
                ok = False
                break
        if ok:
            centers.append(ci)
    for r in range(len(centers)):
        c = centers[r]
        mu = pool[c].copy()
        ms = exact_score(mu, n)
        if ms < best_s - 1e-12:
            best_s, best_x = ms, mu.copy()
        sg = sigmas[r] if r < len(sigmas) else 0.08
        try:
            ex, es = evolve(mu, n, rng, lam, ngen, sg, sdp_cov)
            if es < best_s - 1e-12:
                best_s, best_x = es, ex.copy()
            ex2, es2 = polish(ex, n, rng, 2)
            if es2 < best_s - 1e-12:
                best_s, best_x = es2, ex2.copy()
        except Exception:
            pass
    fx, fs = polish(best_x, n, rng, 3)
    if fs < best_s - 1e-12:
        best_s, best_x = fs, fx.copy()
    final = project_capped(best_x, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
        fs_final = fb_score
    else:
        fs_final = exact_score(final, n)
        if fs_final > fb_score + 1e-12:
            final = fallback.copy()
            fs_final = fb_score
    if not np.isfinite(fs_final):
        final = np.full(n, 0.5)
        fs_final = exact_score(final, n)
    return np.asarray(final, dtype=np.float64), float(fs_final), int(n)
