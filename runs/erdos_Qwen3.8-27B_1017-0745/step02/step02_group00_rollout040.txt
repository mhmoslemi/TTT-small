def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(x, n):

        hi = min(n, n - d)

    return a

def dir_products(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    r = np.zeros(m)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)

                           - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def wtd_grad(x, w, n):
    dx = 2.0 / n
    m = 2 * n - 1
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return g

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    m = v.size
    if m == 0 or not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.arange(1, m + 1, dtype=np.float64)
    cond = u - (cs - 1.0) / idx > 0
    if not np.any(cond):
        return np.zeros(m)
    j = int(np.max(np.where(cond)))
    lam = (cs[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    if not np.all(np.isfinite(w)):
        return None
    r = 1.0 - float(np.sum(w))
    if abs(r) > 1e-12:
        # fix on a positive coordinate

            w[pos[0]] += r
            w = np.maximum(w, 0.0)
        else:
            w[0] = 1.0
            w[:] = 0.0
            w[0] = 1.0
    return w

    m = 2 * n - 1
    L = 8.0 / mu
    w = project_simplex(w_init)
    if w is None:
        w = np.zeros(m)
        w[np.argmax(a)] = 1.0
    z = w.copy()
    t = 1.0
    best_y, best_P = None, float('inf')
    best_w, best_D = w.copy(), -float('inf')

        if wz is None:

        if mid == lo or mid == hi:

            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        if r > 0:

        else:

    h = np.clip(h, 0.0, 1.0)

        return float('inf')

def lag_values(x, n):

        a[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return a

def wtd_grad(x, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:

        lo = max(0, -d); hi = min(n, n - d)
        g[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return g

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    m = v.size
    if m == 0 or not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.arange(1, m + 1, dtype=np.float64)
    cond = u - (cs - 1.0) / idx > 0
    if not np.any(cond):
        return np.zeros(m)
    j = int(np.max(np.where(cond)))
    lam = (cs[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    r = 1.0 - float(np.sum(w))
    if abs(r) > 1e-13:

        if len(pos) > 0:
            w[pos[0]] += r
        else:
            w = np.zeros(m); w[0] = 1.0
    return w

        max_starts = max(1, min(int(kwargs.get('starts', 4)), 8))
    except Exception:
        max_starts = 4
    try:
        max_outer = max(0, min(int(kwargs.get('max_outer', 14)), 30))
    except Exception:
        max_outer = 14
    try:
        K_dual = max(1, min(int(kwargs.get('dual_iter', 100)), 300))
    except Exception:
        K_dual = 100
    try:
        n_mu = max(1, min(int(kwargs.get('mu_retries', 3)), 5))
    except Exception:
        n_mu = 3
    if n > 64:
        max_starts = min(max_starts, 3); max_outer = min(max_outer, 8); K_dual = min(K_dual, 70)
    if n > 128:
        max_starts = min(max_starts, 2); max_outer = min(max_outer, 5); K_dual = min(K_dual, 35)
    if n > 256:
        max_starts = 1; max_outer = min(max_outer, 3); K_dual = min(K_dual, 16)
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        starts.append((1.0 - parent).copy())
        starts.append((1.0 - parent[::-1]).copy())
        for s2 in [0.03, 0.08, 0.15]:
            starts.append(parent + rng.normal(0, s2, n))
    else:
        starts.append(uniform.copy())
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        starts.append(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n))
    starts.append(rng.uniform(0, 1, n))
    starts.append(0.5 + rng.normal(0, 0.1, n))
    dedup = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64).ravel()
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True; break
        if not dup:
            dedup.append(s)
    starts = dedup[:max_starts]
    for st in starts:
        pc = project_capped(st, n)
        if pc is None:
            continue
        ss = exact_score(pc, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = pc.copy(), ss
        if max_outer <= 0:
            continue
        xm, sm = mm_run(pc, n, max_outer, K_dual, n_mu)
        if np.isfinite(sm) and sm < best_score - 1e-12:
            best, best_score = xm.copy(), sm
    if n <= 80:
        polished = polish_2d(best, n, 2)
        polished = project_capped(polished, n)
        if polished is not None:
            ps = exact_score(polished, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = polished, ps
    final = project_capped(best, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best.copy(); fs = best_score
    if not np.isfinite(fs):
        final = uniform.copy(); fs = exact_score(uniform, n)
    return final, float(fs), int(n)
