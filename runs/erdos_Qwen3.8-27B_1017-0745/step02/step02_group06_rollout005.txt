        if xo is None:
            continue
        so = exact_score(xo, n)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()
    sigma = 0.08
    for _ in range(n_restarts):
        improved = False
        for att in range(4):
            prop = project_capped(gb + sigma * rng.normal(0, 1, n), n)
            if prop is None:
                continue
            if np.max(np.abs(prop - gb)) < 1e-5:
                continue
            ps = exact_score(prop, n)
            if ps < gs - 1e-12:
                gs, gb = ps, prop.copy()
            xo = proj_descent(prop, n, tau_list, max_it, 1.0)
            xo = project_capped(xo, n)
            if xo is not None:
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
                    improved = True
            break
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb = project_capped(fallback, n)
    if fb is not None:
        fbs = exact_score(fb, n)
        if fbs < fs - 1e-12:
            final, fs = fb, fbs
    uf = project_capped(uniform, n)
    if uf is not None:
        ufs = exact_score(uf, n)
        if ufs < fs - 1e-12:
            final, fs = uf, ufs
    return final, float(fs), int(n)
