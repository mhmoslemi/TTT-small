    _try_add(pool, project_capped(cos_v, n), n)
    # Two-block pattern
    tb = np.zeros(n)
    tb[:(n // 4)] = 1.0
    tb[(n // 2):(n // 2 + n // 4)] = 1.0
    if n % 4 != 0:
        s_needed = n / 2.0
        s_cur = 2.0 * (n // 4)
        if s_cur < s_needed:
            tb[n // 4] = min(s_needed - s_cur, 1.0)
    _try_add(pool, project_capped(tb, n), n)
    return pool[:count]
