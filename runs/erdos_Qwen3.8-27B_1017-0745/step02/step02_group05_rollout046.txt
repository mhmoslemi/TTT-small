def quad_roots(a, b, c):
    # returns list of real roots
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return []
        return [-c/b]
    disc = b*b - 4*a*c
    if disc < -1e-14*(b*b + abs(4*a*c) + 1e-300):
        return []
    if disc < 0:

    sqrt_d = math.sqrt(disc)
    r1 = (-b + sqrt_d) / (2*a)

    roots = []
    if math.isfinite(r1):

    if math.isfinite(r2) and abs(r2-r1) > 1e-12:

    return roots

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        r = s - np.sum(h)

        B[d+n-1] = dx * v
    return B

def quad_roots(a, b, c):
    if abs(a) < 1e-15:
        if abs(b) < 1e-15: return []
        r = -c / b
        return [r] if math.isfinite(r) else []
    disc = b*b - 4.0*a*c
    if disc < -1e-14*(b*b + abs(4.0*a*c) + 1e-300): return []
    if disc < 0: disc = 0.0
    sq = math.sqrt(disc)
    r1, r2 = (-b+sq)/(2.0*a), (-b-sq)/(2.0*a)
    out = []
    if math.isfinite(r1): out.append(r1)
    if math.isfinite(r2) and abs(r2-r1) > 1e-12: out.append(r2)
    return out

def build_hull(sl, ic, lo, hi):
    n = len(sl)
    if n == 0: return []
    order = np.argsort(sl, kind='stable')
    hull = []
    for oi in order:
        sv, cv = sl[oi], ic[oi]
        if hull and sv == hull[-1][0]:
            if cv > hull[-1][1]: hull[-1] = (sv, cv, hull[-1][3])
            continue
        if not hull:
            hull.append((sv, cv, -np.inf))
        else:
            x_int = (hull[-1][1] - cv) / (sv - hull[-1][0])
            while hull and x_int <= hull[-1][2] + 1e-15:
                hull.pop()
                if not hull: break
                x_int = (hull[-1][1] - cv) / (sv - hull[-1][0])
            if not hull: hull = [(sv, cv, -np.inf)]
            else: hull.append((sv, cv, x_int))
    segs = []
    for k in range(len(hull)):
        ts = max(hull[k][2], lo)
        te = hull[k+1][2] if k+1 < len(hull) else np.inf
        te = min(te, hi)
        if ts <= te + 1e-15: segs.append((ts, te))
    return segs

def pair_line_min(h, A, n, i, j):
    dx = 2.0 / n
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], h[j])
    if hi - lo < 1e-14: return 0.0, float(np.max(A))
    B = compute_B(h, n, i, j)
    D = np.zeros(2*n-1)
    D[n-1] = -2.0*dx
    D[n-1+(i-j)] = dx
    D[n-1+(j-i)] = dx
    sp = [n-1, n-1+(i-j), n-1+(j-i)]
    mask = np.ones(2*n-1, dtype=bool); mask[sp] = False
    aff = np.where(mask)[0]
    pre_cands = [lo, hi, 0.0]
    for sm in sp:
        if D[sm] > 1e-15:
            tv = -B[sm]/(2.0*D[sm])
            if lo <= tv <= hi: pre_cands.append(tv)
    for p in range(3):
        for q in range(p+1, 3):
            for rt in quad_roots(D[sp[p]]-D[sp[q]], B[sp[p]]-B[sp[q]], A[sp[p]]-A[sp[q]]):
                if lo <= rt <= hi: pre_cands.append(rt)
    best_t, best_val = 0.0, float(np.max(A))
    if len(aff) > 0:
        segs = build_hull(B[aff], A[aff], lo, hi)
        for (ts, te) in segs:
            if te < ts: continue
            mid_t = 0.5*(ts+te)
            best_k = 0; best_l = -1e30
            for ki in range(len(aff)):
                lv = A[aff[ki]] + B[aff[ki]]*mid_t
                if lv > best_l: best_l, best_k = lv, ki
            k = aff[best_k]
            cands = [ts, te]
            if ts <= 0.0 <= te: cands.append(0.0)
            for sm in sp:
                for rt in quad_roots(D[sm], B[sm]-B[k], A[sm]-A[k]):
                    if ts-1e-12 <= rt <= te+1e-12: cands.append(min(max(rt,ts),te))
            for pc in pre_cands:
                if ts-1e-12 <= pc <= te+1e-12: cands.append(min(max(pc,ts),te))
            for ct in cands:
                ct = min(max(ct, ts), te)
                vals = [A[k]+B[k]*ct]
                for sm in sp:
                    vals.append(A[sm]+B[sm]*ct+D[sm]*ct*ct)
                mv = max(vals)
                if mv < best_val - 1e-15 or (abs(mv-best_val) <= 1e-15 and abs(ct) < abs(best_t)):
                    best_val, best_t = mv, ct
        if best_val >= float(np.max(A)) - 1e-13:
            best_t, best_val = 0.0, float(np.max(A))
    else:
        for ct in pre_cands:
            ct = min(max(ct, lo), hi)
            vals = [A[sm]+B[sm]*ct+D[sm]*ct*ct for sm in sp]
            mv = max(vals)
            if mv < best_val - 1e-15 or (abs(mv-best_val)<=1e-15 and abs(ct)<abs(best_t)):
                best_val, best_t = mv, ct
    return best_t, best_val

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0-fb, np.roll(fb, n//4), np.roll(fb, -n//4)]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1: blk[n//2] = 0.5
    cands.append(blk); cands.append(blk[::-1].copy())
    for sd in (0.04, 0.12, 0.25):
        cands.append(fb + rng.normal(0, sd, n))
    for fp in (1, 2, 3):
        for ph in range(3):
            arr = 0.5 + 0.35*np.sin(2*np.pi*fp*np.arange(n)/n + ph*2.094)
            cands.append(arr)
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None: continue
        dup = False
        for s in pool:
            if np.max(np.abs(p-s)) < 1e-8: dup = True; break
        if not dup: pool.append(p)
        if len(pool) >= count: break
    return pool[:count]

def sweep(h, A, n, rng, max_sweeps, pair_limit):
    n_tot = n*(n-1)//2

                    nsv = float(np.max(cur_A)) if cur_A is not None else cur_s
                if nsv < cur_s - 1e-13:

                    cur[i] -= t_st; cur[j] += t_st
            n_acc += 1
