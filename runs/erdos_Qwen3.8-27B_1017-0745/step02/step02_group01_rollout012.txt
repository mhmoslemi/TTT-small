import numpy as np
from scipy.optimize import linprog
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def build_model(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    A = np.zeros((L, n + 1))
    for k in range(L):
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        A[k, p:p + m] += dx * (1.0 - h[r:r + m])
        A[k, r:r + m] -= dx * h[p:p + m]
        A[k, n] = -1.0
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(A)):
        return None
    return q, A

def lp_step(h, n, rho, q, A):
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = A.copy()
    b_ub = -q
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-rho, -h[i])
        hi = min(rho, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'maxiter': 2000,
                           'primal_feasibility_tolerance': 1e-9,
                           'dual_feasibility_tolerance': 1e-9})
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if abs(np.sum(u)) > 1e-6 * n:
        return None
    for i in range(n):
        if u[i] < max(-rho, -h[i]) - 1e-8 or u[i] > min(rho, 1.0 - h[i]) + 1e-8:
            return None
    return u

def refine_chain(start, n, max_iter, rho0, best_h, best_score):
    current = start.copy()
    cur_score = exact_score(current, n)
    if not np.isfinite(cur_score):
        return best_h, best_score
    if cur_score < best_score - 1e-12:
        best_h, best_score = current.copy(), cur_score
    rho = rho0
    rho_min, rho_max = 1e-5, 0.32
    no_imp = 0
    fail_count = 0
    for _ in range(max_iter):
        if no_imp >= 6 or fail_count >= 3 or rho < rho_min:
            break
        model = build_model(current, n)
        if model is None:
            fail_count += 1
            rho = max(rho_min, 0.5 * rho)
            no_imp += 1
            continue
        q, A = model
        u = lp_step(current, n, rho, q, A)
        if u is None:
            fail_count += 1
            rho = max(rho_min, 0.5 * rho)
            no_imp += 1
            continue
        fail_count = 0
        best_trial, best_ts = None, cur_score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            z = current + alpha * u
            y = project_capped(z, n)
            if y is None:
                continue
            ts = exact_score(y, n)
            if ts < best_ts - 1e-12:
                best_ts, best_trial = ts, y.copy()
            if ts < best_score - 1e-12:
                best_h, best_score = y.copy(), ts
        if best_trial is None:
            rho = max(rho_min, 0.5 * rho)
            no_imp += 1
            continue
        delta = best_trial - current
        actual = cur_score - best_ts
        model_val = np.max(q + A[:, :n] @ delta)
        predicted = cur_score - model_val
        if predicted > 1e-12 and np.isfinite(predicted) and np.isfinite(actual):
            ratio = actual / predicted
        else:
            ratio = 0.0
        current = best_trial
        cur_score = best_ts
        no_imp = 0
        if ratio < 0.1 or ratio <= 0:
            rho = max(rho_min, 0.5 * rho)
        elif ratio > 0.75 and np.max(np.abs(delta)) >= 0.8 * rho:
            rho = min(1.5 * rho, rho_max)
    return best_h, best_score

def add_start(pool, z, n, limit):
    y = project_capped(z, n)
    if y is None:
        return pool
    for s in pool:
        if np.max(np.abs(y - s)) < 1e-8:
            return pool
    if len(pool) >= limit:
        return pool
    pool.append(y)
    return pool

def build_starts(fallback, n, rng):
    pool = [fallback.copy()]
    pool = add_start(pool, fallback[::-1].copy(), n, 10)
    pool = add_start(pool, 1.0 - fallback[::-1].copy(), n, 10)
    pool = add_start(pool, np.full(n, 0.5), n, 10)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool = add_start(pool, blk, n, 10)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    pool = add_start(pool, alt, n, 10)
    pool = add_start(pool, fallback + 0.05 * rng.standard_normal(n), n, 10)
    pool = add_start(pool, fallback + 0.15 * rng.standard_normal(n), n, 10)
    t = np.arange(n, dtype=np.float64)
    pool = add_start(pool, 0.5 + 0.3 * np.cos(2.0 * np.pi * t / n), n, 10)
    return pool

    n = None

    try:

    except Exception:
        pass
    if n is None or n < 1:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:

        except Exception:
            pass
    if parent is not None and n == parent.shape[0]:

    try:

    if n == 1:

    scored = []

        s = exact_score(pool[i], n)
        scored.append((s, i))
