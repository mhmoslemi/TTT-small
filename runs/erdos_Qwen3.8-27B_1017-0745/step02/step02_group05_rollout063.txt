import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(h[i] + r, 1.0)
        else:
            i = int(np.argmax(h))
            h[i] = max(h[i] + r, 0.0)
        r = s - np.sum(h)
    if h.min() < -1e-14 or h.max() > 1.0 + 1e-14:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):

        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def pair_anneal(h, n, rng, L_total):

    sx = exact_score(hx, n)
    best_h, best_s = h.copy(), sx
    for t in range(L_total):
        q = t / max(1, L_total - 1)
        rho = 1.0 - 0.98 * q
        T = 5e-4 * (1e-7 / 5e-4) ** q
        done = False
        for _ in range(20):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:

            ci = min(1.0 - hx[i], hx[j])
            cj = min(1.0 - hx[j], hx[i])

                amt = ci * rho * (0.05 + 0.95 * rng.uniform())

                hx[j] -= amt
            else:

        hp = project_capped(hx, n)

            hp = hx.copy()
        hs = exact_score(hp, n)
        if not np.isfinite(hs):
            continue
        delta = hs - sx
        if delta < -1e-12:
            acc = True
        elif abs(delta) <= 1e-12:
            acc = rng.uniform() < 0.1
        else:
            ratio = delta / T
            acc = rng.uniform() < (math.exp(-ratio) if ratio < -700 else 0.0)
        if acc:
            sx, hx = hs, hp
        if hs < best_s - 1e-12:

    return best_h, best_s

def pair_polish(h, n, rng, n_sweeps, n_pairs_big):
    hx = h.copy()
    sx = exact_score(hx, n)
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for sweep in range(n_sweeps):

                    hx[dj] += amt
                    if hs < sx - 1e-12:
                        sx = hs
                        hx = hp
                        improved = True
        if not improved:

    return hx, sx
