```python
def slsqp_solve(h0, n):
    dx = 2.0 / n
    s_target = n / 2.0
    t0 = exact_score(h0, n) + 1e-6
    z0 = np.concatenate([h0, [t0]])
    
    bounds = [(0, 1)] * n + [(0, 2)]
    
    # Objective: minimize t
    def obj(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    # Inequality: t - S_r >= 0 for r = 0,...,n-1
    def ineq(z):
        h = z[:n]
        t = z[n]
        return t - lag_scaled(h, n)
    
    def ineq_jac(z):
        h = z[:n]
        G = -lag_grad(h, n) * dx  # shape (n, n)
        J = np.hstack([G, np.ones((n, 1))])
        return J
    
    # Equality: sum(h) - n/2 = 0
    def eq(z):
        return np.sum(z[:n]) - s_target
    
    def eq_jac(z):
        J = np.zeros(n+1)
        J[:n] = 1.0
        return J
    
    constraints = [{'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                   {'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    
    res = minimize(obj, z0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=constraints,
                   options={'ftol': 1e-10, 'maxiter': 220, 'disp': False})
    
    if res.x is not None and len(res.x) == n + 1:
        hr = project_capped(res.x[:n], n)
        if hr is not None and is_feasible(hr, n):
            return hr
    return None
```

```python
def lag_values(h, n):
    """Return Q_r for r=0,...,n-1 (n values)."""
    Q = np.empty(n)
    Q[0] = np.dot(h, 1-h)
    for r in range(1, n):
        m = n - r
        Q[r] = np.dot(h[r:], 1-h[:m])
    return Q

def lag_grad(h, n):
    """Return gradient matrix G where G[r, i] = dQ_r/dh_i."""
    G = np.zeros((n, n))
    G[0] = 1 - 2*h
    for r in range(1, n):
        m = n - r
        G[r, :m] -= h[r:]
        G[r, r:] += 1 - h[:m]
    return G
```

```python
                       jac=_slsqp_obj_grad, bounds=bounds, constraints=cons,
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and len(res.x) == n + 1:
            hr = project_capped(res.x[:n], n)
            if hr is not None and is_feasible(hr, n):
                return hr
    except Exception:
        pass
    return None

def _smooth_val_grad(h, n, tau):
    dx = 2.0 / n
    Q = lag_values(h, n)
    S = Q * dx
    qmax = np.max(S)
    exps = np.exp((S - qmax) / tau)
    Z = np.sum(exps)
    p = exps / Z
    F = qmax + tau * np.log(Z)
    g = np.zeros(n)
    g += dx * p[0] * (1.0 - 2.0 * h)
    for r in range(1, n):
        if p[r] < 1e-16:
            continue
        m = n - r
        pr = dx * p[r]
        g[:m] -= pr * h[r:]
        g[r:] += pr * (1.0 - h[:m])
    return F, g

def smooth_refine(h, n):
    h_best = h.copy()
    s_best = exact_score(h, n)
    taus = [0.02, 0.005, 0.001, 0.0002]
    n_it = 15 if n <= 64 else (10 if n <= 128 else 8)
    for tau in taus:
        h_cur = h_best.copy()
        for _ in range(n_it):
            F, g = _smooth_val_grad(h_cur, n, tau)
            if not np.all(np.isfinite(g)):
                break
            alpha = 1.0
            accepted = False
            for _ in range(20):
                y = project_capped(h_cur - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                Fy, _ = _smooth_val_grad(y, n, tau)
                if Fy <= F + 1e-4 * np.dot(g, y - h_cur) + 1e-14:
                    if np.max(np.abs(y - h_cur)) < 1e-9:
                        return h_best
                    sy = exact_score(y, n)
                    if np.isfinite(sy) and sy < s_best - 1e-12:
                        s_best = sy
                        h_best = y.copy()
                    h_cur = y.copy()
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
        h_best = h_best.copy()
    return h_best

def _try_add(pool, v, th=1e-9):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def build_starts(fb, n, rng, parent):
    starts = []
    max_starts = 10 if n <= 64 else (7 if n <= 128 else 4)
    _try_add(starts, fb.copy())
    _try_add(starts, project_capped(fb[::-1].copy(), n))
    _try_add(starts, project_capped(1.0 - fb, n))
    _try_add(starts, project_capped(1.0 - fb[::-1], n))
    _try_add(starts, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(starts, project_capped(blk, n))
    if parent is not None:
        _try_add(starts, project_capped(parent + rng.normal(0, 0.08, n), n))
        _try_add(starts, project_capped(parent + rng.normal(0, 0.22, n), n))
        _try_add(starts, project_capped(parent + rng.normal(0, 0.40, n), n))
    x = np.arange(n, dtype=np.float64)
    for k in range(1, 4):
        _try_add(starts, project_capped(0.5 + 0.4 * np.cos(2.0 * np.pi * k * x / n), n))
        _try_add(starts, project_capped(0.5 + 0.4 * np.sin(2.0 * np.pi * k * x / n), n))
    _try_add(starts, project_capped(rng.uniform(0, 1, n), n))
    p = rng.permutation(n)
    v = np.zeros(n)
    if n % 2 == 0:
        v[p[:n // 2]] = 1.0
    else:
        v[p[:n // 2]] = 1.0
        v[p[n // 2]] = 0.5
    _try_add(starts, v)
    _try_add(starts, project_capped(np.roll(fb, n // 3), n))
    _try_add(starts, project_capped(np.roll(fb, n // 2), n))
    return starts[:max_starts]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if not (1 <= n <= 999):
            n = 47
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.5, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    starts = build_starts(fb, n, rng, parent)
    best_h = fb.copy()
    best_s = exact_score(fb, n)
    maxiter_sl = 250 if n <= 64 else 150
    for s in starts:
        h_sl = slsqp_solve(s, n, maxiter_sl) if n <= 128 else None
        cands = []
        if h_sl is not None:
            cands.append(h_sl)
            for a in (0.25, 0.5, 0.75):
                cands.append(project_capped((1.0 - a) * s + a * h_sl, n))
        h_sm = smooth_refine(s, n)
        if h_sm is not None:
            cands.append(h_sm)
        if h_sl is not None:
            h_sm2 = smooth_refine(h_sl, n)
            if h_sm2 is not None:
                cands.append(h_sm2)
        for hc in cands:
            if hc is not None and is_feasible(hc, n):
                sc = exact_score(hc, n)
                if np.isfinite(sc) and sc < best_s - 1e-12:
                    best_s = sc
                    best_h = hc.copy()
    if not is_feasible(best_h, n):
        best_h = fb.copy()
    best_s = exact_score(best_h, n)
    if not np.isfinite(best_s):
        best_h = fb.copy()
        best_s = exact_score(fb, n)
    best_h = project_capped(best_h, n)
    if best_h is None:
        best_h = fb.copy()
        best_s = exact_score(fb, n)
    else:
        fs = exact_score(best_h, n)
        if np.isfinite(fs) and fs < best_s - 1e-12:
            best_s = fs
        elif not np.isfinite(best_s):
            best_s = fs
    return best_h.astype(np.float64), float(best_s), int(n)
```
