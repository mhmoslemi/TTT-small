Let me think about the gradient computation for the soft objective:

For lag d (from -(n-1) to n-1), the correlation at that lag is:
q_d = dx * sum_{s=0}^{m-1} h[p+s] * (1 - h[r+s])

where p = max(d,0), r = max(-d,0), m = n - |d|

The gradient of q_d with respect to h:
- For coordinate i = p+s (the h factor): dq_d/dh_i = dx * (1 - h[r+s])
- For coordinate i = r+s (the (1-h) factor): dq_d/dh_i = -dx * h[p+s]

Note: when d=0, p=r=0, so both contributions go to the same coordinate.

For the soft gradient:
grad S_tau = sum_d w_d * grad q_d

where w_d = exp((q_d - q_max)/tau) / sum_j exp((q_j - q_max)/tau)

This is efficient - we don't need to store all lag gradients, just accumulate weighted contributions.

        if m <= 0 or wk < 1e-16:
            continue
        g[p0:p0 + m] += wk * dx * (1.0 - h[r0:r0 + m])
        g[r0:r0 + m] -= wk * dx * h[p0:p0 + m]
    if not np.all(np.isfinite(g)):
        return 1e30, np.zeros(n)
    return val, g

def soft_fun(x, n, tau):
    return soft_obj_grad(x, n, tau)

def mass_eq(x, n, tau):
    return np.sum(x) - n / 2.0

def mass_jac(x, n, tau):
    return np.ones(n)

def slsqp_stage(h0, n, tau, maxiter):
    try:
        res = minimize(soft_fun, h0, jac=True, args=(n, tau),
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': mass_eq, 'jac': mass_jac, 'args': (n, tau)}],
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and hasattr(res.x, 'shape') and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
            return project_capped(res.x, n)
    except Exception:
        pass
    return None

def optimize_chain(h0, n, taus, maxiter):
    cur = h0.copy()
    for tau in taus:
        y = slsqp_stage(cur, n, tau, maxiter)
        if y is None:
            continue
        v0, _ = soft_obj_grad(cur, n, tau)
        v1, _ = soft_obj_grad(y, n, tau)
        if np.isfinite(v1) and np.isfinite(v0) and v1 <= v0 + 1e-8:
            cur = y
        else:
            for a in (0.5, 0.25, 0.125):
                z = project_capped(cur + a * (y - cur), n)
                if z is None:
                    continue
                vz, _ = soft_obj_grad(z, n, tau)
                if np.isfinite(vz) and vz <= v0 + 1e-8:
                    cur = z
                    break
    return cur

def subgrad_polish(h0, n, n_iter=40):
    cur = h0.copy()
    score = exact_score(cur, n)
    dx = 2.0 / n
    eta = 0.5 / n
    failed = 0
    for _ in range(n_iter):
        if failed >= 10 or eta < 1e-14:
            break
        q = np.correlate(cur, 1.0 - cur, mode='full') * dx
        k = int(np.argmax(q))
        d = k - (n - 1)
        p0, r0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - cur[r0:r0 + m])
        g[r0:r0 + m] -= dx * cur[p0:p0 + m]
        z = project_capped(cur - eta * g, n)
        if z is None:
            eta *= 0.5
            failed += 1
            continue
        best_z, best_s = None, score
        for a in (1.0, 0.5, 0.25, 0.125):
            t = z if a >= 1.0 - 1e-15 else project_capped(cur + a * (z - cur), n)
            if t is None:
                continue
            ts = exact_score(t, n)
            if ts < best_s - 1e-13:
                best_s, best_z = ts, t.copy()
        if best_z is None:
            eta *= 0.5
            failed += 1
        else:
            cur, score = best_z, best_s
            failed = 0
            eta = min(eta * 1.12, 2.0 / n)
    return cur, score

def add_candidate(pool, c, n, limit):
    if len(pool) >= limit:
        return
    p = project_capped(c, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p)

def build_starts(fallback, n, rng, limit):
    pool = [fallback.copy()]
    add_candidate(pool, fallback[::-1].copy(), n, limit)
    add_candidate(pool, (1.0 - fallback).copy(), n, limit)
    add_candidate(pool, (1.0 - fallback[::-1]).copy(), n, limit)
    add_candidate(pool, np.full(n, 0.5), n, limit)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_candidate(pool, blk, n, limit)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_candidate(pool, alt, n, limit)
    for sd in (0.05, 0.12, 0.25):
        add_candidate(pool, fallback + rng.normal(0, sd, n), n, limit)
    add_candidate(pool, rng.uniform(0, 1, n), n, limit)
    t = np.arange(n, dtype=np.float64)
    add_candidate(pool, 0.5 + 0.35 * np.cos(2.0 * np.pi * t / n), n, limit)
    add_candidate(pool, 0.5 + 0.3 * np.cos(4.0 * np.pi * t / n) + 0.1 * np.sin(2.0 * np.pi * t / n), n, limit)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)):
                pr = project_capped(p, n)
                if pr is not None:
                    parent = pr
    except Exception:
        pass
    if n is None or n < 1:
        n = 47
    if parent is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
        parent = np.full(n, 0.5)
    if n == 1:
        return parent, exact_score(parent, n), n
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fallback = parent.copy()
    fscore = exact_score(fallback, n)
    pool_limit = 12 if n <= 64 else 8
    starts = build_starts(fallback, n, rng, pool_limit)
    best_h = fallback.copy()
    best_s = fscore
    sscores = []
    for st in starts:
        sc = exact_score(st, n)
        sscores.append(sc)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = st.copy()
    n_chain = 6 if n <= 64 else (3 if n <= 256 else 2)
    idx_order = sorted(range(len(starts)), key=lambda i: sscores[i])
    sel = [0]
    for i in idx_order:
        if len(sel) >= n_chain:
            break
        if i not in sel:
            sel.append(i)
    taus = [0.01, 0.003, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
    maxit = max(30, min(150, int(round(150 * math.sqrt(47.0 / n)))))
    for si in sel:
        h0 = starts[si]
        try:
            x_opt = optimize_chain(h0, n, taus, maxit)
        except Exception:
            x_opt = None
        if x_opt is not None:
            xo = project_capped(x_opt, n)
            if xo is not None:
                so = exact_score(xo, n)
                if so < best_s - 1e-12:
                    best_s = so
                    best_h = xo.copy()
                for a in (0.75, 0.5, 0.25):
                    z = project_capped(h0 + a * (xo - h0), n)
                    if z is not None:
                        zs = exact_score(z, n)
                        if zs < best_s - 1e-12:
                            best_s = zs
                            best_h = z.copy()
        try:
            x_pg, s_pg = subgrad_polish(x_opt if (x_opt is not None and xo is not None) else h0, n, 40)
            if s_pg < best_s - 1e-12:
                best_s = s_pg
                best_h = x_pg.copy()
        except Exception:
            pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fscore + 1e-12:
        final = fallback
        fs = fscore
    return final, float(fs), int(n)
