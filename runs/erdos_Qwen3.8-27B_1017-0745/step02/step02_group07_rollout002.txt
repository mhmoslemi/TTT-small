    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    try_add(pool, project_capped(np.roll(fb, n // 2), n))
    for sd in (0.08, 0.22):
        try_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    for k in (2, 3, 5):
        phase = 0.0 if k % 2 == 0 else math.pi / 3.0
        x = 0.5 + 0.4 * np.cos(2.0 * math.pi * k * np.arange(n) / n + phase)
        try_add(pool, project_capped(x, n))
    p = rng.permutation(n)
    rb = np.zeros(n)
    rb[p[:n // 2]] = 1.0
    if n % 2:
        rb[p[n // 2]] = 0.5
    try_add(pool, project_capped(rb, n))
    max_starts = 10 if n <= 64 else (7 if n <= 128 else 4)
    while len(pool) > max_starts:
        pool.pop()
    return pool

def solve_epigraph(h0, n, rng):
    t0 = exact_score(h0, n) + 1e-8
    if not np.isfinite(t0):
        return None
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
    ]
    mi = 220 if n <= 64 else 140
    try:
        res = minimize(slsqp_obj, z0, jac=slsqp_obj_grad, args=(n,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and len(res.x) == n + 1:
            return project_capped(res.x[:n], n)
    except Exception:
        pass
    return None

def smooth_grad(h, n, tau):
    Q = lag_values(h, n)
    qmax = np.max(Q)
    shifted = (Q - qmax) / tau
    shifted = np.clip(shifted, -700, 0)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    g = np.zeros(n)
    dx = 2.0 / n
    g += p[0] * dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        g[:m] += p[idx] * dx * (1.0 - h[r:])
        g[r:] -= p[idx] * dx * h[:m]
        g[:m] -= p[idx + 1] * dx * h[r:]
        g[r:] += p[idx + 1] * dx * (1.0 - h[:m])
        idx += 2
    return g

def smooth_f(h, n, tau):
    Q = lag_values(h, n)
    qmax = np.max(Q)
    shifted = np.clip((Q - qmax) / tau, -700, 0)
    return qmax + tau * np.log(np.sum(np.exp(shifted)))

def smooth_refine(h, n, iters):
   _best = h.copy()
    best_sc = exact_score(h, n)
    h = h.copy()
    for tau in (0.02, 0.005, 0.001, 0.0001):
        alpha = 1.0
        cur_f = smooth_f(h, n, tau)
        for _ in range(iters):
            g = smooth_grad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _bt in range(20):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                fy = smooth_f(y, n, tau)
                d = np.dot(g, y - h)
                if fy <= cur_f + 0.1 * d + 1e-14:
                    h = y
                    cur_f = fy
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            sc = exact_score(h, n)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = h.copy()
            alpha = min(1.5 * alpha, 10.0)
    sc = exact_score(best, n)
    return best, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_sc = exact_score(fb, n)
    best_h = fb.copy()
    best_sc = fb_sc
    starts = build_starts(fb, n, rng)
    n_smooth = 8 if n <= 128 else 20
    for h0 in starts:
        if n <= 128:
            hr = solve_epigraph(h0, n, rng)
            if hr is not None:
                sc = exact_score(hr, n)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_h = hr.copy()
            for a in (0.25, 0.5, 0.75):
                mix = project_capped((1.0 - a) * h0 + a * best_h, n)
                if mix is not None:
                    sc = exact_score(mix, n)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_h = mix.copy()
        hn, hn_sc = smooth_refine(h0, n, n_smooth)
        if hn_sc < best_sc - 1e-12:
            best_sc = hn_sc
            best_h = hn
    hn, hn_sc = smooth_refine(best_h, n, n_smooth)
    if hn_sc < best_sc - 1e-12:
        best_sc = hn_sc
        best_h = hn
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
        final_sc = fb_sc
    else:
        final_sc = exact_score(final, n)
        if not np.isfinite(final_sc) or final_sc > best_sc + 1e-12:
            final = fb
            final_sc = fb_sc
    return final.astype(np.float64), float(final_sc), int(n)
