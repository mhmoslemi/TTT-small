    best, best_score = uniform.copy(), fb_score
    parent_fallback = None
    if parent is not None:
        if feasible(parent, n):
            ps = exact_score(parent, n)
            if math.isfinite(ps):
                parent_fallback = parent.copy()
                if ps < best_score:
                    best, best_score = parent.copy(), ps
            else:
                parent_fallback = project_capped(parent, n)
        else:
            pp = project_capped(parent, n)
            if pp is not None:
                parent_fallback = pp
                ps = exact_score(pp, n)
                if math.isfinite(ps) and ps < best_score:
                    best, best_score = pp.copy(), ps

    if n <= 1:
        return best, float(best_score), n

    starts = []
    def add_start(v):
        try:
            v = np.asarray(v, dtype=np.float64).ravel()
            if v.shape != (n,) or not np.all(np.isfinite(v)):
                return
            p = project_capped(v, n)
            if p is None:
                return
            for d in starts:
                if float(np.max(np.abs(p - d))) < 1e-8:
                    return
            starts.append(p)
        except Exception:
            pass

    if parent is not None:
        add_start(parent)
        for s2 in [0.03, 0.08, 0.16]:
            add_start(parent + rng.normal(0, s2, n))
        add_start(1.0 - parent)
        add_start(parent[::-1])
        add_start(1.0 - parent[::-1])
    add_start(uniform)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        for phase in [0.0, math.pi / 2]:
            add_start(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n + phase))
    add_start(rng.uniform(0, 1, n))

    if n <= 64:
        d_chains, d_epochs, d_mpe = 5, 28, 180
    elif n <= 128:
        d_chains, d_epochs, d_mpe = 3, 20, 100
    elif n <= 256:
        d_chains, d_epochs, d_mpe = 2, 14, 45
    elif n <= 512:
        d_chains, d_epochs, d_mpe = 2, 10, 18
    else:
        d_chains, d_epochs, d_mpe = 1, 6, 8

    try:
        chains = max(1, min(int(kwargs.get('chains', d_chains)), 8))
    except Exception:
        chains = d_chains
    try:
        epochs = max(1, min(int(kwargs.get('epochs', d_epochs)), 40))
    except Exception:
        epochs = d_epochs
    try:
        mpe = max(1, min(int(kwargs.get('moves_per_epoch', d_mpe)), 300))
    except Exception:
        mpe = d_mpe
    try:
        delta0 = max(0.01, min(float(kwargs.get('proposal_scale', 0.12)), 0.30))
    except Exception:
        delta0 = 0.12
    try:
        pilot_n = max(4, min(int(kwargs.get('pilot_moves', 16)), 40))
    except Exception:
        pilot_n = 16

    cap = max(chains * epochs * mpe, int(80000000.0 / (n * n)))
    total_remaining = cap

    for ci in range(chains):
        if total_remaining <= 0:
            break
        if ci < len(starts):
            cur = starts[ci].copy()
        else:
            scale = 0.04 * (1.5 ** (ci - len(starts)))
            scale = min(scale, 0.20)
            cand = project_capped(best + rng.normal(0, scale, n), n)
            cur = cand if cand is not None else uniform.copy()
        cur_score = exact_score(cur, n)
        if not math.isfinite(cur_score):
            cur, cur_score = uniform.copy(), fb_score

        ups = []
        for _ in range(min(pilot_n, total_remaining)):
            total_remaining -= 1
            p = make_proposal(cur, n, rng, delta0)
            if p is not None:
                ps = exact_score(p, n)
                if math.isfinite(ps) and ps > cur_score + 1e-12:
                    ups.append(ps - cur_score)
        if ups:
            T0 = min(2e-2, max(1e-5, float(np.median(ups)) / (-math.log(0.75))))
        else:
            T0 = 0.002
        Tmin = max(1e-7, 1e-3 * T0)

        for ep in range(epochs):
            if total_remaining <= 0:
                break
            T = T0 * (Tmin / T0) ** (ep / max(1, epochs - 1)) if epochs > 1 else T0
            accepted, attempted = 0, 0
            for _ in range(min(mpe, total_remaining)):
                total_remaining -= 1
                attempted += 1
                prop = make_proposal(cur, n, rng, delta)
                if prop is None:
                    continue
                ps = exact_score(prop, n)
                if not math.isfinite(ps):
                    continue
                if ps < best_score - 1e-12:
                    best, best_score = prop.copy(), ps
                d = ps - cur_score
                if d <= 1e-14:
                    accepted += 1
                    cur, cur_score = prop, ps
                else:
                    u = rng.random()
                    if math.log(u) < -d / T:
                        accepted += 1
                        cur, cur_score = prop, ps
            rate = accepted / max(1, attempted)
            if rate > 0.40:
                delta = min(0.35, delta * 1.25)
            elif rate < 0.15:
                delta = max(0.002, delta * 0.75)

        if math.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best, best_score = cur.copy(), cur_score

    if n <= 128:
        np_sweeps = 4 if n <= 64 else 2
        polished = pair_polish(best, n, np_sweeps)
        ps = exact_score(polished, n)
        if math.isfinite(ps) and ps < best_score - 1e-12:
            best, best_score = polished.copy(), ps
        polished2 = pair_polish(best, n, np_sweeps)
        ps2 = exact_score(polished2, n)
        if math.isfinite(ps2) and ps2 < best_score - 1e-12:
            best, best_score = polished2.copy(), ps2

    final = project_capped(best, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not math.isfinite(fs) or fs > best_score + 1e-9:
        final, fs = best.copy(), best_score
    if not math.isfinite(fs):
        final, fs = uniform.copy(), fb_score
    if parent_fallback is not None:
        pf = project_capped(parent_fallback, n)
        if pf is not None:
            pfs = exact_score(pf, n)
            if math.isfinite(pfs) and pfs < fs:
                final, fs = pf, pfs
    final = project_capped(np.clip(final, 0.0, 1.0), n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not math.isfinite(fs):
        final, fs = uniform.copy(), fb_score
    return np.clip(final, 0.0, 1.0), float(fs), int(n)
