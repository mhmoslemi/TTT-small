    _try_add(pool, project_capped(1.0 - fb, n), n)
    _try_add(pool, project_capped((1.0 - fb)[::-1].copy(), n), n)
    _try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    _try_add(pool, project_capped(alt, n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.05, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.15, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.25, n), n), n)
    P = max(2, n // 3)
    cosv = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P)
    _try_add(pool, project_capped(cosv, n), n)
    _try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.asarray(fb, dtype=np.float64)
    if n == 1:
        return fb, float(exact_score(fb, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, maxiter, n_hops = 10, 300, 10
    elif n <= 128:
        n_starts, maxiter, n_hops = 7, 200, 6
    elif n <= 256:
        n_starts, maxiter, n_hops = 4, 120, 3
    else:
        n_starts, maxiter, n_hops = 2, 60, 0
    starts = build_starts(fb, n, rng, n_starts)
    gb = fb.copy()
    gs = float(exact_score(fb, n))
    for st in starts:
        s0 = float(exact_score(st, n))
        y = solve_slsqp(st, n, maxiter)
        best_x, best_s = st, s0
        if y is not None and is_feasible(y, n):
            sy = float(exact_score(y, n))
            if sy < best_s:
                best_x, best_s = y, sy
        if best_s < gs - 1e-13:
            gs, gb = best_s, best_x.copy()
    # Basin hopping
    T = 3.0e-3
    scales = [0.04, 0.10, 0.20, 0.30]
    cur_x = gb.copy()
    cur_s = gs
    for hop in range(n_hops):
        sig = scales[hop % len(scales)]
        if hop % 5 == 4:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            prop = project_capped(cur_x + sig * rng.normal(0, 1, n), n)
        if prop is None:
            continue
        py = solve_slsqp(prop, n, maxiter)
        cand = py if (py is not None and is_feasible(py, n)) else prop
        cs = float(exact_score(cand, n))
        if not np.isfinite(cs):
            continue
        if cs < gs - 1e-13:
            gs, gb = cs, cand.copy()
        if cs <= cur_s or rng.random() < np.exp(-(cs - cur_s) / max(1e-5, T)):
            cur_x, cur_s = cand, cs
        T *= 0.75
    # Final polish: re-solve from global best
    fp = solve_slsqp(gb, n, maxiter // 2)
    if fp is not None and is_feasible(fp, n):
        fs = float(exact_score(fp, n))
        if fs < gs - 1e-13:
            gs, gb = fs, fp.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fb.copy()
    gs = float(exact_score(gb, n))
    fs = float(exact_score(fb, n))
    if gs > fs + 1e-13:
        gb, gs = fb.copy(), fs
    if gs == float('inf'):
        gb = np.full(n, 0.5)
        gs = float(exact_score(gb, n))
    return gb, gs, int(n)
