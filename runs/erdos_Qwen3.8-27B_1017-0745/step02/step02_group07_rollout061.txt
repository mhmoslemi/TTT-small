import numpy as np
import math
import cvxpy as cp

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-15:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= -1e-15:
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')

def lag_data(h, n):
    dx = 2.0 / n
    m_lags = 2 * n - 1
    q = np.zeros(m_lags)
    G = np.zeros((m_lags, n))
    idx = n - 1
    q[idx] = dx * np.sum(h * (1.0 - h))
    G[idx] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        j = n - 1 + r
        q[j] = dx * np.sum(h[r:r+m] * (1.0 - h[:m]))
        G[j, r:r+m] += dx * (1.0 - h[:m])
        G[j, :m] += -dx * h[r:r+m]
        j2 = n - 1 - r
        q[j2] = dx * np.sum(h[:m] * (1.0 - h[r:r+m]))
        G[j2, :m] += dx * (1.0 - h[r:r+m])
        G[j2, r:r+m] += -dx * h[:m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def build_qp(n):
    dx = 2.0 / n
    L = 2.0 * dx
    n_lags = 2 * n - 1
    d = cp.Variable(n)
    u = cp.Variable()
    q_p = cp.Parameter(n_lags)
    G_p = cp.Parameter((n_lags, n))
    h_p = cp.Parameter(n)
    obj = cp.Minimize(u + (L / 2.0) * cp.sum_squares(d))
    con = [G_p @ d + q_p <= u, cp.sum(d) == 0, d >= -h_p, d <= 1.0 - h_p]
    prob = cp.Problem(obj, con)
    return prob, q_p, G_p, h_p

def get_solver_opts(solver):
    if solver == cp.ECOS:
        return {'iter_max': 300, 'abstol': 1e-8, 'reltol': 1e-8}
    elif solver == cp.MOSEK:
        return {}
    elif solver == cp.GUROBI:
        return {'Threads': 1, 'OutputFlag': 0}
    elif solver == cp.CVXOPT:
        return {'abstol': 1e-8, 'reltol': 1e-8, 'maxiter': 300}
    return {}

def solve_qp(prob, q, G, h, solvers):
    for solver in solvers:
        try:
            opt = get_solver_opts(solver)
            prob.solve(solver=solver, **opt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = prob.variables()[0].value
                if d_val is not None and d_val.shape == (len(h),) and np.all(np.isfinite(d_val)):
                    return d_val
        except Exception:
            continue
    return None

def try_add(pool, v, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def build_pool(fb, n, rng):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n))
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(pool, project_capped(bv, n))
    try_add(pool, project_capped(fb + 0.06 * rng.normal(0, 1, n), n))
    try_add(pool, project_capped(fb + 0.18 * rng.normal(0, 1, n), n))
    try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    for k in range(1, 4):
        for ph in (0.0, math.pi / 2):
            cv = 0.5 + 0.35 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph)
            try_add(pool, project_capped(cv, n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    while len(pool) < 16:
        pool.append(pool[0].copy())
    return pool[:16]

def mm_optimize(h0, n, prob, q_p, G_p, h_p, solvers, max_iter):
    h = h0.copy()
    sc = exact_score(h, n)
    small_cnt = 0
    for it in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        q_p.value = q
        G_p.value = G
        h_p.value = h
        d_val = solve_qp(prob, q, G, h, solvers)
        if d_val is None:
            break
        if np.max(np.abs(d_val)) <= 1e-8:
            break
        best_trial = None
        best_sc = sc
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = project_capped(h + alpha * d_val, n)
            if cand is None:
                continue
            if np.max(np.abs(cand - h)) < 1e-10:
                continue
            cs = exact_score(cand, n)
            if cs < best_sc - 1e-12:
                best_sc = cs
                best_trial = cand
        if best_trial is None:
            break
        reduction = sc - best_sc
        sc = best_sc
        h = best_trial
        if reduction < 1e-10:
            small_cnt += 1
            if small_cnt >= 3:
                break
        else:
            small_cnt = 0
    return h, sc
