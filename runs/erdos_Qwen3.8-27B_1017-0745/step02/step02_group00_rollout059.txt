def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()

        return float('inf')

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_value_grad(h, tau, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        q[row] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    m = float(np.max(q))
    exp_q = np.exp((q - m) / tau)
    S = float(np.sum(exp_q))
    w = exp_q / S
    F = m + tau * float(np.log(S))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        row = d + n - 1
        wd = w[row]
        if wd < 1e-14:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += dx * wd * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= dx * wd * h[lo:hi]
    return F, grad
