    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if (np.all(np.isfinite(p)) and np.all(p >= -1e-8)
                    and np.all(p <= 1.0 + 1e-8)
                    and abs(np.sum(p) - p.size / 2.0) <= 1e-6 * max(1, p.size)):
                parent = project_capped(np.clip(p, 0.0, 1.0), n)
    except Exception:
        pass
    if n is None:
        try:
            nv = int(kwargs.get('n_points', 0))
            if 2 <= nv <= 999:
                n = nv
        except Exception:
            pass
        if n is None:
            n = 47
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    unif = np.full(n, 0.5)
    fallback = parent if parent is not None else unif
    gb = fallback.copy()
    gs = exact_score(gb, n)
    if not np.isfinite(gs):
        gb = unif.copy()
        gs = exact_score(gb, n)
    if n <= 64:
        n_starts, n_refine, maxiter = 8, 4, 220
    elif n <= 128:
        n_starts, n_refine, maxiter = 5, 2, 150
    else:
        n_starts, n_refine, maxiter = 3, 1, 80
    starts = build_starts(parent, n, rng)
    scored = []
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc <= 0.5 + 1e-12:
            scored.append((sc, s))
    scored.sort(key=itemgetter(0))
    us = exact_score(unif, n)
    if np.isfinite(us) and us <= 0.5 + 1e-12:
        dup = False
        for sc, s in scored:
            if np.max(np.abs(s - unif)) < 1e-6:
                dup = True
                break
        if not dup:
            scored.append((us, unif))
    picked = scored[:n_starts] if len(scored) > 0 else [(us, unif)]
    for sc, h0 in picked:
        t0 = max(0.0, min(0.5, sc + 1e-9))
        z0 = np.concatenate([h0, [t0]])
        hc = solve_slsqp(z0, n, maxiter)
        if hc is None:
            continue
        hs = exact_score(hc, n)
        if np.isfinite(hs) and hs < gs - 1e-12:
            gs, gb = hs, hc.copy()
        for alpha in (0.25, 0.5, 0.75):
            blend = alpha * gb + (1.0 - alpha) * hc
            bp = project_capped(blend, n)
            if bp is None:
                continue
            bs = exact_score(bp, n)
            if np.isfinite(bs) and bs < gs - 1e-12:
                gs, gb = bs, bp.copy()
    sigma = 0.04
    for _r in range(n_refine):
        improved = False
        for _a in range(3):
            pert = gb + rng.normal(0, sigma, n)
            pp = project_capped(pert, n)
            if pp is None:
                continue
            ps = exact_score(pp, n)
            if not np.isfinite(ps) or ps > 0.5 + 1e-12:
                continue
            if np.max(np.abs(pp - gb)) < 1e-6:
                continue
            t0 = max(0.0, min(0.5, ps + 1e-9))
            z0 = np.concatenate([pp, [t0]])
            hc = solve_slsqp(z0, n, maxiter)
            if hc is not None:
                hs = exact_score(hc, n)
                if np.isfinite(hs) and hs < gs - 1e-12:
                    gs, gb = hs, hc.copy()
                    improved = True
            break
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fscore = exact_score(final, n)
    if not np.isfinite(fscore):
        final = unif.copy()
        fscore = exact_score(final, n)
    return final.astype(np.float64), float(fscore), int(n)
