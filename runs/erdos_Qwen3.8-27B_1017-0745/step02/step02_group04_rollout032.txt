            if noise is None:

            else:

            else:
                pops.append(mu.copy())

        w = np.array([q - j for j in range(q)], dtype=np.float64)

        new_mu = np.zeros(n)

            new_mu += w[idx] * pops[e]
        new_mu = project_capped(new_mu, n)
        if new_mu is None:
            new_mu = mu.copy()
        new_s = exact_score(new_mu, n)
        # Update best
        if new_s < best_s - 1e-12:
            best, best_s = new_mu.copy(), new_s
            no_improve = 0
        else:
            no_improve += 1
        # Sigma adaptation
        n_succ = sum(1 for s_ in scores[1:] if s_ < cur_s - 1e-12)
        p_succ = n_succ / max(lam, 1)
        if p_succ > 0.25:
            sigma = min(0.25, 1.12 * sigma)
        elif p_succ < 0.08:
            sigma = max(5e-4, 0.85 * sigma)
        # Covariance adaptation
        if eigs is not None and V is not None:

                    d = pops[e] - new_mu
                    E += w[idx] * np.outer(d, d)

                if tr > 1e-10:

                    eigs2, V2 = np.linalg.eigh(0.5 * (E + E.T))

def project_capped(z, n):
    if z is None or z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 4):
        if abs(r) < 1e-13:
            break
        if r > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(h[i] + r, 1.0)
        else:
            i = int(np.argmax(h))
            h[i] = max(h[i] + r, 0.0)
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10:
        return None
    return h

def is_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    return not (np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(np.sum(h) - n / 2.0) > 1e-10)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c))
    except Exception:
        return float('inf')

def solve_sdp(n):
    if not _HAS_CP or n < 2 or n > 64:
        return None, None
    try:
        h = cp.Variable(n)
        Y = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        s = n / 2.0
        dx = 2.0 / n
        M = cp.bmat([[cp.Constant([[1.0]]), h.reshape(1, -1)], [h.reshape(-1, 1), Y]])
        ones_arr = np.ones((n, n))
        cons = [M >> 0, h >= 0, h <= 1, cp.sum(h) == s,
                Y >= 0, Y <= h[:, None] * np.ones((1, n)),
                Y <= np.ones((n, 1)) * h[None, :],
                Y >= h[:, None] + h[None, :] - ones_arr,
                Y @ np.ones((n, 1)) == s * h.reshape(-1, 1)]
        for d in range(-(n - 1), n):
            p0, q0 = max(d, 0), max(-d, 0)
            m = n - abs(d)
            su = sum(h[p0 + j] for j in range(m))
            sy = sum(Y[p0 + j, q0 + j] for j in range(m))
            cons.append(dx * (su - sy) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        for solver in ['MOSEK', 'GUROBI', 'CVXOPT']:
            try:
                kw = {}
                if solver == 'CVXOPT':
                    kw = {'max_iters': 400, 'abstol': 1e-8, 'reltol': 1e-8}
                prob.solve(solver=solver, **kw)
            except Exception:
                continue
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if h.value is not None and Y.value is not None:
                    hv = np.array(h.value, dtype=np.float64).ravel()
                    Yv = np.array(Y.value, dtype=np.float64)
                    if np.all(np.isfinite(hv)) and np.all(np.isfinite(Yv)):
                        return hv, Yv
            break
    except Exception:
        pass
    return None, None

def make_eig(Y, u, n):
    try:
        S = Y - np.outer(u, u)
        S = 0.5 * (S + S.T)
        P = np.eye(n) - np.ones((n, n)) / n
        S = P @ S @ S if False else P @ S @ P
        eigs, V = np.linalg.eigh(0.5 * (S + S.T))
        eigs = np.clip(eigs, 0.0, n)
        if np.sum(eigs) < 1e-15:
            return None, None
        return eigs, V
    except Exception:
        return None, None

def cma_es(mu, sigma, eigs, V, rng, n, n_gen, lam):
    best = mu.copy()
    best_s = exact_score(mu, n)
    cur = mu.copy()
    cur_s = best_s
    I = np.eye(n)
    st = I - np.ones((n, n)) / n
    no_imp = 0
    q = max(2, (lam + 1) // 4)
    base_eigs = eigs.copy() if eigs is not None else np.ones(n) * 0.04
    base_V = V.copy() if V is not None else I
    for gen in range(n_gen):
        pops = [cur.copy()]
        scores = [cur_s]
        for _ in range(lam):

            xi = rng.standard_normal(n)

            if p is not None:
