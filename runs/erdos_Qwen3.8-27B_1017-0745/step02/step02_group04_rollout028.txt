        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_pop(fb, n, rng, P):
    pool = []
    try_add(pool, fb.copy(), n)
    try_add(pool, project_capped(fb[::-1].copy(), n), n)
    try_add(pool, project_capped(1.0 - fb, n), n)
    try_add(pool, project_capped((1.0 - fb)[::-1].copy(), n), n)
    try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n), n)
    Q = max(2, n // 3)
    cosv = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / Q)
    try_add(pool, project_capped(cosv, n), n)
    cosv2 = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / max(2, n // 5))
    try_add(pool, project_capped(cosv2, n), n)
    for sc in [0.04, 0.08, 0.12, 0.20, 0.30]:
        try_add(pool, project_capped(fb + rng.normal(0, sc, n), n), n)
    tries = 0
    while len(pool) < P and tries < 20 * P:
        v = rng.uniform(0, 1, n) if tries % 3 == 0 else fb + rng.normal(0, 0.15, n)
        try_add(pool, project_capped(v, n), n)
        tries += 1
    while len(pool) < P:
        pool.append(pool[len(pool) % max(1, len(pool))].copy())
    X = np.stack(pool[:P])
    sc = np.array([exact_score(X[i], n) for i in range(P)])
    return X, sc

def de_optimize(X, sc, n, rng, n_gen, patience, n_restart, P):
    best_v = X[np.argmin(sc)].copy()
    best_s = float(sc.min())
    stag = 0
    nrest = 0
    for gen in range(n_gen):
        Xn = X.copy(); sn = sc.copy(); gen_imp = False
        for i in range(P):
            others = np.array([j for j in range(P) if j != i])
            perm = rng.permutation(min(3, len(others)))
            r1, r2, r3 = others[perm[0]], others[perm[1]], others[perm[2]]
            F = rng.uniform(0.45, 0.90)
            CR = rng.uniform(0.75, 0.95)
            use_best = rng.random() < 0.15
            if use_best:
                bi = int(np.argmin(sn))
                oth2 = np.array([j for j in range(P) if j != i and j != bi])
                p2 = rng.permutation(min(2, len(oth2)))
                donor = best_v + F * (X[oth2[p2[0]]] - X[oth2[p2[1]]])
            else:
                donor = X[r1] + F * (X[r2] - X[r3])
            dproj = project_capped(donor, n)
            if dproj is None:
                continue
            mask = rng.random(n) < CR
            mask[rng.integers(n)] = True
            z = np.where(mask, dproj, X[i])
            y = project_capped(z, n)
            if y is None or np.max(np.abs(y - X[i])) < 1e-12:
                continue
            sy = exact_score(y, n)
            if sy < sn[i] - 1e-12:
                Xn[i] = y; sn[i] = sy; gen_imp = True
            if sy < best_s - 1e-12:
                best_s = sy; best_v = y.copy()
        X, sc = Xn, sn
        if gen_imp:
            stag = 0
        else:
            stag += 1
            if stag >= patience:
                if nrest >= n_restart:
                    break
                nrest += 1; stag = 0
                X2 = np.empty_like(X); sc2 = np.empty(P)
                X2[0] = best_v; sc2[0] = best_s; cnt = 1
                order = np.argsort(sc)
                for j in order:
                    if cnt >= max(2, P // 3):
                        break
                    dup = False
                    for k in range(cnt):
                        if np.max(np.abs(X[j] - X2[k])) < 1e-8:
                            dup = True; break
                    if not dup:
                        X2[cnt] = X[j]; sc2[cnt] = sc[j]; cnt += 1
                fs = [0.05, 0.10, 0.20]
                while cnt < P:
                    if cnt % 3 == 2:
                        v = rng.uniform(0, 1, n)
                    else:
                        v = best_v + rng.normal(0, fs[cnt % 3], n)
                    pv = project_capped(v, n)
                    if pv is None:
                        v = rng.uniform(0, 1, n)
                        pv = project_capped(v, n)
                    if pv is not None:
                        X2[cnt] = pv; sc2[cnt] = exact_score(pv, n)
                        if sc2[cnt] < best_s - 1e-12:
                            best_s = sc2[cnt]; best_v = pv.copy()
                    cnt += 1
                X, sc = X2, sc2
    return best_v, best_s

def local_polish(x, n, rng, steps):
    best = x.copy(); bs = exact_score(x, n)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.04, -0.04, 0.08, -0.08]
    for _ in range(steps):
        i = int(rng.integers(n))
        improved = False
        for d in deltas:
            y = x.copy(); y[i] = np.clip(y[i] + d, 0.0, 1.0)
            y = project_capped(y, n)
            if y is None:
                continue
            sy = exact_score(y, n)
            if sy < bs - 1e-13:
                x = y.copy(); bs = sy; best = y.copy(); improved = True
                break
        if improved:
            continue
        j = int(rng.integers(n))
        if j == i:
            j = (j + 1) % n
        for d in [0.02, -0.02, 0.05, -0.05]:
            y = x.copy()
            a = min(d, 1.0 - y[i]) if d > 0 else max(d, -y[i])
            b = min(-d, 1.0 - y[j]) if d < 0 else max(-d, -y[j])
            if d > 0:
                y[i] -= min(-b, y[i]); y[j] += min(-b, 1.0 - y[j])
            else:
                y[i] += min(-b, 1.0 - y[i]); y[j] -= min(-b, y[j])
            y = project_capped(y, n)
            if y is None:
                continue
            sy = exact_score(y, n)
            if sy < bs - 1e-13:
                x = y.copy(); bs = sy; best = y.copy(); improved = True
                break
    return best, bs

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 0))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if n is None:
        n = 47
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.clip(np.asarray(fb, dtype=np.float64), 0.0, 1.0)
    fb_s = float(exact_score(fb, n))
    if n == 1:
        return fb, fb_s, int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P, ngen, pat, nr, ps = 48, 250, 30, 2, 400
    elif n <= 128:
        P, ngen, pat, nr, ps = 32, 160, 25, 2, 250
    elif n <= 256:
        P, ngen, pat, nr, ps = 24, 100, 20, 1, 150
    else:
        P, ngen, pat, nr, ps = 16, 60, 15, 1, 80
    X, sc = build_pop(fb, n, rng, P)
    best_v, best_s = de_optimize(X, sc, n, rng, ngen, pat, nr, P)
    if best_s > fb_s + 1e-12:
        best_v, best_s = fb.copy(), fb_s
    best_v, best_s = local_polish(best_v, n, rng, ps)
    if best_s > fb_s + 1e-12:
        best_v, best_s = fb.copy(), fb_s
    best_v = project_capped(best_v, n)
    if best_v is None:
        best_v = np.clip(fb, 0.0, 1.0)
    bs = float(exact_score(best_v, n))
    if not np.isfinite(bs):
        best_v = np.full(n, 0.5)
        bs = float(exact_score(best_v, n))
    return best_v, bs, int(n)
