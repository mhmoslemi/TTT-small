import numpy as np
from scipy.optimize import minimize
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,):
        return None
    z = np.array(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i])
            h[i] += c
            r -= c
        else:
            c = min(-r, h[i])
            h[i] -= c
            r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(float(np.sum(h)) - target) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def soft_obj(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    qm = float(np.max(q))
    e = np.exp((q - qm) / tau)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e6, np.zeros(n)
    w = e / se
    val = qm + tau * math.log(se)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = float(w[k])
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        p0 = d if d > 0 else 0
        r0 = -d if d < 0 else 0
        m = n - abs(d)
        grad[p0:p0 + m] += wk * dx * (1.0 - h[r0:r0 + m])
        grad[r0:r0 + m] -= wk * dx * h[p0:p0 + m]
    return val, grad

def mass_eq(h, n):
    return float(np.sum(h) - n / 2.0)

def mass_jac(h, n):
    return np.ones(n)

def polish(h, n, n_iter=100):
    current = h.copy()
    score = exact_score(current, n)
    dx = 2.0 / n
    eta = 2.5 / (n * dx)
    stalled = 0
    for _ in range(n_iter):
        if eta < 1e-14 or stalled > 25:
            break
        q = np.correlate(current, 1.0 - current, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0 = d if d > 0 else 0
        r0 = -d if d < 0 else 0
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - current[r0:r0 + m])
        g[r0:r0 + m] -= dx * current[p0:p0 + m]
        z = project_capped(current - eta * g, n)
        if z is None:
            eta *= 0.5
            stalled += 1
            continue
        best_z, best_s = None, score
        for k in range(5):
            alpha = 1.0 / (2.0 ** k)
            if alpha >= 1.0 - 1e-15:
                trial = z
            else:
                trial = project_capped(current + alpha * (z - current), n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-13:
                best_s = ts
                best_z = trial.copy()
        if best_z is None:
            eta *= 0.5
            stalled += 1
        else:
            current = best_z
            score = best_s
            stalled = 0
            eta = min(eta * 1.12, 4.0 / (n * dx))
    return current, score

    if pr is None:
        return
    for s in pool:
        if float(np.max(np.abs(pr - s))) < 1e-7:
            return
    pool.append(pr.copy())
