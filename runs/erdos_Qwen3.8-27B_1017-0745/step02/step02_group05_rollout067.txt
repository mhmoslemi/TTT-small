    best = x.copy()

        else:
            i = int(np.argmax(h))
            h[i] = max(h[i] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def add_candidate(pool, c, n):
    p = project_capped(c, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p)

def build_pop(base, n, rng, P):
    pool = []
    add_candidate(pool, base, n)
    add_candidate(pool, base[::-1].copy(), n)
    add_candidate(pool, 1.0 - base, n)
    add_candidate(pool, 1.0 - base[::-1].copy(), n)
    r4 = max(1, n // 4)
    add_candidate(pool, np.roll(base, r4), n)
    add_candidate(pool, np.roll(base, -r4), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_candidate(pool, blk, n)
    add_candidate(pool, blk[::-1].copy(), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_candidate(pool, alt, n)
    for k in (1, 2, 3):
        ph = rng.uniform(0, 2 * np.pi)
        add_candidate(pool, 0.5 + 0.35 * np.sin(2 * np.pi * k * np.arange(n) / n + ph), n)
    for sd in (0.03, 0.08, 0.18):
        add_candidate(pool, base + rng.normal(0, sd, n), n)
    add_candidate(pool, rng.uniform(0, 1, n), n)
    add_candidate(pool, np.full(n, 0.5), n)
    t = 0
    while len(pool) < P and t < 200:
        t += 1
        add_candidate(pool, base + rng.normal(0, 0.15, n), n)
        add_candidate(pool, rng.uniform(0, 1, n), n)
    while len(pool) < P:
        pool.append(pool[0].copy())
    return pool[:P]

def de_search(pop, n, rng, G, stall_lim):
    P = len(pop)
    scores = [exact_score(p, n) for p in pop]
    gi = int(np.argmin(scores))
    bh, bs = pop[gi].copy(), scores[gi]
    stall = 0
    for gen in range(G):
        q = gen / max(1, G - 1)
        Tg = 2e-3 * (1e-5 / 2e-3) ** q
        frozen = [p.copy() for p in pop]
        fs = list(scores)
        gbest = bh.copy()
        npop = [None] * P
        for i in range(P):
            av = [j for j in range(P) if j != i]
            sel = rng.choice(av, size=3, replace=False)
            a, b, c = int(sel[0]), int(sel[1]), int(sel[2])
            F = (0.5 + 0.4 * rng.random()) * (1 - 0.25 * q)
            if rng.random() < 0.25:
                v = frozen[i] + F * (gbest - frozen[i]) + F * (frozen[b] - frozen[c])
            else:
                v = frozen[a] + F * (frozen[b] - frozen[c])
            mask = rng.random(n) < 0.85
            mask[int(rng.integers(n))] = True
            tr_raw = np.where(mask, v, frozen[i])
            tr = project_capped(tr_raw, n)
            if tr is None:
                npop[i] = frozen[i]
                continue
            ts = exact_score(tr, n)
            d = ts - fs[i]
            if d < -1e-12:
                acc = True
            elif abs(d) <= 1e-12:
                acc = rng.random() < 0.1
            elif Tg > 1e-15:
                acc = rng.random() < math.exp(-d / Tg)
            else:
                acc = False
            npop[i] = tr if acc else frozen[i]
            if ts < bs - 1e-12:
                bh, bs = tr.copy(), ts
        npop[0] = bh.copy()
        stall = 0 if bs < min(fs) - 1e-12 else stall + 1
        pop = npop
        scores = [exact_score(p, n) for p in pop]
        if stall >= stall_lim and gen < G - 1:
            keep = max(2, P // 3)
            for j in range(keep, P):
                if rng.random() < 0.5:
                    sg = 0.025 + 0.075 * (1 - q)
                    rep = bh + sg * rng.normal(0, 1, n)
                else:
                    rep = rng.uniform(0, 1, n)
                rp = project_capped(rep, n)
                if rp is not None:
                    rs = exact_score(rp, n)
                    pop[j] = rp
                    if rs < bs - 1e-12:
                        bh, bs = rp.copy(), rs
                else:
                    pop[j] = bh.copy()
            pop[0] = bh.copy()
            scores = [exact_score(p, n) for p in pop]
            stall = 0
    return bh, bs

def pair_anneal(h, n, rng, L):
    x, xs = h.copy(), exact_score(h, n)
    bh, bs = x.copy(), xs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        rho = 1 - 0.98 * q
        y = None
        for _ in range(20):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            c1 = min(1.0 - x[i], x[j])
            if c1 > 1e-10:
                y = x.copy()
                y[i] += c1 * rho * (0.05 + 0.95 * rng.random())
                y[j] -= c1 * rho * (0.05 + 0.95 * rng.random())
                break
            c2 = min(1.0 - x[j], x[i])
            if c2 > 1e-10:
                y = x.copy()
                y[j] += c2 * rho * (0.05 + 0.95 * rng.random())
                y[i] -= c2 * rho * (0.05 + 0.95 * rng.random())
                break
        if y is None:
            continue
        y = project_capped(y, n)
        if y is None:
            continue
        ys = exact_score(y, n)
        d = ys - xs
        if d < -1e-12 or (T > 1e-15 and rng.random() < math.exp(-max(0, d) / T)):
            x, xs = y, ys
            if xs < bs - 1e-12:
                bh, bs = y.copy(), xs
    return bh, bs

def pair_polish(h, n, rng, sweeps, fl):
    x, xs = h.copy(), exact_score(h, n)
    for _ in range(sweeps):
        imp = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            nc = min(300, n * (n - 1) // 2)
            ai = rng.integers(0, n, nc)
            bj = rng.integers(0, n, nc)
            pairs = [(int(a), int(b)) for a, b in zip(ai, bj) if a != b]
        for i, j in pairs:
            for fr in fl:
                c1 = min(1.0 - x[i], x[j])
                if c1 > 1e-10:
                    y2 = x.copy()
                    y2[i] += fr * c1
                    y2[j] -= fr * c1
                    yp = project_capped(y2, n)
                    if yp is not None:
                        yp_s = exact_score(yp, n)
                        if yp_s < xs - 1e-12:
                            x, xs = yp, yp_s
                            imp = True
                c2 = min(1.0 - x[j], x[i])
                if c2 > 1e-10:
                    y2 = x.copy()
                    y2[j] += fr * c2
                    y2[i] -= fr * c2
                    yp = project_capped(y2, n)
                    if yp is not None:
                        yp_s = exact_score(yp, n)
                        if yp_s < xs - 1e-12:
                            x, xs = yp, yp_s
                            imp = True
        if not imp:
            break
    return x, xs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    safe_h, safe_s = fb.copy(), exact_score(fb, n)
    if n <= 64:
        P, G, L, sl, sw = 12, 120, 1600, 20, 2
    elif n <= 256:
        P, G, L, sl, sw = 8, 70, 500, 14, 2
    else:
        P, G, L, sl, sw = 6, 25, 80, 7, 1
    fl = (1.0 / 64, 1.0 / 16, 0.25, 0.5, 1.0)
    pop = build_pop(fb, n, rng, P)
    bh, bs = de_search(pop, n, rng, G, sl)
    if bs > safe_s + 1e-12:
        bh, bs = safe_h.copy(), safe_s
    ah, as_ = pair_anneal(bh, n, rng, L)
    if as_ < bs - 1e-12:
        bh, bs = ah, as_
    ph, ps = pair_polish(bh, n, rng, sw, fl)
    if ps < bs - 1e-12:
        bh, bs = ph, ps
    final = project_capped(bh, n)
    if final is None or not is_feasible(final, n):
        final = safe_h
    fs = exact_score(final, n)
    if fs > safe_s + 1e-12:
        final, fs = safe_h, safe_s
    return final.astype(np.float64), float(fs), int(n)
