        Phi[:, idx] = np.sqrt(2.0 / n) * np.sin(2.0 * np.pi * k * i / n)
        idx += 1
    if n % 2 == 0 and idx < m:
        Phi[:, idx] = np.cos(np.pi * i) / np.sqrt(n)
        idx += 1
    return Phi[:, :idx]

def get_stages(n):
    F = (n - 1) // 2
    cuts = []
    for c in (4, 8, 16, 32, F):
        if 1 <= c <= F and c not in cuts:
            cuts.append(c)
    if not cuts:
        cuts = [F]
    return cuts

def active_cols(Phi, n, stage_k, all_active):
    F = (n - 1) // 2
    cols = []
    if all_active:
        return list(range(Phi.shape[1]))
    for k in range(1, stage_k + 1):
        ci, si = 2 * (k - 1), 2 * k - 1
        if ci < Phi.shape[1]:
            cols.append(ci)
        if si < Phi.shape[1]:
            cols.append(si)
    return cols

def scan_coeff(z, u_j, phi_j, h_cur, s_cur, n, R, n_grid, n_ref, n_rnd):
    grid = np.linspace(-R, R, n_grid)
    scores = np.full(n_grid, float('inf'))
    for gi in range(n_grid):
        zt = z + (grid[gi] - u_j) * phi_j
        ht = project_capped(zt, n)
        if ht is not None:
            scores[gi] = exact_score(ht, n)
    minima = []
    for gi in range(n_grid):
        if not np.isfinite(scores[gi]):
            continue
        lok = (gi == 0) or (scores[gi] <= scores[gi - 1])
        rok = (gi == n_grid - 1) or (scores[gi] <= scores[gi + 1])
        if lok and rok:
            minima.append((scores[gi], gi))
    if not minima:
        gi = int(np.argmin(scores))
        minima = [(scores[gi], gi)]
    minima.sort()
    minima = minima[:n_ref]
    best_s, best_h, best_z = s_cur, h_cur, z
    for sc, gi in minima:
        lo = max(grid[gi - 1], -R) if gi > 0 else -R
        hi = min(grid[gi + 1], R) if gi < n_grid - 1 else R
        if hi <= lo:
            hi = lo + 1e-6
        bm, bms = None, float('inf')
        for _ in range(n_rnd):
            mids = np.linspace(lo, hi, 5)
            bm, bms = None, float('inf')
            for mid in mids:
                zt = z + (mid - u_j) * phi_j
                ht = project_capped(zt, n)
                if ht is None:
                    continue
                st = exact_score(ht, n)
                if st < bms:
                    bms, bm = st, mid
            if bm is None:
                break
            mi = int(np.argmin(np.abs(mids - bm)))
            if mi == 0:
                hi = mids[1]
            elif mi == 4:
                lo = mids[3]
            else:
                lo, hi = mids[mi - 1], mids[mi + 1]
            if hi - lo < 1e-12 * max(1.0, R):
                break
        if bm is not None and bms < best_s - 1e-12:
            best_s, bm_h = bms, project_capped(z + (bm - u_j) * phi_j, n)
            if bm_h is not None:
                best_h, best_z, best_s = bm_h, z + (bm - u_j) * phi_j, bms
    return best_z, best_h, best_s

def optimize_start(h0, Phi, n, rng, stages, sweeps, ng, nref, nrnd):
    h = project_capped(h0, n)
    if h is None:
        return None, float('inf')
    s = exact_score(h, n)
    u = Phi.T @ (h - 0.5)
    z = 0.5 + Phi @ u
    zp = project_capped(z, n)
    if zp is not None:
        sp = exact_score(zp, n)
        if sp < s:
            h, s, z = zp, sp, zp
    R = np.sqrt(n) / 2.0
    nstg = len(stages)
    for si, stage_k in enumerate(stages):
        cols = active_cols(Phi, n, stage_k, si == nstg - 1)
        for _sw in range(sweeps):
            order = list(cols)
            rng.shuffle(order)
            improved = False
            for j in order:
                u_j = u[j]
                phi_j = Phi[:, j]
                nz, nh, ns = scan_coeff(z, u_j, phi_j, h, s, n, R, ng, nref, nrnd)
                if ns < s - 1e-12:
                    z, h, s = nz, nh, ns
                    u[j] = float((z - 0.5) @ phi_j)
                    improved = True
            if not improved:
                break
        zp = project_capped(z, n)
        if zp is not None:
            sp = exact_score(zp, n)
            if sp < s - 1e-12:
                h, s, z = zp, sp, zp
    return h, s

def local_polish(h, n, rng, ntri, sig0):
    h = h.copy()
    s = exact_score(h, n)
    sigma = sig0
    for _ in range(ntri):
        sigma *= 0.965
        if sigma < 1e-8:
            break
        z = h + rng.normal(0, sigma, n)
        zp = project_capped(z, n)
        if zp is None:
            continue
        sz = exact_score(zp, n)
        if sz < s - 1e-13:
            h, s = zp, sz
    return h, s

def build_starts(parent, n, rng, cap):
    base = parent if parent is not None else np.full(n, 0.5)
    cands = [base.copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(1.0 - base)
    cands.append(base[::-1].copy())
    cands.append(base + rng.normal(0, 0.08, n))
    cands.append(base + rng.normal(0, 0.18, n))
    i = np.arange(n, dtype=np.float64)
    for k in (1, 2):
        cands.append(0.5 + 0.3 * np.sin(2.0 * np.pi * k * i / n + rng.uniform(0, 6.28)))
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(rng.uniform(0, 1, n))
    pool = []
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        if any(np.max(np.abs(p - q)) < 1e-8 for q in pool):
            continue
        pool.append(p)
        if len(pool) >= cap:
            break
    return pool

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    if n <= 64:
        nst, sw, ng, nref, nrnd = 6, 2, 9, 2, 3
    elif n <= 256:
        nst, sw, ng, nref, nrnd = 3, 2, 7, 2, 2
    else:
        nst, sw, ng, nref, nrnd = 2, 1, 5, 1, 2
    Phi = build_phi(n)
    stages = get_stages(n)
    starts = build_starts(parent, n, rng, nst)
    if not any(np.max(np.abs(fb - s)) < 1e-8 for s in starts):
        starts.insert(0, fb.copy())
        starts = starts[:nst]
    gb, gs = fb.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, Phi, n, rng, stages, sw, ng, nref, nrnd)
            if xo is not None and so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        kb, ks = local_polish(gb, n, rng, 250, 0.04)
        if ks < gs - 1e-12:
            gs, gb = ks, kb
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fb.copy(), fb_score
    return final.astype(np.float64), float(fs), int(n)
