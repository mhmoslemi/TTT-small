    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        i_start = max(0, -k)
        i_end = min(n - 1, n - 1 - k)
        if i_end < i_start:
            continue
        i = np.arange(i_start, i_end + 1)
        j = i + k
        np.add.at(J, (k + n - 1, j), dx * (1.0 - h[i]))
        np.add.at(J, (k + n - 1, i), -dx * h[j])
    return J

def obj_fn(z, n):
    return z[-1]

def obj_jac(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def ineq_fn(z, n):
    return z[-1] - np.correlate(z[:n], 1.0 - z[:n], mode='full') * (2.0 / n)

def ineq_jac(z, n):
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -lag_jacobian(z[:n], n)
    out[:, n] = 1.0
    return out

def eq_fn(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g

def solve_slsqp(h0, n, maxiter):
    h0 = project_capped(np.clip(h0, 0, 1), n)
    if h0 is None:
        return None
    score0 = exact_score(h0, n)
    t0 = max(0.0, min(0.5, score0 + 1e-8))
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac, 'args': (n,)},
    ]
    try:
        res = minimize(obj_fn, z0, method='SLSQP', jac=obj_jac, args=(n,),
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            return project_capped(res.x[:n], n)
    except Exception:
        pass
    return None

def build_starts(base, n, rng):
    pool = [np.full(n, 0.5)]
    for v in [base, base[::-1].copy(), 1.0 - base, 1.0 - base[::-1].copy()]:
        p = project_capped(v, n)
        if p is not None:
            pool.append(p)
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool.append(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    p = project_capped(alt, n)
    if p is not None:
        pool.append(p)
    for m in range(1, 4):
        for ph in [0.0, 1.5708]:
            cos = 0.5 + 0.4 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
            p = project_capped(cos, n)
            if p is not None:
                pool.append(p)
    for sd in [0.04, 0.10, 0.20, 0.35]:
        p = project_capped(base + rng.normal(0, sd, n), n)
        if p is not None:
            pool.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        pool.append(p)
    dedup = []
    for p in pool:
        if p is None:
            continue
        dup = False
        for s in dedup:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True; break
        if not dup:
            dedup.append(p)
    return dedup

def try_update(hc, best_h, best_s, n):
    if hc is None:
        return best_h, best_s
    hc = project_capped(hc, n)
    if hc is None:
        return best_h, best_s
    hs = exact_score(hc, n)
    if np.isfinite(hs) and hs < best_s - 1e-12:
        return hc.copy(), hs
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = project_capped(np.clip(p, 0, 1), n)
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except Exception:
            n = 47
    if n == 1:
        return np.array([0.5]), 0.0, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    h_uni = np.full(n, 0.5)
    fallback = h_uni
    fb_score = 0.5
    if parent is not None:
        fs = exact_score(parent, n)
        if np.isfinite(fs) and fs < fb_score:
            fallback = parent; fb_score = fs
    if n <= 64:
        n_starts, n_refine, maxiter = 8, 4, 250
    elif n <= 128:
        n_starts, n_refine, maxiter = 5, 2, 150
    else:
        n_starts, n_refine, maxiter = 3, 1, 80
    pool = build_starts(fallback, n, rng)
    scored = []
    for p in pool:
        s = exact_score(p, n)
        if np.isfinite(s) and s <= 0.5 + 1e-12:
            scored.append((s, p))
    order = np.argsort(np.asarray([s for s, _ in scored]))
    starts = [scored[i][1] for i in order if i < len(scored)][:n_starts]
    if not any(np.max(np.abs(p - fallback)) < 1e-6 for p in starts):
        starts.append(fallback.copy())
    best_h = fallback.copy()
    best_s = fb_score
    alphas = [0.25, 0.50, 0.75]
    for st in starts:
        sol = solve_slsqp(st, n, maxiter)
        best_h, best_s = try_update(sol, best_h, best_s, n)
        if sol is not None:
            for a in alphas:
                blend = a * sol + (1.0 - a) * best_h
                best_h, best_s = try_update(blend, best_h, best_s, n)
    sigma = 0.04
    for _ in range(n_refine):
        start_ref = None
        for _ in range(3):
            pert = best_h + rng.normal(0, sigma, n)
            pp = project_capped(pert, n)
            if pp is not None and np.max(np.abs(pp - best_h)) > 1e-6:
                ps = exact_score(pp, n)
                if np.isfinite(ps) and ps <= 0.5 + 1e-12:
                    start_ref = pp; break
        if start_ref is None:
            sigma = min(0.20, 1.5 * sigma); continue
        old_best = best_s
        sol = solve_slsqp(start_ref, n, maxiter)
        best_h, best_s = try_update(sol, best_h, best_s, n)
        if sol is not None:
            for a in alphas:
                blend = a * sol + (1.0 - a) * best_h
                best_h, best_s = try_update(blend, best_h, best_s, n)
        if best_s < old_best - 1e-12:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback; fs = fb_score
    return final, float(fs), int(n)
