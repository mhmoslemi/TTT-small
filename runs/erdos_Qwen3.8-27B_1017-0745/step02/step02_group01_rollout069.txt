        r = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p:p + m] += dx * (1.0 - cur2[r:r + m])
        g[r:r + m] -= dx * cur2[p:p + m]
        z = project_capped(cur2 - eta * g, n)
        if z is None:
            eta *= 0.5
            continue
        improved = False
        for al in (1.0, 0.5, 0.25, 0.125):
            if al >= 0.999:
                zz = z
            else:
                zz = project_capped(cur2 + al * (z - cur2), n)
            if zz is not None and is_feasible(zz, n):
                sz = exact_score(zz, n)
                if sz < sc2 - 1e-12:
                    sc2 = sz
                    best_s = sz
                    best_h = zz.copy()
                    cur2 = zz.copy()
                    improved = True
                    break
        if improved:
            eta = min(eta * 1.1, 2.0 / dx)
        else:
            eta *= 0.7
        if eta < 1e-10 / dx:
            break
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
    fs = exact_score(final, n)
    ff = exact_score(fb, n)
    if fs > ff - 1e-12:
        final, fs = fb, ff
    if not is_feasible(final, n):
        final = fb
        fs = exact_score(fb, n)
    return final, float(fs), int(n)
