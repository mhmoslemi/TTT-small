        n_starts, maxiter, n_hops = 4, 120, 3
    else:
        n_starts, maxiter, n_hops = 2, 60, 0

    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))

    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()

    for st in starts:
        try:
            y = solve_epigraph(st, n, maxiter, 1e-12)
            if y is not None:
                y = project_capped(y, n)
                if y is not None and is_feasible(y, n):
                    sy = float(exact_score(y, n))
                    if sy < gs - 1e-12:
                        gs, gb = sy, y.copy()
        except Exception:
            pass

    if n_hops > 0:
        current = gb.copy()
        cs = gs
        sigmas = [0.04, 0.08, 0.15, 0.25]
        for hop in range(n_hops):
            if hop % 5 == 4:
                proposal = project_capped(rng.uniform(0, 1, n), n)
            else:
                sigma = sigmas[hop % 4]
                proposal = project_capped(current + sigma * rng.standard_normal(n), n)
            if proposal is None:
                continue
            try:
                y = solve_epigraph(proposal, n, maxiter, 1e-12)
                if y is not None:
                    y = project_capped(y, n)
                    if y is not None and is_feasible(y, n):
                        sy = float(exact_score(y, n))
                        if sy < gs - 1e-12:
                            gs, gb = sy, y.copy()
                        T = max(1e-5, 2e-3 * (0.75 ** hop))
                        if sy <= cs or (np.log(rng.random() + 1e-30) < -(sy - cs) / T):
                            current, cs = y.copy(), sy
            except Exception:
                pass

    # Polish: re-solve from best with tighter ftol
    try:
        y = solve_epigraph(gb, n, 400, 1e-14)
        if y is not None:
            y = project_capped(y, n)
            if y is not None and is_feasible(y, n):
                sy = float(exact_score(y, n))
                if sy < gs - 1e-14:
                    gs, gb = sy, y.copy()
    except Exception:
        pass

    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12:
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
