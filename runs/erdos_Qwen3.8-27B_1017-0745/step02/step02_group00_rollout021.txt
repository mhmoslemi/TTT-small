            break
        p = y - x
        r = lag_dirprod(x, p, n)
        gradD = a + r
        P_val = float(np.max(a + r)) + 0.5 * mu * float(np.dot(p, p))
        D_val = float(np.dot(z, a)) + float(np.dot(gbar, p)) + 0.5 * mu * float(np.dot(p, p))
        if np.isfinite(P_val) and P_val < best_P:
            best_P = P_val
            best_candidate = y.copy()
        gap = max(0.0, P_val - D_val)
        w_new = project_simplex(z + gradD / L, m)
        if w_new is None:
            w_new = w.copy()
            break
        if float(np.max(np.abs(w_new - w))) <= 1e-10:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        if gap <= 1e-8 * max(1.0, abs(P_val)):
            gbar2 = weighted_grad(x, w_new, n)
            y2 = project_capped(x - gbar2 / mu, n)
            if y2 is not None:
                p2 = y2 - x
                r2 = lag_dirprod(x, p2, n)
                P2 = float(np.max(a + r2)) + 0.5 * mu * float(np.dot(p2, p2))
                if np.isfinite(P2) and P2 < best_P:
                    best_P = P2
                    best_candidate = y2.copy()
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        t = t_new
        w = w_new
    return best_candidate

def mm_optimize(x, n, best_score, max_outer, dual_K, mu_retries):
    dx = 2.0 / n
    mu_min = 2.0 * dx
    mu = mu_min
    w_warm = None
    cur = x.copy()
    cur_score = exact_score(cur, n)
    for _ in range(max_outer):
        if cur_score >= best_score:
            break
        a = lag_vals(cur, n)
        if not np.all(np.isfinite(a)):
            break
        accepted = False
        for rr in range(mu_retries):
            mu_t = mu_min * (2.0 ** rr)
            wi = w_warm if w_warm is not None and w_warm.shape == (2 * n - 1,) else None
            y = dual_solve(cur, a, mu_t, wi, n, dual_K)
            if y is None:
                continue
            if float(np.linalg.norm(y - cur)) <= 1e-10 * max(1.0, np.sqrt(n)):
                break
            best_z, best_s = cur.copy(), cur_score
            for j in range(9):
                theta = 1.0 / (2.0 ** j)
                zt = (1.0 - theta) * cur + theta * y
                zp = project_capped(zt, n)
                if zp is None:
                    continue
                st = exact_score(zp, n)
                if np.isfinite(st) and st < best_s - 1e-12:
                    best_z, best_s = zp.copy(), st
                    if st < best_score:
                        best_score = st
            if best_s < cur_score - 1e-12:
                cur = best_z
                cur_score = best_s
                accepted = True
                a = lag_vals(cur, n)
                if theta == 1.0:
                    mu = max(mu_min, 0.75 * mu_t)
                else:
                    mu = mu_t
                w_warm = None
            if accepted:
                break
        if not accepted:
            break
    return cur, cur_score, best_score

def pairwise_polish(h, n, sweeps):
    h = h.copy()
    deltas = np.array([0.005, 0.02, 0.05, 0.1, 0.2, 0.35])
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d <= lo_d + 1e-12:
                    continue
                cur_s = exact_score(h, n)
                best_d, best_s = 0.0, cur_s
                for dv in np.concatenate([deltas, -deltas, [lo_d, hi_d]]):
                    if abs(dv) < 1e-14:
                        continue
                    dv = max(lo_d, min(hi_d, dv))
                    h[i] += dv
                    h[j] -= dv
                    se = exact_score(h, n)
                    h[i] -= dv
                    h[j] += dv
                    if np.isfinite(se) and se < best_s - 1e-12:
                        best_s, best_d = se, dv
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        if not improved:
            break
    return h

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= iv.size <= 999:
            n = iv.size
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = np.clip(iv, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(int(kwargs.get('n_points', 47)), 999))
        except Exception:
            n = 47
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**32))
    uniform = np.full(n, 0.5)
    fb_score = exact_score(uniform, n)
    fallback = uniform.copy()
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None:
            ps = exact_score(pp, n)
            if np.isfinite(ps) and ps < fb_score:
                fallback, fb_score = pp, ps
    if n <= 64:
        mx_s, mx_o, du_K, mu_r = 5, 14, 120, 3
    elif n <= 128:
        mx_s, mx_o, du_K, mu_r = 4, 9, 70, 3
    elif n <= 256:
        mx_s, mx_o, du_K, mu_r = 3, 5, 35, 2
    else:
        mx_s, mx_o, du_K, mu_r = 2, 3, 16, 2
    try:
        mx_s = max(1, min(int(kwargs.get('starts', mx_s)), 10))
    except Exception:
        pass
    try:
        mx_o = max(0, min(int(kwargs.get('max_outer', mx_o)), 40))
    except Exception:
        pass
    try:
        du_K = max(1, min(int(kwargs.get('dual_iter', du_K)), 300))
    except Exception:
        pass
    idx = np.arange(n, dtype=np.float64)
    starts = []
    def add_start(v):
        p = project_capped(v, n)
        if p is not None and np.all(np.isfinite(p)):
            dup = False
            for d in starts:
                if float(np.max(np.abs(p - d))) < 1e-8:
                    dup = True
                    break
            if not dup:
                starts.append(p.copy())
    if parent is not None:
        for arr in [parent, parent[::-1], 1.0 - parent, 1.0 - parent[::-1]]:
            add_start(arr)
        for sc in [0.03, 0.06, 0.10, 0.15]:
            add_start(parent + rng.normal(0, sc, n))
    for freq in [1, 2, 3]:
        add_start(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n))
    for p2 in [3, 5]:
        blk = np.zeros(n)
        for k in range(0, n, p2):
            blk[k:k + 2] = 1.0
        if n % p2 >= 2:
            blk[-2:] = 1.0
        add_start(blk)
    add_start(0.5 * np.ones(n))
    for sc in [0.08, 0.15]:
        add_start(0.5 + rng.normal(0, sc, n))
    add_start(rng.uniform(0, 1, n))
    starts = starts[:mx_s]
    best, best_score = fallback.copy(), fb_score
    if n > 1 and mx_o > 0:
        for st in starts:
            ss = exact_score(st, n)
            if np.isfinite(ss) and ss < best_score - 1e-12:
                best, best_score = st.copy(), ss
            try:
                mo = mx_o if ss < best_score + 1e-12 else max(2, mx_o // 2)
                res_h, res_s, best_score = mm_optimize(st.copy(), n, best_score, mo, du_K, mu_r)
                if np.isfinite(res_s) and res_s < best_score:
                    best, best_score = res_h.copy(), res_s
            except Exception:
                pass
    n_sw = 2 if n <= 64 else (1 if n <= 128 else 0)
    if n_sw > 0:
        try:
            pol = pairwise_polish(best, n, n_sw)
            pol = project_capped(pol, n)
            if pol is not None:
                ps = exact_score(pol, n)
                if np.isfinite(ps) and ps < best_score - 1e-12:
                    best, best_score = pol, ps
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-10:
        final = best.copy()
        fs = best_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    final = np.clip(final, 0.0, 1.0)
    fs2 = exact_score(final, n)
    if np.isfinite(fs2):
        fs = fs2
    return final, float(fs), int(n)
