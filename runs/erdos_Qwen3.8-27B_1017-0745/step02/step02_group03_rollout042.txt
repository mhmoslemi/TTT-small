def hit_and_run(h, n, rng):
    for _ in range(5):
        d = rng.standard_normal(n)
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            continue
        d /= norm
        nonzero = np.abs(d) > 1e-14
        if not np.any(nonzero):
            continue
        lo = 0.0
        hi = 0.0
        valid = True
        for i in range(n):
            if abs(d[i]) < 1e-14:
                continue
            r0 = -h[i] / d[i]
            r1 = (1.0 - h[i]) / d[i]
            rlo = min(r0, r1)
            rhi = max(r0, r1)
            if not np.isfinite(rlo) or not np.isfinite(rhi):
                valid = False
                break
            lo = max(lo, rlo)
            hi = min(hi, rhi)
            if hi - lo < 0:
                valid = False
                break
        if not valid or hi - lo < 1e-14:
            continue
        t = rng.uniform(lo, hi)
        cand = h + t * d
        cand = np.clip(cand, 0.0, 1.0)
        if abs(np.sum(cand) - np.sum(h)) > 1e-8 * n:
            continue
        if np.all(np.isfinite(cand)) and np.all(cand >= -1e-10) and np.all(cand <= 1.0 + 1e-10):
            return cand
    return None

def pair_proposal(h, n, rng, local_scale, local_prob):
    p = int(rng.integers(0, n))
    q = int(rng.integers(0, n - 1))
    if q >= p:
        q += 1
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo < 1e-14:
        return None
    if rng.random() < local_prob:
        t = rng.uniform(-local_scale, local_scale)
        if t < lo or t > hi:
            return None
    else:
        t = rng.uniform(lo, hi)
    cand = h.copy()
    cand[p] = h[p] + t
    cand[q] = h[q] - t
    if np.all(np.isfinite(cand)) and np.all(cand >= -1e-10) and np.all(cand <= 1.0 + 1e-10):
        return cand
    return None
