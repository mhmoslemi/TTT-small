            cs = exact_score(cand, n)

                cur, cur_s = cand, cs
                if cur_s < gs - 1e-12:
                    g, gs = cur.copy(), cur_s
    return g, gs

def scan_pair(h, n, i, j, budget):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return h, exact_score(h, n)
    pts = [(0.0, exact_score(h, n))]
    for r in range(1, 7):
        if budget[0] >= budget[1]:
            break
        d = lo + r * (hi - lo) / 6.0
        c = h.copy()
        c[i] += d
        c[j] -= d
        c = project_capped(c, n)
        if c is None:
            continue
        budget[0] += 1
        pts.append((d, exact_score(c, n)))
    bd, bs = _min_score(pts)
    srt = sorted(p[0] for p in pts)
    pos = srt.index(bd)
    ilo = srt[max(pos - 1, 0)]
    ihi = srt[min(pos + 1, len(srt) - 1)]
    for _ in range(8):
        if budget[0] >= budget[1] or ihi - ilo < 1e-14:
            break
        nv = []
        for fr in (0.25, 0.5, 0.75):
            if budget[0] >= budget[1]:
                break
            d = ilo + fr * (ihi - ilo)
            if any(abs(d - p[0]) < 1e-15 for p in pts):
                continue
            c = h.copy()
            c[i] += d
            c[j] -= d
            c = project_capped(c, n)
            if c is None:
                continue
            budget[0] += 1
            s = exact_score(c, n)
            nv.append((d, s))
            pts.append((d, s))
        if not nv:
            break
        bd, bs = _min_score(nv)
        srt = sorted(p[0] for p in pts)
        pos = srt.index(bd)
        ilo = srt[max(pos - 1, 0)]
        ihi = srt[min(pos + 1, len(srt) - 1)]
    if abs(bd) < 1e-15:
        return h, bs
    c = h.copy()
    c[i] += bd
    c[j] -= bd
    c = project_capped(c, n)
    if c is None:
        return h, bs
    return c, bs
