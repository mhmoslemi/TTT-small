    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-rho, -h[i])
        hi = min(rho, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=-q, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'maxiter': 5000,
                               'primal_feasibility_tolerance': 1e-10,
                               'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if abs(np.sum(u)) > 1e-7:
        return None
    for i in range(n):
        lo = max(-rho, -h[i])
        hi = min(rho, 1.0 - h[i])
        if u[i] < lo - 1e-7 or u[i] > hi + 1e-7:
            return None
    return u

def refine_chain(start, n, max_iter, rho0, best_h, best_score):
    cur = project_capped(start, n)
    if cur is None or not is_feasible(cur, n):
        return best_h, best_score
    cs = exact_score(cur, n)
    if not np.isfinite(cs):
        return best_h, best_score
    if cs < best_score - 1e-12:
        best_score, best_h = cs, cur.copy()
    rho = rho0
    rho_min = 1e-5
    rho_max = 0.32
    no_improve = 0
    no_solver = 0
    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625]
    for _it in range(max_iter):
        if rho < rho_min or no_improve >= 6 or no_solver >= 3:
            break
        q, A = lag_model(cur, n)
        if q is None:
            rho = max(rho_min, 0.5 * rho)
            no_improve += 1
            no_solver += 1
            continue
        u = lp_step(cur, n, rho, q, A)
        if u is None:
            rho = max(rho_min, 0.5 * rho)
            no_improve += 1
            no_solver += 1
            continue
        no_solver = 0
        best_trial = None
        best_ts = float('inf')
        for a in alphas:
            z = project_capped(cur + a * u, n)
            if z is None or not is_feasible(z, n):
                continue
            ts = exact_score(z, n)
            if not np.isfinite(ts):
                continue
            if ts < best_ts:
                best_ts = ts
                best_trial = z.copy()
            if ts < best_score - 1e-12:
                best_score = ts
                best_h = z.copy()
        if best_trial is None or best_ts >= cs - 1e-12:
            rho = max(rho_min, 0.5 * rho)
            no_improve += 1
            continue
        delta = best_trial - cur
        actual = cs - best_ts
        model_vals = q + A[:, :n] @ delta
        model_max = float(np.max(model_vals))
        predicted = cs - model_max
        if predicted > 1e-12 and np.isfinite(predicted) and np.isfinite(actual):
            ratio = actual / predicted
        else:
            ratio = -1.0
        cur = best_trial.copy()
        cs = best_ts
        no_improve = 0
        if ratio < 0.1 or ratio == -1.0:
            rho = max(rho_min, 0.5 * rho)
        elif ratio > 0.75 and np.max(np.abs(delta)) >= 0.8 * rho:
            rho = min(1.5 * rho, rho_max)
    if cs < best_score - 1e-12:
        best_score, best_h = cs, cur.copy()
    return best_h, best_score

def add_start(pool, z, n, limit):
    if z is None or z.shape != (n,):
        return pool
    p = project_capped(z, n)
    if p is None:
        return pool
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return pool
    if len(pool) < limit:
        pool.append(p)
    return pool

def build_starts(fallback, n, rng):
    pool = [fallback.copy()]
    pool = add_start(pool, fallback[::-1], n, 8)
    pool = add_start(pool, 1.0 - fallback[::-1], n, 8)
    pool = add_start(pool, np.full(n, 0.5), n, 8)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    pool = add_start(pool, blk, n, 8)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    pool = add_start(pool, alt, n, 8)
    pool = add_start(pool, fallback + rng.normal(0, 0.05, n), n, 8)
    pool = add_start(pool, fallback + rng.normal(0, 0.15, n), n, 8)
    i = np.arange(n)
    cos_v = 0.5 + 0.3 * np.cos(2.0 * math.pi * i / n)
    pool = add_start(pool, cos_v, n, 8)
    shift = np.roll(fallback, n // 3)
    pool = add_start(pool, shift, n, 8)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    if n <= 64:
        n_chains, iters = 4, 30
    elif n <= 256:
        n_chains, iters = 3, 25
    elif n <= 999:
        n_chains, iters = 2, 20
    else:
        n_chains, iters = 0, 0
    starts = build_starts(fallback, n, rng)
    sortable = []
    for i in range(len(starts)):
        s = exact_score(starts[i], n)
        sortable.append((s, i))
    sortable.sort()
    selected = []
    for _, idx in sortable:
        if idx not in selected:
            selected.append(idx)
        if len(selected) >= n_chains:
            break
    if 0 not in selected and len(selected) > 0:
        selected[-1] = 0
    if 0 not in selected:
        selected.append(0)
    selected = selected[:max(n_chains, 1)]
    best_h = fallback.copy()
    best_score = exact_score(fallback, n)
    if not np.isfinite(best_score):
        best_score = float('inf')
    if n_chains > 0:
        rho0 = 0.08
        for idx in selected:
            try:
                best_h, best_score = refine_chain(starts[idx], n, iters, rho0, best_h, best_score)
            except Exception:
                pass
        try:
            best_h, best_score = refine_chain(best_h, n, max(5, iters // 2), 0.03, best_h, best_score)
        except Exception:
            pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, n)
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback.copy()
        fs = fb_score
    if not np.isfinite(fs) or not is_feasible(final, n):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
