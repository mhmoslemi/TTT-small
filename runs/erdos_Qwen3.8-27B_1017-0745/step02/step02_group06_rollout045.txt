import numpy as np
import math

try:
    import cvxpy as cp
    _HAS_CP = True
except Exception:
    _HAS_CP = False

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.size != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    if L.size != 2 * n - 1 or not np.all(np.isfinite(L)):
        return None, None
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        a = max(0, -k)
        b = min(n, n - k)
        if a >= b:
            continue
        i = np.arange(a, b)
        j = i + k
        G[row, j] += dx * (1.0 - h[i])
        G[row, i] -= dx * h[j]
    if not np.all(np.isfinite(G)):
        return None, None
    return L, G

def solve_mm_qp(x, L, G, n, solvers):
    if not _HAS_CP:
        return None
    dx = 2.0 / n
    rho = dx * math.cos(math.pi / (n + 1))
    if rho <= 0 or not math.isfinite(rho):
        return None
    y = cp.Variable(n)
    u = cp.Variable()
    Gx = G @ x
    constraints = [y >= 0, y <= 1, cp.sum(y) == n / 2, G @ y - u <= Gx - L]
    obj = cp.Minimize(u + rho * cp.sum_squares(y - x))
    prob = cp.Problem(obj, constraints)
    for sname in solvers:
        try:
            if sname == 'ECOS':
                prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, abstol=1e-9, reltol=1e-9, feastol=1e-9, max_iters=2000)
            elif sname == 'CVXOPT':
                prob.solve(solver=cp.CVXOPT, warm_start=True, verbose=False, abstol=1e-9, reltol=1e-9, feastol=1e-9, max_iters=2000)
            elif sname == 'MOSEK':
                prob.solve(solver=cp.MOSEK, warm_start=True, verbose=False, msdk_rel_gap=1e-9, msdk_pfeal=1e-9, msdk_dfeal=1e-9, msdk_max_time=60)
            elif sname == 'GUROBI':
                prob.solve(solver=cp.GUROBI, warm_start=True, verbose=False, FeasibilityTol=1e-9, OptimalityTol=1e-9, Threads=1, BarIterLimit=1000)
            else:
                continue
            if prob.status in ('optimal', 'optimal_inaccurate') and y.value is not None:
                yv = np.array(y.value, dtype=np.float64).ravel()
                if yv.shape == (n,) and np.all(np.isfinite(yv)):
                    return yv
        except Exception:
            try:
                prob.solve(solver=sname, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate') and y.value is not None:
                    yv = np.array(y.value, dtype=np.float64).ravel()
                    if yv.shape == (n,) and np.all(np.isfinite(yv)):
                        return yv
            except Exception:
                pass
    return None

def mm_trajectory(x0, n, max_iter, solvers):
    x = x0.copy()
    sc = exact_score(x, n)
    for _ in range(max_iter):
        L, G = lag_data(x, n)
        if L is None:
            break
        yq = solve_mm_qp(x, L, G, n, solvers)
        if yq is None:
            break
        best_x, best_s = x, sc
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            trial = project_capped(x + alpha * (yq - x), n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-12:
                best_s, best_x = ts, trial
        if best_s >= sc - 1e-12:
            break
        prev = x
        x = best_x
        sc = best_s
        if np.max(np.abs(x - prev)) <= 1e-9:
            break
    return x, sc

def pair_polish(x0, n, max_rounds, rng):
    x = x0.copy()
    sc = exact_score(x, n)
    dx = 2.0 / n
    for _ in range(max_rounds):
        D = np.correlate(x, 1.0 - x, mode='full') * dx
        top_k = min(8, 2 * n - 1)
        top_idx = np.argpartition(D, -top_k)[-top_k:]
        if n <= 128:
            ii, jj = np.triu_indices(n, 1)
        else:
            np2 = min(4000, 8 * n)
            ii = rng.integers(0, n, np2)
            jj = rng.integers(0, n, np2)
            m2 = ii < jj
            ii, jj = ii[m2], jj[m2]
        best_t, best_s, best_js = 0.0, sc, None
        for ti in range(len(ii)):
            i, j = int(ii[ti]), int(jj[ti])
            Lp = max(-x[i], x[j] - 1.0)
            Up = min(1.0 - x[i], x[j])
            if Up - Lp < 1e-12:
                continue
            Bv = np.zeros(top_k)
            for ti2 in range(top_k):
                d = int(top_idx[ti2]) - (n - 1)
                a2 = max(d, 0)
                b2 = max(-d, 0)
                m2 = n - abs(d)
                gi = 0.0
                si = i - a2
                if 0 <= si < m2:
                    gi += 1.0 - x[si + b2]
                si2 = i - b2
                if 0 <= si2 < m2:
                    gi -= x[si2 + a2]
                gj = 0.0
                sj = j - a2
                if 0 <= sj < m2:
                    gj += 1.0 - x[sj + b2]
                sj2 = j - b2
                if 0 <= sj2 < m2:
                    gj -= x[sj2 + a2]
                Bv[ti2] = dx * (gi - gj)
            mR = float(np.max(Bv)) if Up > 0 else 1e30
            mL = float(np.max(-Bv)) if Lp < 0 else 1e30
            mer = min(mR, mL)
            if mer > 1e-6:

            fd = float(np.max(Bv * d2 + D[top_idx]))
            for _ in range(20):

                    d2 = c2

            ht = project_capped(ht, n)
            if ht is None or not is_feasible(ht, n):
                continue
            s2 = exact_score(ht, n)
            if s2 < best_s - 1e-13:

        if best_js is None:
            break
        x = best_js[2]
        sc = best_s
    return x, sc

def add_candidate(pool, c, n):
    p = project_capped(c, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-7:
            return
    pool.append(p)

def build_starts(fallback, n, rng, cap):
    pool = []
    add_candidate(pool, np.full(n, 0.5), n)
    add_candidate(pool, fallback, n)
    add_candidate(pool, fallback[::-1].copy(), n)
    add_candidate(pool, 1.0 - fallback[::-1], n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_candidate(pool, blk, n)
    add_candidate(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
    ar = np.arange(n, dtype=np.float64)
    for m in (1, 2, 3):
        for ph in (0.0, math.pi / 2.0):
            add_candidate(pool, 0.5 + 0.35 * np.cos(2.0 * math.pi * m * ar / n + ph), n)
    for sd in (0.04, 0.10, 0.20):
        add_candidate(pool, fallback + rng.normal(0, sd, n), n)
    add_candidate(pool, rng.uniform(0, 1, n), n)
    scored = []
    for idx in range(len(pool)):
        scored.append((exact_score(pool[idx], n), idx))
    scored.sort(key=lambda t: t[0])
    sel = []
    for s0, idx0 in scored:
        if len(sel) >= cap:
            break
        sel.append(pool[idx0])
    return sel
