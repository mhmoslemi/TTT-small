def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(8):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
