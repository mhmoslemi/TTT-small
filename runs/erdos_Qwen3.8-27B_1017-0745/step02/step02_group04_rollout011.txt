def solve_sdp(n, rng):
    """Try moment relaxation, return (u, S) or (None, None)."""
    if n < 2 or n > 64:

    except ImportError:
        return None, None

    dx = 2.0 / n
    h = cp.Variable(n)
    Y = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == s, t >= 0]
    M = cp.bmat([[cp.Constant(1), h.T], [h, Y]])
    cons.append(M >> 0)
    cons.append(Y @ np.ones(n) == s * h)
    ones = np.ones(n)
    cons.append(Y >= 0)
    cons.append(Y >= cp.outer(h, ones) + cp.outer(ones, h) - 1)
    cons.append(Y <= cp.outer(h, ones))
