    J = lag_jacobian(h, n)
    G = np.zeros((2 * n - 1, n + 1))
    G[:, :n] = -J; G[:, n] = 1.0
    return G

def eq_func(z, n):
    return z[:n].sum() - n / 2.0

def eq_grad(z, n):
    g = np.zeros(n + 1); g[:n] = 1.0
    return g

def build_starts(fallback, n, rng, count):
    pool = []
    cands = [fallback.copy(), fallback[::-1].copy(),
             (1.0 - fallback[::-1]).copy(), np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    pi = 3.14159265358979
    for m in range(1, 4):
        for ph in [0.0, 0.5 * pi]:
            c = 0.5 + 0.45 * np.cos(2.0 * pi * m * np.arange(n) / n + ph)
            cands.append(c)
    cands.append(fallback + rng.normal(0, 0.06, n))
    cands.append(fallback + rng.normal(0, 0.15, n))
    cands.append(fallback + rng.normal(0, 0.30, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True; break
        if not dup:
            pool.append(p)
    while len(pool) < count:
        p = project_capped(fallback + rng.normal(0, 0.10, n), n)
        if p is not None and all(np.max(np.abs(p - s)) >= 1e-6 for s in pool):
            pool.append(p)
    return pool[:max(count, 1)]

def slsqp_solve(h0, n, maxiter):
    sc = lag_values(h0, n)
    t0 = min(0.5, float(np.max(sc)) + 1e-9)
    t0 = max(0.0, t0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad, 'args': (n,)}
    ]
    try:
        res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, args=(n,),
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    x = res.x
    if x.shape != (n + 1,) or not np.all(np.isfinite(x)):
        return None
    return project_capped(x[:n], n)

def evaluate_candidate(cand, gb, gs, n):
    if cand is None or not is_feasible(cand, n):
        return gb, gs
    cs = exact_score(cand, n)
    if cs < gs - 1e-12:
        gs, gb = cs, cand.copy()
    for alpha in [0.25, 0.5, 0.75]:
        blend = alpha * cand + (1.0 - alpha) * gb
        bp = project_capped(blend, n)
        if bp is not None and is_feasible(bp, n):
            bs = exact_score(bp, n)
            if bs < gs - 1e-12:
                gs, gb = bs, bp.copy()
    return gb, gs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if (np.all(np.isfinite(p)) and np.all(p >= -1e-8) and
                    np.all(p <= 1.0 + 1e-8)):
                parent = np.clip(p, 0, 1)
    except Exception:
        pass
    if n is None:
        n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n <= 64:
        n_starts, max_iter, n_restart = 8, 220, 5
    elif n <= 128:
        n_starts, max_iter, n_restart = 5, 150, 3
    else:
        n_starts, max_iter, n_restart = 3, 80, 1

    uniform = np.full(n, 0.5)
    starts = build_starts(fallback, n, rng, n_starts)
    if all(np.max(np.abs(s - uniform)) > 1e-6 for s in starts):
        starts[-1] = uniform.copy()
    if parent is not None:
        pp2 = project_capped(parent, n)
        if pp2 is not None and all(np.max(np.abs(s - pp2)) > 1e-6 for s in starts):
            starts[-1] = pp2

    gb = uniform.copy()
    gs = exact_score(uniform, n)
    pf = project_capped(parent, n) if parent is not None else None
    if pf is not None:
        ps = exact_score(pf, n)
        if ps < gs - 1e-12:
            gs, gb = ps, pf.copy()

    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
        cand = slsqp_solve(st, n, max_iter)
        gb, gs = evaluate_candidate(cand, gb, gs, n)

    sigma = 0.04
    for _ in range(n_restart):
        improved = False
        pproj = None
        for _a in range(3):
            pert = gb + rng.normal(0, sigma, n)
            pproj = project_capped(pert, n)
            if pproj is not None and is_feasible(pproj, n):
                if exact_score(pproj, n) <= 0.5 + 1e-12:
                    break
            pproj = None
        if pproj is None:
            sigma = min(0.20, 1.5 * sigma)
            continue
        cand2 = slsqp_solve(pproj, n, max_iter)
        gb, gs = evaluate_candidate(cand2, gb, gs, n)
        if exact_score(gb, n) < 0.5:
            improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = uniform
    fs = exact_score(final, n)
    ub_s = exact_score(uniform, n)
    if fs > ub_s + 1e-12:
        final = uniform; fs = ub_s
    if pf is not None:
        pf_s = exact_score(pf, n)
        if pf_s < fs - 1e-12:
            final, fs = pf, pf_s
    return final.astype(np.float64), float(fs), int(n)
