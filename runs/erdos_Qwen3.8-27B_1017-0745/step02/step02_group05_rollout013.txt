def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def mm_step(x, n, beta, maxiter):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        p = x[p0:p0 + m]
        qv = x[q0:q0 + m]
        q[row] = dx * np.dot(p, 1.0 - qv)
        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    
    def obj(z):
        y = z[:n]
        t = z[n]
        u = y - x
        return t + beta * dx * np.dot(u, u)
    
    def obj_jac(z):
        y = z[:n]
        u = y - x
        g = np.empty(n + 1)
        g[:n] = 2.0 * beta * dx * u
        g[n] = 1.0
        return g
    
    def ineq(z):
        y = z[:n]
        t = z[n]
        return t - q - G @ (y - x)
    
    def ineq_jac(z):
        J = np.empty((L, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    def eq(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    z0 = np.zeros(n + 1)
    z0[:n] = x
    z0[n] = np.max(q) + 1e-8
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        y = res.x[:n]
        if y is None or not np.all(np.isfinite(y)):
            return None
        return y
    except Exception:
        return None

def optimize_mm(x0, n, n_outer, maxiter_qp):

    if x is None or not is_feasible(x, n):
        return None, float('inf')
    score = exact_score(x, n)
    beta = 1.0
    rejs = 0
    for _ in range(n_outer):
        if rejs >= 4:
            break
        y = mm_step(x, n, beta, maxiter_qp)
        if y is None:
            beta = min(beta * 2.0, 64.0)
            rejs += 1

        y = project_capped(y, n)

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    base = fb
    cands = [
        base[::-1].copy(),
        1.0 - base,
        1.0 - base[::-1],
        np.roll(base, 1),
        np.roll(base, -1),
        np.full(n, 0.5),
    ]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.extend([blk, blk[::-1]])
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    cands.append(alt.copy())
    for sd in (0.04, 0.1, 0.2):
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for k in range(1, 4):
        ph = rng.uniform(0, 6.2832)
        cands.append(0.5 + 0.35 * np.sin(2.0 * np.pi * k * np.arange(n) / n + ph))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    while len(pool) < count:
        p = project_capped(base + rng.normal(0, 0.1, n), n)
        if p is not None:
            pool.append(p)
    return pool[:count]

def pair_anneal(x0, n, rng, L, T0, T1):
    x = x0.copy()
    sc = exact_score(x, n)
    best_x = x0.copy()
    best_sc = sc
    for t in range(L):
        q_t = t / max(1, L - 1)
        rho = 1.0 - 0.98 * q_t
        T = T0 * (T1 / T0) ** q_t
        # Pick random pair
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # Pick direction
        if rng.random() < 0.5:
            cap = min(1.0 - x[i], x[j])
        else:
            i, j = j, i
            cap = min(1.0 - x[i], x[j])
        if cap <= 1e-14:
            continue
        amt = cap * rho * (0.05 + 0.95 * rng.random())
        y = x.copy()
        y[i] += amt
        y[j] -= amt
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            continue
        sy = exact_score(y, n)
        delta = sy - sc
        if delta < -1e-13:
            x = y
            sc = sy
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_x = y.copy()
        elif delta < 0:
            if rng.random() < 0.3:
                x = y
                sc = sy
        else:
            try:
                p_acc = np.exp(-delta / max(T, 1e-15))
            except (OverflowError, FloatingPointError):
                p_acc = 0.0
            if rng.random() < p_acc:
                x = y
                sc = sy
    return best_x, best_sc

def greedy_polish(x0, n, sweeps, rng):
    x = x0.copy()
    sc = exact_score(x, n)
    best_x = x0.copy()
    best_sc = sc
    fracs = np.array([0.0625, 0.125, 0.25, 0.375, 0.5, 0.75, 1.0])
    for sw in range(sweeps):
        improved = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            npairs = min(n * (n - 1) // 2, 200)
            pairs = []
            for _ in range(npairs):
                i, j = rng.integers(0, n, 2)
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
        for (i, j) in pairs:
            for (di, dj) in [(i, j), (j, i)]:
                cap = min(1.0 - x[di], x[dj])
                if cap <= 1e-14:
                    continue
                for f in fracs:
                    amt = cap * f
                    y = x.copy()
                    y[di] += amt
                    y[dj] -= amt
                    y = project_capped(y, n)
                    if y is None or not is_feasible(y, n):

                    sy = exact_score(y, n)
                    if sy < sc - 1e-13:
                        x = y
                        sc = sy
                        improved = True
                        if sc < best_sc - 1e-13:
                            best_sc = sc
                            best_x = y.copy()
        if not improved:
            break
    return best_x, best_sc

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

class _QP:
    def __init__(self, x, n, beta, q, G):
        self.x = x
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n
        self.q = q
        self.G = G
    def obj(self, z):
        u = z[:self.n] - self.x
        return z[self.n] + self.beta * self.dx * np.dot(u, u)
    def obj_jac(self, z):
        u = z[:self.n] - self.x
        g = np.empty(self.n + 1)
        g[:self.n] = 2.0 * self.beta * self.dx * u
        g[self.n] = 1.0
        return g
    def ineq(self, z):
        return z[self.n] - self.q - self.G @ (z[:self.n] - self.x)
    def ineq_jac(self, z):
        L = self.q.shape[0]
        J = np.empty((L, self.n + 1))
        J[:, :self.n] = -self.G
        J[:, self.n] = 1.0
        return J
    def eq(self, z):
        return np.sum(z[:self.n]) - self.n / 2.0
    def eq_jac(self, z):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J

def mm_step(x, n, beta, maxiter):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        p = x[p0:p0 + m]
        qv = x[q0:q0 + m]
        q[row] = dx * np.dot(p, 1.0 - qv)
        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None
    prob = _QP(x, n, beta, q, G)
    z0 = np.zeros(n + 1)
    z0[:n] = x
    z0[n] = float(np.max(q)) + 1e-8
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
    ]
    try:
        res = minimize(prob.obj, z0, jac=prob.obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        y = res.x[:n]
        if y is None or not np.all(np.isfinite(y)):
            return None
        return y
    except Exception:
        return None

            di, dj = i, j
        else:
            di, dj = j, i
