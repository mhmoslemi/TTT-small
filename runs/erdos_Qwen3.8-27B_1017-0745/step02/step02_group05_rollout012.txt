    c_sigma = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = max(0.0, min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff)))
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    m = helmert_transpose(start_h - 0.5, n)
    sigma = sigma0
    p_sigma = np.zeros(d)
    p_c = np.zeros(d)
    C = np.eye(d) if use_full else np.ones(d)
    for t in range(G):
        Z = rng.standard_normal((P, d))
        if use_full:
            C_sym = 0.5 * (C + C.T)
            try:
                ev, U = np.linalg.eigh(C_sym)
            except np.linalg.LinAlgError:
                break
            ev = np.clip(ev, 1e-10, 1e4)
            sq = np.sqrt(ev)
            A = U @ np.diag(sq)
            Y = Z @ A.T
            Chalf_inv = U @ np.diag(1.0 / sq) @ U.T
        else:
            C = np.clip(C, 1e-10, 1e4)
            sq = np.sqrt(C)
            Y = Z * sq
        scores = np.full(P, np.inf)
        Xs = np.zeros((P, d))
        for k in range(P):
            xk = m + sigma * Y[k]
            if not np.all(np.isfinite(xk)):
                continue
            Xs[k] = xk
            hk = latent_to_h(xk, n)
            if hk is None:
                continue
            sk = exact_score(hk, n)
            scores[k] = sk
            if sk < best_s - 1e-12:
                best_s = sk
                best_h = hk.copy()
        order = np.argsort(scores, kind='stable')
        n_fin = int(np.sum(np.isfinite(scores)))
        if n_fin < mu:
            break
        idx = order[:mu]
        x_sel = Xs[idx]
        m_old = m.copy()
        m_new = np.sum(w[:, None] * x_sel, axis=0)
        y_w = (m_new - m_old) / sigma
        if use_full:
            p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (Chalf_inv @ y_w)
            rank1 = np.outer(p_c, p_c)
            for i2 in range(mu):
                yi = (x_sel[i2] - m_old) / sigma
                rank1 += w[i2] * np.outer(yi, yi)
        else:
            inv_sq = 1.0 / sq
            p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (y_w * inv_sq)
            rank1_diag = np.zeros(d)
            for i2 in range(mu):
                yi = (x_sel[i2] - m_old) / sigma
                rank1_diag += w[i2] * yi * yi
        norm_ps = np.linalg.norm(p_sigma)
        denom = math.sqrt(max(1e-30, 1.0 - (1.0 - c_sigma) ** (2 * (t + 1))))
        h_sig = 1.0 if norm_ps / denom / chi_d < 1.4 + 2.0 / (d + 1) else 0.0
        p_c = (1 - c_c) * p_c + h_sig * math.sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        if use_full:
            C_sym = 0.5 * (C + C.T)
            b1 = (1 - h_sig) * c_c * (2 - c_c) * C_sym
            C = (1 - c1 - c_mu) * C_sym + c1 * (np.outer(p_c, p_c) + b1) + c_mu * rank1
            C = 0.5 * (C + C.T)
        else:
            b1_diag = (1 - h_sig) * c_c * (2 - c_c) * C
            C = (1 - c1 - c_mu) * C + c1 * (p_c * p_c + b1_diag) + c_mu * rank1_diag
        log_s = (c_sigma / d_sigma) * (np.linalg.norm(p_sigma) / chi_d - 1.0)
        sigma = sigma * math.exp(max(-50.0, min(50.0, log_s)))
        sigma = max(1e-4, min(0.5, sigma))
        m = m_new
        if not np.all(np.isfinite(m)) or not np.all(np.isfinite(C)):
            break
    hm = latent_to_h(m, n)
    if hm is not None:
        sm = exact_score(hm, n)
        if sm < best_s - 1e-12:
            best_s = sm
            best_h = hm.copy()
    return best_h, best_s

def build_starts(base, n, rng, count):
    pool = [base.copy()]
    n2 = n // 2
    blk = np.zeros(n)
    blk[:n2] = 1.0
    if n % 2 == 1:
        blk[n2] = 0.5
    cands = [blk, base[::-1].copy(), 1.0 - base, np.roll(base, n // 4), np.roll(base, -n // 4)]
    for sd in (0.05, 0.15):
        cands.append(base + rng.normal(0, sd, n))
    for k in (1, 2):
        ph = rng.uniform(0, 2 * math.pi)
        cands.append(0.5 + 0.3 * np.sin(2 * math.pi * k * np.arange(n) / n + ph))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    cands.append(alt)
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = p.copy()
    except NameError:
        pass
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
    if n is None:
        n = 47
    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.25, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    const_h = np.full(n, 0.5)
    safe_h = const_h.copy()
    safe_s = exact_score(const_h, n)
    base = const_h
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and is_feasible(pp, n):
            base = pp
            sp = exact_score(pp, n)
            if sp < safe_s:
                safe_h = pp.copy()
                safe_s = sp
    if n <= 64:
        n_starts, P, G, use_full = 5, max(8, min(24, 4 + int(3 * math.log(max(2, n - 1))))), 120, True
        sig_first, sig_other = 0.08, 0.14
    elif n <= 256:
        n_starts, P, G, use_full = 4, max(8, min(20, 4 + int(3 * math.log(max(2, n - 1))))), 40, False
        sig_first, sig_other = 0.06, 0.10
    else:
        n_starts, P, G, use_full = 3, max(8, min(12, 4 + int(3 * math.log(max(2, n - 1))))), 12, False
        sig_first, sig_other = 0.05, 0.08
    starts = build_starts(base, n, rng, n_starts)
    gb = safe_h.copy()
    gs = safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for i, st in enumerate(starts):
        sig = sig_first if i == 0 else sig_other
        try:
            rh, rs = cmaes_run(st, n, sig, P, G, use_full, rng)
            if rh is not None and rs < gs - 1e-12:
                gs, gb = rs, rh.copy()
        except Exception:
            pass
    fin = project_capped(gb, n)
    if fin is None or not is_feasible(fin, n):
        fin = safe_h.copy()
    fs = exact_score(fin, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        fin = safe_h.copy()
        fs = exact_score(fin, n)
    return fin.astype(np.float64), float(fs), int(n)
