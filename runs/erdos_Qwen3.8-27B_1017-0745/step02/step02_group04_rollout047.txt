def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 10):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(r, 1.0 - h[idx])
            if add <= 0.0:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-r, h[idx])
            if sub <= 0.0:
                break
            h[idx] -= sub
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.size == 0 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_q(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.empty(L)
    for d in range(-n + 1, n):
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        q[d + n - 1] = dx * np.dot(h[a:a + m], 1.0 - h[b:b + m])
    return q

def lag_J(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    J = np.empty((L, n))
    J.fill(0.0)
    for d in range(-n + 1, n):
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        idx = d + n - 1
        xp = h[a:a + m]
        y = 1.0 - h[b:b + m]
        J[idx, a:a + m] += dx * y
        J[idx, b:b + m] -= dx * xp
    return J

def set_ctx(n, active):

    n = _ctx['n']
    K = _ctx['K']
    act = _ctx['active']
    h = z[:n]
    t = z[n]

        val = t - q
    else:

        return np.full(K, -1e100)
    return val

def _ineq_jac(z):
    n = _ctx['n']
    K = _ctx['K']
    act = _ctx['active']
    Jout = np.empty((K, n + 1))
    h = z[:n]
    if not np.all(np.isfinite(h)):

        Jout[:, n] = 1.0
        return Jout

def solve_slqp(x, n, maxiter, active=None):
    if n < 1 or maxiter <= 0:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.shape != (n,) or not np.all(np.isfinite(x)):
        return None
    x = np.clip(x, 0.0, 1.0)
    set_ctx(n, active)
    q = lag_q(x, n)
    if not np.all(np.isfinite(q)):
        return None
    t0 = float(np.max(q))
    if not np.isfinite(t0):
        t0 = 0.0
    if t0 < 0.0:
        t0 = 0.0
    t0 += 1e-8
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-11, 'disp': False})
    except Exception:
        return None

        return None

        return None

        return None
    return y

def is_feasible(h, n):

def lag_q(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.empty(L)
    for d in range(-n + 1, n):
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        q[d + n - 1] = dx * np.dot(h[a:a + m], 1.0 - h[b:b + m])
    return q

def lag_J(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    J = np.empty((L, n))
    J.fill(0.0)
    for d in range(-n + 1, n):
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        idx = d + n - 1
        xp = h[a:a + m]
        y = 1.0 - h[b:b + m]
        J[idx, a:a + m] += dx * y
        J[idx, b:b + m] -= dx * xp
    return J
