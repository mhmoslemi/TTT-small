    return float(np.max(q))

def quad_roots(a, b, c, lo, hi):
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    if abs(a) < 1e-14:
        if abs(b) < 1e-14:
            return []
        r = -c / b
        return [max(lo, min(hi, r))] if lo - tol <= r <= hi + tol else []
    disc = b * b - 4.0 * a * c
    if disc < -1e-10:
        return []
    if disc < 0.0:
        disc = 0.0
    sq = math.sqrt(disc)
    qv = -0.5 * (b + sq) if b > 0 else -0.5 * (b - sq)
    r1 = qv / a
    r2 = c / qv if abs(qv) > 1e-30 else float('inf')
    out = []
    for r in (r1, r2):
        if np.isfinite(r) and lo - tol <= r <= hi + tol:
            out.append(max(lo, min(hi, r)))
    return out

def dct_dir(k, n):
    i = np.arange(n)
    v = np.cos(np.pi * k * (i + 0.5) / n)
    v -= np.mean(v)
    v[n - 1] -= np.sum(v)
    mx = np.max(np.abs(v))
    if mx < 1e-14:
        return None
    return v / mx

def line_search(h, n, v, n_cand):
    dx = 2.0 / n
    lo, hi = -1e15, 1e15
    for i in range(n):
        if v[i] > 1e-15:
            lo = max(lo, -h[i] / v[i])
            hi = min(hi, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / v[i])
            hi = min(hi, -h[i] / v[i])
    if hi - lo < 1e-12:
        return None, None
    n_lag = 2 * n - 1
    A = np.zeros(n_lag)
    B = np.zeros(n_lag)
    C = np.zeros(n_lag)
    for idx in range(n_lag):
        s = idx - (n - 1)
        p0, q0, m = max(s, 0), max(-s, 0), n - abs(s)
        hp, hq = h[p0:p0 + m], h[q0:q0 + m]
        vp, vq = v[p0:p0 + m], v[q0:q0 + m]
        A[idx] = dx * np.dot(hp, 1.0 - hq)
        B[idx] = dx * (np.dot(vp, 1.0 - hq) - np.dot(hp, vq))
        C[idx] = -dx * np.dot(vp, vq)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return None, None
    alphas = [lo, hi, 0.0]
    for idx in range(n_lag):
        if abs(C[idx]) > 1e-14:
            sp = -B[idx] / (2.0 * C[idx])
            if lo - 1e-10 <= sp <= hi + 1e-10:
                alphas.append(max(lo, min(hi, sp)))
    for i in range(n_lag):
        for j in range(i + 1, n_lag):
            alphas.extend(quad_roots(C[i] - C[j], B[i] - B[j], A[i] - A[j], lo, hi))
    alphas.sort()
    deduped = [alphas[0]]
    for a in alphas[1:]:
        if a - deduped[-1] > 1e-10 * max(1.0, abs(lo), abs(hi)):
            deduped.append(a)
    alphas = deduped
    aa = np.array(alphas)
    F = np.max(A[None, :] + B[None, :] * aa[:, None] + C[None, :] * aa[:, None] ** 2, axis=1)
    cur_s = float(F[np.argmin(np.abs(aa))])
    order = np.argsort(F)[:n_cand]
    best_h, best_s = None, cur_s
    for oi in order:
        a = alphas[oi]
        if abs(a) < 1e-14:
            continue
        tp = project_capped(h + a * v, n)
        if tp is None:
            continue
        sc = exact_score(tp, n)
        if sc < best_s - 1e-12:
            best_s, best_h = sc, tp
    if best_h is None:
        return None, None
    return best_h, best_s

def add_start(pool, v, n):
    p = project_capped(v, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p)

def run(seed=42, **kwargs):
    n = 47
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = p.size
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    gs = exact_score(fb, n)
    gb = fb.copy()
    if n == 1:
        return fb, gs, 1
    starts = []
    add_start(starts, fb, n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)
    add_start(starts, fb + rng.normal(0, 0.08, n), n)
    add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * (np.arange(n) + 0.5) / n), n)
    add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:3]
    if n <= 48:
        k_list = list(range(1, n))
    elif n <= 64:
        k_list = sorted(set(int(round(x)) for x in np.geomspace(1, n - 1, 24)))
    elif n <= 256:
        k_list = sorted(set(int(round(x)) for x in np.geomspace(1, n - 1, 12)))
    else:
        k_list = sorted(set(int(round(x)) for x in np.geomspace(1, n - 1, 8)))
    k_list = [k for k in k_list if 1 <= k <= n - 1]
    n_polish = 2 if n <= 64 else 1
    nc = 8 if n <= 64 else 4
    for st in starts:
        cur = st.copy()
        cur_s = exact_score(cur, n)
        if cur_s < gs - 1e-12:
            gs, gb = cur_s, cur.copy()
        for _sw in range(n_polish):
            improved = False
            ko = k_list[:]
            rng.shuffle(ko)
            for k in ko:
                v = dct_dir(k, n)
                if v is None:
                    continue
                try:
                    nh, ns = line_search(cur, n, v, nc)
                except Exception:
                    continue
                if nh is not None and ns < cur_s - 1e-12:
                    cur, cur_s = nh, ns
                    improved = True
                    if cur_s < gs - 1e-12:
                        gs, gb = cur_s, cur.copy()
            if not improved and _sw > 0:
                break
    for _ in range(n_polish):
        improved = False
        cur = gb.copy()
        cur_s = gs
        ko = k_list[:]
        rng.shuffle(ko)
        for k in ko:
            v = dct_dir(k, n)
            if v is None:
                continue
            try:
                nh, ns = line_search(cur, n, v, nc)
            except Exception:
                continue
            if nh is not None and ns < cur_s - 1e-12:
                cur, cur_s = nh, ns
                improved = True
                if cur_s < gs - 1e-12:
                    gs, gb = cur_s, cur.copy()
        if not improved:
            break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
