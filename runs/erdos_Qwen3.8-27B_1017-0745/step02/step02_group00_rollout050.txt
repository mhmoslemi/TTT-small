def project_capped(z, n):

            hi = mid

def is_feasible(h, n):

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')

        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        if np.isfinite(val):
            return val

        pass

        s = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
