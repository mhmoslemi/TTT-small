        theta = max((n / 2.0) * tau, 1e-10)
        alpha = 0.2
        for _ in range(max_iter):
            phi_h, p_h, _ = softmax_val(h, n, theta)
            if phi_h is None:
                break
            g = (2.0 / n) * lag_grad(h, n, p_h)
            for bt in range(25):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                step = y - h
                if float(np.max(np.abs(step))) < 1e-10:
                    break
                gd = float(np.dot(g, step))
                if gd > -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _, _ = softmax_val(y, n, theta)
                if phi_y is not None and phi_y <= phi_h + 1e-4 * gd + 1e-12 * max(1.0, abs(phi_h)):
                    h = y
                    alpha = min(0.5, 1.15 * alpha)
                    break
                alpha *= 0.5
            if alpha < 1e-9:
                break
    return h

def add_start(starts, v, n):
    try:
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.shape != (n,):
            return
        h2 = project_capped(v, n)
        if h2 is None:
            return
        for ex in starts:
            if float(np.max(np.abs(h2 - ex))) < 1e-8:
                return
        starts.append(h2)
    except Exception:
        pass

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = max(0, int(seed))
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    n_s_def = 6 if n <= 128 else (4 if n <= 512 else 2)
    mi_def = 60 if n <= 128 else (25 if n <= 512 else 10)
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_s_def)), 8))
    except Exception:
        n_starts = n_s_def
    try:
        max_iter = max(1, min(int(kwargs.get('max_iter', mi_def)), 120))
    except Exception:
        max_iter = mi_def
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, 1.0 - fallback[::-1], n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    for st in starts:
        try:
            opt = project_capped(continuation_opt(st.copy(), n, max_iter), n)
            if opt is None:
                continue
            sc = exact_score(opt, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best, best_score = opt.copy(), sc
        except Exception:
            continue
    n_hops = 4 if n <= 128 else 2
    hop_sds = [0.05, 0.03, 0.02, 0.08]
    for k in range(n_hops):
        try:
            sd_hop = hop_sds[k % len(hop_sds)]
            hop = project_capped(best + rng.normal(0, sd_hop, n), n)
            if hop is None:
                continue
            hs = exact_score(hop, n)
            if np.isfinite(hs) and hs < best_score - 1e-12:
                best, best_score = hop.copy(), hs
            opt = project_capped(continuation_opt(hop, n, max_iter), n)
            if opt is not None:
                sc = exact_score(opt, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = opt.copy(), sc
        except Exception:
            continue
    for k in range(2):
        try:
            hop = project_capped(best + rng.normal(0, 0.015, n), n)
            if hop is None:
                continue
            opt = project_capped(continuation_opt(hop, n, max_iter), n)
            if opt is not None:
                sc = exact_score(opt, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = opt.copy(), sc
        except Exception:
            continue
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-15:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
