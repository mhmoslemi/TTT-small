import numpy as np
from math import copysign, sqrt, inf

def project_capped(z, n):
    s = n / 2.0
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if hi - lo < 1e-16:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(10):
        if abs(r) < 1e-12:
            break
        if r > 0:
            sl = 1.0 - h
            t = float(np.sum(sl))
            if t < 1e-14:
                break
            h += r * sl / t
        else:
            t = float(np.sum(h))
            if t < 1e-14:
                break
            h += r * h / t
        r = s - float(np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return inf
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return inf
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8 * max(1, n):
        return inf
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return inf

def compute_F(h, n):
    ds = np.arange(-(n - 1), n)
    F = np.zeros(2 * n - 1)
    for di in range(2 * n - 1):
        d = ds[di]
        if d >= 0:
            lo, hi = 0, n - d
        else:
            lo, hi = -d, n
        F[di] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return F

def compute_b(h, p, q, n):
    ds = np.arange(-(n - 1), n)
    b = np.zeros(2 * n - 1)
    for r, sgn in ((p, 1.0), (q, -1.0)):
        m1 = (ds >= -r) & (ds <= n - 1 - r)
        m2 = (ds >= r - n + 1) & (ds <= r)
        b[m1] += sgn * (1.0 - h[r + ds[m1]])
        b[m2] -= sgn * h[r - ds[m2]]
    return b

def line_hull(m, c):
    if len(m) == 0:
        return np.array([]), np.array([]), np.array([])
    order = np.argsort(m)
    um, uc = [], []
    for i in order:
        mi, ci = float(m[i]), float(c[i])
        if um and abs(mi - um[-1]) < 1e-15:
            if ci > uc[-1]:
                uc[-1] = ci
        else:
            um.append(mi)
            uc.append(ci)
    hs, hm, hc = [], [], []
    for idx in range(len(um)):
        mi, ci = um[idx], uc[idx]
        while hm:
            x = (hc[-1] - ci) / (mi - hm[-1])
            if x > hs[-1]:
                break
            hs.pop(); hm.pop(); hc.pop()
        if not hm:
            hs.append(-1e30)
        else:
            hs.append(x)
        hm.append(mi); hc.append(ci)
    return np.array(hs), np.array(hm), np.array(hc)

def qroots(A, B, C, lo, hi):
    if abs(A) < 1e-15:
        if abs(B) < 1e-15:
            return []
        r = -C / B
        return [r] if lo <= r <= hi else []
    disc = B * B - 4.0 * A * C
    if disc < 0:
        if disc > -1e-10 * (B * B + abs(4.0 * A * C) + 1.0):
            disc = 0.0
        else:
            return []
    sd = sqrt(max(0.0, disc))
    q = -0.5 * (B + copysign(sd, B))
    r1 = q / A
    r2 = C / q if abs(q) > 1e-30 else -B / (2.0 * A)
    if abs(disc) < 1e-14:
        return [r1] if lo <= r1 <= hi else []
    out = []
    for r in (r1, r2):
        if lo <= r <= hi:
            out.append(r)
    return sorted(set(round(r, 14) for r in out))

def chord_minimize(h, p, q, n, F):
    L = max(-h[p], h[q] - 1.0)
    U = min(1.0 - h[p], h[q])
    cur_val = float(np.max(F))
    if U - L < 1e-15:
        return 0.0, cur_val
    b = compute_b(h, p, q, n)
    d0 = n - 1
    dq = q - p
    i1, i2 = d0 + dq, d0 - dq
    v1, v2 = (0 <= i1 <= 2 * n - 2), (0 <= i2 <= 2 * n - 2)
    excl = {d0}
    if v1:
        excl.add(i1)
    if v2:
        excl.add(i2)
    idx0 = [i for i in range(2 * n - 1) if i not in excl]
    hs, hm, hc = line_hull(b[idx0], F[idx0])
    cands = set()
    cands.update([L, U, 0.0])
    for s in hs:
        if L < s < U:
            cands.add(float(s))
    if v1 and v2 and i1 != i2:
        db = b[i1] - b[i2]
        if abs(db) > 1e-15:
            ts = (F[i2] - F[i1]) / db
            if L <= ts <= U:
                cands.add(float(ts))
    fa_A, fa_B, fa_C = -2.0, b[d0], F[d0]
    # Intersections with each hull line
    for li in range(len(hm)):
        m_k, c_k, s_lo = float(hm[li]), float(hc[li]), float(hs[li])
        s_hi = float(hs[li + 1]) if li + 1 < len(hm) else 1e30
        a_lo = max(L, s_lo)
        a_hi = min(U, s_hi)
        if a_hi - a_lo < 1e-14:
            continue
        cands.update(qroots(fa_A - 0.0, fa_B - m_k, fa_C - c_k, a_lo, a_hi))
        if v1:
            cands.update(qroots(1.0, b[i1] - m_k, F[i1] - c_k, a_lo, a_hi))
        if v2:
            cands.update(qroots(1.0, b[i2] - m_k, F[i2] - c_k, a_lo, a_hi))
    if v1:
        cands.update(qroots(-3.0, b[d0] - b[i1], F[d0] - F[i1], L, U))
        vtx = -b[i1] / 2.0
        if L <= vtx <= U:
            cands.add(vtx)
    if v2:
        cands.update(qroots(-3.0, b[d0] - b[i2], F[d0] - F[i2], L, U))
        vtx = -b[i2] / 2.0
        if L <= vtx <= U:
            cands.add(vtx)
    best_t, best_v = 0.0, cur_val
    a_arr = np.zeros(2 * n - 1)
    a_arr[d0] = -2.0
    if v1:
        a_arr[i1] = 1.0
    if v2:
        a_arr[i2] = 1.0
    for t in cands:
        t = max(L, min(U, t))
        vals = F + b * t + a_arr * t * t
        mv = float(np.max(vals))
        if mv < best_v - 1e-14:
            best_v = mv
            best_t = t
    if abs(best_t) < 1e-13:
        return 0.0, cur_val
    return best_t, best_v

def pair_sweep(h, n, budget, rng):
    best_h = h.copy()
    best_s = exact_score(best_h, n)
    npairs = n * (n - 1) // 2
    if npairs <= budget and npairs <= 50000:
        pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    else:
        pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    attempts = 0
    F = compute_F(best_h, n)
    full_sweep_improved = False
    remaining = budget
    while remaining > 0 and attempts < budget:
        perm = rng.permutation(len(pairs))
        full_sweep_improved = False
        for pi in perm:
            if remaining <= 0:
                break
            p, q = pairs[pi]
            remaining -= 1
            attempts += 1
            L = max(-best_h[p], best_h[q] - 1.0)
            U = min(1.0 - best_h[p], best_h[q])
            if U - L < 1e-14:
                continue
            t_opt, v_opt = chord_minimize(best_h, p, q, n, F)
            if abs(t_opt) < 1e-13:
                continue
            trial = best_h.copy()
            ps = best_h[p] + best_h[q]
            trial[p] = max(0.0, min(1.0, best_h[p] + t_opt))
            trial[q] = ps - trial[p]
            if trial[q] < -1e-12 or trial[q] > 1.0 + 1e-12:
                continue
            sc = exact_score(trial, n)
            if np.isfinite(sc) and sc < best_s - 1e-13:
                best_h = trial
                best_s = sc
                F = compute_F(best_h, n)
                full_sweep_improved = True
        if not full_sweep_improved:
            break
    return best_h, best_s, budget - attempts

def add_start(starts, v, n, tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < tol:
            return
    starts.append(h2)
