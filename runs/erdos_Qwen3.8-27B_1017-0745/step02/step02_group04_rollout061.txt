            gs, gb = so, yo.copy()
        else:
            T = max(1e-5, 1.5e-3 * (0.8 ** r))
            if so < gs or rng.uniform() < np.exp(-(so - gs) / T):
                gb = yo.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12 or fs == float('inf'):
        gb, fs = fallback.copy(), fb
    return gb, fs, int(n)
