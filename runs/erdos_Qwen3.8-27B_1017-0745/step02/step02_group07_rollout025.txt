def anneal(h0, s0, F, n, steps, rng):
    h = h0.copy()
    cs = s0
    dx = 2.0 / n
    T0 = 0.25 * dx
    Tf = 0.01 * dx
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = T0 * (Tf / T0) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        if rng.random() < 0.8:
            # pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n - 1))
            if j >= i: j += 1
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            tlo = max(-h[i], h[j] - 1.0)
            thi = min(1.0 - h[i], h[j])
        else:
            # collective
            if F is not None and F.shape[1] > 0:
                z = rng.standard_normal(F.shape[1])
                d = F @ z
            else:
                d = rng.standard_normal(n)
            d -= d.mean()
            d[-1] = -d[:-1].sum()
            mx = float(np.max(np.abs(d)))
            if mx < 1e-14 or not np.isfinite(mx):
                continue
            d /= mx
            tlo, thi = -delta, delta
            for i in range(n):
                if d[i] > 1e-14:
                    tlo = max(tlo, -h[i] / d[i])
                    thi = min(thi, (1.0 - h[i]) / d[i])
                elif d[i] < -1e-14:
                    tlo = max(tlo, (1.0 - h[i]) / d[i])
                    thi = min(thi, -h[i] / d[i])
        tlo = max(tlo, -delta)
        thi = min(thi, delta)
        if tlo >= thi - 1e-12 or not np.isfinite(tlo) or not np.isfinite(thi):

        t = tlo + rng.random() * (thi - tlo)
        cand = project_capped(h + t * d, n)

        if float(np.max(np.abs(cand - h))) < 1e-10:

        ns = exact_score(cand, n)
        if ns < cs or rng.random() < math.exp(-max(0, ns - cs) / T) if T > 0 else False:
