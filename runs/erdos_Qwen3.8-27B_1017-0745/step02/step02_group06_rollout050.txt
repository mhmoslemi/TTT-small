        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:

            h[idx] -= d
            r += d

    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jacobian(h, n):
    n_lags = 2 * n - 1
    dx = 2.0 / n
    J = np.zeros((n_lags, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        i = np.arange(max(0, -k), min(n, n - k))
        j = i + k
        # d L_k / d h[j] += dx * (1 - h[i])
        # d L_k / d h[i] -= dx * h[j]
        np.add.at(J[row], j, dx * (1.0 - h[i]))
        np.add.at(J[row], i, -dx * h[j])
    return J

                       bounds=bounds, constraints=[eq, ineq],
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
