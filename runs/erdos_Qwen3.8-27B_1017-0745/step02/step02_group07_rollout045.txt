  # We can cache m and coefficients? loop.

```python
import numpy as np
import math
try:
    from scipy.optimize import minimize as sp_minimize
except Exception:
    sp_minimize = None

G_N = 47


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 4):
        if abs(r) < 1e-14:
            break
        if r > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - float(np.sum(h))
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h


def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-8


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if not is_feasible(h, n):
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    v = float(np.max(c))
    return v if math.isfinite(v) else float("inf")


def active_subgrad(h, n, active):
    grad = np.zeros(n, dtype=np.float64)
    ctr = n - 1
    for p in active:
        p = int(p)
        if p == ctr:
            grad += 1.0 - 2.0 * h
        elif p > ctr:
            r = p - ctr
            m = n - r
            grad[:m] += -h[r:]
            grad[r:n] += 1.0 - h[:m]
        else:
            r = ctr - p
            m = n - r
            grad[:m] += 1.0 - h[r:]
            grad[r:n] += -h[:m]
    return grad


def direction_interval(h, d, limit):
    lo = -limit
    hi = limit
    for k in range(h.size):
        a = float(d[k])
        hk = float(h[k])
        if a > 1e-13:
            lo = max(lo, -hk / a)
            hi = min(hi, (1.0 - hk) / a)
        elif a < -1e-13:
            lo = max(lo, (1.0 - hk) / a)
            hi = min(hi, -hk / a)
    if not (math.isfinite(lo) and math.isfinite(hi)) or hi <= lo + 1e-14:
        return None
    return lo, hi


def gen_ts(lo, hi, cnt):
    if hi <= lo + 1e-14:
        return np.array([], dtype=np.float64)
    f = np.linspace(0.0, 1.0, int(cnt), dtype=np.float64)
    t = lo + f * (hi - lo)
    t = t[np.isfinite(t)]
    if t.size == 0:
        return np.array([], dtype=np.float64)
    return np.unique(t)


def subgrad_polish(h, n, rng, iters):
    h = h.copy()
    s = exact_score(h, n)
    if not math.isfinite(s):
        return h, s
    for _ in range(iters):
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        mx = float(np.max(c))
        tol = max(1e-8, 1e-5 * abs(mx))
        act = np.flatnonzero(c > mx - tol)
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        g = active_subgrad(h, n, act)
        g = g - float(np.mean(g))
        nm = float(np.max(np.abs(g)))
        if nm < 1e-14:
            break
        d = g / nm
        moved = False
        for step in (0.04, 0.015, 0.09, 0.006, 0.18, 0.30):
            cands = (
                project_capped(h + step * d, n),
                project_capped(h - 0.5 * step * d, n),
            )
            for cand in cands:
                if cand is None:
                    continue
                ts = exact_score(cand, n)
                if ts < s - 1e-12:
                    h = cand
                    s = ts
                    moved = True
                    break
            if moved:
                break
        if not moved:
            break
    return h, exact_score(h, n)


def as_jacob_arg(x):
    n = G_N
    if x.size != n or not np.all(np.isfinite(x)):
        return 1.0
    bad = float(np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x)))
    if bad > 1e-8:
        return 1.0 + 10.0 * bad
    if abs(float(np.sum(x) - n / 2.0)) > 1e-6:
        return 1.0 + 10.0 * abs(float(np.sum(x) - n / 2.0))
    return exact_score(np.clip(x, 0.0, 1.0), n)


def as_jacob_grad(x):
    n = G_N
    if x.size != n or not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=np.float64)
    if np.any(x < -1e-8) or np.any(x > 1.0 + 1e-8):
        return np.zeros(n, dtype=np.float64)
    if abs(float(np.sum(x) - n / 2.0)) > 1e-6:
        return np.zeros(n, dtype=np.float64)
    h = np.clip(x, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.zeros(n, dtype=np.float64)
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    mx = float(np.max(c))
    tol = max(1e-8, 1e-5 * abs(mx))
    act = np.flatnonzero(c > mx - tol)
    if act.size == 0:
        act = np.array([int(np.argmax(c))])
    return active_subgrad(h, n, act) / float(act.size)


def sum_constraint_as(x):
    return float(np.sum(x) - G_N / 2.0)


def slsqp_opt(h, n):
    if sp_minimize is None:
        return h, exact_score(h, n)
    s0 = exact_score(h, n)
    x0 = np.asarray(h, dtype=np.float64).copy()
    try:
        res = sp_minimize(
            as_jacob_arg,
            x0,
            method="SLSQP",
            jac=as_jacob_grad,
            constraints={"type": "eq", "fun": sum_constraint_as},
            bounds=[(0.0, 1.0)] * n,
            options={"maxiter": 40, "ftol": 1e-11},
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).copy()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.clip(x, 0.0, 1.0)
                pr = project_capped(x, n)
                if pr is not None:
                    ts = exact_score(pr, n)
                    if math.isfinite(ts) and ts < s0 - 1e-13:
                        return pr, ts
    except Exception:
        pass
    return h, s0


def pair_polish(h, n, rng, iters, pairs_per, samp):
    h = h.copy()
    s = exact_score(h, n)
    if not math.isfinite(s):
        return h, s
    for _ in range(iters):
        base = h.copy()
        bs = s
        bm = None
        msk = np.abs(base[:, None] - base[None, :]) > 1e-10
        idx = np.argwhere(msk)[0:2]
        idx = idx[idx[:, 0] < idx[:, 1]]
        if idx.shape[0] == 0:
            break
        if idx.shape[0] > pairs_per:
            idx = idx[rng.permutation(idx.shape[0])[:pairs_per]]
        else:
            idx = idx[rng.permutation(idx.shape[0])]
        for row in idx:
            i = int(row[0])
            j = int(row[1])
            lo = max(-base[i], base[j] - 1.0)
            hi = min(1.0 - base[i], base[j])
            if hi - lo < 1e-12:
                continue
            c = samp if hi - lo > 0.25 else max(9, samp // 2)
            cand = base.copy()
            for t in gen_ts(lo, hi, c):
                t = float(t)
                if abs(t) < 1e-14:
                    continue
                ni = base[i] + t
                nj = base[j] - t
                if ni < -1e-10 or ni > 1.0 + 1e-10:
                    continue
                if nj < -1e-10 or nj > 1.0 + 1e-10:
                    continue
                cand[i] = ni
                cand[j] = nj
                ts = exact_score(cand, n)
                if ts < bs - 1e-12:
                    bs = ts
                    bm = cand.copy()
        if bm is not None:
            h = bm
        else:
            break
    return h, exact_score(h, n)


def anneal_pair(h, n, rng, steps, bh, bbs):
    curr = h.copy()
    cs = exact_score(curr, n)
    if not math.isfinite(cs):
        curr = bh.copy()
        cs = bbs
    T0 = 0.55 / max(1, n)
    Tf = 0.02 / max(1, n)
    for k in range(steps):
        rho = k / float(max(1, steps - 1))
        T = max(1e-15, T0 * (Tf / T0) ** rho)
        delta = 0.20 * (1.0 - 0.70 * rho)
        used = False
        if n >= 5 and rng.random() < 0.22:
            d = rng.normal(0.0, 1.0, n)
            d -= float(np.mean(d))
            if n > 1:
                d[n - 1] = -float(np.sum(d[:-1]))
            nm = float(np.max(np.abs(d)))
            if nm > 1e-13:
                d /= nm
                iv = direction_interval(curr, d, delta)
                if iv is not None:
                    t = iv[0] + float(rng.random()) * (iv[1] - iv[0])
                    cand = project_capped(curr + t * d, n)
                    if cand is not None:
                        ts = exact_score(cand, n)
                        if math.isfinite(ts) and ts < bbs - 1e-12:
                            bh = cand.copy()
                            bbs = ts
                        if ts <= cs + 1e-14 or math.log(max(float(rng.random()), 1e-300)) < -(ts - cs) / T:
                            curr = cand
                            cs = ts
                        used = True
        if not used:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
            lo = max(-curr[i], curr[j] - 1.0)
            hi = min(1.0 - curr[i], curr[j])
            lo = max(lo, -delta)
            hi = min(hi, delta)
            if hi - lo < 1e-12:
                continue
            t = lo + float(rng.random()) * (hi - lo)
            cand = curr.copy()
            cand[i] = curr[i] + t
            cand[j] = curr[j] - t
            ts = exact_score(cand, n)
            if math.isfinite(ts) and ts < bbs - 1e-12:
                bh = cand.copy()
                bbs = ts
            if ts <= cs + 1e-14 or math.log(max(float(rng.random()), 1e-300)) < -(ts - cs) / T:
                curr = cand
                cs = ts
    return bh, bbs, curr, cs


def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n, dtype=np.float64)
    v[p[: n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v


def make_runs(rng, n):
    v = np.zeros(n, dtype=np.float64)
    pos = 0
    bit = 1
    rem1 = n // 2
    rem0 = n - rem1
    while pos < n:
        if bit == 1 and rem1 <= 0:
            bit = 0
        if bit == 0 and rem0 <= 0:
            bit = 1
        if bit == 1:
            maxlen = min(rem1, n - pos)
            if maxlen <= 0:
                bit = 0
                continue
            upper = min(maxlen, 6)
            ln = int(rng.integers(1, upper + 1))
            v[pos:pos + ln] = 1.0
            rem1 -= ln
            pos += ln
            bit = 0
        else:
            maxlen = min(rem0, n - pos)
            if maxlen <= 0:
                bit = 1
                continue
            upper = min(maxlen, 6)
            ln = int(rng.integers(1, upper + 1))
            v[pos:pos + ln] = 0.0
            rem0 -= ln
            pos += ln
            bit = 1
    return v


def try_add_pool(pool, n, v):
    if v is None:
        return
    h = project_capped(v, n)
    if h is None or not is_feasible(h, n):
        return
    for old in pool:
        if np.max(np.abs(h - old)) < 1e-8:
            return
    pool.append(h.copy())


def local_polish(h, n, rng, use_slsqp):
    h, s = subgrad_polish(h, n, rng, 12)
    if use_slsqp and sp_minimize is not None:
        h, s = slsqp_opt(h, n)
    h, s = pair_polish(h, n, rng, 6, 80, 15)
    h, s = subgrad_polish(h, n, rng, 8)
    h, s = pair_polish(h, n, rng, 4, 80, 19)
    h, s = subgrad_polish(h, n, rng, 8)
    h, s = pair_polish(h, n, rng, 4, 80, 19)
    return h, s


def run(seed=42, **kwargs):
    global G_N
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get("n_points", 47))
        except Exception:
            n = 47
        if n < 1:
            n = 1
        if n > 999:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1

    G_N = n
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5, dtype=np.float64)
    if fb is None or not is_feasible(fb, n):
        fb = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if fb is None:
        fb = np.full(n, 0.5, dtype=np.float64)

    pool = []
    try_add_pool(pool, n, fb)
    for sh in (0, n // 4, n // 3, n // 2, -n // 3, -n // 4):
        try_add_pool(pool, n, np.roll(fb, sh))
    try_add_pool(pool, n, 1.0 - fb)
    try_add_pool(pool, n, fb[::-1])
    try_add_pool(pool, n, (1.0 - fb)[::-1])
    try_add_pool(pool, n, np.full(n, 0.5, dtype=np.float64))

    blk = np.zeros(n, dtype=np.float64)
    blk[: n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add_pool(pool, n, blk)

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    try_add_pool(pool, n, alt)

    for _ in range(10):
        try_add_pool(pool, n, make_binary(rng, n))
    for _ in range(8):
        try_add_pool(pool, n, make_runs(rng, n))
    while len(pool) < 6:
        try_add_pool(pool, n, project_capped(make_binary(rng, n), n))

    scored = []
    for pv in pool:
        sv = exact_score(pv, n)
        if math.isfinite(sv):
            scored.append((sv, pv))
    if not scored:
        scored.append((exact_score(fb, n), fb))
    scored.sort(key=lambda x: x[0] if math.isfinite(x[0]) else 1e9)

    max_starts = 5 if n <= 100 else 3
    sel = []
    for sv, pv in scored:
        if len(sel) >= max_starts:
            break
        ok = True
        for q in sel:
            if np.max(np.abs(pv - q)) <= 0.04:
                ok = False
                break
        if ok:
            sel.append(pv.copy())
    for sv, pv in scored:
        if len(sel) >= max_starts:
            break
        ok = True
        for q in sel:
            if np.max(np.abs(pv - q)) < 1e-12:
                ok = False
                break
        if ok:
            sel.append(pv.copy())
    if not sel:
        sel = [fb.copy()]

    gh = fb.copy()
    gs = exact_score(gh, n)
    anneal_steps = 300 if n <= 80 else 180

    for li, h0 in enumerate(sel):
        h0 = project_capped(h0, n)
        if h0 is None:
            continue
        h, s = local_polish(h0, n, rng, li < 3)
        if s < gs - 1e-12:
            gh, gs = h.copy(), s
        h, s, _, _ = anneal_pair(h, n, rng, anneal_steps, h, s)
        if s < gs - 1e-12:
            gh, gs = h.copy(), s
        h, s = pair_polish(h, n, rng, 6, 90, 19)
        h, s = subgrad_polish(h, n, rng, 8)
        if s < gs - 1e-12:
            gh, gs = h.copy(), s

    for _ in range(3):
        h0 = make_binary(rng, n) if rng.random() < 0.5 else make_runs(rng, n)
        h0 = project_capped(h0, n)
        if h0 is None:
            continue
        h, s = local_polish(h0, n, rng, False)
        h, s, _, _ = anneal_pair(h, n, rng, max(60, anneal_steps // 3), h, s)
        h, s = pair_polish(h, n, rng, 5, 80, 17)
        if s < gs - 1e-12:
            gh, gs = h.copy(), s

    gh = project_capped(gh, n)
    if gh is None or not is_feasible(gh, n):
        gh = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if gh is None:
        gh = np.full(n, 0.5, dtype=np.float64)
    gs = exact_score(gh, n)
    if not math.isfinite(gs):
        gh = np.full(n, 0.5, dtype=np.float64)
        gs = exact_score(gh, n)
    return gh.astype(np.float64), float(gs), int(n)
```
