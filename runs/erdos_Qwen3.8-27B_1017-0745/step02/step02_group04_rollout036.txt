        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        ns, mi, nh = 12, 300, 12
    elif n <= 128:
        ns, mi, nh = 8, 200, 6
    elif n <= 256:
        ns, mi, nh = 5, 120, 3
    else:
        ns, mi, nh = 3, 60, 0
    starts = build_starts(fallback, n, rng, ns)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))
    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    # Phase 1: SLSQP from each start
    for st in starts:
        y = solve_slsqp(st, n, mi)
        if y is not None:
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
    # Phase 2: Basin hopping with Metropolis
    cur = gb.copy()
    cur_s = gs
    sigmas = [0.03, 0.08, 0.15, 0.25]
    T0 = 2.0e-3
    decay = 0.78
    for r in range(nh):
        if r % 5 == 4:
            prop = project_capped(rng.uniform(0, 1, n), n)
        elif r % 5 == 3:
            prop = project_capped(cur[::-1].copy() + sigmas[r % 4] * rng.standard_normal(n), n)
        else:
            prop = project_capped(cur + sigmas[r % 4] * rng.standard_normal(n), n)
        if prop is None:
            continue
        y = solve_slsqp(prop, n, mi)
        if y is None:
            continue
        sy = float(exact_score(y, n))
        if sy < gs - 1e-13:
            gs, gb = sy, y.copy()
        T = max(T0 * (decay ** r), 1e-15)
        if sy <= cur_s + 1e-14:
            cur, cur_s = y.copy(), sy
        elif rng.random() < np.exp(-(sy - cur_s) / T):
            cur, cur_s = y.copy(), sy
    # Phase 3: Polish best with more iterations
    y2 = solve_slsqp(gb, n, mi * 2)
    if y2 is not None:
        s2 = float(exact_score(y2, n))
        if s2 < gs - 1e-13:
            gs, gb = s2, y2.copy()
    # Final validation
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb_s = float(exact_score(fallback, n))
    if fs > fb_s + 1e-12:
        gb, fs = fallback.copy(), fb_s
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return np.ascontiguousarray(gb, dtype=np.float64), fs, int(n)
