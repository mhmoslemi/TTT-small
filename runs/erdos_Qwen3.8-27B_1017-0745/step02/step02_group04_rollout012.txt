                if i == j or best[j] >= 1.0 - 1e-6:

                for delta in (0.02, 0.06, 0.15, 0.4):
                    amt = min(delta, best[i], 1.0 - best[j])
                    if amt < 1e-8:
                        continue
                    z = best.copy()
                    z[i] -= amt
                    z[j] += amt
                    if is_feasible(z, n):
                        sc = exact_score(z, n)
                        if sc < best_s - 1e-12:
                            best, best_s = z.copy(), sc
                            improved = True
        if not improved:
            break
    return best, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fs = exact_score(fallback, n)
    if n == 1:
        return fallback, fs, n
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    sdp_u, sdp_cov = solve_sdp(n)
    starts = build_starts(fallback, n, rng, sdp_u, sdp_cov)
    gb, gs = fallback.copy(), fs
    for st in starts:
        sc = exact_score(st, n)
        if sc < gs - 1e-12:
            gs, gb = sc, st.copy()
    centers = [gb]
    for st in starts:
        if st is gb:
            continue
        if np.min([np.max(np.abs(st - c)) for c in centers]) > 1e-3 and len(centers) < 4:
            centers.append(st.copy())
    sigmas = [0.04, 0.09, 0.18, 0.07]
    lam, n_gen = 20, 50
    for ci, ctr in enumerate(centers):
        try:
            sig = sigmas[ci % len(sigmas)]
            res, res_s = evolve(ctr, n, sdp_cov, sig, lam, n_gen, rng)
            if res_s < gs - 1e-12:
                gs, gb = res_s, res.copy()
        except Exception:
            pass
    try:
        pol, pol_s = localize(gb, n, 40)
        if pol_s < gs - 1e-12:
            gs, gb = pol_s, pol.copy()
    except Exception:
        pass
    for ci in range(1, min(len(starts), 6)):
        try:
            pol2, pol2_s = localize(starts[ci], n, 20)
            if pol2_s < gs - 1e-12:
                gs, gb = pol2_s, pol2.copy()
        except Exception:
            pass
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs_final = float(exact_score(gb, n))
    if fs_final > fs + 1e-12:
        gb, fs_final = fallback.copy(), fs
    if not np.isfinite(fs_final):
        gb = np.full(n, 0.5)
        fs_final = float(exact_score(gb, n))
    return gb, fs_final, int(n)
