def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-9 or not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-8

    if not is_feasible(h, n):
        return float('inf')
    L = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if L.shape[0] != 2 * n - 1 or not np.all(np.isfinite(L)):
        return float('inf')
    return float(np.max(L))

def smooth_value_grad(h, n, tau):

    L = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(L)):
        return None, None
    a = np.max(L)

        grad[i0:i1] -= ws * h[j0:j1]
    if not np.all(np.isfinite(grad)):
        return None, None
    return val, grad

        return x, best_h, best_s

        if y is None:

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-9 or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-8

    if not is_feasible(h, n):
        return float('inf')

    if L.shape[0] != 2 * n - 1 or not np.all(np.isfinite(L)):
        return float('inf')
    return float(np.max(L))

def smooth_value_grad(h, n, tau):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(L)):
        return None, None
    a = np.max(L)
    e = np.exp(np.maximum((L - a) / tau, -745))
    se = np.sum(e)
    if se <= 0.0:
        return None, None
    w = e / se
    val = a + tau * (np.log(se) - np.log(2 * n - 1))
    grad = np.zeros(n, dtype=np.float64)
    for p in range(2 * n - 1):
        m = p - (n - 1)
        ws = dx * w[p]
        if ws < 1e-18:
            continue
        i0 = max(0, -m)
        i1 = min(n, n - m)
        j0 = i0 + m
        j1 = j0 + (i1 - i0)
        grad[j0:j1] += ws * (1.0 - h[i0:i1])
        grad[i0:i1] -= ws * h[j0:j1]
    if not np.all(np.isfinite(grad)):
        return None, None
    return val, grad
