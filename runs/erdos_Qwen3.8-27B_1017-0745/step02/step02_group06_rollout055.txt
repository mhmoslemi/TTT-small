def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)

            h[idx] += d
            r -= d

            h[idx] -= d
            r += d

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-9 or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        j_lo = max(0, -k)
        j_hi = min(n - 1, n - 1 - k)
        if j_lo > j_hi:
            continue
        jv = np.arange(j_lo, j_hi + 1)
        i_val = jv + k
        np.add.at(J[k + n - 1], i_val, dx * (1.0 - h[jv]))
        np.add.at(J[k + n - 1], jv, -dx * h[i_val])
    return J

    return x[-1]

def obj_grad(x):
    g = np.zeros(len(x))

    h = x[:n]

    J[:, :n] = -Lk

    return J

    t0 = max(0.0, t0)
