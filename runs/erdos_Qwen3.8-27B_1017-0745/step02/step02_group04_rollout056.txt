    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n - 1), n):
        a, b = max(d, 0), max(-d, 0)
        m = n - abs(d)
        q[d + n - 1] = dx * np.dot(h[a:a+m], 1.0 - h[b:b+m])
        J[d + n - 1, a:a+m] += dx * (1.0 - h[b:b+m])
        J[d + n - 1, b:b+m] -= dx * h[a:a+m]
    return q, J

def _obj(z):
    return z[-1]

def _obj_jac(z):
    g = np.zeros(z.shape)
    g[-1] = 1.0
    return g

def _ineq(z):
    n = _ctx['n']
    q, _ = lag_jac(z[:n], n)
    return z[-1] - q

def _ineq_jac(z):
    n = _ctx['n']
    L = 2 * n - 1
    _, J = lag_jac(z[:n], n)
    out = np.zeros((L, n + 1))
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

def _eq(z):
    n = _ctx['n']
    return np.array([np.sum(z[:n]) - n / 2.0])

def _eq_jac(z):
    n = _ctx['n']
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def solve_epigraph(x, n, maxiter):
    _ctx['n'] = n
    q0, _ = lag_jac(x, n)
    t0 = max(float(np.max(q0)) + 1e-6, 1e-6)
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constr = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
    except Exception:
        return None
    if res.x is None or len(res.x) != n + 1:
        return None
    y = res.x[:n]
    if not np.all(np.isfinite(y)):
        return None
    return project_capped(y, n)

def _try_add(pool, v, n):
    if v is None:
        return
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    p = project_capped(v, n)
    if p is not None:
        pool.append(p.copy())

def build_starts(fallback, n, rng):
    pool = []
    _try_add(pool, fallback, n)
    _try_add(pool, fallback[::-1].copy(), n)
    _try_add(pool, 1.0 - fallback, n)
    _try_add(pool, (1.0 - fallback)[::-1].copy(), n)
    _try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, blk, n)
    _try_add(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
    _try_add(pool, fallback + rng.normal(0, 0.05, n), n)
    _try_add(pool, fallback + rng.normal(0, 0.12, n), n)
    _try_add(pool, fallback + rng.normal(0, 0.22, n), n)
    _try_add(pool, rng.uniform(0, 1, n), n)
    P = max(2, n // 3)
    _try_add(pool, 0.5 + 0.4 * np.cos(2.0 * np.pi * np.arange(n) / P), n)
    P2 = max(2, n // 4)
    _try_add(pool, 0.5 + 0.35 * np.cos(2.0 * np.pi * np.arange(n) / P2 + 1.0), n)
    _try_add(pool, fallback + 0.1 * np.sin(2.0 * np.pi * np.arange(n) * 3.0 / n), n)
    return pool

def basin_hop(cur, cur_sc, n, rng, n_hops, maxiter):
    best = cur.copy()
    best_sc = cur_sc
    sigmas = [0.03, 0.06, 0.10, 0.15, 0.22]
    for i in range(n_hops):
        if i % 7 == 6:
            prop = rng.uniform(0, 1, n)
        else:
            sigma = sigmas[i % len(sigmas)]
            prop = cur + sigma * rng.standard_normal(n)
        p = project_capped(prop, n)
        if p is None:
            continue
        y = solve_epigraph(p, n, maxiter)
        if y is None or not is_feasible(y, n):
            y = p
        sc = float(exact_score(y, n))
        if sc == float('inf'):
            continue
        if sc < best_sc - 1e-12:
            best, best_sc = y.copy(), sc
        T = 1.5e-3 * (0.75 ** i)
        if sc <= cur_sc or rng.random() < np.exp(-(sc - cur_sc) / max(T, 1e-10)):
            cur, cur_sc = y.copy(), sc
    return best, best_sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    rng = np.random.default_rng(int(seed))
    starts = build_starts(fallback, n, rng)
    gs = float(exact_score(fallback, n))
    gb = fallback.copy()
    maxiter = 250
    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
        try:
            y = solve_epigraph(st, n, maxiter)
            if y is not None and is_feasible(y, n):
                sy = float(exact_score(y, n))
                if sy < gs - 1e-12:
                    gs, gb = sy, y.copy()
        except Exception:
            pass
    # Second pass: re-solve from best and perturbed versions
    for st in [gb, gb + rng.normal(0, 0.03, n), gb + rng.normal(0, 0.08, n)]:
        p = project_capped(st, n)
        if p is None:
            continue
        try:
            y = solve_epigraph(p, n, 300)
            if y is not None and is_feasible(y, n):
                sy = float(exact_score(y, n))
                if sy < gs - 1e-12:
                    gs, gb = sy, y.copy()
        except Exception:
            pass
    # Basin hopping
    try:
        hbest, hscore = basin_hop(gb, gs, n, rng, 14, 200)
        if hscore < gs - 1e-12:
            gs, gb = hscore, hbest.copy()
    except Exception:
        pass
    # Final validation
    gb2 = project_capped(gb, n)
    if gb2 is not None and is_feasible(gb2, n):
        gb = gb2
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12 or fs == float('inf'):
        gb, fs = fallback.copy(), fb
        if fs == float('inf'):
            gb = np.full(n, 0.5)
            fs = float(exact_score(gb, n))
    return gb, fs, int(n)
