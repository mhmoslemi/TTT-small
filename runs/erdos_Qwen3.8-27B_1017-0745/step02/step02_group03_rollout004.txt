The LP formulation:

- Objective: minimize t (c[0:n]=0, c[n]=1)
- Inequalities: F_d + G[d,:]·δ ≤ t, i.e., G[d,:]·δ - t ≤ -F_d

- Bounds: max(-r, -h[i]) ≤ δ[i] ≤ min(r, 1-h[i]), t is unbounded

            m_lin = float(np.max(F + G @ res.x[:n]))
            p = float(F.max()) - m_lin
            rho = (cur_sc - trial_sc) / ((2.0 / n) * p) if p > 1e-14 else 0.0
            if rho >= 0.75 and step_sz >= 0.8 * r:
                r = min(r_max, 1.4 * r)
            elif rho < 0.25:
                r = max(r_min, 0.5 * r)
        else:
            r = max(r_min, 0.5 * r)
    return gbest, gscore

def add_start(starts, v, n, tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.abs(h2 - ex).max()) < tol:
            return
    starts.append(h2)

def pairwise_polish(h, n, sweeps):
    best = h.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = set()
                for sgn in (1, -1):
                    for st in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                        dd = sgn * st
                        if lo <= dd <= hi:
                            cands.add(round(dd, 12))
                cands.add(max(lo, 0.0))
                cands.add(min(hi, 0.0))
                bd, bs = 0.0, best_s
                for dd in cands:
                    t = best.copy()
                    t[i] += dd
                    t[j] -= dd
                    s = exact_score(t, n)
                    if s < bs - 1e-13:
                        bd, bs = dd, s
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    best_s = bs
                    imp = True
        if not imp:
            break
    return best, best_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = abs(int(seed))
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    h0 = project_capped(parent, n) if parent is not None else None
    fallback = h0 if h0 is not None else np.full(n, 0.5)
    fb_sc = exact_score(fallback, n)
    if n <= 64:
        ns_def, mi_def, ps_def = 7, 30, 1
    elif n <= 128:
        ns_def, mi_def, ps_def = 5, 20, 1
    elif n <= 256:
        ns_def, mi_def, ps_def = 2, 8, 0
    else:
        ns_def, mi_def, ps_def = 2, 0, 0
    try:
        n_starts = max(1, min(int(kwargs.get('starts', ns_def)), 8))
    except Exception:
        n_starts = ns_def
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mi_def)), 60))
    except Exception:
        max_iter = mi_def
    try:
        psweeps = max(0, min(int(kwargs.get('polish', ps_def)), 3))
    except Exception:
        psweeps = ps_def
    if n > 256:
        max_iter = 0
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 3), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.08, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best, best_sc = fallback.copy(), fb_sc
    for st in starts:
        s = exact_score(st, n)
        if np.isfinite(s) and s < best_sc - 1e-12:
            best, best_sc = st.copy(), s
    if max_iter > 0:
        for st in starts:
            try:
                best, best_sc = tr_lp_search(st, n, max_iter, best, best_sc)
            except Exception:
                pass
        if n <= 128:
            for sds in ([0.02, 0.06] if n <= 64 else [0.04]):
                try:
                    pert = project_capped(best + rng.normal(0, sds, n), n)
                    if pert is not None:
                        best, best_sc = tr_lp_search(pert, n, max_iter, best, best_sc)
                except Exception:
                    pass
    if psweeps > 0 and n <= 64:
        try:
            ph, ps = pairwise_polish(best, n, psweeps)
            fp = project_capped(ph, n)
            if fp is not None:
                fs = exact_score(fp, n)
                if np.isfinite(fs) and fs < best_sc - 1e-12:
                    best, best_sc = fp, fs
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_sc:
        final, fs = fallback.copy(), fb_sc
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
