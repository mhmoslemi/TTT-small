        Q[2 * r] = dx * np.dot(h[r:], 1.0 - h[:m])
    return Q

def compute_lag_grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    G[0, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[2 * r - 1, :m] += dx * (1.0 - h[r:])
        G[2 * r - 1, r:] -= dx * h[:m]
        G[2 * r, :m] -= dx * h[r:]
        G[2 * r, r:] += dx * (1.0 - h[:m])
    return G

def _obj(x, n):
    return x[n]

def _obj_jac(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _ineq(x, n):
    return x[n] - compute_lags(x[:n], n)

def _ineq_jac(x, n):
    J = np.empty((2 * n - 1, n + 1))
    J[:, :n] = -compute_lag_grads(x[:n], n)
    J[:, n] = 1.0
    return J

def _eq(x, n):
    return np.sum(x[:n]) - n / 2.0

def _eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def slsqp_solve(h0, n):
    t0 = exact_score(h0, n) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'fun': _ineq, 'jac': _ineq_jac, 'args': (n,), 'type': 'ineq'},
        {'fun': _eq, 'jac': _eq_jac, 'args': (n,), 'type': 'eq'},
    ]
    maxit = 250 if n <= 64 else 150
    try:
        res = minimize(_obj, z0, jac=_obj_jac, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    hp = project_capped(res.x[:n], n)
    if hp is None or not np.all(np.isfinite(hp)):
        return None
    return hp

def smooth_refine(h0, n, niter_max=15):
    h = h0.copy()
    best_h, best_s = h.copy(), exact_score(h, n)
    taus = [0.02, 0.005, 0.001, 0.0002]
    for tau in taus:
        alpha = 1.0
        for it in range(niter_max):
            Q = compute_lags(h, n)
            qmax = np.max(Q)
            shifted = (Q - qmax) / tau
            exps = np.exp(shifted)
            p = exps / np.sum(exps)
            g = np.zeros(n)
            g += p[0] * (2.0 / n) * (1.0 - 2.0 * h)
            for r in range(1, n):
                m = n - r
                dx_r = (2.0 / n)
                g[:m] += dx_r * p[2*r-1] * (1.0 - h[r:])
                g[r:] -= dx_r * p[2*r-1] * h[:m]
                g[:m] -= dx_r * p[2*r] * h[r:]
                g[r:] += dx_r * p[2*r] * (1.0 - h[:m])
            if not np.all(np.isfinite(g)):
                break
            y = project_capped(h - alpha * g, n)
            if y is None:
                break
            d = y - h
            if np.max(np.abs(d)) < 1e-10:
                break
            F_y = qmax + tau * np.log(np.sum(exps)) if False else None
            Qy = compute_lags(y, n)
            qmaxy = np.max(Qy)
            Fy = qmaxy + tau * np.log(np.sum(np.exp((Qy - qmaxy) / tau)))
            Fh = qmax + tau * np.log(np.sum(exps))
            sufficient = Fy <= Fh + 1e-4 * np.dot(g, d) + 1e-14
            if not sufficient:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
                continue
            h = y
            alpha = min(1.5 * alpha, 10.0)
            s = exact_score(h, n)
            if s < best_s:
                best_s, best_h = s, h.copy()
    return best_h, best_s

def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)
    v[p[:n // 2]] = 1.0
    if n % 2 == 1:
        v[p[n // 2]] = 0.5
    return v

def try_add(pool, v, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def build_starts(fb, n, rng, max_starts):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    try_add(pool, project_capped(make_binary(rng, n), n))
    for sd in (0.06, 0.15, 0.30):
        try_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    for mode in (1, 2, 3):
        for ph in (0.0, math.pi / 2.0, math.pi):
            t = np.arange(n)
            sig = 0.35
            v = project_capped(0.5 + sig * np.cos(2.0 * math.pi * mode * t / n + ph), n)
            try_add(pool, v)
    try_add(pool, project_capped(fb + rng.normal(0, 0.4, n), n))
    try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    while len(pool) < max_starts:
        base = pool[int(rng.integers(len(pool)))]
        try_add(pool, project_capped(base + rng.normal(0, 0.1, n), n))
    return pool[:max_starts]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_s = exact_score(fb, n)
    best_h, best_s = fb.copy(), fb_s
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), exact_score(np.array([0.5]), 1), 1
    max_starts = 10 if n <= 64 else (7 if n <= 128 else 4)
    starts = build_starts(fb, n, rng, max_starts)
    starts_s = [exact_score(sv, n) for sv in starts]
    for si in range(len(starts)):
        if starts_s[si] < best_s - 1e-12:
            best_s, best_h = starts_s[si], starts[si].copy()
    if n <= 128:
        for si in range(len(starts)):
            cand = slsqp_solve(starts[si], n)
            if cand is not None:
                s = exact_score(cand, n)
                if s < best_s - 1e-12:
                    best_s, best_h = s, cand.copy()
            for a in (0.25, 0.5, 0.75):
                mix = project_capped((1.0 - a) * starts[si] + a * cand, n) if cand is not None else None
                if mix is not None:
                    ms = exact_score(mix, n)
                    if ms < best_s - 1e-12:
                        best_s, best_h = ms, mix.copy()
        for si in range(min(3, len(starts))):
            idx = int(np.argmin(starts_s))
            cand, cs = smooth_refine(starts[idx], n, 20)
            if cs < best_s - 1e-12:
                best_s, best_h = cs, cand.copy()
        cand, cs = smooth_refine(best_h, n, 25)
        if cs < best_s - 1e-12:
            best_s, best_h = cs, cand.copy()
    else:
        for si in range(len(starts)):
            cand, cs = smooth_refine(starts[si], n, 25)
            if cs < best_s - 1e-12:
                best_s, best_h = cs, cand.copy()
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
        best_s = fb_s
    else:
        fs = exact_score(final, n)
        if fs < best_s:
            best_s = fs
        if fs > fb_s:
            final = fb.copy()
            best_s = fb_s
    final = np.clip(final, 0.0, 1.0)
    final = project_capped(final, n)
    if final is None:
        final = fb.copy()
        best_s = fb_s
    final_score = exact_score(final, n)
    if not np.isfinite(final_score):
        final = fb.copy()
        final_score = exact_score(final, n)
    return final.astype(np.float64), float(final_score), int(n)
