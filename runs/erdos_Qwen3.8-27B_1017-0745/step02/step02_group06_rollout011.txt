        pool.append(p)
    add(np.full(n, 0.5))
    if parent is not None:
        add(parent)
        add(parent[::-1])
        add(1.0 - parent[::-1])
        for sd in [0.04, 0.10, 0.20, 0.35]:
            add(parent + rng.normal(0, sd, n))
    arr = np.arange(n, dtype=np.float64)
    for m in [1, 2, 3]:
        for ph in [0.0, math.pi / 2]:
            add(0.5 + 0.45 * np.cos(2.0 * math.pi * m * arr / n + ph))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(blk)
    add(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2:
        bb[perm[n // 2]] = 0.5
    add(bb)
    add(rng.uniform(0, 1, n))
    return pool

def mm_trajectory(x0, n, rho, solvers, rng, max_iter):
    x = x0.copy()
    sc = exact_score(x, n)
    for it in range(max_iter):
        L, G = lag_data(x, n)
        if L is None:
            break
        ys = solve_mm_qp(x, L, G, n, rho, solvers)
        if ys is None:
            break
        best_x, best_s = x, sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial = project_capped(x + alpha * (ys - x), n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-12:
                best_s, best_x = ts, trial
        if best_s >= sc - 1e-12:
            break
        x = best_x
        sc = best_s
        if np.max(np.abs(ys - x)) < 1e-9:
            break
    return x, sc

def pair_polish(h, n, rng, n_moves):
    h = h.copy()
    sc = exact_score(h, n)
    dx = 2.0 / n
    for _ in range(n_moves):
        D = np.correlate(h, 1.0 - h, mode='full') * dx
        reliable = D * n
        top_k = min(8, 2 * n - 1)
        top_idx = np.argpartition(reliable, -top_k)[-top_k:]
        best_s = sc
        best_h = h
        ii, jj = np.triu_indices(n, 1)
        rng.shuffle(ii)
        npairs = min(len(ii), 200)
        for pi in range(npairs):
            i, j = int(ii[pi]), int(jj[pi])
            Lp = max(-h[i], h[j] - 1.0)
            Up = min(1.0 - h[i], h[j])
            if Up - Lp < 1e-12:
                continue
            t = 0.5 * (Lp + Up)
            ht = h.copy()
            ht[i] += t; ht[j] -= t
            ht = project_capped(ht, n)
            if ht is None or not is_feasible(ht, n):
                continue
            for tt in [t, 0.0, 0.25 * (Up - Lp), 0.75 * (Up - Lp)]:
                if tt < Lp - 1e-12 or tt > Up + 1e-12:
                    continue
                ht2 = h.copy()
                ht2[i] += tt; ht2[j] -= tt
                ht2 = project_capped(ht2, n)
                if ht2 is None or not is_feasible(ht2, n):
                    continue
                ts = exact_score(ht2, n)
                if ts < best_s - 1e-12:
                    best_s, best_h = ts, ht2.copy()
        if best_h is h or best_s >= sc - 1e-13:
            break
        h = best_h
        sc = best_s
    return h, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            np_ = int(kwargs.get('n_points', 47))
            if 2 <= np_ <= 999 and not isinstance(kwargs.get('n_points'), bool):
                n = np_
        except Exception:
            n = 47
    if n == 1:
        hv = np.array([0.5])
        return hv, exact_score(hv, 1), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    rho = (2.0 / n) * math.cos(math.pi / (n + 1))
    if rho < 1e-6:
        rho = 2.0 / n
    solvers = ['ECOS', 'CVXOPT', 'SCS', 'MOSEK']
    if n <= 64:
        n_starts, max_it = 6, 25
    elif n <= 128:
        n_starts, max_it = 4, 15
    elif n <= 256:
        n_starts, max_it = 3, 10
    else:
        n_starts, max_it = 2, 6
    starts = build_starts(parent, n, rng)
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and all(np.max(np.abs(pp - s)) >= 1e-7 for s in starts):
            starts.insert(0, pp)
    scored = sorted([(exact_score(s, n), s) for s in starts[:n_starts]])
    gb = fallback.copy()
    gs = exact_score(fallback, n)
    for s, st in scored:
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for s, st in scored:
        xo, so = mm_trajectory(st, n, rho, solvers, rng, max_it)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()
    hb, hs = pair_polish(gb, n, rng, 8)
    if hs < gs - 1e-12:
        gs, gb = hs, hb
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb = exact_score(fallback, n)
    if fs > min(fb, gs) + 1e-10:
        final = fallback
        fs = fb
    return final, float(fs), int(n)
