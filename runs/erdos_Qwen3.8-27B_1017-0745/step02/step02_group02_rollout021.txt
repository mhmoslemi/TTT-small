def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1]]
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n)
    wave = np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n), 0, 1)
    cands.append(wave)
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'maxiter': 300})
    if not res.success or res.x is None or len(res.x) != n + 1:
        return None, None, None
    if not np.all(np.isfinite(res.x)):
        return None, None, None
    d = res.x[:n]
    if abs(np.sum(d)) > 1e-7 * max(1, n):
        return None, None, None
    return d, Q, G

def local_search(h0, n, max_iter, Delta0):
    h = h0.copy().astype(np.float64)
    f = exact_score(h, n)
    best_h = h.copy()
    best_f = f
    Delta = Delta0
    lp_fails = 0
    rej_fails = 0
    for _ in range(max_iter):
        if Delta < 1e-6:
            break
        q = lag_values(h, n)
        f = float(np.max(q))
        active = build_active_set(q, n, Delta)
        d, Q, G = solve_lp(h, n, active, q, Delta)
        if d is None:
            Delta *= 0.5
            lp_fails += 1
            if lp_fails >= 5:
                break
            continue
        lp_fails = 0
        if float(np.max(np.abs(d))) <= 1e-10:
            Delta *= 0.5
            continue
        pred = f - float(np.max(Q + G @ d))
        if pred <= 1e-12:
            Delta *= 0.5
            if Delta < 1e-6:
                break
            continue
        best_trial = None
        best_trial_f = f
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            trial = project_capped(h + alpha * d, n)
            if trial is None:
                continue
            sf = exact_score(trial, n)
            if sf < best_trial_f - 1e-12:
                best_trial_f = sf
                best_trial = trial
        if best_trial is None:
            Delta *= 0.5
            rej_fails += 1
            if rej_fails >= 8 or Delta < 1e-6:
                break
            continue
        rej_fails = 0
        old_h = h.copy()
        actual = f - best_trial_f
        h = best_trial
        f = best_trial_f
        if f < best_f - 1e-12:
            best_f = f
            best_h = h.copy()
        rho = actual / pred if pred > 1e-12 else 0.0
        disp = float(np.max(np.abs(h - old_h)))
        if rho < 0.1:
            Delta = max(Delta * 0.5, 1e-6)
        elif rho > 0.75 and disp >= 0.8 * Delta:
            Delta = min(1.5 * Delta, 0.5)
    return best_h, best_f

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1], np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n, dtype=np.float64)
    wave = np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n), 0.0, 1.0)
    cands.append(wave)
    cands.append(rng.uniform(0.0, 1.0, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and len(parent) == n:
        fb = project_capped(parent, n)
        if fb is None:
            fb = np.full(n, 0.5)
    elif parent is not None and len(parent) > 1:
        old_n = len(parent)
        old_x = (np.arange(old_n) + 0.5) / old_n
        new_x = (np.arange(n) + 0.5) / n
        tp = np.interp(new_x, old_x, parent)
        fb = project_capped(tp, n)
        if fb is None:
            fb = np.full(n, 0.5)
    else:
        fb = np.full(n, 0.5)
    if n == 1:
        return (fb, exact_score(fb, 1), 1)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_sc, n_iter, n_rej = 7, 80, 3
    elif n <= 128:
        n_sc, n_iter, n_rej = 4, 45, 2
    elif n <= 256:
        n_sc, n_iter, n_rej = 3, 25, 1
    elif n <= 512:
        n_sc, n_iter, n_rej = 2, 12, 1
    else:
        n_sc, n_iter, n_rej = 2, 6, 0
    starts = build_starts(fb, n, rng, n_sc)
    gs = float('inf')
    gb = fb.copy()
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    scored.sort(key=lambda x: x[0])
    for _, st in scored:
        try:
            bh, bf = local_search(st, n, n_iter, 0.25)
            if bf < gs - 1e-12:
                gs, gb = bf, bh.copy()
        except Exception:
            continue
    fb_score = exact_score(fb, n)
    if n_rej > 0:
        if n <= 64:
            stds = [0.04, 0.10, 0.18] if gs < fb_score - 1e-5 else [0.12, 0.22, 0.08]
        elif n <= 256:
            stds = [0.04, 0.08] if gs < fb_score - 1e-5 else [0.10, 0.18]
        else:
            stds = [0.08]
        for std in stds:
            try:
                pert = project_capped(gb + rng.normal(0.0, std, n), n)
                if pert is None:
                    continue
                bh, bf = local_search(pert, n, n_iter, 0.25)
                if bf < gs - 1e-12:
                    gs, gb = bf, bh.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return (final, float(sf), int(n))
