    cur_s = exact_score(h, n)

        return h, cur_s

            for i in range(n):
                a = min(-r, h[i])
                h[i] -= a
                r += a
                if abs(r) < 1e-14:
                    break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    rmax = float(np.max(r))
    ex = np.exp((r - rmax) / tau)
    W = float(np.sum(ex))
    if W < 1e-300 or not np.isfinite(W):
        return None, None
    w = ex / W
    fval = rmax + tau * np.log(W)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-15:
            continue
        p, q, m = max(d, 0), max(-d, 0), n - abs(d)
        grad[p:p + m] += wd * dx * (1.0 - h[q:q + m])
        grad[q:q + m] -= wd * dx * h[p:p + m]
    if not np.all(np.isfinite(grad)):
        return None, None
    return float(fval), grad

def smooth_optimize(start, n, iters_pt):
    x = start.copy()
    best_h, best_s = start.copy(), exact_score(start, n)
    for tau in [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]:
        alpha = min(1.0, 1.0 / (8.0 / (n * tau) + 4.0 / n + 1e-12))
        for _ in range(iters_pt):
            fv, g = smooth_grad(x, n, tau)
            if fv is None:
                break
            trial = project_capped(x - alpha * g, n)
            if trial is None:
                break
            ok = False
            for _ in range(25):
                v = trial - x
                if np.max(np.abs(v)) < 1e-13:
                    break
                fvt, _ = smooth_grad(trial, n, tau)
                if fvt is not None and fvt < fv + 1e-14:
                    x = trial
                    fs = exact_score(x, n)
                    if fs < best_s - 1e-12:
                        best_s, best_h = fs, x.copy()
                    alpha = min(1.0, 1.5 * alpha)
                    ok = True
                    break
                alpha *= 0.5
                trial = project_capped(x - alpha * g, n)
                if trial is None:
                    break
            if not ok:
                break
    return best_h, best_s

def _try_add(pool, v, n):
    if v is None or v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    if not is_feasible(v, n):
        v = project_capped(v, n)
        if v is None:
            return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fb, n, rng, cnt):
    pool = []
    _try_add(pool, fb, n)
    _try_add(pool, fb[::-1].copy(), n)
    _try_add(pool, 1.0 - fb, n)
    _try_add(pool, (1.0 - fb)[::-1].copy(), n)
    _try_add(pool, fb + rng.normal(0, 0.04, n), n)
    _try_add(pool, fb + rng.normal(0, 0.15, n), n)
    P = max(2, n // 3)
    _try_add(pool, 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, blk, n)
    _try_add(pool, rng.uniform(0, 1, n), n)
    return pool[:cnt]

def pairwise_refine(h0, n, rng, n_passes):
    h = h0.copy()
    cur_s = exact_score(h, n)
    if cur_s == float('inf'):
        return h, cur_s
    if n <= 64:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    else:
        budget = max(16, min(128, 784 // n))
        pairs = [(i, (i + 1) % n) for i in range(budget // 2)]
        seen = set(pairs)
        while len(pairs) < budget:
            a, b = int(rng.integers(0, n)), int(rng.integers(0, n))
            if a == b:
                continue
            if a > b:
                a, b = b, a
            if (a, b) in seen:
                continue
            seen.add((a, b))
            pairs.append((a, b))
    if n <= 64:
        meshes = [0.04, 0.01, 0.0025, 0.0006, 0.00015]
    elif n <= 256:
        meshes = [0.04, 0.01, 0.002, 0.0004]
    else:
        meshes = [0.04, 0.008, 0.0016]
    for _ in range(n_passes):
        imp = False
        for ms in meshes:
            bs, bp, bdir, bcap = cur_s, None, 0, 0.0
            for (i, j) in pairs:
                c1 = min(ms, 1.0 - h[i], h[j])
                if c1 > 1e-12:
                    h[i] += c1
                    h[j] -= c1
                    s = exact_score(h, n)
                    h[i] -= c1
                    h[j] += c1
                    if s < bs - 1e-12:
                        bs, bp, bdir, bcap = s, (i, j), 1, c1
                c2 = min(ms, h[i], 1.0 - h[j])
                if c2 > 1e-12:
                    h[i] -= c2
                    h[j] += c2
                    s = exact_score(h, n)
                    h[i] += c2
                    h[j] -= c2
                    if s < bs - 1e-12:
                        bs, bp, bdir, bcap = s, (i, j), -1, c2
            if bp is not None:
                i, j = bp
                if bdir == 1:
                    h[i] += bcap
                    h[j] -= bcap
                else:
                    h[i] -= bcap
                    h[j] += bcap
                h = np.clip(h, 0.0, 1.0)
                cur_s = bs
                imp = True
        if not imp:
            break
    return h, cur_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    if n < 1:
        n = 1
    if parent is not None and parent.shape == (n,):
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.asarray(fb, dtype=np.float64)
    if n == 1:
        return fb, float(exact_score(fb, n)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        ns, ipt = 9, 60
    elif n <= 128:
        ns, ipt = 8, 40
    elif n <= 256:
        ns, ipt = 5, 25
    else:
        ns, ipt = 3, 12
    starts = build_starts(fb, n, rng, ns)
    gb, gs = fb.copy(), float(exact_score(fb, n))
    for st in starts:
        try:
            hh, ss = smooth_optimize(st, n, ipt)
            ss = float(exact_score(hh, n))
            if ss < gs - 1e-12:
                gs, gb = ss, hh.copy()
        except Exception:
            pass
    pf, ps = pairwise_refine(fb, n, rng, 3)
    ps = float(exact_score(pf, n))
    if ps < gs - 1e-12:
        gs, gb = ps, pf.copy()
    if np.max(np.abs(gb - pf)) > 1e-10:
        pf2, ps2 = pairwise_refine(gb, n, rng, 2)
        ps2 = float(exact_score(pf2, n))
        if ps2 < gs - 1e-12:
            gs, gb = ps2, pf2.copy()
    gbp = project_capped(gb, n)
    if gbp is not None:
        fs = float(exact_score(gbp, n))
        if fs < gs - 1e-14:
            gb, gs = gbp, fs
    fb_s = float(exact_score(fb, n))
    if gs > fb_s + 1e-12 or not np.isfinite(gs):
        gb, gs = fb.copy(), fb_s
    return np.asarray(gb, dtype=np.float64), float(gs), int(n)
