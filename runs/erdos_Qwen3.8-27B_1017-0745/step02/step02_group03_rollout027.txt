        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        jdx = idx + d
        F[row] = float(np.dot(h[idx], 1.0 - h[jdx]))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return F, G

def repair_step(h, delta, r, n):
    lo_b = np.maximum(-r, -h)
    hi_b = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, lo_b, hi_b)
    resid = -float(np.sum(delta))
    for _ in range(n + 4):
        if abs(resid) < 1e-13:
            break
        if resid > 0:
            slack = hi_b - delta
            idx = int(np.argmax(slack))
            if slack[idx] <= 1e-14:
                break
            delta[idx] += min(resid, slack[idx])
            resid -= min(resid, slack[idx])
        else:
            slack = delta - lo_b
            idx = int(np.argmax(slack))
            if slack[idx] <= 1e-14:
                break
            delta[idx] += max(resid, -slack[idx])
            resid -= max(resid, -slack[idx])
    trial = h + delta
    if not np.all(np.isfinite(trial)) or np.max(np.abs(delta)) > r + 1e-8:
        return None
    proj = project_capped(trial, n)
    if proj is None or np.max(np.abs(proj - trial)) > r + 1e-7:
        return None
    return proj

def lp_search(h_start, n, max_iter, best_ref):
    h = project_capped(h_start, n)
    if h is None:
        return best_ref
    sc = exact_score(h, n)
    if np.isfinite(sc) and sc < best_ref[1] - 1e-12:
        best_ref = (h.copy(), sc)
    else:
        h, sc = best_ref[0].copy(), best_ref[1]
    r = 0.07
    r_min, r_max = 1e-5, 0.22
    for it in range(max_iter):
        if r < r_min:
            break
        F, G = lag_grads(h, n)
        if not np.all(np.isfinite(F)) or not np.all(np.isfinite(G)):
            break
        L = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -F
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        lo_b = np.maximum(-r, -h)
        hi_b = np.minimum(r, 1.0 - h)
        bounds = [(lo_b[i], hi_b[i]) for i in range(n)] + [(None, None)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True,
                                   'dual_feasibility_tolerance': 1e-9,
                                   'primal_feasibility_tolerance': 1e-9,
                                   'max_iter': 500})
        except Exception:
            r *= 0.5
            continue
        if res is None or res.status not in (0, 1, 3) or res.x is None:
            r *= 0.5
            continue
        if len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            r *= 0.5
            continue
        delta = res.x[:n]
        trial = repair_step(h, delta, r, n)
        if trial is None:
            r *= 0.5
            continue
        trial_sc = exact_score(trial, n)
        if not np.isfinite(trial_sc):
            r *= 0.5
            continue
        F_max = float(np.max(F))
        m_lin = float(np.max(F + G @ delta))
        pred_dec = F_max - m_lin
        if trial_sc < sc - 1e-13:
            h = trial
            sc = trial_sc
            if sc < best_ref[1] - 1e-12:
                best_ref = (h.copy(), sc)
            if pred_dec > 1e-14:
                rho = (sc - trial_sc) / ((2.0 / n) * pred_dec)
                if rho >= 0.7 and np.max(np.abs(delta)) >= 0.7 * r:
                    r = min(r_max, 1.4 * r)
                elif rho < 0.3:
                    r = max(r_min, 0.5 * r)
            else:
                r = max(r_min, 0.8 * r)
        else:
            r *= 0.5
        if np.max(np.abs(delta)) < 1e-9:
            break
    return best_ref

def add_start(starts, vec, n, tol=1e-8):
    h = project_capped(vec, n)
    if h is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h - ex))) < tol:
            return
    starts.append(h)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = abs(int(seed)) % (2**32)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = None
    if fb is None:
        fb = uniform.copy()
    fb_score = exact_score(fb, n)
    if not np.isfinite(fb_score):
        fb = uniform.copy()
        fb_score = exact_score(fb, n)
    best = (fb.copy(), fb_score)
    n_s_def = 7 if n <= 64 else (5 if n <= 128 else 2)
    mi_def = 35 if n <= 64 else (20 if n <= 128 else 8)
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_s_def)), 8))
    except Exception:
        n_starts = n_s_def
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mi_def)), 55))
    except Exception:
        max_iter = mi_def
    starts = [fb.copy()]
    if parent is not None:
        add_start(starts, fb[::-1].copy(), n)
        add_start(starts, 1.0 - fb, n)
        add_start(starts, np.roll(fb, max(1, n // 3)), n)
        add_start(starts, np.roll(fb, -max(1, n // 4)), n)
        add_start(starts, fb + rng.normal(0, 0.03, n), n)
        add_start(starts, fb + rng.normal(0, 0.08, n), n)
        add_start(starts, fb + rng.normal(0, 0.15, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, np.where(tt % 2 == 0, 1.0, 0.0), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best[1] - 1e-12:
            best = (st.copy(), ss)
    if n <= 256 and max_iter > 0:
        for st in starts:
            try:
                best = lp_search(st, n, max_iter, best)
            except Exception:
                pass
        # Second pass with smaller radius from best
        for _ in range(2):
            pert = best[0] + rng.normal(0, 0.012, n)
            proj = project_capped(pert, n)
            if proj is not None:
                try:
                    best = lp_search(proj, n, max(5, max_iter // 2), best)
                except Exception:
                    pass
    hops = 2 if n <= 128 else (1 if n <= 256 else 0)
    for _ in range(hops):
        for ns_val in ([0.02, 0.06] if n <= 128 else [0.04]):
            pert = best[0] + rng.normal(0, ns_val, n)
            proj = project_capped(pert, n)
            if proj is None:
                continue
            ps = exact_score(proj, n)
            if np.isfinite(ps) and ps < best[1] - 1e-12:
                best = (proj.copy(), ps)
            if n <= 256 and max_iter > 0:
                try:
                    best = lp_search(proj, n, max(5, max_iter // 2), best)
                except Exception:
                    pass
    final = project_capped(best[0], n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = fb.copy()
        fs = fb_score
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
