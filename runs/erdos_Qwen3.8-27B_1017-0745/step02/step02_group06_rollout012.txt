def lag_jacobian(h, n):
    dx = 2.0 / n
    # For lag d, derivative w.r.t. h[j]:
    # dx * ((1-h[j-d]) if 0<=j-d<n else 0) - dx * (h[j+d] if 0<=j+d<n else 0)
    # We can compute this as: for each d, the first part is a shifted (1-h) and the second is a shifted h.
    one_minus_h = 1.0 - h
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        # First term: (1-h[j-d]) valid when 0<=j-d<n -> d<=j<n+d
        j_lo1 = max(0, d)
        j_hi1 = min(n, n + d)
        if j_lo1 < j_hi1:
            J[row, j_lo1:j_hi1] += dx * (1.0 - h[j_lo1 - d: j_hi1 - d])
        # Second term: -h[j+d] valid when 0<=j+d<n -> -d<=j<n-d
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            J[row, j_lo2:j_hi2] -= dx * h[j_lo2 + d: j_hi2 + d]
    return J

    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        a1, b1 = max(0, d), min(n, n + d)
        if a1 < b1:
            J[row, a1:b1] += 1.0 - h[a1 - d:b1 - d]
        a2, b2 = max(0, -d), min(n, n - d)
        if a2 < b2:
            J[row, a2:b2] -= h[a2 + d:b2 + d]
    J *= dx
    return J
