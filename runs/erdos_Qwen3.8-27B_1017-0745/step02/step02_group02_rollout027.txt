def _sortkey_sd(sd):

def project_capped(z, n):

        return None

    s = n / 2.0

            lo = mid
        else:
            hi = mid

    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 5e-13 * n:
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
