import numpy as np
from scipy.optimize import linprog

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 5e-13 * max(1, n):
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 5e-10 * max(1, n):
        return None
    return h

def is_feas(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and
            abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def score(h, n):
    if not is_feas(h, n):
        return float('inf')
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(v)) if np.all(np.isfinite(v)) else float('inf')

def lag_grad(h, n, s):
    dx = 2.0 / n
    p, r, m = max(s, 0), max(-s, 0), n - abs(s)
    g = np.zeros(n)
    g[p:p + m] += dx * (1.0 - h[r:r + m])
    g[r:r + m] -= dx * h[p:p + m]
    return g

    pool = [fb.copy()]
    extras = [fb[::-1].copy(), 1.0 - fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    extras.append(blk)
    j = np.arange(n) + 0.5
    extras.append(np.clip(0.5 + 0.45 * np.cos(2 * np.pi * j / n), 0, 1))
    extras.append(rng.uniform(0, 1, n))
    for sd in (0.15, 0.30):
        extras.append(fb + rng.normal(0, sd, n))
    for c in extras:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s2 in pool:
            if np.max(np.abs(p - s2)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

    h = start.copy().astype(np.float64)
    f = score(h, n)
    bh, bf = h.copy(), f
    Delta = Delta0
    K = min(12, 2 * n - 1)
    dx = 2.0 / n
    lp_f, rj_f = 0, 0
    for _ in range(max_iter):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        mmax = float(np.max(q))
        gap = max(1e-10, 0.05 * Delta)
        act = np.where(q >= mmax - gap)[0]
        if len(act) > K:
            act = act[np.argsort(q[act])[::-1][:K]]
        if len(act) == 0:
            act = np.array([int(np.argmax(q))])
        G = np.array([lag_grad(h, n, int(i) - (n - 1)) for i in act])
        Q = q[act]
        sc = 1.0 / dx

        nv = n + 1

        b_ub = -Qs
        A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0

        if not ok:

            Delta *= 0.5
            if lp_f >= 5 or Delta < 1e-6:

            continue

        rj_f = 0

        actual = old_f - best_s

                Delta = min(Delta * 1.5, 0.5)
        h = best_t
        f = best_s
        if f < bf - 1e-12:
            bf, bh = f, h.copy()
