import numpy as np
import math

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-14:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    if abs(r) > 1e-10:
        if r > 0:
            free = np.clip(1.0 - h, 0.0, None)
            tf = float(np.sum(free))
            if tf > 1e-14:
                h += min(r, tf) * free / tf
        else:
            occ = np.clip(h, 0.0, None)
            to = float(np.sum(occ))
            if to > 1e-14:
                h += max(r, -to) * occ / to
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-7 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals(h, n):
    L = 2 * n - 1
    F = np.zeros(L)
    for d in range(-(n - 1), n):
        idx = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        F[idx] = np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d])
    return F

def surrogate(h, n, theta):
    F = lag_vals(h, n)
    m = float(np.max(F))
    ex = np.exp(np.clip((F - m) / theta, -700.0, 0.0))
    sw = float(np.sum(ex))
    if sw <= 0 or not math.isfinite(sw):
        return (2.0 / n) * m, np.zeros(n)
    p = ex / sw
    s_val = (2.0 / n) * (m + theta * math.log(sw))
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        idx = d + n - 1
        w = p[idx]
        if w < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += (2.0 / n) * w * (1.0 - h[lo + d:hi + d])
        g[lo + d:hi + d] -= (2.0 / n) * w * h[lo:hi]
    return s_val, g

def cont_opt(start, n, itp, taus):
