                    continue
                spans = hi - lo
                dd_cands = [lo + f * spans for f in (0.3, 0.5, 0.7)]
                dd_cands.append(lo + 1e-10)
                dd_cands.append(hi - 1e-10)
                if abs(best[j] - best[i]) < 1e-12:
                    dd_cands.append(0.5 * (best[i] - best[i]))
                for dd in dd_cands:
                    dd = max(lo, min(hi, dd))
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    sc = exact_score(trial, n)
                    if sc < best_s - 1e-13:
                        best = trial
                        best_s = sc
                        improved = True
        if not improved:
            break
    return best, best_s

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 64:
        n_starts_d, n_iter_d, ps_d = 6, 65, 2
    elif n <= 128:
        n_starts_d, n_iter_d, ps_d = 4, 30, 1
    else:
        n_starts_d, n_iter_d, ps_d = 2, 12, 0
    try:
        n_starts = max(1, min(8, int(kwargs.get('starts', n_starts_d))))
    except Exception:
        n_starts = n_starts_d
    try:
        n_iter = max(1, min(120, int(kwargs.get('max_iter', n_iter_d))))
    except Exception:
        n_iter = n_iter_d
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 5), n)
        add_start(starts, np.roll(fallback, -(n // 7 + 1)), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, fallback + rng.normal(0, 0.06, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fallback + rng.normal(0, 0.05, n), n)
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    for st in starts:
        try:
            res_h, res_sc = cont_optimize(st, n, n_iter)
            if np.isfinite(res_sc) and res_sc < best_score - 1e-12:
                best, best_score = res_h.copy(), res_sc
        except Exception:
            pass
    hop_sds = [0.025, 0.08] if n <= 128 else [0.05]
    for hs in hop_sds:
        pert = best + rng.normal(0, hs, n)
        hp = project_capped(pert, n)
        if hp is None:
            continue
        ss = exact_score(hp, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = hp.copy(), ss
        try:
            res_h, res_sc = cont_optimize(hp, n, n_iter)
            if np.isfinite(res_sc) and res_sc < best_score - 1e-12:
                best, best_score = res_h.copy(), res_sc
        except Exception:
            pass
    if n <= 128 and ps_d > 0:
        polished, poly_sc = pair_polish(best, n, ps_d)
        if np.isfinite(poly_sc) and poly_sc < best_score - 1e-12:
            best, best_score = polished.copy(), poly_sc
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
