def build_qp(n):
    if cp is None: return None
    try:
       dx=2.0/n
       L=2.0*dx
       d=cp.Variable(n)
       u=cp.Variable()
       qp=cp.Parameter(2*n-1)
       Gp=cp.Parameter((2*n-1,n))
       hp=cp.Parameter(n)
       cons=[Gp @ d + qp <= u, cp.sum(d)==0, d >= -hp, d <= 1.0 - hp]
       prob=cp.Problem(cp.Minimize(u + (L/2.0)*cp.sum_squares(d)), cons)
       return prob,d,u,qp,Gp,hp
    except Exception:
       return None

import numpy as np
import math
try:
    import cvxpy as cp
except Exception:
    cp = None

def project_capped(z, n):
    z = np.asarray(z, dtype=float).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(z - m, 0.0, 1.0)) > s:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(4):
        if abs(r) <= 1e-13:
            break
        for i in range(n):
            if r > 0.0:
                a = min(r, 1.0 - h[i])
                h[i] += a
                r -= a
            else:
                a = min(-r, h[i])
                h[i] -= a
                r += a
    if (not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or
            abs(s - float(np.sum(h))) > 1e-9):
        return None
    return np.clip(h, 0.0, 1.0)

def add_unique(pool, v, th, n):
    if v is None:
        return
    v = np.asarray(v, dtype=float).ravel()
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    for a in pool:
        if np.max(np.abs(v - a)) < th:
            return
    pool.append(v)

def make_binary(rng, n):
    v = np.zeros(n)
    p = rng.permutation(n)
    v[p[:n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v

def exact_score(h, n):
    try:
        h = np.asarray(h, dtype=float).ravel()
        if h.shape != (n,) or not np.all(np.isfinite(h)):
            return float('inf')
        if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(float(np.sum(h)) - n / 2.0) > 1e-9:
            return float('inf')
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')

def build_pool(fb, n, rng, size):
    raw = [fb.copy(), fb[::-1].copy(), 1.0 - fb, 1.0 - fb[::-1], np.full(n, 0.5)]
    b = np.zeros(n)
    b[:n // 2] = 1.0
    if n % 2:
        b[n // 2] = 0.5
    raw += [b, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), make_binary(rng, n)]
    raw += [fb + rng.normal(0.0, 0.05, n), fb + rng.normal(0.0, 0.15, n),
            rng.uniform(0.0, 1.0, n), np.roll(fb, n // 3), np.roll(fb, n // 2)]
    for k in (1, 2, 3):
        for ph in (0.0, math.pi / 2.0):
            raw.append(0.5 + 0.35 * np.cos(2.0 * math.pi * k * np.arange(n) / float(n) + ph))
    while len(raw) < size:
        base = raw[int(rng.integers(len(raw)))]
        if rng.random() < 0.7:
            sd = float(rng.choice([0.04, 0.12, 0.20]))
            raw.append(base + rng.normal(0.0, sd, n))
        else:
            raw.append(rng.uniform(0.0, 1.0, n))
    pool = []
    for v in raw:
        add_unique(pool, project_capped(v, n), 1e-8, n)
    if not pool:
        p = project_capped(np.full(n, 0.5), n)
        if p is None:
            p = np.full(n, 0.5)
        pool.append(p)
    while len(pool) < size:
        pool.append(pool[0].copy())
    return pool

def build_qp(n):
    if cp is None:
        return None
    try:
        dx = 2.0 / n
        L = 2.0 * dx * (1.0 + 1e-6)
        d = cp.Variable(n)
        u = cp.Variable()
        qp = cp.Parameter(2 * n - 1)
        Gp = cp.Parameter((2 * n - 1, n))
        hp = cp.Parameter(n)
        prob = cp.Problem(
            cp.Minimize(u + 0.5 * L * cp.sum_squares(d)),
            [Gp @ d + qp <= u, cp.sum(d) == 0, d >= -hp, d <= 1.0 - hp]
        )
        return prob, d, u, qp, Gp, hp
    except Exception:
        return None

def solve_qp(prob, d, u, q):
    if cp is None:
        return None
    try:
        d.value = np.zeros(d.shape[0], dtype=float)
        u.value = float(np.max(q))
    except Exception:
        pass
    try:
        avail = cp.installed_solvers()
    except Exception:
        avail = ['ECOS', 'CVXOPT', 'MOSEK', 'GUROBI']
    for s in ['ECOS', 'CVXOPT', 'MOSEK', 'GUROBI']:
        if s not in avail:
            continue
        if s == 'ECOS':
            kw = {'max_iters': 200}
        elif s == 'GUROBI':
            kw = {'threads': 1, 'OutputFlag': 0}
        else:
            kw = {}
        for opts in ([kw, {}] if kw else [{}]):
            ok = False
            try:
                prob.solve(solver=s, **opts)
                ok = True
            except Exception:
                if opts:
                    try:
                        prob.solve(solver=s)
                        ok = True
                    except Exception:
                        pass
            if not ok:
                continue
            if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:
                a = np.asarray(d.value, dtype=float).ravel()
                if a.size == d.shape[0] and np.all(np.isfinite(a)):
                    return a
    return None

def lag_data(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.empty(m)
    G = np.zeros((m, n))
    G[0] = dx * (1.0 - 2.0 * h)
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    idx = 1
    for r in range(1, n):
        a = n - r
        q[idx] = dx * float(np.sum(h[:a] * (1.0 - h[r:])))
        row = G[idx]
        row[:a] += dx
        row[:a] -= dx * h[r:]
        row[r:] -= dx * h[:a]
        idx += 1
        q[idx] = dx * float(np.sum(h[r:] * (1.0 - h[:a])))
        row = G[idx]
        row[r:] += dx
        row[r:] -= dx * h[:a]
        row[:a] -= dx * h[r:]
        idx += 1
    if np.any(~np.isfinite(q)) or np.any(~np.isfinite(G)):
        return None, None
    return q, G

def random_step(cur, n, rng, ref):
    if len(ref) > 1 and rng.random() < 0.6:
        i = int(rng.integers(len(ref)))
        j = int(rng.integers(len(ref)))
        if i == j:
            j = (j + 1) % len(ref)
        v = ref[i] - ref[j]
    else:
        v = rng.normal(0.0, 1.0, n)
    v = v - np.mean(v)
    mx = float(np.max(np.abs(v)))
    if mx <= 1e-14 or not np.isfinite(mx):
        return np.zeros(n)
    return (v / mx) * 0.12

def optimize_start(start, n, rng, qp, ref, miter):
    cur = project_capped(start, n)
    if cur is None:
        return None, float('inf')
    sc = exact_score(cur, n)
    if not np.isfinite(sc):
        return None, sc
    use = qp is not None
    small = 0
    bad = 0
    for _ in range(miter):
        old = sc
        step = None
        if use:
            q, G = lag_data(cur, n)
            if q is not None:
                prob, dv_var, u_var, qp_p, Gp_p, hp_p = qp
                try:
                    q2 = q + 1e-10
                    qp_p.value = q2
                    Gp_p.value = G
                    hp_p.value = cur
                    dv = solve_qp(prob, dv_var, u_var, q2)
                    if dv is not None and float(np.max(np.abs(dv))) > 1e-9:
                        step = dv
                    else:
                        bad += 1
                except Exception:
                    bad += 1
            else:
                bad += 1
            if bad >= 3:
                use = False
        if step is None:
            step = random_step(cur, n, rng, ref)
            if float(np.max(np.abs(step))) <= 1e-12:
                step = rng.normal(0.0, 1.0, n)
                step = step - np.mean(step)
                mx = float(np.max(np.abs(step)))
                if mx <= 1e-14:
                    break
                step = (step / mx) * 0.12
        best = None
        bsc = sc
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = project_capped(cur + alpha * step, n)
            if cand is None or float(np.max(np.abs(cand - cur))) < 1e-11:
                continue
            s2 = exact_score(cand, n)
            if s2 < bsc - 1e-12:
                bsc = s2
                best = cand
        if best is None:
            for _ in range(10):
                st = random_step(cur, n, rng, ref)
                if float(np.max(np.abs(st))) <= 1e-12:
                    continue
                for alpha in (0.5, 0.2, 0.07):
                    cand = project_capped(cur + alpha * st, n)
                    if cand is None:
                        continue
                    s2 = exact_score(cand, n)
                    if s2 < sc - 1e-12:
                        bsc = s2
                        best = cand
                        break
                if best is not None:
                    break
        if best is None:
            break
        cur = best.copy()
        sc = bsc
        small = small + 1 if old - sc < 1e-10 else 0
        if small >= 4:
            break
    return cur, sc

def finalize(bh, bs, fb, fbs, n):
    best = project_capped(bh, n) if bh is not None else None
    sc = exact_score(best, n) if best is not None else float('inf')
    fb2 = project_capped(fb, n)
    scf = exact_score(fb2, n)
    if fb2 is None or not np.isfinite(scf):
        u = project_capped(np.full(n, 0.5), n)
        if u is None:
            u = np.full(n, 0.5)
        fb2 = u
        scf = exact_score(fb2, n)
    if best is not None and np.isfinite(sc) and (not np.isfinite(scf) or sc < scf - 1e-12):
        return np.asarray(best, dtype=float).copy(), float(sc), int(n)
    return np.asarray(fb2, dtype=float).copy(), float(scf), int(n)

def run(seed=42, **kwargs):
    n = None
    par = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                par = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h = np.array([0.5], dtype=float)
        return h, float(exact_score(h, 1)), 1
    if par is not None:
        fb = project_capped(par, n)
    else:
        fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.asarray(fb, dtype=float).copy()
    fsc = exact_score(fb, n)
    size = 16 if n <= 128 else 12
    pool = build_pool(fb, n, rng, size)
    scores = np.empty(len(pool), dtype=float)
    for i in range(len(pool)):
        scores[i] = exact_score(pool[i], n)
    bh = fb.copy()
    bs = fsc
    for i in range(len(pool)):
        if scores[i] < bs - 1e-12:
            bs = float(scores[i])
            bh = pool[i].copy()
    cap = 4 if n <= 64 else 3 if n <= 128 else 2
    sq = np.argsort(scores)
    starts = []
    add_unique(starts, bh.copy(), 1e-8, n)
    add_unique(starts, fb.copy(), 1e-8, n)
    for i in sq:
        if len(starts) >= cap:
            break
        v = pool[int(i)]
        ok = True
        for s in starts:
            if np.max(np.abs(v - s)) < 0.05:
                ok = False
                break
        if ok:
            add_unique(starts, v.copy(), 1e-8, n)
    for i in sq:
        if len(starts) >= cap:
            break
        add_unique(starts, pool[int(i)].copy(), 1e-8, n)
    miter = 25 if n <= 64 else 18 if n <= 128 else 12 if n <= 256 else 8
    qp = build_qp(n) if n <= 300 else None
    for st in starts:
        c, s = optimize_start(st, n, rng, qp, pool, miter)
        if c is not None and s < bs - 1e-12:
            bh = c.copy()
            bs = s
    if qp is not None:
        c, s = optimize_start(bh, n, rng, qp, pool, max(4, miter // 3))
        if c is not None and s < bs - 1e-12:
            bh = c.copy()
            bs = s
    return finalize(bh, bs, fb, fsc, n)
