                gs, gb = bs, bh
        except Exception:
            continue

    # Additional: try slightly different n values by interpolating best
    if n <= 80:
        for dn in (2, -2):
            n2 = n + dn
            if n2 < 40 or n2 > 120:
                continue
            t2 = (np.arange(n2) + 0.5) / n2
            t1 = (np.arange(n) + 0.5) / n
            h2 = np.interp(t2, t1, gb, left=gb[0], right=gb[-1])
            h2p = project_capped(h2, n2)
            if h2p is None:
                continue
            taus2 = [0.01, 0.003, 0.0008, 0.00015, 0.00003]
            mi2 = max_it if n2 <= 64 else 70
            try:
                bh2, bs2 = continuation_search(h2p, n2, taus2, mi2)
                if bs2 < gs - 1e-6:
                    gs, gb, n = bs2, bh2, n2
            except Exception:
                continue

    # Perturbation restarts
    if n_re > 0:
        if gs < fb_s - 1e-5:
            perts = [0.03, 0.08]
        else:
            perts = [0.08, 0.18]
        if n > 64:
            perts = [0.05, 0.12]
        for i in range(min(n_re, len(perts))):
            try:
                pert = project_capped(gb + rng.normal(0.0, perts[i], n), n)
                if pert is None:
                    continue
                bh, bs = continuation_search(pert, n, taus, max_it)
                if bs < gs - 1e-12:
                    gs, gb = bs, bh
            except Exception:
                continue

    # Second round: more aggressive perturbations from global best
    for sd in (0.15, 0.25):
        try:
            pert = project_capped(gb + rng.normal(0.0, sd, n), n)
            if pert is None:
                continue
            bh, bs = continuation_search(pert, n, taus, max_it // 2)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue

    final = project_capped(gb, n)
    if final is None:
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if not np.isfinite(sf) or sf > ss:
        final, sf = fb, ss
    if not np.isfinite(sf):
        final = np.full(n, 0.5, dtype=np.float64)
        sf = exact_score(final, n)

    return final, float(sf), int(n)
