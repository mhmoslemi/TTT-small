    dx = 2.0 / n
    L = 2 * n - 1
    A = lag_vals(h, n)
    B = np.zeros(L)
    C = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        hp, hq = h[p0:p0 + m], h[q0:q0 + m]
        vp, vq = v[p0:p0 + m], v[q0:q0 + m]
        B[row] = dx * (np.dot(vp, 1.0 - hq) - np.dot(hp, vq))
        C[row] = -dx * np.dot(vp, vq)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return None, cur_s
    alphas = [lo, hi, 0.0]
    for s in range(L):
        if abs(C[s]) > 1e-15:
            sp = -B[s] / (2.0 * C[s])
            if lo - 1e-10 <= sp <= hi + 1e-10:
                alphas.append(sp)
    for s in range(L):
        for t in range(s + 1, L):
            roots = quad_roots(C[s] - C[t], B[s] - B[t], A[s] - A[t])
            for r in roots:
                if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
                    alphas.append(r)
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    clipped = sorted(max(lo, min(hi, a)) for a in alphas if np.isfinite(a))
    unique = []
    for a in clipped:
        if not unique or abs(a - unique[-1]) > tol:
            unique.append(a)
    best_a, best_f = 0.0, cur_s
    for st in range(0, len(unique), 512):
        arr = np.array(unique[st:st + 512])
        vals = A[None, :] + arr[:, None] * B[None, :] + (arr * arr)[:, None] * C[None, :]
        F = np.max(vals, axis=1)
        for idx in range(len(arr)):
            if F[idx] < best_f - 1e-14:
                best_f, best_a = F[idx], arr[idx]
    if best_a == 0.0 or best_f >= cur_s - 1e-12:
        return None, cur_s
    ht = h + best_a * v
    if np.any(ht < -1e-10) or np.any(ht > 1.0 + 1e-10):
        ht = project_capped(ht, n)
        if ht is None:
            return None, cur_s
    sc = exact_score(ht, n)
    if sc < cur_s - 1e-12:
        return ht, sc
    return None, cur_s

def box_dir(h, n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    donors = [i for i in idx if h[i] > 0.02]
    recvs = [i for i in idx if h[i] < 0.98 and i not in set(donors)]
    if len(donors) < 2 or len(recvs) < 2:
        return None
    donors, recvs = donors[:8], recvs[:8]
    v = np.zeros(n)
    wd = rng.uniform(0.5, 1.0, len(donors))
    wr = rng.uniform(0.5, 1.0, len(recvs))
    sd, sr = np.sum(wd), np.sum(wr)
    v[donors] = -wd / sd
    v[recvs] = wr / sr
    v[-1] -= np.sum(v)
    m = np.max(np.abs(v))
    if m < 1e-15:
        return None
    return v / m

def make_modes(n, rng, cnt):
    ks = list(range(1, n))
    if len(ks) > cnt:
        fr = np.logspace(0, math.log10(len(ks) - 1), cnt)
        ks = sorted(set(int(round(fr[i])) + 1 for i in range(cnt)) | {1, n - 1})
    modes = []
    for k in ks:
        if 1 <= k < n:
            v = dct2_v(k, n)
            if v is not None:
                modes.append(v)
    rng.shuffle(modes)
    return modes

def run(seed=42, **kwargs):
    n, parent = 47, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n, parent = v, None
        except Exception:
            pass
    if not (1 <= n <= 999):
        n, parent = 47, None
    rng = np.random.default_rng(int(seed))
    h0 = np.full(n, 0.5)
    s0 = exact_score(h0, n)
    hp = project_capped(parent, n) if parent is not None else None
    sp = exact_score(hp, n) if hp is not None else float('inf')
    if n == 1:
        return h0, s0, 1
    if n <= 64:
        n_modes, n_sweeps, n_starts, n_pert = n - 1, 4, 4, 4
    elif n <= 256:
        n_modes, n_sweeps, n_starts, n_pert = 12, 2, 3, 2
    else:
        n_modes, n_sweeps, n_starts, n_pert = 8, 1, 2, 1
    starts = []
    if hp is not None:
        starts.append(hp.copy())
    starts.append(h0.copy())
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    pb = project_capped(blk, n)
    if pb is not None: starts.append(pb)
    i = np.arange(n, dtype=np.float64)
    pw = project_capped(np.clip(0.5 + 0.45 * np.cos(2 * np.pi * (i + 0.5) / n), 0, 1), n)
    if pw is not None: starts.append(pw)
    if hp is not None:
        pp = project_capped(hp + rng.normal(0, 0.1, n), n)
        if pp is not None: starts.append(pp)
    pu = project_capped(rng.uniform(0, 1, n), n)
    if pu is not None: starts.append(pu)
    ded = []
    for s in starts:
        if not any(np.max(np.abs(s - d)) < 1e-8 for d in ded):
            ded.append(s)
    ded = ded[:n_starts]
    gb = min(ded, key=lambda x: exact_score(x, n)).copy()
    gs = exact_score(gb, n)
    if gs > min(s0, sp):
        gb = h0.copy() if s0 <= sp else (hp.copy() if hp is not None else h0.copy())
        gs = min(s0, sp)
    for st in ded:
        cur, cs = st.copy(), exact_score(st, n)
        for sw in range(n_sweeps):
            imp = False
            for v in make_modes(n, rng, n_modes):
                try:
                    res, ns = line_search(cur, n, v, cs)
                    if res is not None:
                        cur, cs = res, ns
                        imp = True
                        if cs < gs - 1e-12:
                            gs, gb = cs, cur.copy()
                except Exception:
                    pass
            for _ in range(2):
                bv = box_dir(cur, n, rng)
                if bv is None:
                    continue
                try:
                    res, ns = line_search(cur, n, bv, cs)
                    if res is not None:
                        cur, cs = res, ns
                        imp = True
                        if cs < gs - 1e-12:
                            gs, gb = cs, cur.copy()
                except Exception:
                    pass
            if not imp:
                break
    for _ in range(n_pert):
        pert = project_capped(gb + rng.normal(0, 0.04, n), n)
        if pert is None:
            continue
        cur, cs = pert.copy(), exact_score(pert, n)
        if cs >= gs - 1e-12:
            continue
        for sw in range(n_sweeps):
            imp = False
            for v in make_modes(n, rng, n_modes):
                try:
                    res, ns = line_search(cur, n, v, cs)
                    if res is not None:
                        cur, cs = res, ns
                        imp = True
                        if cs < gs - 1e-12:
                            gs, gb = cs, cur.copy()
                except Exception:
                    pass
            if not imp:
                break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = h0
    sf = exact_score(final, n)
    if sf > s0:
        final, sf = h0.copy(), s0
    if hp is not None and sf > sp:
        final, sf = hp.copy(), sp
    return final, float(sf), int(n)
