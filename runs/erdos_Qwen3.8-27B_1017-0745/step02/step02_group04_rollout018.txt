import numpy as np

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def smooth_eval(h, n, tau):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    rmax = np.max(r)
    e = np.exp((r - rmax) / tau)
    w = e / np.sum(e)
    fval = rmax + tau * np.log(np.sum(e))
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        p, q = max(d, 0), max(-d, 0)
        m = n - abs(d)
        g[p:p+m] += w[d + n - 1] * dx * (1.0 - h[q:q+m])
        g[q:q+m] -= w[d + n - 1] * dx * h[p:p+m]
    if not np.all(np.isfinite(g)):
        return None, None
    return fval, g

def smooth_optimize(start, n, iters_per_temp):
    x = start.copy()
    best_h = start.copy()
    best_s = exact_score(start, n)
    temps = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    L = 4.0 / n + 16.0 / (n * 0.00001)
    for tau in temps:
        Lt = 4.0 / n + 16.0 / (n * tau)
        alpha = min(1.0, 0.5 / max(Lt, 1e-6))
        fval, grad = smooth_eval(x, n, tau)
        if fval is None:
            break
        for _ in range(iters_per_temp):
            for bt in range(20):
                trial = project_capped(x - alpha * grad, n)
                if trial is None:
                    alpha *= 0.5
                    continue
                v = trial - x
                if np.max(np.abs(v)) < 1e-12:
                    break
                ft, _ = smooth_eval(trial, n, tau)
                if ft is None:
                    alpha *= 0.5
                    continue
                if ft <= fval + np.dot(grad, v) + np.dot(v, v) / (2.0 * alpha) + 1e-12:
                    fval = ft
                    x = trial
                    fval, grad = smooth_eval(x, n, tau)
                    if grad is None:
                        break
                    alpha = min(1.0, 1.5 * alpha)
                    s = exact_score(x, n)
                    if s < best_s - 1e-12:
                        best_s = s
                        best_h = x.copy()
                    break

            else:
                break
    s = exact_score(x, n)
    if s < best_s - 1e-12:
        best_s, best_h = s, x.copy()
    return best_h, best_s

def pairwise_refine(start, n, rng):
    h = start.copy()
    s = exact_score(h, n)
    if s == float('inf'):
        return h, s
    if n <= 64:
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    else:
        pairs = [(i, i+1) for i in range(n-1) if i+1 < n]
        cap = min(128, max(16, n // 4))
        while len(pairs) < cap:
            i, j = rng.integers(0, n), rng.integers(0, n)
            if i != j:
                if i > j:
                    i, j = j, i
                if (i, j) not in pairs:
                    pairs.append((i, j))
    if n <= 64:
        meshes = [0.04, 0.01, 0.0025, 0.0006, 0.00015]
    elif n <= 256:
        meshes = [0.04, 0.01, 0.002, 0.0004]
    else:
        meshes = [0.04, 0.008, 0.0016]
    for mesh in meshes:
        best_h, best_s = h.copy(), s
        for (i, j) in pairs:
            for sign in (1, -1):
                if sign == 1:
                    cap_d = min(mesh, 1.0 - h[i], h[j])
                else:
                    cap_d = min(mesh, h[i], 1.0 - h[j])
                if cap_d < 1e-12:
                    continue

                h[j] -= sign * cap_d
                if is_feasible(h, n):
                    ss = exact_score(h, n)
                    if ss < best_s:
                        best_s, best_h = ss, h.copy()
                h[i] -= sign * cap_d
                h[j] += sign * cap_d
        if best_s < s - 1e-12:
            h, s = best_h, best_s
    return h, exact_score(h, n)

def try_add(pool, v, n):
    if v is None or v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    if np.any(v < -1e-6) or np.any(v > 1.0 + 1e-6):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    try_add(pool, project_capped(fallback[::-1].copy(), n), n)
    try_add(pool, project_capped(1.0 - fallback, n), n)
    try_add(pool, project_capped((1.0 - fallback)[::-1].copy(), n), n)
    try_add(pool, project_capped(fallback + rng.normal(0, 0.04, n), n), n)
    try_add(pool, project_capped(fallback + rng.normal(0, 0.15, n), n), n)
    P = max(2, n // 3)
    cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P)
    try_add(pool, project_capped(cos_v, n), n)
    blk = np.zeros(n)
    blk[:n//2] = 1.0
    if n % 2:
        blk[n//2] = 0.5
    try_add(pool, project_capped(blk, n), n)
    try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    return pool[:count]
