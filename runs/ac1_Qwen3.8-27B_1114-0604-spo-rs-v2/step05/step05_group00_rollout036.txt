```python
"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: multi-peak subgradient with line search, L-BFGS with adaptive beta,
coordinate descent, length exploration, and multi-restart.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L = sum_k p_k * c_k with respect to x
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def multi_peak_subgrad(a, n_iters=5000, top_k=10, line_search=True):
    """
    Multi-peak subgradient method with optional line search.
    Targets all top-K autocorrelation peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_total = 2 * n - 1
    ideal_level = 1.0 / M_total  # after normalization S=1

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_norm = c / (S * S)  # normalized autocorrelation
        M_val = float(c_norm.max())
        sc = 2.0 * n * M_val

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks (exclude those at or below ideal)
        sorted_idx = np.argsort(c_norm)[::-1]
        active_peaks = sorted_idx[:top_k]
        active_c = c_norm[active_peaks]
        
        # Only keep peaks above ideal
        above = active_c > ideal_level * 1.001
        if not above.any():
            break
        active_peaks = active_peaks[above]
        active_c = active_c[above]
        
        if len(active_peaks) == 0:
            break

        # Compute weighted gradient
        # Weight by excess above ideal
        excess = active_c - ideal_level
        weights = excess / excess.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, active_peaks):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                grad[i_start:i_end + 1] += w * 2.0 * a[k - i_start:k - i_end + 1][::-1]

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Step size
        if line_search:
            # Try multiple step sizes, accept the best
            sc_current = sc
            best_step_sc = sc_current
            best_a_try = a.copy()
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                a_try = a - lr * grad / gn  # normalized step
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < best_step_sc:
                    best_step_sc = sc_try
                    best_a_try = a_try.copy()
            a = best_a_try
        else:
            # Fixed step with decay
            step = 0.01 * (0.999 ** (it // 100))
            a -= step * grad / gn
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S

    return best_a


def coord_descent(a, n_sweeps=10, n_top=50):
    """
    Coordinate descent on top contributors at the peak lag.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = float(a[j])
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_aggressive(a, n_iters=200):
    """Aggressive Fienup: strong flattening pressure."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Aggressive flattening
        alpha = 0.3 + 0.5 * (it / n_iters)
        ratio = np.sqrt(P / (P + P_mean))
        # Where P is high, reduce; where low, increase
        A_new = A * (ratio ** (-alpha))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
```

```python
"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: multi-peak subgradient with line search, L-BFGS with adaptive beta schedule,
coordinate descent, aggressive Fienup, and length exploration.
Key improvements over prior work: (1) line-search-based subgradient for reliable progress,
(2) higher beta values in L-BFGS for sharper peak targeting, (3) length exploration,
(4) multi-peak weighting based on excess above ideal flat level.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(x[::-1], M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def multi_peak_subgrad(a, n_iters=5000, top_k=10, line_search=True, deadline=None):
    """
    Multi-peak subgradient with line search.
    Targets all top-K autocorrelation peaks weighted by excess above ideal level.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_total = 2 * n - 1
    ideal_level = 1.0 / M_total

    step_decay = 0.9995
    base_lr = 0.02

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        c_norm = c / (S * S)
        M_val = float(c_norm.max())
        sc = 2.0 * n * M_val

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks above ideal
        sorted_idx = np.argsort(c_norm)[::-1]
        active_peaks = sorted_idx[:top_k]
        active_c = c_norm[active_peaks]
        above = active_c > ideal_level * 1.0005
        if not above.any():
            break
        active_peaks = active_peaks[above]
        active_c = active_c[above]
        if len(active_peaks) == 0:
            break

        # Weight by excess (sharper weights for higher peaks)
        excess = active_c - ideal_level
        weights = excess ** 2
        weights = weights / weights.sum()

        # Compute combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_peaks):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                grad[i_start:i_end + 1] += w * 2.0 * a[k - i_start:k - i_end + 1][::-1]

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Direction (normalize gradient norm)
        direction = -grad / gn
        step_scale = base_lr * (step_decay ** (it // 200))

        if line_search:
            best_step_sc = sc
            best_a_try = a.copy()
            for lr_mult in [0.25, 0.5, 1.0, 2.0, 4.0]:
                lr = step_scale * lr_mult
                a_try = a + lr * direction
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < best_step_sc:
                    best_step_sc = sc_try
                    best_a_try = a_try.copy()
            a = best_a_try
        else:
            a += step_scale * direction
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S

    return best_a


def coord_descent(a, n_sweeps=10, n_top=50, deadline=None):
    """Coordinate descent on top contributors at peak lags."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        # Try top 3 lags
        top_lags = np.argsort(c)[::-1][:3]
        for k_star in top_lags:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k_star - js]
            total = contribs.sum()
            if total < 1e-15:
                continue
            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local
            cur_sc = compute_score(a)
            for j in top_global:
                a_old_j = float(a[j])
                if a_old_j < 1e-14:
                    continue
                candidates = [a_old_j * f for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6]]
                best_val = a_old_j
                best_local = cur_sc
                for cv in candidates:
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old_j
                        continue
                    sc = compute_score(a / S_tmp)
                    if sc < best_local:
                        best_local = sc
                        best_val = cv
                    a[j] = a_old_j
                a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_aggressive(a, n_iters=200):
    """Aggressive Fienup: strong spectral flattening pressure."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        alpha = 0.3 + 0.5 * (it / n_iters)
        ratio = np.sqrt(P / (P + P_mean))
        A_new = A * (ratio ** (-alpha))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile_target(a, n_iters=200):
    """Fienup targeting the worst autocorrelation quantile."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    M_lin = 2 * n - 1
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = c.max()
        # Target: reduce the top 10% of autocorrelation values
        thresh = np.quantile(c, 0.90)
        # Build a "target" autocorrelation
        c_target = np.minimum(c, thresh)

        # Compute FFT of a
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2

        # In frequency domain: P should correspond to c_target
        P_target = np.zeros(len(A))
        P_target[:len(c_target)] = np.fft.rfft(c_target, M_fft)
        # Ensure P_target is real and non-negative (take real part, clip)
        P_target = np.maximum(np.real(P_target), 0)

        # Blend
        alpha = 0.1 + 0.4 * (it / n_iters)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        # Update: keep phase, adjust amplitude
        A_new = A * np.sqrt(P_blend / P)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def perturbed_copy(a, rng, scale=0.02):
    """Create a multiplicative Gaussian perturbation."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def run_fista(a0, beta, n_iters=3000, deadline=None):
    """FISTA on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S

    x = a0.copy()
    z = a0.copy()
    t_k = 1.0

    _, g0 = lse_obj_grad(z, beta)
    gn = float(np.linalg.norm(g0))
    L = max(gn + 1.0, 5.0)

    best_x = a0.copy()
    best_sc = compute_score(a0)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

        f_y, g_y = lse_obj_grad(z, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new

        f_new, _ = lse_obj_grad(x_new, beta)
        quadratic = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quadratic:
            L *= 1.5
            if L > 1e10:
                break
            continue
        else:
            L = max(0.7 * L, 1.0)

        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = z.sum()
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()
    return best_x


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: subgradient + L-BFGS (high beta) + coord descent +
    Fienup + length exploration. Returns best non-negative sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        print("No parent, using uniform init")

    print(f"Starting best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Quick global mixing (15% time)
    # ==================================================================
    phase1_end = t_start + 0.15 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup mixing (until t={phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 30:
        a_f1 = fienup_aggressive(best_a, n_iters=100)
        sc = compute_score(a_f1)
        update(sc, a_f1, "FIENUP-agg")

    if time.monotonic() < phase1_end - 10:
        a_f2 = fienup_quantile_target(best_a, n_iters=100)
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-quant")

    # ==================================================================
    # Phase 2: Multi-peak subgradient with line search (20% time)
    # ==================================================================
    phase2_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Multi-peak subgradient (until t={phase2_end - t_start:.0f}s) ===")

    # Run with increasing top_k
    for tk in [5, 10, 20, 30]:
        if time.monotonic() > phase2_end - 60:
            break
        a_sg = multi_peak_subgrad(best_a, n_iters=2000, top_k=tk,
                                  line_search=True, deadline=phase2_end - 30)
        sc = compute_score(a_sg)
        update(sc, a_sg, f"SUBGRAD-tk{tk}")

    # Quick L-BFGS polish after subgradient
    if time.monotonic() < phase2_end - 30:
        sc, a_opt = run_lbfgs(best_a, 3000, maxiter=5000, maxfun=100000)
        update(sc, a_opt, "LBFGS-3k-after-sg")

    # ==================================================================
    # Phase 3: L-BFGS with high-beta schedule (30% time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS high-beta sweep (until t={phase3_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    # Progressive beta: each higher beta refines the previous
    beta_schedule = [1000, 5000, 10000, 20000, 50000, 100000, 200000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end - 90:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 120:
            break
        if beta <= 10000:
            maxiter = min(20000, int(remaining * 50))
            maxfun = min(400000, int(remaining * 1000))
        else:
            maxiter = min(12000, int(remaining * 30))
            maxfun = min(300000, int(remaining * 600))

        print(f"  L-BFGS beta={beta} (maxiter={maxiter}, maxfun={maxfun})...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # Second pass: moderate beta on best (can escape fine-grained local min)
    if time.monotonic() < phase3_end - 120:
        print("  Second pass: moderate beta...")
        for beta in [2000, 10000, 30000]:
            if time.monotonic() > phase3_end - 30:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
            update(sc, a_opt, f"LBFGS-2 b={beta}")

    # ==================================================================
    # Phase 4: Coordinate descent + FISTA (15% time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Coord descent + FISTA (until t={phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 120:
        a_cd = coord_descent(best_a, n_sweeps=20, n_top=100, deadline=phase4_end - 120)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")

        if sc < best_score and time.monotonic() < phase4_end - 60:
            sc2, a2 = run_lbfgs(a_cd, 20000, maxiter=8000, maxfun=160000)
            update(sc2, a2, "CD+LBFGS-20k")

    if time.monotonic() < phase4_end - 60:
        a_fi = run_fista(best_a, 5000, n_iters=3000, deadline=phase4_end - 30)
        sc = compute_score(a_fi)
        update(sc, a_fi, "FISTA-5k")

    if time.monotonic() < phase4_end - 30:
        a_sg2 = multi_peak_subgrad(best_a, n_iters=1500, top_k=40,
                                   line_search=True, deadline=phase4_end - 10)
        sc = compute_score(a_sg2)
        update(sc, a_sg2, "SUBGRAD-tk40")

    # ==================================================================
    # Phase 5: Length exploration + restarts (10% time)
    # ==================================================================
    phase5_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 5: Length exploration (until t={phase5_end - t_start:.0f}s) ===")

    current_n = len(best_a)
    # Try a few longer lengths
    length_candidates = sorted(set([
        current_n,
        current_n + 256,
        current_n + 512,
        current_n + 1024,
        8192,
        10240,
        12288,
    ]))

    for new_n in length_candidates:
        if time.monotonic() > phase5_end - 60:
            break
        if new_n == current_n:
            continue
        if new_n > 16000:
            continue
        a_interp = interpolate_to_length(best_a, new_n)
        sc_inter = compute_score(a_interp)
        print(f"  Interp to n={new_n}: score={sc_inter:.8f}")
        if time.monotonic() < phase5_end - 30:
            sc, a_opt = run_lbfgs(a_interp, 10000, maxiter=5000, maxfun=100000)
            update(sc, a_opt, f"LEN-{new_n}")

    # Multi-restart with perturbations
    if time.monotonic() < phase5_end - 30:
        for r in range(3):
            if time.monotonic() > phase5_end - 20:
                break
            a_pert = perturbed_copy(best_a, rng, scale=0.01 + 0.005 * r)
            sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=60000)
            update(sc, a_opt, f"RESTART-{r}")

    # ==================================================================
    # Phase 6: Final polish (10% time)
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = int(remaining * 30)
        maxiter = max(5000, min(20000, maxiter))
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 25)
        update(sc, a_final, "FINAL-LBFGS-50k")

    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=8000, maxfun=160000)
        update(sc, a_final, "FINAL-LBFGS-100k")

    # Final subgradient squeeze
    if time.monotonic() < deadline - 20:
        a_final_sg = multi_peak_subgrad(best_a, n_iters=1000, top_k=50,
                                        line_search=True, deadline=deadline - 10)
        sc = compute_score(a_final_sg)
        update(sc, a_final_sg, "FINAL-SUBGRAD")

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_cd = coord_descent(best_a, n_sweeps=5, n_top=120, deadline=deadline - 5)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
