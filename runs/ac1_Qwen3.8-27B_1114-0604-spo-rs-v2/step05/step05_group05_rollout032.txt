                           lr_start=0.01, lr_decay=0.9995, mu_start=0.85, mu_end=0.96):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    it = 0
    beta = beta_start
    lr = lr_start
    mu = mu_start
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(mu_end, mu_start + (mu_end - mu_start) * (it / max(1, it + 100)))
        lr *= lr_decay
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute gradient
        _, grad = lse_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        # Adaptive step
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Increase beta gradually
        beta = min(beta * 1.003, beta_end)

        # Track best
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def peak_squeeze(a, max_time, n_top_peaks=5, n_top_pairs=10):
    """
    Direct peak-squeezing: identify the top peaks and reduce them by
    moving mass from contributing pairs to less harmful positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    stall_count = 0
    
    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        current_max = c.max()
        
        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        
        moved = False
        for k in peak_indices:
            k = int(k)
            if current_max < best_sc * (S*S) / (2.0*n) * 1.0001:
                break
            
            # Find contributing pairs for this peak
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contributions: a[j] * a[k-j] for j in [j_start, j_end]
            pairs = []
            for j in range(j_start, j_end + 1):
                prod = a[j] * a[k - j]
                pairs.append((prod, j, k - j))
            
            pairs.sort(reverse=True)
            
            for prod, j1, j2 in pairs[:n_top_pairs]:
                if prod < 1e-15:
                    continue
                    
                # Try reducing a[j1] and moving that mass to a random position
                # that doesn't contribute to any top peak
                old_val = a[j1]
                epsilon = old_val * 0.1  # move 10% of the mass
                
                # Find a target position that minimizes the new peak
                # Simple heuristic: move to a position where the local convolution is low
                # Try a few random target positions
                best_target = -1
                best_new_score = best_sc + 0.01  # must improve by at least this much
                
                for _ in range(20):
                    target = np.random.randint(0, n)
                    if target == j1:
                        continue
                    
                    a[j1] = old_val - epsilon
                    a[target] += epsilon
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j1] = old_val
                        a[target] -= epsilon
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_new_score:
                        best_new_score = sc
                        best_target = target
                    a[j1] = old_val
                    a[target] -= epsilon
                
                if best_target >= 0 and best_new_score < best_sc:
                    a[j1] = old_val - epsilon
                    a[best_target] += epsilon
                    a = a / a.sum()
                    best_sc = best_new_score
                    best_a = a.copy()
                    moved = True
                    break
            if moved:
                stall_count = 0
                break
            if not moved:
                stall_count += 1
                if stall_count > 100:
                    break
    
    return best_a


def fienup_optimize(a, n_iters=200, thresh_start=2.0, thresh_end=1.05,
                    clip_start=0.3, clip_end=0.8, momentum=0.2):
    """Fienup with momentum for spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if above.any():
            scale = np.ones_like(P)
            scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)
            A_mod = A * scale
        else:
            A_mod = A

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Momentum
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_grad(x, beta)

    res = minimize(
        objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_chirp_init(n):
    """Chirp-based initialization."""
    x = np.arange(n, dtype=np.float64)
    # Quadratic chirp, absolute value, shifted to be non-negative
    a = np.abs(np.cos(np.pi * (x / n) ** 2 * n)) + 0.05
    return a / a.sum()


def build_racem_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_mixed_init(n, rng):
    """Mixed initialization."""
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.3 * np.cos(2 * np.pi * x * (1 + rng.random() * 3)) +
         0.2 * np.cos(4 * np.pi * x * (1 + rng.random() * 2)))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def upsample(a, new_n):
    """Upsample using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def multi_beta_lbfgs(a, time_budget):
    """Run L-BFGS with increasing beta values."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for beta in [100, 300, 1000, 3000, 10000, 30000]:
        if time.monotonic() - t_start > time_budget:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    L-BFGS beta={beta}: {sc:.8f}")
    
    return best_a

"""

and multi-scale L-BFGS polish. Combines:

4. Multi-beta L-BFGS final polish

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_and_grad(a, beta):
    """
    LSE-smoothed objective and gradient.
    Objective: F(a) = 2n * LSE_beta(conv(a,a)) / (sum a)^2
    Returns (F, grad_F).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + np.log(sum_exp) / beta
    p = exp_shifted / sum_exp

    # dL/da[i] = 2 * sum_{m=0}^{n-1} a[m] * p[m+i]
    # Computed as: 2 * conv(a_rev, p)[n-1+i]
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    M_fft = 1 << (conv_len - 1).bit_length()

    a_rev = np.zeros(M_fft)
    a_rev[:n] = a[::-1]
    p_pad = np.zeros(M_fft)
    p_pad[:2 * n - 1] = p

    Aft = np.fft.rfft(a_rev)
    Pft = np.fft.rfft(p_pad)
    conv_result = np.fft.irfft(Aft * Pft)[:conv_len]

    grad_L = 2.0 * conv_result[n - 1:2 * n - 1]

    S2 = S * S
    F = 2.0 * n * L / S2
    grad = (2.0 * n / S2) * (grad_L - 2.0 * L / S)

    return F, grad


def nesterov_lse(a, max_time, beta_start=20.0, beta_end=100000.0,
                 lr_start=0.015, lr_decay=0.9997, mu_start=0.85, mu_end=0.97):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    Adaptive beta with aggressive growth.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    lr = lr_start
    mu = mu_start
    no_improve = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(mu_end, mu_start + (mu_end - mu_start) * min(1.0, it / 500.0))
        lr *= lr_decay

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute gradient
        _, grad = lse_value_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        # Adaptive step size
        eff_lr = lr
        if gnorm > 0:
            max_step = 0.5 / (gnorm + 1e-10)
            eff_lr = min(lr, max_step)

        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Grow beta
        beta_growth = 1.002 + 0.003 * min(1.0, it / 2000.0)
        beta = min(beta * beta_growth, beta_end)

        # Track best
        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                # If stuck, boost beta more aggressively
                if no_improve > 10:
                    beta = min(beta * 2.0, beta_end)
                    no_improve = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def peak_targeted_move(a, max_time, rng, n_top_peaks=5, n_tries=50):
    """
    Direct peak-targeting: identify top peaks and make structured moves
    to reduce them by redistributing mass.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    stall = 0

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        current_max = float(c.max())

        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        peak_vals = c[peak_indices]

        moved = False
        for k in peak_indices:
            k = int(k)
            if c[k] < best_sc * S * S / (2.0 * n) * 1.001:
                break

            # Find contributing pairs
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Get contributions
            contribs = []
            for j in range(j_start, j_end + 1):
                prod = a[j] * a[k - j]
                if prod > 1e-15:
                    contribs.append((prod, j, k - j))

            if not contribs:
                continue
            contribs.sort(reverse=True)

            # Try to reduce top contributors
            for prod, j1, j2 in contribs[:20]:
                old_val = a[j1]
                if old_val < 1e-10:
                    continue

                # Try different fractions to move
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    epsilon = old_val * frac

                    # Try several target positions
                    best_target = -1
                    best_new_sc = best_sc

                    for _ in range(n_tries):
                        # Target: prefer positions far from current peak contributors
                        target = int(rng.integers(0, n))
                        if target == j1:
                            continue

                        a[j1] = old_val - epsilon
                        a[target] += epsilon
                        S_tmp = float(a.sum())
                        if S_tmp < 1e-12:
                            a[j1] = old_val
                            a[target] -= epsilon
                            continue

                        sc = 2.0 * n * float(np.max(fast_conv(a))) / (S_tmp * S_tmp)

                        a[j1] = old_val
                        a[target] -= epsilon

                        if sc < best_new_sc:
                            best_new_sc = sc
                            best_target = target

                    if best_target >= 0:
                        a[j1] = old_val - epsilon
                        a[best_target] += epsilon
                        a = a / a.sum()
                        best_sc = best_new_sc
                        best_a = a.copy()
                        moved = True
                        break

                if moved:
                    break

        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

    return best_a


def fienup_optimize(a, n_iters=300, thresh_start=2.5, thresh_end=1.02,
                    clip_start=0.2, clip_end=0.9, momentum=0.3):
    """Fienup with momentum for spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if above.any():
            scale = np.ones_like(P)
            # Use a softer compression
            ratio = (P_mean * thresh) / P[above]
            scale[above] = 1.0 - clip + clip * np.sqrt(np.maximum(ratio, 0.0))
            A_mod = A * scale
        else:
            A_mod = A

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Momentum
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_value_and_grad, a0, method='L-BFGS-B', jac=True,
        bounds=bounds, args=(beta,),
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_chirp_init(n):
    """Quadratic chirp initialization (non-negative)."""
    x = np.arange(n, dtype=np.float64)
    phase = np.pi * (x / n) ** 2 * n
    a = np.abs(np.cos(phase)) + 0.02 * np.ones(n)
    return a / a.sum()


def build_symmetric_init(n):
    """Raised cosine symmetric initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_random_smooth_init(n, rng):
    """Random smooth initialization using Gaussian bump mixture."""
    a = np.zeros(n)
    n_bumps = int(n // 100) + 3
    for _ in range(n_bumps):
        center = rng.uniform(0, n)
        sigma = rng.uniform(n / 20, n / 5)
        x = np.arange(n)
        a += np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0.005)
    return a / a.sum()


def build_mixed_harmonic_init(n, rng):
    """Mixed harmonic initialization."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for _ in range(4):
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.3 * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def simulated_annealing_diversify(a, rng, temperature):
    """Perturb a sequence with structured noise."""
    n = len(a)
    a_copy = a.copy()

    # Type 1: Gaussian noise
    r = rng.random()
    if r < 0.3:
        noise = temperature * 0.1 * rng.standard_normal(n)
        a_copy += noise
    elif r < 0.6:
        # Type 2: Block perturbation
        n_blocks = rng.integers(3, 15)
        for _ in range(n_blocks):
            start = rng.integers(0, max(1, n - 10))
            length = rng.integers(5, min(50, n - start))
            scale = 1.0 + temperature * 0.5 * rng.standard_normal()
            a_copy[start:start + length] *= max(0.0, scale)
    else:
        # Type 3: Spike perturbation
        n_spikes = rng.integers(1, 5)
        for _ in range(n_spikes):
            pos = rng.integers(0, n)
            a_copy[pos] += temperature * 0.05 * rng.standard_normal()

    a_copy = np.maximum(a_copy, 0.0)
    S = a_copy.sum()
    if S < 1e-12:
        return a.copy()
    return a_copy / S

            a_nf = nesterov_lse(best_a, max_time=min(30, phase3_end - time.monotonic() - 15),
                                beta_start=100.0, beta_end=30000.0)
            update_best(compute_score(a_nf), a_nf, f"nesterov3-{i}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS multi-beta polish (18%)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 90.0)
    print(f"\n=== Phase 4: L-BFGS polish ({phase4_end - time.monotonic():.0f}s budget) ===")

    lbfgs_betas = [200, 500, 1000, 3000, 10000, 30000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase4_end:
            break
        rem = phase4_end - time.monotonic()
        if rem < 20:
            break

        # Scale maxiter/maxfun with available time
        maxiter = min(5000, max(500, int(rem * 50)))
        maxfun = min(100000, max(1000, int(rem * 800)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"lbfgs-b{beta}")

        # Quick Nesterov between L-BFGS runs
        if time.monotonic() < phase4_end - 30:
            a_qn = nesterov_lse(best_a, max_time=20,
                                beta_start=beta, beta_end=beta * 10)
            update_best(compute_score(a_qn), a_qn, f"qn-b{beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final peak squeeze + ultra-high beta L-BFGS (10%)
    # ============================================================
    print(f"\n=== Phase 5: Final refinement ({deadline - time.monotonic():.0f}s budget) ===")

    # Peak targeted move
    if time.monotonic() < deadline - 30:
        t_sq = min(40.0, (deadline - time.monotonic()) * 0.3)
        a_sq = peak_targeted_move(best_a, max_time=t_sq, rng=rng,
                                   n_top_peaks=3, n_tries=50)
        update_best(compute_score(a_sq), a_sq, "final-peak")

    # Ultra-high beta Nesterov
    if time.monotonic() < deadline - 30:
        t_fn = min(40.0, (deadline - time.monotonic()) * 0.3)
        a_fn = nesterov_lse(best_a, max_time=t_fn,
                           beta_start=5000.0, beta_end=200000.0,
                           lr_start=0.003, lr_decay=0.999)
        update_best(compute_score(a_fn), a_fn, "final-nesterov")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 20:
        sc, a_opt = run_lbfgs(best_a, 100000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "final-lbfgs-hibeta")

    # One more Nesterov pass
    if time.monotonic() < deadline - 10:
        a_last = nesterov_lse(best_a, max_time=min(20, deadline - time.monotonic() - 10),
                             beta_start=10000.0, beta_end=500000.0,
                             lr_start=0.002, lr_decay=0.999)
        update_best(compute_score(a_last), a_last, "final-last")

    # ============================================================
    # Output
    # ============================================================
    if best_a is None:
        best_a = np.ones(64) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
