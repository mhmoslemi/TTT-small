    n = len(a)

    N = _next_pow2(M)

    Returns array of length n: result[k] = sum_i a[i] * w[k-i] (valid range)"""

def _subgrad_step(a, eta, k_active=None):
    """One subgradient step: a -= eta * subgrad_F(a), keeping a >= 0."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    
    d = _d_values(a)  # d_k = n * (a*a)_k / S^2
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    
    if k_active is None:
        # Use all positions above some threshold
        w = np.exp(50 * (d - dmax))  # approximate indicator of argmax
        w = w / w.sum()
    else:
        # Use top-k positions with equal weight
        idx = np.argpartition(d, -k_active)[-k_active:]
        w = np.zeros(len(d))
        w[idx] = 1.0 / k_active
    
    h = _cross_corr(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return a
    
    # Subgradient of F = 2 * max(d) is: 2 * sum_k w_k * d(d_k)/da
    # = 2 * [(2n/S^2) * h - (2/S) * LDS]  (from above, with the factor of 2 for F=2L)
    # Actually for the true max (not LSE), the subgradient of max_k d_k is any convex combination of the active gradients.
    # d(max_k d_k)/da[j] = sum_k w_k * d(d_k)/da[j] = (2n/S^2)*h[j] - (2n/S^3)*sum_k w_k*(a*a)_k
    # F = 2*max(d)*n/S^2... wait I defined d_k = n*(a*a)_k/S^2 and F = 2*max(d)
    # So dF/da[j] = 2 * sum_k w_k * d(d_k)/da[j]
    # = 2 * [(2n/S^2)*h[j] - (2n/S^3)*sum_k w_k*(a*a)_k]
    
    # For the subgradient step, we want to move in the negative subgradient direction:
    # a_new = a - eta * subgrad
    # But we need to keep a >= 0. So use multiplicative update:
    # a_new = a * exp(-eta * subgrad / norm(subgrad))
    
    # Simpler: use the multiplicative update a * exp(-eta * h / hmax) which is what the previous code did.
    # This is approximately in the right direction.
    
    q = a * np.exp(-eta * h / hmax)
    return _normalize(q)

def _subgrad_polish(a, max_iters=50, deadline=None):
    """Repeated subgradient steps with decreasing step size."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        for k in (1, 2, 3, 5):
            for eta in (0.05, 0.1, 0.2, 0.4, 0.8):
                if deadline is not None and time.monotonic() > deadline:
                    break
                q = _subgrad_step(a, eta, k)
                sc = _score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < _score(a):
                    a = q
                    improved = True
        
        if not improved:
            break
    
    return best_sc, best_a

def _resample(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)

def _generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        if arr.sum() < 1e-10:
            arr = np.ones(n)
        inits.append(arr / arr.sum())
    
    if parent is not None:
        p = _resample(parent, n)
        add(p)
        add(p[::-1])
    
    # Uniform
    add(np.ones(n))
    
    # V-shapes (high at edges)
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2*x - 1))
    
    # Inverted V (high in middle) - usually worse but try
    for alpha in (0.3, 0.5):
        add(1.0 - alpha * np.abs(2*x - 1))
    
    # Cosine harmonics
    for f in (1, 2, 3, 4, 5):
        for alpha in (0.3, 0.5, 0.7, 1.0):
            add(1.0 + alpha * np.cos(2*np.pi*f*x))
            add(1.0 + alpha * np.cos(np.pi*f*x))
    
    # Sine
    for f in (1, 2, 3, 4):
        add(1.0 + 0.5*np.sin(np.pi*f*x))
        add(1.0 + 0.5*np.abs(np.sin(np.pi*f*x)))
    
    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1-x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1-x)))
    
    # Power law
    for p in (0.5, 1.0, 2.0, 3.0):
        add(1.0 / (1.0 + p * x))
        add(1.0 / (1.0 + p * (1-x)))
    
    # Random smooth (spectral)
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(1.0, 4.0))
        power = 1.0 / (1.0 + np.abs(freqs)) ** gamma
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        noise = np.fft.irfft(power * np.exp(1j * phase), Nf)[:n]
        noise = noise - noise.mean()
        amp = float(rng.uniform(0.3, 1.5))
        add(1.0 + amp * noise)
    
    # Random trigonometric
    for _ in range(5):
        vals = np.ones(n, dtype=np.float64)
        for _ in range(int(rng.integers(2, 7))):
            f = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2*np.pi*f*x + ph)
        add(vals)
    
    # Bimodal
    for w1 in (0.05, 0.1, 0.15, 0.2):
        for c1 in (0.2, 0.25, 0.3):
            c2 = 1 - c1
            add(np.exp(-((x-c1)/w1)**2) + np.exp(-((x-c2)/w1)**2))
    
    # "Comb" patterns
    for period in (4, 8, 16, 32):
        if period > n // 2:
            continue
        pattern = np.ones(n)
        pattern[::period] = 0.1
        add(pattern)
        pattern2 = np.ones(n)
        pattern2[::period] = 2.0
        add(pattern2)
    
    # Pure random
    for _ in range(3):
        add(rng.random(n) + 0.01)
    
    return inits

def _perturb(a, rng, strength=None):
    """Perturb a sequence for diversity."""
    a = _normalize(a)
    n = len(a)
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    
    mode = int(rng.integers(0, 10))
    x = np.linspace(0, 1, n)
    
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1-t)*a + t*(np.ones(n)/n)
    elif mode == 2:
        q = a[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        # Active-set aware
        d = _d_values(a)
        k = int(rng.integers(1, 8))
        idx = np.argpartition(d, -k)[-k:]
        w = np.zeros(len(d))
        w[idx] = 1.0/k
        h = _cross_corr(a, w)
        hmax = max(1e-12, float(np.max(np.abs(h))))
        q = a * np.exp(-strength * h / hmax)
    elif mode == 4:
        # Smooth noise in log domain
        logp = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        k = max(3, min(31, n // 200))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c)**2) / (2*sig*sig))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        noise -= noise.mean()
        q = np.exp(logp + 0.3*strength*noise - np.max(logp + 0.3*strength*noise))
    elif mode == 5:
        # V-shape blend
        vshape = 1.0 + strength * np.abs(2*x - 1)
        q = a * vshape
    elif mode == 6:
        # Cosine perturbation
        f = int(rng.integers(1, max(2, n//20)))
        amp = float(rng.uniform(0.1, 0.4))
        ph = float(rng.uniform(-np.pi, np.pi))
        q = a * (1 + amp * np.cos(2*np.pi*f*x + ph))
    elif mode == 7:
        # Local bump
        center = int(rng.integers(0, n))
        width = int(rng.integers(1, max(2, n//50)))
        amp = float(rng.uniform(-0.5, 0.5))
        bump = np.exp(-((np.arange(n) - center)**2) / (2*width*width))
        q = a * (1 + amp * bump)
    elif mode == 8:
        # Low-pass smooth
        k = int(rng.integers(5, max(7, n//100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n-1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c/2)
        kern = np.exp(-((np.arange(k)-c)**2)/(2*sig*sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    else:
        # Anti-smooth (high-pass)
        q = a - 0.5 * float(strength) * (np.convolve(a, np.ones(max(3, n//500), mode='same') / max(3, n//500)) - a)
    
    return _normalize(np.clip(q, 0, None))

def _run_lbfgs(a0, beta, maxfun=500):
    """Run L-BFGS-B on LSE objective with given beta."""
    a = _normalize(a0)
    n = len(a)
    bounds = [(0.0, 10.0)] * n
    opts = {
        'ftol': 1e-15,
        'gtol': 1e-12,
        'maxiter': max(50, maxfun // 2),
        'maxfun': maxfun,
        'maxls': 50,
    }
    try:
        res = minimize(_fun_lse, a, method='L-BFGS-B', jac=True,
                       args=(beta,), bounds=bounds, options=opts)
        if res.x is not None:
            return _normalize(res.x), _score(_normalize(res.x))
    except Exception:
        pass
    return a, _score(a)

def _optimize_with_continuation(a_start, deadline, max_beta=50000):
    """Adaptive continuation method for LSE optimization."""
    a = _normalize(a_start)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Adaptive beta schedule
    betas = [50, 100, 200, 400, 800, 1500, 3000, 5000, 10000, 20000, 40000, max_beta]
    
    for beta in betas:
        if time.monotonic() > deadline - 30:
            break
        
        # Run L-BFGS with this beta for a while
        n_runs = 0
        beta_deadline = min(deadline - 30, time.monotonic() + 120)
        
        while time.monotonic() < beta_deadline:
            mf = 200
            sc, a2 = _run_lbfgs(a, beta, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  beta={:.0f}: {:.8f}".format(beta, best_sc))
            a = _normalize(a2)
            n_runs += 1
            
            # Check convergence: if score hasn't improved much, break
            if n_runs >= 3 and np.isfinite(sc) and abs(sc - best_sc) < 1e-10:
                break
        
        # Subgradient polish between stages
        if time.monotonic() < deadline - 60:
            sc_sg, a_sg = _subgrad_polish(a, max_iters=15, deadline=deadline - 30)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("  subgrad after beta={:.0f}: {:.8f}".format(beta, best_sc))
            a = _normalize(a_sg)
    
    return best_sc, best_a

def _estimate_time(n):
    """Estimate time for one L-BFGS evaluation at length n."""
    return 0.01 * (n / 8192.0) * (1.0 + 0.1 * np.log2(max(4, n)))
