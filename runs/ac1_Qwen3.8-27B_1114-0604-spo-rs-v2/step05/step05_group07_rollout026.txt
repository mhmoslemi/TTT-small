"""
Multi-strategy optimizer leveraging Adam with LSE smoothing, 
exact coordinate descent, and population-based search to minimize
the autocorrelation peak ratio of non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_value_and_grad(a, beta):
    """
    LSE-smoothed objective and gradient.
    Score = 2n * LSE(c) / S^2 where LSE(c) = (1/beta) * log(sum(exp(beta * c)))
    but we use the shifted version for numerical stability.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    x = beta * (c - c_max)
    x = np.clip(x, -50, 50)
    exp_x = np.exp(x)
    log_sum_exp = c_max + np.log(exp_x.sum()) / beta
    L = log_sum_exp  # This approximates max(c)
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a
    p = exp_x / exp_x.sum()  # softmax weights over the convolution values
    
    # dL/da[i] = 2 * sum_k p[k] * a[k-i] (with appropriate index handling)
    # This is 2 * (p * reversed_a)[n-1 + i]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    # Full gradient: d(score)/da = 2n * (dL_da / S^2 - 2L / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad
```

def adam_optimize(a0, beta, n_iters, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective, with normalization after each step."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    a = a / S  # normalize to sum=1
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t += 1
        sc, grad = lse_value_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        # Center gradient (since we're on the simplex, constant shifts don't matter)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
    
    return best_a, best_sc
```

def projected_subgradient_with_averaging(a0, n_iters, f_lb=1.0, n_active=10):
    """
    Projected subgradient method with:
    - Top-k active set for subgradient
    - Polyak step size
    - Running average for output
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    
    # Running average
    a_avg = np.zeros(n)
    total_weight = 0.0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = a.sum()
        c_max = c.max()
        sc = 2.0 * n * c_max / (S_a * S_a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top-k peaks
        top_k = np.argsort(c)[::-1][:n_active]
        
        # Weighted subgradient
        # For peak at k, the subgradient component for coordinate i is:
        #   2 * a[k-i] if 0 <= k-i < n, else 0
        # So the subgradient vector is 2 * sum_k w_k * rev_a[k-i]
        # = 2 * (w * rev_a)[n-1+i] where w is the weight vector over peaks
        
        w = np.zeros(2*n-1)
        for rank, idx in enumerate(top_k):
            # Weight by proximity to max
            w[idx] = np.exp(-(c[idx] - c_max) * 10)  # exponential weighting
        w /= w.sum()
        
        rev_a = a[::-1]
        M2 = len(w) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        W = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_result = np.fft.irfft(W * RA, M2_fft)[:M2]
        sg = 2.0 * conv_result[n-1:n-1+n]
        
        # Full objective gradient (including the 1/S^2 factor)
        S2 = S_a * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / (S2 * S_a))
        
        # Polyak step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)
        
        a_new = a - step * grad
        a = np.clip(a_new, 0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Running average
        weight = 1.0 / (it + 1)
        total_weight += weight
        a_avg += weight * a
        a_avg /= total_weight
        
    return best_a

    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad

def adam_optimize(a0, beta, n_iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                  check_interval=100):
    """Adam optimizer on LSE-smoothed objective, with periodic normalization."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        return a, compute_score(a)
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t += 1
        sc, grad = lse_value_and_grad(a, beta)
        
        if it % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
        
        # Center gradient
        gc = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * gc
        v = beta2 * v + (1 - beta2) * gc * gc
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a_new, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new
    
    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()
    
    return best_a, best_sc

def coordinate_descent(a0, n_sweeps=5, n_top=20, n_bottom=20):
    """
    Coordinate descent: for each of the top contributors to the peak,
    try moving mass to the bottom contributors.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        L = len(contrib)
        top_count = min(n_top, L // 4)
        bot_count = min(n_bottom, L // 4)
        
        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]
        bot_sum = contrib[bot_local].sum()
        
        if bot_sum < 1e-15:
            break
        
        for li in top_local:
            gi = j_start + li
            old_val = a[gi]
            if old_val < 1e-10:
                continue
            
            # Try different transfer amounts
            best_val = old_val
            best_local_sc = best_sc
            
            for frac in [0.5, 0.7, 0.8, 0.9]:
                transfer = old_val * (1 - frac)
                a[gi] = old_val * frac
                # Redistribute to bottom contributors proportionally
                for bi in bot_local:
                    a[j_start + bi] += transfer * contrib[bi] / bot_sum
                
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = a.copy()
            
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = best_val
                a = best_val.copy()
            else:
                a[gi] = old_val
                for bi in bot_local:
                    a[j_start + bi] -= transfer * contrib[bi] / bot_sum
                S = a.sum()
                if S > 1e-12:
                    a = a / S
    
    return best_a

def exp_weighted_subgrad(a0, n_iters=5000, n_active=10, f_lb=1.0, gamma=5.0):
    """Exponentially-weighted multi-peak subgradient descent."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = a.sum()
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_a * S_a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_k = np.argsort(c)[::-1][:n_active]
        
        # Exponential weights
        w = np.zeros(2 * n - 1)
        for idx in top_k:
            w[idx] = np.exp(-gamma * (c[idx] - c_max))
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w /= w_sum
        
        # Subgradient
        rev_a = a[::-1]
        M2 = (2*n-1) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        W = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_result = np.fft.irfft(W * RA, M2_fft)[:M2]
        sg = 2.0 * conv_result[n-1:n-1+n]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = min((sc - f_lb) / (gnorm * gnorm + 1e-30), 0.3)
        
        a = np.clip(a - step * grad, 0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

def perturbed(a0, rng, scale=0.03):
    """Generate perturbation."""
    n = len(a0)
    strategy = rng.integers(0, 4)
    
    if strategy == 0:
        a_p = a0 * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        x = np.arange(n) / n
        a_p = a0 * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Peak-targeted: reduce at peak contributors, increase at valleys
        c = fast_conv(a0)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a0.copy()
        for i in range(j_start, j_end + 1):
            a_p[i] *= (1.0 - scale * rng.random())
        valley_mask = c < 0.5 * c.max()
        if valley_mask.any():
            valley_idx = np.where(valley_mask[:n])[0]
            for i in valley_idx[:n//10]:
                a_p[i] *= (1.0 + scale * 2.0 * rng.random())
    else:
        # Random block scaling
        n_blocks = 5 + int(rng.integers(0, 15))
        block_size = max(1, n // n_blocks)
        a_p = a0.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    
    a_p = np.clip(a_p, 0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

def multi_resolution_refine(coarse_a, n_fine, n_coarse):
    """Upsample coarse solution to fine resolution."""
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    fine_a = np.interp(x_fine, x_coarse, coarse_a)
    S = fine_a.sum()
    if S > 1e-12:
        fine_a = fine_a / S
    return fine_a

"""
Multi-strategy optimizer using Adam with LSE smoothing, exponentially-weighted
subgradient, targeted coordinate descent, and L-BFGS refinement to minimize
the autocorrelation peak ratio of non-negative sequences.
Key innovations over previous approach:
- Adam optimizer with per-coordinate adaptive learning rates

from scipy.optimize import minimize

# Global variable for L-BFGS callback
_current_beta = 100.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_and_grad(a, beta):
    """
    LSE-smoothed objective and gradient w.r.t. a.
    score = 2n * L / S^2 where L = LSE_beta(c)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    x = beta * (c - c_max)
    x = np.clip(x, -60.0, 60.0)
    exp_x = np.exp(x)
    log_sum = float(np.log(exp_x.sum()))
    L = c_max + log_sum / beta

    score = 2.0 * n * L / (S * S)

    p = exp_x / exp_x.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def lbfgs_callback(x):
    """Top-level callback for L-BFGS using global _current_beta."""
    return lse_value_and_grad(x, _current_beta)


def adam_optimize(a0, beta, n_iters, lr=0.02, beta1=0.9, beta2=0.999,
                  eps=1e-8, check_interval=200):
    """
    Adam optimizer on LSE-smoothed objective with periodic normalization.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a, compute_score(a)
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        t += 1
        sc, grad = lse_value_and_grad(a, beta)

        if it % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        gc = grad - np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * gc
        v = beta2 * v + (1.0 - beta2) * gc * gc
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a_new, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc


def lse_value_and_grad_w(a, beta):
    """
    LSE-smoothed objective and gradient w.r.t. w where a = exp(w).
    Used for log-parameterized optimization.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    x = beta * (c - c_max)
    x = np.clip(x, -60.0, 60.0)
    exp_x = np.exp(x)
    log_sum = float(np.log(exp_x.sum()))
    L = c_max + log_sum / beta

    score = 2.0 * n * L / (S * S)

    p = exp_x / exp_x.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad_a = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    # Chain rule: grad_w = grad_a * a (since da/dw = a)
    grad_w = grad_a * a

    return score, grad_w


def log_param_optimize(a0, beta, n_iters, lr=0.05, beta1=0.9, beta2=0.999,
                       eps=1e-8, check_interval=200):
    """
    Adam optimizer in log-space: a = exp(w), optimize over w.
    Periodically normalizes a to sum=1 and updates w accordingly.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a, compute_score(a)
    a = a / S

    w = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        t += 1
        sc, grad_w = lse_value_and_grad_w(a, beta)

        if it % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * grad_w * grad_w
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.exp(np.clip(w_new, -30.0, 15.0))
        a = np.clip(a_new, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new
        w = np.log(np.maximum(a, 1e-15))

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc


def exp_weighted_subgrad(a0, n_iters=5000, n_active=10, f_lb=1.0, gamma=5.0):
    """
    Exponentially-weighted multi-peak subgradient descent.
    Uses exponential weighting to focus more on the dominant peak(s).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = a.sum()
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_a * S_a)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_k = np.argsort(c)[::-1][:n_active]
        c_top = c[top_k]

        # Exponential weights: peaks closer to max get more weight
        w_vals = np.exp(-gamma * (c_top - c_top[0]))
        w_sum = w_vals.sum()
        if w_sum < 1e-15:
            break
        w_vals /= w_sum

        # Build weight vector in full convolution space
        w = np.zeros(2 * n - 1)
        w[top_k] = w_vals

        # Subgradient: 2 * (w * rev_a)[n-1 + i]
        rev_a = a[::-1]
        M2 = (2 * n - 1) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        W = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_result = np.fft.irfft(W * RA, M2_fft)[:M2]
        sg = 2.0 * conv_result[n - 1:n - 1 + n]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = min((sc - f_lb) / (gnorm * gnorm + 1e-30), 0.4)
        step = max(step, 1e-6)

        a = np.clip(a - step * grad, 0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def nesterov_subgrad(a0, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on exact objective."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def targeted_peak_flattening(a0, n_sweeps=8, n_top=25, n_bottom=25):
    """
    Targeted coordinate descent: reduce top contributors to peak,
    redistribute to bottom contributors proportionally.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        L = len(contrib)
        top_count = min(n_top, max(1, L // 4))
        bot_count = min(n_bottom, max(1, L // 4))

        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]
        bot_sum = contrib[bot_local].sum()
        if bot_sum < 1e-15:
            break
        bot_weights = contrib[bot_local] / bot_sum

        improved = False
        for li in top_local:
            gi = j_start + li
            old_val = a[gi]
            if old_val < 1e-10:
                continue

            best_local_sc = best_sc
            best_snapshot = None

            for frac in [0.4, 0.6, 0.75, 0.85, 0.95]:
                transfer = old_val * (1.0 - frac)
                a[gi] = old_val * frac
                a[j_start + bot_local] += transfer * bot_weights

                S_cur = a.sum()
                if S_cur > 1e-12:
                    a = a / S_cur
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_snapshot = a.copy()

            if best_snapshot is not None and best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = best_snapshot.copy()
                a = best_snapshot
                improved = True
            else:
                a[gi] = old_val
                a[j_start + bot_local] -= transfer * bot_weights
                S_cur = a.sum()
                if S_cur > 1e-12:
                    a = a / S_cur

        if not improved:
            break

    return best_a


def perturbed_restart(a0, rng, scale=0.03):
    """Generate a structure-aware perturbation."""
    n = len(a0)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a0 * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a0 * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Peak-targeted: reduce at peak contributors
        c = fast_conv(a0)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a0.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 20)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random() - 0.05)
        # Boost at low-contribution positions
        bot_local = np.argsort(contrib)[:top_count]
        for li in bot_local:
            a_p[j_start + li] *= (1.0 + 0.1 * rng.random())
    elif strategy == 3:
        # Multiplicative random
        a_p = a0 * np.exp(scale * 1.5 * rng.standard_normal(n))
    else:
        # Block scaling
        n_blocks = 5 + int(rng.integers(0, 12))
        block_size = max(1, n // n_blocks)
        a_p = a0.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * (1.0 / n))
    inits.append(np.maximum(0.01, 1.0 + 0.3 * np.cos(2.0 * np.pi * x)))
    inits.append(1.0 + 0.5 * (2.0 * x - 1.0) ** 2)
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2))
    inits.append(0.5 + 0.5 * np.cos(np.pi * x))
    inits.append(np.maximum(0.01, 1.0 + rng.standard_normal(n) * 0.3))
    return inits


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lbfgs_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Adam + log-param multi-start (15% time)
    Phase 2: L-BFGS with increasing beta (45% time)
    Phase 3: Iterated local search with Adam/subgrad (20% time)
    Phase 4: Final polishing (20% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * (1.0 / 1024)

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        n_work = len(parent)
        parent_norm = parent / parent.sum()
        sc = compute_score(parent_norm)
        print(f"Parent: n={n_work}, score={sc:.10f}")
        report(sc, parent_norm, "parent")
    else:
        n_work = 8192
        best_a = np.ones(n_work) / n_work
        best_score = compute_score(best_a)
        print(f"Initialized uniform: n={n_work}, score={best_score:.10f}")

    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Multi-start with Adam and log-param (15% time)
    # ============================================================
    phase1_deadline = t_start + 0.15 * effective_budget
    print(f"\n--- Phase 1: Adam Multi-Start (n={n_work}) ---")

    inits = build_initializations(n_work, rng, parent)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
        S = a_init.sum()
        if S < 1e-12:
            continue
        a_init = a_init / S

        sc = compute_score(a_init)
        report(sc, a_init, f"init_{i}")

        # Standard Adam
        remaining = phase1_deadline - time.monotonic()
        if remaining > 30:
            n_iters = int(min(4000, max(1000, remaining * 8)))
            a_opt, sc_adam = adam_optimize(a_init, beta=50, n_iters=n_iters, lr=0.05)
            report(sc_adam, a_opt, f"adam_{i}")

            # Log-param Adam from the best
            if sc_adam < best_score:
                a_lp, sc_lp = log_param_optimize(a_opt, beta=100, n_iters=2000, lr=0.03)
                report(sc_lp, a_lp, f"logadam_{i}")

    print(f"Phase 1 done: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: L-BFGS with increasing beta (45% time)
    # ============================================================
    phase2_deadline = t_start + 0.60 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    beta_schedule = [50, 100, 200, 500, 1000, 3000, 10000, 30000, 100000, 300000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 90:
            break

        maxiter = int(min(50000, max(3000, remaining * 25)))
        maxfun = int(min(300000, max(15000, remaining * 80)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS_b={beta}")

        # If significantly better, do a perturbed restart at same beta
        if time.monotonic() < phase2_deadline - 90 and beta >= 1000:
            a_pert = perturbed_restart(best_a, rng, scale=0.008)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS_b={beta}_pert")

    print(f"Phase 2 done: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Iterated local search (20% time)
    # ============================================================
    phase3_deadline = t_start + 0.80 * effective_budget
    print(f"\n--- Phase 3: Iterated Local Search ---")

    trial = 0
    while time.monotonic() < phase3_deadline:
        remaining = phase3_deadline - time.monotonic()
        if remaining < 180:
            break

        scale = 0.008 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Adam with moderate beta
        n_iters = int(min(2000, max(500, remaining * 3)))
        a_adam, sc_adam = adam_optimize(a_pert, beta=200, n_iters=n_iters, lr=0.02)
        report(sc_adam, a_adam, f"ILS{trial}_adam")

        # Subgradient polish
        a_sg = exp_weighted_subgrad(a_adam, n_iters=1500, n_active=8, gamma=5.0)
        sc_sg = compute_score(a_sg)
        report(sc_sg, a_sg, f"ILS{trial}_sg")

        # Pick best from this trial
        if sc_adam <= sc_sg:
            a_trial_best = a_adam
        else:
            a_trial_best = a_sg

        # L-BFGS refinement on trial best
        if time.monotonic() < phase3_deadline - 60:
            sc, a_opt = run_lbfgs(a_trial_best, 5000, maxiter=15000, maxfun=40000)
            report(sc, a_opt, f"ILS{trial}_lbfgs")

        trial += 1

    print(f"Phase 3 done: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Final polishing (20% time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Final Polishing ---")

    # Very high beta L-BFGS
    for beta in [50000, 100000, 300000, 500000]:
        if time.monotonic() > phase4_deadline - 60:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 90:
            break
        sc, a_opt = run_lbfgs(best_a, beta,
                              maxiter=int(remaining * 15),
                              maxfun=int(remaining * 50))
        report(sc, a_opt, f"final_b={beta}")

    # Targeted peak flattening
    if time.monotonic() < phase4_deadline - 90:
        a_tpf = targeted_peak_flattening(best_a, n_sweeps=8, n_top=30, n_bottom=30)
        sc_tpf = compute_score(a_tpf)
        report(sc_tpf, a_tpf, "peak_flat")

        # Subgradient after flattening
        a_sg2 = exp_weighted_subgrad(best_a, n_iters=3000, n_active=12, gamma=3.0)
        report(compute_score(a_sg2), a_sg2, "sg_final")

    # Nesterov on the absolute best
    if time.monotonic() < phase4_deadline - 60:
        a_ns = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
        report(compute_score(a_ns), a_ns, "nesterov_final")

    # One more high-beta L-BFGS pass
    if time.monotonic() < phase4_deadline - 40:
        sc, a_opt = run_lbfgs(best_a, 1000000, maxiter=20000, maxfun=60000)
        report(sc, a_opt, "ultra_final")

    # Log-param final push
    if time.monotonic() < phase4_deadline - 30:
        a_lp, sc_lp = log_param_optimize(best_a, beta=5000, n_iters=2000, lr=0.01)
        report(sc_lp, a_lp, "log_final")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
