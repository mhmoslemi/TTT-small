"""
High-beta LSE optimizer with multi-restart, two-point coordinate descent,
and L-BFGS polish for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def normalize(a):
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_lse_refined(a, max_time, beta_start=50.0, beta_end=100000.0,
                          lr_start=0.01, momentum=0.9):
    """Nesterov-accelerated LSE with backtracking line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    beta = beta_start
    lr = lr_start
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * 1.003, beta_end)
        
        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        a_look = normalize(a_look)
        
        sc_look, grad = lse_score_and_grad(a_look, beta)
        
        # Backtracking line search
        step = lr
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        sc_new = lse_score_and_grad(a_new, beta)[0]
        
        backtracks = 0
        while sc_new > sc_look and backtracks < 10:
            step *= 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            sc_new = lse_score_and_grad(a_new, beta)[0]
            backtracks += 1
        
        v = momentum * v - step * grad
        a = a_new
        
        # Track best (exact score)
        if it % 20 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()
            lr *= 0.9995
        
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    
    return best_a


def two_point_cd(a, n_sweeps=3, top_k=50):
    """Two-point coordinate descent targeting peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find the peak
        k_star = int(np.argmax(c))
        
        # Find top contributing pairs (i, k_star - i) for c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # Contribution of pair (j, k_star-j) is a[j] * a[k_star-j]
        j_max = max(0, k_star - n + 1)
        j_min = min(k_star, n - 1)
        
        pairs = []
        for j in range(j_max, j_min + 1):
            j2 = k_star - j
            if j2 != j:
                pairs.append((j, j2, 2.0 * a[j] * a[j2]))
            else:
                pairs.append((j, j2, a[j] * a[j2]))
        
        pairs.sort(key=lambda x: -x[2])
        
        for j1, j2, _ in pairs[:top_k]:
            if j1 >= n or j2 >= n:
                continue
            if j1 == j2:
                # Single variable: try scaling a[j1]
                old = a[j1]
                for factor in [0.8, 0.9, 1.1, 1.2]:
                    a[j1] = old * factor
                    a_new = normalize(a)
                    sc = compute_score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new
                    else:
                        a[j1] = old
                a[j1] = old
            else:
                # Two variables: try transferring mass between them
                old1, old2 = a[j1], a[j2]
                # c[k_star] contains 2*a[j1]*a[j2], we want to reduce this
                # Try: a[j1] = old1 + t, a[j2] = old2 - t (or vice versa)
                # But we need to consider the effect on OTHER c[k] values too
                # Simple approach: try a few transfers
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    # Transfer from larger to smaller (to reduce product)
                    if old1 > old2:
                        t = frac * (old1 - old2)
                        a[j1] = old1 - t
                        a[j2] = old2 + t
                    else:
                        t = frac * (old2 - old1)
                        a[j1] = old1 + t
                        a[j2] = old2 - t
                    a_new = normalize(a)
                    sc = compute_score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new
                    else:
                        a[j1] = old1
                        a[j2] = old2
        
        if best_sc < 1.503:
            break
    
    return best_a


def fienup_refined(a, n_iters=200, thresh_start=2.0, thresh_end=1.05,
                   clip_start=0.3, clip_end=0.8, momentum_end=0.4):
    """Fienup with momentum for spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t
        momentum = momentum_end * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh
        if above.any():
            scale = np.ones_like(P)
            target = np.sqrt(P_mean * thresh)
            current = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target / np.maximum(current, 1e-10))
            scale[above] = np.clip(factor, 0.01, 1.0)
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a_candidate = a + v
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = normalize(a_candidate)
        a = a_candidate
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def run_lbfgs(a0, beta, maxiter=3000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a0 = normalize(a0)
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):

            objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,

        a_out = np.clip(res.x, 0.0, 1000.0)
        a_out = normalize(a_out)
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0

Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses high-beta LSE with backtracking, subgradient averaging, two-point CD, and L-BFGS.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def normalize(a):
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_lse_bt(a, max_time, beta_start=100.0, beta_end=200000.0,
                     lr_start=0.005, momentum=0.9, beta_growth=1.005):
    """Nesterov LSE with backtracking line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    beta = beta_start
    lr = lr_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * beta_growth, beta_end)
        
        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        a_look = normalize(a_look)
        
        sc_look, grad = lse_score_and_grad(a_look, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        a_try = normalize(a_try)
        sc_try = lse_score_and_grad(a_try, beta)[0]
        
        bt = 0
        while sc_try > sc_look and bt < 15:
            step *= 0.5
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try)
            sc_try = lse_score_and_grad(a_try, beta)[0]
            bt += 1
        
        v = momentum * v - step * grad
        a = a_try
        
        # Track best

            sc_exact = compute_score(a)
            if sc_exact < best_sc - 1e-10:

                best_a = a.copy()
                no_improve = 0

        best_a = a.copy()
        best_sc = sc_final

def subgradient_averaging(a, max_time, lr_start=0.01, lr_decay=0.999):
    """Subgradient method with Polyak averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    a_avg = a.copy() * 1.0
    total_weight = 1.0
    lr = lr_start
    t_start = time.monotonic()
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Subgradient of max(c) w.r.t. a
        # d(c[k*])/da[i] = 2*a[k*-i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Adjust for the normalization constraint (sum(a) = 1)
        # Project gradient onto the tangent space of the simplex
        grad = grad - np.mean(grad)
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        a = a_new
        
        # Polyak average
        weight = float(it)
        a_avg = (total_weight * a_avg + weight * a) / (total_weight + weight)
        total_weight += weight
        
        # Track best
        if it % 50 == 0:
            sc = compute_score(a_avg)
            if sc < best_sc:
                best_sc = sc
                best_a = a_avg.copy()
        
        lr *= lr_decay
    
    sc_final = compute_score(a_avg)
    if sc_final < best_sc:
        best_a = a_avg.copy()
        best_sc = sc_final
    
    return best_a


def two_point_cd(a, n_sweeps=3, top_k=50, max_time=None):
    """Two-point coordinate descent targeting peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if max_time and time.monotonic() - t_start > max_time:
            break
            
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find contributing pairs
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        contributions = []
        for j in range(j_start, j_end + 1):
            j2 = k_star - j
            if j2 != j:
                contributions.append((j, j2, 2.0 * a[j] * a[j2]))
            else:
                contributions.append((j, j2, a[j] * a[j2]))
        
        contributions.sort(key=lambda x: -x[2])
        
        for j1, j2, _ in contributions[:top_k]:
            if max_time and time.monotonic() - t_start > max_time:
                break
            if j1 >= n or j2 >= n:
                continue
            
            if j1 == j2:
                old = a[j1]
                if old < 1e-10:
                    continue
                improved = False
                for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                    a[j1] = old * factor
                    a_new = normalize(a)

                    continue

                    if old1 >= old2:

                        a[j1] = old1 - t
                        a[j2] = old2 + t
                    else:

                        a[j1] = old1 + t

                    sc = compute_score(a_new)

                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        break
                if not improved:
                    a[j1] = old1
                    a[j2] = old2
    
    return best_a
