```python
def solve_lp_for_g(f, M):
    """Solve max sum(b) s.t. (f*b)_k <= M for all k, b >= 0"""
    n = len(f)
    # Variables: b_0, ..., b_{n-1}
    # Constraints: for k = 0, ..., 2n-2: sum_{j: valid} f_j * b_{k-j} <= M
    #              b_j >= 0
    
    # (f*b)_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f_j * b_{k-j}
    #          = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f_{k-i} * b_i  (substituting i=k-j)
    
    # Let's build the constraint matrix A where A[k, i] = f_{k-i} if valid, else 0
    # Then the constraint is A @ b <= M * ones(2n-1)
    
    # For large n, this matrix is sparse (band-like)
    # We can use scipy's sparse solvers or cvxpy
    
    from scipy.optimize import linprog
    from scipy.sparse import sparse
    
    # c = -ones(n) (we want to maximize sum, so minimize -sum)
    c = -np.ones(n)
    
    # Build A matrix (sparse)
    # A[k, i] = f[k-i] if 0 <= k-i < n and 0 <= i < n, else 0
    # i.e., A[k, i] = f[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
    
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_vec = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    return res.x
```

```python
Combines LP projection (state-of-the-art) with GS-Fienup, L-BFGS, and subgradient.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S*S)

def solve_lp(f, M_val):
    """Solve max sum(b) s.t. (f*b)_k <= M_val, b >= 0."""
    n = len(f)
    # Build sparse constraint matrix
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M_val)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 60})
    if res.success:
        return res.x
    return None

def lp_iteration(f, t_blend=0.3, lp_time_limit=60):

    f = np.maximum(f.astype(np.float64), 0.0).copy()
    S = float(f.sum())
    if S < 1e-12:
        return f
    f = f / S
    
    b_conv = fast_conv(f)
```

```python
"""
LP-based iterative optimization for minimizing max(self-convolution) score.
Primary innovation: Uses LP projection (state-of-the-art) to find optimal blending direction,
combined with GS-Fienup spectral shaping, L-BFGS with LSE smoothing, and subgradient polishing.
Multi-length search to exploit the 2n/(2n-1) theoretical lower bound at different sizes.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_sparse_conv_matrix(f):
    """Build sparse matrix A where A[k,i] = f[k-i] if 0<=k-i<n and 0<=i<n."""
    n = len(f)
    n_k = 2 * n - 1
    col_idx = []
    row_idx = []
    data = []
    for i in range(n):
        k_end = min(i + n - 1, 2 * n - 2)
        ks = np.arange(i, k_end + 1)
        col_idx.append(np.full(len(ks), i, dtype=np.int32))
        row_idx.append(ks.astype(np.int32))
        data.append(np.arange(len(ks), dtype=np.float64))
    # data gives us f[0], f[1], ... for each column
    row_flat = np.concatenate(row_idx)
    col_flat = np.concatenate(col_idx)
    # For each entry, value = f[k - i]
    vals = f[row_flat - col_flat]
    return csr_matrix((vals, (row_flat, col_flat)), shape=(n_k, n))


def solve_lp_for_g(f, M_val, time_limit=30):
    """Solve max sum(b) s.t. (f*b)_k <= M_val for all k, b >= 0."""
    n = len(f)
    A = build_sparse_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_blend_step(f, t_blend=0.3, time_limit=20):
    """One step of LP-based iteration: find g via LP, blend with f."""
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    S = float(f.sum())
    if S < 1e-12:
        return f
    f = f / S

    b_conv = fast_conv(f)
    M_val = float(b_conv.max())

    g = solve_lp_for_g(f, M_val, time_limit=time_limit)
    if g is None:
        return f

    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return f

    # Rescale g to same sum as f
    g = g * (S / g_sum)

    # Blend
    f_new = (1.0 - t_blend) * f + t_blend * g
    f_new = np.maximum(f_new, 0.0)
    S_new = float(f_new.sum())
    if S_new < 1e-12:
        return f
    f_new = f_new / S_new
    return f_new


def gs_fienup(a, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with correct flat-convolution target."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_negative_redistribute(a, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """GS variant: instead of clipping negatives to 0, redistribute them proportionally."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_raw = np.fft.irfft(A_new, M)[:n]

        # Redistribution: add negative mass to positive elements proportionally
        neg_sum = float(np.sum(a_raw[a_raw < 0]))
        pos_mask = a_raw >= 0
        if neg_sum < 0 and np.any(pos_mask):
            a_redist = a_raw.copy()
            a_redist[pos_mask] += (-neg_sum) * a_redist[pos_mask] / max(float(a_redist[pos_mask].sum()), 1e-12)
            a_new = np.maximum(a_redist, 0.0)
        else:
            a_new = np.maximum(a_raw, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum_exps = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum_exps / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a_j: sum_k p_k * d(b_k)/da_j
    # d(b_k)/da_j = a_{k-j} + a_{j-k} (for valid indices), but since b is symmetric:
    # Actually b_k = sum_i a_i * a_{k-i}, so db_k/da_j = a_{k-j} (if valid) + a_{j-k} (if valid)
    # For the LSE gradient: grad_L_j = sum_k p_k * (a_{k-j} + a_{j-k} * [j!=k-j])
    # Simpler: convolve p with a (reversed)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapped(x, beta):
    """Wrapper for scipy minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=300):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    t_start = time.monotonic()

    class TimeLimitCB:
        def __init__(self):
            self.stop = False
        def __call__(self, xk):
            if time.monotonic() - t_start > time_limit:
                self.stop = True
                return True  # Stop
            return self.stop

    cb = TimeLimitCB()
    try:
        res = minimize(
            lse_wrapped, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=cb,
            options={'ftol': 1e-14, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_iterate(a, n_iters=5000, top_m=10, step_init=0.03, decay=0.999):
    """Subgradient method using top-M peaks with Polyak step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = min(step * decay, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a

            x_new = np.linspace(0, 1, n_gs)
            a_init = np.interp(x_new, x_old, best_a).copy()

        a_init = np.maximum(a_init, 0.01)
        a_init = a_init / a_init.sum()

        # Multiple GS variants
        a1 = gs_fienup(a_init, n_iters=300, alpha_init=0.6, alpha_final=0.05)
        update_best(compute_score(a1), a1, f"GS n={n_gs} v1")

        a2 = gs_negative_redistribute(a1, n_iters=200, alpha_init=0.4, alpha_final=0.05)
        update_best(compute_score(a2), a2, f"GS-redist n={n_gs} v2")

        # Try with different initialization
        a_rand = np.maximum(a_init + 0.1 * rng.standard_normal(n_gs), 0.01)
        a_rand = a_rand / a_rand.sum()
        a3 = gs_fienup(a_rand, n_iters=200, alpha_init=0.7, alpha_final=0.1)
        update_best(compute_score(a3), a3, f"GS-rand n={n_gs}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (deadline t={phase3_end-t_start:.0f}s) ===")

    n_best = len(best_a)
    print(f"  L-BFGS at n={n_best}")

    betas = [50, 100, 300, 500, 1000, 3000, 5000, 10000, 30000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        t_beta_start = time.monotonic()
        remaining = phase3_end - time.monotonic()
        tl = min(120.0, remaining * 0.3)
        if tl < 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000, time_limit=tl)
        update_best(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    beta={beta}: score={sc:.8f}, time={time.monotonic()-t_beta_start:.1f}s")

    # L-BFGS from a slightly perturbed best
    if time.monotonic() < phase3_end - 60:
        a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(len(best_a)))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=3000, maxfun=50000, time_limit=60)
        update_best(sc, a_opt, "LBFGS-perturb")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient + final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Subgradient + polish ===")

    n_best = len(best_a)
    for top_m in [5, 10, 20, 50]:
        if time.monotonic() > deadline - 30:
            break
        a_sg = subgradient_iterate(best_a, n_iters=3000, top_m=top_m, step_init=0.02)
        update_best(compute_score(a_sg), a_sg, f"SG top_m={top_m}")

    # Final GS polish
    if time.monotonic() < deadline - 30:
        a_final = gs_fienup(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.02)
        update_best(compute_score(a_final), a_final, "FINAL-GS")

    # Final L-BFGS with high beta (very sharp peak detection)
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=2000, maxfun=30000, time_limit=30)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final: try a couple more creative initializations if time remains
    # ==================================================================
    if time.monotonic() < deadline - 40:
        print(f"\n=== Bonus: Creative initializations ===")
        # Try a sequence based on binomial coefficients (approximates flat conv)
        n_try = len(best_a)
        # Central binomial-like
        k_max = min(n_try - 1, 200)
        from math import comb
        binom_vals = []
        for j in range(n_try):
            d = abs(j - n_try // 2)
            if d <= k_max:
                binom_vals.append(comb(2 * d, d) / (4 ** d))
            else:
                binom_vals.append(0.01)
        a_binom = np.array(binom_vals, dtype=np.float64)
        a_binom = np.maximum(a_binom, 0.0)
        a_binom = a_binom / a_binom.sum()
        sc = compute_score(a_binom)
        update_best(sc, a_binom, "binom-init")
        if sc < best_score + 0.1:
            a_ref = gs_fienup(a_binom, n_iters=200, alpha_init=0.7, alpha_final=0.1)
            update_best(compute_score(a_ref), a_ref, "binom+GS")

    # Try LDPC-like structured sequence
    if time.monotonic() < deadline - 20:
        n_try = len(best_a)
        a_struct = np.zeros(n_try)
        for j in range(n_try):
            a_struct[j] = 1.0 + 0.5 * np.cos(2 * np.pi * j * 7.0 / n_try) + \
                         0.3 * np.cos(2 * np.pi * j * 13.0 / n_try)
        a_struct = np.maximum(a_struct, 0.0)
        a_struct = a_struct / a_struct.sum()
        sc = compute_score(a_struct)
        update_best(sc, a_struct, "structured-init")

    # ==================================================================
    # Final verification and output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
```
