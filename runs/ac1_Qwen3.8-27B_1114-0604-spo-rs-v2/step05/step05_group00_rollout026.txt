def run_adaptive_lbfgs(a0, betas, maxiter_per_beta=5000):
    """Run L-BFGS with annealing over beta values."""
    n = len(a0)
    a = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for beta in betas:
        res = minimize(
            lse_and_grad, a, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 20, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new  # Continue from improved point
        # If L-BFGS didn't improve, try normalizing and continuing
        a = normalize(a_new)
    
    return best_sc, best_a

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        if mode == 'flatten':
            alpha = 0.3 + 0.5 * (it / n_iters)
            P_target = np.full_like(P, P_mean)
            P_blend = (1 - alpha) * P + alpha * P_target
        elif mode == 'quantile':
            q = 0.8 + 0.15 * (it / n_iters)
            thresh = np.quantile(P, q)
            P_blend = P.copy()
            above = P > thresh
            P_blend[above] = thresh
        else:
            P_blend = P.copy()

        A_new = A * np.sqrt(P_blend / np.maximum(P, 1e-30))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def simultaneous_peak_shave(a, n_iters=3000, top_k=10, step_init=0.01):
    """Reduce top-K peaks simultaneously using subgradient method."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if sc - 2.0 * n / (2.0 * n - 1) < 1e-6:
            break
        
        # Find top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Build subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            weight = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += weight * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        # Average weight
        avg_w = float(np.mean([c[int(k)] / (M_val + 1e-30) for k in top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, 0.5 * (sc - 2.0 * n / (2.0 * n - 1)) / (gn * gn + 1e-30))
        step = max(step, 1e-10)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        step *= 0.9995
    
    return best_a

def coordinate_line_search(a, n_sweeps=10, n_cand=10):
    """Coordinate descent with line search on top contributors to peak."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        
        # Focus on top contributors
        top_count = min(50, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            
            # Try reducing a[j] (since reducing peak is the goal)
            factors = np.linspace(0.1, 0.95, n_cand)
            best_val = a_old
            best_local = best_sc
            
            for f in factors:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            
            a[j] = best_val
            a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def perturb_and_optimize(a, rng, beta, maxiter=3000):
    """Perturb the sequence and run L-BFGS."""
    n = len(a)
    # Random perturbation
    scale = 0.01 + 0.02 * rng.random()
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    a_pert = np.maximum(a_pert, 0.0)
    a_pert = a_pert / a_pert.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_and_grad, a_pert, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

"""
Adaptive LSE-annealing optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses: spectrum-flattening (Fienup), multi-beta LSE annealing with L-BFGS,
simultaneous peak shaving via subgradient, coordinate descent with line search,
and multi-scale n growth with interpolation. Focuses on squeezing the peak
autocorrelation below 1.503.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score_LSE = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    # Softmax weights
    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a:
    # dL/da_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * conv(p, a_reversed)[n-1+j]
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1  # length of linear conv of p (len 2n-1) and a_rev (len n)
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    # Full gradient of score = 2n * L / S^2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * 1.0 / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs_single(a0, beta, maxiter=5000, maxfun=100000):
    """Run a single L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def lse_anneal(a0, betas, maxiter_per_beta=5000):
    """
    Anneal through a sequence of beta values, always continuing from the best point.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_sc = compute_score(a)
    best_a = a.copy()

    for beta in betas:
        sc, a_new = run_lbfgs_single(a, beta, maxiter=maxiter_per_beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = normalize(a_new)

    return best_sc, best_a


def fienup_flat(a, n_iters=500):
    """Fienup: iteratively flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Gradually increase flattening strength
        alpha = 0.15 + 0.7 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        # Update amplitude, preserve phase
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=500):
    """Fienup with quantile-based spectral clipping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Tightening quantile threshold
        q = 0.70 + 0.25 * progress
        thresh = np.quantile(P, q) * (1.0 + 0.5 * (1.0 - progress))

        P_clipped = P.copy()
        above = P_clipped > thresh
        P_clipped[above] = thresh

        ratio = np.sqrt(P_clipped / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_hybrid(a, n_iters=500):
    """Alternate between flatten and quantile modes."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if it % 3 == 0:
            # Flatten mode
            alpha = 0.2 + 0.5 * progress
            P_target = np.full_like(P, P_mean)
            P_new = (1.0 - alpha) * P + alpha * P_target
        elif it % 3 == 1:
            # Quantile mode
            q = 0.75 + 0.2 * progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        else:
            # Inverse: boost low frequencies
            alpha = 0.1 + 0.3 * progress
            P_low = P < P_mean * 0.5
            P_new = P.copy()
            P_new[P_low] = P[P_low] / (1.0 + alpha)

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave_subgrad(a, n_iters=5000, top_k=8, step_init=0.008, decay=0.9997):
    """
    Subgradient descent targeting top-K autocorrelation peaks simultaneously.
    WithPolyak step size.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-7:
            break

        # Find top-K peaks and build subgradient
        top_idx = np.argsort(c)[::-1][:top_k]

        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with cap
        step_polyak = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.03)
        step = max(step, 1e-10)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def peak_shave_focused(a, n_iters=3000):
    """
    Focused peak shaving: add mass to off-peak positions and remove from peak contributors.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Identify contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution of each pair to the peak
        js = np.arange(j_start, j_end + 1)
        pairs = a[js] * a[k_star - js]
        total_pair = pairs.sum()
        if total_pair < 1e-15:
            break

        # Reduce top contributors, increase non-contributors
        n_reduce = max(5, n // 50)
        top_contrib = np.argsort(pairs)[::-1][:n_reduce]
        reduce_idx = j_start + top_contrib

        # Find positions that don't contribute to the peak
        peak_set = set(js.tolist())
        non_peak = np.array([i for i in range(n) if i not in peak_set])
        if len(non_peak) < 10:
            return best_a
        n_add = min(len(non_peak), n_reduce * 2)
        add_candidates = non_peak[rng_choice(non_peak, n_add)]

        delta = 0.001 / n  # Small transfer amount
        for j in reduce_idx:
            if a[j] > delta * 2:
                a[j] -= delta
        for j in add_candidates:
            a[j] += delta

        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a


def rng_choice(arr, k):
    """Randomly select k elements without replacement."""
    if k >= len(arr):
        return arr
    idx = np.random.choice(len(arr), k, replace=False)
    return arr[idx]


def coord_descent(a, n_sweeps=10, n_top=40):
    """
    Coordinate descent: reduce top contributors to the peak lag.
    Uses multiplicative factors with line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue

            # Try reducing
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using cubic-ish interpolation."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def perturb_rng(a, rng, scale=0.02):
    """Additive Gaussian perturbation."""
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)


def perturb_mult(a, rng, scale=0.03):
    """Multiplicative perturbation with low-frequency structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 6):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)

            if len(add_idx) > n_add:
                add_idx = rng.choice(add_idx, n_add, replace=False)
            delta = 0.002 / len(a_pert)
            a_pert[add_idx] += delta
            a_pert = normalize(a_pert)

        maxiter = max(500, int(remaining * 6))
        beta = 500 + r * 400
        sc, a_opt = run_lbfgs_single(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 6: Larger-n Exploration (8% time)
    # ==================================================================
    phase6_end = t_start + 0.83 * (budget_s - 25.0)
    print(f"\n=== Phase 6: Larger-n Exploration ===")

    current_n = len(best_a)
    target_ns = []
    if current_n < 10000:
        target_ns.append(10000)
    if current_n < 15000:
        target_ns.append(15000)
    if current_n < 20000:
        target_ns.append(20000)

    for new_n in target_ns:
        if time.monotonic() > phase6_end - 30:
            break
        remaining = phase6_end - 30 - time.monotonic()
        if remaining < 30:
            break

        a_large = interpolate_sequence(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  Interpolated to n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        # LSE on the larger sequence
        if time.monotonic() < phase6_end - 15:
            maxiter = max(500, int((phase6_end - 15 - time.monotonic()) * 5))
            sc2, a2 = run_lbfgs_single(a_large, 3000, maxiter=maxiter, maxfun=maxiter * 20)
            update(sc2, a2, f"INTERP+LSE n={new_n}")

            # If improved, try Fienup on the larger seq
            if sc2 < best_score and time.monotonic() < phase6_end - 10:
                a_f = fienup_flat(a2, n_iters=100)
                sc3 = compute_score(a_f)
                update(sc3, a_f, f"INTERP+FIENUP n={new_n}")

    # ==================================================================
    # Phase 7: Final Polish (12% time)
    # ==================================================================
    print(f"\n=== Phase 7: Final Polish ===")

    # High-beta LSE
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = max(1000, int(remaining * 8))
        sc, a_final = run_lbfgs_single(best_a, 20000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_final, "FINAL-LSE-20k")

    if time.monotonic() < deadline - 20:
        maxiter = max(500, int((deadline - 20 - time.monotonic()) * 6))
        sc, a_final = run_lbfgs_single(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-50k")

    # Final peak shave
    if time.monotonic() < deadline - 10:
        remaining_iters = max(500, int((deadline - 10 - time.monotonic()) * 3000))
        a_ps = peak_shave_subgrad(best_a, n_iters=min(remaining_iters, 3000), top_k=5,
                                   step_init=0.003)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd = coord_descent(best_a, n_sweeps=5, n_top=20)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
