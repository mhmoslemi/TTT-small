"""

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr_active(a, w):
    """Compute sum_j w[j] * a[j-i] for i=0..n-1 (subgradient building block)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient for top-k active positions."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr_active(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on active-set LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _nesterov_subgrad(a, deadline, max_iters=2000, active_frac=0.05, verbose=False):
    """
    Nesterov-accelerated subgradient descent with BB step size.
    Active set = top (active_frac * (2n-1)) positions of the convolution.
    """
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    M = 2 * n - 1
    k_active = max(3, int(active_frac * M))

    a_prev = a.copy()
    t_mom = 1.0
    s_prev = None
    y_prev = None
    best_step = 0.02
    improvement_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        if it == 0:
            a_mom = a.copy()
        else:
            a_mom = a + (t_mom - 1.0) / t_next * (a - a_prev)
            a_mom = np.clip(a_mom, 0.0, None)
        t_mom = t_next

        # Compute subgradient at a_mom
        S = float(a_mom.sum())
        if S < 1e-12:
            break
        c = _self_conv(a_mom)
        d = float(n) * c / (S * S)

        # Active set: top k_active positions
        idx = np.argpartition(d, M - k_active)[M - k_active:]
        d_act = d[idx]
        dmax = float(d_act.max())

        # Uniform weights on active set (exact subgradient of max)
        w_full = np.zeros(M)
        w_full[idx] = 1.0 / k_active

        corr = _cross_corr_active(a_mom, w_full)
        LDS = float(w_full[idx] @ d_act)
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # BB step size update
        s_curr = a - a_prev
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_curr, y_prev))
            ss = float(np.dot(s_curr, s_curr))
            if sy > 1e-12:
                bb1 = ss / sy
                bb2 = sy / ss
                # Blend with previous
                best_step = 0.5 * best_step + 0.5 * np.clip(bb1, 1e-6, 1.0)
                best_step = np.clip(best_step, 1e-5, 0.5 / gnorm)
        else:
            best_step = 0.01 / gnorm

        # Take step from a (not a_mom) for stability
        eta = best_step
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-8:
            a_new = a
        else:
            a_new = a_new / S_new

        sc_new = _score(a_new)

        # Store for BB
        y_curr = grad  # approximate y = g_new - g_old (using same grad as approximation)
        s_prev = s_curr
        y_prev = grad - (y_prev if y_prev is not None else 0.0)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            improvement_count += 1
            if verbose and improvement_count % 50 == 0:
                print(f"    nesterov it={it}: {best_sc:.8f}")
        else:
            # If no improvement, don't update a (keep it for Nesterov)
            a_new = a

        a_prev = a
        a = a_new.copy()

    return best_sc, best_a


def _peak_equalization(a, deadline, max_iters=500, top_m=20):
    """
    Directly target the top peaks: push them down and push non-peaks up
    to maintain sum=1. Uses a coordinated approach.
    """
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    M = 2 * n - 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())

        # Get top m peaks
        idx = np.argsort(d)[-top_m:]
        d_top = d[idx]
        d_avg_top = float(d_top.mean())

        # For each peak position, compute the "pressure" gradient
        # grad_i = (1/m) * sum_{j in top} a[j-i]
        w_full = np.zeros(M)
        w_full[idx] = 1.0 / top_m
        corr = _cross_corr_active(a, w_full)

        # Also compute a "mean field" to push up non-peak regions
        # Use the average of all positions as the target
        w_mean = np.ones(M) / M
        corr_mean = _cross_corr_active(a, w_mean)

        # Direction: reduce peaks, increase elsewhere
        # grad direction = corr (pushes elements that contribute to peaks down)
        # We want to move in -corr direction for peak reduction
        # But also want to maintain sum, so subtract the mean contribution
        grad = corr - (corr @ a) / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Try multiple step sizes
        best_step = None
        best_new_sc = best_sc
        for scale in (0.001, 0.005, 0.01, 0.03, 0.08):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        a = best_step
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()

    return best_sc, best_a


def _gs_init(n, rng, n_iters=50):
    """Gerchberg-Saxton style spectral flattening for initialization."""
    Nf = _next_pow2(2 * n - 1)
    a = np.ones(n) / n  # start uniform

    for it in range(n_iters):
        A = np.fft.rfft(a, Nf)
        # Target: flat magnitude spectrum (keep DC)
        mag = np.abs(A)
        mag_target = np.full_like(mag, mag[0] / np.sqrt(n))
        mag_target[0] = mag[0]  # preserve DC (sum)
        # Blend
        alpha = 0.5
        mag_new = (1 - alpha) * mag + alpha * mag_target
        # Preserve phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Nf)[:n]
        a_new = np.clip(a_new, 0.0, None)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S

    return a


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig

        if not scored:

        # Pick top 2 diverse inits
        selected = scored[:2]
        print(f"  Inits: {[f'{s:.6f} ({nm})' for s, nm, _ in selected]}")

        # Optimize each selected init
        n_sel = len(selected)
        sel_time = (cfg_deadline - time.monotonic()) / n_sel

        for si, (sc_init, name, arr_init) in enumerate(selected):
            sel_deadline = time.monotonic() + sel_time - 5
            if time.monotonic() > sel_deadline:
                continue

            sc, seq = _optimize_full(arr_init, sel_deadline, f'cfg{ci}_{name}', rng)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"  *** Global best: {best_score:.8f} (n={len(seq)}) ***")

        # After optimizing, if we have parent and this is the parent-length config,
        # also try a quick GS + optimize
        if ci == 0 and time.monotonic() < cfg_deadline - 60:
            gs_a = _gs_init(target_n, rng, n_iters=100)
            sc_gs, seq_gs = _optimize_full(gs_a, cfg_deadline - 10, 'cfg0_gs', rng)
            if sc_gs < best_score:
                best_score = sc_gs
                best_seq = seq_gs.copy()
                print(f"  *** GS best: {best_score:.8f} ***")

    # === Final global polish ===
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0

    if best_seq is not None:
        # Very high beta LSE
        for k_f, beta_f in [(300, 15000), (600, 40000)]:
            if time.monotonic() > final_deadline - 30:
                break
            remaining = final_deadline - time.monotonic() - 30
            if remaining < 10:
                break
            sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, remaining * 0.5, 800)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final LSE K={k_f}: {best_score:.8f}")

        # Nesterov subgradient with narrow active set
        if time.monotonic() < final_deadline - 20:
            sc_ns, a_ns = _nesterov_subgrad(
                best_seq, final_deadline - 10, max_iters=3000, active_frac=0.005
            )
            if sc_ns < best_score:
                best_score = sc_ns
                best_seq = a_ns.copy()
                print(f"  Final Nesterov: {best_score:.8f}")

        # Peak equalization
        if time.monotonic() < final_deadline - 10:
            sc_pe, a_pe = _peak_equalization(
                best_seq, final_deadline - 5, max_iters=200, top_m=10
            )
            if sc_pe < best_score:
                best_score = sc_pe
                best_seq = a_pe.copy()
                print(f"  Final peak-eq: {best_score:.8f}")

        # Random restarts
        if time.monotonic() < final_deadline - 40:
            for restart in range(3):
                if time.monotonic() > final_deadline - 15:
                    break
                p = _smooth_perturb(best_seq, rng, strength=0.05)
                chunk_dl = min(final_deadline - 15, time.monotonic() + 20)
                sc_r, a_r = _run_lbfgs(p, 20000, 500, max(3, chunk_dl - time.monotonic()), 300)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  Restart {restart}: {best_score:.8f}")

    # Prepare output
    if best_seq is None:
        best_seq = _norm(np.ones(7000))

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
