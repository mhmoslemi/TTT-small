def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m-1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2*n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A*A, N)[:M]

    a = np.asarray(a, dtype=np.float64)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_i a[i]*w[j+i] for j=0..n-1 (positive-lag cross-correlation)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _compute_grad(a, active_mask, S):
    """Compute 2*_cross_corr(a, w) for the given active mask, scaled properly."""
    n = a.size
    M = 2 * n - 1
    n_act = int(active_mask.sum())
    if n_act == 0:
        return np.zeros(n)
    w = active_mask.astype(np.float64) / n_act
    corr = _cross_corr(a, w)
    return 2.0 * corr


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient for L-BFGS-B."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(np.clip(z, -500, 0))
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a.copy(), method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgradient_nesterov(a0, deadline, max_iters=50000):
    """
    Nesterov-momentum subgradient method with BB step size.
    Minimizes 2n*max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()

    # Nesterov state
    y = a.copy()
    alpha = 1.0
    prev_a = None
    prev_grad = None

    # BB step size state
    bb_step = 0.01

    # Active set tolerance (adaptive)
    act_tol = 1e-3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Every 500 iterations, adjust active set tightness
        if it % 500 == 0 and it > 0:
            act_tol = max(1e-5, act_tol * 0.85)

        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        cur_sc = float(2.0 * dmax)

        # Active set
        active = d >= dmax - act_tol * max(dmax, 1e-10)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        # Compute subgradient
        w = active.astype(np.float64) / n_act
        corr = _cross_corr(a, w)
        grad_main = 4.0 * float(n) * corr / (S * S)

        # LSE value for reference
        LDS = float(w[active] @ d[active])
        grad = grad_main - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        # Barzilai-Borwein step size
        if prev_a is not None and prev_grad is not None:
            s = a - prev_a
            r = grad - prev_grad
            ss = float(s @ s)
            sr = float(s @ r)
            if ss > 1e-20 and abs(sr) > 1e-20:
                bb_candidate = ss / abs(sr)
                bb_candidate = np.clip(bb_candidate, 1e-8, 1e-2)
                bb_step = 0.5 * bb_step + 0.5 * bb_candidate

        # Nesterov momentum
        alpha_new = (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha)) / 2.0
        if prev_a is not None:
            y = a + ((alpha - 1.0) / alpha_new) * (a - prev_a)
        alpha = alpha_new

        # Line search from y
        best_step_val = None
        best_new_sc = cur_sc
        eta_base = bb_step / max(gnorm, 1e-16)

        for scale in (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0):
            eta = eta_base * scale
            q = y - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc_q = _score(q)
            if sc_q < best_new_sc:
                best_new_sc = sc_q
                best_step_val = q.copy()

        if best_step_val is None:
            # Try from a directly (no Nesterov)
            for scale in (0.001, 0.003, 0.01, 0.03, 0.1):
                eta = eta_base * scale
                q = a - eta * grad
                q = np.clip(q, 0.0, None)
                Sq = float(q.sum())
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc_q = _score(q)
                if sc_q < best_new_sc:
                    best_new_sc = sc_q
                    best_step_val = q.copy()

        if best_step_val is None:
            # No improvement; tighten step further
            eta = eta_base * 0.001
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                break
            q = q / Sq
            best_step_val = q
            best_new_sc = _score(q)

        prev_a = a.copy()
        prev_grad = grad.copy()
        a = best_step_val

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()

        # Periodic reporting
        if it % 2000 == 0:
            el = time.monotonic() - t0
            print(f"    [SG-nest] it={it}, score={best_sc:.8f}, n_act={n_act}, "
                  f"gnorm={gnorm:.2e}, bb={bb_step:.2e}, t={el:.0f}s")

    return best_sc, best_a


def _coord_descent(a0, deadline, max_iters=200000):
    """
    Coordinate descent: pick coordinate with largest subgradient,
    do 1D line search to minimize the max convolution.
    """
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        cur_sc = float(2.0 * dmax)

        # Active set (tight)
        active = d >= dmax - 1e-5 * max(dmax, 1e-10)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = active.astype(np.float64) / n_act
        corr = _cross_corr(a, w)
        grad = 2.0 * corr / S  # simplified gradient direction

        # Pick coordinate: largest |grad| with some randomization
        gabs = np.abs(grad)
        # Randomized: top 10% of coordinates
        threshold = np.percentile(gabs, 90)
        candidates = np.where(gabs >= threshold)[0]
        if len(candidates) == 0:
            candidates = np.array([np.argmax(gabs)])
        i = candidates[rng.integers(len(candidates))]

        # 1D line search on coordinate i
        best_delta = 0.0
        best_val = dmax

        # Try several delta values (fraction of a[i])
        steps = np.array([0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0])
        for sign in (-1, 1):
            for s in steps:
                delta = sign * s * max(a[i], 1e-6)
                new_ai = a[i] + delta
                if new_ai < 0:
                    continue
                # Compute new max conv efficiently
                # conv[k] changes by delta * (a[i+k] + a[i-k]) for affected k
                # Full recomputation for accuracy
                a[i] = new_ai
                c_new = _self_conv(a)
                a[i] = a[i] - delta  # restore
                new_max = float(c_new.max())
                if new_max < best_val:
                    best_val = new_max
                    best_delta = delta

        if best_delta != 0.0:
            a[i] += best_delta
            a[i] = max(0.0, a[i])
            S_new = float(a.sum())
            if S_new > 1e-8:
                a = a / S_new
                new_sc = _score(a)
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_a = a.copy()

        if it % 5000 == 0 and it > 0:
            el = time.monotonic() - t0
            print(f"    [CoordDesc] it={it}, score={best_sc:.8f}, t={el:.0f}s")

    return best_sc, best_a


def _smooth_perturb(a, rng, strength=0.1):
    """Apply smooth random perturbation."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _lse_trajectory(a0, deadline, label):
    """Progressive LSE optimization with growing K and beta."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.10f}, n={n}")

    total_time = max(10.0, deadline - t0 - 10.0)
    stages = [
        (5, 50.0, 0.05),

        (50, 500.0, 0.10),

    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + min(cum_time, 1.0) * total_time
        stage_end = min(stage_end, deadline - 5.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        n_calls = min(3, max(1, int(stage_budget / 15)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 1.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 500)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k}, beta={beta:.0f}, call{ci}: {best_sc:.10f}")
            a = _norm(a_new)

    print(f"  [{label}] end: score={best_sc:.10f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _peak_targeted_optimize(a0, deadline):
    """
    Targeted optimization: identify peak positions and use a dedicated
    subgradient method with very tight active set and tiny steps.
    """
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()

    for phase in range(5):
        phase_deadline = min(deadline, t0 + (phase + 1) * (deadline - t0) / 5.0)
        tol = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4][phase]
        step_scales = [
            [0.001, 0.003, 0.01, 0.03, 0.1, 0.3],
            [0.0005, 0.001, 0.003, 0.01, 0.03, 0.1],
            [0.0002, 0.0005, 0.001, 0.003, 0.01, 0.03],
            [0.0001, 0.0003, 0.0005, 0.001, 0.003, 0.01],
            [0.00005, 0.0001, 0.0003, 0.0005, 0.001, 0.003],
        ][phase]

        improved = True
        while improved and time.monotonic() < phase_deadline - 5:
            improved = False
            S = float(a.sum())
            c = _self_conv(a)
            d = float(n) * c / (S * S)
            dmax = float(d.max())
            if dmax <= 0:
                break
            cur_sc = float(2.0 * dmax)

            active = d >= dmax - tol * max(dmax, 1e-12)
            if not active.any():
                active[np.argmax(d)] = True
            n_act = int(active.sum())

            w = active.astype(np.float64) / n_act
            corr = _cross_corr(a, w)
            grad = 4.0 * float(n) * corr / (S * S)
            LDS = float(w[active] @ d[active])
            grad = grad - 4.0 * LDS / S
            grad = np.nan_to_num(grad, nan=0.0)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-16:
                break

            eta_base = 1.0 / (gnorm * gnorm)  # Polyak-like
            best_new_sc = cur_sc
            best_step = None

            for scale in step_scales:
                eta = eta_base * scale
                q = a - eta * grad
                q = np.clip(q, 0.0, None)
                Sq = float(q.sum())
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc_q = _score(q)
                if sc_q < best_new_sc:
                    best_new_sc = sc_q
                    best_step = q.copy()

            if best_step is not None and best_new_sc < best_sc:
                a = best_step
                best_sc = best_new_sc
                best_a = a.copy()
                improved = True
                if phase < 4:
                    print(f"    [PeakTgt] phase={phase}, score={best_sc:.10f}, "
                          f"n_act={n_act}, tol={tol:.0e}")

    print(f"  [PeakTgt] final: score={best_sc:.10f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _spectral_flat_init(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization combining LSE, Nesterov subgradient,
    coordinate descent, and peak-targeted methods.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    print(f"=== Starting search, budget={budget_s}s ===")

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
                print(f"Loaded parent: length={len(parent)}, score={_score(parent):.10f}")
    except Exception:
        pass

    if parent is None:
        print("No parent found, generating initial sequence")
        parent = _spectral_flat_init(8192, rng)

    target_n = len(parent)
    print(f"Working with n={target_n}")

    best_score = _score(parent)
    best_seq = parent.copy()

    # === Phase 1: Quick LSE cleanup (120s) ===
    print(f"\n=== Phase 1: LSE cleanup ===")
    p1_deadline = t_start + 120.0
    sc, seq = _lse_trajectory(best_seq, p1_deadline, "LSE1")
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 1 improved: {best_score:.10f}")

    # === Phase 2: Nesterov subgradient (800s) ===
    print(f"\n=== Phase 2: Nesterov subgradient ===")
    p2_deadline = t_start + 920.0
    sc, seq = _subgradient_nesterov(best_seq, p2_deadline, max_iters=100000)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 2 improved: {best_score:.10f}")

    # === Phase 3: LSE with high beta (400s) ===
    print(f"\n=== Phase 3: High-beta LSE ===")
    p3_deadline = t_start + 1320.0
    sc, seq = _lse_trajectory(best_seq, p3_deadline, "LSE2")
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 3 improved: {best_score:.10f}")

    # === Phase 4: Coordinate descent (400s) ===
    print(f"\n=== Phase 4: Coordinate descent ===")
    p4_deadline = t_start + 1720.0
    sc, seq = _coord_descent(best_seq, p4_deadline, max_iters=100000)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 4 improved: {best_score:.10f}")

    # === Phase 5: Peak-targeted optimization (300s) ===
    print(f"\n=== Phase 5: Peak-targeted ===")
    p5_deadline = t_start + 2020.0
    sc, seq = _peak_targeted_optimize(best_seq, p5_deadline)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 5 improved: {best_score:.10f}")

    # === Phase 6: Nesterov subgradient again (200s) ===
    print(f"\n=== Phase 6: Final Nesterov ===")
    p6_deadline = t_start + 2220.0
    sc, seq = _subgradient_nesterov(best_seq, p6_deadline, max_iters=50000)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Phase 6 improved: {best_score:.10f}")

    # === Phase 7: Final ultra-fine LSE + subgradient (150s) ===
    print(f"\n=== Phase 7: Final polish ===")
    p7_deadline = deadline - 10.0

    # Very high beta LSE
    for k_f, beta_f in [(3000, 30000), (5000, 50000)]:
        if time.monotonic() > p7_deadline - 30:
            break
        remaining = p7_deadline - time.monotonic() - 20
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, remaining * 0.5, 800)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_f}, beta={beta_f}: {best_score:.10f}")

    # Tiny-step subgradient
    if time.monotonic() < p7_deadline - 10:
        a = _norm(best_seq)
        n = len(a)
        for tiny_it in range(200):
            if time.monotonic() > p7_deadline - 5:
                break
            S = float(a.sum())
            c = _self_conv(a)
            d = float(n) * c / (S * S)
            dmax = float(d.max())
            cur_sc = float(2.0 * dmax)

            active = d >= dmax - 1e-6 * max(dmax, 1e-14)
            if not active.any():
                active[np.argmax(d)] = True
            n_act = int(active.sum())
            w = active.astype(np.float64) / n_act
            corr = _cross_corr(a, w)
            grad = 4.0 * float(n) * corr / (S * S)
            LDS = float(w[active] @ d[active])
            grad = grad - 4.0 * LDS / S
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-18:
                break

            for s in (1e-5, 5e-5, 1e-4, 5e-4, 1e-3):
                eta = s / (gnorm * gnorm)
                q = a - eta * grad
                q = np.clip(q, 0.0, None)
                if q.sum() < 1e-8:
                    continue
                q = q / q.sum()
                sc_q = _score(q)
                if sc_q < best_score:
                    best_score = sc_q
                    best_seq = q.copy()
                    a = q.copy()
                    break

    # Perturbation restart
    if time.monotonic() < deadline - 30:
        for restart in range(3):
            if time.monotonic() > deadline - 15:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.02)
            sc_r, a_r = _subgradient_nesterov(p, min(deadline - 10, time.monotonic() + 30), max_iters=5000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.10f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.10f}, verified={verified:.10f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
