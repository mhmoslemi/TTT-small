def pairwise_cd(a, n_sweeps=5, n_top=50):
    """
    Pairwise coordinate descent: try exchanging mass between pairs.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for i in range(len(top_global)):
            for j in range(i + 1, len(top_global)):
                j1 = top_global[i]
                j2 = top_global[j]
                a1, a2 = a[j1], a[j2]
                
                # Try transferring mass from higher to lower
                if a1 < a2:
                    j1, j2 = j2, j1
                    a1, a2 = a2, a1
                
                for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    delta = a1 * frac
                    a[j1] = a1 - delta
                    a[j2] = a2 + delta
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j1] = a1
                        a[j2] = a2
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    a[j1] = a1
                    a[j2] = a2
                    if sc < best_sc:
                        best_sc = sc
                        a[j1] = a1 - delta
                        a[j2] = a2 + delta
                        a = a / a.sum()
                        best_a = a.copy()
    
    return best_a

High-beta LSE optimizer with Adam, Fienup spectral annealing,
softmax peak shaving with Nesterov momentum, and pairwise coordinate descent.
Targets minimization of 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key improvements over previous: higher beta sweep (up to 200k), Adam optimizer,
adaptive temperature softmax peak shaving, and pairwise coordinate exchange.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -600, 600)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    a_rev = x[::-1]
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    def func(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    return a_out


def run_adam_lse(a0, beta, n_iters=3000, lr_init=0.005):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a0)
    x = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = x.sum()
    if S < 1e-12:
        x = np.ones(n) / n
    else:
        x = x / S

    m = np.zeros(n)
    v = np.zeros(n)
    best_x = x.copy()
    best_sc = compute_score(x)
    lr = lr_init

    for it in range(n_iters):
        f, g = lse_obj_grad(x, beta)

        b1, b2, eps = 0.85, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))

        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            continue
        x_new = x_new / S_new

        f_new, _ = lse_obj_grad(x_new, beta)
        if f_new < f:
            x = x_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.92

        if it % 300 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()
    return best_x


def nesterov_subgrad(a0, n_iters=5000, lr_init=0.02):
    """
    Nesterov-accelerated projected subgradient on the true max objective.
    Uses the subgradient at the argmax lag.
    """
    n = len(a0)
    x = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = x.sum()
    if S < 1e-12:
        x = np.ones(n) / n
    else:
        x = x / S

    y = x.copy()
    v = np.zeros(n)
    t_k = 1.0
    best_x = x.copy()
    best_sc = compute_score(x)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(y)
        k_star = int(np.argmax(c))
        M_val = c[k_star]
        S_y = float(y.sum())
        score = 2.0 * n * M_val / (S_y * S_y)

        # Subgradient of 2n*max(c)/S^2 at the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        g = np.zeros(n)
        seg = y[j_start:j_end + 1]
        g[j_start:j_end + 1] = seg[::-1]
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (g / S2 - 2.0 * M_val / S3)

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x + ((t_k - 1.0) / t_new) * (x - y)
        y_new = y_new - lr * grad
        y_new = np.maximum(y_new, 0.0)
        S_new = y_new.sum()
        if S_new < 1e-12:
            y_new = y
        else:
            y_new = y_new / S_new

        x = y_new
        y = x + ((t_k - 1.0) / t_new) * (x - y)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y > 1e-12:
            y = y / S_y
        t_k = t_new

        # Adaptive lr
        if score < best_sc:
            best_sc = score
            best_x = x.copy()
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.998

    return best_x


def fienup_anneal(a, n_iters=400):
    """Fienup with annealed spectral flattening exponent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Anneal: start aggressive (gamma=1.5), end gentle (gamma=0.3)
        gamma = 1.5 - 1.2 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        if P_mean < 1e-20:
            break

        ratio = P / (P_mean + 1e-30)
        # Push towards flat: P_new = P * (P/P_mean)^(-gamma) = P_mean * ratio^(1-gamma)
        P_new = P_mean * np.maximum(ratio, 1e-20) ** (1.0 - gamma)

        A_new = A * np.sqrt(P_new / (P + 1e-30))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_phase_random(a, n_iters=200, rng=None):
    """Fienup with random phase perturbation to escape local minima."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Random phase perturbation
        phase_noise = rng.standard_normal(len(A)) * 0.1
        A_noisy = A * np.exp(1j * phase_noise)
        P_noisy = A_noisy.real ** 2 + A_noisy.imag ** 2

        # Flatten
        ratio = P_noisy / (P_mean + 1e-30)
        P_new = P_mean * np.maximum(ratio, 1e-20) ** 0.5
        A_new = A_noisy * np.sqrt(P_new / (P_noisy + 1e-30))

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
    return best_a


def softmax_peak_shave(a, n_iters=5000, T_init=0.05, T_end=0.002, lr_init=0.01):
    """
    Gradient descent on softmax-weighted peaks with annealed temperature.
    Uses FFT-accelerated gradient computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    v = np.zeros(n)  # momentum

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        T = T_init + (T_end - T_init) * progress

        c = fast_conv(a)
        c_sharp = c / T
        c_sharp -= c_sharp.max()
        exp_c = np.exp(c_sharp)
        p = exp_c / exp_c.sum()

        # Weighted sum of peaks
        L_val = float(np.dot(p, c))
        S_cur = float(a.sum())
        score = 2.0 * n * L_val / (S_cur * S_cur)

        # Gradient
        PA = np.fft.rfft(p, M2_fft)
        a_rev = a[::-1]
        RA = np.fft.rfft(a_rev, M2_fft)
        conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
        g = 2.0 * conv_res[n - 1:2 * n - 1]

        S2 = S_cur * S_cur
        S3 = S2 * S_cur
        grad = (2.0 * n) * (g / S2 - 2.0 * L_val / S3)

        # Nesterov-like momentum
        v = 0.9 * v + 0.1 * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new < score:
            a = a_new
            lr = min(lr * 1.005, 0.08)
        else:
            lr *= 0.95

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

    return best_a


def pairwise_cd(a, n_sweeps=3, n_top=40, fracs=None):
    """
    Pairwise coordinate descent: exchange mass between top-contributing pairs.
    """
    if fracs is None:
        fracs = [0.2, 0.4, 0.6]
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Process top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]
        for k_star in top_peaks:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k_star - js]
            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for i in range(min(top_count, 20)):
                for j in range(i + 1, min(top_count, 20)):
                    j1 = int(top_global[i])
                    j2 = int(top_global[j])
                    a1, a2 = a[j1], a[j2]
                    if a1 < a2:
                        j1, j2 = j2, j1
                        a1, a2 = a2, a1
                    if a1 < 1e-10:
                        continue
                    for frac in fracs:
                        delta = a1 * frac
                        a[j1] = a1 - delta
                        a[j2] = a2 + delta
                        S_tmp = a.sum()
                        if S_tmp < 1e-12:
                            a[j1] = a1
                            a[j2] = a2
                            continue
                        a_scaled = a / S_tmp
                        sc = compute_score(a_scaled)
                        a[j1] = a1
                        a[j2] = a2
                        if sc < best_sc:
                            best_sc = sc
                            a[j1] = a1 - delta
                            a[j2] = a2 + delta
                            a = a / a.sum()
                            best_a = a.copy()
    return best_a


def single_cd(a, n_sweeps=5, n_top=60):
    """Single coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.4]:
                a_new_val = a_old * factor
                a[j] = a_new_val
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                a[j] = a_old
                if sc < best_sc:
                    best_sc = sc
                    a[j] = a_new_val
                    a = a / a.sum()
                    best_a = a.copy()
    return best_a


def structured_perturb(a, rng, scale=0.02):
    """Low-frequency structured perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 10):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Fienup spectral shaping (10%)
    Phase 2: LSE + L-BFGS low-beta sweep (25%)
    Phase 3: LSE + L-BFGS/Adam high-beta sweep (30%)
    Phase 4: Softmax peak shaving + Nesterov subgrad (20%)
    Phase 5: Coordinate descent + restarts (10%)
    Phase 6: Final polish (5%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 8192
        best_a = np.ones(n0) / n0
        sc_p = compute_score(best_a)
        update(sc_p, best_a, "uniform-init")

    print(f"=== Starting search from score {best_score:.8f} ===")

    # ============================================================
    # Phase 1: Fienup (10% time)
    # ============================================================
    p1_end = t_start + 0.10 * (total_budget - 30.0)
    print(f"\n--- Phase 1: Fienup (until t={p1_end - t_start:.0f}s) ---")

    a_f = fienup_anneal(best_a, n_iters=300)
    sc = compute_score(a_f)
    update(sc, a_f, "Fienup-anneal")

    a_f2 = fienup_phase_random(best_a, n_iters=100, rng=rng)
    sc = compute_score(a_f2)
    update(sc, a_f2, "Fienup-phase")

    # ============================================================
    # Phase 2: LSE low-beta L-BFGS (25% time)
    # ============================================================
    p2_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n--- Phase 2: LSE low-beta (until t={p2_end - t_start:.0f}s) ---")

    a_work = best_a.copy()
    for beta in [20, 50, 100, 200, 500]:
        if time.monotonic() > p2_end - 60:
            break
        remaining = p2_end - time.monotonic()
        maxiter = min(8000, int(remaining * 15))
        maxfun = min(150000, int(remaining * 200))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
        t0 = time.monotonic()
        a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        if update(sc, a_opt, f"LBFGS-low b={beta}"):
            a_work = a_opt.copy()

    # ============================================================
    # Phase 3: LSE high-beta sweep (30% time)
    # ============================================================
    p3_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n--- Phase 3: LSE high-beta (until t={p3_end - t_start:.0f}s) ---")

    a_work = best_a.copy()
    high_betas = [1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000]
    for beta in high_betas:
        if time.monotonic() > p3_end - 90:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 90:
            break

        # Use L-BFGS for medium beta, Adam for very high beta
        if beta <= 20000:
            maxiter = min(10000, int(remaining * 12))
            maxfun = min(200000, int(remaining * 150))
            print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
            t0 = time.monotonic()
            a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
            sc = compute_score(a_opt)
            dt = time.monotonic() - t0
            print(f"    score={sc:.8f} ({dt:.1f}s)")
            if update(sc, a_opt, f"LBFGS-high b={beta}"):
                a_work = a_opt.copy()
        else:
            # Adam for very high beta
            n_iter = min(3000, int(remaining * 5))
            print(f"  Adam beta={beta} (iters={n_iter})...")
            t0 = time.monotonic()
            a_opt = run_adam_lse(a_work, beta, n_iters=n_iter, lr_init=0.003)
            sc = compute_score(a_opt)
            dt = time.monotonic() - t0
            print(f"    score={sc:.8f} ({dt:.1f}s)")
            if update(sc, a_opt, f"Adam b={beta}"):
                a_work = a_opt.copy()

            # Also try L-BFGS from Adam result
            if time.monotonic() < p3_end - 60:
                a_opt2 = run_lbfgs(a_opt, min(beta, 50000), maxiter=3000, maxfun=60000)
                sc2 = compute_score(a_opt2)
                if update(sc2, a_opt2, f"LBFGS-after-Adam b={beta}"):
                    a_work = a_opt2.copy()

    # ============================================================
    # Phase 4: Softmax peak shaving + Nesterov (20% time)
    # ============================================================
    p4_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n--- Phase 4: Peak shaving (until t={p4_end - t_start:.0f}s) ---")

    # Softmax peak shaving with annealed temperature
    remaining = p4_end - time.monotonic()
    n_iter_shave = min(8000, int(remaining * 4))
    print(f"  Softmax peak shaving ({n_iter_shave} iters)...")
    t0 = time.monotonic()
    a_ps = softmax_peak_shave(best_a, n_iters=n_iter_shave, T_init=0.05, T_end=0.001, lr_init=0.008)
    sc = compute_score(a_ps)
    dt = time.monotonic() - t0
    print(f"    score={sc:.8f} ({dt:.1f}s)")
    if update(sc, a_ps, "SOFTMAX-SHAVE"):
        a_work = a_ps.copy()

    # Nesterov subgradient on true max
    if time.monotonic() < p4_end - 60:
        remaining = p4_end - time.monotonic()
        n_iter_ng = min(5000, int(remaining * 3))
        print(f"  Nesterov subgrad ({n_iter_ng} iters)...")
        t0 = time.monotonic()
        a_ng = nesterov_subgrad(best_a, n_iters=n_iter_ng, lr_init=0.015)
        sc = compute_score(a_ng)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        if update(sc, a_ng, "NESTEROV-SUBGRAD"):
            a_work = a_ng.copy()

    # ============================================================
    # Phase 5: Coordinate descent + restarts (10% time)
    # ============================================================
    p5_end = t_start + 0.95 * (total_budget - 30.0)
    print(f"\n--- Phase 5: CD + restarts (until t={p5_end - t_start:.0f}s) ---")

    # Single coordinate descent
    if time.monotonic() < p5_end - 120:
        a_cd = single_cd(best_a, n_sweeps=5, n_top=80)
        sc = compute_score(a_cd)
        print(f"  Single CD: score={sc:.8f}")
        if update(sc, a_cd, "SINGLE-CD"):
            a_work = a_cd.copy()

    # Pairwise CD
    if time.monotonic() < p5_end - 60:
        a_pcd = pairwise_cd(best_a, n_sweeps=3, n_top=30)
        sc = compute_score(a_pcd)
        print(f"  Pairwise CD: score={sc:.8f}")
        if update(sc, a_pcd, "PAIRWISE-CD"):
            a_work = a_pcd.copy()

    # L-BFGS polish after CD
    if time.monotonic() < p5_end - 30:
        a_polish = run_lbfgs(best_a, 20000, maxiter=3000, maxfun=60000)
        sc = compute_score(a_polish)
        print(f"  LBFGS polish: score={sc:.8f}")
        update(sc, a_polish, "LBFGS-POLISH")

    # Multi-restart
    n_restarts = 4
    for r in range(n_restarts):
        if time.monotonic() > p5_end - 30:
            break
        if r % 2 == 0:
            a_pert = structured_perturb(best_a, rng, scale=0.015 + 0.005 * r)
        else:
            noise = rng.standard_normal(len(best_a)) * (0.01 + 0.005 * r)
            a_pert = np.maximum(best_a * (1.0 + noise), 0.0)
            S_pert = a_pert.sum()
            if S_pert > 1e-12:
                a_pert = a_pert / S_pert

        remaining = p5_end - time.monotonic() - 30
        if remaining < 30:
            break
        maxiter = min(4000, int(remaining * 10))
        a_opt = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=maxiter * 20)
        sc = compute_score(a_opt)
        print(f"  Restart {r}: score={sc:.8f}")
        update(sc, a_opt, f"RESTART-{r}")

    # ============================================================
    # Phase 6: Final polish (5% time)
    # ============================================================
    print(f"\n--- Phase 6: Final polish ---")

    if time.monotonic() < deadline - 20:
        # Very high beta L-BFGS
        a_final = run_lbfgs(best_a, 100000, maxiter=3000, maxfun=50000)
        sc = compute_score(a_final)
        print(f"  Final LBFGS b=100k: score={sc:.8f}")
        update(sc, a_final, "FINAL-LBFGS-100k")

    if time.monotonic() < deadline - 15:
        # Final softmax shave
        remaining = deadline - 15 - time.monotonic()
        n_it = min(3000, int(remaining * 5))
        a_sh = softmax_peak_shave(best_a, n_iters=n_it, T_init=0.01, T_end=0.001, lr_init=0.005)
        sc = compute_score(a_sh)
        print(f"  Final shave: score={sc:.8f}")
        update(sc, a_sh, "FINAL-SHAVE")

    if time.monotonic() < deadline - 10:
        a_cd_final = single_cd(best_a, n_sweeps=3, n_top=100)
        sc = compute_score(a_cd_final)
        print(f"  Final CD: score={sc:.8f}")
        update(sc, a_cd_final, "FINAL-CD")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
