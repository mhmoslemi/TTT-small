def peak_shaving(a, n_sweeps=10, n_top_peaks=10, max_time=60.0):

    Targeted coordinate descent on the top peaks of the convolution.
    For each peak, try to reduce it by adjusting the most contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > max_time:
            break

        # Get top peaks
        top_k = min(n_top_peaks, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

            if time.monotonic() - t_start > max_time:
                break
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Find top contributors to this peak
            n_contrib = j_end - j_start + 1
            contrib = np.array([a[j] * a[k - j] for j in range(j_start, j_end + 1)])
            n_top_local = min(30, n_contrib)
            top_local = np.argsort(contrib)[::-1][:n_top_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:

                # Try several factors to reduce this element
                for factor in [0.6, 0.8, 0.9, 1.1, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
                    S_tmp = float(a.sum())
                    a = a / S_tmp

    return best_a


def fienup_refine(a, n_iters=200, thresh_start=2.5, thresh_end=1.05,
                  clip=0.5, mu=0.2, max_time=120.0):
    """Fienup-based refinement with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > max_time:
            break

        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        mu_t = mu * (1.0 - 0.5 * t)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target = np.sqrt(bin_thresh)
            curr = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target / np.maximum(curr, 1e-10))
            factor = np.clip(factor, 0.05, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        v = mu_t * v + (a_new - a)
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if (it + 1) % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def run_lbfgs_polish(a0, beta, maxiter=2000, maxfun=50000):
    """L-BFGS-B polish on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses adaptive LSE with backtracking, peak shaving, fienup, and L-BFGS.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 6962

    best_score = np.inf
    best_a = np.ones(n_target) / n_target

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: LSE optimization from parent (40% of time)
    # ============================================================
    phase1_end = t_start + 0.40 * (total_budget - 40.0)
    print(f"=== Phase 1: LSE from parent (t < {phase1_end - t_start:.0f}s) ===")

    if parent is not None:
        a_parent = parent.astype(np.float64).copy()
        a_parent = np.maximum(a_parent, 0.0)
        S_p = float(a_parent.sum())
        if S_p > 1e-6:
            a_parent = a_parent / S_p
        sc_p = compute_score(a_parent)
        update_best(sc_p, a_parent, "parent")
    else:
        a_init = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n_target))
        a_init = np.maximum(a_init, 0.01)
        a_init = a_init / a_init.sum()
        a_parent = a_init
        update_best(compute_score(a_parent), a_parent, "default-init")

    # Main LSE with backtracking
    time_left = phase1_end - time.monotonic()
    if time_left > 30:
        print(f"  Running LSE backtracking (time_left={time_left:.0f}s)")
        a_lse = lse_optimize_bt(a_parent, max_time=time_left * 0.95,
                                beta_start=5.0, beta_max=50000.0,
                                beta_growth=1.05, patience_epochs=2,
                                mu_start=0.85, mu_end=0.95, lr_init=0.005)
        update_best(compute_score(a_lse), a_lse, "lse-bt")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE with different parameters (25% of time)
    # ============================================================
    phase2_end = t_start + 0.65 * (total_budget - 40.0)
    print(f"\n=== Phase 2: LSE variants (t < {phase2_end - t_start:.0f}s) ===")

    # Simple LSE with different schedule
    time_left = phase2_end - time.monotonic()
    if time_left > 30:
        print(f"  Running LSE simple (time_left={time_left:.0f}s)")
        a_lse2 = lse_optimize_simple(best_a, max_time=time_left * 0.8,
                                     beta_start=10.0, beta_max=30000.0,
                                     beta_growth=1.003, mu=0.92,
                                     lr_init=0.01, lr_decay=0.9997)
        update_best(compute_score(a_lse2), a_lse2, "lse-simple")

    # Fienup refinement
    time_left = phase2_end - time.monotonic()
    if time_left > 20:
        print(f"  Running Fienup (time_left={time_left:.0f}s)")
        a_f = fienup_refine(best_a, n_iters=300, max_time=time_left * 0.9)
        update_best(compute_score(a_f), a_f, "fienup")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Peak shaving (15% of time)
    # ============================================================
    phase3_end = t_start + 0.80 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Peak shaving (t < {phase3_end - t_start:.0f}s) ===")

    time_left = phase3_end - time.monotonic()
    if time_left > 10:
        a_ps = peak_shaving(best_a, n_sweeps=20, n_top_peaks=8, max_time=time_left * 0.9)
        update_best(compute_score(a_ps), a_ps, "peak-shave")

    # LSE after peak shaving
    time_left = phase3_end - time.monotonic()
    if time_left > 20:
        a_lse3 = lse_optimize_simple(best_a, max_time=time_left * 0.9,
                                     beta_start=200.0, beta_max=50000.0,
                                     beta_growth=1.005, mu=0.93,
                                     lr_init=0.005, lr_decay=0.9998)
        update_best(compute_score(a_lse3), a_lse3, "lse-post-shave")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Random restarts (10% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 4: Random restarts (t < {phase4_end - t_start:.0f}s) ===")

    n_restarts = 3
    time_per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase4_end or time_per_restart < 30:
            break
        # Create perturbation
        noise_scale = 0.02 * (r + 1)
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_pert = np.maximum(a_pert, 0.0)
        S_p = float(a_pert.sum())
        if S_p < 1e-6:
            continue
        a_pert = a_pert / S_p

        print(f"  Restart {r+1}: noise_scale={noise_scale:.4f}, time={time_per_restart:.0f}s")
        a_r = lse_optimize_simple(a_pert, max_time=time_per_restart * 0.9,
                                  beta_start=50.0, beta_max=30000.0,
                                  beta_growth=1.005, mu=0.93,
                                  lr_init=0.005, lr_decay=0.9998)
        update_best(compute_score(a_r), a_r, f"restart-{r+1}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish (5% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: L-BFGS polish (t < {phase5_end - t_start:.0f}s) ===")

    lbfgs_betas = [1000, 5000, 20000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase5_end:
            break
        time_left = phase5_end - time.monotonic()
        if time_left < 10:
            break
        print(f"  L-BFGS with beta={beta}, time_left={time_left:.0f}s")
        sc, a_opt = run_lbfgs_polish(best_a, beta, maxiter=2000, maxfun=40000)
        update_best(sc, a_opt, f"lbfgs-b{beta}")

    # Final attempt: one more LSE with high beta
    if time.monotonic() < deadline - 10:
        time_left = deadline - time.monotonic()
        a_final = lse_optimize_simple(best_a, max_time=time_left * 0.9,
                                      beta_start=500.0, beta_max=80000.0,
                                      beta_growth=1.005, mu=0.94,
                                      lr_init=0.003, lr_decay=0.9998)
        update_best(compute_score(a_final), a_final, "final-lse")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
