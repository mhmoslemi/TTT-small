def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    n_old = len(a)
    if n_old == new_n:
        return a.copy()
    if n_old == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def shape_optimize(n, base_type='hann', n_modes=6, seed=42):
    """Optimize a low-dimensional Fourier parameterization of the window shape."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    # Base window
    if base_type == 'hann':
        base = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif base_type == 'gaussian':
        base = np.exp(-4.0 * (x - 0.5) ** 2)
    else:
        base = np.ones(n) * 0.5

    # Fourier basis (odd modes only for symmetry consideration not needed, use all)
    modes = []
    for k in range(1, n_modes + 1):
        modes.append(np.cos(2.0 * np.pi * k * x))
        modes.append(np.sin(2.0 * np.pi * k * x))

    def objective(params):
        a = base.copy()
        for i in range(n_modes):
            a += params[2 * i] * np.cos(2.0 * np.pi * (i + 1) * x)
            a += params[2 * i + 1] * np.sin(2.0 * np.pi * (i + 1) * x)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-10:
            return 1e10
        a = a / S
        return compute_score(a)

    # Start from zero perturbation
    x0 = np.zeros(2 * n_modes)
    # Small random perturbation to avoid symmetry
    x0 += 0.01 * rng.standard_normal(2 * n_modes)

    res = minimize(objective, x0, method='Nelder-Mead',
                   options={'maxiter': 50000, 'xatol': 1e-8, 'fatol': 1e-12,
                            'adaptive': True})

    a = base.copy()
    for i in range(n_modes):
        a += res.x[2 * i] * np.cos(2.0 * np.pi * (i + 1) * x)
        a += res.x[2 * i + 1] * np.sin(2.0 * np.pi * (i + 1) * x)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-10:
        a = a / S
    return a, objective(res.x)


def active_set_perturbation(a, rng, scale=0.02, n_active_frac=0.3):
    """Perturb based on active set: reduce peak contributors, boost low contributors."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    M = float(c.max())
    threshold = M * (1.0 - 1e-6)

    # Find active positions (convolution at or very near max)
    active_mask = c >= threshold * (1.0 - 1e-4)
    n_active = int(n * n_active_frac)

    a_p = a.copy()

    # For each active convolution position, find top contributors and reduce them
    active_idx = np.where(active_mask)[0]
    # Sample a subset of active positions for efficiency
    if len(active_idx) > 50:
        active_idx = rng.choice(active_idx, size=50, replace=False)

    for k in active_idx:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        seg = a[j_start:j_end + 1]
        contrib = seg * a[k - np.arange(j_start, j_end + 1)][::-1]
        # Actually contrib[i] = a[j_start+i] * a[k-(j_start+i)]
        ii = np.arange(j_start, j_end + 1)
        contrib = a[ii] * a[k - ii]

        n_top = max(3, len(contrib) // 5)
        top_local = np.argsort(contrib)[::-1][:n_top]
        bottom_local = np.argsort(contrib)[:n_top]

        for li in top_local:
            reduction = a[j_start + li] * scale * rng.random()
            a_p[j_start + li] -= reduction
            # Redistribute to bottom contributors
            per = reduction / len(bottom_local)
            for bi in bottom_local:
                a_p[j_start + bi] += per

    a_p = np.clip(a_p, 0.0, 1000.0)
    S_p = a_p.sum()
    if S_p > 1e-12:
        a_p = a_p / S_p
    return a_p


def random_perturbation(a, rng, scale=0.02):
    """Simple multiplicative random perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 5)

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        a_p = a * np.exp(scale * rng.standard_normal(n))
    elif strategy == 3:
        n_blocks = 5 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def direct_peak_reduction(a, n_rounds=5, top_k=30):
    """Directly reduce the peak by scaling down top contributors and redistributing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        ii = np.arange(j_start, j_end + 1)
        contrib = a[ii] * a[k_star - ii]
        n_top = min(top_k, len(contrib) // 3)
        n_bot = min(top_k, len(contrib) // 3)
        top_local = np.argsort(contrib)[::-1][:n_top]
        bot_local = np.argsort(contrib)[:n_bot]

        # Try different reduction factors
        best_frac = 1.0
        best_local_sc = best_sc
        for frac in [0.5, 0.6, 0.7, 0.8, 0.9]:
            a_trial = a.copy()
            total_reduction = 0.0
            for li in top_local:
                reduction = a_trial[j_start + li] * (1.0 - frac)
                a_trial[j_start + li] -= reduction
                total_reduction += reduction
            per = total_reduction / n_bot
            for bi in bot_local:
                a_trial[j_start + bi] += per
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_t = a_trial.sum()
            if S_t > 1e-12:
                a_trial = a_trial / S_t
            sc = compute_score(a_trial)
            if sc < best_local_sc:
                best_local_sc = sc
                best_frac = frac
                best_a = a_trial.copy()

        if best_frac == 1.0:
            break
        a = best_a.copy()
        if best_sc < compute_score(a):
            a = best_a.copy()

    return best_a


def nesterov_subgrad_multi(a, n_iters=5000, f_lb=1.0, top_k=8):
    """Nesterov-accelerated subgradient with multi-peak weighting."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        c_max = float(c.max())

        # Use top-k peaks with softmax weighting
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.exp(-0.5 * (c_max - c[top_idx]) / (c_max * 0.01 + 1e-10))
        weights /= weights.sum()

        Mak_weighted = float(np.sum(weights * c[top_idx]))
        sc = 2.0 * n * c_max / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Weighted subgradient
        S2 = Sy * Sy
        S3 = S2 * Sy
        sg = np.zeros(n)
        for rank, idx in enumerate(top_idx):
            w = weights[rank]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * y[idx - ii]

        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def continuation_lbfgs(a0, beta_schedule, deadline):
    """Run L-BFGS with increasing beta (continuation method)."""
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(50000, max(3000, remaining * 30)))
        maxfun = int(min(200000, max(10000, remaining * 120)))

        sc, a_opt, _ = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a = a_opt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: shape optimization -> continuation L-BFGS -> ILS -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 25.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    if parent is not None:
        parent = parent / parent.sum()
        sc_parent = compute_score(parent)
        print(f"Parent: score={sc_parent:.10f}, n={len(parent)}")
        report(sc_parent, parent, "parent")

        # Try multiple n values
        n_candidates = [len(parent)]
        for n_try in [4096, 6000, 8192, 10000, 12000]:
            if n_try not in n_candidates:
                n_candidates.append(n_try)
        n_candidates = sorted(set(n_candidates))
    else:
        n_candidates = [6962, 8192, 10000]

    # ==================================================================
    # Phase 1: Multi-n initialization + quick L-BFGS (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 1: Multi-n Initialization ---")

    candidates = {}
    for n_try in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        if parent is not None:
            a_init = resample_sequence(parent, n_try)
        else:
            x = np.linspace(0, 1, n_try)
            a_init = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))

        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S
        sc = compute_score(a_init)
        report(sc, a_init, f"init n={n_try}")
        candidates[n_try] = a_init

    # Quick L-BFGS on each candidate with moderate beta
    for n_try, a_init in list(candidates.items()):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        sc, a_opt, _ = run_lbfgs(a_init, 200, maxiter=5000, maxfun=20000)
        report(sc, a_opt, f"quick_lbfgs n={n_try}")
        candidates[n_try] = a_opt

    # Pick the best n to work with
    best_n = min(candidates.keys(), key=lambda k: compute_score(candidates[k]))
    if best_score < np.inf:
        best_n = len(best_a)
    print(f"Best n for refinement: {best_n}")

    # ==================================================================
    # Phase 2: Shape optimization for better template (5% time)
    # ==================================================================
    phase2_deadline = t_start + 0.20 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 2: Shape Optimization ---")

    if time.monotonic() < phase2_deadline:
        for n_try in [best_n]:
            for base_type in ['hann', 'gaussian']:
                if time.monotonic() > phase2_deadline:
                    break
                a_shape, sc_shape = shape_optimize(n_try, base_type=base_type,
                                                   n_modes=8, seed=seed)
                report(sc_shape, a_shape, f"shape_{base_type} n={n_try}")

    # ==================================================================
    # Phase 3: Continuation L-BFGS (40% time)
    # ==================================================================
    phase3_deadline = t_start + 0.60 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 3: Continuation L-BFGS ---")

    a_work = best_a.copy() if best_a.sum() > 1e-12 else candidates.get(best_n, np.ones(best_n) * 0.5)
    S_w = a_work.sum()
    if S_w > 1e-12:
        a_work = a_work / S_w

    # Continuation: gradually increase beta
    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(50000, max(3000, remaining * 25)))
        maxfun = int(min(200000, max(10000, remaining * 100)))

        sc, a_opt, _ = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"cont beta={beta}")
        a_work = a_opt

        # If improvement is significant, try a perturbed restart
        if time.monotonic() < phase3_deadline - 60 and beta >= 1000:
            a_pert = random_perturbation(a_opt, rng, scale=0.01)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter // 2,
                                       maxfun=maxfun // 2)
            report(sc2, a_opt2, f"cont beta={beta} pert")

    # ==================================================================
    # Phase 4: Iterated Local Search with active-set perturbation (25% time)
    # ==================================================================
    phase4_deadline = t_start + 0.85 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 25
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 45:
            break

        # Alternate between random and active-set perturbation
        if trial % 3 == 0:
            a_pert = active_set_perturbation(best_a, rng, scale=0.015)
        elif trial % 3 == 1:
            a_pert = random_perturbation(best_a, rng, scale=0.01 + 0.02 * rng.random())
        else:
            a_pert = random_perturbation(best_a, rng, scale=0.03)
            a_pert = active_set_perturbation(a_pert, rng, scale=0.01)

        # Short L-BFGS with high beta
        beta = 5000 + int(50000 * rng.random())
        maxiter = int(min(20000, max(2000, remaining * 10)))
        maxfun = int(min(80000, max(5000, remaining * 50)))
        sc, a_opt, _ = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} b={beta}")

        # If improved, do a slightly longer refinement
        if sc < best_score:
            if time.monotonic() < phase4_deadline - 30:
                sc2, a_opt2, _ = run_lbfgs(a_opt, beta * 2,
                                           maxiter=maxiter // 2, maxfun=maxfun // 2)
                report(sc2, a_opt2, f"ILS#{trial} refine")

    # ==================================================================
    # Phase 5: Subgradient + Direct Peak Reduction (10% time)
    # ==================================================================
    phase5_deadline = t_start + 0.95 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 5: Subgradient + Peak Reduction ---")

    if time.monotonic() < phase5_deadline:
        a_sg = nesterov_subgrad_multi(best_a, n_iters=3000, f_lb=1.0, top_k=8)
        sc = compute_score(a_sg)
        report(sc, a_sg, "nesterov_multi")

    if time.monotonic() < phase5_deadline - 20:
        a_dpr = direct_peak_reduction(best_a, n_rounds=10, top_k=50)
        sc = compute_score(a_dpr)
        report(sc, a_dpr, "peak_reduction")

    if time.monotonic() < phase5_deadline - 20:
        # Combine: subgradient then peak reduction
        a_combo = nesterov_subgrad_multi(best_a, n_iters=1000, f_lb=1.0, top_k=5)
        a_combo = direct_peak_reduction(a_combo, n_rounds=5, top_k=30)
        sc = compute_score(a_combo)
        report(sc, a_combo, "combo_sg+dpr")

    # ==================================================================
    # Phase 6: Final high-precision L-BFGS polish (5% time)
    # ==================================================================
    print(f"\n--- Phase 6: Final Polish ---")

    for beta in [100000, 300000, 1000000]:
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic() - 15
        if remaining < 20:
            break
        maxiter = int(min(30000, max(2000, remaining * 20)))
        maxfun = int(min(100000, max(5000, remaining * 80)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final beta={beta}")

    # One more peak reduction + L-BFGS cycle
    if time.monotonic() < deadline - 15:
        a_dpr = direct_peak_reduction(best_a, n_rounds=5, top_k=20)
        sc = compute_score(a_dpr)
        report(sc, a_dpr, "final_dpr")
        if time.monotonic() < deadline - 10:
            sc, a_opt, _ = run_lbfgs(a_dpr, 500000, maxiter=10000, maxfun=50000)
            report(sc, a_opt, "final_polish2")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
