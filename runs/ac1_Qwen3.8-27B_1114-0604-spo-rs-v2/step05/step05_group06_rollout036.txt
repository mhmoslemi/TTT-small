"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA with adaptive subgradient,
active-set management, and Nesterov acceleration.
Key innovations: direct FISTA (no L-BFGS-B overhead), adaptive active set
growth, backtracking line search, and multiple diverse initializations.
"""
import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _normalize(a):
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def _cross_corr(a, w):
    """Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    n = a.size
    M = 2 * n - 1
    wc = np.zeros(M)
    wc[:min(w.size, M)] = w[:min(w.size, M)]
    N = _next_pow2(n + M - 1)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(wc, N)
    full = np.fft.irfft(A * W, N)[:n + M - 1]
    return full[n - 1:2 * n - 1]


def _simplex_proj(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u - (cssv - 1.0) / np.arange(1, n + 1) > 0
    if not cond.any():
        u = np.ones(n) / n
        return u.copy()
    idx = np.where(cond)[0][-1]
    tau = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - tau, 0.0)


def _subgrad_objective(a, k=10):
    """Compute (score, subgradient) using top-k active set."""
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _self_conv(a)
    M = c.size
    d = c / (S * S)  # normalized correlation
    k_eff = max(1, min(k, M))
    # Find top-k positions
    if k_eff >= M:
        idx = np.arange(M)
        w = np.ones(M) / M
    else:
        top_idx = np.argpartition(d, M - k_eff)[M - k_eff:]
        # Use softmax weighting over the top-k
        d_top = d[top_idx]
        d_max = d_top.max()
        # Temperature-based weighting
        temp = 0.01 * max(d_max, 1e-10)
        weights = np.exp((d_top - d_max) / temp)
        weights /= weights.sum()
        w = np.zeros(M)
        w[top_idx] = weights
    
    corr = _cross_corr(a, w)
    # Score
    score = float(2.0 * n * c.max() / (S * S))
    # Subgradient of 2n * max(c) / S^2 (with S fixed = 1): 2n * 2 * corr = 4n * corr
    # But we should account for the 1/S^2 factor...
    # Actually, since we maintain S=1 via projection, we can ignore the 1/S^2 in the gradient
    # (it's a constraint, handled by the projection)
    grad = 4.0 * n * corr
    
    return score, grad


def _fista_optimize(a0, max_iters=500, deadline=None, k_start=5, k_end=200, 
                    verbose=False, label=""):
    """FISTA optimization with adaptive active set."""
    a = _normalize(a0).copy()
    n = a.size
    t_k = 1.0
    y = a.copy()  # Nesterov extrapolation point
    best_a = a.copy()
    best_score = _score(a)
    
    if deadline is None:
        deadline = time.monotonic() + 60
    
    # Adaptive step size
    step = 1.0 / (4.0 * n)  # initial conservative step
    # Actually, the Lipschitz constant of the subgradient is related to 4n*||a||_1 = 4n
    # So step should be at most 1/(4n) for non-accelerated, but FISTA can be more aggressive
    # Let's use backtracking
    
    L = 4.0 * n  # Lipschitz constant estimate
    step = 1.0 / L
    c_step = 0.5  # backtracking factor
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adapt k over iterations
        frac = it / max(1, max_iters - 1)
        k_cur = int(k_start + frac * (k_end - k_start))
        k_cur = max(1, min(k_cur, 2 * n - 1))
        
        # Subgradient at y
        score_y, g_y = _subgrad_objective(y, k=k_cur)
        
        if not np.isfinite(score_y):
            break
        
        # Gradient step with backtracking
        x_new = y - step * g_y
        x_new = _simplex_proj(x_new)
        score_new = _score(x_new)
        
        # Backtracking: if score didn't decrease, reduce step
        while score_new > score_y + 1e-12 * score_y and step > 1e-10:
            step *= c_step
            x_new = y - step * g_y
            x_new = _simplex_proj(x_new)
            score_new = _score(x_new)
        
        # Nesterov update for next iteration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = _simplex_proj(y)
        a = x_new
        t_k = t_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    [{label}] iter={it}, score={best_score:.8f}, k={k_cur}, step={step:.2e}")
    
    return best_score, best_a


def _sg_polish(a, max_iters=200, deadline=None):
    """Exact subgradient descent with Polyak step size for final polishing."""
    a = _normalize(a).copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.size
    M = 2 * n - 1
    
    if deadline is None:
        deadline = time.monotonic() + 30
    
    # Polyak step: use (f(x_k) - f*) / ||g||^2, but we don't know f*
    # Use a heuristic: start with a moderate step and adapt
    step = 1e-4
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        
        # Active set: all positions within 1% of max
        threshold = dmax - 0.01 * dmax
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        
        w = np.zeros(M)
        w[active] = 1.0 / active.sum()
        
        corr = _cross_corr(a, w)
        # Subgradient of 2n*max(c)/S^2: approximately 4n*corr (when S~1)
        grad = 4.0 * n * corr
        gnorm = float(np.linalg.norm(grad))
        
        if gnorm < 1e-14:
            break
        
        # Try multiple step sizes
        best_new = None
        best_new_score = best_score
        for scale in (step / 4, step / 2, step, step * 2, step * 4):
            q = a - scale * grad / gnorm  # normalize gradient direction
            q = _simplex_proj(q)
            sc = _score(q)
            if sc < best_new_score:
                best_new_score = sc
                best_new = q.copy()
            if time.monotonic() > deadline:
                break
        
        if best_new is None:
            # Try smaller steps
            for scale in (step / 16, step / 64, step / 256):
                q = a - scale * grad / gnorm
                q = _simplex_proj(q)
                sc = _score(q)
                if sc < best_new_score:
                    best_new_score = sc
                    best_new = q.copy()
        
        if best_new is None:
            # No improvement, try to reduce step
            step *= 0.5
            if step < 1e-8:
                break
            continue
        
        if best_new_score < best_score:
            best_score = best_new_score
            best_a = best_new.copy()
        a = best_new
        step *= 1.01  # slight increase
    
    return best_score, best_a


def _lse_optimize(a0, beta, k, time_limit, maxfun=800):
    """LSE-smoothed optimization using scipy L-BFGS-B."""
    from scipy.optimize import minimize
    a = _normalize(a0)
    n = a.size
    t_end = time.monotonic() + time_limit
    
    if time_limit < 2.0:
        return _score(a), a
    
    PEN = 1e-3
    
    def obj(x):
        S = float(x.sum())
        if S < 1e-8:
            return 1e10, np.zeros(n)
        c = _self_conv(x)
        d = float(n) * c / (S * S)
        M = c.size
        k_eff = max(1, min(k, M))
        idx = np.argpartition(d, M - k_eff)[M - k_eff:]
        d_act = d[idx]
        dmax = float(d_act.max())
        if dmax <= 0:
            return 1e10, np.zeros(n)
        z = beta * (d_act - dmax)
        e = np.exp(z)
        se = float(e.sum())
        if se <= 0:
            e = np.ones(k_eff) / k_eff
            se = 1.0
        w_act = e / se
        L = dmax + np.log(se) / beta
        
        w_full = np.zeros(M)
        w_full[idx] = w_act
        corr = _cross_corr(x, w_full)
        LDS = float(w_act @ d_act)
        
        grad = 4.0 * n * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        val = 2.0 * L + PEN * (S - 1.0) ** 2
        grad = grad + 2.0 * PEN * (S - 1.0)
        return float(val), grad
    
    try:
        maxiter = max(20, maxfun // 3)
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': maxfun,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30}
        )
        out = _normalize(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _spectral_init(n, rng):
    """Generate non-negative sequence with flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + 0.1 * np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = a.size
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _normalize(q)


def _upsample(a, n_target):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n_target) / n_target
    if m == n_target:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n_target)
    return np.interp(x1, x0, a).astype(np.float64)

        if gnorm < 1e-14:
            break
        
        # Direction: -g/||g|| (normalized)
        direction = -g_y / gnorm
        
        # Backtracking line search
        alpha = step
        x_new = None

            candidate = _simplex_proj(y + alpha * direction)
            sc = _score(candidate)

            best_a = a.copy()
        
        # Increase step if we didn't need backtracking
        if alpha == step:
            step = min(step * 1.1, 1.0 / (4.0 * n))

    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _normalize(a):
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def _cross_corr(a, w):
    """Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    n = a.size
    M = 2 * n - 1
    wc = np.zeros(M)
    wc[:min(w.size, M)] = w[:min(w.size, M)]
    N = _next_pow2(n + M - 1)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(wc, N)
    full = np.fft.irfft(A * W, N)[:n + M - 1]
    return full[n - 1:2 * n - 1]


def _simplex_proj(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u - (cssv - 1.0) / np.arange(1, n + 1) > 0
    if not cond.any():
        return np.ones(n) / n
    idx = np.where(cond)[0][-1]
    tau = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - tau, 0.0)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient for top-k active positions."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = c.size
    k_eff = max(1, min(k, M))
    if k_eff >= M:
        idx = np.arange(M)
    else:
        idx = np.argpartition(d, M - k_eff)[M - k_eff:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k_eff) / k_eff
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * n * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=600):
    """Run L-BFGS-B on LSE objective."""
    a = _normalize(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 2.0:
        return _score(a), a
    mf = int(0.8 * time_limit / max(0.003 * max(1, n / 4096.0), 0.001))
    mf = max(20, min(mf, maxfun_cap, 5000))
    maxiter = max(15, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30}
        )
        out = _normalize(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _sg_step(a, k_active=1):
    """Compute subgradient with top-k active set. Returns (score, grad, active_mask)."""
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n), None
    c = _self_conv(a)
    M = c.size
    d = c / (S * S)
    dmax = float(d.max())

    # Active set: positions within tolerance of max
    tol = 1e-8 * max(dmax, 1e-10)
    if k_active >= 1:
        # Use top-k positions
        k_eff = min(k_active, M)
        top_idx = np.argpartition(d, M - k_eff)[M - k_eff:]
        # Weight by proximity to max
        d_top = d[top_idx]
        if k_eff == 1:
            w = np.zeros(M)
            w[top_idx[0]] = 1.0
        else:
            # Softmax weighting
            temperature = max(1e-6, (d_max - d_top.min()) * 0.1)
            weights = np.exp(-(d_max - d_top) / temperature)
            weights /= weights.sum()
            w = np.zeros(M)
            w[top_idx] = weights
    else:
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        w = np.zeros(M)
        w[active] = 1.0 / active.sum()

    corr = _cross_corr(a, w)
    score = 2.0 * n * dmax
    grad = 4.0 * n * corr
    grad = np.nan_to_num(grad, nan=0.0)
    return score, grad, w


def _subgrad_line_search(a, max_iters=2000, deadline=None, k_schedule=None,
                          verbose=False, label=""):
    """Subgradient descent with backtracking line search and Nesterov momentum."""
    a = _normalize(a).copy()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    t_k = 1.0
    y = a.copy()
    momentum = 0.0
    step = 1.0 / (8.0 * n)
    c_bt = 0.5
    no_improve_count = 0
    max_no_improve = 500

    if deadline is None:
        deadline = time.monotonic() + 120

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive k
        if k_schedule is not None:
            frac = it / max(1, max_iters - 1)
            k_cur = int(k_schedule[0] + frac * (k_schedule[1] - k_schedule[0]))
        else:
            k_cur = 1

        score_y, g_y, _ = _sg_step(y, k_active=k_cur)
        if not np.isfinite(score_y):
            break

        gnorm = float(np.linalg.norm(g_y))
        if gnorm < 1e-15:
            break

        # Direction
        direction = -g_y / gnorm

        # Backtracking line search
        alpha = step
        x_new = None
        for _ in range(15):
            candidate = _simplex_proj(y + alpha * direction)
            sc = _score(candidate)
            if sc <= score_y + 1e-12:
                x_new = candidate
                break
            alpha *= c_bt
            if alpha < 1e-16:
                break

        if x_new is None:
            x_new = _simplex_proj(y + alpha * direction)
            sc = _score(x_new)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = _simplex_proj(y)
        a = x_new
        t_k = t_new
        momentum = (t_k - 1.0) / t_new

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adapt step size
        if no_improve_count > max_no_improve:
            step *= 0.5
            no_improve_count = 0
            if step < 1e-15:
                break
        elif alpha == step and it > 0:
            step = min(step * 1.05, 1.0 / (4.0 * n))

        if verbose and it % 200 == 0:
            print(f"    [{label}] it={it}, best={best_score:.8f}, step={step:.2e}, k={k_cur}")

    return best_score, best_a


def _exquisite_polish(a, deadline, max_iters=5000):
    """Very fine-grained subgradient descent focusing on the single worst lag."""
    a = _normalize(a).copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.size

    step = 1.0 / (10.0 * n)
    c_bt = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        S = float(a.sum())
        c = _self_conv(a)
        M = c.size
        dmax_idx = int(np.argmax(c))
        dmax = c[dmax_idx]

        # Weight: focus on top 1-3 lags
        k_eff = 3
        top_idx = np.argpartition(c, M - k_eff)[M - k_eff:]
        weights = c[top_idx].copy()
        # Emphasize the max
        weights = weights ** 10
        weights /= weights.sum()
        w = np.zeros(M)
        w[top_idx] = weights

        corr = _cross_corr(a, w)
        grad = 4.0 * n * corr / (S * S)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        direction = -grad / gnorm

        # Line search
        alpha = step
        improved = False
        for _ in range(12):
            candidate = _simplex_proj(a + alpha * direction)
            sc = _score(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                a = candidate
                improved = True
                break
            alpha *= c_bt

        if improved:
            step = min(step * 1.02, 1.0 / (5.0 * n))
        else:
            step *= 0.5
            if step < 1e-16:
                break

    return best_score, best_a


def _spectral_init(n, rng):
    """Non-negative sequence with controlled spectral shape."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + 0.05 * np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = np.clip(sig - sig.min(), 0, None)
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = a.size
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _normalize(q)


def _upsample(a, n_target):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n_target) / n_target
    if m == n_target:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n_target)
    return np.interp(x1, x0, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: progressive LSE + FISTA subgradient + exquisite polish + mutation.
    """
    t_start = time.monotonic()
    total_deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 128:
                parent = _normalize(p)
    except Exception:
        pass

    if parent is not None:
        target_n = len(parent)
        print(f"Parent: n={target_n}")
    else:
        target_n = 8192
        print(f"No parent, n={target_n}")

    # Generate initializations
    inits = []
    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))
        p = _upsample(parent, target_n)
        for amp in (0.03, 0.08, 0.15):
            inits.append((f'p+{amp}', _smooth_perturb(p, rng, amp)))

    for i in range(3):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # Score all inits
    scored = []
    for name, arr in inits:
        arr = _normalize(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))
    scored.sort(key=lambda x: x[0])

    print(f"Inits: " + ", ".join(f"{nm}:{sc:.6f}" for sc, nm, _ in scored[:5]))

    best_score = scored[0][0] if scored else 2.0
    best_seq = (scored[0][2].copy() if scored else _normalize(np.ones(target_n)))

    # === PHASE 1: Progressive LSE (55% of time) ===
    p1_end = t_start + (total_deadline - t_start) * 0.55
    print(f"\n=== Phase 1: LSE (deadline in {p1_end - t_start:.0f}s) ===")

    # Focus on the best initialization
    a_cur = _normalize(best_seq)

    # LSE stages: (K, beta, fraction_of_phase1_time)
    p1_stages = [
        (3, 30, 0.10),
        (5, 60, 0.12),
        (8, 120, 0.13),
        (12, 250, 0.14),
        (20, 500, 0.15),
        (35, 1000, 0.16),
        (60, 2000, 0.20),
    ]

    cum_frac = 0.0
    for k, beta, frac in p1_stages:
        cum_frac += frac
        stage_end = t_start + (p1_end - t_start) * cum_frac
        stage_end = min(stage_end, p1_end - 3)
        if time.monotonic() > stage_end:
            break

        time_left = max(3.0, stage_end - time.monotonic())
        # Run L-BFGS-B for this stage
        sc_new, a_new = _run_lbfgs(a_cur, beta, k, time_left, maxfun_cap=800)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = _normalize(a_new)
            print(f"  LSE K={k}, beta={beta}: {best_score:.8f} *")
        a_cur = _normalize(a_new)

    # Quick LSE on 2nd best init if time permits
    if len(scored) > 1 and time.monotonic() < p1_end - 30:
        a2 = _normalize(scored[1][2])
        sc2, a2_new = _run_lbfgs(a2, 500, 30, min(60, p1_end - time.monotonic() - 10), 400)
        if sc2 < best_score:
            best_score = sc2
            best_seq = _normalize(a2_new)
            print(f"  Alt init improved: {best_score:.8f}")

    # === PHASE 2: Subgradient with line search (25% of time) ===
    p2_end = t_start + (total_deadline - t_start) * 0.80
    print(f"\n=== Phase 2: Subgrad (score={best_score:.8f}) ===")

    a_sg = _normalize(best_seq)
    sc_sg, a_sg = _subgrad_line_search(
        a_sg, max_iters=5000, deadline=p2_end,
        k_schedule=(1, 5), verbose=True, label="sg"
    )
    if sc_sg < best_score:
        best_score = sc_sg
        best_seq = a_sg.copy()
        print(f"  Subgrad improved: {best_score:.8f}")

    # === PHASE 3: Mutation + re-optimize (10% of time) ===
    p3_end = t_start + (total_deadline - t_start) * 0.90
    print(f"\n=== Phase 3: Mutation (score={best_score:.8f}) ===")

    for mut_i in range(5):
        if time.monotonic() > p3_end:
            break
        # Small mutation
        a_mut = _smooth_perturb(_normalize(best_seq), rng, strength=0.03)
        # Quick LSE re-optimization
        time_left = max(5.0, (p3_end - time.monotonic()) / (5 - mut_i) - 2)
        sc_mut, a_mut = _run_lbfgs(a_mut, 1000, 50, time_left, maxfun_cap=300)
        if sc_mut < best_score:
            best_score = sc_mut
            best_seq = _normalize(a_mut)
            print(f"  Mutation {mut_i}: {best_score:.8f} *")

    # === PHASE 4: Exquisite polish (10% of time) ===
    p4_end = total_deadline - 5.0
    print(f"\n=== Phase 4: Polish (score={best_score:.8f}) ===")

    a_pol = _normalize(best_seq)
    sc_pol, a_pol = _exquisite_polish(a_pol, deadline=p4_end, max_iters=10000)
    if sc_pol < best_score:
        best_score = sc_pol
        best_seq = a_pol.copy()
        print(f"  Polish improved: {best_score:.8f}")

    # Final verification
    out = _normalize(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
