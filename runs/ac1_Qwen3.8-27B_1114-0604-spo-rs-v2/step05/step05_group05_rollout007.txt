"""
Peak-directed exact subgradient + high-beta Nesterov LSE + simplex projection + diverse restarts.
Key differences from prior approach:
1. Proper simplex projection (Duchi et al.) instead of clip+normalize
2. Much higher LSE beta (up to 500K) for sharper max approximation
3. Exact subgradient at argmax with backtracking line search (no smoothing)
4. Stochastic peak selection to escape local minima
5. Diverse multi-length exploration (n=6962, 8192, 10240)
6. Adaptive bundle of active peaks
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using Duchi et al. (2011)."""
    n = len(v)
    if n == 0:
        return v
    v_sum = float(v.sum())
    if v_sum < 1e-15:
        return np.ones(n) / n
    # Normalize first to help numerical stability
    v = v / v_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient using FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-30:
        exp_sum = 1e-30
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_sum

    # Gradient of L w.r.t. a: dL/di = 2 * sum_m a[m] * p[m+i]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def exact_subgrad_at_peak(a, peak_idx=None):
    """
    Compute exact subgradient of max(c) w.r.t. a at a given peak.
    If peak_idx is None, use the argmax.
    Returns (subgradient, peak_value, peak_idx).
    """
    n = len(a)
    c = fast_conv(a)
    if peak_idx is None:
        peak_idx = int(np.argmax(c))
    S = float(a.sum())

    # dc[peak_idx]/da[j] = 2 * a[peak_idx - j] if 0 <= peak_idx - j < n, else 0
    # (factor of 2 because conv(a,a) is symmetric)
    j_start = max(0, peak_idx - n + 1)
    j_end = min(peak_idx, n - 1)
    grad = np.zeros(n)
    for j in range(j_start, j_end + 1):
        idx = peak_idx - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]

    # Full objective gradient: d(2n*max(c)/S^2)/da[j]
    S2 = S * S
    S3 = S2 * S
    peak_val = c[peak_idx]
    obj_grad = (2.0 * n) * (grad / S2 - 2.0 * peak_val / S3)
    return obj_grad, peak_val, peak_idx


def topk_subgrad(a, k=5, gap_threshold=0.0):
    """
    Compute weighted subgradient using top-K peaks.
    Weights are based on how much each peak exceeds the gap_threshold above the max.
    Returns (weighted_subgradient, weights).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    # Get top K peaks
    sorted_idx = np.argsort(c)[::-1]
    top_k = sorted_idx[:min(k, len(c))]
    max_val = c[sorted_idx[0]]

    # Weight: peaks within gap_threshold of max get full weight
    weights = np.zeros(len(c))
    for idx in top_k:
        w = c[idx] / max_val
        if w >= 1.0 - gap_threshold:
            weights[idx] = 1.0

    if weights.sum() < 1e-15:
        weights[sorted_idx[0]] = 1.0

    # Compute weighted subgradient
    grad = np.zeros(n)
    S2 = S * S
    S3 = S2 * S

    for idx in top_k:
        if weights[idx] < 1e-15:
            continue
        j_start = max(0, idx - n + 1)
        j_end = min(idx, n - 1)
        for j in range(j_start, j_end + 1):
            i2 = idx - j
            if 0 <= i2 < n:
                grad[j] += weights[idx] * 2.0 * a[i2]

    # Add the -2*L/S^3 term (where L is weighted average of top peaks)
    L_weighted = float((weights * c).sum()) / float(weights.sum())
    obj_grad = (2.0 * n) * (grad / S2 - 2.0 * L_weighted / S3)
    return obj_grad, float(weights.sum())


def nesterov_lse_simplex(a, beta_start, beta_end, n_iters, lr_start, lr_end,
                          mu_start=0.9, mu_end=0.95):
    """Nesterov LSE with proper simplex projection."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start + (lr_end - lr_start) * t
        mu = mu_start + (mu_end - mu_start) * t

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = project_simplex(a_look)

        _, grad = lse_score_and_grad(a_look, beta)

        # Gradient descent with momentum
        v = mu * v - lr * grad
        a_new = a + v

        # Project to simplex
        a = project_simplex(a_new)

        if it % 20 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def exact_subgrad_optimizer(a, max_time=100.0, use_stochastic=False, rng=None):
    """
    Exact subgradient at argmax with backtracking line search.
    Uses simplex projection. Optionally uses stochastic peak selection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    lr = 0.05
    lr_decay = 0.9995
    patience = 0
    patience_limit = 500

    it = 0
    while time.monotonic() - t_start < max_time:
        it += 1

        # Compute subgradient
        if use_stochastic and rng is not None and it % 10 == 0:
            # Stochastic: pick a random top-5 peak
            c = fast_conv(a)
            top5 = np.argsort(c)[::-1][:5]
            peak_idx = int(top5[rng.integers(0, 5)])
            _, _, peak_idx = exact_subgrad_at_peak(a, peak_idx)
            grad, peak_val, pk = exact_subgrad_at_peak(a, peak_idx)
        else:
            grad, peak_val, pk = exact_subgrad_at_peak(a)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Backtracking line search
        alpha = lr
        a_cur = a.copy()
        sc_cur = compute_score(a_cur) if it % 50 == 0 else best_sc

            a_trial = a_cur - alpha * grad
            a_trial = project_simplex(a_trial)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc_cur * (1.0 - 1e-6):

            alpha *= 0.5

        a = a_trial
        sc = compute_score(a)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= patience_limit:

        lr *= lr_decay
        if lr < 1e-6:
            break

    return best_a


def bundle_optimizer(a, max_time=100.0, bundle_size=10):

    penalty that includes all active peaks. Periodically update the bundle.
    """

    patience = 0

        c = fast_conv(a)
        max_c = float(c.max())

        grad = np.zeros(n)
        total_w = 0.0

            j_start = max(0, idx - n + 1)

                if 0 <= i2 < n:

        S3 = S2 * S
        L_w = float((c[idx] for idx in [int(np.argmax(c))]).__next__())

        L_w = max_c

        a = project_simplex(a + v)

            patience = 0
        else:

def fienup_modified(a, n_iters=200, momentum=0.3, thresh_start=2.5, thresh_end=1.05,
                    clip_start=0.3, clip_end=0.8):
    """Modified Fienup with momentum and annealed parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh

        above = P > bin_thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Momentum
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = project_simplex(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_descent_peaks(a, n_sweeps=5, n_top_peaks=50, n_top_contrib=30):
    """Coordinate descent targeting top peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:n_top_peaks]

        for k in top_peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Get contributions
            contribs = np.array([a[j] * a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(n_top_contrib, len(contribs))
            top_local = np.argsort(contribs)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                for factor in [0.4, 0.6, 0.8, 1.2, 1.5]:
                    a[j] = old_val * factor
                    a = project_simplex(a)
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    else:
                        a[j] = old_val
                        a = project_simplex(a)
                        break

    return best_a


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_window_init(n, window_type='raised_cosine'):
    """Build a window-shaped initialization."""
    x = np.linspace(0, 1, n)
    if window_type == 'raised_cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif window_type == 'hann':
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x)
    elif window_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif window_type == 'chirp':
        a = np.abs(np.sin(2 * np.pi * x * x * 3.0)) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B on LSE objective with simplex-like bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multistage_at_scale(a, time_budget):
    """Run multiple optimization stages at a single scale."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    def try_update(a_new, label):
        nonlocal best_a, best_sc
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    {label}: {sc:.8f}")
        return best_a

    # Quick Fienup
    if time_budget > 5:
        a_f = fienup_modified(best_a, n_iters=100, momentum=0.2,
                              thresh_start=2.5, thresh_end=1.05,
                              clip_start=0.3, clip_end=0.7)
        try_update(a_f, "fienup")

    # Short Nesterov LSE
    if time_budget > 15:
        a_nl = nesterov_lse_simplex(best_a, beta_start=50.0, beta_end=5000.0,
                                    n_iters=300, lr_start=0.01, lr_end=0.003,
                                    mu_start=0.9, mu_end=0.94)
        try_update(a_nl, "nesterov")

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses a combination of exact subgradient, high-beta LSE Nesterov,
    bundle method, Fienup, coordinate descent, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Multi-scale warm start (15% of time)
    # ============================================================
    phase1_end = t_start + 0.15 * (total_budget - 40.0)
    print(f"=== Phase 1: Multi-scale warm start (t < {phase1_end - t_start:.0f}s) ===")

    n_target = len(parent) if parent is not None else 6962
    scales = [64, 128, 256, 512, 1024, 2048, 4096]
    scales = [s for s in scales if s < n_target] + [n_target]
    scales = list(dict.fromkeys(scales))

    current_a = None
    for n in scales:
        if time.monotonic() > phase1_end:
            break
        scale_budget = phase1_end - time.monotonic()
        if scale_budget < 5:
            break

        print(f"  Scale n={n} (budget {scale_budget:.0f}s)")
        if current_a is None:
            a = build_window_init(n, 'raised_cosine')
        else:
            a = upsample(current_a, n)

        time_left = phase1_end - time.monotonic()
        if time_left > 5:
            a = multistage_at_scale(a, time_left)

        current_a = a
        sc = compute_score(a)
        print(f"    Score at n={n}: {sc:.8f}")

    if current_a is not None:
        if len(current_a) != n_target:
            current_a = upsample(current_a, n_target)
        update_best(compute_score(current_a), current_a, "multi-scale")

    # Also try parent
    if parent is not None:
        a_parent = parent.copy()
        sc_p = compute_score(a_parent)
        update_best(sc_p, a_parent, "parent")

        if time.monotonic() < phase1_end - 5:
            a_pf = fienup_modified(a_parent, n_iters=80, momentum=0.2,
                                   thresh_start=2.0, thresh_end=1.05,
                                   clip_start=0.3, clip_end=0.7)
            update_best(compute_score(a_pf), a_pf, "parent+fienup")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: High-beta Nesterov LSE with simplex projection (30%)
    # ============================================================
    phase2_end = t_start + 0.45 * (total_budget - 40.0)
    print(f"\n=== Phase 2: High-beta Nesterov LSE (t < {phase2_end - t_start:.0f}s) ===")

    # Much higher beta than before
    configs = [
        {'beta_start': 50.0, 'beta_end': 100000.0, 'lr_start': 0.01, 'lr_end': 0.002},
        {'beta_start': 100.0, 'beta_end': 200000.0, 'lr_start': 0.008, 'lr_end': 0.001},
        {'beta_start': 20.0, 'beta_end': 50000.0, 'lr_start': 0.015, 'lr_end': 0.003},
    ]

    for i, cfg in enumerate(configs):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        n_iters = int(time_left * 30)  # ~30 iterations per second
        n_iters = max(200, min(n_iters, 50000))
        print(f"  Nesterov LSE config {i}: beta [{cfg['beta_start']}, {cfg['beta_end']}], {n_iters} iters")
        a_nl = nesterov_lse_simplex(best_a,
                                    beta_start=cfg['beta_start'],
                                    beta_end=cfg['beta_end'],
                                    n_iters=n_iters,
                                    lr_start=cfg['lr_start'],
                                    lr_end=cfg['lr_end'])
        update_best(compute_score(a_nl), a_nl, f"nesterov-{i}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Exact subgradient with backtracking (20%)
    # ============================================================
    phase3_end = t_start + 0.65 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Exact subgradient + bundle (t < {phase3_end - t_start:.0f}s) ===")

    # Exact subgradient
    time_left = phase3_end - time.monotonic()
    if time_left > 20:
        a_sg = exact_subgrad_optimizer(best_a, max_time=time_left * 0.5,
                                       use_stochastic=False, rng=rng)
        update_best(compute_score(a_sg), a_sg, "subgrad-exact")

    # Stochastic subgradient
    time_left = phase3_end - time.monotonic()
    if time_left > 20:
        a_sg2 = exact_subgrad_optimizer(best_a, max_time=time_left * 0.5,
                                        use_stochastic=True, rng=rng)
        update_best(compute_score(a_sg2), a_sg2, "subgrad-stoch")

    # Bundle method
    time_left = phase3_end - time.monotonic()
    if time_left > 20:
        a_bun = bundle_optimizer(best_a, max_time=time_left * 0.8, bundle_size=15)
        update_best(compute_score(a_bun), a_bun, "bundle")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Fienup refinement (10%)
    # ============================================================
    phase4_end = t_start + 0.78 * (total_budget - 40.0)
    print(f"\n=== Phase 4: Fienup refinement (t < {phase4_end - t_start:.0f}s) ===")

    fienup_configs = [
        {'n_iters': 300, 'momentum': 0.3, 'thresh_start': 2.5, 'thresh_end': 1.02,
         'clip_start': 0.3, 'clip_end': 0.85},
        {'n_iters': 400, 'momentum': 0.4, 'thresh_start': 2.0, 'thresh_end': 1.0,
         'clip_start': 0.5, 'clip_end': 0.9},
        {'n_iters': 200, 'momentum': 0.2, 'thresh_start': 3.0, 'thresh_end': 1.1,
         'clip_start': 0.2, 'clip_end': 0.6},
    ]

    for i, cfg in enumerate(fienup_configs):
        if time.monotonic() > phase4_end:
            break
        a_f = fienup_modified(best_a, **cfg)
        update_best(compute_score(a_f), a_f, f"fienup-{i}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Diverse restarts (12%)
    # ============================================================
    phase5_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 5: Diverse restarts (t < {phase5_end - t_start:.0f}s) ===")

    n_diverse = 5
    per_restart = max(10, (phase5_end - time.monotonic()) / n_diverse)

    for i in range(n_diverse):
        if time.monotonic() > phase5_end - 10:
            break
        # Create diverse perturbation
        noise_scale = 0.05 + 0.05 * i
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_pert = project_simplex(a_pert)

        # Quick optimization
        a_opt = nesterov_lse_simplex(a_pert, beta_start=100.0, beta_end=100000.0,
                                     n_iters=max(100, int(per_restart * 20)),
                                     lr_start=0.01, lr_end=0.002)
        update_best(compute_score(a_opt), a_opt, f"restart-{i}")

    # Also try different window inits
    for wtype in ['hann', 'gaussian', 'chirp']:
        if time.monotonic() > phase5_end - 10:
            break
        a_w = build_window_init(len(best_a), wtype)
        a_w = project_simplex(a_w)
        a_opt = nesterov_lse_simplex(a_w, beta_start=50.0, beta_end=50000.0,
                                     n_iters=500, lr_start=0.01, lr_end=0.002)
        update_best(compute_score(a_opt), a_opt, f"window-{wtype}")

    print(f"After Phase 5: best = {best_score:.8f}")

    # ============================================================
    # Phase 6: L-BFGS polish + coordinate descent (8%)
    # ============================================================
    phase6_end = deadline
    print(f"\n=== Phase 6: L-BFGS + CD polish (t < {phase6_end - t_start:.0f}s) ===")

    # Coordinate descent
    if time.monotonic() < phase6_end - 30:
        a_cd = coordinate_descent_peaks(best_a, n_sweeps=3, n_top_peaks=30, n_top_contrib=20)
        update_best(compute_score(a_cd), a_cd, "coord-desc")

    # L-BFGS with high beta
    lbfgs_betas = [5000, 20000, 50000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase6_end - 15:
            break
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, f"lbfgs-{beta}")

    # Final coordinate descent
    if time.monotonic() < phase6_end - 10:
        a_cd2 = coordinate_descent_peaks(best_a, n_sweeps=2, n_top_peaks=20, n_top_contrib=15)
        update_best(compute_score(a_cd2), a_cd2, "cd-final")

    # ============================================================
    # Try longer sequences if time permits
    # ============================================================
    if time.monotonic() < deadline - 100:
        print(f"\n=== Trying longer sequence n=8192 ===")
        a_long = upsample(best_a, 8192)
        a_long = project_simplex(a_long)
        a_long_opt = nesterov_lse_simplex(a_long, beta_start=100.0, beta_end=100000.0,
                                          n_iters=2000, lr_start=0.008, lr_end=0.001)
        sc_long = compute_score(a_long_opt)
        if sc_long < best_score:
            update_best(sc_long, a_long_opt, "n8192")
        else:
            print(f"  n=8192: {sc_long:.8f} (not better than {best_score:.8f})")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 6962

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
