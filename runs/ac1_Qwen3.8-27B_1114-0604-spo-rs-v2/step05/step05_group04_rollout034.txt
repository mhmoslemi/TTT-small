"""
Adaptive-beta LSE/L-BFGS optimizer with Fienup hybrid for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key features:
- Continuation method: ramp up LSE beta from low to high
- FFT-based efficient gradient computation
- Hybrid Fienup + L-BFGS for escaping local minima
- Multi-start with diverse spectral initializations
- Adaptive time allocation across phases

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # Use c - c_max for numerical stability
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient: p[k] = exp(beta*(c[k]-c_max)) / sum(exp(beta*(c-j-c_max)))
    p = exp_scaled / exp_sum
    
    # grad_LSE[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # This is 2 * cross-correlation of p and a
    # Compute via FFT: corr[i] = ifft(FFT(p) * conj(FFT(a)))[i]
    M2 = 2 * M - 1  # Need length >= M + n - 1 for the correlation
    # Actually, we need M_fft to be large enough for the correlation of p (length M) and a (length n)
    # The correlation has length M + n - 1
    M_corr = M + n - 1
    M_corr_fft = 1 << (M_corr - 1).bit_length()
    
    P_fft = np.fft.rfft(p, M_corr_fft)
    A_fft = np.fft.rfft(x, M_corr_fft)
    corr = np.fft.irfft(P_fft * np.conj(A_fft), M_corr_fft)
    # corr[i] = sum_k p[k] * a[(k-i) mod M_corr_fft]
    # For i in 0..n-1, and k in 0..M-1, we need k-i in 0..n-1
    # corr[i] = sum_{k=i}^{i+n-1} p[k] * a[k-i] (for the linear part)
    # But we also need k in 0..M-1, so the valid range is k from i to min(M-1, i+n-1)
    # Since M = 2n-1 and i < n, i+n-1 < 2n-1 = M, so the range is valid.
    # Actually wait: corr[i] as computed by the FFT circular correlation includes wraparound.
    # For M_corr_fft >= M + n - 1, there's no wraparound for the linear correlation.
    # corr[i] = sum_{k=0}^{M-1} p[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
    # = sum_{k=max(0,i)}^{min(M-1, i+n-1)} p[k] * a[k-i]
    
    # The gradient is 2 * corr[i] for i = 0, ..., n-1
    g_L = 2.0 * corr[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on the LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_shape(a, n_iters=200, top_frac=0.2, gamma=0.7):
    """Fienup-style spectral shaping to reduce peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)
        
        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if above.any():
            scale = np.ones_like(P)
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 0.001)) * gamma
            A = A * scale
        
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def spectral_eq(a, n_iters=150, alpha=0.5):
    """Spectral equalization: push towards flat magnitude spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        # Relax towards flat
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A = A * amp_ratio
        
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def multi_peak_subgrad(a, n_iters=3000, top_frac=0.05, step_init=0.02, decay=0.9997):
    """Subgradient method using top fraction of peaks with soft weighting."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    n_top = max(3, int(M * top_frac))
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top peaks with soft weighting
        c_sorted = np.sort(c)[::-1]
        thresh = c_sorted[n_top - 1]
        
        # Soft weights: exp((c - thresh) / (M_val - thresh + 1e-10))
        diff = c - thresh
        width = M_val - thresh + 1e-10
        weights = np.exp(np.maximum(0, diff) / width)
        # Only keep top peaks
        weights[c < thresh] = 0
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum
        
        # Gradient: 2 * sum_k weights[k] * a[k-i]
        M_corr = M + n - 1
        M_corr_fft = 1 << (M_corr - 1).bit_length()
        W_fft = np.fft.rfft(weights, M_corr_fft)
        A2_fft = np.fft.rfft(a, M_corr_fft)
        corr = np.fft.irfft(W_fft * np.conj(A2_fft), M_corr_fft)
        sg = 2.0 * corr[:n]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = min(step, max((sc - 1.0) / (gn * gn + 1e-30) * 0.5, 1e-12))
        step = min(step, 0.05)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a


def make_random_init(n, rng):
    """Create a random non-negative initialization with flat-ish spectrum."""
    # Random phase, flat magnitude
    M_fft = 1 << (2 * n - 1).bit_length()
    n_bins = M_fft // 2 + 1
    mag = np.ones(n_bins)
    phase = 2 * np.pi * rng.random(n_bins)
    A = mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    A_full = np.fft.irfft(A, M_fft)
    a = A_full[:n]
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) * 0.5
    else:
        a = a / S
    return a


def make_tapered_init(n, rng, taper_type='hann'):
    """Create a tapered initialization."""
    x = np.arange(n) / n
    if taper_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif taper_type == 'hamming':
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    elif taper_type == 'gauss':
        a = np.exp(-4.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    # Add small random perturbation
    a = a + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

"""
Adaptive-beta LSE/L-BFGS continuation optimizer with hybrid Fienup for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
- Efficient continuation method with adaptive beta ramping
- Correct FFT-based cross-correlation gradient (uses same FFT size as convolution)
- Focused multi-peak subgradient with soft exponential weighting
- Diverse initializations including spectral-domain random and structured patterns
- Better time management with early termination on stagnation
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based linear autocorrelation (a convolved with itself)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and gradient for scipy minimize.
    Objective: 2n * LSE_beta(conv(x,x)) / (sum(x))^2
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Softmax weights on peaks
    p = exp_scaled / exp_sum

    # Gradient: grad_LSE[i] = 2 * sum_k p[k] * a[k-i]
    # Computed as 2 * circular cross-correlation of p and a, using same M_fft
    P_fft = np.fft.rfft(p, M_fft)
    # Reuse A (already computed) - x is zero-padded to M_fft
    corr = np.fft.irfft(P_fft * np.conj(A), M_fft)[:n]
    g_L = 2.0 * corr

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1.0 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if above.any():
            scale = np.ones_like(P)
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-6)) * gamma
            A = A * scale

        a_new = np.fft.irfft(A, M_fft)[:n]

        # Soft non-negativity via sigmoid gate
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def spectral_eq(a, n_iters=150, alpha_init=0.3, alpha_end=0.8):
    """Spectral equalization: relax magnitude spectrum towards flat."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A = A * amp_ratio

        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=3000, top_frac=0.05, step_init=0.02, decay=0.9998):
    """
    Subgradient method using top fraction of peaks with exponential soft weights.
    More robust than single-peak subgradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    n_top = max(3, int(M * top_frac))

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Soft weights on top peaks
        c_sorted = np.sort(c)[::-1]
        thresh = c_sorted[n_top - 1]
        width = M_val - thresh + 1e-10
        weights = np.zeros_like(c)
        mask = c >= thresh
        weights[mask] = np.exp((c[mask] - thresh) / width)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Gradient via cross-correlation
        P_fft = np.fft.rfft(weights, M_fft)
        A2 = np.fft.rfft(a, M_fft)
        corr = np.fft.irfft(P_fft * np.conj(A2), M_fft)[:n]
        sg = 2.0 * corr

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step: cap by score gap
        step = min(step, max(0.5 * (sc - 1.001) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.05)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def make_random_flat_spectrum(n, rng):
    """Random phase, flat magnitude spectrum -> non-negative sequence."""
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    n_bins = M_fft // 2 + 1
    mag = np.ones(n_bins)
    phase = 2.0 * np.pi * rng.random(n_bins)
    A = mag * np.exp(1j * phase)
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) * 0.5
    return a / S


def make_tapered_init(n, rng, taper_type='hann'):
    """Tapered initialization with small random perturbation."""
    x = np.arange(n) / n
    if taper_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif taper_type == 'gauss':
        a = np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.05
    elif taper_type == 'bipeak':
        a = np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.02)
    else:
        a = np.ones(n) * 0.5
    a = a + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def spectral_perturb(a, rng, strength=0.3):
    """Perturb the magnitude spectrum randomly."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    mask = 1.0 + strength * rng.standard_normal(len(A))
    mask = np.maximum(mask, 0.1)
    A = A * mask
    a_new = np.fft.irfft(A, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses continuation L-BFGS (ramping beta), hybrid Fienup, and subgradient.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None
    stale_counter = 0
    improve_threshold = 1e-10

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a, stale_counter
        if sc < best_score - improve_threshold:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            stale_counter = 0
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent score: {sc_p:.8f} (n={len(parent)})")
        update_best(sc_p, parent, "PARENT")

    if best_a is None:
        best_a = np.ones(4096) * 0.5
        update_best(compute_score(best_a), best_a, "INIT-UNIFORM")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Continuation L-BFGS (30% time)
    # Adaptive beta ramping from parent
    # ============================================================
    phase1_end = t_start + 0.30 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Continuation L-BFGS (deadline t={phase1_end - t_start:.0f}s) ===")

    a_current = best_a.copy() / best_a.sum()
    betas = [20, 50, 100, 200, 500]

    for beta in betas:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 20:
            break

        # Estimate function eval time
        eval_time_est = n_work * np.log2(n_work) / 8e5
        maxiters = max(300, min(int(remaining * 0.7 / (eval_time_est * 8)), 15000))
        maxfun = maxiters * 12

        print(f"  L-BFGS beta={beta}, maxiter={maxiters}, time_left={remaining:.0f}s")
        t_call = time.monotonic()
        sc, a_opt = run_lbfgs(a_current, beta, maxiter=maxiters, maxfun=maxfun)
        call_time = time.monotonic() - t_call
        improved = update_best(sc, a_opt, f"P1-b={beta}")

        if improved:
            a_current = a_opt.copy() / a_opt.sum()
        else:
            # Still use as starting point for next beta
            S = a_opt.sum()
            if S > 1e-12:
                a_current = a_opt / S

        print(f"    score={sc:.8f}, call_time={call_time:.1f}s, improved={improved}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Diverse multi-start (30% time)
    # Fienup + spectral eq + L-BFGS from different inits
    # ============================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Multi-start (deadline t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    n_starts = 10

    for s in range(n_starts):
        if time.monotonic() > phase2_end - 30:
            break

        # Generate diverse initialization
        if s == 0:
            a_init = best_a.copy()
        elif s == 1:
            a_init = make_random_flat_spectrum(n_best, rng)
        elif s == 2:
            a_init = make_tapered_init(n_best, rng, 'hann')
        elif s == 3:
            a_init = make_tapered_init(n_best, rng, 'gauss')
        elif s == 4:
            a_init = make_tapered_init(n_best, rng, 'bipeak')
        elif s == 5:
            a_init = spectral_perturb(best_a, rng, strength=0.3)
        elif s == 6:
            a_init = spectral_perturb(best_a, rng, strength=0.6)
        elif s == 7:
            # Perturbed best
            a_init = best_a * (1.0 + 0.08 * rng.standard_normal(n_best))
            a_init = np.clip(a_init, 0.0, 1000.0)
            a_init = a_init / a_init.sum()
        elif s == 8:
            # Alt length
            n_alt = n_best + 499
            a_init = make_random_flat_spectrum(n_alt, rng)
        else:
            # Structured: alternating blocks
            n_block = n_best // 6
            a_init = np.zeros(n_best)
            for b in range(6):
                val = 1.0 if b % 2 == 0 else 0.4
                a_init[b * n_block:(b + 1) * n_block] = val
            a_init += 0.03 * rng.standard_normal(n_best)
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init / a_init.sum()

        n_init = len(a_init)

        # Step 1: Fienup shaping
        a_f = fienup_shape(a_init, n_iters=80, top_frac_init=0.10,
                           top_frac_end=0.40, gamma=0.7, soft_alpha=25.0)
        sc_f = compute_score(a_f)
        update_best(sc_f, a_f, f"P2-FIENUP s={s}")

        # Step 2: Spectral equalization from Fienup result
        if time.monotonic() < phase2_end - 30:
            a_eq = spectral_eq(a_f, n_iters=60, alpha_init=0.3, alpha_end=0.7)
            sc_eq = compute_score(a_eq)
            update_best(sc_eq, a_eq, f"P2-SPEQ s={s}")

        # Step 3: L-BFGS from the best of Fienup/SPEQ
        if time.monotonic() < phase2_end - 30:
            a_seed = a_f if sc_f <= sc_eq else a_eq
            n_seed = len(a_seed)
            remaining = phase2_end - time.monotonic()
            eval_time_est = n_seed * np.log2(n_seed) / 8e5
            maxiters = max(200, min(int(remaining * 0.4 / (eval_time_est * 8)), 5000))
            sc_l, a_l = run_lbfgs(a_seed, 100, maxiter=maxiters, maxfun=maxiters * 10)
            update_best(sc_l, a_l, f"P2-LBFGS s={s}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: High-beta L-BFGS refinement (25% time)
    # Push towards exact max with increasing beta
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: High-beta refinement (deadline t={phase3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_current = best_a.copy() / best_a.sum()
    high_betas = [200, 500, 1000, 2000, 5000]

    for beta in high_betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break

        eval_time_est = n_best * np.log2(n_best) / 8e5
        maxiters = max(500, min(int(remaining * 0.6 / (eval_time_est * 8)), 20000))
        maxfun = maxiters * 12

        print(f"  L-BFGS beta={beta}, maxiter={maxiters}, time_left={remaining:.0f}s")
        sc, a_opt = run_lbfgs(a_current, beta, maxiter=maxiters, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"P3-b={beta}")

        if improved:
            a_current = a_opt.copy() / a_opt.sum()
        else:
            S = a_opt.sum()
            if S > 1e-12:
                a_current = a_opt / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Subgradient + final refinement (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Subgradient + final ===")

    for tf in [0.02, 0.05, 0.10]:
        if time.monotonic() > deadline - 60:
            break
        a_sg = multi_peak_subgrad(best_a, n_iters=2000, top_frac=tf,
                                  step_init=0.015, decay=0.9998)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"P4-SG tf={tf}")

    # Final Fienup + L-BFGS
    if time.monotonic() < deadline - 90:
        a_f = fienup_shape(best_a, n_iters=80, top_frac_init=0.08,
                           top_frac_end=0.35, gamma=0.6, soft_alpha=20.0)
        sc = compute_score(a_f)
        update_best(sc, a_f, "P4-FIENUP")

        if time.monotonic() < deadline - 40:
            sc2, a_opt = run_lbfgs(a_f, 1000, maxiter=3000, maxfun=30000)
            update_best(sc2, a_opt, "P4-LBFGS")

    # Final perturbation restarts
    if time.monotonic() < deadline - 30:
        for trial in range(4):
            if time.monotonic() > deadline - 15:
                break
            noise = 0.02 * (1.0 + trial * 0.5)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            a_f = fienup_shape(a_p, n_iters=60, top_frac_init=0.10,
                               top_frac_end=0.30, gamma=0.7)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"P4-RESTART-{trial}")

            if time.monotonic() < deadline - 15:
                sc2, a_opt = run_lbfgs(a_f, 500, maxiter=2000, maxfun=20000)
                update_best(sc2, a_opt, f"P4-RESTART-{trial}-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
