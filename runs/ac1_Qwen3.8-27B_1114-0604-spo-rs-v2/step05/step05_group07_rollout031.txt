13. **Practical approach: Iterative reconstruction / "peaking" algorithm**

    Here's an idea inspired by iterative algorithms in signal processing:
    
    Algorithm "Peak Reduction":
    1. Start with a (normalized, sum=1)
    2. Compute c = conv(a, a)
    3. Find k* = argmax(c)
    4. The peak c_{k*} = sum_{j} a_j * a_{k*-j}
    5. To reduce c_{k*}, we need to reduce the "overlap" between a and its reversed copy at shift k*
    6. One way: reduce the values of a that contribute most to c_{k*}, and redistribute to positions that don't contribute to c_{k*} (or contribute less to other peaks)
    
    This is essentially what the "targeted peak flattening" in the previous code does, but we can make it smarter.

"""
Optimized sequence search using exact pairwise coordinate descent (1D convex
subproblems), multi-peak subgradient with Nesterov momentum, and high-beta
LSE-smoothed L-BFGS polishing. The key innovation is the exact 1D coordinate
descent that exploits the linearity of c_k in each coordinate pair, enabling
precise peak reduction without approximations.
"""

import numpy as np
from scipy.optimize import minimize
import time

# ============================================================
# Core computation functions
# ============================================================

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


_current_beta = 1000.0

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def pairwise_coord_desc(a, n_passes=50, rng=None, verbose=False, t_deadline=None):
    """
    Exact pairwise coordinate descent. For each pair (j,i), the objective
    max_k c_k is linear in the transfer delta between j and i, so we can
    find the exact optimal delta via 1D convex optimization.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    if verbose:
        print(f"  CoordDesc start: score={best_sc:.10f}")
    
    for p in range(n_passes):
        if t_deadline is not None and time.monotonic() > t_deadline:
            break
        
        c = fast_conv(a)
        max_c = float(np.max(c))
        
        # Random permutation of coordinates
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        improved = 0
        for j in order:
            if t_deadline is not None and time.monotonic() > t_deadline:
                break
            
            # Choose partner i: use a "shifted" coordinate
            i = (j + n // 3) % n
            if i == j:
                i = (j + 1) % n
            
            # Compute d vector: d[k] = 2*(a[k-j] - a[k-i]) for valid indices
            d = np.zeros(2 * n - 1)
            # Contribution from j: for k in [j, j+n-1], d[k] += 2*a[k-j]
            d[j:j + n] += 2.0 * a
            # Contribution from i: for k in [i, i+n-1], d[k] -= 2*a[k-i]
            d[i:i + n] -= 2.0 * a
            
            # Feasible range for delta: a[j] + delta >= 0, a[i] - delta >= 0
            lo = -a[j]
            hi = a[i]
            
            if hi - lo < 1e-14:
                continue
            
            # f(delta) = max_k (c[k] + delta * d[k]) - convex piecewise linear
            # Use golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_t = lo
            b_t = hi
            for _ in range(50):
                m1 = a_t + (1.0 - gr) * (b_t - a_t)
                m2 = a_t + gr * (b_t - a_t)
                f1 = float(np.max(c + m1 * d))
                f2 = float(np.max(c + m2 * d))
                if f1 < f2:
                    b_t = m2
                else:
                    a_t = m1
                if b_t - a_t < 1e-14 * max(1.0, hi - lo):
                    break
            
            delta_star = (a_t + b_t) / 2.0
            delta_star = max(lo, min(hi, delta_star))
            
            if abs(delta_star) > 1e-12:
                a[j] += delta_star
                a[i] -= delta_star
                improved += 1
        
        # Renormalize (should be nearly zero change since sum is preserved)
        S_new = a.sum()
        if abs(S_new - S) > 1e-10:
            a = a / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if verbose and (p % 10 == 0):
            print(f"    Pass {p}: score={best_sc:.10f}, improved={improved}")
    
    if verbose:
        print(f"  CoordDesc end: score={best_sc:.10f}")
    return best_a


def multi_partial_coord_desc(a, n_passes=30, rng=None, t_deadline=None, verbose=False):
    """

    For each j, try several different i values and take the best.
    """
    n = len(a)

        return a

    best_a = a.copy()
    best_sc = compute_score(a)

    offsets = [n // 2, n // 3, n // 4, 1, n // 5, 2 * n // 3]

            break
        
        c = fast_conv(a)
        
        if rng is not None:

        else:
            order = np.arange(n)
        
        for j in order:
            if t_deadline is not None and time.monotonic() > t_deadline:
                break
            
            best_delta = 0.0
            best_f = compute_score(a)
            partner_i = None
            
            for off in offsets:
                i = (j + off) % n
                if i == j:
                    continue
                
                d = np.zeros(2 * n - 1)
                d[j:j + n] += 2.0 * a
                d[i:i + n] -= 2.0 * a
                
                lo = -a[j]
                hi = a[i]

                    continue
                
                # Quick check: evaluate at a few points

                a_t = lo

                    f1 = float(np.max(c + m1 * d))

                    else:

def nesterov_subgrad_v2(a, n_iters=5000, f_lb=1.0, momentum_init=0.9, t_deadline=None):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    mu = momentum_init
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if t_deadline is not None and time.monotonic() > t_deadline:
            break
        
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        # Subgradient of max(c) = 2 * rev_seg at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * y[k_star - idx]
        
        # Full gradient of the score
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)
        
        # Center the gradient (projection onto sum=0 Hyperplane)
        grad = grad - np.mean(grad)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        gap = max(sc - f_lb, 0.01)
        step = gap / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        # Decay momentum
        mu = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / t_new
        
        y_new = y - step * grad
        a_new = y_new + mu * (y_new - a)
        
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a

def multi_peak_subgrad_v2(a, n_iters=3000, f_lb=1.0, top_k=8, t_deadline=None):
    """Subgradient using top-k peaks with exponential weighting."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if t_deadline is not None and time.monotonic() > t_deadline:
            break
        
        c = fast_conv(a)
        S_a = float(a.sum())
        
        # Top-k peaks
        top_idx = np.partition(c, -top_k)[-top_k:]
        top_pos = np.where(c >= top_idx[0])[0]
        if len(top_pos) > top_k:
            # Take top_k
            vals = c[top_pos]
            order = np.argsort(vals)[::-1][:top_k]
            top_pos = top_pos[order]
        
        # Exponential weights
        max_val = float(c[top_pos].max())
        weights = np.exp(-0.5 * (max_val - c[top_pos]))
        weights = weights / weights.sum()
        
        sc = 2.0 * n * max_val / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Weighted subgradient
        sg = np.zeros(n)
        for idx in top_pos:
            w = float(weights[list(top_pos).index(idx)])
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]
        
        weighted_max = float(np.dot(weights, c[top_pos]))
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * weighted_max / S3)
        grad = grad - np.mean(grad)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = max(sc - f_lb, 0.01) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        step *= (1.0 - 0.001 * it)
        
        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    
    return best_a

def newton_active_set(a, n_iters=200, t_deadline=None):
    """
    Newton method targeting the active set of constraints.
    At each step, identify active peaks and solve a projected Newton system.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if t_deadline is not None and time.monotonic() > t_deadline:
            break
        
        c = fast_conv(a)
        max_c = float(c.max())
        sc = 2.0 * n * max_c / 1.0  # since sum=1
        
        # Active set: peaks within 1% of max
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += 2.0 * a[k - idx]
        
        # Center for sum constraint
        grad = grad - np.mean(grad)
        
        # Approximate Hessian: use the dominant active constraint
        k_main = int(np.argmax(c))
        H_diag = np.zeros(n)
        j_start = max(0, k_main - n + 1)
        j_end = min(k_main, n - 1)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            H_diag[idx] = 2.0 * 2.0 * a  # Hessian of c_k is 2*S_k, diagonal approx
        
        # Simple diagonal Newton step with line search
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        # Step: solve (H_diag + lambda*I) * delta = -grad
        lam = gnorm * 0.1  # regularization
        delta = -grad / (H_diag + lam)
        
        # Clip to maintain non-negativity
        delta_max_pos = np.where(delta > 0, a / delta, np.inf)
        delta_max_neg = np.where(delta < 0, -a / (-delta), np.inf)
        max_alpha = np.minimum(delta_max_pos, delta_max_neg).min()
        max_alpha = min(max_alpha, 1.0, 0.5)
        
        if max_alpha < 1e-15:
            # Use gradient direction instead
            step = 2.0 * (sc - 1.0) / (gnorm * gnorm + 1e-30)
            step = min(step, 0.3)
            a_new = a - step * grad
        else:
            a_new = a + max_alpha * delta
        
        a_new = np.clip(a_new, 0.0, 1000.0)
        Sa = a_new.sum()
        if Sa < 1e-12:
            break
        a_new = a_new / Sa
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def perturb(a, rng, scale=0.05):
    """Generate a perturbed version."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 4 + int(rng.integers(0, 12))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + 2.0 * scale * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = a[j_start:j_end + 1][::-1].copy()
        a_p = a.copy()
        top_count = max(10, n // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
    else:
        # Multiplicative noise
        a_p = a * np.exp(scale * rng.standard_normal(n))
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

    else:
        n = 8192
        a = np.ones(n) * 0.5
        print(f"No parent. Using uniform n={n}")
    
    n = len(a)
    
    # ============================================================
    # Phase 1: Pairwise Coordinate Descent (35% time)
    # ============================================================
    phase1_deadline = t_start + 0.35 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Pairwise Coordinate Descent (n={n}) ---")
    
    # First: single-pair coordinate descent
    a1 = pairwise_coord_desc(a, n_passes=80, rng=rng, verbose=True, t_deadline=phase1_deadline)
    report(compute_score(a1), a1, "coord_desc_basic")
    
    # Second: multi-partial coordinate descent
    if time.monotonic() < phase1_deadline - 30:
        a2 = multi_partial_coord_desc(a1, n_passes=30, rng=rng, 
                                       t_deadline=phase1_deadline, verbose=False)
        report(compute_score(a2), a2, "coord_desc_multi")
    
    print(f"After Phase 1: best = {best_score:.10f}")
    
    # ============================================================
    # Phase 2: LSE + L-BFGS (30% time)
    # ============================================================
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LSE + L-BFGS Refinement ---")
    
    beta_schedule = [200, 500, 1000, 3000, 10000, 30000, 100000]
    
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline - 20:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 40:
            break
        maxiter = int(min(40000, max(2000, remaining * 30)))
        maxfun = int(min(150000, max(5000, remaining * 100)))
        
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        
        # Try a perturbed start for higher betas
        if beta >= 1000 and time.monotonic() < phase2_deadline - 60:
            a_pert = perturb(best_a, rng, scale=0.02)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")
    
    print(f"After Phase 2: best = {best_score:.10f}")
    
    # ============================================================
    # Phase 3: Coordinate Descent again from improved point (20% time)
    # ============================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Coordinate Descent Polish ---")
    
    a3 = pairwise_coord_desc(best_a, n_passes=60, rng=rng, verbose=True, t_deadline=phase3_deadline)
    report(compute_score(a3), a3, "coord_desc_phase3")
    
    if time.monotonic() < phase3_deadline - 20:
        a3b = multi_partial_coord_desc(best_a, n_passes=25, rng=rng, t_deadline=phase3_deadline)
        report(compute_score(a3b), a3b, "coord_multi_phase3")
    
    print(f"After Phase 3: best = {best_score:.10f}")
    
    # ============================================================
    # Phase 4: Subgradient + Newton (10% time)
    # ============================================================
    phase4_deadline = t_start + 0.95 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Subgradient + Newton ---")
    
    a_ns = nesterov_subgrad_v2(best_a, n_iters=3000, f_lb=1.0, t_deadline=phase4_deadline)
    report(compute_score(a_ns), a_ns, "nesterov_v2")
    
    if time.monotonic() < phase4_deadline - 20:
        a_mp = multi_peak_subgrad_v2(best_a, n_iters=2000, f_lb=1.0, top_k=10, 
                                       t_deadline=phase4_deadline)
        report(compute_score(a_mp), a_mp, "multi_peak")
    
    if time.monotonic() < phase4_deadline - 15:
        a_newt = newton_active_set(best_a, n_iters=100, t_deadline=phase4_deadline)
        report(compute_score(a_newt), a_newt, "newton")
    
    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_final_cd = pairwise_coord_desc(best_a, n_passes=30, rng=rng, t_deadline=deadline)
        report(compute_score(a_final_cd), a_final_cd, "final_coord")
    
    # ============================================================
    # Iterated local search with different perturbations
    # ============================================================
    if time.monotonic() < deadline - 60:
        print(f"\n--- ILS: Iterated Local Search ---")
        for trial in range(10):
            if time.monotonic() > deadline - 30:
                break
            
            scale = 0.01 + 0.03 * rng.random()
            a_pert = perturb(best_a, rng, scale=scale)
            
            # Quick coordinate descent
            a_cd = pairwise_coord_desc(a_pert, n_passes=20, rng=rng, 
                                         t_deadline=deadline - 30)
            report(compute_score(a_cd), a_cd, f"ILS#{trial} cd")
            
            # Quick L-BFGS
            if time.monotonic() < deadline - 30:
                sc, a_lb = run_lbfgs(a_cd, 30000, maxiter=20000, maxfun=80000)
                report(sc, a_lb, f"ILS#{trial} lbfgs")
    
    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
