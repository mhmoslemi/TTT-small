def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    m = d.max()
    if m > 1e-12:
        d = d / m
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

    return lse_score_grad(a, float(beta))


def frequency_domain_init(n, rng, target_type="flat", n_iters=200):
    """Iterative projection in frequency domain to find non-negative sequence
    with approximately flat convolution."""
    N = next_power_of_two(2 * n - 1)
    
    # Target magnitude spectrum
    if target_type == "flat":
        target_mag = np.ones(N // 2 + 1, dtype=np.float64)
    elif target_type == "raised_cosine":
        x = np.linspace(0, np.pi, N // 2 + 1)
        target_mag = np.cos(0.5 * x) ** 2
        target_mag = np.maximum(target_mag, 0.01)
    elif target_type == "tapered":
        x = np.linspace(0, 1, N // 2 + 1)
        target_mag = 1.0 - 0.5 * x ** 2
    else:
        target_mag = np.ones(N // 2 + 1, dtype=np.float64)
    
    # Initialize with random phase
    phase = rng.uniform(0, 2 * np.pi, size=N // 2 + 1)
    A_full = np.zeros(N // 2 + 1, dtype=np.complex128)
    A_full = target_mag * np.exp(1j * phase)
    A_full[0] = float(np.sqrt(N))  # DC component
    
    for it in range(n_iters):
        # Inverse FFT
        a_full = np.fft.irfft(A_full, N)
        a = a_full[:n]
        
        # Project onto non-negative
        a = np.maximum(a, 0.0)
        
        # Normalize
        S = float(a.sum())
        if S < 1e-12:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / S
        
        # Forward FFT
        A_new = np.fft.rfft(a, N)
        
        # Replace magnitude, keep phase
        mag_new = np.abs(A_new)
        phase_new = np.angle(A_new)
        A_full = target_mag * np.exp(1j * phase_new)
        A_full[0] = float(np.sqrt(N))
        
        # Soft projection: blend to converge
        if it > 50:
            blend = min(1.0, (it - 50) / 100.0)
            A_full = (1 - blend) * A_full + blend * target_mag * np.exp(1j * phase_new)
            A_full[0] = float(np.sqrt(N))
    
    return a

        add(pr)

    Nf = next_power_of_two(n)

        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))

        add(vals)
    
    # Bimodal
    for width in (0.08, 0.12, 0.2):
        for center in (0.25, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1 - center)) / width) ** 2)) + 0.1)
    
    # Pure random
    for _ in range(3):

    return inits


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def subgrad_step(a, eta, k_active=3):
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return a
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return a
    w = active_weights(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def subgrad_polish(a, max_iters=20, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc
        
        for k in (1, 2, 3, 5, 8, 12):
            for mult in (0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step(a, mult, k)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()
        
        if cur_best is None:
            break
        a = cur_best
    
    return float(best_sc), best_a


def coordinate_descent_polish(a, deadline, max_rounds=5):
    """Try pairwise mass transfers to reduce the peak convolution."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    rng = np.random.default_rng(12345)
    
    for round_idx in range(int(max_rounds)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        d, S = d_from_a(a)
        # Find active set (where convolution is near max)
        dmax = float(np.max(d))
        if dmax < 1e-12:
            break
        threshold = dmax * 0.95
        active_lags = np.where(d >= threshold)[0]
        
        # For each active lag, find which pairs contribute most
        # and try to reduce them
        for _ in range(3):
            if deadline is not None and time.monotonic() > float(deadline):
                break
            
            # Pick a random pair of indices
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            # Try transferring mass
            for frac in (0.02, 0.05, 0.1, 0.2, 0.3):
                q = a.copy()
                transfer = q[i] * frac
                q[i] -= transfer
                q[j] += transfer
                if q[i] < 0:
                    continue
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    break
    
    return float(best_sc), best_a

def run_lbfgs(a0, beta, deadline, maxfun):

        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    mf = max(64, min(5000, mf))
    return mf


def optimize_trajectory(a_start, deadline, rng, label):
    """Main optimization loop with LSE continuation, subgradient, and
    frequent interleaving of different polishing methods."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    n = len(a)
    total_time = max(10.0, float(deadline - t0 - 5.0))
    
    print("[{}] start score={:.10f} n={}".format(label, best_sc, n))
    
    # Continuation schedule: geometric progression of beta
    # Start low, end very high
    betas = [50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600, 51200, 100000]
    
    # Allocate time proportionally (more time for higher betas)
    weights = np.array([1.0, 1.0, 1.2, 1.3, 1.5, 1.6, 1.8, 2.0, 2.2, 2.5, 3.0, 4.0])
    weights = weights / weights.sum()
    
    cum_time = 0.0
    for idx, beta in enumerate(betas):
        sub_dead = t0 + (cum_time + weights[idx]) * total_time
        sub_dead = min(float(sub_dead), float(deadline - 5.0))
        
        if time.monotonic() >= sub_dead - 3.0:
            continue
        
        # Multiple LSE iterations at this beta
        n_iters = int(weights[idx] * 4)
        n_iters = max(2, min(n_iters, 20))
        
        for it in range(n_iters):
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.8, n)
            if mf < 64:
                break
            
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] beta={} it={}: {:.10f}".format(label, beta, it, best_sc))
            a = normalize_a(a2)
        
        # Subgradient polish after each beta stage
        if time.monotonic() < sub_dead - 3.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=8, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad after beta={}: {:.10f}".format(label, beta, best_sc))
            a = normalize_a(a_sg)
        
        # Coordinate descent every few stages
        if idx % 3 == 2 and time.monotonic() < sub_dead - 3.0:
            sc_cd, a_cd = coordinate_descent_polish(a, deadline=sub_dead, max_rounds=2)
            if np.isfinite(sc_cd) and sc_cd < best_sc:
                best_sc = sc_cd
                best_a = a_cd.copy()
                print("[{}] coord after beta={}: {:.10f}".format(label, beta, best_sc))
            a = normalize_a(a_cd)
        
        cum_time += weights[idx]
    
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)
    
    # Try to use parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None
    
    print("Parent: {} (len={})".format(
        parent is not None, parent.size if parent is not None else 0))
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        plen = int(min(parent.size, 16384))
        if plen < 2048:
            plen = 4096
        lengths.append(plen)
        for L in [8192, 10240, 12288, 16384, 6144]:
            if L != plen:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288, 16384, 6144]
    
    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:5]
    print("Lengths: {}".format(lengths))
    
    # Phase 1: Generate and evaluate initializations
    all_scores = []
    all_seqs = []
    all_lens = []
    
    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=15)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))
    
    if not all_scores:
        n0 = lengths[0] if lengths else 8192
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)
    
    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init: {:.10f} (len={})".format(all_scores[int(order[0])], all_lens[int(order[0])]))
    
    # Select top diverse starts
    starts = []
    used_lens = set()
    
    if parent is not None:
        plen = int(min(parent.size, 16384))
        if plen < 2048:
            plen = 4096
        ps = normalize_a(resample_sequence(parent, plen))
        starts.append(ps)
        used_lens.add(plen)
    
    for idx in order:
        if len(starts) >= 4:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)
    
    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))
    
    # Evaluate starts
    start_scores = []
    for s in starts:
        sc = exact_score_a(s)
        start_scores.append(sc)
    
    best_seq = starts[0].copy()
    best_score = min(start_scores)
    for i, sc in enumerate(start_scores):
        if sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()
    
    print("Initial best: {:.10f} (len={})".format(best_score, len(best_seq)))
    
    # Phase 2: Main optimization - allocate time to top starts
    starts_sorted = sorted(range(len(starts)), key=lambda i: start_scores[i])
    
    n_starts = len(starts_sorted)
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.7, 0.3]
    elif n_starts == 3:
        fracs = [0.55, 0.28, 0.17]
    else:
        fracs = [0.45, 0.25, 0.18, 0.12]
    
    main_end = deadline - 250.0
    main_budget = max(60.0, main_end - t_start)
    cum = 0.0
    
    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))
        
        if time.monotonic() > sub_dead - 10.0:
            continue
        
        sq = starts[si].copy()
        
        # For secondary starts, blend with current best
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            t_blend = 0.4 if rank == 1 else 0.6
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * base)
        
        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.10f}".format(rank + 1, best_score))
    
    best_seq = normalize_a(best_seq)
    
    # Phase 3: Subgradient + coordinate descent polish
    if time.monotonic() < deadline - 180.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 180.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.10f}".format(best_score))
        
        if time.monotonic() < deadline - 150.0:
            sc, seq = coordinate_descent_polish(best_seq, deadline=deadline - 150.0, max_rounds=5)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Coord descent: {:.10f}".format(best_score))
    
    # Phase 4: Perturbation restarts
    perturb_count = 0
    while time.monotonic() < deadline - 120.0 and perturb_count < 8:
        left = float(deadline - 120.0 - time.monotonic())
        if left < 45.0:
            break
        n_remaining = 8 - perturb_count
        chunk = time.monotonic() + left / float(n_remaining)
        
        p = perturb_best(best_seq, rng)
        
        # Quick optimize: a few LSE + subgrad
        a_cur = p.copy()
        sc_cur = exact_score_a(a_cur)
        
        for beta in (500, 5000, 50000):
            rem = float(chunk - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.7, len(a_cur))
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a_cur, float(beta), chunk, mf)
            if np.isfinite(sc) and sc < sc_cur:
                sc_cur = sc
                a_cur = a2.copy()
            a_cur = normalize_a(a2)
        
        if time.monotonic() < chunk - 3.0:
            sc_sg, a_sg = subgrad_polish(a_cur, max_iters=5, deadline=chunk)
            if np.isfinite(sc_sg) and sc_sg < sc_cur:
                sc_cur = sc_sg
                a_cur = a_sg.copy()
        
        if np.isfinite(sc_cur) and sc_cur < best_score:
            best_score = sc_cur
            best_seq = a_cur.copy()
            print("Perturb restart {}: {:.10f}".format(perturb_count, best_score))
        perturb_count += 1
    
    # Phase 5: Final high-beta LSE polish
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 10.0:
            mf = estimate_maxfun(rem * 0.8, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 200000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.10f}".format(best_score))
    
    # Phase 6: Final subgradient cleanup
    if time.monotonic() < deadline - 10.0:
        sc, seq = subgrad_polish(best_seq, max_iters=8, deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.10f}".format(best_score))
    
    # Final cleanup and output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score
    
    print("FINAL: internal={:.10f} verified={:.10f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
