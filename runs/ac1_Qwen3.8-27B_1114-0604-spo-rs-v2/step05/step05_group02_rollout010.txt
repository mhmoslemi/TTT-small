"""
Hybrid optimizer combining GS-Fienup spectral iteration, adaptive LSE-smoothed L-BFGS,
peak-chasing greedy reduction, and multi-length exploration.
Key innovations: (1) adaptive beta annealing for LSE to progressively sharpen the max,
(2) peak-specific line search that directly targets the largest convolution peaks,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize so sum = 1."""
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    return a / S


def gs_fienup(a, n_iters, alpha_init, alpha_final, target_mag=None):
    """GS-Fienup with annealed alpha."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    
    if target_mag is None:
        b_target = np.zeros(M, dtype=np.float64)
        b_target[:2*n-1] = 1.0 / (2*n - 1)
        T = np.fft.fft(b_target)
        target_mag = np.maximum(np.abs(T), 1e-20)
        target_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * target_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * target_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.minimum(scaled, 0))  # numerical stability
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L with respect to a: L = c_max + log(sum(exp(beta*(c-c_max))))/beta
    # dL/da_i = (1/beta) * sum_k p_k * dc_k/da_i
    # dc_k/da_i = 2 * a[k-i] for i in [k-n+1, k]
    # So dL/da = 2 * conv(a, p) (shifted appropriately)
    
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    # conv(a, p) has length 2n-1, we want the part that gives dc/da
    # Actually, d(conv(a,a))_k / da_i = a[k-i] + a[k-i] (by chain rule, both a's depend on a_i)
    # Wait no: conv(a,a)_k = sum_j a_j * a_{k-j}
    # d/d a_i: sum_j [delta_{j,i} * a_{k-j} + a_j * delta_{k-j,i}] = a_{k-i} + a_{k-i} = 2*a_{k-i} (when valid)
    # So grad_i = 2 * sum_k p_k * a_{k-i} = 2 * (a * p_flipped)_i ... 
    
    # Let me think more carefully.
    # (conv(a,a))_k = sum_{j} a_j * a_{k-j} (linear conv, valid range)
    # d(conv(a,a))_k / da_i = a_{k-i} + a_{k-i} = 2 * a_{k-i} if 0 <= k-i < n
    # dL/da_i = sum_k p_k * 2 * a_{k-i} * (1/beta) * beta = sum_k p_k * 2 * a_{k-i}
    # Wait, L = c_max + (1/beta)*log(sum(exp(beta*(c-c_max))))
    # dL/dc_k = (1/beta) * exp(beta*(c_k - c_max)) / sum(exp(beta*(c_j - c_max))) = p_k
    # dL/da_i = sum_k dL/dc_k * dc_k/da_i = sum_k p_k * 2 * a_{k-i} (for valid k-i)
    
    # sum_k p_k * a_{k-i} = (p * rev_a)_i where * is appropriate convolution
    # Actually: sum_k p_k * a_{k-i} for k from 0 to 2n-2, a_{k-i} valid when 0 <= k-i < n
    # Let m = k-i, then k = m+i, sum_m p_{m+i} * a_m
    # This is (cross_correlation of p and a) at lag i
    # = sum_m a_m * p_{m+i}
    
    # In FFT terms: if we compute ifft(FFT(p) * conj(FFT(a))), we get cross-correlation
    # Or: ifft(FFT(rev_p) * FFT(a)) gives convolution of rev_p and a
    # sum_m a_m * p_{m+i} = (rev_p * a)_{n-1-i} ... this is getting complicated
    
    # Simpler: just compute it as a convolution
    # grad_i = 2 * sum_k p_k * a_{k-i} for valid indices
    # This is 2 * cross_correlation(p, a) at lag i
    # cross_correlation(p, a)[i] = sum_k p[k] * a[k+i] ... no
    # Let's just use the formula: grad = 2 * ifft(FFT(a) * conj(FFT(p)))[:n]
    # Actually: sum_k p_k * a_{k-i} = (a convolved with flipped p) at position ... 
    
    # Let me just do it directly with FFT:
    # We want g_i = sum_{k=0}^{2n-2} p_k * a_{k-i} (where a_j = 0 if j < 0 or j >= n)
    # g_i = sum_{j=0}^{n-1} a_j * p_{j+i}  (substituting j = k-i)
    # This is the cross-correlation of a and p, evaluated at lag i
    # = ifft(FFT(a) * conj(FFT(p)))_i ... let me verify
    # corr[i] = sum_j a[j] * p[j+i]
    # In FFT: corr = ifft(FFT(a) * conj(FFT(p)))
    # Yes, that's the circular cross-correlation. Since we have linear conv of length 2n-1 and we're
    # only looking at lags 0 to n-1, and p is zero-padded, this should work.
    
    n_fft = M2_fft
    FA = np.fft.rfft(a, n_fft)
    FP = np.fft.rfft(p, n_fft)
    corr = np.fft.irfft(FA * np.conj(FP), n_fft)[:n]
    
    # So dL/da_i = 2 * corr_i (without the 1/beta factor since p already includes it)
    # Wait, let me redo. L = c_max + (1/beta)*log(Z) where Z = sum_k exp(beta*(c_k - c_max))
    # p_k = exp(beta*(c_k - c_max)) / Z
    # dL/dc_k = (1/beta) * (1/Z) * exp(beta*(c_k - c_max)) * beta = p_k
    # dL/da_i = sum_k p_k * dc_k/da_i = sum_k p_k * 2 * a_{k-i} = 2 * corr_i
    
    dL_da = 2.0 * corr
    
    # Now score = 2n * L / S^2
    # d(score)/da_i = 2n * (dL_da_i / S^2 - 2*L*dS_da_i / S^3)
    # dS/da_i = 1
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    
    return score, grad


def lse_objective(x, beta):
    return lse_grad(x, beta)


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_chasing(a, n_iters=2000, n_top=5, tol=1e-8):
    """
    Greedy peak-chasing: identify top peaks, compute direction to reduce them,
    take a line-searched step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc - tol:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Line search with backtracking
        best_step = 0.0
        best_trial_sc = sc
        for factor in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            trial = a - factor * grad / gn
            trial = np.clip(trial, 0.0, 1000.0)
            S_t = float(trial.sum())
            if S_t < 1e-8:
                continue
            trial = trial / S_t
            trial_sc = compute_score(trial)
            if trial_sc < best_trial_sc:
                best_trial_sc = trial_sc
                best_step = factor
        
        if best_step == 0.0:
            # Try smaller steps
            for factor in [0.001, 0.003, 0.005]:
                trial = a - factor * grad / gn
                trial = np.clip(trial, 0.0, 1000.0)
                S_t = float(trial.sum())
                if S_t < 1e-8:
                    continue
                trial = trial / S_t
                trial_sc = compute_score(trial)
                if trial_sc < best_trial_sc:
                    best_trial_sc = trial_sc
                    best_step = factor
        
        if best_step == 0.0:
            # Stuck, try random perturbation
            break
        
        a = a - best_step * grad / gn
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def random_restart_refine(a, rng, n_restarts=5, n_gs_iters=200):
    """Add random perturbation, refine with GS, keep if better."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for _ in range(n_restarts):
        noise = rng.standard_normal(n) * 0.1
        perturbed = a + noise
        perturbed = np.maximum(perturbed, 0.0)
        S = float(perturbed.sum())
        if S < 1e-8:
            continue
        perturbed = perturbed / S
        
        refined = gs_fienup(perturbed, n_iters=n_gs_iters, alpha_init=0.5, alpha_final=0.1)
        sc = compute_score(refined)
        if sc < best_sc:
            best_sc = sc
            best_a = refined.copy()
    
    return best_a


def build_initializations(n, rng, parent=None):

"""
Hybrid optimizer: GS-Fienup + Adaptive LSE-LBFGS + Peak-Chasing with line search

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_vec(a):
    """Normalize so sum = 1, ensuring non-negative."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    return a / S


def gs_fienup(a, n_iters, alpha_init, alpha_final):
    """GS-Fienup with annealed alpha and correct flat-conv target."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    target_mag = np.maximum(np.abs(T), 1e-20)
    target_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * target_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * target_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.minimum(scaled, 0.0)
    exp_scaled = np.exp(scaled)
    Z = float(exp_scaled.sum())
    if Z < 1e-30:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    log_sum = float(np.log(Z))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / Z

    # dL/da_i = 2 * sum_k p_k * a_{k-i} = 2 * cross_correlation(a, p)[i]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    FA = np.fft.rfft(a, M2_fft)
    FP = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(FA * np.conj(FP), M2_fft)[:n]

    dL_da = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad

def peak_chasing_ls(a, n_iters, n_top):
    """
    Peak-chasing with backtracking line search.
    At each step: identify top peaks, compute subgradient,
    find best step via line search, take step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 200:
                break

        top_idx = np.argsort(c)[::-1][:n_top]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Line search with multiple trial steps
        best_step = 0.0
        best_trial_sc = sc
        trial_factors = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]
        for factor in trial_factors:
            trial = a - factor * grad / gn
            trial = np.clip(trial, 0.0, 1000.0)
            S_t = float(trial.sum())
            if S_t < 1e-8:
                continue
            trial = trial / S_t
            trial_sc = compute_score(trial)
            if trial_sc < best_trial_sc:
                best_trial_sc = trial_sc
                best_step = factor

        if best_step == 0.0:
            # Try even smaller steps
            for factor in [0.001, 0.002, 0.003]:
                trial = a - factor * grad / gn
                trial = np.clip(trial, 0.0, 1000.0)
                S_t = float(trial.sum())
                if S_t < 1e-8:
                    continue
                trial = trial / S_t
                trial_sc = compute_score(trial)
                if trial_sc < best_trial_sc:
                    best_trial_sc = trial_sc
                    best_step = factor

        if best_step == 0.0:
            break

        a = a - best_step * grad / gn
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def variance_objective_grad(a):
    """
    Sum of squared deviations of convolution from its mean.
    This is a smooth objective that promotes flatness.
    score_var = sum((c_k - mean_c)^2)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    # Work with normalized a (sum=1)
    a_norm = a / S
    c = fast_conv(a_norm)
    mean_c = c.mean()
    deviations = c - mean_c
    var_score = float(np.sum(deviations ** 2))

    # Gradient: d/dc_k of sum((c_j - mean)^2)
    # = 2*(c_k - mean) - 2*mean = 2*c_k - (2/(2n-1))*sum(c_j) ... 
    # Actually d/d c_k [sum_j (c_j - mean)^2] = 2*(c_k - mean) - 2*mean * (1/(2n-1)) * (2n-1)
    # = 2*(c_k - mean) - 2*mean = 2*c_k - 4*mean ... no
    # Let mean = (1/(2n-1)) * sum_j c_j
    # d mean / d c_k = 1/(2n-1)
    # d/d c_k [sum_j (c_j - mean)^2] = 2*(c_k - mean)*1 - 2*sum_j(c_j - mean)*d(mean)/dc_k
    # = 2*(c_k - mean) - 2*0*(1/(2n-1))  [since sum(c_j - mean) = 0]
    # = 2*(c_k - mean)
    # OK so it's just 2*(c_k - mean)

    p = 2.0 * (c - mean_c)

    # d/var_score/da_i = sum_k p_k * dc_k/da_i = sum_k p_k * 2*a_{k-i}
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    FA = np.fft.rfft(a_norm, M2_fft)
    FP = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(FA * np.conj(FP), M2_fft)[:n]

    grad = 2.0 * corr * (2.0 / S)  # chain rule through normalization... 
    # Actually since we work with a_norm = a/S, and we're optimizing over a directly:
    # Let's just use the gradient with respect to a_norm and ignore the S scaling
    # since we normalize after each step anyway.
    grad = 2.0 * corr

    return var_score, grad

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(m_sequence_init(n, rng))
    inits.append(zaden_chu_init(n, rng))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * 2 * x + rng.uniform(0, 2 * np.pi)), 0.01))
    inits.append(np.maximum(0.5 + 0.2 * np.cos(2.0 * np.pi * 3 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    # Chirp-like
    freq = rng.uniform(1, 5)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * (freq * x + 0.5 * x ** 2)), 0.01))
    return inits


def random_perturb_refine(a, rng, n_perturbs, n_gs_iters):
    """Add random perturbation, refine with GS, keep best."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_perturbs):
        scale = rng.uniform(0.02, 0.1)
        noise = rng.standard_normal(n) * scale
        perturbed = a + noise
        perturbed = np.maximum(perturbed, 0.0)
        S = float(perturbed.sum())
        if S < 1e-8:
            continue
        perturbed = perturbed / S

        refined = gs_fienup(perturbed, n_iters=n_gs_iters, alpha_init=0.4, alpha_final=0.05)
        sc = compute_score(refined)
        if sc < best_sc:
            best_sc = sc
            best_a = refined.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for best non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase hybrid: GS-Fienup exploration -> variance minimization ->
    LSE-LBFGS refinement -> peak-chasing with line search -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    # Phase 1: Multi-length GS exploration (35% of time)
    phase1_end = t_start + 0.35 * (total - 30.0)
    print(f"=== Phase 1: GS exploration [deadline t={phase1_end - t_start:.0f}s] ===")

    lengths = [4096, 6962, 8192, 10240]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
    lengths = sorted(set(lengths))

    print(f"Lengths: {lengths}, Parent available: {parent is not None}")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a1 = gs_fienup(a0, n_iters=400, alpha_init=0.7, alpha_final=0.05)
            sc = compute_score(a1)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a1.copy(), 0.0, 1000.0)
                print(f"    GS n={n} i={i}: {sc:.8f}")

            # Magnitude-only GS
            if time.monotonic() < phase1_end - 5:
                a2 = gs_magnitude_only(a1, n_iters=300, alpha=0.3)
                sc2 = compute_score(a2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = np.clip(a2.copy(), 0.0, 1000.0)
                    print(f"    GS-mag n={n} i={i}: {sc2:.8f}")

    print(f"After Phase 1: best={best_score:.8f}")

    # Phase 2: Variance minimization + GS refine (15% of time)
    phase2_end = t_start + 0.50 * (total - 30.0)
    print(f"\n=== Phase 2: Variance + GS refine [deadline t={phase2_end - t_start:.0f}s] ===")

    # Minimize variance of convolution (smooth objective)
    sc_var, a_var = run_lbfgs_variance(best_a, maxiter=3000, maxfun=30000)
    if sc_var < best_score:
        best_score = sc_var
        best_a = np.clip(a_var.copy(), 0.0, 1000.0)
        print(f"  Variance min: {sc_var:.8f}")

    # GS refine after variance
    if time.monotonic() < phase2_end - 10:
        a_gs = gs_fienup(best_a, n_iters=400, alpha_init=0.3, alpha_final=0.02)
        sc_gs = compute_score(a_gs)
        if sc_gs < best_score:
            best_score = sc_gs
            best_a = np.clip(a_gs.copy(), 0.0, 1000.0)
            print(f"  GS after variance: {sc_gs:.8f}")

    # More variance min
    if time.monotonic() < phase2_end - 5:
        sc_var2, a_var2 = run_lbfgs_variance(best_a, maxiter=2000, maxfun=20000)
        if sc_var2 < best_score:
            best_score = sc_var2
            best_a = np.clip(a_var2.copy(), 0.0, 1000.0)
            print(f"  Variance min 2: {sc_var2:.8f}")

    print(f"After Phase 2: best={best_score:.8f}")

    # Phase 3: LSE-LBFGS with adaptive beta (25% of time)
    phase3_end = t_start + 0.75 * (total - 30.0)
    print(f"\n=== Phase 3: LSE-LBFGS [deadline t={phase3_end - t_start:.0f}s] ===")

    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=2000, maxfun=20000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
        elapsed_beta = time.monotonic() - t0
        print(f"    beta={beta:>6d}: score={sc:.8f}, time={elapsed_beta:.1f}s")

        # Alternate with GS between LSE runs
        if time.monotonic() < phase3_end - 30:
            a_gs_mid = gs_fienup(best_a, n_iters=200, alpha_init=0.2, alpha_final=0.03)
            sc_mid = compute_score(a_gs_mid)
            if sc_mid < best_score:
                best_score = sc_mid
                best_a = np.clip(a_gs_mid.copy(), 0.0, 1000.0)
                print(f"      GS mid: {sc_mid:.8f}")

    print(f"After Phase 3: best={best_score:.8f}")

    # Phase 4: Peak-chasing with line search (15% of time)
    phase4_end = t_start + 0.90 * (total - 30.0)
    print(f"\n=== Phase 4: Peak-chasing [deadline t={phase4_end - t_start:.0f}s] ===")

    for n_top in [3, 5, 8, 15, 30]:
        if time.monotonic() > phase4_end:
            break
        t0 = time.monotonic()
        a_pc = peak_chasing_ls(best_a, n_iters=3000, n_top=n_top)
        sc = compute_score(a_pc)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_pc.copy(), 0.0, 1000.0)
        print(f"    top={n_top:>2d}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # Random restarts
    if time.monotonic() < phase4_end - 15:
        t0 = time.monotonic()
        a_rr = random_perturb_refine(best_a, rng, n_perturbs=8, n_gs_iters=200)
        sc_rr = compute_score(a_rr)
        if sc_rr < best_score:
            best_score = sc_rr
            best_a = np.clip(a_rr.copy(), 0.0, 1000.0)
        print(f"    Random restart: {sc_rr:.8f}, time={time.monotonic()-t0:.1f}s")

    print(f"After Phase 4: best={best_score:.8f}")

    # Phase 5: Final polish (10% of time)
    print(f"\n=== Phase 5: Final polish [deadline t={deadline - t_start:.0f}s] ===")

    # Alternating LSE + peak-chasing
    for round_num in range(5):
        if time.monotonic() > deadline - 15:
            break

        # LSE with high beta
        sc, a_opt = run_lbfgs_lse(best_a, 50000, maxiter=2000, maxfun=20000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  Round {round_num} LSE: {sc:.8f}")

        if time.monotonic() > deadline - 10:
            break

        # Peak chasing
        a_pc = peak_chasing_ls(best_a, n_iters=2000, n_top=5)
        sc_pc = compute_score(a_pc)
        if sc_pc < best_score:
            best_score = sc_pc
            best_a = np.clip(a_pc.copy(), 0.0, 1000.0)
            print(f"  Round {round_num} Peak: {sc_pc:.8f}")

    # Final GS
    if time.monotonic() < deadline - 5:
        a_final = gs_fienup(best_a, n_iters=200, alpha_init=0.1, alpha_final=0.01)
        sc_final = compute_score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_a = np.clip(a_final.copy(), 0.0, 1000.0)
            print(f"  Final GS: {sc_final:.8f}")

    # Final peak-chasing
    if time.monotonic() < deadline - 3:
        a_fp = peak_chasing_ls(best_a, n_iters=1000, n_top=3)
        sc_fp = compute_score(a_fp)
        if sc_fp < best_score:
            best_score = sc_fp
            best_a = np.clip(a_fp.copy(), 0.0, 1000.0)
            print(f"  Final Peak: {sc_fp:.8f}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
