"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key improvements over parent:
1. FISTA with exact simplex projection and Polyak-adaptive step size
2. Multi-beta LSE continuation with restarts on stagnation
3. Multiple sequence lengths (interpolated from parent)
4. L-BFGS-B polishing with tight tolerances
5. Peak-targeted coordinate descent with binary search
6. Structured multi-restart in final phase
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1} in O(n log n)."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / ind
    rho = int(np.max(ind[cond])) if cond.any() else 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient on the probability simplex.
    Objective: 2n * LSE_beta(conv(a,a)) where LSE smooths the max.
    Assumes sum(a) = 1.
    Returns (score, grad) where grad = df/da for f(a) = 2n*LSE.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L

    p = exp_s / exp_s.sum()

    # grad_m = 4n * sum_j p[m+j] * a_j  (cross-correlation)
    # = 4n * conv(p, a_rev)[m + n - 1]
    M = 2 * n - 1
    conv_len = M + n - 1  # 3n - 2
    conv_fft = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, conv_fft)
    AR = np.fft.rfft(a[::-1], conv_fft)
    conv_res = np.fft.irfft(P * AR, conv_fft)[:conv_len]
    grad = 4.0 * n * conv_res[n - 1:2 * n - 1]

    return score, grad


def fista_lse(a0, beta, n_iters, lr_init=None, use_polyak=True, verbose=False):
    """
    FISTA on LSE-smoothed objective over the probability simplex.
    Uses exact simplex projection and optional Polyak step size.
    """
    n = len(a0)
    x = a0.copy()
    y = a0.copy()
    t_k = 1.0
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    # Initial Lipschitz estimate
    _, g0 = lse_score_grad(y, beta)
    L = max(float(np.linalg.norm(g0)) + 1.0, 10.0)

    best_x = x.copy()
    best_sc = compute_score(x)
    stagnant = 0

    for it in range(n_iters):
        f_y, g_y = lse_score_grad(y, beta)

        # Adaptive step: try Polyak if available
        if use_polyak and f_y > sc_lb + 1e-10:
            gn2 = float(g_y @ g_y)
            if gn2 > 1e-20:
                step_polyak = (f_y - sc_lb) / (gn2 + 1e-10)
                step = min(step_polyak, 2.0 / L)
                step = max(step, 1e-10)
            else:
                step = 1.0 / L
        else:
            step = 1.0 / L

        # Gradient step + projection
        x_new = y - step * g_y
        x_new = project_simplex(x_new)

        # Backtracking
        f_new, _ = lse_score_grad(x_new, beta)
        quad = 0.5 * (1.0 / step) * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quad:
            L *= 2.0
            if L > 1e12:
                break
            continue
        else:
            L = max(0.7 * L, 1.0)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_simplex(y)

        x = x_new
        t_k = t_new

        # Track best
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(x)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_x = x.copy()
                stagnant = 0
            else:
                stagnant += 1

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()

    return best_x, best_sc


def fienup_iter(a, n_iters=200, mode='flatten'):
    """Fienup frequency-domain iteration for global mixing."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2 + 1e-30

        if mode == 'flatten':
            alpha = 0.3 + 0.5 * progress
            P_mean = float(P.mean())
            P_target = np.full_like(P, P_mean)
            P_blend = (1 - alpha) * P + alpha * P_target
            P_blend = np.maximum(P_blend, 1e-30)
            A_new = A * np.sqrt(P_blend / P) ** alpha
        else:  # quantile
            q = 0.7 + 0.25 * progress
            thresh = np.quantile(P, q)
            scale = np.ones_like(P)
            above = P > thresh
            if above.any():
                scale[above] = np.sqrt(thresh / P[above])
            A_new = A * scale

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B on LSE-smoothed objective with simplex constraint (via log-barrier)."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a0 = a0 / a0.sum()

    def obj_and_grad(x):
        x_pos = np.maximum(x, 1e-15)
        x_norm = x_pos / x_pos.sum()
        sc, g = lse_score_grad(x_norm, beta)
        # Chain rule for normalization
        S = x_pos.sum()
        g_full = (g - sc / S * 2.0 * n / (2.0 * n) * 2.0) / S
        # Actually, let's just use the direct gradient on the simplex
        # and handle bounds separately
        return sc, g

    # Use bounded approach: optimize in log-space
    # x_i = exp(z_i) / sum(exp(z))
    # This is complex; let's just use bounds on the simplex directly.
    bounds = [(1e-15, 1.0)] * n

    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def coord_descent(a, n_sweeps=10, n_top=100):
    """Coordinate descent targeting top-contributing elements at the peak lag."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            # Binary search for best multiplier
            lo, hi = 0.3, 2.0
            best_val = a_old_j
            best_local = compute_score(a)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[j] = a_old_j * mid
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    break
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old_j * mid
                    lo = mid
                else:
                    hi = mid
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolated_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def structured_perturb(a, rng, scale=0.03):
    """Add low-frequency structured perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(6):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / (m + 1)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):
    """Add multiplicative Gaussian noise."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new
