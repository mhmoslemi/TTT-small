"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key innovation: log-space multiplicative subgradient descent for direct peak reduction,
combined with L-BFGS polishing and surgical coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def lse_objective_func(x, beta):
    return lse_obj_grad(x, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out

def log_space_descent(a0, n_iters=100000, top_k=3, step_init=0.01, decay=0.99995, patience=5000):
    """
    Multiplicative subgradient descent in log-space.
    Directly targets the top peaks of the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    iterations_since_improve = 0
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            iterations_since_improve = 0
        else:
            iterations_since_improve += 1
        
        if iterations_since_improve > patience:
            break
        
        # Find top K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        
        # Weight peaks by proximity to max
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            weights[i] = (c[k] / M) ** 10  # sharp weighting
        
        # Compute log-space gradient (drop the constant -2/S term)
        # d_j = sum_k w_k * a_j * a_{k-j} / M_k
        d = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            w = weights[i]
            M_k = c[k]
            if M_k < 1e-30:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # d[j] += w * a[j] * a[k-j] / M_k
                # a[k-j] for j in [j_start, j_end] is a[k-j_end : k-j_start+1][::-1]
                reversed_seg = a[k - j_end : k - j_start + 1][::-1]
                d[j_start:j_end+1] += w * a[j_start:j_end+1] * reversed_seg / M_k
        
        # Multiplicative update: a_j *= exp(-step * d_j)
        # Clip to avoid overflow
        log_update = -step * d
        log_update = np.clip(log_update, -1.0, 1.0)
        a = a * np.exp(log_update)
        
        # Renormalize
        S = float(a.sum())
        if S < 1e-15:
            a = best_a.copy()
            break
        a = a / S
        
        step *= decay
    
    return best_a, best_sc

def lse_fista(a0, beta, n_iters=2000):
    """FISTA on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S

    x = a0.copy()
    z = a0.copy()
    t_k = 1.0
    
    _, g0 = lse_obj_grad(z, beta)
    gn = float(np.linalg.norm(g0))
    L = max(gn + 1.0, 10.0)

    best_x = a0.copy()
    best_sc = compute_score(a0)

    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

        f_y, g_y = lse_obj_grad(z, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new

        f_new, _ = lse_obj_grad(x_new, beta)
        quadratic = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quadratic:
            L *= 1.5
            if L > 1e10:
                break
            continue
        else:
            L = max(0.8 * L, 1.0)

        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = z.sum()
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()
        best_sc = sc

    return best_x, best_sc

def coord_descent(a, n_sweeps=5, n_top=50):
    """Coordinate descent on top-contributing elements at the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            a[j] = a_old_j * 0.7
            S_tmp = a.sum()
            if S_tmp > 1e-12:
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = a_old_j
            else:
                a[j] = a_old_j
        
        # Also try increasing
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            a[j] = a_old_j * 1.3
            S_tmp = a.sum()
            if S_tmp > 1e-12:
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = a_old_j
            else:
                a[j] = a_old_j

    return best_a, best_sc

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key innovations: log-space multiplicative subgradient descent for direct peak reduction,
adaptive step sizing, multi-scale L-BFGS polishing, and surgical coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    """Top-level wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_func, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def log_space_descent(a0, n_iters=50000, top_k=3, step=0.008, patience=5000, reheat_every=10000):
    """
    Multiplicative subgradient descent in log-space.
    Directly targets top peaks of autocorrelation using peak-contribution fractions.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    iterations_since_improve = 0
    cur_step = step

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            iterations_since_improve = 0
        else:
            iterations_since_improve += 1

        if iterations_since_improve > patience:
            # Reheat: increase step and reset patience
            cur_step = step * (1.0 + 0.5 * (iterations_since_improve / patience))
            cur_step = min(cur_step, 0.05)
            iterations_since_improve = 0

        # Find top K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Weight peaks by sharp proximity to max
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            ratio = c[k] / M
            weights[i] = ratio ** 20  # very sharp

        # Compute log-space gradient direction (peak contribution fractions)
        d = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            w = weights[i]
            M_k = c[k]
            if M_k < 1e-30:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                rev_seg = a[k - j_end:k - j_start + 1][::-1]
                d[j_start:j_end + 1] += w * a[j_start:j_end + 1] * rev_seg / M_k

        # Multiplicative update
        log_update = -cur_step * d
        log_update = np.clip(log_update, -0.5, 0.5)
        a = a * np.exp(log_update)

        # Renormalize
        S = float(a.sum())
        if S < 1e-15:
            a = best_a.copy()
            break
        a = a / S

    return best_a, best_sc


def log_space_descent_v2(a0, n_iters=50000, top_k=5, step=0.005, patience=8000):

def fienup_peaks(a0, n_iters=200, target_lags=None):
    """
    Fienup-style iteration targeting specific peak lags.
    In frequency domain: reduce power at frequencies that contribute to peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find peak lags
        top_idx = np.argsort(c)[::-1][:3]

        # In frequency domain: reduce spectral components that create peaks at these lags
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target: flatten peaks in the autocorrelation
        # The autocorrelation peak at lag k is related to the "coherence" at that lag
        # Reduce power where phase alignment creates peaks
        for k in top_idx:
            k = int(k)
            # The peak at lag k corresponds to frequency components where
            # phi(f) has alignment exp(-i*2*pi*f*k/n)
            # Approximate: reduce power at frequencies that maximally contribute
            freqs = np.fft.rfftfreq(M_fft) * M_fft
            penalty = np.cos(2.0 * np.pi * freqs * k / n)
            # Reduce power where penalty is positive (constructive at this lag)
            reduce_mask = penalty > 0.5
            if reduce_mask.any():
                P[reduce_mask] *= (1.0 - 0.1 * (it / n_iters + 0.1))

        P = np.maximum(P, 1e-20)
        # Blend with flat spectrum
        alpha = 0.3 + 0.5 * (it / n_iters)
        P_new = (1 - alpha) * P + alpha * P_mean
        A_new = A * np.sqrt(P_new / P)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 1e-15)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

    return best_a, best_sc


def coord_descent(a, n_sweeps=5, n_top=50):
    """
    Coordinate descent: try multiplicative adjustments to top-contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Process top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]
        for k_star in top_peaks:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k_star - js]
            total = contribs.sum()
            if total < 1e-15:
                continue

            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old_j = a[j]
                if a_old_j < 1e-14:
                    continue
                # Try reducing
                for factor in [0.6, 0.8]:
                    a[j] = a_old_j * factor
                    S_tmp = a.sum()
                    if S_tmp > 1e-12:
                        sc = compute_score(a)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                            break
                    a[j] = a_old_j
                # Try increasing (if element doesn't contribute to this peak)
                a[j] = a_old_j * 1.4
                S_tmp = a.sum()
                if S_tmp > 1e-12:
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    else:
                        a[j] = a_old_j
                else:
                    a[j] = a_old_j

    return best_a, best_sc


def lse_fista(a0, beta, n_iters=2000):
    """FISTA on LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()

    if S < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S

    x = a0.copy()
    z = a0.copy()
    t_k = 1.0

    _, g0 = lse_obj_grad(z, beta)
    gn = float(np.linalg.norm(g0))
    L = max(gn + 1.0, 10.0)

    best_x = a0.copy()
    best_sc = compute_score(a0)

    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

        f_y, g_y = lse_obj_grad(z, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new

        f_new, _ = lse_obj_grad(x_new, beta)
        quadratic = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quadratic:
            L *= 1.5
            if L > 1e10:
                break
            continue
        else:
            L = max(0.8 * L, 1.0)

        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = z.sum()
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x
        best_sc = sc

    return best_x, best_sc

    # Phase 2: L-BFGS multi-beta (25% of time)
    # ================================================================
    phase2_end = t_start + 0.55 * (total_budget - 90.0)
    print(f"\n=== Phase 2: L-BFGS (until t={phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas = [500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic() - 30
        if remaining < 20:
            break
        maxiter = min(8000, int(remaining * 15))
        maxfun = min(150000, int(remaining * 300))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # ================================================================
    # Phase 3: Log-space descent v2 (20% of time)
    # ================================================================
    phase3_end = t_start + 0.75 * (total_budget - 90.0)
    print(f"\n=== Phase 3: Log-space descent v2 (until t={phase3_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    while time.monotonic() < phase3_end - 5:
        t0 = time.monotonic()
        a_ls, sc_ls = log_space_descent_v2(a_work, n_iters=80000, top_k=5,
                                           step=0.005, patience=6000)
        dt = time.monotonic() - t0
        print(f"  LOG-SPACE v2: score={sc_ls:.8f} ({dt:.1f}s)")
        update(sc_ls, a_ls, "LOG-v2")
        if sc_ls < best_score:
            a_work = a_ls.copy()
        if time.monotonic() > phase3_end - 5:
            break

    # ================================================================
    # Phase 4: Fienup + L-BFGS polish (10% of time)
    # ================================================================
    phase4_end = t_start + 0.85 * (total_budget - 90.0)
    print(f"\n=== Phase 4: Fienup + polish (until t={phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 30:
        a_f, sc_f = fienup_peaks(best_a, n_iters=100)
        update(sc_f, a_f, "FIENUP")
        if sc_f < best_score:
            sc2, a2 = run_lbfgs(a_f, 5000, maxiter=5000, maxfun=100000)
            update(sc2, a2, "FIENUP+LBFGS")

    if time.monotonic() < phase4_end - 30:
        a_fi, sc_fi = lse_fista(best_a, beta=2000, n_iters=1500)
        update(sc_fi, a_fi, "FISTA")

    # ================================================================
    # Phase 5: Coordinate descent + final polish (15% of time)
    # ================================================================
    print(f"\n=== Phase 5: Coord descent + final polish ===")

    if time.monotonic() < deadline - 60:
        a_cd, sc_cd = coord_descent(best_a, n_sweeps=8, n_top=60)
        update(sc_cd, a_cd, "COORD")

    if time.monotonic() < deadline - 45:
        sc, a_final = run_lbfgs(best_a, 10000, maxiter=5000, maxfun=100000)
        update(sc, a_final, "FINAL-LBFGS-10k")

    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=60000)
        update(sc, a_final, "FINAL-LBFGS-50k")

    if time.monotonic() < deadline - 15:
        a_ls, sc_ls = log_space_descent(best_a, n_iters=20000, top_k=3,
                                        step=0.003, patience=5000)
        update(sc_ls, a_ls, "FINAL-LOG")

    if time.monotonic() < deadline - 5:
        a_cd, sc_cd = coord_descent(best_a, n_sweeps=3, n_top=100)
        update(sc_cd, a_cd, "FINAL-COORD")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
