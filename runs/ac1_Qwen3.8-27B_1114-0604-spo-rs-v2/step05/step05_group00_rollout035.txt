For non-symmetric sequences, `r[n-1] = sum a[i]*a[n-1-i]` can be smaller. In fact, if `a` is concentrated at one end (say `a[0] ≈ 1`), then `r[n-1] ≈ a[0]*a[n-1] ≈ 0`. But then `r[0] ≈ 1` is the peak, and the score is `2n`.

So there's a trade-off. The optimal sequence balances the peaks across all lags.

I think the key insight for a practical algorithm is:
1. The peak is typically at the center lag (k ≈ n-1) for "spread out" sequences
2. Or at the edge lags (k ≈ 0 or k ≈ 2n-2) for "concentrated" sequences
3. The optimum is somewhere in between

Let me now just write the code. I'll use a practical optimization approach with the power-p objective as the main driver.

Here's my final code plan:

```python
"""
Optimizer using power-p autocorrelation flattening.
Uses sum(r[k]^p) as a smooth proxy for max(r[k]), with gradient
computed via FFT. Combines global exploration (various p, various
starting points) with local refinement (high-p gradient descent,
coordinate descent, L-BFGS with LSE).
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def power_p_obj_grad(a, p):
    """Objective: sum(r[k]^p). Gradient via FFT."""
    n = len(a)
    r = fast_conv(a)
    # Clip r to avoid issues with negative values (shouldn't happen for non-negative a)
    r = np.maximum(r, 0.0)
    R = r ** (p - 1)
    # grad[j] = 2p * sum_{m=0}^{n-1} a[m] * R[j+m]
    # = 2p * conv(a_flip, R)[n-1 : 2n-1]
    a_flip = a[::-1]
    total_len = 2 * n - 1 + n - 1  # = 3n - 2... actually len(a_flip) + len(R) - 1 = n + (2n-1) - 1 = 3n-2
    M_fft = 1 << (3 * n - 2).bit_length()
    AF = np.fft.rfft(a_flip, M_fft)
    RF = np.fft.rfft(R, M_fft)
    conv_res = np.fft.irfft(AF * RF, M_fft)[:3*n-2]
    grad = 2.0 * p * conv_res[n-1:2*n-1]
    obj = float(np.sum(r ** p))
    return obj, grad

def lse_obj_grad(a, beta):
    """LSE-smoothed score with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p_vec = exp_scaled / exp_scaled.sum()
    # Gradient of L with respect to a
    # dL/da[j] = sum_k p_k * dr[k]/da[j]
    # = sum_k p_k * 2*a[k-j] (for valid k-j)
    # = 2 * conv(a_flip, p_vec)[n-1:2n-1]
    a_flip = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    # Wait, I need conv(a_flip, p_vec) where p_vec has length 2n-1
    total_len = n + (2*n-1) - 1
    M_fft = 1 << total_len.bit_length()
    AF = np.fft.rfft(a_flip, M_fft)
    PF = np.fft.rfft(p_vec, M_fft)
    conv_res = np.fft.irfft(AF * PF, M_fft)[:total_len]
    dL_da = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lbfgs_wrapper(x, beta):
    return lse_obj_grad(x, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0/n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lbfgs_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out

def power_p_descent(a, p, n_iters=5000, step_init=0.01):
    """Gradient descent on sum(r^p) with adaptive step and normalization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        obj, grad = power_p_obj_grad(a, p)
        
        # Also add a small regularization to keep a from becoming too sparse
        # proj gradient onto the constraint sum(a) = 1
        mean_grad = grad.mean()
        grad_centered = grad - mean_grad
        
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break
        
        a -= step * grad_centered
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive step
        if obj > 0:
            step = min(step * 1.01, 0.5 * obj / (gn * gn + 1e-30))
        step = max(step, 1e-10)
    
    return best_a

def peak_subgrad(a, n_iters=3000, top_k=3, step_init=0.01):
    """Subgradient descent targeting top-k peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        r = fast_conv(a)
        S = a.sum()
        M_val = r.max()
        
        # Top-k peaks
        top_idx = np.argsort(r)[::-1][:top_k]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # dr[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                sg[indices] += 2.0 * a[k - indices]
        
        # Center the gradient (for the sum=1 constraint)
        mean_sg = sg.mean()
        sg = sg - mean_sg
        
        gn = np.linalg.norm(sg)
        if gn < 1e-15:
            break
        
        a -= step * sg
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive step
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30) * 0.5)
        step = max(step, 1e-10)
        step = min(step, 0.05)
    
    return best_a
