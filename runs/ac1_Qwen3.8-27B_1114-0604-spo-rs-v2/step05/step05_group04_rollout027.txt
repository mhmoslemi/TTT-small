def fast_conv(a):

    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_adaptive(a, n_iters=300, gamma=0.8, top_frac_init=0.1, top_frac_end=0.5,
                    soft_alpha=30.0, sum_preserve=True, peak_target=False):
    """
    Fienup with adaptive top fraction and optional peak targeting.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        
        if peak_target and it > 0:
            # Target specific peaks
            c = np.fft.irfft(A * A, M_fft)[:M]
            top_idx = np.argsort(c)[::-1][:10]
            # Find which frequency bins are responsible for these peaks
            # Peak at lag k is most affected by frequency bin k (approximately)
            target_bins = set()
            for k in top_idx:
                k = int(k)
                # The peak at lag k is most affected by the spectral component
                # at frequency omega = pi*k/(n-1), which corresponds to bin ~ k
                target_bins.add(k % len(P))
                target_bins.add(len(P) - k % len(P))  # symmetric
            
            # Apply stronger compression to target bins
            scale = np.ones_like(P)
            thresh = np.quantile(P, 1.0 - top_frac)
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                scale[above] = ratio ** 0.5 * gamma
            # Extra compression for target bins
            for tb in target_bins:
                if 0 <= tb < len(P) and P[tb] > P.mean():
                    scale[tb] *= 0.7
        else:
            thresh = np.quantile(P, 1.0 - top_frac)
            scale = np.ones_like(P)
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                scale[above] = ratio ** 0.5 * gamma
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        
        # Soft non-negativity
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        if sum_preserve:
            a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""

Key improvements:
1. Peak-targeted spectral editing in Fienup iterations
2. Multi-band spectral equalization
3. Aggressive chained LSE with wider beta range and adaptive iteration budgets
4. Multi-length exploration with length-specific initialization
5. Oscillatory spectral modulation to escape local minima
6. Perturbation-based restarts with re-optimization at each scale

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RXA = np.fft.rfft(rev_x, M2_fft)
    conv_res = np.fft.irfft(PA * RXA, M2_fft)[n - 1:2 * n - 1]

    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma

        if peak_target and it % 10 == 5:
            c = np.fft.irfft(A * A, M_fft)[:M]
            top_idx = np.argsort(c)[::-1][:peak_k]
            n_bins = len(P)
            for k in top_idx:
                k = int(k)
                bin_idx = k % n_bins
                sym_idx = (n_bins - bin_idx) % n_bins
                if P[bin_idx] > P.mean():
                    scale[bin_idx] *= 0.75
                if sym_idx != bin_idx and P[sym_idx] > P.mean():
                    scale[sym_idx] *= 0.75

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_multi_band(a, n_iters=300, n_bands=4):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)

        scale = np.ones_like(P)
        band_size = max(1, n_bins // n_bands)

        for band in range(n_bands):
            start = band * band_size
            end = min((band + 1) * band_size, n_bins)
            if start >= end:
                continue
            band_P = P[start:end].copy()
            band_mean = float(band_P.mean())
            alpha = 0.1 + 0.4 * progress
            target = (1.0 - alpha) * band_P + alpha * band_mean
            target = np.maximum(target, 1e-20)
            ratio = np.sqrt(target / band_P)
            scale[start:end] = ratio

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_oscillation(a, n_iters=300, n_osc=3):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    freqs = [0.3, 0.7, 1.3, 2.1, 3.7][:n_osc]

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)

        top_frac = 0.1 + 0.3 * progress
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * 0.8

        x = np.arange(n_bins) / n_bins
        for f in freqs:
            mod = 1.0 + 0.1 * np.cos(2.0 * np.pi * f * x + it * 0.13)
            scale = scale * mod

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_deep_compression(a, n_iters=300, gamma_schedule=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    if gamma_schedule is None:
        gamma_schedule = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2]

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = 0.05 + 0.6 * progress

        gamma_idx = min(int(progress * len(gamma_schedule)), len(gamma_schedule) - 1)
        gamma_next_idx = min(gamma_idx + 1, len(gamma_schedule) - 1)
        gamma_frac = progress * len(gamma_schedule) - gamma_idx
        gamma = gamma_schedule[gamma_idx] * (1 - gamma_frac) + gamma_schedule[gamma_next_idx] * gamma_frac

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_inits(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(0.3 + 0.7 * (1.0 - x))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.ones(n) * 0.5 + 0.05 * rng.standard_normal(n))
    inits.append(np.exp(-1.0 * x) + 0.1)
    inits.append(np.maximum(0.2 + 0.3 * np.sin(3.0 * np.pi * x) + 0.2 * np.cos(5.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Fienup variants + chained LSE L-BFGS with peak-targeted spectral editing.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    # Determine lengths
    lengths = [6962, 8192, 10240, 12288, 14336]
    if parent is not None:
        lengths = [len(parent)] + [l for l in lengths if l != len(parent)]
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={ps:.8f}")
        best_score = ps
        best_a = parent.copy()

    # ==============================================================
    # Phase 0: Quick LBFGS from parent (5% time)
    # ==============================================================
    phase0_end = t_start + 0.05 * (total_budget - 30.0)
    print(f"\n=== Phase 0: Quick LBFGS (t<{phase0_end - t_start:.0f}s) ===")
    if parent is not None:
        for beta in [100, 500, 2000, 5000]:
            if time.monotonic() > phase0_end:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
                print(f"  [P0 b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ==============================================================
    # Phase 1: Fienup from diverse inits (30% time)
    # ==============================================================
    phase1_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (t<{phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            # Peak-targeted Fienup
            a_f = fienup_adaptive(a0, n_iters=200, gamma=0.7, top_frac_init=0.1,
                                  top_frac_end=0.5, soft_alpha=30.0, peak_target=True, peak_k=15)
            sc = compute_score(a_f)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_f.copy(), 0.0, 1000.0)
                print(f"  [FIENUP n={n} i={i}] NEW BEST: {sc:.8f}")

            # Multi-band
            if time.monotonic() < phase1_end - 30:
                a_mb = fienup_multi_band(a0, n_iters=200, n_bands=4)
                sc2 = compute_score(a_mb)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = np.clip(a_mb.copy(), 0.0, 1000.0)
                    print(f"  [MBAND n={n} i={i}] NEW BEST: {sc2:.8f}")

            # Deep compression
            if time.monotonic() < phase1_end - 30:
                a_dc = fienup_deep_compression(a0, n_iters=200)
                sc3 = compute_score(a_dc)
                if sc3 < best_score:
                    best_score = sc3
                    best_a = np.clip(a_dc.copy(), 0.0, 1000.0)
                    print(f"  [DEEP n={n} i={i}] NEW BEST: {sc3:.8f}")

            # Oscillation (only for first few inits to save time)
            if i < 3 and time.monotonic() < phase1_end - 30:
                a_osc = fienup_oscillation(a0, n_iters=150, n_osc=5)
                sc4 = compute_score(a_osc)
                if sc4 < best_score:
                    best_score = sc4
                    best_a = np.clip(a_osc.copy(), 0.0, 1000.0)
                    print(f"  [OSC n={n} i={i}] NEW BEST: {sc4:.8f}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==============================================================
    # Phase 2: Fienup refinement from best (10% time)
    # ==============================================================
    phase2_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Fienup from best (t<{phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_norm = best_a.copy() / best_a.sum()

    # Chain multiple Fienup variants
    fienup_configs = [
        (0.5, 0.05, 0.7, 20, True),
        (0.7, 0.1, 0.5, 30, True),
        (0.6, 0.08, 0.6, 25, False),
        (0.8, 0.15, 0.4, 40, True),
        (0.4, 0.05, 0.8, 15, True),
    ]
    for cfg_idx, (g, tf_i, tf_e, sa, pt) in enumerate(fienup_configs):
        if time.monotonic() > phase2_end:
            break
        a_out = fienup_adaptive(a_norm, n_iters=250, gamma=g, top_frac_init=tf_i,
                                top_frac_end=tf_e, soft_alpha=sa, peak_target=pt, peak_k=20)
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_out.copy(), 0.0, 1000.0)
            a_norm = a_out.copy() / a_out.sum()
            print(f"  [P2-FIENUP-{cfg_idx}] NEW BEST: {sc:.8f}")

    # Multi-band from best
    if time.monotonic() < phase2_end:
        a_mb = fienup_multi_band(a_norm, n_iters=250, n_bands=6)
        sc = compute_score(a_mb)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_mb.copy(), 0.0, 1000.0)
            a_norm = a_mb.copy() / a_mb.sum()
            print(f"  [P2-MBAND] NEW BEST: {sc:.8f}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==============================================================
    # Phase 3: Chained LSE LBFGS (40% time)
    # ==============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Chained LBFGS (t<{phase3_end - t_start:.0f}s) ===")

    # Adaptive beta schedule: start moderate, go high
    beta_schedule = [500, 1000, 2000, 4000, 8000, 15000, 30000, 60000, 100000, 200000, 400000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break

        # Allocate iterations: more for higher beta
        base_iters = 2000
        beta_factor = 1.0 + 0.5 * np.log10(1 + beta / 1000)
        maxiter = int(min(30000, base_iters * beta_factor))
        maxfun = maxiter * 25

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  [P3-LBFGS b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            # If no improvement, try from perturbed best
            if time.monotonic() < phase3_end - 30:
                a_pert = best_a * (1.0 + 0.015 * rng.standard_normal(len(best_a)))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
                S_p = a_pert.sum()
                if S_p > 1e-6:
                    a_pert = a_pert / S_p
                    sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = np.clip(a_opt2.copy(), 0.0, 1000.0)
                        print(f"  [P3-PERT b={beta}] NEW BEST: {sc2:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==============================================================
    # Phase 4: Final refinement + restarts (15% time)
    # ==============================================================
    print(f"\n=== Phase 4: Final refinement ===")

    # One more round of Fienup from best
    if time.monotonic() < deadline - 90:
        a_f = fienup_adaptive(best_a / best_a.sum(), n_iters=200, gamma=0.5,
                              top_frac_init=0.05, top_frac_end=0.7,
                              soft_alpha=20.0, peak_target=True, peak_k=25)
        sc = compute_score(a_f)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_f.copy(), 0.0, 1000.0)
            print(f"  [P4-FIENUP] NEW BEST: {sc:.8f}")

    # Final high-beta LBFGS
    if time.monotonic() < deadline - 60:
        for beta in [500000, 1000000]:
            if time.monotonic() > deadline - 30:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=120000)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
                print(f"  [P4-LBFGS b={beta}] NEW BEST: {sc:.8f}")

    # Perturbation restarts
    for trial in range(5):
        if time.monotonic() > deadline - 15:
            break
        noise_scale = 0.005 * (1 + trial * 0.5)
        a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S_p = a_p.sum()
        if S_p < 1e-12:
            break
        a_p = a_p / S_p

        # Quick Fienup then LBFGS
        a_r = fienup_adaptive(a_p, n_iters=100, gamma=0.7, top_frac_init=0.1,
                              top_frac_end=0.5, soft_alpha=30.0, peak_target=True, peak_k=15)
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs(a_r, 50000, maxiter=3000, maxfun=40000)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
                print(f"  [P4-RESTART-{trial}] NEW BEST: {sc:.8f}")

    # ==============================================================
    # Output
    # ==============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
