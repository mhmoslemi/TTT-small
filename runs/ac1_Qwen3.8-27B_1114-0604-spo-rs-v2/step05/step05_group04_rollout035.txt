Uses peak-lag targeting, spectral coordinate descent, and adaptive L-BFGS.

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_peak_lag(a, n_iters=500, K=5, soft_alpha=50.0, damp=0.5):
    """
    Fienup iteration that specifically targets the top K peak lags in the autocorrelation.
    For each peak lag p, we compute the "negative" of the lag-p contribution to each spectral bin
    and use it to suppress those bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        
        c = fast_conv(a)
        M_val = float(c.max())
        
        # Find top K peak positions
        top_idx = np.argsort(c)[::-1][:K]
        
        A = np.fft.rfft(a, M_fft)

        P_safe = np.maximum(P, 1e-30)
        
        # For each peak lag, compute the phase contribution
        # R[p] = (1/M_fft) * sum_k P[k] * cos(2*pi*k*p/M_fft)
        # To reduce R[p], we want to decrease P[k] where cos(2*pi*k*p/M_fft) > 0
        # and increase P[k] where cos(2*pi*k*p/M_fft) < 0
        # But we can't increase (would increase total power), so we just suppress positively-contributing bins
        
        k_vals = np.arange(len(P))
        scale = np.ones_like(P)
        
        for p in top_idx:
            p = int(p)

            # Suppress bins that contribute positively to R[p]
            # Use a soft version: scale by (1 - damp * max(cos_phase, 0))
            suppression = 1.0 - damp * np.maximum(cos_phase, 0.0)
            suppression = np.maximum(suppression, 0.1)  # Don't suppress too much

            # Renormalize scale to preserve total power
            scale = scale / (float((scale * P_safe).sum()) / float(P_safe.sum()))

        a = a_new

                # Ensure Hermitian symmetry for real signal
                # rfft only stores positive frequencies, so irfft handles this
                a_try = np.fft.irfft(A_try, M_fft)[:n]
                a_try = np.maximum(a_try, 0.0)
                
                S_try = float(a_try.sum())
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try
                
                sc = compute_score(a_try)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_try.copy()
                    amp = amp_try.copy()
                    # Update A and phase from the new a
                    A = np.fft.rfft(a_try, M_fft)
                    amp = np.sqrt(A.real ** 2 + A.imag ** 2)
                    phase = np.angle(A)
                    break  # Move to next bin
    
    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient of L with respect to x
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_spectral_eq(a, n_iters=300, alpha_init=0.3, alpha_end=0.8):
    """Spectral equalization: relax spectrum towards flat."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))

    inits.append(0.3 + 0.7 * x)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    return inits

"""
Advanced spectral optimization for minimizing peak auto-convolution of non-negative sequences.
Combines peak-lag targeting Fienup, adaptive damping, spectral coordinate descent,
and high-beta L-BFGS polishing. Novel approach: directly targets specific autocorrelation
peak lags via spectral phase analysis rather than generic top-bin compression.

import numpy as np

import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_peak_lag(a, n_iters=500, K=5, soft_alpha=50.0, damp=0.4):
    """
    Fienup iteration targeting top K peak lags in autocorrelation.
    For each peak lag p, suppresses spectral bins that contribute positively
    to R[p] via cosine phase analysis.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    k_vals = np.arange(len(np.fft.rfft(a, M_fft)), dtype=np.float64)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)

        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:K]

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_safe = np.maximum(P, 1e-30)

        scale = np.ones_like(P)
        adaptive_damp = damp * (1.0 - 0.3 * progress)

        for p in top_idx:
            p_int = int(p)
            cos_phase = np.cos(2.0 * np.pi * k_vals * p_int / M_fft)
            suppression = 1.0 - adaptive_damp * np.maximum(cos_phase, 0.0)
            suppression = np.maximum(suppression, 0.05)
            scale *= suppression
            total_current = float((scale * P_safe).sum())
            total_target = float(P_safe.sum())
            if total_current > 1e-30:
                scale = scale * (total_target / total_current)

        A_mod = A * np.sqrt(np.maximum(scale, 0.0))
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_alpha > 0:
            clipped = np.clip(a_new, -5.0, 5.0)
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * clipped))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_adaptive(a, n_iters=500):
    """
    Adaptive Fienup: dynamically adjusts top-fraction and compression strength
    based on progress. Escalates aggressiveness when stuck.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    rng_local = np.random.default_rng(123)

    for it in range(n_iters):
        progress = it / max(1, n_iters)

        if no_improve < 30:
            top_frac = 0.05 + 0.10 * progress
            gamma = 0.7
        elif no_improve < 80:
            top_frac = 0.20 + 0.15 * progress
            gamma = 0.5
        else:
            top_frac = 0.40 + 0.20 * progress
            gamma = 0.3

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = np.maximum(thresh / P[above], 0.001)
            scale[above] = ratio ** 0.5 * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                a = a * (1.0 + 0.02 * rng_local.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a


def fienup_spectral_eq(a, n_iters=300, alpha_init=0.3, alpha_end=0.8):
    """Spectral equalization: relax power spectrum towards uniform."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_dual_domain(a, n_iters=400):
    """
    Dual-domain iteration: alternates between time-domain gradient step
    (reducing peak lag) and frequency-domain flattening.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step_size = 0.02

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Time-domain step: reduce peak lag
        peak_pos = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            idx = peak_pos - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]
        # Normalize gradient
        gnorm = float(np.linalg.norm(grad))
        if gnorm > 1e-15:
            step = min(step_size, (sc - 1.0) / (gnorm * 2.0 + 1e-10))
            a = a - max(step, 1e-6) * grad / gnorm
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S

        # Frequency-domain step: flatten spectrum
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha_f = 0.1 + 0.2 * progress
        P_new = (1.0 - alpha_f) * P + alpha_f * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        step_size *= 0.999

    return best_a


def spectral_coord_descent(a, n_passes=8, n_samples=60, step_size=0.08):
    """
    Coordinate descent on individual spectral magnitudes.
    Tries small perturbations to each sampled bin.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    A = np.fft.rfft(a, M_fft)
    amp = np.sqrt(A.real ** 2 + A.imag ** 2)
    phase = np.angle(A)
    n_bins = len(amp)

    rng = np.random.default_rng(777)

    for p in range(n_passes):
        idx = rng.choice(n_bins, size=min(n_samples, n_bins), replace=False)

        for i in idx:
            for direction in [1.0, -1.0]:
                amp_try = amp.copy()
                amp_try[i] *= (1.0 + direction * step_size)
                if amp_try[i] < 0:
                    amp_try[i] = 0.001

                A_try = amp_try * np.exp(1j * phase)
                a_try = np.fft.irfft(A_try, M_fft)[:n]
                a_try = np.maximum(a_try, 0.0)

                S_try = float(a_try.sum())
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try

                sc = compute_score(a_try)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_try.copy()
                    A = np.fft.rfft(a_try, M_fft)
                    amp = np.sqrt(A.real ** 2 + A.imag ** 2)
                    phase = np.angle(A)
                    break

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective with analytical gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, rng, parent):

        a_final = fienup_peak_lag(best_a, 400, K=3, soft_alpha=60.0, damp=0.2)
        sc_final = compute_score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"    FINAL PEAKLAG: {sc_final:.8f}")

    if time.monotonic() < deadline - 25:
        a_final2 = fienup_adaptive(best_a, 300)
        sc_final2 = compute_score(a_final2)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_a = a_final2.copy()
            print(f"    FINAL ADAPTIVE: {sc_final2:.8f}")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=30000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    FINAL LBFGS: {sc:.8f}")

    # ============ Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
