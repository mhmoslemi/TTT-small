"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: (1) GS-Fienup with exact flat-convolution target, (2) subgradient descent with
correct gradient formula ∂(a*a)[k]/∂a[j] = 2*a[k-j], (3) L-BFGS on LSE objective,
(4) projected subgradient with adaptive active set.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def subgradient_active(a, active_k, n):
    """
    Compute subgradient of max_k (a*a)[k] at a, restricted to active indices.
    Formula: sg[j] = 2 * sum_{k in active_k} a[k-j] * 1[0 <= k-j < n]
    """
    sg = np.zeros(n)
    for k in active_k:
        k = int(k)
        # a[k-j] for j = 0..n-1 is valid when 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # sg[j] += 2 * a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start to k-j_end
            idx_start = k - j_end  # = k - min(k, n-1)
            idx_end = k - j_start  # = k - max(0, k-n+1)
            sg[j_start:j_end + 1] += 2.0 * a[idx_start:idx_end + 1][::-1]
    return sg


def subgradient_descent(a, n_iters=5000, top_m=20, tol=1e-6):
    """Subgradient descent with projected step onto simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    patience = 200
    patience_counter = 0

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / 1.0  # since sum=1, score = 2n*max(c)
        # Actually score = 2n * max(c) / (sum)^2 = 2n * max(c) since sum=1
        sc = 2.0 * n * c_max

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience and it > 200:
                break

        # Determine active set: top_m convolution values
        if top_m >= len(c):
            active_idx = np.arange(len(c))
        else:
            active_idx = np.argsort(c)[::-1][:top_m]
            # Also include all within epsilon of max
            threshold = c_max - 0.05 * c_max
            extra = np.where(c >= threshold)[0]
            active_idx = np.unique(np.concatenate([active_idx, extra]))

        sg = subgradient_active(a, active_idx, n)

        # Project gradient onto tangent space of simplex (sum=1)
        # For equality constraint sum(a)=1: subtract mean
        grad_proj = sg - np.mean(sg)

        gn = float(np.linalg.norm(grad_proj))
        if gn < 1e-15:
            break

        # Polyak step size
        step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.1 / (1.0 + it * 0.001))
        step = max(step, 1e-8)

        a_new = a - step * grad_proj
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        # Gradually increase top_m
        if it % 500 == 0 and top_m < 100:
            top_m = min(top_m + 5, 100)

    return best_a


def gs_fienup_v2(a, n_iters=600, alpha=0.5, M=None, use_phase=False):
    """
    Improved GS-Fienup: target is the DFT of the flat linear convolution pattern.
    For a flat conv of length 2n-1 (value 1/(2n-1)), padded to M:
    target_conv[m] = 1/(2n-1) for m=0..2n-2, 0 otherwise.
    Target for A^2 = DFT(target_conv).
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1 - 1).bit_length() if 2 * n - 1 > 1 else 2
        M = max(M, 2 * n - 1)

    # Build target: flat convolution of length 2n-1, zero-padded to M
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)  # This is the target for A^2

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        if use_phase:
            new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
            new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        else:
            new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
            new_phase = A_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_anneal(a, n_iters=800, M=None):
    """GS with annealed alpha: start high, end low."""
    n = len(a)
    if M is None:

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[j] = sum_k p[k] * 2*a[k-j] (valid range)
    # = 2 * (p reversed convolved with a) ... let me just do it directly
    # Actually: dL/da[j] = sum_k p[k] * d(a*a)[k]/da[j] = sum_k p[k] * 2*a[k-j]
    # = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is 2 * (cross-correlation of p and a evaluated at specific lags)
    # More precisely: sum_k p[k] * a[k-j] = (p * a_rev)[2n-2-j] ... let me use FFT
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    # dL/da[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = 2 * sum_{m: 0<=m<n} p[m+j] * a[m]  (substituting m = k-j)
    # = 2 * (p shifted and multiplied by a, summed)
    # This is 2 * cross-correlation-style. Let's use the formula:
    # sum_k p[k] * a[k-j] for k in [0, 2n-2], j in [0, n-1]
    # Let m = k - j: sum_m p[m+j] * a[m] = (p_rev * a)[2n-2-m] ... 
    # Simplest: for each j, sum_k p[k]*a[k-j] = sum_m a[m]*p[m+j]
    # This is a cross-correlation. Let's compute via FFT.
    # corr[j] = sum_m a[m] * p[m+j] = (a * p_rev)[j + (2n-2)] ... 
    # Actually let me just compute it as: convolve a with p, then take appropriate indices
    # (a * p)[k] = sum_i a[i] * p[k-i]
    # We want sum_m a[m] * p[m+j] = sum_m a[m] * p[(m+j)] 
    # Let k = m+j: this is sum_k a[k-j] * p[k] = (a * p_rev)[k] evaluated... 
    # Let me just do it directly with a different approach:
    # grad_L[j] = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]
    # = 2 * sum_{m=0}^{n-1} p[m+j] * a[m]  (where m = k-j, so k = m+j)
    # Valid when 0 <= m+j <= 2n-2, i.e., -j <= m <= 2n-2-j, but m in [0,n-1]
    # So: grad_L[j] = 2 * sum_{m=0}^{n-1} p[m+j] * a[m]
    # This is a "correlation" between a and shifted p.
    # Can compute as: for each j, dot(a, p[j:j+n])
    # Vectorized: this is a Toeplitz-like operation.
    # Use FFT: define q where q[k] = p[k] for k in [0, 2n-2], else 0.
    # Then grad_L[j]/2 = sum_m a[m] * q[m+j] = (a * q_rev)[(2n-2) + j - (2n-2)] ... 
    # Let me just use the direct formula with the full conv:
    # full_conv = convolve(a, p) has length n + (2n-1) - 1 = 3n-2
    # full_conv[k] = sum_i a[i]*p[k-i] for valid i,k-i
    # We want sum_m a[m]*p[m+j] = full_conv[m+j - m]... no.
    # sum_m a[m]*p[m+j]: let i=m, then = sum_i a[i]*p[i+j]
    # = full_conv[j + i - i + ... ] hmm.
    # Actually full_conv[k] = sum_i a[i]*p[k-i]. Set k-i = i+j => k = 2i+j. Doesn't work cleanly.
    # Let me just use: sum_i a[i]*p[i+j] = sum_i a[i]*p[i+j]
    # If I define r[i] = p[i] for i in [0, 2n-2], then I want sum_i a[i]*r[i+j].
    # This equals the cross-correlation <a, r shifted by -j>.
    # In FFT terms: multiply FFT(a) by conj(FFT(r shifted by -j))... complex.
    # Simplest: just compute sum_i a[i]*p[i+j] for all j simultaneously.
    # p has length 2n-1. p[i+j] for i in [0,n-1], j in [0,n-1].
    # Create matrix P where P[i,j] = p[i+j] (size n x n). Then grad = 2*P.T @ a.
    # But this is O(n^2) which is too slow for large n.
    # Better: use the FFT approach.
    # sum_i a[i]*p[i+j] = (a * p_flipped)[(n-1) + (2n-2) - (n-1) + j - ... ]
    
    # OK let me just use a clean FFT approach:
    # Define: h[j] = sum_i a[i] * p[i+j] for j = 0, ..., 2n-2
    # This is the cross-correlation of a (length n) with p (length 2n-1)
    # = convolve(a_flipped, p)[(n-1) + j] ... let me verify
    # convolve(a_flipped, p)[k] = sum_i a_flipped[i]*p[k-i] = sum_i a[n-1-i]*p[k-i]
    # Let m = n-1-i: = sum_m a[m]*p[k-n+1+m]
    # We want sum_m a[m]*p[m+j], so set k-n+1 = j, i.e., k = j+n-1.
    # So h[j] = convolve(a_flipped, p)[j+n-1]
    
    a_flipped = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    conv_fft_len = 1 << (conv_len - 1).bit_length()
    A_f = np.fft.rfft(a_flipped, conv_fft_len)
    P_f = np.fft.rfft(p, conv_fft_len)
    conv_result = np.fft.irfft(A_f * P_f, conv_fft_len)
    grad_L = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    # Total gradient of score = 2n * L / S^2:
    # d(score)/da[j] = 2n * (grad_L[j]/S^2 - 2*L*a[j]/S^3)
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L * a / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_clipped = np.maximum(x, 0.0)
        S = x_clipped.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        return lse_score_grad(x_clipped, beta)

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def projected_subgradient(a, n_iters=3000, tol=1e-7):
    """
    Projected subgradient with adaptive active set and Polyak step.
    Uses the exact subgradient formula.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    patience = 300
    patience_cnt = 0

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            patience_cnt = 0
        else:
            patience_cnt += 1
            if patience_cnt > patience and it > 100:
                break

        # Active set: all k within relative epsilon of max
        epsilon = max(0.01 * c_max, (c_max - c.min()) * 0.1)
        active_k = np.where(c >= c_max - epsilon)[0]

        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(c))])
        elif len(active_k) > 200:
            # Limit to top 200
            active_k = np.argsort(c)[::-1][:200]

        sg = subgradient_active(a, active_k, n)

        # Project onto tangent space: subtract mean (for sum constraint)
        active_mask = a > 1e-10
        if np.any(active_mask):
            grad_tangent = sg.copy()
            grad_tangent[active_mask] -= np.mean(sg[active_mask])
        else:
            grad_tangent = sg - np.mean(sg)

        gn = float(np.linalg.norm(grad_tangent[active_mask])) if np.any(active_mask) else float(np.linalg.norm(grad_tangent))
        if gn < 1e-12:
            break

        # Polyak step
        gap = sc - sc_lb
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.05 * n / (n + it))  # cap step size
        step = max(step, 1e-10)

        a_new = a - step * grad_tangent
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

    return best_a

    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine primary length
    if parent is not None:
        n_primary = len(parent)
    else:
        n_primary = 6962

    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")
    print(f"Primary length: {n_primary}")
    print(f"Deadline: {deadline - t_start:.0f}s")

    # ==================================================================
    # Phase 1: GS-Fienup with multiple initializations (35% of time)
    # ==================================================================
    phase1_end = t_start + 0.35 * (total_budget - 40.0)
    print(f"\n=== Phase 1: GS-Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    inits = build_diverse_inits(n_primary, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        if time.monotonic() - t_start < 5:
            pass  # Allow first few to run

        t_init = time.monotonic()
        # Multi-alpha GS
        a1 = multi_alpha_gs(a0)
        sc1 = compute_score(a1)
        update_best(sc1, a1, f"GS-multi init={i}")

        if time.monotonic() < phase1_end - 30:
            # Annealed GS on best result
            a2 = gs_anneal(a1, n_iters=500)
            sc2 = compute_score(a2)
            update_best(sc2, a2, f"GS-anneal init={i}")
            print(f"    init={i}: score={sc2:.8f}, time={time.monotonic()-t_init:.1f}s")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Subgradient descent (25% of time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 40.0)
    print(f"\n=== Phase 2: Subgradient (deadline t={phase2_end - t_start:.0f}s) ===")

    # Run subgradient from current best
    a_sg1 = subgradient_descent(best_a, n_iters=3000, top_m=20)
    sc = compute_score(a_sg1)
    update_best(sc, a_sg1, "SG-top20")

    if time.monotonic() < phase2_end - 60:
        a_sg2 = subgradient_descent(best_a, n_iters=3000, top_m=50)
        sc = compute_score(a_sg2)
        update_best(sc, a_sg2, "SG-top50")

    if time.monotonic() < phase2_end - 30:
        a_ps = projected_subgradient(best_a, n_iters=3000)
        sc = compute_score(a_ps)
        update_best(sc, a_ps, "Proj-SG")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: GS refinement + Subgradient interleaved (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 40.0)
    print(f"\n=== Phase 3: GS+SG interleave (deadline t={phase3_end - t_start:.0f}s) ===")

    # Alternate GS and subgradient
    current = best_a.copy()
    for round_num in range(5):
        if time.monotonic() > phase3_end:
            break

        # GS step
        a_gs = gs_fienup_v2(current, n_iters=300, alpha=0.3, use_phase=True)
        sc = compute_score(a_gs)
        if sc < compute_score(current):
            current = a_gs
            update_best(sc, current, f"GS-r{round_num}")

        # SG step
        a_sg = subgradient_descent(current, n_iters=2000, top_m=30)
        sc = compute_score(a_sg)
        if sc < compute_score(current):
            current = a_sg
            update_best(sc, current, f"SG-r{round_num}")

        # Projected SG
        if time.monotonic() < phase3_end - 30:
            a_psg = projected_subgradient(current, n_iters=2000)
            sc = compute_score(a_psg)
            if sc < compute_score(current):
                current = a_psg
                update_best(sc, current, f"Psg-r{round_num}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS refinement (15% of time)
    # ==================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: L-BFGS (deadline t={phase4_end - t_start:.0f}s) ===")

    for beta in [500, 2000, 5000, 10000]:
        if time.monotonic() > phase4_end - 30:
            break
        t_beta = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=60000)
        update_best(sc, a_opt, f"LBFGS-beta={beta}")
        print(f"    beta={beta}: score={sc:.8f}, time={time.monotonic()-t_beta:.1f}s")

        # Also try from SG result
        if time.monotonic() < phase4_end - 60:
            sc2, a_opt2 = run_lbfgs(a_sg1, beta, maxiter=2000, maxfun=40000)
            update_best(sc2, a_opt2, f"LBFGS-beta={beta}-fromSG")

    # Final GS polish
    if time.monotonic() < deadline - 20:
        a_final = gs_anneal(best_a, n_iters=400)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-GS")

    # Final subgradient polish
    if time.monotonic() < deadline - 15:
        a_final_sg = projected_subgradient(best_a, n_iters=2000)
        sc = compute_score(a_final_sg)
        update_best(sc, a_final_sg, "FINAL-SG")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
