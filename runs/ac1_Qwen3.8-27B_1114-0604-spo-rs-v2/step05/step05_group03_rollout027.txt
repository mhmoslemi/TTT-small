        best_t = 0

        if best_t == 0:

        a = (1-best_t)*a + best_t*x_norm
        a = np.maximum(a, 0)

"""
LP-alternating optimizer using exact LP solver (GLOP/MOSEK via CVXPY)
for minimizing the flat-autocorrelation criterion of non-negative sequences.
"""

import numpy as np
import time
import scipy.sparse as sp
from cvxpy import Variable, Maximize, Problem, Solver as CP_Solver
import cvxpy as cp

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix C where C[k,i] = a[i] if max(0,k-n+1)<=i<=min(k,n-1)"""
    n = len(a)
    n_rows = 2 * n - 1
    # Build in CSR directly for efficiency
    # Row k has entries in columns max(0, k-n+1) to min(k, n-1)
    # with values a[max(0,k-n+1):min(k,n-1)+1]
    
    # Use the structure: for k < n, row k = [a[0], a[1], ..., a[k], 0, ..., 0]
    # for k >= n, row k = [0, ..., 0, a[k-n+1], ..., a[n-1]]
    
    indptr = np.zeros(n_rows + 1, dtype=np.int32)
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        indptr[k+1] = indptr[k] + (i_end - i_start + 1)
    
    total_nz = indptr[-1]
    indices = np.empty(total_nz, dtype=np.int32)
    data = np.empty(total_nz, dtype=np.float64)
    
    pos = 0
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        indices[pos:pos+length] = range(i_start, i_end + 1)
        data[pos:pos+length] = a[i_start:i_end+1]
        pos += length
    
    return sp.csr_matrix((data, indices, indptr), shape=(n_rows, n))

def solve_lp(a, M, solver='GLOP'):
    """Solve: max sum(x) s.t. conv(a,x) <= M, x >= 0"""
    n = len(a)
    C = build_conv_matrix(a)
    x = Variable(n)
    constraints = [x >= 0, C @ x <= M]
    prob = Problem(Maximize(cp.sum(x)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='MOSEK', verbose=False)
        except Exception:
            prob.solve(solver='ECOS', verbose=False)
    
    if x.value is None:
        return None
    return np.maximum(x.value, 0)

def lp_alternating(a, max_iters=100, time_limit=None, t_start=None):
    """LP-based alternating method with exact LP solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        x = solve_lp(a, M)
        
        if x is None or x.sum() < 1e-10:
            break
        
        x = x / x.sum()
        
        # Line search
        best_t = 0

"""
LP-alternating optimizer using exact LP solver for non-negative
flat-autocorrelation sequence design. Uses CVXPY with GLOP/MOSEK
for the inner LP, golden-section line search for mixing, and
multiple initializations including parent interpolation.
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp
from cvxpy import Variable, Maximize, Problem, sum as cp_sum


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    # Vectorized construction
    # For row k (0 <= k < 2n-1):
    #   i_start = max(0, k-n+1), i_end = min(k, n-1)
    #   length = i_end - i_start + 1
    k_range = np.arange(n_rows)
    i_start = np.maximum(0, k_range - n + 1)
    i_end = np.minimum(k_range, n - 1)
    lengths = i_end - i_start + 1
    
    indptr = np.zeros(n_rows + 1, dtype=np.int64)
    indptr[1:] = np.cumsum(lengths)
    total_nz = int(indptr[-1])
    
    # Build indices and data
    indices = np.empty(total_nz, dtype=np.int64)
    data = np.empty(total_nz, dtype=np.float64)
    
    for k in range(n_rows):
        s = int(indptr[k])
        l = int(lengths[k])
        indices[s:s+l] = np.arange(int(i_start[k]), int(i_end[k]) + 1)
        data[s:s+l] = a[int(i_start[k]):int(i_end[k]) + 1]
    
    return sp.csr_matrix((data, indices, indptr), shape=(n_rows, n))


def solve_lp(a, M, timeout=120):
    n = len(a)
    C = build_conv_matrix(a)
    x = Variable(n)
    constraints = [x >= 0, C @ x <= M]
    prob = Problem(Maximize(cp_sum(x)), constraints)
    
    solvers = ['GLOP', 'MOSEK', 'ECOS']
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if x.value is not None:
                return np.maximum(x.value, 0.0)
        except Exception:
            continue
    return None


def lp_alternating(a, max_iters=200, time_limit=None):

    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())

        a = a / S

    best_sc = compute_score(a)
    no_improve = 0
    rng = np.random.default_rng(123)

        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        x = solve_lp(a, M)

            break

        # Golden section search for optimal t
        # f(t) is convex, so we just need to find the minimum
        def f_of_t(t):
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                return np.inf
            a_try = a_try / St
            return compute_score(a_try)

        best_t_sc = best_sc
        grid = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9]

            sc = f_of_t(t)
            if sc < best_t_sc:

                best_t = t

def subgradient_peak(a, n_iters=500):
    """Subgradient method targeting the peak of the autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        
        # Subgradient: g[i] = 2*a[k_star - i] for valid i
        g = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        for i in range(i_start, i_end + 1):
            g[i] = 2.0 * a[k_star - i]
        
        # Projected subgradient step
        # Project onto {a >= 0, sum(a) = 1}
        # Use adaptive step size
        step = 1.0 / (it + 10)
        
        a_new = a - step * g
        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a
