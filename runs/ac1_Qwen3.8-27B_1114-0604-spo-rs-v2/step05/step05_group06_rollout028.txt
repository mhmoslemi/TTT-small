def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _bb_descent(a0, max_iters, deadline, label=""):
    """
    Barzilai-Borwein subgradient descent with backtracking.
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    # BB state
    a_prev = a.copy()
    grad_prev = None
    step_prev = None
    gamma = 1.0  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        S = float(a.sum())
        if S < 1e-12:
            break
        
        # Compute active set and subgradient
        c = _self_conv(a)
        cmax = float(c.max())
        if cmax <= 0:
            break
        
        # Active set: positions within tolerance of max
        tol = 1e-4 * cmax
        active = c >= cmax - tol
        if not active.any():
            active[np.argmax(c)] = True
        
        n_act = int(active.sum())
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act
        
        grad, _ = _subgrad_step(a, w, S)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # BB step size update
        if grad_prev is not None and step_prev is not None:
            ys = np.dot(grad - grad_prev, step_prev)
            ss = float(np.dot(step_prev, step_prev))
            if ss > 1e-20:
                if abs(ys) > 1e-20:
                    gamma = ss / max(abs(ys), 1e-20)
                # Clip gamma
                gamma = np.clip(gamma, 0.01, 10.0)
        
        # Try several step sizes (backtracking)
        base_eta = gamma / gnorm
        candidates = [base_eta * s for s in (0.25, 0.5, 1.0, 2.0, 4.0)]
        
        best_new_sc = best_sc
        best_new_a = None
        
        for eta in candidates:
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = q.copy()
        
        if best_new_a is None:
            # All steps failed, try a very small step
            q = a - 0.01 * base_eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq > 1e-8:
                q = q / Sq
                sc = _score(q)
                if sc < best_sc:
                    best_new_sc = sc
                    best_new_a = q.copy()
        
        if best_new_a is None:
            break
        
        # Update BB state
        step_prev = best_new_a - a
        grad_prev = grad
        a = best_new_a
        
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()
            if it % 50 == 0:
                print(f"    [{label}] iter {it}: {best_sc:.8f} (active={n_act})")
    
    return best_sc, best_a


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective with top-k active set."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    
    w_full = np.zeros(M)
    w_full[idx] = w_act
    
    # Cross-correlation: sum_j a[j] * w[i+j]
    M_total = n + M - 1
    N = _next_pow2(M_total)
    flip_a = a[::-1]
    FA = np.fft.rfft(flip_a, N)
    WA = np.fft.rfft(w_full, N)
    conv = np.fft.irfft(FA * WA, N)[:M_total]
    corr = conv[n-1 : 2*n-1]
    
    LDS = float(w_act @ d_act)
    grad = 4.0 * n * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    """L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    
    est_eval_t = 0.005 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 6000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3
    
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _peak_shave(a, max_iters=50, deadline=None):
    """
    Coordinate descent: for the peak position, try reducing individual a[i]
    values that contribute most to the peak.
    """
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _self_conv(a)
        cmax = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find positions i that contribute to c[k_star]:
        # c[k_star] = sum_i a[i] * a[k_star - i]
        # The contribution of a[i] is a[i] * a[k_star - i]
        # Reducing a[i] by delta reduces c[k_star] by delta * a[k_star - i]
        
        # But reducing a[i] also increases some other c values potentially.
        # Let's try: for the top contributors to the peak, reduce them slightly.
        
        i_range = np.arange(n)
        j = k_star - i_range
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i_range[valid]] * a[j[valid]]
        
        # Try reducing top contributors
        top_k = min(20, n)
        top_idx = np.argsort(contrib)[-top_k:]
        
        improved = False
        for idx in top_idx:
            if a[idx] <= 0:
                continue
            for frac in (0.1, 0.2, 0.3, 0.5):
                q = a.copy()
                q[idx] *= (1.0 - frac)
                Sq = float(q.sum())
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc = _score(q)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                    a = q
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    if k < 3:
        k = 3
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-stage LSE optimization (log-sum-exp with growing active set K),
BB subgradient descent with backtracking, and coordinate-descent peak shaving.
Explores U-shape and spectral initializations alongside parent resampling.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _cross_corr_a_w(a, w):
    """
    Compute g[i] = sum_j a[j] * w[i+j] for i=0..n-1,
    where w has length 2n-1 (valid indices 0..2n-2).
    Uses FFT.
    """
    n = a.size
    M = 2 * n - 1
    M_total = n + M - 1
    N = _next_pow2(M_total)
    flip_a = a[::-1]
    FA = np.fft.rfft(flip_a, N)
    WA = np.fft.rfft(w, N)
    conv = np.fft.irfft(FA * WA, N)[:M_total]
    return conv[n - 1:2 * n - 1]


def _lse_obj_grad(a, beta, k):
    """
    LSE-smoothed objective: 2 * (dmax + log(sum(exp(beta*(d-dmax))))/beta)
    where d[i] = n * c[i] / S^2, using top-k positions.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    if k >= M:
        idx = np.arange(M)
    else:
        idx = np.argpartition(d, M - k)[M - k:]

    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    z = np.clip(z, -700, 0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(len(idx)) / len(idx)
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr_a_w(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * n * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    """L-BFGS-B on LSE objective with sum-penalty."""
    a = _norm(a0)
    n = len(a)
    t0 = time.monotonic()
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.80 * time_limit / max(est_eval_t, 1e-4))
    mf = max(20, min(mf, maxfun_cap, 5000))
    maxiter = max(15, mf // 2)
    PEN = 5e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    a_start = a.copy()
    best_sc = _score(a_start)
    best_x = a_start

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        if getattr(res, 'x', None) is not None:
            out = _norm(res.x)
            sc = _score(out)
            if sc < best_sc:
                best_sc = sc
                best_x = out
    except Exception:
        pass

    return best_sc, best_x


def _bb_descent(a0, max_iters, deadline, tol=1e-5):
    """
    Subgradient descent with Barzilai-Borwein step sizes and backtracking.
    Minimizes f(a) = 2n*max(c)/S^2 where c = conv(a,a).
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    grad_prev = None
    step_prev = None
    gamma = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        S = float(a.sum())
        if S < 1e-12:
            break

        c = _self_conv(a)
        cmax = float(c.max())
        if cmax <= 0:
            break

        # Adaptive active set
        tol_act = max(tol * cmax, 1e-10)
        active = c >= cmax - tol_act
        if not active.any():
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act

        corr = _cross_corr_a_w(a, w)
        LDAP = float(c[active].max())
        grad = 2.0 * n * corr / (S * S) - 4.0 * n * LDAP * a / (S ** 3)
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        # BB step size
        if grad_prev is not None and step_prev is not None:
            ys = float(np.dot(grad - grad_prev, step_prev))
            ss = float(np.dot(step_prev, step_prev))
            if ss > 1e-20:
                if ys > 1e-10:
                    gamma_bb = ss / ys
                    gamma = 0.5 * gamma + 0.5 * np.clip(gamma_bb, 0.001, 10.0)
                else:
                    gamma = max(gamma * 0.5, 0.001)

        # Backtracking line search
        base_eta = gamma / gnorm
        tried_steps = [base_eta * s for s in (0.125, 0.25, 0.5, 1.0, 2.0, 4.0)]

        best_new_sc = best_sc
        best_new_a = None

        for eta in tried_steps:
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = q.copy()

        if best_new_a is None:
            # Try even smaller
            for eta in (base_eta * 0.01, base_eta * 0.001):
                q = a - eta * grad
                q = np.clip(q, 0.0, None)
                Sq = float(q.sum())
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc = _score(q)
                if sc < best_sc:
                    best_new_sc = sc
                    best_new_a = q.copy()
                    break

        if best_new_a is None:
            break

        step_prev = best_new_a - a
        grad_prev = grad.copy()
        a = best_new_a

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()
            if it % 100 == 0:
                print(f"      BB iter {it}: {best_sc:.8f} (active={n_act}, gamma={gamma:.4f})")

    return best_sc, best_a


def _peak_shave(a, max_rounds=40, deadline=None):
    """
    Coordinate descent: reduce individual a[i] values that most contribute
    to the maximum convolution peak.
    """
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for rnd in range(max_rounds):
        if deadline and time.monotonic() > deadline:
            break

        c = _self_conv(a)
        cmax = float(c.max())
        k_star = int(np.argmax(c))

        # Contributions to c[k_star]: a[i] * a[k_star - i]
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i_vals[valid]] * a[j_vals[valid]]

        # Try reducing top contributors
        top_count = min(min(30, n), max(5, n // 200))
        top_idx = np.argsort(contrib)[-top_count:]
        top_idx = top_idx[::-1]  # largest first

        improved = False
        for idx in top_idx:
            if a[idx] <= 1e-15:
                continue
            for frac in (0.05, 0.1, 0.2, 0.35, 0.5):
                q = a.copy()
                q[idx] *= (1.0 - frac)
                Sq = float(q.sum())
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc = _score(q)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = q.copy()
                    a = q
                    improved = True
                    break

        if not improved:
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _u_shape_init(n, rng):
    """U-shaped initialization: high at edges, lower in middle."""
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.3, 1.5)
    beta_val = rng.uniform(1.0, 4.0)
    a = 1.0 + alpha * np.abs(2.0 * x - 1.0) ** beta_val
    noise = rng.standard_normal(n) * 0.05
    a = a + noise
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    if k < 3:
        k = 3
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

                    best_score = sc_f
                    best_seq = a_f.copy()
                    print(f"  Final LSE K={k_f}, beta={beta_f:.0f}: {best_score:.8f}")
                best_seq = _norm(a_f)

    # BB subgradient
    if time.monotonic() < final_deadline - 20:
        sc_bb, a_bb = _bb_descent(best_seq, max_iters=100, deadline=final_deadline - 20)
        if sc_bb < best_score:
            best_score = sc_bb
            best_seq = a_bb.copy()
            print(f"  Final BB: {best_score:.8f}")
        best_seq = _norm(a_bb)

    # Peak shave
    if time.monotonic() < final_deadline - 15:
        sc_ps, a_ps = _peak_shave(best_seq, max_rounds=30, deadline=final_deadline - 15)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
            print(f"  Final peak_shave: {best_score:.8f}")
        best_seq = _norm(a_ps)

    # One more BB
    if time.monotonic() < final_deadline - 5:
        sc_final, a_final = _bb_descent(best_seq, max_iters=30, deadline=final_deadline - 3)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        # Scale so sum is reasonable (evaluate_sequence is scale-invariant,
        # but keep values in [0, 1000])
        out = out / S  # sum = 1

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
