"""

Uses LSE-smoothed L-BFGS-B with multi-length exploration, spectral initialization,
Nesterov-accelerated subgradient polishing, and adaptive time allocation.

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute sum_k w[k] * conv(a, delta_k) for each position in a.
    This is the gradient of the weighted sum of convolution values."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    # corr_pair computes: for each position i in a, the "pressure" from the weights
    # grad of sum_k w[k]*conv[k] w.r.t. a[i] = sum_k w[k] * a[k-i] (for valid k-i)
    # This is a correlation: (a * w_flipped)[n-1:2n-1]
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def d_from_a(a):
    n = len(a)

    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)

def soft_active_weights(d, beta_w):
    """Soft weights based on how close each lag is to the max."""
    d = np.asarray(d, dtype=np.float64)
    dmax = float(np.max(d))
    z = beta_w * (d - dmax)
    w = np.exp(z)
    sw = float(w.sum())
    if sw < 1e-300:
        return np.ones_like(d) / len(d)
    return w / sw

def subgrad_step(a, eta, weights):
    """Single subgradient step using given weights over lags."""
    n = len(a)
    h = corr_pair(a, weights)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)

    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n, dtype=np.float64)  # momentum
    
    for t in range(1, max_iters + 1):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        d, S = d_from_a(y)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax < 1e-12:

        h = corr_pair(y, w)
        hmax = float(np.max(np.abs(h)))
        if hmax < 1e-12:
            break
        
        # Adaptive step size
        eta = eta0 / np.sqrt(1.0 + t * 0.01)
        
        # Gradient step
        grad = h / hmax
        v = 0.9 * v - eta * grad
        y = y * np.exp(-0.5 * np.abs(grad))  # multiplicative update
        y = normalize_a(y)
        
        sc = exact_score_a(y)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = y.copy()
    
    return float(best_sc), best_a

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()

        return np.array([], dtype=np.float64)

    S = float(a.sum())

def spectral_init(n, rng):
    """Generate initialization from flat-spectrum random phase."""
    N = next_power_of_two(2 * n)
    half = N // 2 + 1
    phase = rng.uniform(-np.pi, np.pi, size=half)
    mag = np.ones(half, dtype=np.float64)
    # Slightly shape the magnitude to help non-negativity
    freqs = np.arange(half, dtype=np.float64) / half
    mag = 1.0 + 0.3 * np.cos(np.pi * freqs)
    spec = mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    if N % 2 == 0:
        spec_full = np.zeros(N, dtype=np.complex128)
        spec_full[:half] = spec
        spec_full[half:] = np.conj(spec[1:half][::-1])
    else:
        spec_full = np.zeros(N, dtype=np.complex128)
        spec_full[:half] = spec
        spec_full[half:] = np.conj(spec[1:half][::-1])
    sig = np.fft.ifft(spec_full).real[:n]
    sig = np.maximum(sig, 0.0)
    if sig.sum() < 1e-10:
        sig = np.ones(n)
    return normalize_a(sig)


def generate_inits(n, rng, parent, cap=20):
    inits = []
    n = int(n)

    def add(arr):

            return

        if arr.size != n:

            arr = arr - mn + 0.05

        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Arc-sine-like (high at ends)
    for eps in (0.01, 0.05, 0.1):
        add(1.0 / np.sqrt(np.abs(x - 0.5) * 2.0 + eps))

    # Cosine shapes
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # W-shape (two peaks)
    for pos in (0.25, 0.3, 0.35):
        add(np.exp(-(((x - pos) / 0.08) ** 2)) + np.exp(-(((x - (1.0 - pos)) / 0.08) ** 2)))

    # Spectral initializations
    for _ in range(4):
        add(spectral_init(n, rng))

    # Random smooth
    Nf = next_power_of_two(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 0.8)) * noise)

    # Pure random
    for _ in range(2):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):

    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        d, S = d_from_a(p)
        if d.size > 0:

                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 3:
        # Smooth in log domain
        logp = np.log(p + 1e-300)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)

        kern = kern / float(kern.sum())
        noise = rng.standard_normal(n)

        noise = noise - float(noise.mean())

        q = np.exp(logp - float(np.max(logp)))

        # V-shape multiplication
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * 0.5 * np.abs(2.0 * x - 1.0)
        q = p * vshape
    else:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    return normalize_a(np.clip(q, 0.0, None))


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for beta in (100, 500, 2000, 10000):
        rem = float(deadline - time.monotonic())
        if rem < 15.0:
            break
        n = len(a)
        est_time = max(0.02, 0.02 * (n / 8192.0))
        mf = max(64, int(min(2000, (0.7 * rem) / est_time)))
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)
    rem = float(deadline - time.monotonic())
    if rem > 20.0:
        n = len(a)
        est_time = max(0.02, 0.02 * (n / 8192.0))
        mf = max(64, int(min(3000, (0.8 * rem) / est_time)))
        sc, a2 = run_lbfgs(a, 50000.0, deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
    return float(best_sc), best_a


def optimize_trajectory(a_start, deadline, rng, label):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    n = len(a)
    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    stages = [
        (50.0, 0.08),
        (200.0, 0.10),
        (500.0, 0.10),
        (2000.0, 0.12),
        (5000.0, 0.12),
        (15000.0, 0.15),
        (50000.0, 0.15),
        (150000.0, 0.18),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est_time = max(0.02, 0.02 * (n / 8192.0))
            mf = max(64, int(min(5000, (0.8 * rem) / est_time)))
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} elapsed={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)
        # Subgradient polish between stages
        if time.monotonic() < sub_dead - 3.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=15, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad best {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine lengths to try - include longer sequences
    lengths = []
    parent_len = 0
    if parent is not None:
        parent_len = int(min(parent.size, 32768))
        if parent_len < 4096:
            parent_len = 4096
        lengths.append(parent_len)
        for L in [8192, 12288, 16384, 20480, 24576, 28672, 32768]:
            if L != parent_len:
                lengths.append(L)
    else:
        lengths = [16384, 20480, 12288, 24576, 32768]

    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 4096 <= L <= 32768:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:5]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Generate and evaluate initializations
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=12 if n > 16000 else 16)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))

    if not all_scores:
        n0 = lengths[0] if lengths else 16384
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.8f} (from length {})".format(
        all_scores[int(order[0])], all_lens[int(order[0])]))

    # Select top starts (at most 3, from different lengths)
    starts = []
    used_lens = set()

    if parent is not None and parent_len > 0:
        ps = normalize_a(resample_sequence(parent, parent_len))
        starts.append(ps)
        used_lens.add(parent_len)

    for idx in order:
        if len(starts) >= 3:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [0.50, 0.30, 0.20]

    main_end = deadline - 250.0
    main_budget = max(60.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))

        if time.monotonic() > sub_dead - 20.0:
            continue

        sq = starts[si].copy()
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            t_blend = 0.4 if rank == 1 else 0.6
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Nesterov subgradient on the best
    if time.monotonic() < deadline - 150.0:
        sc, seq = nesterov_subgrad(best_seq, deadline=deadline - 150.0,
                                   max_iters=500, beta_w=8.0, eta0=0.03)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Nesterov subgrad: {:.8f}".format(best_score))

    # Phase 4: Standard subgradient polish
    if time.monotonic() < deadline - 120.0:
        sc, seq = subgrad_polish(best_seq, max_iters=30, deadline=deadline - 120.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # Phase 5: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 100.0 and perturb_done < 4:
        left = float(deadline - 100.0 - time.monotonic())
        if left < 50.0:
            break
        chunk = time.monotonic() + left / float(4 - perturb_done)
        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 6: Final high-beta LSE polish
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - time.monotonic() - 3.0)
        if rem > 10.0:
            n = len(best_seq)
            est_time = max(0.02, 0.02 * (n / 8192.0))
            mf = max(64, int(min(5000, (0.85 * rem) / est_time)))
            sc, seq = run_lbfgs(best_seq, 100000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

    # Phase 7: Final subgradient cleanup
    if time.monotonic() < deadline - 10.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
