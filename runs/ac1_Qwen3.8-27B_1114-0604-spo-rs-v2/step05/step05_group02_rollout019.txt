"""
Parametric perturbation + GS-Fienup + L-BFGS hybrid optimizer.
Key innovation: Low-dimensional parametric optimization using cosine basis
perturbations of the parent sequence, which efficiently explores the
manifold of nearby sequences and avoids the expensive full-dimensional
optimization while still finding improvements.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp as _logsumexp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_fienup(a, n_iters=300, alpha_init=0.6, alpha_final=0.1, M=None):
    """GS-Fienup with flat-convolution target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def parametric_score(params, base, basis, n):
    """Score of the parametrically perturbed sequence."""
    perturbation = np.zeros(n)
    for m in range(len(params)):
        perturbation += params[m] * basis[m]
    a = base * (1.0 + perturbation)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    a = a / S
    return compute_score(a)


def parametric_score_grad(params, base, basis, n):
    """Score and its gradient w.r.t. params."""
    perturbation = np.zeros(n)
    for m in range(len(params)):
        perturbation += params[m] * basis[m]
    a = base * (1.0 + perturbation)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(len(params))
    a = a / S
    sc = compute_score(a)
    
    # Numerical gradient
    grad = np.zeros(len(params))
    eps = 1e-7
    for m in range(len(params)):
        params[m] += eps
        sc_plus = parametric_score(params, base, basis, n)
        params[m] -= 2*eps
        sc_minus = parametric_score(params, base, basis, n)
        params[m] += eps
        grad[m] = (sc_plus - sc_minus) / (2*eps)
    return sc, grad


def run_parametric_opt(base, n, n_basis=20, method='L-BFGS-B'):
    """Optimize parameters in the basis expansion."""
    x = np.arange(n, dtype=np.float64) / n
    
    # Build basis: cosine modes
    basis = []
    for m in range(1, n_basis + 1):
        basis.append(np.cos(2.0 * np.pi * m * x))
    basis = np.array(basis)  # shape (n_basis, n)
    
    # Initial params: zeros
    x0 = np.zeros(n_basis)
    
    # Bounds: keep perturbation moderate
    bounds = [(-0.5, 0.5)] * n_basis
    
    best_sc = compute_score(base)
    best_params = x0.copy()
    
    # Try multiple initial points
    rng = np.random.default_rng(42)
    inits = [x0]
    for _ in range(5):
        inits.append(rng.standard_normal(n_basis) * 0.05)
    
    for x_init in inits:
        res = minimize(
            parametric_score, x_init,
            args=(base, basis, n),
            method=method, bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        sc = parametric_score(res.x, base, basis, n)
        if sc < best_sc:
            best_sc = sc
            best_params = res.x.copy()
    
    # Reconstruct
    perturbation = np.zeros(n)
    for m in range(n_basis):
        perturbation += best_params[m] * basis[m]
    a = base * (1.0 + perturbation)
    a = np.maximum(a, 0.0)
    return a, best_sc


def run_lbfgs_lse(a0, beta, maxiter=5000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    def obj(x):
        x = np.maximum(x, 0.0)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros_like(x)
        c = fast_conv(x)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_s = np.exp(scaled)
        log_sum = float(np.log(exp_s.sum()))
        L = c_max + log_sum / beta
        score = 2.0 * n * L / (S * S)
        
        # Gradient
        p = exp_s / exp_s.sum()
        # Convolve reversed a with p to get gradient
        rev_a = x[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
        g = 2.0 * conv_res
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
        return score, grad
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxiter*20}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_refine(a, n_iters=5000, top_m=20):
    """Subgradient method with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)  # momentum
    beta_mom = 0.9
    step = 0.03
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top-m peak indices
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Momentum update
        v = beta_mom * v + (1 - beta_mom) * grad
        a -= step * v
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay step size
        step *= 0.9995
        
        if time.monotonic() > deadline_global:
            break

    return best_a
