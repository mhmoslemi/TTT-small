Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses monotone-sequence-inspired initializations, adaptive LSE smoothing with
progressive active-set growth, FISTA as complementary optimizer, and
perturbation-based exploration to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 2)
    PEN = 5e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _fista_opt(a0, beta, k, time_limit, max_iter=5000):
    a = _norm(a0).copy()
    n = len(a)
    t_end = time.monotonic() + time_limit
    best_a = a.copy()
    best_sc = _score(a)
    x_prev = a.copy()
    # Estimate Lipschitz constant
    _, g0 = _lse_obj_grad(a, beta, k)
    gn = np.linalg.norm(g0)
    if gn < 1e-12:
        return best_sc, best_a
    # Use a conservative step
    S = a.sum()
    c = _self_conv(a)
    d = n * c / (S * S)
    dmax = d.max()
    L_est = max(1.0, 4.0 * n * dmax / (S * S) * 10)
    step = 1.0 / L_est
    tt = 1.0
    for it in range(max_iter):
        if time.monotonic() > t_end:
            break
        y = a + ((tt - 1.0) / (tt + 2.0)) * (a - x_prev)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-8:
            break
        _, g_y = _lse_obj_grad(y, beta, k)
        a_new = y - step * g_y
        a_new = np.maximum(a_new, 0.0)
        a_new = _norm(a_new)
        tt_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tt * tt))
        x_prev = a.copy()
        a = a_new
        tt = tt_new
        if it % 200 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_sc, best_a


def _subgrad_polish(a, max_iters=50, deadline=None):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    momentum = 0.0
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        # Use top few positions for subgradient
        k_top = min(10, len(d))
        idx = np.argpartition(d, -k_top)[-k_top:]
        active = d >= dmax - 1e-4 * max(dmax, 1.0)
        if not active.any():
            active = np.zeros_like(d, dtype=bool)
            active[idx[-1]] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        # Nesterov-style momentum
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0002, 0.0008, 0.003, 0.01, 0.03, 0.08, 0.15):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _generate_inits(n, rng):
    inits = []
    x = np.linspace(0, 1, n)
    # Monotone sequences (key insight: these can beat uniform)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (x + 1e-6) ** p
        inits.append((f'pow_inc_{p}', a / a.sum()))
        a = (1.0 - x + 1e-6) ** p
        inits.append((f'pow_dec_{p}', a / a.sum()))
    # Exponential
    for rate in [0.5, 1.0, 2.0, 5.0, 10.0]:
        a = np.exp(rate * x)
        inits.append((f'exp_inc_{rate}', a / a.sum()))
        a = np.exp(-rate * x)
        inits.append((f'exp_dec_{rate}', a / a.sum()))
    # Sine/cosine modulations (non-negative)
    for f in [1, 2, 3, 5, 8]:
        for phase in [0, np.pi / 2]:
            a = 0.3 + 0.7 * np.sin(np.pi * f * x + phase)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'sin_{f}_{phase:.2f}', a / a.sum()))
    # Two-segment monotone
    for split in [0.3, 0.4, 0.5]:
        a = np.ones(n)
        idx = int(n * split)
        a[:idx] = np.linspace(1, 0.2, idx)
        a[idx:] = np.linspace(0.2, 1, n - idx)
        a = np.clip(a, 0.01, None)
        inits.append((f'V_{split}', a / a.sum()))
    # Spectral inits
    for i in range(4):
        inits.append((f'spec{i}', _spectral_init(n, rng)))
    # Smoothed random
    for i in range(3):
        noise = rng.standard_normal(n)
        k = max(5, n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * max(1, k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'rand_exp{i}', a / a.sum()))
    return inits


def _monotone_init(n, power, direction='inc'):
    x = np.linspace(0, 1, n)
    if direction == 'inc':
        a = (x + 1e-10) ** power
    else:
        a = (1.0 - x + 1e-10) ** power
    return a / a.sum()


def _optimize_from(a0, deadline, label, rng):
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    total_time = max(10.0, deadline - t0 - 15.0)

    # Progressive LSE stages: (K, beta, time_fraction)
    stages = [
        (2, 20.0, 0.04),
        (3, 50.0, 0.06),
        (5, 100.0, 0.07),
        (8, 200.0, 0.08),
        (15, 400.0, 0.10),
        (30, 800.0, 0.12),
        (60, 1500.0, 0.13),
        (120, 3000.0, 0.14),
        (250, 6000.0, 0.14),
        (500, 12000.0, 0.12),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + cum_time * total_time
        stage_end = min(stage_end, deadline - 10.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        # Run L-BFGS-B
        n_calls = min(3, max(1, int(stage_budget / 15)))
        per_call = stage_budget / n_calls
        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 500)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k}, beta={beta:.0f}, c{ci}: {best_sc:.8f}")
            a = _norm(a_new)

        # Alternate with FISTA for diversity
        if time.monotonic() < stage_end - 5.0 and beta >= 400:
            fista_budget = min(5.0, (deadline - time.monotonic()) * 0.05)
            if fista_budget > 3.0:
                sc_f, a_f = _fista_opt(a, beta, k, fista_budget, max_iter=2000)
                if sc_f < best_sc:
                    best_sc = sc_f
                    best_a = a_f.copy()
                    print(f"    [{label}] FISTA K={k}: {best_sc:.8f}")
                a = _norm(a_f)

    # Subgradient polish
    if time.monotonic() < deadline - 5.0:
        sc, a_sg = _subgrad_polish(a, max_iters=25, deadline=deadline - 3.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent available: {parent is not None}, budget={budget_s}s")

    # Determine target length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192

    # Cap length for computational feasibility
    target_n = max(2048, min(target_n, 16384))
    print(f"Target n = {target_n}")

    # --- Generate diverse initializations ---
    inits = _generate_inits(target_n, rng)

    # Add parent-based inits
    if parent is not None:
        p_resampled = _upsample(parent, target_n)
        inits.append(('parent', p_resampled))
        inits.append(('parent_rev', p_resampled[::-1]))
        # Perturbed parent
        for amp in (0.05, 0.15, 0.3, 0.5):
            noise = rng.standard_normal(target_n)
            k_sm = max(5, target_n // 200)
            if k_sm % 2 == 0:
                k_sm += 1
            kern = np.exp(-((np.arange(k_sm) - k_sm // 2) ** 2) / (2 * max(1, k_sm // 4) ** 2))
            kern /= kern.sum()
            noise_s = np.convolve(noise, kern, mode='same')
            q = _norm(p_resampled + amp * noise_s)
            inits.append((f'parent_pert_{amp}', q))

    # Evaluate and rank all inits
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top 10: "
          f"{[(n, f'{s:.6f}') for s, n, _ in scored[:10]]}")

    # Select top diverse candidates (at least 4)
    candidates = []
    for sc, name, arr in scored:
        if len(candidates) >= 5:
            break
        # Check diversity: cosine similarity with already selected
        is_diverse = True
        for _, _, arr2 in candidates:
            sim = float(np.dot(arr, arr2)) / (float(np.linalg.norm(arr)) * float(np.linalg.norm(arr2)) + 1e-12)
            if sim > 0.995:
                is_diverse = False
                break
        if is_diverse:
            candidates.append((sc, name, arr))

    # If not enough diverse, fill with next best
    for sc, name, arr in scored:
        if len(candidates) >= 5:
            break
        if not any(arr is c[2] for c in candidates):
            candidates.append((sc, name, arr))

    if not candidates:
        candidates = [(2.0, 'uniform', np.ones(target_n) / target_n)]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()

    # --- Main optimization phase ---
    # Reserve time for final polish
    final_polish_time = 300.0
    main_time = deadline - t_start - final_polish_time

    print(f"\nMain optimization time: {main_time:.0f}s, candidates: {len(candidates)}")

    # Allocate time to candidates (more time for better initial scores)
    n_cands = len(candidates)
    if n_cands == 1:
        fractions = [1.0]
    elif n_cands == 2:
        fractions = [0.55, 0.45]
    elif n_cands == 3:
        fractions = [0.45, 0.35, 0.20]
    elif n_cands == 4:
        fractions = [0.35, 0.30, 0.20, 0.15]
    else:
        fractions = [0.30, 0.25, 0.20, 0.15, 0.10]

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        sub_deadline = t_start + (sum(fractions[:ci + 1]) * main_time) + 20.0
        sub_deadline = min(sub_deadline, deadline - final_polish_time)

        if time.monotonic() > sub_deadline:
            continue

        print(f"\n=== Optimization #{ci+1}: '{name}', score={sc_init:.8f}, n={len(arr_init)} ===")
        a_work = _norm(arr_init)

        # Multilevel warm start for large sequences
        n = len(a_work)
        if n > 4096 and (sub_deadline - time.monotonic()) > 120:
            for factor in [8, 4]:
                n_small = n // factor
                if n_small < 256:
                    break
                a_small = _upsample(a_work, n_small)
                sc_small = _score(a_small)
                time_per = min(20.0, (sub_deadline - time.monotonic() - 60.0) / (2 * factor))
                for k_s, beta_s in [(3, 100), (10, 500), (30, 2000)]:
                    if time.monotonic() > sub_deadline - 60.0:
                        break
                    sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, time_per, 300)
                    if sc_s < sc_small:
                        sc_small = sc_s
                        a_small = a_s
                a_up = _norm(_upsample(a_small, n))
                sc_up = _score(a_up)
                if sc_up < _score(a_work):
                    a_work = a_up
                    print(f"  Multilevel (factor {factor}): {sc_up:.8f}")

        # Main trajectory optimization
        sc_traj, seq_traj = _optimize_from(a_work, sub_deadline, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** Global best: {best_score:.8f} ***")

    # --- Final polishing phase ---
    print(f"\n=== Final polish (current best: {best_score:.8f}) ===")
    final_deadline = deadline - 20.0

    # Progressive high-beta LSE
    for k_final, beta_final in [(300, 10000), (600, 20000), (1000, 40000), (2000, 80000)]:
        if time.monotonic() > final_deadline - 30:
            break
        remaining = final_deadline - time.monotonic() - 30
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_final, k_final, remaining * 0.35, 800)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_final}, beta={beta_final}: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < final_deadline - 20:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=30, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Random restart perturbation
    if time.monotonic() < final_deadline - 60:
        for restart in range(5):
            if time.monotonic() > final_deadline - 30:
                break
            strength = 0.05 + 0.1 * restart
            p = _norm(best_seq + strength * rng.standard_normal(len(best_seq)))
            # Smooth the perturbation
            k_sm = max(3, len(p) // 500)
            if k_sm % 2 == 0:
                k_sm += 1
            kern = np.exp(-((np.arange(k_sm) - k_sm // 2) ** 2) / (2 * max(1, k_sm // 4) ** 2))
            kern /= kern.sum()
            p = np.convolve(p, kern, mode='same')
            p = _norm(p)
            chunk_budget = min(30.0, final_deadline - time.monotonic() - 30)
            if chunk_budget < 5:
                break
            sc_r, a_r = _run_lbfgs(p, 8000, 300, chunk_budget, 400)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart} (str={strength:.2f}): {best_score:.8f}")
            # Also try FISTA on the perturbed
            if time.monotonic() < final_deadline - 35:
                sc_rf, a_rf = _fista_opt(p, 8000, 300, min(10.0, final_deadline - time.monotonic() - 30), 1000)
                if sc_rf < best_score:
                    best_score = sc_rf
                    best_seq = a_rf.copy()
                    print(f"  Restart {restart} FISTA: {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=15, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
