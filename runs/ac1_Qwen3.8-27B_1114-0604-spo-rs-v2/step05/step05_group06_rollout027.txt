```python
import numpy as np

from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out)

def _cross_corr(a, w):
    n = a.size
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
```

```python
def _lse_value_grad(a, beta, k):
    n = a.size
    S = a.sum()
    c = _self_conv(a)
    d = n * c / (S*S)
    M = 2*n - 1
    k = min(k, M)
    idx = np.argpartition(d, M-k)[M-k:]
    d_act = d[idx]
    dmax = d_act.max()
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = e.sum()
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = w_act @ d_act
    grad = 4.0 * n * corr / (S*S) - 4.0 * LDS / S
    return 2.0 * L, grad
```

```python
    a = _norm(a0).astype(np.float64)
    n = a.size
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    best_a = a.copy()
    best_val = _score(a)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for t in range(steps):
        if time.monotonic() > deadline:
            break
        val, grad = _lse_value_grad(a, beta, k)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0, None)
        a_new = _norm(a_new)
        # Check if it improved
        sc = _score(a_new)
        if sc < best_val:
            best_val = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_val, best_a
```

```python
"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses progressive LSE optimization with L-BFGS-B, mirror descent (multiplicative weights)
polishing, subgradient descent, and diverse initializations.
Key improvements: adaptive beta schedule, mirror descent for efficient polishing,
more aggressive final stages, and time-aware candidate allocation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """corr[i] = sum_k w[k] * a[k - (i - (n-1))] for i=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_func(a, beta, k):
    """Compute LSE value and gradient for 2*LSE_beta(top_k(d)) where d = n*c/S^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=5000):
    """Run L-BFGS-B on LSE objective with sum penalty and non-negativity bounds."""
    a = _norm(a0).astype(np.float64)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_eval_t, 1e-4))
    mf = max(50, min(mf, maxfun_cap, 20000))
    maxiter = max(30, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_func(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _mirror_descent(a0, max_iters=2000, deadline=None):
    """Mirror descent (multiplicative weights) on the exact max-convolution objective."""
    a = _norm(a0).astype(np.float64)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    eta = 0.05
    patience = 0
    max_patience = 80
    best_ever = best_sc

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within a small fraction of max
        tol = max(1e-6, 5e-5 * dmax)
        active = d >= (dmax - tol)
        if not active.any():
            idx = np.argmax(d)
            active = np.zeros_like(d)
            active[idx] = True

        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])

        # Gradient of 2*L w.r.t. a (for the exact max, L = dmax)
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        # Mirror descent (multiplicative weights) update
        log_weight = -eta * grad
        # Clip for numerical stability
        log_weight = np.clip(log_weight, -50, 50)
        new_a = a * np.exp(log_weight)
        new_a = np.clip(new_a, 0, None)
        sc_sum = new_a.sum()
        if sc_sum < 1e-12:
            eta *= 0.5
            patience += 1
            if patience > max_patience * 2:
                break
            continue
        new_a = new_a / sc_sum

        sc = _score(new_a)
        if sc < best_sc:
            best_sc = sc
            best_a = new_a.copy()
            patience = 0
        else:
            patience += 1

        a = new_a

        if patience > max_patience:
            eta *= 0.5
            patience = 0
            if eta < 1e-7:
                break

    return best_sc, best_a


def _subgrad_polish(a0, max_iters=100, deadline=None):
    """Exact subgradient descent with adaptive step size on max-convolution objective."""
    a = _norm(a0).astype(np.float64)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        active = d >= dmax - 1e-6 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0003, 0.001, 0.004, 0.01, 0.03, 0.1, 0.3, 1.0):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.2, 1.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _polynomial_init(n, order, rng):
    """Generate a non-negative polynomial-like sequence."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for _ in range(order):
        coeff = rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = a * (1.0 + coeff * np.cos(2 * np.pi * (rng.uniform(1, 5)) * x + phase))
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def _noise_init(n, rng, smoothness=50):
    """Generate smoothed positive noise."""
    noise = rng.standard_normal(n)
    k = max(5, min(n // 2, smoothness))
    if k % 2 == 0:
        k += 1
    c = k // 2
    sig = max(1.0, c / 3.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    a = np.exp(noise * 0.8)
    return a / a.sum()


def _smooth_perturb(a, rng, strength=0.2):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_trajectory(a0, deadline, label, rng):

    a = _norm(a0).astype(np.float64)

    t0 = time.monotonic()

    # Stages: (K, beta, time_fraction, maxfun_cap)

    stages = [
        (3, 30.0, 0.05, 3000),
        (5, 60.0, 0.07, 4000),

    cum_time = 0.0
    for k, beta, frac, mf_cap in stages:

                best_a = a_new.copy()

            a = _norm(a_new)

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")

        weight_sum += w
    fractions = [w / weight_sum for w in weights]

    print(f"\nCandidates: {[(n2, f'{s:.6f}', f'{f:.2f}') for (s, n2, _), f in zip(candidates, fractions)]}")

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        sub_deadline = t_start + sum(fractions[:ci + 1]) * main_time + 20.0
        sub_deadline = min(sub_deadline, main_deadline - 10.0)

        if time.monotonic() > sub_deadline - 30.0:
            continue

        print(f"\n=== Candidate {ci + 1}/{n_cands}: '{name}', score={sc_init:.8f} ===")

        # Quick multilevel warm start for the first candidate
        a_work = _norm(arr_init)
        n = len(a_work)
        if n > 4096 and ci == 0 and time.monotonic() < sub_deadline - 90.0:
            n_small = max(512, n // 4)
            a_small = _upsample(a_work, n_small)
            sc_small = _score(a_small)
            for k_s, beta_s in [(5, 100), (20, 500)]:
                if time.monotonic() > sub_deadline - 60.0:
                    break
                sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, 20.0, 2000)
                if sc_s < sc_small:
                    sc_small = sc_s
                    a_small = a_s
            a_up = _norm(_upsample(a_small, n))
            sc_up = _score(a_up)
            if sc_up < _score(a_work):
                a_work = a_up
                print(f"  Multilevel: {sc_up:.8f}")

        # Main trajectory
        sc_traj, seq_traj = _optimize_trajectory(a_work, sub_deadline, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** Global best updated: {best_score:.8f} ***")

    # --- Final global polish ---
    print(f"\n=== Final polish (current best score={best_score:.8f}) ===")
    final_deadline = t_start + budget_s - 10.0
    final_start = time.monotonic()
    a_best = _norm(best_seq)

    # High-beta LSE stages
    final_stages = [
        (100, 10000, 0.25),
        (200, 30000, 0.25),
        (500, 80000, 0.25),
        (1000, 200000, 0.25),
    ]
    total_final = max(30.0, final_deadline - final_start - 30.0)

    for k_f, beta_f, frac_f in final_stages:
        stage_budget = frac_f * total_final
        stage_end = min(time.monotonic() + stage_budget, final_deadline - 15.0)
        if time.monotonic() > stage_end:
            continue
        sc_f, a_f = _run_lbfgs(a_best, beta_f, k_f, stage_budget - time.monotonic(), 20000)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            a_best = _norm(a_f)
            print(f"  Final LSE K={k_f}, beta={beta_f}: {best_score:.8f}")

    # Mirror descent polish (long)
    if time.monotonic() < final_deadline - 20.0:
        sc_md, a_md = _mirror_descent(a_best, max_iters=5000, deadline=final_deadline - 15.0)
        if sc_md < best_score:
            best_score = sc_md
            best_seq = a_md.copy()
            a_best = _norm(a_md)
            print(f"  Final mirror_descent: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < final_deadline - 10.0:
        sc_sg, a_sg = _subgrad_polish(a_best, max_iters=100, deadline=final_deadline - 5.0)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            a_best = _norm(a_sg)
            print(f"  Final subgrad: {best_score:.8f}")

    # One more mirror descent pass
    if time.monotonic() < final_deadline - 5.0:
        sc_md2, a_md2 = _mirror_descent(a_best, max_iters=2000, deadline=final_deadline - 2.0)
        if sc_md2 < best_score:
            best_score = sc_md2
            best_seq = a_md2.copy()
            print(f"  Final mirror2: {best_score:.8f}")

    # Random restart perturbations if time remains
    if time.monotonic() < final_deadline - 30:
        for restart in range(3):
            if time.monotonic() > final_deadline - 10:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05)
            chunk_time = min(15.0, final_deadline - 5.0 - time.monotonic())
            if chunk_time < 2:
                break
            sc_r, a_r = _mirror_descent(p, max_iters=300, deadline=time.monotonic() + chunk_time)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
```
