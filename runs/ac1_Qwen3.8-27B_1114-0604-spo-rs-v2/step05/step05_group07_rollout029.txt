"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-guided directional steps with line search, L-BFGS with LSE smoothing,
multiplicative updates, and coordinate descent peak-flattening.
Key innovation: reduced-constraint LP to find descent directions efficiently,
combined with aggressive high-beta L-BFGS refinement and peak-balancing
coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -60))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


_current_beta = 1000.0


def lse_callback(x):
    """Callback for L-BFGS."""
    return lse_score_and_grad(x, _current_beta)


def normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def run_lbfgs(a0, beta, maxiter=30000, maxfun=150000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_callback, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        S = a_out.sum()
        if S < 1e-12:
            a_out = a0 / max(a0.sum(), 1e-12)
        else:
            a_out = a_out / S
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception:
        return compute_score(a0), a0


def reduced_lp_direction(a, M, top_k=300):
    """
    Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for top-k lags, b >= 0.
    Returns the optimal b (normalized) or None on failure.
    """
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:top_k]

    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(top_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_vals = np.arange(j_start, j_end + 1)
        a_vals = a[k - j_vals]
        rows.extend([ii] * len(j_vals))
        cols.extend(j_vals.tolist())
        vals.extend(a_vals.tolist())

    A = csc_matrix((vals, (rows, cols)), shape=(top_k, n))

    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.status != 0:
            return None
        b = np.maximum(result.x, 0)
        if b.sum() < 1e-12:
            return None
        return b / b.sum()
    except Exception:
        return None


def lp_line_search(a, direction, M, n_bisect=20):
    """
    Line search: find best t in [0,1] for a_new = (1-t)*a + t*direction.
    Minimizes the actual peak of the blend.
    """
    best_t = 0.0
    best_sc = compute_score(a)

    # Try a few initial values
    candidates = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    lo = 0.0
    hi = 0.0

    for t in candidates:
        a_trial = (1 - t) * a + t * direction
        a_trial = np.clip(a_trial, 0, 1000)
        S = a_trial.sum()
        if S < 1e-12:
            continue
        a_trial = a_trial / S
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_t = t
            hi = t
            lo = max(0, t - 0.1)

    # Bisection around the best found
    for _ in range(n_bisect):
        mid = (lo + hi) / 2
        a_trial = (1 - mid) * a + mid * direction
        a_trial = np.clip(a_trial, 0, 1000)
        S = a_trial.sum()
        if S < 1e-12:
            lo = mid
            continue
        a_trial = a_trial / S
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_t = mid
            lo = mid
        else:
            hi = mid

    a_new = (1 - best_t) * a + best_t * direction
    a_new = np.clip(a_new, 0, 1000)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new, best_sc


def coordinate_descent_peak(a, n_sweeps=20, n_top=100, n_bottom=100):
    """
    Coordinate descent: iteratively reduce peak by moving mass from
    top contributors to bottom contributors at the peak lag.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        L = len(contrib)

        top_count = min(n_top, max(5, L // 4))
        bot_count = min(n_bottom, max(5, L // 4))

        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]
        top_global = j_start + top_local
        bot_global = j_start + bot_local

        improved = False
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-12:
                continue

            best_val = a_old
            best_local_sc = best_sc

            # Try several reduction fractions
            for frac in [0.3, 0.5, 0.7, 0.8, 0.9, 0.95]:
                new_val = a_old * frac
                reduction = a_old - new_val
                if reduction < 1e-15:
                    continue

                a[j] = new_val
                # Redistribute to bottom positions
                per = reduction / len(bot_global)
                for bj in bot_global:
                    a[bj] += per

                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val

            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            improved = True

        if not improved and sweep > 3:
            break

    return best_a


def peak_balancing_cd(a, n_iters=500):
    """
    Peak-balancing coordinate descent: equalize the top peaks by
    moving mass between peak-contributing positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        # Active peaks: within 1% of max
        active_mask = c >= c_max * 0.99
        active_lags = np.where(active_mask)[0]

        if len(active_lags) < 2:
            # Only one peak, use standard CD
            break

        # Find positions that contribute to the maximum peak but not others
        k_max = int(np.argmax(c))
        j_start = max(0, k_max - n + 1)
        j_end = min(k_max, n - 1)
        seg = a[j_start:j_end + 1]
        contrib_max = seg[::-1].copy()

        # For each other active peak, compute contribution
        # Find positions with high contribution to max peak but low to others
        low_contrib_elsewhere = np.ones(j_end - j_start + 1)
        for k2 in active_lags:
            if k2 == k_max:
                continue
            js2 = max(0, k2 - n + 1)
            je2 = min(k2, n - 1)
            # Map to the same index range if possible
            # This is approximate - just use the overlap
            overlap_start = max(0, js2 - j_start)
            overlap_end = min(j_end - j_start, je2 - j_start)
            if overlap_end >= overlap_start:
                contrib2 = np.zeros(j_end - j_start + 1)
                for ii in range(overlap_start, overlap_end + 1):
                    i2 = j_start + ii
                    if 0 <= k2 - i2 < n:
                        contrib2[ii] = a[i2] * a[k2 - i2]
                low_contrib_elsewhere *= (1.0 + contrib2 / (c_max / (j_end - j_start + 1) + 1e-12))

        # Priority: high contrib to max, low contrib elsewhere
        priority = contrib_max / (low_contrib_elsewhere + 1e-12)
        top_positions = np.argsort(priority)[::-1][:50]
        bot_positions = np.argsort(priority)[:50]

        # Move a small amount from top to bottom
        total_top = a[j_start + top_positions].sum()
        total_bot = a[j_start + bot_positions].sum()
        if total_top < 1e-12:
            break

        frac = 0.05
        reduction = total_top * frac
        per_top = reduction * a[j_start + top_positions] / total_top
        per_bot = reduction / len(bot_positions)

        a[j_start + top_positions] -= per_top
        a[j_start + bot_positions] += per_bot

        a = np.clip(a, 0, 1000)
        S = a.sum()
        if S > 1e-12:
            a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multiplicative_update(a, beta, n_iters=2000, eta0=0.1, decay=0.999):
    """Multiplicative (mirror descent) update with LSE smoothing."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -40, 40)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version for ILS."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 5 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        n_actual = min(n_blocks, n // block_size)
        for b in range(n_actual):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted reduction
        c = fast_conv(a)
        k_star = int(np.abspath(np.argsort(c)[::-1][:5]).mean())
        j_start = max(0, int(k_star) - n + 1)
        j_end = min(int(k_star), n - 1)
        a_p = a.copy()
        seg_len = j_end - j_start + 1
        n_reduce = max(5, seg_len // 10)
        idx = rng.choice(seg_len, size=n_reduce, replace=False)
        a_p[j_start + idx] *= (1.0 - 0.15 * rng.random(n_reduce))
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            n_add = max(5, n - n_reduce)
            add_idx = rng.choice(n, size=n_add, replace=False)
            a_p[add_idx] += deficit / n_add
    else:
        a_p = a * np.exp(scale * 3.0 * rng.standard_normal(n))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Chirp-like (quadratic phase)
    l = 1 + int(rng.integers(0, 5))
    inits.append(np.maximum(0.1, 1.0 + 0.3 * np.cos(2.0 * np.pi * l * x * x)))

    # Linear ramp (asymmetric)
    inits.append(0.1 + 0.9 * x)
    inits.append(0.1 + 0.9 * (1.0 - x))

    # V-shaped (edges higher)
    inits.append(0.1 + 0.9 * np.abs(2.0 * x - 1.0))

    # Cosine window
    inits.append(np.maximum(0.01, np.cos(np.pi * (x - 0.5))))

    # Gaussian
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))

    # Random smooth
    freqs = rng.integers(1, 20, size=5)
    amps = rng.random(5) * 0.3
    phases = rng.random(5) * 2 * np.pi
    sig = np.zeros(n)
    for f, amp, ph in zip(freqs, amps, phases):
        sig += amp * np.cos(2.0 * np.pi * f * x + ph)
    inits.append(np.maximum(0.01, 1.0 + sig))

    # Two-peak
    inits.append(0.1 + 0.9 * (np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)))

    return inits

            else:
                break

    print(f"After Phase 1: best = {best_score:.10f} (LP iters: {n_lp_iters})")

    # ================================================================
    # Phase 2: L-BFGS with increasing beta (40% time)
    # ================================================================
    phase2_deadline = t_start + 0.65 * (effective_budget - 30.0)
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    # Start with moderate beta, increase gradually
    beta_schedule = [500, 2000, 5000, 10000, 30000, 50000, 100000, 200000, 500000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        maxiter = int(min(50000, max(3000, remaining * 30)))
        maxfun = int(min(200000, max(10000, remaining * 120)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")

        # Try a perturbed restart at higher betas
        if beta >= 10000 and time.monotonic() < phase2_deadline - 60:
            a_pert = perturbed_restart(best_a, rng, scale=0.015)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 3, maxfun=maxfun // 3)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ================================================================
    # Phase 3: Coordinate descent + peak balancing (15% time)
    # ================================================================
    phase3_deadline = t_start + 0.80 * (effective_budget - 30.0)
    print(f"\n--- Phase 3: Coordinate Descent + Peak Balancing ---")

    a_cd = coordinate_descent_peak(best_a, n_sweeps=30, n_top=100, n_bottom=100)
    sc = compute_score(a_cd)
    report(sc, a_cd, "CD peak")

    a_pb = peak_balancing_cd(best_a, n_iters=300)
    sc = compute_score(a_pb)
    report(sc, a_pb, "peak balance")

    # Another CD pass on the best
    if time.monotonic() < phase3_deadline - 30:
        a_cd2 = coordinate_descent_peak(best_a, n_sweeps=20, n_top=50, n_bottom=50)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "CD peak 2")

    # L-BFGS after CD
    if time.monotonic() < phase3_deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=20000, maxfun=80000)
        report(sc, a_opt, "LBFGS post-CD")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ================================================================
    # Phase 4: Iterated local search (12% time)
    # ================================================================
    phase4_deadline = t_start + 0.92 * (effective_budget - 30.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 90:
            break

        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Quick mirror descent
        a_md = multiplicative_update(a_pert, beta=200, n_iters=1000, eta0=0.04, decay=0.999)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"ILS#{trial} md")

        # L-BFGS refinement
        beta = 5000 + int(50000 * rng.random())
        maxiter = int(min(20000, max(2000, remaining * 15)))
        maxfun = int(min(60000, max(5000, remaining * 60)))
        sc, a_opt = run_lbfgs(np.maximum(a_md, 1e-10), beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} lbfgs b={beta}")

        # Quick CD if improved
        if sc < best_score * 1.001:
            a_cd = coordinate_descent_peak(a_opt, n_sweeps=10, n_top=50, n_bottom=50)
            sc_cd = compute_score(a_cd)
            report(sc_cd, a_cd, f"ILS#{trial} cd")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ================================================================
    # Phase 5: Final high-precision polish (5% time)
    # ================================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final Polish ---")

    # Very high beta L-BFGS
    for beta in [100000, 300000, 1000000]:
        if time.monotonic() > phase5_deadline - 20:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000)
        report(sc, a_opt, f"final b={beta}")

    # Final coordinate descent
    if time.monotonic() < phase5_deadline - 30:
        a_final_cd = coordinate_descent_peak(best_a, n_sweeps=30, n_top=80, n_bottom=80)
        sc = compute_score(a_final_cd)
        report(sc, a_final_cd, "final CD")

    # One more L-BFGS
    if time.monotonic() < phase5_deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 500000, maxiter=20000, maxfun=80000)
        report(sc, a_opt, "final LBFGS")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
