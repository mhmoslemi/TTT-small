"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a combination of LSE optimization and exact 1-D coordinate descent for polishing.
Key innovation: coordinate descent with analytically-derived 1-D subproblems
provides sharper local minima than subgradient methods.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    w_full = np.zeros(M)
    w_full[idx] = w_act
    # Cross correlation
    N = _next_pow2(M)
    A_rev = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w_full, N)
    full = np.fft.irfft(A_rev * W, N)[:M]
    corr = full[n - 1:2 * n - 1]
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * (dmax + np.log(se) / beta)
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 5000))
    maxiter = max(20, mf // 3)

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += 1e-4 * (Sx - 1.0) ** 2
        g = g + 2e-4 * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _coord_descent_1d(i, a, conv, n, grid_size=50):
    """Find optimal value for coordinate i given all others fixed.
    Returns the optimal x >= 0 for a[i]."""
    x_old = a[i]
    S = float(a.sum())
    S_b = S - x_old
    if S_b < 1e-12:
        return 0.0

    # Compute w[j] = 2 * b[j-i] where b = a with b[i]=0
    # w has length 2n-1
    w = np.zeros(2 * n - 1)
    # b[k] = a[k] for k != i, b[i] = 0
    # w[j] = 2*b[j-i] for valid j-i in [0, n-1]
    # So w[i+k] = 2*b[k] = 2*a[k] if k != i, else 0
    for k in range(n):
        if k != i:
            w[i + k] = 2.0 * a[k]
    # w[i + i] = 0 (already zero since b[i]=0)

    # conv_b[j] = conv[j] - x_old * w[j] / 2 * 2 + x_old^2 * delta(j, 2i)
    # Actually: conv_b[j] = conv[j] - x_old*w[j] + x_old^2*delta(j, 2i)
    # because conv[j] = conv_b[j] + x_old*w[j] - x_old^2*delta(j,2i)
    # Wait: conv_new[j] = conv_b[j] + x*w[j] + x^2*delta(j,2i)
    # conv_old[j] = conv_b[j] + x_old*w[j] + x_old^2*delta(j,2i)
    # So conv_b[j] = conv[j] - x_old*w[j] - x_old^2*delta(j,2i)
    # Hmm wait, let me re-derive.
    # conv with a[i] = x: new_conv[j] = conv_b[j] + x*w[j] + x^2*delta(j,2i)
    # conv with a[i] = x_old: conv[j] = conv_b[j] + x_old*w[j] + x_old^2*delta(j,2i)
    # So conv_b[j] = conv[j] - x_old*w[j] - x_old^2*delta(j,2i)

    quad_pos = 2 * i
    conv_b = conv - x_old * w
    if quad_pos < 2 * n - 1:
        conv_b[quad_pos] -= x_old * x_old

    # Grid search for x that minimizes 2*n*max(conv_b + x*w + x^2*delta) / (S_b+x)^2
    x_low = 0.0
    x_high = max(0.01, 5.0 * S_b / max(n, 1))

    # Coarse grid
    grid = np.linspace(x_low, x_high, grid_size)
    best_x = x_old
    best_g = 2.0 * n * float(conv_b.max() + x_old * w.max() + x_old * x_old) / (S * S) if S > 0 else 1e10

    for x in grid:
        new_max = float(conv_b.max() + x * w.max())  # upper bound check first
        if new_max * 2.0 * n < best_g * (S_b + x) ** 2 * 0.5:
            # Might be worth evaluating
            pass
        # Full evaluation
        peak = float(np.max(conv_b + x * w))
        if quad_pos < 2 * n - 1:
            val_at_quad = conv_b[quad_pos] + x * w[quad_pos] + x * x
            peak = max(peak, val_at_quad)
        denom = (S_b + x) ** 2
        g = 2.0 * n * peak / denom
        if g < best_g:
            best_g = g
            best_x = x

    # Refine: finer grid around best_x
    margin = x_high / grid_size * 3
    x_lo = max(0.0, best_x - margin)
    x_hi = best_x + margin
    grid2 = np.linspace(x_lo, x_hi, 30)
    for x in grid2:
        peak = float(np.max(conv_b + x * w))
        if quad_pos < 2 * n - 1:
            val_at_quad = conv_b[quad_pos] + x * w[quad_pos] + x * x
            peak = max(peak, val_at_quad)
        denom = (S_b + x) ** 2
        g = 2.0 * n * peak / denom
        if g < best_g:
            best_g = g
            best_x = x

    return best_x


def _coord_descent(a, max_sweeps, deadline, verbose=False, restart_every=100):
    """Coordinate descent with exact 1-D updates. Returns (score, sequence)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_sc = _score(a)
    no_improve_count = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Compute convolution
        conv = _self_conv(a)
        S = float(a.sum())

        # Random order
        order = rng.permutation(n)

        sweep_improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = _coord_descent_1d(i, a, conv, n, grid_size=40)
            x_old = a[i]
            if abs(x_opt - x_old) > 1e-12:
                a[i] = x_opt
                # Update conv incrementally
                # new_conv[j] = old_conv[j] + (x_opt - x_old)*w[j] + (x_opt^2 - x_old^2)*delta(j,2i)
                # where w[j] = 2*a[j-i] (with a[i] = 0, i.e., using a without position i)
                # But a has already been updated at position i, so:
                # delta = x_opt - x_old
                # For j where j-i is in [0,n-1] and j-i != i:
                #   conv[j] += delta * a[j-i] (from the k=i term) + delta * a[j-i] (from the j-k=i term)
                #   Wait, this double counts. Let me just recompute.
                # Actually, the incremental update is:
                # conv_new[j] - conv_old[j] = delta*(a[j-i] + a[j+i]) - delta*a[i_old]*delta(j-i,i) - ...
                # This is error-prone. Let me just recompute conv from scratch every few coordinates.
                pass  # We'll recompute conv below if needed

            # Every 500 coordinates, recompute conv to avoid accumulated error
            if (int(np.where(order == i)[0][0]) if n > 0 else 0) % 500 == 499:
                conv = _self_conv(a)

        # Recompute conv at end of sweep
        conv = _self_conv(a)

        # Evaluate
        sc = _score(a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if verbose and (sweep % 50 == 0 or sweep == 0):
            print(f"    CD sweep {sweep}: score={sc:.8f}, best={best_sc:.8f}")

        if no_improve_count >= 20:
            # Try a perturbation
            noise = rng.standard_normal(n) * 0.01 * (best_a.max() + 1e-6)
            a = np.clip(best_a + noise, 0, None)
            a = _norm(a)
            no_improve_count = 0

    return best_sc, best_a


def _fast_coord_descent(a, deadline, verbose=False):
    """Faster coordinate descent that recomputes conv less frequently."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    rng = np.random.default_rng(54321)
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    sweeps_done = 0

    while time.monotonic() < deadline:
        sweeps_done += 1
        conv = _self_conv(a)
        order = rng.permutation(n)
        improved = False

        for idx in range(n):
            if time.monotonic() > deadline:
                break
            i = order[idx]
            x_old = a[i]
            S = float(a.sum())
            S_b = S - x_old
            if S_b < 1e-10:
                continue

            # Compute w
            w = np.zeros(2 * n - 1)
            # w[i+k] = 2*a[k] for k != i
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            w[i + np.where(mask)[0]] = 2.0 * a[mask]

            # conv_b
            quad_pos = 2 * i
            conv_b = conv - x_old * w
            if quad_pos < 2 * n - 1:
                conv_b[quad_pos] -= x_old * x_old

            # Quick 1-D optimization with fewer grid points
            x_max = max(0.01, 4.0 * S_b / n)
            grid = np.linspace(0.0, x_max, 30)
            best_x = x_old
            # Evaluate at x_old (current)
            peak_old = float(np.max(conv_b + x_old * w))
            if quad_pos < 2 * n - 1:
                peak_old = max(peak_old, conv_b[quad_pos] + x_old * w[quad_pos] + x_old * x_old)
            best_g = 2.0 * n * peak_old / (S * S) if S > 0 else 1e10

            for x in grid:
                peak = float(np.max(conv_b + x * w))
                if quad_pos < 2 * n - 1:
                    vq = conv_b[quad_pos] + x * w[quad_pos] + x * x
                    if vq > peak:
                        peak = vq
                g = 2.0 * n * peak / ((S_b + x) ** 2)
                if g < best_g:
                    best_g = g
                    best_x = x

            # Refine
            step = x_max / 30.0
            x_lo = max(0.0, best_x - 2 * step)
            x_hi = best_x + 2 * step
            grid2 = np.linspace(x_lo, x_hi, 15)
            for x in grid2:
                peak = float(np.max(conv_b + x * w))
                if quad_pos < 2 * n - 1:
                    vq = conv_b[quad_pos] + x * w[quad_pos] + x * x
                    if vq > peak:
                        peak = vq
                g = 2.0 * n * peak / ((S_b + x) ** 2)
                if g < best_g:
                    best_g = g
                    best_x = x

            if abs(best_x - x_old) > 1e-10:
                a[i] = best_x
                # Update conv
                delta = best_x - x_old
                conv += delta * w
                if quad_pos < 2 * n - 1:
                    conv[quad_pos] += (best_x * best_x - x_old * x_old)
                improved = True

        sc = _score(a)
        if sc < best_sc - 1e-11:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweeps_done % 20 == 0:
            print(f"    FCD sweep {sweeps_done}: score={sc:.8f}, best={best_sc:.8f}")

        if no_improve >= 30:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.005 * (best_a.max() + 1e-6)
            a = np.clip(best_a + noise, 0, None)
            a = _norm(a)
            no_improve = 0

    return best_sc, best_a


def _subgrad_polish(a, max_iters=20, deadline=None):
    """Subgradient descent with adaptive step size."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        active = d >= dmax - 1e-4 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        # Cross correlation
        M = 2 * n - 1
        N = _next_pow2(M)
        A_rev = np.fft.rfft(a[::-1], N)
        W = np.fft.rfft(w, N)
        full = np.fft.irfft(A_rev * W, N)[:M]
        corr = full[n - 1:2 * n - 1]

        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0002, 0.001, 0.005, 0.02, 0.1, 0.3, 0.8):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        best_sc = best_new_sc
        best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _smooth_perturb(a, rng, strength=0.1):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LSE optimization + exact coordinate descent polishing.
    Uses multi-length exploration and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent loaded: {parent is not None}, budget={budget_s}s")

    # Target lengths to try
    target_n = 8192

    # --- Generate initializations ---
    inits = []

    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))

    for i in range(8):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    if parent is not None:
        p = _upsample(parent, target_n)
        for amp in (0.03, 0.1, 0.2):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3, 5):
        for amp in (0.2, 0.5, 0.8):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'cos{f}_{amp}', a / a.sum()))

    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top scores:")
    for s, nm, _ in scored[:8]:
        print(f"  {nm}: {s:.8f}")

    # Select top candidates for optimization
    candidates = []
    for sc, name, arr in scored[:5]:
        candidates.append((sc, name, arr))

    best_score = candidates[0][0] if candidates else 2.0
    best_seq = candidates[0][2].copy() if candidates else _norm(np.ones(target_n))

    # --- Phase 1: LSE optimization on top candidates ---
    phase1_deadline = t_start + 0.35 * budget_s
    n_cands = min(3, len(candidates))
    per_cand_time = (phase1_deadline - t_start) / n_cands

    for ci in range(n_cands):
        sc_init, name, arr_init = candidates[ci]
        cand_start = t_start + ci * per_cand_time
        cand_end = min(cand_start + per_cand_time, phase1_deadline)

        if time.monotonic() > cand_end:
            continue

        n = len(arr_init)
        print(f"\n=== Phase 1: Candidate '{name}', n={n}, score={sc_init:.8f} ===")

        a_work = _norm(arr_init)

        # Progressive LSE
        stages = [
            (5, 50.0, 0.15),
            (10, 150.0, 0.20),
            (20, 400.0, 0.20),
            (50, 1000.0, 0.25),
            (100, 3000.0, 0.20),
        ]
        total_t = cand_end - cand_start
        cum = 0.0
        for k, beta, frac in stages:
            stage_end = cand_start + (cum + frac) * total_t
            stage_end = min(stage_end, cand_end - 5.0)
            stage_budget = stage_end - time.monotonic()
            if stage_budget < 3.0:
                break
            cum += frac

            # Split into sub-calls
            n_calls = min(3, max(1, int(stage_budget / 15)))
            per_call = stage_budget / n_calls
            for _ in range(n_calls):
                if time.monotonic() > stage_end - 1.0:
                    break
                sc, a_new = _run_lbfgs(a_work, beta, k, per_call, 300)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new.copy()
                    print(f"  LSE K={k}, beta={beta:.0f}: {best_score:.8f}")
                a_work = _norm(a_new)

        # Quick coordinate descent
        if time.monotonic() < cand_end:
            sc_cd, a_cd = _fast_coord_descent(a_work, cand_end - 2.0)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  CD after LSE: {best_score:.8f}")

    # --- Phase 2: Intensive coordinate descent on global best ---
    phase2_deadline = t_start + 0.70 * budget_s
    print(f"\n=== Phase 2: Intensive CD, score={best_score:.8f} ===")

    a_work = _norm(best_seq)
    sc_cd, a_cd = _fast_coord_descent(a_work, phase2_deadline - 10.0, verbose=True)
    if sc_cd < best_score:
        best_score = sc_cd
        best_seq = a_cd.copy()
        print(f"  Phase 2 CD: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < phase2_deadline - 5:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=15, deadline=phase2_deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Phase 2 subgrad: {best_score:.8f}")

    # --- Phase 3: Try coordinate-only optimization from different starts ---
    phase3_deadline = t_start + 0.85 * budget_s
    print(f"\n=== Phase 3: CD from fresh starts, score={best_score:.8f} ===")

    # Try a few different initializations with CD only
    for name, arr in candidates[1:4]:
        if time.monotonic() > phase3_deadline:
            break
        a_try = _norm(arr)
        # Small perturbation for diversity
        a_try = _smooth_perturb(a_try, rng, 0.05)
        chunk_end = min(time.monotonic() + (phase3_deadline - time.monotonic()) / max(1, 4 - candidates.index((name, arr)) if (name, arr) in candidates else 3), phase3_deadline)
        sc_try, a_try = _fast_coord_descent(a_try, chunk_end)
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print(f"  Phase 3 '{name}': {best_score:.8f}")

    # --- Phase 4: Final polishing ---
    print(f"\n=== Phase 4: Final polish, score={best_score:.8f} ===")
    final_deadline = deadline

    # More CD
    a_work = _norm(best_seq)
    sc_cd, a_cd = _fast_coord_descent(a_work, final_deadline - 60.0, verbose=True)
    if sc_cd < best_score:
        best_score = sc_cd
        best_seq = a_cd.copy()
        print(f"  Final CD: {best_score:.8f}")

    # Subgradient
    if time.monotonic() < final_deadline - 30:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=25, deadline=final_deadline - 20)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # One more CD round after subgrad
    if time.monotonic() < final_deadline - 10:
        a_work = _norm(best_seq)
        sc_cd, a_cd = _fast_coord_descent(a_work, final_deadline - 5.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  Final CD2: {best_score:.8f}")

    # Also try high-beta LSE
    if time.monotonic() < final_deadline - 30:
        for k_f, beta_f in [(200, 5000), (500, 15000)]:
            if time.monotonic() > final_deadline - 15:
                break
            rem = final_deadline - 15 - time.monotonic()
            if rem < 10:
                break
            sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, rem * 0.5, 400)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final LSE K={k_f}: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
