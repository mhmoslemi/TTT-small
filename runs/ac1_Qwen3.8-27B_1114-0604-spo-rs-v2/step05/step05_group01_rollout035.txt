import numpy as np

# --- Utility functions ---

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def grad_conv(a, w):
    """Compute grad[m] = 2 * sum_k w[k] * a[k-m] (the gradient of weighted sum of c[k])."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return 2.0 * full[n - 1:2 * n - 1]

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.full(a.size, 1.0 / max(1, a.size), dtype=np.float64)
    return a / S

def exact_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))

    m = int(a.size)
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)

# --- Optimizer ---

def adam_update(state, grad, lr):
    """Standard Adam update. Returns new state and update vector."""
    m, v, t = state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    update = m_hat / (np.sqrt(v_hat) + eps)
    return (m, v, t), lr * update

def lse_and_grad(a, beta, var_weight=0.0):
    """Compute LSE-smoothed max of convolution + variance penalty, and gradient."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    
    if cmax < 1e-15:
        return 0.0, np.zeros(n)
    
    # LSE of c (normalized)
    z = beta * (c - cmax)
    z = np.clip(z, -500, 500)  # prevent overflow
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e < 1e-300:
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w = e / sum_e
    
    L = cmax + np.log(sum_e) / beta
    
    # Gradient of LSE w.r.t. a
    grad = grad_conv(a, w)
    
    # If we have variance penalty: (1/M) * sum c[k]^2
    if var_weight > 0:
        c_norm = c / (S * S)  # normalize by sum^2
        M = 2 * n - 1
        var_penalty = var_weight * float(np.sum(c_norm * c_norm))
        # d/d a[m] of (1/M) * sum_k (c[k]/S^2)^2
        # = (2/(M*S^2)) * sum_k c[k] * dc[k]/da[m]
        # = (2/(M*S^2)) * 2 * sum_k c[k] * a[k-m]
        # = (4/(M*S^2)) * cross(c, a)[m]
        # But we need to be careful: c depends on a, and S = sum(a)
        # For simplicity, let's just use the unnormalized version:
        # var_penalty = var_weight * sum_k c[k]^2
        # d/d a[m] = 2 * var_weight * sum_k c[k] * dc[k]/da[m]
        # = 2 * var_weight * 2 * sum_k c[k] * a[k-m]
        # = 4 * var_weight * grad_conv_weighted(c, a)[m]
        # where grad_conv_weighted(c, a)[m] = sum_k c[k] * a[k-m] / 2 * 2
        # Actually grad_conv(a, c) already gives 2 * sum_k c[k] * a[k-m]
        # So d/d a[m] of var_weight * sum c[k]^2 = 2 * var_weight * grad_conv(a, c)
        
        # Wait, I need to be more careful. Let's define:
        # f(a) = var_weight * sum_k c[k]^2
        # df/da[m] = 2 * var_weight * sum_k c[k] * dc[k]/da[m]
        # = 2 * var_weight * sum_k c[k] * 2 * a[k-m]
        # = 4 * var_weight * sum_k c[k] * a[k-m]
        # = 2 * var_weight * grad_conv(a, c)  [since grad_conv(a,c) = 2*sum_k c[k]*a[k-m]]
        
        grad_var = 2.0 * var_weight * grad_conv(a, c)
        grad = grad + grad_var
        L = L + var_penalty  # Hmm, but L is the LSE of the raw c, not normalized
    
    # Normalize gradient by S^2 (since the actual objective uses c/S^2)
    # Objective: 2n * max(c) / S^2 (approximately, via LSE)
    # d/d a[m] of 2n * LSE(c) / S^2:
    # = 2n * [dLSE/da[m] * S^2 - LSE * 2*S*da_S/da[m]] / S^4
    # = 2n * [dLSE/da[m] / S^2 - 2 * LSE * (1/S) * (1/S)]  ... hmm
    
    # Actually, since the score is scale-invariant, let's just normalize a to sum=1
    # at each step and work with the unnormalized objective: 2n * max(c)
    # The gradient of 2n * LSE(c) w.r.t. a is: 2n * grad
    # But we also need to account for the constraint sum(a)=1.
    # Since we project onto the simplex after each step, we don't need the
    # Lagrange multiplier - the projection handles it.
    
    grad = 2.0 * n * grad
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    return float(L), grad

def adam_optimize(a0, n_steps, beta, var_weight=0.0, lr=0.01, deadline=None, label=""):
    """Run Adam optimization on the LSE+variance objective."""
    a = normalize_a(a0.copy())
    n = len(a)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    state = (m, v, t)
    
    best_a = a.copy()
    best_score = exact_score(a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        _, grad = lse_and_grad(a, beta, var_weight)
        
        # Adaptive learning rate: scale by norm of a
        a_norm = float(np.linalg.norm(a))
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-30:
            break
        
        # Scale learning rate
        effective_lr = lr * a_norm / (grad_norm + 1e-10)
        # Cap the step size
        effective_lr = min(effective_lr, 0.5)
        
        state, update = adam_update(state, grad, effective_lr)
        
        a_new = a - update
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_a(a_new)
        
        sc = exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if step % 500 == 0:
            sc_cur = exact_score(a)
            print("  [{}] step={}, score={:.8f}, best={:.8f}".format(label, step, sc_cur, best_score))
    
    return best_score, best_a

def subgrad_step_active(a, eta, k_top=5):
    """Subgradient step targeting the top-k active constraints."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 1e-15:
        return a
    
    # Pick top-k indices
    m = len(c)
    if k_top >= m:
        w = np.ones(m) / m
    else:
        idx = np.argpartition(c, m - k_top)[m - k_top:]
        w = np.zeros(m)
        w[idx] = 1.0 / k_top
    
    h = grad_conv(a, w)
    hmax = float(np.max(h))
    if
