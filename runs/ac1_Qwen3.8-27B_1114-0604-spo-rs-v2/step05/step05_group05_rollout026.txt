def fast_conv(a):

def fienup_optimize(a, n_iters=500, thresh_start=3.0, thresh_end=1.05, 
                    clip_start=0.3, clip_end=0.9, momentum_end=0.5):
    """Fienup/Gerchberg-Saxton with momentum and annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t
        momentum = momentum_end * t  # linearly increase momentum
        
        A = np.fft.rfft(a, M_fft)
        P = (A.real ** 2 + A.imag ** 2)
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh
        if not above.any():
            # Try to reduce the overall peak
            break
            
        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        current = np.sqrt(P[above])
        factor = 1.0 - clip + clip * (target / np.maximum(current, 1e-15))
        scale[above] = np.clip(factor, 0.01, 1.0)
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        if momentum > 0:
            v = momentum * v + (a_new - a)
            a_curr = a + v
        else:
            a_curr = a_new
        
        a_curr = np.maximum(a_curr, 0.0)
        S_new = float(a_curr.sum())
        if S_new < 1e-12:
            a = a_curr.copy()  # don't normalize, just clip
            break
        a = a_curr / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_descent(a, max_time, beta_start=50.0, beta_end=50000.0, 
                lr_start=0.01, momentum=0.9):
    """LSE-smoothed gradient descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    beta = beta_start
    lr = lr_start
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        
        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        v = momentum * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        # Adaptive beta increase
        beta = min(beta * 1.003, beta_end)
        
        # Learning rate decay
        lr *= 0.9995
        
        # Track best
        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc

def peak_shave(a, n_peaks=10, n_candidates=30, max_time=60.0):
    """Targeted peak shaving via coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # For each j in [j_start, j_end], a[j] contributes a[k-j] to c[k]
            # Try reducing the top contributors
            contribs = np.array([a[k - j] * a[j] for j in range(j_start, j_end + 1)])
            top_local = np.argsort(contribs)[::-1][:n_candidates]
            top_global = j_start + top_local
            
            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                
                # Try several reductions and additions
                for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        improved = True
                        break
                else:
                    a[j] = old_val
                    continue
                break  # Only one factor per j per sweep
        
        if not improved:
            break
    
    return best_a


def run_lbfgs_polish(a0, beta, maxiter=5000):
    """L-BFGS-B polish on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    if a0.sum() < 1e-6:
        return a0, compute_score(a0)
    
    def objective(x):
        return lse_grad(x, beta)
    
    bounds = [(0.0, None)] * n
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxiter * 10}
    )
    a_out = np.clip(res.x, 0.0, None)
    if a_out.sum() < 1e-6:
        return a0, compute_score(a0)
    return a_out, compute_score(a_out)

"""
Peak-balancing optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE gradient descent with very high beta, Fienup spectral flattening,
targeted peak shaving via coordinate descent with pairwise moves, and L-BFGS polish.
Key improvements: much higher beta values, pairwise coordinate transfers,
asymmetric perturbations, and more aggressive time allocation to final polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient via FFT.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-300:
        # Only one active peak
        idx = np.argmax(c)
        p = np.zeros(2 * n - 1)
        p[idx] = 1.0
        L = c_max
    else:
        p = exp_shifted / exp_sum
        log_sum = float(np.log(exp_sum))
        L = c_max + log_sum / beta

    # Gradient computation via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_optimize(a, n_iters=300, thresh_start=3.0, thresh_end=1.05,
                    clip_start=0.3, clip_end=0.9, momentum_end=0.5,
                    randomize=False, rng=None):
    """Fienup/Gerchberg-Saxton with momentum and threshold annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t
        momentum = momentum_end * (t ** 0.5)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        current = np.sqrt(np.maximum(P[above], 1e-30))
        factor = 1.0 - clip + clip * (target / current)
        scale[above] = np.clip(factor, 0.01, 1.0)

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        if momentum > 0.01:
            v = momentum * v + (a_new - a)
            a_curr = a + v
        else:
            a_curr = a_new

        a_curr = np.maximum(a_curr, 0.0)
        S_new = float(a_curr.sum())
        if S_new < 1e-12:
            break
        a = a_curr / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_descent(a, max_time, beta_start=100.0, beta_end=200000.0,
                lr_start=0.005, momentum=0.92, beta_growth=1.005):
    """LSE-smoothed Nesterov gradient descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    beta = beta_start
    lr = lr_start
    it = 0
    stagnant = 0

    while time.monotonic() - t_start < max_time:
        it += 1

        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        v = momentum * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        beta = min(beta * beta_growth, beta_end)
        lr *= 0.9997
        momentum = min(0.97, momentum + 0.00002)

        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 10:
                # Perturb to escape local minimum
                noise = 0.001 * (1.0 + 0.1 * stagnant)
                a = a * (1.0 + noise * np.random.standard_normal(n))
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S > 1e-12:
                    a = a / S
                v = np.zeros(n)
                stagnant = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def peak_shave(a, n_peaks=10, n_candidates=30, max_time=60.0,
               do_pairwise=True):
    """Targeted peak shaving via coordinate descent with optional pairwise moves."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    no_improve_count = 0

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Single coordinate moves
            contribs = []
            for j in range(j_start, j_end + 1):
                contribs.append(a[j] * a[k - j])
            contribs = np.array(contribs)
            top_local = np.argsort(contribs)[::-1][:n_candidates]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-13:
                    continue

                for factor in [0.3, 0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.5]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        improved = True
                        no_improve_count = 0
                        break
                else:
                    a[j] = old_val
                    continue
                break

            # Pairwise moves: transfer from contributor to non-contributor
            if do_pairwise and improved is False:
                for j1 in top_global[:min(5, len(top_global))]:
                    j1 = int(j1)
                    v1 = a[j1]
                    if v1 < 1e-12:
                        continue
                    # Find a non-contributing index to transfer to
                    for j2 in [0, n // 4, n // 2, 3 * n // 4, n - 1]:
                        if j2 == j1 or j2 < j_start or j2 > j_end:
                            for frac in [0.1, 0.3, 0.5]:
                                transfer = v1 * frac
                                a[j1] = v1 - transfer
                                a[j2] = a[j2] + transfer
                                S_tmp = float(a.sum())
                                if S_tmp < 1e-12:
                                    a[j1] = v1
                                    a[j2] = a[j2] - transfer
                                    continue
                                a_norm = a / S_tmp
                                sc = compute_score(a_norm)
                                if sc < best_sc:
                                    best_sc = sc
                                    best_a = a_norm.copy()
                                    a = a_norm
                                    improved = True
                                    no_improve_count = 0
                                    break
                            if improved:
                                break
                    if improved:
                        break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 3:
                break
            # Small random perturbation to escape
            idx = np.random.randint(n)
            a[idx] *= 1.0 + 0.05 * np.random.standard_normal()
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S > 1e-12:
                a = a / S

    return best_a


def run_lbfgs_polish(a0, beta, maxiter=3000):
    """L-BFGS-B polish on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    if a0.sum() < 1e-6:
        return a0, compute_score(a0)

    def objective(x):
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, None)] * n
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
                 'maxfun': maxiter * 15, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    if a_out.sum() < 1e-6:
        return a0, compute_score(a0)
    return a_out, compute_score(a_out)


def upsample(a, new_n):
    """Upsample a sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_diverse_init(n, rng, mode=0):
    """Build diverse initializations."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 1:
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif mode == 2:
        x = np.arange(n, dtype=np.float64) / n
        a = np.abs(np.sin(2 * np.pi * x * 1.5)) + 0.1
    elif mode == 3:
        a = np.ones(n) * 0.5
        a[:n // 4] = 0.3
        a[3 * n // 4:] = 0.3
    else:
        a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()

    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
