def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L = sum_k p_k * c_k with respect to x
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    """Top-level wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):

    exact_sc = compute_score(a_out)

def run_fista(a0, beta, n_iters=5000):
    """FISTA on LSE-smoothed objective with non-negativity and normalization."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()

    z = a0.copy()
    t_k = 1.0

    # Initial Lipschitz estimate
    _, g0 = lse_obj_grad(z, beta)
    gn = float(np.linalg.norm(g0))
    L = max(gn + 1.0, 10.0)

    best_x = a0.copy()
    best_sc = compute_score(a0)

    for it in range(n_iters):
        if it % 1000 == 0:
            sc = compute_score(x)

                best_x = x.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)

        # Gradient step with projection
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new

        # Backtracking line search
        f_new, _ = lse_obj_grad(x_new, beta)
        quadratic = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quadratic:
            L *= 1.5
            if L > 1e10:
                break
            continue
        else:
            L = max(0.8 * L, 1.0)

        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = z.sum()
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()

    return best_x


def fienup_flatten(a, n_iters=300):
    """Fienup: flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Target: blend towards flat spectrum
        alpha = 0.2 + 0.6 * progress
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        # Update amplitude, keep phase
        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=300):
    """Fienup with quantile-based thresholding."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Tightening quantile threshold
        q = 0.75 + 0.2 * (it / n_iters)
        thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - it / n_iters))

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave(a, n_iters=3000, top_k=5, step_init=0.02, decay=0.999):
    """
    Subgradient method targeting top-K peaks with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Subgradient from top peaks
        sg = np.zeros(n)
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            # Weight peaks by how close they are to the max
            weights[i] = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * (weights.sum() / len(top_idx)) / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coord_descent_peaks(a, n_sweeps=10, n_top=100):
    """
    Coordinate descent: for each top-contributing element at the peak lag,
    try multiplicative perturbations.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution of each element to the peak
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        # Focus on top contributors
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def structured_perturb(a, rng, n_modes=8, scale=0.03):
    """Add low-frequency structured perturbations to escape local minima."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = rng.integers(1, 15)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):
    """Add Gaussian noise perturbation."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def interpolated_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
