def project_simplex(v, s):
    v=np.asarray(v,dtype=np.float64)
    if s <= 0: return np.zeros_like(v)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho_candidates=np.nonzero(u * np.arange(1, len(v)+1) > (cssv - s))[0]
    if rho_candidates.size==0: return np.zeros_like(v)
    rho=rho_candidates[-1]
    theta=(cssv[rho]-s)/(rho+1.0)
    return np.maximum(v-theta,0.0)

def hermitian_sym(z):
    L=len(z)
    out=np.empty_like(z)
    out[0]=z[0].real
    mid=L//2
    if L%2==0:
        out[mid]=z[mid].real
        idx=np.arange(1,mid)
        if idx.size:
            left=0.5*(z[idx]+np.conj(z[L-idx]))
            out[idx]=left
            out[L-idx]=np.conj(left)
    else:
        idx=np.arange(1,mid+1)
        left=0.5*(z[idx]+np.conj(z[L-idx]))
        out[idx]=left
        out[L-idx]=np.conj(left)

```python
"""
Optimizer for non-negative sequences minimizing
    score(a) = 2*n*max_k (a*a)_k / (sum a)^2
using a combination of LSE-smoothed maximization, an active-set dual
weight construction, L-BFGS polishing, projected gradient polishing,
and convolution-flattening spectral steps.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _fft_len(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    """Linear convolution a*a by real FFT."""
    n = len(a)
    M = 2 * n - 1
    L = _fft_len(M)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:M]


def compute_score(a):
    """Exact score used by the evaluator."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * float(np.max(c)) / (S * S))


def normalize_to_target(a, target=None):
    """Clip to [0,1000] and scale to target sum sqrt(2n) if target is None."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([0.5])
    a = np.clip(a, 0.0, 1000.0)
    if target is None:
        target = float(np.sqrt(2.0 * len(a)))
    S = float(a.sum())
    if S < 1e-12:
        return np.full(len(a), target / len(a), dtype=np.float64)
    return np.clip(a * (target / S), 0.0, 1000.0)


def project_simplex(v, s):
    """Euclidean projection onto {x >= 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if s <= 0.0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, len(v) + 1) > (cssv - s))[0]
    if idx.size == 0:
        return np.zeros_like(v)
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hermitian_sym(z):
    """Force Hermitian symmetry for a full FFT array."""
    L = len(z)
    out = np.empty_like(z)
    out[0] = z[0].real
    mid = L // 2
    if L % 2 == 0:
        out[mid] = z[mid].real
        idx = np.arange(1, mid)
        if idx.size:
            val = 0.5 * (z[idx] + np.conj(z[L - idx]))
            out[idx] = val
            out[L - idx] = np.conj(val)
    else:
        idx = np.arange(1, mid + 1)
        if idx.size:
            val = 0.5 * (z[idx] + np.conj(z[L - idx]))
            out[idx] = val
            out[L - idx] = np.conj(val)
    return out


def grad_from_p(a, p, L=None):
    """
    Gradient of sum_k p_k (a*a)_k wrt a.
    p has length 2n-1.
    """
    n = len(a)
    M = 2 * n - 1
    if L is None:
        L = _fft_len(M)
    PA = np.fft.rfft(p, L)
    RA = np.fft.rfft(a[::-1], L)
    return 2.0 * np.fft.irfft(PA * RA, L)[n - 1:2 * n - 1]


def h_from_fft(a, P_fft, L):
    """Gradient of sum_k p_k c_k when FFT of p is already available."""
    n = len(a)
    M = 2 * n - 1
    del M
    RA = np.fft.rfft(a[::-1], L)
    return 2.0 * np.fft.irfft(P_fft * RA, L)[n - 1:2 * n - 1]


def prepare_lambda_fft(active_idx, lam, n):
    """Build FFT of a sparse active-lag weight vector."""
    if active_idx is None or len(active_idx) == 0:
        return None, _fft_len(2 * n - 1)
    M = 2 * n - 1
    L = _fft_len(M)
    p = np.zeros(M, dtype=np.float64)
    p[active_idx] = np.asarray(lam, dtype=np.float64)
    P = np.fft.rfft(p, L)
    return P, L


def autocorr_pos_lags(a, L=None):
    """
    Positive-lag autocorrelation r_d = sum_i a_i a_{i+d}, d=0..n-1.
    Used for the dual weights matrix.
    """
    n = len(a)
    M = 2 * n - 1
    if L is None:
        L = _fft_len(M)
    A = np.fft.fft(a, L)
    power = A * np.conj(A)
    cor = np.fft.ifft(power).real[:L]
    return cor[:n]


def dual_weights(a, active_idx, n_iter=200, blend_uniform=0.1):
    """
    For fixed a and active convolution lags, find nonnegative weights lambda
    (sum 1) that make the subgradient 2*(a*lambda) as flat as possible.
    This minimizes lambda^T B lambda, where B is a Gram/autocorrelation matrix.
    """
    active_idx = np.asarray(active_idx, dtype=np.int64)
    K = len(active_idx)
    if K <= 1:
        return np.ones(max(K, 1), dtype=np.float64) / max(K, 1)

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    L = _fft_len(2 * n - 1)
    r = autocorr_pos_lags(a, L)

    diff = np.abs(active_idx[:, None] - active_idx[None, :])
    B = np.zeros((K, K), dtype=np.float64)
    mask = diff <= (n - 1)
    if np.any(mask):
        B[mask] = 4.0 * r[diff[mask]]
    B = 0.5 * (B + B.T)

    if float(np.max(B)) <= 1e-300:
        return np.full(K, 1.0 / K, dtype=np.float64)

    lam = np.full(K, 1.0 / K, dtype=np.float64)
    prev = lam.copy()
    obj_old = float(lam @ B @ lam)

    for it in range(n_iter):
        g = 2.0 * (B @ lam)
        v = int(np.argmin(g))
        step = 2.0 / (it + 2.0)
        new_lam = (1.0 - step) * lam
        new_lam[v] += step
        new_lam /= new_lam.sum()
        obj_new = float(new_lam @ B @ new_lam)
        if obj_new > obj_old + 1e-14:
            # rare numerical case; keep old
            lam = new_lam
            lam /= lam.sum()
        else:
            lam = new_lam
            obj_old = obj_new
        if float(np.max(np.abs(lam - prev))) < 1e-12:
            break
        prev = lam.copy()

    lam = (1.0 - blend_uniform) * lam + (blend_uniform / K)
    lam /= lam.sum()
    return lam


def select_active(c, K, rng=None, mode=0):
    """Choose active max-near lags."""
    c = np.asarray(c, dtype=np.float64)
    K = int(min(K, len(c)))
    if K <= 0:
        return np.array([], dtype=np.int64)
    if K >= len(c):
        return np.arange(len(c), dtype=np.int64)

    order = np.argsort(c)[::-1]
    sample = order[:K].copy().astype(np.int64)

    if rng is not None and mode == 1:
        M = float(c.max())
        thresh = M - 0.05 * M
        near = np.flatnonzero(c >= thresh)
        if near.size > 2 * K:
            chosen = np.sort(rng.choice(near, size=K, replace=False)).astype(np.int64)
            keep = order[:min(16, K)].astype(np.int64)
            if keep.size:
                chosen = np.concatenate([chosen[:K - keep.size], keep])
            sample = np.sort(chosen).astype(np.int64)

    return np.sort(sample).astype(np.int64)


def conv_flatten_step(a, alpha=0.2, threshold_frac=None, threshold_quantile=None, passes=1):
    """
    Spectral projection that flattens the self-convolution.
    For a target flattened c, replace A^2 by (1-alpha)A^2 + alpha*FFT(c_tgt),
    take a square root, force Hermitian symmetry, inverse FFT, clip, normalize.
    """
    a = normalize_to_target(a)
    n = len(a)
    M = 2 * n - 1
    L = _fft_len(M)

    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(passes):
        c = fast_conv(a)
        Mc = float(c.max())

        if threshold_frac is not None:
            thr = Mc * float(threshold_frac)
        elif threshold_quantile is not None:
            thr = float(np.quantile(c, float(threshold_quantile)))
        else:
            thr = 0.95 * Mc

        c_tgt = np.minimum(c, thr)
        S2 = float(a.sum() * a.sum())
        if float(c_tgt.sum()) <= 1e-300:
            c_tgt = c.copy()
        else:
            c_tgt = c_tgt * (S2 / float(c_tgt.sum()))

        A = np.fft.fft(a, L)
        A2 = A * A
        Ctgt = np.fft.fft(c_tgt, L)

        A2_new = (1.0 - alpha) * A2 + alpha * Ctgt
        A2_new = hermitian_sym(A2_new)
        A_new = np.sqrt(A2_new)
        A_new = hermitian_sym(A_new)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-12:
            a_new = normalize_to_target(a_new)

        for gamma in (1.0, 0.75, 0.5, 0.25):
            cand = (1.0 - gamma) * a + gamma * a_new
            cand = normalize_to_target(cand)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()

        sc_work = compute_score(a)
        if best_sc < sc_work:
            a = best_a.copy()

    return best_a


def lbfgs_obj(x, beta, mix, lam, active_idx, P_lam_fft, rho, target):
    """
    Objective for L-BFGS:
       U + rho*(sum(x)-target)^2
    where U is a mixture of an LSE-smoothed maximum and a fixed active-set sum.
    """
    n = len(x)
    S = float(np.sum(x))
    if S < 1e-8:
        grad = np.maximum(target / n - x, 0.0)
        return 1e30, grad

    M = 2 * n - 1
    L = _fft_len(M)
    c = fast_conv(x)
    one = np.ones(n, dtype=np.float64)

    mix = float(mix)
    if (mix < 1.0) and (lam is None or active_idx is None or len(active_idx) == 0):
        mix = 1.0

    U = 0.0
    H = np.zeros(n, dtype=np.float64)

    if mix > 0.0:
        Mc = float(np.max(c))
        scaled = beta * (c - Mc)
        scaled = np.clip(scaled, -500.0, 0.0)
        e = np.exp(scaled)
        se = float(e.sum())
        if se <= 1e-300:
            p = np.full(M, 1.0 / M, dtype=np.float64)
            U_ent = float(np.mean(c))
        else:
            p = e / se
            U_ent = Mc + float(np.log(se)) / float(beta)
        h_ent = grad_from_p(x, p, L)
        U += mix * U_ent
        H += mix * h_ent

    if mix < 1.0:
        dw = 1.0 - mix
        U_lam = float(np.dot(lam, c[active_idx]))
        h_lam = h_from_fft(x, P_lam_fft, L)
        U += dw * U_lam
        H += dw * h_lam

    dev = S - target
    obj = U + rho * dev * dev
    grad = H + 2.0 * rho * dev * one
    grad = np.nan_to_num(grad)
    return float(obj), grad


def run_lbfgs_fixed(a0, beta, mix, lam, active_idx, P_lam_fft, maxiter=1000, maxfun=8000, rho=10.0):
    """L-BFGS-B on U + sum-penalty, with normalization after."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = len(a0)
    target = float(np.sqrt(2.0 * n))
    x0 = normalize_to_target(a0, target)

    maxiter = max(50, int(maxiter))
    maxfun = max(200, int(maxfun))

    bounds = [(0.0, 1000.0)] * n
    args = (float(beta), float(mix), lam, active_idx, P_lam_fft, float(rho), target)

    try:
        res = minimize(
            lbfgs_obj,
            x0,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1e-18,
                "gtol": 1e-12,
                "maxiter": maxiter,
                "maxfun": maxfun,
                "maxls": 50,
            },
        )
        x = res.x if hasattr(res, "x") else x0
    except Exception:
        x = x0

    x = np.asarray(x, dtype=np.float64)
    x = np.clip(x, 0.0, 1000.0)
    x = normalize_to_target(x, target)
    sc = compute_score(x)
    return sc, x


def run_projected_gradient(a0, beta, mix, lam, active_idx, P_lam_fft, target,
                           max_iters=2000, time_limit=60.0):
    """
    Projected gradient on the simplex with surrogate gradient H.
    Accepts only moves that improve the exact score; BB-style steps.
    """
    a = normalize_to_target(a0, target)
    n = len(a)
    t_start = time.monotonic()

    best_a = a.copy()
    best_sc = compute_score(a)

    step = 1.0
    a_prev = None
    g_prev = None

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        S = float(a.sum())
        if S < 1e-12:
            break
        c = fast_conv(a)
        cur_sc = float(2.0 * n * np.max(c) / (S * S))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        H = np.zeros(n, dtype=np.float64)
        if mix > 0.0:
            Mc = float(np.max(c))
            scaled = beta * (c - Mc)
            scaled = np.clip(scaled, -500.0, 0.0)
            e = np.exp(scaled)
            se = float(e.sum())
            p = np.full(len(c), 1.0 / len(c), dtype=np.float64) if se <= 1e-300 else e / se
            h_ent = grad_from_p(a, p, _fft_len(2 * n - 1))
            H += mix * h_ent
        if mix < 1.0 and lam is not None and active_idx is not None and len(active_idx) > 0:
            h_lam = h_from_fft(a, P_lam_fft, _fft_len(2 * n - 1))
            H += (1.0 - mix) * h_lam

        g = H - float(H.mean())
        gg = float(g @ g)
        if gg < 1e-26:
            break

        if a_prev is not None and g_prev is not None:
            dx = a - a_prev
            dy = g - g_prev
            num = float(dx @ dx)
            den = abs(float(dx @ dy))
            if den > 1e-14:
                step = max(1e-8, min(1e5, num / den))

        a_prev = a.copy()
        g_prev = g.copy()

        accepted = False
        accepted_mult = 1.0
        sh = step
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            trial = project_simplex(a - sh * mult * g, target)
            if float(g @ (trial - a)) < -1e-15:
                accepted = True
                accepted_mult = mult
                a = trial
                break

        if accepted:
            step = max(1e-8, step * accepted_mult)
        else:
            step *= 0.5
            if step < 1e-10:
                break

    final_sc = compute_score(best_a)
    return final_sc, best_a


def noise_perturb(a, rng, scale=0.02):
    a = np.asarray(a, dtype=np.float64)
    out = a * (1.0 + scale * rng.standard_normal(len(a)))
    return normalize_to_target(out)


def structured_perturb(a, rng, n_modes=6, scale=0.02):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    x = np.arange(n, dtype=np.float64) / max(1, n)
    pert = np.zeros(n, dtype=np.float64)
    for m in range(1, n_modes + 1):
        freq = int(rng.integers(1, 8 + 2 * m))
        ph = 2.0 * float(np.pi) * float(rng.uniform(0.0, 1.0))
        pert += (scale / float(np.sqrt(m))) * np.cos(2.0 * float(np.pi) * freq * x + ph)
    return normalize_to_target(a * (1.0 + pert))


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, int(new_n))
    out = np.interp(x_new, x_old, a)
    return normalize_to_target(out)


def explore_length(base_a, new_n, rng, time_limit=120.0):
    """Interpolate to another length and do a short local refinement."""
    t_start = time.monotonic()
    a = interpolate_sequence(base_a, int(new_n))
    best_a = a.copy()
    best_sc = compute_score(a)

    try:
        for frac, alpha in ((0.98, 0.2), (0.95, 0.3), (0.92, 0.35)):
            if time.monotonic() - t_start > time_limit * 0.4:
                break
            try:
                cand = conv_flatten_step(best_a, alpha=alpha, threshold_frac=frac, passes=1)
            except Exception:
                continue
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()

        if time.monotonic() - t_start < time_limit * 0.75:
            ncur = len(best_a)
            c = fast_conv(best_a)
            K = 256 if ncur > 5000 else 160
            active = select_active(c, K, rng, mode=0)
            lam = dual_weights(best_a, active, n_iter=60, blend_uniform=0.12)
            P, L = prepare_lambda_fft(active, lam, ncur)
            remaining = max(5.0, time_limit - (time.monotonic() - t_start))
            maxiter = max(200, int(remaining * 10))
            sc, cand = run_lbfgs_fixed(best_a, 800.0, 0.65, lam, active, P,
                                       maxiter=maxiter, maxfun=maxiter * 8, rho=10.0)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
    except Exception:
        pass

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence under the evaluator function.
    Returns the best sequence found within the requested time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 90.0

    # Load a parent construction if available.
    a = None
    try:
        if "height_sequence_1" in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                a = p
    except Exception:
        a = None

    if a is None or len(a) == 0:
        a = np.ones(8192, dtype=np.float64) * 0.5

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    target = float(np.sqrt(2.0 * len(a)))
    a = normalize_to_target(a, target)

    best_a = a.copy()
    best_score = compute_score(a)
    print(f"initial n={len(best_a)} internal_score={best_score:.8f}")

    def maybe_update(sc, cand, label):
        nonlocal best_a, best_score
        if sc < best_score:
            best_score = sc
            best_a = normalize_to_target(cand)
            el = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST {sc:.8f} n={len(best_a)} t={el:.1f}s")

    maybe_update(best_score, best_a, "initial")

    # ------------------------------------------------------------------
    # Phase 1: convolution-flattening starts
    # ------------------------------------------------------------------
    phase1_end = min(deadline, t_start + 90.0)
    flatten_specs = []
    for fr in (0.98, 0.96, 0.94, 0.92, 0.90):
        for al in (0.08, 0.18, 0.32, 0.5):
            flatten_specs.append((al, fr, None))
    for q in (0.80, 0.88, 0.94, 0.98):
        for al in (0.12, 0.25, 0.45):
            flatten_specs.append((al, None, q))

    for al, fr, q in flatten_specs:
        if time.monotonic() > phase1_end:
            break
        try:
            cand = conv_flatten_step(best_a, alpha=al, threshold_frac=fr, threshold_quantile=q, passes=1)
        except Exception:
            continue
        sc = compute_score(cand)
        maybe_update(sc, cand, f"flatten a={al} q={q}")

    # ------------------------------------------------------------------
    # Main loop: active dual weights + L-BFGS on mixed LSE/active surrogate
    # ------------------------------------------------------------------
    main_end = deadline - 820.0
    n_cyc = 0
    no_improve = 0

    # (mix_entropy, beta_unnormalized)
    schedule = [
        (1.0, 80.0),
        (1.0, 300.0),
        (0.85, 200.0),
        (0.75, 700.0),
        (0.65, 1200.0),
        (0.55, 3500.0),
        (0.75, 450.0),
        (0.60, 2500.0),
    ]

    while time.monotonic() < main_end:
        a_work = best_a.copy()
        if no_improve >= 3:
            if no_improve % 2 == 1:
                a_work = noise_perturb(best_a, rng, scale=0.015 + 0.005 * (no_improve % 4))
            else:
                a_work = structured_perturb(best_a, rng, n_modes=6, scale=0.015 + 0.005 * (no_improve % 4))
            a_work = normalize_to_target(a_work)

        ncur = len(a_work)
        c = fast_conv(a_work)

        # Active set size: most cycles 448; every few cycles smaller/larger.
        if ncur < 3000:
            K = 192
        else:
            K = 448
        if no_improve >= 5 and (n_cyc % 3 == 0):
            K = 320
        if no_improve >= 7 and (n_cyc % 3 == 1):
            K = 640
        K = min(K, 2 * ncur - 1)

        active = select_active(c, K, rng, mode=(n_cyc % 2))
        if active.size < 20:
            active = select_active(c, 120, rng, mode=0)

        if time.monotonic() > main_end - 60:
            break

        lam_iter = 80 if active.size <= 512 else 40
        lam = dual_weights(a_work, active, n_iter=lam_iter, blend_uniform=0.10)
        P_lam, L_p = prepare_lambda_fft(active, lam, ncur)

        mix, beta = schedule[n_cyc % len(schedule)]
        if active.size < 20 or P_lam is None:
            mix = 1.0

        remaining_main = max(20.0, main_end - time.monotonic())
        # Keep individual L-BFGS calls modest; the loop is the robustness mechanism.
        slice_s = min(65.0, max(25.0, remaining_main / 12.0))
        maxiter = max(300, int(slice_s * 8))
        maxfun = max(2000, maxiter * 10)

        try:
            sc, a_new = run_lbfgs_fixed(
                a_work,
                beta=beta,
                mix=mix,
                lam=lam,
                active_idx=active,
                P_lam_fft=P_lam,
                maxiter=maxiter,
                maxfun=maxfun,
                rho=10.0,
            )
        except Exception:
            sc, a_new = np.inf, a_work

        if sc < best_score:
            maybe_update(sc, a_new, f"LBFGS mix={mix:.2f} beta={beta:.0f}")
            no_improve = 0
        else:
            no_improve += 1

        # One flatten pass after L-BFGS, if the time model has not overshot.
        if time.monotonic() < main_end - 30:
            try:
                cand2 = conv_flatten_step(a_new, alpha=0.18, threshold_frac=0.985, passes=1)
                sc2 = compute_score(cand2)
                maybe_update(sc2, cand2, "flatten-post-LBFGS")
            except Exception:
                pass

        n_cyc += 1

    # ------------------------------------------------------------------
    # Optional length diversification
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 520.0:
        base_n = len(best_a)
        candidates = []
        for factor in (0.90, 1.05, 1.15, 1.30):
            nn = int(round(base_n * factor))
            if 2500 <= nn <= 20000 and nn != base_n:
                candidates.append(nn)
        if 8192 not in candidates and base_n != 8192:
            candidates.append(8192)
        if 12288 not in candidates and base_n != 12288:
            candidates.append(12288)
        candidates = sorted(set(candidates))[:2]

        for nn in candidates:
            if time.monotonic() > deadline - 520.0:
                break
            if len(best_a) > 16000:
                continue
            tl = min(160.0, deadline - time.monotonic() - 520.0)
            if tl < 60.0:
                break
            try:
                sc, cand = explore_length(best_a, nn, rng, time_limit=tl)
            except Exception:
                continue
            maybe_update(sc, cand, f"length-{int(nn)}")

    # ------------------------------------------------------------------
    # Rebuild active/dual for polishing best
    # ------------------------------------------------------------------
    def make_polish_kwargs():
        ncur = len(best_a)
        c = fast_conv(best_a)
        K = 384 if ncur < 8000 else 320
        active = select_active(c, K, rng, mode=0)
        lam = dual_weights(best_a, active, n_iter=80, blend_uniform=0.10)
        P, L = prepare_lambda_fft(active, lam, ncur)
        return active, lam, P, L

    # Projected gradient polish
    if time.monotonic() < deadline - 300.0:
        active, lam, P, L = make_polish_kwargs()
        tl = min(150.0, deadline - time.monotonic() - 300.0)
        if tl > 30.0:
            try:
                sc, cand = run_projected_gradient(
                    best_a,
                    beta=700.0,
                    mix=0.6,
                    lam=lam,
                    active_idx=active,
                    P_lam_fft=P,
                    target=float(np.sqrt(2.0 * len(best_a))),
                    max_iters=5000,
                    time_limit=tl,
                )
            except Exception:
                sc, cand = compute_score(best_a), best_a
            maybe_update(sc, cand, "PG-active")

        active, lam, P, L = make_polish_kwargs()
        tl = min(90.0, deadline - time.monotonic() - 150.0)
        if tl > 30.0:
            try:
                sc, cand = run_projected_gradient(
                    best_a,
                    beta=3000.0,
                    mix=0.35,
                    lam=lam,
                    active_idx=active,
                    P_lam_fft=P,
                    target=float(np.sqrt(2.0 * len(best_a))),
                    max_iters=3000,
                    time_limit=tl,
                )
            except Exception:
                sc, cand = compute_score(best_a), best_a
            maybe_update(sc, cand, "PG-high-beta")

    # High-beta L-BFGS polish
    if time.monotonic() < deadline - 120.0:
        active, lam, P, L = make_polish_kwargs()
        tl = min(110.0, deadline - time.monotonic() - 120.0)
        if tl > 40.0:
            maxiter = max(600, int(tl * 8))
            try:
                sc, cand = run_lbfgs_fixed(
                    best_a,
                    beta=6000.0,
                    mix=0.45,
                    lam=lam,
                    active_idx=active,
                    P_lam_fft=P,
                    maxiter=maxiter,
                    maxfun=maxiter * 8,
                    rho=10.0,
                )
            except Exception:
                sc, cand = compute_score(best_a), best_a
            maybe_update(sc, cand, "final-LBFGS-high-beta")

    if time.monotonic() < deadline - 60.0:
        try:
            cand = conv_flatten_step(best_a, alpha=0.12, threshold_frac=0.99, passes=2)
        except Exception:
            cand = best_a
        sc = compute_score(cand)
        maybe_update(sc, cand, "final-flatten")

    # Last tiny entropy-only L-BFGS, if there is any time left.
    if time.monotonic() < deadline - 20.0:
        tl = deadline - 20.0 - time.monotonic()
        if tl > 25.0:
            maxiter = max(300, int(tl * 6))
            try:
                sc, cand = run_lbfgs_fixed(
                    best_a,
                    beta=20000.0,
                    mix=1.0,
                    lam=None,
                    active_idx=None,
                    P_lam_fft=None,
                    maxiter=maxiter,
                    maxfun=maxiter * 6,
                    rho=10.0,
                )
            except Exception:
                sc, cand = compute_score(best_a), best_a
            maybe_update(sc, cand, "last-entropy-LBFGS")

    # Safety: guaranteed feasible output.
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    s_out = sum(out)
    if s_out < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("=" * 70)
    print(f"final internal={best_score:.10f}")
    print(f"final verified={verified if np.isfinite(verified) else 'inf':.10f}")
    print(f"length={len(out)} elapsed={time.monotonic()-t_start:.1f}s/{total_budget:s}")
    print("=" * 70)

    return out
```
