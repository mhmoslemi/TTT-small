import numpy as np

from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    # Numerically stable: subtract max
    z_max = np.max(z)
    e = np.exp(z - z_max)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + (z_max + np.log(sum_e)) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 3.0:
        return 32
    est = estimate_eval_time(n)
    mf = int((0.8 * t_left) / max(est, 1e-4))
    mf = max(32, min(5000, mf))
    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 32:
            cap = 32
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 500)

    maxfun = max(32, maxfun)
    maxiter = max(20, maxfun)
    bounds = [(0.0, 100.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    # Smooth the weights slightly
    w = w / float(k)
    return w


def subgrad_direction(a, k_active):
    """Compute subgradient direction for the max-convolution objective."""
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return None
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return None
    w = active_weights(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return None
    return h / hmax, w


def subgrad_step_backtrack(a, direction, max_iter=20, eta0=0.1):
    """Subgradient step with backtracking line search."""
    base_score = exact_score_a(a)
    eta = eta0
    best_a = a.copy()
    best_sc = base_score

    for _ in range(max_iter):
        q = a * np.exp(-eta * direction)
        q = normalize_a(q)
        sc = exact_score_a(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()
        if sc < base_score * (1.0 - 0.001 * eta):
            # Sufficient decrease
            return sc, best_a
        eta *= 0.5
        if eta < 1e-6:
            break

    if best_sc < base_score:
        return best_sc, best_a
    return base_score, a


def subgrad_polish(a, max_iters=50, deadline=None):
    """Polish using subgradient with backtracking, trying various k values."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    k_values = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30]

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        improved = False
        for k in k_values:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            result = subgrad_direction(a, k)
            if result is None:
                continue
            direction, _ = result
            sc, q = subgrad_step_backtrack(a, direction, max_iter=15, eta0=0.05)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best_a = q.copy()
                a = q.copy()
                improved = True

        if not improved:
            # Try smaller steps with more k values
            for k in [1, 2, 3, 5, 8, 13]:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                result = subgrad_direction(a, k)
                if result is None:
                    continue
                direction, _ = result
                sc, q = subgrad_step_backtrack(a, direction, max_iter=15, eta0=0.01)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True

        if not improved:
            break

    return float(best_sc), best_a


def component_wise_local_search(a, deadline, max_rounds=5):
    """Try to improve by adjusting small groups of elements locally."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    # Work with groups of consecutive elements
    group_size = max(2, min(8, n // 1000))
    num_groups = n // group_size + 1

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for g in range(0, num_groups, max(1, num_groups // 200)):
            if deadline is not None and time.monotonic() > float(deadline):
                break
            start = g * group_size
            end = min(start + group_size, n)
            if start >= n:
                break

            # Try scaling this group
            for factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
                q = a.copy()
                q[start:end] *= factor
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True

        if not improved:
            break

    return float(best_sc), best_a

        A = np.fft.rfft(p, Nf)
        num_bins = max(2, n // 100)
        bins_to_modify = rng.choice(len(A), size=min(num_bins, len(A)), replace=False)
        A[bins_to_modify] *= np.exp(rng.standard_normal(len(bins_to_modify)) * 0.1)
        q = np.fft.irfft(A, Nf)[:n]
    elif mode == 8:
        # Anti-correlation perturbation: reduce where autocorrelation is high
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                # Use all positions weighted by their contribution
                w = d / float(d.sum())
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * 0.5 * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    else:
        # Random block rescaling
        num_blocks = int(rng.integers(4, 16))
        block_size = n // num_blocks
        q = p.copy()
        for b in range(num_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            q[start:end] *= float(rng.uniform(0.5, 1.5))

    return normalize_a(np.clip(q, 0.0, None))


def quick_optimize(a, deadline):
    """Rapid LSE optimization with increasing beta."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    betas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for beta in betas:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            break
        mf = estimate_maxfun(rem * 0.8, n)
        if mf < 32:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    return float(best_sc), best_a


def optimize_trajectory(a_start, deadline, rng, label):
    """Main optimization trajectory with adaptive beta continuation."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    n = len(a)

    total = max(5.0, float(deadline - t0 - 5.0))
    print("[{}] start score={:.10f} n={}".format(label, best_sc, n))

    # Adaptive beta schedule: spend more time at higher betas
    stages = [
        (30.0, 0.04),
        (80.0, 0.05),
        (200.0, 0.06),
        (500.0, 0.07),
        (1000.0, 0.08),
        (2000.0, 0.09),
        (5000.0, 0.10),
        (10000.0, 0.12),
        (20000.0, 0.12),
        (50000.0, 0.10),
        (100000.0, 0.08),
        (200000.0, 0.05),
        (500000.0, 0.04),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            break
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))

        iterations = 0
        while time.monotonic() < sub_dead - 1.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 2.0:
                break
            mf = estimate_maxfun(rem * 0.85, n)
            if mf < 32:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a2.copy()
                if iterations % 5 == 0:
                    print("[{}] new best {:.10f} beta={:.0f} iter={} elapsed={:.0f}s".format(
                        label, best_sc, beta, iterations, time.monotonic() - t0))
            a = normalize_a(a2)
            iterations += 1

        # Subgradient polish between stages
        if time.monotonic() < sub_dead - 3.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=15, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad best {:.10f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Load parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = int(min(parent.size, 16384))
        if parent_len < 2048:
            parent_len = 2048
        # Round to nearest multiple of 128 for efficiency
        parent_len = (parent_len // 128) * 128
        lengths = [parent_len]
        # Try a few nearby lengths
        for L in [parent_len // 2, parent_len * 3 // 2, parent_len * 2]:
            L = int(max(2048, min(L, 16384)))
            L = (L // 128) * 128
            if L not in lengths:
                lengths.append(L)
        # Also try some standard lengths
        for L in [4096, 8192, 12288]:
            if L not in lengths and 2048 <= L <= 16384:
                lengths.append(L)
    else:
        lengths = [8192, 6144, 10240, 4096, 12288]

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 1024 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:4]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Generate and evaluate initializations
    all_scores = []
    all_seqs = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=15 if n > 8000 else 20)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)

    if not all_scores:
        n0 = lengths[0] if lengths else 8192
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.10f} (length {})".format(
        all_scores[int(order[0])], len(all_seqs[int(order[0])])))

    # Select top starts from different lengths
    starts = []
    used_lens = set()

    # Always include parent if available
    if parent is not None:
        pl = int(min(parent.size, 16384))
        pl = (pl // 128) * 128
        ps = normalize_a(resample_sequence(parent, pl))
        starts.append(ps)
        used_lens.add(len(ps))

    for idx in order:
        if len(starts) >= 3:
            break
        L = len(all_seqs[int(idx)])
        if L not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(L)

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    # Find initial best
    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.10f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.60, 0.25, 0.15]

    # Reserve time for final polish
    main_end_offset = 400.0
    main_end = deadline - main_end_offset
    main_budget = max(60.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))

        if time.monotonic() > sub_dead - 10.0:
            continue

        sq = starts[si].copy()

        # For secondary starts, blend with best
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            t_blend = 0.4 if rank == 1 else 0.6
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.10f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Intensive subgradient polish
    if time.monotonic() < deadline - 250.0:
        sc, seq = subgrad_polish(best_seq, max_iters=40, deadline=deadline - 250.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgradient polish: {:.10f}".format(best_score))

    # Phase 4: Component-wise local search
    if time.monotonic() < deadline - 200.0:
        sc, seq = component_wise_local_search(best_seq, deadline - 200.0, max_rounds=3)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Component-wise: {:.10f}".format(best_score))

    # Phase 5: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 120.0 and perturb_done < 8:
        left = float(deadline - 120.0 - time.monotonic())
        if left < 40.0:
            break
        n_remaining = 8 - perturb_done
        chunk = time.monotonic() + left / float(max(1, n_remaining))

        # Gradient-guided perturbation
        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.10f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 6: Final high-beta LSE refinement
    if time.monotonic() < deadline - 40.0:
        for final_beta in [200000.0, 500000.0, 1000000.0]:
            if time.monotonic() < deadline - 30.0:
                rem = float(deadline - time.monotonic() - 5.0)
                if rem > 8.0:
                    mf = estimate_maxfun(rem * 0.85, len(best_seq))
                    sc, seq = run_lbfgs(best_seq, final_beta, deadline, mf)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print("Final beta={:.0f}: {:.10f}".format(final_beta, best_score))

    # Phase 7: Final subgradient cleanup
    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 15.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.10f}".format(best_score))

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
