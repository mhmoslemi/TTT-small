"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: spectral error reduction, LP alternating with FISTA (tighter Lipschitz bound),
top-k subgradient descent, and L-BFGS with LSE smoothing.

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def spectral_error_reduction(a, n_iters=200, alpha_start=0.5, alpha_end=0.9):
    """Spectral error reduction with annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        frac = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]
        
        c_max = float(conv.max())
        c_mean = float(conv.mean())
        if c_max - c_mean < 1e-15:
            break
        
        thresh = c_mean + (1.0 - alpha) * (c_max - c_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv.copy()
        conv_mod[excess > 0] -= 0.5 * excess[excess > 0]
        # Also add to below-mean to preserve sum
        below = conv - c_mean
        deficit = np.maximum(-below, 0.0)
        if deficit.sum() > 0:
            conv_mod[deficit > 0] += 0.5 * excess.sum() / deficit.sum() * deficit[deficit > 0]
        
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        mask = np.abs(A) < 1e-8
        d[mask] = 0.0
        
        # Adaptive damping
        damp = 0.3 + 0.5 * (1.0 - frac)  # Start strong, decay
        
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    call_count = [0]
    t_start = time.monotonic()
    
    def obj_and_grad(x):
        call_count[0] += 1
        if time_limit is not None and time.monotonic() > time_limit:
            raise TimeoutError()
        return lse_score_and_grad(x, beta)
    
    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
    except (TimeoutError, Exception):
        # Return current best from L-BFGS partial
        # Actually we can't recover partial, so just return input
        return compute_score(a0), a0, None
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

    A = np.fft.rfft(a_pad)
    max_A_sq = float(np.max(np.abs(A))**2)
    # L = beta * max_A_sq (this is the bound on ||B^T B||_op / something)
    # Actually the Lipschitz constant of grad^2 is beta * ||B^T B||_op
    # And ||B||_op <= max|FFT(a_padded)| for circulant approximation
    # So ||B^T B||_op <= max|FFT(a_padded)|^2
    lip_base = max_A_sq
    
    for it in range(n_fista_iters):
        if it < int(n_fista_iters * ramp_frac):
            frac = it / max(1, int(n_fista_iters * ramp_frac))
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end
        
        L = beta * lip_base + 1e-12
        step = 1.0 / L
        
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        # grad = B^T p - 1
        # (B^T p)_i = sum_k p_k * a_{k-i}
        # This is cross-correlation of a and p
        rev_a = a[::-1]
        M2 = n + len(p) - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
        grad = conv_res[n - 1:2 * n - 1] - 1.0
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x

def lp_alternating_improved(a, max_outer_iters=40, fista_iters=10000, time_limit=None,
                            beta_end=300.0):
    """LP alternating with improved FISTA solver and better line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None
    
    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        x = lp_solve_fista_improved(a, M, n_fista_iters=fista_iters,
                                     beta_start=5.0, beta_end=beta_end, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Line search with Golden section
        # f(t) = score((1-t)*a + t*x), find t in [0, 1] that minimizes
        a_try_50 = (1.0 - 0.5) * a + 0.5 * x
        a_try_50 = np.maximum(a_try_50, 0.0)
        S50 = float(a_try_50.sum())
        if S50 > 1e-12:
            sc_50 = compute_score(a_try_50 / S50)
        else:
            sc_50 = np.inf
        
        a_try_25 = (1.0 - 0.25) * a + 0.25 * x
        a_try_25 = np.maximum(a_try_25, 0.0)
        S25 = float(a_try_25.sum())
        if S25 > 1e-12:
            sc_25 = compute_score(a_try_25 / S25)
        else:
            sc_25 = np.inf
        
        a_try_75 = (1.0 - 0.75) * a + 0.75 * x
        a_try_75 = np.maximum(a_try_75, 0.0)
        S75 = float(a_try_75.sum())
        if S75 > 1e-12:
            sc_75 = compute_score(a_try_75 / S75)
        else:
            sc_75 = np.inf
        
        # Pick best t from {0.25, 0.5, 0.75}
        candidates = [(0.25, sc_25), (0.5, sc_50), (0.75, sc_75)]
        candidates.sort(key=lambda x: x[1])
        best_t = candidates[0][0]
        
        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 15:
                # Diversify
                rng_local = np.random.default_rng(outer)
                a_new = a_new * (1.0 + 0.05 * rng_local.standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                no_improve_count = 0
        
        a = a_new
    
    return best_a, best_sc, x_warm

def topk_subgradient_descent(a, n_iters=300, k=50, lr_decay=0.999, rng=None):
    """Top-k subgradient descent for the score."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = 0.01
    
    for it in range(n_iters):
        c = self_conv(a)
        # Get top-k indices
        top_k_idx = np.argsort(c)[::-1][:k]
        
        # Subgradient: average of subgradients at each top-k peak
        grad = np.zeros(n)
        for k_idx in top_k_idx:
            k_val = int(k_idx)
            j_start = max(0, k_val - n + 1)
            j_end = min(k_val, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
        grad /= k
        
        # Normalize gradient for the score function
        # score = 2n * max(c) / S^2
        # d(score)/da = 2n * (grad / S^2 - 2*max(c)*a / S^3) for single peak
        # For top-k: use the average
        c_max = float(c.max())
        S2 = S * S
        S3 = S2 * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * c_max * a / S3)
        
        # Step
        a_new = a - lr * grad_score
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        lr *= lr_decay
    
    return best_a

def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Cosine power
    for p in [2, 4, 8]:
        inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x) ** p)
    
    # Gaussian
    inits.append(np.exp(-2.0 * (2.0 * x - 1.0) ** 2))
    
    # Random smooth
    noise = rng.standard_normal(n)
    # Smooth with running average
    k = max(3, n // 100)
    smoothed = np.convolve(noise, np.ones(k) / k, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smoothed, 0.01, 10.0))
    
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x) 
                + 0.05 * np.cos(8 * np.pi * x))
    
    # Triangular
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 10.0))
    
    # Plateau
    inits.append(np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.3))
    
    return inits

The Hessian of g is: `beta * B^T * diag(p) * B - beta * B^T * p * p^T * B`
The eigenvalues of the Hessian are bounded by `beta * ||B^T||^2 = beta * ||B||^2`.

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: spectral error reduction, LP alternating with FISTA (improved Lipschitz bound + restarts),
top-k subgradient descent, and L-BFGS with LSE smoothing.
Key improvements over previous: tighter step sizes, FISTA restarts, better time allocation,
spectral projection initialization, and adaptive beta scheduling.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def spectral_error_reduction(a, n_iters=200, alpha_start=0.4, alpha_end=0.9):
    """Spectral error reduction with annealing threshold."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        c_max = float(conv.max())
        c_mean = float(conv.mean())
        if c_max - c_mean < 1e-15:
            break

        thresh = c_mean + (1.0 - alpha) * (c_max - c_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv.copy()
        conv_mod[excess > 0] -= 0.6 * excess[excess > 0]

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        mask = np.abs(A) < 1e-8
        d[mask] = 0.0

        damp = 0.3 + 0.5 * (1.0 - frac)

        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def spectral_projection(a, n_iters=100, anneal_rate=0.995):
    """Spectral projection: alternate between flattening |A| and non-negativity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Target: |A(k)|^2 should make conv flat
    # For flat conv with sum=1: c[k]=1/(2n-1), C(w) = (1/(2n-1))*Dirichlet(w)
    # So target |A(w)| = sqrt(|Dirichlet(w)|/(2n-1)) where positive
    # Simpler: just try to equalize |A| for non-DC bins
    target_mag = 1.0 / np.sqrt(2.0 * n - 1.0)

    temperature = 1.0
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        A_mod = A.copy()
        for k in range(1, M // 2 + 1):
            mag = abs(A[k])
            target = target_mag
            # Blend towards target
            blend = 0.3
            new_mag = (1 - blend) * mag + blend * target
            scale = new_mag / (mag + 1e-15)
            A_mod[k] *= scale
            A_mod[M - k] *= scale
        A_mod[0] = A[0]

        a_new = np.fft.ifft(A_mod).real[:n]

        temperature *= anneal_rate
        if temperature > 0.01:
            a_new = a_new / (1.0 + np.exp(-a_new / (temperature * 0.05)))
            S_new = float(a_new.sum())
            if S_new > 1e-12:
                a_new = a_new / S_new
        else:
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000, time_limit=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t_start = time.monotonic()

    def obj_and_grad(x):
        if time_limit is not None and time.monotonic() > time_limit:
            raise TimeoutError()
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def compute_lip_bound(a):
    """Compute tighter Lipschitz bound using FFT."""
    n = len(a)
    a_pad = np.zeros(2 * n, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    return float(np.max(np.abs(A)) ** 2)


def lp_solve_fista(a, M, n_fista_iters=10000, beta_start=5.0, beta_end=300.0,
                   x_init=None, ramp_frac=0.3, restart_interval=500):
    """
    FISTA LP solver with tighter Lipschitz bound and momentum restarts.
    Solves: min -sum(x) + (1/beta)*LSE(beta*(conv(a,x)-M)) s.t. x >= 0
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    lip_base = compute_lip_bound(a)
    if lip_base < 1e-10:
        lip_base = 1.0

    best_obj = np.inf
    no_improve = 0

    for it in range(n_fista_iters):
        # Adaptive beta: ramp then hold
        if it < int(n_fista_iters * ramp_frac):
            frac = it / max(1, int(n_fista_iters * ramp_frac))
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * lip_base + 1e-12
        step = 1.0 / L

        # Compute C @ x = conv(a, x)
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # Grad = B^T p - 1 (cross-correlation of a and p, minus 1)
        rev_a = a[::-1]
        M2 = n + len(p) - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
        grad = conv_res[n - 1:2 * n - 1] - 1.0

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        # Restart check (every restart_interval iterations)
        if (it + 1) % restart_interval == 0:
            obj_val = -float(x.sum()) + (1.0 / beta) * (z_max + np.log(exp_sum) / beta)
            if obj_val < best_obj:
                best_obj = obj_val
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 3:
                # Restart momentum
                u = x.copy()
                t_mom = 1.0
                no_improve = 0

    return x


def lp_alternating(a, max_outer_iters=40, fista_iters=10000, time_limit=None,
                   beta_end=300.0, line_search=True):
    """
    LP-based alternating method with improved FISTA solver and line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None
    rng_local = np.random.default_rng(7)

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters,
                           beta_start=5.0, beta_end=beta_end, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        if line_search:
            # Try several t values
            t_candidates = [0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85]
            best_t = 0.3
            best_t_sc = best_sc
            for t in t_candidates:
                a_try = (1.0 - t) * a + t * x
                a_try = np.maximum(a_try, 0.0)
                St = float(a_try.sum())
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
            t = best_t
            a_new = (1.0 - t) * a + t * x
        else:
            t = 0.4
            a_new = (1.0 - t) * a + t * x

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 12:
                # Diversify
                a_new = a_new * (1.0 + 0.03 * rng_local.standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                S_d = float(a_new.sum())
                if S_d > 1e-12:
                    a_new = a_new / S_d
                no_improve_count = 0

        a = a_new

    return best_a, best_sc, x_warm


def topk_subgradient_descent(a, n_iters=300, k=30, lr_init=0.005, lr_decay=0.9995, rng=None):
    """Top-k subgradient descent specifically targeting the largest convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        c = self_conv(a)
        top_k_idx = np.argsort(c)[::-1][:k]

        grad = np.zeros(n)
        for k_idx in top_k_idx:
            k_val = int(k_idx)
            j_start = max(0, k_val - n + 1)
            j_end = min(k_val, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
        grad /= k

        c_max = float(c.max())
        S2 = S * S
        S3 = S2 * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * c_max * a / S3)

        a_new = a - lr * grad_score
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        lr *= lr_decay

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.3 + 0.6 * np.cos(2.0 * np.pi * x) ** 4)
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2))

    noise = rng.standard_normal(n)
    k = max(3, n // 80)
    smoothed = np.convolve(noise, np.ones(k) / k, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smoothed, 0.01, 10.0))

    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
                + 0.05 * np.cos(8 * np.pi * x))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 10.0))
    inits.append(0.3 + 0.7 * (np.cos(2.0 * np.pi * x) ** 6))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x) * np.cos(2.0 * np.pi * x))

    return inits

    # Phase 1: Spectral Methods (18% time)
    # ==================================================================
    phase1_deadline = t_start + 0.18 * (total_budget - 30.0)
    print(f"\n--- Phase 1: Spectral Methods ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")

        inits = build_initializations(n, rng, parent)
        max_inits = min(len(inits), max(3, int(remaining / 12.0)))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = np.maximum(a0, 0.0)
            S0 = float(a0.sum())
            if S0 < 1e-12:
                continue
            a0 = a0 / S0

            a_er = spectral_error_reduction(a0, n_iters=120, alpha_start=0.4, alpha_end=0.85)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

            # Additional spectral projection on the best result
            if sc < best_score + 0.03 and time.monotonic() < phase1_deadline - 15:
                a_sp = spectral_projection(a_er, n_iters=60, anneal_rate=0.995)
                sc_sp = compute_score(a_sp)
                report(sc_sp, a_sp, f"SP n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP Alternating (45% time) - MAIN PHASE
    # ==================================================================
    phase2_deadline = t_start + 0.63 * (total_budget - 30.0)
    print(f"\n--- Phase 2: LP Alternating (Main) ---")

    n_lp = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP with increasing sophistication
    stages = [
        (5000, 200, 40),
        (8000, 300, 35),
        (12000, 400, 30),
        (18000, 500, 25),
        (25000, 600, 20),
        (35000, 800, 15),
    ]

    x_warm = None
    for fi in range(len(stages)):
        if time.monotonic() > phase2_deadline - 40:
            break
        fista_it, beta_e, max_outer = stages[fi]
        remaining = phase2_deadline - time.monotonic()
        # Estimate time per outer iteration
        est_per_outer = fista_it * n_lp * 2.5e-7 + 0.05
        effective_outer = min(max_outer, max(2, int(remaining / max(1.0, est_per_outer))))

        print(f"  LP stage {fi}: n={n_lp}, fista={fista_it}, beta={beta_e}, "
              f"outer={effective_outer}, rem={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating(
            a_work,
            max_outer_iters=effective_outer,
            fista_iters=fista_it,
            time_limit=phase2_deadline - 25,
            beta_end=beta_e,
            line_search=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP s{fi}")
        a_work = a_out.copy()

    # Try other lengths if time permits
    for n in lengths:
        if n == n_lp:
            continue
        if time.monotonic() > phase2_deadline - 150:
            break
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S_t = float(a_test.sum())
        if S_t > 1e-12:
            a_test = a_test / S_t
        a_out, sc_out, _ = lp_alternating(
            a_test, max_outer_iters=12, fista_iters=10000,
            time_limit=phase2_deadline - 80, beta_end=300
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Top-k Subgradient (12% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n--- Phase 3: Top-k Subgradient ---")

    a_sub = best_a.copy()
    S = float(a_sub.sum())
    if S > 0:
        a_sub = a_sub / S

    for k_val in [5, 15, 30, 60, 100]:
        if time.monotonic() > phase3_deadline - 50:
            break
        print(f"  Top-k={k_val}...", end=" ", flush=True)
        t0 = time.monotonic()
        a_out = topk_subgradient_descent(a_sub, n_iters=400, k=k_val,
                                         lr_init=0.004, lr_decay=0.9995, rng=rng)
        sc = compute_score(a_out)
        print(f"score={sc:.8f}, t={time.monotonic()-t0:.1f}s")
        report(sc, a_out, f"TopK {k_val}")
        a_sub = a_out.copy()

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n--- Phase 4: L-BFGS Refinement ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [80, 200, 500, 1200, 3000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=4000, maxfun=60000,
                              time_limit=phase4_deadline - 30)
        print(f"score={sc:.8f}, t={time.monotonic()-t0:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final Polish (remaining)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    if time.monotonic() < deadline - 200:
        print("  Final LP round...")
        a_fin = best_a.copy()
        S = float(a_fin.sum())
        if S > 0:
            a_fin = a_fin / S
        a_out, sc_out, _ = lp_alternating(
            a_fin, max_outer_iters=8, fista_iters=20000,
            time_limit=deadline - 150, beta_end=600
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")

    if time.monotonic() < deadline - 80:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 6000, maxiter=3000, maxfun=50000,
                              time_limit=deadline - 50)
        report(sc, a_opt, "final LBFGS")

    if time.monotonic() < deadline - 30:
        a_fin = topk_subgradient_descent(best_a, n_iters=200, k=10,
                                         lr_init=0.003, rng=rng)
        sc = compute_score(a_fin)
        report(sc, a_fin, "final TopK")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
