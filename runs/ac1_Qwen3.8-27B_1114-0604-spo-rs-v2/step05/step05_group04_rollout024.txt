def fft_size(n):
    M = 2 * n - 1
    return 1 << (M - 1).bit_length()

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = fft_size(n)
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())

def top_peak_descent(a, n_iters, top_m, step_init, decay):
    """Subgradient descent targeting top-m autocorrelation peaks."""
    n = len(a)
    a = normalize_nonneg(a)
    best_a = a.copy()

    step = step_init
    M = get_fft_size(n)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        step = min(step, max((sc - 2.0 * n / (2.0 * n - 1.0)) / (gn * gn + 1e-30), 1e-12))

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())

        a = a / S
        step *= decay
    return best_a

        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        if _try_update(sc, a_opt, f"LBFGS b={beta}"):
            a_cur = normalize_nonneg(a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Peak descent + final polish (10% time)
    print(f"\n=== Phase 4: Peak Descent + Polish ===")
    for tm in [3, 5, 10, 20]:
        if time.monotonic() > deadline - 30:
            break
        a_sg = top_peak_descent(normalize_nonneg(best_a), n_iters=2000,
                                top_m=tm, step_init=0.02, decay=0.9999)
        sc = score_of(a_sg)
        _try_update(sc, a_sg, f"PEAK m={tm}")

    # Final spectral + L-BFGS combo
    if time.monotonic() < deadline - 60:
        a_fl = spectral_flatten_v2(normalize_nonneg(best_a), 150, 0.5, 0.95, top_frac=0.2)
        sc = score_of(a_fl)
        if _try_update(sc, a_fl, "FINAL-FLAT"):
            pass
        if time.monotonic() < deadline - 30:
            sc2, a_opt = run_lbfgs(a_fl, 20000, maxiter=5000, maxfun=60000)
            _try_update(sc2, a_opt, "FINAL-LBFGS")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.015 * (1 + trial * 0.5)
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            if a_p.sum() < 1e-12:
                continue
            a_p = a_p / a_p.sum()
            a_r = spectral_flatten_v2(a_p, 100, 0.5, 0.9, top_frac=0.3)
            sc = score_of(a_r)
            _try_update(sc, a_r, f"RESTART-{trial}")
            if time.monotonic() < deadline - 15:
                sc2, a_opt = run_lbfgs(a_r, 5000, maxiter=2000, maxfun=25000)
                _try_update(sc2, a_opt, f"RESTART-{trial}-LBFGS")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
