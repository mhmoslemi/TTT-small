```python
"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses power-sum approximation with Nesterov acceleration, multi-scale optimization,
and coordinate descent polishing.
"""
import numpy as np
import time

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def _cross_corr(a, w):
    """Compute cross-correlation: result[i] = sum_k w[k] * a[k-i] for i=0..n-1"""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))

def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _power_sum_obj_grad(a, P, k):
    """
    Power-sum approximation to 2*max(d) where d = n*conv(a,a)/S^2.
    Uses top-k positions with power P.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    
    # Power-mean approximation: (mean(d_act^P))^(1/P)
    # Shift to avoid numerical issues
    d_shifted = d_act / dmax  # in (0, 1]
    d_p = d_shifted ** P
    mean_p = float(d_p.mean())
    if mean_p <= 0 or not np.isfinite(mean_p):
        val = 2.0 * dmax
        # uniform weights
        w_act = np.ones(k) / k
    else:
        pm_val = dmax * mean_p ** (1.0 / P)
        val = 2.0 * pm_val
        # weights proportional to d_i^(P-1)
        w_act = d_p  # d_shifted^P is proportional to (d_i/dmax)^P
        # Actually for the gradient of (sum d_i^P)^(1/P):
        # d/d(a) [(sum d_i^P)^(1/P)] = (1/P)(sum d_i^P)^(1/P-1) * P * sum d_i^(P-1) * dd_i/da
        # = (sum d_i^P)^(1/P-1) * sum d_i^(P-1) * dd_i/da
        # weights should be d_i^(P-1) / sum d_j^(P-1)
        w_act = d_shifted ** (P - 1)
        w_sum = float(w_act.sum())
        if w_sum < 1e-15:
            w_act = np.ones(k) / k
        else:
            w_act = w_act / w_sum
        # The actual gradient factor is (sum d_i^P)^(1/P-1) * dmax^(P-1)
        # = (dmax^P * mean_p)^(1/P-1) * dmax^(P-1)
        # = dmax^(P-1+1) * mean_p^(1/P-1)
        # = dmax^P * mean_p^((1-P)/P)
        # This is a scalar that multiplies the weighted cross-correlation
        # But for the purpose of gradient direction, we can just use the weights
        # and adjust the step size accordingly.
    
    w_full = np.zeros(M)
    w_full[idx] = w_act
    
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    
    # For the power-sum, the gradient of val = 2*(sum d_i^P/M)^(1/P) w.r.t. a is:
    # 2 * (1/P) * (sum d_i^P/M)^(1/P - 1) / M * P * sum_i d_i^(P-1) * dd_i/da
    # = 2 * (sum d_i^P/M)^(1/P-1) / M * sum_i d_i^(P-1) * dd_i/da
    # 
    # dd_i/da = (2n * corr_i / S^2 - 2*d_i*LDS/S) ... wait, let me recompute.
    # 
    # d_i = n * c_i / S^2
    # dd_i/da_j = n * (dc_i/da_j) / S^2 - 2*n*c_i * S_j / S^3
    # where S_j = 1 (since S = sum a, dS/da_j = 1)
    # = n * (dc_i/da_j) / S^2 - 2*n*c_i / S^3
    # = n * (dc_i/da_j) / S^2 - 2*d_i / S
    #
    # dc_i/da_j = 2 * (cross-correlation of a with e_i at position j)
    # Actually, dc_k/da_j = a_{k-j} + a_{k-j} (when valid) = 2*a_{k-j} (when valid)
    # So sum_i w_i * dc_i/da_j = 2 * (cross_corr(a, w))_j
    #
    # Therefore: sum_i w_i * dd_i/da_j = 2n * corr_j / S^2 - 2 * sum_i w_i * d_i / S
    # = 2n * corr_j / S^2 - 2 * LDS_w / S
    # where LDS_w = sum_i w_i * d_i
    
    # For the power-sum with weights w_i = d_i^(P-1) / sum d_j^(P-1):
    # grad_val_j = 2 * (sum d^P/M)^(1/P-1) / M * sum_i d_i^(P-1) * dd_i/da_j
    # = 2 * (sum d^P/M)^(1/P-1) / M * (sum_i d_i^(P-1) * (2n*corr_i/S^2 - 2*d_i/S))
    # = 2 * scale * (2n/S^2 * sum_i d_i^(P-1) * corr_i - 2/S * sum_i d_i^P)
    # = 2 * scale * (2n/S^2 * (2 * cross_corr(a, d^(P-1))_j) - 2/S * sum d_i^P)
    #
    # Hmm, this is getting complex. Let me simplify by just using the LSE-style gradient
    # but with power-sum weights. The key insight is that the gradient direction is:
    # c_j = 2n * (cross_corr(a, w))_j / S^2 - 2 * (w @ d) / S
    # where w are the "importance weights" for each convolution position.
    
    # For power-sum: w_i prop to d_i^(P-1)
    # For LSE: w_i prop to exp(beta * d_i)
    # Both give valid subgradient directions for the max.
    
    grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    
    return float(val), grad
```

Wait, I realize I'm overcomplicating the gradient. Let me just use the same gradient formula as LSE (which is a valid subgradient of the max function) but with different weights. The key point is:

For any weights $w \geq 0$, $\sum w = 1$ supported on the "active" positions, the vector:
$$g = \sum_k w_k \nabla d_k = \frac{2n}{S^2} \text{cross\_corr}(a, w) - \frac{2}{S}(w \cdot d)$$

is a subgradient of $\max_k d_k$ (when $w$ is supported on argmax positions) or a valid descent direction for the power-sum/LSE approximation.

So the gradient formula is the same regardless of whether we use LSE or power-sum weights. The only difference is **which weights we use**:
- LSE: $w_k \propto \exp(\beta \cdot d_k)$
- Power-sum: $w_k \propto d_k^{P-1}$

For the optimization, I'll use the gradient direction with these weights, combined with Nesterov acceleration.

Let me now write a cleaner, more complete solution:

```python
"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient descent with power-sum and LSE weightings,
multi-scale optimization, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _compute_grad(a, beta=None, P=None, k=50):
    """
    Compute subgradient of 2*max(d) where d = n*conv(a,a)/S^2.
    Uses either LSE (beta) or power-sum (P) weighting.
    Returns (score, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return np.inf, np.zeros(n)
    
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    
    if dmax <= 0:
        # uniform weights
        w_full = np.ones(M) / M
    else:
        if beta is not None:
            z = beta * (d_act - dmax)
            e = np.exp(z)
            w_act = e / e.sum()
        elif P is not None:
            if P == 1:
                w_act = np.ones(k) / k
            else:
                d_shifted = d_act / dmax
                w_act = d_shifted ** (P - 1)
                ws = w_act.sum()
                if ws < 1e-30:
                    w_act = np.ones(k) / k
                else:
                    w_act = w_act / ws
        else:
            w_act = np.ones(k) / k
        
        w_full = np.zeros(M)
        w_full[idx] = w_act
    
    corr = _cross_corr(a, w_full)
    LDS = float(w_full @ d)
    
    grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    score = 2.0 * dmax
    return score, grad


def _nesterov_descent(a0, deadline, beta=None, P=None, k=50, 
                       step_scale=0.01, momentum_init=0.9):
    """
    Nesterov accelerated subgradient descent on the max-convolution objective.
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    # Initialize
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    momentum = momentum_init
    
    # Estimate initial step size
    sc, g = _compute_grad(y, beta=beta, P=P, k=k)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return best_sc, best_a
    
    # Backtracking line search helper
    def _line_search(a_cur, g, step0):
        best_a_try = a_cur.copy()
        best_sc_try = _score(a_cur)
        step = step0
        for _ in range(15):
            new_a = a_cur - step * g
            new_a = np.clip(new_a, 0, None)
            new_a = _norm(new_a)
            sc_try = _score(new_a)
            if sc_try < best_sc_try:
                best_a_try = new_a.copy()
                best_sc_try = sc_try
            else:
                step *= 0.5
            if step < 1e-12:
                break
        return best_a_try, best_sc_try, step
    
    it = 0
    plateau_count = 0
    while time.monotonic() < deadline:
        it += 1
        # Nesterov momentum update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        theta = (t - 1) / t_new  # momentum parameter
        
        sc_y, g_y = _compute_grad(y, beta=beta, P=P, k=k)
        gnorm = np.linalg.norm(g_y)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking
        step = step_scale / gnorm
        
        # Try the step
        y_new = y - step * g_y
        y_new = np.clip(y_new, 0, None)
        y_new = _norm(y_new)
        
        sc_new = _score(y_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = y_new.copy()
            plateau_count = 0
        else:
            plateau_count += 1
            # Reduce step
            for factor in (0.5, 0.25, 0.1):
                y_try = y - step * factor * g_y
                y_try = np.clip(y_try, 0, None)
                y_try = _norm(y_try)
                sc_try = _score(y_try)
                if sc_try < _score(y):
                    y_new = y_try
                    break
            
        # Update y with momentum
        y = y_new + theta * (y_new - a)
        y = np.clip(y, 0, None)
        y = _norm(y)
        a = y_new
        
        t = t_new
        momentum = min(0.999, momentum * 0.999 + 0.001)  # gradually increase
        
        # Periodic check
        if it % 100 == 0:
            print(f"    nesterov it={it}, sc={best_sc:.8f}, gnorm={gnorm:.4e}")
        
        if plateau_count > 200:
            break
    
    return best_sc, best_a


def _coord_descent(a0, deadline, max_sweeps=50):
    """
    Coordinate descent: optimize one element at a time.
    For each element a_i, the optimal value given all others is found
    by solving a quadratic subproblem.
    """
    a = _norm(a0).copy()
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    # Precompute: for each i, the contribution to max(conv) is complex,
    # so we just do a simple 1D search for each coordinate.
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try changing a[i]
            orig_val = a[i]
            if orig_val < 1e-15:
                # Try setting it to a small positive value
                a[i] = 1e-4 / n
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.0
                continue
            
            # Try scaling a[i]
            for scale in (0.5, 0.75, 1.25, 1.5, 2.0, 0.25, 0.1):
                a[i] = orig_val * scale
                a = _norm(a)
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[i] = orig_val
                a = _norm(a)
        
        if not improved:
            break
    
    return best_sc, best_a


def _lbfgs_smoothed(a0, beta, k, time_limit, maxfun=500):
    """L-BFGS-B on the LSE-smoothed objective."""
    a = _norm(a0)
    n = len(a)
    
    def obj(x):
        a_x = np.clip(x, 0, None)
        S = a_x.sum()
        if S < 1e-8:
            return 1e10, np.zeros(n)
        c = _self_conv(a_x)
        d = float(n) * c / (S * S)
        M = 2 * n - 1
        k_eff = max(1, min(k, M))
        idx = np.argpartition(d, M - k_eff)[M - k_eff:]
        d_act = d[idx]
        dmax = float(d_act.max())
        if dmax <= 0:
            return 1e10, np.zeros(n)
        
        z = beta * (d_act - dmax)
        e = np.exp(z)
        se = e.sum()
        w_act = e / se
        L = dmax + np.log(se) / beta
        
        w_full = np.zeros(M)
        w_full[idx] = w_act
        corr = _cross_corr(a_x, w_full)
        LDS = float(w_act @ d_act)
        
        grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        return 2.0 * L, grad
    
    t_end = time.monotonic() + time_limit
    mf = int(0.7 * time_limit / max(0.003, 0.003 * n / 4096))
    mf = max(20, min(mf, maxfun))
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': max(10, mf // 3), 'maxfun': mf,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        out = _norm(res.x) if hasattr(res, 'x') and res.x is not None else a
    except Exception:
        out = a
    
    return _score(out), out


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

"""
```

```python
import numpy as np

from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _raised_cosine_init(n, alpha, rng):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    a = a ** alpha
    a = np.clip(a, 1e-10, None)
    if rng is not None:
        noise = rng.standard_normal(n) * 0.01
        a = a + noise
        a = np.clip(a, 0, None)
    return a / a.sum()


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_standard if hasattr(rng, 'standard_standard') else rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _lbfgs_lse(a0, beta, k, time_limit, maxfun=500):
    a = _norm(a0)
    n = len(a)

    def obj(x):
        a_x = np.clip(x, 0, None)

        M = 2 * n - 1
        k_eff = max(1, min(k, M))

        corr = _cross_corr(a_x, w_full)

def _deflation_polish(a0, deadline, max_iters=200):

    best_a = a.copy()

        w[active] = 1.0 / n_act

        LDS = float(w @ c) / (S * S) * float(n)

        grad = 2.0 * float(n) * corr / (S * S) - 2.0 * float(n) * LDS / (float(n) * S)
        grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        best_step_a = None
        best_step_sc = best_sc

        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = best_step_a.copy()
            a = best_step_a
        else:
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = np.clip(sig - sig.min(), 0, None)
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _optimize_at_length(a_init, deadline, label, rng):
    a = _norm(a_init)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    print(f"    [{label}] n={n}, init_score={best_sc:.8f}")

    total_time = max(5.0, deadline - t0 - 10.0)

    # Stage 1: L-BFGS with increasing K
    lse_stages = [
        (5, 50, 0.05),
        (10, 100, 0.07),
        (20, 250, 0.08),
        (40, 500, 0.10),
        (80, 1000, 0.12),
        (150, 2500, 0.14),
        (300, 5000, 0.14),
        (600, 10000, 0.14),
        (1200, 20000, 0.12),
        (2500, 40000, 0.14),
    ]
    cum = 0.0
    for k_s, beta_s, frac in lse_stages:
        cum += frac
        if cum >= 0.80:
            break
        stage_end = t0 + cum * total_time
        stage_end = min(stage_end, deadline - 15.0)
        remaining = stage_end - time.monotonic()
        if remaining < 8.0:
            continue
        n_calls = min(3, max(1, int(remaining / 25.0)))
        per_call = remaining / n_calls
        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _lbfgs_lse(a, beta_s, k_s, per_call, 350)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"      [{label}] LSE K={k_s},b={beta_s}: {best_sc:.8f}")
            a = _norm(a_new)

    # Stage 2: Nesterov exact subgradient
    if time.monotonic() < deadline - 20.0:
        nesterov_deadline = min(deadline - 10.0, time.monotonic() + total_time * 0.10)
        for k_ns in (20, 50, 100):
            if time.monotonic() > nesterov_deadline:
                break
            sc_ng, a_ng = _nesterov_exact(a, nesterov_deadline, k=k_ns, step_scale=0.03)
            if sc_ng < best_sc:
                best_sc = sc_ng
                best_a = a_ng.copy()
                print(f"      [{label}] Nesterov k={k_ns}: {best_sc:.8f}")
            a = _norm(a_ng)

    # Stage 3: Deflation polish
    if time.monotonic() < deadline - 10.0:
        sc_def, a_def = _deflation_polish(best_a, deadline - 5.0, max_iters=100)
        if sc_def < best_sc:
            best_sc = sc_def
            best_a = a_def.copy()
            print(f"      [{label}] Deflation: {best_sc:.8f}")

    # Final L-BFGS push
    if time.monotonic() < deadline - 5.0:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            sc_fin, a_fin = _lbfgs_lse(best_a, 20000, min(5000, 2 * n - 1), remaining, 400)
            if sc_fin < best_sc:
                best_sc = sc_fin
                best_a = a_fin.copy()
                print(f"      [{label}] Final LSE: {best_sc:.8f}")

    print(f"    [{label}] done: {best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    if parent is not None:
        parent_n = len(parent)
    else:
        parent_n = 8192

    print(f"Parent length: {parent_n}, budget: {budget_s}s")

    # Determine lengths to try
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(parent_n)
        for d in (512, -512, 1024, -1024):
            nl = parent_n + d
            if 2048 <= nl <= 20000:
                lengths_to_try.append(nl)
    else:
        lengths_to_try = [4096, 6144, 8192]

    # Deduplicate and sort
    lengths_to_try = sorted(set(lengths_to_try))
    # Prioritize parent length first
    if parent_n in lengths_to_try:
        lengths_to_try.remove(parent_n)
        lengths_to_try.insert(0, parent_n)

    print(f"Lengths to try: {lengths_to_try}")

    best_score = np.inf
    best_seq = None
    best_len = 0

    # Phase 1: Multi-length exploration (60% of time)
    phase1_deadline = t_start + (deadline - t_start) * 0.55
    n_lens = len(lengths_to_try)
    # Allocate more time to smaller lengths (faster per-iteration)
    weights = []
    for ln in lengths_to_try:
        w = 1.0 / max(1.0, ln / 4096.0)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    cum_w = 0.0
    for li, (target_n, wt) in enumerate(zip(lengths_to_try, weights)):
        seg_start = t_start + cum_w * (phase1_deadline - t_start - 30)
        seg_end = t_start + (cum_w + wt) * (phase1_deadline - t_start - 30)
        cum_w += wt
        seg_end = min(seg_end, phase1_deadline - 5)

        if time.monotonic() > seg_end:
            continue

        # Generate initializations for this length
        inits = []
        if parent is not None:
            p_up = _upsample(parent, target_n)
            inits.append(('parent', p_up))
            inats = _upsample(parent[::-1], target_n)
            inits.append(('parent_rev', inats))
            for amp in (0.08, 0.2):
                inits.append((f'pnoise{amp}', _smooth_perturb(p_up, rng, amp)))

        for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
            inits.append((f'rc{alpha}', _raised_cosine_init(target_n, alpha, rng)))

        for i in range(2):
            inits.append((f'spec{i}', _spectral_init(target_n, rng)))

        # Popular init
        x = np.linspace(0, 1, target_n)
        for f in (1, 2):
            for amp in (0.15, 0.3, 0.5):
                arr = 1.0 + amp * np.cos(2 * np.pi * f * x)
                arr = np.clip(arr, 0, None)
                if arr.sum() > 0:
                    inits.append((f'cos{f}a{amp}', arr / arr.sum()))

        # Score and rank
        scored = []
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr))
        scored.sort(key=lambda x: x[0])

        top_k = min(3, len(scored))
        for ci in range(top_k):
            sc_init, name, arr_init = scored[ci]
            sub_deadline = min(seg_end - 2, time.monotonic() + (seg_end - seg_start) / top_k - 1)
            if time.monotonic() > sub_deadline:
                break

            print(f"  [L{target_n}/{name}] score={sc_init:.8f}")
            sc_opt, seq_opt = _optimize_at_length(arr_init, sub_deadline, f"{target_n}_{name}", rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = seq_opt.copy()
                best_len = target_n
                print(f"  *** New best: {best_score:.8f} (len={target_n}) ***")

    print(f"\nPhase 1 done. Best: {best_score:.8f} (len={best_len})")

    # Phase 2: Focused optimization on global best (30% of time)
    if best_seq is not None:
        phase2_start = max(time.monotonic(), phase1_deadline)
        phase2_deadline = t_start + (deadline - t_start) * 0.85

        print(f"\n=== Phase 2: Focused (deadline={phase2_deadline - t_start:.0f}s) ===")
        a_best = _norm(best_seq)

        # Multi-scale: downsample, optimize, upsample
        n_best = len(a_best)
        if n_best > 4096 and phase2_deadline - phase2_start > 60:
            for div in (4, 2):
                n_small = max(256, n_best // div)
                a_small = _upsample(a_best, n_small)
                sc_small = _score(a_small)
                small_deadline = phase2_start + (phase2_deadline - phase2_start) * 0.15
                sc_s, a_s = _optimize_at_length(a_small, small_deadline, f"small{div}", rng)
                a_up = _norm(_upsample(a_s, n_best))
                sc_up = _score(a_up)
                if sc_up < best_score:
                    best_score = sc_up
                    best_seq = a_up.copy()
                    a_best = a_up.copy()
                    print(f"  Multi-scale (div={div}): {best_score:.8f}")

        # Main optimization trajectory
        if time.monotonic() < phase2_deadline - 20:
            sc_p2, seq_p2 = _optimize_at_length(a_best, phase2_deadline - 5, "focused", rng)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_seq = seq_p2.copy()
                print(f"  Phase 2 focused: {best_score:.8f}")

    # Phase 3: Perturbation restarts (15% of time)
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Phase 3: Perturbations ===")
        phase3_deadline = deadline - 10
        n_restarts = 6
        per_restart = (phase3_deadline - time.monotonic()) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > phase3_deadline - 5:
                break
            strength = 0.03 + 0.05 * ri
            p = _smooth_perturb(best_seq, rng, strength)
            r_deadline = time.monotonic() + max(10, per_restart - 5)
            r_deadline = min(r_deadline, phase3_deadline - 5)

            sc_r, a_r = _lbfgs_lse(p, 5000, 300, max(3, r_deadline - time.monotonic() - 10), 250)
            if sc_r < best_score and time.monotonic() < phase3_deadline - 15:
                sc_r2, a_r2 = _deflation_polish(a_r, phase3_deadline - 5, max_iters=50)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_seq = a_r2.copy()
                    print(f"  Restart {ri}: {best_score:.8f}")
                elif sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  Restart {ri} (LSE): {best_score:.8f}")

    # Final deflation
    if best_seq is not None and time.monotonic() < deadline - 5:
        sc_fin, a_fin = _deflation_polish(best_seq, deadline - 2, max_iters=50)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
```
