def next_pow2(m):
    return 1 << (int(m) - 1).bit_length()

def self_conv(a):
    n = len(a)
    N = next_pow2(2*n-1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A*A, N)[:2*n-1]

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    if beta <= 1e-6:
        beta = 100.0

    conv_a = self_conv(a)
    d = (float(n) * conv_a) / (S * S)
    dmax = float(np.max(d))
    if dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0:
        return 1e10, np.zeros(n)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    # h[k] = sum_j w[j] * a[j-k] (cross-correlation)
    n = int(a.size)
    m = len(w)
    N = next_pow2(n + m - 1)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:n + m - 1]
    h = full[n - 1:2 * n - 1]

    obj_val = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(obj_val), grad

    def fun(a_in):
        return lse_score_grad(a_in, beta)

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return score(a), a

    a_out = normalize(res.x) if res.x is not None else a
    sc = score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = score(a)
    return float(sc), a_out


def subgrad_step(a, eta, k_active=3):
    n = len(a)
    conv_a = self_conv(a)
    dmax = float(np.max(conv_a))
    if dmax < 1e-15:
        return a
    K = min(k_active, 2 * n - 1)
    top = np.argpartition(conv_a, -K)[-K:]
    w = np.zeros(2 * n - 1)
    w[top] = 1.0 / K
    h = cross_conv(a, w)
    h = h[:n]
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def subgrad_polish(a, max_iters=50, deadline=None):
    a = normalize(a)
    best_a = a.copy()
    best_sc = score(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        cur_best = None
        cur_sc = best_sc

        for k in [1, 2, 3, 5, 8, 13]:
            for mult in [0.01, 0.03, 0.08, 0.15, 0.30, 0.60]:
                if deadline is not None and time.monotonic() > deadline:
                    break
                q = subgrad_step(a, mult, k)
                q = normalize(q)
                sc = score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=25):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        p = normalize(arr)
        inits.append(p)

    if parent is not None and len(parent) > 0:
        pr = resample_seq(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Uniform
    add(np.ones(n))

    # V-shapes
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Power V
    for p in [2, 4, 6]:
        add(0.1 + np.abs(2.0 * x - 1.0) ** p)

    # Sinusoidal
    for f in [1, 2, 3, 4]:
        for alpha in [0.3, 0.6, 1.0, 1.5]:
            add(1.0 + alpha * np.sin(np.pi * x) ** 2)
            add(1.0 - alpha * 0.5 * np.cos(2 * np.pi * f * x))

    # Gaussian bumps (bimodal)
    for width in [0.08, 0.12, 0.2, 0.3]:
        for c in [0.25, 0.3, 0.35]:
            add(np.exp(-((x - c) / width) ** 2) + np.exp(-((x - (1 - c)) / width) ** 2))

    # Cosine-based
    for f in [1, 2, 3]:
        add(0.5 + 0.5 * np.cos(np.pi * f * x) ** 2)
        add(1.0 + 0.5 * np.cos(2 * np.pi * f * x))

    # Chirp-like (non-negative)
    for rate in [1.0, 2.0, 5.0, 10.0]:
        phase = np.pi * rate * x ** 2
        add(0.5 + 0.5 * np.cos(phase))

    # Exponential tapers
    for r in [1.0, 2.0, 3.0, 5.0]:
        add(np.exp(-r * x) + np.exp(-r * (1 - x)) + 0.05)
        add(np.exp(-r * (x - 0.5) ** 2 * 4))

    # Random smooth
    Nf = next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.2)) * noise)

    # Random trigonometric
    for _ in range(6):
        vals = np.ones(n)
        K = int(rng.integers(2, 8))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.1, 0.6))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns
    for nb in [4, 8, 16, 32, 64]:
        if nb > n:
            nb = max(2, n // 2)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_seq(a, rng, strength=None):
    p = normalize(a)
    n = len(p)
    if strength is None:
        strength = float(rng.uniform(0.2, 0.7))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        q = p[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d = self_conv(p)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            K = int(rng.integers(1, 8))
            w = np.zeros(2 * n - 1)
            top = np.argpartition(d, -K)[-K:]
            w[top] = 1.0 / K
            h = cross_conv(p, w)[:n]
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * h / hmax)
                return normalize(np.clip(q, 0, None))
        q = p.copy()
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp = logp + strength * 0.4 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        x = np.linspace(0, 1, n)
        v = 1.0 + strength * np.abs(2 * x - 1)
        q = p * v
    elif mode == 6:
        k = int(rng.integers(7, max(9, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Shift mass from high-conv to low-conv
        d = self_conv(p)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            w = np.zeros(2 * n - 1)
            top = np.argpartition(d, -3)[-3:]
            w[top] = 1.0 / 3
            h = cross_conv(p, w)[:n]
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * 0.5 * h / hmax)
                return normalize(np.clip(q, 0, None))
        q = p.copy()
    elif mode == 8:
        # Sinusoidal modulation
        f = float(rng.uniform(0.5, 5.0))
        ph = float(rng.uniform(0, 2 * np.pi))
        q = p * (1.0 + strength * 0.3 * np.sin(2 * np.pi * f * np.linspace(0, 1, n) + ph))
    else:
        # Random coordinate perturbation
        q = p.copy()
        idx = rng.choice(n, size=n // 4, replace=False)
        q[idx] *= np.exp(strength * 0.5 * rng.standard_normal(len(idx)))

    return normalize(np.clip(q, 0.0, None))


def quick_optimize(a, deadline):
    a = normalize(a)
    best_a = a.copy()
    best_sc = score(a)

    for beta in [200, 600, 2000, 6000]:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        mf = max(50, int(rem * 0.8 / 0.05))
        mf = min(mf, 2000)
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize(a2)

    rem = deadline - time.monotonic()
    if rem > 15.0:
        mf = max(50, int(rem * 0.8 / 0.05))
        sc, a2 = run_lbfgs(a, 20000.0, deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    return float(best_sc), best_a


def optimize_trajectory(a_start, deadline, rng, label="opt"):
    a = normalize(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    total = max(10.0, deadline - t0 - 10.0)

    print("[{}] n={} initial score={:.10f}".format(label, n, best_sc))

    # Phase A: FW iterations (majority of time)
    fw_deadline = t0 + 0.70 * total
    sc_fw, a_fw = fw_optimize(a, fw_deadline, max_iters=1000, verbose=True)
    if np.isfinite(sc_fw) and sc_fw < best_sc:
        best_sc = sc_fw
        best_a = a_fw.copy()
    a = normalize(a_fw)

    # Phase B: LSE refinement
    lse_deadline = t0 + 0.88 * total
    rem = lse_deadline - time.monotonic()
    if rem > 10.0:
        for beta in [1000, 5000, 20000, 80000]:
            rem = lse_deadline - time.monotonic()
            if rem < 8.0:
                break
            mf = max(50, min(3000, int(rem * 0.8 / 0.05)))
            sc, a2 = run_lbfgs(a, float(beta), lse_deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] LSE beta={}: {:.10f}".format(label, beta, best_sc))
            a = normalize(a2)

    # Phase C: Subgradient polish
    sg_deadline = t0 + 0.96 * total
    rem = sg_deadline - time.monotonic()
    if rem > 5.0:
        sc_sg, a_sg = subgrad_polish(a, max_iters=30, deadline=sg_deadline)
        if np.isfinite(sc_sg) and sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()
            print("[{}] subgrad: {:.10f}".format(label, best_sc))
        a = normalize(a_sg)

    # Phase D: Final FW + LSE
    rem = deadline - time.monotonic()
    if rem > 15.0:
        sc2, a2 = fw_optimize(a, deadline - 10.0, max_iters=50, verbose=False)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
        rem2 = deadline - time.monotonic()
        if rem2 > 10.0:
            mf = max(30, int(rem2 * 0.8 / 0.05))
            sc3, a3 = run_lbfgs(a2, 100000.0, deadline - 5.0, mf)
            if np.isfinite(sc3) and sc3 < best_sc:
                best_sc = sc3
                best_a = a3.copy()

    print("[{}] final score={:.10f}".format(label, best_sc))
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("START: parent={} (len={}), HAS_GROBI={}".format(
        parent is not None, parent.size if parent is not None else 0, HAS_GROBI))

    # Determine lengths
    lengths = []
    parent_len = 0
    if parent is not None:
        parent_len = int(min(parent.size, 16384))
        lengths.append(parent_len)
        for L in [8192, 10240, 12288, 16384]:
            if L != parent_len and L not in lengths:
                lengths.append(L)
    else:
        lengths = [8192, 10240, 12288, 6144]

    # Filter to valid range
    lengths = [L for L in lengths if 2048 <= L <= 20000]
    if not lengths:
        lengths = [8192]
    lengths = lengths[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate and evaluate inits (fast, ~60s)
    init_deadline = t_start + 80.0
    all_cands = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=18 if n > 10000 else 25)
        for arr in inits:
            if time.monotonic() > init_deadline:
                break
            p = normalize(arr)
            sc = score(p)
            if np.isfinite(sc):
                all_cands.append((sc, p, n))

    if not all_cands:
        n0 = lengths[0]
        all_cands.append((2.0, np.ones(n0) / n0, n0))

    all_cands.sort(key=lambda x: x[0])
    print("Best init: {:.10f} (n={})".format(all_cands[0][0], all_cands[0][2]))

    # Select top starts (max 3, prefer diversity in length)
    starts = []
    used_lens = set()

    if parent is not None and parent_len > 0:
        ps = normalize(resample_seq(parent, parent_len))
        starts.append(ps)
        used_lens.add(parent_len)

    for sc, arr, n in all_cands:
        if len(starts) >= 3:
            break
        if n not in used_lens:
            starts.append(arr.copy())
            used_lens.add(n)

    while len(starts) < 2:
        starts.append(perturb_seq(starts[0], rng))

    # Evaluate starts
    best_start_sc = min(score(s) for s in starts)
    best_start = min(starts, key=score)
    print("Best start: {:.10f} (n={})".format(best_start_sc, len(best_start)))

    # Phase 2: Main optimization (allocate time by priority)
    main_deadline = deadline - 200.0
    main_budget = main_deadline - t_start
    starts_sorted = sorted(range(len(starts)), key=lambda i: score(starts[i]))

    if len(starts_sorted) == 1:
        fracs = [1.0]
    elif len(starts_sorted) == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.55, 0.28, 0.17]

    best_seq = best_start.copy()
    best_score = best_start_sc
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_deadline)

        if time.monotonic() > sub_dead - 30.0:
            continue

        sq = starts[si].copy()
        if rank > 0:
            # Blend with current best for secondary starts
            base = best_seq
            if len(base) != len(sq):
                base = resample_seq(base, len(sq))
            base = normalize(base)
            t_blend = 0.5 if rank == 1 else 0.7
            sq = normalize((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.10f} (n={})".format(rank + 1, best_score, len(seq)))

    best_seq = normalize(best_seq)

    # Phase 3: Perturbation restarts
    perturb_deadline = deadline - 100.0
    perturb_done = 0
    while time.monotonic() < perturb_deadline and perturb_done < 5:
        left = perturb_deadline - time.monotonic()
        if left < 40.0:
            break
        n_left = 5 - perturb_done
        chunk_dead = time.monotonic() + left / float(n_left)
        p = perturb_seq(best_seq, rng)
        sc, seq = quick_optimize(p, chunk_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.10f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 4: Final high-precision FW + LSE
    if time.monotonic() < deadline - 40.0:
        final_deadline = deadline - 10.0
        sc, seq = fw_optimize(best_seq, final_deadline - 20.0, max_iters=50, verbose=True)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

        if time.monotonic() < deadline - 15.0:
            rem = deadline - time.monotonic() - 5.0
            if rem > 8.0:
                mf = max(30, int(rem * 0.8 / 0.05))
                sc, seq = run_lbfgs(best_seq, 200000.0, deadline - 5.0, mf)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()

    # Phase 5: Final subgradient
    if time.monotonic() < deadline - 10.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
