def nesterov_subgradient(a, n_iters=5000, top_m=20):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    
    M_fft = 1 << (2*n-1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    y = a.copy()
    t_ = 1.0
    
    for it in range(n_iters):
        t_new = (1 + np.sqrt(1 + 4 * t_**2)) / 2
    
        # Compute subgradient at y
        X = np.fft.rfft(y, M_fft)
        c = np.fft.irfft(X * X, M_fft)[:2*n-1]
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        c_max = float(c.max())
        
        # Top-m peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Compute subgradient (averaged over top peaks)
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # g[j] += 2 * a[k-j] for valid j
            # This is equivalent to: g = 2 * cross_correlation
            # conv(rev(y), ones)[...] 
            # Actually, for each k, the contribution to g[j] is 2*y[k-j] if 0<=k-j<n
            # This is the same as (rev(y) * delta_k) or cross-correlation
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end+1]
                g[j_start:j_end+1] += 2.0 * seg[::-1]
        
        g /= len(top_idx)
        
        # Add the (sum a)^{-2} and (sum a)^{-3} terms
        S2 = S_y * S_y
        S3 = S2 * S_y
        g = (2.0 * n) * (g / S2 - 2.0 * c_max / S3)
        
        # Nesterov update
        step = 2.0 / (1.0 + t_new)  # or some adaptive step
        a_new = y - step * g
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Nesterov momentum
        beta = (t_ - 1) / t_new
        y = a_new + beta * (a_new - y)
        y = np.maximum(y, 0.0)
        
        t_ = t_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
    
    return best_a

def coordinate_optimize(a, j, v_range=(0.0, 1.0), n_evals=50):
    """Optimize coordinate j of a using golden section search."""
    n = len(a)
    a_fixed = a.copy()
    a_fixed[j] = 0.0
    S_0 = float(a_fixed.sum())
    c_0 = fast_conv(a_fixed)
    
    # h[k] = coefficient of v in c[k](v), and h2[k] = coefficient of v^2
    h = np.zeros(2*n-1)
    for k in range(2*n-1):
        k_minus_j = k - j
        if 0 <= k_minus_j < n:
            h[k] = 2.0 * a_fixed[k_minus_j]
    # Quadratic term at k = 2j
    h2 = np.zeros(2*n-1)
    if 2*j < 2*n-1:
        h2[2*j] = 1.0
    
    def f_of_v(v):
        c = c_0 + v * h + v * v * h2
        S = S_0 + v
        if S < 1e-12:
            return 1e10
        return 2.0 * n * float(np.max(c)) / (S * S)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = v_range
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f_of_v(c1), f_of_v(c2)
    
    for _ in range(n_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f_of_v(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f_of_v(c2)
    
    v_opt = (lo + hi) / 2
    a[j] = v_opt
    return a

"""
Advanced optimizer combining X^2 spectral shaping, L-BFGS, subgradient,
and coarse-to-fine strategies for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def x2_shape(a, n_iters=300, alpha_init=0.05, alpha_end=0.4, warmup=50):
    """Spectral shaping operating on X^2 = FFT(a)^2.
    
    This directly targets the convolution peak (IFFT(X^2)) rather than
    the auto-correlation peak (IFFT(|X|^2)).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if it < warmup:
            alpha = alpha_init
        else:
            progress = (it - warmup) / max(1, n_iters - warmup)
            alpha = alpha_init + (alpha_end - alpha_init) * progress
        
        X = np.fft.rfft(a, M_fft)
        Y = X * X
        c = np.fft.irfft(Y, M_fft)[:2 * n - 1]
        c = np.maximum(c, 0.0)  # numerical safety
        
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        # Softly compress peaks above the mean
        excess = np.maximum(c - c_mean, 0.0)
        c_mod = c - alpha * excess
        c_mod = np.maximum(c_mod, 0.0)
        
        # Recover modified X via complex square root
        Y_mod = np.fft.rfft(c_mod, M_fft)
        X_new = np.sqrt(Y_mod)
        
        a_new = np.fft.irfft(X_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def x2_waterfill(a, n_iters=300, n_levels=5, alpha_base=0.1):
    """Multi-level water-filling on X^2 spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        
        X = np.fft.rfft(a, M_fft)
        Y = X * X
        c = np.fft.irfft(Y, M_fft)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        # Multi-level compression
        c_mod = c.copy()
        for lv in range(n_levels):
            frac = 0.2 * (lv + 1) / n_levels
            thresh = np.quantile(c, 1.0 - frac)
            excess = np.maximum(c_mod - thresh, 0.0)
            strength = alpha_base * (0.5 + 0.5 * progress)
            c_mod -= strength * excess
        
        c_mod = np.maximum(c_mod, 0.0)
        
        Y_mod = np.fft.rfft(c_mod, M_fft)
        X_new = np.sqrt(Y_mod)
        a_new = np.fft.irfft(X_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_momentum(a, n_iters=3000, top_m=20, step_init=0.02, decay=0.9999):
    """Subgradient with momentum (heavy ball)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    v = np.zeros(n)  # momentum
    momentum = 0.9
    
    for it in range(n_iters):
        X = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(X * X, M_fft)[:2 * n - 1]
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg /= len(top_idx)
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        step *= decay
    
    return best_a


def downsample(a, factor):
    """Downsample by averaging groups of `factor` elements."""
    n = len(a)
    m = n // factor
    a_trunc = a[:m * factor]
    return a_trunc.reshape(m, factor).mean(axis=1)


def upsample(a, n):
    """Upsample a to length n via linear interpolation."""
    m = len(a)
    if m == n:
        return a.copy()
    if m == 1:
        return np.ones(n) * a[0]
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, a)


def coarse_to_fine(parent, n_target, budget_s):
    """Optimize at multiple scales, upsample, and refine."""
    t_start = time.monotonic()
    best = parent.copy()
    best_sc = compute_score(parent)
    
    # Try different downsample factors
    for factor in [2, 4, 8]:
        if time.monotonic() - t_start > budget_s:
            break
        n_coarse = n_target // factor
        a_coarse = downsample(parent, factor)
        a_coarse = np.maximum(a_coarse, 0.0)
        S = a_coarse.sum()
        if S < 1e-12:
            continue
        a_coarse = a_coarse / S
        
        # Optimize the coarse sequence
        a_opt = x2_shape(a_coarse, n_iters=200, alpha_init=0.05, alpha_end=0.3)
        a_opt = x2_waterfill(a_opt, n_iters=100, n_levels=4)
        
        # Upsample and evaluate
        a_full = upsample(a_opt, n_target)
        a_full = np.maximum(a_full, 0.0)
        S = a_full.sum()
        if S < 1e-12:
            continue
        a_full = a_full / S
        
        sc = compute_score(a_full)
        if sc < best_sc:
            best_sc = sc
            best = a_full.copy()
            print(f"    C2F factor={factor}: {sc:.8f}")
    
    return best
