```python
"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses GS-Fienup (flat spectrum target), LSE-LBFGS with adaptive beta continuation,
peak-focused local search, and perturbation-based refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient.
    score = 2n * L / S^2 where L = (1/beta) * log(sum(exp(beta * c[k])))
    Returns (score, gradient) as a tuple.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(np.max(c))
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = c_max + float(np.log(exp_shifted.sum())) / beta

    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a[i] = 2 * sum_m a[m] * p[i+m]
    # Computed via FFT: r[i] = (a_flipped * p)[n-1+i]
    a_flipped = a[::-1].copy()
    conv_len = n + len(p) - 1
    M_fft = 1 << (conv_len - 1).bit_length()
    A_f = np.fft.rfft(a_flipped, M_fft)
    P = np.fft.rfft(p, M_fft)
    conv_result = np.fft.irfft(A_f * P, M_fft)[:conv_len]
    r = conv_result[n - 1:n - 1 + n]

    dL_da = 2.0 * r
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L * a / S3)

    return score, grad


def lse_objective(x, beta):
    """Objective + gradient for scipy.minimize."""
    a = np.maximum(x, 0.0)
    return lse_score_and_grad(a, beta)


def gs_fienup(a, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """
    GS-Fienup iteration with correct flat-convolution target.
    Target: |A(f)|^2 should approximate DFT(flat_convolution).
    """
    n = len(a)
    M.fft_len = 0  # avoid shadowing
    M_fft = 1 << (2 * n - 1 - 1).bit_length()

    # Target: flat convolution of length 2n-1, padded to M_fft
    b_target = np.zeros(M_fft, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        # Blend magnitude toward target
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
        A_new = new_mag * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_relaxed(a, n_iters=500, relax=0.5):
    """
    GS with relaxation: a_new = (1-relax)*a + relax*GS_step(a).
    More stable for difficult problems.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1 - 1).bit_length()

    b_target = np.zeros(M_fft, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        A_mag = np.maximum(np.abs(A), 1e-20)

        new_mag = A_mag * (T_mag / A_mag) ** 0.5  # sqrt blend
        # Actually use the standard GS: replace mag with target
        # But with relaxation:
        new_mag_gs = T_mag  # full GS step target
        A_gs = new_mag_gs * np.exp(1j * np.angle(A))
        a_gs = np.fft.irfft(A_gs, M_fft)[:n]
        a_gs = np.maximum(a_gs, 0.0)
        S_gs = float(a_gs.sum())
        if S_gs < 1e-12:
            break
        a_gs = a_gs / S_gs

        # Relaxation
        a_new = (1.0 - relax) * a + relax * a_gs
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_focused_optimize(a, n_iters=2000, top_k=10):
    """
    Iteratively reduce the top-k peaks of the convolution.
    For each peak, decrease contributing elements and increase others.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(np.max(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-k peaks
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]

        # Compute reduction direction
        reduce = np.zeros(n)
        increase = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            for j in range(n):
                other = ki - j
                if 0 <= other < n:
                    # a[j]*a[other] contributes to c[ki]
                    # Reduce the larger of the two
                    if a[j] >= a[other]:
                        reduce[j] += a[other]
                    else:
                        reduce[other] += a[j]

        # Normalize and apply
        r_norm = float(np.linalg.norm(reduce))
        if r_norm < 1e-15:
            break

        # Decrease elements that contribute to peaks
        eta = 0.01
        a -= eta * reduce / r_norm * M_val
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def subgradient_top_m(a, n_iters=3000, top_m=10):
    """Subgradient method targeting top-m peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.05
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(np.max(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:min(top_m, len(c))]

        sg = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max(0.01, (sc - sc_lb)) / (gn * gn + 1e-30))
        step = min(step, 0.08)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999

    return best_a


def perturbation_search(a, n_trials=2000, sigma_init=0.01):
    """Simple perturbation-based local search."""
    n = len(a)
    best_a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(best_a.sum())
    if S < 1e-12:
        return best_a
    best_a = best_a / S
    best_sc = compute_score(best_a)
    sigma = sigma_init
    rng = np.random.default_rng(12345)

    for it in range(n_trials):
        # Adaptively decrease sigma
        sigma = sigma_init * (0.995 ** it)

        # Random perturbation
        perturbation = rng.standard_normal(n) * sigma
        a_new = best_a + perturbation
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        sc_new = compute_score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            # Slightly increase sigma on success
            sigma = min(sigma * 1.1, sigma_init)

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.01).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.2 * np.sin(10.0 * np.pi * x) * np.cos(4.0 * np.pi * x), 0.01))
    # Exponential taper
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-3.0 * np.abs(2.0 * x - 1.0)), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses GS-Fienup, LSE-LBFGS with adaptive beta,
    peak-focused optimization, and perturbation search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [6962, 8192]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")
    print(f"Budget: {total_budget}s, deadline at t={deadline - t_start:.0f}s")

    # ============================================================
    # Phase 1: GS-Fienup from diverse initializations (35% of time)
    # ============================================================
    phase1_end = t_start + 0.35 * (total_budget - 60.0)
    print(f"\n=== Phase 1: GS-Fienup (until t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  GS at n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            # Standard GS
            a1 = gs_fienup(a0, n_iters=400, alpha_init=0.7, alpha_final=0.1)
            sc = compute_score(a1)
            update_best(sc, a1, f"GS n={n} init={i}")

            # Relaxed GS from the result
            if time.monotonic() < phase1_end - 10:
                a2 = gs_relaxed(a1, n_iters=300, relax=0.4)
                sc2 = compute_score(a2)
                update_best(sc2, a2, f"GS-relax n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS with adaptive beta continuation (35% of time)
    # ============================================================
    phase2_end = t_start + 0.70 * (total_budget - 60.0)
    print(f"\n=== Phase 2: LSE-LBFGS adaptive beta (until t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_work = best_a.copy()

    # Adaptive beta schedule: start low for shape, increase for peak focus
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        t_b = time.monotonic()
        # Scale maxiter based on beta (higher beta = harder = fewer iters needed per improvement)
        maxiter = max(500, 5000 - int(400 * np.log10(beta + 1)))
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 20)
        elapsed_b = time.monotonic() - t_b
        improved = update_best(sc, a_opt, f"LBFGS beta={beta}")
        if improved:
            a_work = a_opt.copy()
        print(f"    beta={beta}: score={sc:.8f}, time={elapsed_b:.1f}s, improved={improved}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Peak-focused + subgradient (20% of time)
    # ============================================================
    phase3_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Peak-focused + Subgradient (until t={phase3_end - t_start:.0f}s) ===")

    # Peak-focused optimization
    if time.monotonic() < phase3_end - 30:
        a_pf = peak_focused_optimize(best_a, n_iters=1000, top_k=5)
        sc = compute_score(a_pf)
        update_best(sc, a_pf, "peak-focused k=5")

        a_pf2 = peak_focused_optimize(best_a, n_iters=1000, top_k=10)
        sc = compute_score(a_pf2)
        update_best(sc, a_pf2, "peak-focused k=10")

    # Subgradient
    for top_m in [5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        a_sg = subgradient_top_m(best_a, n_iters=2000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish - perturbation + high-beta LSE (10% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Perturbation search
    if time.monotonic() < deadline - 60:
        a_pert = perturbation_search(best_a, n_trials=1000, sigma_init=0.005)
        sc = compute_score(a_pert)
        update_best(sc, a_pert, "perturbation")

    # High-beta LSE polish
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "FINAL-LBFGS beta=20000")

    # One more GS round from best
    if time.monotonic() < deadline - 15:
        a_gs_final = gs_fienup(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.05)
        sc = compute_score(a_gs_final)
        update_best(sc, a_gs_final, "FINAL-GS")

    # Very high beta LSE
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "FINAL-LBFGS beta=50000")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
