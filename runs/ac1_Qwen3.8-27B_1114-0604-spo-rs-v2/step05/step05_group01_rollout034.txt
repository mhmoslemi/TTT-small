"""

Uses power-law parameterized initializations, L-BFGS with LSE smoothing in

import numpy as np

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """FFT-based self-convolution (full, length 2n-1)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Correlation of a with w: result has length n, result[j] = sum_i a[i+j]*w[i]."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(max(n, m), dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[m - 1:m - 1 + n]


def normalize_a(a):
    """Normalize sequence to sum=1, clip negatives."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / S


def exact_score(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def d_array(a):
    """Compute normalized convolution d_k = n*conv[k]/S^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    return np.maximum(d, 0.0), S


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (with sum=1 constraint handled by penalty)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    d, S = d_array(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)

    if beta <= 1e-6:
        beta = 10.0

    z = beta * (d - dmax)
    # Clip for numerical stability
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    # Gradient of 2*L w.r.t. a:
    # d(conv[k])/d(a[j]) involves corr_pair
    # d(d[k])/d(a[j]) = n/S^2 * d(conv[k])/d(a[j]) - 2*n*conv[k]*a[j]/S^3
    # Summing with w[k]:
    # grad[j] = n/S^2 * sum_k w[k]*d(conv[k])/d(a[j]) - 2/S^3 * sum_k w[k]*conv[k]*a[j]*n
    #         = n/S^2 * h[j] - 2*n*a[j]/S^3 * S^2 * LDS/n
    #         = n/S^2 * h[j] - 2*a[j]*LDS/S
    # Multiplied by 2 for the score 2*L:
    # grad_score[j] = 2*n/S^2 * h[j] - 4*a[j]*LDS/S

    grad = (2.0 * float(n) / (S * S)) * h - (4.0 * LDS / S) * a

    # Penalty for sum != 1
    PEN = 1e-3
    score = 2.0 * L + PEN * (S - 1.0) ** 2
    grad += 2.0 * PEN * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def subgrad_step_active(a, k_active, step_size):
    """Active-set subgradient step: multiply a by exp(-eta * h/hmax) where h is correlation with active weights."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    d, S = d_array(a)
    if S < 1e-12 or d.size == 0:
        return a.copy()
    dmax = float(np.max(d))
    if dmax <= 1e-12:
        return a.copy()

    m = d.size
    if k_active <= 0 or k_active >= m:
        w = np.ones(m) / m
    else:
        idx = np.argpartition(d, m - k_active)[m - k_active:]
        w = np.zeros(m)
        # Use softmax weighting within active set for smoother steps
        w_vals = d[idx]
        w_max = float(np.max(w_vals))
        w_exp = np.exp(10.0 * (w_vals - w_max))
        w_sum = float(w_exp.sum())
        if w_sum < 1e-12:
            w_exp = np.ones(len(idx))
            w_sum = float(w_exp.sum())
        w[idx] = w_exp / w_sum

    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a.copy()
    return a * np.exp(-step_size * h / hmax)


def subgrad_polish(a, max_iters=50, deadline=None):
    """Iterative subgradient polish with multiple active-set sizes and step sizes."""
    a = normalize_a(a.copy())
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        for k in (1, 2, 3, 5, 8, 12):
            for mult in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0):
                if deadline is not None and time.monotonic() > deadline:
                    break
                q = subgrad_step_active(a, k, mult)
                q = normalize_a(q)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < exact_score(a) - 1e-12:
                    a = q.copy()
                    improved = True

        if not improved:
            # Try smaller steps
            found = False
            for k in (1, 2, 3):
                for mult in (0.005, 0.01, 0.02):
                    q = subgrad_step_active(a, k, mult)
                    q = normalize_a(q)
                    sc = exact_score(q)
                    if np.isfinite(sc) and sc < exact_score(a) - 1e-13:
                        a = q.copy()
                        found = True

            if not found:
                break

    return float(best_sc), best_a


def lse_minimize(a0, beta, deadline, maxfun=500):
    """Run L-BFGS-B with LSE objective."""
    a = normalize_a(a0.copy())
    n = len(a)

    def fun(x):
        x = np.clip(x, 0.0, None)
        sc, g = lse_score_and_grad(x, beta)
        return sc, g

    bounds = [(0.0, 10.0)] * n
    options = {
        'ftol': 1e-16,
        'gtol': 1e-14,
        'maxiter': min(200, maxfun),
        'maxfun': maxfun,
        'maxls': 40,
    }

    try:
        res = minimize(fun, a, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, None)
    except Exception:
        a_out = a

    a_out = normalize_a(a_out)
    sc = exact_score(a_out)
    if not np.isfinite(sc):
        a_out = a
        sc = exact_score(a)
    return sc, a_out


def generate_power_law(n, alpha):
    """Generate a[i] = (i+1)^alpha, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)

def generate_param_family(n):
    """Generate diverse parameterized initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # Power laws: a[i] = (i+1)^alpha
    for alpha in [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]:
        inits.append(generate_power_law(n, alpha))
        # Reversed power law
        inits.append(generate_power_law(n, alpha)[::-1].copy())

    # Exponential: a[i] = exp(r*x)
    for r in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(normalize_a(np.exp(r * x)))
        inits.append(normalize_a(np.exp(-r * x)))

    # Quadratic: a[i] = (x)^2 or (1-x)^2 or x*(1-x)
    inits.append(normalize_a(x ** 2))
    inits.append(normalize_a((1 - x) ** 2))
    inits.append(normalize_a(x * (1 - x) + 0.01))
    inits.append(normalize_a(x ** 3))

    # Mixed: power law with bump
    for alpha in [1.0, 1.5, 2.0]:
        for bump_loc in [0.3, 0.5, 0.7]:
            for bump_w in [0.1, 0.2, 0.3]:
                a = (x + 0.01) ** alpha + 0.5 * np.exp(-((x - bump_loc) / bump_w) ** 2)
                inits.append(normalize_a(a))

    # Cosine-modulated power law
    for alpha in [0.5, 1.0, 1.5]:
        for f in [1, 2, 3]:
            a = (x + 0.01) ** alpha * (1.0 + 0.3 * np.cos(2 * np.pi * f * x))
            inits.append(normalize_a(np.maximum(a, 0.0)))

    return inits


def resample_seq(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n_new) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)


def quick_param_sweep(n, deadline):

def perturb(a, rng, strength=0.3):
    """Perturb a sequence for diversification."""
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    mode = int(rng.integers(0, 6))

    if mode == 0:
        # Multiplicative Gaussian noise
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.05, 0.2))
        q = (1 - t) * a + t * np.ones(n) / n
    elif mode == 2:
        # Reverse
        q = a[::-1].copy()
    elif mode == 3:
        # Active-set directed perturbation
        d, S = d_array(a)
        if d.size > 0:
            k = int(rng.integers(1, 6))
            q = subgrad_step_active(a, k, float(rng.uniform(0.1, 0.5)))
        else:
            q = a.copy()
    elif mode == 4:
        # Smooth perturb (Gaussian kernel)
        k = int(rng.integers(5, max(7, n // 500)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.5)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    else:
        # Add small power-law component
        x = np.linspace(0, 1, n)
        alpha = float(rng.uniform(0.5, 3.0))
        bump = (x + 0.01) ** alpha
        q = a + strength * 0.1 * bump
        q = np.maximum(q, 0)

    return normalize_a(np.clip(q, 0.0, None))

        fracs = [1.0]
    elif len(selected) == 2:
        fracs = [0.65, 0.35]
    elif len(selected) == 3:
        fracs = [0.55, 0.30, 0.15]
    else:
        fracs = [0.50, 0.25, 0.15, 0.10]

    cum = 0.0
    for i, (sc_init, a_init) in enumerate(selected):
        if i >= len(fracs):
            break
        cum += fracs[i]
        stage_dead = time.monotonic() + (1.0 - cum) * main_budget
        stage_dead = min(stage_dead, main_dead - 10.0)

        if time.monotonic() > stage_dead - 20.0:
            continue

        print("Optimizing candidate {} (score={:.8f}, len={})".format(i, sc_init, len(a_init)))
        sc, a_opt = optimize_with_continuation(a_init, stage_dead, label="cand{}".format(i))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print("  New global best: {:.8f}".format(best_sc))

    best_a = normalize_a(best_a)
    print("After Phase 2: {:.8f} (len={})".format(best_sc, len(best_a)))

    # Phase 3: Subgradient polish
    print("Phase 3: Subgradient polish")
    sg_dead = deadline - 200.0
    if time.monotonic() < sg_dead:
        sc, a = subgrad_polish(best_a, max_iters=80, deadline=sg_dead)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  SG polish: {:.8f}".format(best_sc))

    # Phase 4: Perturbation restarts
    print("Phase 4: Perturbation restarts")
    pert_dead = deadline - 100.0
    n_perts = 0
    while time.monotonic() < pert_dead and n_perts < 8:
        left = pert_dead - time.monotonic()
        if left < 90.0:
            break
        n_perts += 1
        # Perturb and do a shorter optimization
        p = perturb(best_a, rng, strength=float(rng.uniform(0.1, 0.4)))
        pert_dead_i = time.monotonic() + left * 0.8
        sc, a = optimize_with_continuation(p, pert_dead_i, label="pert{}".format(n_perts))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  Perturb {}: {:.8f}".format(n_perts, best_sc))

    # Phase 5: Final high-precision polish
    print("Phase 5: Final polish")
    final_dead = deadline - 20.0
    if time.monotonic() < final_dead - 30.0:
        # Very high beta LSE
        rem = final_dead - time.monotonic()
        if rem > 30.0:
            n = len(best_a)
            est_tpf = max(0.001, n / 2000000.0)
            mf = int(min(300, max(50, rem * 0.7 / (est_tpf * 2.5))))
            sc, a = lse_minimize(best_a, 100000.0, final_dead - 10.0, maxfun=mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print("  Final LSE: {:.8f}".format(best_sc))

    # Final subgradient cleanup
    if time.monotonic() < final_dead - 10.0:
        sc, a = subgrad_polish(best_a, max_iters=20, deadline=final_dead - 5.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  Final SG: {:.8f}".format(best_sc))

    # Final output preparation
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
