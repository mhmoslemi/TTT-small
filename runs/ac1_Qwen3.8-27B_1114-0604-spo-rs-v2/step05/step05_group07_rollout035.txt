def binomial_init(n, edge_beta=0.0, p_power=2):
    """Initialize with central binomial coefficients, optionally with edge boosting."""
    k = np.arange(n, dtype=np.float64)
    # C(2k, k) / 4^k using log for numerical stability
    from math import lgamma
    log_a = np.array([lgamma(2*int(ki)+1) - 2*lgamma(int(ki)+1) - int(ki)*np.log(4) for ki in range(n)])
    a = np.exp(log_a)

        x = k / (n - 1) if n > 1 else np.array([0.5])
        window = 1.0 + edge_beta * np.abs(2*x - 1)**p_power
        a = a * window
    
    S = a.sum()
    if S > 0:
        a = a / S
    return a

def active_set_equalize(a, n_active=10, n_steps=100):
    """
    Active-set peak equalization:
    - Identify the top n_active peaks in the convolution
    - Compute the subgradient from these peaks
    - Take a damped step to reduce the peaks while preserving non-negativity
    - Renormalize
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    step = 0.01
    for s in range(n_steps):
        c = fast_conv(a)
        S_a = float(a.sum())
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_active]
        c_max = c[top_idx[0]]
        
        # Compute weighted subgradient
        # Weights proportional to peak height (normalized)
        weights = c[top_idx] / c[top_idx].sum()
        
        sg = np.zeros(n)
        for w, k_val in zip(weights, top_idx):
            j_start = max(0, k_val - n + 1)
            j_end = min(k_val, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w * 2.0 * a[k_val - idx]
        
        # Gradient of score w.r.t. a
        S2 = S_a * S_a
        S3 = S2 * S_a
        Mak = float(c.max())
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Take step
        a_new = a - step * grad / gnorm
        a_new = np.clip(a_new, 0.0, 1000.0)
        Sa = a_new.sum()
        if Sa < 1e-12:
            break
        a = a_new / Sa
        
        step *= 0.995
    
    return best_a

def coordinate_descent_peak(a, n_sweeps=5, n_candidates=20):
    """
    Coordinate descent: for each position in the peak's support,
    try reducing it and redistributing to non-peak positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end <= j_start:
            break
        
        # Positions contributing to the peak
        contrib = a[j_start:j_end+1][::-1].copy()
        
        # Top contributors (to reduce)
        top_count = min(n_candidates, len(contrib) // 4)
        top_local = np.argsort(contrib)[::-1][:top_count]
        
        # Bottom contributors (to increase)
        bot_local = np.argsort(contrib)[top_count:]
        
        for li in top_local:
            global_idx = j_start + li
            old_val = a[global_idx]
            if old_val < 1e-10:
                continue
            
            best_val = old_val
            best_local_sc = best_sc
            
            # Try reducing by various fractions
            for frac in [0.5, 0.7, 0.8, 0.9, 0.95]:
                new_val = old_val * frac
                reduction = old_val - new_val
                if reduction < 1e-12:
                    continue
                a[global_idx] = new_val
                # Redistribute to bottom positions
                per = reduction / len(bot_local)
                for bi in bot_local:
                    a[j_start + bi] += per
                S_new = a.sum()
                if S_new > 1e-12:
                    a = a / S_new
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            
            a[global_idx] = best_val
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

- Identify the peak position k*

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

_current_beta = 1000.0

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def binomial_init(n):
    """Central binomial coefficients C(2k,k)/4^k normalized to sum=1."""
    k = np.arange(n)
    log_a = np.array([lgamma(2 * int(ki) + 1) - 2 * lgamma(int(ki) + 1) - int(ki) * mlog(4) for ki in range(n)])
    a = np.exp(log_a)
    S = a.sum()
    if S > 0:
        a = a / S
    return a

def edge_boost_init(n, beta_val, p_power=2):
    """Binomial with edge-boosting window."""
    base = binomial_init(n)
    k = np.arange(n, dtype=np.float64)
    if n > 1:
        x = k / (n - 1)
    else:
        x = np.array([0.5])
    window = 1.0 + beta_val * np.abs(2.0 * x - 1.0) ** p_power
    a = base * window
    S = a.sum()
    if S > 0:
        a = a / S
    return a

def symmetric_binomial_init(n):
    """Symmetric sum of binomial coefficients from both ends."""
    k = np.arange(n)
    log_a1 = np.array([lgamma(2 * int(ki) + 1) - 2 * lgamma(int(ki) + 1) - int(ki) * mlog(4) for ki in range(n)])
    log_a2 = np.array([lgamma(2 * int(n - 1 - ki) + 1) - 2 * lgamma(int(n - 1 - ki) + 1) - int(n - 1 - ki) * mlog(4) for ki in range(n)])
    a = np.exp(log_a1) + np.exp(log_a2)
    S = a.sum()
    if S > 0:
        a = a / S
    return a

def gaussian_edge_init(n, gamma):
    """Binomial with Gaussian edge boost: a_k = C(2k,k)/4^k * exp(gamma*(k/(n-1)-0.5)^2)."""
    base = binomial_init(n)
    if n > 1:
        x = np.arange(n, dtype=np.float64) / (n - 1)
    else:
        x = np.array([0.5])
    boost = np.exp(gamma * (x - 0.5) ** 2)
    a = base * boost
    S = a.sum()
    if S > 0:
        a = a / S
    return a

def coordinate_descent_peak(a, n_sweeps=10, n_top=30):
    """Targeted coordinate descent: reduce peak contributors, redistribute to non-peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg_len = j_end - j_start + 1
        contrib = a[j_start:j_end + 1][::-1].copy()
        top_count = min(n_top, max(1, seg_len // 5))
        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[top_count:]
        if len(bot_local) == 0:
            break

        for li in top_local:
            global_idx = j_start + li
            old_val = a[global_idx]
            if old_val < 1e-10:
                continue
            best_val = old_val
            best_local_sc = best_sc
            for frac in [0.3, 0.5, 0.7, 0.85, 0.95]:
                new_val = old_val * frac
                reduction = old_val - new_val
                if reduction < 1e-12:
                    continue
                a[global_idx] = new_val
                per = reduction / len(bot_local)
                for bi in bot_local:
                    a[j_start + bi] += per
                S_new = a.sum()
                if S_new > 1e-12:
                    a = a / S_new
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[global_idx] = best_val
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    a
