"""
Optimized LSE-smoothed gradient descent with Nesterov acceleration,
combined with L-BFGS refinement and exact subgradient polishing.
Uses adaptive beta scheduling and periodic restarts for robustness.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient computation via FFT
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def nesterov_lse(a0, betas, iters_per_beta, lr_init=0.01):
    """Nesterov-accelerated gradient descent on LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = normalize_nonneg(a)
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for beta in betas:
        for it in range(iters_per_beta):
            sc, grad = lse_score_and_grad(y, beta)
            
            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            theta = (t - 1.0) / t_new
            t = t_new
            
            # Backtracking line search
            lr = lr_init / (1.0 + 0.1 * it)
            c1 = 1e-4
            max_bt = 20
            
            for bt in range(max_bt):

                a_new = y_new + theta * (y_new - y)
                a_new = normalize_nonneg(a_new)
                
                sc_new, _ = lse_score_and_grad(a_new, beta)

"""
Multi-phase optimizer for minimizing 2n*max(conv)/sum^2 over non-negative sequences.
Strategy: LSE-smoothed L-BFGS with aggressive beta schedule, Nesterov subgradient
polish with backtracking, and periodic perturbation-restart cycles.
Uses FFT-based convolution for O(n log n) gradient evaluations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # dL/da[j] = 2 * sum_m p[m+j] * a[m] = 2 * (p * rev_a)[n-1+j]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def lbfgs_lse(a0, beta, maxiter=20000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective with bounds a >= 0."""
    n = len(a0)
    a0c = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0c.sum() < 1e-6:
        a0c[:] = 1.0 / n

    bounds = [(0.0, 1000.0)] * n

    def func_jac(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        func_jac, a0c,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100,
                 'factr': 1e8, 'pgtol': 1e-12}
    )
    a_out = normalize_nonneg(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_subgrad(a0, n_iters=5000, f_lb=1.4, use_backtrack=True):
    """Nesterov-accelerated subgradient on the exact objective with backtracking."""
    n = len(a0)
    a = normalize_nonneg(a0.astype(np.float64).copy())
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    base_lr = 0.05

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc_y = 2.0 * n * Mak / (Sy * Sy)

        # Subgradient of 2n * max_k (a*a)[k] / (sum a)^2
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        lr = base_lr / (1.0 + 0.05 * it)
        if use_backtrack:
            lr = min(lr, 2.0 * max(0.0, sc_y - f_lb) / (gnorm * gnorm + 1e-30))
            lr = min(lr, 0.5)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - lr * grad
        x_new = y_new + theta * (y_new - x_prev)
        x_new = normalize_nonneg(x_new)

        x_prev = y
        y = x_new
        a = x_new

        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

    return best_a, best_sc


def mirror_descent(a0, beta, n_iters=3000, eta0=0.06):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""

    best_sc = compute_score(a)
    eta = eta0
    decay = 0.9995

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -40, 40)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        if it % 200 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        eta *= decay

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc


def multi_peak_smooth(a0, n_iters=3000, top_k=3, lr=0.02):
    """Gradient step using weighted top-k peaks for smoother descent."""
    n = len(a0)
    a = normalize_nonneg(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.array([top_k - r for r in range(top_k)], dtype=np.float64)
        weights /= weights.sum()

        # Weighted subgradient
        sg = np.zeros(n)
        weighted_max = 0.0
        for wi, idx in zip(weights, top_idx):
            cval = float(c[idx])
            weighted_max += wi * cval
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += wi * 2.0 * a[idx - ii]

        S2 = S_a * S2 if False else S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * weighted_max / S3)

        a_new = a - lr * grad
        a = normalize_nonneg(a_new)

        if it % 500 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        lr *= 0.9995

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc
