```python
"""
Multi-strategy convex optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Uses FFT-accelerated LSE gradients, mirror descent,
Nesterov acceleration, peak-targeted coordinate descent, and L-BFGS polish.
Exploits the convexity of the problem and the structure of autocorrelation gradients.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    Compute LSE-smoothed score and exact gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum

    # grad_L[i] = 2 * sum_j a[j] * p[i+j] = 2 * conv_a_rev_p[i+n-1]
    a_rev = a[::-1]
    len_conv = 3 * n - 2
    M_conv = 1 << (len_conv - 1).bit_length()
    Arev_ft = np.fft.rfft(a_rev, M_conv)
    P_ft = np.fft.rfft(p, M_conv)
    conv_ap = np.fft.irfft(Arev_ft * P_ft, M_conv)[:len_conv]
    grad_L = 2.0 * conv_ap[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    return score, grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_lse(a, beta, lr, n_iters=1000):
    """Mirror descent (multiplicative weights) on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        score, grad = lse_score_grad(a, beta)
        # Mirror descent update: a <- a * exp(-lr * grad) / Z
        log_update = -lr * grad
        # Subtract mean for numerical stability
        log_update -= log_update.mean()
        a_new = a * np.exp(np.clip(log_update, -50, 50))
        a_new = np.maximum(a_new, 1e-15)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def nesterov_lse_fast(a, max_time=60.0, beta_start=5.0, beta_end=20000.0,
                       lr_start=0.05, momentum_start=0.8, momentum_end=0.95):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    Uses backtracking line search for adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    lr = lr_start
    mu = momentum_start

    while time.monotonic() - t_start < max_time:
        it += 1
        # Adaptive schedules
        progress = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        mu = momentum_start + (momentum_end - momentum_start) * progress
        lr *= 0.9995

        # Nesterov lookahead
        y = a + mu * v
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-12:
            break
        y = y / S_y

        # Gradient at lookahead
        _, grad = lse_score_grad(y, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step: scale by inverse gradient norm
        effective_lr = lr / (gnorm + 1e-10) * max(0.1, best_sc - 2.0 * n / (2.0 * n - 1.0))

        # Update
        v = mu * v - effective_lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a.copy()
            break
        v = a_new - a
        a = a_new

        # Track best
        if it % 30 == 0 or it == 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def nesterov_lse_fixed(a, n_iters=3000, beta=100.0, lr=0.01, mu=0.9):
    """Nesterov LSE with fixed parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

    best_a = a.copy()
    best_sc = compute_score(a)

        S_y = y.sum()

        y = y / S_y

        _, grad = lse_score_grad(y, beta)
        v = mu * v - lr * grad
        a_new = project_simplex(a + v)
        S_new = a_new.sum()
        if S_new < 1e-12:

def peak_targeted_cd(a, n_rounds=20, n_top_peaks=10, n_top_coords=30):
    """
    Coordinate descent targeting top peak contributors.
    For each peak, reduce the coordinates that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution of a[j] to c[k] is a[k-j]
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(n_top_coords, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j_idx in top_local:
                j = int(top_global[j_idx])
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try reducing this coordinate
                for factor in [0.7, 0.8, 0.9, 0.5, 0.95]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc - 1e-10:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        improved = True
                        break
                else:
                    a[j] = old_val

        if not improved:
            break

    return best_a


def fienup_peaks(a, n_iters=200):
    """
    Fienup variant that specifically targets the peak bins.
    More aggressive reduction of the highest spectral bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_sorted = np.sort(P)[::-1]
        # Target: make top 10% of bins equal to the median
        n_top = max(1, len(P) // 10)
        target = P_sorted[n_top] * 0.9
        P_min = P_sorted[len(P) // 2] * 0.5

        # Scale down high bins, scale up low bins
        scale = np.ones_like(P)
        high = P > target
        low = P < P_min
        scale[high] = (target / P[high]) ** 0.4
        scale[low] = (P_min / P[low]) ** 0.2

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def run_lbfgs_lse(a0, beta, maxiter=5000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        x = np.maximum(x, 1e-15)
        S = x.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xx = x / S
        return lse_score_grad(xx, beta)

    res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-12, 'maxiter': maxiter,
                            'maxfun': 80000, 'maxls': 40})
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    return compute_score(a_out), a_out


def build_init_from_parent(parent, n):
    """Create initial sequence from parent, resampled to length n."""
    if parent is None or len(parent) == 0:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.01) / 0.9
    p = np.asarray(parent, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, None)
    if p.sum() < 1e-10:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x)
        return np.maximum(a, 0.01) / 0.8
    p = p / p.sum()
    if len(p) == n:
        return p
    x_old = np.linspace(0, 1, len(p))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, p)
```
