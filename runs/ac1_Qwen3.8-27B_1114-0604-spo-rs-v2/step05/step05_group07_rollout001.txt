LP-based Frank-Wolfe optimizer for low-peak autocorrelation sequences.
Uses the SOTA LP direction-finding combined with L-BFGS refinement 
and subgradient polishing.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix(f, n):
    """Build sparse convolution matrix A where A[k,i] = f[k-i] if valid."""
    rows = np.zeros(2*n-1, dtype=np.int32)  # will be overwritten
    # Actually, let me build it properly
    # Number of nonzeros = n^2
    nnz = n * n
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        count = i_end - i_start + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(i_start, i_end+1)
        vals[idx:idx+count] = f[k - np.arange(i_start, i_end+1)]
        idx += count
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0."""
    # Build sparse matrix
    A = build_conv_matrix(f, n)
    
    # Maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    b_vec = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_vec, 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        return res.x
    return None

def lp_frank_wolfe(f_init, n_iters=200, max_lp_time=30):
    """LP-based Frank-Wolfe iterations."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        f[:] = 1.0/n
        return f
    f = f / S  # normalize
    
    best_f = f.copy()
    best_sc = compute_score(f)
    
    for it in range(n_iters):
        c = fast_conv(f)
        M = float(np.max(c))
        
        # Solve LP for direction
        b = solve_lp_direction(f, M, n)
        if b is None:
            break
        
        S_b = b.sum()
        if S_b < 1e-12:
            break
        
        # Rescale b to match normalization
        b = b / S_b
        
        # Frank-Wolfe step size
        # Optimal step: minimize ||(1-t)f + t*b||^2 ... actually we want to minimize the score
        # Use the standard FW step size: t = 2/(it+2) or backtracking
        t = 2.0 / (it + 2)
        
        f_new = (1 - t) * f + t * b
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-12:
            break
        f_new = f_new / S_new
        
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        
        f = f_new
        print(f"  LP-FW it {it}: score={sc:.10f}, M={M:.6f}, S_b={S_b:.6f}")
    
    return best_f

def solve_lp_active_set(f, M, n, active_set=None, max_inner=50):
    """Solve LP with active set constraint generation."""
    if active_set is None:
        active_set = [n-1]  # start with the peak lag
    
    for inner in range(max_inner):
        # Build constraint matrix for active set
        m = len(active_set)
        A = np.zeros((m, n))
        for i, k in enumerate(active_set):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for j in range(i_start, i_end + 1):
                A[i, j] = f[k - j]
        
        # Solve LP: max sum(b) s.t. A b <= M*ones(m), b >= 0
        c = -np.ones(n)
        b_vec = M * np.ones(m)
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        
        # Check for violated constraints
        conv = fast_conv_generic(f, b)  # Need to compute f * b
        violations = np.where(conv > M * (1 + 1e-10))[0]
        
        if len(violations) == 0:
            return b, active_set
        
        # Add worst violated constraint
        worst = violations[np.argmax(conv[violations])]
        active_set.append(worst)
    
    return b, active_set
