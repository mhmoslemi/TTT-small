Optimized step function search using:
1. GS-Fienup with cosine annealing
2. Projected Gradient Descent (PGD) with LSE smoothing
3. L-BFGS-B refinement
4. Peak-repulsion local search

Key innovation: PGD on the probability simplex with LSE-smoothed max-convolution,
combined with adaptive L-BFGS and targeted peak reduction.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def simplex_project(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    p = np.maximum(v - theta, 0.0)
    p /= p.sum()  # numerical safety
    return p


def lse_score_grad_simplex(a, beta):
    """
    LSE-smoothed max-convolution score and gradient.
    Assumes a is on the simplex (sum=1, a>=0).
    Returns (L, grad) where L ≈ max(conv(a,a)) and grad = dL/da.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical safety
    exp_scaled = np.exp(scaled - scaled.max())
    p = exp_scaled / exp_scaled.sum()
    log_sum = float(np.log(np.sum(np.exp(scaled))))  # = c_max*beta + log(sum exp(...))
    # Actually: log(sum exp(beta*(c-c_max))) = beta*c_max + log(sum exp(beta*(c-c_max))) - beta*c_max
    # = log(sum exp(beta*c)) - beta*c_max
    # L = c_max + (1/beta) * log(sum exp(beta*(c-c_max)))
    #   = c_max + (1/beta) * (log(sum exp(beta*c)) - beta*c_max)
    #   = c_max + (1/beta)*log(sum exp(beta*c)) - c_max
    #   = (1/beta) * log(sum exp(beta*c))
    # So L = logsumexp(beta * c) / beta
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    # FFT size for cross-correlation of p (len 2n-1) and a (len n):

    P_fft = np.fft.rfft(p, M_fft)
    A_fft = np.fft.rfft(a, M_fft)
    cc = np.fft.irfft(P_fft * np.conj(A_fft), M_fft)
    grad = 2.0 * cc[:n]
    
    return L, grad


def pgd_simplex(a0, beta, n_iters, lr0, lr_decay, M=None):
    """
    Projected Gradient Descent on the simplex.
    Minimizes LSE normal approximation of max(conv).
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        lr = lr0 * (lr_decay ** it)
        L, grad = lse_score_grad_simplex(a, beta)
        a_new = a - lr * grad
        a = simplex_project(a_new)
        
        # Periodic exact evaluation
        if it % 20 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a


def gs_fienup_v2(a, n_iters, alpha_init, alpha_final, M=None):
    """
    Improved GS-Fienup with cosine annealing.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()
    
    # Target: flat convolution
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Cosine annealing
        alpha = alpha_final + 0.5 * (alpha_init - alpha_final) * (1.0 + np.cos(np.pi * it / max(1, n_iters-1)))
        
        A = np.fft.rfft(a, M)
        A_full = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A_full), 1e-20)
        A_phase = np.angle(A_full)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def peak_repulsion(a0, n_iters, top_m, eta0, eta_decay):
    """
    Peak repulsion: iteratively reduce elements that contribute to the top peaks.
    Works on the simplex.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        eta = eta0 * (eta_decay ** it)
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top M peaks
        top_idx = np.argsort(c)[::-1][:top_m]
        
        # Compute contribution of each element to top peaks
        # contrib[j] = sum over k in top_idx of a[j] * a[k-j]
        contrib = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # c[k] = sum_j a[j] * a[k-j]
                # contribution of a[j] to c[k] is a[j] * a[k-j]
                # We want to reduce a[j] proportionally to its contribution
                for j in range(j_start, j_end + 1):
                    contrib[j] += a[j] * a[k - j]
        
        # Normalize contrib
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        contrib_norm = contrib / total_contrib
        
        # Update: reduce elements with high contribution, increase others
        # a_new[j] = a[j] * (1 - eta * contrib_norm[j] / mean_contrib)
        mean_contrib = 1.0 / n
        a_new = a * (1.0 - eta * (contrib_norm - mean_contrib))
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
    
    return best_a


def run_lbfgs_simplex(a0, beta, maxiter=5000):
    """L-BFGS-B on LSE-smoothed objective, on the simplex."""
    n = len(a0)
    a0 = simplex_project(a0.copy())
    
    def obj(x):
        # Project to simplex for evaluation
        a_p = simplex_project(x)
        L, _ = lse_score_grad_simplex(a_p, beta)
        return 2.0 * n * L
    
    def obj_grad(x):
        a_p = simplex_project(x)
        L, grad = lse_score_grad_simplex(a_p, beta)
        return 2.0 * n * L, 2.0 * n * grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj, a0, jac=obj_grad,
        method='L-BFGS-B', bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': 100000, 'maxls': 50}
    )
    a_out = simplex_project(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a0, n_sweeps=10, n_top=50):
    """Coordinate descent: optimize each element via 1D line search."""
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Top contributing elements to the peak
        if k_star >= 0 and k_star < 2*n-1:
            contrib = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]
            top_idx = np.argsort(contrib)[::-1][:n_top]
            
            for j in top_idx:
                j = int(j)
                a_old = a[j]
                if a_old < 1e-10:
                    continue
                # Try several values
                best_val = a_old
                cur_sc = best_sc  # approximate
                for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.1]:
                    cv = a_old * f
                    a[j] = cv
                    a_proj = simplex_project(a)
                    sc = compute_score(a_proj)
                    if sc < best_sc:
                        best_sc = sc
                        best_val = cv
                        best_a = a_proj.copy()
                    a = a_proj
                a[j] = best_val
                a = simplex_project(a)
            
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(simplex_project(parent.astype(np.float64).copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(simplex_project(np.interp(x_new, x_old, parent).copy()))
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    inits.append(simplex_project(np.maximum(0.5 + 0.4*np.cos(2*np.pi*x), 0.01)))
    inits.append(simplex_project(0.3 + 0.7*np.cos(2*np.pi*x)**2))
    inits.append(simplex_project(np.clip(0.5 + 0.3*rng.standard_normal(n), 0.01, None)))
    inits.append(simplex_project(0.2 + 0.8*(2*x-1)**2))
    inits.append(simplex_project(np.maximum(0.3 + 0.5*np.cos(4*np.pi*x), 0.01)))
    inits.append(simplex_project(0.5 + 0.3*rng.standard_normal(n) * np.exp(-2*(x-0.5)**2)))
    
    return inits
