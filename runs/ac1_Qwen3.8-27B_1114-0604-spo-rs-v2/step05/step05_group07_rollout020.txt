import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS callback
_current_beta = 1000.0
_current_topk = 1


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_smoothed_max_and_grad(a, beta, topk=10):
    """
    Top-k smoothed max: smooth approximation that focuses on top-k values.
    Uses a two-level smoothing: LSE of the top-k values.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top-k values
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def lse_objective_and_grad(x):
    """Combined objective and gradient for L-BFGS."""
    return topk_smoothed_max_and_grad(x, _current_beta, _current_topk)


def run_lbfgs(a0, beta, topk, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the smoothed objective."""
    global _current_beta, _current_topk
    _current_beta = beta
    _current_topk = topk
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_objective_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=10000, eta0=0.05, decay=0.9995):
    """Mirror descent with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    
    for it in range(n_iters):
        sc, grad = topk_smoothed_max_and_grad(a, beta, 1)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay
    
    return best_a


def subgradient_polish(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a

def peak_deflation(a, n_rounds=20):
    """
    Peak deflation: iteratively identify the peak, reduce its contributors,
    and redistribute to fill in the valleys.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for round_idx in range(n_rounds):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        
        # Find top contributors to the peak
        top_frac = 0.3
        top_count = max(1, int(total_len * top_frac))
        top_local = np.argsort(contrib)[::-1][:top_count]
        
        # Reduce top contributors
        reduction_total = 0.0
        for li in top_local:
            gi = j_start + li
            a[gi] *= 0.85  # reduce by 15%
            reduction_total += contrib[li] * 0.15
        
        # Redistribute to bottom contributors (valleys)
        bottom_count = max(1, total_len - top_count)
        bottom_local = np.argsort(contrib)[:bottom_count]
        per_bottom = reduction_total / bottom_count
        for li in bottom_local:
            a[j_start + li] += per_bottom
        
        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Revert if it got worse
            a = best_a.copy()
            break
    
    return best_a


def block_coordinate_descent(a, n_sweeps=5, block_size=64):
    """
    Block coordinate descent: optimize blocks of coordinates using L-BFGS.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    n_blocks = n // block_size + 1
    
    for sweep in range(n_sweeps):
        for b in range(n_blocks):
            start = b * block_size
            end = min((b + 1) * block_size, n)
            if end - start < 2:
                continue
            
            # Fix all but the block, optimize the block
            a_block = a[start:end].copy()
            
            def block_obj(x):
                a_temp = a.copy()
                a_temp[start:end] = np.maximum(x, 0.0)
                S_temp = a_temp.sum()
                if S_temp < 1e-12:
                    return 1e10
                c = fast_conv(a_temp)
                return 2.0 * n * float(np.max(c)) / (S_temp * S_temp)
            
            res = minimize(block_obj, a_block, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            
            a_new = np.maximum(res.x, 0.0)
            S_new = a.sum() - a[start:end].sum() + a_new.sum()
            if S_new < 1e-12:
                continue
            a[start:end] = a_new
            a = a / a.sum()
            
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def perturbe(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 8)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)

        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)

        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 5:
        # Twin peak: create a dip in the middle
        center = n // 2
        width = n // 8
        dip = 1.0 - 0.5 * np.exp(-((np.arange(n) - center) / width) ** 2)
        a_p = a * (1.0 + 0.3 * (1.0 - dip))
    elif strategy == 6:
        # Random phase modulation (spectral perturbation)
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A) + scale * rng.standard_normal(len(A))
        A_new = mag * np.exp(1j * phase)
        a_p = np.fft.irfft(A_new, len(a))
    else:
        n_sel = max(1, n // 4)

        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:

    return a_p


def spectral_init(n, rng):
    """
    Spectral initialization: create a sequence whose DFT has a specific structure.
    The idea is to make |DFT(a)|^2 have low peak in time domain.
    """
    # Strategy: make DFT(a) have a specific magnitude profile
    strategies = [
        'uniform_mag',  # |A| = constant (gives delta in time, bad)
        'linear_phase',  # linear phase = shifted delta (bad)
        'random_phase',  # random phase with uniform magnitude
        'chirp',  # quadratic phase (chirp signal)
        'multi_sine',  # sum of several sinusoids
    ]
    
    strat = strategies[rng.integers(0, len(strategies))]
    M = 1 << (2*n - 2).bit_length()
    
    if strat == 'random_phase':
        mag = np.ones(M // 2 + 1)
        phase = 2 * np.pi * rng.random(M // 2 + 1)
        A = mag * np.exp(1j * phase)
        # Make it Hermitian symmetric
        A_full = np.zeros(M, dtype=complex)
        A_full[:M//2+1] = A
        for k in range(1, M//2):
            A_full[M-k] = np.conj(A[k])
        a = np.fft.irfft(A_full, M)[:n]
    elif strat == 'chirp':
        # Chirp: a[i] = exp(i*pi*i^2/N)
        i = np.arange(n)
        a = np.cos(np.pi * i**2 / (4*n) + rng.random() * np.pi)
        a = np.abs(a) + 0.1
    elif strat == 'multi_sine':
        n_sines = 3 + int(rng.integers(0, 5))
        a = np.zeros(n)
        for _ in range(n_sines):
            freq = rng.random() * 0.5 + 0.05
            phase = 2 * np.pi * rng.random()
            a += np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.abs(a) + 0.1
    elif strat == 'linear_phase':
        a = np.ones(n) * 0.5
    else:  # uniform_mag
        a = np.ones(n) * 0.5
    
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a

from scipy.optimize import minimize
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -60))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    rev_a = x[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x):
    """Wrapper for L-BFGS using global beta."""
    return lse_obj_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_wrapper, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.9995):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_obj_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def subgradient_polish(a, n_iters=5000, f_lb=1.0, top_k=1):
    """Nesterov-accelerated subgradient descent (optionally multi-peak)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        if top_k == 1:
            k_star = int(np.argmax(c))
            Mak = float(c[k_star])
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg_M = np.zeros(n)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg_M[idx] = 2.0 * y[k_star - idx]
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            w_sum = top_k * (top_k + 1) / 2.0
            Mak = 0.0
            sg_M = np.zeros(n)
            for rank, idx in enumerate(top_idx):
                w = (top_k - rank) / w_sum
                Mak += w * c[idx]
                j_s = max(0, idx - n + 1)
                j_e = min(idx, n - 1)
                if j_e >= j_s:
                    ii = np.arange(j_s, j_e + 1)
                    sg_M[ii] += 2.0 * w * y[idx - ii]
            k_star = int(np.argmax(c))
            Mak_for_sc = float(c[k_star])

        sc = 2.0 * n * float(np.max(c)) / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def peak_deflation(a, n_rounds=30, reduce_frac=0.12, top_frac=0.25):
    """
    Peak deflation: iteratively reduce top contributors to the main peak
    and redistribute mass to the valleys of that peak's support.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for round_idx in range(n_rounds):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)

        top_count = max(2, int(total_len * top_frac))
        top_local = np.argsort(contrib)[::-1][:top_count]
        bottom_count = total_len - top_count
        if bottom_count < 1:
            bottom_count = max(1, total_len // 2)
            top_count = total_len - bottom_count
            top_local = np.argsort(contrib)[::-1][:top_count]
        bottom_local = np.argsort(contrib)[:bottom_count]

        reduction_total = 0.0
        for li in top_local:
            gi = j_start + li
            old_val = a[gi]
            a[gi] = old_val * (1.0 - reduce_frac)
            reduction_total += old_val * reduce_frac

        per_bottom = reduction_total / bottom_count
        for li in bottom_local:
            a[j_start + li] += per_bottom

        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            # Try smaller reduction
            reduce_frac *= 0.5
            if reduce_frac < 0.01:
                break

    return best_a


def perturbed_restart(a, rng, scale=0.02):
    """Generate a perturbed version of a for ILS."""
    n = len(a)
    strategy = rng.integers(0, 8)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 40)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 5:
        # Spectral phase perturbation
        M_fft = 1 << (len(a) - 1).bit_length()
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A) + scale * 3.0 * rng.standard_normal(len(A))
        A_new = mag * np.exp(1j * phase)
        a_p = np.fft.irfft(A_new, M_fft)[:n]
    elif strategy == 6:
        # Center dip (twin peak)
        center = n // 2
        width = max(1, n // 10)
        dip = np.exp(-((np.arange(n) - center) / width) ** 2)
        a_p = a * (1.0 + 0.4 * scale * (1.0 - dip) * 10)
    else:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    else:
        a_p = a.copy()
    return a_p


def spectral_init(n, rng):
    """Spectral initialization exploiting Fourier structure."""
    strategies = ['random_phase', 'chirp', 'multi_sine', 'broadband']
    strat = strategies[rng.integers(0, len(strategies))]
    M = 1 << (2 * n - 2).bit_length()

    if strat == 'random_phase':
        mag = np.ones(M // 2 + 1)
        # Add some structure: low-pass
        freqs = np.arange(M // 2 + 1) / M
        mag *= np.exp(-freqs * (1.0 + 2.0 * rng.random()))
        phase = 2 * np.pi * rng.random(M // 2 + 1)
        A = mag * np.exp(1j * phase)
        A_full = np.zeros(M, dtype=complex)
        A_full[:M // 2 + 1] = A
        for k in range(1, M // 2):
            A_full[M - k] = np.conj(A[k])
        a = np.fft.irfft(A_full, M)[:n]
    elif strat == 'chirp':
        i = np.arange(n, dtype=np.float64)
        chirp_rate = 0.5 + rng.random()
        a = 0.5 + 0.5 * np.cos(np.pi * chirp_rate * i**2 / n + rng.random() * np.pi)
    elif strat == 'multi_sine':
        n_sines = 3 + int(rng.integers(0, 5))
        a = np.zeros(n)
        for _ in range(n_sines):
            freq = 0.1 + 0.8 * rng.random()
            phase = 2 * np.pi * rng.random()
            amp = 0.3 + 0.7 * rng.random()
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.abs(a) + 0.05
    else:  # broadband
        a = np.random.default_rng(rng.integers(0, 2**31)).standard_normal(n)
        a = np.abs(a) + 0.1

    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.05, np.cos(np.pi * x)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.exp(-3.0 * (x - 0.5) ** 2) + 0.05)
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init(n, rng))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Multi-start + mirror descent exploration
    Phase 2: L-BFGS with progressive beta (main refinement)
    Phase 3: Peak deflation + subgradient cycle
    Phase 4: ILS with spectral perturbations
    Phase 5: Final ultra-high-beta L-BFGS + subgradient polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def try_report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}, Budget: {budget_s}s")

    if parent is not None:
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        try_report(sc_parent, parent_norm, "parent")

    # ==================================================================
    # Phase 1: Multi-start (12% time)
    # ==================================================================
    phase1_dl = t_start + 0.12 * effective_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-Start (deadline t={phase1_dl - t_start:.0f}s)")
    print(f"{'='*60}")

    inits = build_initializations(n_work, rng, parent)
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_dl:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S
        sc = compute_score(a_init)
        try_report(sc, a_init, f"init_{i}")

        if time.monotonic() < phase1_dl - 10:
            a_md = mirror_descent(a_init, beta=80, n_iters=3000, eta0=0.07)
            sc = compute_score(a_md)
            try_report(sc, a_md, f"mirror_{i}")

    # Extra mirror descent from current best
    while time.monotonic() < phase1_dl:
        a_md = mirror_descent(best_a, beta=150, n_iters=4000, eta0=0.04)
        sc = compute_score(a_md)
        try_report(sc, a_md, "mirror_best")
        if not try_report(sc, a_md, "mirror_stall_check"):
            break

    print(f"  Phase 1 done. Best: {best_score:.10f}")

    # ==================================================================
    # Phase 2: L-BFGS progressive beta (43% time)
    # ==================================================================
    phase2_dl = t_start + 0.55 * effective_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Refinement (deadline t={phase2_dl - t_start:.0f}s)")
    print(f"{'='*60}")

    beta_schedule = [30, 80, 200, 500, 1500, 5000, 15000, 50000, 150000, 500000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_dl:
            break
        remaining = phase2_dl - time.monotonic()
        if remaining < 20:
            break
        maxiter = int(min(100000, max(4000, remaining * 60)))
        maxfun = int(min(400000, max(15000, remaining * 250)))

        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_report(sc, a_opt, f"LBFGS b={beta}")

        # Perturbation + re-optimize for higher betas
        if beta >= 1000 and time.monotonic() < phase2_dl - 30:
            a_pert = perturbed_restart(best_a, rng, scale=0.003)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            try_report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"  Phase 2 done. Best: {best_score:.10f}")

    # ==================================================================
    # Phase 3: Peak deflation + subgradient cycle (15% time)
    # ==================================================================
    phase3_dl = t_start + 0.70 * effective_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak Deflation + Subgradient (deadline t={phase3_dl - t_start:.0f}s)")
    print(f"{'='*60}")

    # Cycle: peak deflation -> subgradient -> L-BFGS
    n_cycles = 5
    for cyc in range(n_cycles):
        if time.monotonic() > phase3_dl:
            break
        remaining = phase3_dl - time.monotonic()
        if remaining < 30:
            break

        # Peak deflation
        a_pd = peak_deflation(best_a, n_rounds=30, reduce_frac=0.08, top_frac=0.2)
        sc = compute_score(a_pd)
        try_report(sc, a_pd, f"PD cyc{cyc}")

        # Multi-peak subgradient
        a_sg = subgradient_polish(a_pd, n_iters=3000, f_lb=1.0, top_k=5)
        sc = compute_score(a_sg)
        try_report(sc, a_sg, f"SG5 cyc{cyc}")

        # Quick L-BFGS
        beta = 50000
        maxiter = int(min(50000, max(2000, remaining * 30)))
        maxfun = int(min(200000, max(5000, remaining * 100)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_report(sc, a_opt, f"LBFGS_after_PD cyc{cyc}")

    print(f"  Phase 3 done. Best: {best_score:.10f}")

    # ==================================================================
    # Phase 4: ILS with diverse perturbations (13% time)
    # ==================================================================
    phase4_dl = t_start + 0.83 * effective_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Iterated Local Search (deadline t={phase4_dl - t_start:.0f}s)")
    print(f"{'='*60}")

    max_trials = 40
    no_improve_count = 0
    for trial in range(max_trials):
        if time.monotonic() > phase4_dl:
            break
        remaining = phase4_dl - time.monotonic()
        if remaining < 45:
            break

        # Adaptive perturbation scale: smaller as we converge
        base_scale = 0.003 + 0.015 * (1.0 - trial / max_trials)
        scale = base_scale * (0.5 + rng.random())
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Quick mirror descent
        a_md = mirror_descent(a_pert, beta=300, n_iters=1500, eta0=0.03)
        sc_md = compute_score(a_md)
        improved = try_report(sc_md, a_md, f"ILS#{trial} md")

        # L-BFGS refinement
        beta = 5000 + int(50000 * rng.random())
        maxiter = int(min(50000, max(2000, remaining * 25)))
        maxfun = int(min(200000, max(5000, remaining * 100)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-12), beta, maxiter=maxiter, maxfun=maxfun)
        improved2 = try_report(sc, a_opt, f"ILS#{trial} lbfgs")

        # Track stagnation
        if improved or improved2:
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count >= 15:
            print(f"  ILS stagnating, breaking at trial {trial}")
            break

    print(f"  Phase 4 done. Best: {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final ultra-high-beta L-BFGS + polish (17% time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Refinement (deadline t={deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Very high beta L-BFGS
    for beta in [100000, 300000, 1000000, 3000000]:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(100000, max(5000, remaining * 50)))
        maxfun = int(min(500000, max(10000, remaining * 200)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_report(sc, a_opt, f"FINAL b={beta}")

        # Subgradient polish after each
        if time.monotonic() < deadline - 25:
            a_sg = subgradient_polish(a_opt, n_iters=3000, f_lb=1.05, top_k=3)
            sc = compute_score(a_sg)
            try_report(sc, a_sg, f"FINAL b={beta}+sg3")

            a_sg1 = subgradient_polish(a_opt, n_iters=2000, f_lb=1.0, top_k=1)
            sc = compute_score(a_sg1)
            try_report(sc, a_sg1, f"FINAL b={beta}+sg1")

    # Final peak deflation
    if time.monotonic() < deadline - 20:
        a_pd_final = peak_deflation(best_a, n_rounds=50, reduce_frac=0.03, top_frac=0.15)
        sc = compute_score(a_pd_final)
        try_report(sc, a_pd_final, "FINAL_PD")

    # One more ultra-high beta pass
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 5000000, maxiter=50000, maxfun=200000)
        try_report(sc, a_opt, "FINAL б=5M")

    # Final mirror descent with very high beta
    if time.monotonic() < deadline - 15:
        a_final = mirror_descent(best_a, beta=50000, n_iters=5000, eta0=0.003)
        sc = compute_score(a_final)
        try_report(sc, a_final, "FINAL_mirror")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Sequence length: {len(out)}")
    print(f"Time used: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
