"""
Optimized step function sequence search using adaptive LSE-smoothed L-BFGS,
subgradient descent with momentum, and targeted peak redistribution.
Combines multiple strategies to beat the 1.5027 target.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global state for L-BFGS callback
_g_beta = 100.0
_g_use_topk = False
_g_topk = 3


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, topk=1):
    """Multi-peak LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if topk > 1:
        # Top-k LSE: penalize top-k peaks
        k = min(topk, len(c))
        top_vals = np.sort(c)[::-1][:k]
        # Use LSE on the top-k values with a smaller effective beta
        shifted = c - top_vals[0]
        scaled = beta * shifted
        exp_scaled = np.exp(np.clip(scaled, -60, 60))
        log_sum = float(np.log(exp_scaled.sum() + 1e-300))
        L = top_vals[0] + log_sum / beta
        p = exp_scaled / (exp_scaled.sum() + 1e-300)
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(np.clip(scaled, -60, 60))
        exp_sum = float(exp_scaled.sum())
        log_sum = float(np.log(exp_sum + 1e-300))
        L = c_max + log_sum / beta
        p = exp_scaled / (exp_sum + 1e-300)

    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da[i] = 2 * (p * rev_a)[n-1+i]
    rev_a = a[::-1]
    conv_len = len(p) + n - 1
    M2_fft = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:conv_len]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    return lse_score_grad(x, _g_beta)


def run_lbfgs(a0, beta, topk=1, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _g_beta
    _g_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def momentum_subgrad(a, n_iters=50000, f_lb=1.0, momentum=0.9):
    """Momentum-based subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    t = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        sc = 2.0 * n * M / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = a[k_star - idx]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-18:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5 / (1.0 + np.sqrt(it + 1)))

        v = momentum * v - step * grad
        a_candidate = a + v
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        Sa_cand = a_candidate.sum()
        if Sa_cand < 1e-12:
            break
        a_candidate = a_candidate / Sa_cand
        a = a_candidate

    return best_a


def topk_subgrad(a, n_iters=30000, f_lb=1.0, k=3, momentum=0.85):
    """Subgradient using top-k peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:k]
        M = float(c[top_idx[0]])
        sc = 2.0 * n * M / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Weighted subgradient over top-k peaks
        weights = np.array([(k - i) for i in range(k)], dtype=np.float64)
        weights /= weights.sum()

        sg = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * a[idx - ii]

        S2 = S_a * S2 if False else S_a * S_a
        S3 = S2 * S_a
        M_weighted = float(np.sum(weights * c[top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_weighted / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-18:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3 / (1.0 + 0.1 * np.sqrt(it + 1)))

        v = momentum * v - step * grad
        a_candidate = a + v
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        Sa_cand = a_candidate.sum()
        if Sa_cand < 1e-12:
            break
        a = a_candidate / Sa_cand

    return best_a


def peak_redistribute(a, n_rounds=200):
    """Directly redistribute mass to reduce the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        sc = 2.0 * n * M / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Who contributes to the peak?
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        n_seg = j_end - j_start + 1
        if n_seg == 0:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Reduce top contributors, add to bottom
        n_reduce = max(1, n_seg // 8)
        n_add = max(1, n_seg // 4)

        top_local = np.argsort(contrib)[::-1][:n_reduce]
        bot_local = np.argsort(contrib)[:n_add]

        # Fraction to redistribute
        frac = 0.05 + 0.03 * np.random.random()
        reduction = 0.0
        for li in top_local:
            old = a[j_start + li]
            a[j_start + li] = old * (1.0 - frac)
            reduction += old * frac

        # Distribute to bottom contributors
        per = reduction / n_add
        for li in bot_local:
            a[j_start + li] += per

        a = np.clip(a, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new

    return best_a


def coordinate_descent(a, n_passes=5, top_frac=0.1):
    """Coordinate descent: for each coordinate, find optimal 1D value."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        order = rng.permutation(n)
        # Focus on "active" coordinates (those contributing to peaks)
        c = fast_conv(a)
        # Get active positions
        threshold = np.percentile(c, 80)
        active_coords = set()
        for k in np.where(c > threshold)[0]:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            active_coords.update(range(js, je + 1))

        active_list = [j for j in order if j in active_coords]
        if len(active_list) < n // 2:
            active_list = list(order)

        for j in active_list:
            old_val = a[j]
            if old_val < 1e-12:
                continue
            # Try a few multiplier values
            best_frac = 1.0
            best_local_sc = best_sc
            for frac in [0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3]:
                a[j] = old_val * frac
                a = np.clip(a, 0.0, 1000.0)
                S_t = a.sum()
                if S_t < 1e-12:
                    a[j] = old_val
                    break
                a = a / S_t
                sc_t = compute_score(a)
                if sc_t < best_local_sc:
                    best_local_sc = sc_t
                    best_frac = frac
            a[j] = old_val * best_frac
            a = np.clip(a, 0.0, 1000.0)
            S_t = a.sum()
            if S_t > 1e-12:
                a = a / S_t
                sc_t = compute_score(a)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_a = a.copy()

    return best_a


def perturb(a, rng, scale=0.02):
    """Generate a perturbed version for ILS."""
    n = len(a)
    strategy = rng.integers(0, 8)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:

    elif strategy == 1:

        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * 1.5 * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 5 + int(rng.integers(0, 15))
        a_p = a.copy()
        block_size = max(1, n // n_blocks)
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        a_p = a.copy()
        top_count = max(10, n // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = 1.0 - S
            if deficit > 0:
                bot_local = np.argsort(contrib)[:top_count]
                per = deficit / len(bot_local)
                for li in bot_local:
                    a_p[j_start + li] += per
        a_p = np.clip(a_p, 0.0, 1000.0)
        S = a_p.sum()
        if S > 1e-12:
            a_p = a_p / S
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.5 * rng.standard_normal(n))
    elif strategy == 5:
        # Random subset rescale
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)

        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()

            sc2, a_opt2 = run_lbfgs(a_pert, beta, topk=topk,
                                     maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Iterated Local Search (30% time)
    # ============================================================
    p3_end = t_start + 0.83 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: ILS (until t={p3_end - t_start:.0f}s) ---")

    n_trials = 40
    for trial in range(n_trials):
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 30:
            break

        # Perturb with decreasing scale
        scale = 0.01 + 0.03 * (trial / n_trials)
        a_pert = perturb(best_a, rng, scale=max(scale, 0.015))

        # Quick subgradient
        a_sg = momentum_subgrad(a_pert, n_iters=3000, f_lb=1.0, momentum=0.9)
        sc = compute_score(a_sg)
        report(sc, a_sg, f"ILS{trial} sg")

        # L-BFGS refinement
        beta = 300 + int(2000 * (1.0 + rng.random()))
        maxiter = int(min(20000, max(1000, remaining * 10)))
        maxfun = int(min(60000, max(3000, remaining * 40)))
        sc, a_opt = run_lbfgs(a_sg, beta, topk=1, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS{trial} lbfgs b={beta}")

        # Peak redistribution
        a_pr = peak_redistribute(a_opt, n_rounds=100)
        sc = compute_score(a_pr)
        report(sc, a_pr, f"ILS{trial} peak")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Final polishing (17% time)
    # ============================================================
    p4_end = deadline
    print(f"\n--- Phase 4: Final polish (until t={p4_end - t_start:.0f}s) ---")

    # Coordinate descent
    a_cd = coordinate_descent(best_a, n_passes=3, top_frac=0.1)
    sc = compute_score(a_cd)
    report(sc, a_cd, "coord_desc")

    # High-beta L-BFGS
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > p4_end - 20:
            break
        remaining = p4_end - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(30000, max(2000, remaining * 20)))
        maxfun = int(min(100000, max(5000, remaining * 60)))
        sc, a_opt = run_lbfgs(best_a, beta, topk=1, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final LBFGS b={beta}")

    # More subgradient
    if time.monotonic() < p4_end - 30:
        remaining = p4_end - time.monotonic()
        n_it = int(min(30000, max(5000, remaining * 100)))
        a_sg = momentum_subgrad(best_a, n_iters=n_it, f_lb=1.0, momentum=0.92)
        sc = compute_score(a_sg)
        report(sc, a_sg, "final subgrad")

    # Final peak redistribution
    if time.monotonic() < p4_end - 10:
        a_pr = peak_redistribute(best_a, n_rounds=200)
        sc = compute_score(a_pr)
        report(sc, a_pr, "final peak")

    # One more L-BFGS on the very best
    if time.monotonic() < p4_end - 10:
        sc, a_opt = run_lbfgs(best_a, 100000, topk=1, maxiter=20000, maxfun=60000)
        report(sc, a_opt, "absolute final")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
