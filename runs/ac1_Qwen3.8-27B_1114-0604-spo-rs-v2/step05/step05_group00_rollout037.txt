"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines KKT-based balancing, Nesterov-accelerated subgradient, L-BFGS with log-sum-exp smoothing,
Fienup frequency-domain iterations, and coordinate descent with line search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_r(w, a):
    """
    Compute r[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    a_rev = a[::-1]
    M = len(w) + n - 1  # 3n - 2
    M_fft = 1 << (M - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv = np.fft.irfft(W * AR, M_fft)[:M]
    return conv[n - 1:n - 1 + n]

    KKT-based balancing iteration.
    Makes (w * a) approximately constant on the support,
    where w = softmax(beta * conv).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        conv = fast_conv(a)
        # Soft peak weights
        scaled = beta * (conv - conv.max())
        scaled = np.clip(scaled, -500, 500)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute r
        r = compute_r(w, a)

        # Balance: make r more uniform on support
        support = a > 1e-10
        if support.sum() < 2:

        target = r[support].mean()
        ratio = target / (r + 1e-15)
        ratio = np.clip(ratio, 0.01, 100.0)
        a = a * (ratio ** alpha)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    sc = compute_score(a)
    if sc < best_sc:

    return best_a

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines KKT-based balancing iterations, Nesterov-accelerated subgradient on the true max,
L-BFGS with log-sum-exp smoothing, Fienup frequency-domain iterations, and coordinate descent
with geometric line search. Explores multiple sequence lengths and structured initializations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_r(w, a):
    """
    Compute r[i] = sum_k w[k] * a[k-i] for i = 0,...,n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    M = len(w) + n - 1
    M_fft = 1 << (M - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv = np.fft.irfft(W * AR, M_fft)[:M]
    return conv[n - 1:n - 1 + n]


def kkt_iterate(a, n_iters, beta, alpha, deadline=None):
    """
    KKT-based balancing: makes (w * a) ~ constant on support,
    where w = softmax(beta * (conv - max_conv)).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        conv = fast_conv(a)
        scaled = beta * (conv - conv.max())
        scaled = np.clip(scaled, -400, 400)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            break
        w = w / ws

        r = compute_r(w, a)
        support = a > 1e-10
        n_sup = support.sum()
        if n_sup < 2:
            break
        target = float(r[support].mean())
        ratio = target / (r + 1e-15)
        ratio = np.clip(ratio, 0.05, 20.0)
        a = a * (ratio ** alpha)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def nesterov_subgrad(a, n_iters, top_k, deadline=None):
    """Nesterov-accelerated subgradient on the true max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1.0)

    x_prev = a.copy()
    x_curr = a.copy()
    t_k = 1.0
    sc0 = best_sc
    step = max(sc0 - lb, 0.01) / (5.0 * np.sqrt(n))

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        if it % 500 == 0:
            sc = compute_score(x_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = x_curr.copy()
            gap = max(sc - lb, 0.005)
            step = gap / (5.0 * np.sqrt(n) * (1.0 + 0.01 * it))

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_curr + ((t_k - 1.0) / t_new) * (x_curr - x_prev)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy > 1e-12:
            y = y / Sy

        conv_y = fast_conv(y)
        M_val = float(conv_y.max())
        k_actual = min(top_k, len(conv_y))
        top_idx = np.argsort(conv_y)[::-1][:k_actual]
        pk = conv_y[top_idx]
        psum = pk.sum()
        if psum < 1e-30:
            break
        wk = pk / psum

        sg = np.zeros(n)
        for idx in range(k_actual):
            k = int(top_idx[idx])
            wv = float(wk[idx])
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                j_vals = k - np.arange(i_start, i_end + 1)
                sg[i_start:i_end + 1] += 2.0 * wv * y[j_vals]

        grad = 2.0 * n * (sg - 2.0 * M_val)
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        Sn = x_new.sum()
        if Sn > 1e-12:
            x_new = x_new / Sn
        else:
            x_new = y.copy()

        x_prev = x_curr
        x_curr = x_new
        t_k = t_new

    sc = compute_score(x_curr)
    if sc < best_sc:
        best_a = x_curr.copy()
    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for scipy."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -400, 400)
    exp_s = np.exp(scaled)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = x[::-1].copy()
    M2 = 2 * n - 1
    M2f = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2f)
    RA = np.fft.rfft(a_rev, M2f)
    cr = np.fft.irfft(PA * RA, M2f)[:M2]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (cr[n - 1:2 * n - 1] / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, deadline=None):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1e4)] * n
    opts = {'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
            'maxfun': maxfun, 'maxls': 30}
    res = minimize(lse_obj_grad, a0, args=(beta,),
                   method='L-BFGS-B', jac=True, bounds=bounds, options=opts)
    a_out = np.maximum(res.x, 0.0)
    So = a_out.sum()
    if So < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / So
    return compute_score(a_out), a_out


def coord_descent(a, n_sweeps, n_top, n_factors, deadline=None):
    """Coordinate descent with geometric line search on top contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            if deadline is not None and time.monotonic() > deadline:
                break
            aj = a[j]
            if aj < 1e-14:
                continue
            factors = np.logspace(np.log10(0.4), np.log10(2.0), n_factors)
            best_val = aj
            best_local = cur_sc
            for f in factors:
                nv = max(0.0, aj * f)
                if abs(nv - aj) < 1e-14:
                    continue
                a[j] = nv
                tmp = a.sum()
                if tmp < 1e-12:
                    a[j] = aj
                    continue
                normalized = a / tmp
                sc = compute_score(normalized)
                if sc < best_local:
                    best_local = sc
                    best_val = nv
                    cur_sc = sc
                a[j] = aj
            a[j] = best_val

        S2 = a.sum()
        if S2 > 1e-12:
            a = a / S2

    return best_a
