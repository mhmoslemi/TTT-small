    p_rev = p[::-1]
    
    A = np.fft.rfft(a, M_fft)
    A_rev = np.fft.rfft(a_rev, M_fft)
    P = np.fft.rfft(p, M_fft)
    P_rev = np.fft.rfft(p_rev, M_fft)
    
    # f1 = convolve(p, a_rev)[n-1:2n-1]
    f1 = np.fft.irfft(P * A_rev, M_fft)[n-1:2*n-1]
    # f2 = convolve(p_rev, a)[n-1:2n-1]
    f2 = np.fft.irfft(P_rev * A, M_fft)[n-1:2*n-1]
    
    grad_lse = f1 + f2
    return L, grad_lse

def score_and_gradient(a, beta):
    """Compute the LSE-smoothed score and full gradient (including the 1/s^2 factor)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, grad_lse = lse_and_grad(a, beta)
    score = 2.0 * n * L / (S * S)
    # d(score)/d(a[j]) = 2n * (grad_lse[j]/S^2 - 2L/(S^3))
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_lse / S2 - 2.0 * L / S3)
    return score, grad
