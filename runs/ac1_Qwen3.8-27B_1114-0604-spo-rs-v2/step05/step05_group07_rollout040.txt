```python
"""
Optimizer for non-negative sequences minimizing
    2 * n * max(convolve(a, a)) / (sum(a)^2).

Algorithmic outline:
- Normalize all searches to sum = 1.
- Use a log-sum-exp smoothed max-convolution objective.
- Optimize both in the non-negative linear parametrization (L-BFGS-B with
  projected gradients) and in a softmax/log parametrization (L-BFGS-B).
- Use adaptive FFT-based gradients, with sparse direct back-propagation when
  the active convolution lags are few.
- Include mirror-descent polishing, perturbed restarts, a multiplicative
  "LP-direction" polishing procedure, and multi-scale upsampling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Global state used only by the SciPy callbacks that cannot receive arguments.
_CURRENT_BETA = 1000.0


def normalize_sequence(a):
    """Clip to [0,1000], make non-negative, and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    nfft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    out = np.fft.irfft(A * A, nfft)[:M]
    if out.size:
        out = np.maximum(out, 0.0)
    return out


def fast_conv2(a, b):
    """Linear convolution of two real arrays via real FFT."""
    na = len(a)
    nb = len(b)
    if na == 0 or nb == 0:
        return np.empty(0, dtype=np.float64)
    M = na + nb - 1
    nfft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    out = np.fft.irfft(A * B, nfft)[:M]
    if out.size:
        out = np.maximum(out, 0.0)
    return out


def cross_c(p, a):
    """
    Return d_i = sum_k p_k * a_{k-i}, for i = 0..len(a)-1.
    Uses direct computation when p is sparse, otherwise FFT correlation.
    """
    n = len(a)
    if n == 0 or len(p) == 0:
        return np.zeros(n, dtype=np.float64)

    p = np.asarray(p, dtype=np.float64)
    maxp = float(np.max(p)) if p.size else 0.0
    if maxp <= 0.0 or not np.isfinite(maxp):
        return np.zeros(n, dtype=np.float64)

    thresh = maxp * 1e-5
    sig = p > thresh
    idx = np.nonzero(sig)[0]

    if idx.size <= 20:
        out = np.zeros(n, dtype=np.float64)
        for k in idx:
            pk = float(p[k])
            js = k - (n - 1)
            if js < 0:
                js = 0
            je = k
            if je > n - 1:
                je = n - 1
            if js <= je:
                start = k - je
                end = k - js
                if 0 <= start and end < n:
                    out[js:je + 1] += pk * (a[start:end + 1][::-1])
        return out

    M = len(p) + n - 1
    nfft = 1 << (M - 1).bit_length()
    P = np.fft.rfft(p, nfft)
    RA = np.fft.rfft(a[::-1], nfft)
    out = np.fft.irfft(P * RA, nfft)[:M]
    return out[n - 1:n - 1 + n]


def lse_L_p(c, beta):
    """
    Stable log-sum-exp approximating max(c):
        L = c_max + (1/beta) log sum exp(beta(c - c_max))
    Returns (L, p), where p_j = exp(beta(c_j-c_max)) / sum.
    """
    beta = float(beta)
    if beta <= 0.0 or not np.isfinite(beta):
        idx = int(np.argmax(c))
        p = np.zeros(len(c), dtype=np.float64)
        p[idx] = 1.0
        return float(c[idx]), p

    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.empty(0, dtype=np.float64)

    cmax = float(np.max(c))
    if cmax <= 0.0:
        return 0.0, np.zeros_like(c, dtype=np.float64)

    scaled = beta * (c - cmax)
    scaled = np.where(scaled > 0.0, 0.0, scaled)
    scaled = np.where(scaled < -50.0, -50.0, scaled)
    e = np.exp(scaled)
    z = float(np.sum(e))
    if z <= 1e-300:
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return cmax, p

    p = e / z
    L = cmax + np.log(z) / beta
    return float(L), p


def smooth_value_grad(b, beta):
    """
    Smoothed objective and gradient with b normalized to sum 1.
    Objective returned is n * LSE_beta(conv(b)), not 2n; correct constant
    is irrelevant for minimization.
    """
    b = np.asarray(b, dtype=np.float64)
    n = len(b)
    if n == 0:
        return 0.0, np.empty(0, dtype=np.float64)

    c = fast_conv(b)
    L, p = lse_L_p(c, beta)
    val = float(n * L)
    cross = cross_c(p, b)
    grad = float(n) * (2.0 * cross)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return val, grad


def score_exact(a):
    """Exact internal score corresponding to evaluate_sequence for nonnegative a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = fast_conv(a)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def objective_gradient_linear(x, beta):
    """
    Objective/gradient for L-BFGS in linear coordinates.
    The function is the smoothed objective restricted to the simplex:
        F(x) = n * LSE_beta(conv(x / sum(x))).
    Gradient is projected onto the tangent space of the simplex.
    """
    x = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.0)
    n = len(x)
    if n == 0:
        return np.inf, np.empty(0, dtype=np.float64)

    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-15:
        return 1e100, np.zeros(n, dtype=np.float64)

    b = x / s
    val, gb = smooth_value_grad(b, beta)
    dot = float(np.dot(b, gb))
    grad = (gb - b * dot) / s
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def _cb_linear(x):
    return objective_gradient_linear(x, _CURRENT_BETA)


def softmax_seq(theta):
    """Stable softmax for a 1D real vector."""
    t = np.asarray(theta, dtype=np.float64)
    if t.size == 0:
        return np.empty(0, dtype=np.float64)
    if not np.all(np.isfinite(t)):
        t = np.where(np.isfinite(t), t, 0.0)
    tmax = float(np.max(t))
    e = np.exp(np.clip(t - tmax, -50.0, 0.0))
    s = float(np.sum(e))
    if s <= 1e-300:
        return np.ones(t.size, dtype=np.float64) / float(t.size)
    return e / s


def linear_to_log(a):
    """Convert nonnegative sequence to log-softmax parameters."""
    b = normalize_sequence(a)
    p = np.clip(b, 1e-300, 1.0)
    th = np.log(p)
    th -= float(np.max(th))
    return th


def objective_gradient_log(theta, beta):
    """
    Objective/gradient for L-BFGS in softmax/log parameters.
    a = softmax(theta), sum a = 1.
    """
    theta = np.asarray(theta, dtype=np.float64)
    if not np.all(np.isfinite(theta)):
        theta = np.where(np.isfinite(theta), theta, 0.0)
    n = len(theta)
    if n == 0:
        return np.inf, np.empty(0, dtype=np.float64)

    b = softmax_seq(theta)
    val, gb = smooth_value_grad(b, beta)

    dot = float(np.dot(b, gb))
    gt = b * (gb - dot)

    gmax = float(np.max(np.abs(gt))) if n else 0.0
    if not np.isfinite(gmax):
        gt = np.zeros(n, dtype=np.float64)
    elif gmax > 1e6:
        gt = gt * (1e6 / gmax)

    if not np.all(np.isfinite(gt)):
        gt = np.zeros(n, dtype=np.float64)

    return float(val), gt


def _cb_log(x):
    return objective_gradient_log(x, _CURRENT_BETA)


def estimate_linear_eval_time(a, beta, trials=2):
    """Rough timing for one objective/gradient evaluation."""
    a = normalize_sequence(a)
    n = len(a)
    beta_use = float(min(max(beta, 0), max(1000.0, float(n))))
    t0 = time.monotonic()
    for _ in range(int(max(1, trials))):
        _, _ = objective_gradient_linear(a, beta_use)
    dt = (time.monotonic() - t0) / float(max(1, trials))
    if not np.isfinite(dt) or dt <= 0.0:
        dt = 0.002
    return float(min(1.0, max(1e-4, dt)))


def run_linear_lbfgs(a0, beta, alloc_s, maxfun_cap=8000):
    """Run L-BFGS-B on the linear normalized objective for a time allocation."""
    if alloc_s < 8.0:
        return None

    a0 = normalize_sequence(a0)
    n = len(a0)
    if n == 0:
        return None

    dt = estimate_linear_eval_time(a0, beta, trials=2)
    maxfun = int(min(maxfun_cap, max(500, float(alloc_s) / (dt * 1.8))))
    maxiter = int(min(8000, max(400, maxfun)))

    global _CURRENT_BETA
    _CURRENT_BETA = float(beta)

    x0 = a0.copy()
    bounds = [(0.0, 1000.0)] * n
    opts = {
        'ftol': 1e-14,
        'gtol': 1e-8,
        'maxiter': maxiter,
        'maxfun': maxfun,
        'maxls': 20,
    }

    try:
        res = minimize(_cb_linear, x0, method='L-BFGS-B', jac=True,
                       bounds=bounds, options=opts)
        x = np.asarray(res.x, dtype=np.float64)
    except Exception:
        return None

    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-12:
        x = a0
    else:
        x = x / s

    sc = score_exact(x)
    if not np.isfinite(sc):
        return None
    return float(sc), x

def interpolate_to(a, n):
    """Linearly resample a on [0,1] to length n and normalize."""
    m = len(a)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return normalize_sequence(a)
    a = np.asarray(a, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    return normalize_sequence(out)


def perturb(a, rng, scale=0.05):
    """Generate a normalized perturbation of a using several random strategies."""
    a = normalize_sequence(a)
    n = len(a)
    if n == 0:
        return a

    strategy = int(rng.integers(0, 6))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    if strategy == 0:
        a2 = a * (1.0 + float(scale) * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * float(rng.random())
        phase = 2.0 * np.pi * float(rng.random())
        amp = float(scale) * (1.0 + float(rng.random()))
        a2 = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        nblocks = 3 + int(rng.integers(0, 8))
        bs = max(1, n // nblocks)
        a2 = a.copy()
        did = False
        for s in range(0, n, bs):
            e = min(s + bs, n)
            if e > s:
                a2[s:e] *= 1.0 + 2.0 * float(scale) * (0.5 - float(rng.random()))
                did = True
        if not did:
            a2 = a.copy()
    elif strategy == 3:
        # Reduce high contributors to the current main peak and redistribute.
        c = fast_conv(a)
        if c.size:
            k = int(np.argmax(c))
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                a2 = a.copy()
                idx = np.arange(js, je + 1)
                contrib = a[idx]
                top_count = max(1, min(max(5, n // 50), len(idx)))
                top_order = np.argsort(contrib)[::-1][:top_count]
                for li in top_order:
                    a2[js + li] *= 1.0 - 0.05 * float(rng.random())
                s2 = float(a2.sum())
                if s2 > 1e-12:
                    deficit = float(a.sum()) - s2
                    bottom_order = np.argsort(contrib)[:max(1, top_count)]
                    per = deficit / float(len(bottom_order))
                    for li in bottom_order:
                        a2[js + li] += per
            else:
                a2 = a.copy()
        else:
            a2 = a.copy()
    elif strategy == 4:
        a2 = a * np.exp(2.0 * float(scale) * rng.standard_normal(n))
    else:
        nsel = max(1, n // 4)
        sel = rng.choice(n, size=nsel, replace=False)
        a2 = a.copy()
        a2[sel] *= 1.0 + 3.0 * float(scale) * (float(rng.random(nsel)) - 0.5)

    a2 = np.clip(a2, 0.0, 1000.0)
    s = float(a2.sum())
    if s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a2 / s


def choose_betas(n, count=8):
    """Choose an increasing beta schedule for the LSE smoother."""
    n = max(1, int(n))
    cands = [
        500,
        max(500, int(0.1 * n)),
        max(500, int(0.5 * n)),
        int(max(500.0, float(n))),
        int(3.0 * n),
        int(10.0 * n),
        int(30.0 * n),
        int(100.0 * n),
        int(300.0 * n),
    ]
    cands = [max(500, int(v)) for v in cands if np.isfinite(v)]
    cands = [v for v in cands if v <= 5000000]
    cands.sort()

    uniq = []
    for v in cands:
        if not uniq or v > uniq[-1]:
            uniq.append(v)

    if not uniq:
        uniq = [1000]

    if len(uniq) > count:
        idxs = np.linspace(0, len(uniq) - 1, int(count)).astype(int)
        uniq = [uniq[int(i)] for i in idxs]

    return uniq


def mirror_polish(a, beta, iters, eta0=0.05):
    """Multiplicative/mirror descent polish on the simplex using LSE weights."""
    a = normalize_sequence(a)
    n = len(a)
    best = a.copy()
    best_sc = score_exact(best)

    eta = float(eta0)
    for _ in range(int(max(0, iters))):
        c = fast_conv(a)
        L, p = lse_L_p(c, beta)
        cross = cross_c(p, a)
        gb = float(n) * (2.0 * cross)
        avg = float(np.dot(a, gb))
        gt = gb - a * avg

        exp_arg = np.clip(-eta * gt, -5.0, 5.0)
        a = a * np.exp(exp_arg)
        s = float(a.sum())
        if s <= 1e-12:
            a = best.copy()
            break
        a = a / s

        sc = score_exact(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

        eta *= 0.999

    return best, best_sc


def lp_direction(f, target, iters=50, eta=1.0):
    """
    Multiplicative approximation to maximizing sum(g) subject to f*g <= target.
    Returns a normalized nonnegative g.
    """
    n = len(f)
    if n == 0:
        return None
    f = normalize_sequence(f)
    if target <= 0.0 or not np.isfinite(target):
        return np.ones(n, dtype=np.float64) / float(n)

    g = np.ones(n, dtype=np.float64) / float(n)
    best_g = g.copy()
    best_viol = np.inf

    for _ in range(int(max(0, iters))):
        h = fast_conv2(f, g)
        v = h - target
        v = np.maximum(v, 0.0)
        viol = float(np.sum(v))
        if viol < best_viol:
            best_viol = viol
            best_g = g.copy()

        vmax = float(np.max(v)) if v.size else 0.0
        if vmax <= 1e-12 + 1e-12 * target:
            break

        d = cross_c(v, f)
        dmax = float(np.max(d)) if d.size else 0.0
        if dmax <= 1e-30:
            break

        factor = np.exp(np.clip(-float(eta) * d / dmax, -8.0, 0.0))
        g_new = g * factor
        s = float(g_new.sum())
        if s <= 1e-300:
            break
        g_new /= s
        g = g_new

    if float(np.sum(best_g)) <= 1e-300:
        best_g = np.ones(n, dtype=np.float64) / float(n)
    return best_g


def lp_polish(a, budget_s, rng):
    """
    Use LP-direction polishing: find g feasible/low-impact for f*a <= target,
    then mix f toward g and accept only exact improvements.
    """
    if budget_s <= 0.0:
        return normalize_sequence(a), score_exact(a)

    a = normalize_sequence(a)
    best = a.copy()
    best_sc = score_exact(best)
    t0 = time.monotonic()

    eps_list = [0.002, 0.005, 0.01, 0.02, 0.05, 0.10, 0.20]
    mix_list = [0.05, 0.10, 0.20, 0.40, 0.70, 1.0]
    iters = 45
    eta = 1.0
    outer = 0

    while (time.monotonic() - t0) < budget_s - 5.0:
        c = fast_conv(a)
        M0 = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(M0) or M0 <= 0.0:
            break

        improved = False
        for eps in eps_list:
            if (time.monotonic() - t0) > budget_s - 5.0:
                break
            target = M0 * (1.0 - float(eps))
            if target <= 0.0:
                target = 1e-300
            g = lp_direction(a, target, iters=iters, eta=eta)
            if g is None:
                continue

            for mix in mix_list:
                cand = (1.0 - float(mix)) * a + float(mix) * g
                cand = normalize_sequence(cand)
                sc = score_exact(cand)
                if np.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best = cand.copy()
                    a = cand.copy()
                    improved = True
                    print(f"    [lp] mix={mix:.2f} eps={eps:.4f} score={sc:.10f}")
                    break
            if improved:
                break

        if not improved:
            ap = perturb(best, rng, 0.04)
            sc = score_exact(ap)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = ap.copy()
                a = ap.copy()
            else:
                break

        outer += 1
        if outer > 25:
            break

    if float(np.sum(best)) <= 1e-12:
        best = a.copy()
    return best, best_sc


def refine_length(a0, deadline, rng, tag, schedule):
    """
    Refine a sequence of fixed length using a list of (kind, beta) runs.
    kind is 'L' for linear L-BFGS and 'G' for log/softmax L-BFGS.
    """
    a = normalize_sequence(a0)
    n = len(a)
    best = a.copy()
    best_sc = score_exact(best)
    print(f"  [{tag}] start n={n} score={best_sc:.10f}")

    if n == 0:
        return best, best_sc

    now = time.monotonic()
    if schedule and now < deadline - 120.0:
        b0 = int(schedule[0][1]) if schedule else 10 * n
        b_mirror = int(min(max(b0, n), 20 * n))
        m, msc = mirror_polish(a, beta=b_mirror, iters=200, eta0=0.05)
        if msc < best_sc:
            best_sc = msc
            best = m.copy()
            a = m.copy()
            print(f"  [{tag}] mirror polish score={best_sc:.10f}")

    total = len(schedule)
    if total == 0:
        return best, best_sc

    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        return best, best_sc
    base_alloc = float(max(1.0, remaining)) / float(total)

    for idx, item in enumerate(schedule):
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 30.0:
            break

        kind = item[0]
        beta = int(item[1])

        if idx == total - 1:
            alloc = max(30.0, remaining - 5.0)
        else:
            alloc = min(remaining - 5.0, max(30.0, base_alloc * (1.35 if kind == 'G' else 1.0)))
        if alloc < 20.0:
            continue

        if kind == 'L':
            out = run_linear_lbfgs(a, beta, alloc, maxfun_cap=8000)
        elif kind == 'G':
            out = run_log_lbfgs(a, beta, alloc, maxfun_cap=6000)
        else:
            out = None

        if out is not None:
            sc, x = out
            if np.isfinite(sc) and sc + 1e-12 < best_sc:
                best_sc = sc
                best = x.copy()
                a = x.copy()
                print(f"  [{tag}] {kind} beta={beta} NEW BEST score={best_sc:.10f}")
            else:
                print(f"  [{tag}] {kind} beta={beta} no improvement best={best_sc:.10f}")

    # Iterated local search: perturb best and run short L-BFGS.
    while time.monotonic() < deadline - 35.0:
        remaining = deadline - time.monotonic()
        if remaining < 60.0:
            break

        scale = 0.02 + 0.05 * float(rng.random())
        ap = perturb(best, rng, scale)
        sc0 = score_exact(ap)
        if np.isfinite(sc0) and sc0 < best_sc:
            best_sc = sc0
            best = ap.copy()
            a = ap.copy()
            print(f"  [{tag}] perturb NEW BEST score={best_sc:.10f}")

        beta = int(schedule[-1][1]) if schedule else 10 * n
        alloc = min(90.0, 0.45 * remaining)
        if alloc > 25.0 and time.monotonic() < deadline - 60.0:
            out = run_linear_lbfgs(ap, beta, alloc, maxfun_cap=2500)
            if out is not None:
                sc, x = out
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best = x.copy()
                    a = x.copy()
                    print(f"  [{tag}] ILS L-BFGS beta={beta} NEW BEST score={best_sc:.10f}")

    return best, best_sc


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    t_start = time.monotonic()
    budget_s = max(120.0, float(budget_s))
    total_available = max(60.0, float(budget_s) - 20.0)
    deadline = t_start + total_available
    rng = np.random.default_rng(seed)

    # Load parent construction if available.
    parent = None
    try:
        pobj = globals().get('height_sequence_1')
        if pobj is not None:
            p = np.asarray(pobj, dtype=np.float64).ravel()
            if p.size > 0:
                p = np.clip(p, 0.0, 1000.0)
                ps = float(np.sum(p))
                if ps > 1e-12:
                    parent = p / ps
    except Exception:
        parent = None

    # Choose initial working length and start.
    if parent is not None:
        lp = len(parent)
        if lp < 2048:
            N0 = 8192
            a0 = interpolate_to(parent, N0)
        elif lp <= 12000:
            N0 = lp
            a0 = parent.copy()
        else:
            N0 = 16384
            a0 = interpolate_to(parent, N0)
    else:
        N0 = 8192
        a0 = np.ones(N0, dtype=np.float64) / float(N0)

    best_a = normalize_sequence(a0)
    best_sc = score_exact(best_a)
    fallback_a = best_a.copy()
    print(f"Initial length N0={N0}, score={best_sc:.10f}")

    # Without parent, try a few quick structured starts.
    if parent is None:
        c1 = perturb(best_a, rng, 0.25)
        s1 = score_exact(c1)
        if np.isfinite(s1) and s1 < best_sc:
            best_sc = s1
            best_a = c1.copy()
            fallback_a = best_a.copy()
            print(f"Random start improved: {best_sc:.10f}")

    large_start = None
    large_sc = np.inf

    # ------------------------------------------------------------------
    # Phase 0: refine initial length.
    # ------------------------------------------------------------------
    phase0_deadline = t_start + 0.32 * total_available
    if time.monotonic() < phase0_deadline:
        betas0 = choose_betas(N0, count=8)
        schedule0 = [('L', int(b)) for b in betas0]
        schedule0 += [('G', int(b)) for b in betas0[-4:]]
        new_a, new_sc = refine_length(best_a, phase0_deadline, rng, 'N0', schedule0)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = new_a.copy()
            fallback_a = best_a.copy()

    # ------------------------------------------------------------------
    # Phase 1: larger length.
    # ------------------------------------------------------------------
    base = max(N0, 8192)
    np2 = 1 << (max(1, base - 1).bit_length())
    if np2 >= 1.5 * base:
        N1 = np2
    else:
        N1 = 2 * np2
    N1 = min(int(N1), 20480)
    N1 = max(int(N1), int(1.4 * base))

    phase1_deadline = t_start + 0.78 * total_available
    if N1 > N0 and time.monotonic() < phase1_deadline:
        a1 = interpolate_to(best_a, N1)
        betas1 = choose_betas(N1, count=8)
        schedule1 = [('L', int(b)) for b in betas1[:5]]
        schedule1 += [('G', int(b)) for b in betas1[-4:]]
        large_start, large_sc = refine_length(a1, phase1_deadline, rng, 'N1', schedule1)
        if large_sc < best_sc:
            best_sc = large_sc
            best_a = large_start.copy()
            fallback_a = best_a.copy()
            print(f"Length N1={N1} improved global best: {best_sc:.10f}")

    # ------------------------------------------------------------------
    # Phase 2: even larger length if time allows.
    # ------------------------------------------------------------------
    N2 = int(max(20480, min(20480, 1.5 * N1)))
    phase2_deadline = t_start + 0.93 * total_available
    if N2 > best_a.size and time.monotonic() < phase2_deadline - 180.0:
        if (large_start is not None and len(large_start) == N1 and large_sc < best_sc + 0.05):
            a2 = interpolate_to(large_start, N2)
            tag2 = 'N2_from_N1'
        else:
            a2 = interpolate_to(best_a, N2)
            tag2 = 'N2_from_best'
        betas2 = choose_betas(N2, count=6)
        schedule2 = [('L', int(b)) for b in betas2[:4]]
        schedule2 += [('G', int(b)) for b in betas2[-3:]]
        new_a, new_sc = refine_length(a2, phase2_deadline, rng, tag2, schedule2)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = new_a.copy()
            fallback_a = best_a.copy()

    # ------------------------------------------------------------------
    # Final polishing.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 45.0:
        lp_budget = max(0.0, min(300.0, deadline - time.monotonic() - 20.0))
        if lp_budget > 30.0:
            lp_a, lp_sc = lp_polish(best_a, lp_budget, rng)
            if np.isfinite(lp_sc) and lp_sc < best_sc:
                best_sc = lp_sc
                best_a = lp_a.copy()
                fallback_a = best_a.copy()

    if time.monotonic() < deadline - 60.0:
        beta_final = int(min(5000000, max(1000, 30 * len(best_a))))
        m_a, m_sc = mirror_polish(best_a, beta=beta_final, iters=300, eta0=0.02)
        if np.isfinite(m_sc) and m_sc < best_sc:
            best_sc = m_sc
            best_a = m_a.copy()
            fallback_a = best_a.copy()

    if time.monotonic() < deadline - 25.0:
        alloc = max(0.0, min(60.0, deadline - time.monotonic() - 5.0))
        if alloc > 20.0:
            beta_final = int(min(5000000, max(1000, 50 * len(best_a))))
            out = run_linear_lbfgs(best_a, beta_final, alloc, maxfun_cap=2000)
            if out is not None:
                sc, a_out = out
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_out.copy()
                    fallback_a = best_a.copy()

    # If using a very large sequence made no verified improvement, prefer the
    # best previously verified/fallback candidate if it is meaningfully better.
    if fallback_a is not None and float(np.sum(fallback_a)) > 1e-9:
        fsc = score_exact(fallback_a)
        bsc = score_exact(best_a)
        if (not np.isfinite(bsc) or (np.isfinite(fsc) and fsc < bsc)):
            best_a = fallback_a.copy()
            best_sc = fsc

    # Final sanitize and optional verification.
    out = [float(max(0.0, min(1000.0, float(v)))) for v in np.asarray(best_a, dtype=np.float64)]
    if sum(out) < 0.01:
        out = [0.5, 0.5]

    # Call evaluate_sequence only when it is likely to be cheap or very likely to finish in time.
    try:
        if len(out) <= 12000 or time.monotonic() < deadline - 120.0:
            verified = float(evaluate_sequence(out))
            if not np.isfinite(verified):
                fb = [float(max(0.0, min(1000.0, float(v)))) for v in fallback_a]
                if sum(fb) > 0.01:
                    out = fb
                else:
                    out = [0.5, 0.5]
            print(f"FINAL verified={verified:.10f} internal={best_sc:.10f} len={len(out)}")
        else:
            print(f"FINAL internal={best_sc:.10f} len={len(out)} (skipped evaluate for time)")
    except Exception:
        print(f"FINAL internal={best_sc:.10f} len={len(out)} (evaluate_sequence skipped)")

    return out
```
