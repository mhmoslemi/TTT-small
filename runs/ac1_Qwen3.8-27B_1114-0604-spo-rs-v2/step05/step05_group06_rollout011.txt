import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e4, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _coord_descent(a, conv, n, max_sweeps=5, time_limit=None, t_start=None):
    """
    Coordinate descent: for each position i, find optimal a[i] via golden section.
    conv is the current self-convolution (recomputed each sweep).
    """
    a = a.copy().astype(np.float64)
    S = float(a.sum())
    best_sc = _score(a)
    t_start = time.monotonic() if t_start is None else t_start
    deadline = t_start + time_limit if time_limit else np.inf
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Recompute convolution
        conv = _self_conv(a)
        S = float(a.sum())
        
        # Randomize order for diversity
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            ai_old = a[i]
            if ai_old < 1e-15:
                # Try setting to a small positive value
                x_try = S / (10 * n)
                a[i] = x_try
                c_new = _self_conv(a)
                sc_new = _score(a)
                if sc_new < best_sc:
                    best_sc = sc_new
                    S = float(a.sum())
                else:
                    a[i] = 0.0
                continue
            
            # Determine affected convolution positions
            # k = i + j for j = 0..n-1, clipped to [0, 2n-2]
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max < 0:
                continue
            
            j_arr = np.arange(j_max + 1, dtype=np.int64)
            k_arr = i + j_arr  # affected conv positions
            
            # Build B, alpha, beta for: new_conv[k] = B + alpha*x + beta*x^2
            B = conv[k_arr].astype(np.float64).copy()
            alpha = np.zeros(j_max + 1, dtype=np.float64)
            beta = np.zeros(j_max + 1, dtype=np.float64)
            
            # Remove old contribution: 2*ai_old*a[j] for all j
            B -= 2.0 * ai_old * a[j_arr]
            
            # Add new contribution: 2*x*a[j] for j != i
            alpha[:] = 2.0 * a[j_arr]
            
            # Special case: j == i (k = 2i), where old contrib was ai_old^2 not 2*ai_old^2
            if i <= j_max:
                # Fix: we subtracted 2*ai_old*a[i] = 2*ai_old^2, but should have subtracted ai_old^2
                B[i] += ai_old * ai_old  # add back the extra subtraction
                alpha[i] = 0.0
                beta[i] = 1.0
            
            # Unmodified max
            km = 2 * n - 1
            unmod_max = 0.0
            if k_arr[0] > 0:
                unmod_max = max(unmod_max, float(conv[:k_arr[0]].max()))
            if k_arr[-1] < km - 1:
                unmod_max = max(unmod_max, float(conv[k_arr[-1]+1:].max()))
            
            # Golden section search for x in [0, x_hi]
            x_lo = 0.0
            x_hi = max(2.0 * ai_old, 10.0 * S / n)
            
            def _f(x):
                mod_vals = B + alpha * x + beta * x * x
                mval = max(unmod_max, float(mod_vals.max()))
                Sx = S - ai_old + x
                if Sx < 1e-12:
                    return 1e10
                return 2.0 * n * mval / (Sx * Sx)
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            f1 = _f(c1)
            f2 = _f(c2)
            
            for _ in range(40):
                if f1 < f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    f1 = _f(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    f2 = _f(c2)
            
            x_opt = (x_lo + x_hi) / 2.0
            f_opt = _f(x_opt)
            
            if f_opt < best_sc - 1e-12:
                a[i] = max(0.0, x_opt)
                best_sc = f_opt
                S = float(a.sum())
            else:
                a[i] = max(0.0, x_opt)  # still set to optimal (might be same)
                S = float(a.sum())
        
        # End of sweep
    
    return best_sc, a


def _lse_obj_grad(a, beta, k_top):
    """LSE-smoothed objective with top-k active positions."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k_top, M))
    
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    
    z = beta * (d_act - dmax)
    e = np.exp(np.clip(z, -50, 0))
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    
    w_full = np.zeros(M)
    w_full[idx] = w_act
    
    # Cross-correlation: corr[i] = sum_k w_full[k] * a[k-i] (for valid indices)
    # Use FFT
    Nf = _next_pow2(M + n - 1)
    A_rev = np.fft.rfft(a[::-1], Nf)
    W = np.fft.rfft(w_full, Nf)
    full = np.fft.irfft(A_rev * W, Nf)[:M + n - 1]
    corr = full[n - 1:2 * n - 1]
    
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    val = 2.0 * L
    return float(val), grad


def _run_lse(a0, beta, k_top, time_limit, maxfun=500):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    
    est_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_t, 1e-4))
    mf = max(20, min(mf, maxfun, 6000))
    maxiter = max(15, mf // 3)
    PEN = 5e-4
    
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k_top)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_nesterov(a, max_iters=50, deadline=None):
    """Nesterov-accelerated subgradient descent on the max-convolution objective."""
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Nesterov momentum
    y = a.copy()
    beta_mom = 0.0
    step_base = 0.002
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        S = float(y.sum())
        if S < 1e-12:
            break
        c = _self_conv(y)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        
        # Active set: positions within 1% of max
        thr = dmax - 0.01 * max(dmax, 1e-10)
        active = d >= thr
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        # Gradient via cross-correlation
        Nf = _next_pow2(2 * n - 1 + n - 1)
        A_rev = np.fft.rfft(y[::-1], Nf)
        W = np.fft.rfft(w, Nf)
        full = np.fft.irfft(A_rev * W, Nf)[:2 * n - 1 + n - 1]
        corr = full[n - 1:2 * n - 1]
        
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        # Adaptive step size with backtracking
        eta = step_base / max(gnorm, 1e-10)
        for _ in range(20):
            q = y - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                eta *= 0.5
                continue
            q = q / Sq
            sc_q = _score(q)
            if sc_q <= best_sc + eta * gnorm * 1.5:
                break
            eta *= 0.5
        
        t = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        beta_mom = (1.0 + t) / (1.0 + t + 1.0)  # Nesterov
        beta_mom = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / (1.0 + np.sqrt(1.0 + 4.0 * (it+1) * (it+1)))
        
        new_y = q + beta_mom * (q - y)
        new_y = np.clip(new_y, 0.0, None)
        if new_y.sum() < 1e-8:
            new_y = q
        else:
            new_y = new_y / new_y.sum()
        
        y = new_y
        sc_y = _score(y)
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()
    
    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with flat-ish power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs) * 10.0) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = np.clip(sig, 0, None)
    if sig.sum() < 1e-10:
        sig = np.ones(n)
    return sig / sig.sum()


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(61, max(5, n // 100))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig_k = max(1.0, c / 3.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig_k * sig_k))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: coordinate descent + LSE hybrid with Nesterov polishing.
    Uses exact 1D coordinate updates as the primary optimizer, combined with
    LSE-smoothed L-BFGS for broader convergence and Nesterov subgradient for
    final polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e4, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass
    
    print(f"Parent loaded: {parent is not None}", flush=True)
    
    # Target length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192
    
    # If parent length is odd or unusual, try to use it or round to power of 2
    print(f"Target n = {target_n}", flush=True)
    
    # --- Phase 1: Generate candidates ---
    candidates = []
    
    if parent is not None:
        candidates.append(('parent', _norm(parent)))
        # Reversed
        candidates.append(('parent_rev', _norm(parent[::-1])))
        # Smoothed perturbations
        for amp in (0.05, 0.1, 0.2):
            candidates.append((f'pn{amp}', _smooth_perturb(parent, rng, amp)))
    
    # Spectral inits
    for i in range(4):
        candidates.append((f'spec{i}', _spectral_init(target_n, rng)))
    
    # Evaluate candidates
    scored = []
    for name, arr in candidates:
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))
    
    scored.sort(key=lambda x: x[0])
    print(f"Candidates: {len(scored)}, top scores: {[f'{s:.6f}' for s,_,_ in scored[:5]]}", flush=True)
    
    best_score = scored[0][0] if scored else 2.0
    best_seq = scored[0][2].copy() if scored else _norm(np.ones(target_n))
    
    # --- Phase 2: Main optimization loop ---
    # Strategy: alternate between coordinate descent and LSE, with Nesterov polish
    
    # Select top 2-3 candidates to optimize
    n_opt = min(3, len(scored))
    time_per_candidate = (deadline - t_start - 100.0) / n_opt
    
    for ci in range(n_opt):
        if time.monotonic() > deadline - 60:
            break
        cand_sc, cand_name, cand_arr = scored[ci]
        cand_deadline = t_start + (ci + 1) * time_per_candidate
        cand_deadline = min(cand_deadline, deadline - 60)
        
        print(f"\n=== Optimizing candidate {ci+1}: '{cand_name}', init={cand_sc:.8f} ===", flush=True)
        
        a_work = _norm(cand_arr)
        sc_work = cand_sc
        n = len(a_work)
        
        # Iteration: alternate CD and LSE
        num_iters = max(3, int((cand_deadline - time.monotonic() - 30) / 30))
        
        for it in range(num_iters):
            if time.monotonic() > cand_deadline - 10:
                break
            
            rem = cand_deadline - time.monotonic() - 10
            if rem < 5:
                break
            
            # Part A: Coordinate descent
            cd_time = min(rem * 0.5, 60.0)
            sc_cd, a_cd = _coord_descent(a_work, _self_conv(a_work), n,
                                         max_sweeps=2, time_limit=cd_time,
                                         t_start=time.monotonic())
            if sc_cd < sc_work:
                sc_work = sc_cd
                a_work = a_cd
                if sc_work < best_score:
                    best_score = sc_work
                    best_seq = a_work.copy()
                    print(f"  CD it{it}: NEW BEST {best_score:.8f}", flush=True)
            
            if time.monotonic() > cand_deadline - 10:
                break
            
            # Part B: LSE optimization
            lse_time = min(rem * 0.3, 30.0)
            # Increase beta and k over iterations
            beta = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000][min(it, 7)]
            k_top = [3, 5, 10, 20, 50, 100, 200, 500][min(it, 7)]
            sc_lse, a_lse = _run_lse(a_work, beta, k_top, lse_time, maxfun=300)
            if sc_lse < sc_work:
                sc_work = sc_lse
                a_work = a_lse
                if sc_work < best_score:
                    best_score = sc_work
                    best_seq = a_work.copy()
                    print(f"  LSE it{it}: NEW BEST {best_score:.8f}", flush=True)
        
        # Nesterov polish
        if time.monotonic() < cand_deadline - 5:
            sc_ne, a_ne = _subgrad_nesterov(a_work, max_iters=30,
                                           deadline=cand_deadline - 2)
            if sc_ne < sc_work:
                sc_work = sc_ne
                a_work = a_ne
                if sc_work < best_score:
                    best_score = sc_work
                    best_seq = a_work.copy()
                    print(f"  Nesterov: NEW BEST {best_score:.8f}", flush=True)
    
    # --- Phase 3: Final intensive polishing on global best ---
    print(f"\n=== Final polish: score={best_score:.8f} ===", flush=True)
    final_deadline = deadline - 10
    
    # Multiple CD sweeps
    while time.monotonic() < final_deadline - 20:
        rem = final_deadline - time.monotonic() - 20
        if rem < 5:
            break
        sc_cd, a_cd = _coord_descent(best_seq, _self_conv(best_seq), len(best_seq),
                                     max_sweeps=2, time_limit=min(rem, 40.0),
                                     t_start=time.monotonic())
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd
            print(f"  Final CD: {best_score:.8f}", flush=True)
        else:
            break
    
    # Nesterov
    if time.monotonic() < final_deadline - 5:
        sc_ne, a_ne = _subgrad_nesterov(best_seq, max_iters=30,
                                       deadline=final_deadline - 3)
        if sc_ne < best_score:
            best_score = sc_ne
            best_seq = a_ne
            print(f"  Final Nesterov: {best_score:.8f}", flush=True)
    
    # More CD
    if time.monotonic() < final_deadline - 3:
        sc_cd, a_cd = _coord_descent(best_seq, _self_conv(best_seq), len(best_seq),
                                     max_sweeps=1, time_limit=final_deadline - time.monotonic() - 2,
                                     t_start=time.monotonic())
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd
            print(f"  Final CD2: {best_score:.8f}", flush=True)
    
    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S
    
    result = [float(x) for x in out]
    
    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score
    
    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return result
