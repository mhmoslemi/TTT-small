def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_hyp(a, n_iters=300, alpha_init=1.0, alpha_end=20.0, t_relax=0.5):
    """Hyperbolic spectral compression with error feedback and relaxation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress
        relax = t_relax * (1.0 - 0.5 * progress)  # decrease relaxation over time

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        if P_mean < 1e-30:
            break

        # Hyperbolic compression: P_new = P / (1 + alpha * P / P_mean)
        P_new = P / (1.0 + alpha * (P / P_mean))
        P_new = np.maximum(P_new, 1e-30)

        # Preserve phase, update magnitude
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio

        a_raw = np.fft.irfft(A_mod, M_fft)[:n]

        # Error feedback: keep the "error" from projection
        a_proj = np.maximum(a_raw, 0.0)
        S_proj = float(a_proj.sum())
        if S_proj < 1e-12:
            break
        a_proj = a_proj / S_proj
        error = a_proj - a_raw  # error in spatial domain

        # Feed error back in frequency domain
        E = np.fft.rfft(error, M_fft)
        A_new = A_mod + E

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Relaxation
        a = (1.0 - relax) * a + relax * a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_multi_thresh(a, n_iters=300, n_levels=5, strength_init=0.3, strength_end=0.8):
    """Multi-level threshold compression with progressive strength."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        strength = strength_init + (strength_end - strength_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        scale = np.ones_like(P)
        for lv in range(n_levels):
            frac = 0.08 * (lv + 1) / n_levels
            thresh = np.quantile(P, 1.0 - frac)
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                scale[above] *= (1.0 - strength + strength * np.sqrt(np.maximum(ratio, 0.001)))

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B on the LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def top_m_subgradient(a, n_iters=3000, top_m=20, step_init=0.02, decay=0.9999):
    """Adaptive top-m subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if it % 200 == 0:
            sc_lb = max(sc_lb, best_sc - 0.01)

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        adaptive_step = max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12)
        step = min(step, adaptive_step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
        if step < 1e-15:
            step = 1e-8  # restart step

    return best_a


def coordinate_refine(a, n_trials=5000):
    """Try small perturbations of individual coordinates to improve score."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)

    c = fast_conv(a)
    S = float(a.sum())
    M_val = float(c.max())

    for trial in range(n_trials):
        # Find peak lag
        peak_k = int(np.argmax(c))
        # Find contributing indices
        j_start = max(0, peak_k - n + 1)
        j_end = min(peak_k, n - 1)
        # Pick a random index in the contributing range
        idx = rng.integers(j_start, j_end + 1)

        # Try small positive and negative perturbation
        for delta in [0.001, 0.005, 0.01, -0.001, -0.005, -0.01]:
            a_test = a.copy()
            a_test[idx] = max(0.0, a_test[idx] + delta)
            S_test = float(a_test.sum())
            if S_test < 1e-12:
                continue
            c_test = fast_conv(a_test)
            sc_test = 2.0 * n * float(c_test.max()) / (S_test * S_test)
            if sc_test < best_sc - 1e-12:
                best_sc = sc_test
                best_a = a_test.copy()
                a = a_test.copy()
                c = c_test
                S = S_test
                break

    return best_a

            sc = compute_score(a_h)
            update_best(sc, a_h, f"HYP n={n} i={i}")

            # Flat target
            if time.monotonic() < phase0_end:
                a_f = fienup_flat_target(a0, n_iters=150, alpha_init=0.1, alpha_end=0.9)
                sc2 = compute_score(a_f)
                update_best(sc2, a_f, f"FLAT n={n} i={i}")

            # Multi-thresh
            if time.monotonic() < phase0_end:
                a_m = fienup_multi_thresh(a0, n_iters=100, n_levels=6, strength_init=0.3, strength_end=0.7)
                sc3 = compute_score(a_m)
                update_best(sc3, a_m, f"MULTI n={n} i={i}")

    print(f"After Phase 0: best = {best_score:.8f}")

    # ==================================================================
    # Phase 1: Extended spectral refinement from best (15%)
    # ==================================================================
    phase1_end = t_start + 0.30 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Deep spectral refinement (deadline t={phase1_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    S = a_cur.sum()
    if S < 1e-12:
        a_cur = np.ones(len(a_cur)) / len(a_cur)
    a_cur = a_cur / S

    chains = [
        ("hyp_deep", lambda x: fienup_hyp(x, 400, alpha_init=2.0, alpha_end=50.0, t_relax=0.4)),
        ("flat_deep", lambda x: fienup_flat_target(x, 400, alpha_init=0.3, alpha_end=0.99)),
        ("multi_deep", lambda x: fienup_multi_thresh(x, 400, n_levels=8, strength_init=0.4, strength_end=0.9)),
        ("hyp_aggr", lambda x: fienup_hyp(x, 300, alpha_init=5.0, alpha_end=100.0, t_relax=0.3)),
    ]

    for name, func in chains:
        if time.monotonic() > phase1_end:
            break
        a_out = func(a_cur)
        sc = compute_score(a_out)
        improved = update_best(sc, a_out, f"P1-{name}")
        if improved:
            a_cur = a_out.copy()
            S = a_cur.sum()
            if S > 1e-12:
                a_cur = a_cur / S

    # Chain: alternate between different spectral methods
    if time.monotonic() < phase1_end - 30:
        for cycle in range(3):
            if time.monotonic() > phase1_end:
                break
            a_out = fienup_hyp(a_cur, 200, alpha_init=3.0, alpha_end=30.0, t_relax=0.5)
            sc = compute_score(a_out)
            update_best(sc, a_out, f"P1-CYCLE{cycle}-hyp")
            if a_out.sum() > 1e-12:
                a_cur = a_out / a_out.sum()
            a_out = fienup_flat_target(a_cur, 200, alpha_init=0.2, alpha_end=0.95)
            sc = compute_score(a_out)
            update_best(sc, a_out, f"P1-CYCLE{cycle}-flat")
            if a_out.sum() > 1e-12:
                a_cur = a_out / a_out.sum()

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS with progressively increasing beta (45%)
    # ==================================================================
    phase2_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS (deadline t={phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 1e-12:
        a_work = a_work / S

    # Progressive beta schedule
    beta_schedule = [
        (50, 5000),       # (beta, maxiter)
        (200, 8000),
        (500, 10000),
        (1000, 12000),
        (3000, 15000),
        (10000, 15000),
        (30000, 10000),
        (100000, 5000),
    ]

    for beta, maxiter in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break

        # Scale maxiter by remaining time
        est_time_per_iter = len(a_work) / 500000.0  # rough estimate
        maxiter_use = min(maxiter, max(500, int(remaining * 0.8 / max(est_time_per_iter, 0.001))))
        maxfun_use = maxiter_use * 30

        print(f"  beta={beta}, maxiter={maxiter_use}, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter_use, maxfun=maxfun_use)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            S = a_opt.sum()
            if S > 1e-12:
                a_work = a_opt / S
        else:
            # Still update a_work if it's closer to optimum in LSE sense
            S = a_opt.sum()
            if S > 1e-12:
                a_work = a_opt / S

    # L-BFGS from parent at native length if different
    if parent is not None and len(parent) != len(a_work):
        if time.monotonic() < phase2_end - 60:
            a_p = parent.copy()
            S = a_p.sum()
            if S > 1e-12:
                a_p = a_p / S
            for beta in [500, 5000, 50000]:
                if time.monotonic() > phase2_end:
                    break
                remaining = phase2_end - time.monotonic()
                if remaining < 30:
                    break
                maxiter_use = min(10000, int(remaining * 0.7 / max(len(a_p) / 500000.0, 0.001)))
                sc, a_opt = run_lbfgs(a_p, beta, maxiter=maxiter_use, maxfun=maxiter_use * 30)
                improved = update_best(sc, a_opt, f"LBFGS-PARENT b={beta}")
                if improved:
                    S = a_opt.sum()
                    if S > 1e-12:
                        a_p = a_opt / S

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Subgradient refinement (10%)
    # ==================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Subgradient (deadline t={phase3_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 1e-12:
        a_work = a_work / S

    for top_m in [5, 10, 20, 50]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        n_iters = min(3000, int(remaining * 100))
        a_sg = top_m_subgradient(a_work, n_iters=n_iters, top_m=top_m, step_init=0.01, decay=0.99995)
        sc = compute_score(a_sg)
        improved = update_best(sc, a_sg, f"SG m={top_m}")
        if improved:
            S = a_sg.sum()
            if S > 1e-12:
                a_work = a_sg / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish - perturbation + L-BFGS + coordinate (10%)
    # ==================================================================
    print(f"\n=== Phase 4: Final polish (deadline t={deadline - t_start:.0f}s) ===")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 1e-12:
        a_work = a_work / S

    # Perturbation restarts
    for trial in range(5):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        noise_scales = [0.005, 0.01, 0.02, 0.03, 0.05]
        ns = noise_scales[trial % len(noise_scales)]
        a_pert = a_work * (1.0 + ns * rng.standard_normal(len(a_work)))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_p = a_pert.sum()
        if S_p < 1e-12:
            continue
        a_pert = a_pert / S_p

        # Quick spectral shape
        a_shaped = fienup_flat_target(a_pert, n_iters=100, alpha_init=0.2, alpha_end=0.8)
        sc = compute_score(a_shaped)
        update_best(sc, a_shaped, f"PERT-{trial}")

        # L-BFGS polish
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                maxiter_use = min(5000, int(remaining * 0.5 / max(len(a_shaped) / 500000.0, 0.001)))
                sc2, a_opt = run_lbfgs(a_shaped, 10000, maxiter=max(1000, maxiter_use), maxfun=max(30000, maxiter_use * 30))
                update_best(sc2, a_opt, f"PERT-{trial}-LBFGS")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_work = best_a.copy()
            S = a_work.sum()
            if S > 1e-12:
                a_work = a_work / S
            maxiter_use = min(5000, int(remaining * 0.7 / max(len(a_work) / 500000.0, 0.001)))
            sc, a_opt = run_lbfgs(a_work, 50000, maxiter=max(2000, maxiter_use), maxfun=max(60000, maxiter_use * 30))
            update_best(sc, a_opt, "FINAL-LBFGS-50k")

    # Coordinate refinement
    if time.monotonic() < deadline - 5:
        a_coord = coordinate_refine(best_a, n_trials=3000)
        sc = compute_score(a_coord)
        update_best(sc, a_coord, "COORD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
