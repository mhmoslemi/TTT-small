```
def _project_simplex(v):
   v = np.asarray(v,dtype=np.float64).ravel()
   v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=-1.0)
   n=v.size
   if n==0: return v
   x = np.sort(v)[::-1]
   cssum = np.cumsum(x)
   rho = np.arange(1,n+1)
   cond = (x - cssum/rho) > 0
   if not cond.any():
       # theta = max(v)
       u = v - max(v); u = np.maximum(u,0); s=u.sum(); if s<=0: u=np.ones(n)/n; else u/=s; return u
   theta = cssum[cond][-1]/rho[cond][-1]
   u = np.maximum(v - theta, 0.0)
   s = u.sum()
   if s <= 0 or not finite: u=np.ones(n)/n
   else: u /= s
   return u
```

```
def _fft_self(a):
    a = np.nan_to_num(np.asarray(a,dtype=np.float64).ravel(), nan=0.0, posinf=1e6, neginf=0.0)
    n = a.size
    if n == 0:

    M = 2*n - 1
    N = 1 << (M - 1).bit_length()
    P = np.fft.rfft(a, N)
    c = np.fft.irfft(P * P, N)[:M]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
```

```python
"""
Search for a non-negative sequence minimizing
    2 * n * max(k) conv(a, a)[k] / sum(a)^2.

The implementation combines several complementary local methods:
  * full-length log-sum-exp smoothing of the max-convolution objective,
  * projected LSE subgradient/BB polishing on the probability simplex,
  * L-BFGS polishing in a centered log-softmax parameter space,
  * a local linear-programming direction method that tries to increase
    the sum while respecting active convolution constraints,
  * exact top-k subgradient polishing,
  * multilevel warm-starts and diverse initializations.

The main new idea over the previous program is the LP direction step
and the full LSE / centered-log-softmax polish, plus a stronger
multi-length search schedule.
"""

import numpy as np
import time
from scipy.optimize import minimize

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _first_key(x):
    return x[0]


def _clean_array(a):
    return np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(),
                         nan=0.0, posinf=1e6, neginf=0.0)


def _norm(a, target=1.0):
    a = np.maximum(_clean_array(a), 0.0)
    if a.size == 0:
        return a
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fft_self(a):
    a = _clean_array(a)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64), None, 0
    M = 2 * n - 1
    N = _next_pow2(M)
    P = np.fft.rfft(a, N)
    c = np.fft.irfft(P * P, N)[:M]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    return c, P, N


def _corr_from_P(P, w, N, n):
    w = _clean_array(w)
    M = 2 * n - 1
    if w.size != M:
        full = np.zeros(M, dtype=np.float64)
        m = min(w.size, M)
        if m > 0:
            full[:m] = w[:m]
        w = full
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(P * np.conj(W), N)[:n]
    return np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    a = np.maximum(_clean_array(a), 0.0)
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c, _, _ = _fft_self(a)
    if c.size == 0:
        return np.inf
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    if mc < 0.0 and mc > -1e-12:
        mc = 0.0
    return float(2.0 * n * mc / (S * S))


def _lse_compute(a, beta):
    """
    Returns (actual_score, lse_value, gradient_of_lse_value).
    The LSE value is a smooth approximation of the official score.
    """
    a = np.maximum(_clean_array(a), 0.0)
    n = a.size
    if n == 0:
        return np.inf, np.inf, np.empty(0, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.inf, np.zeros(n, dtype=np.float64)

    c, P, N = _fft_self(a)
    if c.size == 0:
        return np.inf, np.inf, np.zeros(n, dtype=np.float64)

    d = float(n) * c / (S * S)
    dmax = float(d.max())
    if not np.isfinite(dmax) or dmax <= 0.0:
        return np.inf, np.inf, np.zeros(n, dtype=np.float64)

    beta = max(1e-6, float(beta))
    z = beta * (d - dmax)
    e = np.exp(z)
    se = float(e.sum())

    if np.isfinite(se) and se > 0.0:
        w = e / se
        logse = np.log(se)
    else:
        tol = max(1e-12, 1e-12 * abs(dmax))
        active = d >= (dmax - tol)
        if not active.any():
            active[int(np.argmax(d))] = True
        w = active.astype(np.float64)
        w /= w.sum()
        logse = np.log(float(w.shape[0]))

    corr = _corr_from_P(P, w, N, n)
    LDS = float(w @ d)
    val = 2.0 * (dmax + logse / beta)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(2.0 * dmax), float(val), grad


def _project_simplex(v):
    v = _clean_array(v)
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    cssum = np.cumsum(x)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (x - cssum / rho) > 0.0
    if np.any(cond):
        idx = int(np.nonzero(cond)[0][-1])
        theta = cssum[idx] / rho[idx]
    else:
        theta = float(v.max())
    u = np.maximum(v - theta, 0.0)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return u / s


def _softmax_centered(y):
    y = _clean_array(y)
    n = y.size
    if n == 0:
        return y
    y = y - float(y.mean())
    rng = float(y.max() - y.min())
    if np.isfinite(rng) and rng > 100.0:
        y = y * (100.0 / rng)
    m = float(y.max())
    a = np.exp(y - m)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def _init_y(a):
    a = _norm(a)
    a = np.maximum(a, 1e-12)
    y = np.log(a)
    y -= float(y.mean())
    return np.clip(y, -50.0, 50.0)


def _lse_lsm_obj(y, beta):
    y = _clean_array(y)
    n = y.size
    if n == 0:
        return 0.0, y
    a = _softmax_centered(y)
    actual, val, g = _lse_compute(a, beta)
    if not np.isfinite(val):
        return 1e10, np.zeros(n, dtype=np.float64)

    alpha = float(g @ a)
    gy = a * (g - alpha)
    gy -= float(gy.mean())

    reg = 1e-8
    val = val + 0.5 * reg * float(y @ y)
    gy = gy + reg * y
    gy = np.nan_to_num(gy, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), gy


def _lbfgs_maxfun(n, time_limit, cap=None):
    if cap is None:
        if n <= 4096:
            cap = 900
        elif n <= 8192:
            cap = 650
        elif n <= 16384:
            cap = 450
        elif n <= 32768:
            cap = 260
        else:
            cap = 160
    est = int(max(40, float(time_limit) * 7.0))
    return int(min(cap, est))


def _run_lsm_lbfgs(a, beta, time_limit, maxfun_cap=None):
    a = _norm(a)
    n = a.size
    sc0 = _score(a)
    if not np.isfinite(sc0):
        return sc0, a
    if time_limit < 4.0:
        return sc0, a

    cap = _lbfgs_maxfun(n, time_limit, maxfun_cap)
    maxiter = max(10, min(180, cap // 4))
    y0 = _init_y(a)

    try:
        res = minimize(
            _lse_lsm_obj,
            y0,
            method="L-BFGS-B",
            jac=True,
            args=(float(beta),),
            bounds=None,
            options={
                "maxiter": maxiter,
                "maxfun": cap,
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 40
            }
        )
        if res.x is not None:
            a_new = _norm(_softmax_centered(res.x))
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new <= sc0 + 1e-9:
                return float(sc_new), a_new
    except Exception as e:
        print("  L-BFGS log-softmax error:", e)

    return float(sc0), a


def _binom_init(n):
    n = int(max(1, n))
    q = np.empty(n, dtype=np.float64)
    q[0] = 1.0
    if n > 1:
        i = np.arange(1, n, dtype=np.float64)
        ratios = (2.0 * i - 1.0) / (2.0 * i)
        q[1:] = np.cumprod(ratios)
    return _norm(q)


def _upsample(a, n):
    a = _norm(a)
    m = a.size
    n = int(max(1, n))
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, n)
    out = np.interp(x1, x0, a)
    return _norm(out)


def _smooth_perturb(a, rng, strength=0.1):
    a = _norm(a)
    n = a.size
    if n <= 1:
        return a.copy()
    noise = rng.standard_normal(n)
    k = int(min(61, max(3, n // 100)))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = n if n % 2 == 1 else max(1, n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k, dtype=np.float64) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode="same")
    q = a + strength * noise
    return _norm(q)


def _plse(a, deadline, rng, beta_start=1.0, beta_end=2000.0, max_iter=None, label=""):
    a = _norm(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score(a)
    if max_iter is None:
        max_iter = int(max(300, min(6000, 50000000.0 / max(1, n))))
    beta = float(beta_start)
    log_growth = np.log(max(float(beta_end), beta * 1.1)) / max(1, max_iter)
    prev_a = None
    prev_g = None
    eta = 0.02

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        actual, val, g = _lse_compute(a, beta)
        if not np.isfinite(actual):
            g = np.zeros_like(a)
            actual = _score(a)
        if not np.all(np.isfinite(g)):
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

        if actual < best_sc - 1e-12:
            best_sc = float(actual)
            best_a = a.copy()

        if prev_a is not None:
            s = a - prev_a
            y = g - prev_g
            ss = float(s @ s)
            sy = float(s @ y)
            yy = float(y @ y)
            if ss > 1e-14 and sy > 1e-12 and yy > 1e-12:
                e1 = ss / sy
                e2 = sy / yy
                eta_new = 0.5 * (e1 + e2)
                if np.isfinite(eta_new) and eta_new > 0.0:
                    eta = float(np.clip(eta_new, 1e-6, 0.12))
                else:
                    eta *= 0.7
            else:
                eta *= 0.75

        cand = _project_simplex(a - eta * g)
        sc_cand = _score(cand)
        if sc_cand < best_sc - 1e-12:
            best_sc = float(sc_cand)
            best_a = cand.copy()

        if sc_cand < actual - 1e-12:
            a = cand
        else:
            half = _project_simplex(a - 0.5 * eta * g)
            sc_half = _score(half)
            if sc_half < best_sc - 1e-12:
                best_sc = float(sc_half)
                best_a = half.copy()
            if sc_half < actual - 1e-12:
                a = half
            else:
                a = best_a.copy()
                eta *= 0.5
                if eta < 1e-7:
                    a = _smooth_perturb(best_a, rng, 0.03)
                    eta = 0.04
                    prev_a = None
                    prev_g = None

        if prev_a is not None:
            prev_a = a.copy()
            prev_g = g
        else:
            prev_a = a.copy()
            prev_g = g

        beta *= float(np.exp(log_growth))
        if it % 300 == 0:
            print(f"    PLSE {label} it={it} best={best_sc:.8f}")

    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return float(best_sc), best_a


def _exact_score_grad(a, smooth, topk):
    a = _norm(a)
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n, dtype=np.float64)
    c, P, N = _fft_self(a)
    d = float(n) * c / (S * S)
    M = d.size
    dmax = float(d.max())
    if M == 0 or not np.isfinite(dmax):
        return np.inf, np.zeros(n, dtype=np.float64)

    K = int(min(M, max(16, int(topk))))
    if K <= M:
        pos = M - K
        idx = np.argpartition(d, pos)[pos:]
    else:
        idx = np.arange(M)
    d_act = d[idx]
    gap = dmax - d_act
    w_act = np.exp(-gap / max(1e-8, float(smooth)))
    sw = float(w_act.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w_act = np.ones_like(d_act)
        sw = float(w_act.size)
    weights = w_act / sw

    w_full = np.zeros(M, dtype=np.float64)
    w_full[idx] = weights
    corr = _corr_from_P(P, w_full, N, n)
    LDS = float(weights @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(2.0 * dmax), grad


def _subgrad_polish(a, deadline, rng, max_iter=None, topk_frac=0.05,
                    smooth_start=0.01, smooth_end=0.0001):
    a = _norm(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score(a)
    M = max(1, 2 * n - 1)
    topk = int(max(32, min(M, int(topk_frac * M))))
    if max_iter is None:
        max_iter = int(max(60, min(2500, 15000000.0 / max(1, n))))
    smooth = float(smooth_start)
    log_growth = np.log(max(smooth_end, smooth_start * 0.5)) / max(1, max_iter)
    prev_a = None
    prev_g = None
    eta = 0.005

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        actual, g = _exact_score_grad(a, smooth, topk)
        if not np.isfinite(actual):
            break
        if not np.all(np.isfinite(g)):
            g = np.zeros_like(a)

        if actual < best_sc - 1e-12:
            best_sc = float(actual)
            best_a = a.copy()

        if prev_a is not None:
            s = a - prev_a
            y = g - prev_g
            ss = float(s @ s)
            sy = float(s @ y)
            yy = float(y @ y)
            if ss > 1e-14 and sy > 1e-12 and yy > 1e-12:
                e1 = ss / sy
                e2 = sy / yy
                eta_new = 0.5 * (e1 + e2)
                if np.isfinite(eta_new) and eta_new > 0.0:
                    eta = float(np.clip(eta_new, 1e-7, 0.08))
                else:
                    eta *= 0.7
            else:
                eta *= 0.75

        best_cand_sc = actual
        best_cand_a = a
        for f in (1.0, 0.5, 0.25):
            q = _project_simplex(a - eta * f * g)
            sc_q = _score(q)
            if np.isfinite(sc_q) and sc_q < best_cand_sc:
                best_cand_sc = float(sc_q)
                best_cand_a = q

        if best_cand_sc < best_sc - 1e-12:
            best_sc = best_cand_sc
            best_a = best_cand_a.copy()

        if best_cand_sc < actual - 1e-12:
            a = best_cand_a
        else:
            a = best_a.copy()
            eta *= 0.5
            if eta < 1e-8:
                a = _smooth_perturb(best_a, rng, 0.02)
                eta = 0.004
                prev_a = None
                prev_g = None

        prev_a = a.copy()
        prev_g = g
        smooth *= float(np.exp(log_growth))
        if it % 100 == 0:
            print(f"    SUBGRAD best={best_sc:.8f} it={it}")

    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return float(best_sc), best_a


def _lp_direction(a, K, delta, time_limit=20.0):
    if linprog is None:
        return None
    a = _norm(a)
    n = a.size
    if n < 16:
        return None
    c, _, _ = _fft_self(a)
    M = c.size
    if M == 0:
        return None
    Mmax = float(c.max())
    if not np.isfinite(Mmax) or Mmax <= 0.0:
        return None

    K = int(min(M, max(8, K)))
    pos = M - K
    idx = np.argpartition(c, pos)[pos:]
    c_act = c[idx]
    slacks = (1.0 + float(delta)) * Mmax - c_act
    slacks = np.maximum(slacks, 0.0)
    b = 0.5 * slacks

    R = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(idx):
        lo = max(0, int(k) - (n - 1))
        hi = min(n - 1, int(k))
        if hi >= lo:
            R[r, lo:hi + 1] = a[int(k) - hi:int(k) - lo + 1][::-1]

    c_obj = -np.ones(n, dtype=np.float64)
    lower = -a
    upper = np.maximum(a, 1e-4)
    bounds = list(zip(lower, upper))

    try:
        res = linprog(
            c_obj,
            A_ub=R,
            b_ub=b,
            bounds=bounds,
            method="highs",
            disp=False,
            options={"presolve": True, "time_limit": float(max(0.5, time_limit))}
        )
    except Exception:
        try:
            res = linprog(
                c_obj,
                A_ub=R,
                b_ub=b,
                bounds=bounds,
                method="highs",
                disp=False,
                options={"presolve": True}
            )
        except Exception:
            return None

    if res is None or not getattr(res, "success", False):
        return None
    h = _clean_array(res.x)
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    h = np.clip(h, lower, upper)
    if float(h.sum()) <= 1e-12:
        return None
    return h


def _lp_line_search(a, h):
    a = _norm(a)
    h = _clean_array(h)
    n = a.size
    best_sc = _score(a)
    best_a = a.copy()
    if n == 0 or h.size != n:
        return best_sc, best_a

    tmax = 1.0
    neg = h < 0.0
    if np.any(neg):
        ratios = a[neg] / (-h[neg])
        if ratios.size > 0 and np.all(np.isfinite(ratios)):
            tmax = min(1.0, 0.99 * float(np.min(ratios)))
    if not np.isfinite(tmax) or tmax <= 0.0:
        tmax = 0.01

    base = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0)
    ts = [t for t in base if t <= tmax]
    if not ts:
        ts = [0.01 * tmax if tmax > 0 else 0.001]

    for t in ts:
        q = a + t * h
        q = np.maximum(q, 0.0)
        q = _norm(q)
        sc = _score(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = q.copy()

    return best_sc, best_a


def _lp_polish(a, deadline, rng):
    if linprog is None:
        return _score(a), a
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    max_iters = 45
    fails = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / float(max(1, max_iters - 1))
        K = int(16 + (48 - 16) * frac)
        delta = 0.20 * (0.003 / 0.20) ** frac
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break
        try_t = min(14.0, max(3.0, remaining * 0.25))
        h = _lp_direction(a, K, delta, try_t)
        if h is None:
            fails += 1
            if fails > 5:
                break
            continue

        sc, q = _lp_line_search(a, h)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = float(sc)
            best_a = q.copy()
            a = q.copy()
            fails = 0
            print(f"    LP dir improves: {best_sc:.8f} it={it}")
        else:
            fails += 1
            if fails > 8:
                break

    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return float(best_sc), best_a


def _generate_candidates(parent, lengths, rng):
    best_by_len = {}
    for L0 in lengths:
        L = int(max(512, L0))
        inits = []
        if parent is not None:
            p = _upsample(parent, L)
            inits.append(("parent", p))
            inits.append(("parent_rev", _upsample(parent[::-1], L)))
            for amp in (0.04, 0.10, 0.20):
                inits.append((f"smooth_{amp}", _smooth_perturb(p, rng, amp)))

        b = _binom_init(L)
        inits.append(("binom", b))

        x = np.linspace(0.0, 1.0, L)
        taper1 = b * (1.0 - 0.35 * x)
        taper2 = b * (1.0 + 0.55 * x)
        tail1 = b * np.exp(-0.7 * x)
        tail2 = b * np.exp(0.8 * x)
        for nm, arr in (("taper1", taper1), ("taper2", taper2),
                        ("tail1", tail1), ("tail2", tail2)):
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                inits.append((nm, _norm(arr)))

        # A few simple power-law shapes for diversity.
        ii = np.arange(1, L + 1, dtype=np.float64)
        for pe in (0.45, 0.50, 0.55):
            arr = ii ** (-pe)
            arr = arr + 0.2 * np.exp(-0.5 * x)
            inits.append((f"pow_{pe}", _norm(arr)))

        scored = []
        for nm, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((float(sc), nm, arr))
        if not scored:
            b = _binom_init(L)
            scored.append((float(_score(b)), "binom", b))
        scored.sort(key=_first_key)
        best_by_len[L] = scored[0]

    candidates = []
    seen = set()
    preferred = [16384, 8192]
    for L in preferred:
        if L in best_by_len:
            candidates.append(best_by_len[L])
            seen.add(L)
    for L in lengths:
        if L in best_by_len and L not in seen:
            candidates.append(best_by_len[L])
            seen.add(L)

    final = []
    for item in candidates:
        if len(final) < 3:
            final.append(item)
        else:
            worst = 0
            for i in range(1, 3):
                if final[i][0] > final[worst][0]:
                    worst = i
            if item[0] < final[worst][0]:
                final[worst] = item
    final.sort(key=_first_key)
    return final


def _optimize_candidate(a0, sub_deadline, label, rng):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = a.size
    print(f"  [{label}] start score={best_sc:.8f}, n={n}")

    t_start = time.monotonic()
    if sub_deadline - t_start < 80.0:
        return float(best_sc), best_a
    total = sub_deadline - t_start

    # Multilevel warm start for larger lengths.
    if n > 2500 and total > 320.0:
        small_n = max(1024, n // 4)
        ml_deadline = t_start + min(70.0, 0.08 * total)
        a_small = _upsample(a, small_n)
        sc_small = _score(a_small)
        a_small = _norm(_upsample(a, small_n))
        sc_small, a_small = _plse(a_small, ml_deadline * 0.65, rng,
                                  beta_start=1.0, beta_end=500.0,
                                  max_iter=400, label=label + "_ml")
        for beta in (50.0, 500.0):
            if time.monotonic() < ml_deadline:
                sc_small, a_small = _run_lsm_lbfgs(a_small, beta,
                                                   max(4.0, (ml_deadline - time.monotonic()) * 0.45),
                                                   maxfun_cap=180)
        a_up = _upsample(a_small, n)
        sc_up = _score(a_up)
        if np.isfinite(sc_up) and sc_up < best_sc:
            best_sc = float(sc_up)
            best_a = a_up.copy()
            a = a_up.copy()
            print(f"  [{label}] multilevel improved: {best_sc:.8f} (small={sc_small:.8f})")

    # Projected LSE exploration.
    now = time.monotonic()
    if sub_deadline - now > 90.0:
        plse_deadline = now + min(150.0, 0.14 * (sub_deadline - now))
        sc, a2 = _plse(a, plse_deadline, rng, beta_start=0.8, beta_end=3000.0,
                       label=label)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a2.copy()
        a = a2 if np.all(np.isfinite(a2)) else best_a.copy()

    # L-BFGS in log-softmax space, with increasing smoothness precision.
    now = time.monotonic()
    if sub_deadline - now > 120.0:
        lb_end = now + 0.42 * (sub_deadline - now)
        betas = (30.0, 150.0, 750.0, 4000.0, 18000.0)
        for bi, beta in enumerate(betas):
            if time.monotonic() >= lb_end - 5.0:
                break
            remaining = lb_end - time.monotonic()
            per_stage = remaining / float(len(betas) - bi)
            if per_stage < 8.0:
                continue
            sc, a2 = _run_lsm_lbfgs(a, beta, max(8.0, per_stage * 0.92))
            if np.isfinite(sc):
                a = a2
                if sc < best_sc:
                    best_sc = float(sc)
                    best_a = a2.copy()
                    print(f"  [{label}] L-BFGS beta={beta:.0f} -> {best_sc:.8f}")

    # LP direction polishing.
    now = time.monotonic()
    if sub_deadline - now > 70.0:
        lp_deadline = now + min(260.0, 0.27 * (sub_deadline - now))
        sc, a2 = _lp_polish(a, lp_deadline, rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a2.copy()
        a = a2 if np.all(np.isfinite(a2)) else best_a.copy()

    # Exact subgradient final polish.
    now = time.monotonic()
    if sub_deadline - now > 40.0:
        sg_deadline = now + min(220.0, 0.90 * (sub_deadline - now))
        sc, a2 = _subgrad_polish(a, sg_deadline, rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a2.copy()

    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = float(final_sc)
    print(f"  [{label}] end score={best_sc:.8f}, time={time.monotonic() - t_start:.1f}s")
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = float(max(120.0, float(budget_s)))
    deadline = t0 + budget - 45.0
    final_reserve = min(240.0, 0.12 * budget)
    main_end = max(t0 + 60.0, deadline - final_reserve)
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = _clean_array(p)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 256 and np.isfinite(p).size == p.size:
                parent = p
    except Exception:
        parent = None

    lengths = []
    if parent is not None:
        pn = int(parent.size)
        if 512 <= pn <= 20000:
            lengths.append(pn)
    for L in (8192, 16384):
        if L not in lengths:
            lengths.append(L)
    if not lengths:
        lengths = [8192, 16384]

    print(f"Parent available: {parent is not None}, lengths={lengths}, budget={budget:.0f}s")

    cands = _generate_candidates(parent, lengths, rng)
    if not cands:
        cands = [(1.6, "binom", _binom_init(8192))]

    best_seq = cands[0][2].copy()
    best_score = float(cands[0][0])
    print("Candidate starts:",
          [(nm, ln, f"{sc:.8f}") for (sc, nm, ln) in cands])

    fracs = [0.45, 0.32, 0.23][:len(cands)]
    if len(cands) == 2:
        fracs = [0.55, 0.45]

    for i, (sc0, nm, a0) in enumerate(cands):
        left_sum = 0.0
        for j in range(i, len(cands)):
            left_sum += fracs[j]
        remaining_main = max(0.0, main_end - time.monotonic())
        alloc = (fracs[i] / left_sum) * remaining_main if left_sum > 0 else 0.0
        alloc = max(60.0, min(alloc, remaining_main))
        sub_deadline = time.monotonic() + alloc
        sub_deadline = min(sub_deadline, main_end)

        if sub_deadline <= time.monotonic() + 30.0:
            continue

        sc, a = _optimize_candidate(a0, sub_deadline, f"{nm}_len{len(a0)}", rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = a.copy()
            print(f"  *** global best updated: {best_score:.8f} from {nm}_len{len(a0)} ***")

    # Final global polish on the best sequence.
    print(f"Final polish starting, best={best_score:.8f}, n={best_seq.size}")
    fp_deadline = deadline - 25.0

    # A few high-smoothness L-BFGS runs.
    for beta in (6000.0, 40000.0, 180000.0):
        rem = fp_deadline - time.monotonic()
        if rem < 60.0:
            break
        sc, a2 = _run_lsm_lbfgs(best_seq, beta, min(150.0, rem * 0.28),
                                maxfun_cap=350)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = a2.copy()
            print(f"  Final L-BFGS beta={beta:.0f} -> {best_score:.8f}")

    # Final LP attempt.
    rem = fp_deadline - time.monotonic()
    if rem > 90.0:
        sc, a2 = _lp_polish(best_seq, time.monotonic() + min(220.0, rem * 0.55), rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = a2.copy()
            print(f"  Final LP -> {best_score:.8f}")

    # Final subgradient attempt.
    rem = fp_deadline - time.monotonic()
    if rem > 50.0:
        sc, a2 = _subgrad_polish(best_seq, time.monotonic() + min(200.0, rem * 0.85), rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = a2.copy()
            print(f"  Final SUBGRAD -> {best_score:.8f}")

    # If the best is very long, consider a safer 8192-length if it stays almost as good.
    if best_seq.size > 12000:
        small = _upsample(best_seq, 8192)
        small_sc = _score(small)
        if np.isfinite(small_sc) and small_sc < best_score + 0.00025:
            best_score = float(small_sc)
            best_seq = small
            print(f"  Switched to len=8192 with score={best_score:.8f}")

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    if float(out.sum()) < 0.01:
        out = np.ones(8192, dtype=np.float64) * 0.5

    result = [float(x) for x in out]

    try:
        if "evaluate_sequence" in globals():
            verified = float(evaluate_sequence(result))
            if np.isfinite(verified) and verified < best_score:
                best_score = verified
            print(f"FINAL internal={best_score:.10f}, verified={verified:.10f}, "
                  f"len={len(result)}, time={time.monotonic() - t0:.1f}s")
        else:
            print(f"FINAL internal={best_score:.10f}, len={len(result)}, "
                  f"time={time.monotonic() - t0:.1f}s")
    except Exception as e:
        print("Verification skipped:", e)
        print(f"FINAL internal={best_score:.10f}, len={len(result)}, "
              f"time={time.monotonic() - t0:.1f}s")

    return result
```
