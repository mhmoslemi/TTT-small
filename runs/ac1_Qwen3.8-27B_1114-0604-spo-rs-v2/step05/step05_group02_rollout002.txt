"""
Improved step function optimization using GS-Fienup with error reduction,
high-beta LSE-LBFGS, and peak-specific refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_fienup_err_red(a, n_iters=1000, beta_err=1.0):
    """
    GS-Fienup with Fienup error reduction function for non-negativity.
    Instead of hard clipping, uses a smooth error function.
    """
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()

    # Target: flat convolution -> flat |A|^2
    # b_target = 1/(2n-1) for all 2n-1 lags
    b_target = np.ones(2 * n - 1) / (2 * n - 1)
    B_target = np.fft.rfft(b_target, M)
    target_mag_sq = np.abs(B_target) ** 2  # This is |B_target|^2, but we want |A|^2 = |B_target| for flat conv
    # Actually: if a*a = b_target (flat), then |A|^2 = FFT(b_target) = B_target
    # So target |A|^2 = B_target (which may be complex!)
    # We can only control |A|, not the phase of A^2
    # So target |A| = sqrt(|B_target|)
    target_mag = np.sqrt(np.maximum(np.abs(B_target), 1e-30))

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        alpha = 0.3 * (1.0 - progress * 0.5)  # Decay from 0.3 to 0.15

        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-30)

        # Update magnitude: blend current with target
        new_mag = A_mag ** (1.0 - alpha) * target_mag ** alpha
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, M)[:n]

        # Fienup error reduction: smooth non-negativity
        # a_err = a_new - max(a_new, 0) = min(a_new, 0) (negative part)
        # Error reduction: a_corrected = a_new - gamma * a_err / |a_err|_norm^2 * <a_new, a_err>

        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

    n = len(a)

    target_mag = np.sqrt(np.maximum(np.abs(B_target), 1e-30))

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-30)

        new_mag = A_mag ** (1.0 - alpha) * target_mag ** alpha
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed max of convolution, with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    # Shift for numerical stability
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(exp_scaled.sum())
    L = c_max + np.log(sum_exp) / beta  # LSE approximation of max
    # score = 2n * L / S^2 (scaled version; we optimize 2n*L/S^2 subject to a>=0)
    # Actually for optimization, since S is also variable, let's just use 2n*L/S^2
    # But for L-BFGS with bounds [0, 1000], S will be determined by the optimizer
    
    # Gradient of L with respect to a:
    # L = max(c) approx, and c_k = sum_j a_j * a_{k-j}
    # dL/da_j = sum_k (dc_k/da_j) * (dL/dc_k)
    # dL/dc_k = exp(beta*(c_k - c_max)) / sum_exp (softmax weights)
    # dc_k/da_j = a_{k-j} + a_{k-j'} where j' is the other index... 
    # Actually dc_k/da_j = a_{k-j} (since c_k = sum_i a_i * a_{k-i}, dc_k/da_j = a_{k-j} when k-j is valid, plus a_j when k-j = j i.e. k=2j)
    # More precisely: c_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} if 0<=k-j<n (i= j, k-i = k-j)
    #              + a_j if 0<=k-j<n and k-j != j (i=k-j, k-i = j) -- wait this is the same term
    # Actually: c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j: the term a_j appears when i=j (giving a_{k-j}) and when k-i=j i.e. i=k-j (giving a_j)
    # So dc_k/da_j = a_{k-j} (if valid) + a_j (if valid and j != k-j, i.e., k != 2j)
    # If k = 2j: dc_k/da_j = 2*a_j
    
    # This is essentially: the gradient of c with respect to a is related to the "cross-correlation"
    # dc/da_j = (a * p)_j where p is... hmm let me think differently.
    
    # c = a * a (linear conv), so dc/da = 2 * (a * da) in some sense.
    # More precisely: if c = conv(a, a), then dc_j/da_i = a_{j-i} + a_{j-i} (with appropriate boundary handling)
    # = 2 * a_{j-i} for i != j-i, and a_j for i = j-i (i.e., 2i = j)
    
    # The gradient of L = sum_k w_k * c_k where w_k = exp(beta*(c_k-cmax))/sum_exp
    # dL/da_j = sum_k w_k * dc_k/da_j
    # dc_k/da_j = a_{k-j} (if 0 <= k-j < n) + a_{k-j} (if 0 <= k-j < n and k-j != j)
    #           = 2*a_{k-j} (if 0 <= k-j < n and k != 2j) + a_j (if k = 2j)
    # Wait, I think I'm overcomplicating this. Let me use the chain rule more carefully.
    
    # c_k = sum_i a_i * a_{k-i} (sum over valid i)
    # Partial derivative of c_k w.r.t. a_j:
    #   = a_{k-j} (from the term where i = j) + a_{k-j} (from the term where k-i = j, i.e., i = k-j, IF k-j != j)
    #   = 2 * a_{k-j} if k-j is valid and k-j != j
    #   = a_j if k = 2j (only one term)
    #   = 0 if k-j is not valid
    
    # So dL/da_j = sum_k w_k * [2*a_{k-j} * 1(0<=k-j<n, k!=2j) + a_j * 1(k=2j)]
    #            = 2 * sum_k w_k * a_{k-j} - sum_k w_k * a_{k-j} * 1(k=2j) + sum_k w_k * a_j * 1(k=2j)
    # Hmm, this is getting messy. Let me use a different approach.
    
    # Alternative: dL/da_j = sum_k w_k * dc_k/da_j
    # Let's define: for each k, dc_k/da_j = a_{k-j} + a_{k-j} (with the understanding that if k-j = j, it's just 2*a_j, and if k-j is out of bounds, it's 0)
    # Actually the standard result is: if c = conv(a, b), then dc/da = conv(b, delta) * stuff...
    # Let me just use the fact that conv(a, a) has gradient 2 * conv(a, e_j) = 2 * a reversed and shifted.
    
    # The cleaner way: d(conv(a,a))/da_j = 2 * a convolved with delta_j, but with the self-term adjustment.
    # For linear convolution: (conv(a,a))_k = sum_i a_i a_{k-i}
    # d/d a_j of this = a_{k-j} (from i=j) + a_{k-j} (from k-i=j, i.e., i=k-j, if k-j≠j)
    # = 2 a_{k-j} if k-j is in [0,n-1] and k≠2j; = a_j if k=2j; = 0 otherwise.
    
    # So: grad_j = sum_k w_k * (2 a_{k-j} * [0<=k-j<n, k!=2j] + a_j * [k=2j])
    # = 2 sum_k w_k a_{k-j} [0<=k-j<n] - sum_k w_k a_{k-j} [k=2j, 0<=k-j<n] + sum_k w_k a_j [k=2j]
    # Note: when k=2j, k-j=j which is in [0,n-1], so [0<=k-j<n] is true.
    # = 2 sum_k w_k a_{k-j} [0<=k-j<n] - sum_k w_k a_j [k=2j] + sum_k w_k a_j [k=2j]
    # = 2 sum_k w_k a_{k-j} [0<=k-j<n]
    
    # So grad_j = 2 * sum_{k: max(0,j) <= k <= min(2n-2, j+n-1)} w_k * a_{k-j}
    #           = 2 * (w * a)_j where * denotes the appropriate cross-correlation
    
    # More precisely: grad_j = 2 * sum_i w_{j+i} * a_i for i = 0, ..., n-1 (with w out of range = 0)
    # This is 2 * (a reflected, convolved with w) evaluated at position... 
    
    # Let me just compute it directly:
    # grad_j = 2 * sum_{i=0}^{n-1} a_i * w_{j+i}  (where w_k = 0 if k < 0 or k >= 2n-1)
    
    # This can be computed as: 2 * conv(a, w_pad) where w_pad is w with appropriate zero padding
    # Actually: sum_i a_i * w_{j+i} = (reversed_a convolved with w)_j ... let me just do it with FFT.
    
    # Define: r_i = a_i (i=0..n-1), w_k (k=0..2n-2)
    # grad_j / 2 = sum_{i=0}^{n-1} r_i * w_{j+i}
    # This is like a cross-correlation. If we define w_padded of length 2n-1, then:
    # grad_j/2 = sum_i a_i * w[j+i] = (correlation of a and w) at lag j
    # = sum_i a_i * w[j+i]
    
    # Using FFT: if we compute IFFT(FFT(a) * conj(FFT(w_reversed)))... 
    # Actually, the cross-correlation (a corr w)_j = sum_i a_i * conj(w_{i-j})
    # That's not quite what we want.
    
    # Let me define: result_j = sum_{i=0}^{n-1} a_i * w_{j+i}
    # This is a "sliding dot product" - a sliding window of w against a.
    # We can compute this as: reverse a, then convolve with w.
    # Let ar = a[::-1] (reversed a, length n)
    # conv(ar, w)[j + n - 1] = sum_i ar_i * w[j+n-1-i] = sum_i a[n-1-i] * w[j+n-1-i]
    # Let m = n-1-i: = sum_m a[m] * w[j+m] = result_j.
    # So result_j = conv(ar, w)[j + n - 1]
    
    # Great! So grad = 2 * conv(a_reversed, w)[n-1 : 2n-1]
    
    w = exp_scaled / sum_exp
    a_rev = a[::-1]
    # conv(a_rev, w): a_rev has length n, w has length 2n-1, result has length n + 2n-1 - 1 = 3n-2
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    AR = np.fft.rfft(a_rev, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_result = np.fft.irfft(AR * W, M2_fft)[:3*n-2]
    grad_conv = 2.0 * conv_result[n-1:2*n-1]  # This gives grad of L w.r.t. a
    
    # Now the score is 2n * L / S^2
    # d(score)/da_j = 2n * (grad_conv_j * S^2 - L * 2 * S) / S^4
    #               = 2n * (grad_conv_j / S^2 - 2L/S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_conv / S2 - 2.0 * L / S3)
    
    score = 2.0 * n * L / S2
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=10000, maxfun=200000, ftol=1e-18):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        sc, g = lse_score_and_grad(x, beta)
        return sc, g
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_reduction(a, n_iters=500, top_k=20, step_size=0.01):
    """
    Directly reduce peaks in the convolution by adjusting a at positions
    that contribute most to those peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # For each peak, find which a_j contribute most and reduce them
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contribution of a[j] to c[k] is a[j] * a[k-j]
            contrib = np.array([a[j] * a[k - j] for j in range(j_start, j_end + 1)])
            
            # Reduce the element with highest contribution
            j_max = j_start + int(np.argmax(contrib))
            if a[j_max] > 1e-8:
                # Reduce it
                a[j_max] *= (1.0 - step_size)
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a


def random_perturb_optimize(a, n_perturbs=50, n_trials=5, rng=None):
    """Try random perturbations and keep improvements."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_perturbs):
        a_trial = best_a.copy()
        # Perturb a random subset
        n_change = rng.integers(10, max(11, n // 10))
        indices = rng.choice(n, size=n_change, replace=False)
        factors = rng.uniform(0.5, 1.5, size=n_change)
        a_trial[indices] *= factors
        a_trial = np.maximum(a_trial, 0.0)
        S = a_trial.sum()
        if S < 1e-12:
            continue
        a_trial = a_trial / S
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.01))

    x = np.arange(n, dtype=np.float64) / n
    
    # Various structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0))
    
    # "Windowed" inits - concentrated in middle
    center = n // 2
    width = n // 3
    sig = np.zeros(n)
    sig[center - width:center + width] = 1.0
    # Smooth edges
    edge = width // 5
    sig[center - width:center - width + edge] = np.linspace(0, 1, edge)
    sig[center + width - edge:center + width] = np.linspace(1, 0, edge)
    inits.append(sig)
    
    # Two bumps
    bump1 = np.zeros(n)
    bump1[:n//2] = 1.0
    bump2 = np.zeros(n)
    bump2[n//2:] = 1.0
    inits.append(bump1 + 0.5 * bump2)
    
    return inits
