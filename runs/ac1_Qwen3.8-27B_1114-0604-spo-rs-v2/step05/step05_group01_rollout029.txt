"""

Uses LSE-smoothed optimization with adaptive beta, FISTA acceleration,

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute cross-correlation: result[i] = sum_j w[i+j] * a[j] for valid indices."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    # Shift for numerical stability
    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L  # Since we normalize to sum=1, the penalty is just 2*L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def estimate_eval_time(n):
    base = 0.025
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    mf = max(64, min(5000, mf))
    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.85 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    maxfun = max(64, maxfun)
    maxiter = max(80, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 80,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def fista_optimize(a0, beta, deadline, max_iters=500):
    """FISTA (Nesterov accelerated gradient) for the LSE-smoothed objective.
    Uses backtracking line search and projection onto {a >= 0}."""
    a = normalize_a(a0).astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = exact_score_a(a)

    # Compute initial gradient
    f, g = lse_score_grad(a, beta)
    if not np.isfinite(f):
        return best_score, best_a

    # Initial step size
    L = float(np.linalg.norm(g, ord=np.inf)) + 1e-8
    t = 1.0
    x = a.copy()
    y = a.copy()

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        # Gradient step
        alpha = 1.0 / L
        x_new = y - alpha * g
        x_new = np.clip(x_new, 0.0, None)
        x_new = normalize_a(x_new)

        # Backtracking line search
        f_new, g_new = lse_score_grad(x_new, beta)
        if not np.isfinite(f_new):
            f_new = 1e10
            g_new = np.zeros(n)

        num_iter = 0
        while f_new > f + 0.5 * L * float(np.sum((x_new - y) ** 2)) and num_iter < 20:
            L *= 2.0
            alpha = 1.0 / L
            x_new = y - alpha * g
            x_new = np.clip(x_new, 0.0, None)
            x_new = normalize_a(x_new)
            f_new, g_new = lse_score_grad(x_new, beta)
            if not np.isfinite(f_new):
                f_new = 1e10
                g_new = np.zeros(n)
            num_iter += 1

        if f_new < f:
            L = max(1e-10, L / 1.5)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, None)
        y = normalize_a(y)
        t = t_new
        x = x_new
        f, g = f_new, g_new_new = g_new

        # Track best
        sc = exact_score_a(x)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = x.copy()

    return float(best_score), best_a


def active_weights_topk(d, k):
    """Get weights concentrated on top-k lags of d."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    # Soft top-k: use temperature-based weighting
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    vals = d[idx]
    v_max = float(np.max(vals))
    # Soft weighting within the top-k
    temp = max(0.1 * (v_max - float(np.min(vals)) + 1e-12), 1e-12)
    w[idx] = np.exp((vals - v_max) / temp)
    w_norm = float(w.sum())
    if w_norm > 1e-12:
        w /= w_norm
    return w


def subgrad_step_weighted(a, eta, w):
    """Subgradient step using given weights w over the convolution lags."""
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def active_set_equalize(a, deadline, max_outer=20):
    """Try to equalize the top active lags by iterative subgradient steps
    focused on the most prominent peaks."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for outer in range(int(max_outer)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        d, S = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 1e-12:
            break

        # Try different active set sizes
        improved = False
        for k in (1, 2, 3, 5, 8, 12):
            if deadline is not None and time.monotonic() > float(deadline):
                break
            w = active_weights_topk(d, k)
            for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step_weighted(a, eta, w)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True

        if not improved:
            break
        a = best_a.copy()

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=30):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes (high at ends)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Asymmetric ramps (important for breaking the symmetry barrier)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(1.0 + alpha * x)
        add(1.0 + alpha * (1.0 - x))
        add(np.exp(alpha * x))
        add(np.exp(-alpha * x))

    # Cosine shapes
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x + np.pi / 3))

    # Sine-based
    for f in (1, 2, 3, 4, 5):
        add(1.0 + 0.7 * np.sin(np.pi * f * x))
        add(1.0 + 0.7 * np.abs(np.sin(np.pi * f * x)))
        add(1.0 + 0.7 * np.sin(2.0 * np.pi * f * x + 0.5))

    # Bimodal
    for width in (0.08, 0.12, 0.18, 0.25):
        for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7), (0.35, 0.65)]:
            add(np.exp(-(((x - c1) / width) ** 2)) + np.exp(-(((x - c2) / width) ** 2)))

    # Power law shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(x ** p)
        add((1.0 - x) ** p)
        add(x ** p + (1.0 - x) ** p)

    # Random smooth via low-pass noise
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)

    # Random trigonometric sums
    for _ in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 7))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns with smooth transitions
    for nb in (4, 8, 16, 32):
        if nb > n:
            nb = max(2, n // 4)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        # Smooth the block values
        if nb >= 3:
            kern = np.array([1, 2, 1], dtype=np.float64) / 4.0
            block_vals = np.convolve(block_vals, kern, mode='same')
        per = max(1, n // nb)
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    # "Anti-symmetric" shapes (high in one half, low in other)
    for frac in (0.3, 0.4, 0.5):
        arr = np.ones(n)
        split = int(frac * n)
        arr[:split] *= 2.0
        arr[split:] *= 0.5
        add(arr)
        add(arr[::-1])

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        # Multiplicative noise
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        # Reverse + noise
        q = p[::-1].copy()
        q = q * np.exp(0.08 * rng.standard_normal(n))
    elif mode == 3:
        # Subgradient-inspired
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                k = int(rng.integers(1, 8))
                w = active_weights_topk(d, k)
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed noise in log space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 150))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape modulation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 80)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Split and swap halves
        mid = n // 2
        q = p.copy()
        t = float(rng.uniform(0.1, 0.4))
        q[:mid] = (1 - t) * p[:mid] + t * p[mid:]
        q[mid:] = (1 - t) * p[mid:] + t * p[:mid]
    elif mode == 8:
        # Frequency-domain perturbation
        Nf = 1 << (n - 1).bit_length()
        spec = np.fft.rfft(p, Nf)
        noise_spec = (rng.standard_normal(len(spec)) + 1j * rng.standard_normal(len(spec))) * 0.05
        spec = spec + noise_spec * np.abs(spec)
        q = np.fft.irfft(spec, Nf)[:n]
    else:
        # Random block scaling
        nb = int(rng.integers(3, max(4, n // 200)))
        scales = 1.0 + strength * rng.standard_nb = 1.0 + strength * rng.standard_normal(nb)
        block_size = n // nb
        q = p.copy()
        for b in range(nb):
            start = b * block_size
            end = start + block_size if b < nb - 1 else n
            q[start:end] *= scales[b]

    return normalize_a(np.clip(q, 0.0, None))


def quick_optimize(a, deadline):
    """Rapid multi-beta optimization using L-BFGS-B and FISTA."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    betas = [100, 300, 1000, 3000, 10000, 30000, 80000]
    for beta in betas:
        rem = float(deadline - time.monotonic())
        if rem < 15.0:
            break
        # L-BFGS-B
        mf = estimate_maxfun(rem * 0.6, n)
        if mf >= 64:
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)

        # FISTA (complementary to L-BFGS-B)
        rem = float(deadline - time.monotonic())
        if rem > 30.0:
            max_it = int(min(300, rem * 20))
            sc_f, a_f = fista_optimize(a, float(beta * 1.5), deadline, max_iters=max_it)
            if np.isfinite(sc_f) and sc_f < best_sc:
                best_sc = sc_f
                best_a = a_f.copy()
            a = normalize_a(a_f)

    return float(best_sc), best_a


def optimize_trajectory(a_start, deadline, rng, label):
    """Main optimization trajectory with adaptive beta continuation."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    n = len(a)

    total = max(10.0, float(deadline) - t0 - 15.0)

    print("[{}] start score={:.10f} n={}".format(label, best_sc, n))

    # Stages: (beta, fraction_of_time)
    stages = [
        (50.0, 0.04),
        (150.0, 0.06),
        (400.0, 0.08),
        (1000.0, 0.08),
        (3000.0, 0.10),
        (8000.0, 0.12),
        (20000.0, 0.14),
        (50000.0, 0.14),
        (120000.0, 0.14),
        (300000.0, 0.10),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline) - 10.0)

        # Multiple L-BFGS-B calls at this beta
        while time.monotonic() < sub_dead - 8.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 10.0:
                break
            mf = estimate_maxfun(rem * 0.8, n)
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.10f} beta={:.0f} elapsed={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0
                ))
            a = normalize_a(a2)

        # Active set equalization
        if time.monotonic() < sub_dead - 5.0:
            sc_eq, a_eq = active_set_equalize(a, deadline=sub_dead, max_outer=15)
            if np.isfinite(sc_eq) and sc_eq < best_sc:
                best_sc = sc_eq
                best_a = a_eq.copy()
                print("[{}] equalize best {:.10f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_eq)

        # FISTA at higher effective beta
        rem = float(sub_dead - time.monotonic())
        if rem > 20.0:
            max_it = int(min(200, rem * 5))
            sc_f, a_f = fista_optimize(a, float(beta * 2), sub_dead, max_iters=max_it)
            if np.isfinite(sc_f) and sc_f < best_sc:
                best_sc = sc_f
                best_a = a_f.copy()
                print("[{}] FISTA best {:.10f} at beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_f)

    return float(best_sc), best_a


def multi_resolution_refine(coarse_a, n_fine, deadline):
    """Upsample a coarse solution and refine on the fine grid."""
    fine_a = resample_sequence(coarse_a, n_fine)
    fine_a = normalize_a(fine_a)
    sc, refined = quick_optimize(fine_a, deadline)
    return sc, refined


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns the best sequence found within budget_s."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    # Try to get parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0
    ))

    # Determine lengths to explore
    if parent is not None:
        parent_len = int(min(parent.size, 16384))
        if parent_len < 2048:
            parent_len = 2048
        lengths = [parent_len]
        for L in [4096, 6144, 8192, 10240, 12288, 16384]:
            if L != parent_len:
                lengths.append(L)
    else:
        lengths = [8192, 10240, 6144, 4096, 12288]

    # Deduplicate, filter, sort by priority
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:5]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Generate and evaluate diverse initializations
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=25 if n > 8000 else 35)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))

    if not all_scores:
        n0 = lengths[0] if lengths else 8192
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.10f} (length {})".format(
        all_scores[int(order[0])], all_lens[int(order[0])]))

    # Select top starts (max 3, prefer different lengths)
    starts = []
    used_lens = set()

    if parent is not None:
        ps = normalize_a(resample_sequence(parent, min(parent.size, 10240)))
        starts.append(ps)
        used_lens.add(len(ps))

    for idx in order:
        if len(starts) >= 3:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    # Evaluate initial best
    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.10f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization trajectories
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [0.50, 0.28, 0.22]

    main_end = deadline - 400.0
    main_budget = max(60.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))

        if time.monotonic() > sub_dead - 30.0:
            continue

        sq = starts[si].copy()
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            t_blend = 0.4 if rank == 1 else 0.6
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.10f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Active set equalization on best
    if time.monotonic() < deadline - 300.0:
        sc, seq = active_set_equalize(best_seq, deadline=deadline - 300.0, max_outer=30)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Active set equalization: {:.10f}".format(best_score))

    # Phase 4: Multi-resolution refinement
    if time.monotonic() < deadline - 250.0:
        # Try refining at a different resolution
        n_fine = min(16384, max(8192, len(best_seq)))
        if n_fine != len(best_seq):
            sc_mr, seq_mr = multi_resolution_refine(best_seq, n_fine, deadline - 200.0)
            if np.isfinite(sc_mr) and sc_mr < best_score:
                best_score = sc_mr
                best_seq = seq_mr.copy()
                print("Multi-resolution: {:.10f}".format(best_score))
        else:
            # Same length, just do more optimization
            sc_mr, seq_mr = quick_optimize(best_seq, deadline - 200.0)
            if np.isfinite(sc_mr) and sc_mr < best_score:
                best_score = sc_mr
                best_seq = seq_mr.copy()
                print("Re-optimization: {:.10f}".format(best_score))

    # Phase 5: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 200.0 and perturb_done < 8:
        left = float(deadline - 200.0 - time.monotonic())
        if left < 80.0:
            break
        n_remaining = 8 - perturb_done
        chunk = time.monotonic() + left / float(n_remaining)
        chunk = min(chunk, deadline - 180.0)

        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.10f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 6: Final high-beta polish
    if time.monotonic() < deadline - 60.0:
        rem = float(deadline - time.monotonic() - 10.0)
        if rem > 15.0:
            mf = estimate_maxfun(rem * 0.7, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 200000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High-beta final: {:.10f}".format(best_score))

        if time.monotonic() < deadline - 40.0:
            rem = float(deadline - time.monotonic() - 5.0)
            if rem > 10.0:
                # FISTA at very high beta
                max_it = int(min(200, rem * 5))
                sc_f, a_f = fista_optimize(best_seq, 500000.0, deadline, max_iters=max_it)
                if np.isfinite(sc_f) and sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_f.copy()
                    print("FISTA very high beta: {:.10f}".format(best_score))

    # Phase 7: Final active set cleanup
    if time.monotonic() < deadline - 20.0:
        sc, seq = active_set_equalize(best_seq, deadline=deadline - 15.0, max_outer=20)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final active set: {:.10f}".format(best_score))

    # Final output preparation
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        # Keep as is (scale-invariant objective)
        pass

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start
    ))
    return out
