        # For each peak at k, the contributing pairs are (i, k-i) for valid i
        # We want to reduce a[i]*a[k-i] for high-contributing pairs

        if h_max < 1e-15:
            break

        # Reduce hot positions

        alpha = 0.3 + 0.3 * (it / n_iters)

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()

"""
Peak-tracking LSE gradient with backtracking line search + L-BFGS polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. Backtracking line search for guaranteed monotone decrease
2. Adaptive beta schedule with periodic resets to escape local minima
3. Structured perturbation (chirp-based) instead of random
4. Multi-seed L-BFGS polish from diverse starting points
5. Peak-directed coordinate descent for final micro-optimization

import numpy as np

import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and its gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_quick(a, n_iters=80, thresh=1.5, clip=0.5):
    """Quick Fienup iterations for warm start."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_backtracking(a, beta, lr_init=0.01, max_backtrack=10, c1=1e-4):
    """
    Single LSE gradient step with backtracking line search.
    Returns updated a.
    """
    n = len(a)
    f0, grad = lse_score_and_grad(a, beta)
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-15:
        return a

    lr = lr_init
    for _ in range(max_backtrack):
        step = lr
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            lr *= 0.5
            continue
        a_new = a_new / S_new

        f_new, _ = lse_score_and_grad(a_new, beta)
        if f_new <= f0 - c1 * step * gnorm * gnorm:
            return a_new
        lr *= 0.5

    # If no acceptable step found, return a small step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new > 1e-12:
        a_new = a_new / S_new
    else:
        a_new = a.copy()
    return a_new


def lse_adaptive_descent(a, max_time, beta_start=30.0, beta_end=20000.0):
    """
    LSE gradient descent with adaptive beta and backtracking.
    Periodic structured perturbation to escape local minima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    beta = beta_start
    lr = 0.008
    stall_count = 0
    patience = 80
    it = 0
    v = np.zeros(n)
    mu = 0.0  # No momentum initially, add later

    while time.monotonic() - t_start < max_time:
        it += 1

        # Gradually introduce momentum
        if it > 100:
            mu = min(0.7, 0.3 + 0.002 * (it - 100))

        # Get gradient
        f_curr, grad = lse_score_and_grad(a, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Nesterov-style lookahead with small momentum
        if mu > 0:
            a_look = a + mu * v
            a_look = np.maximum(a_look, 0.0)
            S_look = a_look.sum()
            if S_look > 1e-12:
                a_look = a_look / S_look
                _, grad = lse_score_and_grad(a_look, beta)
                gnorm = float(np.linalg.norm(grad))

        # Backtracking line search
        lr_step = lr
        accepted = False
        for _ in range(8):
            if mu > 0:
                v_new = mu * v - lr_step * grad
                a_trial = a + v_new
            else:
                a_trial = a - lr_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                lr_step *= 0.5
                continue
            a_trial = a_trial / S_trial

            f_trial, _ = lse_score_and_grad(a_trial, beta)
            if f_trial <= f_curr - 1e-5 * lr_step * gnorm * gnorm:
                accepted = True
                break
            lr_step *= 0.4

        if not accepted:
            # Take a very small step in negative gradient direction
            lr_step = lr * 0.01
            a_trial = a - lr_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = float(a_trial.sum())
            if S_trial > 1e-12:
                a_trial = a_trial / S_trial
                a = a_trial
            break

        if mu > 0:
            v = v_new
        a = a_trial

        # Track best (exact score periodically)
        if it % 20 == 0 or it == 1:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()
                stall_count = 0
                # Increase learning rate slightly on progress
                lr = min(lr * 1.03, 0.03)
            else:
                stall_count += 1
                if stall_count > patience:
                    # Structured perturbation: add small chirp component
                    x = np.arange(n, dtype=np.float64) / n
                    chirp = np.cos(2 * np.pi * x * x * np.sqrt(2.0))
                    a = a + 0.005 * chirp
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    stall_count = 0
                    lr *= 0.7
                    mu = 0.0
                    v = np.zeros(n)

        # Increase beta (accelerating schedule)
        if it % 50 == 0:
            beta = min(beta * 1.3, beta_end)
            lr *= 0.85

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final

    return best_a

    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 0.5

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coord_descent(a, n_sweeps=5, n_top=30):
    """Coordinate descent targeting top peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_top_local = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                for factor in [0.6, 0.75, 0.9, 1.1, 1.25]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_scaled.copy()
                        a = a_scaled.copy()
                        break
                else:
                    a[j] = old_val

    return best_a


def build_chirp_nonneg(n):
    """Non-negative chirp-based initialization."""
    x = np.arange(n, dtype=np.float64) / n
    # Use |cos| of quadratic phase to get non-negative with structure
    a = np.abs(np.cos(2 * np.pi * x * x * 1.5)) + 0.05
    return a / a.sum()


def build_mixed_init(n, rng):
    """Mixed initialization with multiple frequency components."""
    x = np.linspace(0, 1, n)
    a = 0.5
    for f, amp in [(1.0, 0.3), (2.0, 0.15), (3.0, 0.1), (5.0, 0.05)]:
        a += amp * np.cos(2 * np.pi * f * x + rng.standard_normal() * 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LSE gradient with backtracking, adaptive beta, L-BFGS polish,
    and peak-directed coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Warm start (10% of time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 20.0)
    print(f"=== Phase 1: Warm start (budget {phase1_end - t_start:.0f}s) ===")

    n_target = len(parent) if parent is not None else 6962

    # Start from parent if available
    if parent is not None and len(parent) == n_target:
        a_cur = parent.copy()
        update_best(compute_score(a_cur), a_cur, "parent")
    elif parent is not None:
        a_cur = upsample(parent, n_target)
        update_best(compute_score(a_cur), a_cur, "parent-upsampled")
    else:
        a_cur = build_mixed_init(n_target, rng)

    # Quick Fienup warm start
    a_f = fienup_quick(a_cur, n_iters=60, thresh=1.5, clip=0.4)
    update_best(compute_score(a_f), a_f, "fienup-warm")
    a_cur = best_a.copy()

    # Also try from a different init
    a_alt = build_chirp_nonneg(n_target)
    a_alt = fienup_quick(a_alt, n_iters=40, thresh=1.8, clip=0.3)
    sc_alt = compute_score(a_alt)
    if sc_alt < best_score * 1.01:  # Close enough to be useful
        print(f"  Alternative init score: {sc_alt:.8f}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Main LSE optimization with backtracking (45% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * (total_budget - 20.0)
    print(f"\n=== Phase 2: LSE backtracking (budget {phase2_end - t_start:.0f}s) ===")

    # Run multiple configurations
    configs = [
        {'beta_start': 30.0, 'beta_end': 10000.0, 'time_frac': 0.4},
        {'beta_start': 10.0, 'beta_end': 5000.0, 'time_frac': 0.3},
        {'beta_start': 50.0, 'beta_end': 30000.0, 'time_frac': 0.3},
    ]

    for cfg in configs:
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        alloc = time_left * cfg['time_frac']
        if alloc < 10:
            alloc = time_left * 0.8
        print(f"  LSE: beta [{cfg['beta_start']}, {cfg['beta_end']}], alloc={alloc:.0f}s")

        a_res = lse_adaptive_descent(best_a, max_time=alloc,
                                      beta_start=cfg['beta_start'],
                                      beta_end=cfg['beta_end'])
        update_best(compute_score(a_res), a_res, "lse-backtrack")

    # Try from perturbed version
    if time.monotonic() < phase2_end - 30:
        a_pert = best_a.copy()
        x = np.arange(len(a_pert), dtype=np.float64) / len(a_pert)
        a_pert += 0.01 * np.cos(2 * np.pi * 7 * x)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        time_left = phase2_end - time.monotonic()
        a_res2 = lse_adaptive_descent(a_pert, max_time=time_left * 0.8,
                                       beta_start=20.0, beta_end=8000.0)
        update_best(compute_score(a_res2), a_res2, "lse-perturbed")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS polish (30% of time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n=== Phase 3: L-BFGS polish (budget {phase3_end - t_start:.0f}s) ===")

    lbfgs_betas = [200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase3_end:
            break
        if phase3_end - time.monotonic() < 15:
            break
        print(f"  L-BFGS beta={beta}")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=4000, maxfun=80000)
        update_best(sc, a_opt, f"lbfgs-b{beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak coordinate descent + final LSE (15% of time)
    # ============================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Coordinate descent + final polish (budget {phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 20:
        a_cd = peak_coord_descent(best_a, n_sweeps=4, n_top=25)
        update_best(compute_score(a_cd), a_cd, "coord-descent")

    # Final LSE refinement
    if time.monotonic() < phase4_end - 15:
        time_left = phase4_end - time.monotonic() - 10
        a_final = lse_adaptive_descent(best_a, max_time=time_left,
                                        beta_start=500.0, beta_end=50000.0)
        update_best(compute_score(a_final), a_final, "final-lse")

    # One last L-BFGS with very high beta
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 30000, maxiter=2000, maxfun=40000)
        update_best(sc, a_opt, "final-lbfgs-hibeta")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
