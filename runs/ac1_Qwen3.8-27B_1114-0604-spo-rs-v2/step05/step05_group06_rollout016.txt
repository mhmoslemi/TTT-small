Key innovations over prior approach: Adam optimizer with per-parameter adaptive

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""

    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(64) * (target / 64)
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_value_grad_full(a, beta, k):
    """
    Compute LSE of top-k convolution values and its gradient.
    a is normalized to sum=1. Returns (lse_value, gradient_of_2n*lse).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    M = 2 * n - 1
    k = max(1, min(k, M))

    # Find top-k indices
    if k >= M:
        idx = np.arange(M)
    else:
        idx = np.argpartition(c, M - k)[M - k:]
    c_act = c[idx]
    cmax = float(c_act.max())
    if cmax <= 0 or not np.isfinite(cmax):
        return 1e10, np.zeros(n)

    z = beta * (c_act - cmax)
    e = np.exp(np.clip(z, -700, 0))
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = cmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    # Gradient of 2*n*L / S^2 w.r.t. a: 4*n*corr/S^2 - 4*n*L/S^3
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * float(n) * L / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    val = 2.0 * float(n) * L / (S * S)
    return float(val), grad


def _adam_optimize(a0, stages, time_limit, lr_base=0.005, barrier_mu=0.001):
    """
    Adam optimization with staged K and beta.
    stages: list of (K, beta) tuples.
    Returns (best_score, best_a).
    """
    a = _norm(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    t_start = time.monotonic()
    t_end = t_start + time_limit

    # Adam state
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    total_stages = len(stages)

    for si, (K, beta) in enumerate(stages):
        if time.monotonic() > t_end:
            break

        # Time per stage
        stage_frac = 1.0 / total_stages
        stage_time = time_limit * stage_frac * (0.7 + 0.3 * (si / max(1, total_stages - 1)))
        stage_end = time.monotonic() + stage_time

        # Learning rate for this stage: higher for early stages
        stage_lr = lr_base * (1.0 + 0.5 * (1.0 - si / max(1, total_stages - 1)))

        # Capped iterations per stage
        n_iters = max(50, min(5000, int(stage_time / 0.005)))

        for it in range(n_iters):
            if time.monotonic() > stage_end - 0.5:
                break
            if time.monotonic() > t_end:
                break

            t += 1
            val, grad = _lse_value_grad_full(a, beta, K)

            # Add barrier for non-negativity
            if barrier_mu > 0:
                barrier_grad = -barrier_mu / (a + 1e-4)
                grad = grad + barrier_grad

            # Cosine annealing within stage
            progress = it / max(1, n_iters)
            lr = stage_lr * 0.5 * (1.0 + np.cos(np.pi * progress))

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Enforce non-negativity and normalize
            a_new = np.clip(a_new, 0.0, None)
            S = a_new.sum()
            if S < 1e-10:
                a_new = a
            else:
                a_new = a_new / S
            a = a_new

            # Check score periodically
            if (it + 1) % 200 == 0:
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

        # End of stage: check score
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        a = _norm(a)

    # Final score check
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_sc, best_a


def _subgrad_polish(a, max_iters=50, deadline=None):
    """Exact subgradient descent with adaptive step size on max-convolution."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        cmax = float(c.max())
        if cmax <= 0:
            break

        # Active set: positions within 0.01% of max
        tol = max(1e-8, 1e-4 * cmax)
        active = c >= cmax - tol
        if not active.any():
            active[np.argmax(c)] = True
        n_act = int(active.sum())

        w = np.zeros_like(c)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        obj_val = 2.0 * n * cmax / (S * S)
        grad = 4.0 * n * corr / (S * S) - 4.0 * n * obj_val / (2.0 * S * S * S)
        # Simplified: grad of 2*n*max(c)/S^2
        # = 2*n * (2*corr)/S^2 - 2*n*max(c)*2/S^3
        grad = 4.0 * n * corr / (S * S) - 4.0 * n * cmax / (S * S * S)
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        # Try multiple step sizes
        for scale in (0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            # Try a very small step
            eta = 1e-6 / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq > 1e-8:
                q = q / Sq
                sc = _score(q)
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
            break

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _structured_init(n, rng, mode=0):
    """Generate structured initializations with good autocorrelation properties."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Multi-frequency cosine
        f1, f2 = rng.integers(1, 5), rng.integers(5, 15)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        # Wigner-like distribution
        a = np.sqrt(np.clip(1.0 - x ** 2, 0.01, None))
    elif mode == 2:
        # Bimodal
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.05, 0.2)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.5
    elif mode == 3:
        # Power-law
        a = np.power(x + 0.01, rng.uniform(-2, 2)) * np.power(1 - x + 0.01, rng.uniform(-2, 2))
    else:
        # Smooth random
        noise = rng.standard_normal(n)
        k = max(5, n // 50)
        if k % 2 == 0:
            k += 1
        kk = k // 2
        kern = np.exp(-((np.arange(k) - kk) ** 2) / (2.0 * max(1, kk // 2) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise * 0.5)

    a = np.clip(a, 1e-6, None)
    return a / a.sum()


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _high_beta_polish(a, deadline, max_beta=50000):
    """Run very high beta LSE optimization for precise max minimization."""
    a_cur = _norm(a)
    best_a = a_cur.copy()
    best_sc = _score(a_cur)
    n = len(a_cur)

    betas = [5000, 10000, 20000, 40000, max_beta]
    k_values = [50, 100, 200, 400, 800]

    for i, (beta, k) in enumerate(zip(betas, k_values)):
        if time.monotonic() > deadline - 10:
            break
        stage_time = min(60.0, (deadline - time.monotonic()) / max(1, len(betas) - i))
        sc, a_new = _adam_optimize(a_cur, [(k, beta)], stage_time, lr_base=0.001, barrier_mu=0.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    High-beta polish: beta={beta}, K={k}: {best_sc:.8f}")
        a_cur = _norm(a_new)

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: Adam-based LSE with staged growth and multi-start."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    # Determine target length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192

    # Cap for computational feasibility
    target_n = min(target_n, 16384)
    target_n = max(target_n, 2048)

    print(f"Parent available: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    # --- Generate diverse initializations ---
    inits = []

    # 1. Parent
    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))

    # 2. Spectral inits
    for i in range(5):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # 3. Structured inits
    for mode in range(5):
        inits.append((f'struct{mode}', _structured_init(target_n, rng, mode)))

    # 4. Parent + noise
    if parent is not None:
        p = _upsample(parent, target_n)
        for amp in (0.05, 0.1, 0.2, 0.4):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    # 5. Cosine modulations
    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3, 5, 7):
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 1e-6, None)
        inits.append((f'cos{f}', a / a.sum()))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    top_scores = [f'{s:.6f}({nm})' for s, nm, _ in scored[:8]]
    print(f"Generated {len(scored)} inits. Top: {top_scores}")

    # Select top 4 diverse candidates
    candidates = []
    for sc, name, arr in scored:
        if len(candidates) >= 4:
            break
        # Check diversity from already selected
        diverse = True
        for _, _, existing in candidates:
            corr = float(np.dot(arr, existing))
            if corr > 0.98:
                diverse = False
                break
        if diverse:
            candidates.append((sc, name, arr))

    # Fill if not enough diverse ones
    for sc, name, arr in scored:
        if len(candidates) >= 4:
            break
        if (sc, name, arr) not in candidates:
            candidates.append((sc, name, arr))

    if not candidates:
        candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()
    print(f"Selected {len(candidates)} candidates. Initial best: {best_score:.8f}")

    # --- Main optimization ---
    main_deadline = deadline - 350.0  # Reserve 350s for final polish
    main_time = main_deadline - t_start
    n_cands = len(candidates)

    # Time allocation: give more time to better initial candidates
    weights = [1.0 / (i + 1) for i in range(n_cands)]
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    # Optimization stages: (K, beta)
    # Gradual growth for better convergence
    stages = [
        (3, 20),
        (5, 40),
        (8, 80),
        (12, 150),
        (20, 300),
        (40, 600),
        (80, 1200),
        (150, 2500),
        (300, 5000),
    ]

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        sub_deadline = t_start + (sum(weights[:ci + 1]) * main_time)
        sub_deadline = min(sub_deadline, main_deadline - 10.0)

        if time.monotonic() > sub_deadline:
            print(f"  Skipping candidate {ci+1} ({name}): out of time")
            continue

        n = len(arr_init)
        print(f"\n=== Candidate {ci+1}: '{name}', n={n}, score={sc_init:.8f} ===")

        a_work = _norm(arr_init)

        # Multilevel: optimize at smaller size first
        if n > 2048 and time.monotonic() < sub_deadline - 60.0:
            n_small = max(512, n // 4)
            a_small = _upsample(a_work, n_small)
            sc_small = _score(a_small)
            print(f"  Multilevel: n_small={n_small}, score={sc_small:.8f}")

            # Quick stages at small level
            small_stages = [(3, 30), (10, 100), (30, 400)]
            small_time = min(30.0, (sub_deadline - time.monotonic()) * 0.2)
            if small_time > 10:
                sc_s, a_s = _adam_optimize(a_small, small_stages, small_time, lr_base=0.01)
                a_up = _norm(_upsample(a_s, n))
                sc_up = _score(a_up)
                if sc_up < _score(a_work):
                    a_work = a_up
                    print(f"  Multilevel improved: {sc_up:.8f}")

        # Main trajectory
        traj_time = sub_deadline - time.monotonic() - 5.0
        if traj_time > 10:
            sc_traj, seq_traj = _adam_optimize(a_work, stages, traj_time, lr_base=0.008)
            if sc_traj < best_score:
                best_score = sc_traj
                best_seq = seq_traj.copy()
                print(f"  *** Global best updated: {best_score:.8f} ***")
        else:
            sc_traj, seq_traj = _score(a_work), a_work

        print(f"  End candidate {ci+1}: score={sc_traj:.8f}")

    # --- Final polish phase ---
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 15.0

    # 1. High-beta LSE polish
    if time.monotonic() < final_deadline - 60:
        sc_hb, a_hb = _high_beta_polish(best_seq, final_deadline - 60)
        if sc_hb < best_score:
            best_score = sc_hb
            best_seq = a_hb.copy()
            print(f"  After high-beta: {best_score:.8f}")

    # 2. Subgradient polish
    if time.monotonic() < final_deadline - 20:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=30, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  After subgrad: {best_score:.8f}")

    # 3. Random restarts from best
    if time.monotonic() < final_deadline - 30:
        for restart in range(4):
            if time.monotonic() > final_deadline - 15:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05 + 0.05 * restart)
            chunk_time = min(30.0, final_deadline - time.monotonic() - 15)
            if chunk_time < 5:
                break
            sc_r, a_r = _adam_optimize(p, [(20, 500), (50, 1500), (100, 3000)], chunk_time, lr_base=0.003)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    # 4. Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=20, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
