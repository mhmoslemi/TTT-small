def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)

    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S

    return score, grad

# Adam optimizer state
class AdamState:
    def __init__(self, n, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, x, grad, beta):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        x_new = x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return x_new

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (unnormalized)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -80))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S

    return score, grad

def peak_directed_cd(a, n_sweeps=20, top_k=30, n_candidates=7):
    """
    Coordinate descent that targets the peak of the convolution.
    At each step, identify the peak position, find its main contributors,
    reduce them, and redistribute mass to low-contribution positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top few peaks to reduce concentration
        n_peaks = min(3, n)
        peak_positions = np.argsort(c)[::-1][:n_peaks]

        for k_star in peak_positions:
            j_start = max(0, int(k_star) - n + 1)
            j_end = min(int(k_star), n - 1)
            if j_end <= j_start:
                continue

            indices = np.arange(j_start, j_end + 1)
            # Contribution of a[j] to (a*a)_{k_star} is a[j] * a[k_star - j]
            contrib = a[indices] * a[k_star - indices]
            total_len = len(contrib)
            top_count = min(top_k, max(1, total_len // 4))
            bottom_count = max(1, total_len - top_count)

            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[:bottom_count]

            for li in top_local:
                j = indices[li]
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                best_frac = 1.0
                best_local_sc = best_sc
                fracs = np.linspace(0.3, 0.95, n_candidates)
                for frac in fracs:
                    a[j] = old_val * frac
                    reduction = old_val * (1.0 - frac)
                    if reduction > 1e-15:
                        per = reduction / len(bottom_local)
                        for bli in bottom_local:
                            a[indices[bli]] += per
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_frac = frac
                a[j] = old_val * best_frac
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        _report(sc, a_opt, f"restart#{trial} b={beta}")

        # If improved, do a quick CD polish
        if sc < best_score + 1e-8:
            a_cd = peak_directed_cd(a_opt, n_sweeps=10, top_k=20)
            _report(compute_score(a_cd), a_cd, f"restart#{trial} CD")

    # ==================================================================
    # Final: One more high-beta pass
    # ==================================================================
    if time.monotonic() < deadline - 30:
        print(f"\nFinal high-beta pass...")
        sc, a_opt = run_lbfgs(best_a, 2000000, maxiter=30000, maxfun=100000)
        _report(sc, a_opt, "final ultra")

    # Final CD polish
    if time.monotonic() < deadline - 10:
        a_cd = peak_directed_cd(best_a, n_sweeps=15, top_k=30)
        _report(compute_score(a_cd), a_cd, "final CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
