    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p_sum = float(exp_scaled.sum())
    L = c_max + float(np.log(p_sum)) / beta
    p = exp_scaled / p_sum
    
    # Gradient of LSE w.r.t. a:
    # dL/d(a[i]) = 2 * sum_m a[m] * p[m+i]  (cross-correlation of a with p)
    # Computed via FFT convolution with reversed p
    M_conv = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(a, M_conv)
    P_fft = np.fft.rfft(p[::-1], M_conv)
    conv_res = np.fft.irfft(A_fft * P_fft, M_conv)
    # g[i] = conv_res[2n-2-i] for i=0..n-1
    g = conv_res[n-1:2*n-1][::-1]
    grad_L = 2.0 * g
    
    # Gradient of score = 2n*L/S^2
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad_score = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    
    return score, grad_score


def gs_fienup(a, n_iters=500, alpha_start=0.7, alpha_end=0.1, M=None):
    """GS-Fienup with flat-convolution target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()
    
    b_target = np.zeros(M)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_objective(x, beta):
    return lse_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_subgradient(a0, n_iters=5000, top_m=15, momentum=0.9, step_init=0.02):
    """Nesterov-accelerated subgradient on top-M peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    v = np.zeros(n)  # velocity for Nesterov
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    prev_grad = None
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute subgradient from top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Weight peaks by their excess over mean
        c_mean = S * S / len(c)
        weights = np.zeros(len(c))
        for k in top_idx:
            weights[k] = max(c[k] - c_mean, 0) + 0.1 * c_mean
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights[top_idx] = 1.0 / top_m
            w_sum = 1.0
        weights = weights / w_sum
        
        # Weighted gradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = weights[k]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * float(np.sum(weights * c)) / S3)
        
        # Nesterov update
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        gap = max(sc - sc_lb, 1e-10)
        step = min(step, 2.0 * gap / (gn * gn + 1e-30))
        step = min(step, 0.05)
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            v = np.zeros(n)
            break
        a = a_new / S_new
        step *= 0.9999
        
        # Decay momentum slightly
        if it % 500 == 0 and it > 0:
            momentum = max(0.5, momentum * 0.95)
    
    return best_a


def single_peak_gradient(a0, n_iters=5000, momentum=0.7, step_init=0.01):
    """Projected gradient descent targeting only the single worst peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        
        # Gradient of c[k_star] w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end + 1]
        grad_c = np.zeros(n)
        grad_c[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        gap = max(sc - sc_lb, 1e-10)
        step = min(step, 2.0 * gap / (gn * gn + 1e-30))
        step = min(step, 0.03)
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            v = np.zeros(n)
            break
        a = a_new / S_new
        step *= 0.9999
    
    return best_a


def coordinate_descent(a0, n_sweeps=3, n_top=40):
    """Coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j
            for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.15]:
                cv = a_old_j * f
                cv = max(0.0, min(1000.0, cv))
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def perturb_and_refine(a0, n_perturb=20, perturb_scale=0.01, refine_iters=200):
    """Perturb random elements and re-optimize with short GS."""
    n = len(a0)
    best_a = a0.copy()
    best_sc = compute_score(a0)
    rng = np.random.default_rng(123)
    
    for _ in range(n_perturb):
        a = best_a.copy()
        # Perturb a random subset
        n_change = max(5, n // 100)
        indices = rng.choice(n, size=n_change, replace=False)
        a[indices] *= (1.0 + perturb_scale * rng.standard_normal(n_change))
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            continue
        a = a / S
        
        # Quick GS refinement
        M = 1 << (2 * n - 1).bit_length()
        b_target = np.zeros(M)
        b_target[:2*n-1] = 1.0 / (2*n - 1)
        T = np.fft.fft(b_target)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)
        
        for it in range(refine_iters):
            alpha = 0.5 * (1.0 - it / refine_iters) + 0.1
            A = np.fft.fft(a, M)
            A_mag = np.maximum(np.abs(A), 1e-20)
            A_phase = np.angle(A)
            new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
            new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
            A_new = new_mag * np.exp(1j * new_phase)
            a_new = np.fft.ifft(A_new, M).real[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
