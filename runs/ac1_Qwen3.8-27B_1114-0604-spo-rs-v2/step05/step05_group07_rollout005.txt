"""
Optimized step function search using coordinate descent with exact line search,
combined with LSE-smoothed L-BFGS for global refinement.
Key innovation: efficient 1D line search per coordinate exploiting the structure
of the self-convolution, with adaptive coordinate ordering and restart strategy.

import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS
_current_beta = 1000.0

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

                # Update c (using old a)
                c[j:j+n] += 2.0 * d * a
                if 0 <= 2*j < 2*n-1:
                    c[2*j] += d * d
                a[j] = v + d
                S += d
                sweep_improved = True
        
        new_score = 2.0 * n * float(c.max()) / (S * S)
        if new_score < best_score:
            best_score = new_score
        
        if not sweep_improved:
            break
    
    return best_score

```python
def _score_delta(S, n, c_seg, a, max_unchanged, d, local_2j):
    """Score for a given delta in coordinate descent."""
    Sn = S + d
    if Sn < 1e-12:
        return np.inf
    ns = c_seg + 2.0 * d * a
    if 0 <= local_2j < len(ns):
        ns[local_2j] += d * d
    nm = max(max_unchanged, float(np.max(ns)))
    return 2.0 * n * nm / (Sn * Sn)
```

This is a top-level function, no closures. Good.

But wait, `ns = c_seg + 2.0 * d * a` creates a new array of length n each time. For n=7000, that's 56KB per call, and we call it ~35 times per coordinate, 7000 coordinates per sweep. Total memory allocation per sweep: 35 * 7000 * 56KB ≈ 13.7 GB. That's a lot of allocation!
