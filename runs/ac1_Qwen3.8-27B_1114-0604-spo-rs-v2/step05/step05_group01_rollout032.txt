"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS-B with adaptive continuation, active-set equalization,
log-space reparameterization, and structured perturbations.
Key improvements over baseline: better continuation schedule, active-set
targeted subgradient, feasibility-based binary search on T, and diverse
initializations informed by the U-shape structure.

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Compute cross-correlation: result[i] = sum_k w[k] * a[k-i] (valid range)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    nw = int(w.size)
    if n == 0 or nw == 0:
        return np.empty(n, dtype=np.float64)
    M = n + nw - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_d(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64)
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    return np.maximum(d, 0.0)


def lse_score_grad(a, beta, pen_sum=1e-4):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d = compute_d(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = cross_corr(a, w)
    score = 2.0 * L + pen_sum * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * pen_sum * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun, pen_sum=1e-4):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = int(max(maxfun, 64))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 5.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    def fun(x):
        return lse_score_grad(x, beta, pen_sum)

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def active_weights_soft(d, k):
    """Soft active set: weight peaks by their excess over the k-th largest."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0:
        return np.ones(m, dtype=np.float64) / float(m)
    sorted_d = np.sort(d)[::-1]
    threshold = sorted_d[min(k, m) - 1]
    excess = np.maximum(d - threshold, 0.0)
    total = float(excess.sum())
    if total < 1e-15:
        return active_weights_topk(d, k)
    return excess / total


def subgrad_step(a, eta, weights):
    """Multiplicative subgradient step using given weights on peak lags."""
    d = compute_d(a)
    n = len(a)
    if d.size == 0:
        return a
    h = cross_corr(a, weights)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return a
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    a_new = np.exp(log_a)
    return normalize_a(a_new)


def equalize_peaks(a, max_iters=20, deadline=None):
    """Iteratively equalize top peaks using subgradient steps with adaptive weights."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        d = compute_d(a)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax < 1e-10:
            break

        improved = False
        for k in (1, 2, 3, 5, 8):
            for wtype in ('topk', 'soft'):
                if wtype == 'topk':
                    w = active_weights_topk(d, k)
                else:
                    w = active_weights_soft(d, k)
                for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                    if deadline is not None and time.monotonic() > deadline:
                        break
                    a_new = subgrad_step(a, eta, w)
                    sc = exact_score_a(a_new)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        improved = True
                    if np.isfinite(sc) and sc < exact_score_a(a):
                        a = a_new
                        d = compute_d(a)
                        dmax = float(np.max(d))
        if not improved:
            break

    return best_sc, best_a


def generate_inits(n, rng, parent=None, cap=25):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # U-shapes (high at ends) - key structure
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(0.1 + alpha * np.abs(2.0 * x - 1.0))
        add(0.1 + alpha * (0.5 - np.cos(np.pi * x)))

    # Squared U-shape
    for alpha in (1.0, 2.0, 3.0):
        add(0.1 + alpha * (np.abs(2.0 * x - 1.0) ** 2))

    # Power U-shapes
    for p in (1.5, 2.0, 3.0, 4.0):
        add(0.1 + np.abs(2.0 * x - 1.0) ** p)

    # Sine-based U (zero at center)
    add(0.1 + np.sin(np.pi * x) ** 2)
    add(0.1 + (1.0 - np.cos(2.0 * np.pi * x)) ** 2)
    add(0.1 + np.abs(np.sin(np.pi * x)))

    # Asymmetric U-shapes
    for t in (0.1, 0.3, 0.5, 0.7):
        add(0.1 + np.abs(2.0 * x - 1.0) + t * (2.0 * x - 1.0))
        add(0.1 + np.exp(-((x - 0.5) / 0.3) ** 2) + np.exp(-((x) / 0.15) ** 2) + np.exp(-(((x-1)) / 0.15) ** 2))

    # Bimodal peaks near edges
    for width in (0.05, 0.1, 0.15, 0.2):
        for pos in (0.15, 0.25, 0.35):
            add(np.exp(-(((x - pos) / width) ** 2)) + np.exp(-(((x - (1 - pos)) / width) ** 2)) + 0.01)

    # Cosine shapes
    for f in (1, 2, 3, 4):
        for alpha in (0.5, 1.0, 2.0):
            add(0.5 + alpha * np.cos(2.0 * np.pi * f * x))
            add(0.5 + alpha * np.cos(np.pi * f * x + np.pi / 2))

    # Linear ramps
    for alpha in (0.5, 1.0, 2.0):
        add(0.1 + alpha * x)
        add(0.1 + alpha * (1.0 - x))

    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)

    # Random smooth (low-pass)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(0.5 + 0.5 * noise)

    # Random trig
    for _ in range(3):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Pure random
    add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d = compute_d(p)
        if d.size > 0:
            w = active_weights_topk(d, int(rng.integers(1, 6)))
            q = subgrad_step(p, strength * 0.5, w)
        else:
            q = p.copy()
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # U-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(7, max(9, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Cut off a fraction of the smallest elements
        frac = float(rng.uniform(0.05, 0.2))
        thresh = np.percentile(p, frac * 100)
        q = np.where(p > thresh, p, 0.0)
    elif mode == 8:
        # Random frequency-domain perturbation
        N = next_power_of_two(2 * n)
        A = np.fft.rfft(np.concatenate([p, np.zeros(N - n)]), N)
        mask = np.ones(len(A), dtype=bool)
        n_cut = int(n * 0.3)
        freqs = np.abs(np.fft.rfftfreq(N, d=1.0 / N))
        high_freq = freqs > float(np.sort(freqs)[n_cut])
        mask[high_freq] = False
        A[mask] *= (1.0 + 0.1 * rng.standard_normal(mask.sum()))
        A[~mask] *= (1.0 + 0.3 * rng.standard_normal((~mask).sum()))
        q = np.fft.irfft(A, N)[:n]
    else:
        # Blend with a U-shape
        x = np.linspace(0.0, 1.0, n)
        u = 0.1 + 2.0 * np.abs(2.0 * x - 1.0)
        t = float(rng.uniform(0.3, 0.7))
        q = (1.0 - t) * p + t * normalize_a(u)

    return normalize_a(np.clip(q, 0.0, None))


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.75 * t_left) / max(est, 1e-4))
    mf = max(64, min(5000, mf))
    return mf


def optimize_trajectory(a_start, deadline, rng, label="traj"):
    """Run continuation LSE optimization with adaptive beta and equalization."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 10.0))

    # Continuation schedule: more time at high beta
    stages = [
        (50.0, 0.04),
        (150.0, 0.06),
        (400.0, 0.08),
        (1000.0, 0.10),
        (3000.0, 0.12),
        (8000.0, 0.14),
        (20000.0, 0.16),
        (50000.0, 0.14),
        (100000.0, 0.16),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        # Multiple LSE runs at this beta
        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.9, n)
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)

        # Equalize peaks between stages
        if time.monotonic() < sub_dead - 5.0:
            sc_eq, a_eq = equalize_peaks(a, max_iters=15, deadline=sub_dead)
            if np.isfinite(sc_eq) and sc_eq < best_sc:
                best_sc = sc_eq
                best_a = a_eq.copy()
                print("[{}] equalize best {:.8f} beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_eq)

    return float(best_sc), best_a

            best_seq = starts[i].copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main optimization on best start (majority of time)
    main_deadline = deadline - 300.0
    main_budget = max(60.0, main_deadline - t_start)

    # Allocate time: 60% to best, 25% to second, 15% to third
    if len(starts) >= 3:
        fracs = [0.60, 0.25, 0.15]
    elif len(starts) == 2:
        fracs = [0.75, 0.25]
    else:
        fracs = [1.0]

    cum = 0.0
    for rank, si in enumerate(range(len(starts))):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_deadline))

        if time.monotonic() > sub_dead - 15.0:
            continue

        sq = starts[si].copy()
        # For secondary starts, blend with current best
        if rank > 0:
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            t_blend = 0.4 if rank == 1 else 0.6
            sq = normalize_a((1.0 - t_blend) * normalize_a(sq) + t_blend * normalize_a(base))

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Equalization on the best
    if time.monotonic() < deadline - 200.0:
        sc_eq, a_eq = equalize_peaks(best_seq, max_iters=30, deadline=deadline - 200.0)
        if np.isfinite(sc_eq) and sc_eq < best_score:
            best_score = sc_eq
            best_seq = a_eq.copy()
            print("Equalization: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts
    perturb_done = 0
    max_perturb = 8
    while time.monotonic() < deadline - 120.0 and perturb_done < max_perturb:
        left = float(deadline - 120.0 - time.monotonic())
        if left < 45.0:
            break
        n_remaining = max_perturb - perturb_done
        chunk = t_start + total_budget - 120.0 - (left / n_remaining) * 0.5
        chunk = min(chunk, t_start + total_budget - 120.0)

        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb {}: {:.8f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 5: Final high-beta LSE polish
    if time.monotonic() < deadline - 40.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 10.0:
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 80000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

            if time.monotonic() < deadline - 25.0:
                rem = float(deadline - time.monotonic() - 5.0)
                if rem > 8.0:
                    mf = estimate_maxfun(rem * 0.85, len(best_seq))
                    sc, seq = run_lbfgs(best_seq, 200000.0, deadline, mf)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print("Very high beta: {:.8f}".format(best_score))

    # Phase 6: Final equalization
    if time.monotonic() < deadline - 10.0:
        sc_eq, a_eq = equalize_peaks(best_seq, max_iters=15, deadline=deadline - 5.0)
        if np.isfinite(sc_eq) and sc_eq < best_score:
            best_score = sc_eq
            best_seq = a_eq.copy()
            print("Final equalize: {:.8f}".format(best_score))

    # Final cleanup and verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
