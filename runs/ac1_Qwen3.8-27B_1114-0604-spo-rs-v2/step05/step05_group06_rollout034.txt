"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LSE optimization with high-beta sharp approximation,
active-set Newton polishing, multi-length exploration, and diverse restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters, deadline):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set with adaptive threshold
        threshold = dmax - 1e-6 * max(dmax, 1.0)
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _newton_polish(a, max_iters, deadline):
    """Newton-type polish: use quasi-Newton on the active set objective."""
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Use a running estimate of the Hessian via BFGS-like updates
    # Since n can be large, we'll use L-BFGS style memory
    m_mem = 10  # memory for L-BFGS
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _self_conv(a)
        d = float(n) * c / (a.sum() ** 2)
        dmax = float(d.max())
        
        # Very tight active set
        active = d >= dmax - 1e-8 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        S = float(a.sum())
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Try a "Newton-like" step: scale gradient by an effective curvature estimate
        # Use the ratio of recent gradient changes to estimate curvature
        best_step = None
        best_sc_new = best_sc
        
        for alpha in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
            eta = alpha / (gnorm * max(1.0, gnorm))
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_sc_new:
                best_sc_new = sc
                best_step = q.copy()
        
        if best_step is None:
            break
        if best_sc_new < best_sc:
            best_sc = best_sc_new
            best_a = best_step.copy()
        a = best_step
    
    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _run_trajectory(a0, deadline, label, rng):
    """Optimize with progressive K and beta."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    total_time = max(10.0, deadline - t0 - 15.0)
    
    # More stages with higher betas
    stages = [
        (2, 20.0, 0.04),
        (4, 50.0, 0.06),

        (15, 200.0, 0.10),

        (60, 1000.0, 0.14),

        (500, 10000.0, 0.14),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:

        if stage_budget < 5.0:

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based circulant-embedding Newton steps on the active set (exploiting
the sum-Toeplitz/Hankel structure of the Hessian), combined with progressive
LSE smoothing, multi-length exploration, and diverse restarts for polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 12000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 60}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _newton_direction(a, w_full):
    """
    Compute Newton direction for minimizing 2n * w^T conv(a,a) with sum(a)=1.
    Exploits the sum-Toeplitz (Hankel) structure of the Hessian H[p,q] = 8n*w[p+q]
    via circulant embedding solved by FFT.
    Returns the direction d such that a - d is the Newton update (before projection).
    """
    n = len(a)
    M = 2 * n - 1
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)

    # Gradient of 2n * w^T conv(a,a) / S^2 (for normalized a, S=1)
    corr = _cross_corr(a, w_full)
    LDS = float(w_full @ _self_conv(a)) / (S * S)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0)

    # Newton equation: H * d = -grad
    # H[p,q] = 8n * w_full[p+q] (sum-Toeplitz)
    # Use circulant embedding of size 2n-1
    Nft = 2 * n - 1
    Nft_p2 = _next_pow2(Nft)

    # h[k] = H's "generating function": h[k] = 8n * w_full[k] for k=0..2n-2
    h = 8.0 * float(n) * w_full  # length 2n-1
    rhs_full = np.zeros(Nft)
    rhs_full[:n] = -grad

    H_fft = np.fft.fft(h, Nft_p2)
    RHS_fft = np.fft.fft(rhs_full, Nft_p2)

    # Regularized solve
    threshold = 1e-25 * max(1.0, float(np.max(np.abs(H_fft))))
    X_fft = np.zeros(Nft_p2, dtype=np.complex128)
    mask = np.abs(H_fft) > threshold
    X_fft[mask] = RHS_fft[mask] / H_fft[mask]

    x = np.fft.ifft(X_fft, Nft_p2).real[:n]
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    return x


def _newton_polish(a, max_iters, deadline):
    """
    Projected Newton method using FFT-based circulant embedding for the
    active-set objective. Falls back to gradient if Newton direction is
    not a descent direction.
    """
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S = float(a.sum())
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within tiny epsilon of max
        eps = max(1e-10, 1e-8 * dmax)
        active = d >= dmax - eps
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        if n_act > 5000:
            # Cap active set for numerical stability
            idx = np.where(active)[0]
            vals = d[idx]
            top_k = 5000
            top_idx = idx[np.argsort(vals)[-top_k:]]
            active = np.zeros_like(d)
            active[top_idx] = True
            n_act = top_k

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act

        # Try Newton direction
        newton_d = _newton_direction(a, w)
        newton_d = np.nan_to_num(newton_d, nan=0.0)
        nd_norm = float(np.linalg.norm(newton_d))

        # Gradient for fallback
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        # Determine search direction
        use_newton = False
        if nd_norm > 1e-15:
            # Check if Newton direction is descent: grad^T * (-d_newton) < 0
            descent = float(np.dot(grad, -newton_d))
            if descent > 0:
                use_newton = True

        if use_newton:
            direction = -newton_d / nd_norm
        else:
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            direction = -grad / gnorm

        # Line search
        best_step = None
        best_new_sc = best_sc
        scale_base = 1.0 / max(nd_norm, 1e-10) if use_newton else 1.0 / max(float(np.linalg.norm(grad)), 1e-10)

        for scale in (0.01, 0.03, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
            eta = scale * scale_base
            q = a - eta * (-direction if not use_newton else -newton_d)
            # Simplify: q = a + eta * direction_to_go
            q = a + eta * direction
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            # Try gradient-only with smaller steps
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            for scale in (0.0001, 0.0003, 0.001, 0.003, 0.01):
                eta = scale / gnorm
                q = a - eta * grad
                q = np.clip(q, 0.0, None)
                Sq = q.sum()
                if Sq < 1e-8:
                    continue
                q = q / Sq
                sc = _score(q)
                if sc < best_new_sc:
                    best_new_sc = sc
                    best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _subgrad_polish(a, max_iters, deadline):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        threshold = dmax - 1e-7 * max(dmax, 1.0)
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _run_trajectory(a0, deadline, label, rng):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    total_time = max(10.0, deadline - t0 - 20.0)

    stages = [
        (2, 20.0, 0.04),
        (4, 50.0, 0.05),
        (8, 100.0, 0.07),
        (15, 200.0, 0.08),
        (30, 500.0, 0.10),
        (60, 1000.0, 0.12),
        (120, 2500.0, 0.12),
        (250, 5000.0, 0.12),
        (500, 10000.0, 0.13),
        (1000, 20000.0, 0.13),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + cum_time * total_time
        stage_end = min(stage_end, deadline - 20.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        n_calls = min(5, max(1, int(stage_budget / 12)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 600)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = _norm(a_new)

    if time.monotonic() < deadline - 20.0:
        sc, a_sg = _subgrad_polish(a, max_iters=15, deadline=deadline - 15.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()

    if time.monotonic() < deadline - 15.0:
        sc, a_nt = _newton_polish(a, max_iters=20, deadline=deadline - 10.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_nt.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    if parent is not None:
        parent_n = len(parent)
        print(f"Parent loaded: n={parent_n}, score={_score(parent):.8f}")
        # Try a few lengths around the parent
        lengths = sorted(set([
            parent_n,
            max(2048, parent_n // 2),
            min(parent_n, 16384),
        ]))
    else:
        lengths = [4096, 8192]

    print(f"Lengths to try: {lengths}")
    print(f"Total budget: {budget_s}s")

    # === Phase 1: Multi-length quick optimization ===
    print("\n=== Phase 1: Multi-length exploration ===")
    phase1_time = min(350.0, (deadline - t_start) * 0.18)
    phase1_deadline = t_start + phase1_time

    candidates = []
    per_length = phase1_time / max(1, len(lengths))

    for ln in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()

        if parent is not None and ln == len(parent):
            arr = parent.copy()
        elif parent is not None:
            arr = _norm(_upsample(parent, ln))
        else:
            arr = _spectral_init(ln, rng)

        init_sc = _score(arr)
        print(f"  n={ln}: init={init_sc:.8f}")

        # Quick LSE
        t_q = min(time.monotonic() + per_length * 0.5, phase1_deadline - 10)
        if t_q - time.monotonic() > 8:
            sc, a_opt = _run_lbfgs(arr, 1000, 200, t_q - time.monotonic(), 400)
            if sc < init_sc:
                arr, init_sc = a_opt, sc

        # Quick Newton
        if time.monotonic() < phase1_deadline - 8:
            sc, a_nt = _newton_polish(arr, max_iters=10, deadline=phase1_deadline - 8)
            if sc < init_sc:
                arr, init_sc = a_nt, sc

        candidates.append((init_sc, ln, arr))
        print(f"  n={ln}: best={init_sc:.8f} ({time.monotonic()-t0:.0f}s)")

    candidates.sort(key=lambda x: x[0])
    best_sc, best_n, best_a = candidates[0]
    print(f"Phase 1 winner: n={best_n}, score={best_sc:.8f}")

    # === Phase 2: Main trajectory on best ===
    print(f"\n=== Phase 2: Main optimization (n={best_n}) ===")
    phase2_end = t_start + phase1_time + (deadline - t_start - phase1_time) * 0.50
    sc2, a2 = _run_trajectory(best_a, phase2_end, "main", rng)
    if sc2 < best_sc:
        best_sc, best_a = sc2, a2
    print(f"Phase 2: {best_sc:.8f}")

    # Also try second-best length if time allows
    if len(candidates) > 1 and time.monotonic() < phase2_end - 60:
        sc2b, best_n2, best_a2 = candidates[1]
        if best_n2 != best_n:
            sub_dl = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            sc2b, a2b = _run_trajectory(best_a2, sub_dl, "alt", rng)
            if sc2b < best_sc:
                best_sc, best_a, best_n = sc2b, a2b, best_n2
                print(f"  Alt length won: n={best_n2}, score={best_sc:.8f}")

    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse restarts ===")
    phase3_end = deadline - 250.0
    n_restarts = 10
    per_restart = max(5.0, (phase3_end - time.monotonic()) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase3_end:
            break
        if per_restart < 8:
            break

        # Diverse perturbation strategies
        n = len(best_a)
        if r % 4 == 0:
            p = _smooth_perturb(best_a, rng, strength=0.03 + 0.05 * (r // 4))
        elif r % 4 == 1:
            # Block perturbation
            p = best_a.copy()
            bs = max(20, n // 40)
            start = int(rng.integers(0, max(1, n - bs)))
            p[start:start + bs] *= rng.uniform(0.3, 2.0, bs)
            p = _norm(p)
        elif r % 4 == 2:
            # Low-freq modulation
            x = np.linspace(0, 2 * np.pi, n)
            f = int(rng.integers(1, 15))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            p = best_a * (1.0 + amp * np.cos(f * x + phase))
            p = _norm(p)
        else:
            # Spectral re-initialization blended with current
            spec_init = _spectral_init(n, rng)
            mix = rng.uniform(0.2, 0.5)
            p = (1 - mix) * best_a + mix * spec_init
            p = _norm(p)

        # Optimize the perturbed candidate
        t_r_end = time.monotonic() + per_restart * 0.6
        sc_r, a_r = _run_lbfgs(p, 3000, 300, max(5, t_r_end - time.monotonic()), 400)

        if sc_r < best_sc:
            # Polish it
            if time.monotonic() < phase3_end - 5:
                sc_r2, a_r2 = _newton_polish(a_r, max_iters=10, deadline=phase3_end - 5)
                if sc_r2 < best_sc:
                    best_sc, best_a = sc_r2, a_r2
                    print(f"  Restart {r} (newton): {best_sc:.8f}")
                else:
                    best_sc, best_a = sc_r, a_r
                    print(f"  Restart {r}: {best_sc:.8f}")
            else:
                best_sc, best_a = sc_r, a_r
                print(f"  Restart {r}: {best_sc:.8f}")

    # === Phase 4: Final aggressive polish ===
    print(f"\n=== Phase 4: Final polish (score={best_sc:.8f}) ===")
    final_dl = deadline - 25.0

    # Newton polish (multiple rounds)
    for round_id in range(3):
        if time.monotonic() > final_dl - 30:
            break
        sc_f, a_f = _newton_polish(best_a, max_iters=25, deadline=final_dl - 25)
        if sc_f < best_sc:
            best_sc, best_a = sc_f, a_f
            print(f"  Newton round {round_id}: {best_sc:.8f}")
        else:
            break

    # High-beta LSE
    for k_f, beta_f in [(500, 10000), (1500, 30000), (3000, 60000)]:
        if time.monotonic() > final_dl - 30:
            break
        remaining = final_dl - time.monotonic() - 25
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_a, beta_f, k_f, remaining * 0.4, 1000)
        if sc_f < best_sc:
            best_sc, best_a = sc_f, a_f
            print(f"  LSE K={k_f}, beta={beta_f}: {best_sc:.8f}")

    # Finest subgradient
    if time.monotonic() < final_dl - 15:
        sc_sg, a_sg = _subgrad_polish(best_a, max_iters=30, deadline=final_dl - 10)
        if sc_sg < best_sc:
            best_sc, best_a = sc_sg, a_sg
            print(f"  Final subgrad: {best_sc:.8f}")

    # One more Newton
    if time.monotonic() < final_dl - 8:
        sc_nt, a_nt = _newton_polish(best_a, max_iters=15, deadline=final_dl - 3)
        if sc_nt < best_sc:
            best_sc, best_a = sc_nt, a_nt
            print(f"  Very final Newton: {best_sc:.8f}")

    # Prepare and verify output
    out = _norm(best_a)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_sc

    print(f"\nFINAL: internal={best_sc:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
