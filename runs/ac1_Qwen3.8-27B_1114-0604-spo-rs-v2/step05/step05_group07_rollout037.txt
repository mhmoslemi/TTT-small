import numpy as np
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Score = 2n * max(conv) / sum^2"""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

"""
Exact Coordinate Descent (ECD) optimizer for the step function scoring problem.
Key idea: For each coordinate, solve the 1D subproblem EXACTLY by finding
all breakpoints of the piecewise-quotient objective and evaluating at critical points.
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def solve_1d_subproblem(c0, p, q, S_0, x_max):
    """
    Solve: min_{x >= 0} max_k (c0[k] + p[k]*x + q[k]*x^2) / (S_0 + x)^2
    Returns the optimal x in [0, x_max].
    """
    K = len(c0)
    
    candidates = [0.0, x_max]
    
    # Interior critical points for q=1 lags
    for i in range(K):
        if q[i] > 0.5:
            denom = 2.0 * S_0 - p[i]
            if abs(denom) > 1e-10:
                xs = (2.0 * c0[i] - p[i] * S_0) / denom
                if 0.0 < xs < x_max:
                    candidates.append(xs)
    
    # Breakpoints between all pairs
    for i in range(K):
        for l in range(i + 1, K):
            A = q[i] - q[l]
            B = p[i] - p[l]
            C = c0[i] - c0[l]
            
            if abs(A) < 1e-10:
                if abs(B) > 1e-10:
                    xp = -C / B
                    if 0.0 < xp < x_max:
                        candidates.append(xp)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sqrt_d = np.sqrt(disc)
                    for s in [1, -1]:
                        xp = (-B + s * sqrt_d) / (2.0 * A)
                        if 0.0 < xp < x_max:
                            candidates.append(xp)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= 0) & (candidates <= x_max)]
    
    if len(candidates) == 0:
        return 0.0
    
    # Evaluate at all candidates
    c_vals = c0[None, :] + p[None, :] * candidates[:, None] + q[None, :] * candidates[:, None] ** 2
    S_vals = S_0 + candidates
    obj = c_vals.max(axis=1) / (S_vals * S_vals)
    
    best_idx = int(np.argmin(obj))
    return float(candidates[best_idx])


def exact_coord_descent_step(a, c, S, j, K=30):
    """
    Perform one exact coordinate descent step for coordinate j.
    Returns new value of a[j] and the change in c.
    """
    n = len(a)
    a_j = a[j]
    
    # Get active lags (top K by current c value)
    top_k = np.argsort(c)[::-1][:K]
    
    # Compute c0, p, q for each active lag
    c0 = np.zeros(K)
    p = np.zeros(K)
    q = np.zeros(K)
    
    for idx, k in enumerate(top_k):
        k_int = int(k)
        if k_int == 2 * j and 0 <= j < n:
            c0[idx] = c[k_int] - a_j * a_j
            p[idx] = 0.0
            q[idx] = 1.0
        elif 0 <= k_int - j < n:
            c0[idx] = c[k_int] - 2.0 * a_j * a[k_int - j]
            p[idx] = 2.0 * a[k_int - j]
            q[idx] = 0.0
        else:
            c0[idx] = c[k_int]
            p[idx] = 0.0
            q[idx] = 0.0
    
    S_0 = S - a_j
    x_max = max(20.0 * a_j, 5.0 * S / n, 1.0)
    
    x_new = solve_1d_subproblem(c0, p, q, S_0, x_max)
    
    return x_new


def exact_coord_descent(a, n_passes=20, K=30, seed=42, verbose=True, deadline=None):
    """
    Run exact coordinate descent for n_passes full passes.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    c = fast_conv(a)
    
    best_a = a.copy()
    best_c = c.copy()
    best_sc = 2.0 * n * float(c.max()) / (S * S)
    
    rng = np.random.default_rng(seed)
    
    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            x_new = exact_coord_descent_step(a, c, S, j, K)
            a_j = a[j]
            
            if abs(x_new - a_j) < 1e-12 * max(1.0, a_j):
                continue
            
            # Update a[j]
            delta = x_new - a_j
            a[j] = x_new
            S += delta
            
            # Update convolution c
            # k = 2j: c[2j] += x_new^2 - a_j^2
            if 2 * j < 2 * n - 1:
                c[2 * j] += x_new * x_new - a_j * a_j
            
            # k where 0 <= k-j < n and k != 2j: c[k] += 2*delta*a[k-j]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_hi >= k_lo:
                k_range = np.arange(k_lo, k_hi + 1)
                mask = k_range != 2 * j
                if mask.any():
                    c[k_range[mask]] += 2.0 * delta * a[k_range[mask] - j]
            
            # Check improvement
            c_max = float(np.max(c))
            sc = 2.0 * n * c_max / (S * S)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
                best_c = c.copy()
        
        if verbose and (p % 5 == 0 or p == n_passes - 1):
            print(f"  ECD pass {p}: score = {best_sc:.12f}")
    
    return best_a, best_sc

            if 0.0 < xs < x_max:
                candidates.append(xs)

    # Breakpoints between all pairs (vectorized)
    if K >= 2:
        i_idx, l_idx = np.triu_indices(K, k=1)
        A = q[i_idx] - q[l_idx]
        B = p[i_idx] - p[l_idx]
        C = c0[i_idx] - c0[l_idx]

        # Linear cases
        lin_mask = np.abs(A) < 1e-12
        if lin_mask.any():
            Bl = B[lin_mask]
            Cl = C[lin_mask]
            valid = np.abs(Bl) > 1e-12
            if valid.any():
                xps = -Cl[valid] / Bl[valid]
                xps = xps[(xps > 1e-12) & (xps < x_max - 1e-12)]
                if len(xps) > 0:
                    candidates.extend(xps.tolist())

        # Quadratic cases
        quad_mask = ~lin_mask
        if quad_mask.any():
            Aq = A[quad_mask]
            Bq = B[quad_mask]
            Cq = C[quad_mask]
            disc = Bq * Bq - 4.0 * Aq * Cq
            valid_q = disc >= 0
            if valid_q.any():
                sq_d = np.sqrt(disc[valid_q])
                r1 = (-Bq[valid_q] + sq_d) / (2.0 * Aq[valid_q])
                r2 = (-Bq[valid_q] - sq_d) / (2.0 * Aq[valid_q])
                for r in [r1, r2]:
                    r = r[(r > 1e-12) & (r < x_max - 1e-12)]
                    if len(r) > 0:
                        candidates.extend(r.tolist())

    candidates = np.asarray(candidates, dtype=np.float64)
    candidates = candidates[(candidates >= 0.0) & (candidates <= x_max)]

    if len(candidates) == 0:
        return 0.0

    # Deduplicate (sort and unique)
    candidates = np.sort(candidates)
    if len(candidates) > 1:
        keep = np.concatenate([[True], np.diff(candidates) > 1e-10])
        candidates = candidates[keep]

    # Evaluate objective at all candidates
    c_vals = c0[None, :] + p[None, :] * candidates[:, None] + q[None, :] * (candidates[:, None] ** 2)
    S_vals = S_0 + candidates
    obj = c_vals.max(axis=1) / (S_vals * S_vals)

    best_idx = int(np.argmin(obj))
    return float(candidates[best_idx])


def exact_coord_descent(a, n_passes=50, K=60, seed=42, deadline=None, verbose=True):
    """
    Exact coordinate descent: for each coordinate, solve the 1D subproblem exactly.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    c = fast_conv(a)

    best_a = a.copy()
    best_sc = 2.0 * n * float(np.max(c)) / (S * S)

    rng = np.random.default_rng(seed)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)

        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break

            a_j = a[j]
            if a_j < 1e-15 and S > 1e-12:
                # Very small: try setting to a small value
                # The best is likely x=0 or a small positive value
                # For x=0: no change. Skip if already ~0
                continue

            # Get active lags
            top_k = np.argsort(c)[::-1][:K]

            # Compute c0, p, q for active lags
            c0 = np.zeros(K)
            pv = np.zeros(K)
            qv = np.zeros(K)

            for idx in range(K):
                k_int = int(top_k[idx])
                if k_int == 2 * j:
                    c0[idx] = c[k_int] - a_j * a_j
                    pv[idx] = 0.0
                    qv[idx] = 1.0
                elif 0 <= k_int - j < n:
                    c0[idx] = c[k_int] - 2.0 * a_j * a[k_int - j]
                    pv[idx] = 2.0 * a[k_int - j]
                    qv[idx] = 0.0
                else:
                    c0[idx] = c[k_int]
                    pv[idx] = 0.0
                    qv[idx] = 0.0

            S_0 = S - a_j
            x_max = max(30.0 * a_j, 10.0 * S / n, 5.0)

            x_new = solve_1d_subproblem(c0, pv, qv, S_0, x_max)

            if abs(x_new - a_j) < 1e-13 * max(1.0, a_j):
                continue

            # Update a[j] and convolution
            delta = x_new - a_j
            a[j] = x_new
            S += delta

            # Update c[2j]
            if 2 * j < 2 * n - 1:
                c[2 * j] += x_new * x_new - a_j * a_j

            # Update c[k] for k where 0 <= k-j < n and k != 2j
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_hi >= k_lo:
                k_range = np.arange(k_lo, k_hi + 1)
                mask = k_range != 2 * j
                if mask.any():
                    c[k_range[mask]] += 2.0 * delta * a[k_range[mask] - j]

            # Check improvement
            c_max = float(np.max(c))
            sc = 2.0 * n * c_max / (S * S)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

        if verbose and (p % 5 == 0 or p == n_passes - 1):
            elapsed = time.monotonic() - (deadline - float(n_passes)) if False else 0
            print(f"  ECD pass {p+1}/{n_passes}: score = {best_sc:.12f}")

    # Final verification
    c_verify = fast_conv(best_a)
    S_verify = float(best_a.sum())
    best_sc = 2.0 * n * float(np.max(c_verify)) / (S_verify * S_verify)

    return best_a, best_sc


def spectral_flatten(a, n_iters=200, gamma=0.7, deadline=None):
    """
    Spectral flattening: iteratively modify the magnitude spectrum to reduce
    peak sidelobe ratio, then project back to non-negative.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current aperiodic autocorrelation spectrum
        # |A|^2 is the DTFT of the circular autocorrelation
        # We want to flatten the time-domain autocorrelation
        power = mag * mag

        # Peak reduction in time domain
        # Invert to get circular autocorrelation
        circ_corr = np.fft.irfft(power, M_fft)
        # Extract aperiodic part
        aperiodic = circ_corr[:2 * n - 1]

        # Modify power spectrum to reduce peaks
        # Replace peaks in time domain, then re-FFT
        peak_val = float(np.max(aperiodic))
        mean_val = float(np.mean(aperiodic))
        # Soft threshold: compress peaks
        aperiodic_mod = np.where(aperiodic > gamma * peak_val,
                                 gamma * peak_val + (1 - gamma) * aperiodic[aperiodic > gamma * peak_val],
                                 aperiodic)
        # Pad back to full length
        full_mod = np.zeros(M_fft)
        full_mod[:len(aperiodic_mod)] = aperiodic_mod
        # Make it real and symmetric (autocorrelation is symmetric)
        full_mod = full_mod * 0.5  # scale

        # Re-FFT to get modified power spectrum
        new_power = np.fft.rfft(full_mod)[:n + 1]
        new_power = np.real(new_power)
        new_power = np.maximum(new_power, 0)

        # New magnitude
        new_mag = np.sqrt(new_power)

        # IFFT to get new sequence
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Blend with current for stability
        a = 0.5 * a + 0.5 * a_new
        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a, best_sc


def perturb_sequence(a, rng, scale=0.05):
    """Generate a perturbed version of the sequence for ILS."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = int(rng.integers(0, 8))

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + 2.0 * rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Gaussian perturbation
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 3:
        # Block perturbation
        n_blocks = 5 + int(rng.integers(0, 15))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 4:
        # Tapered perturbation (more at edges)
        edge_weight = 1.0 + 3.0 * np.abs(x - 0.5) * 2.0
        a_p = a * (1.0 + scale * edge_weight * rng.standard_normal(n))
    elif strategy == 5:
        # Low-frequency perturbation
        n_pert = max(1, n // 20)
        pert = rng.standard_normal(n_pert)
        pert_full = np.interp(np.arange(n), np.arange(n_pert), pert)
        a_p = a * (1.0 + scale * 2.0 * pert_full)
    elif strategy == 6:
        # Random subset scaling
        n_sel = max(1, n // 3)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))
    else:
        # Mirror-like perturbation
        half = n // 2
        a_p = a.copy()
        scale_half = scale * (rng.random() - 0.3)
        a_p[:half] *= (1.0 + scale_half)
        a_p[-half:] *= (1.0 - scale_half)

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def multi_scale_ecl(a, n_passes, K, seed, deadline):
    """
    Multi-scale ECD: gradually increase the active set size K for better
    convergence in later stages.
    """
    result = a.copy()
    best_sc = compute_score(result)

    # Start with small K (fast), increase to large K (accurate)
    k_schedule = []
    total_passes = 0
    for k_val in [K // 3, K // 2, K, K]:
        passes = n_passes // 4
        if k_val == K and k_val == k_schedule[-1] if k_schedule else True:
            passes = n_passes - total_passes
        k_schedule.append(k_val)
        total_passes += passes

    for k_val, passes in zip(k_schedule, [n_passes // 4, n_passes // 4, n_passes // 4, n_passes // 4]):
        if deadline is not None and time.monotonic() > deadline:
            break
        if passes < 1:
            continue
        result, sc = exact_coord_descent(result, n_passes=passes, K=k_val,
                                         seed=seed, deadline=deadline, verbose=False)
        if sc < best_sc:
            best_sc = sc
        print(f"    ECD K={k_val}: score = {sc:.12f}")

    return result, best_sc


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses exact coordinate descent with iterated local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.12f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============ Phase 0: Initialize ============
    print(f"{'='*60}")
    print(f"Starting ECD optimizer (budget: {budget_s}s)")
    print(f"{'='*60}")

    init_candidates = []

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent length: {n_parent}, score: {compute_score(parent):.12f}")
        parent_norm = parent / parent.sum()
        init_candidates.append((parent_norm, "parent"))
        report(compute_score(parent_norm), parent_norm, "parent")
    else:
        n_parent = 8192
        init_candidates.append((np.ones(n_parent) * 0.5, "uniform"))

    # Add alternative initializations
    n_work = init_candidates[0][0].shape[0]
    x = np.arange(n_work, dtype=np.float64) / n_work

    init_candidates.append((np.maximum(0.1, 0.5 + 0.5 * np.cos(2 * np.pi * x)), "cosine"))
    init_candidates.append((0.3 + 0.7 * np.sin(np.pi * x) ** 2, "sin_sq"))
    init_candidates.append((np.exp(-3.0 * (x - 0.5) ** 2), "gaussian"))
    init_candidates.append((0.5 * (1.0 + np.cos(2.0 * np.pi * x)), "hann"))
    init_candidates.append((np.clip(0.5 + 0.3 * rng.standard_normal(n_work), 0.01, None), "noise"))

    # Try a different length
    n_alt = int(n_work * 1.3)
    x_alt = np.arange(n_alt, dtype=np.float64) / n_alt
    init_candidates.append((0.5 * (1.0 + np.cos(2.0 * np.pi * x_alt)), "hann_alt"))
    init_candidates.append((np.ones(n_alt) * 0.3, "uniform_alt"))

    # ============ Phase 1: Quick ECD on best initializations ============
    print(f"\n--- Phase 1: Quick ECD (short passes) ---")
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 30.0)

    for a_init, label in init_candidates:
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S < 1e-12:
            continue
        a_init = a_init / S

        sc = compute_score(a_init)
        report(sc, a_init, f"init_{label}")

        # Quick ECD with small K and few passes
        if time.monotonic() < phase1_deadline - 30:
            a_ecl, sc_ecl = exact_coord_descent(a_init, n_passes=5, K=30,
                                                seed=seed, deadline=phase1_deadline, verbose=False)
            report(sc_ecl, a_ecl, f"ECD_{label}")

    print(f"After Phase 1: best = {best_score:.12f}")

    # ============ Phase 2: Main ECD on the best solution ============
    print(f"\n--- Phase 2: Main ECD (long passes, increasing K) ---")
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 30.0)

    # Use the best solution from Phase 1
    main_a = best_a.copy()

    # Progressive ECD with increasing K
    k_phases = [30, 50, 80, 120, 150]
    remaining = phase2_deadline - time.monotonic()
    passes_per_phase = max(5, int(remaining / (len(k_phases) * 20.0)))

    for k_val in k_phases:
        if time.monotonic() > phase2_deadline - 10:
            break
        a_main, sc_main = exact_coord_descent(main_a, n_passes=passes_per_phase, K=k_val,
                                              seed=seed + k_val, deadline=phase2_deadline, verbose=True)
        report(sc_main, a_main, f"ECD_K{k_val}")
        main_a = a_main

    print(f"After Phase 2: best = {best_score:.12f}")

    # ============ Phase 3: Iterated Local Search ============
    print(f"\n--- Phase 3: Iterated Local Search ---")
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 30.0)

    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline - 60:
            break

        # Progressive perturbation (start small, grow larger)
        scale = 0.02 + 0.08 * (trial / max(1, n_trials)) * (1.0 + rng.random())
        a_pert = perturb_sequence(main_a, rng, scale=scale)

        # ECD refinement
        remaining_s = phase3_deadline - time.monotonic()
        n_passes = max(3, int(remaining_s / (30.0 * (n_trials - trial))))
        K_use = 60 + int(40 * rng.random())

        a_refined, sc_refined = exact_coord_descent(a_pert, n_passes=n_passes, K=K_use,
                                                    seed=seed + 1000 + trial * 7,
                                                    deadline=phase3_deadline, verbose=False)
        report(sc_refined, a_refined, f"ILS#{trial} K={K_use}")

        if sc_refined < best_score:
            main_a = a_refined.copy()

    print(f"After Phase 3: best = {best_score:.12f}")

    # ============ Phase 4: Try different lengths ============
    print(f"\n--- Phase 4: Length exploration ---")
    phase4_deadline = t_start + 0.93 * (float(budget_s) - 30.0)

    if time.monotonic() < phase4_deadline - 60:
        for n_try in [int(len(main_a) * 0.8), int(len(main_a) * 1.2)]:
            if time.monotonic() > phase4_deadline - 30:
                break
            a_resampled = resample_sequence(main_a, n_try)
            S = a_resampled.sum()
            if S < 1e-12:
                continue
            a_resampled = a_resampled / S
            sc = compute_score(a_resampled)
            report(sc, a_resampled, f"resample_n{n_try}")

            # Quick ECD on resampled
            if time.monotonic() < phase4_deadline - 30 and sc < best_score + 0.05:
                a_r, sc_r = exact_coord_descent(a_resampled, n_passes=3, K=50,
                                                seed=seed + 5000, deadline=phase4_deadline, verbose=False)
                report(sc_r, a_r, f"resample_n{n_try}_ECD")

    # ============ Phase 5: Final high-precision ECD ============
    print(f"\n--- Phase 5: Final Polish ---")
    final_deadline = deadline

    remaining_s = final_deadline - time.monotonic()
    if remaining_s > 60:
        n_final_passes = max(5, int(remaining_s / 40.0))
        a_final, sc_final = exact_coord_descent(best_a, n_passes=n_final_passes, K=200,
                                               seed=seed + 9999, deadline=final_deadline, verbose=True)
        report(sc_final, a_final, "final_polish")

    # One more pass with the very best K
    if time.monotonic() < final_deadline - 30:
        a_final2, sc_final2 = exact_coord_descent(best_a, n_passes=10, K=250,
                                                 seed=seed + 8888, deadline=final_deadline, verbose=False)
        report(sc_final2, a_final2, "final_K250")

    # ============ Final verification ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.12f}, verified={verified:.12f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
