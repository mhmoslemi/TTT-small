import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute gradient of sum_m w[m]*c[m] w.r.t. a, where c = a*a."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a[::-1], N)
    full = np.fft.irfft(W * AR, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=100.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def pnorm_score_grad(a, p):
    """Compute p-norm of n*c/S^2 and its gradient w.r.t. a (with normalization penalty)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)

    p = float(max(1.0, p))
    # Compute (sum d^p)^(1/p)
    dp = np.power(d, p)
    F = float(np.sum(dp))
    if F < 1e-30:
        return 0.0, np.zeros(n)
    score = F ** (1.0 / p)

    # Gradient: F^(1/p - 1) * sum_m d[m]^(p-1) * dd/da[i]
    # dd/da[i] = n * dc[m]/da[i] / S^2 = n * a[m-i] / S^2
    # So sum_m d[m]^(p-1) * dd[m]/da[i] = (n/S^2) * sum_m d[m]^(p-1) * a[m-i]
    # = (n/S^2) * corr_pair(a, d^(p-1))

    w = np.power(d, p - 1.0)
    h = corr_pair(a, w)
    coeff = (float(n) / (S * S)) * (F ** (1.0 / p - 1.0))
    grad = coeff * h

    # Normalization penalty: 0.5*(S-1)^2
    PEN = 0.01
    grad = grad + PEN * (S - 1.0)
    score = score + 0.5 * PEN * (S - 1.0) ** 2

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(score), grad


def lse_score_grad(a, beta):
    """LSE-smoothed max of n*c/S^2, with normalization penalty."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = (float(n) * np.maximum(c, 0.0)) / (S * S)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return 0.0, np.zeros(n)

    beta = float(max(10.0, beta))
    z = beta * (d - dmax)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    sum_e = float(np.sum(e))
    if sum_e < 1e-30:
        w = np.zeros_like(d)
        w[np.argmax(d)] = 1.0
        L = dmax
    else:
        w = e / sum_e
        L = dmax + np.log(sum_e) / beta

    h = corr_pair(a, w)
    coeff = float(n) / (S * S)
    grad = coeff * h
    PEN = 0.01
    grad = grad + PEN * (S - 1.0)
    score = L + 0.5 * PEN * (S - 1.0) ** 2

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(score), grad


def run_lbfgs(a0, score_fn, deadline, maxfun=500):
    """Run L-BFGS-B with a given score function."""
    a = a0.copy()
    n = len(a)
    if deadline is not None and time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(30, maxfun),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(score_fn, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        if res.x is not None:
            a = normalize_a(res.x)
    except Exception:
        pass
    sc = exact_score(a)
    if not np.isfinite(sc) or sc > 1e6:
        a = a0.copy()
        sc = exact_score(a)
    return float(sc), a


def nesterov_subgrad(a, max_iters=200, deadline=None):
    """Nesterov-accelerated multiplicative weights on the active set."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    y = a.copy()
    v = np.zeros(n)
    t_val = 1.0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline - 5.0:
            break

        c = fast_self_conv(y)
        d = (float(n) * np.maximum(c, 0.0)) / (float(np.sum(y)) ** 2)
        dmax = float(np.max(d))
        if dmax < 1e-15:
            break

        # Active set: top k positions
        k = min(5, len(d))
        idx = np.argpartition(d, len(d) - k)[len(d) - k:]
        w = np.zeros(len(d))
        w[idx] = 1.0
        w = w / k

        h = corr_pair(y, w)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            break

        # Step size
        step = 2.0 / (t_val + 1.0) ** 2
        # Multiplicative update in log space
        log_y = np.log(y + 1e-300)
        log_y_new = log_y - step * 0.5 * h / hmax
        y_new = np.exp(log_y_new - np.max(log_y_new))
        y_new = normalize_a(y_new)

        # Nesterov interpolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val ** 2)) / 2.0
        theta = (t_val - 1.0) / t_new
        v = y_new
        y = (1.0 + theta) * y_new - theta * y
        y = np.clip(y, 1e-15, None)
        y = normalize_a(y)
        t_val = t_new

        sc = exact_score(y)
        sc_v = exact_score(v)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        if sc_v < best_sc:
            best_sc = sc_v
            best_a = v.copy()

    return float(best_sc), best_a


def plain_subgrad_polish(a, max_iters=50, deadline=None):
    """Plain multiplicative subgradient with multiple step sizes."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline - 5.0:
            break
        cur_best = None
        cur_sc = best_sc

        c = fast_self_conv(a)
        d = (float(n) * np.maximum(c, 0.0)) / (float(np.sum(a)) ** 2)
        dmax = float(np.max(d))
        if dmax < 1e-15:
            break

        for k in (1, 2, 3, 5, 8, 12):
            idx = np.argpartition(d, len(d) - k)[len(d) - k:]
            w = np.zeros(len(d))
            w[idx] = 1.0
            w /= k

            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-15:
                continue

            for mult in (0.01, 0.03, 0.08, 0.15, 0.3, 0.5, 0.8):
                if deadline is not None and time.monotonic() > deadline - 2.0:
                    break
                q = a * np.exp(-mult * h / hmax)
                q = normalize_a(q)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def generate_inits(n, rng, parent=None, cap=25):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.1
        arr = np.clip(arr, 1e-6, None)
        inits.append(normalize_a(arr))

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    # V-shapes with varying sharpness
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    # Sharp V (power)
    for p in (2.0, 3.0, 4.0, 6.0):
        add(1.0 + 1.0 * np.power(np.abs(2.0 * x - 1.0), p))
    # Inverse-V (U-shape with divergence)
    for alpha in (0.1, 0.05, 0.02):
        add(1.0 / (np.abs(2.0 * x - 1.0) + alpha))

    # Cosine shapes
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x + np.pi / 2.0))

    # Linear / polynomial
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))
        # Symmetric exponential
        add(np.exp(-rate * np.abs(x - 0.5)))

    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2):
        for center in (0.2, 0.25, 0.3):
            g1 = np.exp(-(((x - center) / width) ** 2))
            g2 = np.exp(-(((x - (1.0 - center)) / width) ** 2))
            add(g1 + g2 + 0.1)

    # Sine-based
    for f in (1, 2, 3, 4, 5, 6):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.8 * np.abs(np.sin(np.pi * f * x)))
        add(np.power(np.abs(np.sin(np.pi * f * x)), 0.5))

    # Random smooth
    Ny = next_power_of_two(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Ny) * Ny
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Ny)[:n].real
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 1.0)) * noise)

    # Random trig sums
    for _ in range(5):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 7))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns
    for nb in (4, 8, 16, 32, 64):
        if nb > n // 2:
            continue
        block_vals = 0.05 + 0.95 * rng.random(nb)
        per = n // nb
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(3):
        add(0.05 + 0.95 * rng.random(n))

    return inits


def perturb_sequence(a, rng, strength=None):
    """Perturb a sequence with various strategies."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1].copy()
        q *= np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Active-set-based
        c = fast_self_conv(p)
        d = (n * np.maximum(c, 0.0)) / (np.sum(p) ** 2)
        k = int(rng.integers(1, 8))
        idx = np.argpartition(d, len(d) - k)[len(d) - k:]
        w = np.zeros(len(d))
        w[idx] = 1.0
        w /= k
        h = corr_pair(p, w)
        hmax = float(np.max(h))
        if hmax > 1e-12:
            q = p * np.exp(-strength * h / hmax)
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed Gaussian noise in log space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 200))
        if k % 2 == 0:
            k += 1
        if 3 <= k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - np.max(logp))
    elif mode == 5:
        # V-shape blend
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Sharpen the V
        x = np.linspace(0.0, 1.0, n)
        sharp = np.power(1.0 + strength * np.abs(2.0 * x - 1.0), 2.0)
        q = p * sharp
    elif mode == 7:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 8:
        # Random mask perturbation
        mask = (rng.random(n) < 0.3).astype(float)
        q = p * (1.0 + strength * mask * rng.standard_normal(n))
    else:
        # Power perturbation
        alpha = float(rng.uniform(0.5, 1.5))
        q = np.power(p / np.max(p + 1e-300), alpha)
        q = np.where(q > 1e-15, q, 1e-10)

    return normalize_a(np.clip(q, 0.0, None))


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def make_score_fn(beta):
    return lambda x: lse_score_grad(x, beta)


def make_pnorm_fn(p):
    return lambda x: pnorm_score_grad(x, p)


def optimize_full(a0, deadline, rng, label="opt"):
    """Full optimization pipeline: p-norm continuation + LSE + subgradient."""
    a = normalize_a(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Phase 1: p-norm continuation
    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
    # Time allocation: give more time to higher p
    total_time = max(10.0, deadline - t0 - 60.0)
    p_fracs = [0.06, 0.07, 0.08, 0.09, 0.1, 0.12, 0.15, 0.15]

    cum = 0.0
    for pi, p in enumerate(p_schedule):
        cum += p_fracs[pi]
        sub_dead = t0 + cum * total_time
        sub_dead = min(sub_dead, deadline - 60.0)
        if time.monotonic() > sub_dead - 3.0:
            break

        mf = max(50, int((sub_dead - time.monotonic()) / max(estimate_eval_time(n), 0.001) * 0.7))
        mf = min(mf, 2000)
        if mf < 50:
            break

        sc, a2 = run_lbfgs(a, make_pnorm_fn(p), sub_dead, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] p={}: score={:.8f} t={:.0f}s".format(label, p, best_sc, time.monotonic() - t0))
        a = normalize_a(a2)

    # Phase 2: LSE refinement
    lse_schedule = [(200, 0.08), (500, 0.1), (1000, 0.1), (3000, 0.12), (10000, 0.15)]
    for beta, frac in lse_schedule:
        sub_dead = t0 + (cum + frac) * total_time
        sub_dead = min(sub_dead, deadline - 60.0)
        cum += frac
        if time.monotonic() > sub_dead - 3.0:
            break

        mf = max(50, int((sub_dead - time.monotonic()) / max(estimate_eval_time(n), 0.001) * 0.7))
        mf = min(mf, 3000)
        if mf < 50:
            break

        sc, a2 = run_lbfgs(a, make_score_fn(beta), sub_dead, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] lse beta={}: score={:.8f} t={:.0f}s".format(label, beta, best_sc, time.monotonic() - t0))
        a = normalize_a(a2)

    # Phase 3: Subgradient polish
    if time.monotonic() < deadline - 30.0:
        sc, a_sg = nesterov_subgrad(a, max_iters=100, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print("[{}] nesterov: score={:.8f}".format(label, best_sc))

        sc, a_sg2 = plain_subgrad_polish(best_a, max_iters=30, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_sg2.copy()
            print("[{}] subgrad: score={:.8f}".format(label, best_sc))

    return float(best_sc), best_a


def quick_optimize(a, deadline):
    """Quick LSE + subgradient optimization."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for beta in (500, 2000, 10000, 30000):
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        mf = max(40, int(rem * 0.7 / max(estimate_eval_time(len(a)), 0.001)))
        mf = min(mf, 1500)
        if mf < 40:
            break
        sc, a2 = run_lbfgs(a, make_score_fn(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    if time.monotonic() < deadline - 5.0:
        sc, a_sg = plain_subgrad_polish(best_a, max_iters=15, deadline=deadline)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=100.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and np.sum(p) > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("found" if parent is not None else "none",
                                        parent.size if parent is not None else 0))

    # Determine lengths
    lengths = []
    if parent is not None:
        pl = int(parent.size)
        if 2048 <= pl <= 16384:
            lengths.append(pl)
        for L in [8192, 10240, 12288, 16384, 6144]:
            if L not in lengths and 2048 <= L <= 16384:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288, 6144, 16384]

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Evaluate diverse inits
    all_results = []  # (score, length, sequence)
    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=15 if n > 8000 else 20)
        for arr in inits:
            sc = exact_score(arr)
            if np.isfinite(sc) and sc < 2.5:
                all_results.append((sc, n, arr))

    if not all_results:
        n0 = lengths[0]
        all_results.append((2.0, n0, np.ones(n0) / n0))

    all_results.sort(key=lambda x: x[0])
    print("Best init: {:.8f} (len={})".format(all_results[0][0], all_results[0][1]))

    # Select top starts from different lengths
    starts = []
    used_lens = set()
    for sc, n, arr in all_results:
        if len(starts) >= 3:
            break
        if n not in used_lens:
            starts.append((sc, n, arr))
            used_lens.add(n)
    while len(starts) < 2:
        sc, n, arr = all_results[0]
        starts.append((sc, n, arr + 1e-10))

    best_score = min(s[0] for s in starts)
    best_seq = starts[0][2].copy()

    # Phase 2: Main optimization on top starts
    # Time allocation: 50% to best, 30% to second, 20% to third
    main_end = deadline - 200.0
    main_budget = max(60.0, main_end - t_start)
    fracs = [0.5, 0.3, 0.2][:len(starts)]

    cum = 0.0
    for rank, (sc0, n, arr) in enumerate(starts):
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)
        if time.monotonic() > sub_dead - 10.0:
            continue

        # For secondary starts, blend with current best
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != n:
                base = resample_sequence(base, n)
            t_blend = 0.6 if rank == 1 else 0.7
            arr = normalize_a((1.0 - t_blend) * arr + t_blend * normalize_a(base))

        sc, seq = optimize_full(arr, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(rank + 1, best_score))

    # Phase 3: Subgradient polish on best
    if time.monotonic() < deadline - 100.0:
        sc, seq = nesterov_subgrad(best_seq, max_iters=80, deadline=deadline - 50.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Nesterov polish: {:.8f}".format(best_score))

        sc, seq = plain_subgrad_polish(best_seq, max_iters=20, deadline=deadline - 50.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 100.0 and perturb_done < 8:
        left = deadline - 100.0 - time.monotonic()
        if left < 50.0:
            break
        n_remaining = 8 - perturb_done
        chunk_end = time.monotonic() + left / float(max(1, n_remaining))
        p = perturb_sequence(best_seq, rng)
        sc, seq = quick_optimize(p, min(chunk_end, deadline - 50.0))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb {:d}: {:.8f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 5: Final high-beta LSE
    if time.monotonic() < deadline - 30.0:
        rem = deadline - time.monotonic() - 5.0
        if rem > 15.0:
            mf = max(40, int(rem * 0.7 / max(estimate_eval_time(len(best_seq)), 0.001)))
            sc, seq = run_lbfgs(best_seq, make_score_fn(50000.0), deadline - 5.0, min(mf, 2000))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High-beta final: {:.8f}".format(best_score))

    # Phase 6: Final subgradient
    if time.monotonic() < deadline - 5.0:
        sc, seq = plain_subgrad_polish(best_seq, max_iters=10, deadline=deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
