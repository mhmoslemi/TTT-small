"""
Adaptive Peak-Chasing + LSE-Nesterov + Fienup Hybrid Optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations over previous approach:
1. Novel "peak-chasing" projected gradient with Nesterov momentum and adaptive peak count
2. Frank-Wolfe (conditional gradient) hybrid for larger steps
3. Multi-phase with aggressive parameter sweeps
4. Iterated local search with diverse perturbations
5. FFT-based gradient computation exploiting convolution structure

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_sum

    # Gradient via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def peak_gradient_direct(a, peak_indices, weights):
    """Compute gradient as weighted sum of peak gradients. O(n * n_peaks)."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        idx = k - np.arange(j_start, j_end + 1)
        grad[j_start:j_end + 1] += 2.0 * w * a[idx]
    return grad

    return None  # Fallback to direct

        if it % 50 == 0 or it == 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                sc_current = sc

        # Early stop if no improvement for a long time
        if it % 5000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final

    return best_a


def peak_chase_fw(a, max_time, n_peaks=5):
    """
    Peak-chasing with Frank-Wolfe (conditional gradient) steps.
    Combines gradient info with extreme point moves.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    v = np.zeros(n)
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = min(1.0, (time.monotonic() - t_start) / max_time)

        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[top_idx]
        total_val = peak_vals.sum()
        if total_val < 1e-30:
            break
        weights = peak_vals / total_val

        grad = peak_gradient_direct(a, top_idx, weights)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Frank-Wolfe: find the minimizer of <grad, x> over the simplex
        j_min = int(np.argmin(grad))

        # Step size: use exact line search (simplified)
        # Try a few values of lambda (mixing parameter)
        best_lambda = 0
        best_trial_sc = compute_score(a)
        for lam in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - lam) * a + lam * np.zeros(n)
            a_trial[j_min] = lam
            # Mix with gradient step direction
            a_trial2 = (1 - lam) * a
            a_trial2 -= lam * 0.1 * grad / (gnorm + 1e-30)
            a_trial2 = np.maximum(a_trial2, 0.0)
            S2 = a_trial2.sum()
            if S2 > 1e-12:
                a_trial2 = a_trial2 / S2
                sc = compute_score(a_trial2)
                if sc < best_trial_sc:
                    best_trial_sc = sc
                    best_lambda = lam
                    a_trial2_save = a_trial2.copy()

        if best_lambda > 0:
            a = a_trial2_save
        else:
            # Fallback: projected gradient
            alpha = c[top_idx[0]] / (gnorm + 1e-30)
            alpha = min(alpha, 0.3)
            v = mu * v - alpha * grad
            a = a + v
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def fienup_spectral_eq(a, n_iters=300, method='power'):
    """
    Fienup with spectral equalization.
    method='power': P_new = P^alpha (compresses dynamic range)
    method='threshold': standard clipping
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    mu = 0.3

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        if method == 'power':
            # Compress dynamic range: P^alpha with alpha decreasing from 0.5 to 0.8
            alpha = 0.5 + 0.3 * t
            P_mean = P.mean()
            P_norm = P / (P_mean + 1e-30)
            P_new = P_mean * np.power(P_norm, alpha)
        else:
            P_mean = float(P.mean())
            thresh = 1.5 + 0.5 * t
            P_new = P.copy()
            above = P_new > P_mean * thresh
            if above.any():
                P_new[above] = P_mean * thresh + (P_new[above] - P_mean * thresh) * 0.3

        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Momentum blend
        a = (1 - mu) * a + mu * a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def nesterov_lse_adaptive(a, max_time, beta_start=10.0, beta_end=10000.0):
    """Nesterov LSE with time-adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    beta_growth = 1.003
    lr = 0.01
    lr_decay = 0.9995
    mu = 0.88

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.88 + 0.00003 * it)
        lr *= lr_decay

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        beta = min(beta * beta_growth, beta_end)

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def diverse_init(n, rng, method='mixed'):
    """Create diverse initializations."""
    if method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif method == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif method == 'chirp':
        x = np.arange(n, dtype=np.float64) / n
        a = np.abs(np.sin(2 * np.pi * x * x * 3.0)) + 0.05
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8.0
        a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif method == 'random_smooth':
        a = np.abs(rng.standard_normal(n)) + 0.1
        # Smooth it
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n)
        S = n
    return a / S


def perturb_sequence(a, rng, strength=0.02):
    """Perturb a sequence while maintaining non-negativity."""
    n = len(a)
    perturbation = strength * rng.standard_normal(n)
    a_new = a * (1.0 + perturbation)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S


def swap_perturb(a, rng, n_swaps=50):
    """Perturb by swapping random pairs of elements."""
    a_new = a.copy()
    n = len(a)
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, size=2)
        if i != j:
            a_new[i], a_new[j] = a_new[j], a_new[i]
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses peak-chasing + LSE-Nesterov + Fienup hybrid approach.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 6962

    best_score = np.inf
    best_a = np.ones(n_target) * 0.5

    def update_best(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a_cand)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Initialization + Quick Fienup (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 20.0)
    print(f"=== Phase 1: Init + Fienup (budget {phase1_end - t_start:.0f}s) ===")

    candidates = []

    if parent is not None:
        a_parent = parent.copy()
        S = a_parent.sum()
        a_parent = a_parent / S
        candidates.append(('parent', a_parent))
        sc_p = compute_score(a_parent)
        print(f"  Parent score: {sc_p:.10f}")
        update_best(sc_p, a_parent, "parent")

    # Try diverse initializations at target scale
    if n_target <= 10000:
        for method in ['gaussian', 'cosine', 'chirp', 'bimodal', 'random_smooth']:
            a_init = diverse_init(n_target, rng, method)
            candidates.append((f"init-{method}", a_init))
            sc = compute_score(a_init)
            update_best(sc, a_init, f"init-{method}")

    # Quick Fienup on best candidate
    if time.monotonic() < phase1_end - 5:
        a_f = fienup_spectral_eq(best_a, n_iters=200, method='power')
        update_best(compute_score(a_f), a_f, "fienup-power")

        a_f2 = fienup_spectral_eq(best_a, n_iters=200, method='threshold')
        update_best(compute_score(a_f2), a_f2, "fienup-thresh")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Peak-chasing with Nesterov (30% time)
    # ============================================================
    phase2_end = t_start + 0.40 * (total_budget - 20.0)
    print(f"\n=== Phase 2: Peak-chasing Nesterov (budget {phase2_end - t_start:.0f}s) ===")

    time_left_p2 = phase2_end - time.monotonic()

    # Run peak-chasing with different peak counts
    pc_configs = [
        {'start_n_peaks': 1, 'end_n_peaks': 5, 'frac': 0.4},
        {'start_n_peaks': 3, 'end_n_peaks': 10, 'frac': 0.3},
        {'start_n_peaks': 5, 'end_n_peaks': 20, 'frac': 0.3},
    ]

    for cfg in pc_configs:
        if time.monotonic() > phase2_end:
            break
        alloc = time_left_p2 * cfg['frac']
        if alloc < 30:
            break
        print(f"  Peak-chase: peaks [{cfg['start_n_peaks']}->{cfg['end_n_peaks']}], time={alloc:.0f}s")
        a_pc = peak_chase_nesterov(best_a, max_time=alloc,
                                    start_n_peaks=cfg['start_n_peaks'],
                                    end_n_peaks=cfg['end_n_peaks'])
        update_best(compute_score(a_pc), a_pc, f"peakchase-{cfg['start_n_peaks']}-{cfg['end_n_peaks']}")

    # Also run from perturbed versions
    if time.monotonic() < phase2_end - 60:
        a_pert = perturb_sequence(best_a, rng, strength=0.01)
        alloc = phase2_end - time.monotonic() - 10
        if alloc > 30:
            a_pc2 = peak_chase_nesterov(a_pert, max_time=alloc * 0.8,
                                         start_n_peaks=1, end_n_peaks=10)
            update_best(compute_score(a_pc2), a_pc2, "peakchase-pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: LSE Nesterov with adaptive beta (25% time)
    # ============================================================
    phase3_end = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n=== Phase 3: LSE Nesterov (budget {phase3_end - t_start:.0f}s) ===")

    nesterov_configs = [
        {'beta_start': 5.0, 'beta_end': 8000.0},
        {'beta_start': 20.0, 'beta_end': 15000.0},
        {'beta_start': 50.0, 'beta_end': 20000.0},
        {'beta_start': 100.0, 'beta_end': 30000.0},
    ]

    for cfg in nesterov_configs:
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 30:
            break
        alloc = min(time_left * 0.7, 300)
        print(f"  Nesterov LSE: beta [{cfg['beta_start']}, {cfg['beta_end']}], time={alloc:.0f}s")
        a_nl = nesterov_lse_adaptive(best_a, max_time=alloc,
                                      beta_start=cfg['beta_start'],
                                      beta_end=cfg['beta_end'])
        update_best(compute_score(a_nl), a_nl, f"nesterov-{int(cfg['beta_start'])}")

    # Also try from a perturbed start
    if time.monotonic() < phase3_end - 30:
        a_pert2 = perturb_sequence(best_a, rng, strength=0.005)
        time_left = phase3_end - time.monotonic()
        if time_left > 30:
            a_nl2 = nesterov_lse_adaptive(a_pert2, max_time=time_left * 0.8,
                                           beta_start=30.0, beta_end=20000.0)
            update_best(compute_score(a_nl2), a_nl2, "nesterov-pert")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: L-BFGS polish (20% time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n=== Phase 4: L-BFGS polish (budget {phase4_end - t_start:.0f}s) ===")

    lbfgs_betas = [100, 300, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase4_end:
            break
        time_left = phase4_end - time.monotonic()
        if time_left < 30:
            break
        print(f"  L-BFGS: beta={beta}, time_left={time_left:.0f}s")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, f"lbfgs-{beta}")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Iterated Local Search (10% time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Iterated Local Search (budget {phase5_end - t_start:.0f}s) ===")

    ils_count = 0
    while time.monotonic() < phase5_end - 30:
        ils_count += 1
        # Perturb
        strength = 0.01 / (1 + ils_count * 0.1)
        a_pert = perturb_sequence(best_a, rng, strength=strength)

        # Quick optimization: peak chase (short) + LSE (short)
        time_left = phase5_end - time.monotonic() - 10
        if time_left < 20:
            break

        # Short peak chase
        t_pc = min(60, time_left * 0.4)
        if t_pc > 10:
            a_tmp = peak_chase_nesterov(a_pert, max_time=t_pc,
                                         start_n_peaks=1, end_n_peaks=5)
        else:
            a_tmp = a_pert

        # Short LSE
        t_lse = min(60, time_left * 0.4)
        if t_lse > 10:
            a_tmp = nesterov_lse_adaptive(a_tmp, max_time=t_lse,
                                           beta_start=100.0, beta_end=20000.0)

        update_best(compute_score(a_tmp), a_tmp, f"ils-{ils_count}")

    print(f"After Phase 5: best = {best_score:.10f}")

    # ============================================================
    # Final: one more L-BFGS with highest beta
    # ============================================================
    if time.monotonic() < deadline - 5:
        print("\n=== Final L-BFGS ===")
        sc, a_opt = run_lbfgs(best_a, 30000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "final-lbfgs")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
