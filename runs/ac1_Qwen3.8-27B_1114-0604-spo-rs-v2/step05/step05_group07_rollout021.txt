import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -200, 200)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    p = exp_shifted / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # dL/da[i] = 2 * sum_k p[k] * a[k-i]
    N = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(P) * A, N)[:n]
    dL_da = 2.0 * cross
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L * a / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_grad(x):
        return lse_obj_grad(x, beta)
    
    res = minimize(
        obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out > 1e-12:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        return a

    S = a.sum()
    if S < 1e-12:
        return a

        k_star = int(np.argmax(c))
        # The peak is at lag k_star. The contributing elements are a[j] and a[k_star-j]
        # To reduce the peak, we should reduce the largest contributors
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Compute contributions
        idx = np.arange(j_start, j_end + 1)
        contrib = a[idx] * a[k_star - idx]
        
        # Find the largest contributor
        max_local = np.argmax(contrib)
        gi = idx[max_local]
        
        # Try reducing this element
        a_old = a[gi]
        for frac in [0.7, 0.5, 0.3, 0.1]:

            min_local = np.argmin(contrib)
            a[idx[min_local]] += a_old * (1 - frac)

            if S > 1e-12:
                a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                break
        else:
            a[gi] = a_old  # Revert
