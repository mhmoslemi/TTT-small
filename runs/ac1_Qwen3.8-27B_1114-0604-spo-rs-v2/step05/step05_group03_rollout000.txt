def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def solve_lp_cvxpy(a, M, solver_list=None, timeout=45):
    """Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using cvxpy."""
    n = len(a)
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    
    x = cp.Variable(n, nonneg=True)
    
    # Build convolution matrix C (2n-1) x n
    # C[k, i] = a[i] if i <= k <= i+n-1
    C = lil_matrix((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    C = csr_matrix(C)
    
    constraints = [C @ x <= M]
    objective = cp.Maximize(cp.sum(x))
    prob = cp.Problem(objective, constraints)
    
    t0 = time.monotonic()
    for solver in solver_list:
        if time.monotonic() - t0 > timeout:
            break
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.maximum(np.array(x.value).flatten(), 0.0)
        except Exception:
            continue
    
    return None


def solve_lp_fista(a, M, n_iters=15000, beta_start=10.0, beta_end=2000.0,
                   x_init=None, ramp_frac=0.3):
    """Solve LP subproblem using FISTA on LSE-smoothed surrogate."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_iters):
        if it < int(n_iters * ramp_frac):
            frac = it / max(1, int(n_iters * ramp_frac))
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=50, fista_iters=15000, time_limit=None,
                   use_cvxpy=False, cvxpy_timeout=30, multiplicative=False):
    """LP-based alternating method with adaptive mixing and optional multiplicative update."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None
    rng = np.random.default_rng(123)

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        
        if use_cvxpy:
            x = solve_lp_cvxpy(a, M, timeout=cvxpy_timeout)
            if x is None:
                x = solve_lp_fista(a, M, n_fista_iters=fista_iters,
                                   x_init=x_warm, beta_end=1500.0)
        else:
            x = solve_lp_fista(a, M, n_iters=fista_iters,
                               x_init=x_warm, beta_end=1500.0)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Try multiple update strategies
        candidates = []
        
        # Linear mixing
        for t in [0.2, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St > 1e-12:
                a_try = a_try / St
                candidates.append(a_try)
        
        # Multiplicative update
        if multiplicative:
            a_mult = a * x
            Sm = float(a_mult.sum())
            if Sm > 1e-12:
                candidates.append(a_mult / Sm)
        
        # Geometric mean
        a_geo = np.sqrt(np.maximum(a * x, 0.0))
        Sg = float(a_geo.sum())
        if Sg > 1e-12:
            candidates.append(a_geo / Sg)

        best_try = best_a
        best_try_sc = best_sc
        for a_try in candidates:
            sc_try = compute_score(a_try)
            if sc_try < best_try_sc:
                best_try_sc = sc_try
                best_try = a_try

        if best_try_sc < best_sc:
            best_sc = best_try_sc
            best_a = best_try.copy()
            a = best_try.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Try perturbation
            a_pert = a * (1.0 + 0.03 * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            Sp = float(a_pert.sum())
            if Sp > 1e-12:
                a_pert = a_pert / Sp
                sc_pert = compute_score(a_pert)
                if sc_pert < best_sc:
                    best_sc = sc_pert
                    best_a = a_pert.copy()
                    a = a_pert
                    no_improve_count = 0
                elif no_improve_count >= 10:
                    a = a_pert
                    no_improve_count = 0

    return best_a, best_sc, x_warm


def lse_lbfgs(a0, beta, maxiter=8000, maxfun=300000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def multi_peak_reduction(a, n_steps=300, top_k=5, rng=None):
    """Targeted reduction of the top-k convolution peaks simultaneously."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        S2 = S * S
        S3 = S2 * S

        # Get top-k peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_vals = c[peak_indices]
        max_val = float(c.max())

        # Composite subgradient direction
        direction = np.zeros(n)
        for ki in peak_indices:
            w = peak_vals[int(np.where(peak_indices == ki)[0][0])] / max_val
            j_start = max(0, int(ki) - n + 1)
            j_end = min(int(ki), n - 1)
            if j_end >= j_start:
                direction[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        grad = 2.0 * n * (direction / S2 - 2.0 * max_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Backtracking line search
        base_step = min(0.05, max(0.001, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3))
        improved = False
        for mult in [0.2, 0.4, 0.6, 1.0, 1.5, 2.0, 3.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 10 == 0:
            # Randomized perturbation to escape local minimum
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    return inits


def propose_candidate(budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing the evaluation function.
    Uses LP-alternating (cvxpy + FISTA), LSE-L-BFGS polishing, and
    multi-peak targeted reduction.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 45.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Test cvxpy LP speed on a small example
    use_cvxpy = False
    try:
        small_a = np.ones(200) / 200
        t0 = time.monotonic()
        sol = solve_lp_cvxpy(small_a, float(self_conv(small_a).max()), timeout=10)
        lp_time = time.monotonic() - t0
        if sol is not None:
            use_cvxpy = True
            print(f"Cvxpy LP test: time={lp_time:.2f}s for n=200")
        else:
            print("Cvxpy LP test: no solution found")
    except Exception as e:
        print(f"Cvxpy LP test failed: {e}")
        use_cvxpy = False

    # ============================================================
    # Phase 1: Quick exploration (8% time)
    # ============================================================
    phase1_dl = t_start + 0.08 * (float(budget_s) - 45.0)
    print(f"\n--- Phase 1: Quick Exploration ---")

    n_main = len(parent) if parent is not None else 6962
    lengths_to_try = [n_main]
    # Try a couple of nearby lengths
    for dl in [int(n_main * 0.9), int(n_main * 1.1)]:
        if dl not in lengths_to_try and dl >= 500:
            lengths_to_try.append(dl)

    for n_try in lengths_to_try[:3]:
        if time.monotonic() > phase1_dl:
            break
        remaining = phase1_dl - time.monotonic()
        if remaining < 15:
            break
        print(f"  n={n_try}, remaining={remaining:.0f}s")
        inits = build_initializations(n_try, rng, parent)
        for i, a0 in enumerate(inits[:3]):
            if time.monotonic() > phase1_dl:
                break
            a_er = error_reduction(a0, n_iters=50, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n_try} init={i}")

    # Use the best found so far as main working sequence
    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S
    print(f"Main length: {n_main}, starting score: {best_score:.8f}")

    # ============================================================
    # Phase 2: LP-Alternating (45% time)
    # ============================================================
    phase2_dl = t_start + 0.53 * (float(budget_s) - 45.0)
    print(f"\n--- Phase 2: LP-Alternating ---")

    x_warm = None
    # Progressive FISTA budgets
    fista_budgets = [6000, 10000, 15000, 20000]
    
    for fi in range(len(fista_budgets)):
        if time.monotonic() > phase2_dl - 20:
            break
        remaining = phase2_dl - time.monotonic()
        if remaining < 30:
            break
        
        # Estimate time per LP solve and determine max outer iterations
        est_per_solve = fista_budgets[fi] * n_main * 5e-7
        if use_cvxpy:
            est_per_solve = max(2.0, est_per_solve / 3.0)
        max_outer = max(3, int(remaining / max(1.0, est_per_solve * 1.5)))
        max_outer = min(max_outer, 60)
        
        print(f"  LP round {fi}: fista={fista_budgets[fi]}, "
              f"max_outer={max_outer}, cvxpy={use_cvxpy}, remaining={remaining:.0f}s")
        
        a_out, sc_out, x_warm = lp_alternating(
            a_work,
            max_outer_iters=max_outer,
            fista_iters=fista_budgets[fi],
            time_limit=phase2_dl - 10,
            use_cvxpy=use_cvxpy and fi < 2,
            cvxpy_timeout=25,
            multiplicative=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()
        S = float(a_work.sum())
        if S > 1e-12:
            a_work = a_work / S

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: LSE-L-BFGS Polishing (25% time)
    # ============================================================
    phase3_dl = t_start + 0.78 * (float(budget_s) - 45.0)
    print(f"\n--- Phase 3: LSE-L-BFGS ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [100, 300, 800, 2000, 5000, 10000, 20000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_dl - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = lse_lbfgs(a_norm, beta, maxiter=6000, maxfun=200000)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Multi-peak targeting (12% time)
    # ============================================================
    phase4_dl = t_start + 0.90 * (float(budget_s) - 45.0)
    print(f"\n--- Phase 4: Multi-Peak Targeting ---")

    # Try different top_k values
    for top_k in [3, 5, 8]:
        if time.monotonic() > phase4_dl - 10:
            break
        n_steps = int(max(50, (phase4_dl - 15 - time.monotonic()) / 1.0))
        a_pk = multi_peak_reduction(best_a, n_steps=min(n_steps, 400),
                                    top_k=top_k, rng=rng)
        sc = compute_score(a_pk)
        report(sc, a_pk, f"MultiPeak k={top_k}")

    # ============================================================
    # Phase 5: Final polish (remaining time)
    # ============================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Final LP round with high FISTA
    if time.monotonic() < deadline - 60:
        print("  Final LP...")
        a_fin_in = best_a.copy()
        S = float(a_fin_in.sum())
        if S > 1e-12:
            a_fin_in = a_fin_in / S
        a_fin, sc_fin, _ = lp_alternating(
            a_fin_in, max_outer_iters=20, fista_iters=25000,
            time_limit=deadline - 40, use_cvxpy=use_cvxpy,
            cvxpy_timeout=20, multiplicative=True
        )
        sc = compute_score(a_fin)
        report(sc, a_fin, "final LP")

    # Final very high beta L-BFGS
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt = lse_lbfgs(a_norm, 50000, maxiter=8000, maxfun=400000)
        report(sc, a_opt, "final LBFGS b=50000")

    # Final peak reduction
    if time.monotonic() < deadline - 10:
        a_pk = multi_peak_reduction(best_a, n_steps=200, top_k=3, rng=rng)
        sc = compute_score(a_pk)
        report(sc, a_pk, "final peak")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
