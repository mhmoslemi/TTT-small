"""
Peak-balancing optimizer combining:
1. High-precision L-BFGS with adaptive LSE smoothing (beta up to 1M)
2. Exact coordinate descent with 1D linesearch on peak contributors
3. Multi-peak weighted subgradient with adaptive step
4. Periodic L-BFGS refinement between CD phases
5. Diverse multi-start with creative perturbations

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coordinate_descent_phase(f, n_moves=2000, top_k_contrib=3, time_limit=None, t_start=None):

    rng = np.random.default_rng()
    n = len(f)

    S = f.sum()

def multi_peak_subgrad(f, n_iters=5000, top_k=5, f_lb=1.0, time_limit=None, t_start=None):
    """Multi-peak weighted subgradient descent."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S
    best_f = f.copy()
    best_sc = compute_score(f)
    step0 = 0.15
    decay = 0.9995

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        c = fast_conv(f)
        S_a = float(f.sum())
        c_max = float(c.max())

        # Weight top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        # Exponential weights: heavier weight on higher peaks
        peak_vals = c[top_idx] - c[top_idx[-1]] + 1e-10
        weights = np.exp(peak_vals / (c_max / (top_k + 1) + 1e-30))
        weights /= weights.sum()

        sc = 2.0 * n * c_max / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()

        # Composite subgradient
        sg = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * f[idx - ii]

        S2 = S_a * S_a
        S3 = S2 * S_a
        Mak_weighted = float(np.sum(weights * c[top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (decay ** it)
        step = min(step, 2.0 * max(0, sc - f_lb) / (gnorm + 1e-30))
        step = max(step, 1e-6)

        f_new = f - step * grad
        f = np.clip(f_new, 0.0, 1000.0)
        Sa = f.sum()
        if Sa < 1e-12:
            f = best_f.copy()
            break
        f = f / Sa

    return best_f

def block_coordinate_descent(f, block_size=50, n_sweeps=5, time_limit=None, t_start=None):
    """Optimize blocks of coordinates at a time using L-BFGS on sub-problems."""
    n = len(f)
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_sc = compute_score(f)

    for sweep in range(n_sweeps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block_idx = np.arange(start, end)
            block = f[block_idx].copy()
            S_block = block.sum()
            S_rest = f.sum() - S_block

            if S_block < 1e-12:
                continue

            # Define objective for this block
            def block_obj(x_block):
                fb = f.copy()
                fb[block_idx] = np.maximum(x_block, 0)
                return compute_score(fb)

            # Use L-BFGS on the block
            n_block = len(block_idx)
            bounds = [(0.0, 1.0)] * n_block
            try:
                res = minimize(block_obj, block, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'maxfun': 2000,
                                        'ftol': 1e-15, 'gtol': 1e-10})
                new_block = np.clip(res.x, 0, 1.0)

                sc = compute_score(f)

    if time.monotonic() < deadline - 10:
        a_cd = coordinate_descent_phase(best_a, n_moves=1000, top_k_contrib=5,
                                        time_limit=20.0, t_start=time.monotonic())
        sc = compute_score(a_cd)
        report(sc, a_cd, "final_CD")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
