Actually, let me reconsider the whole approach. The key insight is:

score(a) = 2n * max(conv(a,a)) / (sum(a))^2

This is scale-invariant: score(t*a) = score(a) for any t > 0. So we can normalize without loss of generality. Let's set sum(a) = 1 and minimize max(conv(a,a)).

With sum(a) = 1, we minimize max(conv(a,a)). The LSE approximation is:
F(a; beta) = (1/beta) * log(sum_k exp(beta * c_k))

where c = conv(a,a).

The gradient is:
dF/da_j = sum_k (exp(beta*c_k) / sum_l exp(beta*c_l)) * d(c_k)/da_j
= sum_k p_k * 2 * a_{k-j} (where valid)
= 2 * cross_corr(p, a)_j

This is much simpler! No S^2 terms.

But we need to maintain sum(a) = 1. We can either:
(a) Project after each step (clip to >= 0, then normalize)
(b) Use a Lagrangian approach

Option (a) is simpler and works well in practice. The normalization (dividing by sum) is a mild perturbation that doesn't hurt much.

So the algorithm is:
1. Normalize a to have sum = 1
2. Compute LSE and gradient
3. Take a gradient step: a_new = a - step * grad
4. Project: a_new = max(a_new, 0), then normalize to sum = 1
5. Repeat

- Maintain a "momentum" array m
- a_new = a - step * (m + grad)  [approximately]

Actually, standard Nesterov is:

"""
Nesterov-LSE with Multi-peak Subgradient and Strategic Perturbation.
Improves on previous approach by:
1. Nesterov momentum for accelerated convergence
2. Geometric beta annealing for LSE
3. Multi-peak subgradient targeting top-K peaks simultaneously
4. Strategic random perturbation to escape local minima
5. Working in normalized (sum=1) space for cleaner gradients
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_grad_normalized(a, beta):
    """
    LSE of conv(a,a) (assuming sum(a)=1) and its gradient.
    F = (1/beta) * log(sum_k exp(beta * c_k))
    grad_j = 2 * sum_k p_k * a_{k-j} (valid indices only)
    """
    n = len(a)
    c = fast_conv(a)
    
    # Softmax weights
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    p = exp_scaled / exp_sum
    
    LSE = c_max + float(np.log(exp_sum)) / beta
    
    # Gradient: 2 * cross_corr(p, a)
    # cross_corr_j = sum_{l=0}^{n-1} a_l * p_{j+l}
    # Compute via FFT: cross_corr = conv(a, p_rev)[n-1:2n-1][::-1]
    n2 = 2 * n - 1  # length of c and p
    conv_len = n + n2 - 1  # = 3n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    P_rev = np.fft.rfft(p[::-1], M_fft)
    cross_full = np.fft.irfft(A * P_rev, M_fft)[:conv_len]
    cross_corr = cross_full[n-1:2*n-1][::-1]
    
    grad = 2.0 * cross_corr
    return LSE, grad


def nesterov_lse(a0, betas, steps_per_beta, momentum=0.9, step_scale=0.5):
    """
    Nesterov-accelerated gradient descent on LSE with geometric beta annealing.
    a0: initial sequence (will be normalized)
    betas: array of beta values (increasing)
    steps_per_beta: number of gradient steps for each beta
    """
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)  # momentum
    
    for beta in betas:
        for step in range(steps_per_beta):
            val, grad = lse_grad_normalized(a, beta)
            
            # Adaptive step size
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            # Step size: scale by 1/gradient_norm, capped
            ls = min(step_scale / gnorm, 0.1)
            
            # Nesterov step
            v = momentum * v + ls * grad
            a_new = a - v
            a_new = normalize(a_new)
            
            # Check improvement
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            
            a = a_new
        
        # Periodic check
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def multipic_peak_subgrad(a, n_iters=5000, top_k=10, step_init=0.02, decay=0.9999):
    """
    Multi-peak subgradient method for normalized sequences.
    Targets top-K peaks simultaneously.
    """
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    n2 = 2 * n - 1
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max  # since sum(a)=1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-K peaks
        if top_k >= n2:
            top_idx = np.arange(n2)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Multi-peak subgradient: g_j = (2/K) * sum_i a_{k_i - j}
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                g[j_start:j_end + 1] += 2.0 * seg[::-1]
        g /= top_k
        
        # Project gradient onto tangent space (remove mean to preserve sum)
        g = g - g.mean()
        
        # Step
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            break
        ls = min(step, 0.05)
        a_new = a - ls * g
        a_new = normalize(a_new)
        a = a_new
        step *= decay
    
    return best_a, best_sc


def single_peak_subgrad(a, n_iters=10000, step_init=0.01, decay=0.9999):
    """Single-peak subgradient (targets only the max peak)."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        
        g = np.zeros(n)
        seg = a[j_start:j_end + 1]
        g[j_start:j_end + 1] = 2.0 * seg[::-1]
        g = g - g.mean()  # tangent space
        
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            break
        
        ls = min(step, 0.03)
        a_new = a - ls * g
        a_new = normalize(a_new)
        a = a_new
        step *= decay
    
    return best_a, best_sc


def gs_flat_target(a, n_iters=300, alpha=0.5):
    """
    GS-Fienup: project in frequency domain to approximate flat convolution.
    a: sequence (will be normalized)
    """
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()  # FFT size
    
    # Target: flat convolution of total sum 1
    # If a is normalized (sum=1), ideal conv is 1/(2n-1) at each position
    b_target = np.zeros(M)
    b_target[:2*n-1] = 1.0 / (2*n-1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        # GS update: |A_new| = |A|^(1-alpha) * |T|^alpha, phase mixed
        # We want A^2 ≈ T, so |A| ≈ |T|^(1/2)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def random_perturb_and_reopt(a, n_perturbations=20, rng=None):
    """
    Try random perturbations and re-optimize.
    Returns the best found.

"""
Nesterov-LSE with Multi-peak Subgradient and Strategic Perturbation.
Combines:
1. Nesterov-accelerated gradient descent on LSE-smoothed max-convolution
2. Geometric beta annealing for progressive sharpening
3. Multi-peak subgradient targeting top-K peaks simultaneously
4. GS-Fienup frequency-domain projection
5. Random perturbation + re-optimization for escaping local minima
All work in normalized (sum=1) space for cleaner gradients.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_grad_normalized(a, beta):
    """
    LSE of conv(a,a) for normalized a (sum=1) and its gradient.
    F = (1/beta) * log(sum_k exp(beta * c_k))
    grad_j = 2 * sum_l a_l * p_{j+l}  (cross-correlation of p and a)
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    LSE = c_max + np.log(exp_sum) / beta

    # Gradient via cross-correlation computed with FFT
    n2 = 2 * n - 1
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    P_rev = np.fft.rfft(p[::-1], M_fft)
    cross_full = np.fft.irfft(A * P_rev, M_fft)[:conv_len]
    cross_corr = cross_full[n - 1:2 * n - 1][::-1]
    grad = 2.0 * cross_corr
    return LSE, grad


def nesterov_lse(a0, beta_schedule, steps_per_beta, base_momentum=0.9,
                 step_scale=0.5, max_step=0.08):
    """
    Nesterov-momentum gradient descent on LSE with beta annealing.
    Works in normalized space (sum a = 1).
    """
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    stagnant = 0
    scale = step_scale

    for beta_idx, beta in enumerate(beta_schedule):
        steps = steps_per_beta[beta_idx] if beta_idx < len(steps_per_beta) else 100
        # Reduce momentum for very high beta to avoid oscillation
        if beta > 5000:
            mom = 0.5 + 0.4 * max(0.0, 1.0 - beta / 200000.0)
        else:
            mom = base_momentum

        for _ in range(steps):
            val, grad = lse_grad_normalized(a, beta)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            ls = min(scale / max(gnorm, 1e-10), max_step)
            v = mom * v + ls * grad
            a_new = a - v
            a_new = normalize(a_new)
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 200:
                scale = max(scale * 0.8, 0.02)
                stagnant = 0
            a = a_new
    return best_a, best_sc


def multi_peak_subgrad(a0, n_iters=8000, top_k=10, step_init=0.03, decay=0.99995):
    """
    Multi-peak subgradient: targets top-K peaks simultaneously.
    Update direction: g_j = (2/K) * sum_i a_{k_i - j} (mean-centered for tangent).
    """
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for _ in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        K = min(top_k, 2 * n - 1)
        top_idx = np.argpartition(c, -K)[-K:]
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                g[j_start:j_end + 1] += 2.0 * seg[::-1]
        g /= K
        g -= g.mean()  # project to tangent space (preserve sum)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            break
        ls = min(step, 0.06)
        a = normalize(a - ls * g)
        step *= decay
    return best_a, best_sc


def single_peak_subgrad(a0, n_iters=15000, step_init=0.015, decay=0.99995):

    a = normalize(a0.astype(np.float64).copy())

        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        g = np.zeros(n)
        seg = a[j_start:j_end + 1]
        g[j_start:j_end + 1] = 2.0 * seg[::-1]
        g -= g.mean()
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            break
        ls = min(step, 0.04)
        a = normalize(a - ls * g)
        step *= decay
    return best_a, best_sc


def gs_flat_target(a0, n_iters=300, alpha=0.5):
    """
    GS-Fienup: project in frequency domain toward flat convolution target.
    """
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    b_target = np.zeros(M)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def perturb_reopt(a, n_attempts, rng, betas=(500, 2000, 10000), steps=(200, 200, 300)):
    """Random perturbation of random subset, then quick LSE re-optimization."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_attempts):
        n_pert = int(rng.integers(5, max(10, n // 100)))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_pert = a.copy()
        a_pert[idx] *= (1.0 + rng.standard_normal(n_pert) * 0.4)
        a_pert = normalize(a_pert)
        a_opt, sc_opt = nesterov_lse(a_pert, list(betas), list(steps),
                                     base_momentum=0.9, step_scale=0.3)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
    return best_a, best_sc


def build_inits(n, rng, parent=None):
    """Build diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None:
        if len(parent) == n:
            inits.append(normalize(parent.copy()))
        else:
            inits.append(normalize(np.interp(x, np.linspace(0, 1, len(parent)), parent)))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.sin(np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.2 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * 2 * x) * np.sin(np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.15 * rng.standard_normal(n), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Hybrid: Nesterov-LSE + multi-peak subgrad + GS + perturbation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    n_primary = len(parent) if parent is not None else 6962
    print(f"n_primary={n_primary}, parent={'yes' if parent is not None else 'no'}, "
          f"budget={budget_s-60:.0f}s")

    # Phase 1: GS-Fienup warmup (15%)
    p1_end = t_start + 0.15 * (deadline - t_start)
    print(f"\n=== Phase 1: GS warmup ===")
    a_cur = normalize(parent) if parent is not None else np.ones(n_primary) / n_primary
    for alpha in [0.6, 0.4, 0.3]:
        if time.monotonic() > p1_end:
            break
        a_cur, sc = gs_flat_target(a_cur, n_iters=300, alpha=alpha)
        update_best(sc, a_cur, f"GS a={alpha}")

    # Phase 2: Nesterov LSE main optimization (45%)
    p2_end = t_start + 0.60 * (deadline - t_start)
    print(f"\n=== Phase 2: Nesterov LSE ===")
    # Long annealing schedule
    betas1 = [20, 100, 500, 2000, 10000, 50000]
    steps1 = [400, 300, 300, 300, 200, 150]
    a_opt, sc_opt = nesterov_lse(best_a, betas1, steps1,
                                  base_momentum=0.92, step_scale=0.5)
    update_best(sc_opt, a_opt, "Nesterov-main")

    # Second schedule with higher starting beta
    if time.monotonic() < p2_end - 120:
        betas2 = [200, 1000, 5000, 20000, 100000, 500000]
        steps2 = [300, 300, 300, 200, 150, 100]
        a_opt, sc_opt = nesterov_lse(best_a, betas2, steps2,
                                      base_momentum=0.9, step_scale=0.4)
        update_best(sc_opt, a_opt, "Nesterov-high")

    # Try from different init
    if time.monotonic() < p2_end - 60:
        inits = build_inits(n_primary, rng, parent)
        for i in range(1, min(4, len(inits))):
            if time.monotonic() > p2_end:
                break
            a_opt, sc_opt = nesterov_lse(inits[i], betas1[:4], steps1[:4],
                                          base_momentum=0.9, step_scale=0.4)
            update_best(sc_opt, a_opt, f"Nesterov-init{i}")

    print(f"After Phase 2: best={best_score:.8f}")

    # Phase 3: Multi-peak subgradient refinement (20%)
    p3_end = t_start + 0.80 * (deadline - t_start)
    print(f"\n=== Phase 3: Multi-peak subgradient ===")
    for K in [3, 5, 8, 15, 30]:
        if time.monotonic() > p3_end:
            break
        a_sg, sc_sg = multi_peak_subgrad(best_a, n_iters=6000, top_k=K,
                                         step_init=0.025, decay=0.99995)
        update_best(sc_sg, a_sg, f"MultiPeak K={K}")

    if time.monotonic() < p3_end - 30:
        a_sp, sc_sp = single_peak_subgrad(best_a, n_iters=12000,
                                           step_init=0.012, decay=0.99996)
        update_best(sc_sp, a_sp, "SinglePeak")

    print(f"After Phase 3: best={best_score:.8f}")

    # Phase 4: Random perturbation + re-optimize (10%)
    p4_end = deadline
    print(f"\n=== Phase 4: Perturbation ===")
    for round_idx in range(6):
        if time.monotonic() > p4_end:
            break
        a_p, sc_p = perturb_reopt(best_a, n_attempts=4, rng=rng,
                                   betas=(1000, 5000, 20000), steps=(150, 150, 200))
        update_best(sc_p, a_p, f"Perturb r{round_idx}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 40:
        a_gs, sc_gs = gs_flat_target(best_a, n_iters=200, alpha=0.2)
        update_best(sc_gs, a_gs, "Final GS")

    if time.monotonic() < deadline - 30:
        a_fin, sc_fin = nesterov_lse(best_a, [100000, 500000], [100, 80],
                                      base_momentum=0.6, step_scale=0.2)
        update_best(sc_fin, a_fin, "Final LSE")

    if time.monotonic() < deadline - 15:
        a_sg2, sc_sg2 = multi_peak_subgrad(best_a, n_iters=5000, top_k=5,
                                            step_init=0.01, decay=0.99996)
        update_best(sc_sg2, a_sg2, "Final Subgrad")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
