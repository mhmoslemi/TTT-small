Combines GS-Fienup, Top-M LP (sequential linearization), subgradient peak-chasing,

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_fienup(a, n_iters, alpha_init, alpha_final, M=None):
    """GS-Fienup iteration with flat spectral target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Cosine annealing for smoother alpha schedule
        alpha = alpha_final + 0.5 * (alpha_init - alpha_final) * (1.0 + np.cos(np.pi * progress))

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

    c = fast_conv(a)
    if top_m >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[::-1][:top_m]

    # Build LP: minimize t
    # Variables: x = [a_0, ..., a_{n-1}, t]
    # Constraints: for each k in top_idx: sum_j a[k-j]*a_j - t <= 0
    #              sum_j a_j = 1
    #              a_j >= 0
    # Objective: [0, 0, ..., 0, 1] (minimize t)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = np.zeros((len(top_idx), n + 1))
    b_ub = np.zeros(len(top_idx))

    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A_ub[i, j_start:j_end+1] = a[j_start:j_end+1][::-1]
        A_ub[i, -1] = -1.0
        # b_ub[i] = 0 means: linearized_conv[k] - t <= 0, i.e., t >= linearized_conv[k]

    # Equality constraints: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            a_new = res.x[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                return a
            a_new = a_new / S_new
            return a_new
        else:
            return a
    except Exception:
        return a


def top_m_lp_sequence(a, n_iters=200, top_m=30):
    """Iterate top-M LP steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_new = top_m_lp_step(a, top_m)
        if np.array_equal(a_new, a):
            break
        # Damped update for stability
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_peak_chase(a, n_iters, top_m, step_init=0.04, step_decay=0.997):
    """Subgradient method chasing top-M peaks with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Weight peaks by excess over median
        c_median = np.median(c)
        weights = np.maximum(c[top_idx] - c_median, 0.0)
        w_sum = weights.sum()
        if w_sum < 1e-20:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights = weights / w_sum

        sg = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * (2.0 * seg[::-1])

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak-inspired step with cap
        step_candidate = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, step_candidate, 0.06)
        step = max(step, 1e-7)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step = step_init * (step_decay ** it)

    return best_a


def lse_objective_and_grad(x, beta):
    """LSE-smoothed score and its gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -400, 400)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient of L w.r.t. x:
    # d(L)/d(x_j) = sum_k p[k] * d(conv[k])/d(x_j) = sum_k p[k] * x[k-j]
    # Use FFT for efficiency
    rev_x = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * P, M2_fft)[:M2]
    # conv_res[m] = sum_i x[n-1-i] * p[m-i]
    # We need g_j = sum_k p[k] * x[k-j] for j=0..n-1
    # g_j = conv_res[j + n - 1] (by the indexing above)
    g_conve = conv_res[n-1:2*n-1]
    g = 2.0 * g_conve

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
    else:
        a0 = a0 / S
    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            lse_objective_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-11,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_local_opt(a, n_rounds=200, top_k=3, n_candidates=7):
    """Local optimization: for top peak lags, try scaling down key contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Focus on top few lags
        top_lags = np.argsort(c)[::-1][:top_k]

        for k in top_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1]

            # Top contributors
            sorted_idx = np.argsort(contrib)[::-1][:8]
            for local_j in sorted_idx[:4]:
                global_j = j_start + local_j
                old_val = a[global_j]
                if old_val < 1e-10:
                    continue
                for factor in [0.7, 0.8, 0.9, 1.1, 1.2]:
                    a[global_j] = old_val * factor
                    S_new = float(a.sum())
                    if S_new < 1e-12:
                        a[global_j] = old_val
                        continue
                    a_norm = a / S_new
                    sc_new = compute_score(a_norm)
                    if sc_new < sc:
                        sc = sc_new
                        a = a_norm
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        break
                    else:
                        a[global_j] = old_val
                        a = a * S / S_new if S_new > 0 else a  # restore (a was already normalized)
                        a = a / a.sum() if a.sum() > 0 else a

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.copy(), 0.0, 1000.0))
        else:
            inits.append(np.interp(np.linspace(0, 1, n),
                                    np.linspace(0, 1, len(parent)), parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.sin(4.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Hybrid search combining GS-Fienup, Top-M LP, subgradient, and L-BFGS.
    The Top-M LP is a new key ingredient that provides more principled updates
    by solving a linearized subproblem at each step.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    n_primary = 6962
    if parent is not None:
        n_primary = len(parent)

    print(f"N primary={n_primary}, Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: GS-Fienup to get a good starting point (30% time)
    # ==================================================================
    phase1_end = t_start + 0.30 * (budget_s - 30.0)
    print(f"\n=== Phase 1: GS (until t={phase1_end - t_start:.0f}s) ===")

    inits = build_inits(n_primary, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break

        # Multiple GS configurations, chained
        a_cur = a0.copy()
        for cfg_idx, (iters, a_init, a_final) in enumerate([
            (400, 0.8, 0.15),
            (400, 0.5, 0.05),
            (300, 0.3, 0.05),
        ]):
            if time.monotonic() > phase1_end:
                break
            a_cur = gs_fienup(a_cur, iters, a_init, a_final)
            sc = compute_score(a_cur)
            update_best(sc, a_cur, f"GS{i}-cfg{cfg_idx}")

        # Also try magnitude-only
        if time.monotonic() < phase1_end:
            a_mag = gs_magnitude_only(a0, 400, 0.5)
            sc = compute_score(a_mag)
            update_best(sc, a_mag, f"GSmag{i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Top-M LP (20% time) - NEW KEY METHOD
    # ==================================================================
    phase2_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Top-M LP (until t={phase2_end - t_start:.0f}s) ===")

    for top_m in [20, 30, 50, 80]:
        if time.monotonic() > phase2_end:
            break
        t_lp_start = time.monotonic()
        a_lp = top_m_lp_sequence(best_a, n_iters=300, top_m=top_m)
        sc = compute_score(a_lp)
        update_best(sc, a_lp, f"LP-m{top_m}")
        print(f"    top_m={top_m}: score={sc:.8f}, time={time.monotonic()-t_lp_start:.1f}s")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Subgradient refinement (15% time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Subgradient (until t={phase3_end - t_start:.0f}s) ===")

    for top_m in [10, 30, 60, 100, 200]:
        if time.monotonic() > phase3_end:
            break
        a_sg = subgradient_peak_chase(best_a, 5000, top_m, 0.04, 0.998)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG-m{top_m}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS with multiple betas (20% time)
    # ==================================================================
    phase4_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 4: L-BFGS (until t={phase4_end - t_start:.0f}s) ===")

    beta_list = [100, 300, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in beta_list:
        if time.monotonic() > phase4_end:
            break
        t_beta = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=4000, maxfun=80000)
        update_best(sc, a_opt, f"LB-b{beta}")
        print(f"    beta={beta}: score={sc:.8f}, time={time.monotonic()-t_beta:.1f}s")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final refinement cycles (15% time)
    # ==================================================================
    print(f"\n=== Phase 5: Final refinement ===")

    cycle_budget = deadline - time.monotonic()
    for cycle in range(5):
        if time.monotonic() > deadline:
            break

        # GS polish
        a_gs = gs_fienup(best_a, 200, 0.3, 0.05)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"C{cycle}-GS")

        # LP polish
        if time.monotonic() < deadline - 60:
            a_lp = top_m_lp_sequence(best_a, n_iters=100, top_m=40)
            sc = compute_score(a_lp)
            update_best(sc, a_lp, f"C{cycle}-LP")

        # Subgradient
        if time.monotonic() < deadline - 60:
            a_sg = subgradient_peak_chase(best_a, 3000, 40, 0.03, 0.997)
            sc = compute_score(a_sg)
            update_best(sc, a_sg, f"C{cycle}-SG")

        # L-BFGS final
        if time.monotonic() < deadline - 30:
            sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"C{cycle}-LB")

        # Peak local opt
        if time.monotonic() < deadline - 30:
            a_local = peak_local_opt(best_a, n_rounds=50, top_k=2, n_candidates=5)
            sc = compute_score(a_local)
            update_best(sc, a_local, f"C{cycle}-Loc")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
