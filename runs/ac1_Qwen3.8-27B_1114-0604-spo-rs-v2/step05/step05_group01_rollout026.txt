import numpy as np

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _corr_pair(a, w):
    """Compute (a * w)[n-1 : 2n-1] where a has length n."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1].astype(np.float64), N)
    W = np.fft.rfft(w.astype(np.float64), N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-14:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    return a * (target / S)


def _resample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    n = int(n)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a, left=0.0, right=0.0).astype(np.float64)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    S = float(a.sum())
    if S < 1e-12:

    c = _fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:

    return float(2.0 * n * cmax / (S * S))

    return np.maximum(d, 0.0), S
