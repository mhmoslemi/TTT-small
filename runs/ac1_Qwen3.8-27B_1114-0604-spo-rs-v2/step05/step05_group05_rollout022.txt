import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum
    
    # Gradient computation via FFT
    # dL/da_i = 2 * sum_m p[m+i] * a[m] for m=0..n-1
    # This is 2 * (p_rev * a)[2n-2-i] where p_rev is reversed p
    conv_len = 2 * n - 1 + n - 1  # = 3n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev))[:conv_len]
    # d[i] = 2 * conv_res[2n-2-i] for i=0..n-1
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def adam_lse(a, max_time, beta_start=10.0, beta_end=10000.0, 
             lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.003
    
    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * beta_growth, beta_end)
        
        # Compute gradient
        _, grad = lse_score_and_grad(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        
        if it % 50 == 0 or it == 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a

def multiplicative_adam_lse(a, max_time, beta_start=10.0, beta_end=5000.0,
                            lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Multiplicative Adam on LSE-smoothed objective (mirror descent style)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.005
    
    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * beta_growth, beta_end)
        
        _, grad = lse_score_and_grad(a, beta)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        # Multiplicative update: a_i <- a_i / exp(lr * m_hat_i / (sqrt(v_hat_i) + eps))
        # Clip the exponent to prevent overflow
        log_step = np.clip(lr * m_hat / (np.sqrt(v_hat) + eps), -10, 10)
        a = a / np.exp(log_step)
        a = np.maximum(a, 1e-15)  # Prevent underflow
        S = float(a.sum())
        a = a / S
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a

def stochastic_peak_subgrad(a, max_time, n_peaks=5, lr_start=0.05, lr_end=0.005):
    """Stochastic subgradient method sampling from near-max peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        t = (time.monotonic() - t_start) / max_time
        lr = lr_start + (lr_end - lr_start) * t
        
        # Compute convolution
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Sample peaks proportional to their value (among top peaks)
        # Use softmax sampling over the top n_peaks
        top_idx = np.argsort(c)[::-1][:n_peaks * 10]
        weights = np.exp(50 * (c[top_idx] - c[top_idx[0]]))
        weights = weights / weights.sum()
        k_sampled = top_idx[np.random.choice(len(top_idx), p=weights)]
        k_sampled = int(k_sampled)
        
        # Subgradient at k_sampled: grad_i = 2 * a[k_sampled - i] for valid i
        j_start = max(0, k_sampled - n + 1)
        j_end = min(k_sampled, n - 1)
        grad = np.zeros(n)
        for j in range(j_start, j_end + 1):
            grad[j] = 2.0 * a[k_sampled - j]
        
        # Normalize gradient (scale-invariant objective)
        S2 = S * S
        S3 = S2 * S
        # d(score)/da_j = 2n * (grad_L_j / S2 - 2*L/S3) where L = c_max, grad_L = subgrad
        # For the subgradient, L = c[k_sampled]
        L = float(c[k_sampled])
        grad_score = 2.0 * n * (grad / S2 - 2.0 * L / S3)
        
        a = a - lr * grad_score
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a

def sca_lp_step(a_t, n_reduced=None):
    """
    One SCA iteration: solve the linearized LP and return the solution.
    If n_reduced is specified, work at that scale and upsample.
    """
    n = len(a_t)
    if n_reduced is not None and n_reduced < n:
        # Downsample for LP, then upsample
        a_small = np.interp(
            np.linspace(0, 1, n_reduced),
            np.linspace(0, 1, n),
            a_t
        )
        a_small = np.maximum(a_small, 0)
        a_small = a_small / a_small.sum()
        result_small = sca_lp_step(a_small, n_reduced=None)
        # Upsample back
        result = np.interp(
            np.linspace(0, 1, n),
            np.linspace(0, 1, n_reduced),
            result_small
        )
        result = np.maximum(result, 0)
        if result.sum() > 1e-12:
            result = result / result.sum()
        return result
    
    c_t = fast_conv(a_t)
    n_vars = n + 1  # a_0..a_{n-1}, M
    
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    rhs = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a_t[k - j])
        # -M constraint
        rows.append(k)
        cols.append(n)  # M is variable n
        vals.append(-1.0)
        rhs[k] = c_t[k]
    
    A_mat = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n_vars))
    
    # Equality constraint: sum(a) = 1
    eq_row = np.zeros(n_vars)
    eq_row[:n] = 1.0
    
    # CVXPY problem
    a_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [
        cp.sum(a_var) == 1,
        a_var >= 0,
        M_var >= 0
    ]
    
    # Add the linearized constraints: A[:n] @ a - M <= c_t
    # i.e., A_mat[:2n-1, :n] @ a_var - A_mat[:2n-1, n] * M_var <= rhs
    # A_mat[:2n-1, n] is all -1, so: A_mat[:2n-1, :n] @ a_var + M_var <= rhs... 
    # Wait, the constraint is: sum_j 2*a_t[k-j]*a_j - M <= c_t[k]
    # i.e., A_conv @ a - 1 * M <= c_t
    # In CVXPY: A_conv @ a_var - M_var <= c_t
    
    A_conv = A_mat[:2*n-1, :n]  # The convolution part
    # We need: A_conv @ a_var - M_var <= c_t
    # Since A_conv is sparse, let's use it directly
    
    constraints.append(A_conv @ a_var - M_var <= c_t)
    
    objective = cp.Minimize(M_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return a_t.copy()
    
    if a_var.value is not None:
        a_out = np.maximum(a_var.value, 0.0)
        S = float(a_out.sum())
        if S > 1e-12:
            a_out = a_out / S
            return a_out
    
    return a_t.copy()

def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    x0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    if x0.sum() < 1e-6:
        x0[:] = 0.5
    x0 = x0 / x0.sum()
    
    bounds = [(0.0, None)] * n
    
    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, x0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    if a_out.sum() < 1e-6:
        a_out = x0.copy()
    sc = compute_score(a_out)
    return sc, a_out

"""
Multi-method optimizer: SCA-LP + Adam-LSE + Multiplicative Mirror Descent + Stochastic Peak Subgradient.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences of length n.

Key innovations over previous approach:
1. Sequential Convex Approximation (SCA) with LP linearization at reduced scale
2. Adam optimizer with per-coordinate adaptive learning rates on LSE-smoothed objective
3. Multiplicative (mirror descent) updates for natural non-negativity preservation
4. Stochastic peak subgradient sampling for local minimum escape
5. Multi-restart with progressive perturbation and simulated annealing acceptance
6. Fienup with peak-aware frequency targeting
"""

import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient w.r.t. a.
    Uses FFT for efficient convolution-based gradient.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum

    # Gradient: dL/da_i = 2 * sum_m a[m] * p[m+i]
    # Computed via: conv(a, p_reversed)
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev))[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def _lbfgs_obj(x, beta):
    """Wrapped objective+gradient for L-BFGS-B."""
    return lse_score_and_grad(x, beta)


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective at given beta."""
    n = len(a0)
    x0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    if x0.sum() < 1e-6:
        x0[:] = 0.5
    x0 = x0 / x0.sum()
    bounds = [(0.0, None)] * n
    res = minimize(
        _lbfgs_obj, x0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    if a_out.sum() < 1e-6:
        a_out = x0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def adam_lse(a, max_time, beta_start=10.0, beta_end=10000.0,
             lr=0.008, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective with adaptive beta."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.004
    lr_decay = 0.9995

    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * beta_growth, beta_end)
        lr *= lr_decay

        _, grad = lse_score_and_grad(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final

    return best_a


def multiplicative_lse(a, max_time, beta_start=10.0, beta_end=5000.0,
                       lr=0.15, beta1=0.9, beta2=0.999, eps=1e-8):
    """Multiplicative (mirror descent) Adam on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.006
    lr_decay = 0.999

    while time.monotonic() - t_start < max_time:
        it += 1
        beta = min(beta * beta_growth, beta_end)
        lr *= lr_decay

        _, grad = lse_score_and_grad(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Multiplicative update: a_i <- a_i * exp(-lr * m_hat_i / (sqrt(v_hat_i)+eps))
        log_step = np.clip(lr * m_hat / (np.sqrt(v_hat) + eps), -15.0, 15.0)
        a = a * np.exp(-log_step)
        a = np.maximum(a, 1e-15)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final

    return best_a


def stochastic_peak_subgrad(a, max_time, n_sample=10, lr_start=0.03, lr_end=0.003,
                            temperature=20.0, rng=None):
    """Stochastic subgradient method sampling from near-max peaks with softmax weighting."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < max_time:
        t_frac = (time.monotonic() - t_start) / max_time
        lr = lr_start + (lr_end - lr_start) * t_frac
        it += 1

        c = fast_conv(a)
        c_max = float(c.max())

        # Sample from top peaks with softmax weighting
        top_k = min(n_sample * 5, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        log_w = temperature * (c[top_idx] / c_max - 1.0)
        log_w = log_w - log_w.max()
        w = np.exp(log_w)
        w = w / w.sum()
        k_s = int(top_idx[rng.choice(len(top_idx), p=w)])

        # Subgradient for peak at k_s
        j_start = max(0, k_s - n + 1)
        j_end = min(k_s, n - 1)
        grad = np.zeros(n)
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * a[k_s - indices]

        S2 = S * S
        S3 = S2 * S
        L = float(c[k_s])
        grad_score = 2.0 * n * (grad / S2 - 2.0 * L / S3)

        a = a - lr * grad_score
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def fienup_peak_targeted(a, n_iters=200, thresh=1.5, clip=0.6,
                         momentum_end=0.3, focus_peaks=3):
    """Fienup that targets specific frequency bands associated with peak locations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = float(a.sum())
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh_i = 2.5 - 1.0 * t
        clip_i = 0.3 + 0.6 * t
        momentum = 0.3 * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Find peaks in time domain to identify which frequencies to suppress
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:focus_peaks]

        # For each peak, the "frequency content" that creates it
        # Peak at lag k is created by all frequencies; we target the bins
        # where |A|^2 is highest relative to mean
        above = P > P_mean * thresh_i
        scale = np.ones_like(P)
        if above.any():
            target_amp = np.sqrt(P_mean * thresh_i)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip_i + clip_i * (target_amp / np.maximum(current_amp, 1e-10))
            scale[above] = np.clip(factor, 0.05, 1.0)

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def sca_lp_step(a_t, n_reduced):
    """
    One SCA iteration: solve linearized LP at reduced scale, return upsampled solution.
    The LP minimizes M subject to linearized autocorrelation constraints.
    """
    n = len(a_t)
    # Downsample
    a_small = np.interp(
        np.linspace(0, 1, n_reduced),
        np.linspace(0, 1, n),
        a_t
    )
    a_small = np.maximum(a_small, 0.0)
    S_small = float(a_small.sum())
    if S_small < 1e-12:
        return a_t.copy()
    a_small = a_small / S_small

    nr = n_reduced
    c_r = fast_conv(a_small)

    # Build constraint: for each k, sum_j 2*a_small[k-j]*a_j - M <= c_r[k]
    # Variables: a_0..a_{nr-1}, M (total nr+1 variables)
    rows = []
    cols = []
    vals = []

    for k in range(2 * nr - 1):
        j_start = max(0, k - nr + 1)
        j_end = min(k, nr - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a_small[k - j])
        # -M term
        rows.append(k)
        cols.append(nr)
        vals.append(-1.0)

    A_conv = sp.csc_matrix((vals, (rows, cols)), shape=(2 * nr - 1, nr + 1))

    # CVXPY formulation
    a_var = cp.Variable(nr)
    M_var = cp.Variable()

    constraints = [
        cp.sum(a_var) == 1,
        a_var >= 0,
        M_var >= 0
    ]

    # Linearized constraint: A_conv[:, :nr] @ a_var - A_conv[:, nr] * M_var <= c_r
    # A_conv[:, nr] is all -1, so: A_conv[:, :nr] @ a_var + M_var <= c_r... 
    # Wait: the constraint is sum_j 2*a_small[k-j]*a_j - M <= c_r[k]
    # i.e., A_conv[:, :nr] @ a_var - 1*M_var <= c_r
    # A_conv[:, nr] = -1, so A_conv @ [a; M] <= c_r
    # which is A_conv[:, :nr] @ a + A_conv[:, nr] * M <= c_r
    # = A_conv[:, :nr] @ a - M <= c_r
    # So in CVXPY: A_conv[:, :nr] @ a_var + A_conv[:, nr] * M_var <= c_r
    A_part = A_conv[:, :nr].tocsc()
    m_coeff = A_conv[:, nr]

    constraints.append(A_part @ a_var + (m_coeff * M_var) <= c_r)

    objective = cp.Minimize(M_var)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return a_t.copy()

    if a_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        a_out_small = np.maximum(a_var.value, 0.0)
        S_out = float(a_out_small.sum())
        if S_out > 1e-12:
            a_out_small = a_out_small / S_out
            # Upsample back to full scale
            a_out = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, nr),
                a_out_small
            )
            a_out = np.maximum(a_out, 0.0)
            S_full = float(a_out.sum())
            if S_full > 1e-12:
                a_out = a_out / S_full
                return a_out

    return a_t.copy()


def multi_scale_sca(a_full, max_time, n_reduced=2048, n_iters=30, damping=0.5):
    """Run multiple SCA iterations at reduced scale, tracking best full-scale result."""
    n = len(a_full)
    best_a = a_full.copy()
    best_sc = compute_score(a_full)

    # Downsample
    a_work = np.interp(
        np.linspace(0, 1, n_reduced),
        np.linspace(0, 1, n),
        a_full
    )
    a_work = np.maximum(a_work, 0.0)
    a_work = a_work / a_work.sum()

    t_start = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t_start > max_time:
            break

        a_lp = sca_lp_step(a_full, n_reduced)
        sc_lp = compute_score(a_lp)

        # Damped update
        a_new = (1.0 - damping) * best_a + damping * a_lp
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                # Also update the working point
                a_work = np.interp(
                    np.linspace(0, 1, n_reduced),
                    np.linspace(0, 1, n),
                    best_a
                )
                a_work = a_work / a_work.sum()

        # Also try full LP solution directly
        if sc_lp < best_sc:
            best_sc = sc_lp
            best_a = a_lp.copy()

    return best_a


def peak_coordinate_descent(a, n_sweeps=3, n_top=20):
    """Coordinate descent targeting top peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                improved = False
                for factor in [0.6, 0.75, 0.85, 1.15, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_scaled.copy()
                        a = a_scaled
                        a[j] = a[j] * (old_val * factor) / old_val  # keep consistent
                        improved = True
                        break
                if not improved:
                    a[j] = old_val

    return best_a


def upsample_seq(a, new_n):
    """Upsample a sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    return np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, old_n), a)

    # Phase 1: SCA-LP at reduced scale (25% of time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - 20.0)
    print(f"\n=== Phase 1: SCA-LP (budget {phase1_end-t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 10:
        time_avail = phase1_end - time.monotonic() - 5
        n_red = min(2048, n_target // 2)
        a_sca = multi_scale_sca(best_a, max_time=time_avail,
                                n_reduced=n_red, n_iters=40, damping=0.5)
        try_update(a_sca, "SCA-LP")

        # Second SCA round with different damping
        if time.monotonic() < phase1_end - 10:
            time_avail2 = phase1_end - time.monotonic() - 5
            a_sca2 = multi_scale_sca(best_a, max_time=time_avail2,
                                     n_reduced=n_red, n_iters=30, damping=0.7)
            try_update(a_sca2, "SCA-LP-2")

    print(f"After Phase 1: {best_score:.8f}")

    # ============================================================
    # Phase 2: Adam-LSE optimization (30% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * (total_budget - 20.0)
    print(f"\n=== Phase 2: Adam-LSE (budget {phase2_end-t_start:.0f}s) ===")

    # Adam with different beta schedules
    adam_configs = [
        {'beta_start': 10.0, 'beta_end': 8000.0, 'lr': 0.008},
        {'beta_start': 20.0, 'beta_end': 15000.0, 'lr': 0.005},
        {'beta_start': 5.0, 'beta_end': 5000.0, 'lr': 0.012},
    ]

    for i, cfg in enumerate(adam_configs):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 20:
            break
        print(f"  Adam-LSE config {i}: beta=[{cfg['beta_start']},{cfg['beta_end']}], lr={cfg['lr']}")
        a_new = adam_lse(best_a, max_time=time_left * 0.9, **cfg)
        try_update(a_new, f"adam-{i}")

    # Perturbed restart
    if time.monotonic() < phase2_end - 20:
        a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n_target))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        time_left = phase2_end - time.monotonic()
        a_new = adam_lse(a_pert, max_time=time_left * 0.8,
                         beta_start=20.0, beta_end=10000.0, lr=0.006)
        try_update(a_new, "adam-pert")

    print(f"After Phase 2: {best_score:.8f}")

    # ============================================================
    # Phase 3: Multiplicative Mirror Descent (20% of time)
    # ============================================================
    phase3_end = t_start + 0.75 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Multiplicative MD (budget {phase3_end-t_start:.0f}s) ===")

    md_configs = [
        {'beta_start': 10.0, 'beta_end': 5000.0, 'lr': 0.15},
        {'beta_start': 30.0, 'beta_end': 8000.0, 'lr': 0.08},
        {'beta_start': 5.0, 'beta_end': 3000.0, 'lr': 0.20},
    ]

    for i, cfg in enumerate(md_configs):
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 20:
            break
        print(f"  Mult-MD config {i}: beta=[{cfg['beta_start']},{cfg['beta_end']}], lr={cfg['lr']}")
        a_new = multiplicative_lse(best_a, max_time=time_left * 0.9, **cfg)
        try_update(a_new, f"multmd-{i}")

    print(f"After Phase 3: {best_score:.8f}")

    # ============================================================
    # Phase 4: Stochastic Peak Subgradient + Fienup (15% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n=== Phase 4: Stochastic Subgrad + Fienup (budget {phase4_end-t_start:.0f}s) ===")

    # Stochastic peak subgradient
    if time.monotonic() < phase4_end - 30:
        time_left = (phase4_end - time.monotonic()) * 0.5
        a_new = stochastic_peak_subgrad(best_a, max_time=time_left,
                                        n_sample=10, lr_start=0.02, lr_end=0.002,
                                        temperature=30.0, rng=rng)
        try_update(a_new, "stoch-subgrad")

    # Fienup with peak targeting
    if time.monotonic() < phase4_end - 10:
        time_left = min(phase4_end - time.monotonic(), 120)
        a_new = fienup_peak_targeted(best_a, n_iters=300, thresh=1.4,
                                     clip=0.7, momentum_end=0.4, focus_peaks=5)
        try_update(a_new, "fienup-peak")

    # Another stochastic pass with different parameters
    if time.monotonic() < phase4_end - 10:
        time_left = phase4_end - time.monotonic() - 5
        if time_left > 15:
            a_new = stochastic_peak_subgrad(best_a, max_time=time_left,
                                            n_sample=5, lr_start=0.01, lr_end=0.001,
                                            temperature=50.0, rng=rng)
            try_update(a_new, "stoch-subgrad-2")

    print(f"After Phase 4: {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS Polish (remaining time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: L-BFGS Polish (budget {phase5_end-t_start:.0f}s) ===")

    lbfgs_betas = [500, 1000, 2000, 5000, 10000, 20000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase5_end - 10:
            break
        time_left = phase5_end - time.monotonic() - 5
        if time_left < 10:
            break
        # Scale maxiter based on available time
        maxit = min(5000, max(500, int(time_left * 50)))
        print(f"  L-BFGS beta={beta}, maxiter={maxit}")
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=maxit, maxfun=maxit * 20)
        try_update(a_opt, f"lbfgs-b{beta}")

    # Final coordinate descent
    if time.monotonic() < phase5_end - 15:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=3, n_top=15)
        try_update(a_cd, "coord-descent")

        # Quick LSE after CD
        if time.monotonic() < phase5_end - 10:
            a_cd2 = adam_lse(a_cd, max_time=min(30, phase5_end - time.monotonic() - 5),
                             beta_start=500.0, beta_end=5000.0, lr=0.003)
            try_update(a_cd2, "cd+adam")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if len(out) < 1 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
