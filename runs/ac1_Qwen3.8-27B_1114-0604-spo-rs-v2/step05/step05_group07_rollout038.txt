def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

"""
LP-guided optimization with L-BFGS polishing for minimizing 2n*max(a*a)/(sum a)^2.
Uses alternating LP direction-finding (block coordinate descent), smooth
L-BFGS refinement, and iterated local search with diverse perturbations.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import diags, csr_matrix
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

_current_beta = 100.0

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # dL/da_i = 2 * sum_k p_k * a_{k-i} (for valid k-i)
    # = 2 * (p * rev_a)[n-1+i] where rev_a = a[::-1]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def solve_lp_direction(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (conv(f,g))_k <= M for all k, g >= 0."""
    from scipy.optimize import linprog
    n = len(f)
    
    # Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n
    # Using scipy.sparse.diags
    M_mat, N_mat = 2 * n - 1, n
    diagonals = []
    offsets = []
    for d in range(n):
        diagonals.append(np.full(N_mat, f[d]))
        offsets.append(-d)
    A_ub = diags(diagonals, offsets, shape=(M_mat, N_mat), format='csr')
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None

def lp_bcd_iteration(f, time_limit=15):
    """One iteration of LP-based block coordinate descent with backtracking."""
    n = len(f)
    c = fast_conv(f)
    S = float(f.sum())
    M = float(np.max(c))
    current_score = 2.0 * n * M / (S * S)
    
    g = solve_lp_direction(f, M, time_limit=time_limit)
    if g is None:
        return f, current_score, False
    
    Sg = g.sum()
    if Sg < 1e-12:
        return f, current_score, False
    g = g / Sg
    
    # Backtracking line search
    best_f = f
    best_sc = current_score
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            continue
        f_new = f_new / S_new
        sc_new = compute_score(f_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_f = f_new
            if sc_new < current_score * (1 - 1e-6):
                break
    
    return best_f, best_sc, (best_f is not f)

def mirror_descent(a0, beta_list, n_iters_total=10000):
    """Mirror descent with adaptive beta."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    iters_per_beta = n_iters_total // len(beta_list)
    
    for beta in beta_list:
        eta = 0.05
        for it in range(iters_per_beta):
            sc, grad = lse_score_and_grad(a, beta)
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
            
            g_centered = grad - np.mean(grad)
            exponent = -eta * g_centered
            exponent = np.clip(exponent, -30, 30)
            a_new = a * np.exp(exponent)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a = a_new / S_new
            eta *= 0.999
    
    return best_a

Or better yet: **derive a multiplicative update from the LP KKT conditions.**

The KKT conditions for the LP are:
1. g >= 0
2. A@g <= M
3. y >= 0 (dual variables for A@g <= M)
4. 1 - A^T*y - mu = 0 (mu >= 0 is dual for g >= 0)
5. y_k * (M - (A@g)_k) = 0 for all k
6. mu_j * g_j = 0 for all j

From 4 and 6: if g_j > 0, then mu_j = 0, so (A^T*y)_j = 1.
From 5: y_k > 0 only if (A@g)_k = M.

At the optimum, if the support of g is S (set of j with g_j > 0), then:
For j in S: sum_k A[k,j] * y_k = 1
For k not active: y_k = 0

This is a system of |S| equations in |S| unknowns (the active y_k's).

A multiplicative update to solve this:
y_k <- y_k * ((A@g)_k / M)^alpha (for the active k's)

And:
g_j <- g_j / ((A^T*y)_j) (for j in S)

This is similar to the Sinkhorn-Knopp algorithm or the multiplicative weights update.

def multiplicative_update(f, n_iters=100, alpha=0.5):
    """Multiplicative update derived from LP KKT conditions."""
    n = len(f)
    c = fast_conv(f)
    S = float(f.sum())
    M = float(np.max(c))
    
    # Initialize y (dual variables)
    y = np.zeros(2*n-1)
    y[np.argmax(c)] = 1.0  # Start at the peak
    
    # Initialize g
    g = np.ones(n)
    Ng = g.sum()
    g = g / Ng
    
    for it in range(n_iters):
        # Compute A@g = conv(f, g)
        cg = np.convolve(f, g)  # length 2n-1
        
        # Update y: increase y_k where (A@g)_k is close to M
        ratio = cg / M
        # y_k <- y_k * ratio^alpha (for active k's)
        # But only for k where ratio is close to 1 (active constraints)
        active_mask = ratio > 0.99
        if np.any(active_mask):
            y_new = np.where(active_mask, y * np.power(ratio[active_mask], alpha)[...], 0)
            # Actually this is getting complicated. Let me simplify.
            pass
        
        # Update g: g_j <- g_j / (A^T*y)_j
        # A^T*y has component j = sum_k A[k,j] * y_k = sum_k f[k-j] * y_k (for valid k-j)
        Ay = np.convolve(f, y, mode='full')  # Wait, this isn't right
        # Actually A^T*y at position j = sum_k A[k,j] * y_k = sum_k f[k-j] * y_k
        # This is the cross-correlation of f and y, not convolution!
        # A[k,j] = f[k-j], so (A^T*y)_j = sum_k f[k-j] * y_k
        # Let l = k-j, then k = j+l, so (A^T*y)_j = sum_l f[l] * y_{j+l}
        # This is the cross-correlation <f, shift_{-j}(y)>
        
        # In FFT terms: cross_corr(f, y)[j] = IFFT(FFT(f) * conj(FFT(y)))[j]
        # But since f and y are real, conj(FFT(y)) = FFT(y[::-1]) (with padding)
        
        # Actually, let me just compute it directly:
        # (A^T*y)_j = sum_{k=0}^{2n-2} f[k-j] * y[k] (where 0<=k-j<n)
        # = sum_{l=0}^{n-1} f[l] * y[j+l] (where l = k-j, and j+l <= 2n-2)
        
        # This is the cross-correlation of f (length n) and y (length 2n-1)
        # Specifically, it's the "valid" part of the correlation

"""
LP-guided Block Coordinate Descent with L-BFGS refinement for minimizing
2n*max(a*a)/(sum a)^2 over non-negative sequences.
Combines: (1) structured initializations, (2) LP-based direction finding
with backtracking, (3) smooth L-BFGS with escalating beta, (4) mirror
descent for exploration, (5) iterated local search with diverse perturbations.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags
import time

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

_current_beta = 100.0

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(S0, 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def build_conv_sparse(n, f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n."""
    M_mat, N_mat = 2 * n - 1, n
    diagonals = [np.full(N_mat, f[d]) for d in range(n)]
    offsets = [-d for d in range(n)]
    return diags(diagonals, offsets, shape=(M_mat, N_mat), format='csr')

def solve_lp_direction(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    A_ub = build_conv_sparse(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None

def lp_bcd_iteration(f, time_limit=10):
    """One LP-BCD iteration with backtracking line search."""
    n = len(f)
    c = fast_conv(f)
    S = float(f.sum())
    M = float(np.max(c))
    current_score = 2.0 * n * M / (S * S)

    g = solve_lp_direction(f, M, time_limit=time_limit)
    if g is None:
        return f, current_score, False

    Sg = float(g.sum())
    if Sg < 1e-12:
        return f, current_score, False
    g = g / Sg

    best_f = f.copy()
    best_sc = current_score
    for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            continue
        f_new = f_new / S_new
        sc_new = compute_score(f_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_f = f_new
    return best_f, best_sc, (best_sc < current_score - 1e-10)

def mirror_descent(a0, beta_list, n_iters_total=5000, eta0=0.05):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    iters_per = n_iters_total // len(beta_list)
    eta = eta0
    for beta in beta_list:
        eta = eta0
        for it in range(iters_per):
            sc_s, grad = lse_score_and_grad(a, beta)
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
            g_centered = grad - np.mean(grad)
            gn = np.linalg.norm(g_centered)
            if gn < 1e-15:
                break
            exponent = -eta * g_centered
            exponent = np.clip(exponent, -30, 30)
            a_new = a * np.exp(exponent)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a = a_new / S_new
            eta *= 0.9995
    return best_a

def perturbation(a, rng, scale=0.03):
    """Diverse perturbation strategies."""
    n = len(a)
    strategy = rng.integers(0, 7)
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

            e = min(s + bsize, n)
            a_p[s:e] *= (1.0 + scale * 2.5 * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 4:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    elif strategy == 5:
        # Peak-targeted: reduce top contributors, boost bottom
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_s = max(0, k_star - n + 1)
        j_e = min(k_star, n - 1)
        seg = a[j_s:j_e+1]
        contrib = seg[::-1].copy()
        top_k = max(5, len(contrib) // 8)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bot_idx = np.argsort(contrib)[:top_k]
        a_p = a.copy()
        for li in top_idx:
            a_p[j_s + li] *= (1.0 - 0.15 * rng.random())
        S_before = a_p.sum()
        deficit = a.sum() - S_before

            a_p[j_s + bot_idx] += deficit / len(bot_idx)

        # Random bridge
        i = rng.integers(0, n - 1)
        j = rng.integers(i + 1, n)

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    xm = x - 0.5  # centered
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Raised cosine to various powers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        base = 0.5 * (1.0 + np.cos(2.0 * np.pi * xm))
        inits.append(np.maximum(base**p, 0.001))
    # Super-Gaussian
    for p in [2.0, 4.0, 6.0, 8.0]:
        inits.append(npclip_power = np.maximum(1.0 - (2*xm)**p, 0.001))
    # Gaussian
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(-alpha * (2 * xm)**2))
    # Flat-top (Gauss-windowed sinc-like)
    inits.append(np.maximum(1.0 - 4*(xm)**4 + 3*(xm)**2, 0.001))
    # Random smooth
    inits.append(np.maximum(0.3 + 0.7 * rng.standard_normal(n).cumsum() / np.sqrt(n), 0.001))
    # Two-peak
    for pos in [0.3, 0.35, 0.4]:
        inits.append(0.5 * np.exp(-20*(xm-pos+0.5)**2) + 0.5*np.exp(-20*(xm+pos-0.5)**2))
    return inits

    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
