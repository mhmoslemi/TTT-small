"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Adam-based LSE gradient descent, peak-cycling, improved Fienup, and L-BFGS-B polish.

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def adam_lse(a, max_time, beta_start=20.0, beta_end=5000.0,
             lr_start=0.01, lr_end=0.001, batch_check=50):
    """Adam-based LSE optimization with adaptive beta."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t_start = time.monotonic()
    beta = beta_start
    lr = lr_start
    n_stuck = 0
    prev_best = best_sc
    
    while time.monotonic() - t_start < max_time:
        t += 1
        # Schedule
        progress = min(1.0, (time.monotonic() - t_start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (1.0 - 0.8 * progress) + 0.02 * lr_start * progress
        
        # Compute gradient
        _, grad = lse_score_and_grad(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        
        # Check for improvement
        if t % batch_check == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                n_stuck = 0
            else:
                n_stuck += 1
                # If stuck for a while, perturb
                if n_stuck > 50:
                    a = best_a * (1.0 + 0.01 * np.random.default_rng().standard_normal(n))
                    a = np.maximum(a, 0.0)
                    S = a.sum()
                    if S < 1e-12:
                        break
                    a = a / S
                    n_stuck = 0
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def peak_cycling(a, max_time, n_top=5):
    """Cycle through top peaks and reduce each one."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(123)
    lr = 0.01
    
    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        # Get top peaks
        peak_idx = np.argsort(c)[::-1][:n_top]
        
        for k in peak_idx:
            k = int(k)
            # Gradient of b[k] w.r.t. a
            # db[k]/da[i] = 2*a[k-i] if 0<=k-i<=n-1
            grad_k = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j <= n - 1:
                    grad_k[i] = 2.0 * a[j]
            
            # Line search: try several step sizes
            gnorm = float(np.linalg.norm(grad_k))
            if gnorm < 1e-15:
                continue
            
            for factor in [1.0, 0.5, 0.25, 0.1]:
                step = lr * factor * grad_k / gnorm
                a_trial = a - step
                a_trial = np.maximum(a_trial, 0.0)
                S_trial = float(a_trial.sum())
                if S_trial < 1e-12:
                    continue
                a_trial = a_trial / S_trial
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    break
        
        # Decay learning rate
        lr *= 0.999
        
        # Periodic check
        if (time.monotonic() - t_start) % 1.0 < 0.01:
            pass
    
    return best_a

            factor = np.clip(factor, 0.01, 1.0)

        a = a + v

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_random_init(n, rng):
    """Random non-negative initialization with some structure."""
    # Mix of smooth and random components
    x = np.linspace(0, 1, n)
    smooth = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + rng.random() * 2 * np.pi)
    smooth = np.maximum(smooth, 0.01)
    noise = 0.05 * rng.standard_normal(n)
    a = smooth + noise
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.ones(n)
    return a / a.sum()


def build_tapered_init(n, rng):
    """Tapered (raised cosine) initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.001)
    a += 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    return a / a.sum()

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses Adam-LSE gradient descent, adaptive Fienup, peak cycling, and L-BFGS-B polish.
Key innovations: adaptive beta schedule, periodic perturbation restarts,
and a peak-cycling method that directly targets the top autocorrelation peaks.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient using FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum

    # Gradient via FFT cross-correlation
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a, max_time, beta_start=20.0, beta_end=5000.0, lr_start=0.01):
    """Adam-based LSE optimization with adaptive beta and periodic restarts."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t_start = time.monotonic()
    n_stuck = 0
    rng = np.random.default_rng(t_start % 10000)

    while time.monotonic() - t_start < max_time:
        t += 1
        progress = min(1.0, (time.monotonic() - t_start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (1.0 - 0.7 * progress) + 0.0005 * progress

        _, grad = lse_score_and_grad(a, beta)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        a = a / S

        if t % 40 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                n_stuck = 0
            else:
                n_stuck += 1
                if n_stuck > 80:
                    a = best_a * (1.0 + 0.02 * rng.standard_normal(n))
                    a = np.maximum(a, 0.0)
                    S = float(a.sum())
                    if S < 1e-12:
                        break
                    a = a / S
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    n_stuck = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_adaptive(a, max_time):
    """Fienup with adaptive threshold, momentum, and clip schedule."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 40)))
        momentum = 0.0 + 0.4 * progress
        thresh = 3.0 - 1.95 * progress
        clip = 0.2 + 0.7 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v

        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a = a / S_new

        if it % 25 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def peak_cycling(a, max_time, n_top=5):
    """Target top peaks with gradient descent steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    lr = 0.005

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:n_top]

        for k in peak_idx:
            k = int(k)
            grad_k = np.zeros(n)
            # db[k]/da[i] = 2*a[k-i] if 0 <= k-i <= n-1
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            if hi >= lo:
                indices = np.arange(lo, hi + 1)
                grad_k[indices] = 2.0 * a[k - indices]

            gnorm = float(np.linalg.norm(grad_k))
            if gnorm < 1e-15:
                continue

            for factor in [1.0, 0.5, 0.25]:
                step = lr * factor * grad_k / gnorm
                a_trial = a - step
                a_trial = np.maximum(a_trial, 0.0)
                S_trial = float(a_trial.sum())
                if S_trial < 1e-12:
                    continue
                a_trial = a_trial / S_trial
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    break

        lr *= 0.9995

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
