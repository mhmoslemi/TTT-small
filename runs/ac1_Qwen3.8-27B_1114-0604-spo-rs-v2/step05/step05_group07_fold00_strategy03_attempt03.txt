            res = minimize(lambda a: float(evaluate_sequence(a)), base_candidate, method='L-BFGSB',
                           bounds=[(0.0, 1000.0)]*n_pop, options={'disp':True, 'max_iter':50000})
            best_candidate = res.x
            return best_candidate
        
    except Exception as e:
        # Fallback to a normalized sequence if optimization fails
        print("Raising child strategies failed:", e)
        if len(parent)==seq_length:
            best_candidate = np.clip(parent.copy(), 0.0, 1000.0)
            best_candidate = best_candidate / sum(best_candidate**0.5)
            return best_candidate
        else:
            uniform_seq = np.ones(seq_length) * 0.5
            return np.clip(uniform_seq, 0.0, 1000.0)
