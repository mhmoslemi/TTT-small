def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

            break

        progress = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_eff = lr_start * (lr_end / lr_start) ** progress

        L, grad = lse_grad_max_b(a, beta)

        # Multiplicative update: a_new = a * exp(-lr * grad), then normalize
        log_a = np.log(a) - lr_eff * grad
        # Clamp to avoid underflow
        log_a = np.clip(log_a, -50.0, 50.0)
        a_new = np.exp(log_a - log_a.max())
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        if (it + 1) % check_interval == 0 or it == n_iters - 1:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if verbose and (it + 1) % (check_interval * 10) == 0:
                    print(f"      iter {it+1}: beta={beta:.0f}, score={sc:.8f}")

        a = a_new

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    if verbose:
        print(f"    Mirror-Descent end: score={best_sc:.8f}")

    return best_a


def frank_wolfe_away(a, n_iters=50000, top_m=30, verbose=True, deadline=None,
                     check_interval=5000):
    """Frank-Wolfe with away steps for minimizing max(conv) on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)
    # Track support for away steps
    support = np.zeros(n, dtype=bool)
    support[np.where(a > 1e-10)[0]] = True

    if verbose:
        print(f"    FW-Away start: score={best_sc:.8f}")

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max  # sum(a)=1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and (it + 1) % (check_interval * 2) == 0:
                print(f"      iter {it+1}: score={sc:.8f}, |sup|={support.sum()}")

        # Subgradient from top-M
        top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        # Linear minimization over simplex: argmin sg_j
        j_min = int(np.argmin(sg))

        # Away step: check if we should move away from a support element
        gamma = 2.0 / (it + 2)

        if support.sum() > 1:
            sup_idx = np.where(support)[0]
            # Find the support element with largest sg (most "expensive")
            j_away = sup_idx[np.argmax(sg[sup_idx])]
            a_away = a.copy()
            a_away[j_away] -= gamma * a[j_away]
            a_away = np.maximum(a_away, 0.0)
            S_away = a_away.sum()
            if S_away > 1e-12:
                a_away = a_away / S_away
                c_away = fast_conv(a_away)
                if c_away.max() < c_max:
                    a = a_away
                    support[j_away] = False
                    continue

        # Standard FW step
        a[j_min] += gamma * (1.0 - a[j_min])
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        support[j_min] = True

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    if verbose:
        print(f"    FW-Away end: score={best_sc:.8f}")

    return best_a


def gs_fienup_flat(a, n_iters=500, alpha_init=0.8, alpha_final=0.05, M=None):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_adaptive_target(a, n_iters=400, M=None):
    """GS with adaptive target: use current max as target level, spread energy away from peaks."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        # Target: slightly above mean
        target = c_mean * 1.5 + c_max * 0.5

        b_target = np.zeros(M, dtype=np.float64)
        # Shape the target: flat at 'target' level
        b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
        T = np.fft.fft(b_target)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)

        alpha = 0.3 * (1.0 - it / n_iters) + 0.05

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective_and_grad(x, beta):
    """LSE objective and gradient for L-BFGS (full score including 1/S^2)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + (float(np.log(exp_scaled.sum())) + max_scaled - max_scaled) / beta
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(x[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    # Custom callback for time checking
    call_count = [0]

    func_and_jac = lambda x: lse_objective_and_grad(x, beta)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        func_and_jac, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_topm(a, n_iters=20000, top_m=15, step_init=0.1, decay=0.9999,
                     verbose=True, deadline=None, check_interval=5000):
    """Subgradient method with adaptive step size using top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    if verbose:
        print(f"    SubGrad start: score={best_sc:.8f}")

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and (it + 1) % (check_interval * 2) == 0:
                print(f"      iter {it+1}: score={sc:.8f}")

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step: Polyak step size
        step_polyak = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, step_polyak)
        step = min(step, 0.1)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    if verbose:
        print(f"    SubGrad end: score={best_sc:.8f}")

    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=50, verbose=True, deadline=None):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    if verbose:
        print(f"    CoordDesc start: score={best_sc:.8f}")

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            if deadline is not None and time.monotonic() > deadline:
                break
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                cv = max(0.0, min(1000.0, a_old_j * factor))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = 2.0 * n * fast_conv(a).max() / (S_tmp * S_tmp)
                if sc < cur_sc:
                    cur_sc = sc
                else:
                    a[j] = a_old_j
                    break
            # Reduce overall max by checking if we should increase a[j]
            if a[j] == a_old_j:
                for factor in [1.2, 1.5, 2.0]:
                    cv = max(0.0, min(1000.0, a_old_j * factor))
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old_j
                        continue
                    sc = 2.0 * n * fast_conv(a).max() / (S_tmp * S_tmp)
                    if sc < cur_sc:
                        cur_sc = sc
                    else:
                        a[j] = a_old_j
                        break

        S = a.sum()
        if S > 1e-12:
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    if verbose:
        print(f"    CoordDesc end: score={best_sc:.8f}")

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            p = np.clip(parent.astype(np.float64), 0.0, 1000.0).copy()
            if p.sum() > 1e-6:
                inits.append(p / p.sum())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, parent)
            interp = np.maximum(interp, 0.0)
            if interp.sum() > 1e-6:
                inits.append(interp / interp.sum())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    # Prolate-spheroidal-like: concentrated in center
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    # Bimodal
    inits.append(np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2))
    # Polynomial
    inits.append((x * (1 - x)) ** 2)
    # Random smooth
    freqs = rng.integers(1, 6, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.5, size=5)
    smooth = 0.5 + sum(amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i]) for i in range(5))
    inits.append(np.maximum(smooth, 0.01))

    for init in inits:
        init = np.maximum(init, 0.0)
        if init.sum() > 1e-6:
            init = init / init.sum()
        else:
            init = np.ones(n) / n
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses Adam-LSE with curriculum, GS-Fienup, Frank-Wolfe, and L-BFGS in a hybrid strategy.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) / 6962

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine main length
    lengths = [4096, 6962, 8192]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))

    print(f"Lengths: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: GS-Fienup for global structure (15% time)
    # ============================================================
    phase1_end = t_start + 0.15 * (total_budget - 60.0)
    print(f"\n=== Phase 1: GS-Fienup (deadline +{phase1_end - t_start:.0f}s) ===")

    candidates = []
    for n in lengths[:3]:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}:")
        inits = build_initializations(n, rng, parent)
        for i, a0 in enumerate(inits[:6]):
            if time.monotonic() > phase1_end:
                break
            a1 = gs_fienup_flat(a0, n_iters=300, alpha_init=0.7, alpha_final=0.05)
            sc = compute_score(a1)
            update_best(sc, a1, f"GS n={n} i={i}")
            candidates.append((sc, a1))
            # Second pass with adaptive target
            if time.monotonic() < phase1_end:
                a2 = gs_adaptive_target(a1, n_iters=200)
                sc2 = compute_score(a2)
                update_best(sc2, a2, f"GS-adapt n={n} i={i}")
                candidates.append((sc2, a2))

    # Sort candidates and keep top ones
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:8]]
    print(f"Phase 1 done. Best so far: {best_score:.8f}")

    # ============================================================
    # Phase 2: Adam-LSE with curriculum (35% time)
    # ============================================================
    phase2_end = t_start + 0.50 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Adam-LSE curriculum (deadline +{phase2_end - t_start:.0f}s) ===")

    # Work on the best candidate
    n_best = len(best_a)
    work_a = best_a.copy()

    # Multiple Adam runs with different parameters
    adam_configs = [
        {'beta_start': 10, 'beta_end': 30000, 'n_iters': 150000, 'lr': 0.01},
        {'beta_start': 5, 'beta_end': 10000, 'n_iters': 150000, 'lr': 0.005},
        {'beta_start': 20, 'beta_end': 50000, 'n_iters': 100000, 'lr': 0.02},
        {'beta_start': 10, 'beta_end': 20000, 'n_iters': 100000, 'lr': 0.01, 'beta1': 0.85, 'beta2': 0.999},
    ]

    for cfg_idx, cfg in enumerate(adam_configs):
        if time.monotonic() > phase2_end:
            break
        print(f"  Adam config {cfg_idx}:")
        a_res = adam_lse(work_a, verbose=True, deadline=phase2_end, **cfg)
        sc = compute_score(a_res)
        update_best(sc, a_res, f"Adam cfg={cfg_idx}")
        work_a = a_res

    print(f"Phase 2 done. Best so far: {best_score:.8f}")

    # ============================================================
    # Phase 3: Mirror Descent + more Adam (15% time)
    # ============================================================
    phase3_end = t_start + 0.65 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Mirror Descent + Adam (deadline +{phase3_end - t_start:.0f}s) ===")

    a_md = mirror_descent_lse(best_a, beta_start=5, beta_end=20000,
                              n_iters=80000, lr_start=0.05, lr_end=0.001,
                              verbose=True, deadline=phase3_end)
    sc = compute_score(a_md)
    update_best(sc, a_md, "MirrorDescent")

    if time.monotonic() < phase3_end - 30:
        # Adam from mirror descent result
        a_a2 = adam_lse(a_md, beta_start=100, beta_end=50000,
                        n_iters=80000, lr=0.005, verbose=True, deadline=phase3_end)
        sc = compute_score(a_a2)
        update_best(sc, a_a2, "MD+Adam")

    # Also try Adam from second-best GS candidate
    if len(top_candidates) > 1 and time.monotonic() < phase3_end - 30:
        alt_a = top_candidates[1]
        if len(alt_a) == len(best_a):
            a_alt = adam_lse(alt_a, beta_start=10, beta_end=30000,
                             n_iters=60000, lr=0.01, verbose=True, deadline=phase3_end)
            sc = compute_score(a_alt)
            update_best(sc, a_alt, "Adam-alt-init")

    print(f"Phase 3 done. Best so far: {best_score:.8f}")

    # ============================================================
    # Phase 4: Frank-Wolfe + Subgradient (15% time)
    # ============================================================
    phase4_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 4: FW-Away + SubGrad (deadline +{phase4_end - t_start:.0f}s) ===")

    a_fw = frank_wolfe_away(best_a, n_iters=40000, top_m=30,
                            verbose=True, deadline=phase4_end)
    sc = compute_score(a_fw)
    update_best(sc, a_fw, "FW-Away")

    if time.monotonic() < phase4_end - 30:
        a_sg = subgradient_topm(best_a, n_iters=20000, top_m=10,
                                step_init=0.05, decay=0.9999,
                                verbose=True, deadline=phase4_end)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "SubGrad")

    # Combine: FW then Adam
    if time.monotonic() < phase4_end - 60:
        a_comb = adam_lse(a_fw, beta_start=200, beta_end=30000,
                          n_iters=40000, lr=0.003, verbose=True, deadline=phase4_end)
        sc = compute_score(a_comb)
        update_best(sc, a_comb, "FW+Adam")

    print(f"Phase 4 done. Best so far: {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish (12% time)
    # ============================================================
    phase5_end = t_start + 0.92 * (total_budget - 60.0)
    print(f"\n=== Phase 5: L-BFGS (deadline +{phase5_end - t_start:.0f}s) ===")

    for beta in [5000, 10000, 20000, 50000]:
        if time.monotonic() > phase5_end:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        dt = time.monotonic() - t0
        update_best(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    beta={beta}: score={sc:.8f}, time={dt:.1f}s")

    # L-BFGS from FW result
    if time.monotonic() < phase5_end - 60:
        sc2, a_opt2 = run_lbfgs(a_fw, 20000, maxiter=2000, maxfun=40000)
        update_best(sc2, a_opt2, "FW+LBFGS")

    print(f"Phase 5 done. Best so far: {best_score:.8f}")

    # ============================================================
    # Phase 6: Final refinements (8% time)
    # ============================================================
    print(f"\n=== Phase 6: Final polish ===")

    # Coordinate descent
    a_cd = coordinate_descent(best_a, n_sweeps=3, n_top=80,
                              verbose=True, deadline=deadline)
    sc = compute_score(a_cd)
    update_best(sc, a_cd, "CoordDesc")

    # Final Adam with very high beta
    if time.monotonic() < deadline - 60:
        a_fin = adam_lse(best_a, beta_start=5000, beta_end=100000,
                         n_iters=50000, lr=0.002, verbose=True, deadline=deadline)
        sc = compute_score(a_fin)
        update_best(sc, a_fin, "FinalAdam")

    # Final L-BFGS
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 100000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "FinalLBFGS")

    # One more GS pass
    if time.monotonic() < deadline - 20:
        a_gs_final = gs_fienup_flat(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.02)
        sc = compute_score(a_gs_final)
        update_best(sc, a_gs_final, "FinalGS")

    # ============================================================
    # Multi-length exploration (if time remaining)
    # ============================================================
    if time.monotonic() < deadline - 120:
        print(f"\n=== Multi-length exploration ===")
        for n_try in [5000, 8192, 10000]:
            if time.monotonic() > deadline - 60:
                break
            if n_try == len(best_a):
                continue
            # Interpolate best to new length
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_a)
            a_interp = np.maximum(a_interp, 0.01)
            a_interp = a_interp / a_interp.sum()

            a_try = adam_lse(a_interp, beta_start=10, beta_end=20000,
                             n_iters=50000, lr=0.01, verbose=False,
                             deadline=deadline - 60)
            sc = compute_score(a_try)
            update_best(sc, a_try, f"MultiLen n={n_try}")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
