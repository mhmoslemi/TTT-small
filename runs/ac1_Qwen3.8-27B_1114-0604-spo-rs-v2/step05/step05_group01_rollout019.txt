def active_weights(d, k):

def subgrad_step_with_linesearch(a, k_active, max_steps=50):
    """Subgradient step with backtracking line search."""
    n = len(a)

    for k in [k_active]:

        eta = 0.5

def peak_shave(a, n_peaks=5, strength=0.1):
    """Apply targeted reduction at peak autocorrelation positions."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return normalize_a(a)
    
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    
    # Get top peak positions
    top_k = n_peaks
    idx = np.argpartition(d, -top_k)[-top_k:]
    w = np.zeros_like(d)
    w[idx] = 1.0 / top_k
    
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return normalize_a(a)
    
    # Reduce a at positions that contribute most to peaks
    a_new = a * np.exp(-strength * h / hmax)
    return normalize_a(a_new)

def perturb(a, rng, strength=0.3):
    """Random perturbation of the sequence."""
    n = len(a)
    mode = int(rng.integers(0, 6))

        q = a * np.exp(strength * rng.standard_normal(n))

    elif mode == 2:

        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:

    elif mode == 4:
        # Smoothed random perturbation
        logp = np.log(a + 1e-300)
        noise = rng.standard_normal(n)

        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - logp.max())

        v = 1.0 + strength * 0.5 * np.abs(2.0 * x - 1.0)

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute sum_j a[j]*w[i+j] for each i (cross-correlation with +1 shift)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_obj_grad(a, beta):
    """LSE-smoothed objective with gradient. Objective = 2*max(n*c/S^2)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = max(float(beta), 10.0)
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax < 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)

    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S) - (4.0 * LDS) / S

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_obj_grad(a, float(beta))


def top_k_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def subgrad_step_linesearch(a, k_active, n_trials=40):
    """Subgradient step with exponential backtracking line search."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return exact_score(a), a

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if dmax < 1e-12:
        return exact_score(a), a

    w = top_k_weights(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return exact_score(a), a

    best_sc = exact_score(a)
    best_a = a.copy()
    eta = 0.3
    for _ in range(n_trials):
        a_try = a * np.exp(-eta * h / hmax)
        a_try = np.maximum(a_try, 0.0)
        a_try = normalize_a(a_try)
        sc = exact_score(a_try)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
        eta *= 0.65
        if eta < 1e-7:
            break
    return best_sc, best_a


def subgrad_polish(a, max_iters=40, deadline=None):
    """Iterative subgradient polishing with multiple active set sizes."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    k_list = (1, 2, 3, 5, 8, 12, 20)
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            sc, a_new = subgrad_step_linesearch(a, k, n_trials=35)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
        if not improved:
            break
    return best_sc, best_a

    sc = exact_score(a_out)

        sc = exact_score(a)
    return float(sc), a_out


def peak_shave(a, n_peaks=5, strength=0.1):
    """Reduce sequence values at positions contributing most to autocorrelation peaks."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return normalize_a(a)
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    top_k = min(n_peaks, len(d))
    idx = np.argpartition(d, -top_k)[-top_k:]
    w = np.zeros_like(d)
    w[idx] = 1.0 / float(top_k)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return normalize_a(a)
    a_new = a * np.exp(-strength * h / hmax)
    return normalize_a(a_new)

            sig = max(1.0, c / 2.0)
