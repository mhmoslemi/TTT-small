Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-length sweeping, diverse non-palindromic initializations,
progressive LSE-LBFGS optimization with Nesterov-accelerated subgradient polish.
Key innovations: one-sided (anti-palindromic) initializations to reduce center
convolution, length adaptation, and Nesterov acceleration for faster convergence.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad

def _nesterov_subgrad(a, max_iters=50, deadline=None):
    """Nesterov-accelerated subgradient descent on the max-conv objective."""
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Nesterov momentum
    x = a.copy()  # current
    y = a.copy()  # lookahead (Nesterov point)
    mu = 1.0  # momentum parameter
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
            
        # Compute subgradient at y (the Nesterov point)
        S = float(y.sum())
        if S < 1e-12:
            break
        c = _self_conv(y)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        
        # Active set
        active = d >= dmax - 1e-6 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(y, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Nesterov step
        t_k = 2.0 / (it + 2.0 + mu)  # or (1+sqrt(1+4*mu*mu))/2 variant
        # Standard Nesterov: beta_k = (1 - sqrt(1 + 4/mu)) / 2 ... 
        # Simpler: beta_k = (k-1)/(k+1+mu)
        beta_k = (it) / (it + 1.0 + mu) if it > 0 else 0.0
        
        # Step size: L-smoothness based or fixed
        # Use polyak-like: step = (f(y) - f*) / ||g||^2, but f* unknown
        # Use fixed decay: eta = c / (it + 1)
        eta = 0.5 / (gnorm * (1.0 + 0.1 * it))
        
        # Gradient step from y
        x_new = y - eta * grad
        x_new = np.clip(x_new, 0.0, None)
        x_new = _norm(x_new)
        
        # Nesterov extrapolation
        y = x_new + beta_k * (x_new - x)
        y = np.clip(y, 0.0, None)
        y = _norm(y)
        
        x = x_new
        
        sc = _score(x)
        if sc < best_sc:
            best_sc = sc
            best_a = x.copy()
    
    return best_sc, best_a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: multi-length scan with diverse non-palindromic initializations,
progressive LSE-LBFGS optimization, Nesterov-accelerated subgradient polish,
and large-perturbation restarts to escape local minima.
Key innovations: one-sided exponential/bump initializations to reduce center
convolution peak, adaptive length selection, and combined LSE+Nesterov polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 2.0:
        return _score(a), a
    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.80 * time_limit / max(est_eval_t, 1e-4))
    mf = max(20, min(mf, maxfun_cap, 6000))
    maxiter = max(15, mf // 3)
    PEN = 5e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _nesterov_subgrad(a, max_iters=40, deadline=None):
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    x = a.copy()
    y = a.copy()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        Sy = float(y.sum())
        if Sy < 1e-12:
            break
        c = _self_conv(y)
        d = float(n) * c / (Sy * Sy)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(y, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (Sy * Sy) - 4.0 * LDS / Sy
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        beta_k = float(it) / (it + 1.0 + 1.0) if it > 0 else 0.0
        best_step = None
        best_new_sc = best_sc
        for scale in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            eta = scale / gnorm
            x_new = y - eta * grad
            x_new = np.clip(x_new, 0.0, None)
            Sx = x_new.sum()
            if Sx < 1e-8:
                continue
            x_new = x_new / Sx
            sc = _score(x_new)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = x_new.copy()
        if best_step is None:
            break
        y_new = best_step + beta_k * (best_step - x)
        y_new = np.clip(y_new, 0.0, None)
        Sy2 = y_new.sum()
        if Sy2 < 1e-8:
            y_new = best_step
        else:
            y_new = y_new / Sy2
        x = best_step
        y = y_new
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
    return best_sc, best_a


def _plain_subgrad(a, max_iters=30, deadline=None):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0005, 0.002, 0.008, 0.02, 0.08, 0.2):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    return best_sc, best_a


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _generate_inits(n, parent, rng):
    inits = []
    x = np.arange(n)
    # Parent resampled
    if parent is not None:
        inits.append(('parent', _upsample(parent, n)))
        inits.append(('parent_rev', _upsample(parent[::-1], n)))
        # Parent + noise
        p = _upsample(parent, n)
        for amp in (0.05, 0.1, 0.2, 0.4):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))
    # One-sided exponential (non-palindromic, center of conv is small)
    for rate in (0.5, 1.0, 2.0, 3.0):
        a = np.exp(-rate * x / n)
        inits.append((f'expR{rate}', a / a.sum()))
    # Bump in first quarter
    for center_frac in (0.15, 0.25, 0.35):
        for width in (0.05, 0.1, 0.2):
            mu = center_frac * n
            sig = width * n
            a = np.exp(-(x - mu) ** 2 / (2.0 * sig * sig))
            inits.append((f'bump{center_frac}_{width}', a / a.sum()))
    # Power law
    for p in (0.5, 1.0, 2.0, 3.0):
        a = (x + 1) ** p
        inits.append((f'pow{p}', a / a.sum()))
    # Cosine modulations
    xt = np.linspace(0, 1, n)
    for f in (1, 2, 3, 5):
        for amp in (0.2, 0.5, 0.8):
            a = 1.0 + amp * np.cos(2.0 * np.pi * f * xt)
            a = np.clip(a, 0.01, None)
            inits.append((f'cos{f}_{amp}', a / a.sum()))
    # Spectral init
    for i in range(4):
        Nf = _next_pow2(2 * n - 1)
        freqs = np.fft.rfftfreq(Nf) * Nf
        exponent = rng.uniform(0.3, 2.0)
        power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - sig.min()
        if sig.max() <= 0:
            sig = np.ones(n) / n
        else:
            sig = sig / sig.sum()
        inits.append((f'spec{i}', sig))
    # Random smooth (log-normal)
    for i in range(3):
        noise = rng.standard_normal(n)
        k = max(5, n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * max(1, k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'ln{k}_{i}', a / a.sum()))
    # "Sawtooth" (very non-palindromic)
    a = np.linspace(0.01, 1.0, n)
    inits.append(('sawtooth', a / a.sum()))
    # "Anti-sawtooth"
    a = np.linspace(1.0, 0.01, n)
    inits.append(('antiSaw', a / a.sum()))
    return inits


def _optimize_trajectory(a0, deadline, label, rng):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    total_time = max(5.0, deadline - t0 - 5.0)
    print(f"  [{label}] start: score={best_sc:.8f}, n={n}, budget={total_time:.0f}s")

    stages = [
        (3, 50.0, 0.05),
        (5, 100.0, 0.07),
        (8, 200.0, 0.09),
        (15, 400.0, 0.10),
        (25, 800.0, 0.12),
        (50, 1600.0, 0.14),
        (100, 3200.0, 0.16),
        (200, 6400.0, 0.15),
        (400, 12800.0, 0.12),
    ]
    cum_frac = 0.0
    for k, beta, frac in stages:
        cum_frac += frac
        stage_end = t0 + cum_frac * total_time
        stage_end = min(stage_end, deadline - 10.0)
        remaining = stage_end - time.monotonic()
        if remaining < 5.0:
            break
        n_calls = min(3, max(1, int(remaining / 25)))
        per_call = remaining / n_calls
        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 400)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k},b={beta:.0f}: {best_sc:.8f}")
            a = _norm(a_new)
            # Occasional perturbation to escape local minima
            if ci < n_calls - 1 and rng.random() < 0.3:
                a = _smooth_perturb(a, rng, strength=0.05)
    # Nesterov subgradient polish
    if time.monotonic() < deadline - 15.0:
        sc_ng, a_ng = _nesterov_subgrad(a, max_iters=20, deadline=deadline - 10.0)
        if sc_ng < best_sc:
            best_sc = sc_ng
            best_a = a_ng.copy()
            print(f"    [{label}] nesterov: {best_sc:.8f}")
    # Plain subgradient
    if time.monotonic() < deadline - 5.0:
        sc_pg, a_pg = _plain_subgrad(best_a, max_iters=10, deadline=deadline - 2.0)
        if sc_pg < best_sc:
            best_sc = sc_pg
            best_a = a_pg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")
    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a

            if np.isfinite(sc):
                scored.append((sc, name, arr))
        scored.sort(key=lambda x: x[0])
        print(f"  Screened {len(scored)} inits. Top: " +
              ", ".join(f"{nm}:{sc:.6f}" for sc, nm, _ in scored[:5]))

        # Select top diverse candidates (max 4)
        candidates = scored[:min(4, len(scored))]
        n_cands = len(candidates)
        per_cand_time = (length_end - time.monotonic() - 30) / max(1, n_cand_time if False else n_cands)
        per_cand_time = (length_end - time.monotonic() - 30) / max(1, n_cands)

        for ci, (sc_init, name, arr_init) in enumerate(candidates):
            cand_end = time.monotonic() + per_cand_time
            cand_end = min(cand_end, length_end - 30)
            if time.monotonic() > cand_end - 10:
                continue
            print(f"\n  --- Candidate {ci+1}/{n_cands}: '{name}', score={sc_init:.8f} ---")

            # Multilevel: if n > 4096, try small level first
            a_work = _norm(arr_init)
            if n > 4096 and (cand_end - time.monotonic()) > 120:
                n_small = max(512, n // 4)
                a_small = _upsample(a_work, n_small)
                # Quick optimize small
                for k_s, b_s in [(5, 100), (20, 500)]:
                    if time.monotonic() > cand_end - 100:
                        break
                    sc_s, a_s = _run_lbfgs(a_small, b_s, k_s, 20, 100)
                    a_small = a_s
                a_up = _norm(_upsample(a_small, n))
                sc_up = _score(a_up)
                if sc_up < _score(a_work):
                    a_work = a_up
                    print(f"    Multilevel improved: {sc_up:.8f}")

            sc_traj, seq_traj = _optimize_trajectory(a_work, cand_end, f"L{n}_{name}", rng)
            if sc_traj < best_score:
                best_score = sc_traj
                best_seq = seq_traj.copy()
                print(f"  *** Global best: {best_score:.8f} (n={n}, {name}) ***")

    # Final polish
    if best_seq is not None:
        print(f"\n{'='*60}")
        print(f"=== FINAL POLISH: score={best_score:.8f}, n={len(best_seq)} ===")
        print(f"{'='*60}")
        sc_final, seq_final = _final_polish(best_seq, deadline - 5, rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = seq_final.copy()

    # Prepare output
    if best_seq is None:
        best_seq = np.ones(8192) / 8192

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
