    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _fienup_flat(a, n_iters, relaxation=0.7, deadline=None):
    """
    Fienup/Gerchberg-Saxton style iterations to flatten the auto-convolution.
    Alternates between time-domain (non-negativity, normalization) and
    frequency-domain (push |A|^2 toward flat) projections.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        # Forward FFT
        A = np.fft.rfft(a, N)
        abs_A = np.abs(A)

        # Target: make |A|^2 flat (geometric mean of |A|^2)
        log_abs = np.log(np.maximum(abs_A, 1e-30))
        geo_mean = np.exp(np.mean(log_abs))
        # Blend: push toward geometric mean
        new_abs = (1.0 - relaxation) * abs_A + relaxation * geo_mean
        new_abs[0] = abs_A[0]  # Preserve DC = sum(a)

        # Update spectrum
        phase = A / np.maximum(abs_A, 1e-30)
        A_new = new_abs * phase

        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.clip(a_new, 0.0, None)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_sc, best_a


def _fienup_loopless(a1, a2, n_iters, martinet=1.1, deadline=None):
    """
    Loopless spectral phase retrieval for flat auto-convolution.
    Uses two iterates with overshoot for faster convergence.
    """
    n = len(a1)
    N = _next_pow2(2 * n - 1)
    best_a = _norm(a1.copy())
    best_sc = _score(best_a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        A1 = np.fft.rfft(a1, N)
        A2 = np.fft.rfft(a2, N)
        # Interpolate in frequency
        A_mid = 0.5 * (A1 + A2)
        # Push toward flat magnitude
        abs_mid = np.abs(A_mid)
        log_abs = np.log(np.maximum(abs_mid, 1e-30))
        geo_mean = np.exp(np.mean(log_abs))
        new_abs = 0.7 * abs_mid + 0.3 * geo_mean
        new_abs[0] = abs_mid[0]
        phase = A_mid / np.maximum(abs_mid, 1e-30)
        A_new = new_abs * phase
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.clip(a_new, 0.0, None)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S

        # Marten's overshoot update
        a1, a2 = a2, a_new + martinet * (a_new - a1)
        a2 = np.clip(a2, 0.0, None)
        S2 = a2.sum()
        if S2 > 1e-12:
            a2 = a2 / S2

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_sc, best_a


def _lse_obj_grad(a, beta, k):
    """
    LSE-smoothed objective using top-k positions.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    # grad of 2*L w.r.t. a:
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=8000):
    """Run L-BFGS-B on active-set LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap))
    maxiter = max(20, mf // 3)
    PEN = 1e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=50, deadline=None, momentum_beta=0.7):
    """
    Enhanced subgradient descent with Polyak step size and momentum.
    Uses the theoretical lower bound f* >= 2n/(2n-1) for step sizing.
    """
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    f_star = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    velocity = np.zeros(n)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        sc = 2.0 * dmax
        if dmax <= 0:
            break

        # Active set: positions within small tolerance of max
        tol = max(1e-6 * dmax, 1e-10)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        # Use exponential weighting for smoother active set
        z = (d[active] - dmax) / max(1e-10, tol)
        w_act = np.exp(z * 5.0)
        w_act /= w_act.sum()

        w = np.zeros_like(d)
        w[active] = w_act

        corr = _cross_corr(a, w)
        LDS = float(w_act @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        # Remove mean component (since we're on sum=1 constraint)
        grad -= np.mean(grad)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Polyak step size with damping
        gap = max(sc - f_star, 1e-10)
        eta_polyak = gap / (gnorm * gnorm)
        eta = min(eta_polyak, 5.0 / gnorm)

        # Momentum
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        vnorm = float(np.linalg.norm(velocity))
        if vnorm < 1e-14:
            break
        step = eta * vnorm / gnorm  # scale to match gradient direction

        # Try multiple step sizes
        best_step_a = None
        best_step_sc = sc
        for scale in (0.1, 0.3, 0.5, 0.8, 1.0, 1.5):
            q = a - scale * eta * velocity / vnorm * gnorm
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc_q = _score(q)
            if sc_q < best_step_sc:
                best_step_sc = sc_q
                best_step_a = q.copy()

        if best_step_a is None:
            # Fallback: small plain subgradient step
            q = a - 0.01 * eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq > 1e-8:
                q = q / Sq
                sc_q = _score(q)
                if sc_q < best_sc:
                    best_sc = sc_q
                    best_a = q.copy()
            break

        a = best_step_a
        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = a.copy()

    return best_sc, best_a


def _subgrad_exact(a, max_iters=30, deadline=None):
    """Exact subgradient with uniform active set and careful step sizes."""
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        sc = 2.0 * dmax
        if dmax <= 0:
            break

        # Exact active set
        tol = 1e-8 * max(dmax, 1.0)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = 1.0  # since w is uniform over active set, LDS = dmax
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * dmax / S
        grad = np.nan_to_num(grad, nan=0.0)
        grad -= np.mean(grad)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_new_a = None
        best_new_sc = sc
        for scale in (0.0001, 0.0005, 0.002, 0.005, 0.01, 0.02, 0.05):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc_q = _score(q)
            if sc_q < best_new_sc:
                best_new_sc = sc_q
                best_new_a = q.copy()

        if best_new_a is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
            a = best_new_a
        else:
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(0, 2 * np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _u_shape_init(n, rng):
    """U-shaped initialization - concentrates energy at edges."""
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.5, 2.0)
    a = np.cos(np.pi * x) ** 2 + alpha
    a = np.clip(a, 0, None)
    a = a / a.sum()
    noise = rng.standard_normal(n) * 0.05
    a = np.clip(a + noise, 0, None)
    return a / a.sum()


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(51, max(5, n // 100))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_full(a0, deadline, label, rng):
    """Full optimization pipeline: Fienup -> LSE progressive -> subgradient."""
    a = _norm(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    total_budget = max(10.0, deadline - t0 - 5.0)

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Phase 1: Fienup warm-start (fast, O(n log n) per iter)
    fienup_budget = total_budget * 0.15
    fienup_deadline = t0 + fienup_budget
    n_fienup = min(500, int(fienup_budget / (n * 0.00002 + 0.005)))
    n_fienup = max(20, n_fienup)
    sc_f, a_f = _fienup_flat(a, n_fienup, relaxation=0.5, deadline=fienup_deadline)
    if sc_f < best_sc:
        best_sc = sc_f
        best_a = a_f.copy()
        a = a_f
        print(f"    Fienup: {best_sc:.8f}")

    # Phase 2: Progressive LSE optimization
    lse_deadline = t0 + total_budget * 0.75
    stages = [
        (5, 50.0, 0.10),
        (10, 150.0, 0.12),
        (20, 400.0, 0.14),
        (50, 1000.0, 0.14),
        (100, 2500.0, 0.14),
        (200, 5000.0, 0.14),
        (400, 10000.0, 0.14),
        (800, 20000.0, 0.14),
    ]

    for k, beta, frac in stages:
        if time.monotonic() > lse_deadline - 10:
            break
        stage_budget = frac * total_budget * 1.2
        stage_budget = min(stage_budget, lse_deadline - time.monotonic() - 5)
        if stage_budget < 5:
            break

        n_calls = min(3, max(1, int(stage_budget / 15)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > lse_deadline - 5:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 500)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k}, beta={beta:.0f}: {best_sc:.8f}")
            a = _norm(a_new)

    # Phase 3: Subgradient polishing
    if time.monotonic() < deadline - 10:
        sg_deadline = deadline - 5
        sc_sg, a_sg = _subgrad_polish(a, max_iters=40, deadline=sg_deadline)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad_momentum: {best_sc:.8f}")
        a = _norm(a_sg)

        sc_sg2, a_sg2 = _subgrad_exact(a, max_iters=30, deadline=sg_deadline)
        if sc_sg2 < best_sc:
            best_sc = sc_sg2
            best_a = a_sg2.copy()
            print(f"    [{label}] subgrad_exact: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: Fienup + progressive LSE + Polyak subgradient, multi-length."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent loaded: {parent is not None}, budget={budget_s}s")

    # Determine lengths to try
    if parent is not None:
        parent_n = len(parent)
    else:
        parent_n = 4096

    # Strategy: primarily optimize at parent length, also try 2x
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(parent_n)
        if parent_n <= 8192:
            lengths_to_try.append(min(16384, parent_n * 2))
    else:
        lengths_to_try = [8192]

    # Time allocation: 70% to primary length, 30% to secondary (if exists)
    best_score = np.inf
    best_seq = None

    # Generate candidates for each length
    all_candidates = []

    for n_target in lengths_to_try:
        cands = []

        # Parent based
        if parent is not None and n_target == len(parent):
            cands.append(('parent', parent.copy()))
            cands.append(('parent_rev', parent[::-1].copy()))
            for amp in (0.05, 0.1, 0.2, 0.4):
                cands.append((f'pn{amp}', _smooth_perturb(parent, rng, amp)))

        # Upsampled parent
        if parent is not None and n_target != len(parent):
            cands.append(('parent_up', _upsample(parent, n_target)))
            cands.append(('parent_up_rev', _upsample(parent[::-1], n_target)))

        # Spectral inits
        for i in range(4):
            cands.append((f'spec{i}', _spectral_init(n_target, rng)))

        # U-shape inits
        for i in range(3):
            cands.append((f'u{i}', _u_shape_init(n_target, rng)))

        # Smooth noise
        for i in range(2):
            noise = rng.standard_normal(n_target)
            k = max(7, n_target // 80)
            if k % 2 == 0:
                k += 1
            kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * (k // 4 + 1) ** 2))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode='same')
            sn = np.exp(noise)
            cands.append((f'noise{i}', sn / sn.sum()))

        # Cosine modulations
        x = np.linspace(0, 1, n_target)
        for f_rep in (1, 2, 3, 5):
            for amp in (0.2, 0.4, 0.7):
                cc = 1.0 + amp * np.cos(2 * np.pi * f_rep * x)
                cc = np.clip(cc, 0, None)
                if cc.sum() > 0:
                    cands.append((f'cos{f_rep}_{amp}', cc / cc.sum()))

        # Evaluate and select top candidates
        scored = []
        for name, arr in cands:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr, n_target))

        scored.sort(key=lambda x: x[0])
        print(f"\nLength {n_target}: {len(scored)} candidates. "
              f"Top: {[(name, f'{sc:.6f}') for sc, name, _, _ in scored[:5]]}")

        # Select top 3 diverse (by structure)
        selected = []
        for sc, name, arr, nt in scored:
            if len(selected) >= 3:
                break
            selected.append((sc, name, arr))
        all_candidates.extend(selected)

    if not all_candidates:
        n_fallback = lengths_to_try[0] if lengths_to_try else 4096
        all_candidates = [(2.0, 'uniform', _norm(np.ones(n_fallback)))]

    # Sort all candidates by score
    all_candidates.sort(key=lambda x: x[0])
    best_score = all_candidates[0][0]
    best_seq = all_candidates[0][2].copy()

    print(f"\nBest initial: {all_candidates[0][1]} with score={best_score:.8f}")

    # Main optimization: allocate time to top candidates
    # Top candidate gets 50%, second gets 25%, third gets 15%, rest share 10%
    main_deadline = deadline - 200.0  # reserve 200s for final polish
    main_budget = main_deadline - t_start

    n_cands = min(3, len(all_candidates))
    fractions = [0.50, 0.30, 0.20][:n_cands]

    for ci in range(n_cands):
        sc_init, name, arr_init = all_candidates[ci]
        n = len(arr_init)
        sub_deadline = t_start + (sum(fractions[:ci + 1]) * main_budget) + 20
        sub_deadline = min(sub_deadline, main_deadline)

        if time.monotonic() > sub_deadline - 10:
            continue

        print(f"\n=== Main candidate {ci+1}: '{name}', n={n}, score={sc_init:.8f} ===")

        sc_opt, seq_opt = _optimize_full(arr_init, sub_deadline, name, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"  *** Global best: {best_score:.8f} ***")

    # Final polish phase
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0
    remaining = final_deadline - time.monotonic()

    if remaining > 30:
        # Very high beta LSE
        n_best = len(best_seq)
        for k_fin, beta_fin in [(300, 15000), (600, 30000), (1200, 60000)]:
            if time.monotonic() > final_deadline - 20:
                break
            chunk = min(30.0, (final_deadline - time.monotonic() - 20) * 0.3)
            if chunk < 5:
                break
            sc_f, a_f = _run_lbfgs(best_seq, beta_fin, k_fin, chunk, 400)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final LSE K={k_fin}, beta={beta_fin}: {best_score:.8f}")

        # Multi-round subgradient
        for round_i in range(3):
            if time.monotonic() > final_deadline - 10:
                break
            chunk_dl = min(final_deadline - 10, time.monotonic() + remaining * 0.15)
            sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=30, deadline=chunk_dl)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  Final SG round {round_i}: {best_score:.8f}")

            sc_sg2, a_sg2 = _subgrad_exact(_norm(a_sg), max_iters=20,
                                           deadline=min(final_deadline - 5, time.monotonic() + 10))
            if sc_sg2 < best_score:
                best_score = sc_sg2
                best_seq = a_sg2.copy()
                print(f"  Final SG exact round {round_i}: {best_score:.8f}")

        # Random restarts
        if time.monotonic() < final_deadline - 30:
            for ri in range(4):
                if time.monotonic() > final_deadline - 15:
                    break
                p = _smooth_perturb(best_seq, rng, strength=0.05 + 0.05 * ri)
                # Quick Fienup
                sc_fr, a_fr = _fienup_flat(p, 50, relaxation=0.4,
                                           deadline=time.monotonic() + 10)
                # Quick LSE
                chunk = min(15.0, final_deadline - time.monotonic() - 15)
                if chunk < 5:
                    break
                sc_r, a_r = _run_lbfgs(a_fr, 10000, 400, chunk, 300)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  Restart {ri}: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
