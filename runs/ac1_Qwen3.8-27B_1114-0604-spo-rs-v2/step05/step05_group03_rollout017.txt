def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):

    if S < 1e-12:

    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp_cvxpy(a, M):

    n = len(a)
    x = cp.Variable(n)

    first_col = np.zeros(2*n - 1)

    if x.value is None:

    return x_val / S

"""
Multi-phase optimizer with exact LP solving (HiGHS) for flat-autocorrelation
non-negative sequences. Key improvements over previous approach:
1. Exact LP solving via SciPy HiGHS instead of FISTA approximation
2. Golden section line search for mixing parameter
3. Adaptive diversification with perturbation-based restarts
4. Progressive L-BFGS polish with high beta
5. Peak-targeted final refinement
6. Frequency-domain error reduction for initialization
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix_sparse(a):
    """Build sparse matrix B where (B @ x) = np.convolve(a, x, 'full')."""
    n = len(a)
    m = 2 * n - 1
    n_entries = n * n
    rows = np.zeros(n_entries, dtype=np.int32)
    cols = np.zeros(n_entries, dtype=np.int32)
    vals = np.zeros(n_entries, dtype=np.float64)
    idx = 0
    for j in range(n):
        r_start = j
        r_end = min(j + n, m)
        length = r_end - r_start
        rows[idx:idx+length] = range(r_start, r_end)
        cols[idx:idx+length] = j
        vals[idx:idx+length] = a[:length]
        idx += length
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def solve_lp_highs(a, M, timeout=120.0):
    """Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using HiGHS."""
    n = len(a)
    B = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    A_ub = B
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    elapsed = time.monotonic() - t0
    if res.success and res.x is not None:
        x = np.maximum(res.x, 0.0)
        S = float(x.sum())
        if S < 1e-12:
            return np.ones(n) / n, elapsed
        return x / S, elapsed
    return np.ones(n) / n, elapsed


def solve_lp_fista(a, M, n_fista_iters=20000, beta_end=500.0, x_init=None):
    """Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0 using FISTA."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_fista_iters):
        if it < 5000:
            beta = 10.0 + (beta_end - 10.0) * (it / 5000.0)
        else:
            beta = beta_end

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(c) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    S = float(x.sum())
    if S < 1e-12:
        return np.ones(n) / n
    return x / S


def golden_section_line_search(a, x, best_sc, n_evals=30):
    """Golden section search for best mixing parameter t."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi
    inv_phi2 = inv_phi * inv_phi
    lo = 0.0
    hi = 1.0
    c_pt = hi - inv_phi * (hi - lo)
    d_pt = lo + inv_phi * (hi - lo)

    def sc_at(t):
        a_try = (1.0 - t) * a + t * x
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            return np.inf
        a_try = a_try / St
        return compute_score(a_try)

    sc_c = sc_at(c_pt)
    sc_d = sc_at(d_pt)
    best_t = c_pt if sc_c < sc_d else d_pt
    best_sc_ls = min(sc_c, sc_d)

    for _ in range(n_evals - 2):
        if sc_c < sc_d:
            hi = d_pt
            d_pt = c_pt
            sc_d = sc_c
            c_pt = hi - inv_phi * (hi - lo)
            sc_c = sc_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            sc_c = sc_d
            d_pt = lo + inv_phi * (hi - lo)
            sc_d = sc_at(d_pt)
        cur_best = min(sc_c, sc_d)
        if cur_best < best_sc_ls:
            best_sc_ls = cur_best
            best_t = c_pt if sc_c < sc_d else d_pt

    if best_sc_ls < best_sc:
        a_try = (1.0 - best_t) * a + best_t * x
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St > 1e-12:
            a_try = a_try / St
            return a_try, best_sc_ls, best_t
    return a, best_sc, 0.0


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=100, alpha=0.75):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break
        conv_mod = conv - 0.85 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_local_search(a, n_steps=300, rng=None):
    """Targeted local search that specifically reduces the convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.05, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3)
        improved = False
        for mult in [0.2, 0.4, 0.6, 1.0, 1.5, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try random perturbation
            if rng.random() < 0.5:
                a_try = a * (1.0 + 0.01 * rng.standard_normal(n))
                a_try = np.maximum(a_try, 0.0)
                S_try = float(a_try.sum())
                if S_try > 1e-12:
                    a_try = a_try / S_try
                    sc_try = compute_score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try

    return best_a

        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())

        if use_highs:
            x, lp_time = solve_lp_highs(a, M, timeout=120.0)
        else:
            x = solve_lp_fista(a, M, n_fista_iters=fista_iters)
            lp_time = 0.0

        if line_search:
            a_new, sc_new, t_used = golden_section_line_search(
                a, x, best_sc, n_evals=25)
        else:
            t = 0.5
            a_new = (1.0 - t) * a + t * x
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)

        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 8:
                # Diversify
                a_new = a * (1.0 + 0.03 * rng.standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                S_new = float(a_new.sum())
                if S_new > 1e-12:
                    a_new = a_new / S_new
                no_improve_count = 0
                continue

        a = a_new

    return best_a, best_sc


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: error reduction, LP alternating (HiGHS), L-BFGS polish,
    peak local search.
    """
    rng = np.random.default_rng(123)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def check_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        check_update(ps, parent, "parent")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    if 6962 not in lengths:
        lengths.append(6962)
    if 8192 not in lengths:
        lengths.append(8192)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Quick global search with error reduction (10% time)
    # ==================================================================
    phase1_deadline = t_start + 0.10 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Global Search (Error Reduction) ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 15:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, min(len(inits), int(remaining / 10.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=60, alpha=0.75)
            sc = compute_score(a_er)
            check_update(sc, a_er, f"ER n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP-based alternating method (58% time) - MAIN PHASE
    # ==================================================================
    phase2_deadline = t_start + 0.68 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LP-Based Alternating (Main) ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # First: try HiGHS LP for a few iterations
    print(f"  Using HiGHS LP, n={n_main}")
    a_out, sc_out = lp_alternating_improved(
        a_work, max_outer_iters=20, time_limit=phase2_deadline - 60,
        use_highs=True, line_search=True, rng=rng)
    sc = compute_score(a_out)
    check_update(sc, a_out, "HiGHS LP")
    a_work = a_out.copy()

    # Continue with FISTA-based LP if time permits
    if time.monotonic() < phase2_deadline - 60:
        print(f"  Continuing with FISTA LP...")
        a_out, sc_out = lp_alternating_improved(
            a_work, max_outer_iters=40, time_limit=phase2_deadline - 30,
            use_highs=False, fista_iters=25000, line_search=True, rng=rng)
        sc = compute_score(a_out)
        check_update(sc, a_out, "FISTA LP")
        a_work = a_out.copy()

    # Try another length if time permits
    if time.monotonic() < phase2_deadline - 120:
        for n_alt in lengths:
            if n_alt == n_main:
                continue
            if time.monotonic() > phase2_deadline - 60:
                break
            print(f"  Trying alternative length n={n_alt}")
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            S_alt = float(a_alt.sum())
            if S_alt > 1e-12:
                a_alt = a_alt / S_alt
            a_alt, sc_alt = lp_alternating_improved(
                a_alt, max_outer_iters=15, time_limit=phase2_deadline - 30,
                use_highs=True, line_search=True, rng=rng)
            sc = compute_score(a_alt)
            check_update(sc, a_alt, f"LP n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (17% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [30, 80, 200, 500, 1000, 2000, 4000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=8000, maxfun=400000)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        check_update(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Local search + final polish (remaining time)
    # ==================================================================
    print(f"\n--- Phase 4: Local Search & Polish ---")

    # Peak-specific local search
    a_peak = peak_local_search(best_a, n_steps=400, rng=rng)
    sc = compute_score(a_peak)
    check_update(sc, a_peak, "peak search")

    # Another round of HiGHS LP if time permits
    if time.monotonic() < deadline - 180:
        print("  Final HiGHS LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating_improved(
            a_final_in, max_outer_iters=10, time_limit=deadline - 120,
            use_highs=True, line_search=True, rng=rng)
        sc = compute_score(a_final)
        check_update(sc, a_final, "final HiGHS LP")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 60:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 8000, maxiter=15000, maxfun=800000)
        check_update(sc, a_opt, "final LBFGS b=8000")

    # Final peak search
    if time.monotonic() < deadline - 15:
        a_final = peak_local_search(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_final)
        check_update(sc, a_final, "final peak")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
