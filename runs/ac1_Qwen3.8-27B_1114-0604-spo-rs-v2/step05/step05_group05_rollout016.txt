This is a "semi-infinite" optimization problem (infinitely many constraints if we consider continuous `k`). But in practice, `k` is discrete (0 to 2n-2), so it's a finite-dimensional problem with O(n) constraints and O(n) variables.

The "dual" problem is: maximize `-t * sum(lambda) - nu` subject to certain constraints. This is a QP (quadratic program) in the dual variables.

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LSE smoothing with very high beta, L-BFGS polish,
and adaptive coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def fienup_momentum(a, n_iters=500, momentum_init=0.0, momentum_end=0.5,
                    thresh_init=3.0, thresh_end=1.0, clip_init=0.2, clip_end=0.9):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.001, 1.0)
            scale[above] = factor
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def nesterov_lse_adaptive(a, max_time=100.0, beta_start=50.0, beta_end=100000.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.01
    lr = 0.01
    lr_decay = 0.9995
    mu = 0.9
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.00002 * it)
        lr *= lr_decay
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        sc_actual = compute_score(a) if it % 50 == 0 else best_sc
        lb = 2.0 * n / (2.0 * n - 1.0)
        max_step = max(0.005, (sc_actual - lb) * 0.3)
        effective_lr = min(lr, max_step / (gnorm + 1e-10))
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        beta = min(beta * beta_growth, beta_end)
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    def objective(x):
        return lse_score_and_grad(x, beta)
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_coordinate_descent(a, n_sweeps=3, n_top=50, max_coords=500):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        # Collect all coordinates that contribute to top peaks
        coord_set = set()
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                if a[j] > 1e-8:
                    coord_set.add(j)
        # Limit to max_coords
        coords = sorted(coord_set)[:max_coords]
        for j in coords:
            old_val = a[j]
            if old_val < 1e-10:
                continue
            for factor in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                a[j] = old_val * factor
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = old_val
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_scaled.copy()
                    a = a_scaled
                    break
            else:
                a[j] = old_val
    return best_a

def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)

def build_chirp_init(n):
    x = np.arange(n, dtype=np.float64) / n
    # Superposition of chirps at different rates
    a = np.zeros(n)
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a += np.abs(np.sin(np.pi * rate * x * x))
    a += 0.1
    return a / a.sum()

def build_gaussian_init(n):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()

def build_mixed_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(5 * np.pi * x) + 0.05 * np.cos(11 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()

Let me also think about whether there's a fundamentally different approach that could give a bigger improvement.

**Alternative: Use a "circular" formulation**

If we zero-pad `a` to length `N = 2n-1` (or a power of 2 >= 2n-1), then the circular autocorrelation of the zero-padded sequence is exactly the linear self-convolution. We can then use any "flat spectrum" algorithm designed for circular signals.

The "Optimum" algorithm (from the PAPR literature) works as follows:
1. Start with `a` (zero-padded to length N).
2. Compute the DFT: `A = DFT(a)`.
3. The power spectrum is `P = |A|^2`.
4. We want to minimize `max|r[k]|` where `r = IDFT(P)`.
5. This is equivalent to minimizing `max|A|^2` in the "time domain" of the spectrum... wait, that's not right.

Actually, the circular autocorrelation `r[k] = IDFT(|A|^2)[k]` and we want to minimize `max_k r[k]`. Since `r[0] = sum|A|^2 = N * mean(|A|^2)`, the "DC" term of the autocorrelation is fixed (it's the total power). We want to minimize the "side lobes".

This is the "PAPR" problem: minimize the peak of the time-domain signal (which is the autocorrelation in this case) given a constraint on the frequency-domain signal (the power spectrum).

The standard PROBER algorithm for this is:
1. Compute `r = IDFT(|A|^2)`.
2. Find peaks in `r` where `r[k] > threshold`.
3. Modify `A` to reduce these peaks: set `A[m] = A[m] * (1 - alpha * w[m])` where `w` is a "window" that targets the peaks.
4. Repeat.

But this is essentially what Fienup does! The Fienup algorithm IS a variant of PROBER.

So I think the key to improvement is:
1. **More aggressive peak reduction** (lower threshold, higher clip).
2. **Better momentum scheduling**.
3. **More iterations**.
4. **Higher LSE beta for the final polish**.

Let me also consider: **What if we use a "dual" Fienup that works on the autocorrelation directly?**

Instead of modifying the frequency domain (Fienup), we could modify the time domain directly:
1. Compute `b = conv(a, a)`.
2. Find the peak `k*`.
3. Reduce the contribution to `b[k*]` by decreasing some `a[i]` values.
4. The reduction should be "spread out" to minimize the increase in other `b[k]` values.

This is a "minimize the max" problem in the time domain. We can solve it with a "water-filling" approach:
1. Identify the peak `k*`.
2. The peak value is `b[k*] = sum_i a[i] * a[k*-i]`.
3. To reduce `b[k*]`, we need to reduce some `a[i]` values where both `a[i]` and `a[k*-i]` are large.
4. The "cost" of reducing `a[i]` is that it also reduces (or increases) other `b[k]` values.

Let me also think about the "lr" (learning rate) for Nesterov. The issue is that with very high beta, the gradient is very concentrated (it's essentially the gradient of the max function), so the step size needs to be very small to avoid oscillation.

A better approach: use a "backtracking line search" for the Nesterov step. After computing the gradient, try a step, and if the score increases, halve the step size.

Actually, for the LSE objective, the objective is smooth (with smoothness parameter proportional to beta), so we can use a theoretical step size `lr = 1/(L * sqrt(n))` where `L` is the Lipschitz constant of the gradient. But this is hard to estimate in practice.

A practical approach: use "Adagrad" or "RMSProp" to adapt the step size per coordinate. This would help when the gradient is concentrated at a few coordinates.

Let me implement an "Adagrad-Nesterov" hybrid:

```python
def nesterov_adagrad(a, max_time=100.0, beta_start=50.0, beta_end=200000.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # Nesterov velocity
    G = np.zeros(n)  # Adagrad accumulator
    lr_base = 0.1
    eps = 1e-8
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.01
    mu = 0.9
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.00001 * it)
        
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_score_and_grad(a_look, beta)
        
        # Adagrad step
        G += grad ** 2
        lr = lr_base / np.sqrt(G + eps)
        
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        beta = min(beta * beta_growth, beta_end)
        
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a
```

"""
Multi-phase optimizer minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: aggressive Fienup -> Nesterov+Adagrad LSE (very high beta) -> L-BFGS escalating beta
-> adaptive coordinate descent -> final L-BFGS polish.
Key innovations: Adagrad-adaptive Nesterov, much higher LSE beta (up to 200k),
escalating L-BFGS beta, peak-targeted coordinate descent with 1D line search.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_momentum(a, n_iters=500, momentum_init=0.0, momentum_end=0.5,
                    thresh_init=3.0, thresh_end=1.0, clip_init=0.2, clip_end=0.95):
    """Fienup iterative FFT amplitude modification with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.001, 1.0)
            scale[above] = factor
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def nesterov_adagrad(a, max_time=100.0, beta_start=50.0, beta_end=200000.0, lr_base=0.05):
    """Nesterov acceleration with Adagrad step-size adaptation on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    G = np.zeros(n)
    eps = 1e-8
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = 1.012
    mu = 0.85
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.85 + 0.00003 * it)
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break
        G += grad * grad
        lr = lr_base / np.sqrt(G + eps)
        step = lr * grad
        step_norm = float(np.linalg.norm(step))
        if step_norm > 0.5:
            step = step * (0.5 / step_norm)
        v = mu * v - step
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        beta = min(beta * beta_growth, beta_end)
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                # Reset some Adagrad state to allow further progress
                G *= 0.5
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def nesterov_simple(a, max_time=100.0, beta_start=100.0, beta_end=100000.0, lr_start=0.005, lr_end=0.0001):
    """Nesterov with linear LR decay, simple fixed momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:

    mu = 0.92
    while time.monotonic() - t_start < max_time:
        it += 1
        t = (time.monotonic() - t_start) / max_time
        lr = lr_start * (lr_end / lr_start) ** t
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)

            break
        v = mu * v - lr * grad / max(gnorm / (n ** 0.5), 1e-10)
        v = v / max(float(np.linalg.norm(v)), 1e-10) * min(float(np.linalg.norm(v)), 0.3)
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())

            break
        a = a / S
        beta = min(beta * beta_growth, beta_end)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coordinate_descent(a, max_sweeps=5, n_top=40, max_coords=800):
    """Coordinate descent targeting top peak contributors with multiplicative scaling."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(max_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        coord_set = set()
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            # Only add high-contribution coords
            for j in range(j_start, j_end + 1):
                if a[j] > 1e-7:
                    coord_set.add(j)
        coords = sorted(coord_set)
        if len(coords) > max_coords:
            # Prioritize coords with highest a[j]
            coords = sorted(coords, key=lambda j: -a[j])[:max_coords]
            coords = sorted(coords)
        for j in coords:
            j = int(j)
            old_val = a[j]
            if old_val < 1e-10:
                continue
            improved = False
            # Try geometrically spaced factors
            for factor in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]:
                a[j] = old_val * factor
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = old_val
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_scaled.copy()
                    a = a_scaled
                    improved = True
                    break
            if not improved:
                a[j] = old_val
    return best_a

def upsample(a, new_n):
    """Upsample a sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_chirp_init(n):
    """Chirp-based initialization with good autocorrelation properties."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.zeros(n)
    for rate in [1.0, 2.0, 3.0, 5.0, 7.0]:
        a += np.abs(np.sin(np.pi * rate * x * x))
    a += 0.05
    return a / a.sum()


def build_gaussian_init(n):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_mixed_init(n):
    """Mixed harmonic initialization."""
    x = np.linspace(0, 1, n)
    a = (0.4 + 0.25 * np.cos(2 * np.pi * x) + 0.15 * np.cos(5 * np.pi * x)
         + 0.1 * np.cos(11 * np.pi * x) + 0.05 * np.cos(19 * np.pi * x))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_window_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    print(f"End Phase 2: best = {best_score:.8f}")

    # Phase 3: L-BFGS with escalating beta (38% of time)
    # ============================================================
    phase3_end = t_start + 0.78 * (total_budget - 60.0)
    print(f"\n=== Phase 3: L-BFGS polish (t < {phase3_end - t_start:.0f}s) ===")

    lbfgs_betas = [500, 2000, 5000, 10000, 30000, 80000, 150000]
    for i, beta in enumerate(lbfgs_betas):
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 20:
            break
        # Allocate more time to later (higher beta) runs
        frac = 0.15 if i < 3 else 0.20
        budget = time_left * frac
        maxfunc = max(5000, int(budget * 80))
        print(f"  L-BFGS beta={beta}, budget~{budget:.0f}s, maxfun={maxfunc}")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=maxfunc)
        update_best(sc, a_opt, f"lbfgs-{beta}")

    print(f"End Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Adaptive coordinate descent (10% of time)
    # ============================================================
    phase4_end = t_start + 0.88 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Coordinate descent (t < {phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 10:
        time_left = phase4_end - time.monotonic()
        budget = min(time_left, 120.0)
        print(f"  Adaptive coord descent, time={budget:.0f}s")
        a_cd = adaptive_coord_desc(best_a, max_time=budget)
        update_best(compute_score(a_cd), a_cd, "coord-desc")

    print(f"End Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final L-BFGS polish + verification (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final polish ===")

    # One more L-BFGS with very high beta
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 200000, maxiter=3000, maxfun=20000)
        update_best(sc, a_opt, "final-lbfgs-200k")

    # Quick Nesterov from the result
    if time.monotonic() < deadline - 20:
        a_final = nesterov_adagrad(best_a, max_time=min(30.0, deadline - time.monotonic() - 15),
                                   beta_start=50000.0, beta_end=200000.0, lr_base=0.02)
        update_best(compute_score(a_final), a_final, "final-nest")

    # One more L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 300000, maxiter=2000, maxfun=10000)
        update_best(sc, a_opt, "final-lbfgs-300k")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
