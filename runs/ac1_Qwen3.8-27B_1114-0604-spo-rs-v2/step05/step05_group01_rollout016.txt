def next_power_of_two(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)

    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]

def normalize(a, target=1.0):

def gs_iteration(a, n_iter=50, alpha=1.0):
    """
    Gerchberg-Saxton style iteration to flatten the power spectrum.
    Alternates between:
    - Frequency domain: flatten |FFT(a)|
    - Time domain: enforce non-negativity and normalization
    """
    a = normalize(a.copy())
    n = len(a)
    M = 2 * n - 1
    N = next_power_of_two(M)
    
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        mags = np.abs(A)
        
        # Target: flat magnitude (geometric mean of non-DC components)
        non_dc = mags[1:]
        non_dc_pos = non_dc[non_dc > 1e-10]
        if len(non_dc_pos) > 0:
            target = np.mean(non_dc_pos) * alpha
        else:
            target = 1.0
        
        # Modify magnitudes to target
        for k in range(1, len(A)):
            if mags[k] > 1e-10:
                A[k] = A[k] / mags[k] * target
            # else: leave as 0
        
        # Preserve DC (sum constraint)
        A[0] = float(a.sum())
        
        a_new = np.fft.irfft(A, N)[:n]
        a = np.maximum(a_new, 0.0)
        a = normalize(a)
    
    return a

def gs_iteration_advanced(a, n_iter=50):

            target = np.mean(non_dc_pos)
        else:
            target = 1.0

        gamma = 0.5  # Strength of flattening
        for k in range(1, len(A)):
            if mags[k] > 1e-10:
                new_mag = (1 - gamma) * mags[k] + gamma * target
                A[k] = A[k] / mags[k] * new_mag

        a_new = np.fft.irfft(A, N)[:n]
        a = np.maximum(a_new, 0.0)
        a = normalize(a)

def adam_optimize(a0, beta, deadline, lr=0.01, max_iters=5000):

    a = normalize(a0.copy())
    n = len(a)
    log_a = np.log(a + 1e-30)  # Work in log space for non-negativity
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = exact_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        
        # Compute gradient in the "a" space
        score, grad = lse_score_grad(a, beta)
        
        # Transform gradient to log space: d(score)/d(log_a) = a * d(score)/d(a)
        log_grad = a * grad
        
        # But we also need to handle the non-negativity constraint
        # In log space, the constraint is automatically satisfied
        
        # Adam update
        m = b1 * m + (1 - b1) * log_grad
        v = b2 * v + (1 - b2) * log_grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step
        
        # Convert back and clip
        a = np.exp(log_a - np.max(log_a))  # Normalize for numerical stability
        a = np.clip(a, 0.0, None)
        S = float(a.sum())
        if S < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / S
        
        sc = exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return float(best_score), best_a

def multiplicative_update(a, k_active=3, eta=0.1):
    """Multiplicative weight update based on active set."""
    a = normalize(a.copy())
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    
    c = fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    
    # Get active set (top-k)
    if k_active >= len(d):

        w = np.zeros(len(d))

    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:

    a_new = a * np.exp(-eta * h / hmax)
    return normalize(a_new)

def resample(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n_new) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

def perturb(a, rng, strength=None):

The inverse DFT of |FFT_padded(a)|^2 gives the circular convolution of the padded a with itself, which equals the linear convolution (since we padded enough).
