"""

import numpy as np

import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient via FFT.
    L = max(c) + (1/beta)*log(sum(exp(beta*(c - max(c)))))
    Score = 2n * L / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da_i = 2 * sum_{m} a[m] * p[m+i]
    # Computed via cross-correlation with FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_adaptive(a, n_iters=500, with_momentum=True):
    """
    Fienup with adaptive threshold and momentum.
    Flattens the power spectrum iteratively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        # Adaptive threshold: start wide, narrow down
        thresh = 2.5 - 1.5 * t  # 2.5 -> 1.0
        clip = 0.3 + 0.6 * t    # 0.3 -> 0.9
        momentum = 0.3 * t if with_momentum else 0.0

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(np.maximum(P[above], 1e-20))
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        if with_momentum:
            a_update = a_new - a
            v = momentum * v + a_update
            a = a + v
        else:
            a = a_new

        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def nesterov_lse_backtrack(a, beta_start=50.0, beta_end=50000.0, max_time=60.0,
                           lr_init=0.01, min_lr=1e-6):
    """
    Nesterov-accelerated projected gradient with backtracking line search.
    Uses LSE smoothing with exponentially increasing beta.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    beta_growth = 1.02
    lr = lr_init
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.00002 * it)

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute gradient
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search
        f_curr = compute_score(a)
        step = lr
        a_trial = None
        for _ in range(15):
            v = mu * v - step * grad
            a_trial = a + v
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                step *= 0.5
                continue
            a_trial = a_trial / S_trial
            f_trial = compute_score(a_trial)
            if f_trial < f_curr:
                break
            step *= 0.5
        else:
            # Accept anyway if step is very small
            if step < min_lr:
                break

        a = a_trial

        # Increase beta
        beta = min(beta * beta_growth, beta_end)
        # Decay lr slightly
        lr *= 0.9995

        # Track best
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a, best_sc


def peak_shaving(a, n_rounds=20, n_top_peaks=10, shrink=0.7):
    """
    Directly identify coordinates contributing most to the peak and reduce them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:n_top_peaks]

        for k in peak_idx:
            k = int(k)
            # Coordinates contributing to c[k]: a[j] * a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution of a[j] to c[k] is a[k-j]
            contrib = a[k - j_start: k - j_end + n]  # a[k-j] for j in range
            # Actually: for j in j_start..j_end, contribution is a[k-j]
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            if contrib.max() < 1e-10:
                continue

            # Find top contributors and shrink them
            top_count = min(8, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            for j_rel in top_local:
                j = j_start + int(j_rel)
                old = a[j]
                if old < 1e-10:
                    continue
                a[j] = old * shrink
                S = float(a.sum())
                if S < 1e-12:
                    a[j] = old
                    continue
                a = a / S
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    # Revert
                    a = best_a.copy()
                    break

    return best_a


def run_lbfgs(a0, beta, maxiter=5000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': 20000, 'maxls': 40,
                 'iprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_shaping(a, n_iters=100, floor=0.3):
    """
    Shape the power spectrum to be flat with a floor.
    This directly targets reducing the peak of the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_target = float(P.sum()) / (M_fft // 2 + 1)  # flat target

        # Scale down bins above floor * P_target
        above = P > P_target * (2.0 - floor)
        if not above.any():
            # Also try to boost below-target bins
            below = P < P_target * floor
            if below.any():
                scale = np.ones_like(P)
                scale[below] = (P_target * floor) / np.maximum(P[below], 1e-20)
                P[below] = P_target * floor
            else:
                break
        else:
            scale = np.ones_like(P)
            # Bring high bins down toward target
            scale[above] = (P_target / P[above]) ** 0.5
            P[above] = P_target

        # Reconstruct with scaled power (preserve phase)
        P_new = np.clip(P, P_target * 0.1, P_target * 3.0)
        A_mag = np.sqrt(P_new)
        A_phase = A / np.maximum(np.abs(A), 1e-20)
        A_new = A_mag * A_phase

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def diverse_init(n, seed=0):
    """Generate a diverse set of initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.001)
    inits.append(a / a.sum())

    # Gaussian center
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 3.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    inits.append(a / a.sum())

    # Multiple Gaussians
    centers = [n * 0.25, n * 0.5, n * 0.75]
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((x - c) / (n / 8.0)) ** 2)
    inits.append(a / a.sum())

    # Random (smoothed)
    a = rng.uniform(0.5, 1.5, n)
    # Smooth with a box filter
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    return inits
