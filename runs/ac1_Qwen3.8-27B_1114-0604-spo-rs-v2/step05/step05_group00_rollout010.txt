        if S_new < 1e-12:

        a = a_new / S_new
    
    return best_a

"""
Hybrid optimizer combining multiplicative weights (mirror descent) on the simplex,
adaptive LSE-smoothed L-BFGS, frequency-domain diverse initializations, and
basin hopping. Targets improvement over 1.506 by escaping local minima through
diverse restarts and using the geometrically-natural MW update for the simplex.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient (scale-invariant, sum normalized)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_norm = c / (S * S)
    c_max = float(c_norm.max())
    scaled = beta * (c_norm - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. x (before normalization)
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    # Score = 2n * L, but L depends on x through c and S
    # Since we normalize x to have sum=1 before calling, S=1 in practice
    # But for safety, compute full gradient
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    """Wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=100000):
    """L-BFGS-B on normalized LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lse_objective_func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1.0)
    S_out = a_out.sum()
    if S_out < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multiplicative_weights_optimize(a0, n_iters, eta0=0.5, eta_decay=0.001, top_k=5,
                                     use_averaging=True, check_interval=100):
    """
    Mirror descent with multiplicative weights update on the simplex.
    Uses top-K peak subgradient with exponential weighting.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Weighted Polyak averaging
    a_avg = a.copy()
    weight_sum = 1.0
    best_avg_a = a.copy()
    best_avg_sc = best_sc

    t_start = time.monotonic()

    for t in range(n_iters):
        # Adaptive step size: starts large, decays
        eta = eta0 / (1.0 + eta_decay * t)
        eta = min(eta, 10.0)

        c = fast_conv(a)
        S_curr = a.sum()
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S_curr * S_curr)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Update weighted Polyak average (weight = 1/eta)
        if use_averaging:
            w = 1.0 / eta
            a_avg = (a_avg * weight_sum + a * w) / (weight_sum + w)
            weight_sum += w
            if t % check_interval == 0:
                avg_sc = compute_score(a_avg)
                if avg_sc < best_avg_sc:
                    best_avg_sc = avg_sc
                    best_avg_a = a_avg.copy()
                    if avg_sc < best_sc:
                        best_sc = avg_sc
                        best_a = a_avg.copy()

        # Find top-K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Weighted subgradient from top peaks (exponential weighting by height)
        c_top = c[top_idx]
        c_max = c_top[0] + 1e-30
        peak_weights = np.exp(-0.5 * (c_max - c_top) / (c_max / n))
        peak_weights = peak_weights / peak_weights.sum()

        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            k = int(k)
            w_k = peak_weights[idx]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            # grad[m] += 2 * w_k * a[k-m] for m in [lo, hi]
            # This is equivalent to: grad[lo:hi+1] += 2*w_k * a[k-hi:k-lo+1][::-1]
            segment = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * w_k * segment

        # Center the gradient (remove mean to handle sum constraint)
        grad_centered = grad - grad.mean()

        # Multiplicative weights update
        log_scale = -eta * grad_centered
        log_scale -= log_scale.max()  # numerical stability
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 1e-30)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

    # Final check on average
    if use_averaging:
        avg_sc = compute_score(a_avg)
        if avg_sc < best_sc:
            best_sc = avg_sc
            best_a = a_avg.copy()

    return best_a


def frequency_domain_init(n, rng, mode='random_phase'):
    """Generate diverse initial sequences using frequency domain methods."""
    if mode == 'random_phase':
        # Random complex spectrum with flat magnitude
        M_fft = 1 << (2 * n - 1).bit_length()
        freqs = np.fft.rfftfreq(M_fft)
        # Low-pass filter to ensure smoothness
        cutoff = rng.uniform(0.3, 0.8)
        mask = np.ones_like(freqs)
        mask[np.abs(freqs) > cutoff * 0.5] = 0.0
        A = mask * (rng.standard_normal(M_fft // 2 + 1) + 1j * rng.standard_normal(M_fft // 2 + 1))
        a = np.fft.irfft(A, M_fft)[:n]
        a = np.maximum(a, 0.0)
    elif mode == 'lowpass_noise':
        a = rng.standard_normal(n)
        # Apply low-pass filter
        A = np.fft.rfft(a, 2 * n)
        freqs = np.fft.rfftfreq(2 * n)
        A[np.abs(freqs) > 0.3] = 0
        a = np.fft.irfft(A, 2 * n)[:n]
        a = np.maximum(a, 0.0)
    elif mode == 'bandpass':
        a = rng.standard_normal(n)
        A = np.fft.rfft(a, 2 * n)
        freqs = np.fft.rfftfreq(2 * n)
        lo = rng.uniform(0.05, 0.2)
        hi = rng.uniform(0.4, 0.7)
        A[(np.abs(freqs) < lo) | (np.abs(freqs) > hi)] = 0
        a = np.fft.irfft(A, 2 * n)[:n]
        a = np.maximum(a, 0.0)
    elif mode == 'structured_lowfreq':
        # Low-frequency cosine mixture
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for m in range(1, 6):
            freq = rng.integers(1, 8) * m
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            amp = 0.3 / np.sqrt(m)
            a += amp * np.cos(2.0 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n) * 0.5

    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def fienup_flatten(a, n_iters=200):
    """Fienup iteration to flatten autocorrelation peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        alpha = 0.15 + 0.5 * progress
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=200):
    """Fienup with quantile thresholding."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        q = 0.80 + 0.15 * (it / n_iters)
        thresh = np.quantile(P, q)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave_subgrad(a, n_iters=3000, top_k=5, step_init=0.02):
    """Subgradient peak shaving with Polyak averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    # Polyak averaging
    a_avg = a.copy()
    w_sum = 1.0
    best_avg_sc = best_sc

    sc_lb = 2.0 * n / (2.0 * n - 1.0) + 0.1  # practical lower bound

    for t in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        sc = 2.0 * n * M_val

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Polyak average update
        w = 1.0 / step
        a_avg = (a_avg * w_sum + a * w) / (w_sum + w)
        w_sum += w
        if t % 200 == 0:
            avg_sc = compute_score(a_avg)
            if avg_sc < best_avg_sc:
                best_avg_sc = avg_sc
                if avg_sc < best_sc:
                    best_sc = avg_sc
                    best_a = a_avg.copy()

        # Top-K peaks with exponential weighting
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        c_top = c[top_idx]
        spread = c_top[0] - c_top[-1] + 1e-30
        weights = np.exp(-2.0 * (c_top[0] - c_top) / spread)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            segment = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * weights[idx] * segment

        # Normalize gradient and take step
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        # Adaptive step (Polyak-Hobo)
        step = min(step * 0.999, (sc - sc_lb) / (gn * gn + 1e-30), 0.05)
        step = max(step, 1e-8)

    return best_a


def coord_descent_peaks(a, n_sweeps=10, n_top=50):
    """Coordinate descent targeting top contributors to the peak lag."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.3]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def basin_hoppable_perturb(a, rng, scale=0.03, mode='mixed'):
    """Generate a perturbation that can escape local minima."""
    n = len(a)
    if mode == 'multiplicative_noise':
        a_new = a * (1.0 + scale * rng.standard_normal(n))
        a_new = np.maximum(a_new, 0.0)
    elif mode == 'additive_noise':
        a_new = a + scale * 0.1 * rng.standard_normal(n)
        a_new = np.maximum(a_new, 0.0)
    elif mode == 'structured':
        x = np.linspace(0, 1, n)
        pert = np.zeros(n)
        for m in range(1, 6):
            freq = rng.integers(1, 12)
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            amp = scale / np.sqrt(m)
            pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
        a_new = a * (1.0 + pert)
        a_new = np.maximum(a_new, 0.0)
    elif mode == 'swap':
        # Randomly swap blocks
        a_new = a.copy()
        for _ in range(3):
            i = rng.integers(0, n - 64)
            j = rng.integers(0, n - 64)
            blen = rng.integers(16, 128)
            blen = min(blen, n - i, n - j)
            a_new[i:i+blen], a_new[j:j+blen] = a[j:j+blen].copy(), a[i:i+blen].copy()
    else:
        a_new = a.copy()

    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy optimizer: diverse initialization -> MW optimization ->
    L-BFGS polish -> basin hopping -> final refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            old = best_score
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old-sc:.6f})")
            return True
        return False

    # Initialize
    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        print("No parent available, using n=4096")

    print(f"Starting best: {best_score:.8f}")

    # ================================================================
    # Phase 1: Diverse Initialization (15% of time)
    # ================================================================
    phase1_end = t_start + 0.15 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Diverse init (until t={phase1_end - t_start:.0f}s) ===")

    candidate_pool = [best_a.copy()]

    # Try several frequency-domain inits
    init_modes = ['random_phase', 'lowpass_noise', 'bandpass', 'structured_lowfreq',
                  'random_phase', 'lowpass_noise', 'random_phase', 'structured_lowfreq']
    for i, mode in enumerate(init_modes):
        if time.monotonic() > phase1_end:
            break
        a_init = frequency_domain_init(n0, rng, mode=mode)
        # Quick Fienup to improve
        a_f = fienup_flatten(a_init, n_iters=80)
        sc = compute_score(a_f)
        print(f"  Init {i} ({mode}): score={sc:.6f}")
        candidate_pool.append(a_f)
        update(sc, a_f, f"INIT-{mode}")

    # Fienup on parent
    if parent is not None:
        a_f = fienup_flatten(parent, n_iters=150)
        sc = compute_score(a_f)
        update(sc, a_f, "FIENUP-parent")
        candidate_pool.append(a_f)

        a_f2 = fienup_quantile(parent, n_iters=150)
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-q-quantile")
        candidate_pool.append(a_f2)

    # Sort candidates by score
    candidate_pool.sort(key=lambda x: compute_score(x))
    print(f"  Phase 1 done. Best: {best_score:.8f}")

    # ================================================================
    # Phase 2: MW + L-BFGS on top candidates (35% of time)
    # ================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: MW + L-BFGS (until t={phase2_end - t_start:.0f}s) ===")

    top_candidates = candidate_pool[:4]
    if best_a not in top_candidates:
        top_candidates[0] = best_a.copy()

    for ci, a_cand in enumerate(top_candidates):
        if time.monotonic() > phase2_end - 60:
            break
        print(f"  Candidate {ci}: score={compute_score(a_cand):.6f}")

        # MW optimization with different parameters
        remaining = phase2_end - time.monotonic()
        n_mw_iters = min(5000, int(remaining * 8))

        # MW pass 1: moderate eta
        a_mw = multiplicative_weights_optimize(a_cand, n_iters=n_mw_iters,
                                                 eta0=0.3, eta_decay=0.0005, top_k=5)
        sc = compute_score(a_mw)
        update(sc, a_mw, f"MW-c{ci}-1")

        # MW pass 2: larger eta for more exploration
        if time.monotonic() < phase2_end - 60:
            a_mw2 = multiplicative_weights_optimize(a_mw, n_iters=n_mw_iters // 2,
                                                     eta0=1.0, eta_decay=0.001, top_k=3)
            sc = compute_score(a_mw2)
            update(sc, a_mw2, f"MW-c{ci}-2")

        # L-BFGS polish on best MW result
        if time.monotonic() < phase2_end - 30:
            a_best_cand = a_mw if compute_score(a_mw) < compute_score(a_mw2) else a_mw2
            sc, a_opt = run_lbfgs(a_best_cand, 1000, maxiter=5000, maxfun=80000)
            update(sc, a_opt, f"LBFGS-c{ci}")

            # Higher beta L-BFGS
            if time.monotonic() < phase2_end - 30 and sc < best_score + 0.01:
                sc2, a_opt2 = run_lbfgs(a_opt, 5000, maxiter=3000, maxfun=50000)
                update(sc2, a_opt2, f"LBFGS-hb-c{ci}")

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # ================================================================
    # Phase 3: Basin Hopping (30% of time)
    # ================================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Basin hopping (until t={phase3_end - t_start:.0f}s) ===")

    n_hops = 12
    for hop in range(n_hops):
        if time.monotonic() > phase3_end - 60:
            break

        # Generate perturbation (varying scale and mode)
        scale = 0.02 + 0.02 * (hop % 4)
        mode = ['multiplicative_noise', 'structured', 'swap', 'additive_noise'][hop % 4]
        a_pert = basin_hoppable_perturb(best_a, rng, scale=scale, mode=mode)

        # Quick optimization: short MW + short L-BFGS
        sc_pert = compute_score(a_pert)

        # Short MW
        a_mw = multiplicative_weights_optimize(a_pert, n_iters=1500,
                                                 eta0=0.5, eta_decay=0.001, top_k=5)
        sc_mw = compute_score(a_mw)
        update(sc_mw, a_mw, f"HOP{hop}-MW")

        # Short L-BFGS from best of perturbation
        a_start = a_mw if sc_mw < sc_pert else a_pert
        sc, a_opt = run_lbfgs(a_start, 2000, maxiter=3000, maxfun=50000)
        update(sc, a_opt, f"HOP{hop}-LBFGS")

        # If we found something good, do a longer optimization
        if sc < best_score + 1e-6 and time.monotonic() < phase3_end - 120:
            sc2, a_opt2 = run_lbfgs(a_opt, 8000, maxiter=5000, maxfun=80000)
            update(sc2, a_opt2, f"HOP{hop}-deep")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # ================================================================
    # Phase 4: Peak shaving + final polish (15% of time)
    # ================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Final polish ===")

    # Peak shaving with Polyak averaging
    if time.monotonic() < phase4_end - 180:
        a_ps = peak_shave_subgrad(best_a, n_iters=3000, top_k=5, step_init=0.015)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")

    # L-BFGS with high beta
    if time.monotonic() < phase4_end - 120:
        remaining = phase4_end - 120 - time.monotonic()
        maxiter = min(8000, int(remaining * 20))
        sc, a_final = run_lbfgs(best_a, 10000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LBFGS-10k")

    # Very high beta L-BFGS (close to exact max)
    if time.monotonic() < phase4_end - 60:
        sc, a_final = run_lbfgs(best_a, 30000, maxiter=3000, maxfun=50000)
        update(sc, a_final, "FINAL-LBFGS-30k")

    # Coordinate descent
    if time.monotonic() < phase4_end - 30:
        a_cd = coord_descent_peaks(best_a, n_sweeps=8, n_top=100)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # Final MW with small eta for fine-tuning
    if time.monotonic() < phase4_end - 10:
        a_mw_final = multiplicative_weights_optimize(best_a, n_iters=2000,
                                                      eta0=0.1, eta_decay=0.0002, top_k=3)
        sc = compute_score(a_mw_final)
        update(sc, a_mw_final, "FINAL-MW")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
