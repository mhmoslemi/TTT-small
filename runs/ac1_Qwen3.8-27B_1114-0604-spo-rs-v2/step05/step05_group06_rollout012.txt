def _lp_direction(a, M, K=50):
    """
    Find the LP-optimal direction for reducing max convolution.
    Maximize sum(b) subject to (a*b)[k] <= M for top-K k, b >= 0.
    Returns b/sum(b) as the direction.
    """
    n = len(a)
    conv = _self_conv(a)
    # Get top K positions
    top_k = np.argpartition(conv, -K)[-K:]
    
    # Build constraint matrix A_ub (K x n)
    # A_ub[k, i] = a[i] if 0 <= k-i < n, else 0
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        # b[j] appears in constraint k when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[idx, j_start:j_end+1] = a[j_start:j_end+1]
    
    b_ub = np.full(K, M)
    
    # Solve LP: maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='hihps')
    
    if res.status != 0:
        return None
    
    b_opt = res.x
    S = b_opt.sum()
    if S < 1e-12:
        return None
    
    return b_opt / S

2. LP-based active-set direction finding (cutting-plane style)
3. Intensive subgradient polishing with momentum and backtracking line search
4. Simplex projection for feasibility
5. Local coordinate search for final refinement
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm_simplex(b):
    """Project onto {x >= 0, sum(x) = 1}."""
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0:
        return b
    # Sort in decreasing order
    u = np.sort(b)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(b - theta, 0.0)
    S = out.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return out / S


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.ones(a.size) / a.size
    return a / S

    Compute grad[j] = sum_k w[k] * (b[k-j] + b[j-k]) for j=0..n-1.
    b has length n, w has length 2n-1.

    n = b.size
    M = 2 * n - 1
    N = _next_pow2(3 * n - 2)
    B = np.fft.rfft(b, N)
    W = np.fft.rfft(w, N)
    # A[j] = sum_k w[k] * b[k-j] = cross-correlation
    A = np.fft.irfft(np.conj(W) * B, N)[:n]
    # B[j] = sum_k w[k] * b[j-k] = convolution
    Bc = np.fft.irfft(W * B, N)[:n]
    grad = A + Bc
    return np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)


def _subgrad_step(b, eta, active_set=None, tol=1e-6):
    """
    One subgradient step with simplex projection.
    b is already normalized (sum=1, b>=0).
    Returns new b and the score.
    """
    n = b.size
    c = _self_conv(b)
    d = float(n) * c  # S=1
    dmax = float(d.max())
    if dmax <= 0:
        return b, float(2.0 * dmax)

    if active_set is None:
        thresh = dmax - tol * max(dmax, 1e-10)
        active = d >= thresh
        if not active.any():
            active[np.argmax(d)] = True
    else:
        active = active_set

    n_act = int(active.sum())
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / n_act

    grad = _cross_corr_grad(b, w)
    grad = 2.0 * float(n) * grad

    # Step
    b_new = b - eta * grad
    b_out = _norm_simplex(b_new)
    sc = _score(b_out)
    return b_out, sc


def _subgrad_polish_initiatives(b, max_iters=500, deadline=None, verbose=True, momentum=0.9):
    """
    Subgradient polishing with Nesterov momentum and backtracking.
    """
    b = _normalize(b)
    best_b = b.copy()
    best_sc = _score(b)
    n = len(b)
    v = np.zeros(n)

    # Adaptive step size
    c0 = 0.5
    t0 = 50

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _self_conv(b)
        d = float(n) * c
        dmax = float(d.max())
        cur_sc = 2.0 * dmax

        # Active set: positions within 0.01% of max
        thresh = dmax - 1e-4 * max(dmax, 1e-10)
        active = d >= thresh
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act

        raw_grad = _cross_corr_grad(b, w)
        grad = 2.0 * float(n) * raw_grad
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Momentum
        v = momentum * v + grad / gnorm

        # Step size with diminishing schedule
        eta = c0 / (it + t0)

        # Backtracking: try eta, eta/2, eta/4, ...
        best_eta = 0
        best_new_sc = cur_sc
        for trial in range(10):
            t = eta / (2 ** trial)
            if t < 1e-10:
                break
            b_new = _norm_simplex(b - t * v)
            sc_new = _score(b_new)
            if sc_new < best_new_sc:
                best_new_sc = sc_new
                best_eta = t
            if sc_new < cur_sc:
                break

        if best_eta == 0:
            # Try without momentum
            for trial in range(5):
                t = eta / (2 ** trial)
                if t < 1e-12:
                    break
                b_new = _norm_simplex(b - t * grad)
                sc_new = _score(b_new)
                if sc_new < cur_sc:
                    b = b_new
                    v = 0.9 * v + grad
                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_b = b.copy()
                        if verbose and it % 50 == 0:
                            print(f"    SG it={it}: {best_sc:.10f}")
                    break
            else:
                if it > 50:
                    break
        else:
            b = _norm_simplex(b - best_eta * v)
            if best_new_sc < best_sc:
                best_sc = best_new_sc
                best_b = b.copy()
                if verbose and it % 50 == 0:
                    print(f"    SG it={it}: {best_sc:.10f}")

    return best_sc, best_b


def _lp_direction(a, top_k_positions, M_val):
    """
    Solve LP: maximize sum(b) s.t. (a*b)[k] <= M_val for k in top_k_positions, b >= 0.
    Returns b/sum(b) or None.
    """
    n = len(a)
    K = len(top_k_positions)
    if K == 0:
        return None

    # Build sparse constraint matrix
    # A_ub has K rows, n columns
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(top_k_positions):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[idx, j_start:j_end + 1] = a[j_start:j_end + 1]

    b_ub = np.full(K, M_val)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='hihps')
        if res.status != 0:
            return None
        b_opt = res.x
        S = float(b_opt.sum())
        if S < 1e-12:
            return None
        return b_opt / S
    except Exception:
        return None


def _lp_refine(b, max_iters=1000, deadline=None, K=50, verbose=True):
    """
    Active-set LP refinement.
    """
    b = _normalize(b)
    best_b = b.copy()
    best_sc = _score(b)
    n = len(b)

    # Damping
    alpha = 0.3

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _self_conv(b)
        d = float(n) * c
        dmax = float(d.max())
        cur_sc = 2.0 * dmax

        # Top K positions
        k = min(K, len(c))
        top_k = np.argpartition(c, len(c) - k)[len(c) - k:]

        # Solve LP
        b_dir = _lp_direction(b, top_k, dmax)
        if b_dir is None:
            break

        # Take damped step
        b_new = (1.0 - alpha) * b + alpha * b_dir
        b_new = _normalize(b_new)
        sc_new = _score(b_new)

        if sc_new < cur_sc:
            b = b_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_b = b.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP it={it}: {best_sc:.10f} (K={K}, alpha={alpha})")
            # Increase K slowly
            K = min(200, K + 1)
        else:
            # Reduce alpha
            alpha *= 0.9
            if alpha < 0.01:
                alpha = 0.3
                K = max(10, K // 2)
            # Try smaller step
            b_new2 = (1.0 - alpha * 0.1) * b + alpha * 0.1 * b_dir
            b_new2 = _normalize(b_new2)
            sc_new2 = _score(b_new2)
            if sc_new2 < cur_sc:
                b = b_new2
                if sc_new2 < best_sc:
                    best_sc = sc_new2
                    best_b = b.copy()

    return best_sc, best_b


def _local_search(b, max_evals=2000, deadline=None, verbose=True):
    """
    Coordinate-wise local search: try modifying individual elements.
    """
    b = _normalize(b)
    best_b = b.copy()
    best_sc = _score(b)
    n = len(b)
    rng = np.random.default_rng(12345)

    # Focus on elements with moderate values (not too small, not too large)
    order = np.argsort(b)[::-1]

    evals = 0
    stalled = 0

    while evals < max_evals:
        if deadline is not None and time.monotonic() > deadline:
            break
        if stalled > 200:
            break

        # Pick a random coordinate (weighted toward larger elements)
        idx = order[rng.integers(0, n)]

        # Try several multipliers
        improved = False
        orig_val = b[idx]
        for mult in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0):
            new_val = orig_val * mult
            if new_val < 0:
                continue
            b_test = b.copy()
            b_test[idx] = new_val
            b_test = _normalize(b_test)
            sc = _score(b_test)
            evals += 1
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_b = b_test.copy()
                b = b_test.copy()
                improved = True
                break

        if improved:
            stalled = 0
            # Re-sort occasionally
            if evals % 100 == 0:
                order = np.argsort(b)[::-1]
        else:
            stalled += 1

        if verbose and evals % 500 == 0 and evals > 0:
            print(f"    LS evals={evals}: {best_sc:.10f}")

    return best_sc, best_b


def _lse_obj_grad(a, beta, k, n):
    """Active-set LSE objective and gradient."""
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act

    # Compute cross-correlation gradient
    corr = _cross_corr_grad(a, w_full)
    LDS = float(w_act @ d_act)
    # grad of 2*L: 4n*corr/S^2 - 4*LDS/S
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    """Run L-BFGS-B on active-set LSE."""
    a = _normalize(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.005 * max(0.3, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 6000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj_and_grad(x):
        v, g = _lse_obj_grad(x, beta, k, n)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj_and_grad, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _normalize(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _normalize(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization combining LSE, LP active-set,
    subgradient with momentum, and local search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _normalize(p)
    except Exception:
        pass

    # Target length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192

    print(f"Parent: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    # Generate initial candidates
    inits = []
    if parent is not None:
        inits.append(('parent', _normalize(_upsample(parent, target_n))))
    inits.append(('uniform', np.ones(target_n) / target_n))
    for i in range(4):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # Evaluate and pick best init
    best_init_sc = np.inf
    best_init = None
    for name, arr in inits:
        arr = _normalize(arr)
        sc = _score(arr)
        if sc < best_init_sc:
            best_init_sc = sc
            best_init = arr
    print(f"Best init: score={best_init_sc:.8f}")

    best_score = best_init_sc
    best_seq = best_init.copy()
    n = len(best_seq)

    # === Phase 1: LSE optimization (coarse) ===
    print("\n=== Phase 1: LSE optimization ===")
    p1_deadline = t_start + 300.0
    a_work = best_seq.copy()

    stages_lse = [
        (5, 100.0, 0.15),
        (10, 300.0, 0.20),
        (20, 800.0, 0.25),
        (50, 2000.0, 0.30),
        (100, 5000.0, 0.10),
    ]
    cum = 0.0
    for k, beta, frac in stages_lse:
        cum += frac
        stage_end = t_start + cum * 300.0
        stage_end = min(stage_end, p1_deadline - 5.0)
        remaining = stage_end - time.monotonic()
        if remaining < 5.0:
            break
        sc, a_new = _run_lbfgs(a_work, beta, k, remaining, 400)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"  LSE K={k}, beta={beta:.0f}: {best_score:.10f}")
        a_work = _normalize(a_new)

    a_work = best_seq.copy()

    # === Phase 2: LP-based active-set refinement ===
    print(f"\n=== Phase 2: LP refinement (score={best_score:.10f}) ===")
    p2_deadline = t_start + 1200.0

    # Do LP refinement in segments, increasing K
    K_start = 20
    for seg in range(5):
        seg_deadline = t_start + 1200.0 * (seg + 1) / 5.0
        seg_deadline = min(seg_deadline, p2_deadline - 10.0)
        K_seg = K_start + seg * 20
        sc, a_new = _lp_refine(a_work, max_iters=200, deadline=seg_deadline,
                                K=K_seg, verbose=(seg == 0))
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"  LP seg{seg} K={K_seg}: {best_score:.10f}")
        a_work = _normalize(a_new)
        if time.monotonic() > p2_deadline - 10.0:
            break

    a_work = best_seq.copy()

    # === Phase 3: Subgradient polishing with momentum ===
    print(f"\n=== Phase 3: Subgradient polishing (score={best_score:.10f}) ===")
    p3_deadline = t_start + 1800.0
    sc, a_new = _subgrad_polish_initiatives(a_work, max_iters=1000,
                                             deadline=p3_deadline, verbose=True)
    if sc < best_score:
        best_score = sc
        best_seq = a_new.copy()
    a_work = _normalize(a_new)

    # === Phase 4: Another round of LP + subgradient ===
    print(f"\n=== Phase 4: Second LP+SG round (score={best_score:.10f}) ===")
    p4_deadline = t_start + 2100.0
    sc, a_new = _lp_refine(a_work, max_iters=200, deadline=min(p4_deadline, p3_deadline + 200),
                            K=50, verbose=False)
    if sc < best_score:
        best_score = sc
        best_seq = a_new.copy()
        print(f"  LP round2: {best_score:.10f}")
    a_work = _normalize(a_new)

    sc, a_new = _subgrad_polish_initiatives(a_work, max_iters=500,
                                             deadline=p4_deadline, verbose=False)
    if sc < best_score:
        best_score = sc
        best_seq = a_new.copy()
        print(f"  SG round2: {best_score:.10f}")
    a_work = _normalize(a_new)

    # === Phase 5: Local search ===
    print(f"\n=== Phase 5: Local search (score={best_score:.10f}) ===")
    p5_deadline = t_start + 2300.0
    sc, a_new = _local_search(a_work, max_evals=3000, deadline=p5_deadline, verbose=True)
    if sc < best_score:
        best_score = sc
        best_seq = a_new.copy()
    a_work = _normalize(a_new)

    # === Phase 6: Final intensive subgradient ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.10f}) ===")
    final_deadline = deadline - 10.0
    sc, a_new = _subgrad_polish_initiatives(a_work, max_iters=300,
                                             deadline=final_deadline, verbose=False)
    if sc < best_score:
        best_score = sc
        best_seq = a_new.copy()
        print(f"  Final SG: {best_score:.10f}")

    # === Prepare output ===
    out = _normalize(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.10f}, verified={verified:.10f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
