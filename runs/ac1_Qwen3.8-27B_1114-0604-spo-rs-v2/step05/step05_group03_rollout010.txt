        target = np.full(2*n-1, c_max * (1.0 - 0.3))  # target below current max
        # Or: target = c_mean (flat at the mean)
        target = np.full(2*n-1, c_mean)
        
        # Solve: min ||W^{1/2} (2 * T_a @ da + (c - target))||^2
        # Normal equations: (T_a^T W T_a) da = -2 * T_a^T
        # W (c - target)
        # Where W is diagonal with entries w_k / (2*n-1) (to normalize)
        
        # Actually, let me formulate it as:
        # min_da ||(c + 2*T_a@da - target)||_W^2
        # = min_da (c + 2*T_a@da - target)^T W (c + 2*T_a@da - target)
        # Gradient: 4 * T_a^T W (c + 2*T_a@da - target) = 0
        # So: (T_a^T W T_a) da = -T_a^T W (c - target)
        
        # Use CG to solve the linear system
        
        # Precompute rhs = -T_a^T @ (w * (c - target))
        residual = w * (c - target)
        rhs = -T_a_adj(a, residual)
        
        # CG iteration
        da = np.zeros(n)
        r = rhs.copy()
        p_vec = r.copy()
        r_norm = float(r @ r)
        
        for cg_it in range(100):
            Ap = T_a_W_T_a(a, w, p_vec)
            p_Ap = float(p_vec @ Ap)
            if abs(p_Ap) < 1e-30:
                break
            alpha_cg = r_norm / p_Ap
            da += alpha_cg * p_vec
            r -= alpha_cg * Ap
            new_r_norm = float(r @ r)
            if new_r_norm < tol * r_norm:
                break
            p_vec = r + (new_r_norm / r_norm) * p_vec
            r_norm = new_r_norm
        
        # Line search
        best_alpha = 0.0
        best_sc = best_score
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_try = a + alpha * da
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
        
        if best_alpha == 0.0:
            # Try smaller steps
            for alpha in [0.01, 0.02, 0.05]:
                a_try = a + alpha * da
                a_try = np.maximum(a_try, 0.0)
                St = float(a_try.sum())
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc = compute_score(a_try)
                if sc < best_sc:
                    best_sc = sc
                    best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = a + best_alpha * da
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        a = a / S
        
        if best_sc < best_score:
            best_score = best_sc
            best_a = a.copy()
    
    return best_a, best_score
