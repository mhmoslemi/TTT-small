import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * L / S^2 where L = (1/beta) * log(sum_k exp(beta * c_k))
    Grad_j = (4n/S^2) * (a cross-corr p)_j - 4nL/S^3
    where p_k = exp(beta*c_k) / sum_m exp(beta*c_m)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Compute (a cross-corr p)_j = sum_m a_m * p_{m+j} for j=0,...,n-1
    # = conv(a_flipped, p)[n-1 : 2n-1]
    a_flipped = a[::-1]
    # a_flipped has length n, p has length 2n-1
    # conv length = n + (2n-1) - 1 = 3n - 2
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_f = np.fft.rfft(a_flipped, M_fft)
    P = np.fft.rfft(p, M_fft)
    cross = np.fft.irfft(A_f * P, M_fft)[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * cross / S2 - 2.0 * L / S3)
    
    return score, grad

    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    a0 = a0 / a0.sum()  # normalize to sum=1 for better conditioning
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:

    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_attack(a, n_iters=2000, n_top=10, step_size=0.01):
    """
    Targeted peak reduction: identify the top peaks in the convolution
    and use gradient descent specifically to reduce them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_size
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute subgradient from top peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        M_val = float(c[top_idx[0]])
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Normalize and step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step size
        sc_lower = 2.0 * n / (2.0 * n - 1.0)
        sc_cur = 2.0 * n * M_val / S2
        step = min(step, max(1e-6, (sc_cur - sc_lower) / (gn * gn + 1e-30) * 0.5))
        step = min(step, 0.05)
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def coordinate_ascent_peak(a, n_sweeps=10, n_top=50):
    """
    Coordinate ascent: for each element that contributes to the peak,
    try to reduce the peak by adjusting it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Elements contributing to c[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Contribution of each element to c[k_star]
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = 2.0 * a[j] * a[k_star - j] if 0 <= k_star - j <= n-1 else 0.0
        
        # Sort by contribution (descending)
        top_elems = np.argsort(contrib)[::-1][:n_top]
        
        for j in top_elems:
            j = int(j)
            if a[j] < 1e-10:
                continue
            a_old = a[j]
            # Try reducing this element
            for factor in [0.1, 0.3, 0.5, 0.7, 0.85, 0.95]:
                a[j] = a_old * factor
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = a_old
                    break
            # Try increasing too (might reduce other peaks)
            for factor in [1.1, 1.3, 1.5]:
                a[j] = a_old * factor
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = a_old
                    break
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    return best_a

"""
Multi-phase optimizer for minimum-peak self-convolution sequences.
Phases: (1) GS-Fienup for good init, (2) L-BFGS with exact FFT-based gradient,
(3) peak-attack subgradient, (4) coordinate descent + final L-BFGS polish.
Key: correct gradient via cross-correlation identity grad_j = 2*(a cross-corr p)_j.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    L = (1/beta)*log(sum_k exp(beta*c_k))  [shifted by c_max for stability]
    score = 2n * L / S^2
    grad_j = (4n/S^2)*(a cross-corr p)_j - 4nL/S^3
    where p_k = exp(beta*(c_k - c_max)) / sum_m exp(beta*(c_m - c_max))
    and (a cross-corr p)_j = sum_m a_m * p_{m+j}  [computed via FFT]
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    psum = float(exp_scaled.sum())
    p = exp_scaled / psum
    L = c_max + np.log(psum) / beta

    score = 2.0 * n * L / (S * S)

    # Compute (a cross-corr p)_j for j=0..n-1
    # = conv(a_flipped, p)[n-1 : 2n-1]
    a_flipped = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_f = np.fft.rfft(a_flipped, M_fft)
    P = np.fft.rfft(p, M_fft)
    cross = np.fft.irfft(A_f * P, M_fft)[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * cross / S2 - 2.0 * L / S3)

    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize (returns f and grad)."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000):
    """Run L-BFGS-B on the LSE-smoothed objective with bounds [0, 1000]."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n

    a0 = a0 / a0.sum() * (n * 0.5)

    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    T = np.fft.rfft(b_target, M)

    S = float(a.sum())

        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_magnitude_only(a, n_iters=400, M=None, alpha=0.4):
    """GS variant updating only magnitude, preserving phase."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target, M)
    T_mag = np.maximum(np.abs(T), 1e-20)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_adaptive(a, n_iters=500, M=None):
    """GS with adaptive step size (accept/reject based on score)."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.rfft(b_target, M)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = 0.7

    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new <= best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(0.8, alpha * 1.03)
        else:
            alpha = max(0.03, alpha * 0.75)

    return best_a


def peak_attack(a, n_iters=3000, n_top=10, step_init=0.04):
    """Subgradient method targeting top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_top]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max(1e-8, (sc - sc_lb) / (gn * gn + 1e-30) * 0.5))
        step = min(step, 0.06)

        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9992

    return best_a


def coordinate_ascent_peak(a, n_sweeps=8, n_top=40):
    """
    Coordinate descent on peak-contributing elements.
    For the dominant peak lag, try scaling down top-contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution of each a[j] to c[k_star]
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            kj = k_star - j
            if 0 <= kj <= n - 1:
                contrib[j] = 2.0 * a[j] * a[kj]

        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local
        top_global = top_global[top_global < n]

        for j in top_global:
            j = int(j)
            if a[j] < 1e-10:
                continue
            a_old = a[j]
            cur_sc = best_sc

            # Try reducing
            for factor in [0.5, 0.7, 0.85, 0.95]:
                a[j] = a_old * factor
                sc = compute_score(a)
                if sc < cur_sc:
                    cur_sc = sc
                else:
                    a[j] = a_old
                    break
            else:
                a[j] = a_old

            # Try increasing (might reduce secondary peaks)
            for factor in [1.2, 1.5]:
                a[j] = a_old * factor
                sc = compute_score(a)
                if sc < cur_sc:
                    cur_sc = sc
                else:
                    a[j] = a_old
                    break
            else:
                a[j] = a_old

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
            inits.append(np.clip(parent + 0.05 * rng.standard_normal(n), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, 3.0))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * x) ** 2 - 0.2, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.sin(2.0 * np.pi * 3.0 * x) ** 2, 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(6.0 * np.pi * x + 0.5) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.8 + 0.4 * np.cos(2.0 * np.pi * 5.0 * x) * np.cos(2.0 * np.pi * x), 0.01))

    for i in range(len(inits)):
        s = inits[i].sum()
        if s > 1e-12:
            inits[i] = inits[i] / s

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phase 1: GS-Fienup from diverse inits (25%)
    Phase 2: L-BFGS with exact gradient, multiple beta (45%)
    Phase 3: Peak-attack subgradient + L-BFGS (20%)
    Phase 4: Coordinate descent + final polish (10%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                if parent.sum() > 1e-6:
                    parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] score={sc:.8f} n={len(a)} t={time.monotonic()-t_start:.0f}s")
            return True
        return False

    n_primary = 6962
    if parent is not None and len(parent) >= 1024:
        n_primary = len(parent)

    print(f"Parent: {'yes n=' + str(n_primary) if parent is not None else 'no'}")
    print(f"Primary length: {n_primary}")

    # ============================================================
    # Phase 1: GS-Fienup (25% of time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total - 30.0)
    print(f"\n=== Phase 1: GS-Fienup (until t={phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n_primary, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        print(f"  Init {i}/{len(inits)}...")

        a1 = gs_fienup(a0, n_iters=300, alpha_init=0.8, alpha_final=0.08)
        sc = compute_score(a1)
        update_best(sc, a1, f"GS init={i}")

        if time.monotonic() < phase1_end - 5:
            a2 = gs_magnitude_only(a0, n_iters=300, alpha=0.4)
            sc = compute_score(a2)
            update_best(sc, a2, f"GS-mag init={i}")

        if time.monotonic() < phase1_end - 5:
            a3 = gs_adaptive(a1, n_iters=300)
            sc = compute_score(a3)
            update_best(sc, a3, f"GS-adapt init={i}")

    if best_a is None:
        best_a = np.ones(n_primary) / n_primary
        best_score = compute_score(best_a)

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: L-BFGS with exact gradient (45% of time)
    # ============================================================
    phase2_end = t_start + 0.70 * (total - 30.0)
    print(f"\n=== Phase 2: L-BFGS (until t={phase2_end - t_start:.0f}s) ===")

    # Strategy: start with large beta (smooth), decrease beta (sharper)
    # For each beta, run L-BFGS and also a perturbed restart
    betas = [5000, 2000, 1000, 500, 300, 200, 100, 50, 30]

    for beta in betas:
        if time.monotonic() > phase2_end:
            break

        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations with remaining time
        maxiter = min(5000, int(remaining * 50))
        maxfun = min(80000, int(remaining * 200))

        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        update_best(sc, a_opt, f"LBFGS b={beta} ({dt:.1f}s)")

        # Perturbed restart
        if time.monotonic() < phase2_end - 15:
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_primary))
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 1e-6:
                a_pert = a_pert / a_pert.sum()
            sc2, a_opt2 = run_lbfgs(a_pert, beta=beta,
                                    maxiter=max(500, maxiter // 2),
                                    maxfun=max(1000, maxfun // 2))
            update_best(sc2, a_opt2, f"LBFGS-pert b={beta}")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Peak-attack subgradient + L-BFGS (20% of time)
    # ============================================================
    phase3_end = t_start + 0.90 * (total - 30.0)
    print(f"\n=== Phase 3: Peak-attack (until t={phase3_end - t_start:.0f}s) ===")

    for top_m in [3, 5, 10, 20]:
        if time.monotonic() > phase3_end:
            break

        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break

        n_iters_pa = min(5000, int(remaining * 30))
        a_pa = peak_attack(best_a, n_iters=n_iters_pa, n_top=top_m)
        sc = compute_score(a_pa)
        update_best(sc, a_pa, f"PA top={top_m}")

        # Follow up with L-BFGS on the peak-attack result
        if time.monotonic() < phase3_end - 15 and sc < best_score + 0.01:
            sc2, a_opt = run_lbfgs(a_pa, beta=500,
                                   maxiter=3000, maxfun=50000)
            update_best(sc2, a_opt, f"PA+LBFGS top={top_m}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Coordinate descent + final polish (10% of time)
    # ============================================================
    print(f"\n=== Phase 4: Coord descent + Final polish ===")

    if time.monotonic() < deadline - 60:
        a_ca = coordinate_ascent_peak(best_a, n_sweeps=6, n_top=30)
        sc = compute_score(a_ca)
        update_best(sc, a_ca, "CoordAscent")

    # Final GS with gentle steps
    if time.monotonic() < deadline - 40:
        a_gs = gs_fienup(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.05)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, "FinalGS")

    # Final L-BFGS with very tight beta
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, beta=2000, maxiter=4000, maxfun=60000)
        update_best(sc, a_opt, "FinalLBFGS")

    # ============================================================
    # Try a different length if time permits
    # ============================================================
    if time.monotonic() < deadline - 120 and parent is not None:
        n_alt = 8192
        if abs(n_alt - n_primary) > 100:
            print(f"\n=== Bonus: Trying n={n_alt} ===")
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            a_alt = a_alt / a_alt.sum()

            a_alt_gs = gs_fienup(a_alt, n_iters=200, alpha_init=0.6, alpha_final=0.1)
            sc = compute_score(a_alt_gs)
            update_best(sc, a_alt_gs, f"Alt n={n_alt} GS")

            if time.monotonic() < deadline - 30:
                sc2, a_opt = run_lbfgs(a_alt_gs, beta=1000, maxiter=3000, maxfun=50000)
                update_best(sc2, a_opt, f"Alt n={n_alt} LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 6962

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
