"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: multi-peak subgradient with momentum, adaptive Fienup with spectral shaping,
Nesterov LSE with restarts, peak-targeted coordinate descent, and linear-system polish.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_sum

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def multi_peak_subgrad(a, K=10, alpha=0.01, momentum=0.9):
    """Subgradient targeting top K peaks simultaneously."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(2000):
        c = fast_conv(a)
        c_max = float(c.max())
        # Get top K peak lags
        top_k = np.argsort(c)[::-1][:K]
        # Compute subgradient: average of gradients at top peaks
        # grad of (conv)_k w.r.t. a[i] = 2 * a[(k-i) mod range] (if valid)
        grad = np.zeros(n)
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * a[k - j]
        # Normalize gradient by K
        grad /= K

        # Remove mean component (to preserve sum)
        grad_mean = grad.mean()
        grad = grad - grad_mean

        # Momentum update
        v = momentum * v - alpha * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Decay alpha
        alpha *= 0.999
        if alpha < 1e-6:
            break

    return best_a


def fienup_adaptive(a, n_iters=500):
    """Fienup with adaptive threshold and spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()
    # Target: slightly non-flat spectrum to encourage non-negativity
    # Boost low frequencies slightly
    freqs = np.arange(M_fft // 2 + 1)
    target_shape = 1.0 + 0.1 * np.exp(-freqs / (n * 0.1))

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = 2.0 - 1.0 * t  # 2.0 -> 1.0
        clip_power = 0.3 + 0.5 * t  # 0.3 -> 0.8

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target spectrum: mean * shape
        target_P = P_mean * target_shape
        above = P > target_P * thresh
        if above.any():
            scale = np.ones_like(P)
            # Reduce to target with soft clipping
            ratio = (target_P[above] * thresh) / P[above]
            scale[above] = 1.0 - clip_power * (1.0 - ratio)
            scale = np.clip(scale, 0.05, 1.0)

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def nesterov_lse_restart(a, max_time=60.0, n_restarts=3, beta_start=10.0, beta_end=20000.0):
    """Nesterov LSE with periodic restarts at higher beta."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    time_per_restart = max_time / n_restarts

    for restart in range(n_restarts):
        v = np.zeros(n)
        beta = beta_start * (beta_end / beta_start) ** (restart / max(1, n_restarts - 1))
        lr = 0.01 / (1 + restart * 0.5)
        mu = 0.9 + 0.02 * restart
        t_restart_start = time.monotonic()

        it = 0
        while time.monotonic() - t_restart_start < time_per_restart:
            it += 1
            a_look = a + mu * v
            a_look = np.maximum(a_look, 0.0)
            S_look = a_look.sum()
            if S_look < 1e-12:
                break
            a_look = a_look / S_look

            _, grad = lse_score_and_grad(a_look, beta)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break

            v = mu * v - lr * grad
            a = a + v
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S

            if it % 30 == 0:
                beta *= 1.01
                lr *= 0.998

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Perturb for next restart
        if restart < n_restarts - 1:
            a = best_a * (1.0 + 0.05 * np.random.default_rng(42 + restart).standard_normal(n))
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        if time.monotonic() - t_start > max_time:
            break

    return best_a


def peak_coordinate_descent(a, n_sweeps=10, n_top_peaks=20, n_top_coords=15):
    """Coordinate descent targeting top peak contributors with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(n_top_coords, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                best_local_sc = best_sc
                best_local_val = old_val
                for factor in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_local_val = a[j]
                        best_local_a = a_norm.copy()
                    a[j] = old_val

                if best_local_val != old_val:
                    a = best_local_a.copy()
                    best_sc = best_local_sc
                    best_a = a.copy()

        # Also try increasing small values at non-peak-contributing positions
        if sweep % 3 == 2:
            c = fast_conv(a)
            min_lag = np.argmin(c[:n])  # Look at first half
            # Try adding mass at positions that don't contribute to top peaks
            low_contrib = np.zeros(n)
            for k in peak_indices[:5]:
                k = int(k)
                for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                    low_contrib[j] += a[k - j]
            # Positions with low contribution to peaks
            candidates = np.argsort(low_contrib)[:20]
            for j in candidates:
                j = int(j)
                if a[j] > 0.1 * a.mean():
                    continue
                a[j] = a[j] + 0.01 * a.mean()
                S_tmp = float(a.sum())
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm
                else:
                    a[j] = max(0, a[j] - 0.01 * a.mean())

    return best_a


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_u_shape(n, depth=0.3):
    """U-shaped initialization: large at edges, small in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 - depth * np.sin(np.pi * x)
    return np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()


def build_comb(n, period=8, depth=0.5):
    """Comb-like initialization."""
    x = np.arange(n)
    a = 1.0 - depth * np.cos(2 * np.pi * x / period)
    return np.maximum(a, 0.05) / np.maximum(a, 0.05).sum()


def build_cosine_mod(n, freq=1, amp=0.4):
    """Cosine-modulated initialization."""
    x = np.arange(n) / n
    a = 1.0 + amp * np.cos(2 * np.pi * freq * x)
    return np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()


def build_bimodal(n, center_sep=0.3):
    """Two Gaussian bumps."""
    x = np.arange(n) / n
    c1 = 0.5 - center_sep / 2
    c2 = 0.5 + center_sep / 2
    sigma = 0.15
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    return a / a.sum()


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def linear_system_polish(a, n_iters=20, K=5):
    """
    Try to solve the KKT linear system for the top K peaks.
    For peak set P, find a such that (C_P * a)_i = const for all i, sum(a) = S.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S * S  # keep original scale
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        peak_lags = np.argsort(c)[::-1][:K]

        # Build matrix C: C[i,j] = sum_{k in P} 1[i+j == k]
        # This is sparse, so build it as a dense matrix only if n is small enough
        # For large n, use iterative methods
        C = np.zeros((n, n))
        for k in peak_lags:
            k = int(k)
            for i in range(max(0, k - n + 1), min(k, n - 1) + 1):
                j = k - i
                C[i, j] = 1.0

        # Solve: C @ a_new = lambda * 1, sum(a_new) = S, a_new >= 0
        # Use iterative approach: a_new = a - alpha * (C @ a - mean(C @ a))
        # This drives C@a toward being constant
        for sub_iter in range(50):
            Ca = C @ a
            Ca_mean = Ca.mean()
            residual = Ca - Ca_mean
            # Gradient of (Ca - mean)^2 w.r.t. a is 2*C*(Ca - Ca_mean)
            grad = 2.0 * C @ residual
            # Remove mean to preserve sum
            grad = grad - grad.mean()
            # Step
            alpha = S / (float(np.sum(residual * residual)) + 1e-10) * 0.5
            alpha = min(alpha, 0.1)
            a = a - alpha * grad
            a = np.maximum(a, 0.0)
            S_cur = float(a.sum())
            if S_cur < 1e-12:
                a = best_a.copy()
                break
            a = a * (S / S_cur)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # No improvement, break
            break

    return best_a


def simulated_annealing_step(a, current_sc, temperature, rng):
    """Try a random perturbation, accept if better or with probability exp(-delta/T)."""
    n = len(a)
    a_try = a.copy()
    # Perturb a random subset
    n_perturb = max(10, n // 10)
    indices = rng.choice(n, size=n_perturb, replace=False)
    a_try[indices] *= np.exp(rng.standard_normal(n_perturb) * temperature)
    a_try = np.maximum(a_try, 0.0)
    S = float(a_try.sum())
    if S < 1e-12:
        return a, current_sc
    a_try = a_try / S
    new_sc = compute_score(a_try)

    if new_sc < current_sc:
        return a_try, new_sc
    else:
        delta = new_sc - current_sc
        if rng.random() < np.exp(-delta / max(temperature, 1e-10)):
            return a_try, new_sc
        return a, current_sc

    print(f"\n=== Phase 3: Fienup variants ===")

    # Standard Fienup with momentum
    a = best_a.copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    M_fft = 1 << (2 * n - 1).bit_length()
    v = np.zeros(n)

    n_iters_fienup = 800
    for it in range(n_iters_fienup):
        if time.monotonic() > phase3_end:
            break
        t = it / max(1, n_iters_fienup - 1)
        momentum = 0.3 * t
        thresh = 2.5 - 1.5 * t
        clip = 0.2 + 0.7 * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    Fienup it={it}: {sc:.8f}")

    update_best(compute_score(a), a, "fienup-momentum")

    # Adaptive Fienup
    if time.monotonic() < phase3_end - 10:
        a_fa = fienup_adaptive(best_a, n_iters=300)
        update_best(compute_score(a_fa), a_fa, "fienup-adaptive")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak coordinate descent (15% time)
    # ============================================================
    phase4_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Peak coordinate descent ===")

    a_cd = peak_coordinate_descent(best_a, n_sweeps=15, n_top_peaks=15, n_top_coords=10)
    update_best(compute_score(a_cd), a_cd, "peak-cd")

    # Multi-peak subgradient
    if time.monotonic() < phase4_end - 20:
        a_ms = multi_peak_subgrad(best_a, K=10, alpha=0.005, momentum=0.85)
        update_best(compute_score(a_ms), a_ms, "multi-peak-subgrad")

    # Linear system polish
    if time.monotonic() < phase4_end - 30:
        a_ls = linear_system_polish(best_a, n_iters=10, K=3)
        update_best(compute_score(a_ls), a_ls, "linear-system")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish with increasing beta (15% time)
    # ============================================================
    phase5_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 5: L-BFGS polish ===")

    betas = [500, 2000, 5000, 15000, 30000]
    for beta in betas:
        if time.monotonic() > phase5_end:
            break
        remaining = phase5_end - time.monotonic()
        if remaining < 20:
            break
        maxiter = min(5000, int(remaining * 50))
        maxfun = min(100000, int(remaining * 500))
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"lbfgs-b={beta}")

    print(f"After Phase 5: best = {best_score:.8f}")

    # ============================================================
    # Phase 6: Simulated annealing exploration (5% time)
    # ============================================================
    phase6_end = deadline - 10
    print(f"\n=== Phase 6: SA exploration ===")

    if time.monotonic() < phase6_end - 10:
        a_sa = best_a.copy()
        sc_sa = compute_score(a_sa)
        temp = max(1e-4, (sc_sa - 1.0) * 0.01)
        n_steps = 2000

        for step in range(n_steps):
            if time.monotonic() > phase6_end:
                break
            a_sa, sc_sa = simulated_annealing_step(a_sa, sc_sa, temp, rng)
            temp *= 0.999
            if step % 200 == 0:
                update_best(sc_sa, a_sa, f"SA-step{step}")

        # Final Nesterov after SA
        if time.monotonic() < phase6_end - 5:
            a_final = nesterov_lse_restart(a_sa, max_time=10.0, n_restarts=2,
                                            beta_start=100.0, beta_end=30000.0)
            update_best(compute_score(a_final), a_final, "SA+nesterov")

    # ============================================================
    # Final polish
    # ============================================================
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs_lse(best_a, 50000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "final-lbfgs")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
