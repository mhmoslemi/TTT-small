Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
uses active-set subgradient with momentum, LSE-optimized L-BFGS-B, 
simulated annealing restarts, cosine-modulation parameterization, and multi-length search.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))

def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):

    PEN = 1e-3
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())

        g = g + 2.0 * PEN * (Sx - 1.0)

def _spectral_init(n, rng, style='flat'):

    return sig

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) LSE-smoothed L-BFGS-B with aggressive beta sweeps, (2) momentum-based
active-set subgradient with backtracking, (3) peak-targeting combinatorial moves,
(4) simulated annealing with convolution-aware perturbations, (5) multi-length search,
(6) cosine-modulation parameterization for local refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v_pen = PEN * (Sx - 1.0) ** 2
        g_pen = 2.0 * PEN * (Sx - 1.0)
        return v + v_pen, g + g_pen

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _momentum_subgrad(a0, max_iters, deadline, momentum=0.8, k_init=5, k_grow=3):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0 or not np.isfinite(dmax):
            break

        M = 2 * n - 1
        k = min(M, k_init + k_grow * it)
        idx = np.argpartition(d, M - k)[M - k:]
        active = np.zeros(M, dtype=bool)
        active[idx] = True
        n_act = int(active.sum())

        # Use softmax weighting for smoother active set
        z = d[active] - dmax
        e = np.exp(100.0 * z)
        se = e.sum()
        if se < 1e-30:
            w_act = np.ones(n_act) / n_act
        else:
            w_act = e / se

        w = np.zeros(M)
        w[active] = w_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_new = best_sc
        best_step_a = None
        best_v = v.copy()

        for scale in (0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1):
            eta = scale / (gnorm + 1e-12)
            v_new = momentum * v - eta * grad
            q = a + v_new
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new:
                best_new = sc
                best_step_a = q.copy()
                best_v = v_new.copy()

        if best_step_a is not None and best_new < best_sc:
            best_sc = best_new
            best_a = best_step_a.copy()
            a = best_step_a
            v = best_v
            no_improve = 0
        else:
            no_improve += 1
            v = 0.5 * v
            if no_improve > 10:
                break

    return best_sc, best_a


def _peak_targeting(a0, max_iters, deadline, rng):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())

        # Find top peaks
        n_peaks = min(20, 2 * n - 1)
        peak_idx = np.argsort(d)[-n_peaks:]

        best_new = best_sc
        best_step = None

        # Try reducing contributions to the highest peak
        j_star = int(peak_idx[-1])
        # Elements contributing to peak j_star: a[k] * a[j_star - k]
        valid_k = np.arange(max(0, j_star - n + 1), min(j_star + 1, n))
        partners = j_star - valid_k

        # Find the element with largest contribution
        contribs = a[valid_k] * a[partners]
        if contribs.max() < 1e-15:
            break

        max_contrib_idx = int(np.argmax(contribs))
        k_main = int(valid_k[max_contrib_idx])
        k_partner = int(partners[max_contrib_idx])

        # Try shifting mass from the main contributor
        for frac in (0.01, 0.02, 0.05, 0.1, 0.2):
            delta = frac * a[k_main]
            # Move to a random low-contribution position
            low_contrib = np.where(contribs < 0.1 * contribs.max())[0]
            if len(low_contrib) == 0:
                low_contrib = np.where(contribs < contribs.max())[0]
            if len(low_contrib) == 0:
                continue
            target = int(low_contrib[rng.integers(len(low_contrib))])
            k_target = int(valid_k[target])

            q = a.copy()
            q[k_main] = max(0.0, a[k_main] - delta)
            q[k_target] = a[k_target] + delta
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new:
                best_new = sc
                best_step = q.copy()

        # Also try uniform perturbation of high contributors
        for frac in (0.05, 0.1, 0.2):
            q = a.copy()
            for ki in valid_k:
                for pi in partners[valid_k == ki]:
                    pass  # skip complex logic
            # Simpler: reduce all high contributors proportionally
            high_mask = contribs > 0.5 * contribs.max()
            for ki_idx in np.where(high_mask)[0]:
                ki = int(valid_k[ki_idx])
                q[ki] = a[ki] * (1.0 - frac)
            # Distribute the removed mass to low contributors
            removed = a[high_mask.astype(int)].sum() - q[high_mask.astype(int)].sum() if high_mask.any() else 0
            low_mask = ~high_mask
            if low_mask.any() and removed > 0:
                q[valid_k[low_mask]] += removed / low_mask.sum()
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq > 1e-8:
                q = q / Sq
                sc = _score(q)
                if sc < best_new:
                    best_new = sc
                    best_step = q.copy()

        if best_step is not None and best_new < best_sc:
            best_sc = best_new
            best_a = best_step.copy()
            a = best_step
        else:
            if it > 5:
                break

    return best_sc, best_a


def _sa_restart(a0, deadline, rng, n_restarts=6):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    total_time = max(1.0, deadline - time.monotonic())
    time_per = total_time / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Convolution-aware perturbation
        c = _self_conv(a)
        n_peaks = min(20, len(c))
        peak_positions = np.argsort(c)[-n_peaks:]

        # Generate smooth noise
        noise = rng.standard_normal(n)
        k_smooth = max(3, n // 300)
        if k_smooth % 2 == 0:
            k_smooth += 1
        kern = np.exp(-0.5 * ((np.arange(k_smooth) - k_smooth // 2) / (k_smooth // 4 + 1)) ** 2)
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')

        # Cooling schedule
        temp = 0.08 * (1.0 - 0.5 * r / max(1, n_restarts))
        q = a + temp * noise
        q = _norm(q)

        # Quick optimize
        opt_end = min(deadline, time.monotonic() + time_per * 0.7)
        sc, a_new = _run_lbfgs(q, 5000, 200, max(1.0, opt_end - time.monotonic()), 400)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    SA restart {r}: {sc:.8f}")
        a = _norm(a_new)

    return best_sc, best_a


def _cosine_refine(a0, deadline, rng):
    n = len(a0)
    a0 = _norm(a0)
    best_a = a0.copy()
    best_sc = _score(a0)
    x = np.linspace(0, 1, n)

    def build_seq(params):
        mod = np.ones(n)
        n_terms = len(params) // 3
        for i in range(n_terms):
            amp = params[3 * i]
            freq = params[3 * i + 1]
            phase = params[3 * i + 2]
            mod = mod * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        mod = np.clip(mod, 0.01, None)
        result = a0 * mod
        S = result.sum()
        if S < 1e-12:
            return _norm(a0)
        return result / S

    for n_terms in [2, 4, 6, 8]:
        if time.monotonic() > deadline:
            break
        params0 = np.zeros(3 * n_terms)
        for i in range(n_terms):
            params0[3 * i] = 0.05
            params0[3 * i + 1] = float(i + 1)
            params0[3 * i + 2] = rng.uniform(-np.pi, np.pi)

        def cos_obj(p):
            seq = build_seq(p)
            return _score(seq)

        try:
            res = minimize(cos_obj, params0, method='Nelder-Mead',
                           options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-10,
                                    'adaptive': True})
            best_seq = build_seq(res.x)
            sc = _score(best_seq)
            if sc < best_sc:
                best_sc = sc
                best_a = best_seq.copy()
                print(f"    Cosine n_terms={n_terms}: {sc:.8f}")
        except Exception:
            pass

    return best_sc, best_a


def _multi_length(a0, deadline, rng):
    best_a = a0.copy()
    best_sc = _score(a0)
    n0 = len(a0)
    times = (deadline - time.monotonic())

    factors = [0.6, 0.8, 1.25, 1.5, 2.0]
    lengths = []
    for f in factors:
        nl = int(n0 * f)
        nl = max(2048, min(40000, nl))
        if nl != n0 and nl not in lengths:
            lengths.append(nl)

    if not lengths:
        return best_sc, best_a

    time_per = times / (len(lengths) + 1)

    for nl in lengths:
        if time.monotonic() > deadline:
            break
        a_r = _norm(_upsample(a0, nl))
        sc_init = _score(a_r)
        opt_end = min(deadline, time.monotonic() + time_per * 0.8)
        sc, a_new = _run_lbfgs(a_r, 3000, 100, max(1.0, opt_end - time.monotonic()), 500)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    Multi-length n={nl}: {sc:.8f}")

    return best_sc, best_a


def _spectral_init(n, rng, style=0):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    abs_f = np.abs(freqs)

    if style == 0:
        power = 1.0 / (1.0 + 0.05 * abs_f) ** 0.5
    elif style == 1:
        power = 1.0 / (1.0 + 0.1 * abs_f) ** 1.5
    elif style == 2:
        center = Nf // 4
        power = np.exp(-((abs_f - center) ** 2) / (2.0 * (Nf / 6) ** 2)) + 0.1
    elif style == 3:
        power = 1.0 + 0.4 * np.cos(abs_f * Nf / 15.0)
    else:
        power = np.ones(len(freqs))

    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.1):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

    print(f"Init scores (top 5): {[f'{s:.6f}({nm})' for s, nm, _ in scored[:5]]}")

    best_score = scored[0][0] if scored else 2.0
    best_seq = scored[0][2].copy() if scored else _norm(np.ones(target_n))

    total_budget = deadline - t_start

    # === Phase 1: LSE + L-BFGS-B (40% of time) ===
    print(f"\n=== Phase 1: LSE+LBFGS (budget={0.4*total_budget:.0f}s) ===")
    p1_end = t_start + 0.40 * total_budget

    top_cands = scored[:min(4, len(scored))]
    cfrac = [0.50, 0.30, 0.12, 0.08]
    for ci, (sc_i, nm_i, arr_i) in enumerate(top_cands):
        sub_end = t_start + (sum(cfrac[:ci + 1]) * (p1_end - t_start)) + t_start
        sub_end = min(sub_end, p1_end - 5)
        if time.monotonic() > sub_end:
            break
        print(f"  [{nm_i}] init={sc_i:.8f}")
        a_w = _norm(arr_i)
        stages = [(30, 3), (100, 8), (400, 25), (1500, 80), (5000, 250), (20000, 800)]
        for beta, k in stages:
            if time.monotonic() > sub_end - 5:
                break
            rem = sub_end - time.monotonic() - 5
            if rem < 8:
                break
            sc, a_new = _run_lbfgs(a_w, beta, k, rem * 0.8, 600)
            if sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                print(f"    [{nm_i}] beta={beta},k={k}: {sc:.8f} *** BEST ***")
            a_w = _norm(a_new)

    # === Phase 2: Momentum subgradient (15%) ===
    print(f"\n=== Phase 2: Momentum subgrad (score={best_score:.8f}) ===")
    p2_end = t_start + 0.55 * total_budget
    sc2, a2 = _momentum_subgrad(best_seq, 300, p2_end, momentum=0.85, k_init=8, k_grow=4)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  Momentum: {sc2:.8f} *** BEST ***")

    # === Phase 3: Peak targeting (10%) ===
    print(f"\n=== Phase 3: Peak targeting (score={best_score:.8f}) ===")
    p3_end = t_start + 0.65 * total_budget
    sc3, a3 = _peak_targeting(best_seq, 200, p3_end, rng)
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  Peak-target: {sc3:.8f} *** BEST ***")

    # === Phase 4: SA restarts (15%) ===
    print(f"\n=== Phase 4: SA restarts (score={best_score:.8f}) ===")
    p4_end = t_start + 0.80 * total_budget
    sc4, a4 = _sa_restart(best_seq, p4_end, rng, n_restarts=8)
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()
        print(f"  SA: {sc4:.8f} *** BEST ***")

    # === Phase 5: Cosine refinement (5%) ===
    print(f"\n=== Phase 5: Cosine refine (score={best_score:.8f}) ===")
    p5_end = t_start + 0.85 * total_budget
    sc5, a5 = _cosine_refine(best_seq, p5_end, rng)
    if sc5 < best_score:
        best_score = sc5
        best_seq = a5.copy()
        print(f"  Cosine: {sc5:.8f} *** BEST ***")

    # === Phase 6: Multi-length (5%) ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    p6_end = t_start + 0.90 * total_budget
    sc6, a6 = _multi_length(best_seq, p6_end, rng)
    if sc6 < best_score:
        best_score = sc6
        best_seq = a6.copy()
        print(f"  Multi-len: {sc6:.8f} *** BEST ***")

    # === Phase 7: Final high-precision polish (5%) ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    p7_end = deadline - 10
    for kf, bf in [(500, 20000), (1500, 100000), (3000, 300000)]:
        if time.monotonic() > p7_end - 20:
            break
        rem = p7_end - time.monotonic() - 20
        if rem < 8:
            break
        scf, af = _run_lbfgs(best_seq, bf, kf, rem * 0.6, 800)
        if scf < best_score:
            best_score = scf
            best_seq = af.copy()
            print(f"  Final LSE k={kf},b={bf}: {scf:.8f} *** BEST ***")

    if time.monotonic() < p7_end - 5:
        scf2, af2 = _momentum_subgrad(best_seq, 40, p7_end - 2, momentum=0.5, k_init=100, k_grow=50)
        if scf2 < best_score:
            best_score = scf2
            best_seq = af2.copy()
            print(f"  Final subgrad: {scf2:.8f} *** BEST ***")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
