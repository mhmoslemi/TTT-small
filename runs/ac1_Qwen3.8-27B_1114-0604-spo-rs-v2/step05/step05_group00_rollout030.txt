def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key improvements over previous:
1. Heavier emphasis on high-beta LSE refinement (70% compute)
2. Interleaved subgradient steps targeting the exact peak
3. Pairwise coordinate descent at peak lags
4. Simpler, more focused structure for reliable convergence
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and its gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lbfgs_fun(x, beta):
    """Wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lbfgs_fun, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def fienup_quick(a, n_iters=50):
    """Quick Fienup for spectral mixing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.03 + 0.12 * progress
        P_blend = (1.0 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * np.sqrt(P_blend / (P + 1e-30))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def subgradient_step(a, n_steps=500, step_init=0.01, decay=0.999, deadline=None):
    """
    Projected subgradient method directly targeting the peak lag.
    The subgradient of max(c) at the argmax point k* is:
    dc_{k*}/da_i = a_{k*-i} + a_{i-k*} (for valid indices)
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute subgradient of c_{k_star} w.r.t. a
        sg = np.zeros(n)
        if k_star >= 0:
            # c_k = sum_{j=0}^{n-1-k} a_j * a_{j+k}
            # dc_k/da_i = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
            sg[:n - k_star] += a[k_star:]
            sg[k_star:] += a[:n - k_star]
        else:
            k_abs = -k_star
            sg[n - k_abs:] += a[:k_abs]
            sg[:k_abs] += a[n - k_abs:]

        # Score gradient includes the S-dependence
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step = min(step, max(1e-10, (sc - sc_lb) / (gn * gn + 1e-30)))
        step = min(step, 0.03)

        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def peak_pairwise_cd(a, n_sweeps=20, n_top=120, deadline=None):
    """
    Pairwise coordinate descent at the peak lag.
    For peak lag k*, targets pairs (j, k*-j) with high a_j*a_{k*-j}.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        any_improved = False
        for idx in top_global:
            j = int(idx)
            partner = k_star - j
            if partner < 0 or partner >= n:
                continue
            a_j = a[j]
            a_p = a[partner]
            if a_j < 1e-14 and a_p < 1e-14:
                continue

            cur_sc = compute_score(a)

            # Try modifying element j
            for f in [0.4, 0.6, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5]:
                a[j] = a_j * f
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = a_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < cur_sc:
                    cur_sc = sc
                    a[j] = a_j * f
                    any_improved = True
                else:
                    a[j] = a_j

            # Try modifying partner element
            a_p_cur = a[partner]
            for f in [0.4, 0.6, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5]:
                a[partner] = a_p_cur * f
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[partner] = a_p_cur
                    continue
                sc = compute_score(a / S_tmp)
                if sc < cur_sc:

                    a[partner] = a_p_cur * f

                else:
                    a[partner] = a_p_cur

        if S > 1e-12:

        sc = compute_score(a)

        # Occasional Fienup mixing to escape local minima
        if beta in [100, 1000, 5000] and time.monotonic() < p2_end - 15:
            a_mix = fienup_quick(work, n_iters=20)
            sc_mix = compute_score(a_mix)
            if sc_mix < compute_score(work):
                try_update(sc_mix, a_mix, "FIENUP@b{}".format(beta))
                work = a_mix.copy()

    # ============================================================
    # Phase 3: Intensive coordinate descent + polish (20%)
    # ============================================================
    p3_end = t_start + 0.90 * (total - 10)
    print("\n=== Phase 3: Peak CD (20%) ===")

    for rnd in range(5):
        if time.monotonic() > p3_end:
            break

        # Pairwise CD
        remaining = p3_end - time.monotonic()
        n_swp = max(5, int(remaining / 15))
        a_cd = peak_pairwise_cd(best_a, n_sweeps=n_swp, n_top=120, deadline=p3_end)
        sc = compute_score(a_cd)
        try_update(sc, a_cd, "PAIR-CD r{}".format(rnd))

        # Single CD on more elements
        if time.monotonic() < p3_end - 10:
            a_cd2 = peak_single_cd(best_a, n_sweeps=10, n_top=200, deadline=p3_end)
            sc2 = compute_score(a_cd2)
            try_update(sc2, a_cd2, "SINGLE-CD r{}".format(rnd))

        # L-BFGS polish at very high beta
        if time.monotonic() < p3_end - 15:
            sc3, a3 = run_lbfgs(best_a, 50000, maxiter=2000, maxfun=20000)
            try_update(sc3, a3, "POLISH r{}".format(rnd))

        # Subgradient step
        if time.monotonic() < p3_end - 10:
            a_sg = subgradient_step(best_a, n_steps=200, step_init=0.005, deadline=p3_end)
            sc_sg = compute_score(a_sg)
            try_update(sc_sg, a_sg, "SG-polish r{}".format(rnd))

    # ============================================================
    # Phase 4: Restarts + final (10%)
    # ============================================================
    print("\n=== Phase 4: Restarts + Final ===")

    n_restarts = 4
    for r in range(n_restarts):
        if time.monotonic() > deadline - 20:
            break

        n = len(best_a)
        x = np.arange(n, dtype=np.float64) / n
        scale = 0.006 + 0.003 * r
        pert = np.zeros(n)
        for m in range(1, 5):
            freq = int(rng.integers(1, 30))
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            pert += (scale / np.sqrt(float(m))) * np.cos(2.0 * np.pi * freq * x + phase)
        a_pert = best_a * (1.0 + pert)
        a_pert = np.maximum(a_pert, 0.0)
        S = float(a_pert.sum())
        if S > 1e-12:
            a_pert = a_pert / S

        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=2000, maxfun=20000)
        print("  R{}: {:.8f}".format(r, sc))
        try_update(sc, a_opt, "RESTART {}".format(r))

        if sc < best_score + 0.002 and time.monotonic() < deadline - 20:
            a_cd = peak_pairwise_cd(a_opt, n_sweeps=8, n_top=100, deadline=deadline - 5)
            sc_cd = compute_score(a_cd)
            try_update(sc_cd, a_cd, "R{}+CD".format(r))

    # Final L-BFGS at highest beta
    if time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        maxiter = max(500, int(remaining * 4))
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 5)
        try_update(sc, a_fin, "FINAL-LBFGS-100k")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd = peak_pairwise_cd(best_a, n_sweeps=5, n_top=150, deadline=deadline - 2)
        sc = compute_score(a_cd)
        try_update(sc, a_cd, "FINAL-CD")

    # Final subgradient
    if time.monotonic() < deadline - 3:
        a_sg = subgradient_step(best_a, n_steps=100, step_init=0.003, deadline=deadline - 1)
        sc = compute_score(a_sg)
        try_update(sc, a_sg, "FINAL-SG")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print("\n" + "=" * 60)
    print("FINAL: internal={:.8f}, verified={:.8f}".format(best_score, verified))
    print("Length: {}, Time: {:.1f}s / {}s".format(len(out), elapsed, total))
    print("=" * 60)
    return out
