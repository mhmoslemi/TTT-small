from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)

    S = float(a.sum())

        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_lp(a, M):

"""
LP-based multiplicative weights optimizer with L-BFGS polishing.
Key innovation: uses the von Neumann minimax theorem to transform the 
inner LP into a zero-sum game, solved via multiplicative weights (exponential 
smoothing). This avoids the need for a large LP solver and converges 
naturally to the optimal non-palindromic structure.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def fast_corr(p, a):
    """Compute (A^T p) where A is the convolution matrix with 'a'.
    (A^T p)[j] = sum_k p[k] * a[k-j] for valid k,j
    = np.correlate(p, a, 'valid')
    """
    return np.correlate(p, a, mode='valid')

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

# Global for L-BFGS callback
_current_beta = 1000.0

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def multiplicative_weights(a_init, n_total_iters, eta0, eta_decay, a_update_rate, 
                           deadline, rng, label="MW"):
    """
    Main multiplicative weights optimizer.
    
    Maintains three sequences: a (current), g (LP direction), p (dual distribution).
    Each iteration:
    1. Compute v = convolve(a, g)
    2. Update p: p ∝ exp(eta * v)
    3. Compute u = A^T p (correlation)
    4. Update g: g ∝ exp(-eta * u)  
    5. Update a: a = (1-rate)*a + rate*g, normalize
    
    Returns best a found.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    
    g = a.copy()
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    rate = a_update_rate
    
    t_start = time.monotonic()
    for it in range(n_total_iters):
        if time.monotonic() > deadline:
            break
        
        # Step 1: v = convolve(a, g)
        v = fast_conv(a, g)
        
        # Step 2: Update p
        v_shift = v - v.max()
        p = np.exp(eta * v_shift)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = p / p_sum
        
        # Step 3: u = A^T p
        u = fast_corr(p, a)
        
        # Step 4: Update g
        u_shift = u - u.min()
        g = np.exp(-eta * u_shift)
        g_sum = g.sum()
        if g_sum < 1e-300:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        # Step 5: Update a
        a = (1 - rate) * a + rate * g
        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Track best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 1000 == 0 or sc < 1.51:
                print(f"    [{label}] {sc:.10f} (it={it}, t={time.monotonic()-t_start:.0f}s)")
        
        # Decay
        eta *= eta_decay
        rate = max(0.01, rate * 0.999)
    
    return best_a
