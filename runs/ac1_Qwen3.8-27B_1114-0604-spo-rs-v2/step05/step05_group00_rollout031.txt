    # This is the cross-correlation.

    # where x_rev = x flipped
    # Actually, I think the correct formula is:
    # grad_i = 2 * sum_k p_k x[k-i] for valid k-i
    # = 2 * (p cross-correlated with x) at lag i
    # Using FFT: cross_corr(p, x)[i] = ifft(fft(p) * conj(fft(x)))[i]

    # 
    # And R[i] = sum_m p[m+i] x[m] (where 0 <= m < n and 0 <= m+i < 2n-1)
    # = sum_{m=0}^{n-1} p[m+i] x[m] (since p is only nonzero for [0, 2n-2])
    # 
    # So R[i] = conv(p, x_rev)[n-1+i] for i >= 0? Let me check:
    # conv(p, x_rev)[n-1+i] = sum_j p[j] x[j-i] (for j from i to min(n-1, 2n-2) with j >= i)
    # = sum_{j=i}^{n-1} p[j] x[j-i]
    # Let m = j-i: = sum_{m=0}^{n-1-i} p[m+i] x[m]
    # 
    # R[i] = sum_{m=0}^{n-1} p[m+i] x[m] (valid when 0 <= m+i < 2n-1, i.e., m < 2n-1-i)
    # For i >= 0: m ranges from 0 to min(n-1, 2n-2-i)
    # If i <= n-1: m goes up to n-1 (since n-1 < 2n-2-i for i >= 0)
    # So R[i] = sum_{m=0}^{n-1} p[m+i] x[m] for i in [0, n-1]
    # 
    # And conv(p, x_rev)[n-1+i] = sum_{m=0}^{n-1-i} p[m+i] x[m]
    # 
    # These are different! R[i] sums m from 0 to n-1, while the convolution only goes to n-1-i.
    # The difference is: R[i] - conv(n-1+i) = sum_{m=n-i}^{n-1} p[m+i] x[m]
    # 
    # Hmm, so the formula in the previous code might be slightly off? Let me think...
    # 
    # Actually wait, I think I need to be more careful. p has length 2n-1 (indices 0 to 2n-2).
    # x has length n (indices 0 to n-1).
    # x_rev has length n (indices 0 to n-1), x_rev[j] = x[n-1-j].
    # 
    # Linear convolution of p (length 2n-1) and x_rev (length n) has length 3n-2.
    # conv[k] = sum_j p[j] * x_rev[k-j] (where 0 <= j < 2n-1 and 0 <= k-j < n)
    # 
    # For k = n-1+i (where i ranges from 0 to n-1):
    # conv[n-1+i] = sum_j p[j] * x_rev[n-1+i-j]
    # Valid when 0 <= n-1+i-j < n, i.e., i-j+1 <= n-1+i-j... 
    # 0 <= n-1+i-j means j <= n-1+i
    # n-1+i-j < n means j > i-1, i.e., j >= i
    # So j ranges from max(0, i) to min(2n-2, n-1+i)
    # = i to n-1+i (since i <= n-1, n-1+i <= 2n-2)
    # 
    # x_rev[n-1+i-j] = x[n-1-(n-1+i-j)] = x[j-i]
    # 
    # So conv[n-1+i] = sum_{j=i}^{n-1+i} p[j] x[j-i]
    # Let m = j-i: = sum_{m=0}^{n-1} p[m+i] x[m]
    # 
    # This equals R[i]! (Since p[m+i] is defined for 0 <= m+i < 2n-1, and m < n, i < n, so m+i < 2n-1 is automatic.)
    # 
    # Great, so the formula is correct:
    # grad_i = 2 * conv(p, x_rev)[n-1+i] for i = 0, ..., n-1
    # = 2 * conv_res[n-1:2n-1]
    
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(a, M, max_time=60):
    """
    Solve the LP: max sum(b) s.t. b >= 0, ||a*b||_inf <= M.
    Returns the optimal b (normalized to sum=1) or None if failed.
    """
    n = len(a)
    m = 2 * n - 1
    
    # Build sparse constraint matrix
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # For row k, non-zero columns are i in [max(0, k-n+1), min(k, n-1)]
    
    # For large n, this might be too big. Use a threshold.
    if n > 4096:
        # Use a reduced version: only use every other constraint
        # Or just skip the LP
        return None
    
    A = lil_matrix((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # A[k, i_start:i_end+1] = a[k - i_end : k - i_start + 1][::-1]
        # For i from i_start to i_end: A[k,i] = a[k-i]
        # a[k-i_end], a[k-i_end+1], ..., a[k-i_start]
        vals = a[k - i_start:k - i_end + 1][::-1]
        for idx, v in enumerate(vals):
            if v > 1e-15:
                A[k, i_start + idx] = v
    
    A = A.tocsr()
    c = -np.ones(n)
    b_ub = np.full(m, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max_time})
    
    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0)
        S = b.sum()
        if S > 1e-12:
            return b / S
    return None


def peak_coord_descent_bisection(a, n_sweeps=20, n_top=200, tol=1e-10):
    """
    Aggressive coordinate descent with bisection on optimal multiplier.
    For each top-contributing element at the peak lag, find the optimal
    multiplier via bisection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_val = c[k_star]
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        improved = False
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            
            # Bisection to find optimal multiplier
            lo, hi = 0.01, 3.0
            best_mult = 1.0
            best_local_sc = compute_score(a)
            
            for _ in range(20):
                mid = (lo + hi) / 2.0
                a[j] = a_old_j * mid
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    hi = mid
                    continue
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_mult = mid
                    hi = mid
                else:
                    lo = mid
                a[j] = a_old_j
            
            if best_mult != 1.0:
                a[j] = a_old_j * best_mult
                improved = True
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc - tol:
            best_sc = sc
            best_a = a.copy()
        
        if not improved:
            break

    return best_a


def spectral_shaping(a, n_iters=200, strength=0.3):
    """
    Directly shape the spectrum to reduce autocorrelation peaks.
    In the frequency domain, the autocorrelation peak corresponds to
    spectral concentration. We flatten the spectrum iteratively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_sqrt = np.sqrt(P + 1e-30)
        P_mean = float(P_sqrt.mean())
        
        # Blend towards flat spectrum
        alpha = strength * (1.0 - 0.5 * it / n_iters)
        P_target = np.ones_like(P_sqrt) * P_mean
        P_new = (1 - alpha) * P_sqrt + alpha * P_target
        P_new = np.maximum(P_new, 1e-20)
        
        # Preserve phase
        A_new = A * (P_new / P_sqrt)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolated_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return result


def multi_scale_optimize(a_best, budget_s, t_start, deadline):
    """
    Try different lengths and find the best one.
    """
    n_orig = len(a_best)
    candidate_lengths = []
    
    # Try powers of 2 near the original length
    for factor in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        n_try = int(n_orig * factor)
        # Round to nearest power of 2 for FFT efficiency
        n_pow2 = 1 << max(10, (n_try - 1).bit_length())
        if n_pow2 not in candidate_lengths and n_pow2 <= 32768:
            candidate_lengths.append(n_pow2)
    
    # Also try the original length
    if n_orig not in candidate_lengths:
        candidate_lengths.append(n_orig)
    
    candidate_lengths = sorted(set(candidate_lengths))
    
    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_orig:
            continue  # already optimized
        
        # Interpolate
        a_interp = interpolated_to_length(a_best, n_try)
        a_interp = np.maximum(a_interp, 0.0)
        S = a_interp.sum()
        if S > 1e-12:
            a_interp = a_interp / S
        
        # Quick L-BFGS
        remaining = deadline - 60 - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 10))
        sc, a_opt = run_lbfgs(a_interp, 2000, maxiter=maxiter, maxfun=maxiter * 15)
        
        # Compare
        sc_orig = compute_score(a_best)
        if sc < sc_orig:
            print(f"  Multi-scale: n={n_try} gives {sc:.8f} < {sc_orig:.8f}")
            a_best = a_opt
    
    return a_best


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    1. Load parent, initial assessment
    2. Spectral shaping for quick global improvement
    3. Multi-beta L-BFGS with LSE (main workhorse)
    4. LP-based direction finding (for n <= 4096)
    5. Aggressive peak coordinate descent with bisection
    6. Multi-scale exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0  # 60s safety margin

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent available, starting from uniform n=4096")

    print(f"Starting best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Spectral shaping (5% of time)
    # ==================================================================
    phase1_end = t_start + 0.05 * (total_budget - 60.0)
    print(f"\n=== Phase 1: Spectral shaping ===")

    a_sp = spectral_shaping(best_a, n_iters=100, strength=0.2)
    sc = compute_score(a_sp)
    update(sc, a_sp, "SPECTRAL")

    # ==================================================================
    # Phase 2: Multi-beta L-BFGS (35% of time)
    # ==================================================================
    phase2_end = t_start + 0.40 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Multi-beta L-BFGS ===")

    a_work = best_a.copy()
    
    # Beta schedule: start moderate, increase
    betas = [200, 500, 1000, 2000, 4000, 8000, 15000, 25000, 50000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(20000, int(remaining * 20))
        maxfun = min(400000, int(remaining * 400))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)", flush=True)
        if update(sc, a_opt, f"LBFGS b={beta}"):
            a_work = a_opt.copy()

    # Second pass on best
    if time.monotonic() < phase2_end - 120:
        for beta in [5000, 20000, 80000]:
            if time.monotonic() > phase2_end - 60:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=15000, maxfun=300000)
            update(sc, a_opt, f"LBFGS-2 b={beta}")

    # ==================================================================
    # Phase 3: LP-based direction (15% of time, only for n <= 4096)
    # ==================================================================
    phase3_end = t_start + 0.55 * (total_budget - 60.0)
    print(f"\n=== Phase 3: LP-based direction ===")

    if len(best_a) <= 4096 and time.monotonic() < phase3_end:
        a_work = best_a.copy()
        S = a_work.sum()
        a_work = a_work / S
        c = fast_conv(a_work)
        M_val = float(c.max())
        
        print(f"  Running LP (n={len(a_work)}, M={M_val:.6e})...")

"""
Multi-phase optimizer combining adaptive LSE gradient descent, LP-based direction finding,
aggressive peak coordinate descent with bisection, spectral shaping, and multi-scale exploration.
Key innovation: LP direction finding from the state-of-the-art formulation, combined with
very high-beta LSE optimization and coordinate descent with bisection for peak reduction.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via FFT: grad_i = 2 * sum_k p_k x[k-i]
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(a, M, max_time=120):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns normalized b or None.
    """
    n = len(a)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            val = a[k - i]
            if val > 1e-15:
                A[k, i] = val
    A = A.tocsr()
    c_vec = -np.ones(n)
    b_ub = np.full(m, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            S_b = b.sum()
            if S_b > 1e-12:
                return b / S_b
    except Exception:
        pass
    return None


def peak_coord_bisection(a, n_sweeps=25, n_top=300):
    """Aggressive coordinate descent with bisection on optimal multiplier per element."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        if contribs.sum() < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        improved = False
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue

            # Bisection for optimal multiplier in [0.01, 3.0]
            lo, hi = 0.01, 3.0
            best_mult = 1.0
            best_local_sc = compute_score(a)

                mid = (lo + hi) / 2.0
                a[j] = a_old_j * mid
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    hi = mid
                    continue
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_mult = mid
                    hi = mid
                else:
                    lo = mid
                a[j] = a_old_j

            if best_mult != 1.0:
                a[j] = a_old_j * best_mult
                improved = True

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        if not improved:
            break

    return best_a


def spectral_shaping(a, n_iters=150, strength=0.25):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P_sqrt = np.sqrt(A.real ** 2 + + A.imag ** 2 + 1e-30)
        P_mean = float(P_sqrt.mean())
        alpha = strength * (1.0 - 0.3 * it / n_iters)
        P_target = np.full_like(P_sqrt, P_mean)
        P_new = (1.0 - alpha) * P_sqrt + alpha * P_target
        P_new = np.maximum(P_new, 1e-20)
        A_new = A * (P_new / P_sqrt)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=100):
    """Fienup with quantile-based spectral thresholding."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:

    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        q = 0.80 + 0.15 * (it / n_iters)
        thresh = np.quantile(P, q)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

                if sc < best_score:
                    best_score = sc
                    best_a = a_opt.copy()
                    print(f"  LP+LBFGS: {sc:.8f} ** NEW BEST **", flush=True)

    # ==================================================================
    # Phase 4: Peak coordinate descent with bisection (20%)
    # ==================================================================
    phase4_end = t_start + 0.70 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Peak coord descent (t<{phase4_end - t_start:.0f}s) ===", flush=True)

    # Multiple rounds with increasing aggressiveness
    for round_idx in range(3):
        if time.monotonic() > phase4_end - 60:
            break
        n_sweeps = [30, 50, 80][round_idx]
        n_top = [200, 400, 600][round_idx]
        a_cd = peak_coord_bisection(best_a, n_sweeps=n_sweeps, n_top=n_top)
        sc = compute_score(a_cd)
        label = f"CD-r{round_idx}(sweeps={n_sweeps},top={n_top})"
        print(f"  {label}: {sc:.8f}", flush=True)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
            print(f"    ** NEW BEST **", flush=True)

        # L-BFGS polish after coordinate descent
        if time.monotonic() < phase4_end - 90:
            sc2, a2 = run_lbfgs(best_a, 20000, maxiter=10000, maxfun=200000)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
                print(f"  CD+LBFGS: {sc2:.8f} ** NEW BEST **", flush=True)

    # ==================================================================
    # Phase 5: Multi-scale + restarts (15%)
    # ==================================================================
    phase5_end = t_start + 0.85 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Multi-scale + restarts (t<{phase5_end - t_start:.0f}s) ===", flush=True)

    # Try different lengths
    n_orig = len(best_a)
    test_lengths = set()
    for factor in [0.5, 1.0, 1.5, 2.0]:
        n_try = int(n_orig * factor)
        n_pow2 = 1 << max(12, (n_try - 1).bit_length())
        if 1024 <= n_pow2 <= 32768:
            test_lengths.add(n_pow2)
    test_lengths.discard(n_orig)
    
    for n_try in sorted(test_lengths):
        if time.monotonic() > phase5_end - 60:
            break
        a_interp = interpolate_to_length(best_a, n_try)
        a_interp = np.maximum(a_interp, 0.0)
        S_i = a_interp.sum()
        if S_i > 1e-12:
            a_interp = a_interp / S_i
        
        remaining_md = phase5_end - time.monotonic() - 30
        if remaining_md < 30:
            break
        maxiter = min(5000, int(remaining_md * 10))
        sc, a_opt = run_lbfgs(a_interp, 5000, maxiter=maxiter, maxfun=maxiter * 12)
        label = f"MULTI-SCALE n={n_try}"
        print(f"  {label}: {sc:.8f}", flush=True)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    ** NEW BEST (length change!) **", flush=True)

    # Multi-restart
    n_restarts = 10
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 30:
            break
        n = len(best_a)
        if r % 3 == 0:
            a_pert = best_a * (1.0 + 0.008 * (r + 1) * rng.standard_normal(n))
        elif r % 3 == 1:
            x = np.arange(n) / n
            pert = np.zeros(n)
            for m in range(1, 5):
                freq = rng.integers(1, 25)
                ph = 2.0 * np.pi * rng.uniform(0, 1)
                pert += (0.012 / np.sqrt(m)) * np.cos(2.0 * np.pi * freq * x + ph)
            a_pert = best_a * (1.0 + pert)
        else:
            # Peak-targeted
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_s = max(0, k_star - n + 1)
            j_e = min(k_star, n - 1)
            n_idx = min(100, j_e - j_s + 1)
            indices = rng.choice(range(j_s, j_e + 1), size=n_idx, replace=False)
            for idx in indices:
                a_pert[idx] *= rng.uniform(0.6, 1.4)
        
        a_pert = np.maximum(a_pert, 0.0)
        S_p = a_pert.sum()
        if S_p > 1e-12:
            a_pert = a_pert / S_p
        
        remaining_rr = phase5_end - time.monotonic() - 30
        if remaining_rr < 20:
            break
        maxiter = min(4000, int(remaining_rr * 12))
        if maxiter < 500:
            break
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=maxiter * 12)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  RESTART {r}: {sc:.8f} ** NEW BEST **", flush=True)
        else:
            print(f"  RESTART {r}: {sc:.8f}", flush=True)

    # ==================================================================
    # Phase 6: Final high-beta polish (15%)
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===", flush=True)

    # Highest beta L-BFGS
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = int(remaining * 15)
        maxiter = max(maxiter, 5000)
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 15)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
            print(f"  FINAL-LBFGS-100k: {sc:.8f} ** NEW BEST **", flush=True)
        else:
            print(f"  FINAL-LBFGS-100k: {sc:.8f}", flush=True)

    # One more very high beta
    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 300000, maxiter=5000, maxfun=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
            print(f"  FINAL-LBFGS-300k: {sc:.8f} ** NEW BEST **", flush=True)

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_cd = peak_coord_bisection(best_a, n_sweeps=15, n_top=500)
        sc = compute_score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
            print(f"  FINAL-CD: {sc:.8f} ** NEW BEST **", flush=True)

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return out
