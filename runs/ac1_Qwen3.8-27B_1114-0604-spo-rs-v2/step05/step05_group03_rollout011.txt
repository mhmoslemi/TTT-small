High-precision optimizer for flat-autocorrelation non-negative sequences.
Uses intensive LP-based alternating with adaptive FISTA, high-beta L-BFGS,
and peak-specific subgradient refinement to achieve scores near 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of LSE w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fista_lp_solve(a, M, n_fista_iters, beta, x_init=None):
    """Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on LSE-smoothed surrogate."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) * (M * n / float(cross_conv(a, np.ones(n)).sum() + 1e-30))
        x = np.clip(x, 0, None)
    
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    best_x = x.copy()
    best_sum = float(x.sum())
    
    for it in range(n_fista_iters):
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L
        
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
        
        # Track best
        x_sum = float(x.sum())
        if x_sum > best_sum:
            best_sum = x_sum
            best_x = x.copy()
    
    return best_x

    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        L_val = c_max
        p = np.ones(len(c)) / len(c)
    else:
        log_sum = z_shifted.max() + np.log(exp_sum)
        L_val = c_max + log_sum / beta
        p = exp_z / exp_sum
    score = 2.0 * n * L_val / (S * S)
    rev_a = a[::-1]
    M2 = 2 * len(c) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1: 2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def _lbfgs_fn(x):
    return lse_score_and_grad(x, _GLB_BETA)

def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    global _GLB_BETA
    _GLB_BETA = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(_lbfgs_fn, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def fista_lp(a, M, n_iters, beta, x_init=None):
    """FISTA solver for: max sum(x) s.t. cross_conv(a,x) <= M, x >= 0."""
    n = len(a)
    M_conv = 2 * n - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64).copy(), 0.0)
    else:
        x = np.ones(n) / n
    
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.dot(a, a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    best_x = x.copy()
    best_sum = float(x.sum())
    
    for it in range(n_iters):
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L
        
        Xf = np.fft.rfft(x, M_fft)
        c = np.fft.irfft(A_fft * Xf, M_fft)[:M_conv]
        
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(M_conv) / M_conv
        else:
            p = exp_z / exp_sum
        
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:M_conv - n + 1]
        grad = a * sw - 1.0
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
        
        s = float(x.sum())
        if s > best_sum:
            best_sum = s
            best_x = x.copy()
    
    return best_x

def golden_search_t(a, x):
    """Find best mixing parameter t for a_new = (1-t)*a + t*x."""
    t_lo, t_hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    t1 = t_hi - gr * (t_hi - t_lo)
    t2 = t_lo + gr * (t_hi - t_lo)
    
    def sc_at(t):
        a_try = (1.0 - t) * a + t * x
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            return np.inf
        a_try = a_try / St
        return compute_score(a_try)
    
    sc1 = sc_at(t1)
    sc2 = sc_at(t2)
    
    for _ in range(12):
        if sc1 < sc2:
            t_hi = t2
            t2 = t1
            sc2 = sc1
            t1 = t_hi - gr * (t_hi - t_lo)
            sc1 = sc_at(t1)
        else:
            t_lo = t1
            t1 = t2
            sc1 = sc2
            t2 = t_lo + gr * (t_hi - t_lo)
            sc2 = sc_at(t2)
    
    return (t_lo + t_hi) / 2.0
