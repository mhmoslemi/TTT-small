"""
Optimized step function search using LSE-smoothed Nesterov acceleration with
backtracking line search, peak-directed coordinate descent, and Fienup refinement.
Targets score < 1.5027 for sequences of length ~6962.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient via FFT.
    Returns (lse_score, gradient) for the true score 2n*LSE/(sum)^2.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum

    # Gradient: 2n * [2*(a cross-corr p)_i / S^2 - 2*L / S^3]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    cross = conv_res[n - 1:2 * n - 1][::-1]  # (a cross-corr p)_i

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (4.0 * n / S2) * cross - (4.0 * n * L / S3)
    return score, grad


def lse_grad_only(a, beta):
    """Compute only the gradient (faster when score not needed)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    cross = conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n / S2) * cross - (4.0 * n * L / S3)
    return grad


def nesterov_lse_bt(a, max_time, beta_start=15.0, beta_end=8000.0,
                    lr_init=0.02, mu_init=0.85, mu_max=0.96):
    """
    Nesterov-accelerated projected gradient with backtracking line search
    on the TRUE (non-smoothed) objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    beta = beta_start
    lr = lr_init
    mu = mu_init
    no_improve_count = 0
    patience = 300

    it = 0
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(mu_max, mu_init + 0.00002 * it)
        beta = min(beta * 1.003, beta_end)

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Gradient at lookahead
        grad = lse_grad_only(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search on true objective
        step = lr / (gnorm + 1e-15) * gnorm  # initial step = lr (in gradient direction)
        # Actually, let's use: trial = a - lr * grad (normalized)
        # We'll do: a_trial = a - step * grad, project, normalize
        found = False
        sc_current = compute_score(a)
        for _ in range(30):
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_t = float(a_trial.sum())
            if S_t < 1e-12:
                break
            a_trial = a_trial / S_t
            sc_trial = compute_score(a_trial)
            if sc_trial < sc_current - 1e-14:
                found = True
                break
            step *= 0.5
            if step < 1e-15:
                break

        if not found:
            # Stuck - try a different step
            step = lr * 0.01
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_t = float(a_trial.sum())
            if S_t < 1e-12:
                break
            a_trial = a_trial / S_t
            sc_trial = compute_score(a_trial)
            if sc_trial < sc_current:
                found = True
            else:
                no_improve_count += 1
                if no_improve_count > patience:
                    # Restart momentum, increase beta more aggressively
                    v = np.zeros(n)
                    beta = min(beta * 2.0, beta_end)
                    lr *= 0.5
                    no_improve_count = 0
                lr *= 0.95
                continue

        no_improve_count = 0
        # Update velocity and position
        v = mu * v + (a_trial - a)
        a = a_trial

        # Decay learning rate slowly
        lr *= 0.9997

        # Track best
        if it % 20 == 0 or found:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def nesterov_lse_simple(a, n_iters=2000, beta_start=20.0, beta_end=5000.0,
                        lr_start=0.015, lr_end=0.001, mu_start=0.85, mu_end=0.95):
    """Simpler Nesterov LSE with fixed iteration count (no line search)."""

    if S < 1e-12:

    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start + (lr_end - lr_start) * t
        mu = mu_start + (mu_end - mu_start) * t

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)

        a = np.maximum(a, 0.0)
        S = float(a.sum())

            break
        a = a / S

        if it % 50 == 0:

                best_a = a.copy()

    sc = compute_score(a)

    return best_a


def peak_shave(a, max_time, n_top_pairs=8, n_trials=6):
    """
    Targeted peak reduction: identify top contributing pairs to the peak lag,
    and try reducing the larger element in each pair.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - t_start < max_time and no_improve < max_no_improve:
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Find contributing pairs for this peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        # Products a[j] * a[j+k_star] for j in range
        js = np.arange(j_start, j_end + 1)
        products = a[js] * a[js + k_star]
        top_idx = np.argsort(products)[::-1][:n_top_pairs]

        improved = False
        for idx in top_idx:
            j = int(js[idx])
            j2 = int(js[idx] + k_star)

            if a[j] < 1e-10 and a[j2] < 1e-10:
                continue

            # Try reducing the larger element
            candidates = []
            if a[j] >= a[j2]:
                candidates.append(j)
                if a[j2] > 1e-10:
                    candidates.append(j2)
            else:
                candidates.append(j2)
                if a[j] > 1e-10:
                    candidates.append(j)

            for target in candidates:
                old_val = a[target]
                if old_val < 1e-12:
                    continue

                # Try several reduction factors
                for factor in [0.8, 0.6, 0.4, 0.2, 1.2, 1.5]:
                    a[target] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[target] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Try a different peak (second highest)
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > 1 and c_sorted[1] > 0.9 * c_sorted[0]:
                # Try the second peak
                mask = np.zeros(len(c), dtype=bool)
                mask[k_star] = False
                k_star2 = int(np.argmax(np.where(mask, c, -np.inf)))
                j_start2 = max(0, k_star2 - n + 1)
                j_end2 = min(k_star2, n - 1)
                if j_end2 >= j_start2:
                    js2 = np.arange(j_start2, j_end2 + 1)
                    products2 = a[js2] * a[js2 + k_star2]
                    top_idx2 = np.argsort(products2)[::-1][:3]
                    for idx2 in top_idx2:
                        j = int(js2[idx2])
                        j2v = int(js2[idx2] + k_star2)
                        target = j if a[j] >= a[j2v] else j2v
                        old_val = a[target]
                        if old_val < 1e-12:
                            continue
                        for factor in [0.7, 0.5, 0.3]:
                            a[target] = old_val * factor
                            S_tmp = float(a.sum())
                            if S_tmp < 1e-12:
                                a[target] = old_val
                                continue
                            a_norm = a / S_tmp
                            sc = compute_score(a_norm)
                            if sc < best_sc - 1e-12:
                                best_sc = sc
                                best_a = a_norm.copy()
                                a = a_norm
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        no_improve = 0
                        continue

    return best_a


def fienup_refine(a, n_iters=200, thresh=1.5, clip=0.5, momentum=0.2):
    """Fienup algorithm with momentum for frequency-domain refinement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh_t = thresh + (1.05 - thresh) * t
        clip_t = clip + (0.9 - clip) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        bin_thresh = P_mean * thresh_t
        above = P > bin_thresh

        scale = np.ones_like(P)
        if above.any():
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip_t + clip_t * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Momentum update
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v

        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    def objective(x):
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample a sequence using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_diverse_init(n, variant=0):
    """Build various structured initializations."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif variant == 1:
        # Multi-freq
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif variant == 2:
        # Gaussian
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif variant == 3:
        # Chirp-like
        a = np.abs(np.sin(2 * np.pi * x ** 2 * 3.0)) + 0.1
    elif variant == 4:
        # Two-lobe
        a = np.abs(np.sin(np.pi * x)) * (1 + 0.5 * np.cos(2 * np.pi * x))
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * (variant + 1))
    a = np.maximum(a, 0.01)
    return a / a.sum()

    best_score = compute_score(best_a)
    best_a = best_a.copy()
    print(f"Initial: n={n_target}, score={best_score:.8f}")

    def update_best(a_new, label):
        nonlocal best_score, best_a
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Nesterov LSE with backtracking (40% of time)
    # ============================================================
    phase1_end = t_start + 0.40 * (total_budget - 15.0)
    print(f"\n=== Phase 1: Nesterov LSE + BT (budget {phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 30:
        time_left = (phase1_end - time.monotonic()) * 0.85
        a1 = nesterov_lse_bt(best_a, max_time=time_left,
                            beta_start=10.0, beta_end=5000.0,
                            lr_init=0.03, mu_init=0.80)
        update_best(a1, "nesterov-bt-1")

    if time.monotonic() < phase1_end - 10:
        # Second run with different params
        time_left = (phase1_end - time.monotonic()) * 0.9
        a2 = nesterov_lse_bt(best_a, max_time=time_left,
                            beta_start=30.0, beta_end=10000.0,
                            lr_init=0.01, mu_init=0.88)
        update_best(a2, "nesterov-bt-2")

    print(f"After Phase 1: {best_score:.8f}")

    # ============================================================
    # Phase 2: Peak shaving (20% of time)
    # ============================================================
    phase2_end = t_start + 0.60 * (total_budget - 15.0)
    print(f"\n=== Phase 2: Peak shaving (budget {phase2_end - t_start:.0f}s) ===")

    if time.monotonic() < phase2_end - 5:
        time_left = (phase2_end - time.monotonic()) * 0.9
        a3 = peak_shave(best_a, max_time=time_left, n_top_pairs=10, n_trials=6)
        update_best(a3, "peak-shave-1")

    if time.monotonic() < phase2_end - 5:
        # Peak shave from a slightly perturbed version
        a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n_target))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        time_left = (phase2_end - time.monotonic())
        if time_left > 10:
            a4 = peak_shave(a_pert, max_time=time_left, n_top_pairs=5, n_trials=4)
            update_best(a4, "peak-shave-pert")

    print(f"After Phase 2: {best_score:.8f}")

    # ============================================================
    # Phase 3: Nesterov LSE (simpler, more iterations) (15% of time)
    # ============================================================
    phase3_end = t_start + 0.75 * (total_budget - 15.0)
    print(f"\n=== Phase 3: Nesterov LSE simple (budget {phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end - 10:
        # Estimate iterations based on time
        time_left = (phase3_end - time.monotonic()) * 0.9
        n_iters = int(time_left * 80)  # ~80 iters/sec estimate
        n_iters = max(500, min(n_iters, 5000))
        a5 = nesterov_lse_simple(best_a, n_iters=n_iters,
                                beta_start=100.0, beta_end=8000.0,
                                lr_start=0.008, lr_end=0.0005,
                                mu_start=0.90, mu_end=0.96)
        update_best(a5, "nesterov-simple")

    if time.monotonic() < phase3_end - 10:
        time_left = (phase3_end - time.monotonic())
        if time_left > 20:
            a6 = nesterov_lse_simple(best_a, n_iters=int(time_left * 60),
                                    beta_start=50.0, beta_end=3000.0,
                                    lr_start=0.01, lr_end=0.001,
                                    mu_start=0.85, mu_end=0.93)
            update_best(a6, "nesterov-simple-2")

    print(f"After Phase 3: {best_score:.8f}")

    # ============================================================
    # Phase 4: Fienup refinement (10% of time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 15.0)
    print(f"\n=== Phase 4: Fienup (budget {phase4_end - t_start:.0f}s) ===")

    fienup_configs = [
        {'n_iters': 300, 'thresh': 1.8, 'clip': 0.4, 'momentum': 0.2},
        {'n_iters': 400, 'thresh': 1.5, 'clip': 0.6, 'momentum': 0.3},
        {'n_iters': 200, 'thresh': 2.0, 'clip': 0.3, 'momentum': 0.1},
        {'n_iters': 300, 'thresh': 1.3, 'clip': 0.7, 'momentum': 0.25},
    ]

    for i, cfg in enumerate(fienup_configs):
        if time.monotonic() > phase4_end - 5:
            break
        # Scale iterations to available time
        cfg_adj = cfg.copy()
        if time.monotonic() < phase4_end - 30:
            a_f = fienup_refine(best_a, **cfg_adj)
            update_best(a_f, f"fienup-{i}")

    print(f"After Phase 4: {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish (10% of time)
    # ============================================================
    phase5_end = deadline - 5
    print(f"\n=== Phase 5: L-BFGS polish (budget {phase5_end - t_start:.0f}s) ===")

    lbfgs_betas = [300, 800, 2000, 5000, 12000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase5_end - 30:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=2000, maxfun=30000)
        update_best(a_opt, f"lbfgs-b{beta}")

    print(f"After Phase 5: {best_score:.8f}")

    # ============================================================
    # Final: One more peak shave on the best result
    # ============================================================
    if time.monotonic() < deadline - 10:
        time_left = (deadline - time.monotonic()) - 5
        if time_left > 5:
            a_final = peak_shave(best_a, max_time=min(time_left, 60.0),
                                n_top_pairs=5, n_trials=4)
            update_best(a_final, "final-peak-shave")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
