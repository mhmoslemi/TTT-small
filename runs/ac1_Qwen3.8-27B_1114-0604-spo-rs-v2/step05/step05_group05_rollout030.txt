import cvxpy as cp

def lp_direction(f, M, solver='CLARABEL'):
    """Solve LP to find best direction b."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Compute f*b for all lags using the constraint structure
    # (f*b)[k] = sum_i f[i] * b[k-i] for valid i
    # This is a linear constraint in b
    
    # For efficiency, use the convolution structure
    # We need (f*b)[k] <= M for k = 0, ..., 2n-2
    
    # Build the constraint matrix implicitly
    # Actually for cvxpy, we can express this as:
    # cp.convolve(f, b) <= M  # but cp.convolve might not handle variable b
    
    # Alternative: express each constraint explicitly
    # (f*b)[k] = sum_{i} f[i] * b[k-i]
    # For k=0: f[0]*b[0] <= M
    # For k=1: f[0]*b[1] + f[1]*b[0] <= M
    # ...
    
    # This is equivalent to: for each k, the dot product of f (reversed and shifted) with b <= M
    
    # Let's use a different approach: since f is fixed, we can precompute
    # the constraint matrix
    
    # Actually, the most efficient way in cvxpy is:
    # constraint: for each k in range(2n-1):
    #   sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[i] * b[k-i] <= M
    
    # For large n, this creates a sparse matrix. Let's build it.
    
    # Actually, let's just use the fact that this is a Toeplitz-like system
    # and express it as a matrix product

    # Then constraint is C @ b <= M * ones(2n-1)
    
    from scipy.sparse import lil_matrix
    C = lil_matrix((2*n-1, n))
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(k, n-1) + 1):
            C[k, j] = f[k-j]
    C = C.tocsr()

Refined optimizer for minimum peak self-convolution of non-negative sequences.
Uses: adaptive Fienup, high-beta LSE gradient, active-set peak equalization,
and multi-restart perturbation strategy.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient computation via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_lse_score_and_grad(a, k, beta):
    """
    Compute LSE of top-k convolution values and gradient.
    Focuses optimization on the most important peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    # Get top-k indices
    if k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:k]
    c_k = c[idx]
    c_max = float(c_k.max())
    shifted = beta * (c_k - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(np.sum(exp_shifted))
    L = c_max + np.log(exp_sum) / beta
    p_k = exp_shifted / exp_sum

    # Build full p vector
    p = np.zeros(2 * n - 1)
    p[idx] = p_k

    # Gradient via FFT: dL/da[i] = 2 * sum_k p[k] * a[k-i]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def full_lse_score_and_grad(a, beta):
    """Full LSE score and gradient (all convolution entries)."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(np.sum(exp_shifted))
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_adaptive(a, n_iters, alpha_start=0.8, alpha_end=0.4,
                    thresh_start=2.5, thresh_end=1.05,
                    momentum_end=0.4):
    """
    Adaptive Fienup with spectral compression exponent.
    Instead of hard thresholding, uses |A|^alpha compression.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        thresh = thresh_start + (thresh_end - thresh_start) * t
        momentum = momentum_end * t

        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        amp_max = amp.max()
        amp_mean = amp.mean()

        # Combined: compress dynamically and threshold high bins
        # Compression: amp_new = amp^alpha * amp_max^(1-alpha) (normalizes max)
        amp_norm = amp / amp_max
        amp_compressed = amp_max * (amp_norm ** alpha)

        # Additional thresholding on very high bins
        bin_thresh = amp_mean * thresh
        high_mask = amp_compressed > bin_thresh
        if high_mask.any():
            # Soft reduction: bring high bins toward threshold
            ratio = bin_thresh / amp_compressed[high_mask]
            amp_compressed[high_mask] = amp_compressed[high_mask] * (0.5 + 0.5 * ratio)

        # Preserve phase
        phase = A / np.maximum(amp, 1e-30)
        A_new = amp_compressed * phase

        # Ensure Hermitian symmetry for real output
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Momentum
        update = a_new - a
        v = momentum * v + update
        a_candidate = a + v

        # Project
        a_candidate = np.maximum(a_candidate, 0.0)
        S_new = float(np.sum(a_candidate))
        if S_new < 1e-12:
            break
        a_candidate = a_candidate / S_new

        sc = compute_score(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()
        a = a_candidate

    return best_a


def nesterov_topk(a, k_start=50, k_end=5, beta_start=20.0, beta_end=20000.0,
                  n_iters=2000, lr_start=0.015, lr_end=0.001,
                  momentum_init=0.9, momentum_end=0.95):
    """
    Nesterov-accelerated gradient descent on top-k LSE objective.
    k decreases over time (from smooth to sharp focus).
    beta increases over time (from smooth to precise).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        k = max(k_end, int(k_start + (k_end - k_start) * t))
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start + (lr_end - lr_start) * t
        mu = momentum_init + (momentum_end - momentum_init) * t

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(np.sum(a_look))
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = topk_lse_score_and_grad(a_look, k, beta)

        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def nesterov_topk_timed(a, max_time=100.0, k_start=40, k_end=3,
                        beta_start=10.0, beta_end=30000.0):
    """Time-limited Nesterov top-k with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    lr = 0.012
    lr_decay = 0.9985
    mu = 0.9
    beta = beta_start
    k = k_start
    stagnation = 0
    best_sc_stag = best_sc

    while time.monotonic() - t_start < max_time:
        # Adaptive k: decrease as we get closer to convergence
        improvement = (best_sc_stag - best_sc) / best_sc_stag
        if improvement < 1e-6:
            stagnation += 1
        else:
            stagnation = 0
            best_sc_stag = best_sc

        if stagnation > 50 and k > k_end:
            k = max(k_end, k // 2)
            lr = max(lr * 0.7, 0.0005)
            stagnation = 0

        mu = min(0.96, 0.9 + 0.0001 * (it_count()))
        lr *= lr_decay

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(np.sum(a_look))
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = topk_lse_score_and_grad(a_look, k, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        effective_lr = min(lr, 0.01 / (gnorm + 1e-10))

        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        a = a / S

        beta = min(beta * 1.003, beta_end)

        if (it_count() + 1) % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def it_count():
    """Global iteration counter (simplified)."""
    global _it_counter
    _it_counter = getattr(_it_counter_holder, 'val', 0)
    _it_counter_holder.val = _it_counter + 1
    return _it_counter


class _it_counter_holder:
    val = 0


def peak_equalization_cd(a, n_sweeps=10, n_top_peaks=8, n_top_contrib=15):
    """
    Peak equalization via coordinate descent.
    For each of the top peaks, find the element that most contributes
    and try scaling it to reduce the peak while not increasing others.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = c.max()
        # Get top peaks (within 0.1% of max for "active" peaks)
        threshold = c_max * 0.9995
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        peak_indices = peak_indices[:n_top_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution of a[j] to c[k] is a[k-j]
            contrib = np.array([a[k - j] * a[j] for j in range(j_start, j_end + 1)])
            if contrib.max() < 1e-15:
                continue

            # Top contributors
            n_local = min(n_top_contrib, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try multiple scaling factors
                for factor in [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1]:
                    a[j] = old_val * factor
                    S_tmp = float(np.sum(a))
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        improved = True
                        break
                else:
                    a[j] = best_a[j]  # restore
                    a = a / float(np.sum(a))

        if not improved and sweep > 3:
            break

    return best_a


def run_lbfgs_topk(a0, k, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B on top-k LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return topk_lse_score_and_grad(x, k, beta)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample using cubic-like interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def fienup_compression(a, n_iters=200, alpha=0.6, thresh=1.2):
    """
    Fienup with pure spectral compression (no momentum).
    Uses |A|^alpha to compress dynamic range.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        amp_max = amp.max()
        if amp_max < 1e-30:
            break
        amp_norm = amp / amp_max
        amp_new = amp_max * (amp_norm ** alpha)

        # Slight thresholding
        amp_mean = amp_new.mean()
        high = amp_new > amp_mean * thresh
        if high.any():
            amp_new[high] = np.sqrt(amp_new[high] * amp_mean * thresh)

        phase = A / np.maximum(amp, 1e-30)
        A_new = amp_new * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_gaussian_init(n):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_mixed_init(n):
    """Mixed initialization with multiple harmonics."""
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
         + 0.05 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-strategy approach: adaptive Fienup, top-k LSE Nesterov,
    peak equalization CD, and multi-restart perturbation.
    """
    global _it_counter_holder
    _it_counter_holder.val = 0

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Initialization
    # ============================================================
    if parent is not None and len(parent) >= 64:
        n_target = len(parent)
        a_init = parent.copy()
    else:
        n_target = 6962
        a_init = build_gaussian_init(n_target)

    sc_init = compute_score(a_init)
    update_best(sc_init, a_init, "init")
    print(f"Initial score: {sc_init:.8f} (n={n_target})")

    # ============================================================
    # Phase 1: Quick Fienup warm-up (200s)
    # ============================================================
    phase1_end = t_start + 200.0
    print(f"\n=== Phase 1: Quick Fienup (t < {phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 10:
        a_f1 = fienup_adaptive(best_a, n_iters=150,
                               alpha_start=0.7, alpha_end=0.5,
                               thresh_start=2.0, thresh_end=1.1,
                               momentum_end=0.3)
        update_best(compute_score(a_f1), a_f1, "fienup-warm")

    a_f2 = fienup_compression(best_a, n_iters=100, alpha=0.65, thresh=1.3)
    update_best(compute_score(a_f2), a_f2, "fienup-compact")

    print(f"After Phase 1: {best_score:.8f}")

    # ============================================================
    # Phase 2: Long adaptive Fienup (400s)
    # ============================================================
    phase2_end = t_start + 600.0
    print(f"\n=== Phase 2: Adaptive Fienup (t < {phase2_end - t_start:.0f}s) ===")

    fienup_configs = [
        {'n_iters': 300, 'alpha_start': 0.8, 'alpha_end': 0.4,
         'thresh_start': 2.5, 'thresh_end': 1.05, 'momentum_end': 0.5},
        {'n_iters': 300, 'alpha_start': 0.6, 'alpha_end': 0.35,
         'thresh_start': 1.8, 'thresh_end': 1.0, 'momentum_end': 0.4},
        {'n_iters': 200, 'alpha_start': 0.9, 'alpha_end': 0.5,
         'thresh_start': 3.0, 'thresh_end': 1.1, 'momentum_end': 0.3},
    ]

    for i, cfg in enumerate(fienup_configs):
        if time.monotonic() > phase2_end - 30:
            break
        a_f = fienup_adaptive(best_a, **cfg)
        update_best(compute_score(a_f), a_f, f"fienup-adapt-{i}")

    print(f"After Phase 2: {best_score:.8f}")

    # ============================================================
    # Phase 3: Top-k LSE Nesterov (800s)
    # ============================================================
    phase3_end = t_start + 1400.0
    print(f"\n=== Phase 3: Top-k LSE Nesterov (t < {phase3_end - t_start:.0f}s) ===")

    nesterov_configs = [
        {'k_start': 50, 'k_end': 5, 'beta_start': 20.0, 'beta_end': 10000.0},
        {'k_start': 30, 'k_end': 3, 'beta_start': 50.0, 'beta_end': 20000.0},
        {'k_start': 20, 'k_end': 2, 'beta_start': 100.0, 'beta_end': 30000.0},
    ]

    for i, cfg in enumerate(nesterov_configs):
        if time.monotonic() > phase3_end - 60:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 60:
            break
        print(f"  Nesterov config {i}: k=[{cfg['k_start']},{cfg['k_end']}], "
              f"beta=[{cfg['beta_start']},{cfg['beta_end']}], time={time_left:.0f}s")
        a_nl = nesterov_topk(best_a, n_iters=int(time_left * 8), **cfg)
        update_best(compute_score(a_nl), a_nl, f"nesterov-{i}")

    # Full LSE with high beta for precision
    if time.monotonic() < phase3_end - 60:
        time_left = phase3_end - time.monotonic()
        print(f"  Full LSE high-beta, time={time_left:.0f}s")
        a_full = nesterov_topk(best_a, k_start=5, k_end=1,
                               beta_start=500.0, beta_end=50000.0,
                               n_iters=int(time_left * 6),
                               lr_start=0.005, lr_end=0.0005)
        update_best(compute_score(a_full), a_full, "lse-hibeta")

    print(f"After Phase 3: {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak equalization CD (300s)
    # ============================================================

    print(f"\n=== Phase 4: Peak equalization CD (t < {phase4_end - t_start:.0f}s) ===")

    cd_configs = [
        {'n_sweeps': 8, 'n_top_peaks': 6, 'n_top_contrib': 20},
        {'n_sweeps': 12, 'n_top_peaks': 10, 'n_top_contrib': 10},
        {'n_sweeps': 5, 'n_top_peaks': 20, 'n_top_contrib': 15},
    ]

    for i, cfg in enumerate(cd_configs):
        if time.monotonic() > phase4_end - 30:
            break
        a_cd = peak_equalization_cd(best_a, **cfg)
        update_best(compute_score(a_cd), a_cd, f"peak-cd-{i}")

    # Short LSE after CD
    if time.monotonic() < phase4_end - 30:
        a_cd_lse = nesterov_topk(best_a, k_start=10, k_end=2,
                                  beta_start=200.0, beta_end=10000.0,
                                  n_iters=500, lr_start=0.003, lr_end=0.0005)
        update_best(compute_score(a_cd_lse), a_cd_lse, "cd+lse")

    print(f"After Phase 4: {best_score:.8f}")

    # ============================================================
    # Phase 5: Multi-restart perturbation (400s)
    # ============================================================
    phase5_end = t_start + 2100.0
    print(f"\n=== Phase 5: Multi-restart (t < {phase5_end - t_start:.0f}s) ===")

    n_restarts = 3
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 60:
            break
        time_left = phase5_end - time.monotonic()
        if time_left < 60:
            break

        # Perturb
        noise_scale = 0.02 * (r + 1)
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_pert = np.maximum(a_pert, 0.0)
        S_pert = float(np.sum(a_pert))
        if S_pert < 1e-12:
            continue
        a_pert = a_pert / S_pert

        print(f"  Restart {r}: noise_scale={noise_scale}, time_left={time_left:.0f}s")

        # Quick Fienup
        a_r = fienup_adaptive(a_pert, n_iters=80,
                              alpha_start=0.7, alpha_end=0.45,
                              thresh_start=2.0, thresh_end=1.1,
                              momentum_end=0.3)
        sc_r = compute_score(a_r)
        if sc_r < best_score:
            update_best(sc_r, a_r, f"restart-{r}-fienup")

        # Short LSE
        if time.monotonic() < phase5_end - 30:
            a_r2 = nesterov_topk(a_r, k_start=20, k_end=3,
                                  beta_start=50.0, beta_end=10000.0,
                                  n_iters=int(time_left * 0.4 * 8),
                                  lr_start=0.008, lr_end=0.001)
            update_best(compute_score(a_r2), a_r2, f"restart-{r}-lse")

    print(f"After Phase 5: {best_score:.8f}")

    # ============================================================
    # Phase 6: Final L-BFGS polish (200s)
    # ============================================================
    phase6_end = deadline
    print(f"\n=== Phase 6: L-BFGS polish (t < {phase6_end - t_start:.0f}s) ===")

    lbfgs_configs = [
        {'k': 5, 'beta': 2000, 'maxiter': 2000},
        {'k': 3, 'beta': 5000, 'maxiter': 1500},
        {'k': 10, 'beta': 1000, 'maxiter': 2000},
    ]

    for i, cfg in enumerate(lbfgs_configs):
        if time.monotonic() > phase6_end - 10:
            break
        try:
            sc, a_opt = run_lbfgs_topk(best_a, cfg['k'], cfg['beta'],
                                        maxiter=cfg['maxiter'], maxfun=30000)
            update_best(sc, a_opt, f"lbfgs-{i}")
        except Exception as e:
            print(f"    L-BFGS config {i} failed: {e}")

    # Final: one more peak CD
    if time.monotonic() < phase6_end - 10:
        a_final_cd = peak_equalization_cd(best_a, n_sweeps=5,
                                           n_top_peaks=5, n_top_contrib=20)
        update_best(compute_score(a_final_cd), a_final_cd, "final-cd")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
