def next_power_of_two(m):

    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    """LSE-smoothed objective with gradient. Returns (score, gradient)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    # Numerical stability
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B with LSE-smoothed objective."""
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
        est_per_eval = 0.01 * max(1.0, n / 4096.0)
        cap = int((0.85 * rem) / max(est_per_eval, 1e-4))
        if cap < 32:
            return exact_score_a(a), a
        if maxfun is not None:
            maxfun = min(int(maxfun), cap)
        else:
            maxfun = max(64, cap)
    else:
        maxfun = int(maxfun if maxfun else 500)

    maxfun = max(32, maxfun)
    maxiter = max(30, maxfun)

    # Use wider bounds to allow exploration
    bounds = [(0.0, 20.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
        "factr": 1e12,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    a_out = normalize_a(res.x) if hasattr(res, 'x') and res.x is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights(d, k):
    """Get weights on top-k elements of d (normalized to sum to 1)."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def subgrad_step(a, eta, k_active=3):
    """One subgradient step: multiply a by exp(-eta * h / hmax)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if dmax <= 1e-12:
        return a
    w = active_weights(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def subgrad_polish(a, max_iters=50, deadline=None):
    """Iterative subgradient with multiple k and step sizes."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        cur_a = a.copy()
        cur_sc = best_sc

        for k in (1, 2, 3, 5, 8, 13):
            for eta in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step(cur_a, eta, k)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()

        if not improved:
            break
        a = cur_a

    return float(best_sc), best_a


def log_space_optimize(a0, beta, deadline, maxfun=200):
    """Optimize in log-space: let x = log(a + eps), optimize x with L-BFGS-B."""
    a = normalize_a(a0)
    n = len(a)
    eps = 1e-10
    x0 = np.log(a + eps)

    def fun_log(x):
        a_trial = np.exp(x)
        a_trial = a_trial / a_trial.sum()  # normalize back
        sc, g = lse_score_grad(a_trial, beta)
        # Chain rule: d(a)/dx = a * ones (since a = exp(x)/sum(exp(x)))
        # da_i/dx_j = a_i * (delta_ij - a_j)
        # grad_x_j = sum_i grad_a_i * da_i/dx_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
        g_log = g * a_trial - a_trial * float(np.dot(g, a_trial))
        g_log = np.nan_to_num(g_log, nan=0.0, posinf=0.0, neginf=0.0)
        return sc, g_log

    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
        maxfun = min(maxfun, max(32, int(rem / 0.5)))

    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(30, maxfun // 2),
        "maxfun": maxfun,
        "maxls": 30,
    }

    try:
        res = minimize(fun_log, x0, method="L-BFGS-B", jac=True, options=opts)
    except Exception:
        return exact_score_a(a), a

    if hasattr(res, 'x') and res.x is not None:
        a_out = np.exp(res.x)
        a_out = normalize_a(a_out)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        return exact_score_a(a), a
    return float(sc), a_out


def generate_inits(n, rng, parent=None, cap=25):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    # Uniform
    add(np.ones(n))

    # U-shapes (high at edges) - key insight from sum-distribution flattening
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(0.1 + alpha * np.abs(2.0 * x - 1.0))
        add(0.1 + alpha * (np.abs(2.0 * x - 1.0) ** 2))
        add(0.1 + alpha * (np.abs(2.0 * x - 1.0) ** 0.5))

    # V-shapes (low in middle)
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 - alpha * (1.0 - np.abs(2.0 * x - 1.0)))

    # Cosine modulation
    for f in (1, 2, 3, 4):
        for amp in (0.3, 0.5, 0.8, 1.0):
            add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
            add(1.0 + amp * np.cos(np.pi * f * x))

    # Power-law edges
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        add(np.abs(2.0 * x - 1.0) ** p + 0.01)

    # Bimodal
    for c in (0.2, 0.3, 0.35):
        for w in (0.05, 0.1, 0.15, 0.2):
            add(np.exp(-(((x - c) / w) ** 2)) + np.exp(-(((x - (1-c)) / w) ** 2)))

    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Random smooth
    Nf = next_power_of_two(2 * n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 0.8)) * noise)

    # Random trigonometric mixture
    for _ in range(4):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(3, 8))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(np.abs(vals) + 0.01)

    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb(a, rng, strength=None):
    """Apply a random perturbation to the sequence."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.2, 0.8))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        # Multiplicative Gaussian noise
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.1, 0.5))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        # Reverse + noise
        q = p[::-1] * np.exp(0.2 * rng.standard_normal(n))
    elif mode == 3:
        # Subgradient-inspired perturbation
        S = float(p.sum())
        c = fast_self_conv(p)
        d = (float(n) * c) / (S * S)
        d = np.maximum(d, 0.0)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            k = int(rng.integers(1, 7))
            w = active_weights(d, k)
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Low-pass filtered noise (smooth perturbation)
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 100))
        if k % 2 == 0:
            k += 1
        c_k = k // 2
        sig = max(1.0, c_k / 2.0)
        kern = np.exp(-((np.arange(k) - c_k) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.5 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # U-shape modulation
        x = np.linspace(0.0, 1.0, n)
        u = 0.5 + strength * np.abs(2.0 * x - 1.0)
        q = p * u
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 80)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c_k = k // 2
        sig = max(1.0, c_k / 2.0)
        kern = np.exp(-((np.arange(k) - c_k) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # FFT phase randomization (preserves magnitude spectrum)
        N = next_power_of_two(2 * n)
        F = np.fft.rfft(p, N)
        mag = np.abs(F)
        ph = rng.uniform(-np.pi, pi, size=len(mag)) if False else rng.uniform(-np.pi, np.pi, size=len(mag))
        F_new = mag * np.exp(1j * ph)
        F_new[0] = mag[0]  # keep DC
        q = np.fft.irfft(F_new, N)[:n]
        q = np.abs(q) + 0.01
    elif mode == 8:
        # Coordinate-wise random reset
        q = p.copy()
        n_reset = max(1, int(n * float(rng.uniform(0.01, 0.1))))
        idx = rng.choice(n, n_reset, replace=False)
        q[idx] = float(rng.uniform(0.0, 2.0 * float(np.mean(p))))
    else:
        # Sharpen: amplify high values, suppress low
        p_max = float(np.max(p))
        if p_max > 1e-12:
            q = p ** (1.0 + strength)
        else:
            q = p.copy()

    return normalize_a(np.clip(q, 0.0, None))


def optimize_trajectory(a_start, deadline, rng, label=""):
    """Main optimization trajectory with adaptive LSE beta schedule."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 10.0))
    n = len(a)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Adaptive beta schedule: start low, increase gradually
    # Use more stages for better convergence
    stages = [
        (50.0, 0.05),
        (150.0, 0.10),
        (400.0, 0.12),
        (1000.0, 0.15),
        (3000.0, 0.18),
        (8000.0, 0.15),
        (20000.0, 0.20),
        (50000.0, 0.10),
        (100000.0, 0.15),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        # Multiple L-BFGS runs at this beta
        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est_per_eval = 0.01 * max(1.0, n / 4096.0)
            mf = max(32, min(500, int(rem * 0.8 / max(est_per_eval, 1e-4))))
            if mf < 32:
                break

            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)

            # Also try log-space at higher betas
            if beta > 1000 and time.monotonic() < sub_dead - 5.0:
                sc_log, a_log = log_space_optimize(a, float(beta), deadline, max(32, mf // 2))
                if np.isfinite(sc_log) and sc_log < best_sc:
                    best_sc = sc_log
                    best_a = a_log.copy()
                    print("[{}] log-space best {:.8f} beta={:.0f}".format(label, best_sc, beta))
                a = normalize_a(a_log)

        # Subgradient between LSE stages
        if time.monotonic() < sub_dead - 5.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=15, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def quick_optimize(a, deadline):
    """Quick multi-beta optimization."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for beta in (100, 500, 2000, 10000, 30000, 80000):
        rem = float(deadline - time.monotonic())
        if rem < 8.0:
            break
        est_per_eval = 0.01 * max(1.0, len(a) / 4096.0)
        mf = max(32, min(300, int(rem * 0.7 / max(est_per_eval, 1e-4))))
        if mf < 32:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    return float(best_sc), best_a


def multi_scale_optimize(a_start, deadline, rng, target_n):
    """Multi-scale: optimize at coarser scales, then refine."""
    t0 = time.monotonic()
    scales = []
    cur = target_n
    while cur > 256:
        scales.append(cur)
        cur = max(256, cur // 2)
    scales.sort()  # from coarse to fine

    print("Multi-scale: {} -> {}".format(scales[0], scales[-1]))

    a = resample_sequence(a_start, scales[0])
    a = normalize_a(a)
    best_overall = exact_score_a(a)
    best_a_overall = a.copy()

    for i, n in enumerate(scales):
        a = resample_sequence(a, n)
        a = normalize_a(a)

        # Allocate time proportionally (more time for finer scales)
        if i < len(scales) - 1:
            frac = 0.3 / len(scales)
        else:
            frac = 0.7 / len(scales)
        sub_dead = t0 + frac * (deadline - t0 - 20.0)
        sub_dead = min(sub_dead, deadline - 15.0)

        if time.monotonic() > sub_dead - 10.0:
            sc = exact_score_a(a)
            if sc < best_overall:
                best_overall = sc
                best_a_overall = a.copy()
            continue

        sc, a_opt = optimize_trajectory(a, sub_dead, rng, "ms_n{}".format(n))
        if np.isfinite(sc) and sc < best_overall:
            best_overall = sc
            best_a_overall = a_opt.copy()

        a = normalize_a(a_opt)

    # Final resample to target
    final_a = resample_sequence(best_a_overall, target_n)
    final_a = normalize_a(final_a)
    final_sc = exact_score_a(final_a)
    if final_sc < best_overall:
        best_overall = final_sc
        best_a_overall = final_a.copy()

    return float(best_overall), best_a_overall


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(p.sum()) > 0.01:
                parent = normalize_a(p)
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                         parent.size if parent is not None else 0))

    # Determine target length
    if parent is not None:
        target_n = int(parent.size)
    else:
        target_n = 8192

    # Phase 1: Quick initialization evaluation
    inits = generate_inits(target_n, rng, parent, cap=30)
    best_init_sc = np.inf
    best_init = None
    for arr in inits:
        sc = exact_score_a(arr)
        if np.isfinite(sc) and sc < best_init_sc:
            best_init_sc = sc
            best_init = arr.copy()

    if parent is not None:
        ps = normalize_a(resample_sequence(parent, target_n))
        ps_sc = exact_score_a(ps)
        if ps_sc < best_init_sc:
            best_init_sc = ps_sc
            best_init = ps

    print("Best init: {:.8f}".format(best_init_sc))

    best_seq = best_init.copy()
    best_score = best_init_sc

    # Phase 2: Multi-scale optimization (use ~50% of budget)
    ms_dead = t_start + 0.50 * (deadline - t_start)
    try:
        sc_ms, seq_ms = multi_scale_optimize(best_init, ms_dead, rng, target_n)
        if np.isfinite(sc_ms) and sc_ms < best_score:
            best_score = sc_ms
            best_seq = seq_ms.copy()
            print("After multi-scale: {:.8f}".format(best_score))
    except Exception as e:
        print("Multi-scale error: {}".format(e))

    # Phase 3: Direct optimization from best (use ~25% of budget)
    direct_dead = t_start + 0.75 * (deadline - t_start)
    if time.monotonic() < direct_dead - 30.0:
        try:
            sc_d, seq_d = optimize_trajectory(best_seq, direct_dead, rng, "direct")
            if np.isfinite(sc_d) and sc_d < best_score:
                best_score = sc_d
                best_seq = seq_d.copy()
                print("After direct: {:.8f}".format(best_score))
        except Exception as e:
            print("Direct error: {}".format(e))

    # Phase 4: Perturbation restarts (use ~15% of budget)
    perturb_dead = t_start + 0.90 * (deadline - t_start)
    n_perturb = 0
    while time.monotonic() < perturb_dead - 30.0 and n_perturb < 8:
        left = float(perturb_dead - 30.0 - time.monotonic())
        if left < 40.0:
            break
        n_left = 8 - n_perturb
        chunk = time.monotonic() + left / float(n_left)

        p = perturb(best_seq, rng)
        try:
            sc_p, seq_p = quick_optimize(p, chunk)
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = seq_p.copy()
                print("Perturb {} : {:.8f}".format(n_perturb, best_score))
        except Exception:
            pass
        n_perturb += 1

    # Phase 5: Final high-beta polish
    final_dead = deadline - 10.0
    if time.monotonic() < final_dead - 20.0:
        for beta in (50000, 150000, 300000):
            rem = float(final_dead - time.monotonic())
            if rem < 10.0:
                break
            est_per_eval = 0.01 * max(1.0, len(best_seq) / 4096.0)
            mf = max(32, min(200, int(rem * 0.8 / max(est_per_eval, 1e-4))))
            sc, seq = run_lbfgs(best_seq, float(beta), final_dead, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final beta={}: {:.8f}".format(beta, best_score))

        # Log-space final
        if time.monotonic() < final_dead - 15.0:
            rem = float(final_dead - time.monotonic())
            if rem > 10.0:
                sc, seq = log_space_optimize(best_seq, 200000.0, final_dead, min(100, max(32, int(rem / 0.5))))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print("Log-space final: {:.8f}".format(best_score))

    # Phase 6: Final subgradient
    if time.monotonic() < deadline - 5.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 3.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Final cleanup and verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = float(out_arr.sum())
    out_arr = out_arr / S  # normalize to sum=1 (scale-invariant, but safe)
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
