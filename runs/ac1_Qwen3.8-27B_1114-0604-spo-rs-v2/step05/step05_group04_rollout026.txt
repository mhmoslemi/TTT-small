"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: adaptive Fienup, LSE-smoothed L-BFGS, Nesterov subgradient, and coordinate refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time

# Top-level helper functions (no closures, no lambdas)

def fast_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_spectral_compress(a, n_iters=500, gamma=0.8, top_frac_init=0.1,
                             top_frac_end=0.5, soft_alpha=30.0):
    """Fienup iteration with spectral compression."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2*n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        thresh = np.quantile(P, 1.0 - top_frac)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(ratio) * gamma
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        
        # Soft non-negativity
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def lse_obj_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. x
    p = exp_scaled / exp_sum  # softmax weights on convolution peaks
    # dL/dx_i = 2 * sum_k p_k * (c_k / dx_i) = 2 * sum_k p_k * a[k-i] (for valid k,i)
    # This is 2 * conv(p, x) evaluated at specific indices
    rev_x = x[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RP = np.fft.rfft(np.concatenate([p, np.zeros(M2_fft - len(p))]))
    RX = np.fft.rfft(np.concatenate([rev_x, np.zeros(M2_fft - len(rev_x))]))
    conv_res = np.fft.irfft(RP * RX, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L * x / (S2 * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out

def nesterov_subgrad(a, n_iters=5000, top_m=20, step_init=0.02):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy() / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    
    for it in range(n_iters):
        t = it + 1
        beta_nest = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        momentum = beta_nest * momentum
        
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        # Subgradient of max(c_k) w.r.t. a
        top_idx = np.argsort(c)[::-1][:top_m]
        weights = c[top_idx] - c[top_m - 1] + 1e-10
        weights = np.maximum(weights, 0)
        weights = weights / (weights.sum() + 1e-30)
        
        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]
        
        # Full gradient including sum term
        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * a / (S2 * S))
        
        # Adaptive step size
        step = step_init / np.sqrt(t + 1)
        
        a = y - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Nesterov update
        alpha = (1.0 + 1.0 / beta_nest) / 2.0
        y = a + alpha * (a - y) if momentum > 0 else a.copy()
        y = np.clip(y, 0.0, 1000.0)
        y = y / y.sum()
    
    return best_a

def coordinate_refine(a, n_iters=10000, top_m=5):
    """Coordinate descent focusing on reducing peak convolution values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find peak positions
        peak_idx = np.argmax(c)
        
        # Try to reduce the peak by adjusting coordinates
        k = int(peak_idx)
        j_start = max(0, k - n + 1)

def fast_conv(a):

def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500.0, 0.0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RP = np.fft.rfft(np.concatenate([p, np.zeros(M2_fft - M)]))
    RX = np.fft.rfft(np.concatenate([rev_x, np.zeros(M2_fft - n)]))
    conv_res = np.fft.irfft(RP * RX, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L * x / (S2 * S))
    return score, grad

                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]
        
        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * a / (S2 * S))
        
        step = step_init / np.sqrt(t)
        
        a = y - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        # Nesterov momentum
        alpha = 1.0 - 1.0 / beta_n
        y = a + alpha * (a - y)
        y = np.clip(y, 0.0, 1000.0)
        S_y = float(y.sum())
        if S_y < 1e-12:
            y = a.copy()
        else:
            y = y / S_y
    
    return best_a


def peak_target_refine(a, n_iters=5000, n_peaks=10):

    n = len(a)

    if S < 1e-12:
        return a

        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the peak position
        peak_k = int(np.argmax(c))
        
        # The peak at position k is contributed by a[j]*a[k-j] for valid j
        j_start = max(0, peak_k - n + 1)
        j_end = min(peak_k, n - 1)
        
        # Compute sensitivity: how much each a[i] contributes to the peak
        # dc_peak/da[i] = 2 * a[peak_k - i] (if valid)
        sensitivity = np.zeros(n)
        for i in range(j_start, j_end + 1):
            sensitivity[i] = 2.0 * a[peak_k - i]
        
        # Also consider contribution to other peaks
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        for pk in top_peaks[1:]:
            pk = int(pk)
            ps = max(0, pk - n + 1)
            pe = min(pk, n - 1)
            for i in range(ps, pe + 1):
                sensitivity[i] += c[pk] / M_val * 2.0 * a[pk - i]
        
        # Move mass from high-sensitivity to low-sensitivity positions
        # that don't contribute to any peak
        for pk in top_peaks:
            pk = int(pk)
            ps = max(0, pk - n + 1)
            pe = min(pk, n - 1)
            for i in range(ps, pe + 1):
                sensitivity[i] += 1e-10  # small positive so they're always "high"
        
        # Find positions with low sensitivity (don't contribute to peaks)
        low_thresh = np.percentile(sensitivity, 50)
        high_mask = sensitivity > low_thresh
        low_mask = sensitivity <= low_thresh
        
        if not high_mask.any() or not low_mask.any():
            # Random perturbation
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            a = a / a.sum()
            continue
        
        # Move a small fraction of mass
        frac = 0.005 + 0.005 * (1.0 - it / n_iters)
        mass_to_move = frac * a[high_mask].sum()
        a[high_mask] -= frac * a[high_mask]
        # Distribute to low-sensitivity positions proportional to current value
        a[low_mask] += mass_to_move * a[low_mask] / max(a[low_mask].sum(), 1e-30)
        
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses adaptive Fienup spectral compression, LSE-smoothed L-BFGS with beta sweeping,
Nesterov-accelerated subgradient, and peak-targeted mass redistribution.
Key improvements: longer Fienup runs, higher LSE betas, chained optimization,
and a novel peak-targeting refinement that redistributes mass away from peak-contributing positions.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute linear autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_compress(a, n_iters=500, gamma=0.8, top_frac_init=0.10,
                    top_frac_end=0.50, soft_alpha=30.0):
    """Fienup iteration with spectral power compression of top-frequency bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * \
            0.5 * (1.0 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(ratio) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat(a, n_iters=400, alpha_init=0.2, alpha_end=0.9):
    """Spectral equalization: relax power spectrum towards flat."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_waterfill(a, n_iters=300, n_levels=6):
    """Multi-level water-filling compression in the spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2

        scale = np.ones_like(P)
        for lv in range(n_levels):
            frac = 0.1 * (lv + 1) / n_levels
            thresh = np.quantile(P, 1.0 - frac)
            above = P > thresh
            if above.any():
                strength = 0.3 + 0.5 * progress
                ratio = thresh / P[above]
                scale[above] *= 1.0 - strength + strength * np.sqrt(np.maximum(ratio, 0.001))

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500.0, 0.0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RP = np.fft.rfft(np.concatenate([p, np.zeros(M2_fft - M)]))
    RX = np.fft.rfft(np.concatenate([rev_x, np.zeros(M2_fft - n)]))
    conv_res = np.fft.irfft(RP * RX, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L * x / (S2 * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def nesterov_subgrad(a, n_iters=3000, top_m=20, step_init=0.015):
    """Nesterov-accelerated subgradient descent on the original (non-smooth) objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    y = a.copy()

    for it in range(n_iters):
        t = it + 1
        beta_n = (1.0 + np.sqrt(1.0 + 4.0 * (t / max(n_iters, 1)) ** 2)) / 2.0

        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c))
        else:
            sorted_idx = np.argsort(c)[::-1]
            top_idx = sorted_idx[:top_m]
            vals = c[top_idx]
            weights = np.maximum(vals - vals[-1], 0.0)
            if weights.sum() < 1e-30:
                weights = np.ones(top_m)
            weights = weights / weights.sum()

        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]

        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * a / (S2 * S))

        step = step_init / np.sqrt(t)
        a = y - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        alpha_mom = 1.0 - 1.0 / beta_n
        y = a + alpha_mom * (a - y)
        y = np.clip(y, 0.0, 1000.0)
        S_y = float(y.sum())
        if S_y < 1e-12:
            y = a.copy()
        else:
            y = y / S_y

    return best_a

            print(f"  Fienup-C n={n_use} i={i}: {sc:.8f}")

        # Fienup flat
        if time.monotonic() < phase1_end:
            a_fl = fienup_flat(a0, n_iters=500, alpha_init=0.2, alpha_end=0.9)
            sc2 = compute_score(a_fl)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_fl.copy()
                print(f"  Fienup-F n={n_use} i={i}: {sc2:.8f}")

        # Water-filling
        if time.monotonic() < phase1_end:
            a_wf = fienup_waterfill(a0, n_iters=400, n_levels=6)
            sc3 = compute_score(a_wf)
            if sc3 < best_score:
                best_score = sc3
                best_a = a_wf.copy()
                print(f"  Fienup-W n={n_use} i={i}: {sc3:.8f}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: L-BFGS with beta sweep (45% time)
    # ============================================================
    phase2_end = t_start + 0.60 * (total - 30.0)
    print(f"\n=== Phase 2: L-BFGS beta sweep (until t={phase2_end - t_start:.0f}s) ===")

    betas = [50, 100, 200, 500, 1000, 3000, 10000, 30000, 100000]
    for beta in betas:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 45:
            break

        # Scale maxiter based on remaining time and sequence length
        maxiter = min(8000, max(500, int(remaining * 100 / n_use * 50)))
        maxfun = maxiter * 15

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  L-BFGS beta={beta}: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  L-BFGS beta={beta}: {sc:.8f} (no improvement)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Nesterov subgradient (20% time)
    # ============================================================
    phase3_end = t_start + 0.80 * (total - 30.0)
    print(f"\n=== Phase 3: Nesterov subgradient (until t={phase3_end - t_start:.0f}s) ===")

    for top_m in [10, 20, 40]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        n_iters = min(3000, max(500, int(remaining * 50)))
        a_sg = nesterov_subgrad(best_a, n_iters=n_iters, top_m=top_m, step_init=0.012)
        sc = compute_score(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()
            print(f"  Nesterov top_m={top_m}: {sc:.8f}")
        else:
            print(f"  Nesterov top_m={top_m}: {sc:.8f} (no improvement)")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak targeting + final polish (20% time)
    # ============================================================
    print(f"\n=== Phase 4: Peak targeting + Final polish ===")

    # Peak-targeted refinement
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_pt = min(5000, max(1000, int(remaining * 30)))
        a_pt = peak_target_refine(best_a, n_iters=n_pt, n_peaks=15, rng_seed=42)
        sc = compute_score(a_pt)
        if sc < best_score:
            best_score = sc
            best_a = a_pt.copy()
            print(f"  Peak-target: {sc:.8f}")

    # Final Fienup + L-BFGS chain
    if time.monotonic() < deadline - 60:
        a_fin = fienup_compress(best_a, n_iters=300, gamma=0.7,
                                top_frac_init=0.10, top_frac_end=0.40, soft_alpha=20.0)
        sc = compute_score(a_fin)
        if sc < best_score:
            best_score = sc
            best_a = a_fin.copy()
            print(f"  Final Fienup: {sc:.8f}")

    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=45000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final L-BFGS: {sc:.8f}")

    # Perturbation restarts
    for trial in range(3):
        if time.monotonic() > deadline - 20:
            break
        noise_scale = 0.02 * (1.0 + trial * 0.5)
        a_p = best_a * (1.0 + noise_scale * rng.standard_normal(n_use))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S = float(a_p.sum())
        if S < 1e-12:
            break
        a_p = a_p / S
        a_r = fienup_compress(a_p, n_iters=200, gamma=0.8,
                              top_frac_init=0.10, top_frac_end=0.50, soft_alpha=30.0)
        sc = compute_score(a_r)
        if sc < best_score:
            best_score = sc
            best_a = a_r.copy()
            print(f"  Restart {trial}: {sc:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
