"""

import numpy as np
import time
from scipy.optimize import minimize


PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute the gradient of (w @ conv(a,a)) with respect to a."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad_a(a, beta):
    """LSE-smoothed objective with gradient, in direct a-space."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def lse_score_grad_log(b, beta):
    """LSE-smoothed objective in log-space: a = exp(b), optimize over b."""
    b = np.asarray(b, dtype=np.float64)
    n = int(b.size)
    a = np.exp(b - float(np.max(b)))
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    a = a / S  # normalize

    score, grad_a = lse_score_grad_a(a, beta)
    # Chain rule: grad_b = grad_a * a (since a = exp(b), da/db = a)
    grad_b = grad_a * a
    grad_b = np.nan_to_num(grad_b, nan=0.0, posinf=0.0, neginf=0.0)
    grad_b = np.clip(grad_b, -10.0, 10.0)
    return float(score), grad_b

def fun_lse_log(x, beta):

def estimate_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 32
    est = estimate_eval_time(n)
    mf = int((0.8 * t_left) / max(est, 1e-4))
    mf = max(32, min(5000, mf))
    return mf


def run_lbfgs_a(a0, beta, deadline, maxfun):
    """L-BFGS-B in a-space with non-negativity bounds."""
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.85 * rem) / max(est, 1e-4))
        if cap < 32:
            cap = 32
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 500)

    maxfun = max(32, maxfun)
    maxiter = max(30, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
        "factr": 1e10,
    }

    try:
        res = minimize(
            fun_lse_a, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def run_lbfgs_log(b0, beta, deadline, maxfun):
    """L-BFGS-B in log-space (no bounds needed since exp is always positive)."""
    b = np.asarray(b0, dtype=np.float64).ravel()
    n = len(b)
    a_check = np.exp(b - float(np.max(b)))
    a_check = normalize_a(a_check)

    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a_check), a_check
        est = estimate_eval_time(n)
        cap = int((0.85 * rem) / max(est, 1e-4))
        if cap < 32:
            cap = 32
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 500)

    maxfun = max(32, maxfun)
    maxiter = max(30, maxfun // 2)
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(
            fun_lse_log, b, method="L-BFGS-B", jac=True,
            args=(float(beta),), options=opts
        )
    except Exception:
        return exact_score_a(a_check), a_check

    if getattr(res, "x", None) is not None:
        b_out = res.x
        a_out = np.exp(b_out - float(np.max(b_out)))
        a_out = normalize_a(a_out)
    else:
        a_out = a_check

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a_check
        sc = exact_score_a(a_check)
    return float(sc), a_out


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def peak_subgrad_step(a, eta, k_peaks=5):
    """Subgradient step targeting only the top-k peaks of the convolution."""
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return a
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 1e-12:
        return a
    w = active_weights(d, k_peaks)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def peak_equalize_step(a, n_iter=50, eta=0.05):
    """Iteratively equalize the top peaks of the convolution."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for _ in range(n_iter):
        d, S = d_from_a(a)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax <= 1e-12:
            break
        # Use top peaks with soft weighting
        k = int(min(10, max(3, d.size // 50)))
        thresh = np.quantile(d, 1.0 - 1.0 / k)
        w = np.zeros(d.size)
        mask = d >= thresh
        w[mask] = (d[mask] - thresh) / (dmax - thresh + 1e-12)
        w = w / float(w.sum())
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            break
        a = a * np.exp(-eta * h / hmax)
        a = normalize_a(a)
        sc = exact_score_a(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_sc, best_a


def subgrad_polish(a, max_iters=40, deadline=None):
    """Polish using peak-subgradient with various step sizes and peak counts."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc

        for k in (3, 5, 8, 12, 20):
            for mult in (0.01, 0.03, 0.06, 0.10, 0.20, 0.40):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = peak_subgrad_step(a, mult, k)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def generate_structured_inits(n, rng, parent, cap=40):
    """Generate diverse structured initializations."""
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        if float(np.sum(arr)) < 1e-12:
            arr = np.ones(n, dtype=np.float64)
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Slight perturbations of parent
        for s in (0.05, 0.15, 0.3):
            add(pr * np.exp(s * rng.standard_normal(n)))

    x = np.linspace(0.0, 1.0, n)
    xc = 2.0 * x - 1.0  # centered [-1, 1]

    # Uniform
    add(np.ones(n, dtype=np.float64))

    # V-shapes (high at ends, low middle) - key for reducing peak
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(xc))
        add(1.0 + alpha * xc ** 2)
        add(1.0 + alpha * xc ** 4)

    # Inverse V (peaked in middle)
    for alpha in (0.5, 1.0, 2.0):
        add(np.exp(-alpha * xc ** 2))

    # Cosine shapes
    for f in (1, 2, 3, 4):
        for alpha in (0.2, 0.5, 0.8, 1.0):
            add(1.0 + alpha * np.cos(np.pi * f * x))
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(np.abs(1.0 + alpha * np.cos(np.pi * f * x)) + 0.01)

    # Linear ramps
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(0.1 + alpha * x)
        add(0.1 + alpha * (1.0 - x))
        add(0.1 + alpha * np.abs(xc))

    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + 0.01)
        add(np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        # Symmetric exponential (U-shape)
        add(np.exp(-rate * xc ** 2) + np.exp(-rate * (1.0 - np.abs(xc)) ** 2))

    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2):
        for sep in (0.15, 0.25, 0.35):
            add(np.exp(-(((x - (0.5 - sep)) / width) ** 2)) +
                np.exp(-(((x - (0.5 + sep)) / width) ** 2)) + 0.01)

    # Sine-based
    for f in (1, 2, 3, 4, 6, 8):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))
        add(1.0 + 0.8 * np.sin(2.0 * np.pi * f * x))

    # Raised cosine window
    add(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    add(0.5 * (1.0 + np.cos(np.pi * x)))

    # Hamming/Hanning-like
    add(0.54 + 0.46 * np.cos(2.0 * np.pi * x))
    add(0.54 + 0.46 * np.cos(np.pi * x))

    # Blackman-like
    add(0.42 + 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x))
    add(np.abs(0.42 + 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x)) + 0.05)

    # Random smooth (spectral synthesis)
    Nf = next_power_of_two(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(1.0, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)

    # Random trigonometric
    for _ in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(3, 10))
        for _ in range(K):
            f = int(rng.integers(1, max(2, min(n // 4, 64))))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns (piecewise constant)
    for nb in (8, 16, 32, 64, 128):
        if nb > n:
            nb = max(2, n // 2)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        # Make it V-shaped at block level
        block_order = np.argsort(np.abs(np.arange(nb) - nb // 2))[::-1]
        block_vals_v = np.zeros(nb)
        for rank, idx in enumerate(block_order):
            block_vals_v[idx] = 1.0 - 0.8 * rank / nb
        per = max(1, n // nb)
        arr = np.repeat(block_vals_v, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals_v[-1], n - arr.size)])
        add(arr[:n])

    # "Spike at ends" - very aggressive V
    add(np.ones(n, dtype=np.float64) * 0.1)
    for width in (5, 10, 20, 50):
        arr = np.ones(n, dtype=np.float64) * 0.1
        arr[:width] = 1.0
        arr[-width:] = 1.0
        add(arr)

    # Pure random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):
    """Create a perturbed version of the best sequence."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights(d, int(rng.integers(2, 10)))
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smooth perturbation in log-space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(5, int(n) // 200))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.5 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape boost
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Anti-smoothing (sharpen)
        k = int(rng.integers(3, max(5, n // 200)))
        if k % 2 == 0:
            k += 1
        c = k // 2
        kern = np.zeros(k)
        kern[c] = 1.0 + strength
        kern[c-1] = -strength / 2.0
        if c - 2 >= 0:
            kern[c-2] = -strength / 4.0
        if c + 1 < k:
            kern[c+1] = -strength / 2.0
        if c + 2 < k:
            kern[c+2] = -strength / 4.0
        q = np.convolve(p, kern, mode="same")
    elif mode == 8:
        # Random frequency modulation
        x = np.linspace(0.0, 1.0, n)
        f = float(rng.uniform(1, max(2, n // 20)))
        amp = float(rng.uniform(0.05, 0.3))
        q = p * (1.0 + amp * np.cos(2.0 * np.pi * f * x + rng.uniform(0, 2 * np.pi)))
    else:
        # Random block scale
        nb = int(rng.integers(4, max(5, n // 100)))
        per = max(1, n // nb)
        scales = 1.0 + strength * rng.standard_normal(nb)
        scales = np.clip(scales, 0.5, 1.5)
        s = np.repeat(scales, per)
        if s.size < n:
            s = np.concatenate([s, np.ones(n - s.size)])
        q = p * s[:n]

    return normalize_a(np.clip(q, 0.0, None))


def optimize_with_continuation(a_start, deadline, rng, label, use_log=True):
    """Multi-stagecontinuation optimization with LSE."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(20.0, float(deadline - t0) - 10.0)
    n = len(a)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Continuation schedule: start smooth, get sharp
    stages = [
        (50.0, 0.04),
        (150.0, 0.06),
        (400.0, 0.08),
        (1000.0, 0.10),
        (3000.0, 0.10),
        (8000.0, 0.10),
        (20000.0, 0.10),
        (50000.0, 0.10),
        (100000.0, 0.10),
        (200000.0, 0.10),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, n)
            if mf < 32:
                break

            if use_log:
                b = np.log(a + 1e-15)
                sc, a2 = run_lbfgs_log(b, float(beta), deadline, mf)
            else:
                sc, a2 = run_lbfgs_a(a, float(beta), deadline, mf)

            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} elapsed={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0
                ))
            a = normalize_a(a2)

        # Subgradient polish between stages
        if time.monotonic() < sub_dead - 5.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=15, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad best {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def quick_optimize(a, deadline, use_log=True):
    """Rapid multi-beta optimization."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for beta in (100, 300, 1000, 3000, 10000, 30000, 100000):
        rem = float(deadline - time.monotonic())
        if rem < 8.0:
            break
        mf = estimate_maxfun(rem * 0.8, len(a))
        if mf < 32:
            break
        if use_log:
            b = np.log(a + 1e-15)
            sc, a2 = run_lbfgs_log(b, float(beta), deadline, mf)
        else:
            sc, a2 = run_lbfgs_a(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    # Peak equalization
    rem = float(deadline - time.monotonic())
    if rem > 5.0:
        sc_pe, a_pe = peak_equalize_step(a, n_iter=30, eta=0.05)
        if np.isfinite(sc_pe) and sc_pe < best_sc:
            best_sc = sc_pe
            best_a = a_pe.copy()

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Get parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(p.sum()) > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine lengths to try - emphasize longer sequences
    lengths = []
    if parent is not None:
        parent_len = int(parent.size)
        lengths.append(parent_len)
        for L in [parent_len, 12288, 16384, 20480, 24576, 28672, 32768]:
            if L != parent_len and L >= 4096 and L <= 32768:
                lengths.append(L)
    else:
        lengths = [10240, 12288, 16384, 20480, 14336]

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        L = int(L)
        if L not in seen and 2048 <= L <= 32768:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:6]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Quick evaluation of diverse initializations
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        n_cap = int(min(n, 24576))  # cap for memory
        if n_cap < 1024:
            continue
        inits = generate_structured_inits(n_cap, rng, parent, cap=25 if n_cap > 12000 else 35)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n_cap))

    if not all_scores:
        n0 = lengths[0] if lengths else 10240
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.8f} (length {})".format(
        all_scores[int(order[0])], all_lens[int(order[0])]))
    print("Top 5 init scores:", ["{:.6f}(n={})".format(all_scores[int(order[i])], all_lens[int(order[i])]) for i in range(min(5, len(order)))])

    # Select top starts (at most 4, from different lengths)
    starts = []
    used_lens = set()

    if parent is not None and parent.size > 0:
        target_len = int(min(parent.size, 16384))
        ps = normalize_a(resample_sequence(parent, target_len))
        starts.append(ps)
        used_lens.add(target_len)

    for idx in order:
        if len(starts) >= 4:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)

    while len(starts) < 2:
        if starts:
            starts.append(perturb_best(starts[0], rng))
        else:
            starts.append(np.ones(10240, dtype=np.float64) / 10240.0)

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.60, 0.40]
    elif len(starts) == 3:
        fracs = [0.50, 0.30, 0.20]
    else:
        fracs = [0.45, 0.25, 0.18, 0.12]

    main_end = deadline - 400.0
    main_budget = max(100.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))

        if time.monotonic() > sub_dead - 30.0:
            continue

        sq = starts[si].copy()
        # For secondary starts, slightly blend with current best
        if rank > 0 and np.isfinite(best_score) and len(starts[si]) == len(best_seq):
            t_blend = 0.3 if rank == 1 else 0.5
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * best_seq)

        use_log = (rank == 0 and len(sq) <= 16384)
        sc, seq = optimize_with_continuation(sq, sub_dead, rng, "traj{}".format(rank + 1), use_log=use_log)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f} (n={})".format(rank + 1, best_score, len(seq)))

    best_seq = normalize_a(best_seq)
    print("After main phase: {:.8f} (n={})".format(best_score, len(best_seq)))

    # Phase 3: Peak equalization on the best
    if time.monotonic() < deadline - 250.0:
        sc, seq = peak_equalize_step(best_seq, n_iter=100, eta=0.04)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Peak equalize: {:.8f}".format(best_score))

    # Phase 4: Subgradient polish
    if time.monotonic() < deadline - 200.0:
        sc, seq = subgrad_polish(best_seq, max_iters=50, deadline=deadline - 200.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # Phase 5: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 200.0 and perturb_done < 8:
        left = float(deadline - 200.0 - time.monotonic())
        if left < 45.0:
            break
        n_remaining = 8 - perturb_done
        chunk = time.monotonic() + left / float(n_remaining)
        p = perturb_best(best_seq, rng, strength=float(rng.uniform(0.1, 0.4)))
        sc, seq = quick_optimize(p, chunk, use_log=(len(p) <= 16384))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 6: Final high-beta polish (multiple passes)
    if time.monotonic() < deadline - 40.0:
        for final_beta in (50000, 100000, 200000):
            rem = float(deadline - time.monotonic() - 10.0)
            if rem < 8.0:
                break
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            if mf < 32:
                break
            if len(best_seq) <= 16384:
                b = np.log(best_seq + 1e-15)
                sc, seq = run_lbfgs_log(b, float(final_beta), deadline, mf)
            else:
                sc, seq = run_lbfgs_a(best_seq, float(final_beta), deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final beta {}: {:.8f}".format(final_beta, best_score))

            # Quick subgrad after each
            if time.monotonic() < deadline - 20.0:
                sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 15.0)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()

    # Phase 7: Very final peak equalization
    if time.monotonic() < deadline - 10.0:
        sc, seq = peak_equalize_step(best_seq, n_iter=50, eta=0.02)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final peak equalize: {:.8f}".format(best_score))

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
