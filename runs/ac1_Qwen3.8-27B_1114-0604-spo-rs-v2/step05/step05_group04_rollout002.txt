def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_adaptive(a, n_iters=300, gamma=0.8, top_frac_init=0.10, top_frac_end=0.50,
                    soft_gate_alpha=30.0, use_optimal_thresh=False):
    """Fienup with optional optimal-threshold binary search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)

        if use_optimal_thresh:
            # Binary search for threshold that minimizes negative mass after IFFT
            lo = 0.0
            hi = float(P.max())
            best_thresh = float(np.quantile(P, 1.0 - top_frac))
            for _ in range(4):
                mid = 0.5 * (lo + hi)
                scale = np.ones_like(P)
                above = P > mid
                if above.any():
                    ratio = mid / P[above]
                    scale[above] = ratio ** 0.5 * gamma
                A_mod = A * scale
                a_test = np.fft.irfft(A_mod, M_fft)[:n]
                neg_mass = float(np.sum(np.minimum(a_test, 0.0) ** 2))
                if neg_mass > best_sc * 1e-6:  # heuristic: too much negative
                    hi = mid
                else:
                    lo = mid
                    best_thresh = mid
            thresh = best_thresh
        else:
            thresh = float(np.quantile(P, 1.0 - top_frac))

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_gate_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_gate_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def fienup_multilevel(a, n_iters=200, n_levels=6):

        A = np.fft.rfft(a, M_fft)

def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=3000, top_m=20, step_init=0.04, decay=0.9999):
    """Subgradient descent on top-m peaks of autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def peak_chasing(a, n_iters=2000, n_peaks=10, lr_init=0.02, lr_decay=0.9997):
    """
    Iterative peak chasing: directly reduce the worst autocorrelation peaks
    by adjusting the time-domain signal via targeted gradient steps.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_c = c[top_idx]
        M_val = float(top_c[0])

        # Compute gradient contribution from top peaks
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += 2.0 * seg[::-1]

        # Subtract uniform gradient for normalization
        S2 = S * S
        S3 = S2 * S
        total_grad = (2.0 * n) * (grad / S2 - 2.0 * M_val / S3)

        # Adaptive step
        gn = float(np.linalg.norm(total_grad))
        if gn < 1e-15:
            break
        step = min(lr, max((sc - 2.0 * n / (2.0 * n - 1.0)) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)

        a -= step * total_grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        lr *= lr_decay

    return best_a


def build_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(0.3 + 0.7 * (1.0 - x))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses multi-phase approach: L-BFGS, adaptive Fienup, peak chasing, subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ', score=' + f'{compute_score(parent):.6f}' + ')' if parent is not None else 'no'}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ==================================================================
    # Phase 1: Aggressive L-BFGS with LSE (first ~400s)
    # ==================================================================
    phase1_end = t_start + 400.0
    print(f"\n=== Phase 1: L-BFGS LSE (deadline t={phase1_end - t_start:.0f}s) ===")

    # Start from parent
    a_current = best_a.copy()
    a_current = a_current / a_current.sum()

    betas_phase1 = [50, 100, 200, 500, 1000, 2000]
    for beta in betas_phase1:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(8000, int(remaining * 0.8 * 20))
        maxfun = maxiter * 15
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}", flush=True)
        sc, a_opt = run_lbfgs(a_current, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"L1-B={beta}")
        if improved:
            a_current = a_opt.copy() / a_opt.sum()
        else:
            # If no improvement, try from a slightly perturbed version
            a_pert = a_current * (1.0 + 0.03 * rng.standard_normal(len(a_current)))
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = a_pert / a_pert.sum()
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            if update_best(sc2, a_opt2, f"L1-B={beta}-PERT"):
                a_current = a_opt2.copy() / a_opt2.sum()

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Adaptive spectral methods (~400-1000s)
    # ==================================================================
    phase2_end = t_start + 1000.0
    print(f"\n=== Phase 2: Spectral Methods (deadline t={phase2_end - t_start:.0f}s) ===")

    a_start = best_a.copy() / best_a.sum()

    # Variant A: Standard adaptive Fienup
    if time.monotonic() < phase2_end:
        a_f = fienup_adaptive(a_start, n_iters=300, gamma=0.7, top_frac_init=0.08,
                              top_frac_end=0.55, soft_gate_alpha=25.0)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FIENUP-ADAPT-A")

    # Variant B: Fienup with optimal threshold
    if time.monotonic() < phase2_end:
        a_f2 = fienup_adaptive(a_start, n_iters=250, gamma=0.85, top_frac_init=0.10,
                               top_frac_end=0.45, soft_gate_alpha=40.0, use_optimal_thresh=True)
        sc = compute_score(a_f2)
        update_best(sc, a_f2, "FIENUP-OPT-THRESH")

    # Variant C: Flat spectrum
    if time.monotonic() < phase2_end:
        a_flat = fienup_flat(a_start, n_iters=250, alpha_init=0.2, alpha_end=0.85)
        sc = compute_score(a_flat)
        update_best(sc, a_flat, "FIENUP-FLAT-C")

    # Variant D: Multi-level
    if time.monotonic() < phase2_end:
        a_ml = fienup_multilevel(a_start, n_iters=250, n_levels=8)
        sc = compute_score(a_ml)
        update_best(sc, a_ml, "FIENUP-MULTI-D")

    # Chain: apply multiple Fienup variants sequentially
    if time.monotonic() < phase2_end:
        a_chain = best_a.copy() / best_a.sum()
        a_chain = fienup_adaptive(a_chain, n_iters=150, gamma=0.6, top_frac_init=0.10,
                                  top_frac_end=0.60, soft_gate_alpha=20.0)
        sc = compute_score(a_chain)
        update_best(sc, a_chain, "CHAIN-1")
        if sc < best_score:
            a_chain = fienup_flat(a_chain, n_iters=150, alpha_init=0.3, alpha_end=0.9)
            sc = compute_score(a_chain)
            update_best(sc, a_chain, "CHAIN-2")

    # Try from perturbed starting points
    if time.monotonic() < phase2_end - 60:
        for trial in range(2):
            noise = 0.05 * rng.standard_normal(n_work)
            a_pert = best_a * (1.0 + noise)
            a_pert = np.clip(a_pert, 0.0, None)
            S = a_pert.sum()
            if S < 1e-12:
                continue
            a_pert = a_pert / S
            a_f = fienup_adaptive(a_pert, n_iters=200, gamma=0.8, top_frac_init=0.12,
                                  top_frac_end=0.50, soft_gate_alpha=30.0)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"FIENUP-PERT-{trial}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Peak chasing + subgradient (~1000-1600s)
    # ==================================================================
    phase3_end = t_start + 1600.0
    print(f"\n=== Phase 3: Peak Chasing + Subgradient (deadline t={phase3_end - t_start:.0f}s) ===")

    a_start3 = best_a.copy() / best_a.sum()

    # Peak chasing with different numbers of peaks
    for n_peaks in [5, 10, 20, 40]:
        if time.monotonic() > phase3_end:
            break
        a_pc = peak_chasing(a_start3, n_iters=2000, n_peaks=n_peaks, lr_init=0.02)
        sc = compute_score(a_pc)
        update_best(sc, a_pc, f"PEAK-CHASE m={n_peaks}")

    # Subgradient with various top_m
    for top_m in [10, 25, 50]:
        if time.monotonic() > phase3_end:
            break
        a_sg = top_m_subgradient(best_a, n_iters=2000, top_m=top_m, step_init=0.03)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG m={top_m}")

    # Alt: L-BFGS after peak chasing
    if time.monotonic() < phase3_end - 60:
        for beta in [500, 2000]:
            if time.monotonic() > phase3_end:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=75000)
            update_best(sc, a_opt, f"P3-LBFGS b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: High-beta L-BFGS refinement (~1600-2200s)
    # ==================================================================
    phase4_end = t_start + 2200.0
    print(f"\n=== Phase 4: High-beta L-BFGS (deadline t={phase4_end - t_start:.0f}s) ===")

    betas_phase4 = [3000, 5000, 10000, 20000]
    for beta in betas_phase4:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 60:
            break
        maxiter = min(10000, int(remaining * 0.7 * 15))
        maxfun = maxiter * 12
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}", flush=True)
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"L4-B={beta}")
        if improved:
            # Small perturbation and retry
            a_pert = a_opt * (1.0 + 0.01 * rng.standard_normal(len(a_opt)))
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = a_pert / a_pert.sum()
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            update_best(sc2, a_opt2, f"L4-B={beta}-PERT")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final combination attempts (~2200-2400s)
    # ==================================================================
    print(f"\n=== Phase 5: Final Refinement ===")

    if time.monotonic() < deadline - 60:
        # Fienup + L-BFGS combo
        a_combo = fienup_adaptive(best_a, n_iters=100, gamma=0.7, top_frac_init=0.10,
                                  top_frac_end=0.40, soft_gate_alpha=30.0)
        sc = compute_score(a_combo)
        update_best(sc, a_combo, "FINAL-FIENUP")

        if time.monotonic() < deadline - 30:
            sc2, a_opt = run_lbfgs(a_combo, 10000, maxiter=3000, maxfun=40000)
            update_best(sc2, a_opt, "FINAL-LBFGS")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(4):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.02 + 0.01 * trial
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(n_work))
            a_p = np.clip(a_p, 0.0, None)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            # Quick L-BFGS from perturbed
            sc, a_r = run_lbfgs(a_p, 5000, maxiter=2000, maxfun=30000)
            update_best(sc, a_r, f"RESTART-{trial}")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
