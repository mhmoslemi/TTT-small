"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using Adam optimization with LSE surrogate, multi-stage continuation,

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def compute_H(a, w):
    """Compute H[i] = sum_j a[j]*(w[j+i] + w[i-j]) for i=0..n-1."""
    n = len(a)
    len_w = len(w)
    N = next_power_of_two(n + len_w + 4)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv1 = np.fft.irfft(A_rev * W, N)  # conv(a_rev, w)
    conv2 = np.fft.irfft(A * W, N)       # conv(a, w)
    term1 = conv1[n-1:n-1+n]
    term2 = conv2[0:n]
    return term1 + term2

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_grad(a, beta):
    """Compute LSE surrogate value and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    
    H = compute_H(a, w)
    grad = (2.0 * float(n) * H) / (S * S) - (2.0 * float(n) * LDS * a) / (S ** 3)
    
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad

class AdamOptimizer:
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = None
        self.v = None
    
    def reset(self, n):
        self.t = 0
        self.m = np.zeros(n)
        self.v = np.zeros(n)
    
    def step(self, x, grad):
        self.t += 1
        n = len(x)
        if self.m is None or len(self.m) != n:
            self.reset(n)
        
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad ** 2
        
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        
        update = m_hat / (np.sqrt(v_hat) + self.eps)
        return x - self.lr * update

def adam_optimize(a, beta, n_steps, lr, deadline, lr_decay=1.0):
    """Run Adam optimization for n_steps or until deadline."""
    a = normalize_a(a)
    n = len(a)
    opt = AdamOptimizer(lr=lr)
    opt.reset(n)
    
    best_sc = exact_score_a(a)
    best_a = a.copy()
    
    for step in range(int(n_steps)):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        val, grad = lse_grad(a, beta)
        a_new = opt.step(a, grad)
        a_new = np.clip(a_new, 0.0, None)
        
        # Adaptive lr: reduce if not making progress
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic exact evaluation and lr adjustment
        if step % 50 == 0:
            sc = exact_score_a(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            # Decay lr slightly
            opt.lr *= lr_decay
    
    return best_sc, best_a

def active_set_equalize(a, n_top=5, n_steps=50, deadline=None):
    """Iteratively reduce the top autocorrelation peaks."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for step in range(int(n_steps)):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        d = c / c[0] if c[0] > 0 else c  # normalize by first element
        # Actually let's use the raw d = n*c/S^2
        n = len(a)
        S = float(a.sum())
        d = (float(n) * c) / (S * S)
        dmax = float(np.max(d))
        
        # Find top-k lags
        k = min(n_top, len(d))
        top_idx = np.argpartition(d, -k)[-k:]
        top_val = d[top_idx].max()
        
        # Compute gradient contributions from these top lags
        w = np.zeros(len(d))
        w[top_idx] = 1.0 / k
        H = compute_H(a, w)
        
        # Step: reduce a where it contributes to top lags
        eta = 0.01  # small step
        a_new = a - eta * H / (float(np.max(np.abs(H))) + 1e-12)
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_a(a_new)
        
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # Try smaller step
            a_new2 = a - eta * 0.3 * H / (float(np.max(np.abs(H))) + 1e-12)
            a_new2 = np.clip(a_new2, 0.0, None)
            a_new2 = normalize_a(a_new2)
            sc2 = exact_score_a(a_new2)
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break
    
    return best_sc, best_a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def perturb(a, rng, strength=0.3):
    """Perturb a sequence for diversity."""
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1-t) * a + t * np.ones(n) / n
    elif mode == 2:
        q = a[::-1].copy()
    elif mode == 3:

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using Adam optimization with LSE surrogate, multi-stage continuation,
active-set equalization, power-mean surrogates, and diverse multi-start.
"""

import numpy as np
import time

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def compute_H(a, w):
    """Compute H[i] = sum_j a[j]*(w[j+i] + w[i-j]) for i=0..n-1."""
    n = len(a)
    len_w = len(w)
    N = next_power_of_two(n + len_w + 4)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv1 = np.fft.irfft(A_rev * W, N)
    conv2 = np.fft.irfft(A * W, N)
    term1 = conv1[n - 1:n - 1 + n]
    term2 = conv2[0:n]
    return term1 + term2

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    H = compute_H(a, w)
    grad = (2.0 * float(n) * H) / (S * S) - (2.0 * float(n) * LDS * a) / (S ** 3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad

def power_mean_grad(a, p):
    """Power mean surrogate: M_p = (1/K * sum d_k^p)^(1/p)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    p = float(p)
    if p < 2:
        p = 4.0
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 1e-20)
    dp = d ** p
    sum_dp = float(dp.sum())
    K = len(d)
    if sum_dp < 1e-30:
        return 1e10, np.zeros(n, dtype=np.float64)
    M = (sum_dp / K) ** (1.0 / p)
    w = dp / sum_dp
    LDS = float(np.dot(w, d))
    H = compute_H(a, w)
    grad = (2.0 * float(n) * H) / (S * S) - (2.0 * float(n) * LDS * a) / (S ** 3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * M), grad

class AdamOpt:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = None
        self.v = None

    def reset(self, n):
        self.t = 0
        self.m = np.zeros(n)
        self.v = np.zeros(n)

    def step(self, x, grad):
        self.t += 1
        n = len(x)
        if self.m is None or len(self.m) != n:
            self.reset(n)
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * (grad * grad)
        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def adam_run(a, beta, n_steps, lr, deadline, use_pmean=False, p_power=6.0):
    a = normalize_a(a)
    n = len(a)
    opt = AdamOpt(lr=lr)
    opt.reset(n)
    best_sc = exact_score_a(a)
    best_a = a.copy()
    warmup = max(10, n_steps // 10)

    for step in range(int(n_steps)):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Learning rate warmup
        if step < warmup:
            eff_lr = lr * (step + 1) / float(warmup)
        else:
            eff_lr = lr
        opt.lr = eff_lr

        if use_pmean:
            val, grad = power_mean_grad(a, p_power)
        else:
            val, grad = lse_grad(a, beta)

        a_new = opt.step(a, grad)
        a_new = np.clip(a_new, 0.0, None)

        if step % 100 == 0 or step == n_steps - 1:
            a_new = normalize_a(a_new)
            sc = exact_score_a(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        a = a_new

    sc = exact_score_a(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_sc, best_a

def active_set_eq(a, n_top=5, n_steps=40, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for step in range(int(n_steps)):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_self_conv(a)
        S = float(a.sum())
        d = (float(n) * c) / (S * S)
        k = min(n_top, len(d))
        top_idx = np.argpartition(d, -k)[-k:]
        w = np.zeros(len(d))
        w[top_idx] = 1.0 / k
        H = compute_H(a, w)
        hmax = float(np.max(np.abs(H)))
        if hmax < 1e-15:
            break
        for eta_mult in (0.02, 0.005, 0.001):
            a_try = a - eta_mult * H / hmax
            a_try = np.clip(a_try, 0.0, None)
            a_try = normalize_a(a_try)
            sc = exact_score_a(a_try)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
                a = a_try
                break

    return best_sc, best_a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

    elif mode == 4:

        d = (float(n) * c) / (S * S)
        n_top = int(rng.integers(3, 10))

        std = np.std(noise) + 1e-8
        q = a * np.exp(0.2 * strength * noise / std)
    return normalize_a(np.clip(q, 0.0, None))

def generate_inits(n, rng, parent):
    inits = []
    x = np.linspace(0, 1, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.001, None)
        inits.append(normalize_a(arr))

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        add(np.abs(2.0 * x - 1.0) ** p + 0.05)

    for width in (0.03, 0.05, 0.08, 0.12, 0.2):
        add(np.exp(-((x - 0.05) / width) ** 2) + np.exp(-((x - 0.95) / width) ** 2) + 0.05)

    for f in (1, 2, 3, 4):
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * f * x))
        add(1.0 + 0.8 * np.abs(np.cos(np.pi * f * x)))

    Nf = next_power_of_two(n)
    for _ in range(10):
        gamma = float(rng.uniform(1.0, 3.5))
        power = 1.0 / ((np.abs(np.fft.rfftfreq(Nf) * Nf) + 1.0) ** gamma)
        power[0] = 0.0
        noise = np.fft.irfft(power * np.exp(1j * rng.uniform(-np.pi, np.pi, len(power))), Nf)[:n]
        noise -= noise.mean()
        amp = float(rng.uniform(0.3, 1.0))
        add(1.0 + amp * noise / (np.std(noise) + 1e-8))

    for _ in range(4):
        add(0.05 + 0.95 * rng.random(n))

    for _ in range(3):
        vals = np.ones(n)
        K = int(rng.integers(2, 8))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    return inits

def optimize_trajectory(a_start, deadline, rng, label, n_stages=15):
    a = normalize_a(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total_time = max(10.0, deadline - t0 - 5.0)

    print("[{}] start: {:.8f}, n={}".format(label, best_sc, n))

    # Mix of LSE and power-mean stages
    betas = np.geomspace(30, 200000, n_stages)
    weights = np.linspace(0.4, 2.5, n_stages)
    weights /= weights.sum()

    for stage in range(n_stages):
        if stage == 0:
            cum_w = weights[0]
        else:
            cum_w = np.cumsum(weights)[stage]
        stage_dead = t0 + cum_w * total_time
        stage_dead = min(stage_dead, deadline - 5.0)

        if time.monotonic() > stage_dead - 3.0:
            continue

        beta = float(betas[stage])
        # Alternating: even stages use LSE, odd use power mean
        use_pm = (stage % 3 == 2)
        lr = 0.02 * (1.0 / (1.0 + 0.05 * stage))
        n_steps = 5000

        sc, a_new = adam_run(a, beta, n_steps, lr, stage_dead, use_pmean=use_pm, p_power=8.0)

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print("[{}] stage {} beta={:.0f} {}: {:.8f}".format(label, stage, beta, "PM" if use_pm else "LSE", best_sc))

        a = normalize_a(a_new)

        # Active set equalization
        if time.monotonic() < stage_dead - 8.0:
            sc_eq, a_eq = active_set_eq(a, n_top=3, n_steps=20, deadline=stage_dead - 3.0)
            if np.isfinite(sc_eq) and sc_eq < best_sc:
                best_sc = sc_eq
                best_a = a_eq.copy()
                print("[{}] eq stage {}: {:.8f}".format(label, stage, best_sc))
            a = normalize_a(a_eq)

    return best_sc, best_a

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    n_target = 10240
    print("Parent: {} (len={}), target n={}".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0, n_target))

    inits = generate_inits(n_target, rng, parent)
    scores = []
    for init in inits:
        sc = exact_score_a(init)
        scores.append(sc)

    valid = [(s, i) for s, i in zip(scores, range(len(inits))) if np.isfinite(s)]
    valid.sort(key=lambda x: x[0])

    print("Best init: {:.8f} (idx={})".format(valid[0][0], valid[0][1]))
    for s, i in valid[:5]:
        print("  Init[{}]: {:.8f}".format(i, s))

    # Select diverse top starts
    starts = []
    for s, i in valid[:6]:
        if len(starts) >= 3:
            break
        starts.append(inits[i].copy())

    if parent is not None:
        pr = normalize_a(resample_sequence(parent, n_target))
        in_starts = False
        for st in starts:
            if np.allclose(pr, st, atol=0.01):
                in_starts = True
                break
        if not in_starts:
            if len(starts) >= 3:
                starts[2] = pr
            else:
                starts.append(pr)

    best_seq = starts[0].copy()
    best_score = min(exact_score_a(s) for s in starts)
    print("Initial best: {:.8f}".format(best_score))

    # Time allocation
    main_end = deadline - 250.0
    main_budget = max(60.0, main_end - t_start)

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.55, 0.28, 0.17]

    cum = 0.0
    for rank, start in enumerate(starts):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)
        if time.monotonic() > sub_dead - 30.0:
            continue

        sc, seq = optimize_trajectory(start, sub_dead, rng, "traj{}".format(rank + 1), n_stages=15)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Perturbation restarts
    pc = 0
    while time.monotonic() < deadline - 120.0 and pc < 5:
        left = deadline - 120.0 - time.monotonic()
        if left < 50.0:
            break
        chunk = time.monotonic() + left / float(5 - pc)
        p = perturb_seq(best_seq, rng, strength=float(rng.uniform(0.05, 0.3)))
        sc, seq = optimize_trajectory(p, chunk, rng, "pert{}".format(pc), n_stages=8)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb {}: {:.8f}".format(pc, best_score))
        pc += 1

    # Symmetry check
    sym = normalize_a(0.5 * (best_seq + best_seq[::-1]))
    sc_sym = exact_score_a(sym)
    if np.isfinite(sc_sym) and sc_sym < best_score:
        best_score = sc_sym
        best_seq = sym
        print("Symmetry improved: {:.8f}".format(best_score))

    # Final active set equalization
    if time.monotonic() < deadline - 40.0:
        sc, seq = active_set_eq(best_seq, n_top=5, n_steps=50, deadline=deadline - 30.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final eq: {:.8f}".format(best_score))

    # Final high-beta Adam
    if time.monotonic() < deadline - 20.0:
        rem = deadline - time.monotonic() - 5.0
        if rem > 10.0:
            sc, seq = adam_run(best_seq, 500000.0, 3000, 0.005, deadline - 5.0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High-beta final: {:.8f}".format(best_score))

    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
