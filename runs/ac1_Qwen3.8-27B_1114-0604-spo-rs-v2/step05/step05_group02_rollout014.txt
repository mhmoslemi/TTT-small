"""
Adaptive LSE + GS-Fienup hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key improvements: adaptive beta schedule, momentum subgradient, peak-directed coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        # All exp values underflowed; use uniform
        p = np.ones(len(c)) / len(c)
        L = c_max
    else:
        p = exp_scaled / exp_sum
        L = c_max + float(np.log(exp_sum)) / beta
    
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * conv_res / S2 - 2.0 * L * a / S3)
    
    return score, grad


def lse_wrapper(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs_adaptive(a0, betas, maxiter_per_beta=5000):
    """Run L-BFGS-B with a sequence of increasing betas."""
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 1.0 / len(a)
    a = a / a.sum()
    
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for beta in betas:
        bounds = [(0.0, 1000.0)] * len(a)
        res = minimize(
            lse_wrapper, a.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 5, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        if a_new.sum() < 1e-6:
            a_new = a.copy()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
