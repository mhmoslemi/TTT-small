        return np.empty(2*n-1, dtype=np.float64)
    N = _next_pow2(2*n-1)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[:2*n-1]
    return np.nan_to_num(conv, nan=0.0, posinf=0.0, neginf=0.0)

def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n < 1 or m < 1:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = _next_pow2(M)
    # Invert the filter: need to reverse a for correlation
    # FFT - Circular cross-correlation to full convolution
    # Mode: we want full convolution, so use the same method as _self_conv but with two different sequences.
    a_rev = np.flip(a)
    A = np.fft.rfft(a_rev, N)
    B = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * B, N)[:M]
    return corr[:n]

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    s = float(np.sum(a))
    if abs(s) < 1e-12:
        return np.ones_like(a) * target / float(max(1, a.size))
    return a * target / s

def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(np.sum(a))
    if abs(S) < 1e-12:
        return np.infty, np.zeros(n)
    c = _self_conv(a)
    M = 2*n - 1
    k = int(k)
    k = max(1, min(k, M))
    # Select the k largest values in the convolution (excluding the extremes)
    indices = np.argpartition(-c, k, axis=0)[:k]
    if indices.shape[0] == 0:
        return np.infty, np.zeros(n)
    c_active = c[indices]
    c_max_active = float(c_active.max())
    if c_max_active <= 0:
        return np.infty, np.zeros(n)
    z = np.log(c_active - c_max_active + 1e-15) * beta
    e = np.exp(z)
    if z.max() > 0:
        # Apply Lagrange multiplier to prevent divergence
        return np.infty, np.zeros(n)
    S_e = np.sum(e)
    if S_e > 0:
        w_active = e / S_e
    else:
        w_active = np.ones(k) / k
    # Set weights that contribute to the correlation
    grad_term = np.zeros(n)
    grad_term[indices] = w_active * c_active[indices]
    # The gradient of the score: derivative of 2*n*conv_max / S^2
    # We are using the active set to approximate the gradient
    grad = _cross_corr(grad_term, a) * beta
    grad = 4.0 * float(grad.max()) - 4.0 * float(grad.min())
    grad = np.nan_to_num(grad, nan=0.0, posinf=1e10, neginf=-1e10)
    return 2.0 * float(c_active.max()), grad

def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on active-set LSE objective with sum penalty"""
    try:
        obj = lambda x, beta=beta, k=k: _lse_obj_grad(x, beta, k)[0]
        grad = lambda x, beta=beta, k=k: _lse_obj_grad(x, beta, k)[1]
        bounds = [(0.0, None)] * len(a0)
        res = minimize(obj, a0, method='L-BFGS-B', jac=grad, options={
            'maxiter': int(0.8 * time_limit),
            'maxfun': int(0.8 * time_limit),
            'ftol': 1e-15,
            'gtol': 1e-11,
            'maxls': 40,
            '_restarts': 3,
            '_cooldown': 5.0,
            '_numentary': 200})
        out = np.asarray(res.x, dtype=np.float64)
        out = _norm(out)
    except Exception:
        out = np.asarray(a0)
    return _lse_obj_grad(out, beta, k)[0], out

def _subgrad_polish(a, max_iters=30, deadline=None):
    """Exact subgradient descent on the max-convolution objective"""
    if deadline is None:
        deadline = np.infty
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    c = _self_conv(a)
    M_val = float(c.max())
    if M_val <= 0.0 or abs(S) < 1e-8:
        if S == 0.0:
            a = np.ones(1) / 2048.0
        else:
            a = np.zeros(n)
        return _lse_obj_grad(_norm(a), 8000, n)[0], _norm(a)
    best_a = a.copy()
    best_score = _lse_obj_grad(a, 8000, n)[0]
    active_positions = (c == M_val)
    active_count = int(active_positions.sum())
    w_active = np.zeros(active_count)
    for i_iter in range(max_iters):
        if time() > deadline:
            break
        grad = _cross_corr(active_positions.reshape(active_count, 1), a)
        grad = grad.reshape(n)
        grad = grad / (1 + np.abs(grad).max() / 200.0)
        eta = 0.2 / (1 + i_iter)
        new_a = a - eta * grad
        new_a = np.clip(new_a, 0.0, 1000.0)
        new_a = _norm(new_a)
        score = _lse_obj_grad(new_a, 20000, active_count)[0]
        if score < best_score:
            best_score = score
            best_a = new_a.copy()
        a = new_a
    best_a = np.clip(best_a, 0.0, 1000.0)
    return best_score, best_a

def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks."""
    if not isinstance(sequence, (list, tuple, np.ndarray)):
        return np.inf
    for x in sequence:
        if not isinstance(x, (int, float)) or isinstance(x, bool) or (isinstance(x, complex) and np.imag(x) > 0):
            return np.inf
    sequence = [float(x) for x in sequence]
    b_sequence = np.convolve(sequence, sequence)
    max_b = float(np.max(b_sequence))
    sum_a = float(np.sum(sequence))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * float(len(sequence)) * max_b / (sum_a**2)
