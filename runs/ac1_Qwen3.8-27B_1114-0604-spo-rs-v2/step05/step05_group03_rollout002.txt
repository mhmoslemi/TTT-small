"""
Multi-phase optimizer for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses improved LP alternating (with precomputed FFT), golden-section line search,
adaptive FISTA with duality-gap monitoring, frequency-domain peak shaving,
and high-beta LSE L-BFGS polish. Explores multiple diverse trajectories.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=400000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fast_lp_fista(a, M, n_iters=8000, beta_start=20.0, beta_end=800.0,
                  x_init=None, ramp_frac=0.2):
    """
    Fast FISTA for: max sum(x) s.t. conv(a,x) <= M, x >= 0.
    Uses precomputed FFT of a for speed.
    Returns (x, duality_gap_estimate).
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, M_fft)

    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.dot(a, a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    ramp_end = int(n_iters * ramp_frac)
    last_gap = np.inf

    for it in range(n_iters):
        if it < ramp_end:
            frac = it / max(1, ramp_end)
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        # Compute c = conv(a, x) using precomputed A_fft
        X_fft = np.fft.rfft(x, M_fft)
        c = np.fft.irfft(A_fft * X_fft, M_fft)[:2 * n - 1]

        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # Duality gap estimate
        last_gap = M * exp_sum / beta - float(x.sum())
        if last_gap < 1e-6 * max(1.0, float(x.sum())):
            break

        # Gradient: (B^T p)_j = sum_k A[k,j] * p[k] = sum_k a[|k-j| boundary] * p[k]
        # More precisely: grad_j = sum_{k: valid} a_{k-j} * p_k where 0<=k-j<n and 0<=j<n
        # This equals: for each j, sum over valid k of a[(k-j)] * p[k]
        # Use correlation approach: grad = correlation(a, p) restricted to valid indices
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x, last_gap


def golden_section_search(f, a, b, tol=1e-4, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0


def lp_alternating_improved(a, max_outer=60, fista_iters=8000, time_limit=None,
                           x_warm=None):
    """
    Improved LP alternating with:
    - Precomputed FFT in FISTA
    - Golden section line search
    - Duality gap monitoring
    - Periodic diversification
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    rng = np.random.default_rng(123)

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x, gap = fast_lp_fista(a, M, n_iters=fista_iters, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Golden section line search on t in [0, 1]
        def obj_t(t):
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                return np.inf
            a_try = a_try / St
            return compute_score(a_try)

        t_best = golden_section_search(obj_t, 0.01, 0.99, tol=0.02, max_iter=30)
        a_new = (1.0 - t_best) * a + t_best * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                # Diversify
                a_new = a_new * (1.0 + 0.05 * rng.standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                no_improve = 0

        a = a_new

        if best_sc - 1.0 < 0.01:
            break

    return best_a, best_sc, x_warm


def freq_peak_shave(a, n_iters=100, alpha=0.5):
    """
    Frequency-domain peak shaving: modify |A(omega)|^2 to reduce peaks
    while maintaining a >= 0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-14:
            break

        # Identify and reduce peak positions in time domain
        thresh = conv_mean + alpha * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        # Newton step in frequency domain
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-14)
        small_mask = np.abs(A) < 1e-6 * np.max(np.abs(A))
        d[small_mask] = 0.0

        damp = min(1.0, 0.3 + 0.7 * it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_peak_descent(a, n_steps=300, rng=None):
    """
    Subgradient method targeting the maximum convolution coefficient.
    Uses multiple peak positions for robustness.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        M_val = float(c.max())

        # Pick top few peaks (not just the single max)
        top_k = max(3, len(c) // 20)
        sorted_idx = np.argsort(c)[::-1][:top_k]
        k_star = sorted_idx[rng.integers(top_k)]

        # Subgradient: d/d a_j of c_k_star = sum_j a_{k-star - j} * delta
        # c_k = sum_j a_{k-j} * a_j, so grad_wrt a_m = sum_j: a_{k-j} * delta_{j,m} + a_{k-m} * delta_{j,m}
        # = a_{k-m} + a_{k-m} (if k-m != m) ... actually:
        # c_k = sum_j a_j * a_{k-j}
        # d c_k / d a_m = a_{k-m} + sum_j a_j * d(a_{k-j})/d(a_m) = a_{k-m} + a_{k-m} (if k-m != m) or a_{k-m} (if k-m == m)
        # Actually: c_k = a_0*a_k + a_1*a_{k-1} + ... + a_k*a_0 (with boundary)
        # d c_k / d a_m = a_{k-m} + a_{k-m} if m != k-m, else a_{k-m}
        # More carefully: if we write c_k = sum_{j=0}^{n-1} a_j a_{k-j} (where a_i = 0 for i<0 or i>=n)
        # then d c_k / d a_m = a_{k-m} + (1 if m == k-m else 0) * a_m ... no
        # c_k = sum_j a_j * a_{k-j}, so the terms involving a_m are: j=m gives a_m * a_{k-m},
        # and k-j=m gives j=k-m, so a_{k-m} * a_m. If m != k-m, both terms appear.
        # So d c_k / d a_m = a_{k-m} + (1 if m != k-m else 0) * a_{k-m} ... 
        # Wait: c_k = sum_j a_j a_{k-j}. Terms with a_m: when j=m: a_m a_{k-m}; when k-j=m i.e. j=k-m: a_{k-m} a_m.
        # If m != k-m: derivative = a_{k-m} + a_{k-m} = 2 a_{k-m}
        # If m == k-m (i.e., k = 2m): derivative = a_m (only one term)
        # So grad_m = 2*a_{k-m} if k-m in [0,n-1] and k-m != m, else a_{k-m} if k-m == m, else 0
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                if j != m:
                    grad[m] = 2.0 * a[j]
                else:
                    grad[m] = a[j]

        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Adaptive step size
        base_step = min(0.02, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3)
        improved = False
        for mult in [0.3, 0.5, 1.0, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, None)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc - 1e-10:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        # Periodic diversification
        if it % 50 == 0 and not improved and rng.random() < 0.5:
            a = a * (1.0 + 0.03 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.01, 1000.0).copy())
        else:
            inits.append(np.interp(np.linspace(0, 1, n),
                                    np.linspace(0, 1, len(parent)),
                                    parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 40))
    inits.append(np.clip(0.5 + 0.4 * smooth, 0.01, 1000.0))
    inits.append(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    return inits


def propose_candidate(budget_s=2400):
    """
    Multi-phase search for optimal non-negative sequence.
    Phase 1: Diverse initialization + quick refinement (15%)
    Phase 2: LP alternating with golden section (45%)
    Phase 3: Frequency peak shaving + subgradient (25%)
    Phase 4: L-BFGS high-beta polish (15%)
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    total_time = max(60.0, float(budget_s))
    deadline = t_start + total_time - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine main length
    n_main = len(parent) if parent is not None else 6962

    # ================================================================
    # Phase 1: Diverse initialization + quick refinement (15% time)
    # ================================================================
    phase1_end = t_start + 0.15 * total_time
    print(f"\n=== Phase 1: Initialization (deadline t={phase1_end - t_start:.0f}s) ===")

    inits = build_diverse_inits(n_main, rng, parent)
    candidates = []
    for i, a0 in enumerate(inits[:4]):
        if time.monotonic() > phase1_end - 5:
            break
        a0 = np.maximum(a0, 0.0)
        S0 = float(a0.sum())
        if S0 > 1e-12:
            a0 = a0 / S0
        # Quick freq shaving
        a_ref = freq_peak_shave(a0, n_iters=40, alpha=0.5)
        sc = compute_score(a_ref)
        report(sc, a_ref, f"init{i}")
        candidates.append(a_ref)

    # Use best candidate as starting point for Phase 2
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    print(f"Phase 1 done: best={best_score:.8f}")

    # ================================================================
    # Phase 2: LP Alternating (45% time)
    # ================================================================
    phase2_end = t_start + 0.60 * total_time
    print(f"\n=== Phase 2: LP Alternating (deadline t={phase2_end - t_start:.0f}s) ===")

    x_warm = None
    # Progressive: start with fewer FISTA iters, increase
    fista_schedules = [
        (6000, 25),   # (fista_iters, max_outer)
        (8000, 30),
        (10000, 30),
        (12000, 25),
        (15000, 20),
    ]

    for fi, (fista_it, max_out) in enumerate(fista_schedules):
        if time.monotonic() > phase2_end - 20:
            break
        remaining = phase2_end - time.monotonic()
        # Estimate: each FISTA iter ~ 0.3ms for n~7000, so ~3s per LP solve
        est_per_outer = fista_it * 0.3e-3 + 0.5  # + line search
        max_out_use = min(max_out, max(3, int(remaining / max(1.0, est_per_outer))))
        print(f"  LP round {fi}: fista={fista_it}, max_outer={max_out_use}, "
              f"remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating_improved(
            a_work, max_outer=max_out_use, fista_iters=fista_it,
            time_limit=phase2_end - 10, x_warm=x_warm
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

        # Check convergence
        if best_score < 1.503:
            print(f"  Approaching target, continuing for polish...")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ================================================================
    # Phase 3: Peak shaving + subgradient (25% time)
    # ================================================================
    phase3_end = t_start + 0.85 * total_time
    print(f"\n=== Phase 3: Peak Shaving & Subgradient (deadline t={phase3_end - t_start:.0f}s) ===")

    # Alternate between freq peak shaving and subgradient
    a_cur = best_a.copy()
    S = float(a_cur.sum())
    if S > 0:
        a_cur = a_cur / S

    for round_id in range(6):
        if time.monotonic() > phase3_end - 10:
            break

        # Freq peak shaving
        a_shaved = freq_peak_shave(a_cur, n_iters=80, alpha=0.4)
        sc = compute_score(a_shaved)
        report(sc, a_shaved, f"shave{round_id}")
        a_cur = a_shaved

        # Subgradient descent
        a_sg = subgradient_peak_descent(a_cur, n_steps=200, rng=rng)
        sc = compute_score(a_sg)
        report(sc, a_sg, f"sg{round_id}")
        a_cur = a_sg

    # One more LP round if time allows
    if time.monotonic() < phase3_end - 60:
        print("  Extra LP round...")
        a_out, sc_out, _ = lp_alternating_improved(
            a_cur, max_outer=15, fista_iters=12000,
            time_limit=phase3_end - 10
        )
        report(sc_out, a_out, "LP extra")
        a_cur = a_out

    print(f"Phase 3 done: best={best_score:.8f}")

    # ================================================================
    # Phase 4: L-BFGS high-beta polish (15% time)
    # ================================================================
    print(f"\n=== Phase 4: L-BFGS Polish ===")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [100, 300, 1000, 3000, 8000, 15000]
    for beta in beta_schedule:
        if time.monotonic() > deadline - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        try:
            sc, a_opt = run_lbfgs(a_norm, beta, maxiter=8000, maxfun=300000)
        except Exception as e:
            print(f"error: {e}")
            break
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    # Final subgradient pass
    if time.monotonic() < deadline - 20:
        a_final = subgradient_peak_descent(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final SG")

    # Final freq shaving
    if time.monotonic() < deadline - 10:
        a_final = freq_peak_shave(best_a, n_iters=50, alpha=0.3)
        sc = compute_score(a_final)
        report(sc, a_final, "final shave")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
