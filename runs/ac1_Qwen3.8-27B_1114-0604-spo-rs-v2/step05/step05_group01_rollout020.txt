- Short sequences are fast to optimize (small FFT)

This is a QCQP (quadratically constrained quadratic program) because `c[k] = a^T Q_k a <= t` is a quadratic constraint. CVXPY might be able to handle this if Q_k is PSD (which it's not in general, as I showed earlier).

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using multiresolution LSE optimization, Nesterov-accelerated subgradient,
and local coordinate refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))

def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if n == 0:
        return 1e10, a
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a), a
    maxfun = max(32, int(maxfun))
    maxiter = max(20, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def subgrad_step_nesterov(a, velocity, eta, k_active=3):
    """Nesterov-accelerated subgradient step."""
    n = len(a)
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return a, velocity
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return a, velocity
    # Get top-k active indices
    m = d.size
    if k_active <= 0 or k_active >= m:
        w = np.ones(m, dtype=np.float64) / float(m)
    else:
        idx = np.argpartition(d, m - k_active)[m - k_active:]
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0
        w /= float(k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-12:
        return a, velocity
    # Gradient direction (we want to minimize, so go opposite to gradient)
    g = h / hmax
    # Nesterov: new_velocity = momentum * old_velocity - eta * gradient
    momentum = 0.9
    velocity = momentum * velocity - eta * g
    new_a = a + velocity
    new_a = np.clip(new_a, 0.0, None)
    return new_a, velocity

def subgrad_polish(a, max_iters=50, deadline=None, k_list=(1, 2, 3, 5)):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            for eta in (0.005, 0.01, 0.02, 0.05, 0.1):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q, _ = subgrad_step_nesterov(a, np.zeros(n), eta, k)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q.copy()
        if not improved and it > 10:
            break
    return float(best_sc), best_a

def generate_inits(n, rng, parent, cap=40):
    inits = []
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)
    
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    add(np.ones(n, dtype=np.float64))
    
    # V-shapes with various parameters
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    
    # Sharper V (power of distance from center)
    for p in (0.5, 1.0, 2.0, 3.0):
        add(1.0 + np.abs(2.0 * x - 1.0) ** p)
    
    # U-shape (1/distance from center, clipped)
    for alpha in (0.5, 1.0, 2.0):
        d = np.abs(2.0 * x - 1.0)
        add(1.0 + alpha / (d + 0.1))
    
    # Cosine patterns
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(np.abs(np.cos(2.0 * np.pi * f * x)) + 0.1)
    
    # Sine patterns
    for f in (1, 2, 3):
        add(np.abs(np.sin(np.pi * x * f)) + 0.1)
        add(1.0 / (0.5 + 0.5 * np.sin(np.pi * x * f)))
    
    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(2.0 * x - 1.0)) + 0.01)
    
    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2):
        for center in (0.2, 0.25, 0.3, 0.35):
            g1 = np.exp(-(((x - center) / width) ** 2))
            g2 = np.exp(-(((x - (1.0 - center)) / width) ** 2))
            add(g1 + g2 + 0.01)
    
    # Linear ramps
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))
    
    # "M" shape (two peaks with dip in middle)
    for dip in (0.1, 0.3, 0.5):
        m_shape = 1.0 - dip * np.cos(2.0 * np.pi * x)
        add(np.maximum(m_shape, 0.01))
    
    # Random smooth
    Nf = next_power_of_two(2 * n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(1.0, 4.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.3, 1.5))
        add(1.0 + amp * noise)
    
    # Random trig
    for _ in range(8):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 7))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)
    
    # Block patterns
    for nb in (2, 4, 8, 16, 32):
        if nb > n:
            continue
        per = max(1, n // nb)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])
    
    # Pure random
    for _ in range(5):
        add(0.01 + 0.99 * rng.random(n))
    
    return inits
