"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed optimization with geometric annealing, coordinate descent,
push-pull active-set heuristics, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute correlation of a with weight vector w, centered."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_score_grad(a, beta, pen_sum=0.0):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 10.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L
    if pen_sum > 0:
        score += pen_sum * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    if pen_sum > 0:
        grad += 2.0 * pen_sum * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def fun_lse(a, beta, pen_sum):
    return lse_score_grad(a, beta, pen_sum)

def run_lbfgs(a0, beta, deadline, maxfun, pen_sum=0.0):

    n = len(a)
    if deadline is not None:

    }

            fun_lse, a, method="L-BFGS-B", jac=True,

    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def coordinate_descent(a, active_lags, max_sweeps=3, deadline=None):
    """
    Coordinate descent: for each position i, find the optimal value of a[i]
    given all other positions fixed, considering only the active lags.
    """
    a = normalize_a(a).copy()
    n = len(a)
    S = float(a.sum())
    best_sc = exact_score_a(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Try a few values around current
            # The contribution of a[i] to lag k is a[i] * a[k-i] (or a[i] * a[i-k])
            # We want to find a[i] that minimizes max over active lags
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
                
            # Compute current contributions to active lags from position i
            # For lag k, contribution is a[i]*a[k-i] if 0<=k-i<n, plus a[i]*a[i-k] if 0<=i-k<n (for k != 0)
            # Actually for autocorrelation: c(k) = sum_j a[j]*a[j+k] (for k>=0)
            # So a[i] appears in c(k) as: a[i]*a[i+k] (if i+k < n) + a[i-k]*a[i] (if i-k >= 0)
            # For k=0: a[i]^2
            
            # For simplicity, try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5, old_val * 0.25]
            best_val = old_val
            best_local = None
            
            for cv in candidates:
                cv = max(0.0, cv)
                a[i] = cv
                S_new = S - old_val + cv
                if S_new < 1e-12:
                    a[i] = old_val
                    continue
                # Quick check: compute score
                c = fast_self_conv(a)
                cmax = float(np.max(c))
                sc = 2.0 * n * cmax / (S_new * S_new)
                if best_local is None or sc < best_local:
                    best_local = sc
                    best_val = cv
            
            a[i] = best_val
            S = S - old_val + best_val
            
            if deadline is not None and time.monotonic() > deadline:
                break
    
    a = normalize_a(a)
    sc = exact_score_a(a)
    if sc < best_sc:
        return a
    return a  # Always return the updated version

def push_pull_step(a, strength=0.1, top_k=5):
    """
    Identify positions that contribute most to the peak convolution lags,
    reduce them; positions that contribute least, increase them.
    """
    a = normalize_a(a).copy()
    n = len(a)
    S = float(a.sum())
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    
    # Find top k lags
    k = min(top_k, len(d))
    idx = np.argpartition(d, -k)[-k:]
    w = np.zeros(len(d))
    w[idx] = 1.0 / k
    
    # Correlate a with these active lags
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a
    
    # Positions with high correlation with peak lags should be reduced
    # Positions with low correlation should be increased (if possible)
    h_norm = h / hmax
    reduction = np.exp(-strength * np.maximum(h_norm, 0))
    # For positions with negative correlation (they reduce the peak), don't reduce
    q = a * reduction
    
    # Also add a small increase to low-correlation positions
    low_mask = h_norm < -0.3
    q[low_mask] *= (1.0 + strength * 0.5)
    
    return normalize_a(q)

                    continue
                w_idx = np.argpartition(d, -k)[-k:]
                w = np.zeros(len(d))
                w[w_idx] = 1.0 / k
                h = corr_pair(a, w)
                hmax = float(np.max(h))
                if hmax < 1e-15:
                    continue
                q = a * np.exp(-mult * h / hmax)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
        
        if cur_best is None:
            break
        a = cur_best

def generate_structured_inits(n, rng, parent=None, cap=25):
    """Generate diverse initial sequences."""
    inits = []

        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)
    
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == n:
            add(p)
            add(p[::-1])
        else:
            # Resample
            x_old = np.linspace(0, 1, p.size)
            p_res = np.interp(x, x_old, p)
            add(p_res)
            add(p_res[::-1])
    
    # Uniform
    add(np.ones(n))
    
    # V-shapes (high at edges)
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    
    # Inverted V (high in middle) - usually worse but worth trying
    for alpha in (0.3, 0.5, 0.7):
        add(1.0 - alpha * np.abs(2.0 * x - 1.0))
    
    # Cosine variants
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.5, 0.8, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * (x - 0.5)))
    
    # Sine absolute
    for f in (1, 2, 3):
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))
    
    # Power shapes
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        add(np.power(np.abs(2.0 * x - 1.0), p) + 0.1)
    
    # Exponential edge concentration
    for rate in (2.0, 4.0, 6.0, 8.0, 12.0):
        edge = np.exp(-rate * np.minimum(x, 1.0 - x))
        add(1.0 + edge)
    
    # Two-peak
    for width in (0.05, 0.1, 0.15, 0.2):
        for gap in (0.2, 0.3, 0.4):
            add(np.exp(-(((x - (0.5-gap)) / width)**2)) + np.exp(-(((x - (0.5+gap)) / width)**2)) + 0.1)
    
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 1.0)) * noise)
    
    # Random trigonometric
    for _ in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 8))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)
    
    # Block patterns (designed to have periodic low-correlation structure)
    for period in (2, 3, 4, 5, 8, 13, 21):
        if period > n // 2:
            continue
        # Alternating high/low
        pattern = np.zeros(period)
        pattern[0] = 1.0
        if period > 1:
            pattern[period // 2] = 0.3
        arr = np.tile(pattern, n // period + 1)[:n]
        add(arr + 0.1)
    
    # Golden ratio spacing (low autocorrelation property)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    frac = np.mod(np.arange(n) * (1.0 / golden), 1.0)
    add(1.0 + 0.5 * frac)
    add(1.0 + 0.5 * (1.0 - frac))
    
    return inits

def perturb_sequence(a, rng, strength=None):
    """Perturb a sequence for diversity."""
    p = normalize_a(a)
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))
    
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q *= np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Correlation-based reduction
        c = fast_self_conv(p)
        S = float(p.sum())
        d = (float(n) * c) / (S * S)
        dmax = float(np.max(d))
        if dmax > 1e-15:
            k = int(rng.integers(1, 8))
            w_idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(len(d))
            w[w_idx] = 1.0 / k
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-15:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smooth noise in log domain
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise -= float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape multiplication
        x = np.linspace(0.0, 1.0, n)
        v = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * v
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Block swap: swap two random blocks
        b1 = int(rng.integers(0, n - 10))
        b2 = int(rng.integers(0, n - 10))
        blen = int(rng.integers(5, max(6, n // 100)))
        q = p.copy()
        q[b1:b1+blen], q[b2:b2+blen] = q[b2:b2+blen].copy(), q[b1:b1+blen].copy()
    elif mode == 8:
        # Taper edges
        taper_len = int(rng.integers(n//20, n//5))
        q = p.copy()
        taper = np.linspace(0.3, 1.0, taper_len)
        q[:taper_len] *= taper
        q[-taper_len:] *= taper[::-1]
    else:
        # Shift
        shift = int(rng.integers(-n//10, n//10 + 1))
        q = np.roll(p, shift)
    
    return normalize_a(np.clip(q, 0.0, None))

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using geometric LSE annealing, subgradient polishing, push-pull active-set
heuristics, coordinate refinement, and perturbation-based diversity.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta, pen_sum=0.0):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 10.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L
    if pen_sum > 0.0:
        score += pen_sum * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    if pen_sum > 0.0:
        grad += 2.0 * pen_sum * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta, pen_sum):
    return lse_score_grad(a, beta, pen_sum)


def estimate_eval_time(n):
    return 0.02 * max(0.5, float(n) / 8192.0)


def run_lbfgs(a0, beta, deadline, maxfun, pen_sum=0.0):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.85 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    maxfun = max(64, maxfun)
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta), float(pen_sum)), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

                S = float(a.sum())
                d = (float(len(a)) * c) / (S * S)
                dmax = float(np.max(d))
                if dmax < 1e-15:
                    continue
                kk = min(k, len(d) - 1)
                w_idx = np.argpartition(d, -kk)[-kk:]
                w = np.zeros(len(d))
                w[w_idx] = 1.0 / kk
                h = corr_pair(a, w)
                hmax = float(np.max(h))
                if hmax < 1e-15:
                    continue
                q = a * np.exp(-mult * h / hmax)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def push_pull_step(a, strength=0.05, top_k=5):
    a = normalize_a(a).copy()
    n = len(a)
    S = float(a.sum())
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return a

    k = min(top_k, len(d) - 1)
    idx = np.argpartition(d, -k)[-k:]
    w = np.zeros(len(d))
    w[idx] = 1.0 / k

    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a

    h_norm = h / hmax
    reduction = np.exp(-strength * np.maximum(h_norm, 0.0))
    q = a * reduction

    low_mask = h_norm < -0.3
    if np.any(low_mask):
        q[low_mask] *= (1.0 + strength * 0.3)

    return normalize_a(q)


def coordinate_refine(a, max_rounds=5, deadline=None):
    """Iteratively nudge individual positions to reduce the peak."""
    a = normalize_a(a).copy()
    n = len(a)
    best_sc = exact_score_a(a)

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_self_conv(a)
        S = float(a.sum())
        d = (float(n) * c) / (S * S)
        dmax = float(np.max(d))
        if dmax < 1e-15:
            break

        k = min(5, len(d) - 1)
        idx = np.argpartition(d, -k)[-k:]
        w = np.zeros(len(d))
        w[idx] = 1.0 / k
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            break

        # Only modify top 50% most contributing positions
        threshold = np.median(h)
        modify_mask = h > threshold

        for i in range(n):
            if not modify_mask[i]:
                continue
            if a[i] < 1e-14:
                continue
            if deadline is not None and time.monotonic() > deadline:
                break

            old = a[i]
            best_val = old
            best_local = exact_score_a(a)

            for factor in (0.5, 0.7, 0.85, 1.15, 1.3):
                cv = old * factor
                a[i] = cv
                sc = exact_score_a(a)
                if np.isfinite(sc) and sc < best_local:
                    best_local = sc
                    best_val = cv

            a[i] = best_val

        a = normalize_a(a)
        sc = exact_score_a(a)
        if sc < best_sc:
            best_sc = sc
            a = a.copy()
        else:
            break

    return a, best_sc


def generate_structured_inits(n, rng, parent=None, cap=25):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == n:
            add(p)
            add(p[::-1])
        else:
            x_old = np.linspace(0, 1, p.size)
            p_res = np.interp(x, x_old, p)
            add(p_res)
            add(p_res[::-1])

    add(np.ones(n))

    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    for alpha in (0.3, 0.5, 0.7):
        add(1.0 - alpha * np.abs(2.0 * x - 1.0))

    for f in (1, 2, 3, 4, 5):
        for alpha in (0.3, 0.5, 0.8, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * (x - 0.5)))

    for f in (1, 2, 3):
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0, 8.0):
        add(np.power(np.abs(2.0 * x - 1.0), p) + 0.05)

    for rate in (2.0, 4.0, 6.0, 8.0, 12.0, 20.0):
        edge = np.exp(-rate * np.minimum(x, 1.0 - x))
        add(0.1 + edge)

    for width in (0.05, 0.08, 0.1, 0.15, 0.2):
        for gap in (0.15, 0.2, 0.25, 0.3, 0.4):
            add(0.1 + np.exp(-(((x - (0.5 - gap)) / width) ** 2))
                + np.exp(-(((x - (0.5 + gap)) / width) ** 2)))

    Nf = 1 << (n - 1).bit_length()
    for _ in range(10):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 6.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 1.0)) * noise)

    for _ in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 10))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    for period in (2, 3, 4, 5, 7, 8, 13, 21):
        if period > n // 3:
            continue
        pattern = np.full(period, 0.2)
        pattern[0] = 1.0
        if period > 2:
            pattern[period // 2] = 0.5
        arr = np.tile(pattern, n // period + 1)[:n]
        add(arr)

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    frac = np.mod(np.arange(n) * (1.0 / golden), 1.0)
    add(0.5 + frac)
    add(0.5 + (1.0 - frac))

    return inits


def perturb_sequence(a, rng, strength=None):
    p = normalize_a(a)
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 12))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q *= np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        c = fast_self_conv(p)
        S = float(p.sum())
        d = (float(n) * c) / (S * S)
        dmax = float(np.max(d))
        if dmax > 1e-15:
            k = int(rng.integers(1, 10))
            w_idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(len(d))
            w[w_idx] = 1.0 / k
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-15:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise -= float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        v = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * v
    elif mode == 6:
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        b1 = int(rng.integers(0, max(1, n - 10)))
        b2 = int(rng.integers(0, max(1, n - 10)))
        blen = int(rng.integers(5, max(6, n // 80)))
        q = p.copy()
        tmp = q[b1:b1 + blen].copy()
        q[b1:b1 + blen] = q[b2:b2 + blen]
        q[b2:b2 + blen] = tmp
    elif mode == 8:
        taper_len = int(rng.integers(max(1, n // 20), max(2, n // 5)))
        q = p.copy()
        taper = np.linspace(0.2, 1.0, taper_len)
        q[:taper_len] *= taper
        q[-taper_len:] *= taper[::-1]
    elif mode == 9:
        shift = int(rng.integers(-n // 8, n // 8 + 1))
        q = np.roll(p, shift)
    elif mode == 10:
        # Split into two halves, randomly permute within each half
        q = p.copy()
        half = n // 2
        q[:half] = q[rng.permutation(half)[:half]]
        q[half:] = q[half:][rng.permutation(n - half)]
    else:
        # Mix two random points
        t = float(rng.uniform(0.2, 0.8))
        q = t * p + (1.0 - t) * (np.ones(n) / n)

    return normalize_a(np.clip(q, 0.0, None))


def geometric_annealing(a_start, deadline, rng, label, n_stages=10):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()

    betas = np.geomspace(5.0, 200000.0, n_stages)

    total_time = max(10.0, float(deadline - t0 - 5.0))
    times = np.linspace(0.0, 1.0, n_stages + 1)
    times = times ** 0.6

    for i, beta in enumerate(betas):
        stage_end = t0 + times[i + 1] * total_time
        stage_end = min(stage_end, float(deadline - 5.0))

        if time.monotonic() >= stage_end - 2.0:
            continue

        prev_sc = best_sc
        while time.monotonic() < stage_end - 2.0:
            rem = float(stage_end - time.monotonic())
            if rem < 3.0:
                break
            n = len(a)
            est = estimate_eval_time(n)
            maxfun = int((0.8 * rem) / max(est, 1e-4))
            maxfun = max(64, min(3000, maxfun))

            sc, a2 = run_lbfgs(a, float(beta), stage_end, maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] {:.8f} beta={:.1f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)

            if abs(sc - prev_sc) / max(1.0, abs(prev_sc)) < 1e-8:
                break

        if time.monotonic() < stage_end - 3.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=5, deadline=stage_end)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad {:.8f} beta={:.1f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    # Push-pull
    if time.monotonic() < deadline - 10.0:
        for i in range(15):
            if time.monotonic() > deadline - 5.0:
                break
            q = push_pull_step(a, strength=0.03, top_k=5)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q
            else:
                q = push_pull_step(a, strength=0.01, top_k=3)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q

    # Coordinate refinement
    if time.monotonic() < deadline - 10.0:
        a_ref, sc_ref = coordinate_refine(a, max_rounds=3, deadline=deadline - 5.0)
        if np.isfinite(sc_ref) and sc_ref < best_sc:
            best_sc = sc_ref
            best_a = a_ref.copy()

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: geometric LSE annealing + subgradient + push-pull +
    coordinate refinement + perturbation restarts.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {}, len={}".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    if parent is not None:
        target_len = int(parent.size)
        lengths = [target_len]
        for L in [8192, 10240, 12288, 16384, 6144]:
            if L != target_len and L not in lengths:
                lengths.append(L)
        lengths = lengths[:4]
    else:
        lengths = [10240, 8192, 12288]

    print("Lengths: {}".format(lengths))

    # Phase 1: Quick init evaluation
    all_starts = []
    for n in lengths:
        inits = generate_structured_inits(n, rng, parent, cap=12 if n > 8000 else 18)
        for arr in inits:
            sc = exact_score_a(arr)
            if np.isfinite(sc):
                all_starts.append((sc, arr))

    all_starts.sort(key=lambda x: x[0])
    print("Best init: {:.8f}".format(all_starts[0][0]))

    starts = []
    used_lengths = set()
    if parent is not None:
        starts.append(normalize_a(parent.astype(np.float64)))
        used_lengths.add(len(starts[0]))

    for sc, arr in all_starts:
        if len(starts) >= 3:
            break
        if len(arr) not in used_lengths:
            starts.append(arr)
            used_lengths.add(len(arr))

    while len(starts) < 2:
        if all_starts:
            starts.append(all_starts[0][1].copy())
        else:
            starts.append(np.ones(8192, dtype=np.float64) / 8192.0)

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for s in starts[1:]:
        sc = exact_score_a(s)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main geometric annealing optimization
    main_end = deadline - 350.0
    main_budget = max(60.0, main_end - t_start)

    n_starts = min(len(starts), 3)
    fracs = [0.60, 0.25, 0.15] if n_starts >= 3 else [0.70, 0.30] if n_starts == 2 else [1.0]

    for rank in range(n_starts):
        if time.monotonic() > main_end - 20.0:
            break

        sub_dead = t_start + sum(fracs[:rank + 1]) * main_budget
        sub_dead = min(sub_dead, float(main_end))

        arr = starts[rank].copy()
        if rank > 0 and len(best_seq) != len(arr):
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, len(arr))
            blended = np.interp(x_new, x_old, best_seq)
            arr = normalize_a(0.3 * arr + 0.7 * blended)
        elif rank > 0:
            arr = normalize_a(0.3 * arr + 0.7 * best_seq)

        sc, seq = geometric_annealing(arr, sub_dead, rng, "main{}".format(rank + 1), n_stages=10)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after main{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Perturbation restarts
    perturb_end = deadline - 250.0
    n_perturb = 0
    while time.monotonic() < perturb_end and n_perturb < 10:
        left = float(perturb_end - time.monotonic())
        if left < 80.0:
            break
        remaining = 10 - n_perturb
        chunk = time.monotonic() + left / float(remaining) - 15.0

        p = perturb_sequence(best_seq, rng)
        sc, seq = geometric_annealing(p, chunk, rng, "pert{}".format(n_perturb + 1), n_stages=6)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb {}: {:.8f}".format(n_perturb + 1, best_score))
        n_perturb += 1

    best_seq = normalize_a(best_seq)

    # Phase 4: Final intensive polish
    if time.monotonic() < deadline - 60.0:
        rem_dead = deadline - 30.0
        sc, seq = geometric_annealing(best_seq, rem_dead, rng, "final", n_stages=6)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    best_seq = normalize_a(best_seq)

    # Phase 5: Final subgradient + coordinate refinement
    if time.monotonic() < deadline - 20.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
        best_seq = normalize_a(best_seq)

    if time.monotonic() < deadline - 10.0:
        a_ref, sc_ref = coordinate_refine(best_seq, max_rounds=5, deadline=deadline - 5.0)
        if np.isfinite(sc_ref) and sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()

    best_seq = normalize_a(best_seq)

    # Final output preparation
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
