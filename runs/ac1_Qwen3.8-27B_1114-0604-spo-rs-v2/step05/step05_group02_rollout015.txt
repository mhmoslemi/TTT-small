"""
Hybrid optimization for minimizing max(self-convolution) / (sum)^2 of non-negative sequences.
Uses Nesterov subgradient, FISTA-LSE, GS-Fienup, and active-set refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact evaluation score."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    Z = float(exp_scaled.sum())
    p = exp_scaled / Z
    L = c_max + float(np.log(Z)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da_i = sum_k p_k * (a_{k-i} + a_{i-k})
    # = cc[i] + cc[M-i] where cc = ifft(FFT(p) * conj(FFT(a)))
    Mx = 3 * n - 2
    Mx_fft = 1 << (Mx - 1).bit_length()
    P_fft = np.fft.rfft(p, Mx_fft)
    A_fft = np.fft.rfft(a, Mx_fft)
    cc = np.fft.irfft(P_fft * np.conj(A_fft), Mx_fft)
    
    grad_L = np.zeros(n)
    grad_L[0] = 2.0 * cc[0]
    if n > 1:
        grad_L[1:] = cc[1:n] + cc[Mx_fft - 1:Mx_fft - n + 1:-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    def fun(x):
        return lse_score_grad(x, beta)
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        fun, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 20}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def gs_fienup(a, n_iters=400, alpha_init=0.7, alpha_final=0.1):

    n = len(a)

    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()

        return np.ones(n) / n

        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)

            break

        a = a_new

        if sc < best_sc:

            best_a = a.copy()
    
    return best_a


def nesterov_subgrad(a, n_iters=5000, top_k_schedule=None, mu=0.9):
    """
    Nesterov accelerated subgradient method with adaptive top-K.
    Uses Polyak step size with running best as lower bound proxy.
    """

    a = normalize(a.astype(np.float64)).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    y = a.copy()
    t_mom = 1.0
    prev_y = a.copy()
    
    # Polyak step size parameters
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    for it in range(n_iters):
        # Compute subgradient at y
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc_y = 2.0 * n * c_max / (S * S)
        
        # Adaptive top-K
        if top_k_schedule is not None:
            progress = it / n_iters
            k = max(2, int(top_k_schedule(progress)))
        else:
            k = 5
        
        # Get top-k peaks
        if k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -k)[-k:]
        
        # Compute subgradient (average of gradients at top-k peaks)
        sg = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            j_start = max(0, kk - n + 1)
            j_end = min(kk, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg /= len(top_idx)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        # Polyak step size
        step = (sc_y - sc_lb) / (float(np.dot(grad, grad)) + 1e-30)
        step = min(step, 0.1)
        step = max(step, 1e-8)
        
        # Nesterov update
        new_x = project_simplex(y - step * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

def fista_lse(a, n_iters=3000, beta_start=50, beta_end=5000, L=None):
    """
    FISTA (accelerated proximal gradient) on LSE-smoothed objective.
    """
    n = len(a)

    best_sc = compute_score(a)

    prev_y = a.copy()
    
    # Estimate Lipschitz constant L
    if L is None:
        sc0, g0 = lse_score_grad(a, beta_start)
        # Rough estimate: L ~ max|grad| / typical_step
        L = float(np.max(np.abs(g0))) * 10.0 + 1.0

        progress = it / n_iters
        beta = beta_start * (beta_end / beta_start) ** progress
        
        sc_y, grad_y = lse_score_grad(y, beta)
        
        # Proximal gradient step
        step = 1.0 / L
        new_x = project_simplex(y - step * grad_y)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        new_y = new_x + ((t_fista - 1.0) / t_new) * (new_x - prev_y)
        new_y = project_simplex(new_y)
        
        prev_y = y
        y = new_y
        t_fista = t_new
        
        sc = compute_score(y)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        # Periodically re-estimate L

            sc_test, g_test = lse_score_grad(new_x + 0.01 * g_test, beta) if 'g_test' in dir() else (sc_y, grad_y)
    
    return best_a
