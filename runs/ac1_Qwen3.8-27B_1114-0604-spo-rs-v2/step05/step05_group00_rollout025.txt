import time

def fast_conv(a):
    """Linear convolution via FFT."""
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S*S)

def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = x.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S*S)
    p = exp_s / exp_s.sum()
    
    # Gradient: raw_grad[m] = 2 * sum_j p[m+j] * a[j]
    M_fft = 1 << (2*n - 1).bit_length()
    PA = np.fft.rfft(p, M_fft)

    S2 = S*S
    S3 = S2*S
    grad = 2.0*n * (raw_grad/S2 - 2.0*L/S3)
    return score, grad

    # Gradient via cross-correlation: raw_grad[m] = 2 * sum_j p[m+j] * a[j]
    M_fft = 1 << (2 * n - 1).bit_length()
    PA = np.fft.rfft(p, M_fft)
    RA = np.fft.rfft(x, M_fft)
    cross = np.fft.irfft(np.conj(PA) * RA, M_fft)[:n]
    raw_grad = 2.0 * cross

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (raw_grad / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def continuation_lbfgs(a0, betas, time_budget, label=""):
    """
    Continuation (annealing) method with LSE:
    optimize with increasing beta, warm-starting each level.
    """
    a = np.maximum(a0, 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        elapsed = time.monotonic() - t_start
        remaining = time_budget - elapsed
        if remaining < 20:
            break
        maxiter = min(8000, int(remaining * 15))
        maxfun = min(150000, int(remaining * 300))
        if maxiter < 100:
            break
        t0 = time.monotonic()
        sc, a = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"  [{label}] Continuation beta={beta}: score={sc:.8f} ({dt:.1f}s)")
    return a


def coord_descent_fast(a0, n_sweeps=100, n_top=200):
    """
    Coordinate descent with O(n) convolution updates per coordinate.
    Focuses on top-contributing elements at the peak lag.
    Tries setting to zero and multiplicative adjustments.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    c = fast_conv(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())

    for sweep in range(n_sweeps):
        k_star = int(np.argmax(c))
        # Valid j range for lag k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        improved = False
        for j in top_global:
            old = a[j]
            if old < 1e-14:
                continue
            # Try several candidate values
            candidates = [0.0, old * 0.1, old * 0.3, old * 0.5, old * 0.7,
                          old * 1.3, old * 1.5, old * 2.0]
            best_val = old
            # Current score contribution
            S_old = S
            cur_max = float(c.max())
            cur_score = 2.0 * n * cur_max / (S_old * S_old)

            for cv in candidates:
                if abs(cv - old) < 1e-14:
                    continue
                cv = max(0.0, cv)
                # Update convolution in O(n)
                delta = cv - old
                k_lo = j
                k_hi = j + n - 1
                # Only update valid range
                k_lo = max(0, k_lo)
                k_hi = min(2 * n - 2, k_hi)
                if k_hi < k_lo:
                    continue
                ks = np.arange(k_lo, k_hi + 1)
                js_idx = ks - j  # = k - j for each k
                # Check validity: 0 <= k-j < n
                valid = (js_idx >= 0) & (js_idx < n)
                if not valid.any():
                    continue
                c_new = c.copy()
                ks_v = ks[valid]
                js_v = js_idx[valid]
                # For k = 2*j: quadratic change
                is_diag = (ks_v == 2 * j)
                if is_diag.any():
                    c_new[ks_v[is_diag]] = c[ks_v[is_diag]] - old * old + cv * cv
                if (~is_diag).any():
                    c_new[ks_v[~is_diag]] = c[ks_v[~is_diag]] + 2.0 * delta * a[js_v[~is_diag]]

                new_S = S_old - old + cv
                if new_S < 1e-12:
                    continue
                new_max = float(np.max(c_new))
                new_score = 2.0 * n * new_max / (new_S * new_S)
                if new_score < cur_score:
                    best_val = cv
                    cur_score = new_score

            if best_val != old:
                delta = best_val - old
                a[j] = best_val
                S = S + delta
                # Update convolution
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)
                if k_hi >= k_lo:
                    ks = np.arange(k_lo, k_hi + 1)
                    js_idx = ks - j
                    valid = (js_idx >= 0) & (js_idx < n)
                    if valid.any():
                        ks_v = ks[valid]
                        js_v = js_idx[valid]
                        is_diag = (ks_v == 2 * j)
                        if is_diag.any():
                            c[ks_v[is_diag]] += best_val * best_val - old * old
                        if (~is_diag).any():
                            c[ks_v[~is_diag]] += 2.0 * delta * a[js_v[~is_diag]]
                improved = True

        if not improved:
            break

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def sparsify(a0, n_rounds=20):
    """
    Iteratively set small elements to zero if it improves the score.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    best_sc = compute_score(a)
    best_a = a.copy()

    for rnd in range(n_rounds):
        S = a.sum()
        # Sort by value
        order = np.argsort(a)
        improved = False
        for j in order:
            if a[j] < 1e-6:
                continue
            old = a[j]
            a[j] = 0.0
            sc = compute_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                improved = True
            else:
                a[j] = old
            if not improved:
                # If we've gone through all small elements without improvement, stop
                if a[j] > 0.01 * S / n:
                    break
        if not improved:
            break

    return best_a


def fienup_target(a0, n_iters=300):
    """
    Fienup iteration targeting the spectrum that would give flat convolution.
    The target is: |A[m]|^2 proportional to |FFT(rect_window)|^2
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    # Target: spectrum of rectangular window of length 2n-1
    rect = np.zeros(M_fft)
    rect[:2 * n - 1] = 1.0 / (2 * n - 1)
    target_spec = np.abs(np.fft.rfft(rect))

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Blend current and target amplitude
        alpha = 0.3 + 0.5 * (it / n_iters)
        curr_amp = np.sqrt(P)
        curr_amp = np.maximum(curr_amp, 1e-30)
        target_amp = target_spec / target_spec.max() * curr_amp.max()

        blend_amp = (1 - alpha) * curr_amp + alpha * target_amp

        # Update: set amplitude to blend, keep phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = blend_amp * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flatten(a0, n_iters=200):
    """Fienup: flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.2 + 0.6 * (it / n_iters)
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / np.maximum(P, 1e-20)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multiplicative_update(a0, n_iters=5000, top_k=10):
    """
    Exponentiated gradient / multiplicative weight update targeting top peaks.
    a[j] <- a[j] * exp(-eta * blame[j]) where blame measures contribution to peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.02

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())

        # Top-k peak lags
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = c[top_idx] / (top_idx.size * (M_val + 1e-30))

        # Compute blame: for each j, how much does it contribute to top peaks?
        blame = np.zeros(n)
        M_fft = 1 << (2 * n - 1).bit_length()
        p = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            p[int(k)] = weights[i]
        # blame[m] = sum_k p[k] * 2*a[k-m] (the subgradient weighted by peak importance)
        PA = np.fft.rfft(p, M_fft)
        RA = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(np.conj(PA) * RA, M_fft)[:n]
        blame = 2.0 * cross / (S * S)

        # Normalized blame
        blame = np.maximum(blame, 0)
        b_mean = blame.mean()
        if b_mean > 1e-15:
            blame = blame / b_mean

        # Multiplicative update
        a = a * np.exp(-eta * (blame - 1.0))
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Decay eta
        eta *= 0.999

    return best_a


def peak_shave_adaptive(a0, n_iters=3000, top_k=5):
    """
    Subgradient method targeting top-K peaks with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.02
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            weights[i] = c[int(k)] / (M_val + 1e-30)

        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * (weights.sum() / len(top_idx)) / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max(0, sc - sc_lb) / (gn * gn + 1e-30))
        step = np.clip(step, 1e-8, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(new_n) / new_n
    a = a / S
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0)
    S_new = result.sum()
    if S_new > 1e-12:
        result = result / S_new
    return result


def noise_perturb(a, rng, scale=0.02):
    """Add Gaussian noise perturbation."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def structured_perturb(a, rng, n_modes=8, scale=0.03):
    """Add low-frequency structured perturbations."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = int(rng.integers(1, 15))
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-length Fienup initialization
    Phase 2: Continuation LSE (main optimizer)
    Phase 3: Coordinate descent + sparsification
    Phase 4: Multiplicative update + peak shaving
    Phase 5: Multi-restart + final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent or default
    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        print("No parent available, starting fresh")

    print(f"Starting best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Multi-length initialization (15% of time)
    # ==================================================================
    phase1_end = t_start + 0.15 * (total_budget - 20.0)
    print(f"\n=== Phase 1: Initialization (until t={phase1_end - t_start:.0f}s) ===")

    # Try Fienup on parent
    if parent is not None:
        a_f = fienup_flatten(parent, n_iters=150)
        sc = compute_score(a_f)
        update(sc, a_f, "FIENUP-parent")

        a_f2 = fienup_target(parent, n_iters=150)
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-target")

    # Try fresh starts at power-of-2 lengths
    for n_try in [4096, 8192]:
        if time.monotonic() > phase1_end - 30:
            break
        a_init = np.ones(n_try) / n_try
        # Add some structure
        a_init = a_init * (1.0 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n_try) / n_try))
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        a_f = fienup_flatten(a_init, n_iters=100)
        sc = compute_score(a_f)
        update(sc, a_f, f"FIENUP-fresh-{n_try}")

        a_f2 = fienup_target(a_init, n_iters=100)
        sc = compute_score(a_f2)
        update(sc, a_f2, f"FIENUP-target-{n_try}")

    # Also try interpolating parent to 8192
    if parent is not None and len(parent) != 8192:
        if time.monotonic() < phase1_end - 30:
            a_interp = interpolate_to_length(parent, 8192)
            sc = compute_score(a_interp)
            update(sc, a_interp, "INTERP-8192")
            a_f = fienup_flatten(a_interp, n_iters=100)
            sc = compute_score(a_f)
            update(sc, a_f, "FIENUP-interp-8192")

    print(f"  After Phase 1: best={best_score:.8f}")

    # ==================================================================
    # Phase 2: Continuation LSE (45% of time) - main workhorse
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 20.0)
    print(f"\n=== Phase 2: Continuation LSE (until t={phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 40000]
    a_cont = continuation_lbfgs(a_work, betas, time_budget=phase2_end - time.monotonic(),
                                label="main")
    sc = compute_score(a_cont)
    update(sc, a_cont, "CONTINUATION")

    # Second continuation from a different starting point
    if time.monotonic() < phase2_end - 120:
        print("  Second continuation from perturbed start...")
        a_pert = noise_perturb(best_a, rng, scale=0.01)
        betas2 = [10, 50, 200, 1000, 5000, 10000]
        a_cont2 = continuation_lbfgs(a_pert, betas2,
                                     time_budget=phase2_end - time.monotonic() - 10,
                                     label="perturbed")
        sc = compute_score(a_cont2)
        update(sc, a_cont2, "CONTINUATION-2")

    print(f"  After Phase 2: best={best_score:.8f}")

    # ==================================================================
    # Phase 3: Coordinate descent + sparsification (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Coord descent + sparsify (until t={phase3_end - t_start:.0f}s) ===")

    # Coordinate descent
    if time.monotonic() < phase3_end - 60:
        a_cd = coord_descent_fast(best_a, n_sweeps=50, n_top=100)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")

    # Sparsification
    if time.monotonic() < phase3_end - 60:
        a_sp = sparsify(best_a, n_rounds=10)
        sc = compute_score(a_sp)
        update(sc, a_sp, "SPARIFY")

        # Coord descent after sparsify
        if sc < best_score and time.monotonic() < phase3_end - 30:
            a_cd2 = coord_descent_fast(a_sp, n_sweeps=30, n_top=80)
            sc2 = compute_score(a_cd2)
            update(sc2, a_cd2, "SPARIFY+CD")

    # Short continuation on the result
    if time.monotonic() < phase3_end - 30:
        a_c3 = continuation_lbfgs(best_a, [5000, 10000, 20000],
                                  time_budget=phase3_end - time.monotonic() - 10,
                                  label="polish")
        sc = compute_score(a_c3)
        update(sc, a_c3, "CONT-POLISH")

    print(f"  After Phase 3: best={best_score:.8f}")

    # ==================================================================
    # Phase 4: Multiplicative update + peak shaving (10% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n=== Phase 4: Mult update + peak shave ===")

    if time.monotonic() < phase4_end - 30:
        a_mu = multiplicative_update(best_a, n_iters=2000, top_k=8)
        sc = compute_score(a_mu)
        update(sc, a_mu, "MULT-UPDATE")

    if time.monotonic() < phase4_end - 30:
        a_ps = peak_shave_adaptive(best_a, n_iters=2000, top_k=5)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")

    # Coord descent on results
    if time.monotonic() < phase4_end - 10:
        a_cd3 = coord_descent_fast(best_a, n_sweeps=30, n_top=150)
        sc = compute_score(a_cd3)
        update(sc, a_cd3, "CD-Phase4")

    print(f"  After Phase 4: best={best_score:.8f}")

    # ==================================================================
    # Phase 5: Multi-restart + final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Multi-restart + final polish ===")

    # Try a few restarts with different perturbations
    for r in range(6):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - 30 - time.monotonic()
        if remaining < 30:
            break

        if r % 3 == 0:
            a_pert = noise_perturb(best_a, rng, scale=0.005 + 0.002 * r)
        elif r % 3 == 1:
            a_pert = structured_perturb(best_a, rng, n_modes=6, scale=0.01 + 0.005 * r)
        else:
            # Try a different length
            new_n = 8192 if len(best_a) != 8192 else 4096
            a_pert = interpolate_to_length(best_a, new_n)

        maxiter = min(3000, int(remaining * 10))
        if maxiter < 200:
            break
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=maxiter * 30)
        update(sc, a_opt, f"RESTART-{r}")

        # Quick coord descent
        if time.monotonic() < deadline - 20:
            a_cd_r = coord_descent_fast(a_opt, n_sweeps=10, n_top=60)
            sc = compute_score(a_cd_r)
            update(sc, a_cd_r, f"CD-RESTART-{r}")

    # Final L-BFGS polish with very high beta
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        maxiter = min(3000, int(remaining * 8))
        if maxiter > 100:
            sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 20)
            update(sc, a_final, "FINAL-LBFGS-50k")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd_final = coord_descent_fast(best_a, n_sweeps=20, n_top=120)
        sc = compute_score(a_cd_final)
        update(sc, a_cd_final, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
