```python
"""
Search for non-negative step-function heights minimizing

    2 * n * max(conv(a, a)) / sum(a)^2

using a continuation strategy with a differentiable log-sum-exp surrogate,
log-softmax parameterization (with smooth saturation), L-BFGS-B local
optimization, length re-sampling, smooth perturbations, and an alternating
fixed-left / free-right polishing phase inspired by the LP active-set idea.
The best sequence found within the time budget is returned.
"""

import numpy as np
import time
from scipy.optimize import minimize

TIMING = {}
Y_LIMIT = 50.0


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def record_timing(n, dt):
    key = int(n)
    entry = TIMING.get(key)
    if entry is None:
        TIMING[key] = [1, float(dt)]
    else:
        entry[0] += 1
        entry[1] += float(dt)


def get_avg_timing(n):
    entry = TIMING.get(int(n))
    if entry is not None and entry[0] > 0:
        return max(1e-5, entry[1] / float(entry[0]))
    # Conservative default.  The dominant FFT size is ~ nextpowerof(3n).
    N = next_power_of_two(max(1, 3 * int(n) - 2))
    return 0.015 + 0.025 * (float(N) / 32768.0)


def estimate_maxfun(t_left, n):
    if t_left <= 0.0:
        return 0
    est = get_avg_timing(n)
    if TIMING.get(int(n)) is None:
        est *= 1.5
    mf = int((0.85 * float(t_left)) / max(est, 1e-4))
    if mf < 16:
        mf = 16
    if mf > 20000:
        mf = 20000
    return mf


def normalize_a(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == 0:
        return np.ones(64, dtype=np.float64) / 64.0
    s = float(arr.sum())
    if s < 1e-12:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = int(max(1, n))
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return a.copy()
    old = np.linspace(0.0, 1.0, int(a.size))
    new = np.linspace(0.0, 1.0, int(n))
    out = np.interp(new, old, a).astype(np.float64)
    return normalize_a(out)


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    N = next_power_of_two(m)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:m]
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(out, 0.0)


def exact_score_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    if c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0
    return float(2.0 * n * cmax / (S * S))


def corr_pair(a, w):
    """
    For h_j = sum_{t=0}^{n-1-j} a_t w_{t+j}, j=0..n-1.
    This is the one-sided Jacobian factor used by convolution gradients.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    mw = int(w.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if mw == 0:
        return np.zeros(n, dtype=np.float64)
    m = n + mw - 1
    N = next_power_of_two(m)
    Ar = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(Ar * W, N)[:m]
    out = full[n - 1:2 * n - 1]
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(out, 0.0)


def y_to_soft(y):
    """
    Map raw logit vector y to a simplex vector using a smooth, nearly
    identity saturation.  Also returns d z_i / d y_i for the unscored
    centered-saturated coordinates.
    """
    y = np.asarray(y, dtype=np.float64).ravel()
    n = int(y.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), np.empty(0, dtype=np.float64)

    y = np.nan_to_num(y, nan=0.0, posinf=Y_LIMIT, neginf=-Y_LIMIT)
    y = y - float(np.mean(y))

    t = np.tanh(y / Y_LIMIT)
    z = Y_LIMIT * t
    deriv = 1.0 - t * t

    mz = float(np.max(z))
    e = np.exp(z - mz)
    S = float(e.sum())
    if not np.isfinite(S) or S <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
        deriv = np.ones(n, dtype=np.float64)
    else:
        a = e / S
        S2 = float(a.sum())
        if S2 <= 0.0 or not np.isfinite(S2):
            a = np.ones(n, dtype=np.float64) / float(n)
        else:
            a = a / S2

    return a, deriv


def softmax_from_y(y):
    a, _ = y_to_soft(y)
    return a


def log_from_a(a):
    """
    Inverse of y_to_soft in the usual log-softmax sense, with clipping to
    keep the pp-inverse finite.
    """
    a = normalize_a(a)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)

    floor = 1e-200
    c = np.maximum(a, floor)
    z = np.log(c)
    z = z - float(np.mean(z))

    lim = 0.95 * Y_LIMIT
    z = np.clip(z, -lim, lim)
    y = Y_LIMIT * np.arctanh(z / Y_LIMIT)
    y = np.nan_to_num(y, nan=0.0, posinf=Y_LIMIT, neginf=-Y_LIMIT)
    y = y - float(np.mean(y))
    return y


def lse_weights(d, beta, dmax):
    """
    Returns (weights, value) for a scaled log-sum-exp of d.
    dmax is provided.  The value returned is
        dmax + log(sum(exp(beta*(d-dmax)))) / beta
    """
    beta = float(beta)
    if not np.isfinite(beta) or beta < 1.0:
        beta = 10.0

    z = beta * (d - dmax)
    z = np.nan_to_num(z, nan=-740.0, posinf=0.0, neginf=-740.0)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -740.0)

    ez = np.exp(z)
    sume = float(ez.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        w = np.zeros_like(d, dtype=np.float64)
        active = d >= (dmax - 1e-12)
        if int(active.sum()) == 0:
            w[0] = 1.0
        else:
            w[active] = 1.0
            w = w / float(w.sum())
        return w, float(dmax)

    w = ez / sume
    val = float(dmax) + float(np.log(sume)) / beta
    return w, val


def soft_objective(y, beta):
    """
    Smooth objective in log-softmax coordinates:
      f(y) ~= 2 * LSE_beta(n * conv(softmax(y), softmax(y)))
    Returns (f, grad wrt y).
    """
    t0 = time.monotonic()
    y = np.asarray(y, dtype=np.float64).ravel()
    n = int(y.size)
    if n == 0:
        return 1e100, np.zeros(0, dtype=np.float64)

    if not np.all(np.isfinite(y)):
        return 1e100, np.zeros(n, dtype=np.float64)

    a, deriv = y_to_soft(y)
    c = fast_self_conv(a)
    d = float(n) * c
    if d.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)

    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    w, L = lse_weights(d, beta, dmax)
    fval = 2.0 * L
    if not np.isfinite(fval):
        fval = 1e100

    # derivative wrt a: 4 n h
    h = corr_pair(a, w)
    grad_a = 4.0 * float(n) * h

    total = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - total)
    grad_y = grad_z * deriv
    grad_y = grad_y - float(np.mean(grad_y))
    grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=1e6, neginf=-1e6)

    record_timing(n, time.monotonic() - t0)
    return float(fval), grad_y


def run_y_lbfgs(y0, beta, deadline, maxfun):
    y = np.asarray(y0, dtype=np.float64).ravel().copy()
    n = int(y.size)
    rem = float(max(0.0, deadline - time.monotonic()))
    if n <= 0 or rem < 5.0:
        return y

    mf = int(estimate_maxfun(rem * 0.9, n))
    if maxfun is not None:
        mf = min(mf, int(maxfun))
    mf = max(16, mf)
    maxiter = int(max(30, mf // 2))

    opts = {
        "ftol": 1e-14,
        "gtol": 1e-12,
        "maxiter": maxiter,
        "maxfun": mf,
        "maxls": 25,
    }

    try:
        res = minimize(
            soft_objective,
            y,
            method="L-BFGS-B",
            jac=True,
            args=(float(beta),),
            options=opts,
        )
    except Exception:
        return y

    y_out = getattr(res, "x", None)
    if y_out is None:
        return y

    y_out = np.asarray(y_out, dtype=np.float64).ravel()
    if not np.all(np.isfinite(y_out)):
        return y

    y_out = y_out - float(np.mean(y_out))
    t = np.tanh(y_out / (2.0 * Y_LIMIT))
    y_out = 2.0 * Y_LIMIT * t
    y_out = y_out - float(np.mean(y_out))
    return y_out


def quick_y_polish(a0, deadline, label):
    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    y = log_from_a(a)

    betas = [500.0, 2000.0, 10000.0, 50000.0]
    for beta in betas:
        rem = float(deadline - time.monotonic())
        if rem < 40.0:
            break
        mf = estimate_maxfun(rem * 0.8, int(a.size))
        if mf < 32:
            break
        sub_dead = time.monotonic() + min(rem * 0.9, deadline - time.monotonic())
        y = run_y_lbfgs(y, beta, sub_dead, mf)
        a2 = softmax_from_y(y)
        sc = exact_score_a(a2)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a2.copy()
            print("[{}] polish beta={} score={:.8f}".format(label, int(beta), best_sc))

    return float(best_sc), best_a


def optimize_sequence_y(a0, time_budget, label, rng):
    t0 = time.monotonic()
    total = float(max(1.0, time_budget))
    a = normalize_a(a0)
    if a.size < 256:
        a = resample_sequence(a, 4096)

    best_a = a.copy()
    best_sc = exact_score_a(a)
    y = log_from_a(a)

    # Calibrate FFT timings for this length.
    for _ in range(2):
        try:
            soft_objective(y, 100.0)
        except Exception:
            pass

    print("[{}] start n={} score={:.8f}".format(label, int(a.size), best_sc))

    stages = [
        (20.0, 0.07),
        (60.0, 0.13),
        (200.0, 0.17),
        (700.0, 0.22),
        (2500.0, 0.18),
        (8000.0, 0.13),
        (25000.0, 0.10),
    ]

    cum = 0.0
    stagnant = 0

    for beta, frac in stages:
        cum += frac
        cum = min(1.0, cum)
        sub_dead = t0 + cum * total
        sub_dead = min(sub_dead, t0 + total - 5.0)
        rem = float(sub_dead - time.monotonic())
        if rem < 15.0:
            break

        mf = estimate_maxfun(rem * 0.9, int(a.size))
        if mf < 32:
            break

        y_res = run_y_lbfgs(y, beta, sub_dead, mf)
        a_res = softmax_from_y(y_res)
        sc_res = exact_score_a(a_res)

        if np.isfinite(sc_res) and sc_res < best_sc - 1e-12:
            best_sc = float(sc_res)
            best_a = a_res.copy()
            stagnant = 0
            print(
                "[{}] new best {:.8f} beta={:.0f} elapsed={:.1f}s".format(
                    label, best_sc, beta, time.monotonic() - t0
                )
            )
        else:
            stagnant += 1

        if (
            np.isfinite(sc_res)
            and sc_res <= best_sc + 1e-9 + 0.004 * max(1.0, best_sc)
        ):
            a = normalize_a(a_res)
            y = log_from_a(a)
        else:
            a = best_a.copy()
            y = log_from_a(a)

        if (
            stagnant >= 3
            and rng is not None
            and time.monotonic() < sub_dead - 10.0
        ):
            y = smooth_perturb_y(y, 0.12, rng)
            a2 = softmax_from_y(y)
            sc2 = exact_score_a(a2)
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = float(sc2)
                best_a = a2.copy()
                print(
                    "[{}] perturbed best {:.8f}".format(label, best_sc)
                )
            stagnant = 0

    return float(best_sc), best_a


def smooth_perturb_y(y, scale, rng):
    y = np.asarray(y, dtype=np.float64).ravel()
    n = int(y.size)
    if n < 8 or rng is None:
        return y
    noise = rng.standard_normal(n)
    A = np.fft.rfft(noise)
    freqs = np.fft.rfftfreq(n) * float(n)
    power = 1.0 / (1.0 + freqs * freqs)
    power[0] = 0.0
    A = A * power
    noise = np.fft.irfft(A, n)
    std = float(noise.std())
    if not np.isfinite(std) or std <= 1e-14:
        return y
    noise = noise / std * float(scale)
    out = y + noise
    out = out - float(np.mean(out))
    return out


def perturb_sequence(a, rng, scale=0.25):
    a = normalize_a(a)
    if rng is None:
        return a
    y = log_from_a(a)
    y = smooth_perturb_y(y, float(scale), rng)
    return normalize_a(softmax_from_y(y))


def smooth_fft_sequence(a, sigma):
    a = normalize_a(a)
    n = int(a.size)
    if n < 4:
        return a
    A = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n) * float(n)
    kernel = np.exp(-0.5 * (freqs / float(max(1.0, sigma))) ** 2)
    b = np.fft.irfft(A * kernel, n)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.maximum(b, 0.0)
    return normalize_a(b)


def make_parametric_inits(n, rng):
    out = []
    n = int(n)
    if n < 16 or rng is None:
        out.append(np.ones(n, dtype=np.float64))
        return out

    x = np.linspace(0.0, 1.0, n)

    out.append(np.ones(n, dtype=np.float64))

    for alpha in (0.35, 0.8, 1.3):
        out.append(1.0 + alpha * np.abs(2.0 * x - 1.0))
        out.append(1.0 + alpha * (2.0 * x - 1.0))
        out.append(1.0 - alpha * (2.0 * x - 1.0))

    for f in (1, 2, 3):
        out.append(1.0 + 0.55 * np.cos(2.0 * np.pi * f * x))
        out.append(1.0 + 0.55 * np.sin(np.pi * f * x))

    freqs = np.fft.rfftfreq(n) * float(n)
    power = 1.0 / (1.0 + freqs * freqs)
    power[0] = 0.0
    re = rng.standard_normal(len(power))
    im = rng.standard_normal(len(power))
    spec = power * (re + 1j * im)
    spec[0] = 0.0
    noise = np.fft.irfft(spec, n)
    noise = noise - float(noise.mean())
    out.append(1.0 + 0.60 * noise)

    cleaned = []
    for arr in out:
        arr = normalize_a(arr)
        if arr.size == n and np.all(np.isfinite(arr)):
            cleaned.append(arr)
    return cleaned


def scored_key(item):
    return item[0]


def build_starts(parent, rng):
    starts = []

    if parent is not None:
        p = normalize_a(parent)
        if p.size == 0:
            p = np.ones(8192, dtype=np.float64) / 8192.0
        if p.size < 2048 or p.size > 16384:
            p = resample_sequence(p, 10240)

        plen = int(p.size)
        starts.append(("parent{:d}".format(plen), p))

        targets = [8192, 10240, 12288, 16384]
        for n in targets:
            if int(n) != plen and 2048 <= n <= 16384:
                starts.append(("resamp{:d}".format(n), resample_sequence(p, n)))

        sigma = max(10.0, 0.05 * float(plen))
        starts.append(("smooth_parent", smooth_fft_sequence(p, sigma)))
        starts.append(("perturb_parent", perturb_sequence(p, rng, 0.35)))
    else:
        for n in (6144, 8192, 10240, 12288):
            for i, arr in enumerate(make_parametric_inits(n, rng)):
                starts.append(("param{}_{}_{}".format(n, i, arr.size), arr))

    scored = []
    for label, a in starts:
        a = normalize_a(a)
        if a.size < 1024:
            a = resample_sequence(a, 4096)
        if a.size > 16384:
            a = resample_sequence(a, 16384)
        sc = exact_score_a(a)
        if np.isfinite(sc):
            scored.append((float(sc), label, a))

    scored.sort(key=scored_key)

    selected = []
    used_lens = set()
    for sc, label, a in scored:
        n = int(a.size)
        if n in used_lens:
            continue
        selected.append((label, a.copy()))
        used_lens.add(n)
        if len(selected) >= 4:
            break

    if parent is not None and len(selected) < 5:
        pl = int(normalize_a(parent).size)
        if pl not in used_lens:
            pr = resample_sequence(parent, pl)
            if pr.size not in used_lens:
                selected.append(("parent_direct", pr))

    while len(selected) < 3 and rng is not None and len(selected) > 0:
        base = selected[0][1]
        q = perturb_sequence(base, rng, 0.45)
        selected.append(("extra_perturb", q))

    if len(selected) == 0:
        selected.append(("fallback_uniform", np.ones(8192, dtype=np.float64) / 8192.0))

    return selected


def linear_objective(y, fixed, F, Frev, N, n, beta):
    """
    For fixed non-negative f (sum 1), minimize a smoothed version of
        max_l conv(f, softmax(y))_l.
    This inner problem is convex; it is used as an alternating polishing step.
    """
    t0 = time.monotonic()
    y = np.asarray(y, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return 1e100, np.zeros(0, dtype=np.float64)

    if not np.all(np.isfinite(y)):
        return 1e100, np.zeros(n, dtype=np.float64)

    g, deriv = y_to_soft(y)

    G = np.fft.rfft(g, N)
    b = np.fft.irfft(F * G, N)[:2 * n - 1]
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.maximum(b, 0.0)

    d = float(n) * b
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    w, L = lse_weights(d, beta, dmax)
    fval = 2.0 * L
    if not np.isfinite(fval):
        fval = 1e100

    W = np.fft.rfft(w, N)
    full = np.fft.irfft(Frev * W, N)[:3 * n - 2]
    h = full[n - 1:2 * n - 1]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    h = np.maximum(h, 0.0)

    # derivative wrt g: 2 n h
    grad_g = 2.0 * float(n) * h
    total = float(np.dot(g, grad_g))
    grad_z = g * (grad_g - total)
    grad_y = grad_z * deriv
    grad_y = grad_y - float(np.mean(grad_y))
    grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=1e6, neginf=-1e6)

    record_timing(n, time.monotonic() - t0)
    return float(fval), grad_y


def run_linear_g(f, beta, deadline, maxfun):
    f = normalize_a(f)
    n = int(f.size)
    if n <= 0:
        return f

    N = next_power_of_two(max(1, 3 * n - 2))
    try:
        F = np.fft.rfft(f, N)
        Frev = np.fft.rfft(f[::-1], N)
    except Exception:
        return f

    y = log_from_a(f)

    # Calibrate this length / fixed-f cost once.
    try:
        linear_objective(y, f, F, Frev, N, n, beta)
    except Exception:
        pass

    rem = float(deadline - time.monotonic())
    if rem < 10.0:
        return softmax_from_y(y)

    mf = estimate_maxfun(rem * 0.85, n)
    if maxfun is not None:
        mf = min(mf, int(maxfun))
    mf = max(16, mf)
    maxiter = int(max(30, mf // 2))

    opts = {
        "ftol": 1e-14,
        "gtol": 1e-12,
        "maxiter": maxiter,
        "maxfun": mf,
        "maxls": 25,
    }

    try:
        res = minimize(
            linear_objective,
            y,
            method="L-BFGS-B",
            jac=True,
            args=(f, F, Frev, N, n, float(beta)),
            options=opts,
        )
    except Exception:
        return softmax_from_y(y)

    y_out = getattr(res, "x", None)
    if y_out is None:
        y_out = y

    y_out = np.asarray(y_out, dtype=np.float64).ravel()
    if not np.all(np.isfinite(y_out)):
        return softmax_from_y(y)

    return normalize_a(softmax_from_y(y_out))


def alternating_polish(a0, time_budget, rng):
    t0 = time.monotonic()
    total = float(max(1.0, time_budget))
    f = normalize_a(a0)
    best_a = f.copy()
    best_sc = exact_score_a(f)

    if not np.isfinite(best_sc):
        best_a = np.ones(8192, dtype=np.float64) / 8192.0
        best_sc = exact_score_a(best_a)
        f = best_a.copy()

    print("[alt] start n={} score={:.8f}".format(int(f.size), best_sc))

    betas = [100.0, 400.0, 2000.0, 10000.0]
    pass_idx = 0
    no_improve = 0
    ts = [0.05, 0.10, 0.20, 0.35, 0.50, 0.75, 1.00]

    while (
        time.monotonic() < t0 + total - 15.0
        and pass_idx < len(betas)
        and no_improve < 3
    ):
        rem = float(t0 + total - time.monotonic())
        if rem < 60.0:
            break

        beta = float(betas[pass_idx])
        pass_budget = min(180.0, rem * 0.40)
        if pass_budget < 20.0:
            break

        g = run_linear_g(
            f,
            beta,
            time.monotonic() + pass_budget,
            estimate_maxfun(pass_budget * 0.75, int(f.size)),
        )

        if g is None or int(g.size) != int(f.size):
            break

        f_old = f.copy()
        best_cand = None
        best_cand_sc = best_sc

        for t in ts:
            q = (1.0 - float(t)) * f_old + float(t) * g
            q = normalize_a(q)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_cand_sc:
                best_cand_sc = float(sc)
                best_cand = q

        if best_cand is not None:
            best_sc = float(best_cand_sc)
            best_a = best_cand.copy()
            f = best_cand.copy()
            no_improve = 0
            print(
                "[alt] improve beta={:.0f} score={:.8f}".format(beta, best_sc)
            )
        else:
            no_improve += 1

        pass_idx += 1

    return float(best_sc), best_a


def get_parent_sequence():
    h = globals().get("height_sequence_1", None)
    if h is None:
        return None
    try:
        arr = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == 0 or float(arr.sum()) < 1e-12:
        return None
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total = float(max(1.0, budget_s))
    deadline = t_start + total - 45.0
    rng = np.random.default_rng(int(seed))

    print("Starting propose_candidate with budget={:.1f}s".format(total))
    parent = get_parent_sequence()
    print(
        "Parent available: {}, length={}".format(
            parent is not None,
            int(parent.size) if parent is not None else 0,
        )
    )

    starts = build_starts(parent, rng)
    if len(starts) == 0:
        starts.append(("fallback_uniform", np.ones(8192, dtype=np.float64) / 8192.0))

    global_seq = normalize_a(starts[0][1])
    global_score = exact_score_a(global_seq)
    print(
        "Initial global score={:.8f} n={}".format(global_score, int(global_seq.size))
    )

    # Main phase: optimize the most promising starts.
    main_reserve = 600.0 if total >= 1800.0 else 0.25 * total
    main_end = max(t_start + 120.0, deadline - main_reserve)
    main_budget = max(60.0, main_end - t_start)

    k = len(starts)
    if k == 1:
        fracs = [1.0]
    elif k == 2:
        fracs = [0.65, 0.35]
    elif k == 3:
        fracs = [0.55, 0.28, 0.17]
    else:
        fracs = [0.50, 0.24, 0.16, 0.10]

    cum = 0.0
    for i, (label, a0) in enumerate(starts):
        if i >= len(fracs):
            break
        cum += fracs[i]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)
        if time.monotonic() >= sub_dead - 30.0:
            continue

        alloc = max(30.0, sub_dead - time.monotonic())
        try:
            sc, a2 = optimize_sequence_y(a0, alloc, "main_{}".format(i + 1), rng)
        except Exception:
            sc, a2 = exact_score_a(a0), a0

        if np.isfinite(sc) and sc < global_score:
            global_score = float(sc)
            global_seq = normalize_a(a2)
            print(
                "Global best after main_{}: {:.8f} n={}".format(
                    i + 1, global_score, int(global_seq.size)
                )
            )

    # Alternating fixed-left polish phase.
    if time.monotonic() < deadline - 280.0:
        alt_budget = min(240.0, deadline - 280.0 - time.monotonic())
        if alt_budget > 60.0:
            try:
                sc, a2 = alternating_polish(global_seq, alt_budget, rng)
            except Exception:
                sc, a2 = global_score, global_seq
            if np.isfinite(sc) and sc < global_score:
                global_score = float(sc)
                global_seq = normalize_a(a2)
                print("Global best after alternating: {:.8f}".format(global_score))

    # Final direct high-beta polish.
    if time.monotonic() < deadline - 35.0:
        try:
            sc, a2 = quick_y_polish(global_seq, deadline - 10.0, "final")
        except Exception:
            sc, a2 = global_score, global_seq
        if np.isfinite(sc) and sc < global_score:
            global_score = float(sc)
            global_seq = normalize_a(a2)
            print("Global best after final polish: {:.8f}".format(global_score))

    # Prepare output.
    out_arr = np.asarray(global_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    if out_arr.size > 16384:
        out_arr = resample_sequence(out_arr, 16384)
        out_arr = np.asarray(out_arr, dtype=np.float64).ravel()
        out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(8192, dtype=np.float64) * (0.5 / 8192.0)
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(v) for v in out_arr]

    verified = np.nan
    if time.monotonic() < deadline - 60.0 and "evaluate_sequence" in globals():
        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = np.nan

    print(
        "FINAL internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
            global_score,
            verified if np.isfinite(verified) else float("nan"),
            len(out),
            time.monotonic() - t_start,
        )
    )

    return out
```
