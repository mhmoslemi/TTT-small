Actually, here's another idea: **iteratively reweighted least squares (IRLS)** for the max.

The idea: to minimize max(c[k]), we can use IRLS:
1. Start with weights w[k] = 1 for all k
2. Compute the weighted sum: L = sum_k w[k] * c[k]
3. Update weights: w[k] = w[k] * (c[k] / L)^p for some p > 0 (or w[k] = 1/c[k]^p)
4. Minimize the new weighted sum
5. Repeat

**Critical issue with LSE gradient descent**: The gradient can be noisy when beta is large (because p[k] concentrates on a few peaks, making q[j] a noisy estimate). This can cause the optimization to get stuck.

- Start with small beta (smooth, well-conditioned)
- Gradually increase beta
- At each beta, do several iterations of projected gradient
- Use the result as initialization for the next beta

This is similar to "continuation methods" in optimization.

"""
Hybrid LSE-continuation + GS-Fienup + Peak-directed optimization
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_and_grad(a, beta):
    """
    Compute LSE-smoothed max of conv(a,a) and its gradient.
    Returns (lse_value, gradient_array).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    LSE = c_max + np.log(exp_sum) / beta
    
    # q[j] = sum_k p[k] * a[k-j]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.fft(p, M2_fft)
    A = np.fft.fft(a, M2_fft)
    q = np.fft.ifft(P * np.conj(A)).real[:n]
    
    # score = 2n * LSE / S^2
    # d score / da[j] = (4n/S^2) * (q[j] - LSE/S)
    grad = (4.0 * n / (S * S)) * (q - LSE / S)
    return LSE, grad


def projected_grad_descent(a0, beta, n_iters=5000, step_init=None):
    """Projected gradient descent on LSE-smoothed score."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    
    if step_init is None:
        # Estimate initial step size
        _, grad = lse_and_grad(a, beta)
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            step = 1.0 / (beta * gn + 1e-10)
            step = min(step, 0.5 / S)
        else:
            step = 0.01 / S
    else:
        step = step_init
    
    for it in range(n_iters):
        if it % 2000 == 0 and n_iters > 5000:
            print(f"      PD iter {it}, score={best_score:.8f}")
        
        LSE, grad = lse_and_grad(a, beta)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-16:
            break
        
        # Backtracking
        alpha = step
        a_new = np.maximum(a - alpha * grad, 0.0)
        for _ in range(30):
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                alpha *= 0.5
                a_new = np.maximum(a - alpha * grad, 0.0)
                continue
            sc_new = compute_score(a_new)
            if sc_new < cur_score - 1e-15:
                break
            alpha *= 0.5
            a_new = np.maximum(a - alpha * grad, 0.0)
        else:
            break
        
        a = a_new
        cur_score = sc_new
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Adapt step size
        step = alpha * 1.05
    
    return best_a


def lse_continuation(a0, betas, iters_per_beta=3000):
    """Multi-scale LSE optimization."""
    a = a0.copy()
    for beta in betas:
        a = projected_grad_descent(a, beta, n_iters=iters_per_beta)
    return a


def gs_fienup(a0, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup iteration with flat convolution target."""
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()
    
    b_target = np.zeros(M)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init * (1 - progress) + alpha_final
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a


def peak_subgradient(a0, n_iters=5000, top_m=10):
    """Subgradient method focusing on top-M peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    
    best_a = a.copy()
    best_score = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    step = 0.03
    for it in range(n_iters):
        if it % 2000 == 0:
            print(f"      SG iter {it}, score={best_score:.8f}, step={step:.6f}")
        
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient: sum over top peaks of 2*a[k*-j]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        
        # Full gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        max_step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step * 1.01, max_step, 0.08 / S)
        
        a = np.maximum(a - step * grad, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
    
    return best_a


def multiplicative_peak_update(a0, n_iters=3000, top_m=5, rate=0.1):
    """
    Multiplicative update focusing on peak reduction.
    a[j] *= exp(-rate * (a[partner] - target))
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        if it % 1000 == 0:
            print(f"      MU iter {it}, score={best_score:.8f}")
        
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        target = S * S / (2.0 * n)  # target for max(c) to achieve score=1
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_m]
        
        # Compute the "excess" at each position
        excess = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # a[j] contributes a[j]*a[k-j] to c[k]
                # The "partner" of a[j] at peak k is a[k-j]
                partners = a[j_start:j_end+1][::-1]
                # Reduce a[j] proportionally to partner
                excess[j_start:j_end+1] += partners
        
        # Normalized excess
        excess_max = float(np.max(excess)) if np.max(excess) > 0 else 1.0
        excess_norm = excess / excess_max
        
        # Multiplicative update
        a = a * np.exp(-rate * excess_norm)
        a = np.maximum(a, 1e-15)
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a

"""
Hybrid optimization: GS-Fienup + LSE-continuation with Adam + Peak subgradient
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: Adam optimizer for LSE phase, multi-scale beta continuation,
and multiplicative peak-reduction updates.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_and_grad(a, beta):
    """
    Compute LSE-smoothed max of conv(a,a) and gradient of the score.
    Returns (lse_value, score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.inf, np.zeros(n)

    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    LSE = c_max + np.log(exp_sum) / beta

    # q[j] = sum_k p[k] * a[k-j], computed via FFT
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.fft(p, M2_fft)
    A = np.fft.fft(a, M2_fft)
    q = np.fft.ifft(P * np.conj(A)).real[:n]

    score = 2.0 * n * LSE / (S * S)
    grad = (4.0 * n / (S * S)) * (q - LSE / S)
    return LSE, score, grad


def adam_lse_descent(a0, beta, n_iters=10000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed score with non-negativity projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)

    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score

    for it in range(1, n_iters + 1):
        if it % 5000 == 0:
            print(f"        Adam iter {it}, score={best_score:.8f}")

        LSE, score, grad = lse_and_grad(a, beta)
        if not np.isfinite(score):
            break

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new < cur_score:
            a = a_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
        else:
            # Try smaller step
            a_try = a - 0.1 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try > 1e-12:
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < cur_score:
                    a = a_try
                    cur_score = sc_try
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a.copy()

    return best_a


def projected_grad_descent(a0, beta, n_iters=10000, step_init=None):
    """Projected gradient descent on LSE-smoothed score with backtracking."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score

    if step_init is None:
        _, _, grad = lse_and_grad(a, beta)
        gn = float(np.linalg.norm(grad))
        step = 1.0 / (gn + 1e-10) * 0.1
        step = min(step, 0.1)
    else:
        step = step_init

    for it in range(n_iters):
        if it % 5000 == 0:
            print(f"        PGD iter {it}, score={best_score:.8f}, step={step:.8f}")

        LSE, score, grad = lse_and_grad(a, beta)
        if not np.isfinite(score):
            break
        gn = float(np.linalg.norm(grad))
        if gn < 1e-16:
            break

        alpha = step
        success = False
        for _ in range(25):
            a_new = np.maximum(a - alpha * grad, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                alpha *= 0.5
                continue
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < cur_score - 1e-14:
                success = True
                break
            alpha *= 0.5
        if not success:
            break

        a = a_new
        cur_score = sc_new
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        step = alpha * 1.1

    return best_a


def gs_fienup(a0, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with flat convolution target."""
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init * (1 - progress) + alpha_final

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a


def peak_subgradient(a0, n_iters=5000, top_m=10):
    """Subgradient method focusing on top-M peaks with adaptive step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    step = 0.02
    stall_count = 0
    for it in range(n_iters):
        if it % 3000 == 0:
            print(f"        SG iter {it}, score={best_score:.8f}, step={step:.6f}")

        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            break

        top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        max_step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step * 1.005, max_step, 0.05 / S)

        a = np.maximum(a - step * grad, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

    return best_a


def multiplicative_peak_update(a0, n_iters=3000, top_m=5, rate=0.05):
    """Multiplicative update reducing peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if it % 2000 == 0:
            print(f"        MU iter {it}, score={best_score:.8f}")

        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:top_m]

        excess = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partners = a[j_start:j_end + 1][::-1]
                excess[j_start:j_end + 1] += partners

        exc_max = float(np.max(excess))
        if exc_max > 0:
            excess_norm = excess / exc_max
            a = a * np.exp(-rate * excess_norm)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a


def lbfgs_refine(a0, beta=2000, maxiter=3000, maxfun=50000):
    """L-BFGS-B refinement on LSE-smoothed objective."""
    n = len(a0)

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        LSE, score, grad = lse_and_grad(x, beta)
        if not np.isfinite(score):
            return 1e10, np.zeros(n)
        return score, grad

    a0 = np.maximum(a0, 0.0).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    a0 = a0 / a0.sum()

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.copy(), 0.01, 1000.0))
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5), 0.01))

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Strategy: GS-Fienup for initialization, then LSE-continuation with Adam,
    then peak subgradient, multiplicative updates, and L-BFGS polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None:
        n = len(parent)

    print(f"n={n}, parent={'yes' if parent is not None else 'no'}")

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def try_update(a, label):
        nonlocal best_score, best_a
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={elapsed:.0f}s)")

    # ================================================================
    # Phase 1: GS-Fienup initialization (10% time)
    # ================================================================
    phase1_end = t_start + 0.10 * (budget_s - 30)
    print(f"\n=== Phase 1: GS-Fienup (until t={phase1_end - t_start:.0f}s) ===")

    inits = build_inits(n, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a_gs = gs_fienup(a0, n_iters=800, alpha_init=0.8, alpha_final=0.1)
        try_update(a_gs, f"GS-{i}")

    print(f"  After GS: best={best_score:.8f}")

    # ================================================================
    # Phase 2: LSE Continuation with Adam (55% time)
    # ================================================================
    phase2_end = t_start + 0.65 * (budget_s - 30)
    print(f"\n=== Phase 2: LSE Continuation + Adam (until t={phase2_end - t_start:.0f}s) ===")

    betas = [30, 60, 100, 200, 400, 800, 1500, 3000, 6000, 12000]
    a_cur = best_a.copy()

    for bi, beta in enumerate(betas):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        n_betas_left = len(betas) - bi
        time_per_beta = remaining / n_betas_left

        # Estimate iterations possible: ~50000 iter/s for this problem size
        est_iters = int(time_per_beta * 30000)
        est_iters = max(2000, min(est_iters, 50000))

        print(f"  Beta={beta}, est_iters={est_iters}, time_budget={time_per_beta:.0f}s")

        # Adam
        t_beta_start = time.monotonic()
        a_adam = adam_lse_descent(a_cur, beta, n_iters=est_iters, lr=0.05)
        try_update(a_adam, f"Adam-β{beta}")
        if time.monotonic() - t_beta_start < time_per_beta * 0.4:
            # Still have time, do more with smaller lr
            a_adam2 = adam_lse_descent(a_adam, beta, n_iters=est_iters // 2, lr=0.01)
            try_update(a_adam2, f"Adam2-β{beta}")
            a_cur = a_adam2 if compute_score(a_adam2) < compute_score(a_adam) else a_adam
        else:
            a_cur = a_adam

    print(f"  After LSE continuation: best={best_score:.8f}")

    # ================================================================
    # Phase 3: Peak Subgradient + Multiplicative (20% time)
    # ================================================================
    phase3_end = t_start + 0.85 * (budget_s - 30)
    print(f"\n=== Phase 3: Peak SG + MU (until t={phase3_end - t_start:.0f}s) ===")

    for top_m in [5, 10, 20]:
        if time.monotonic() > phase3_end - 60:
            break
        a_sg = peak_subgradient(best_a, n_iters=5000, top_m=top_m)
        try_update(a_sg, f"SG-m{top_m}")

    if time.monotonic() < phase3_end - 30:
        a_mu = multiplicative_peak_update(best_a, n_iters=3000, top_m=5, rate=0.05)
        try_update(a_mu, "MU")

    # ================================================================
    # Phase 4: L-BFGS polish + final Adam (10% time)
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    if time.monotonic() < deadline - 60:
        sc, a_lb = lbfgs_refine(best_a, beta=5000, maxiter=3000, maxfun=50000)
        try_update(a_lb, "LBFGS-5000")

    if time.monotonic() < deadline - 30:
        a_fin = adam_lse_descent(best_a, beta=20000, n_iters=5000, lr=0.01)
        try_update(a_fin, "Adam-final")

    # ================================================================
    # Final
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
