"""
Optimized step function construction using spectral shaping with geometric
amplitude equalization, peak-directed time-domain refinement, and L-BFGS
polish. Key improvements: (1) full-spectrum geometric shaping instead of
top-fraction clipping, (2) peak-directed coordinate descent in time domain,
(3) adaptive multi-phase strategy with better time allocation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_geometric(a, n_iters=400, alpha_start=0.1, alpha_end=0.9,
                     use_sigmoid=True, sigmoid_alpha=50.0):
    """
    Fienup iteration with geometric amplitude shaping applied to ALL bins.
    amp_new = amp * (C/amp)^alpha where C is a target amplitude level.
    This smoothly pushes the spectrum toward flatness.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        # Target: use median of non-DC amplitudes
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        # Geometric shaping: push amp toward C
        # amp_new = amp * (C/amp)^alpha = amp^(1-alpha) * C^alpha
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        # Reconstruct complex spectrum
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Non-negativity
        if use_sigmoid:
            gate = 1.0 / (1.0 + np.exp(-sigmoid_alpha * np.clip(a_new, -50, 50)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_power_law(a, n_iters=400, gamma=0.5):
    """
    Power-law spectral shaping: for bins above target, apply
    amp_new = amp * (target/amp)^gamma. Below target, leave unchanged.
    This is more conservative than full geometric shaping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_new = amp.copy()
        above = amp > C * (1.0 + 1e-10)
        if above.any():
            amp_new[above] = amp[above] * (C / amp[above]) ** gamma

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_ellip(a, n_iters=400, beta_shape=2.0):
    """
    Elliptical spectral shaping: amp_new = amp / (1 + beta*(amp/C - 1)^2)^0.5
    This provides a smooth, bounded compression of high-amplitude bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        beta = beta_shape * (0.5 + progress)

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        ratio = amp / C
        amp_new = amp / np.sqrt(1.0 + beta * (ratio - 1.0) ** 2)

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_directed_descent(a, n_iters=3000, n_peaks=5, step_init=0.02, decay=0.9999):
    """
    Directly target the peaks of conv(a,a) by reducing the elements of a
    that contribute most to those peaks. Uses a coordinate-descent-like
    approach with FFT-based convolution for efficiency.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        # Compute subgradient: for each peak at position k,
        # d c[k]/d a[m] = a[k-m] (if 0<=k-m<n)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective with gradient for L-BFGS."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: p_k = exp(beta*(c_k - c_max)) / sum(exp(...))
    p = exp_scaled / exp_sum
    # dL/dx_i = 2 * sum_k p_k * (d c_k / d x_i)
    # d c_k / d x_i = x[k-i] (if 0<=k-i<n) * + x[i] (if 0<=k-i... wait)
    # Actually c_k = sum_j x[j]*x[k-j], so d c_k/d x_i = x[k-i] + x[k-i]... 
    # No: d/dx_i [sum_j x[j]*x[k-j]] = x[k-i] (from j=i term) + x[i] (from k-j=i term, j=k-i)
    # = x[k-i] + (x[i] if k-i != i else 0) ... actually it's just x[k-i] counted once if k != 2i
    # More carefully: c_k = sum_j x[j] x[k-j]
    # d c_k / d x_i = x[k-i] (when j=i) + sum over j where k-j=i, i.e., j=k-i: x[k-i]
    # If i != k-i (i.e., 2i != k): d c_k/d x_i = 2*x[k-i]
    # If i == k-i (i.e., 2i == k): d c_k/d x_i = x[i]
    # For simplicity, use the convolution: conv(p, x_reversed) handles this
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    # conv_res[i] = sum_k p[k] * x[k-i+... ] -- need to verify indexing
    # Actually conv(p, rev_x)[i + n - 1] = sum_k p[k] * rev_x[i+n-1-k] = sum_k p[k] * x[n-1-(i+n-1-k)] = sum_k p[k] * x[k-i]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_hybrid(a, n_iters=300, alpha_start=0.1, alpha_end=0.8,
                  n_peaks=3, step_peak=0.01):
    """
    Hybrid: alternate between geometric spectral shaping and
    peak-directed time-domain descent.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # Spectral shaping step
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Peak-directed step (every 3rd iteration)
        if it % 3 == 0:
            A2 = np.fft.rfft(a, M_fft)
            c = np.fft.irfft(A2 * A2, M_fft)[:2 * n - 1]
            S2 = float(a.sum())
            M_val = float(c.max())
            sc_curr = 2.0 * n * M_val / (S2 * S2)

            if n_peaks >= len(c):

            else:
                top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

            sg = np.zeros(n)

            S2v = S2 * S2
            S3v = S2v * S2
            grad = (2.0 * n) * (sg / S2v - 2.0 * M_val / S3v)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                step = min(step_peak, max((sc_curr - sc_lb) / (gn * gn + 1e-30), 1e-8))
                step = min(step, 0.02)
                a -= step * grad
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S < 1e-12:
                    break
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Edge-enhanced (high at edges, low in middle)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))

    # Inverse sinc-like (suppression at edges)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))

    # Gaussian at center
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)

    # Two peaks
    inits.append(np.maximum(0.2 + 0.6 * np.cos(4.0 * np.pi * (x - 0.25)) +
                            0.6 * np.cos(4.0 * np.pi * (x - 0.75)), 0.01))

    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.5, size=5)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    # Power law
    inits.append(np.maximum(0.1 + 0.5 * x ** 0.5 + 0.5 * (1 - x) ** 0.5, 0.01))

    # Cosine window (Hann-like)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits

    # ============================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Peak-Directed Descent (deadline t={phase2_end - t_start:.0f}s) ===")

    for n_peaks in [3, 5, 8, 15, 30]:
        if time.monotonic() > phase2_end:
            break
        print(f"  n_peaks={n_peaks}")
        a_pd = peak_directed_descent(best_a, n_iters=2000, n_peaks=n_peaks,
                                     step_init=0.015, decay=0.99995)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

    # Hybrid (spectral + peak)
    if time.monotonic() < phase2_end:
        a_h = fienup_hybrid(best_a, n_iters=400, alpha_start=0.05, alpha_end=0.7,
                            n_peaks=5, step_peak=0.01)
        sc = compute_score(a_h)
        update_best(sc, a_h, "HYBRID")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS refinement (25% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    betas = [100, 300, 1000, 3000, 10000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 120:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(8000, int(remaining * 30))
        maxfun = maxiter * 25
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            # Continue with next beta from improved solution
            pass

    # Also try L-BFGS from parent at different betas
    if parent is not None and len(parent) == len(best_a):
        if time.monotonic() < phase3_end:
            for beta in [500, 2000]:
                if time.monotonic() > phase3_end:
                    break
                remaining = phase3_end - time.monotonic()
                if remaining < 60:
                    break
                sc, a_opt = run_lbfgs(parent, beta, maxiter=5000, maxfun=100000)
                update_best(sc, a_opt, f"LBFGS-P b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # More iterations of the best method
    if time.monotonic() < deadline - 60:
        a_final = fienup_geometric(best_a, n_iters=200, alpha_start=0.02, alpha_end=0.5)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-GEO")

    if time.monotonic() < deadline - 60:
        a_final2 = fienup_ellip(best_a, n_iters=200, beta_shape=5.0)
        sc = compute_score(a_final2)
        update_best(sc, a_final2, "FINAL-ELLI")

    if time.monotonic() < deadline - 30:
        a_final3 = peak_directed_descent(best_a, n_iters=1500, n_peaks=10,
                                         step_init=0.005, decay=0.99999)
        sc = compute_score(a_final3)
        update_best(sc, a_final3, "FINAL-PEAK")

    # Final L-BFGS if time
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.02 * (1 + trial)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            a_r = fienup_geometric(a_p, n_iters=100, alpha_start=0.05, alpha_end=0.6)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
