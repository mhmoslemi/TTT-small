def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 2 * n
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

Multi-resolution spectral optimization with LSE-LBFGS for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines Fienup-style iterative spectral equalization at multiple resolutions
with L-BFGS local refinement using log-sum-exp smoothing of the max operator.

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_flat_iter(a, n_iters, mode='compress', clip_mode='soft', soft_alpha=30.0):
    """Iterative spectral flattening with multiple modes."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        amp = np.sqrt(np.maximum(P, 1e-30))

        if mode == 'replace':
            target_amp = np.full_like(amp, float(amp.mean()))
            s = 0.2 + 0.6 * progress
            new_amp = (1.0 - s) * amp + s * target_amp
        elif mode == 'power':
            norm_P = P / (float(P.mean()) + 1e-30)
            exponent = 1.0 - (0.2 + 0.5 * progress)
            new_P = np.maximum(norm_P, 0.0) ** exponent
            new_amp = np.sqrt(new_P * float(P.mean()))

            top_frac = 0.15 + 0.35 * progress
            thresh_val = np.quantile(P, max(0.01, 1.0 - top_frac))
            new_P = P.copy()

            new_amp = np.sqrt(np.maximum(new_P, 0.0))

        a = a_new

        sc = compute_score(a)

    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    # Smooth random
    r = rng.standard_normal(n)
    Ar = np.fft.rfft(r)
    freqs = np.fft.rfftfreq(n)
    Ar[freqs > 0.05] *= 0.01
    smooth_r = np.fft.irfft(Ar, n)
    inits.append(np.abs(smooth_r) + 0.1)
    # Two-lobe
    inits.append(np.maximum(0.3 + 0.6 * np.cos(4.0 * np.pi * (x - 0.25)), 0.01))

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(upsample_interp(parent, n))

    return inits


def peak_chase_subgrad(a, n_iters=5000, top_m=10, step_init=0.02, decay=0.9995):
    """Subgradient method targeting the top peaks of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.04)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-resolution spectral + L-BFGS optimizer for minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    n_target = 6962
    if parent is not None:
        n_target = len(parent)

    print(f"Target n={n_target}, parent={'yes n='+str(len(parent)) if parent is not None else 'no'}")

    if parent is not None:
        sc = compute_score(parent)
        if sc < best_score:
            best_score = sc
            best_a = parent.copy()
            print(f"  [PARENT] score={sc:.8f}")

    # ================================================================
    # PHASE 1: Quick L-BFGS from parent for fast baseline improvement
    # ================================================================
    p1_end = t_start + 0.06 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Quick LBFGS (t<{p1_end - t_start:.0f}s) ===")

    if parent is not None and time.monotonic() < p1_end:
        for beta in [100, 500]:
            if time.monotonic() > p1_end:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_opt, 0, 1000)
                print(f"  [P1 LBFGS b={beta}] score={sc:.8f}")

    # ================================================================
    # PHASE 2: Multi-resolution spectral optimization
    # ================================================================
    p2_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Multi-res spectral (t<{p2_end - t_start:.0f}s) ===")

    # Work at multiple resolutions from coarse to fine
    resolutions = sorted(set([n_target, n_target // 2, n_target // 4]))

    current_coarse = None
    for n_res in resolutions:
        if time.monotonic() > p2_end:
            break
        print(f"  Resolution n={n_res}")

        inits = make_initializations(n_res, rng, parent)
        # Add downscaled current best as additional init
        if current_coarse is not None and len(current_coarse) != n_res:
            inits.append(upsample_interp(current_coarse, n_res))
        elif current_coarse is not None:
            inits.append(current_coarse.copy())

        best_at_res = None
        best_at_res_sc = np.inf

        for i, a0 in enumerate(inits):
            if time.monotonic() > p2_end:
                break

            for mode in ['replace', 'compress', 'power']:
                if time.monotonic() > p2_end:
                    break
                a_out = spectral_flat_iter(a0, 250, mode=mode, clip_mode='soft')
                sc = compute_score(a_out)
                if sc < best_at_res_sc:
                    best_at_res_sc = sc
                    best_at_res = a_out.copy()
                if sc < best_score:
                    best_score = sc
                    best_a = np.clip(a_out, 0, 1000)
                    print(f"    [{mode} n={n_res} i={i}] score={sc:.8f}")

        if best_at_res is not None:
            current_coarse = best_at_res

    print(f"  After Phase 2: best={best_score:.8f}")

    # ================================================================
    # PHASE 3: Extended spectral from current best
    # ================================================================
    p3_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Extended spectral (t<{p3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_cur = best_a.copy() / best_a.sum()

    configs = [
        ('replace', 300, 'soft'),
        ('compress', 300, 'soft'),
        ('power', 300, 'waterfill'),
        ('adaptive', 400, 'soft'),
        ('replace', 300, 'hard'),
        ('compress', 400, 'waterfill'),
        ('power', 300, 'soft'),
        ('adaptive', 400, 'hard'),
    ]

    for mode, n_it, clip in configs:
        if time.monotonic() > p3_end:
            break
        a_out = spectral_flat_iter(a_cur, n_it, mode=mode, clip_mode=clip)
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_out, 0, 1000)
            a_cur = a_out / a_out.sum()
            print(f"  [P3 {mode}/{clip}] score={sc:.8f}")

    # Perturbation restarts
    for trial in range(4):
        if time.monotonic() > p3_end:
            break
        noise = rng.standard_normal(n_best)
        a_p = best_a * (1.0 + 0.03 * (trial + 1) * noise)
        a_p = np.clip(a_p, 0, 1000)
        S = a_p.sum()
        if S < 1e-12:
            continue
        a_p = a_p / S
        a_out = spectral_flat_iter(a_p, 200, mode='replace', clip_mode='soft')
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_out, 0, 1000)
            print(f"  [P3 pert{trial}] score={sc:.8f}")

    print(f"  After Phase 3: best={best_score:.8f}")

    # ================================================================
    # PHASE 4: L-BFGS with increasing beta
    # ================================================================
    p4_end = t_start + 0.82 * (total_budget - 30.0)
    print(f"\n=== Phase 4: LBFGS (t<{p4_end - t_start:.0f}s) ===")

    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() > p4_end:
            break
        remaining = p4_end - time.monotonic()
        if remaining < 60:
            break
        min_per_eval = len(best_a) / 3000.0
        maxiter = min(20000, int(remaining * 0.7 / max(min_per_eval, 0.005)))
        maxiter = max(maxiter, 1000)
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt, 0, 1000)
            print(f"  [LBFGS b={beta}] score={sc:.8f}")

    # L-BFGS from a fresh spectral-optimized starting point
    if time.monotonic() < p4_end - 90:
        a_fresh = spectral_flat_iter(best_a, 150, mode='replace', clip_mode='soft')
        sc, a_opt = run_lbfgs(a_fresh, 2000, maxiter=5000, maxfun=80000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt, 0, 1000)
            print(f"  [LBFGS-from-spec] score={sc:.8f}")

    print(f"  After Phase 4: best={best_score:.8f}")

    # ================================================================
    # PHASE 5: Peak-chasing subgradient + final polish
    # ================================================================
    p5_end = deadline
    print(f"\n=== Phase 5: Subgradient + polish (t<{p5_end - t_start:.0f}s) ===")

    for top_m in [5, 10, 20]:
        if time.monotonic() > p5_end - 60:
            break
        a_sg = peak_chase_subgrad(best_a, n_iters=3000, top_m=top_m,
                                   step_init=0.02, decay=0.9995)
        sc = compute_score(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_sg, 0, 1000)
            print(f"  [SG m={top_m}] score={sc:.8f}")

    # Final L-BFGS polish with high beta
    if time.monotonic() < p5_end - 30:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt, 0, 1000)
            print(f"  [FINAL-LBFGS] score={sc:.8f}")

    # Final spectral sweep from best
    if time.monotonic() < p5_end - 15:
        for mode in ['replace', 'adaptive']:
            a_out = spectral_flat_iter(best_a, 100, mode=mode, clip_mode='soft')
            sc = compute_score(a_out)
            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_out, 0, 1000)
                print(f"  [FINAL-SPEC {mode}] score={sc:.8f}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
