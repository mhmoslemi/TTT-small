    # In our convolution: conv_res[k] = sum_j a[j] * p[2n-2-k+j]
    # Set j = m: sum_m a[m] * p[2n-2-k+m]
    # We want sum_m p[m] * a[m-i] = sum_j p[j+i] * a[j] (letting j = m-i, m = j+i)
    # = sum_j a[j] * p[j+i]
    # Set 2n-2-k+m = j+i where m=j: 2n-2-k+j = j+i => 2n-2-k = i => k = 2n-2-i
    # So dL/da_i = 2 * conv_res[2n-2-i]
    
    grad_L = 2.0 * conv_res[2*n-2 : 3*n-2][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def variance_score_and_grad(a):
    """
    Score based on sum(c²) and its gradient.
    f(a) = 2n * sum(c²) / ((2n-1) * (sum a)²)
    This is a lower bound on the actual score (by Cauchy-Schwarz).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    sum_c2 = float(np.dot(c, c))
    S2 = S * S
    score = 2.0 * n * sum_c2 / ((2.0*n - 1.0) * S2)
    
    # Gradient: d/d(a_i) [sum c² / (sum a)²]
    # = (2 * grad_sumc2 * S² - sum_c2 * 2 * S) / S⁴
    # = (2 * grad_sumc2 / S) - 2 * sum_c2 / S³
    # where grad_sumc2[i] = d(sum c²)/d(a_i) = 4 * (c ⋆ a)[i]
    
    # Compute cross-correlation c ⋆ a via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    C_pad = np.zeros(M_fft)
    C_pad[:2*n-1] = c
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    Cft = np.fft.rfft(C_pad)
    Aft = np.fft.rfft(A_pad)
    # Cross-correlation: IFFT(FFT(c) * conj(FFT(a)))
    cross = np.fft.irfft(Cft * np.conj(Aft), M_fft)[:n]
    
    grad_sumc2 = 4.0 * cross
    
    S3 = S2 * S
    grad = (2.0 * n / (2.0*n - 1.0)) * (2.0 * grad_sumc2 / S - 2.0 * sum_c2 / S3)
    
    return score, grad

def nesterov_var(a, n_iters=2000, lr_start=0.5, lr_end=0.05, mu_start=0.9, mu_end=0.98):
    """Nesterov accelerated gradient on variance objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        lr = lr_start + (lr_end - lr_start) * t
        mu = mu_start + (mu_end - mu_start) * t
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = variance_score_and_grad(a_look)
        
        # Adaptive step
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        effective_lr = min(lr, 0.5 / (gnorm + 1e-10))
        
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:

                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:

        best_sc = sc
    
    return best_a

def hybrid_fienup_lse(a, n_iters=500, fienup_every=3, beta=500.0, lr=0.005):
    """
    Hybrid: alternate between Fienup steps and LSE gradient steps.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    mu = 0.9
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        # Increase beta over time
        beta_t = max(50.0, 100.0 / (0.1 + t))
        
        if it % fienup_every == 0:
            # Fienup step
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P_mean = float(P.mean())
            thresh = P_mean * 1.3
            above = P > thresh
            if above.any():
                scale = np.ones_like(P)
                scale[above] = ((P_mean * 1.3) / P[above]) ** 0.3
                A_mod = A * scale
                a_new = np.fft.irfft(A_mod, M_fft)[:n]
                a_new = np.maximum(a_new, 0.0)
                S_new = a_new.sum()
                if S_new > 1e-12:
                    a = a_new / S_new
        else:
            # LSE gradient step
            _, grad = lse_score_and_grad(a, beta_t)
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-15:
                eff_lr = min(lr, 0.5 / (gnorm + 1e-10))
                v = mu * v - eff_lr * grad
                a = a + v
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S > 1e-12:
                    a = a / S
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a

def build_tukey_init(n, r=0.3):
    """Tukey (tapered cosine) window."""
    x = np.arange(n, dtype=np.float64)
    half = n / 2.0
    d = np.abs(x - half) / half
    a = np.ones(n)
    taper = d > (1 - r)
    a[taper] = 0.5 * (1 + np.cos(np.pi * (d[taper] - (1-r)) / r))
    a = np.maximum(a, 0.01)
    return a / a.sum()

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))² over non-negative sequences.
Key innovations over previous approach:
1. Variance-based pre-optimization (smooth, fast convergence)
2. Nesterov LSE with adaptive beta and gradient-norm-based step sizing
3. Hybrid Fienup + LSE gradient alternating
4. Multi-start with diverse initializations (m-sequence, chirp, spectral random)
5. Aggressive L-BFGS polish with beta sweep
6. Peak-targeted coordinate descent for final micro-optimization

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    es_sum = float(exp_shifted.sum())
    L = c_max + np.log(es_sum) / beta
    p = exp_shifted / es_sum

    # Gradient: dL/da_i = 2 * sum_m p[m] * a[m-i]
    # Computed as cross-correlation of p and a
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[2*n-2:3*n-2][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def variance_score_and_grad(a):
    """Variance-based smooth score and gradient. f = 2n*sum(c²)/((2n-1)*S²)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    sum_c2 = float(np.dot(c, c))
    S2 = S * S
    score = 2.0 * n * sum_c2 / ((2.0 * n - 1.0) * S2)

    # Gradient: d(sum c²)/d(a_i) = 4 * (c ⋆ a)[i]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    C_pad = np.zeros(M_fft)
    C_pad[:2*n-1] = c
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    Cft = np.fft.rfft(C_pad)
    Aft = np.fft.rfft(A_pad)
    cross = np.fft.irfft(Cft * np.conj(Aft), M_fft)[:n]
    grad_sumc2 = 4.0 * cross

    S3 = S2 * S
    coeff = 2.0 * n / (2.0 * n - 1.0)
    grad = coeff * (2.0 * grad_sumc2 / S - 2.0 * sum_c2 / S3)
    return score, grad


def nesterov_var(a, n_iters=5000, lr_start=0.3, lr_end=0.01, mu_start=0.9, mu_end=0.97):
    """Nesterov accelerated gradient on variance objective (fast, smooth)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        lr = lr_start * (lr_end / lr_start) ** t
        mu = mu_start + (mu_end - mu_start) * t

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = variance_score_and_grad(a_look)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        eff_lr = min(lr, 0.3 / (gnorm + 1e-10))

        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def nesterov_lse_long(a, beta_start=10.0, beta_end=50000.0, n_iters=20000,
                      lr_start=0.02, lr_end=0.001, mu_start=0.9, mu_end=0.97):
    """Extended Nesterov LSE with exponential beta schedule."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        mu = mu_start + (mu_end - mu_start) * t

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            a = best_a.copy()
            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        eff_lr = min(lr, 0.3 / (gnorm + 1e-10))

        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_steps(a, n_iters=200, thresh=1.5, clip=0.4):
    """Pure Fienup spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        above = P > P_mean * thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def hybrid_fienup_lse(a, n_iters=5000, fienup_every=5, lr=0.003, mu=0.92):
    """Alternate Fienup and LSE gradient steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        beta_t = 50.0 + 5000.0 * t

        if it % fienup_every == 0:
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P_mean = float(P.mean())
            thresh = 1.0 + 0.5 * (1.0 - t)
            above = P > P_mean * thresh
            if above.any():
                scale = np.ones_like(P)
                scale[above] = np.sqrt(P_mean * thresh / P[above]) ** 0.4
                A_mod = A * scale
                a_f = np.fft.irfft(A_mod, M_fft)[:n]
                a_f = np.maximum(a_f, 0.0)
                S_f = a_f.sum()
                if S_f > 1e-12:
                    a = a_f / S_f
        else:
            _, grad = lse_score_and_grad(a, beta_t)
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-14:
                eff_lr = min(lr, 0.2 / (gnorm + 1e-10))
                v = mu * v - eff_lr * grad
                a = a + v
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S > 1e-12:
                    a = a / S

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=5000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    def objective(x):
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                 'maxfun': 200000, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / a_out.sum()
    return compute_score(a_out), a_out


def peak_cd(a, n_sweeps=5, n_top=30):
    """Peak-targeted coordinate descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_tl = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_tl]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                improved = False
                for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_tmp = a / S_tmp
                    sc = compute_score(a_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_tmp.copy()
                        a = a_tmp
                        improved = True
                        break
                if not improved:
                    a[j] = old_val

    return best_a


def build_mseq_init(n, rng):
    """M-sequence based initialization (smoothed)."""
    x = np.linspace(0, 1, n, endpoint=False)
    # Sum of chirps at different rates (approximates m-sequence spreading)
    a = np.ones(n) * 0.5
    for rate in [1, 3, 7, 15]:
        a += 0.3 * np.abs(np.sin(np.pi * rate * x * (1 + 0.5 * x)))
    a = np.maximum(a, 0.01)
    # Add small random perturbation
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def build_chirp_init(n):
    """Quadratic chirp absolute value."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.abs(np.sin(2 * np.pi * 2.0 * x * x)) + 0.3
    return a / a.sum()


def build_tukey_init(n, r=0.4):
    """Tukey window."""
    x = np.arange(n, dtype=np.float64)
    half = n / 2.0
    d = np.abs(x - half) / half
    a = np.ones(n)
    taper = d > (1 - r)
    a[taper] = 0.5 * (1 + np.cos(np.pi * (d[taper] - (1 - r)) / r))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_spectral_flat_init(n, rng, n_fienup=50):
    """Random init followed by aggressive Fienup spectral flattening."""
    a = np.abs(rng.standard_normal(n)) + 0.5
    a = a / a.sum()
    M_fft = 1 << (2 * n - 1).bit_length()
    for i in range(n_fienup):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        # Drive spectrum toward flat
        target = np.full_like(P, P_mean)
        blend = 0.3 + 0.5 * (i / n_fienup)
        P_new = (1 - blend) * P + blend * target
        A_new = A * np.sqrt(np.maximum(P_new, 1e-10) / np.maximum(P, 1e-10))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return a


def upsample(a, new_n):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: variance pre-opt -> Nesterov LSE -> hybrid -> L-BFGS -> peak CD.
    Uses multi-start with diverse initializations and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent if available
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 7000

    best_score = np.inf
    best_a = np.ones(n_target) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Starting optimization. n={n_target}, budget={total_budget}s")

    # ============================================================
    # Phase 1: Multi-start initialization + variance pre-optimization
    # ============================================================
    phase1_end = t_start + 0.15 * (total_budget - 20.0)
    print(f"\n=== Phase 1: Multi-start + variance pre-opt ===")

    initializations = []

    # Parent (if available)
    if parent is not None:
        initializations.append(('parent', parent.copy()))

    # Diverse random initializations
    initializations.append(('mseq', build_mseq_init(n_target, rng)))
    initializations.append(('chirp', build_chirp_init(n_target)))
    initializations.append(('tukey', build_tukey_init(n_target, 0.3)))
    initializations.append(('spec-flat', build_spectral_flat_init(n_target, rng, 30)))

    candidates = []
    for name, init_a in initializations:
        if time.monotonic() > phase1_end:
            break
        # Quick variance optimization
        a_var = nesterov_var(init_a, n_iters=3000, lr_start=0.3, lr_end=0.01)
        sc = compute_score(a_var)
        candidates.append((sc, a_var, name))
        print(f"  {name}: variance-opt score = {sc:.8f}")

    # Sort by score and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: {candidates[0][2]} with score {candidates[0][0]:.8f}")

    # Update best from top candidate
    for sc, a_c, name in candidates[:3]:
        update_best(sc, a_c, f"var-{name}")

    # ============================================================
    # Phase 2: Nesterov LSE deep optimization (40% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * (total_budget - 20.0)
    print(f"\n=== Phase 2: Nesterov LSE deep opt ===")

    # Run from the best candidate with different beta schedules
    lse_configs = [
        {'beta_start': 10.0, 'beta_end': 50000.0, 'n_iters': 15000,
         'lr_start': 0.02, 'lr_end': 0.001, 'mu_start': 0.9, 'mu_end': 0.97},
        {'beta_start': 50.0, 'beta_end': 30000.0, 'n_iters': 15000,
         'lr_start': 0.015, 'lr_end': 0.0005, 'mu_start': 0.92, 'mu_end': 0.97},
        {'beta_start': 5.0, 'beta_end': 100000.0, 'n_iters': 15000,
         'lr_start': 0.025, 'lr_end': 0.0005, 'mu_start': 0.88, 'mu_end': 0.96},
    ]

    for i, cfg in enumerate(lse_configs):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        # Scale iterations to fit time budget (estimate ~0.5ms per iter)
        max_iters_fit = int(time_left * 1000 / 0.8)
        n_iters = min(cfg['n_iters'], max_iters_fit)
        if n_iters < 1000:
            break

        print(f"  LSE config {i}: beta [{cfg['beta_start']}, {cfg['beta_end']}], iters={n_iters}")
        a_lse = nesterov_lse_long(best_a, **{**cfg, 'n_iters': n_iters})
        update_best(compute_score(a_lse), a_lse, f"lse-{i}")

    # Restart from perturbed best
    if time.monotonic() < phase2_end - 30:
        perturbations = [0.01, 0.03, 0.05]
        for idx, pert in enumerate(perturbations):
            if time.monotonic() > phase2_end:
                break
            a_pert = best_a * (1.0 + pert * rng.standard_normal(len(best_a)))
            a_pert = np.maximum(a_pert, 0.0)
            S_p = a_pert.sum()
            if S_p > 1e-12:
                a_pert = a_pert / S_p
            time_left = phase2_end - time.monotonic()
            n_iters = min(10000, int(time_left * 1000 / 0.8))
            if n_iters < 1000:
                break
            a_restart = nesterov_lse_long(a_pert, beta_start=50.0, beta_end=30000.0,
                                          n_iters=n_iters, lr_start=0.01,
                                          lr_end=0.001, mu_start=0.92, mu_end=0.97)
            update_best(compute_score(a_restart), a_restart, f"lse-restart-{idx}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Hybrid Fienup + LSE (15% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Hybrid Fienup + LSE ===")

    time_left = phase3_end - time.monotonic()
    if time_left > 10:
        n_iters = min(10000, int(time_left * 1000 / 1.5))
        a_hyb = hybrid_fienup_lse(best_a, n_iters=n_iters, fienup_every=5, lr=0.003)
        update_best(compute_score(a_hyb), a_hyb, "hybrid")

    # Pure Fienup with various params
    if time.monotonic() < phase3_end - 5:
        for thresh, clip in [(1.5, 0.4), (1.3, 0.5), (2.0, 0.3)]:
            a_f = fienup_steps(best_a, n_iters=200, thresh=thresh, clip=clip)
            update_best(compute_score(a_f), a_f, f"fienup-t{thresh}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS polish (20% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n=== Phase 4: L-BFGS polish ===")

    lbfgs_betas = [500, 2000, 5000, 10000, 20000, 50000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase4_end:
            break
        time_left = phase4_end - time.monotonic()
        if time_left < 60:
            break
        maxiter = min(5000, int(time_left * 5))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter)
        update_best(sc, a_opt, f"lbfgs-b{beta}")

    # Second round of L-BFGS with higher betas if we improved
    if best_score < 1.51 and time.monotonic() < phase4_end - 60:
        for beta in [50000, 100000, 200000]:
            if time.monotonic() > phase4_end:
                break
            maxiter = min(3000, int((phase4_end - time.monotonic()) * 3))
            if maxiter < 500:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter)
            update_best(sc, a_opt, f"lbfgs2-b{beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Peak coordinate descent + final LSE (5% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Peak CD + final polish ===")

    if time.monotonic() < phase5_end - 10:
        a_cd = peak_cd(best_a, n_sweeps=5, n_top=30)
        update_best(compute_score(a_cd), a_cd, "peak-cd")

    # Final LSE micro-optimization
    if time.monotonic() < phase5_end - 5:
        time_left = phase5_end - time.monotonic()
        n_iters = min(5000, int(time_left * 1000 / 0.8))
        if n_iters > 500:
            a_final = nesterov_lse_long(best_a, beta_start=10000.0, beta_end=200000.0,
                                        n_iters=n_iters, lr_start=0.003,
                                        lr_end=0.0001, mu_start=0.95, mu_end=0.98)
            update_best(compute_score(a_final), a_final, "final-lse")

    # Final L-BFGS
    if time.monotonic() < phase5_end - 30:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=2000)
        update_best(sc, a_opt, "final-lbfgs")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
