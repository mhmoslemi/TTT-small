def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_wild(a, n_iters=300, gamma=1.0, top_frac=0.3, mode='wild', beta=1.0):
    """Fienup with wild magicking for non-negativity."""
    n = len(a)

        S = 1.0

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        n_bins = len(P)
        
        # Adaptive top fraction
        tf = top_frac * (0.5 + 0.5 * progress)
        thresh = np.quantile(P, 1.0 - tf)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():

            scale[above] = ratio ** (0.5 * gamma)

            # a_new = |a_complex| * sign(a_complex) weighted by beta

def lse_obj_grad(w, beta, mode='squared'):
    """LSE-smoothed objective. If mode='squared', a = w^2."""
    if mode == 'squared':
        a = w * w
    else:
        a = np.maximum(w, 0.0)
    
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros_like(w)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros_like(w)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via chain rule
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad_a = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    
    if mode == 'squared':
        grad_w = 2.0 * w * grad_a
    else:
        grad_w = np.where(w > 0, grad_a, 0.0)
    
    return score, grad_w

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    n = len(a)

        for k in peak_idx:
            k = int(k)
            # The Fourier transform of the impulse at lag k has specific structure
            # We notch the bins that have high amplitude AND are "aligned" with the peak
            freq_idx = (k * np.arange(n_bins)) % M_fft
            freq_idx = freq_idx[freq_idx < n_bins]
            if len(freq_idx) > 0:
                weights = np.exp(-progress * 2.0)  # Stronger notch early
                scale[freq_idx] *= (1.0 - 0.3 * weights)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_hestnes(a, n_iters=5000, top_m=10):
    """Hestenes' accelerated subgradient (heavy ball method)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    a_prev = a.copy()
    v = np.zeros(n)
    momentum = 0.0

    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        # Momentum update (increases linearly then saturates)
        momentum = min(0.7, 0.01 * (it + 1))

        c = fast_conv(a)
        S_val = float(a.sum())
        if S_val < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S_val * S_val)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute subgradient
        if top_m >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            sorted_idx = np.argsort(c)[::-1]
            top_idx = sorted_idx[:top_m]
            # Weight by proximity to max
            peak_vals = c[top_idx]
            weights = (peak_vals / peak_vals[0]) / weights.sum()

        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]

        S2 = S_val * S_val
        S3 = S2 * S_val
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.08)

        v = momentum * v + grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a_prev = a
        a = a_new

    return best_a


def lse_direct_obj_grad(x, beta):
    """LSE-smoothed objective with direct parameterization and projection."""
    n = len(x)
    a = np.maximum(x, 0.0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    # Gradient: sum_k p[k] * 2*a[k-i] with bounds
    # This is 2 * cross_correlation(p, a) computed via FFT
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    # We need sum_j p[m+j] * a[j] for each m
    # = conv reverse: let's compute via the formula
    rev_a = a[::-1]
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)

    # Project: zero out gradient where x <= 0 (for L-BFGS-B compatibility)
    grad = np.where(x > 0, grad, 0.0)
    return score, grad


def lse_squared_obj_grad(w, beta):
    """LSE-smoothed objective with squared parameterization a = w^2."""
    n = len(w)
    a = w * w
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad_a = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    grad_w = 2.0 * w * grad_a
    return score, grad_w


def run_lbfgs_direct(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS with direct parameterization and bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_direct_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def run_lbfgs_squared(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS with squared parameterization a = w^2."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    w0 = np.sqrt(a0)
    bounds = [(0.0, 31.6)] * n  # w max = sqrt(1000)
    res = minimize(
        lse_squared_obj_grad, w0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = res.x * res.x
    a_out = np.clip(a_out, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def build_initializations(n, rng, parent):
    """Generate diverse initializations including anti-palindromic structures."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Linear ramp (anti-palindromic)
    inits.append(0.1 + 0.9 * x)
    # Inverse ramp
    inits.append(0.1 + 0.9 * (1.0 - x))
    # Asymmetric bump
    inits.append(np.maximum(0.1 + 0.8 * np.exp(-8.0 * (x - 0.3) ** 2), 0.01))
    # Two-peak asymmetric
    inits.append(np.maximum(0.1 + 0.6 * np.exp(-10.0 * (x - 0.25) ** 2) + 0.4 * np.exp(-10.0 * (x - 0.75) ** 2), 0.01))
    # Oscillatory with ramp envelope
    inits.append(np.maximum(0.1 * x * (1.0 + 0.5 * np.cos(20.0 * np.pi * x)), 0.01))
    # Cosine window
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (x - 0.5)), 0.01))
    # Random
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 50.0))
    # Periodic with anti-palindromic bias
    inits.append(np.maximum(0.2 + 0.5 * x * (1.0 + 0.3 * np.cos(30.0 * np.pi * x)), 0.01))

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        n = len(parent)
        print(f"Parent: n={n}, score={sc:.8f}")
        best_a = parent.copy()
        update_best(sc, parent, "PARENT")
    else:
        n = 6962

    print(f"Starting with n={n}")

    # ==================================================================
    # Phase 0: Quick Fienup from parent (10% time)
    # ==================================================================
    phase0_end = t_start + 0.10 * (budget_s - 20.0)
    print(f"\n=== Phase 0: Quick Fienup (until t={phase0_end-t_start:.0f}s) ===")

    a_norm = best_a / best_a.sum()

    # Wild magicking
    a_w = fienup_wild(a_norm, n_iters=200, gamma=0.7, top_frac_init=0.1,
                      top_frac_end=0.4, mode='wild', blend_alpha=0.7)
    update_best(compute_score(a_w), a_w, "P0-WILD")

    # Hard clip
    if time.monotonic() < phase0_end:
        a_h = fienup_wild(a_norm, n_iters=200, gamma=0.8, top_frac_init=0.15,
                          top_frac_end=0.5, mode='hard')
        update_best(compute_score(a_h), a_h, "P0-HARD")

    # Notch
    if time.monotonic() < phase0_end:
        a_n = fienup_notch(a_norm, n_iters=150, n_notches=3, notch_width=0.03)
        update_best(compute_score(a_n), a_n, "P0-NOTCH")

    print(f"After Phase 0: best = {best_score:.8f}")

    # ==================================================================
    # Phase 1: Diverse initializations + Fienup (25% time)
    # ==================================================================
    phase1_end = t_start + 0.35 * (budget_s - 20.0)
    print(f"\n=== Phase 1: Diverse inits (until t={phase1_end-t_start:.0f}s) ===")

    inits = build_initializations(n, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            continue
        a0 = a0 / a0.sum()

        # Wild Fienup
        a_f = fienup_wild(a0, n_iters=150, gamma=0.7, top_frac_init=0.1,
                          top_frac_end=0.45, mode='wild', blend_alpha=0.6)
        update_best(compute_score(a_f), a_f, f"P1-WILD-{i}")

        # Notch Fienup (only for some inits)
        if i % 3 == 0 and time.monotonic() < phase1_end:
            a_nt = fienup_notch(a_f, n_iters=100, n_notches=5)
            update_best(compute_score(a_nt), a_nt, f"P1-NOTCH-{i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Accelerated subgradient (25% time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (budget_s - 20.0)
    print(f"\n=== Phase 2: Subgradient (until t={phase2_end-t_start:.0f}s) ===")

    a_cur = best_a.copy() / best_a.sum()
    for top_m in [3, 5, 10, 20]:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        n_iters = int(remaining * 8)  # estimate iterations
        n_iters = max(1000, min(n_iters, 8000))
        a_sg = subgradient_hestnes(a_cur, n_iters=n_iters, top_m=top_m)
        improved = update_best(compute_score(a_sg), a_sg, f"P2-SG-m{top_m}")
        if improved:
            a_cur = a_sg.copy() / a_sg.sum()

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% time)
    # ==================================================================
    phase3_end = t_start + 0.85 * (budget_s - 20.0)
    print(f"\n=== Phase 3: LBFGS (until t={phase3_end-t_start:.0f}s) ===")

    # Direct L-BFGS with increasing beta
    for beta in [100, 500, 2000]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        maxiter = min(3000, int(remaining * 3))
        sc, a_opt = run_lbfgs_direct(best_a, beta, maxiter=maxiter)
        if update_best(sc, a_opt, f"P3-DIRECT b={beta}"):
            pass

    # Squared-parameterization L-BFGS
    for beta in [200, 1000]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        maxiter = min(2000, int(remaining * 2))
        sc, a_opt = run_lbfgs_squared(best_a, beta, maxiter=maxiter)
        if update_best(sc, a_opt, f"P3-SQ b={beta}"):
            pass

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polishing (10% time)
    # ==================================================================
    print(f"\n=== Phase 4: Final polishing ===")

    # Chain: Fienup -> Subgradient -> Fienup
    if time.monotonic() < deadline - 90:
        a_p = fienup_wild(best_a, n_iters=150, gamma=0.6, top_frac_init=0.1,
                          top_frac_end=0.35, mode='wild', blend_alpha=0.8)
        update_best(compute_score(a_p), a_p, "P4-FIENUP")

    if time.monotonic() < deadline - 60:
        a_p = subgradient_hestnes(best_a, n_iters=int((deadline - time.monotonic() - 30) * 6), top_m=5)
        update_best(compute_score(a_p), a_p, "P4-SG")

    # One more L-BFGS with high beta if time allows
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs_direct(best_a, 5000, maxiter=2000)
        update_best(sc, a_opt, "P4-LBFGS-HIGH")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
