def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def constraint_clip(a, M, n_sweeps=200, rng=None, recompute_every=20):
    """Maximize sum(a) subject to conv(a,a)[k] <= M, a >= 0, a <= 1000."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    c = fast_conv(a)
    if rng is None:
        rng = np.random.default_rng(0)
    
    for sweep in range(n_sweeps):
        if sweep > 0 and sweep % recompute_every == 0:
            c = fast_conv(a)  # Recompute to avoid drift
        
        changed = False
        order = rng.permutation(n)
        for j in order:
            aj = a[j]
            c_slice = c[j:j+n]
            
            # rest = c_slice - 2*aj*a (for k != 2j positions)
            # For k = 2j (position j in slice): rest = c[2j] - aj^2
            rest = c_slice - 2.0 * aj * a
            rest[j] = c[2*j] - aj * aj  # self-term
            
            # Compute bounds
            idx = np.arange(n)
            bounds = np.full(n, 1000.0)

            # Quadratic constraint (k = 2j)

            else:
                bounds[j] = 0.0

def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
