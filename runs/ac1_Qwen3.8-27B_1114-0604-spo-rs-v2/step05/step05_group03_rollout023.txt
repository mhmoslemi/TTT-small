    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def spectral_init(n, n_iters=300, seed=0):
    """Alternating projections: flat magnitude spectrum + non-negativity constraint."""
    rng = np.random.default_rng(seed)
    M_fft = 1 << (2 * n - 1).bit_length()

    # Start with a smooth initial guess
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.cos(2.0 * np.pi * x)
    a = np.maximum(a, 0.01)
    a = a / a.sum()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        # Target flat magnitude = sqrt(n) (so that sum of |A|^2 = n*n, matching sum(a)^2 = 1 via Parseval)
        # Actually by Parseval: sum(|A[k]|^2) / M_fft = sum(a^2) for the padded version
        # We want conv(a,a) flat, which means |A|^2 flat (approximately)
        target_mag = np.sqrt(n)
        # Keep DC component
        A_new = A.copy()
        for i in range(len(A)):
            if mag[i] > 1e-15:
                A_new[i] = A[i] * (target_mag / mag[i])
        A_new[0] = 1.0  # sum(a) = 1
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

    return a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x))
    # Slightly U-shaped (higher at boundaries)
    inits.append(0.3 + 0.4 * (2.0 * x - 1.0) ** 2)
    # More U-shaped
    inits.append(0.2 + 0.6 * (2.0 * x - 1.0) ** 4)
    # Spectral init
    inits.append(spectral_init(n, n_iters=100, seed=42 + int(n % 100)))
    return inits


def local_peak_search(a, n_steps=300, rng=None):
    """Targeted local search reducing the convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Subgradient direction for reducing peak at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.05, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3)
        improved = False
        for mult in [0.2, 0.4, 0.6, 1.0, 1.5, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 50 == 0:
            # Randomize: pick a different high-conv position
            top_k = max(5, len(c) // 20)
            candidates = np.argsort(c)[::-1][:top_k]
            k_star = candidates[rng.integers(len(candidates))]
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            direction = np.zeros(n)
            if j_end >= j_start:
                direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
            grad = 2.0 * n * (direction / (S * S) - 2.0 * float(c[k_star]) / (S * S * S))
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                step = 0.02
                a_try = a - step * grad
                a_try = np.clip(a_try, 0.0, 1000.0)
                St = float(a_try.sum())
                if St > 1e-12:
                    a_try = a_try / St
                    sc_try = compute_score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try

    return best_a


def frequency_domain_flatten(a, n_iters=100, alpha=0.8):
    """Frequency-domain error reduction to flatten the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-18:
            break

        # Reduce excess above threshold
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-18:
            break

        conv_mod = conv.copy()
        conv_mod -= 0.7 * excess
        # Zero-pad back
        err = conv_mod[:2 * n - 1] - conv[:2 * n - 1]
        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad)

        # Newton step: dA = err_spec / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = err_spec / (2.0 * A)
        dA[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * dA
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: spectral init, LP alternating, frequency flatten, L-BFGS, local search.
    """
    rng = np.random.default_rng(123)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)", flush=True)
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}", flush=True)
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}", flush=True)
        report(ps, parent, "parent")

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [6962, 7500, 8192, 10000, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to explore: {lengths}", flush=True)

    # ============================================================
    # Phase 1: Initialization & Quick Screening (10% time)
    # ============================================================
    phase1_deadline = t_start + 0.10 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Initialization & Screening ---", flush=True)

    # For each length, try a few initializations with brief LP
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 15:
            break
        print(f"  Screening n={n}, remaining={remaining:.0f}s", flush=True)

        inits = build_initializations(n, rng, parent)
        max_inits = min(len(inits), max(3, int(remaining / 10.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 2:
                break
            a0 = np.maximum(a0, 0.0)
            S0 = float(a0.sum())
            if S0 < 1e-12:
                continue
            a0 = a0 / S0

            # Brief LP alternating (3 outer iters, 5000 FISTA iters)
            a_out, sc_out = lp_alternating(a0, max_outer=3, fista_iters=5000,
                                           time_limit=phase1_deadline - 2, beta_end=100.0)
            report(sc_out, a_out, f"P1 n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}", flush=True)

    # ============================================================
    # Phase 2: Main LP Alternating (50% time)
    # ============================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Main LP Alternating ---", flush=True)

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive FISTA budgets
    fista_budgets = [10000, 15000, 20000, 30000, 40000]
    beta_schedules = [100.0, 200.0, 300.0, 500.0, 700.0]

    for fi in range(len(fista_budgets)):
        if time.monotonic() > phase2_deadline - 20:
            break
        remaining = phase2_deadline - time.monotonic()
        # Estimate time per outer iteration
        est_per_iter = fista_budgets[fi] * (n_main + 1) * 3e-7
        max_outer = max(3, int(remaining / max(1.0, est_per_iter * 1.1)))
        max_outer = min(max_outer, 100)
        print(f"  LP round {fi}: n={n_main}, fista={fista_budgets[fi]}, beta={beta_schedules[fi]}, "
              f"max_outer={max_outer}, remaining={remaining:.0f}s", flush=True)

        a_out, sc_out = lp_alternating(
            a_work, max_outer_iters=max_outer, fista_iters=fista_budgets[fi],
            time_limit=phase2_deadline - 10, beta_end=beta_schedules[fi]
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"P2 r{fi}")
        a_work = a_out.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    # Try other lengths if time permits
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_deadline - 40:
            break
        print(f"  LP alt length: n={n}", flush=True)
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating(
            a_test, max_outer=15, fista_iters=15000,
            time_limit=phase2_deadline - 5, beta_end=300.0
        )
        report(sc_out, a_out, f"P2 n={n}")

    print(f"After Phase 2: best = {best_score:.8f}", flush=True)

    # ============================================================
    # Phase 3: Frequency Domain Flattening (10% time)
    # ============================================================
    phase3_deadline = t_start + 0.70 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: Frequency Domain Flattening ---", flush=True)

    a_flat = frequency_domain_flatten(best_a, n_iters=200, alpha=0.7)
    sc = compute_score(a_flat)
    report(sc, a_flat, "flatten")

    a_flat2 = frequency_domain_flatten(best_a, n_iters=200, alpha=0.9)
    sc = compute_score(a_flat2)
    report(sc, a_flat2, "flatten2")

    # Combine best flatten with LP result
    if time.monotonic() < phase3_deadline - 10:
        a_mix = 0.5 * best_a + 0.5 * a_flat
        a_mix = np.maximum(a_mix, 0.0)
        S = a_mix.sum()
        if S > 1e-12:
            a_mix = a_mix / S
            # Brief LP polish on the mix
            a_out, sc_out = lp_alternating(a_mix, max_outer=5, fista_iters=10000,
                                           time_limit=phase3_deadline - 5, beta_end=200.0)
            report(sc_out, a_out, "mix+LP")

    print(f"After Phase 3: best = {best_score:.8f}", flush=True)

    # ============================================================
    # Phase 4: L-BFGS Refinement (15% time)
    # ============================================================
    phase4_deadline = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 4: L-BFGS Refinement ---", flush=True)

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [50, 150, 400, 1000, 2500, 5000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=5000, maxfun=150000)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s", flush=True)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}", flush=True)

    # ============================================================
    # Phase 5: Local Search & Final Polish (remaining time)
    # ============================================================
    print(f"\n--- Phase 5: Local Search & Final Polish ---", flush=True)

    # Peak-specific local search
    a_peak = local_peak_search(best_a, n_steps=500, rng=rng)
    sc = compute_score(a_peak)
    report(sc, a_peak, "peak search")

    # Another LP round on the current best
    if time.monotonic() < deadline - 120:
        print("  Final LP polish...", flush=True)
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating(
            a_final_in, max_outer=20, fista_iters=30000,
            time_limit=deadline - 60, beta_end=500.0
        )
        report(sc_final, a_final, "final LP")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 40:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 10000, maxiter=8000, maxfun=300000)
        report(sc, a_opt, "final LBFGS b=10000")

    # Final peak search
    if time.monotonic() < deadline - 15:
        a_final = local_peak_search(best_a, n_steps=500, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final peak")

    # One more frequency flatten + LP
    if time.monotonic() < deadline - 30:
        a_f = frequency_domain_flatten(best_a, n_iters=100, alpha=0.85)
        a_out, sc_out = lp_alternating(a_f, max_outer=5, fista_iters=10000,
                                       time_limit=deadline - 15, beta_end=300.0)
        report(sc_out, a_out, "final flatten+LP")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return out
