    cmax = float(c_act.max())
    if cmax <= 0 or not np.isfinite(cmax):
        return 0.0, np.zeros(n)

    z = beta * (c_act - cmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se

    w_full = np.zeros(M)
    w_full[idx] = w_act

    N_fft = _next_pow2(M)
    A_fft = np.fft.rfft(a, N_fft)
    W_fft = np.fft.rfft(w_full, N_fft)
    corr = np.fft.irfft(W_fft * np.conj(A_fft), N_fft)[0:n]
    corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)

    grad = 2.0 * corr
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    L = cmax + np.log(se) / beta
    return float(L), grad


def _adam_log_descent(a0, beta_start, beta_end, k_start, k_end, budget, lr=0.01, epsilon=1e-8):
    """
    Adam optimizer in log-domain with adaptive beta and k.
    a = softmax(x) ensures positivity and sum=1.
    """
    a = _norm(a0)
    n = len(a)
    # Initialize log-domain: x = log(a + eps)
    x = np.log(a + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999

    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    t_end = t_start + budget

    plateau_count = 0
    plateau_threshold = 50
    total_iters = 0

    while time.monotonic() < t_end - 1.0:
        total_iters += 1

        # Adaptive beta and k based on progress
        progress = min(1.0, total_iters / max(1, int(budget / 0.05)))
        beta = beta_start + (beta_end - beta_start) * progress
        k = int(k_start + (k_end - k_start) * progress)
        k = max(1, min(k, 2 * n - 1))

        # Compute a from x (softmax for normalization)
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        S = float(e.sum())
        a = e / S

        # Compute gradient
        val, grad_a = _compute_smoothed_max_grad(a, k, beta)

        # Chain rule: d(val)/dx = a * grad_a
        grad_x = a * grad_a
        grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # Score check (every few iterations to save time)
        if total_iters % 5 == 0 or time.monotonic() > t_end - 5.0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
                plateau_count = 0
            else:
                plateau_count += 1

        # Plateau handling: perturb x
        if plateau_count >= plateau_threshold:
            x = x + np.random.standard_normal(n) * 0.05
            plateau_count = 0

    return best_sc, best_a


def _direct_gradient_descent(a0, budget, k_start=3, k_end=200, beta_start=20.0, beta_end=5000.0):
    """
    Direct gradient descent in a-space with backtracking line search.
    """
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    t_end = t_start + budget
    total_iters = 0
    plateau_count = 0

    while time.monotonic() < t_end - 1.0:
        total_iters += 1

        progress = min(1.0, total_iters / max(1, int(budget / 0.03)))
        beta = beta_start + (beta_end - beta_start) * progress
        k = max(1, int(k_start + (k_end - k_start) * progress))

        val, grad = _compute_smoothed_max_grad(a, k, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search
        lr = min(0.1, 2.0 * val / (gnorm * gnorm + 1e-12))
        improved = False
        for trial in range(10):
            a_try = a - lr * grad
            a_try = np.clip(a_try, 0.0, None)
            St = float(a_try.sum())
            if St < 1e-8:
                lr *= 0.5
                continue
            a_try = a_try / St
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                improved = True
                break
            lr *= 0.5

        if not improved:
            plateau_count += 1
            if plateau_count > 30:
                # Perturb
                a = a + np.random.standard_normal(n) * 0.02 * a
                a = _norm(a)
                plateau_count = 0
        else:
            plateau_count = 0

    return best_sc, best_a


def _subgrad_steps(a, max_iters=20, deadline=None):
    """Exact subgradient descent on the non-smooth max-convolution objective."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)

    for _ in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        c = _self_conv(a)
        dmax = float(c.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.01% of max
        thresh = dmax - 1e-4 * max(dmax, 1e-10)
        active = c >= thresh
        if not active.any():
            active[np.argmax(c)] = True
        n_act = int(active.sum())
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act

        # Gradient
        M = 2 * len(a) - 1
        N_fft = _next_pow2(M)
        A_fft = np.fft.rfft(a, N_fft)
        W_fft = np.fft.rfft(w, N_fft)
        corr = np.fft.irfft(W_fft * np.conj(A_fft), N_fft)[0:len(a)]
        corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)
        grad = 2.0 * corr

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Try multiple step sizes
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0001, 0.0005, 0.002, 0.008, 0.02, 0.05, 0.1, 0.2):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        a = best_step
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.2, 1.2)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _symmetric_init(base, n):
    """Create a symmetric (palindromic) sequence from base."""
    a = _upsample(base, n)
    half = n // 2
    a_sym = np.zeros(n)
    a_sym[:half + 1] = a[:half + 1]
    a_sym[half + 1:] = a_sym[half - 1::-1] if half > 0 else a_sym[:half]
    if n % 2 == 0:
        a_sym[half:] = a_sym[half - 1::-1]
    else:
        a_sym[half + 1:] = a_sym[half - 1::-1]
    return _norm(a_sym)


def _refine_candidate(a0, budget, rng, label=""):
    """
    Multi-stage refinement: Adam log-domain -> direct GD -> subgradient polish.
    """
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    t_end = t_start + budget

    print(f"    [{label}] start: score={best_sc:.8f}, n={n}, budget={budget:.0f}s")

    # Stage 1: Adam in log-domain (60% of budget)
    b1 = budget * 0.5
    if b1 > 5.0:
        sc1, a1 = _adam_log_descent(a, 20.0, 2000.0, 5, 100, b1, lr=0.005)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1.copy()
        a = _norm(a1)
        print(f"    [{label}] after Adam: {best_sc:.8f}")

    # Stage 2: Direct gradient descent (25% of budget)
    b2 = budget * 0.25
    if b2 > 5.0 and time.monotonic() < t_end - 10.0:
        sc2, a2 = _direct_gradient_descent(a, b2, k_start=10, k_end=200,
                                           beta_start=50.0, beta_end=5000.0)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
        a = _norm(a2)
        print(f"    [{label}] after GD: {best_sc:.8f}")

    # Stage 3: More Adam with higher lr for fine-tuning
    remaining = t_end - 10.0 - time.monotonic()
    if remaining > 10.0:
        sc3, a3 = _adam_log_descent(a, 5000.0, 50000.0, 50, 200, remaining * 0.6, lr=0.002)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3.copy()
        a = _norm(a3)
        print(f"    [{label}] after fine Adam: {best_sc:.8f}")

    # Stage 4: Subgradient polish
    remaining = t_end - time.monotonic()
    if remaining > 5.0:
        sc4, a4 = _subgrad_steps(a, max_iters=30, deadline=t_end)
        if sc4 < best_sc:
            best_sc = sc4
            best_a = a4.copy()
        print(f"    [{label}] after subgrad: {best_sc:.8f}")

    print(f"    [{label}] done: score={best_sc:.8f}, elapsed={time.monotonic()-t_start:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"Parent length: {len(parent)}, parent score: {_score(parent):.8f}")

    # Determine candidate lengths
    if parent is not None:
        pn = len(parent)
        lengths = sorted(set([max(1024, pn // 2), pn, max(1024, pn * 3 // 2)]))
    else:
        lengths = [4096, 8192, 12288]

    # Generate initial candidates
    candidates = []

    for target_n in lengths:
        # From parent
        if parent is not None:
            base_p = _upsample(parent, target_n)
            candidates.append((f"parent_n{target_n}", base_p, _score(base_p)))

            # Symmetric version
            sym_p = _symmetric_init(parent, target_n)
            candidates.append((f"sym_n{target_n}", sym_p, _score(sym_p)))

            # Perturbed
            for amp in (0.05, 0.15, 0.3):
                pert = _smooth_perturb(base_p, rng, amp)
                candidates.append((f"pert{amp}_n{target_n}", pert, _score(pert)))

        # Spectral inits
        for i in range(2):
            spec = _spectral_init(target_n, rng)
            candidates.append((f"spec{i}_n{target_n}", spec, _score(spec)))

    # Rank candidates
    candidates.sort(key=lambda x: x[2])
    print(f"\nGenerated {len(candidates)} candidates. Top 5:")
    for name, _, sc in candidates[:5]:
        print(f"  {name}: {sc:.8f}")

    # Select top diverse candidates (limit to 4 to have time)
    selected = []
    for name, arr, sc in candidates:
        if len(selected) >= 4:
            break
        selected.append((name, arr, sc))

    best_score = selected[0][2] if selected else 2.0
    best_seq = selected[0][1].copy() if selected else _norm(np.ones(4096))

    # Main optimization: allocate time proportionally
    main_deadline = deadline - 60.0  # Reserve 60s for final polish
    n_sel = len(selected)
    # Give more time to better candidates
    fractions = []
    if n_sel == 1:
        fractions = [1.0]
    elif n_sel == 2:
        fractions = [0.55, 0.45]
    elif n_sel == 3:
        fractions = [0.45, 0.35, 0.20]
    else:
        fractions = [0.35, 0.30, 0.20, 0.15]

    elapsed = time.monotonic() - t_start
    total_budget = main_deadline - t_start
    print(f"\nMain optimization: {n_sel} candidates, total budget={total_budget:.0f}s")

    for i, (name, arr, sc_init) in enumerate(selected):
        sub_budget = fractions[i] * total_budget
        sub_deadline = min(time.monotonic() + sub_budget, main_deadline - 10.0)

        if time.monotonic() > sub_deadline:
            continue

        print(f"\n  --- Candidate {i+1}/{n_sel}: '{name}', init_score={sc_init:.8f}, budget={sub_budget:.0f}s ---")
        sc, seq = _refine_candidate(arr, sub_budget, rng, label=name)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"  *** Global best: {best_score:.8f} ---")

    # Final polish on the best
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0
    remaining = final_deadline - time.monotonic()

    if remaining > 30.0:
        # High-precision Adam
        b_final = remaining * 0.5
        sc_f, a_f = _adam_log_descent(best_seq, 10000.0, 100000.0, 100, 500, b_final, lr=0.001)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final Adam: {best_score:.8f}")
        best_seq = _norm(a_f)

    if time.monotonic() < final_deadline - 10.0:
        sc_sg, a_sg = _subgrad_steps(best_seq, max_iters=30, deadline=final_deadline - 5.0)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Random restart perturbation if time allows
    if time.monotonic() < final_deadline - 30.0:
        for restart in range(2):
            if time.monotonic() > final_deadline - 15.0:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05)
            chunk_budget = min(20.0, final_deadline - 15.0 - time.monotonic())
            if chunk_budget > 5.0:
                sc_r, a_r = _adam_log_descent(p, 5000.0, 50000.0, 50, 200, chunk_budget, lr=0.003)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  Restart {restart}: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
