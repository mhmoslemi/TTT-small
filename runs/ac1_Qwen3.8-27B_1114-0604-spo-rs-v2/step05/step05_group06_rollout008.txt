def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]

def _cross_corr(a, w):
    """Compute gradient contribution: grad[i] = sum_k w[k]*a[k-i] for i=0..n-1."""

def _upsample(a, n):

def _lse_obj_grad(a, beta, k):
    """LSE-smoothed 2*max(d) using top-k positions. Returns (value, gradient)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    S = w.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return w / S

def _spectral_init(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig

                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k}, beta={beta:.0f}, {method}, call{ci}: {best_sc:.8f}")
            a = _norm(a_new)

    # Subgradient polish
    if time.monotonic() < deadline - 10.0:
        sc, a_sg = _subgrad_polish(a, max_iters=20, deadline=deadline - 5.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    # Peak shaving
    if time.monotonic() < deadline - 5.0:
        sc, a_ps = _peak_shave(a, max_iters=20, deadline=deadline - 3.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_ps.copy()
            print(f"    [{label}] peak_shave: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _combine_optimize(a1, a2, deadline, label, rng):
    """Combine two sequences and optimize the result."""
    best_a = _norm((a1 + a2) / 2.0)
    best_sc = _score(best_a)
    n = len(best_a)
    print(f"  [{label}] combined: score={best_sc:.8f}")

    # Quick optimization
    for k, beta in [(20, 500), (80, 2000), (200, 8000)]:
        if time.monotonic() > deadline:
            break
        sc, a_new = _run_lbfgs(best_a, beta, k, min(30.0, deadline - time.monotonic() - 5.0), 300)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    [{label}] K={k}: {best_sc:.8f}")

    sc, a_sg = _subgrad_polish(best_a, max_iters=10, deadline=deadline)
    if sc < best_sc:
        best_sc = sc
        best_a = a_sg.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: progressive LSE + momentum + peak shaving."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    # Determine target length: use parent's native length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192

    print(f"Parent: {parent is not None}, native_n={target_n if parent is not None else 'N/A'}, budget={budget_s}s")

    # --- Generate initializations ---
    inits = []

    if parent is not None:
        inits.append(('parent', _norm(parent)))
        inits.append(('parent_rev', _norm(parent[::-1])))
        for amp in (0.03, 0.08, 0.15, 0.25):
            inits.append((f'pn{amp}', _smooth_perturb(parent, rng, amp)))

    for i in range(6):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3, 5):
        for amp in (0.1, 0.2, 0.4):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'cos{f}_{amp}', a / a.sum()))

    for i in range(4):
        noise = rng.standard_normal(target_n)
        k = max(5, target_n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * max(1, k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'noise{i}', a / a.sum()))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top 5: {[f'{s:.6f}({nm})' for s, nm, _ in scored[:5]]}")

    # Select top candidates (ensuring diversity)
    candidates = []
    used_names = set()
    for sc, name, arr in scored:
        if len(candidates) >= 4:
            break
        # Ensure some diversity: skip if too similar to already selected
        if candidates and any(
            np.corrcoef(arr, ca)[0, 1] > 0.999 for _, _, ca in candidates
        ):
            continue
        candidates.append((sc, name, arr))
        used_names.add(name)

    if not candidates:
        candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()

    # --- Main optimization phase ---
    # Allocate time: 60% to top candidate, 20% to second, 10% to third, 10% to fourth
    main_deadline = deadline - 300.0  # reserve 300s for final polish
    n_cands = len(candidates)

    if n_cands == 1:
        fractions = [1.0]
    elif n_cands == 2:
        fractions = [0.65, 0.35]
    elif n_cands == 3:
        fractions = [0.55, 0.25, 0.20]
    else:
        fractions = [0.45, 0.25, 0.18, 0.12]

    results = []  # (score, seq) for combination

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        sub_deadline = t_start + t_start * 0 + (sum(fractions[:ci + 1]) * (main_deadline - t_start))
        sub_deadline = min(sub_deadline, main_deadline)

        if time.monotonic() > sub_deadline - 10:
            continue

        n = len(arr_init)
        print(f"\n=== Candidate {ci + 1}: '{name}', n={n}, score={sc_init:.8f} ===")

        a_work = _norm(arr_init)

        # Multilevel warm start for large n
        if n > 4096 and time.monotonic() < sub_deadline - 60.0:
            n_small = max(1024, n // 4)
            a_small = _upsample(a_work, n_small)
            sc_small = _score(a_small)
            print(f"  Multilevel: n_small={n_small}, score={sc_small:.8f}")

            time_per_stage = min(15.0, (sub_deadline - 60.0 - time.monotonic()) / 5.0)
            for k_s, beta_s in [(3, 50), (10, 300), (30, 1500)]:
                if time.monotonic() > sub_deadline - 60.0:
                    break
                sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, time_per_stage, 150)
                if sc_s < sc_small:
                    sc_small = sc_s
                    a_small = a_s

            a_up = _norm(_upsample(a_small, n))
            sc_up = _score(a_up)
            if sc_up < _score(a_work):
                a_work = a_up
                print(f"  Multilevel improved: {sc_up:.8f}")

        # Main trajectory optimization
        sc_traj, seq_traj = _optimize_trajectory(a_work, sub_deadline, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** Global best updated: {best_score:.8f} ***")
        results.append((sc_traj, seq_traj))

    # --- Combination phase ---
    if len(results) >= 2 and time.monotonic() < deadline - 200.0:
        print(f"\n=== Combination phase ===")
        # Try combining top results
        results.sort(key=lambda x: x[0])
        for i in range(min(3, len(results) - 1)):
            for j in range(i + 1, min(i + 3, len(results))):
                if time.monotonic() > deadline - 150.0:
                    break
                sc_c, seq_c = _combine_optimize(
                    results[i][1], results[j][1],
                    min(deadline - 150.0, time.monotonic() + 40.0),
                    f'comb{i}{j}', rng
                )
                if sc_c < best_score:
                    best_score = sc_c
                    best_seq = seq_c.copy()
                    print(f"  *** Combination best: {best_score:.8f} ***")

    # --- Final global polish ---
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 15.0

    # High-beta LSE
    for k_final, beta_final in [(100, 3000), (300, 10000), (600, 25000), (1000, 50000)]:
        if time.monotonic() > final_deadline - 30:
            break
        remaining = final_deadline - time.monotonic() - 30
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_final, k_final, remaining * 0.5, 500)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_final}, beta={beta_final}: {best_score:.8f}")

    # Momentum-based high-smoothing
    if time.monotonic() < final_deadline - 30:
        sc_m, a_m = _momentum_optimize(best_seq, 30000, 500, 100, final_deadline - 30)
        if sc_m < best_score:
            best_score = sc_m
            best_seq = a_m.copy()
            print(f"  Final momentum: {best_score:.8f}")

    # Subgradient polish (multiple rounds)
    for round_num in range(3):
        if time.monotonic() > final_deadline - 15:
            break
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=20, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad round{round_num}: {best_score:.8f}")
        else:
            break

    # Peak shaving
    if time.monotonic() < final_deadline - 10:
        sc_ps, a_ps = _peak_shave(best_seq, max_iters=20, deadline=final_deadline - 5)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
            print(f"  Final peak_shave: {best_score:.8f}")

    # Random restart perturbation
    if time.monotonic() < final_deadline - 40:
        for restart in range(5):
            if time.monotonic() > final_deadline - 20:
                break
            strength = 0.02 * (restart + 1)
            p = _smooth_perturb(best_seq, rng, strength)
            sc_r, a_r = _subgrad_polish(p, max_iters=15, deadline=min(final_deadline - 20, time.monotonic() + 20))
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart} (str={strength}): {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=15, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
