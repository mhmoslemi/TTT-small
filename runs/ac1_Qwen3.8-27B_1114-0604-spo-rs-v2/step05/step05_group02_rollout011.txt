def fast_conv(a):

def compute_score(a):

        # Anneal step size

        sc = 2.0 * n * M_val / (S * S)

            best_sc = sc
            best_a = a.copy()

        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)

        S3 = S2 * S

        if gn < 1e-15:
            break

        gap = sc - lb
        if gap < 1e-12:
            step = 0.001
        else:
            step = min(step, gap / (gn * gn + 1e-30))
        step = min(step, 0.03)

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        step *= 0.9997

def gs_project(a, n_iters=300, alpha=0.5, M=None, deadline=None):

    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S

    best_sc = compute_score(a)

                a_trial = np.maximum(a_trial, 0)
                S_t = a_trial.sum()
                if S_t < 1e-12:
                    continue
                a_trial = a_trial / S_t
                sc_t = compute_score(a_trial)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) * np.exp(-2.0 * (2*x-1)**2), 0.01))
    inits.append(0.1 + 0.9 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.7) ** 2))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses mirror descent, peak-targeted subgradient, multi-scale L-BFGS,
    perturbation escape, and block-swap optimization.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent sequence, length={len(parent)}")
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [6962, 6144, 8192, 5120, 7680]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: GS initialization at best length (15% of time)
    # ==================================================================
    phase1_end = t_start + 0.15 * (total_budget - 40.0)
    print(f"\n=== Phase 1: GS initialization (deadline +{phase1_end - t_start:.0f}s) ===")

    # Focus on n=6962 (or parent length)
    n_main = len(parent) if parent is not None else 6962

    inits = build_diverse_inits(n_main, rng, parent)
    for i, a0 in enumerate(inits[:6]):
        if time.monotonic() > phase1_end:
            break
        a_gs = gs_project(a0, n_iters=300, alpha=0.5, deadline=phase1_end)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"GS n={n_main} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Mirror descent with LSE (25% of time)
    # ==================================================================
    phase2_end = t_start + 0.40 * (total_budget - 40.0)
    print(f"\n=== Phase 2: Mirror descent LSE (deadline +{phase2_end - t_start:.0f}s) ===")

    # Multiple runs with different parameters
    md_configs = [
        {'beta_start': 30, 'beta_end': 3000, 'eta_start': 0.15, 'eta_end': 0.003, 'n_iters': 3000},
        {'beta_start': 50, 'beta_end': 10000, 'eta_start': 0.1, 'eta_end': 0.005, 'n_iters': 3000},
        {'beta_start': 20, 'beta_end': 5000, 'eta_start': 0.2, 'eta_end': 0.01, 'n_iters': 2000},
        {'beta_start': 100, 'beta_end': 20000, 'eta_start': 0.05, 'eta_end': 0.002, 'n_iters': 2000},
    ]

    for ci, cfg in enumerate(md_configs):
        if time.monotonic() > phase2_end:
            break
        print(f"  MD config {ci}: beta=[{cfg['beta_start']},{cfg['beta_end']}], eta=[{cfg['eta_start']},{cfg['eta_end']}]")
        a_md = mirror_descent_lse(best_a, deadline=phase2_end, **cfg)
        sc = compute_score(a_md)
        update_best(sc, a_md, f"MD config={ci}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Multi-scale L-BFGS (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Multi-scale L-BFGS (deadline +{phase3_end - t_start:.0f}s) ===")

    # Run multi-scale L-BFGS
    a_ms = multi_scale_lbfgs(best_a,
                             betas=[50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000],
                             maxiter_each=2000, deadline=phase3_end)
    sc = compute_score(a_ms)
    update_best(sc, a_ms, "Multi-scale L-BFGS")

    # Also try from a GS-refined starting point
    if time.monotonic() < phase3_end - 60:
        a_gs2 = gs_project(best_a, n_iters=200, alpha=0.3, deadline=phase3_end)
        a_ms2 = multi_scale_lbfgs(a_gs2,
                                   betas=[200, 500, 1000, 2000, 5000, 10000],
                                   maxiter_each=2000, deadline=phase3_end)
        sc = compute_score(a_ms2)
        update_best(sc, a_ms2, "GS + Multi-scale L-BFGS")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak-targeted subgradient (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 40.0)
    print(f"\n=== Phase 4: Peak-targeted subgradient (deadline +{phase4_end - t_start:.0f}s) ===")

    for top_m in [3, 5, 8, 12, 20]:
        if time.monotonic() > phase4_end:
            break
        a_sg = peak_targeted_subgradient(best_a, n_iters=3000, top_m=top_m, deadline=phase4_end)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Perturbation escape + re-optimization (10% of time)
    # ==================================================================
    phase5_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 5: Perturbation escape (deadline +{phase5_end - t_start:.0f}s) ===")

    a_pert = perturb_and_reoptimize(best_a, n_perturb=15, scale=0.03, deadline=phase5_end)
    sc = compute_score(a_pert)
    update_best(sc, a_pert, "Perturbation escape")

    # Try with larger scale
    if time.monotonic() < phase5_end:
        a_pert2 = perturb_and_reoptimize(best_a, n_perturb=10, scale=0.08, deadline=phase5_end)
        sc = compute_score(a_pert2)
        update_best(sc, a_pert2, "Perturbation escape large")

    print(f"After Phase 5: best = {best_score:.8f}")

    # ==================================================================
    # Phase 6: Block swap + final L-BFGS polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 6: Block swap + Final polish ===")

    if time.monotonic() < deadline - 60:
        a_bs = block_swap_optimize(best_a, n_sweeps=2, deadline=deadline - 60)
        sc = compute_score(a_bs)
        update_best(sc, a_bs, "Block swap")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs_lse(best_a, beta=50000, maxiter=3000, maxfun=50000)
        update_best(sc, a_final, "Final L-BFGS high-beta")

    # One more GS pass
    if time.monotonic() < deadline - 15:
        a_gs_final = gs_project(best_a, n_iters=100, alpha=0.2, deadline=deadline - 10)
        sc = compute_score(a_gs_final)
        update_best(sc, a_gs_final, "Final GS")

    # ==================================================================
    # Try other lengths from the best solution
    # ==================================================================
    if time.monotonic() < deadline - 20:
        n_best = len(best_a)
        for n_alt in lengths:
            if n_alt == n_best or time.monotonic() > deadline - 15:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_alt),
                                  np.linspace(0, 1, n_best), best_a)
            a_interp = np.maximum(a_interp, 0.01)
            a_interp = a_interp / a_interp.sum()
            a_opt = mirror_descent_lse(a_interp, n_iters=500, beta_start=50,
                                        beta_end=2000, eta_start=0.1, eta_end=0.01,
                                        patience=100, deadline=deadline - 15)
            sc = compute_score(a_opt)
            update_best(sc, a_opt, f"Alt length n={n_alt}")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
