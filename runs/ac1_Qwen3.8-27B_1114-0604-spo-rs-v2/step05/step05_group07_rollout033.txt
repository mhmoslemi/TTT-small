"""
Multi-strategy optimizer combining high-precision LSE L-BFGS, peak-directed
anti-mirror subgradient, simulated annealing with targeted moves, and
frequency-domain initialization for minimizing 2n*max(conv)/sum^2.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS callback
_current_beta = 1000.0
_current_n = 0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -60))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # dL/da[i] = 2 * (p * rev_a)[n-1+i]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_step(a, n_steps=500, step_scale=0.01):
    """
    Peak-directed subgradient: move a to reduce the maximum autocorrelation.
    Uses the 'anti-mirror' insight: reduce a[j] where a[k*-j] is large.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        sc = 2.0 * n * M / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of c[k_star] w.r.t. a
        # d(c[k*])/da[j] = a[k*-j] + a[j-k*] (for valid indices)
        sg = np.zeros(n)
        for j in range(n):
            val = 0.0
            idx1 = k_star - j
            if 0 <= idx1 < n:
                val += a[idx1]
            idx2 = j - k_star
            if 0 <= idx2 < n:
                val += a[idx2]
            sg[j] = val

        # Full subgradient of the score
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_scale * (1.0 + it * 0.001)
        step = min(step, 0.5 * (sc - 1.0) / (gnorm + 1e-10))
        step = max(step, 1e-6)

        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def multi_peak_directed(a, n_steps=500, top_k=3, step_scale=0.01):
    """Peak-directed using top-k peaks for smoother gradient."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:top_k]
        total_w = sum(range(1, top_k + 1))
        
        sg = np.zeros(n)
        for rank, k_val in enumerate(top_idx):
            w = (top_k - rank) / total_w
            for j in range(n):
                val = 0.0
                idx1 = k_val - j
                if 0 <= idx1 < n:
                    val += a[idx1]
                idx2 = j - k_val
                if 0 <= idx2 < n:
                    val += a[idx2]
                sg[j] += w * val

        M_max = float(c[top_idx[0]])
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_max / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step_scale * (0.995 ** it)
        step = min(step, 0.3 * (2.0 * n * M_max / S2 - 1.0) / (gnorm + 1e-10))
        step = max(step, 1e-7)

        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def simulated_annealing_peak(a, n_iters=5000, T0=0.05, T1=0.001):
    """Simulated annealing with peak-directed moves."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Peak-directed move: reduce contributors to peak, increase others
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            # Random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            delta = 0.01 * a.mean() * (rng.random() - 0.5)
            a_new = a.copy()
            a_new[i] = max(0, a_new[i] + delta)
            a_new[j] = max(0, a_new[j] - delta)
            S_new = a_new.sum()
            if S_new > 1e-12:
                a_new /= S_new
                sc_new = compute_score(a_new)
                if sc_new < cur_sc or rng.random() < np.exp(-(sc_new - cur_sc) / T):
                    a = a_new
                    cur_sc = sc_new
                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_a = a.copy()
            continue

        # Pick a top contributor to reduce
        top_count = max(3, n // 20)
        top_local = np.argsort(contrib)[::-1][:top_count]
        li = top_local[rng.integers(0, len(top_local))]
        i = j_start + li

        # Pick a bottom contributor to increase
        bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
        lj = bottom_local[rng.integers(0, len(bottom_local))]
        j = j_start + lj

        # Transfer
        transfer = 0.02 * a[i] * (0.5 + rng.random())
        a_new = a.copy()
        a_new[i] = max(0, a_new[i] - transfer)
        a_new[j] += transfer
        S_new = a_new.sum()
        if S_new > 1e-12 and a_new[i] > 0:
            a_new /= S_new
            sc_new = compute_score(a_new)
            if sc_new < cur_sc or rng.random() < np.exp(-(sc_new - cur_sc) / T):
                a = a_new
                cur_sc = sc_new
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a.copy()

    return best_a


def freq_domain_init(n, rng, target_flat=False):
    """Initialize from frequency domain with flat magnitude target."""
    N = 1 << (2 * n - 1).bit_length()
    # Start with random phases
    phases = rng.standard_normal(N // 2 + 1)
    if target_flat:
        mags = np.ones(N // 2 + 1)
        mags[0] = 1.0  # DC
    else:
        mags = np.ones(N // 2 + 1) * (0.8 + 0.4 * rng.random(N // 2 + 1))

    F = mags * np.exp(1j * phases)
    # Make it Hermitian for real signal
    F_full = np.zeros(N, dtype=np.complex128)
    F_full[:N // 2 + 1] = F
    for k in range(1, N // 2):
        F_full[N - k] = np.conj(F[k])
    if N % 2 == 0:
        F_full[N // 2] = float(F[N // 2].real)

    a_full = np.fft.ifft(F_full).real[:n]
    a = np.clip(a_full, 0, 1000.0)
    S = a.sum()
    if S > 1e-12:
        a /= S
    return a


def freq_flatten_iter(a, n_iters=200, rng=None):
    """Iteratively flatten the power spectrum while maintaining a >= 0."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        F = np.fft.rfft(a_pad)
        mag = np.abs(F)
        mag_target = np.sqrt(np.sum(mag**2) / len(mag))

        # Smooth the magnitude toward the target
        alpha = 0.1 * (1.0 - it / n_iters)
        new_mag = (1 - alpha) * mag + alpha * mag_target
        # Keep DC unchanged
        new_mag[0] = mag[0]

        # Reconstruct with new magnitudes, same phases
        phases = np.angle(F)
        F_new = new_mag * np.exp(1j * phases)

        # Make Hermitian
        F_full = np.zeros(N, dtype=np.complex128)
        F_full[:N // 2 + 1] = F_new
        for k in range(1, N // 2):
            F_full[N - k] = np.conj(F_full[k])
        if N % 2 == 0:
            F_full[N // 2] = float(F_full[N // 2].real)

        a_new_full = np.fft.ifft(F_full).real[:n]
        a_new = np.clip(a_new_full, 0, 1000.0)
        S_new = a_new.sum()
        if S_new > 1e-12:
            a_new /= S_new
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new

    return best_a


def block_optimize(a, block_size=32, n_sweeps=5):
    """Optimize within blocks: redistribute energy within each block to reduce local peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            if e - s < 2:
                continue
            block_sum = a[s:e].sum()
            if block_sum < 1e-15:
                continue

            # Within the block, try to make it more uniform
            # (reduces local energy concentration)
            a[s:e] = a[s:e] * 0.7 + (block_sum / (e - s)) * 0.3
            S = a.sum()
            a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def targeted_coord_descent(a, n_sweeps=20, n_top=30):

    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / max(len(bottom_local), 1)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: quick LSE convergence -> peak-directed optimization ->
    simulated annealing -> high-precision LSE polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        n_work = 8192
        parent_norm = np.ones(n_work) / n_work
        report(compute_score(parent_norm), parent_norm, "uniform")

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: Quick LSE convergence (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: Quick LSE L-BFGS (n={n_work}) ---")

    current_a = parent_norm if parent is not None else best_a.copy()

    for beta in [30, 100, 500]:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(20000, max(2000, remaining * 50)))
        maxfun = int(min(80000, max(5000, remaining * 200)))
        sc, a_opt = run_lbfgs(current_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"P1 LBFGS b={beta}")
        current_a = best_a.copy()

    # ==================================================================
    # Phase 2: Peak-directed optimization (25% time)
    # ==================================================================
    phase2_deadline = t_start + 0.40 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: Peak-Directed Optimization ---")

    # Alternating LSE + peak-directed
    for iteration in range(10):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        # Peak-directed step
        a_pd = peak_directed_step(best_a, n_steps=300, step_scale=0.015)
        sc = compute_score(a_pd)
        report(sc, a_pd, f"P2 peak_dir #{iteration}")

        # Multi-peak directed
        a_mp = multi_peak_directed(best_a, n_steps=200, top_k=3, step_scale=0.01)
        sc = compute_score(a_mp)
        report(sc, a_mp, f"P2 multi_peak #{iteration}")

        # LSE refinement
        beta = 500 + 500 * iteration
        maxiter = int(min(15000, max(2000, remaining * 30)))
        maxfun = int(min(50000, max(3000, remaining * 100)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"P2 LBFGS b={beta} #{iteration}")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Simulated annealing + frequency domain (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.60 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: SA + Freq Domain ---")

    # Simulated annealing
    if time.monotonic() < phase3_deadline - 60:
        a_sa = simulated_annealing_peak(best_a, n_iters=3000, T0=0.03, T1=0.001)
        sc = compute_score(a_sa)
        report(sc, a_sa, "P3 SA")

    # Frequency domain initialization and refinement
    if time.monotonic() < phase3_deadline - 60:
        rng2 = np.random.default_rng(seed + 100)
        for trial in range(3):
            a_fd = freq_domain_init(n_work, rng2, target_flat=True)
            a_ff = freq_flatten_iter(a_fd, n_iters=100, rng=rng2)
            sc = compute_score(a_ff)
            report(sc, a_ff, f"P3 freq_{trial}")
            # Refine with LSE
            if time.monotonic() < phase3_deadline - 30:
                sc2, a_opt2 = run_lbfgs(a_ff, 500, maxiter=10000, maxfun=30000)
                report(sc2, a_opt2, f"P3 freq_refine_{trial}")

    # Block optimization
    if time.monotonic() < phase3_deadline - 30:
        a_block = block_optimize(best_a, block_size=64, n_sweeps=3)
        sc = compute_score(a_block)
        report(sc, a_block, "P3 block")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: High-precision LSE (30% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: High-Precision LSE ---")

    beta_schedule = [2000, 5000, 10000, 30000, 100000, 300000]

    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(50000, max(3000, remaining * 40)))
        maxfun = int(min(200000, max(10000, remaining * 150)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"P4 LBFGS b={beta}")

        # Perturbed restart for high beta
        if beta >= 10000 and time.monotonic() < phase4_deadline - 60:
            a_pert = perturbed_restart(best_a, rng, scale=0.005, strategy=0)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"P4 LBFGS b={beta} pert")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final polishing (10% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polishing ---")

    # Targeted coordinate descent
    if time.monotonic() < deadline - 60:
        a_tcd = targeted_coord_descent(best_a, n_sweeps=15, n_top=40)
        sc = compute_score(a_tcd)
        report(sc, a_tcd, "P5 coord_desc")

    # One more peak-directed pass
    if time.monotonic() < deadline - 60:
        a_pd2 = peak_directed_step(best_a, n_steps=500, step_scale=0.005)
        sc = compute_score(a_pd2)
        report(sc, a_pd2, "P5 peak_dir")

    # Final very high beta LSE
    if time.monotonic() < deadline - 30:
        for beta in [500000, 1000000]:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            maxiter = int(min(30000, max(2000, remaining * 30)))
            maxfun = int(min(100000, max(5000, remaining * 100)))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            report(sc, a_opt, f"P5 final b={beta}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
