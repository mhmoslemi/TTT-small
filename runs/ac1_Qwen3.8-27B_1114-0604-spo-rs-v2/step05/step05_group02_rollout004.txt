"""
Multi-beta adaptive LSE L-BFGS with correct diagonal gradient correction,
GS-Fienup refinement, and peak shaving. Improvements over previous:
(1) Correct diagonal term in LSE gradient, (2) much higher beta values,
(3) adaptive beta scheduling with restarts, (4) perturbation-based diversification.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient with correct diagonal correction.
    f(a) = 2n * LSE_beta(conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: subtract max already done via c_max
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta

    score = 2.0 * n * L / (S * S)

    # Cross-correlation: h_m = sum_j p[m+j] * a[j]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_full = np.fft.irfft(RA * P, M2_fft)
    h = conv_full[n - 1:2 * n - 1].copy()

    # Diagonal correction: dc_{2m}/da_m = 2*a_m (not a_m)
    # The cross-correlation already gives p[2m]*a[m], need to add one more p[2m]*a[m]
    h += p[0::2][:n] * a

    # Full gradient: df/da_m = (2n/S^2)*(1/beta)*h_m - 4nL/S^3
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (h / (beta * S2) - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000, ftol=1e-18, gtol=1e-14):
    """Run L-BFGS-B on the LSE-smoothed objective with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def gs_refine(a, n_iters=300, alpha_init=0.5, alpha_final=0.05, M=None):
    """GS-Fienup with correct flat-convolution target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave(a, n_iters=2000, n_peaks=10, step_init=0.02, decay=0.999):
    """Subgradient peak shaving using top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def perturb_sequence(a, rng, scale=0.01):
    """Add small Gaussian perturbation and re-normalize."""
    n = len(a)
    a_pert = a.copy() + rng.standard_normal(n) * scale
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = float(a_pert.sum())
    if S < 1e-6:
        return a.copy()
    return a_pert / S

        a_gs_final = gs_refine(a_alt, n_iters=100, alpha_init=0.3, alpha_final=0.05)
        sc_final = compute_score(a_gs_final)
        update_best(sc_final, a_gs_final, "FINAL GS perturb")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
