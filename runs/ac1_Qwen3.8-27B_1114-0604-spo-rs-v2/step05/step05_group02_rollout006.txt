"""
Adaptive LSE-LBFGS with momentum subgradient and structured restarts for minimizing
max self-convolution of non-negative sequences. Key improvements: geometric beta
annealing schedule, Nesterov momentum subgradient, structured perturbations for
restarts, and optimized time allocation across phases.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score_val = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    gLSE = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (gLSE / S2 - 2.0 * L / S3)
    return score_val, grad


def lse_wrap(x, beta):
    """Wrapper for scipy minimize."""
    return lse_objective_and_grad(x, beta)


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000, m=30):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (score, a_out)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_wrap, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                     'm': m}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def gs_iterate(a, n_iters=300, alpha_init=0.6, alpha_final=0.1):
    """Gerchberg-Saxton with correct flat-convolution target."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def momentum_subgradient(a, n_iters=5000, top_m=10, step_init=0.03,
                         momentum_init=0.9, momentum_growth=1.0001):
    """Subgradient with Nesterov momentum on top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    momentum = momentum_init
    v = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.5 * (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)

        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        momentum = min(momentum * momentum_growth, 0.999)
        step = max(step * 0.9999, 1e-6)

    return best_a


def structured_perturb(a, rng, strength=0.1):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    mode = rng.integers(0, 5)
    if mode == 0:
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        pert = 1.0 + strength * np.cos(2 * np.pi * freq * x + phase)
    elif mode == 1:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        pert = 1.0 + strength * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 2:
        idx = rng.integers(0, n)
        width = max(1, int(rng.integers(10, n // 10)))
        sl = slice(max(0, idx - width), min(n, idx + width))
        pert = np.ones(n)
        pert[sl] *= (1.0 + strength)
    elif mode == 3:
        noise = rng.standard_normal(n) * strength
        pert = 1.0 + 0.1 * noise
    else:
        freq = rng.uniform(2, 20)
        pert = 1.0 + strength * 0.5 * np.sin(2 * np.pi * freq * x)
    a_new = a * pert
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S


def make_initializations(n, rng, parent):

        else:

            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.2 * np.sin(10.0 * np.pi * x) * np.cos(4.0 * np.pi * x), 0.01))

    # Symmetrized versions
    for i in range(min(3, len(inits))):
        sym = inits[i].copy()
        sym = 0.5 * (sym + sym[::-1])
        sym = np.maximum(sym, 0.01)
        inits.append(sym)

    return inits


def adaptive_lse_lbfgs(a0, deadline, rng, tag=""):
    """
    Adaptive LSE with geometric beta increase. Runs L-BFGS at each beta level.
    Restarts with structured perturbation when stuck.
    """
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    if S < 1e-6:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_sc = compute_score(a)
    best_a = a.copy()
    print(f"  [{tag}] Start: score={best_sc:.8f}")

    betas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]
    max_iter_per_beta = [3000, 3000, 3000, 3000, 2500, 2500, 2000, 2000, 1500, 1500, 1000, 1000, 800, 500]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        t_beta_start = time.monotonic()

        # Try up to 2 restarts at this beta
        for restart in range(2):
            if time.monotonic() > deadline:
                break
            sc, a_opt = run_lbfgs_lse(a, beta, maxiter=max_iter_per_beta[i],
                                       maxfun=max_iter_per_beta[i] * 20)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                a = a_opt.copy()
                S = a.sum()
                if S < 1e-6:
                    a = a_opt
                    S = a.sum()
                a = a / S
                print(f"  [{tag}] beta={beta} r{restart}: score={best_sc:.8f} "

            else:

            if time.monotonic() > phase4_end - 10:
                break
            a_pert = structured_perturb(best_a, rng, strength=0.03 + 0.02 * restart_i * 0.1)
            sc_pert = compute_score(a_pert)
            if sc_pert < best_score * 1.02:
                # Perturbation didn't hurt much, try to improve
                sc, a_opt = run_lbfgs_lse(a_pert, 50000, maxiter=1000, maxfun=20000)
                if update_best(sc, a_opt):
                    print(f"  Restart {restart_i}: NEW BEST {sc:.8f}")

    # ==================================================================
    # Phase 5: Final subgradient polish
    # ==================================================================
    if time.monotonic() < deadline - 15:
        a_final = momentum_subgradient(best_a, n_iters=3000, top_m=10,
                                       step_init=0.01, momentum_init=0.9)
        sc = compute_score(a_final)
        if update_best(sc, a_final):
            print(f"  Final SG: NEW BEST {sc:.8f}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start_global
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
