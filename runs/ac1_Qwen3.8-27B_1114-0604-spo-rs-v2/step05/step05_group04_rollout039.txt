    best_a = a.copy()

        B = A * A  # complex square, length N//2+1
        c_full = np.fft.irfft(B, N)
        c = c_full[:2*n-1]  # linear convolution
        
        T = c.max()
        c_mean = S * S / (2*n - 1)

"""
Optimized step function search using corrected Fienup (A^2 spectral shaping),
LSE-smoothed L-BFGS, and subgradient refinement.
Key innovation: uses A^2 (squared complex spectrum) instead of |A|^2 (power spectrum)
for the Fienup iteration, correctly targeting the convolution rather than autocorrelation.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_a2(a, n_iters=500, gamma=0.5, clip_frac=0.05, soft_alpha=30.0):
    """
    Fienup iteration using A^2 (squared spectrum) to target convolution flatness.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        T = float(c.max())
        if T < 1e-15:
            break

        # Create desired c: clip peaks
        thresh = T * (1.0 - clip_frac)
        c_desired = np.minimum(c, thresh)

        # Pad to length N
        c_desired_full = np.zeros(N)
        c_desired_full[:2 * n - 1] = c_desired

        B_desired = np.fft.rfft(c_desired_full)

        # Amplitude update on A
        abs_B = np.abs(B)
        abs_B_des = np.abs(B_desired)
        ratio = np.sqrt(np.maximum(abs_B_des, 1e-30) / np.maximum(abs_B, 1e-30))
        A_new = A * (ratio ** gamma)

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_conv_obj_grad(x, beta):
    """
    LSE-smoothed max of conv(x,x), with gradient.
    Score = 2n * LSE_beta(conv(x,x)) / (sum(x))^2
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. x
    p = exp_scaled / exp_sum  # weights, length M = 2n-1
    # g[i] = 2 * sum_j x[j] * p[j+i] for i=0..n-1
    # = 2 * ifft(conj(fft(x,N)) * fft(p,N))[:n] where N >= 3n-1
    N_grad = 1 << (3 * n - 1).bit_length()
    X = np.fft.fft(x, N_grad)
    P = np.fft.fft(p, N_grad)
    g_L = 2.0 * np.real(np.fft.ifft(np.conj(X) * P))[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_conv_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_k_subgradient(a, n_iters=3000, top_k=10, step_init=0.02, decay=0.9999):
    """Subgradient method on the top-k peaks of the convolution."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    N_grad = 1 << (3 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-k subgradient
        if top_k >= M:
            top_idx = np.arange(M)
            p = np.ones(M) / M
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros(M)
            p[top_idx] = 1.0 / top_k

        # Gradient: g[i] = 2 * sum_j a[j] * p[j+i]
        X = np.fft.fft(a, N_grad)
        P = np.fft.fft(p, N_grad)
        g_M = 2.0 * np.real(np.fft.ifft(np.conj(X) * P))[:n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g_M / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_a2_waterfill(a, n_iters=400, n_levels=5, strength=0.5):
    """Multi-level water-filling Fienup using A^2."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        T = float(c.max())
        if T < 1e-15:
            break

        # Multi-level water-filling: create desired c
        c_desired = c.copy()
        for lv in range(n_levels):
            frac = 0.1 * (lv + 1) / n_levels
            thresh = np.quantile(c, 1.0 - frac)
            above = c > thresh
            if above.any():
                # Compress towards threshold
                c_desired[above] = thresh + (c[above] - thresh) * (1.0 - strength)

        c_desired_full = np.zeros(N)
        c_desired_full[:2 * n - 1] = c_desired
        B_desired = np.fft.rfft(c_desired_full)

        abs_B = np.abs(B)
        abs_B_des = np.abs(B_desired)
        ratio = np.sqrt(np.maximum(abs_B_des, 1e-30) / np.maximum(abs_B, 1e-30))
        A_new = A * (ratio ** 0.5)

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_a2_relax(a, n_iters=300, alpha_init=0.1, alpha_end=0.6):
    """
    Spectral relaxation: B_new = (1-alpha)*B + alpha*B_mean_spectrum
    where B_mean_spectrum is a flat spectrum (gives delta in time domain).
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        # Compute target: flat at the mean level
        c_mean = float(np.mean(c))
        # B_target corresponds to a flat c
        c_target = np.full(N, 0.0)
        c_target[:2 * n - 1] = c_mean
        B_target = np.fft.rfft(c_target)

        # Relax
        B_new = (1.0 - alpha) * B + alpha * B_target

        # Take square root (preserving phase of A)
        abs_ratio = np.sqrt(np.maximum(np.abs(B_new), 1e-30) / np.maximum(np.abs(B), 1e-30))
        A_new = A * abs_ratio

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_a2(a, n_iters=500, gamma=0.5, clip_frac=0.05):
    """Fienup using A^2 (squared spectrum) to target convolution flatness."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())

        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        T = float(c.max())
        if T < 1e-15:
            break

        thresh = T * (1.0 - clip_frac)
        c_desired = np.minimum(c, thresh)

        c_desired_full = np.zeros(N)
        c_desired_full[:2 * n - 1] = c_desired
        B_desired = np.fft.rfft(c_desired_full)

        abs_B = np.abs(B)
        abs_B_des = np.abs(B_desired)
        ratio = np.sqrt(np.maximum(abs_B_des, 1e-30) / np.maximum(abs_B, 1e-30))
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** gamma)

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def fienup_a2_target(a, n_iters=400, target_ratio=1.01):

        B = A * A

        a_new = np.maximum(a_new, 0.0)

def lse_conv_obj_grad(x, beta):
    """LSE-smoothed max of conv(x,x) with gradient. Score = 2n*LSE/(sum(x))^2."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    N_grad = 1 << (3 * n - 1).bit_length()
    X = np.fft.fft(x, N_grad)
    P = np.fft.fft(p, N_grad)
    g_L = 2.0 * np.real(np.fft.ifft(np.conj(X) * P))[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_conv_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_k_subgradient(a, n_iters=3000, top_k=10, step_init=0.02, decay=0.9999):
    """Subgradient method targeting top-k peaks of the convolution."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    N_grad = 1 << (3 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_k >= M:
            p = np.ones(M) / M
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros(M)
            p[top_idx] = 1.0 / top_k

        X = np.fft.fft(a, N_grad)
        P = np.fft.fft(p, N_grad)
        g_M = 2.0 * np.real(np.fft.ifft(np.conj(X) * P))[:n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g_M / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_a2_soft(a, n_iters=400, gamma=0.4, soft_power=2.0):
    """
    Fienup A^2 with soft peak reduction:
    c_desired[k] = c[k] - (c[k] - c_mean)^+ / (1 + (c[k]-c_mean)^+ * soft_power)
    This gives a smoother target than hard clipping.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        c_mean = 1.0 / (2 * n - 1)
        excess = np.maximum(c - c_mean, 0.0)
        # Soft reduction: divide excess by (1 + excess * soft_power)
        reduction = excess / (1.0 + excess * soft_power * n)
        c_desired = c - reduction

        c_desired_full = np.zeros(N)
        c_desired_full[:2 * n - 1] = c_desired
        B_desired = np.fft.rfft(c_desired_full)

        abs_B = np.abs(B)
        abs_B_des = np.abs(B_desired)
        ratio = np.sqrt(np.maximum(abs_B_des, 1e-30) / np.maximum(abs_B, 1e-30))
        ratio = np.clip(ratio, 0.01, 50.0)
        A_new = A * (ratio ** gamma)

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_a2_phased(a, n_iters=300, gamma=0.5, clip_frac=0.08, phase_flip_every=50):
    """
    Fienup A^2 with occasional phase randomization to escape local minima.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    rng_local = np.random.default_rng(12345)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        B = A * A
        c_full = np.fft.irfft(B, N)
        c = c_full[:2 * n - 1]

        T = float(c.max())
        if T < 1e-15:
            break

        thresh = T * (1.0 - clip_frac)
        c_desired = np.minimum(c, thresh)

        c_desired_full = np.zeros(N)
        c_desired_full[:2 * n - 1] = c_desired
        B_desired = np.fft.rfft(c_desired_full)

        abs_B = np.abs(B)
        abs_B_des = np.abs(B_desired)
        ratio = np.sqrt(np.maximum(abs_B_des, 1e-30) / np.maximum(abs_B, 1e-30))
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** gamma)

        # Occasional phase perturbation
        if it > 0 and it % phase_flip_every == 0:
            n_flip = max(1, len(A_new) // 20)
            flip_idx = rng_local.choice(len(A_new), size=n_flip, replace=False)
            A_new[flip_idx] *= -1.0

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def update_best_state(state, sc, a, label, t_start):
    """Update the best solution state. Returns True if improved."""
    if sc < state['score']:
        state['score'] = sc
        state['a'] = np.clip(a.copy(), 0.0, 1000.0)
        print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
        return True
    return False


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses corrected Fienup (A^2-based), LSE-LBFGS, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    state = {'score': np.inf, 'a': np.ones(4096, dtype=np.float64) * 0.5}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine lengths
    lengths = [6962, 8192, 12288, 16384]
    if parent is not None:
        lengths = [len(parent)] + [l for l in lengths if l != len(parent)]
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    if parent is not None:
        par_sc = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={par_sc:.8f}")
        update_best_state(state, par_sc, parent, "PARENT", t_start)

    # === Phase 0: Quick Fienup A^2 from parent (8% time) ===
    phase0_end = t_start + 0.08 * (total_budget - 30.0)
    print(f"\n=== Phase 0: Quick Fienup A^2 (deadline t={phase0_end-t_start:.0f}s) ===")

    if parent is not None and time.monotonic() < phase0_end:
        a_init = parent.copy() / parent.sum()
        for clip in [0.05, 0.08]:
            if time.monotonic() > phase0_end:
                break
            a_f = fienup_a2(a_init, n_iters=150, gamma=0.5, clip_frac=clip)
            sc = compute_score(a_f)
            update_best_state(state, sc, a_f, f"P0-A2 c={clip}", t_start)

        # Target-based Fienup
        if time.monotonic() < phase0_end:
            a_t = fienup_a2_target(a_init, n_iters=150, target_ratio=1.005)
            sc = compute_score(a_t)
            update_best_state(state, sc, a_t, "P0-TARGET", t_start)

    # === Phase 1: Extended Fienup A^2 variants (32% time) ===
    phase1_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Extended Fienup A^2 (deadline t={phase1_end-t_start:.0f}s) ===")

    a_cur = state['a'].copy() / state['a'].sum()

    # Chain Fienup A^2 with increasing aggressiveness
    for variant in range(6):
        if time.monotonic() > phase1_end:
            break
        clip = 0.03 + 0.015 * variant
        gamma = 0.3 + 0.08 * variant
        n_it = 250
        a_out = fienup_a2(a_cur, n_iters=n_it, gamma=gamma, clip_frac=clip)
        sc = compute_score(a_out)
        if update_best_state(state, sc, a_out, f"P1-A2 v={variant} g={gamma:.2f} c={clip:.3f}", t_start):
            a_cur = a_out.copy() / a_out.sum()

    # Water-filling variant
    if time.monotonic() < phase1_end:
        for strength in [0.4, 0.6, 0.8]:
            if time.monotonic() > phase1_end:
                break
            a_wf = fienup_a2_waterfill(a_cur, n_iters=250, n_levels=6, strength=strength)
            sc = compute_score(a_wf)
            if update_best_state(state, sc, a_wf, f"P1-WF s={strength:.1f}", t_start):
                a_cur = a_wf.copy() / a_wf.sum()

    # Target-based
    if time.monotonic() < phase1_end:
        for tr in [1.001, 1.005, 1.01]:
            if time.monotonic() > phase1_end:
                break
            a_t = fienup_a2_target(a_cur, n_iters=250, target_ratio=tr)
            sc = compute_score(a_t)
            if update_best_state(state, sc, a_t, f"P1-TGT r={tr:.3f}", t_start):
                a_cur = a_t.copy() / a_t.sum()

    # Soft reduction
    if time.monotonic() < phase1_end:
        for sp in [1.0, 3.0, 10.0]:
            if time.monotonic() > phase1_end:
                break
            a_soft = fienup_a2_soft(a_cur, n_iters=250, gamma=0.4, soft_power=sp)
            sc = compute_score(a_soft)
            if update_best_state(state, sc, a_soft, f"P1-SOFT p={sp:.0f}", t_start):
                a_cur = a_soft.copy() / a_soft.sum()

    # Phased Fienup
    if time.monotonic() < phase1_end:
        a_ph = fienup_a2_phased(state['a'], n_iters=200, gamma=0.5, clip_frac=0.08)
        sc = compute_score(a_ph)
        update_best_state(state, sc, a_ph, "P1-PHASED", t_start)

    print(f"After Phase 1: best = {state['score']:.8f}")

    # === Phase 2: LSE-smoothed L-BFGS (30% time) ===
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LSE LBFGS (deadline t={phase2_end-t_start:.0f}s) ===")

    betas = [50, 100, 300, 1000, 3000, 5000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 45:
            break
        n_try = len(state['a'])
        maxiter = min(8000, int(remaining // max(1, n_try / 3000) * 8))
        maxfun = maxiter * 12
        sc, a_opt = run_lbfgs(state['a'], beta, maxiter=maxiter, maxfun=maxfun)
        update_best_state(state, sc, a_opt, f"LBFGS b={beta}", t_start)

    print(f"After Phase 2: best = {state['score']:.8f}")

    # === Phase 3: Subgradient refinement (12% time) ===
    phase3_end = t_start + 0.82 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Subgradient (deadline t={phase3_end-t_start:.0f}s) ===")

    for top_k in [5, 10, 20, 50, 100]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        n_it = int(min(3000, remaining * 50))
        a_sg = top_k_subgradient(state['a'], n_iters=n_it, top_k=top_k)
        sc = compute_score(a_sg)
        if update_best_state(state, sc, a_sg, f"SG k={top_k}", t_start):
            pass

    print(f"After Phase 3: best = {state['score']:.8f}")

    # === Phase 4: Final refinement (10% time) ===
    phase4_end = deadline
    print(f"\n=== Phase 4: Final (deadline t={phase4_end-t_start:.0f}s) ===")

    # Fienup + LBFGS combo
    if time.monotonic() < phase4_end - 90:
        a_f = fienup_a2(state['a'], n_iters=200, gamma=0.4, clip_frac=0.04)
        sc = compute_score(a_f)
        if update_best_state(state, sc, a_f, "FINAL-A2", t_start):
            pass

    if time.monotonic() < phase4_end - 45:
        sc, a_opt = run_lbfgs(state['a'], 2000, maxiter=4000, maxfun=40000)
        update_best_state(state, sc, a_opt, "FINAL-LBFGS", t_start)

    # Perturbation restarts
    for trial in range(4):
        if time.monotonic() > phase4_end - 15:
            break
        noise = 0.02 + 0.015 * trial
        a_p = state['a'] * (1.0 + noise * rng.standard_normal(len(state['a'])))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S = float(a_p.sum())
        if S < 1e-12:
            break
        a_p = a_p / S
        a_r = fienup_a2(a_p, n_iters=100, gamma=0.5, clip_frac=0.05)
        sc = compute_score(a_r)
        update_best_state(state, sc, a_r, f"RESTART-{trial} n={noise:.3f}", t_start)

    # One more LBFGS if time permits
    if time.monotonic() < phase4_end - 10:
        sc, a_opt = run_lbfgs(state['a'], 5000, maxiter=3000, maxfun=30000)
        update_best_state(state, sc, a_opt, "ULTIMATE-LBFGS", t_start)

    # === Output ===
    out = [float(max(0.0, min(1000.0, x))) for x in state['a']]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={state['score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
