import numpy as np

PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def subgrad_polish(a, max_iters=20, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc

        for k in (1, 2, 3, 5, 8):
            for mult in (0.01, 0.03, 0.08, 0.15, 0.30):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                d, S = (fast_self_conv(a) * len(a) / max(S, 1e-12) ** 2), S
                dmax = float(np.max(d))
                if dmax <= 1e-12:
                    break
                w = active_weights(d, k)
                h = corr_pair(a, w)
                hmax = float(np.max(h))
                if hmax < 1e-12:
                    continue
                q = a * np.exp(-mult * h / hmax)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
        maxfun = int(min(int(maxfun), int((0.9 * rem) / 0.05)))
        maxfun = max(64, maxfun)
    else:
        maxfun = max(64, int(maxfun))

    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def optimize_from(a_start, deadline):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    t0 = time.monotonic()

    with np.errstate(over='ignore', invalid='ignore'):
        # Continuation schedule
        stages = [
            (100.0, 0.20),
            (300.0, 0.20),
            (1000.0, 0.20),
            (3000.0, 0.15),
            (10000.0, 0.15),
            (30000.0, 0.10),
        ]

        total_time = max(5.0, float(deadline - t0 - 10.0))
        cum = 0.0
        for beta, frac in stages:
            cum += frac
            if cum >= 1.0:
                cum = 1.0
            sub_dead = t0 + cum * total_time
            sub_dead = min(float(sub_dead), float(deadline - 5.0))

            while time.monotonic() < sub_dead - 3.0:
                rem = float(sub_dead - time.monotonic())
                if rem < 5.0:
                    break
                est_time = 0.02 * max(1.0, n / 8192.0)
                mf = max(64, min(2000, int((0.8 * rem) / est_time)))
                sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a2.copy()
                    print("  opt improved: {:.8f} (beta={:.0f})".format(best_sc, beta))
                a = normalize_a(a2)

            # Subgradient polish between stages
            if time.monotonic() < sub_dead - 5.0:
                sc_sg, a_sg = subgrad_polish(a, max_iters=8, deadline=sub_dead)
                if np.isfinite(sc_sg) and sc_sg < best_sc:
                    best_sc = sc_sg
                    best_a = a_sg.copy()
                    print("  subgrad improved: {:.8f}".format(best_sc))
                a = normalize_a(a_sg)

    return float(best_sc), best_a


def generate_structured_inits(n, rng, parent):
    inits = []

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        p = normalize_a(arr)
        inits.append(p)

    # Uniform (optimal by Cauchy-Schwarz)
    add(np.ones(n, dtype=np.float64))

    # Parent
    if parent is not None and parent.size > 0:
        add(resample_sequence(parent, n))
        add(resample_sequence(parent, n)[::-1])

    x = np.linspace(0.0, 1.0, n)

    # V-shapes
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Cosine
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.5, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Linear ramps
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * (2.0 * x - 1.0))

    # Exponential
    for rate in (1.0, 2.0, 4.0, 8.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Bimodal
    for width in (0.1, 0.2, 0.3):
        add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)))
        add(np.exp(-(((x - 0.5) / width) ** 2)))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 10240
    print("Parent available: {}, length={}".format(parent is not None, n_parent))

    # Determine target lengths
    if parent is not None:
        lengths = [n_parent]
    else:
        lengths = [10240]
    # Add a few alternatives
    for L in [4096, 8192, 16384]:
        if L not in lengths and 1024 <= L <= 16384:
            lengths.append(L)
    lengths = lengths[:3]

    # Phase 1: Evaluate initial candidates
    best_seq = None
    best_score = np.inf

    for n in lengths:
        inits = generate_structured_inits(n, rng, parent)
        for arr in inits:
            sc = exact_score_a(arr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = arr.copy()
                print("n={}: init score={:.8f}".format(n, sc))
        if time.monotonic() > deadline - 100.0:
            break

    if best_seq is None:
        best_seq = np.ones(10240, dtype=np.float64) / 10240.0
        best_score = 2.0

    print("Best initial: {:.8f}".format(best_score))

    # Phase 2: Optimize from best starting points
    # Generate top diverse starts
    n_best = len(best_seq)
    starts = [best_seq.copy()]

    # Add parent resampled
    if parent is not None:
        p_res = normalize_a(resample_sequence(parent, n_best))
        starts.append(p_res)

    # Add a few structured inits
    inits = generate_structured_inits(n_best, rng, parent)
    scored = [(exact_score_a(x), x) for x in inits]
    scored.sort(key=lambda t: t[0])
    for sc, arr in scored[:4]:
        if np.isfinite(sc) and sc < best_score + 0.5:
            starts.append(arr)

    starts = starts[:4]

    # Allocate time to each start
    total_opt_time = max(60.0, float(deadline - time.monotonic() - 60.0))
    fracs = [0.45, 0.25, 0.18, 0.12][:len(starts)]
    # Normalize fracs
    fs = float(sum(fracs))
    fracs = [f / fs for f in fracs]

    cum = 0.0
    for i, st in enumerate(starts):
        if i >= len(fracs):
            break
        cum += fracs[i]
        sub_dead = t_start + cum * (total_opt_time) + (time.monotonic() - t_start)
        sub_dead = min(sub_dead, deadline - 30.0)

        if time.monotonic() > sub_dead - 10.0:
            continue

        print("Optimizing start {} (score={:.8f}), deadline in {:.0f}s".format(
            i, exact_score_a(st), sub_dead - time.monotonic()))
        sc, seq = optimize_from(st, sub_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("  New best: {:.8f}".format(best_score))

    # Phase 3: Final subgradient polish
    if time.monotonic() < deadline - 20.0:
        sc, seq = subgrad_polish(best_seq, max_iters=15, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Phase 4: One more LSE pass with very high beta
    if time.monotonic() < deadline - 15.0:
        rem = deadline - time.monotonic() - 5.0
        if rem > 10.0:
            mf = max(100, int((rem * 0.8) / 0.05))
            sc, seq = run_lbfgs(best_seq, 50000.0, deadline - 10.0, min(mf, 1000))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = 512.0
    out_arr = out_arr / S
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
