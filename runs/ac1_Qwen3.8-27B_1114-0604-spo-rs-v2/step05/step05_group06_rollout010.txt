    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _compute_subgrad(a, n_top=5, sharpen=50.0):
    """Compute weighted subgradient of 2n*max(conv)/S^2. Returns (score, grad)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _self_conv(a)
    d = c / (S * S)
    dmax = float(d.max())
    M = 2 * n - 1
    k = min(n_top, M)
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    z = (d_act - d_act.max()) * sharpen
    e = np.exp(z)
    se = e.sum()
    w_act = e / se
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = (2.0 * float(n) / (S * S)) * (corr - 2.0 * LDS)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    score = float(2.0 * n * dmax)
    return score, grad


def _nesterov_subgrad(a0, max_iters=500, deadline=None, mom_base=0.7, init_step=0.02):
    """Nesterov-accelerated subgradient with backtracking on simplex."""
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    step = init_step
    patience = 0
    no_improve = 0

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        sc_y, grad_y = _compute_subgrad(y, n_top=5, sharpen=50.0)
        if not np.isfinite(sc_y):
            break
        gnorm = float(np.linalg.norm(grad_y))
        if gnorm < 1e-14:
            break
        # Backtracking: try several step sizes
        best_trial = None
        best_trial_sc = best_sc
        for scale in (1.0, 0.5, 0.25, 0.125):
            s = step * scale / (np.sqrt(t) * 0.1 + 1.0)
            v = mom_base * v + s * grad_y
            trial = y - v
            trial = np.clip(trial, 0.0, None)
            St = trial.sum()
            if St < 1e-12:
                continue
            trial = trial / St
            sc_t = _score(trial)
            if sc_t < best_trial_sc:
                best_trial_sc = sc_t
                best_trial = trial.copy()
        if best_trial is None:
            step *= 0.5
            if step < 1e-10:
                break
            continue
        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = best_trial + beta * (best_trial - y)
        y = np.clip(y, 0.0, None)
        Sy = y.sum()
        if Sy > 1e-12:
            y = y / Sy
        t = t_new
        a = best_trial
        if best_trial_sc < best_sc:
            best_sc = best_trial_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            step *= 0.7
            v = np.zeros(n)
            y = a.copy()
            no_improve = 0
    return best_sc, best_a


def _plain_subgrad(a0, max_iters=300, deadline=None, init_step=0.01):
    """Plain subgradient with adaptive step (no momentum) for stability."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    step = init_step
    no_improve = 0

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        sc, grad = _compute_subgrad(a, n_top=3, sharpen=100.0)
        if not np.isfinite(sc):
            break
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        best_trial = None
        best_trial_sc = best_sc
        for scale in (1.0, 0.5, 0.25, 0.1):
            s = step * scale
            trial = a - s * grad
            trial = np.clip(trial, 0.0, None)
            St = trial.sum()
            if St < 1e-12:
                continue
            trial = trial / St
            sc_t = _score(trial)
            if sc_t < best_trial_sc:
                best_trial_sc = sc_t
                best_trial = trial.copy()
        if best_trial is None:
            step *= 0.5
            if step < 1e-12:
                break
            continue
        a = best_trial
        if best_trial_sc < best_sc:
            best_sc = best_trial_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 20:
            step *= 0.6
            no_improve = 0
    return best_sc, best_a


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient for L-BFGS-B."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun=400):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_t, 1e-4))
    mf = max(20, min(mf, maxfun, 6000))
    maxiter = max(15, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _peak_shave(a0, deadline, n_peaks=2, max_rounds=50):
    """Targeted peak shaving: identify top peaks, directly reduce them."""
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if deadline and time.monotonic() > deadline:
            break
        c = _self_conv(a)
        d = c / (a.sum() ** 2)
        dmax = float(d.max())
        M = 2 * n - 1
        k = min(n_peaks, M)
        idx = np.argpartition(d, M - k)[M - k:]
        # Use uniform weights on top peaks
        w_full = np.zeros(M)
        w_full[idx] = 1.0 / k
        corr = _cross_corr(a, w_full)
        S = float(a.sum())
        LDS = float(w_full[idx] @ d[idx])
        # Gradient direction to reduce peaks
        g_dir = corr - 2.0 * LDS
        # Try to apply: a_new = a - eta * g_dir, projected
        best_eta = 0
        best_trial_sc = best_sc
        for eta in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            trial = a - eta * g_dir
            trial = np.clip(trial, 0.0, None)
            St = trial.sum()
            if St < 1e-12:
                continue
            trial = trial / St
            sc_t = _score(trial)
            if sc_t < best_trial_sc:
                best_trial_sc = sc_t
                best_eta = eta
        if best_eta == 0:
            break
        a = a - best_eta * g_dir
        a = np.clip(a, 0.0, None)
        Sa = a.sum()
        if Sa > 1e-12:
            a = a / Sa
        if best_trial_sc < best_sc:
            best_sc = best_trial_sc
            best_a = a.copy()
        else:
            break
    return best_sc, best_a


def _spectral_init(n, rng, exponent=None):
    """Generate non-negative seq with controlled spectral shape."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    if exponent is None:
        exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 300))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_at_length(a0, deadline, rng, label=""):
    """Multi-phase optimization at a fixed length."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    print(f"  [{label}] n={n}, start score={best_sc:.8f}")

    remaining = deadline - t0
    if remaining < 10:
        return best_sc, best_a

    # Phase 1: Nesterov subgradient (60% of time)
    t1 = t0 + 0.60 * remaining
    if t1 > t0 + 5:
        sc, a1 = _nesterov_subgrad(a, max_iters=2000, deadline=t1, mom_base=0.7, init_step=0.015)
        if sc < best_sc:
            best_sc = sc
            best_a = a1.copy()
            print(f"  [{label}] after nesterov: {best_sc:.8f}")
        a = _norm(a1)

    # Phase 2: L-BFGS-B with escalating beta (25% of time)
    t2 = t0 + 0.85 * remaining
    if t2 > t1 + 5:
        stages = [(5, 100), (10, 300), (25, 1000), (50, 3000), (100, 8000)]
        stage_time = (t2 - t1) / len(stages)
        for k, beta in stages:
            if time.monotonic() > t2 - 2:
                break
            sc, a2 = _run_lbfgs(a, beta, k, min(stage_time, t2 - time.monotonic()), 300)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print(f"  [{label}] LBP k={k} b={beta}: {best_sc:.8f}")
            a = _norm(a2)

    # Phase 3: Plain subgradient polish (10% of time)
    t3 = t0 + 0.95 * remaining
    if t3 > t2 + 3:
        sc, a3 = _plain_subgrad(a, max_iters=200, deadline=t3, init_step=0.005)
        if sc < best_sc:
            best_sc = sc
            best_a = a3.copy()
            print(f"  [{label}] plain_subgrad: {best_sc:.8f}")
        a = _norm(a3)

    # Phase 4: Peak shave
    if time.monotonic() < deadline - 2:
        sc, a4 = _peak_shave(a, deadline, n_peaks=2, max_rounds=30)
        if sc < best_sc:
            best_sc = sc
            best_a = a4.copy()
            print(f"  [{label}] peak_shave: {best_sc:.8f}")

    # Final Nesterov with smaller steps
    if time.monotonic() < deadline - 1:
        sc, a5 = _nesterov_subgrad(best_a, max_iters=100, deadline=deadline, mom_base=0.5, init_step=0.005)
        if sc < best_sc:
            best_sc = sc
            best_a = a5.copy()
            print(f"  [{label}] final nesterov: {best_sc:.8f}")

    print(f"  [{label}] end: {best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: multi-length Nesterov + LSE + peak shaving."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 16:
                parent = _norm(p)
                print(f"Loaded parent: n={p.size}, score={_score(parent):.8f}")
    except Exception:
        pass

    # Determine lengths to try
    lengths_to_try = []
    if parent is not None:
        pn = len(parent)
        # Try parent length and neighbors
        candidates = [pn]
        # Round to power of 2 or nearby
        for target in [4096, 8192, 16384]:
            if abs(target - pn) / pn < 0.5:
                candidates.append(target)
        # Add a shorter and longer option
        candidates.append(max(2048, pn // 2))
        candidates.append(min(32768, pn * 2))
        # Deduplicate and sort by proximity to parent
        seen = set()
        for c in candidates:
            c = int(c)
            if c not in seen and c >= 1024 and c <= 32768:
                seen.add(c)
                lengths_to_try.append(c)
    else:
        lengths_to_try = [4096, 8192, 16384]

    # Limit to at most 3 lengths based on time
    max_lengths = min(3, len(lengths_to_try))
    lengths_to_try = lengths_to_try[:max_lengths]
    print(f"Lengths to try: {lengths_to_try}")

    # Generate initial candidates for each length
    inits_by_len = {}
    for L in lengths_to_try:
        inits = []
        if parent is not None:
            inits.append(('parent_up', _upsample(parent, L)))
            inits.append(('parent_rev', _upsample(parent[::-1], L)))
            # Perturbed parent
            p_up = _upsample(parent, L)
            inits.append(('parent_perturb', _smooth_perturb(p_up, rng, 0.1)))
            inits.append(('parent_perturb2', _smooth_perturb(p_up, rng, 0.3)))
        # Spectral inits
        for i in range(3):
            inits.append((f'spec{i}', _spectral_init(L, rng)))
        # Noisy init
        noise = rng.standard_normal(L)
        k = max(5, L // 200)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        inits.append(('noise', _norm(np.exp(noise))))
        inits_by_len[L] = inits

    # Quick evaluation to rank inits per length
    best_per_len = {}
    for L, inits in inits_by_len.items():
        best_sc = np.inf
        best_arr = None
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_arr = arr.copy()
        if best_arr is not None:
            best_per_len[L] = (best_sc, best_arr)
            print(f"  n={L}: best init score={best_sc:.8f}")

    if not best_per_len:
        # Fallback
        L = 8192
        best_per_len[L] = (2.0, _norm(np.ones(L)))

    # Allocate time: give more time to the most promising lengths
    sorted_lengths = sorted(best_per_len.keys(), key=lambda L: best_per_len[L][0])
    total_opt_time = deadline - t_start - 60.0

    if len(sorted_lengths) == 1:
        fractions = [1.0]
    elif len(sorted_lengths) == 2:
        fractions = [0.65, 0.35]
    else:
        fractions = [0.50, 0.30, 0.20]

    best_score = float('inf')
    best_seq = None

    for i, L in enumerate(sorted_lengths):
        frac = fractions[i] if i < len(fractions) else 0.1
        sub_deadline = t_start + sum(fractions[:i+1]) * total_opt_time + 10.0
        sub_deadline = min(sub_deadline, deadline - 60.0)

        if time.monotonic() > sub_deadline - 10:
            continue

        sc_init, arr_init = best_per_len[L]
        print(f"\n=== Length {L} (budget={frac:.0%}, init={sc_init:.8f}) ===")

        # Main optimization
        sc, seq = _optimize_at_length(arr_init, sub_deadline, rng, label=f"n{L}")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"  *** Global best: n={L}, score={best_score:.8f} ***")

        # Try a second init for this length if time remains
        if time.monotonic() < sub_deadline - 30:
            # Use the best result as new init for a second pass
            sc2, seq2 = _optimize_at_length(_smooth_perturb(seq, rng, 0.15), sub_deadline, rng, label=f"n{L}_restart")
            if sc2 < best_score:
                best_score = sc2
                best_seq = seq2.copy()
                print(f"  *** Global best (restart): n={L}, score={best_score:.8f} ***")

    # Final global polish on best sequence
    print(f"\n=== Final polish: score={best_score:.8f}, n={len(best_seq)} ===")
    final_deadline = deadline - 20.0

    # L-BFGS-B with high beta
    for k_f, beta_f in [(50, 5000), (100, 10000), (200, 20000)]:
        if time.monotonic() > final_deadline - 20:
            break
        rem = final_deadline - time.monotonic() - 20
        if rem < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, rem * 0.4, 500)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LBP k={k_f}: {best_score:.8f}")

    # Nesterov polish
    if time.monotonic() < final_deadline - 10:
        sc_ng, a_ng = _nesterov_subgrad(best_seq, max_iters=200, deadline=final_deadline - 5, mom_base=0.5, init_step=0.005)
        if sc_ng < best_score:
            best_score = sc_ng
            best_seq = a_ng.copy()
            print(f"  Final nesterov: {best_score:.8f}")

    # Peak shave
    if time.monotonic() < final_deadline - 3:
        sc_ps, a_ps = _peak_shave(best_seq, final_deadline, n_peaks=2, max_rounds=20)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
            print(f"  Final peak_shave: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S
    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
