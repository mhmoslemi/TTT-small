"""
Multi-phase optimizer for flat-autocorrelation non-negative sequences.
Combines improved LP-based alternating method, adaptive LSE-LBFGS polishing,
spectral peak shaving, and targeted subgradient local search.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a: 2 * cross-correlation(a, p) at appropriate offsets
    rev_a = a[::-1]
    conv_res = cross_conv(rev_a, p)
    # The gradient of sum_k c[k]*p[k] where c = conv(a,a) is 2*conv(a, p)[n-1:2n-1]
    # Actually: d/d a_i [sum_k p_k * sum_j a_j * a_{k-j}] = 2 * sum_k p_k * a_{k-i}
    # = 2 * (a * p_reversed)[i]... let me use the cross_conv approach
    # d L / d a[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # This equals 2 * (a cross-correlated with p) at lag i
    # cross_conv(rev_a, p)[i + n - 1] = sum_j rev_a[j] * p[i + n - 1 - j]
    # = sum_j a[n-1-j] * p[i+n-1-j]
    # Let m = n-1-j, then = sum_m a[m] * p[i+m]
    # So cross_conv(rev_a, p)[i + n - 1] = sum_m a[m] * p[i+m]
    # That's what we want: grad_L[i] = 2 * sum_m a[m] * p[i+m]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val * (np.ones(n) * S) / S3)
    # Wait: d/d a_i [2n * L / S^2] = 2n * (grad_L_i / S^2 - 2 * L * 1 / S^3)
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_solve_fista(a, M, n_fista_iters=20000, beta_start=5.0, beta_end=1000.0,
                   x_init=None, ramp_frac=0.2):
    """
    Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on the LSE-smoothed surrogate with adaptive beta.
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    iter_int = int(n_fista_iters * ramp_frac)
    for it in range(n_fista_iters):
        if it < iter_int:
            frac = it / max(1, iter_int)
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * n * sum_a2 + 1e-10
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # grad = a * sliding_window_sum(p) - 1
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=60, fista_iters=20000, time_limit=None,
                   line_search=True):
    """
    LP-based alternating method with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters,
                           beta_start=5.0, beta_end=1000.0, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        if line_search:
            # Backtracking line search
            t = 1.0
            best_t_sc = best_sc
            best_t = 0.0
            for _ in range(15):
                a_try = (1.0 - t) * a + t * x
                a_try = np.maximum(a_try, 0.0)
                St = float(a_try.sum())
                if St < 1e-12:
                    t *= 0.5
                    continue
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
                t *= 0.5
                if t < 0.01:
                    break
            t = best_t if best_t > 0 else 0.3
            a_new = (1.0 - t) * a + t * x
        else:
            t = 0.5
            a_new = (1.0 - t) * a + t * x

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 15:
                # Diversify
                rng = np.random.default_rng(outer)
                a = a * (1.0 + 0.03 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve_count = 0
                if no_improve_count >= 30:
                    break

    return best_a, best_sc, x_warm


def error_reduction(a, n_iters=100, alpha=0.6):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def spectral_peak_shaving(a, n_iters=200, time_limit=None):
    """
    Spectral peak shaving: iteratively reduce the max of conv(a,a)
    by modifying the amplitude spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        # Compute convolution
        spec = A * A
        conv = np.fft.ifft(spec).real[:2 * n - 1]
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Compute the "error": we want to reduce the peak
        target = float(np.median(conv))
        err = np.zeros(2 * n - 1)
        # Focus on top peaks
        top_thresh = target + 0.3 * (peak_val - target)
        err = np.where(conv > top_thresh, conv - top_thresh, 0.0)
        if err.max() < 1e-15:
            break

        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        # Newton step: dA = -eta * err_spec / (2 * conj(A))
        # But we need to be careful: we want A_new such that |A_new|^2 has less peak
        # Using: A_new = A + dA where dA ≈ -eta * err_spec / (2 * A.conjugate())
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = -0.3 * err_spec / (2.0 * A.conjugate() + 1e-15)
        dA[np.abs(A) < 1e-10] = 0.0

        A_new = A + dA
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_subgradient_descent(a, n_steps=300, eta_init=0.02, rng=None):
    """
    Subgradient descent targeting the max convolution element.
    Uses multiple random peaks for diversification.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Subgradient direction
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Normalize gradient
        gn = float(np.linalg.norm(direction))
        if gn < 1e-15:
            break
        direction = direction / gn

        # Adaptive step size with backtracking
        step = eta
        improved = False
        for _ in range(5):
            a_try = a - step * direction
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                step *= 0.5
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                eta = min(eta * 1.2, 0.05)
                break
            step *= 0.5

        if not improved:
            eta = max(eta * 0.7, 0.001)

        # Randomize peak selection occasionally
        if it % 30 == 0 and rng.random() < 0.4:
            top_k = max(5, len(c) // 20)
            candidates = np.argsort(c)[::-1][:top_k]
            k_star = int(candidates[rng.integers(len(candidates))])

    return best_a


def multiplicative_weights_update(a, n_iters=100, eta=0.01):
    """
    Multiplicative weights / mirror descent update.
    a_i <- a_i * exp(-eta * grad[i])
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Subgradient of max(a*a)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        grad = np.zeros(n)
        if j_end >= j_start:
            grad[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.35 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    smooth = uniform_filter1d(rng.standard_normal(n), size=max(3, n // 40))
    inits.append(np.clip(0.5 + 0.35 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(8.0 * np.pi * x))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    # Power-law shaped
    inits.append(np.clip(0.5 * (x ** 0.3 + (1 - x) ** 0.3), 0.01, 1000.0))
    # Gaussian-like
    inits.append(np.clip(0.5 * np.exp(-2.0 * (x - 0.5) ** 2), 0.01, 1000.0))
    return [np.maximum(init, 0.0) for init in inits]

    ]

    x_warm = None
    for fi, (fista_it, max_out) in enumerate(fista_schedules):
        if time.monotonic() > phase3_deadline - 30:
            break
        remaining = phase3_deadline - time.monotonic()
        # Estimate time per LP solve: fista_it * n * ~5e-7 seconds
        n_main = len(a_work)
        est_per_solve = fista_it * n_main * 5e-7
        max_out_actual = max(5, min(max_out, int(remaining / max(1.0, est_per_solve)) - 2))
        print(f"  LP round {fi}: n={n_main}, fista={fista_it}, "
              f"max_outer={max_out_actual}, remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating(
            a_work,
            max_outer_iters=max_out_actual,
            fista_iters=fista_it,
            time_limit=phase3_deadline - 10,
            line_search=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    # Try slightly different lengths
    for n in lengths:
        if n == len(best_a) or time.monotonic() > phase3_deadline - 90:
            continue
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out, _ = lp_alternating(
            a_test, max_outer_iters=25, fista_iters=15000,
            time_limit=phase3_deadline - 15, line_search=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: LSE-LBFGS polishing (22% time)
    # ==================================================================
    phase4_deadline = t_start + 0.82 * total_budget
    print(f"\n--- Phase 4: LSE-LBFGS Polishing ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [30, 80, 200, 500, 1000, 2000, 4000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 20:
            break
        remaining = phase4_deadline - time.monotonic()
        maxiter = min(15000, max(2000, int(remaining / max(1, len(a_norm)) * 200)))
        maxfun = min(800000, max(50000, maxiter * 30))
        print(f"  L-BFGS beta={beta}, maxfun={maxfun}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(a_norm, beta, maxiter=maxiter, maxfun=maxfun)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, t={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final local search and polish (remaining time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Local Search ---")

    # Peak subgradient
    a_pk = peak_subgradient_descent(best_a, n_steps=400, eta_init=0.015, rng=rng)
    sc = compute_score(a_pk)
    report(sc, a_pk, "peak SGD")

    # Spectral again on best
    if time.monotonic() < deadline - 120:
        a_sp2 = spectral_peak_shaving(best_a, n_iters=100, time_limit=deadline - 60)
        sc = compute_score(a_sp2)
        report(sc, a_sp2, "spectral2")

    # Final LP polish
    if time.monotonic() < deadline - 90:
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final, _ = lp_alternating(
            a_final_in, max_outer_iters=20, fista_iters=25000,
            time_limit=deadline - 45, line_search=True
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # Final high-beta LBFGS
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 10000, maxiter=8000, maxfun=400000)
        report(sc, a_opt, "final LBFGS b=10000")

    # Final peak search
    if time.monotonic() < deadline - 10:
        a_final = peak_subgradient_descent(best_a, n_steps=200, eta_init=0.01, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final peak")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
