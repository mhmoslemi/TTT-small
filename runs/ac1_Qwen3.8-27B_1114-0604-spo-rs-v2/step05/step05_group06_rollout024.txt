import numpy as np

import cvxpy as cp
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(c) / (S * S))

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[idx, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    return A


def _lp_direction(f, M, active_indices):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in active_indices, g >= 0."""
    n = len(f)
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n
    A_mat = _build_constraint_matrix(f, active_indices)
    b = np.full(K, M)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ g_var <= b]
    obj = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if g_var.value is None:
        return None
    result = np.clip(g_var.value, 0.0, None)
    if result.sum() < 1e-12:
        return None
    return result


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def _obj_penalty(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            _obj_penalty, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=30, deadline=None):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0005, 0.002, 0.008, 0.02, 0.08, 0.2, 0.5):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _lop_optimize(a, deadline, label, rng):
    """LP-based iterative optimization with progressive active set growth."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    active_sizes = [10, 20, 40, 80, 150, 250, 400, 600]
    steps = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

    for stage, (K, step) in enumerate(zip(active_sizes, steps)):
        if time.monotonic() > deadline:
            break
        stage_budget = (deadline - t0) * (1.0 / (len(active_sizes) - stage))
        n_iters = min(5, max(1, int(stage_budget / 25)))

        for it in range(n_iters):
            if time.monotonic() > deadline - 10:
                break
            c = _self_conv(a)
            M = float(c.max())
            S = float(a.sum())
            if S < 1e-12 or M <= 0:
                break
            top_k = min(K, 2 * n - 1)
            active_idx = np.argsort(c)[-top_k:]
            active_idx = sorted(set(active_idx.tolist()))

            g_raw = _lp_direction(a, M, active_idx)
            if g_raw is None:
                break
            g = _norm(g_raw)
            g_sc = _score(g)

            best_candidate = a
            best_candidate_sc = best_sc
            for t_val in (step, step * 0.5, step * 2.0):
                t_val = min(t_val, 1.0)
                blended = _norm((1 - t_val) * a + t_val * g)
                bsc = _score(blended)
                if bsc < best_candidate_sc:
                    best_candidate_sc = bsc
                    best_candidate = blended

            if best_candidate_sc < best_sc:
                best_sc = best_candidate_sc
                best_a = best_candidate.copy()
                a = best_candidate.copy()
                print(f"    [{label}] stage{stage} K={K} it{it}: {best_sc:.8f}")
            else:
                a = _norm((1 - step * 0.5) * a + step * 0.5 * g)

    if time.monotonic() < deadline:
        sc, a_sg = _subgrad_polish(a, max_iters=10, deadline=deadline)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _lse_optimize(a, deadline, label):
    """LSE-based optimization with progressive K growth."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    print(f"  [{label}] LSE start: score={best_sc:.8f}, n={n}")

    stages = [
        (5, 50.0),
        (10, 100.0),
        (20, 250.0),
        (50, 500.0),
        (100, 1000.0),
        (200, 2000.0),
        (500, 5000.0),
    ]
    total_time = max(10.0, deadline - t0 - 10.0)
    cum_frac = 0.0
    fractions = [0.05, 0.08, 0.12, 0.18, 0.22, 0.25, 0.10]

    for i, ((k, beta), frac) in enumerate(zip(stages, fractions)):
        cum_frac += frac
        stage_end = min(t0 + cum_frac * total_time, deadline - 10)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break
        n_calls = min(3, max(1, int(stage_budget / 20)))
        per_call = stage_budget / n_calls
        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 300)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] LSE K={k} b={beta:.0f} c{ci}: {best_sc:.8f}")
            a = _norm(a_new)

    if time.monotonic() < deadline - 10:
        sc, a_sg = _subgrad_polish(a, max_iters=15, deadline=deadline - 5)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] LSE+subgrad: {best_sc:.8f}")

    print(f"  [{label}] LSE end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: LP peak shaving + LSE polishing with multilevel and diversity."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    if parent is not None:
        target_n = len(parent)
        if target_n < 4096:
            target_n = 8192
        elif target_n > 12000:
            target_n = 8192
    else:
        target_n = 8192

    print(f"Parent: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    inits = []
    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))
    for i in range(4):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))
    if parent is not None:
        p = _upsample(parent, target_n)
        for amp in (0.05, 0.2, 0.4):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))
    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3, 5):
        for amp in (0.2, 0.5, 0.8):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'cos{f}_{amp}', a / a.sum()))
    for i in range(2):
        noise = rng.standard_normal(target_n)
        k = max(7, target_n // 80)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'noise{i}', a / a.sum()))

    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))
    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top: {[f'{s:.6f}' for s, _, _ in scored[:5]]}")

    n_cands = 4
    candidates = []
    for sc, name, arr in scored:
        if len(candidates) >= n_cands:
            break
        candidates.append((sc, name, arr))
    if not candidates:
        candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()
    print(f"Initial best: {best_score:.8f} ({candidates[0][1]})")

    main_time = deadline - t_start - 200.0
    per_cand_time = main_time / n_cands

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        cand_start = t_start + ci * per_cand_time
        cand_end = t_start + (ci + 1) * per_cand_time - 5.0
        cand_end = min(cand_end, deadline - 150.0)
        if time.monotonic() > cand_end:
            continue

        n = len(arr_init)
        print(f"\n=== Candidate {ci+1}/'{name}', n={n}, score={sc_init:.8f} ===")

        a_work = _norm(arr_init)

        # Multilevel: optimize at smaller size first
        if n > 4096 and (cand_end - time.monotonic()) > 120:
            n_small = max(1024, n // 4)
            a_small = _upsample(a_work, n_small)
            small_deadline = min(cand_end - 60, time.monotonic() + 60)
            sc_s, a_s = _lop_optimize(a_small, small_deadline, f'{name}_small', rng)
            a_up = _norm(_upsample(a_s, n))
            sc_up = _score(a_up)
            if sc_up < _score(a_work):
                a_work = a_up
                print(f"  Multilevel improved: {sc_up:.8f}")

        # Main LP optimization
        trep = cand_end - 0.4 * (cand_end - cand_start)
        if time.monotonic() < trep:
            sc_lp, seq_lp = _lop_optimize(a_work, trep, f'{name}_lp', rng)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = seq_lp.copy()
                print(f"  *** LP best: {best_score:.8f} ***")

        # LSE optimization
        if time.monotonic() < cand_end - 10:
            a_for_lse = seq_lp if sc_lp < best_score else a_work
            sc_lse, seq_lse = _lse_optimize(a_for_lse, cand_end - 5, f'{name}_lse')
            if sc_lse < best_score:
                best_score = sc_lse
                best_seq = seq_lse.copy()
                print(f"  *** LSE best: {best_score:.8f} ***")

    # Final global polish
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0

    for k_final, beta_final in [(200, 5000), (500, 10000), (1000, 20000)]:
        if time.monotonic() > final_deadline - 30:
            break
        remaining = final_deadline - time.monotonic() - 30
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_final, k_final, remaining * 0.35, 500)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_final}: {best_score:.8f}")

    if time.monotonic() < final_deadline - 20:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=25, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Random restarts
    if time.monotonic() < final_deadline - 60:
        for restart in range(4):
            if time.monotonic() > final_deadline - 30:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.08)
            chunk_deadline = min(final_deadline - 30, time.monotonic() + 20)
            sc_r, a_r = _run_lbfgs(p, 5000, 200, max(2, chunk_deadline - time.monotonic()), 200)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=10, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]
    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score
    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
