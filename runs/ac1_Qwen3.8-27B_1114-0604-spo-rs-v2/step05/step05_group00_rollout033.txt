3. Redistribute the "saved" mass to positions that are far from the peak (i.e., positions i where a_i * a_{k*-i} is small)

```python
def pair_coord_descent(a, n_sweeps=10, n_top_pairs=20, n_target=50):
    n = len(a)
    a = normalize(np.maximum(a, 0.0))
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_conv(a).max())
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find top pairs
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        js = np.arange(j_start, j_end + 1)
        pairs = a[js] * a[k_star - js]
        
        # Top contributors
        top_local = np.argsort(pairs)[::-1][:n_top_pairs]
        top_js = js[top_local]
        
        # Target positions (low contribution to peak)
        all_pairs = a[js] * a[k_star - js]
        # Sort by contribution, take the bottom ones
        bot_local = np.argsort(pairs)[:n_target]
        bot_js = js[bot_local]
        
        # For each top pair, try to reduce both elements and increase targets
        for j in top_js:
            j_complement = k_star - j
            if j_complement < 0 or j_complement >= n:
                continue
            if j == j_complement:
                # Self-pair: reduce a[j]
                for factor in [0.5, 0.7, 0.85, 0.95]:
                    a_old_j = a[j]
                    a[j] = a_old_j * factor
                    # Redistribute
                    saved = a_old_j * (1 - factor)
                    a[bot_js] += saved / len(bot_js)
                    a = normalize(a)
                    sc = 2.0 * n * float(fast_conv(a).max())
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    else:
                        a[j] = a_old_j
                        a[bot_js] -= saved / len(bot_js)
                        a = normalize(a)
                        break
            else:
                # Pair: reduce both a[j] and a[j_complement]
                for factor in [0.5, 0.7, 0.85, 0.95]:
                    a_old_j = a[j]
                    a_old_c = a[j_complement]
                    a[j] = a_old_j * factor
                    a[j_complement] = a_old_c * factor
                    saved = (a_old_j - a[j]) + (a_old_c - a[j_complement])
                    a[bot_js] += saved / len(bot_js)
                    a = normalize(a)
                    sc = 2.0 * n * float(fast_conv(a).max())
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    else:
                        a[j] = a_old_j
                        a[j_complement] = a_old_c
                        a[bot_js] -= saved / len(bot_js)
                        a = normalize(a)
                        break
    
    return best_a
```
