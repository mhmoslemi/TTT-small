"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses p-norm continuation, Nesterov-accelerated multi-peak subgradient,
and local coordinate refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr(a, b):
    """
    Compute R[m] = sum_{j=0}^{n-1} a[j] * b[j+m] for m=0..n-1.
    where b has length >= n (typically 2n-1 for convolution).
    Uses FFT.
    """
    n = len(a)
    nb = len(b)
    N_fft = 1 << (n + nb - 1).bit_length()
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N_fft)
    B = np.fft.rfft(b, N_fft)
    full = np.fft.irfft(AR * B, N_fft)
    return full[n-1:n-1+n]


def pnorm_value_and_grad(a, p):
    """
    Compute score_p = 2n * ||conv||_p / S^2 and its gradient.
    """
    n = len(a)
    conv = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    if p == 2:
        # ||conv||_2^2 = sum conv^2, grad = 4 * cross_corr(a, conv)
        M_p_sq = float(np.dot(conv, conv))
        M_p = np.sqrt(M_p_sq)
        grad_Mp = 2.0 * cross_corr(a, conv)  # d/da[m] (1/2)||conv||_2^2 = cross_corr(a,conv)
        # Actually: d/da[m] ||conv||_2^2 = 4 * cross_corr(a, conv)[m]
        # And d/da[m] ||conv||_2 = (1/(2||conv||_2)) * d/da[m] ||conv||_2^2 = 2*cross_corr(a,conv)/||conv||_2
        grad_Mp = 2.0 * cross_corr(a, conv) / M_p
    else:
        # ||conv||_p = (sum conv^p)^(1/p)
        conv_p = conv ** p
        sum_conv_p = float(np.sum(conv_p))
        M_p = sum_conv_p ** (1.0/p)
        if M_p < 1e-30:
            return np.inf, np.zeros(n)
        # d/da[m] ||conv||_p = (1/p) * (sum conv^p)^(1/p - 1) * p * sum_k conv[k]^(p-1) * dconv[k]/da[m]
        # = (sum conv^p)^(1/p-1) * sum_k conv[k]^(p-1) * 2*a[k-m] * [valid]
        # = (sum conv^p)^(1/p-1) * 2 * cross_corr(a, conv^(p-1))[m]
        conv_pm1 = conv ** (p - 1)
        factor = 2.0 * sum_conv_p ** (1.0/p - 1.0)
        grad_Mp = factor * cross_corr(a, conv_pm1)
    
    # score = 2n * M_p / S^2
    # d(score)/da[m] = 2n * (grad_Mp[m] * S^2 - M_p * 2S) / S^4
    # = 2n * (grad_Mp[m] * S - 2*M_p) / S^3
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * M_p / S2
    grad = 2.0 * n * (grad_Mp * S - 2.0 * M_p) / S3
    
    return score, grad


def gradient_descent_pnorm(a, p, n_steps=50, lr_init=0.01, verbose=False):
    """
    Gradient descent on p-norm score with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for step in range(n_steps):
        sc, grad = pnorm_value_and_grad(a, p)
        
        if not np.isfinite(sc):
            break
        
        # Also compute exact score for tracking
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        # Backtracking line search
        lr = lr_init
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad_normed = grad / gn
        
        for bt in range(20):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                lr *= 0.5
                continue
            a_trial_norm = a_trial / S_trial
            sc_trial, _ = pnorm_value_and_grad(a_trial_norm, p)
            if sc_trial < sc or not np.isfinite(sc):
                a = a_trial_norm
                break
            lr *= 0.5
        
        if verbose and step % 10 == 0:
            print(f"    p={p} step={step}: pnorm_score={sc:.6f}, exact={compute_score(a):.6f}")
    
    return best_a


def nesterov_multi_peak_subgrad(a, n_iters=5000, top_m=10, momentum=0.9, 
                                  step_init=0.01, verbose=False):
    """
    Nesterov-accelerated multi-peak subgradient method.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    y = a.copy()  # Nesterov ghost point
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    for it in range(n_iters):
        # Compute gradient at ghost point y
        c_y = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M_y = float(c_y.max())
        
        # Find top-m peaks
        if top_m >= len(c_y):
            top_idx = np.arange(len(c_y))
            weights = np.ones(len(c_y))
        else:
            top_idx = np.argsort(c_y)[::-1][:top_m]
            # Weight by proximity to max
            vals = c_y[top_idx]
            weights = vals / vals[0]  # normalized, max weight = 1
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]
        
        # Full subgradient of score
        S2_y = S_y * S_y
        S3_y = S2_y * S_y
        gradient = 2.0 * n * (sg / S2_y - 2.0 * M_y / S3_y)
        
        # Nesterov update
        x_old = a.copy()
        a = y - step * gradient
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            a = x_old
            break
        a = a / S
        y = a + momentum * (a - x_old)
        y = np.clip(y, 0.0, 1000.0)
        
        # Update step size (Polyak-like)
        sc = compute_score(y) if compute_score(y) != np.inf else compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = (y if sc == compute_score(y) else a).copy()
        
        # Adaptive step
        sc_current = compute_score(a)
        gap = max(sc_current - sc_lb, 1e-10)
        gn = float(np.linalg.norm(gradient))
        if gn > 1e-15:
            step = min(step_init, gap / (gn * gn + 1e-30))
            step = max(step, gap / (gn * 200.0 + 1e-30))
        
        if verbose and it % 500 == 0:
            print(f"    NG step={it}: score={sc_current:.6f}, best={best_sc:.6f}, step={step:.6f}")
    
    return best_a


def peak_shaving(a, n_rounds=100, verbose=False):
    """
    Local peak shaving: for the current peak, try to reduce it by
    adjusting individual coordinates in the peak's support.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Get the support indices for this peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end < j_start:
            break
        
        # For each coordinate in the support, try reducing it
        # and putting the mass at a random other position
        improved = False
        for _ in range(5):  # Try a few random modifications per round
            # Pick a coordinate to reduce
            i = j_start + np.random.randint(j_end - j_start + 1)
            if a[i] < 1e-10:
                continue
            
            # Pick a target to increase
            j = np.random.randint(n)
            if j == i:
                continue
            
            # Try transferring delta from i to j
            delta = a[i] * np.random.uniform(0.1, 0.5)
            
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            
            sc_new = compute_score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved and verbose:
            if rnd % 20 == 0:
                print(f"    PeakShave round={rnd}: stuck at {best_sc:.6f}")
    
    return best_a


def lse_subgradient(a, n_iters=3000, top_m=20, step_init=0.03, verbose=False):
    """
    Subgradient on the LSE-smoothed max with top-m peaking.
    Uses Polyak step size and momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    momentum = 0.9
    v = np.zeros(n)  # velocity for momentum
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-m peaks with smooth weighting
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        vals = c[top_idx]
        # Exponential weighting: peaks closer to max get higher weight
        w = np.exp(-10.0 * (vals[0] - vals) / (vals[0] + 1e-20))
        w = w / w.sum()
        
        sg = np.zeros(n)
        for k, wi in zip(top_idx, w):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += wi * 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        gradient = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        
        # Momentum update
        v = momentum * v - step * gradient
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            v *= 0.5
            continue
        a = a_new / S_new
        
        # Adaptive step
        gap = max(sc - sc_lb, 1e-10)
        gn = float(np.linalg.norm(gradient))
        if gn > 1e-15:
            step = min(step_init, gap / (gn * gn + 1e-30))
            step = max(step, 1e-8)
        
        if verbose and it % 500 == 0:
            print(f"    LSE-SG step={it}: score={sc:.6f}, best={best_sc:.6f}")
    
    return best_a

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LSE gradient descent with continuation (varying sharpness),
Nesterov-accelerated multi-peak subgradient, and local peak shaving.
Key innovation: continuation from smooth (low-beta LSE) to sharp (high-beta)
objective to guide the optimizer through the landscape without getting stuck.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr(a, b):
    """
    Compute R[m] = sum_{j=0}^{n-1} a[j] * b[j+m] for m = 0, 1, ..., n-1.
    a has length n, b has length >= n (typically 2n-1).
    Uses FFT-based linear convolution with reversal trick.
    """
    n = len(a)
    nb = len(b)
    N_fft = 1 << (n + nb - 2).bit_length()
    if N_fft < n + nb - 1:
        N_fft = n + nb - 1
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N_fft)
    B = np.fft.rfft(b, N_fft)
    full = np.fft.irfft(AR * B, N_fft)
    return full[n - 1:n - 1 + n]


def lse_value_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score = 2n * LSE_beta(conv) / S^2
    LSE_beta(x) = (1/beta) * log(sum(exp(beta * x)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    LSE = c_max + log_sum / beta

    # Softmax weights: p_k = exp(beta * c_k) / sum(exp(beta * c_j))
    p = exp_shifted / exp_shifted.sum()

    # Grad of LSE w.r.t. a[m] = sum_k p[k] * d(conv[k])/da[m] = 2 * cross_corr(a, p)[m]
    grad_LSE = 2.0 * cross_corr(a, p)

    # score = 2n * LSE / S^2
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * LSE / S2
    grad = 2.0 * n * (grad_LSE * S - 2.0 * LSE) / S3

    return score, grad


def lse_gradient_descent(a, beta, n_steps=200, lr_init=0.05):
    """
    Gradient descent on LSE-smoothed score with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for step in range(n_steps):
        sc, grad = lse_value_grad(a, beta)
        if not np.isfinite(sc):
            break

        # Track best exact score
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        # Backtracking line search
        accepted = False
        lr_try = lr
        for bt in range(15):
            a_trial = a - lr_try * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                lr_try *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial, _ = lse_value_grad(a_trial, beta)
            if sc_trial < sc * (1.0 - 1e-7) or not np.isfinite(sc):
                a = a_trial
                lr = lr_try
                accepted = True
                break
            lr_try *= 0.5

        if not accepted:
            # Try a smaller fixed step
            a_trial = a - 0.01 * lr * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = float(a_trial.sum())
            if S_trial > 1e-12:
                a = a_trial / S_trial
            lr *= 0.5

    return best_a


def nesterov_subgradient(a, n_iters=3000, top_m=10, momentum=0.9,
                          step_init=0.01, weight_mode="linear"):
    """
    Nesterov-accelerated multi-peak subgradient method.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    y = a.copy()
    x_prev = a.copy()
    step = step_init

    for it in range(n_iters):
        c_y = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M_y = float(c_y.max())

        # Find top-m peaks
        if top_m >= len(c_y):
            top_idx = np.arange(len(c_y))
        else:
            top_idx = np.argpartition(c_y, -top_m)[-top_m:]
            top_idx = top_idx[np.argsort(c_y[top_idx])[::-1]]

        vals = c_y[top_idx]
        if weight_mode == "exponential":
            w = np.exp(-5.0 * (vals[0] - vals) / (vals[0] + 1e-20))
        else:
            w = (vals / vals[0]) ** 2
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Weighted subgradient
        sg = np.zeros(n)
        for k, wi in zip(top_idx, w):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += wi * 2.0 * seg[::-1]

        S2_y = S_y * S_y
        S3_y = S2_y * S_y
        gradient = 2.0 * n * (sg / S2_y - 2.0 * M_y / S3_y)

        # Nesterov update
        x_curr = y - step * gradient
        x_curr = np.clip(x_curr, 0.0, 1000.0)
        S_curr = float(x_curr.sum())
        if S_curr < 1e-12:
            step *= 0.5
            continue
        x_curr = x_curr / S_curr
        y = x_curr + momentum * (x_curr - x_prev)
        y = np.clip(y, 0.0, 1000.0)
        S_y = float(y.sum())
        if S_y > 1e-12:
            y = y / S_y
        x_prev = x_curr
        a = x_curr

        # Track best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Polyak step size
        gap = max(sc - sc_lb, 1e-10)
        gn = float(np.linalg.norm(gradient))
        if gn > 1e-15:
            step_candidate = gap / (gn * gn + 1e-30)
            step = min(step_init, step_candidate)
            step = max(step, gap / (200.0 * gn + 1e-30))
            step = max(step, 1e-8)

    return best_a


def momentum_subgradient(a, n_iters=3000, top_m=20, momentum=0.85,
                          step_init=0.02, weight_decay=0.999):
    """
    Standard momentum subgradient (not Nesterov) for stability.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        vals = c[top_idx]
        w = np.exp(-8.0 * (vals[0] - vals) / (vals[0] + 1e-20))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        sg = np.zeros(n)
        for k, wi in zip(top_idx, w):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += wi * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        gradient = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)

        v = momentum * v - step * gradient
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            v *= 0.5
            continue
        a = a_new / S_new

        gap = max(sc - sc_lb, 1e-10)
        gn = float(np.linalg.norm(gradient))
        if gn > 1e-15:
            step = min(step_init, gap / (gn * gn + 1e-30))
            step = max(step, 1e-8)
            step *= weight_decay

    return best_a


def peak_shaving(a, n_rounds=200, rng=None):
    """
    Local peak shaving: randomly transfer mass from peak-contributing
    coordinates to random other positions.
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end < j_start:
            break

        # Try several random modifications
        improved = False
        for _ in range(8):
            # Pick source from peak support (weighted by contribution)
            seg_len = j_end - j_start + 1
            i_rel = rng.integers(0, seg_len)
            i = j_start + i_rel
            if a[i] < 1e-10:
                continue

            # Pick target (prefer positions far from current peaks)
            j = rng.integers(0, n)
            if j == i:
                continue

            # Determine transfer amount
            frac = rng.uniform(0.05, 0.4)
            delta = a[i] * frac

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new

            sc_new = compute_score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved and rnd % 50 == 0:
            pass  # No verbose to save time

    return best_a


def random_perturb_reopt(a, rng, n_perturbs=5, perturb_scale=0.1):
    """
    Try several random perturbations and keep the best.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_perturbs):
        a_pert = a.copy()
        # Random Gaussian perturbation
        noise = rng.standard_normal(n) * perturb_scale
        a_pert = a_pert + noise * a_pert  # relative perturbation
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = float(a_pert.sum())
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick LSE refinement
        a_ref = lse_gradient_descent(a_pert, 100, n_steps=50, lr_init=0.03)
        sc = compute_score(a_ref)
        if sc < best_sc:
            best_sc = sc
            best_a = a_ref.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """

    Phase 2: Nesterov + momentum subgradient
    Phase 3: Peak shaving + random perturb
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Choose length
    n = 6962
    if parent is not None and 4000 <= len(parent) <= 12000:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) / n

    def _update(sc, a, label):
        """Inline best-tracking (avoids closure issues)."""
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print("  [{}] NEW BEST: {:.8f} (n={}, t={:.0f}s)".format(
                label, sc, len(a), time.monotonic() - t_start))
            return True
        return False

    print("n={}, parent={}".format(n, "yes" if parent is not None else "no"))

    # ===== Phase 1: LSE continuation (50% of time) =====
    p1_end = t_start + 0.50 * (total_budget - 30.0)
    print("\n=== Phase 1: LSE continuation (deadline={:.0f}s) ===".format(p1_end - t_start))

    # Start from parent if available
    if parent is not None:
        a_cur = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        a_cur = np.maximum(a_cur, 0.0)
        S = float(a_cur.sum())
        if S > 1e-12:
            a_cur = a_cur / S
        sc = compute_score(a_cur)
        _update(sc, a_cur, "parent")
    else:
        a_cur = np.ones(n) / n

    # Continuation schedule
    beta_schedule = [10, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > p1_end:
            break
        t_beta_start = time.monotonic()
        remaining = p1_end - time.monotonic()
        # Estimate: each gradient step takes ~3ms for n~7000
        steps = max(30, min(500, int(remaining / 20 / 0.003)))
        lr = 0.05 / (1.0 + beta / 100.0)  # Smaller lr for larger beta
        a_cur = lse_gradient_descent(a_cur, beta, n_steps=steps, lr_init=lr)
        sc = compute_score(a_cur)
        _update(sc, a_cur, "LSE beta={}".format(beta))
        print("  beta={}: score={:.8f}, time={:.1f}s".format(
            beta, sc, time.monotonic() - t_beta_start))

    print("After Phase 1: best = {:.8f}".format(best_score))

    # ===== Phase 2: Subgradient methods (30% of time) =====
    p2_end = t_start + 0.80 * (total_budget - 30.0)
    print("\n=== Phase 2: Subgradient (deadline={:.0f}s) ===".format(p2_end - t_start))

    # Nesterov with various top_m
    for top_m in [5, 10, 20]:
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        # Estimate: each subgradient iter takes ~5ms
        n_iters = max(500, min(10000, int(remaining / 3 / 0.005)))
        a_sg = nesterov_subgradient(best_a, n_iters=n_iters, top_m=top_m,
                                     momentum=0.9, step_init=0.015,
                                     weight_mode="exponential")
        sc = compute_score(a_sg)
        _update(sc, a_sg, "NesterovSG top_m={}".format(top_m))

    # Momentum subgradient
    if time.monotonic() < p2_end - 60:
        remaining = p2_end - time.monotonic() - 60
        n_iters = max(500, min(8000, int(remaining / 0.005)))
        a_ms = momentum_subgradient(best_a, n_iters=n_iters, top_m=30,
                                     momentum=0.85, step_init=0.015)
        sc = compute_score(a_ms)
        _update(sc, a_ms, "MomentumSG")

    print("After Phase 2: best = {:.8f}".format(best_score))

    # ===== Phase 3: Peak shaving + perturbation (12% of time) =====
    p3_end = t_start + 0.92 * (total_budget - 30.0)
    print("\n=== Phase 3: Peak shaving (deadline={:.0f}s) ===".format(p3_end - t_start))

    if time.monotonic() < p3_end - 30:
        remaining = p3_end - time.monotonic() - 30
        n_rounds = max(100, int(remaining / 0.05))
        a_ps = peak_shaving(best_a, n_rounds=n_rounds, rng=rng)
        sc = compute_score(a_ps)
        _update(sc, a_ps, "peak-shave")

    # Random perturbation + re-optimize
    if time.monotonic() < p3_end - 10:
        a_rp = random_perturb_reopt(best_a, rng, n_perturbs=3, perturb_scale=0.05)
        sc = compute_score(a_rp)
        _update(sc, a_rp, "perturb-reopt")

    # More LSE refinement on best
    if time.monotonic() < p3_end:
        a_lse = lse_gradient_descent(best_a, 2000, n_steps=100, lr_init=0.01)
        sc = compute_score(a_lse)
        _update(sc, a_lse, "LSE-refine-2000")

    print("After Phase 3: best = {:.8f}".format(best_score))

    # ===== Phase 4: Final polish (8% of time) =====
    print("\n=== Phase 4: Final polish ===")

    if time.monotonic() < deadline - 20:
        # Short Nesterov with high momentum for fine-tuning
        a_fin = nesterov_subgradient(best_a, n_iters=2000, top_m=5,
                                      momentum=0.95, step_init=0.003,
                                      weight_mode="exponential")
        sc = compute_score(a_fin)
        _update(sc, a_fin, "final-nesterov")

    if time.monotonic() < deadline - 10:
        # One more peak shaving
        a_ps2 = peak_shaving(best_a, n_rounds=100, rng=rng)
        sc = compute_score(a_ps2)
        _update(sc, a_ps2, "final-shave")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print("\n" + "=" * 60)
    print("FINAL: internal={:.8f}, verified={:.8f}".format(best_score, verified))
    print("Length: {}, Time: {:.1f}s / {}s".format(len(out), elapsed, total_budget))
    print("=" * 60)
    return out
