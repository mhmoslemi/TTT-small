"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses continuation LSE optimization with momentum subgradient polishing,
adaptive active-set management, and multi-length exploration.
Key improvements: Nesterov momentum for polish phase, geometric beta continuation,
adaptive tolerance for active set, and multi-length strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k, tol_frac=0.01):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3
    best_x = a.copy()
    best_v = 1e10

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        if getattr(res, 'x', None) is not None:
            out = _norm(res.x)
            sc = _score(out)
            if sc < _score(best_x):
                best_x = out
    except Exception:
        pass
    return _score(best_x), best_x


def _momentum_subgrad(a, max_iters=100, deadline=None):
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    velocity = np.zeros(n)
    gamma = 0.7
    base_eta = 0.1

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        active = d >= dmax - 0.005 * max(dmax, 1e-10)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        velocity = gamma * velocity + (1.0 - gamma) * (grad / gnorm)
        vnorm = float(np.linalg.norm(velocity))
        if vnorm < 1e-14:
            break

        eta = base_eta / vnorm
        q = a - eta * velocity
        q = np.clip(q, 0.0, None)
        Sq = q.sum()
        if Sq < 1e-8:
            break
        q = q / Sq
        sc = _score(q)

        if sc < best_sc:
            best_sc = sc
            best_a = q.copy()

        a = q
    return best_sc, best_a

    a = _norm(a)

def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _anti_sym_perturb(a, rng, strength=0.2):

    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')

def _taper_init(n, rng, center_frac=0.6):
    x = np.linspace(0, 1, n)
    center = 0.5
    width = center_frac / 2.0
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    noise = rng.standard_normal(n) * 0.05
    a = a + noise
    a = np.clip(a, 0, None)
    return _norm(a)


def _sine_init(n, rng):
    x = np.linspace(0, 1, n)
    p = rng.uniform(0.5, 3.0)
    a = np.sin(np.pi * x) ** p
    a = a + rng.standard_normal(n) * 0.02
    a = np.clip(a, 0, None)
    return _norm(a)
