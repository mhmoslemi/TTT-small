Improved optimizer for flat-autocorrelation non-negative sequences.
Key improvements:
1. Backtracking FISTA LP solver with adaptive step size
2. Proper 1D golden-section line search for mixing parameter
3. Peak-shaving local search as complementary method
4. Multi-scale approach: optimize at multiple sequence lengths
5. High-beta L-BFGS polish with multiple restarts
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time
import math

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def fista_lp(a, M, n_iters, x_init=None, beta_max=1000.0):
    """FISTA LP solver with backtracking step size."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    # Adaptive step size with backtracking
    step = 1.0 / (10.0 * n * sum_a2)
    
    for it in range(n_iters):
        # Adaptive beta
        beta = min(beta_max, 10.0 + 100.0 * (it / n_iters))
        
        # Compute C @ x = conv(a, x)
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        # Gradient
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        
        # Backtracking step size
        new_step = step
        for _ in range(20):
            v = u - new_step * grad
            x_new = np.maximum(v, 0.0)
            # Check if objective improved (proxy: sum(x) increased)
            if float(x_new.sum()) >= float(x.sum()) - 1e-10 or new_step < 1e-30:
                break
            new_step *= 0.5
        step = new_step
        
        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x

"""
Improved optimizer for flat-autocorrelation non-negative sequences.

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_max = float(z.max())
    exp_z = np.exp(z - z_max)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        log_sum = 0.0
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        log_sum = z_max - z_max + np.log(exp_sum)
        log_sum = np.log(exp_sum)
        p = exp_z / exp_sum
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def fista_lp_improved(a, M, n_iters, x_init=None, beta_max=2000.0):
    """FISTA LP solver with corrected Lipschitz bound for n× faster convergence."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    norm_a_sq = float(np.sum(a * a))
    if norm_a_sq < 1e-15:
        norm_a_sq = 1.0 / n

    for it in range(n_iters):
        frac = it / max(1, n_iters)
        if frac < 0.1:
            beta = 20.0 + 80.0 * (frac / 0.1)
        elif frac < 0.3:
            beta = 100.0 + 400.0 * ((frac - 0.1) / 0.2)
        elif frac < 0.6:
            beta = 500.0 + 900.0 * ((frac - 0.3) / 0.3)
        else:
            beta = 1400.0 + (beta_max - 1400.0) * ((frac - 0.6) / 0.4)

        step = 1.0 / (beta * norm_a_sq + 1e-15)

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def golden_section_search(a, direction, n=20):
    """Golden section search for optimal mixing parameter t in [0, 1]."""
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c = a_hi - inv_phi * (a_hi - a_lo)
    d = a_lo + inv_phi * (a_hi - a_lo)

    S_a = float(a.sum())
    if S_a < 1e-12:
        return 0.5

    def obj(t):
        a_try = a + t * (direction - a)
        a_try = np.maximum(a_try, 0.0)
        S_t = float(a_try.sum())
        if S_t < 1e-12:
            return np.inf
        a_try = a_try / S_t
        return compute_score(a_try)

    f_c = obj(c)
    f_d = obj(d)

    for _ in range(n):
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - inv_phi * (a_hi - a_lo)
            f_c = obj(c)
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + inv_phi * (a_hi - a_lo)
            f_d = obj(d)

    t_opt = (a_lo + a_hi) / 2.0
    a_try = a + t_opt * (direction - a)
    a_try = np.maximum(a_try, 0.0)
    S_t = float(a_try.sum())
    if S_t > 1e-12:
        a_try = a_try / S_t
        return t_opt, a_try
    return 0.5, a

def peak_shaving(a, n_steps=300, rng=None):
    """Direct peak reduction via mass transfer from peak-contributing elements."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Elements contributing to peak at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        # Contribution of each element a[j] to the peak
        contributions = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contributions[j] = a[k_star - j]

        # Find element with highest contribution (reduce it)
        # And element with lowest contribution to peak (increase it)
        peak_elements = np.arange(j_start, j_end + 1)
        other_elements = np.setdiff1d(np.arange(n), peak_elements)

        if len(other_elements) == 0:
            break

        # Weight by current value (only reduce elements with mass)
        contrib_weighted = contributions[peak_elements] * a[peak_elements]
        if contrib_weighted.max() < 1e-15:
            break

        src = peak_elements[np.argmax(contrib_weighted)]

        # Choose destination: element that doesn't contribute to peak
        # and has room to absorb mass
        dest_candidates = other_elements
        if len(dest_candidates) > 100:
            # Pick from top candidates by "safety" (low value, far from peak)
            dist_from_peak = np.abs(dest_candidates.astype(np.float64) - (k_star - n + 1).astype(np.float64))
            dist_from_peak2 = np.abs(dest_candidates.astype(np.float64) - (k_star).astype(np.float64))
            safety = 1.0 / (a[dest_candidates] + 0.01) * (dist_from_peak + dist_from_peak2 + 1.0)
            top_idx = np.argsort(safety)[::-1][:50]
            dest_candidates = dest_candidates[top_idx]
        dest = dest_candidates[rng.integers(len(dest_candidates))]

        # Try several transfer amounts
        base_transfer = a[src] * 0.02
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            transfer = a[src] * frac
            if transfer < 1e-10:
                continue
            a_try = a.copy()
            a_try[src] -= transfer
            a_try[dest] += transfer
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a

        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=100, alpha=0.7):
    """Frequency-domain error reduction for flat autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(budget_s=2400):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2 using improved FISTA LP, golden section
    line search, peak shaving, and L-BFGS polish.
    """
    rng = np.random.default_rng(123)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")
        n_main = len(parent)
    else:
        n_main = 6962

    # ============ Phase 1: Quick ER + initial LP (15% time) ============
    phase1_end = t_start + 0.15 * total_budget
    print(f"\n--- Phase 1: Initial Optimization (t_end={phase1_end - t_start:.0f}s) ---")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Error reduction for quick improvement
    a_er = error_reduction(a_work, n_iters=60, alpha=0.7)
    sc = compute_score(a_er)
    report(sc, a_er, "ER-init")
    a_work = a_er.copy()

    # Quick LP with moderate iterations
    remaining_p1 = phase1_end - time.monotonic()
    if remaining_p1 > 30:
        fista_it = int(min(15000, max(5000, remaining_p1 / (n_main * 1e-5))))
        print(f"  LP quick: fista={fista_it}, remaining={remaining_p1:.0f}s")
        a_lp, sc_lp, x_warm = lp_alternating_improved(
            a_work, max_outer_iters=5, fista_iters=fista_it,
            time_limit=phase1_end - 5, x_warm=None, use_golden=True
        )
        sc = compute_score(a_lp)
        report(sc, a_lp, "LP-quick")
        a_work = a_lp.copy()
    else:
        x_warm = None
        a_work = best_a / max(best_a.sum(), 1e-12)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============ Phase 2: Main LP optimization (55% time) ============
    phase2_end = t_start + 0.70 * total_budget
    print(f"\n--- Phase 2: Main LP Optimization ---")
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    if x_warm is None:
        x_warm = np.ones(n_main) / n_main

    # Progressive FISTA iterations
    fista_budgets = [10000, 20000, 30000, 40000, 50000, 60000]
    x_warm = None

    for fi, fista_it in enumerate(fista_budgets):
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        # Estimate time per LP solve: ~n_iters * n * 2e-7 seconds
        est_per_solve = fista_it * n_main * 2.5e-7
        max_outer = max(2, min(30, int(remaining / max(1.0, est_per_solve))))
        max_outer = min(max_outer, 15 + fi * 5)

        print(f"  LP round {fi}: fista={fista_it}, max_outer={max_outer}, remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating_improved(
            a_work, max_outer_iters=max_outer, fista_iters=fista_it,
            time_limit=phase2_end - 10, x_warm=x_warm, use_golden=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP-r{fi}")
        a_work = a_out.copy()

    # Interleave peak shaving
    if time.monotonic() < phase2_end - 60:
        print("  Peak shaving mid...")
        a_ps = peak_shaving(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_ps)
        report(sc, a_ps, "peak-mid")
        if sc < best_score:
            S = float(a_work.sum())
            if S > 1e-12:
                a_work = a_work / S

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============ Phase 3: L-BFGS refinement (15% time) ============
    phase3_end = t_start + 0.85 * total_budget
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [100, 300, 800, 2000, 5000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(a_norm, beta, maxiter=5000, maxfun=200000)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, t={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============ Phase 4: Final polish (remaining time) ============
    print(f"\n--- Phase 4: Final Polish ---")

    # Peak shaving
    a_ps = peak_shaving(best_a, n_steps=400, rng=rng)
    sc = compute_score(a_ps)
    report(sc, a_ps, "peak-final")

    # One more LP round with best settings
    if time.monotonic() < deadline - 120:
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        remaining = deadline - time.monotonic() - 60
        fista_it_final = int(min(50000, max(10000, remaining / (n_main * 1e-5))))
        a_final, sc_final, _ = lp_alternating_improved(
            a_final_in, max_outer_iters=10, fista_iters=fista_it_final,
            time_limit=deadline - 60, x_warm=None, use_golden=True
        )
        sc = compute_score(a_final)
        report(sc, a_final, "LP-final")

    # Final L-BFGS
    if time.monotonic() < deadline - 40:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 8000, maxiter=6000, maxfun=300000)
        report(sc, a_opt, "LBFGS-final")

    # Final peak shaving
    if time.monotonic() < deadline - 10:
        a_final = peak_shaving(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "peak-ult")

    # ============ Final verification ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
