```python
def lp_step(f, M, max_constraints=5000, max_rounds=50):
    """
    Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for k in active_set
    Uses cutting plane to add violated constraints.
    """
    n = len(f)
    # Start with top constraints
    fb = np.convolve(f, np.ones(n)/n)  # initial guess for active set
    active = np.argsort(fb)[::-1][:min(max_constraints, 2*n-1)]
    
    for round in range(max_rounds):
        # Build constraint matrix A where A[j, i] = f coefficient for constraint j, variable i
        # Constraint j: (f*b)[active[j]] <= M
        # (f*b)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[k-i] * b[i]
        # So A[j, i] = f[k-j - i] if i is in range for constraint j
        
        # This is a banded matrix! Each constraint involves at most n consecutive variables.
        # Actually, for constraint at position k, it involves b[max(0,k-n+1)] through b[min(k,n-1)]
        # The coefficient for b[i] in constraint k is f[k-i]
        
        # For efficiency, we can use the fact that this is a Toeplitz-like structure
        
        # Solve LP using CVXPY with PDLP
        b = cvx.Variable(n, nonneg=True)
        constraints = []
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            # (f*b)[k] = sum_{i=i_start}^{i_end} f[k-i] * b[i]
            idx = np.arange(i_start, i_end + 1)
            coef = f[k - idx]
            constraints.append(coef @ b[idx] <= M)
        
        prob = cvx.Problem(cvx.Maximize(cvx.sum(b)), constraints)
        prob.solve(solver=cvx.PDLP, verbose=False)
        
        b_opt = b.value
        if b_opt is None:
            break
        b_opt = np.maximum(b_opt, 0)
        
        # Check for violated constraints
        fb_opt = np.convolve(f, b_opt)
        violations = np.where(fb_opt > M * (1 + 1e-6))[0]
        
        if len(violations) == 0:
            break
        
        # Add most violated constraints
        violations = violations[np.argsort(fb_opt[violations])[::-1][:100]]
        active = np.unique(np.concatenate([active, violations]))
        active = active[:max_constraints]
    
    return b_opt
```

```python
"""
Multi-strategy optimizer combining LP-based direction finding (cutting plane),
high-precision L-BFGS with adaptive LSE smoothing, spectral reshaping, and
peak-pair coordinate reduction. Designed to improve upon the parent construction
by escaping local minima through diverse exploration strategies.

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import cvxpy as cp
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_direction(f, M, max_constraints=300, max_rounds=30, timeout_s=60):
    """
    LP-based direction finding using cutting plane.
    Solves: max sum(b) s.t. b>=0, (f*b)[k] <= M for active constraints.
    Returns the optimal b (normalized to same sum as f) or None on failure.
    """
    n = len(f)
    S_f = float(f.sum())
    t_start = time.monotonic()

    # Initial active set: top peak locations of f*f
    ff = fast_conv(f)
    active = np.argsort(ff)[::-1][:min(max_constraints, len(ff))]

    b_opt = np.ones(n) / n  # fallback

    for round_idx in range(max_rounds):
        if time.monotonic() - t_start > timeout_s:
            break

        # Build constraint matrix (sparse)
        # Constraint k: (f*b)[k] = sum_i f[k-i] * b[i] <= M
        # For each active constraint k, valid i range is [max(0,k-n+1), min(k,n-1)]
        rows = []
        cols = []
        vals = []

        for ci, k in enumerate(active):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end < i_start:
                continue
            idx = np.arange(i_start, i_end + 1)
            coef = f[k - idx]
            # Only add non-zero coefficients
            nz = coef > 1e-15
            rows.extend([ci] * nz.sum())
            cols.extend(idx[nz].tolist())
            vals.extend(coef[nz].tolist())

        if len(rows) == 0:
            break

        A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(active), n))
        rhs = np.full(len(active), M)

        # Solve LP using CVXPY
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=cp.PDLP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
            except Exception:
                break

        if b_var.value is None:
            break

        b_opt = np.maximum(b_var.value, 0.0)
        if b_opt.sum() < 1e-12:
            break

        # Check for violated constraints
        fb = fast_conv_cross(f, b_opt)
        violations = fb > M * (1 + 1e-5)
        n_viol = int(violations.sum())

        if n_viol == 0:
            break

        # Add most violated constraints to active set
        viol_idx = np.where(violations)[0]
        viol_order = viol_idx[np.argsort(fb[viol_idx])[::-1][:max(50, n_viol)]]
        active = np.unique(np.concatenate([active, viol_order]))
        # Keep only top max_constraints*2 by violation magnitude
        if len(active) > max_constraints * 2:
            active_vals = fb[active]
            active = active[np.argsort(active_vals)[::-1][:max_constraints * 2]]

    S_b = b_opt.sum()
    if S_b < 1e-12:
        return None
    return b_opt / S_b * S_f  # rescale to match sum of f


def spectral_reshapel(a, alpha=0.1, n_iters=50):
    """
    Iterative spectral flattening: alternate between flattening the power spectrum
    and enforcing non-negativity.
    """
    n = len(a)
    a_work = a.astype(np.float64).copy()
    S = a_work.sum()
    if S < 1e-12:
        return a
    a_work = a_work / S

    best_a = a_work.copy()
    best_sc = compute_score(a_work)

    for it in range(n_iters):
        # Compute DFT
        F = np.fft.rfft(a_work)
        power = np.abs(F) ** 2

        # Target: flatten power spectrum (geometric mean)
        log_power = np.log(power + 1e-30)
        target_log = np.mean(log_power)
        target_power = np.exp(target_log)

        # Blend: move power spectrum toward flat
        new_power = power * (1 - alpha) + target_power * alpha
        new_mag = np.sqrt(new_power)

        # Keep phase, update magnitude
        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)

        # Inverse DFT
        a_new = np.fft.irfft(F_new, n)

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a_work = a_new

        # Decrease alpha over time
        alpha *= 0.98

    return best_a


def peak_pair_reduction(a, n_sweeps=20, top_k=10):
    """
    Directly reduce peaks by identifying top contributing pairs and
    redistributing mass to low-activity positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Find contributing pairs for the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        idx = np.arange(i_start, i_end + 1)
        contributions = a[idx] * a[k_star - idx]

        # Get top contributors
        local_order = np.argsort(contributions)[::-1][:top_k]
        top_local = i_start + local_order

        # Find "valley" positions: positions with low autocorrelation contribution
        # Use the complement: positions not in the top contributors
        all_local = np.arange(i_end - i_start + 1)
        bottom_local = np.setdiff1d(all_local, local_order)
        if len(bottom_local) == 0:
            break

        # Reduce top contributors, add to bottom
        total_reduction = 0.0
        for li in local_order:
            gi = i_start + li
            reduction = 0.15 * a[gi]  # reduce by 15%
            a[gi] -= reduction
            total_reduction += reduction

        # Distribute to bottom positions proportionally to their current value (or uniformly)
        per = total_reduction / len(bottom_local)
        for li in bottom_local[:min(len(bottom_local), 50)]:
            gi = i_start + li
            a[gi] += per

        a = np.maximum(a, 0.0)
        S_a = a.sum()
        if S_a < 1e-12:
            break
        a = a / S_a

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        elif best_sc - sc > 0.01:
            # Too much degradation, revert
            a = best_a.copy()
            break

    return best_a


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.9998):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break

        exponent = -eta * g_centered / gnorm * 0.5
        exponent = np.clip(exponent, -20, 20)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def perturbed_restart(a, rng, scale=0.03):

    n = len(a)
    strategy = rng.integers(0, 8)

    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):

            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg * seg[::-1]
        top_count = max(10, n // 20)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.2 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 5:
        # Swap-based: swap values between random pairs
        a_p = a.copy()
        n_swaps = max(1, n // 10)
        for _ in range(n_swaps):
            i, j = rng.choice(n, 2, replace=False)
            delta = scale * a.mean() * rng.random()
            a_p[i] += delta
            a_p[j] -= delta
    elif strategy == 6:
        # Spectral perturbation: modify random frequency bins
        F = np.fft.rfft(a)
        n_bins = max(1, len(F) // 20)
        bins = rng.choice(len(F), size=n_bins, replace=False)
        for b in bins:
            F[b] *= (1.0 + scale * rng.standard_normal())
        a_p = np.fft.irfft(F, n)
    else:
        # Local window perturbation
        w = max(1, n // 50)
        s = rng.integers(0, n - w)
        a_p = a.copy()
        a_p[s:s+w] *= (1.0 + scale * 2.0 * rng.standard_normal(w))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def coordinate_descent_polish(a, n_sweeps=20):
    """
    Coordinate descent: for each coordinate, find the best value
    that minimizes the peak autocorrelation (with renormalization).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            if a[i] < 1e-12:
                continue
            # Try reducing a[i] and redistributing
            old_val = a[i]
            best_local_sc = best_sc
            best_local_val = old_val

            for frac in [0.5, 0.0, 0.9, 0.7]:
                new_val = old_val * frac
                reduction = old_val - new_val
                if reduction < 1e-14:
                    continue

                a[i] = new_val
                # Redistribute to a few random positions (or peak-valley)
                c = fast_conv(a)
                k_min = int(np.argmin(c))
                # Add to position that minimizes the resulting peak
                # Simple heuristic: add to position far from current peak
                j_start = max(0, k_min - n + 1)
                j_end = min(k_min, n - 1)
                n_add = max(1, (j_end - j_start + 1) // 2)
                add_idx = np.random.default_rng(i + sweep * n).choice(
                    np.setdiff1d(np.arange(j_start, j_end + 1), i, assume_unique=False),
                    size=min(n_add, max(1, j_end - j_start)), replace=False)
                per = reduction / len(add_idx)
                for j in add_idx:
                    a[j] += per

                a = np.maximum(a, 0.0)
                S_a = a.sum()
                if S_a > 1e-12:
                    a = a / S_a
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local_val = new_val
                    improved = True

            a[i] = best_local_val
            if best_local_val != old_val:
                S_a = a.sum()
                if S_a > 1e-12:
                    a = a / S_a

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: LP direction + L-BFGS + spectral reshaping + peak reduction.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_s = max(60.0, float(budget_s))
    deadline = t_start + effective_s - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Start from parent
    if parent is not None:
        n_parent = len(parent)
        print(f"Parent length: {n_parent}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        n_parent = 4096
        parent_norm = np.ones(n_parent) * 0.5

    n_work = n_parent
    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: LP-based direction finding (20% time)
    # ==================================================================
    phase1_deadline = t_start + 0.20 * effective_s
    print(f"\n--- Phase 1: LP Direction Finding (n={n_work}) ---")

    a_curr = best_a.copy()
    n_lp_rounds = 5
    for r in range(n_lp_rounds):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break

        c = fast_conv(a_curr)
        M = float(np.max(c))
        S = float(a_curr.sum())

        # Solve LP for direction
        b_dir = lp_direction(a_curr, M, max_constraints=200, max_rounds=20,
                            timeout_s=min(45, remaining - 10))
        if b_dir is not None:
            # Convex combination update
            t_blend = 0.1 + 0.05 * rng.random()
            a_new = (1 - t_blend) * a_curr + t_blend * b_dir
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new > 1e-12:
                a_new = a_new / S_new
            sc = compute_score(a_new)
            report(sc, a_new, f"LP_round{r}")
            if sc < compute_score(a_curr):
                a_curr = a_new

        # Quick L-BFGS refinement after LP step
        if time.monotonic() < phase1_deadline - 30:
            sc, a_opt, _ = run_lbfgs(a_curr, beta=500, maxiter=5000, maxfun=20000)
            report(sc, a_opt, f"LP_post_LBFGS_{r}")
            a_curr = best_a.copy()

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: Aggressive L-BFGS with increasing beta (40% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * effective_s
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    beta_schedule = [200, 500, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        maxiter = int(min(50000, max(5000, remaining * 30)))
        maxfun = int(min(200000, max(10000, remaining * 100)))

        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")

        # If high beta, also try from a slightly perturbed start
        if beta >= 10000 and time.monotonic() < phase2_deadline - 120:
            a_pert = perturbed_restart(best_a, rng, scale=0.005)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            report(sc2, a_opt2, f"LBFGS beta={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Spectral reshaping + peak reduction (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * effective_s
    print(f"\n--- Phase 3: Spectral + Peak Reduction ---")

    # Spectral reshaping
    if time.monotonic() < phase3_deadline:
        a_spec = spectral_reshapel(best_a, alpha=0.05, n_iters=100)
        sc = compute_score(a_spec)
        report(sc, a_spec, "spectral")

    # Peak pair reduction
    if time.monotonic() < phase3_deadline - 30:
        a_ppr = peak_pair_reduction(best_a, n_sweeps=30, top_k=15)
        sc = compute_score(a_ppr)
        report(sc, a_ppr, "peak_pair")

    # Coordinate descent polish
    if time.monotonic() < phase3_deadline - 30:
        a_cd = coordinate_descent_polish(best_a, n_sweeps=5)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_descent")

    # L-BFGS after spectral/peak steps
    if time.monotonic() < phase3_deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, beta=100000, maxiter=20000, maxfun=100000)
        report(sc, a_opt, "LBFGS_post_phase3")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated local search (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * effective_s
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 12
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 60:
            break

        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Mix of methods
        method = rng.integers(0, 3)
        if method == 0:
            # Mirror descent + L-BFGS
            a_md = mirror_descent(a_pert, beta=200, n_iters=2000, eta0=0.04)
            sc_md = compute_score(a_md)
            report(sc_md, a_md, f"ILS#{trial} mirror")
            if time.monotonic() < phase4_deadline - 30:
                sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-10),
                                         beta=5000 + int(5000*rng.random()),
                                         maxiter=min(15000, int(remaining*10)),
                                         maxfun=min(50000, int(remaining*40)))
                report(sc, a_opt, f"ILS#{trial} lbfgs")
        elif method == 1:
            # LP + L-BFGS
            c = fast_conv(a_pert)
            M = float(np.max(c))
            b_dir = lp_direction(a_pert, M, max_constraints=100, max_rounds=10,
                                timeout_s=min(20, remaining - 20))
            if b_dir is not None:
                a_new = 0.8 * a_pert + 0.2 * b_dir
                a_new = np.maximum(a_new, 0.0)
                S_new = a_new.sum()
                if S_new > 1e-12:
                    a_new = a_new / S_new
                sc = compute_score(a_new)
                report(sc, a_new, f"ILS#{trial} LP")
            if time.monotonic() < phase4_deadline - 30:
                sc, a_opt, _ = run_lbfgs(best_a, beta=100000,
                                         maxiter=min(10000, int(remaining*8)),
                                         maxfun=min(50000, int(remaining*30)))
                report(sc, a_opt, f"ILS#{trial} lbfgs_high")
        else:
            # Peak reduction + spectral + L-BFGS
            a_pr = peak_pair_reduction(a_pert, n_sweeps=10, top_k=8)
            sc_pr = compute_score(a_pr)
            report(sc_pr, a_pr, f"ILS#{trial} peak")
            if time.monotonic() < phase4_deadline - 40:
                a_sp = spectral_reshapel(a_pr, alpha=0.03, n_iters=30)
                sc_sp = compute_score(a_sp)
                report(sc_sp, a_sp, f"ILS#{trial} spec")
                if time.monotonic() < phase4_deadline - 30:
                    sc, a_opt, _ = run_lbfgs(best_a, beta=300000,
                                             maxiter=min(8000, int(remaining*6)),
                                             maxfun=min(40000, int(remaining*25)))
                    report(sc, a_opt, f"ILS#{trial} lbfgs_vhigh")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final high-precision polishing (10% time)
    # ==================================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final Polishing ---")

    # Very high beta L-BFGS
    for beta in [500000, 2000000]:
        if time.monotonic() > phase5_deadline - 30:
            break
        remaining = phase5_deadline - time.monotonic()
        if remaining < 30:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta,
                                 maxiter=min(30000, int(remaining * 20)),
                                 maxfun=min(100000, int(remaining * 60)))
        report(sc, a_opt, f"final_beta={beta}")

    # One more LP step at the end
    if time.monotonic() < phase5_deadline - 60:
        c = fast_conv(best_a)
        M = float(np.max(c))
        b_dir = lp_direction(best_a, M, max_constraints=300, max_rounds=30,
                            timeout_s=40)
        if b_dir is not None:
            for t_blend in [0.05, 0.1, 0.2, 0.3]:
                a_new = (1 - t_blend) * best_a + t_blend * b_dir
                a_new = np.maximum(a_new, 0.0)
                S_new = a_new.sum()
                if S_new > 1e-12:
                    a_new = a_new / S_new
                sc = compute_score(a_new)
                report(sc, a_new, f"final_LP_t={t_blend}")

    # Final coordinate descent
    if time.monotonic() < phase5_deadline - 30:
        a_cd = coordinate_descent_polish(best_a, n_sweeps=3)
        sc = compute_score(a_cd)
        report(sc, a_cd, "final_coord")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
