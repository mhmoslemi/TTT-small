"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    # Gradient computation via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_direction(f, M):
    """
    Solve the LP: maximize sum(g) subject to f*g <= M (element-wise), g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    # Build constraint matrix A where (A @ g)[k] = (f*g)[k]
    # A is (2n-1) x n, with A[k,j] = f[k-j] if 0 <= k-j < n
    
    # Use sparse representation
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None


def solve_lp_cvxpy(f, M):
    """Solve LP using cvxpy."""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M, g >= 0]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.PDDLp)
    
    if g.value is not None:
        return np.maximum(g.value, 0)
    return None

    beta_growth = 1.003

    lr_decay = 0.9995
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.00005 * it)
        lr *= lr_decay

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)

            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)

        if gnorm < 1e-15:
            break

        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        beta = min(beta * beta_growth, beta_end)

def fienup_momentum(a, n_iters=200, momentum_init=0.0, momentum_end=0.4,
                    thresh_init=2.0, thresh_end=1.05, clip_init=0.3, clip_end=0.8):
    """Fienup with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_step(a, lr, n_peaks=5):
    """
    Subgradient step targeting top peaks.
    For each peak position, compute the subgradient and take a step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_conv(a)
    # Find top peak positions
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    
    # Weighted subgradient: weight each peak by its excess over mean
    c_mean = c.max()  # use max as reference
    weights = np.zeros(len(c))
    for k in peak_indices:
        k = int(k)
        # Subgradient contribution: for each i, grad[i] += 2 * a[k-i] (if valid)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            weights[i] += 2.0 * a[k - i]
    
    # Normalize and apply
    a_new = a - lr * weights / (S * S)
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new / S_new
    return a_new

def run_lbfgs(a0, beta, maxiter=2000, maxfun=30000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)

"""
LP-direction peak reduction + Nesterov LSE + Fienup + targeted coordinate descent.
Key innovation: Uses reduced linear programming (top-K autocorrelation constraints)
to find a "dual" direction that maximizes mass under bounded cross-correlation,
providing a fundamentally different optimization direction than gradient methods.
Combined with Nesterov-accelerated LSE smoothing and spectral Fienup for comprehensive
peak reduction.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da_i = 2 * sum_m a[m] * p[m+i]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def solve_reduced_lp(f, M, n_constraints=50, seed=0):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for top K peak positions, g >= 0.
    Uses only the top n_constraints autocorrelation peaks as active constraints.
    """
    n = len(f)
    c = fast_conv(f)
    # Select top K constraint indices
    if n_constraints >= len(c):
        k_indices = np.arange(len(c))
    else:
        # Mix of top peaks and some random for diversity
        top_k = np.argsort(c)[::-1][:int(n_constraints * 0.7)]
        rng = np.random.default_rng(seed)
        remaining = np.setdiff1d(np.arange(len(c)), top_k)
        if len(remaining) > 0:
            rand_k = rng.choice(remaining, size=n_constraints - len(top_k), replace=False)
            k_indices = np.concatenate([top_k, rand_k])
        else:
            k_indices = top_k[:n_constraints]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0 and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def nesterov_lse_adaptive(a, max_time=60.0, beta_start=10.0, beta_end=10000.0,
                           lr_init=0.01, lr_decay=0.9995, mu_init=0.9, mu_max=0.97):
    """Nesterov-accelerated projected gradient on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    beta_growth = 1.004
    lr = lr_init
    mu = mu_init

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(mu_max, mu_init + 0.00005 * it)
        lr *= lr_decay

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:

        a = a / S

def fienup_momentum(a, n_iters=200, momentum_init=0.0, momentum_end=0.4,
                    thresh_init=2.0, thresh_end=1.05, clip_init=0.3, clip_end=0.8):
    """Fienup/GER with momentum for spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        if it % 10 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def targeted_peak_descent(a, n_iters=200, n_peaks=10, n_top_contrib=20):
    """
    For each top peak, identify the coordinates that contribute most and
    reduce them while slightly boosting others. Adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    reduction_base = 0.03
    for it in range(n_iters):
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        # Collect all top contributors across peaks
        reduce_set = {}
        for k in peak_indices:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max < j_min:
                continue
            # Contribution of a[j] to c[k] is a[k-j]*a[j], but we're adjusting a[j]
            # So the "sensitivity" is a[k-j]
            sens = np.array([a[k - j] for j in range(j_min, j_max + 1)])
            n_local = min(n_top_contrib, len(sens))
            top_local = np.argsort(sens)[::-1][:n_local]
            for tl in top_local:
                j = j_min + int(tl)
                reduce_set[j] = reduce_set.get(j, 0.0) + 1.0
        
        if not reduce_set:
            break
        
        # Apply reduction
        reduction = reduction_base * (1.0 - 0.3 * it / n_iters)
        keys = list(reduce_set.keys())
        for j in keys:
            a[j] *= (1.0 - reduction * 0.5)
        
        # Slight boost to non-reduced elements
        reduce_arr = np.array(keys)
        other_mask = np.ones(n, dtype=bool)
        other_mask[reduce_arr] = False
        n_other = other_mask.sum()
        if n_other > 0:
            boost = reduction * len(keys) / n_other * 0.1
            a[other_mask] *= (1.0 + boost)
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        
        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=2000, maxfun=30000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    """Upsample a sequence using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def coarse_to_fine_optimize(a, n_coarse, max_time=30.0):
    """Optimize at coarse scale, then upsample back."""
    n = len(a)
    if n_coarse >= n:
        return a
    
    a_coarse = a[::n // n_coarse][:n_coarse] if n % n_coarse != 0 else a[::n // n_coarse]
    # Actually use interpolation for proper downsampling
    x_fine = np.linspace(0, 1, n)
    x_coarse = np.linspace(0, 1, n_coarse)
    a_coarse = np.interp(x_coarse, x_fine, a)
    a_coarse = a_coarse / a_coarse.sum()
    
    # Optimize at coarse scale
    a_opt = nesterov_lse_adaptive(a_coarse, max_time=max_time * 0.6,
                                   beta_start=20.0, beta_end=500.0)
    a_opt = fienup_momentum(a_opt, n_iters=100, thresh_init=2.0, thresh_end=1.1,
                            clip_init=0.3, clip_end=0.7)
    
    # Upsample back
    x_coarse2 = np.linspace(0, 1, len(a_opt))
    x_fine2 = np.linspace(0, 1, n)
    a_fine = np.interp(x_fine2, x_coarse2, a_opt)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine = a_fine / a_fine.sum()
    return a_fine


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Fienup warmup for spectral flattening
    2. LP-direction iteration using reduced constraints (key innovation)
    3. Nesterov LSE for smooth gradient refinement
    4. Targeted peak coordinate descent
    5. L-BFGS polish
    6. Coarse-to-fine multi-scale optimization
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        a = parent.copy()
        n = len(a)
    else:
        n = 6962
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2*np.pi*x) + 0.1 * np.cos(4*np.pi*x)
        a = np.maximum(a, 0.01)

    a = a / a.sum()
    update_best(compute_score(a), a, "init")

    # ============================================================
    # Phase 1: Fienup warmup (8% of time)
    # ============================================================
    phase1_end = t_start + 0.08 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup warmup ===")

    a_f = fienup_momentum(a, n_iters=200, momentum_init=0.0, momentum_end=0.3,
                          thresh_init=2.0, thresh_end=1.1, clip_init=0.3, clip_end=0.7)
    update_best(compute_score(a_f), a_f, "fienup-warm")

    # ============================================================
    # Phase 2: LP-direction iteration (37% of time)
    # ============================================================
    phase2_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LP-direction iteration ===")

    a_curr = best_a.astype(np.float64).copy()
    a_curr = a_curr / a_curr.sum()
    
    lp_iters_done = 0
    while time.monotonic() < phase2_end - 20:
        c_curr = fast_conv(a_curr)
        M_curr = float(c_curr.max())
        
        # Vary number of constraints for diversity
        k_cycle = 30 + 20 * (lp_iters_done % 3)
        seed_lp = seed + lp_iters_done * 7
        
        t_lp_s = time.monotonic()
        g = solve_reduced_lp(a_curr, M_curr, n_constraints=k_cycle, seed=seed_lp)
        t_lp_e = time.monotonic()
        
        if g is None:
            break
        
        S_g = float(g.sum())
        if S_g < 1e-12:
            break
        g = g / S_g
        
        # Adaptive blend with line search
        best_local_sc = compute_score(a_curr)
        best_local_t = 0.0
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.05, 0.4]:
            a_try = (1 - t_try) * a_curr + t_try * g
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try < 1e-12:
                continue
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local_t = t_try
        
        if best_local_t > 0:
            a_curr = (1 - best_local_t) * a_curr + best_local_t * g
            a_curr = np.maximum(a_curr, 0.0)
            a_curr = a_curr / a_curr.sum()
        
        sc_curr = compute_score(a_curr)
        if sc_curr < best_score:
            best_score = sc_curr
            best_a = a_curr.copy()
            print(f"  [LP iter {lp_iters_done}] NEW BEST: {sc_curr:.8f} (t_lp={t_lp_e-t_lp_s:.1f}s, blend={best_local_t:.3f})")
        
        lp_iters_done += 1
        
        # Occasionally apply Fienup to break symmetry
        if lp_iters_done % 5 == 0 and time.monotonic() < phase2_end - 30:
            a_fmix = fienup_momentum(a_curr, n_iters=50,
                                     momentum_init=0.0, momentum_end=0.2,
                                     thresh_init=2.5, thresh_end=1.5,
                                     clip_init=0.2, clip_end=0.5)
            sc_fmix = compute_score(a_fmix)
            if sc_fmix < best_score:
                a_curr = a_fmix
                best_score = sc_fmix
                best_a = a_curr.copy()
                print(f"  [LP+Fienup iter {lp_iters_done}] NEW BEST: {sc_fmix:.8f}")

    print(f"  LP phase done: {lp_iters_done} iterations")
    
    # Fill remaining phase 2 time with Nesterov
    if time.monotonic() < phase2_end - 15:
        time_left = phase2_end - time.monotonic() - 5
        a_nl = nesterov_lse_adaptive(best_a, max_time=time_left,
                                      beta_start=20.0, beta_end=5000.0)
        update_best(compute_score(a_nl), a_nl, "nesterov-p2")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Multi-config Nesterov LSE + perturbation (25% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Nesterov LSE refinement ===")

    configs = [
        {'beta_start': 50.0, 'beta_end': 10000.0, 'lr_init': 0.008},
        {'beta_start': 20.0, 'beta_end': 5000.0, 'lr_init': 0.012},
        {'beta_start': 100.0, 'beta_end': 20000.0, 'lr_init': 0.005},
    ]

    for cfg in configs:
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 30:
            break
        a_nl = nesterov_lse_adaptive(best_a, max_time=time_left * 0.85,
                                      beta_start=cfg['beta_start'],
                                      beta_end=cfg['beta_end'],
                                      lr_init=cfg['lr_init'])
        update_best(compute_score(a_nl), a_nl, "nesterov-p3")

    # Perturbation + Nesterov
    if time.monotonic() < phase3_end - 60:
        for perturb_scale in [0.02, 0.05]:
            if time.monotonic() > phase3_end - 30:
                break
            a_pert = best_a * (1.0 + perturb_scale * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, None)
            S_pert = a_pert.sum()
            if S_pert < 1e-12:
                continue
            a_pert = a_pert / S_pert
            time_left = phase3_end - time.monotonic()
            a_nl2 = nesterov_lse_adaptive(a_pert, max_time=time_left * 0.8,
                                           beta_start=30.0, beta_end=8000.0,
                                           lr_init=0.01)
            update_best(compute_score(a_nl2), a_nl2, f"nesterov-pert-{perturb_scale}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Fienup + Targeted Peak Descent + LP (20% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Fienup + Peak Descent + LP ===")

    # Fienup with different params
    a_f2 = fienup_momentum(best_a, n_iters=300,
                           momentum_init=0.0, momentum_end=0.5,
                           thresh_init=1.8, thresh_end=1.02,
                           clip_init=0.4, clip_end=0.9)
    update_best(compute_score(a_f2), a_f2, "fienup-p4a")

    a_f3 = fienup_momentum(best_a, n_iters=200,
                           momentum_init=0.1, momentum_end=0.3,
                           thresh_init=3.0, thresh_end=1.3,
                           clip_init=0.2, clip_end=0.6)
    update_best(compute_score(a_f3), a_f3, "fienup-p4b")

    # Targeted peak descent
    if time.monotonic() < phase4_end - 30:
        a_cd = targeted_peak_descent(best_a, n_iters=200, n_peaks=10, n_top_contrib=15)
        update_best(compute_score(a_cd), a_cd, "peak-descent")

    # Another LP round from the new best
    if time.monotonic() < phase4_end - 20:
        a_curr4 = best_a.copy()
        a_curr4 = a_curr4 / a_curr4.sum()
        for it4 in range(10):
            if time.monotonic() > phase4_end - 5:
                break
            c4 = fast_conv(a_curr4)
            M4 = float(c4.max())
            g4 = solve_reduced_lp(a_curr4, M4, n_constraints=40, seed=seed+100+it4)
            if g4 is None:
                break
            g4 = g4 / g4.sum()
            a_try4 = 0.7 * a_curr4 + 0.3 * g4
            a_try4 = np.maximum(a_try4, 0.0)
            a_try4 = a_try4 / a_try4.sum()
            sc4 = compute_score(a_try4)
            if sc4 < best_score:
                best_score = sc4
                best_a = a_try4.copy()
                a_curr4 = a_try4
                print(f"  [LP-p4 iter {it4}] NEW BEST: {sc4:.8f}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish + Coarse-to-fine (10% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: L-BFGS + Multi-scale ===")

    for beta in [500, 2000, 8000]:
        if time.monotonic() > phase5_end - 30:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=1500, maxfun=20000)
        update_best(sc, a_opt, f"lbfgs-{beta}")

    # Coarse-to-fine
    if time.monotonic() < phase5_end - 30:
        n_coarse = max(512, len(best_a) // 8)
        a_cf = coarse_to_fine_optimize(best_a, n_coarse,
                                        max_time=min(30, phase5_end - time.monotonic() - 10))
        update_best(compute_score(a_cf), a_cf, "coarse-fine")

    # Final micro-optimization: LSE with very high beta
    if time.monotonic() < phase5_end - 10:
        time_left = phase5_end - time.monotonic() - 5
        a_final = nesterov_lse_adaptive(best_a, max_time=min(20, time_left),
                                         beta_start=2000.0, beta_end=50000.0,
                                         lr_init=0.003, lr_decay=0.9999,
                                         mu_init=0.95, mu_max=0.98)
        update_best(compute_score(a_final), a_final, "final-lse-hibeta")

    # One last L-BFGS with highest beta
    if time.monotonic() < phase5_end - 10:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=1000, maxfun=15000)
        update_best(sc, a_opt, "final-lbfgs")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
