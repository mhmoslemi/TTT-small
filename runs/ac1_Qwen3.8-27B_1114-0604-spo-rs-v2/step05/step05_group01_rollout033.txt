"""

Uses LSE-smoothed L-BFGS-B, Adam optimizer, momentum subgradient with
adaptive step size, and O(n^2) coordinate descent polish.
Multi-length exploration with continuation in beta.

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

def run_lbfgs(a0, beta, deadline, maxfun):

    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = int(max(64, maxfun))

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 80,
    }

    def fun(a_in):
        return lse_score_grad(a_in, float(beta))

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def adam_lse_optimize(a0, beta, deadline, max_iters=5000):
    a = normalize_a(np.asarray(a0, dtype=np.float64).ravel()).copy()
    n = len(a)
    lr = 0.01
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = exact_score_a(a)

    # Adaptive LR schedule
    lr_start = 0.02
    lr_end = 0.001
    decay = max_iters

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        t += 1
        cur_lr = lr_end + (lr_start - lr_end) * np.exp(-3.0 * t / decay)

        sc, grad = lse_score_grad(a, beta)
        if not np.isfinite(sc):
            break

        m_hat = b1 * m_hat + (1 - b1) * grad
        v_hat = b2 * v_hat + (1 - b2) * (grad * grad)
        m_hat_corr = m_hat / (1 - b1 ** t)
        v_hat_corr = v_hat / (1 - b2 ** t)

        step = cur_lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        a_new = np.maximum(0.0, a - step)
        a_new = normalize_a(a_new)

        sc_new = exact_score_a(a_new)
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new

    return float(best_sc), best_a


def subgrad_momentum_optimize(a0, deadline, max_iters=2000, k_active=5):
    a = normalize_a(np.asarray(a0, dtype=np.float64).ravel()).copy()
    n = len(a)
    v = np.zeros(n)
    mu = 0.9
    best_a = a.copy()
    best_sc = exact_score_a(a)
    eta = 0.05
    eta_min = 0.001
    patience = 0
    patience_max = 200

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 3.0:
            break
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        if cmax <= 0:
            break

        # Get active weights (top-k)
        if k_active >= len(c):
            w = np.ones(len(c)) / len(c)
        else:
            idx = np.argpartition(c, len(c) - k_active)[len(c) - k_active:]
            w = np.zeros(len(c))
            top_vals = c[idx]
            w_top = top_vals / float(top_vals.sum())
            w[idx] = w_top

        h = corr_pair(a, w)
        hmax = float(np.max(np.abs(h)))
        if hmax < 1e-15:
            break

        # Nesterov momentum step
        v = mu * v - eta * h / (hmax + 1e-15)
        a_new = a + v
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_a(a_new)

        sc_new = exact_score_a(a_new)

        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_max:
                # Reset momentum, reduce step size
                v = np.zeros(n)
                eta = max(eta_min, eta * 0.5)
                patience = 0

        a = a_new

    return float(best_sc), best_a


def coordinate_descent_polish(a0, deadline, max_passes=50):
    a = normalize_a(np.asarray(a0, dtype=np.float64).ravel()).copy()
    n = len(a)
    b = fast_self_conv(a)
    S = float(a.sum())
    best_a = a.copy()
    best_sc = 2.0 * n * float(np.max(b)) / (S * S)

    for pass_i in range(max_passes):
        if deadline is not None and time.monotonic() > float(deadline) - 5.0:
            break

        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > float(deadline) - 5.0:
                break

            old_val = a[i]
            if old_val < 1e-15:
                # Try a small positive value
                t_trials = [1e-4 * S / n, 5e-4 * S / n, 1e-3 * S / n]
            else:
                t_trials = [0.0, old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                            old_val * 0.3, old_val * 2.0]

            # Find affected range
            k_lo = max(0, i - n + 1)
            k_hi = min(2 * n - 2, i + n - 1)
            affected_range = np.arange(k_lo, k_hi + 1)

            # Precompute max of unaffected
            if k_lo > 0:
                unaff_lo = b[:k_lo]
            else:
                unaff_lo = np.array([])
            if k_hi < 2 * n - 2:
                unaff_hi = b[k_hi + 1:]
            else:
                unaff_hi = np.array([])
            max_unaffected = 0.0
            if unaff_lo.size > 0:
                max_unaffected = max(max_unaffected, float(np.max(unaff_lo)))
            if unaff_hi.size > 0:
                max_unaffected = max(max_unaffected, float(np.max(unaff_hi)))

            best_t = old_val
            best_max_val = 2.0 * n * max(float(np.max(b)), max_unaffected) / (S * S)

            for t in t_trials:
                if t < 0:
                    continue
                # Compute affected b values at new t
                # b[k] = B0[k] + 2*t*a[k-i] (for linear terms where k-i valid and k != 2i)
                # b[2i] += t^2 - old_val^2 (quadratic term)

                ki = k_lo + 0  # k - i ranges from k_lo - i to k_hi - i
                k_indices = affected_range
                di = k_indices - i  # k - i values

                # Valid linear terms: 0 <= di < n and di != i (i.e., k != 2i)
                valid_linear = (di >= 0) & (di < n) & (di != i)
                # Quadratic term: k == 2i, i.e., di == i
                is_quad = (di == i) & (k_indices == 2 * i)

                # Current b values for affected range
                b_curr_affected = b[k_indices].copy()

                # Remove old contribution
                # Old linear contribution: 2 * old_val * a[di] where valid_linear
                if np.any(valid_linear):
                    b_curr_affected[valid_linear] -= 2.0 * old_val * a[di[valid_linear]]
                if np.any(is_quad):
                    b_curr_affected[is_quad] -= old_val * old_val

                # Add new contribution
                if np.any(valid_linear):
                    b_curr_affected[valid_linear] += 2.0 * t * a[di[valid_linear]]
                if np.any(is_quad):
                    b_curr_affected[is_quad] += t * t

                max_affected = float(np.max(b_curr_affected))
                max_overall = max(max_affected, max_unaffected)
                sc = 2.0 * n * max_overall / (S * S)

                if sc < best_max_val:
                    best_max_val = sc
                    best_t = t

            if best_t != old_val:
                # Update a and b
                a[i] = best_t
                k_indices = affected_range
                di = k_indices - i
                valid_linear = (di >= 0) & (di < n) & (di != i)
                is_quad = (di == i) & (k_indices == 2 * i)

                if np.any(valid_linear):
                    b[k_indices[valid_linear]] += 2.0 * (best_t - old_val) * a[di[valid_linear]]
                if np.any(is_quad):
                    b[k_indices[is_quad]] += best_t * best_t - old_val * old_val

                improved = True
                if best_max_val < best_sc:
                    best_sc = best_max_val
                    best_a = a.copy()
            else:
                # Try a finer search around old_val
                for mult in [0.9, 0.95, 1.05, 1.1]:
                    t = old_val * mult
                    if t < 0:
                        continue
                    k_indices = affected_range
                    di = k_indices - i
                    valid_linear = (di >= 0) & (di < n) & (di != i)
                    is_quad = (di == i) & (k_indices == 2 * i)

                    b_curr_affected = b[k_indices].copy()

                    if np.any(is_quad):
