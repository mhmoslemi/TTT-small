"""
Focused optimizer for step function height minimization.
Uses adaptive LSE-smoothed L-BFGS, LP-based free-direction search,
peak-structure-aware perturbation, and coordinate descent for iterated local search.
Key innovations over previous: (1) LP direction for constraint-aware exploration,
(2) coordinate descent on peak contributors, (3) adaptive multi-scale beta scheduling,
(4) targeted active-set growth heuristics.

import numpy as np
from scipy.optimize import minimize, linprog
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x):
    """LSE-smoothed score and gradient using global _current_beta."""
    beta = _current_beta
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -60.0))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_x = x[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RX = np.fft.rfft(rev_x, M2_fft)
    conv_result = np.fft.irfft(P * RX, M2_fft)[:M2]
    dL_dx = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_dx / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective. Returns (score, a_opt, result)."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-25, 'gtol': 1e-18,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 500})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out, res


def lp_direction(f, max_constraints=300):
    """Find g >= 0 maximizing sum(g) s.t. (f*g)[k] <= M for active lags k."""
    n = len(f)
    c_f = fast_conv(f)
    M = float(c_f.max())
    if M < 1e-15:
        return None
    # Select top active lags
    sorted_lags = np.argsort(c_f)[::-1]
    active_lags = sorted_lags[:max_constraints]
    # Only keep lags where c_f[k] > 0.1 * M to limit constraints
    active_lags = active_lags[c_f[active_lags] > 0.1 * M]
    if len(active_lags) == 0:
        return None
    # Build constraint matrix
    A_ub = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A_ub[i, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    b_ub = np.full(len(active_lags), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            g = res.x / res.x.sum()
            return np.clip(g, 0, 1000)
    except Exception:
        pass
    return None

            old_val = a[idx]

            best_val = old_val

                a = a / S

def subgradient_polish(a, n_iters=3000, deadline=None):

        if gnorm < 1e-15:
            break

        t = t_new
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    return best_a


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.9998):
    """Mirror descent with LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    for it in range(n_iters):
        sc, grad = lse_obj_grad(a)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30.0, 30.0)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay
    return best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS convergence to local minimum
    Phase 2: Iterated local search with diverse perturbations + LP directions
    Phase 3: High-beta L-BFGS polishing
    Phase 4: Subgradient + coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = compute_score(best_a)
        print(f"Parent: n={len(best_a)}, score={best_score:.10f}")
    else:
        n = 7000
        best_a = np.ones(n) * 0.5
        best_score = np.inf
        print(f"No parent found. Using uniform n={n}")

    n_work = len(best_a)
    elapsed = lambda: time.monotonic() - t_start

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed():.0f}s)")
            return True
        return False

    # Phase 1: Converge to local minimum (15% time)
    p1_deadline = t_start + 0.15 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Initial L-BFGS convergence (target: {p1_deadline - t_start:.0f}s) ===")

    beta_schedule_p1 = [200, 1000, 5000, 20000]
    for beta in beta_schedule_p1:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        mi = int(min(50000, max(5000, remaining * 25)))
        mf = int(min(200000, max(20000, remaining * 80)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=mi, maxfun=mf)
        report(sc, a_opt, f"P1 LBFGS b={beta}")

    print(f"Phase 1 done. Score: {best_score:.10f}")

    # Phase 2: Iterated local search (55% time)
    p2_deadline = t_start + 0.70 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Iterated Local Search (target: {p2_deadline - t_start:.0f}s) ===")

    iteration = 0
    lp_cooldown = 0  # Don't call LP every iteration (expensive)

    while time.monotonic() < p2_deadline:
        iteration += 1
        remaining = p2_deadline - time.monotonic()
        if remaining < 150:
            break

        # Diversify perturbation strategy
        r = rng.random()
        if r < 0.20:
            a_pert = peak_aware_perturb(best_a, rng, scale=0.008 + 0.025 * rng.random())
            label = "peak"
        elif r < 0.35:
            a_pert = smooth_perturb(best_a, rng, scale=0.005 + 0.02 * rng.random())
            label = "smooth"
        elif r < 0.50:
            a_pert = multiplicative_perturb(best_a, rng, scale=0.01 + 0.03 * rng.random())
            label = "mult"
        elif r < 0.65:
            a_pert = block_perturb(best_a, rng, scale=0.02 + 0.04 * rng.random())
            label = "block"
        elif r < 0.80:
            a_pert = gaussian_perturb(best_a, rng, scale=0.005 + 0.015 * rng.random())
            label = "gauss"
        else:
            # LP direction (expensive, use less frequently)
            if lp_cooldown > 0:
                lp_cooldown -= 1
                a_pert = peak_aware_perturb(best_a, rng, scale=0.015)
                label = "lp_fallback"
            else:
                lp_cooldown = 5
                a_pert = lp_direction(best_a, max_constraints=200)
                if a_pert is None:
                    a_pert = peak_aware_perturb(best_a, rng, scale=0.015)
                    label = "lp_none"
                else:
                    t_mix = 0.05 + 0.2 * rng.random()
                    a_pert = (1.0 - t_mix) * best_a + t_mix * a_pert
                    a_pert = np.clip(a_pert, 0.0, 1000.0)
                    a_pert = a_pert / a_pert.sum()
                    label = "lp"

        # L-BFGS refinement
        beta = int(2000 + 15000 * rng.random())
        mi = int(min(30000, max(5000, remaining * 8)))
        mf = int(min(100000, max(20000, remaining * 30)))
        sc, a_opt, _ = run_lbfgs(a_pert, beta, maxiter=mi, maxfun=mf)
        improved = report(sc, a_opt, f"ILS#{iteration} {label} b={beta}")

        # If improved, do quick polish
        if improved and time.monotonic() < p2_deadline - 90:
            # Quick coordinate descent
            cd_deadline = time.monotonic() + min(60, remaining * 0.1)
            a_cd = coordinate_descent(a_opt, n_rounds=3, top_k=15, deadline=cd_deadline)
            sc_cd = compute_score(a_cd)
            report(sc_cd, a_cd, f"ILS#{iteration} CD")

            # Quick high-beta L-BFGS
            if time.monotonic() < p2_deadline - 30:
                sc2, a_opt2, _ = run_lbfgs(best_a, 50000, maxiter=10000, maxfun=50000)
                report(sc2, a_opt2, f"ILS#{iteration} polish50k")

        # Every 5 iterations, try mirror descent for exploration
        if iteration % 5 == 0 and time.monotonic() < p2_deadline - 120:
            a_md = mirror_descent(a_pert, beta=200, n_iters=2000, eta0=0.04)
            sc_md = compute_score(a_md)
            report(sc_md, a_md, f"ILS#{iteration} mirror")
            if sc_md < best_score + 0.001:
                sc_md2, a_md2, _ = run_lbfgs(a_md, 10000, maxiter=15000, maxfun=80000)
                report(sc_md2, a_md2, f"ILS#{iteration} mirror+lbfgs")

    # Try a few different lengths (if time permits)
    if time.monotonic() < p2_deadline - 300:
        print(f"\n--- Trying alternative lengths ---")
        for n_try in [n_work - 200, n_work + 200, n_work + 500]:
            if n_try < 64 or time.monotonic() > p2_deadline - 180:
                continue
            # Interpolate
            x_old = np.linspace(0, 1, n_work)
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S < 1e-12:
                continue
            a_interp = a_interp / S
            sc_int = compute_score(a_interp)
            # Quick L-BFGS
            sc_alt, a_alt, _ = run_lbfgs(a_interp, 5000, maxiter=15000, maxfun=60000)
            report(sc_alt, a_alt, f"alt_len n={n_try}")

    print(f"Phase 2 done. Score: {best_score:.10f}")

    # Phase 3: High-beta L-BFGS polishing (10% time)
    p3_deadline = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 3: High-beta Polishing ===")

    for beta in [50000, 200000, 1000000]:
        if time.monotonic() > p3_deadline - 30:
            break
        remaining = p3_deadline - time.monotonic()
        if remaining < 30:
            break
        mi = int(min(50000, max(10000, remaining * 15)))
        mf = int(min(200000, max(50000, remaining * 50)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=mi, maxfun=mf)
        report(sc, a_opt, f"P3 b={beta}")

    # Phase 4: Final polish with subgradient + coordinate descent (10% time)
    p4_deadline = deadline
    print(f"\n=== Phase 4: Final Polish ===")

    # Subgradient
    a_sg = subgradient_polish(best_a, n_iters=3000, deadline=p4_deadline - 60)
    sc_sg = compute_score(a_sg)
    report(sc_sg, a_sg, "subgrad")

    # Coordinate descent on best
    a_cd = coordinate_descent(best_a, n_rounds=8, top_k=30, deadline=p4_deadline - 30)
    sc_cd = compute_score(a_cd)
    report(sc_cd, a_cd, "final_CD")

    # One more L-BFGS on the best found
    if time.monotonic() < p4_deadline - 40:
        sc_final, a_final, _ = run_lbfgs(best_a, 500000, maxiter=20000, maxfun=100000)
        report(sc_final, a_final, "final_LBFGS_500k")

    # Final verification
    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.10f}")
    print(f"Length: {len(best_a)}")
    print(f"Time used: {elapsed():.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    print(f"Verified score: {verified:.10f}")
    return out
