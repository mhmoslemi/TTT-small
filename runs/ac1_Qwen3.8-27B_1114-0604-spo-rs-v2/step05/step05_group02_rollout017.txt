    return lse_objective_and_grad(a, beta)

def peak_targeted_gradient(a, n_iters=5000, step_init=0.02):
    """
    Gradient steps targeting the specific peak.
    Uses the exact subgradient of the max function.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M_val = float(c[k_star])
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max at the peak: d/dc[da] at k_star
        # dc[k_star]/da[j] = 2*a[k_star-j] if 0<=k_star-j<n else 0
        # (as derived above, since it's the bilinear form)
        # Actually for the peak, we just need dc[k_star]/da
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end+1]
            sg = np.zeros(n)
            sg[j_start:j_end+1] = 2.0 * seg[::-1]
        else:
            sg = np.zeros(n)
        
        # Gradient of 2n*M/S^2
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Adaptive step size (descent direction check)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            step *= 0.5
            continue
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        
        if sc_new < sc:
            a = a_new
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7
            if step < 1e-10:
                break
    
    return best_a

def random_perturbation_search(a, n_trials=10000, perturb_radius=0.01, n_perturb=10):
    """
    Randomly perturb a few coordinates and keep improvements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        # Pick random coordinates to perturb
        idx = rng.integers(0, n, size=min(n_perturb, n))
        a_trial = a.copy()
        for j in idx:
            delta = rng.uniform(-perturb_radius, perturb_radius) * max(0.01, a[j])
            a_trial[j] = max(0.0, a[j] + delta)

        a_trial = a_trial / S_t
        sc = compute_score(a_trial)

            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()  # accept and continue from here
            # Decrease radius when improving
            perturb_radius *= 0.99
        elif trial % 1000 == 0:
            # Occasionally increase radius to explore
            perturb_radius *= 1.01
    
    return best_a

def block_coordinate_descent(a, block_size=8, n_sweeps=100):

    if S < 1e-12:

        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            # Try scaling the block
            block = a[start:end].copy()
            cur_sc = best_sc
            best_scale = 1.0

                a_trial = a.copy()
                a_trial[start:end] = block * scale

                sc = compute_score(a_trial)
                if sc < cur_sc:
                    cur_sc = sc
                    best_scale = scale
            a[start:end] = block * best_scale
            S = float(a.sum())
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

Multi-stage optimization for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: GS-Fienup (flat convolution target), L-BFGS with LSE continuation (beta annealing),

Key innovation: LSE beta continuation from smooth to sharp, with numerical gradient verification.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """
    LSE-smoothed objective and gradient for L-BFGS.
    score_smooth = 2n * L / S^2 where L = logsumexp(beta*c)/beta approximates max(c).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_clipped = np.clip(shifted, -700, 700)
    exp_v = np.exp(shifted_clipped)
    exp_sum = float(exp_v.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # p_k = exp(beta * c_k) / sum(exp(beta * c_j)) = softmax weights
    p = exp_v / exp_sum
    
    # dL/da[j] = 2 * sum_{m=0}^{n-1} a[m] * p[m+j]
    # Use FFT: cross-correlation of a (len n) and p (len 2n-1)
    M3 = 3 * n - 2
    M3_fft = 1 << (M3 - 1).bit_length()
    FA = np.fft.rfft(a, M3_fft)
    FP = np.fft.rfft(p, M3_fft)
    cross = np.fft.irfft(FA * np.conj(FP), M3_fft)[:n]
    dL_da = 2.0 * cross
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad

    """Wrapper for scipy minimize."""
    return lse_objective_and_grad(a, beta)

    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def peak_subgradient(a, n_iters=3000, step_init=0.015):
    """
    Subgradient method targeting the current peak directly.
    Uses Polyak-like adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M_val = float(c[k_star])
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) at the peak k_star:
        # dc[k_star]/da[j] = 2*a[k_star-j] if 0<=k_star-j<n else 0
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end+1]
            sg[j_start:j_end+1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            step *= 0.5
            continue
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        
        if sc_new < sc:
            a = a_new
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.7
            if step < 1e-12:
                break
    
    return best_a


def multi_peak_subgradient(a, n_iters=3000, top_m=5, step_init=0.01):
    """Subgradient using top-M peaks with LSE-like weighting."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Use LSE weights for smoothed subgradient
        beta = 50.0
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Weighted subgradient
        sg = np.zeros(n)
        for k in range(len(c)):
            w = weights[k]
            if w < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end+1]
                sg[j_start:j_end+1] += w * 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            step *= 0.5
            continue
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        
        if sc_new < sc:
            a = a_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.8
    
    return best_a


def random_perturb_refine(a, n_trials=5000, radius_init=0.005, n_perturb=8, rng=None):
    """
    Random coordinate perturbation with acceptance and gradual radius reduction.
    """
    if rng is None:
        rng = np.random.default_rng(999)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    radius = radius_init
    no_improve_count = 0
    
    for trial in range(n_trials):
        idx = rng.integers(0, n, size=min(n_perturb, n))
        a_trial = a.copy()
        for j in idx:
            delta = rng.uniform(-radius, radius) * max(0.01, a[j])
            a_trial[j] = max(0.0, a[j] + delta)
        
        S_t = float(a_trial.sum())
        if S_t < 1e-12:
            continue
        a_trial = a_trial / S_t
        sc = compute_score(a_trial)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            radius *= 0.995
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 500:
                # Increase radius to escape
                radius = min(radius * 1.5, 0.1)
                no_improve_count = 0
    
    return best_a


def try_different_n(a, n_target, method='interp'):
    """Resample a sequence to a different length."""
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def verify_gradient(a, beta, eps=1e-7):
    """Numerically verify gradient correctness (for debugging)."""
    f0, g0 = lse_objective_and_grad(a, beta)
    g_num = np.zeros(len(a))
    for j in range(min(10, len(a))):
        a_p = a.copy()
        a_p[j] += eps
        f_p, _ = lse_objective_and_grad(a_p, beta)
        a_m = a.copy()
        a_m[j] = max(0, a[j] - eps)
        f_m, _ = lse_objective_and_grad(a_m, beta)
        g_num[j] = (f_p - f_m) / (2 * eps)
    diff = np.max(np.abs(g0[:10] - g_num))
    return diff


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage optimization:
    1. GS-Fienup for quick initial improvement
    2. L-BFGS with LSE continuation (beta: 50 -> 100000)
    3. Peak-targeted subgradient
    4. Random perturbation search
    5. Final high-beta L-BFGS polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(6962) * 0.5
    
    best_score = np.inf
    best_a = parent.copy()
    rng = np.random.default_rng(seed)
    
    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False
    
    n = len(best_a)
    parent_score = compute_score(best_a)
    print(f"Starting: n={n}, parent_score={parent_score:.8f}")
    update_best(parent_score, best_a, "parent")
    
    # === Phase 1: GS refinement (fast, ~1-2 min) ===
    print(f"\n=== Phase 1: GS refinement ===")
    for alpha in [0.6, 0.4, 0.2, 0.1]:
        if time.monotonic() > deadline - 1200:
            break
        t_gs = time.monotonic()
        a_gs = gs_flat_conv(best_a, n_iters=200, alpha=alpha)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"GS-flat a={alpha}")
        
        if time.monotonic() < deadline - 1200:
            a_gs2 = gs_phase_update(a_gs, n_iters=100, alpha=alpha*0.5)
            sc2 = compute_score(a_gs2)
            update_best(sc2, a_gs2, f"GS-phase a={alpha}")
    
    # Also try from a fresh random initialization at same length
    if time.monotonic() < deadline - 1200:
        a_rand = np.maximum(0.5 + 0.3 * rng.standard_normal(n), 0.01)
        a_rand = a_rand / a_rand.sum()
        a_gs_rand = gs_flat_conv(a_rand, n_iters=300, alpha=0.5)
        sc = compute_score(a_gs_rand)
        update_best(sc, a_gs_rand, "GS-random-init")
    
    # === Phase 2: L-BFGS continuation (main phase, ~60% of time) ===
    print(f"\n=== Phase 2: L-BFGS continuation ===")
    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 300:
            break
        
        # Determine maxfun based on remaining time
        time_left = deadline - time.monotonic()
        # Each L-BFGS eval ~5ms, so maxfun ~ time_left / 0.005 * 0.8 (safety factor)
        maxfun = int(min(200000, max(5000, time_left * 150)))
        maxiter = min(20000, maxfun // 3)
        
        t_beta = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        elapsed = time.monotonic() - t_beta
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        print(f"  beta={beta}: score={sc:.8f}, time={elapsed:.1f}s, improved={improved}")
        
        # If no improvement and beta is large, try a small random perturbation first
        if not improved and beta >= 5000 and time.monotonic() < deadline - 600:
            a_perturbed = best_a.copy()
            perturb_idx = rng.integers(0, n, size=n // 10)
            a_perturbed[perturb_idx] *= (1.0 + rng.uniform(-0.02, 0.02, size=len(perturb_idx)))
            a_perturbed = np.maximum(a_perturbed, 0.0)
            a_perturbed = a_perturbed / a_perturbed.sum()
            sc_p, a_opt_p = run_lbfgs(a_perturbed, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            update_best(sc_p, a_opt_p, f"LBFGS-perturbed b={beta}")
    
    # === Phase 3: Peak-targeted subgradient ===
    print(f"\n=== Phase 3: Subgradient methods ===")
    if time.monotonic() < deadline - 120:
        t_sg = time.monotonic()
        a_sg = peak_subgradient(best_a, n_iters=5000, step_init=0.01)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "peak-SG")
        print(f"  Peak SG: {sc:.8f} in {time.monotonic()-t_sg:.1f}s")
    
    if time.monotonic() < deadline - 120:
        t_mp = time.monotonic()
        a_mp = multi_peak_subgradient(best_a, n_iters=3000, top_m=5, step_init=0.008)
        sc = compute_score(a_mp)
        update_best(sc, a_mp, "multi-peak-SG")
        print(f"  Multi-peak SG: {sc:.8f} in {time.monotonic()-t_mp:.1f}s")
    
    # === Phase 4: Random perturbation search ===
    print(f"\n=== Phase 4: Random perturbation ===")
    if time.monotonic() < deadline - 60:
        t_rp = time.monotonic()
        a_rp = random_perturb_refine(best_a, n_trials=10000, radius_init=0.003,
                                     n_perturb=5, rng=rng)
        sc = compute_score(a_rp)
        update_best(sc, a_rp, "random-perturb")
        print(f"  Random perturb: {sc:.8f} in {time.monotonic()-t_rp:.1f}s")
    
    # === Phase 5: Final high-beta L-BFGS polish ===
    print(f"\n=== Phase 5: Final polish ===")
    for beta in [100000, 200000, 500000]:
        if time.monotonic() > deadline - 30:
            break
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break
        maxfun = int(min(100000, time_left * 100))
        maxiter = min(10000, maxfun // 3)
        
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        print(f"  Final L-BFGS beta={beta}: {sc:.8f}")
        update_best(sc, a_opt, f"final b={beta}")
    
    # === Phase 6: One more GS + L-BFGS cycle ===
    if time.monotonic() < deadline - 60:
        a_cycle = gs_flat_conv(best_a, n_iters=100, alpha=0.15)
        sc_gs = compute_score(a_cycle)
        update_best(sc_gs, a_cycle, "cycle-GS")
        
        if time.monotonic() < deadline - 30:
            sc, a_opt = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=50000)
            update_best(sc, a_opt, "cycle-LBFGS")
    
    # === Final output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
