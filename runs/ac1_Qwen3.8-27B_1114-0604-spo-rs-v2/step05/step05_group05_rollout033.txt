    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)

    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)

"""
Adam-based LSE and Top-K peak minimization with multi-start and spectral shaping.
Improvements over previous:
1. Adam optimizer (per-coordinate adaptive LR) instead of Nesterov
2. Soft Top-K objective for denser gradient signal
3. Perturbation-annealing to escape local minima
4. Wider hyperparameter sweep
5. Better time allocation
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev))[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    L = c_max + float(np.log(exp_s.sum())) / beta
    S2 = S * S
    S3 = S2 * S
    return 2.0 * n * L / S2, (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)


def topk_score_grad(a, K, sharpness):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    w = np.exp(sharpness * (c - c_max))
    # Keep only top K
    thresh = np.partition(w, -K)[-K]
    w[w < thresh] = 0.0
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    w = w / w_sum
    L = float(w @ c)
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    w_rev = np.zeros(M_fft)
    w_rev[:2*n-1] = w[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(w_rev))[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    S2 = S * S
    S3 = S2 * S
    return 2.0 * n * L / S2, (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)


def adam_optimize(a, max_time, lr, beta_init, beta_final, K=1, sharpness=5.0,
                  m1=0.9, m2=0.999, eps=1e-8, use_topk=False):
    """Adam optimizer on LSE or Top-K smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < max_time:
        t += 1
        # Beta schedule (exponential ramp)
        frac = min(1.0, t / max(1, int(max_time / 0.002)))
        beta = beta_init * (beta_final / beta_init) ** frac if use_topk is False else sharpness
        
        if use_topk:
            _, grad = topk_score_grad(a, K, sharpness)
        else:
            _, grad = lse_score_grad(a, beta)
        
        # Adam update
        m = m1 * m + (1.0 - m1) * grad
        v = m2 * v + (1.0 - m2) * grad * grad
        m_hat = m / (1.0 - m1**t)
        v_hat = v / (1.0 - m2**t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        
        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def fienup_shaping(a, n_iters, thresh, clip, momentum_end=0.0):
    """Fienup spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2*n - 1).bit_length()
    v = np.zeros(n)
    
    for it in range(n_iters):
        momentum = momentum_end * (it / max(1, n_iters - 1))
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        above = P > P_mean * thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        cur = np.sqrt(P[above])
        factor = 1.0 - clip + clip * (target / np.maximum(cur, 1e-10))
        factor = np.clip(factor, 0.01, 1.0)
        scale[above] = factor
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=2000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxiter*10}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out

    print(f"\n=== Phase 1: Quick screening (to {p1_end - t0:.0f}s) ===")

    # 1a: Adam + LSE on parent
    a1 = adam_run(best_a, 12, 0.01, 'lse', 20, 3000, 1, 0)
    update_best(a1, "adam-lse-parent")

    # 1b: Adam + LSE on perturbed parent
    a_pert = perturb_seq(best_a, rng, 0.03)
    a2 = adam_run(a_pert, 12, 0.01, 'lse', 20, 3000, 1, 0)
    update_best(a2, "adam-lse-pert")

    # 1c: Adam + TopK on parent
    a3 = adam_run(best_a, 12, 0.01, 'topk', 0, 0, K=10, sharpness=3.0)
    update_best(a3, "adam-topk-parent")

    # 1d: Fienup
    a4 = fienup_run(best_a, 100, 1.5, 0.5, 0.0)
    update_best(a4, "fienup-quick")

    print(f"After Phase 1: best={best_sc:.8f}")

    # ================================================================
    # Phase 2: Extended Adam optimization (35% time)
    # ================================================================
    p2_end = t0 + 0.47 * (total - 20.0)
    print(f"\n=== Phase 2: Extended Adam (to {p2_end - t0:.0f}s) ===")

    # Multiple Adam+LSE configs
    lse_configs = [
        (0.01, 10, 5000),
        (0.005, 30, 10000),
        (0.02, 5, 2000),
        (0.008, 50, 8000),
    ]

    for i, (lr, b_init, b_fin) in enumerate(lse_configs):
        if time.monotonic() > p2_end - 15:
            break
        t_left = p2_end - time.monotonic() - 5
        a_new = adam_run(best_a, t_left, lr, 'lse', b_init, b_fin, 1, 0)
        update_best(a_new, f"adam-lse-{i}")

    # Adam + TopK configs
    topk_configs = [
        (0.01, 5, 5.0),
        (0.01, 10, 3.0),
        (0.005, 20, 2.0),
        (0.008, 8, 4.0),
    ]

    for i, (lr, K, sharp) in enumerate(topk_configs):
        if time.monotonic() > p2_end - 15:
            break
        t_left = p2_end - time.monotonic() - 5
        a_new = adam_run(best_a, t_left, lr, 'topk', 0, 0, K, sharp)
        update_best(a_new, f"adam-topk-{i}")

    print(f"After Phase 2: best={best_sc:.8f}")

    # ================================================================
    # Phase 3: Perturbation + re-optimize cycle (20% time)
    # ================================================================
    p3_end = t0 + 0.67 * (total - 20.0)
    print(f"\n=== Phase 3: Perturbation cycles (to {p3_end - t0:.0f}s) ===")

    n_pert_rounds = 6
    for round_i in range(n_pert_rounds):
        if time.monotonic() > p3_end - 20:
            break
        t_left = p3_end - time.monotonic() - 5
        if t_left < 15:
            break
        scale_per_iter = t_left / n_pert_rounds * 0.8

        # Perturb with decreasing scale
        scale = 0.005 + 0.03 * (1.0 - round_i / n_pert_rounds)
        a_pert = perturb_seq(best_a, rng, scale)

        # Alternate between LSE and TopK
        if round_i % 2 == 0:
            a_new = adam_run(a_pert, scale_per_iter, 0.01, 'lse', 20, 5000, 1, 0)
        else:
            a_new = adam_run(a_pert, scale_per_iter, 0.01, 'topk', 0, 0, 10, 3.0)

        update_best(a_new, f"pert-round-{round_i}")

        # Also try interpolation with second best
        if second_a is not None and second_sc < best_sc + 0.05:
            for t_interp in [0.2, 0.5, 0.8]:
                if time.monotonic() > p3_end - 10:
                    break
                a_interp = interpolate_seq(best_a, second_a, t_interp)
                a_interp_opt = adam_run(a_interp, min(5.0, p3_end - time.monotonic() - 5),
                                         0.005, 'lse', 100, 5000, 1, 0)
                update_best(a_interp_opt, f"interp-{t_interp}")

    print(f"After Phase 3: best={best_sc:.8f}")

    # ================================================================
    # Phase 4: Fienup + Adam cycle (13% time)
    # ================================================================
    p4_end = t0 + 0.80 * (total - 20.0)
    print(f"\n=== Phase 4: Fienup+Adam cycle (to {p4_end - t0:.0f}s) ===")

    fienup_params = [
        (1.3, 0.4, 200, 0.0),
        (1.5, 0.6, 150, 0.2),
        (1.8, 0.7, 100, 0.3),
        (2.0, 0.5, 120, 0.0),
        (1.6, 0.8, 80, 0.4),
    ]

    for i, (thresh, clip, niter, mom) in enumerate(fienup_params):
        if time.monotonic() > p4_end - 15:
            break
        t_left = p4_end - time.monotonic() - 10
        if t_left < 10:
            break
        t_split = t_left * 0.5
        a_f = fienup_run(best_a, int(niter * min(1.0, t_split / 60.0)), thresh, clip, mom)
        sc_f = update_best(a_f, f"fienup-4-{i}")
        # Adam polish after Fienup
        t_adam = min(10.0, p4_end - time.monotonic() - 5)
        if t_adam > 3:
            a_pol = adam_run(a_f, t_adam, 0.01, 'lse', 100, 5000, 1, 0)
            update_best(a_pol, f"fienup4-adam-{i}")

    print(f"After Phase 4: best={best_sc:.8f}")

    # ================================================================
    # Phase 5: Peak shaving + final polish (15% time)
    # ================================================================
    p5_end = deadline
    print(f"\n=== Phase 5: Peak shaving + L-BFGS polish ===")

    # Peak shaving
    if time.monotonic() < p5_end - 30:
        t_shave = min(20.0, p5_end - time.monotonic() - 25)
        n_sweeps = int(t_shave / 2.0)
        a_shave = peak_shave(best_a, n_iters=n_sweeps * 3, n_peaks=5, strength=0.2)
        update_best(a_shave, "peak-shave")
        a_shave2 = peak_shave(best_a, n_iters=n_sweeps * 2, n_peaks=10, strength=0.1)
        update_best(a_shave2, "peak-shave-10")

    # L-BFGS polish
    lbfgs_betas = [500, 2000, 5000, 10000, 20000, 50000]
    for beta in lbfgs_betas:
        if time.monotonic() > p5_end - 10:
            break
        sc, a_opt = lbfgs_run(best_a, beta, maxiter=1500)
        update_best(a_opt, f"lbfgs-b{beta}")

    # Final Adam with very high beta
    if time.monotonic() < p5_end - 5:
        t_final = p5_end - time.monotonic() - 3
        a_fin = adam_run(best_a, t_final, 0.003, 'lse', 10000, 100000, 1, 0)
        update_best(a_fin, "final-adam-hibeta")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t0
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_sc:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
