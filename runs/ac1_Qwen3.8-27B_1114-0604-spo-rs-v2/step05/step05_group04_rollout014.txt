Optimized search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses hybrid spectral Fienup + LSE-smoothed L-BFGS + peak-directed refinement.

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_peak_directed(a, n_iters=400, strength=0.2, top_lags=5):
    """Fienup with peak-directed spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    m_vals = np.arange(len(a) + 1)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idxs = np.argsort(c)[::-1][:top_lags]
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        P_mean = float(P.mean())
        
        # Peak-directed weights
        total_weight = np.zeros_like(P)
        for k_star in top_idxs:
            cos_w = np.cos(2.0 * np.pi * m_vals * int(k_star) / M_fft)
            total_weight += np.maximum(cos_w, 0)
        if total_weight.max() > 0:
            total_weight = total_weight / total_weight.max()
        
        # Combine with standard flattening
        scale = 1.0 - strength * total_weight
        above = P > P_mean * 2.0
        scale[above] *= (1.0 - 0.3 * strength)
        scale = np.clip(scale, 0.1, 1.5)
        
        A_new = A * np.sqrt(scale)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
    
    return best_a

def fienup_standard(a, n_iters=300, gamma=0.8, top_frac=0.2):
    """Standard Fienup: reduce top P bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale = np.ones_like(P)
            scale[above] = np.sqrt(np.maximum(ratio, 0.001)) * gamma
            A_mod = A * scale
            a_new = np.fft.irfft(A_mod, M_fft)[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
