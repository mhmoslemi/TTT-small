        a = a / S_new

        # Adaptive step
        gap = max(sc - sc_lb, 1e-8)
        grad_norm2 = float(np.dot(grad, grad))
        step_adaptive = gap / (grad_norm2 + 1e-20)
        step = min(step, min(0.05, max(step_adaptive, 1e-6)))
        step *= step_decay

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_chase(a0, n_iters=3000, top_k=10, lr=0.01):
    """Directly minimize the peak by projecting gradient of max onto feasible set."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    beta_mom = 0.9

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-k peaks with strong weighting
        top_idx = np.argsort(c)[::-1][:top_k]
        c_max = c[top_idx[0]]
        # Exponential weighting
        w = np.exp(-5.0 * (c_max - c[top_idx]) / (c_max + 1e-15))
        w = w / w.sum() * top_k  # normalize so sum = top_k

        sg = np.zeros(n)
        for j, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * float(w[j]) * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Adam-like update
        v = beta_mom * v + (1 - beta_mom) * grad
        v_hat = v / (1 - beta_mom ** (it + 1))
        a -= lr * v_hat
        a = np.maximum(a, 0.0)

        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new

    return best_a


def build_diverse_inits(n, rng, parent):
    """Build diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.05, 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * x * 3), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 5.0))
    # Chirp-like
    inits.append(np.abs(np.exp(2j * np.pi * 5.0 * x * x)) + 0.1)
    # Random phase
    phases = rng.standard_normal(n)
    phases = np.maximum(phases, 0.01)
    inits.append(phases / phases.sum())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Budget: {budget_s}s, deadline at t={deadline - t_start:.0f}s")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ', score=' + str(compute_score(parent)) + ')' if parent is not None else 'no'}")

    # Evaluate parent
    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # Determine lengths to try
    n_parent = len(parent) if parent is not None else 6962
    lengths = [n_parent]
    if n_parent < 16384:
        lengths.append(16384)
    lengths.append(12288)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to try: {lengths}")

    # ================================================================
    # Phase 1: Fienup from multiple inits (35% time)
    # ================================================================
    phase1_end = t_start + 0.35 * (budget_s - 20.0)
    print(f"\n=== Phase 1: Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_diverse_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()

            # DC-shift Fienup
            a_f = fienup_dc_shift(a0, n_iters=300, top_frac_init=0.05,
                                  top_frac_end=0.5, bot_frac=0.08,
                                  compress_strength=0.6, expand_strength=0.15,
                                  dc_shift_alpha=1.0)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"DCFIENUP n={n} i={i}")

            # Spectral equalization
            if time.monotonic() < phase1_end:
                a_eq = fienup_spectral_eq(a0, n_iters=200, alpha_init=0.2,
                                          alpha_end=0.8, target_dc_boost=1.3)
                sc2 = compute_score(a_eq)
                update_best(sc2, a_eq, f"SPEQ n={n} i={i}")

            # DC-shift with different params
            if time.monotonic() < phase1_end:
                a_f2 = fienup_dc_shift(a0, n_iters=200, top_frac_init=0.1,
                                       top_frac_end=0.3, bot_frac=0.15,
                                       compress_strength=0.4, expand_strength=0.3,
                                       dc_shift_alpha=0.8)
                sc3 = compute_score(a_f2)
                update_best(sc3, a_f2, f"DCFIENUP2 n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Chained Fienup + Subgradient (25% time)
    # ================================================================
    phase2_end = t_start + 0.60 * (budget_s - 20.0)
    print(f"\n=== Phase 2: Chain + Subgradient (deadline t={phase2_end - t_start:.0f}s) ===")

    a_cur = best_a.copy() / best_a.sum()

    # Chain: Fienup -> Subgradient -> Fienup -> Subgradient
    for round_idx in range(3):
        if time.monotonic() > phase2_end:
            break

        # Fienup refinement
        a_f = fienup_dc_shift(a_cur, n_iters=200, top_frac_init=0.05,
                              top_frac_end=0.4, compress_strength=0.5,
                              expand_strength=0.1, dc_shift_alpha=1.0)
        sc = compute_score(a_f)
        improved = update_best(sc, a_f, f"CHAIN-F n={len(a_cur)} r={round_idx}")
        if improved:
            a_cur = a_f.copy() / a_f.sum()

        # Momentum subgradient
        if time.monotonic() < phase2_end:
            a_sg = momentum_subgradient(a_cur, n_iters=3000, top_m=20,
                                        momentum=0.85, step_init=0.02)
            sc = compute_score(a_sg)
            improved = update_best(sc, a_sg, f"CHAIN-SG n={len(a_cur)} r={round_idx}")
            if improved:
                a_cur = a_sg.copy() / a_sg.sum()

        # Peak chase
        if time.monotonic() < phase2_end:
            a_pc = peak_chase(a_cur, n_iters=2000, top_k=10, lr=0.005)
            sc = compute_score(a_pc)
            update_best(sc, a_pc, f"CHAIN-PC n={len(a_cur)} r={round_idx}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: L-BFGS with beta annealing (30% time)
    # ================================================================
    phase3_end = t_start + 0.90 * (budget_s - 20.0)
    print(f"\n=== Phase 3: LBFGS annealing (deadline t={phase3_end - t_start:.0f}s) ===")

    betas = [50, 100, 300, 1000, 3000, 10000, 30000]
    a_cur = best_a.copy() / best_a.sum()

    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break

        # Scale iterations based on available time and sequence length
        n = len(a_cur)
        est_time_per_fun = n / 1e6 * 2  # rough estimate
        maxfun = max(500, int(remaining * 0.8 / max(est_time_per_fun, 0.001)))
        maxiter = min(maxfun // 3, 20000)

        print(f"  beta={beta}, maxiter={maxiter}, maxfun={maxfun}")
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            a_cur = a_opt.copy() / a_opt.sum()

    # Also try from parent at native length
    if parent is not None and len(parent) != len(best_a):
        if time.monotonic() < phase3_end:
            for beta in [200, 1000, 5000]:
                if time.monotonic() > phase3_end:
                    break
                remaining = phase3_end - time.monotonic()
                if remaining < 30:
                    break
                maxfun = max(500, int(remaining * 0.7 / max(len(parent) / 1e6 * 2, 0.001)))
                maxiter = min(maxfun // 3, 10000)
                sc, a_opt = run_lbfgs(parent, beta, maxiter=maxiter, maxfun=maxfun)
                update_best(sc, a_opt, f"LBFGS-PARENT b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish (remaining time)
    # ================================================================
    print(f"\n=== Phase 4: Final polish ===")

    if time.monotonic() < deadline:
        # Short Fienup from best
        a_f = fienup_dc_shift(best_a, n_iters=100, top_frac_init=0.05,
                              top_frac_end=0.3, compress_strength=0.4,
                              dc_shift_alpha=1.0)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FINAL-FIENUP")

        # Short subgradient
        if time.monotonic() < deadline - 10:
            a_sg = momentum_subgradient(best_a, n_iters=2000, top_m=15,
                                        momentum=0.9, step_init=0.01)
            sc = compute_score(a_sg)
            update_best(sc, a_sg, "FINAL-SG")

        # Peak chase
        if time.monotonic() < deadline - 10:
            a_pc = peak_chase(best_a, n_iters=1500, top_k=5, lr=0.003)
            sc = compute_score(a_pc)
            update_best(sc, a_pc, "FINAL-PC")

        # One more LBFGS with high beta
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic() - 10
            if remaining > 30:
                n = len(best_a)
                maxfun = max(300, int(remaining / max(n / 1e6 * 2, 0.001)))
                maxiter = min(maxfun // 3, 10000)
                sc, a_opt = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxfun)
                update_best(sc, a_opt, "FINAL-LBFGS-50k")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
