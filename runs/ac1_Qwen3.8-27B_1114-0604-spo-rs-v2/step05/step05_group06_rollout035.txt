def _cross_corr_fft(a, W):
    """g[j] = sum_k W[k] * a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    nw = len(W)
    if n == 0 or nw == 0:
        return np.empty(n, dtype=np.float64)
    M = n + nw - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    WF = np.fft.rfft(W, N)
    full = np.fft.irfft(A * WF, N)[:M]
    return np.nan_to_num(full[n-1:2*n-1], nan=0.0, posinf=0.0, neginf=0.0)


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr_fft(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.006 * max(0.5, float(n) / 4096.0)
    mf = int(0.80 * time_limit / max(est_eval_t, 1e-4))
    mf = max(20, min(mf, maxfun_cap, 5000))
    maxiter = max(15, mf // 3)
    PEN = 5e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(obj, a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter, 'maxfun': mf,
                                'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30})
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=40, deadline=None, momentum_init=0.7):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    momentum = np.zeros(n)
    mu = momentum_init
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        # Adaptive active set: positions within relative tolerance of max
        tol = max(1e-4, 0.001 * dmax)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr_fft(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        # Nesterov momentum
        momentum = mu * momentum - grad / gnorm
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
            q = a + scale * momentum
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
                mu_new = min(0.95, mu + 0.01)
        if best_step is None:
            mu = max(0.3, mu * 0.9)
            continue
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    return best_sc, best_a


def _gs_init(n, rng, n_iters=150):
    """Gerchberg-Saxton: alternating projections for flat spectrum + non-negativity."""
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = float(n)
    target_mag = float(n) / np.sqrt(float(N))
    for k in range(1, N // 2 + 1):
        A[k] = target_mag * np.exp(1j * rng.uniform(0, 2.0 * np.pi))
    a = np.ones(n) / n
    for it in range(n_iters):
        A_cur = np.fft.rfft(a, N)
        A_new = np.zeros_like(A_cur)
        A_new[0] = float(n)
        for k in range(1, len(A_cur)):
            mag = abs(A_cur[k])
            if mag > 1e-12:
                phase = np.angle(A_cur[k])
                A_new[k] = target_mag * np.exp(1j * phase)
            else:
                A_new[k] = target_mag * np.exp(1j * rng.uniform(0, 2.0 * np.pi))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S
        a = a_new
    return a


def _spectral_init(n, rng):
    """Random spectral initialization with shaped power."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.2, 0.8)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _adaptive_optimize(a0, deadline, label, rng):
    """Adaptive K growth LSE optimization with periodic perturbations."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Adaptive K schedule: start small, grow based on landscape
    stages = [
        (2, 50.0),
        (5, 100.0),
        (10, 200.0),
        (20, 400.0),
        (40, 800.0),
        (80, 1600.0),
        (160, 3200.0),
        (320, 6400.0),
        (640, 12800.0),
    ]

    total_time = max(5.0, deadline - t0 - 15.0)
    # Allocate more time to later stages (they do more meaningful work)
    weights = [0.03, 0.04, 0.06, 0.08, 0.10, 0.12, 0.14, 0.16, 0.27]
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    perturbation_interval = 3  # perturb every 3 stages

    for si, (k, beta) in enumerate(stages):
        if time.monotonic() > deadline - 15.0:
            break
        stage_budget = weights[si] * total_time
        stage_end = min(time.monotonic() + stage_budget, deadline - 15.0)

        # Periodic perturbation to escape local minima
        if si > 0 and si % perturbation_interval == 0:
            a = _smooth_perturb(a, rng, strength=0.1)
            a = _norm(a)

        # Run L-BFGS-B in chunks
        n_calls = max(1, min(5, int(stage_budget / 8.0)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 1.0:
                break
            t_rem = stage_end - time.monotonic()
            if t_rem < 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 300)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if si >= 3:
                    print(f"    [{label}] K={k}, beta={beta:.0f}: {best_sc:.8f}")
            a = _norm(a_new)

    # Final subgradient polish
    if time.monotonic() < deadline - 10.0:
        sc_sg, a_sg = _subgrad_polish(a, max_iters=20, deadline=deadline - 5.0)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    elapsed = time.monotonic() - t0
    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={elapsed:.0f}s")
    return best_sc, best_a


def _multilevel_optimize(a0, n_target, deadline, label, rng):
    """Coarse-to-fine: optimize at n/4 first, then upscale and refine at n_target."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)

    if n_target > 4096 and time.monotonic() < deadline - 90.0:
        n_small = max(1024, n_target // 4)
        a_small = _upsample(a, n_small)
        sc_small = _score(a_small)
        print(f"  [{label}] multilevel: n_small={n_small}, score={sc_small:.8f}")

        # Quick optimization at small scale
        t0 = time.monotonic()
        time_per_stage = min(15.0, (deadline - 90.0 - t0) / 5.0)
        for k_s, beta_s in [(3, 80), (10, 300), (30, 1200)]:
            if time.monotonic() > deadline - 90.0:
                break
            sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, time_per_stage, 150)
            if sc_s < sc_small:
                sc_small = sc_s
                a_small = a_s

        # Upscale back
        a_up = _norm(_upsample(a_small, n_target))
        sc_up = _score(a_up)
        if sc_up < best_sc:
            best_sc = sc_up
            best_a = a_up.copy()
            print(f"  [{label}] multilevel improved: {sc_up:.8f}")
        a = a_up

    # Full-scale optimization
    sc_full, a_full = _adaptive_optimize(a, deadline, label, rng)
    if sc_full < best_sc:
        best_sc = sc_full
        best_a = a_full.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent available: {parent is not None}, budget={budget_s}s")

    # Try multiple lengths
    lengths_to_try = [8192, 4096, 12288]
    if parent is not None:
        pn = len(parent)
        # Put parent's length first
        lengths_to_try = [ln for ln in lengths_to_try if ln == pn] + \
                         [ln for ln in lengths_to_try if ln != pn]

    best_overall_sc = float('inf')
    best_overall_seq = None

    # Allocate time across lengths
    # Give more time to the first (parent) length
    n_lengths = len(lengths_to_try)
    if n_lengths == 1:
        time_fracs = [1.0]
    elif n_lengths == 2:
        time_fracs = [0.70, 0.30]
    else:
        time_fracs = [0.50, 0.25, 0.25]

    main_time = deadline - t_start - 120.0  # reserve 120s for final polish

    for li, n_target in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 60.0:
            break
        sub_deadline = t_start + (sum(time_fracs[:li+1]) * main_time) + 60.0
        sub_deadline = min(sub_deadline, deadline - 120.0)

        print(f"\n=== Length n={n_target} (budget to {time.monotonic() + (sub_deadline - t_start):.0f}s) ===")

        # Generate diverse initializations for this length
        inits = []

        # 1. Parent (resampled)
        if parent is not None:
            inits.append(('parent', _upsample(parent, n_target)))
            inits.append(('parent_rev', _upsample(parent[::-1], n_target)))

        # 2. GS flat-spectrum inits
        for gi in range(4):
            try:
                inits.append((f'gs{gi}', _gs_init(n_target, rng, n_iters=100)))
            except Exception:
                pass

        # 3. Spectral inits
        for si in range(3):
            inits.append((f'spec{si}', _spectral_init(n_target, rng)))

        # 4. Parent + noise
        if parent is not None:
            p = _upsample(parent, n_target)
            for amp in (0.1, 0.25):
                inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

        # 5. Cosine-modulated
        x = np.linspace(0, 1, n_target)
        for f in (1, 3):
            for amp in (0.2, 0.5):
                arr = 1.0 + amp * np.cos(2 * np.pi * f * x)
                arr = np.clip(arr, 0, None)
                if arr.sum() > 0:
                    inits.append((f'cos{f}_{amp}', arr / arr.sum()))

        # Evaluate and rank
        scored = []
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr))
        scored.sort(key=lambda x: x[0])

        if not scored:
            scored = [(2.0, 'uniform', _norm(np.ones(n_target)))]

        print(f"  {len(scored)} inits. Top: {[f'{s:.6f}({nm})' for s, nm, _ in scored[:4]]}")

        # Select top 2 diverse candidates
        candidates = scored[:2]

        for ci, (sc_init, name, arr_init) in enumerate(candidates):
            if time.monotonic() > sub_deadline - 30.0:
                break
            cand_deadline = sub_deadline if ci == len(candidates) - 1 else \
                           t_start + (sum(time_fracs[:li+1]) * main_time * (ci+1) / len(candidates)) + 30.0
            cand_deadline = min(cand_deadline, sub_deadline - 10.0)

            print(f"\n  --- Candidate '{name}' (init={sc_init:.8f}) ---")
            sc, seq = _multilevel_optimize(arr_init, n_target, cand_deadline, f'n{n_target}_{name}', rng)

            if sc < best_overall_sc:
                best_overall_sc = sc
                best_overall_seq = seq.copy()
                print(f"  *** Global best: {best_overall_sc:.8f} (n={n_target}) ***")

    # Final global polish
    if best_overall_seq is not None and time.monotonic() < deadline - 20.0:
        print(f"\n=== Final polish (score={best_overall_sc:.8f}) ===")
        final_deadline = deadline - 15.0
        n_best = len(best_overall_seq)

        # High-beta LSE
        for k_f, beta_f in [(200, 8000), (500, 20000), (1000, 40000)]:
            if time.monotonic() > final_deadline - 20:
                break
            remaining = final_deadline - time.monotonic() - 20
            if remaining < 10:
                break
            sc_f, a_f = _run_lbfgs(best_overall_seq, beta_f, k_f, remaining * 0.4, 400)
            if sc_f < best_overall_sc:
                best_overall_sc = sc_f
                best_overall_seq = a_f.copy()
                print(f"  Final LSE K={k_f}: {best_overall_sc:.8f}")

        # Subgradient
        if time.monotonic() < final_deadline - 10:
            sc_sg, a_sg = _subgrad_polish(best_overall_seq, max_iters=25, deadline=final_deadline - 5.0)
            if sc_sg < best_overall_sc:
                best_overall_sc = sc_sg
                best_overall_seq = a_sg.copy()
                print(f"  Final subgrad: {best_overall_sc:.8f}")

        # Random restart
        for restart in range(4):
            if time.monotonic() > final_deadline - 20:
                break
            p = _smooth_perturb(best_overall_seq, rng, strength=0.08)
            chunk = min(final_deadline - 20, time.monotonic() + 10)
            sc_r, a_r = _run_lbfgs(p, 10000, 500, max(2, chunk - time.monotonic()), 150)
            if sc_r < best_overall_sc:
                best_overall_sc = sc_r
                best_overall_seq = a_r.copy()
                print(f"  Restart {restart}: {best_overall_sc:.8f}")

        # Final subgrad
        if time.monotonic() < final_deadline - 5:
            sc_sg2, a_sg2 = _subgrad_polish(best_overall_seq, max_iters=15, deadline=final_deadline)
            if sc_sg2 < best_overall_sc:
                best_overall_sc = sc_sg2
                best_overall_seq = a_sg2.copy()

    # Prepare output
    if best_overall_seq is None:
        best_overall_seq = np.ones(1024) / 1024

    out = _norm(best_overall_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_overall_sc

    print(f"\nFINAL: internal={best_overall_sc:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
