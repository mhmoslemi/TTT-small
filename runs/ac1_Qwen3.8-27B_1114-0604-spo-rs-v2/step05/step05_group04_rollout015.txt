```python
def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0"""
    n = len(f)
    # Build constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n
    # This is sparse, use scipy.sparse
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n and 0 <= j < n
    
    # Actually, let's build it row by row
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = M * np.ones(2*n - 1)
    c = -np.ones(n)  # minimize -sum(g) = max sum(g)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_iter(a, gamma, top_frac, soft_gate_alpha):
    """Single Fienup iteration (stateless for reuse)."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = A.real ** 2 + A.imag ** 2
    n_bins = len(P)
    thresh = np.quantile(P, 1.0 - top_frac)
    scale = np.ones_like(P)
    above = P > thresh
    if above.any():
        ratio = thresh / P[above]
        scale[above] = ratio ** 0.5 * gamma
    A_mod = A * scale
    a_new = np.fft.irfft(A_mod, M_fft)[:n]
    if soft_gate_alpha > 0:
        gate = 1.0 / (1.0 + np.exp(-soft_gate_alpha * a_new))
        a_new = a_new * gate
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a
    return a_new / S_new


def fienup_run(a, n_iters, gamma, top_frac_init, top_frac_end, soft_gate_alpha=30.0,
               timeout=None, t_start=None):
    """Run Fienup iterations with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if timeout is not None and time.monotonic() - t_start > timeout:
            break
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        a = fienup_iter(a, gamma, top_frac, soft_gate_alpha)
        # Periodic spectral diffusion
        if it % 20 == 19:
            M_fft = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P_smooth = np.convolve(P, np.ones(5) / 5.0, mode='same')
            P_mix = 0.7 * P + 0.3 * P_smooth
            P_mix = np.maximum(P_mix, 1e-20)
            amp_ratio = np.sqrt(P_mix / np.maximum(P, 1e-20))
            A_mod = A * amp_ratio
            a_diff = np.fft.irfft(A_mod, M_fft)[:n]
            a_diff = np.maximum(a_diff, 0.0)

            if S_d > 1e-12:

def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

    best_sc = compute_score(a)
    step = 0.02
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        if timeout is not None and time.monotonic() - t_start > timeout:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.03)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9995
    return best_a


def build_initializations(n, rng, parent):
    """Build diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.05, 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 50.0))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.sin(3.0 * np.pi * x) + 0.2 * np.cos(np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * x * (1 - x) * 4)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    Uses hybrid Fienup + LP-direction + L-BFGS + subgradient approach.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5
    n_main = 6962

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Budget: {total_budget}s, Deadline: {deadline - t_start:.0f}s")

    if parent is not None:
        n_main = len(parent)
        sc = compute_score(parent)
        print(f"Parent: n={n_main}, score={sc:.8f}")
        update_best(sc, parent, "PARENT")
    else:
        print("No parent available, starting from scratch.")

    best_a = best_a[:n_main].copy() if len(best_a) != n_main else best_a.copy()

    # ==================================================================
    # Phase 0: Quick Fienup from parent (5% time)
    # ==================================================================
    phase0_end = t_start + 0.05 * (total_budget - 40.0)
    print(f"\n=== Phase 0: Quick Fienup ({phase0_end - t_start:.0f}s) ===")

    if parent is not None:
        a_f = fienup_run(parent, 100, gamma=0.8, top_frac_init=0.1, top_frac_end=0.4,
                         soft_gate_alpha=30.0, timeout=phase0_end - t_start, t_start=t_start)
        update_best(compute_score(a_f), a_f, "P0-FIENUP")

    # ==================================================================
    # Phase 1: Fienup from diverse inits (20% time)
    # ==================================================================
    phase1_end = t_start + 0.25 * (total_budget - 40.0)
    print(f"\n=== Phase 1: Diverse Fienup ({phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n_main, rng, parent)
    for i, a0 in enumerate(inits[:6]):
        if time.monotonic() > phase1_end:
            break
        label = f"init{i}"
        # Fienup variant 1
        a_f = fienup_run(a0, 150, gamma=0.7, top_frac_init=0.08, top_frac_end=0.5,
                         soft_gate_alpha=25.0, timeout=phase1_end - t_start, t_start=t_start)
        update_best(compute_score(a_f), a_f, f"P1-F{label}")
        if time.monotonic() < phase1_end - 30:
            # Fienup variant 2 (spectral eq)
            a_eq = fienup_spectral_eq_run(a0, 120, 0.3, 0.8,
                                          timeout=phase1_end - t_start, t_start=t_start)
            update_best(compute_score(a_eq), a_eq, f"P1-E{label}")

    # Fienup from best so far with different params
    if time.monotonic() < phase1_end:
        a_cur = best_a.copy() / best_a.sum()
        a_f2 = fienup_run(a_cur, 200, gamma=0.5, top_frac_init=0.15, top_frac_end=0.6,
                          soft_gate_alpha=15.0, timeout=phase1_end - t_start, t_start=t_start)
        update_best(compute_score(a_f2), a_f2, "P1-BEST-F2")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP-based iterations (35% time) -- KEY NEW COMPONENT
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 40.0)
    print(f"\n=== Phase 2: LP-based iterations ({phase2_end - t_start:.0f}s) ===")

    a_cur = best_a.copy() / best_a.sum()
    n_lp_iters = 0
    lp_fail_count = 0

    while time.monotonic() < phase2_end - 10:
        # Timed out check
        if time.monotonic() >= phase2_end - 10:
            break

        # Compute current state
        c = fast_conv(a_cur)
        M = float(c.max())
        S = float(a_cur.sum())
        sc = 2.0 * n_main * M / (S * S)

        # Try LP direction
        # Adaptive active set: start large, reduce over time
        progress = n_lp_iters / max(1, 60)
        n_active = max(50, int(200 * (1 - 0.6 * progress)))

        g = lp_direction(a_cur, M, n_active=n_active,
                         timeout=phase2_end - t_start, t_start=t_start)

        if g is not None:
            lp_fail_count = 0
            # Line search: try different step sizes
            best_t = 0.3
            best_sc_trial = sc
            for t_try in [0.5, 0.3, 0.2, 0.1, 0.7, 0.4, 0.15]:
                a_trial = (1 - t_try) * a_cur + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                S_t = a_trial.sum()
                if S_t < 1e-12:
                    continue
                a_trial = a_trial / S_t
                sc_trial = compute_score(a_trial)
                if sc_trial < best_sc_trial:
                    best_sc_trial = sc_trial
                    best_t = t_try

            a_cur = (1 - best_t) * a_cur + best_t * g
            a_cur = np.maximum(a_cur, 0.0)
            S_cur = a_cur.sum()
            if S_cur > 1e-12:
                a_cur = a_cur / S_cur

            n_lp_iters += 1
            sc_new = compute_score(a_cur)
            if n_lp_iters % 5 == 0:
                print(f"    LP iter {n_lp_iters}: score={sc_new:.8f}, t={best_t:.3f}, "
                      f"n_active={n_active}, time={time.monotonic()-t_start:.0f}s")
            update_best(sc_new, a_cur, f"LP-{n_lp_iters}")

        else:
            lp_fail_count += 1

        # Every few iterations, do Fienup polishing
        if n_lp_iters > 0 and n_lp_iters % 8 == 0:
            a_fpol = fienup_run(a_cur, 30, gamma=0.8, top_frac_init=0.1, top_frac_end=0.35,
                                soft_gate_alpha=30.0, timeout=phase2_end - t_start, t_start=t_start)
            sc_pol = compute_score(a_fpol)
            if sc_pol < compute_score(a_cur):
                a_cur = a_fpol
                update_best(sc_pol, a_cur, f"LP-POLISH-{n_lp_iters}")

        # If LP keeps failing, switch to Fienup for a while
        if lp_fail_count >= 3:
            lp_fail_count = 0
            a_f = fienup_run(a_cur, 50, gamma=0.7, top_frac_init=0.1, top_frac_end=0.4,
                             soft_gate_alpha=25.0, timeout=phase2_end - t_start, t_start=t_start)
            sc_f = compute_score(a_f)
            if sc_f < compute_score(a_cur):
                a_cur = a_f
                update_best(sc_f, a_cur, f"LP-FALLBACK-{n_lp_iters}")

    # Also try QP-based direction a few times
    if time.monotonic() < phase2_end - 20:
        print("  Trying QP directions...")
        for qp_iter in range(5):
            if time.monotonic() > phase2_end - 10:
                break
            g_qp = active_qp_direction(a_cur, float(fast_conv(a_cur).max()), n_active=80, eps=0.02,
                                       timeout=phase2_end - t_start, t_start=t_start)
            if g_qp is not None:
                best_t = 0.5
                best_sc_t = compute_score(a_cur)
                for t_try in [0.7, 0.5, 0.3, 0.2]:
                    a_t = (1 - t_try) * a_cur + t_try * g_qp
                    a_t = np.maximum(a_t, 0.0)
                    S_t = a_t.sum()
                    if S_t < 1e-12:
                        continue
                    a_t = a_t / S_t
                    sc_t = compute_score(a_t)
                    if sc_t < best_sc_t:
                        best_sc_t = sc_t
                        best_t = t_try
                a_cur = (1 - best_t) * a_cur + best_t * g_qp
                a_cur = np.maximum(a_cur, 0.0) / a_cur.sum()
                update_best(compute_score(a_cur), a_cur, f"QP-{qp_iter}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS with increasing beta (30% time)
    # ==================================================================
    phase3_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 3: L-BFGS ({phase3_end - t_start:.0f}s) ===")

    a_work = best_a.copy() / best_a.sum()
    betas = [200, 500, 1000, 2000, 5000, 10000]

    for beta in betas:
        if time.monotonic() > phase3_end - 60:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 45:
            break

        # Allocate iterations based on remaining time and beta
        # Higher beta = harder problem = fewer iterations needed
        maxiter = min(10000, max(500, int(remaining * 2 * (2000 / max(beta, 200)))))
        maxfun = maxiter * 15

        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun,
                              timeout=phase3_end - t_start, t_start=t_start)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            a_work = a_opt.copy() / a_opt.sum()
            # Quick Fienup polish between L-BFGS calls
            if time.monotonic() < phase3_end - 30:
                a_pol = fienup_run(a_work, 40, gamma=0.8, top_frac_init=0.08,
                                   top_frac_end=0.3, soft_gate_alpha=40.0,
                                   timeout=phase3_end - t_start, t_start=t_start)
                sc_pol = compute_score(a_pol)
                if sc_pol < sc:
                    a_work = a_pol
                    update_best(sc_pol, a_pol, f"LBFGS-{beta}-POLISH")
        else:
            # If no improvement, try from best_a directly
            if time.monotonic() < phase3_end - 30:
                sc2, a_opt2 = run_lbfgs(best_a, beta, maxiter=max(500, maxiter // 2),
                                        maxfun=max(1000, maxfun // 2),
                                        timeout=phase3_end - t_start, t_start=t_start)
                update_best(sc2, a_opt2, f"LBFGS b={beta} fromBest")
                if sc2 < sc:
                    a_work = a_opt2.copy() / a_opt2.sum()

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polishing (10% remaining time)
    # ==================================================================
    print(f"\n=== Phase 4: Final polishing ===")

    # Subgradient polishing
    for top_m in [10, 20, 50]:
        if time.monotonic() > deadline - 10:
            break
        a_sg = subgradient_polish(best_a, n_iters=300, top_m=top_m,
                                  timeout=deadline - t_start, t_start=t_start)
        update_best(compute_score(a_sg), a_sg, f"SG m={top_m}")

    # Final Fienup + L-BFGS combo
    if time.monotonic() < deadline - 40:
        a_fin = fienup_run(best_a, 80, gamma=0.7, top_frac_init=0.08, top_frac_end=0.35,
                           soft_gate_alpha=35.0, timeout=deadline - t_start, t_start=t_start)
        sc_fin = compute_score(a_fin)
        update_best(sc_fin, a_fin, "FIN-FIENUP")
        if time.monotonic() < deadline - 20:
            sc_fin2, a_fin2 = run_lbfgs(a_fin, 5000, maxiter=2000, maxfun=30000,
                                        timeout=deadline - t_start, t_start=t_start)
            update_best(sc_fin2, a_fin2, "FIN-LBFGS")

    # Perturbation + re-optimize
    if time.monotonic() < deadline - 30:
        for trial in range(2):
            if time.monotonic() > deadline - 15:
                break
            noise = 0.01 * (1 + trial)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = a_p.sum()
            if S_p < 1e-12:
                continue
            a_p = a_p / S_p
            a_r = fienup_run(a_p, 60, gamma=0.8, top_frac_init=0.1, top_frac_end=0.4,
                             soft_gate_alpha=30.0, timeout=deadline - t_start, t_start=t_start)
            sc_r = compute_score(a_r)
            update_best(sc_r, a_r, f"RESTART-{trial}")
            if sc_r < best_score and time.monotonic() < deadline - 15:
                sc_r2, a_r2 = run_lbfgs(a_r, 3000, maxiter=1500, maxfun=20000,
                                        timeout=deadline - t_start, t_start=t_start)
                update_best(sc_r2, a_r2, f"RESTART-{trial}-LBFGS")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
