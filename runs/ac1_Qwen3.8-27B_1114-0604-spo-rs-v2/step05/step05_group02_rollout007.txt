def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def greedy_peak_reduction(a, n_iters=5000):

            break
        
        k_star = int(np.argmax(c))

        if sc < best_sc:
            best_sc = sc

        # Find contributing elements

        # The contribution to c[k_star] from a[j] is a[k_star - j]

        a[top_global] = np.maximum(a[top_global], 0.0)
        
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a
