"""
Multi-phase optimizer with improved LP quality for flat-autocorrelation non-negative sequences.
Key improvements over previous: higher beta FISTA for LP, adaptive steps, 
better line search, more aggressive L-BFGS polish, and perturbation restarts.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a: 2 * cross-correlation(a, p) at the right indices
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}

def lp_solve_fista(a, M, n_fista_iters=25000, beta_start=100.0, beta_end=3000.0,
                   x_init=None, ramp_frac=0.15):
    """
    Solve: max sum(x) s.t. conv(a,x)_k <= M for all k, x >= 0
    using FISTA on LSE-smoothed surrogate.
    
    KEY FIX: Correct gradient computation using cross-correlation.
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a = float(a.sum())
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    # Step size: use 1/(beta * max(sum_a^2, n*sum_a2)) to be safe
    # Actually, let's use the conservative bound: 1/(beta * n * sum_a2)
    # Since n*sum_a2 >= sum_a^2, this is conservative
    L_base = n * sum_a2

    n_ramp = max(1, int(n_fista_iters * ramp_frac))

    for it in range(n_fista_iters):
        if it < n_ramp:
            frac = it / n_ramp
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * L_base + 1e-12
        step = 1.0 / L

        # Compute c = conv(a, x)
        c = cross_conv(a, x)
        
        # Compute softmax weights
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # CORRECT GRADIENT: grad_j = sum_i a_i * p_{i+j} - 1
        # Using FFT-based cross-correlation
        grad = cross_correl_for_grad(a, p) - 1.0

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x

        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters,
                           beta_start=beta_start, beta_end=beta_end, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Line search using minimize_scalar
        def objective_t(t):
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                return np.inf
            a_try = a_try / St
            return compute_score(a_try)
        
        # Use a few initial points to guide the search
        t_results = minimize_scalar(objective_t, bounds=(0.05, 0.95), method='bounded',
                                     options={'xatol': 0.01})
        t = t_results.x if t_results.success else 0.3
        t = max(0.05, min(0.95, t))

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 15:
                # Diversify
                a_new = a_new * (1.0 + 0.01 * np.random.default_rng(outer).standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                sn = float(a_new.sum())
                if sn > 1e-12:
                    a_new = a_new / sn
                no_improve_count = 0

        a = a_new

    return best_a, best_sc, x_warm
