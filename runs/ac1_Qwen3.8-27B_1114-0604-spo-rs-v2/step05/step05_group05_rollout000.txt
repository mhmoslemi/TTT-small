import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def focused_lse_score_and_grad(a, K, beta):
    """
    Compute focused LSE score and gradient.
    Only smooth over the top K peaks of conv(a,a).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top K indices
    K_actual = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:K_actual]
    
    # LSE over top K
    c_top = c[top_idx]
    c_max = c_top.max()
    shifted = beta * (c_top - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    
    # Weights p over the top K
    p_top = exp_shifted / exp_shifted.sum()
    
    # Build full p vector (length 2n-1)
    p = np.zeros(2*n - 1)
    p[top_idx] = p_top
    
    # Gradient: grad = 2 * (p * a)[:n]
    conv_len = 3*n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    conv_res = np.fft.irfft(P * A, M_fft)[:n]
    grad_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def fast_conv(a):

    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad, top_idx


def mirror_descent_focused(a, max_time, K_start=10, K_end=200, beta_start=50.0, beta_end=20000.0, lr=0.05):
    """
    Mirror descent (multiplicative updates) on focused LSE objective.
    Alternates K and beta adaptively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    K = K_start
    beta = beta_start
    lr_cur = lr

    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = it / max(1, (it + 100))  # approximate progress

        # Adapt K and beta
        K = int(min(K_end, K_start + (K_end - K_start) * min(1.0, it / 500.0)))
        beta = min(beta_end, beta_start * (1.003 ** it))
        lr_cur = lr * (0.9995 ** it)

        sc, grad, top_idx = focused_lse_grad(a, K, beta)

        # Multiplicative update: a_m *= exp(-eta * grad_m)
        # Scale gradient to control step size
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-15:
            break

        # Adaptive step: limit the log-change per element
        eta = lr_cur / (gnorm + 1e-10)
        # Clamp the exponent to avoid overflow
        log_update = -np.clip(eta * grad, -3.0, 3.0)
        a = a * np.exp(log_update)

        # Normalization and non-negativity (multiplicative updates preserve non-negativity)
        a = np.maximum(a, 1e-15)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Periodic exact evaluation
        if it % 20 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def nesterov_focused_lse(a, max_time, K_start=20, K_end=100, beta_start=30.0, beta_end=5000.0):
    """
    Nesterov-accelerated projected gradient on focused LSE.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    lr = 0.02
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        K = int(min(K_end, K_start + (K_end - K_start) * min(1.0, it / 300.0)))
        beta = min(beta_end, beta_start * (1.005 ** it))
        lr *= 0.9997
        mu = min(0.96, 0.9 + 0.00003 * it)

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        sc, grad, _ = focused_lse_grad(a_look, K, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def fienup_enhanced(a, n_iters=200, momentum_end=0.4, thresh_start=2.0, thresh_end=1.05, clip_start=0.3, clip_end=0.8):
    """Enhanced Fienup with momentum and cosine-annealed parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_end * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        v = momentum * v + (a_new - a)
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def direct_subgradient_max(a, max_time, lr=0.01):
    """
    Direct subgradient method on the exact max objective.
    Uses empirical distribution over argmax positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    lr_cur = lr
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        lr_cur *= 0.9998
        mu = min(0.95, 0.9 + 0.00002 * it)

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())

        # Find positions near the max (within 1% of max)
        threshold = c_max * 0.99
        active = c >= threshold
        n_active = active.sum()
        if n_active == 0:
            active[np.argmax(c)] = True
            n_active = 1

        # Uniform weights over active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active

        # Gradient: 2 * (p * a)[:n]
        conv_len = 3 * n - 2
        M_fft = 1 << (conv_len - 1).bit_length()
        P_ft = np.fft.rfft(p, M_fft)
        A_ft = np.fft.rfft(a, M_fft)
        conv_res = np.fft.irfft(P_ft * A_ft, M_fft)[:n]
        grad_L = 2.0 * conv_res

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_L / S2 - 2.0 * c_max / S3)

        # Nesterov step
        v = mu * v - lr_cur * grad
        a = a + v
        a = np.maximum(a, 1e-15)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def hybrid_fienup_lse(a, max_time, fienup_frac=0.4):
    """
    Alternate between Fienup (global) and focused LSE (local) optimization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    cycle = 0

    while time.monotonic() - t_start < max_time:
        time_left = max_time - (time.monotonic() - t_start)
        if time_left < 5:
            break

        cycle += 1
        if (cycle % 3) < 2:
            # Fienup step (global)
            sub_time = min(15.0, time_left * 0.3)
            a_f = fienup_enhanced(a, n_iters=int(30 + 50 * sub_time / 15),
                                   momentum_end=0.3 + 0.2 * min(1.0, cycle / 10.0))
        else:
            # Focused LSE step (local)
            sub_time = min(20.0, time_left * 0.4)
            a_f = nesterov_focused_lse(a, max_time=sub_time, K_start=10, K_end=80,
                                        beta_start=50.0, beta_end=5000.0)

        sc = compute_score(a_f)
        if sc < best_sc:
            best_sc = sc
            best_a = a_f.copy()
            a = a_f.copy()
        else:
            # Revert if no improvement
            a = best_a.copy()
            # Add small perturbation to escape
            pert = 0.01 * (cycle % 5 + 1) / 5.0
            a = a * (1.0 + pert * np.random.default_rng(cycle).standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def run_lbfgs_focused(a0, K, beta, maxiter=3000):
    """L-BFGS-B on focused LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        sc, g, _ = focused_lse_grad(x, K, beta)
        return sc, g

    res = minimize(objective, a0, method='L-BFGS-B', bounds=bounds, jac=True,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': 20000, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def build_diverse_init(n, rng, variant=0):
    """Build diverse initializations."""
    if variant == 0:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        return np.maximum(a, 0.01)
    elif variant == 1:
        # Chirp-like
        x = np.arange(n, dtype=np.float64) / n
        a = np.abs(np.sin(2 * np.pi * 2.0 * x * x)) + 0.1
        return a
    elif variant == 2:
        # Gaussian
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a
    elif variant == 3:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.01)
    else:
        # Wigner-like (random smooth)
        a = 0.5 + 0.2 * np.sin(2 * np.pi * 3.7 * x) + 0.1 * np.cos(2 * np.pi * 7.3 * x)
        return np.maximum(a, 0.01)


def upsample(a, new_n):
    """Upsample sequence via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Uses focused active-set LSE with mirror descent,
    hybrid Fienup-LSE alternation, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 6962
    best_score = np.inf
    best_a = np.ones(n_target) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Target n={n_target}, budget={total_budget}s")

    # ============================================================
    # Phase 1: Multi-start diversity (25% of time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Multi-start (budget {phase1_end - t_start:.0f}s) ===")

    candidates = []

    # From parent
    if parent is not None:
        a_p = parent.copy()
        a_p = a_p / a_p.sum()
        sc = compute_score(a_p)
        candidates.append((sc, a_p.copy(), "parent"))
        print(f"  Parent score: {sc:.8f}")

        # Quick Fienup on parent
        a_pf = fienup_enhanced(a_p, n_iters=80, momentum_end=0.3,
                                thresh_start=1.8, thresh_end=1.1,
                                clip_start=0.3, clip_end=0.7)
        sc_pf = compute_score(a_pf)
        candidates.append((sc_pf, a_pf, "parent_fienup"))
        print(f"  Parent+Fienup score: {sc_pf:.8f}")

        # Quick mirror descent on parent
        time_left = phase1_end - time.monotonic()
        if time_left > 20:
            a_pm = mirror_descent_focused(a_p, max_time=min(30.0, time_left * 0.5),
                                           K_start=10, K_end=100,
                                           beta_start=30.0, beta_end=5000.0, lr=0.04)
            sc_pm = compute_score(a_pm)
            candidates.append((sc_pm, a_pm, "parent_mirror"))
            print(f"  Parent+Mirror score: {sc_pm:.8f}")

    # Diverse random initializations
    for var in range(5):
        if time.monotonic() > phase1_end:
            break
        a_init = build_diverse_init(n_target, rng, variant=var)
        a_init = a_init / a_init.sum()
        # Quick Fienup
        a_rq = fienup_enhanced(a_init, n_iters=50, momentum_end=0.2,
                                thresh_start=2.0, thresh_end=1.2,
                                clip_start=0.3, clip_end=0.6)
        sc_rq = compute_score(a_rq)
        candidates.append((sc_rq, a_rq, f"diverse_{var}"))
        print(f"  Diverse {var}: {sc_rq:.8f}")

    # Sort by score and keep top 3
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:min(3, len(candidates))]]
    for i, c in enumerate(candidates[:5]):
        update_best(c[0], c[1], c[2])

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Focused LSE optimization (35% of time)
    # ============================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Focused LSE (budget {phase2_end - t_start:.0f}s) ===")

    for ci, cand in enumerate(top_candidates):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        print(f"  Candidate {ci}: focused LSE (time={time_left:.0f}s)")

        # Nesterov focused LSE
        a_nl = nesterov_focused_lse(cand, max_time=time_left * 0.5,
                                     K_start=10, K_end=120,
                                     beta_start=20.0, beta_end=10000.0)
        update_best(compute_score(a_nl), a_nl, f"nesterov_focus_{ci}")

        # Mirror descent focused LSE
        time_left = phase2_end - time.monotonic()
        if time_left > 30:
            a_md = mirror_descent_focused(a_nl, max_time=time_left * 0.5,
                                           K_start=20, K_end=200,
                                           beta_start=100.0, beta_end=30000.0, lr=0.03)
            update_best(compute_score(a_md), a_md, f"mirror_focus_{ci}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Hybrid Fienup-LSE (20% of time)
    # ============================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Hybrid Fienup-LSE (budget {phase3_end - t_start:.0f}s) ===")

    time_left = phase3_end - time.monotonic()
    if time_left > 30:
        a_h = hybrid_fienup_lse(best_a, max_time=time_left * 0.8, fienup_frac=0.4)
        update_best(compute_score(a_h), a_h, "hybrid")

        # Also try from a perturbed version
        if time.monotonic() < phase3_end - 30:
            a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n_target))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            time_left2 = phase3_end - time.monotonic()
            if time_left2 > 20:
                a_h2 = hybrid_fienup_lse(a_pert, max_time=time_left2 * 0.7)
                update_best(compute_score(a_h2), a_h2, "hybrid_pert")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS polish (10% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: L-BFGS polish (budget {phase4_end - t_start:.0f}s) ===")

    lbfgs_configs = [
        (30, 500), (50, 2000), (80, 5000), (100, 10000), (150, 20000),
    ]
    for K, beta in lbfgs_configs:
        if time.monotonic() > phase4_end:
            break
        time_left = phase4_end - time.monotonic()
        if time_left < 10:
            break
        maxiter = int(min(5000, time_left * 50))
        sc, a_opt = run_lbfgs_focused(best_a, K=K, beta=beta, maxiter=maxiter)
        update_best(sc, a_opt, f"lbfgs_K{K}_b{beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Direct subgradient + final polish (5% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Final polish (budget {phase5_end - time.monotonic():.0f}s) ===")

    time_left = phase5_end - time.monotonic()
    if time_left > 15:
        a_ds = direct_subgradient_max(best_a, max_time=time_left * 0.5, lr=0.008)
        update_best(compute_score(a_ds), a_ds, "subgrad_max")

        # Final L-BFGS with high beta
        time_left = phase5_end - time.monotonic()
        if time_left > 15:
            sc, a_opt = run_lbfgs_focused(best_a, K=50, beta=30000, maxiter=int(time_left * 30))
            update_best(sc, a_opt, "final_lbfgs")

    # ============================================================
    # Final: one more round of mirror descent on the best
    # ============================================================
    time_left = deadline - time.monotonic()
    if time_left > 10:
        a_final = mirror_descent_focused(best_a, max_time=time_left * 0.7,
                                          K_start=5, K_end=50,
                                          beta_start=500.0, beta_end=50000.0, lr=0.02)
        update_best(compute_score(a_final), a_final, "final_mirror")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
