```python
"""
Hybrid optimizer combining L-BFGS with LSE smoothing, efficient 1D coordinate
descent with golden-section search, and peak-aware greedy refinement for
minimizing 2*n*max(autocorrelation)/(sum)^2 over non-negative sequences.
Key innovation: vectorized coordinate descent exploiting the structure that
each coordinate affects a contiguous block of autocorrelation values linearly
(or quadratically at one point), enabling fast 1D golden-section optimization.
"""

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

        perm = np.arange(n)
        rng.shuffle(perm)
        blocks = [perm[i:i+block_size] for i in range(0, n, block_size)]

        for block in blocks:
            if len(block) == 0:
                continue
            m = len(block)
            x_old = a[block].copy()
            S_prime = S - x_old.sum()

            # Determine affected k range (union over all j in block)
            k_lo = min(block)
            k_hi = min(max(block) + n - 1, 2 * n - 2)
            k_range = np.arange(k_lo, k_hi + 1)

            # For each x in the block, compute how b changes
            # This is a quadratic form in x (m-dimensional)
            # b[k] = sum_{j in block} (2*x[j]*a[k-j] + ...) for k != 2j
            # For efficiency with small m, just do 1D line search along stochastic direction

            # Simple approach: optimize one coordinate at a time within the block
            for jj in block:
                x_j_old = a[jj]
                Sp_j = S - x_j_old
                k_lo_j = jj
                k_hi_j = min(jj + n - 1, 2 * n - 2)
                kr_j = np.arange(k_lo_j, k_hi_j + 1)
                a_sh_j = a[kr_j - jj]
                m2_j = (kr_j == 2 * jj)

                c_j = b[kr_j].copy()
                if (~m2_j).any():
                    c_j[~m2_j] = b[kr_j[~m2_j]] - 2.0 * x_j_old * a_sh_j[~m2_j]
                if m2_j.any():
                    c_j[m2_j] = b[kr_j[m2_j]] - x_j_old * x_j_old

                a_aff_j = a_sh_j.copy()
                a_aff_j[m2_j] = 0.0

                if k_lo_j > 0 and k_hi_j < 2*n-2:
                    Mo_j = max(float(b[:k_lo_j].max()), float(b[k_hi_j+1:].max()))
                elif k_lo_j > 0:
                    Mo_j = float(b[:k_lo_j].max())
                elif k_hi_j < 2*n-2:
                    Mo_j = float(b[k_hi_j+1:].max())
                else:
                    Mo_j = -1e30

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo, hi = 0.0, max(x_j_old * 4.0, S - x_j_old, 1e-6)
                for _ in range(20):
                    x1 = hi - gr * (hi - lo)
                    x2 = lo + gr * (hi - lo)
                    bl1 = 2.0 * x1 * a_aff_j + c_j
                    ml1 = float(np.max(bl1))
                    if m2_j.any():
                        bq1 = x1*x1 + c_j[m2_j][0]
                        ma1 = max(Mo_j, ml1, bq1)
                    else:
                        ma1 = max(Mo_j, ml1)
                    f1 = 2.0*n*ma1/((S-x_j_old+x1)**2 + 1e-30)

                    bl2 = 2.0 * x2 * a_aff_j + c_j
                    ml2 = float(np.max(bl2))
                    if m2_j.any():
                        bq2 = x2*x2 + c_j[m2_j][0]
                        ma2 = max(Mo_j, ml2, bq2)
                    else:
                        ma2 = max(Mo_j, ml2)
                    f2 = 2.0*n*ma2/((S-x_j_old+x2)**2 + 1e-30)

                    if f1 < f2:
                        hi = x2
                    else:
                        lo = x1

                x_j_new = (lo + hi) * 0.5
                delta = x_j_new - x_j_old
                if abs(delta) < 1e-14:
                    continue
                a[jj] = x_j_new
                S += delta
                if (~m2_j).any():
                    b[kr_j[~m2_j]] += 2.0 * delta * a_sh_j[~m2_j]
                if m2_j.any():
                    b[2*jj] += delta * (x_j_new + x_j_old)

            if S > 1e-12:
                ratio = 1.0 / S
                a *= ratio
                b *= ratio * ratio
                S = 1.0

        sc = 2.0 * n * float(b.max())
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b_idx in range(min(n_blocks, n // block_size)):
            s = b_idx * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.9995):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.1, np.cos(np.pi * x)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2))
    # Spectral flatness target: raised cosine variants
    inits.append(0.3 + 0.7 * np.sin(np.pi * x) ** 2)
    inits.append(np.maximum(0.05, 0.5 + 0.3 * np.cos(2*np.pi*(x-0.5))))
    # Two-lobe
    inits.append(np.maximum(0.05, 0.8*np.exp(-8*(x-0.3)**2) + 0.8*np.exp(-8*(x-0.7)**2)))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: diverse multi-start -> L-BFGS refinement -> coordinate descent
    polishing -> iterated local search -> final high-precision refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_parent = len(parent)
        print(f"Parent length: {n_parent}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
        # Try a slightly larger length for more optimization room
        n_work = min(12000, max(n_parent, 8000))
    else:
        n_work = 8192

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: Multi-start with L-BFGS (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: Multi-Start L-BFGS (n={n_work}) ---")

    inits = build_initializations(n_work, rng, parent)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S

        sc_init = compute_score(a_init)
        report(sc_init, a_init, f"init_{i}")

        # Quick L-BFGS
        beta = 100
        maxiter = int(min(10000, max(1000, (phase1_deadline - time.monotonic()) * 20)))
        if maxiter < 500:
            break
        sc, a_opt, _ = run_lbfgs(a_init, beta, maxiter=maxiter, maxfun=maxiter*4)
        report(sc, a_opt, f"LBFGS_init_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: L-BFGS with increasing beta (25% time)
    # ==================================================================
    phase2_deadline = t_start + 0.40 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    beta_schedule = [100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining_s = phase2_deadline - time.monotonic()
        if remaining_s < 30:
            break
        maxiter = int(min(50000, max(3000, remaining_s * 30)))
        maxfun = int(min(200000, max(10000, remaining_s * 100)))

        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")

        # Perturbed restart for diversity
        if time.monotonic() < phase2_deadline - 45 and beta >= 300:
            a_pert = perturbed_restart(best_a, rng, scale=0.01)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Coordinate Descent Polishing (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.70 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: Coordinate Descent ---")

    n_cd_sweeps = int(min(80, max(20, (phase3_deadline - time.monotonic()) / 3.0)))
    print(f"  Running {n_cd_sweeps} sweeps of coordinate descent")
    a_cd = coordinate_descent(best_a, n_sweeps=n_cd_sweeps, seed=seed, verbose=True)
    sc = compute_score(a_cd)
    report(sc, a_cd, "coord_descent")

    # Another pass from slightly different starting point
    if time.monotonic() < phase3_deadline - 60:
        a_pert2 = perturbed_restart(best_a, rng, scale=0.005)
        a_cd2 = coordinate_descent(a_pert2, n_sweeps=n_cd_sweeps//2, seed=seed+1)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "coord_descent_pert")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated Local Search (20% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 12
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining_s = phase4_deadline - time.monotonic()
        if remaining_s < 90:
            break

        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # L-BFGS refinement
        beta = 300 + int(2000 * rng.random())
        maxiter = int(min(30000, max(2000, remaining_s * 15)))
        maxfun = int(min(100000, max(5000, remaining_s * 50)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_pert, 1e-10), beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} lbfgs b={beta}")

        # Coordinate descent polish
        if time.monotonic() < phase4_deadline - 30:
            n_sweeps = int(min(30, max(5, (phase4_deadline - time.monotonic()) / 4.0)))
            a_cd_ils = coordinate_descent(a_opt, n_sweeps=n_sweeps, seed=seed+trial)
            sc_cd = compute_score(a_cd_ils)
            report(sc_cd, a_cd_ils, f"ILS#{trial} cd")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final High-Precision Polishing (10% time)
    # ==================================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final Polishing ---")

    # High-beta L-BFGS
    for beta in [100000, 300000, 1000000]:
        if time.monotonic() > phase5_deadline - 30:
            break
        remaining = phase5_deadline - time.monotonic()
        if remaining < 20:
            break
        maxiter = int(min(20000, max(1000, remaining * 20)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter*5)
        report(sc, a_opt, f"final LBFGS b={beta}")

    # Final coordinate descent
    if time.monotonic() < phase5_deadline - 30:
        n_final = int(min(40, max(5, (phase5_deadline - time.monotonic()) / 3.0)))
        a_final_cd = coordinate_descent(best_a, n_sweeps=n_final, seed=seed+100)
        sc = compute_score(a_final_cd)
        report(sc, a_final_cd, "final CD")

    # One more L-BFGS on the very best
    if time.monotonic() < phase5_deadline - 20:
        sc, a_vf, _ = run_lbfgs(best_a, 500000, maxiter=10000, maxfun=50000)
        report(sc, a_vf, "very final LBFGS")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
