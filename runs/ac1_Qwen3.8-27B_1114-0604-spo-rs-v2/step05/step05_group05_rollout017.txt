"""
Spectral compression + adaptive Nesterov LSE + targeted peak reduction.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key differences from previous approach:
1. Full-spectrum multiplicative compression (not just thresholding)
2. Restart Nesterov with backtracking line search
3. Targeted peak-reduction coordinate descent
4. Multi-beta parallel refinement
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def spectral_compression(a, n_iters=500, gamma_start=0.3, gamma_end=0.9,
                         momentum_init=0.0, momentum_end=0.5):
    """
    Full-spectrum multiplicative compression.
    At each iteration:
    1. Compute spectrum A = FFT(a)
    2. Compute power P = |A|^2
    3. Compress: |A_new| = |A| * (P/P_mean)^(-gamma/2)
       This reduces peaks and boosts valleys, making P flatter.
    4. Reconstruct a_new = IFFT(A_new)
    5. Project to non-negative
    6. Normalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * t
        momentum = momentum_init + (momentum_end - momentum_init) * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        # Multiplicative compression: reduce high P, boost low P
        ratio = P / (P_mean + 1e-30)
        # Compression factor: for P > P_mean, factor < 1; for P < P_mean, factor > 1
        factor = ratio ** (-gamma / 2.0)
        # Cap the boosting to avoid numerical issues
        factor = np.clip(factor, 0.01, 100.0)
        
        A_new = A * factor
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Update with momentum
        delta = a_new - a
        v = momentum * v + delta
        a_candidate = a + v
        
        # Project to non-negative and normalize
        a_candidate = np.maximum(a_candidate, 0.0)
        S_new = float(a_candidate.sum())
        if S_new < 1e-12:
            a_candidate = a
        else:
            a_candidate = a_candidate / S_new
        
        sc = compute_score(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()
        
        a = a_candidate
    
    return best_a


def nesterov_restart(a, max_time=100.0, beta_start=20.0, beta_end=10000.0,
                     lr_start=0.02, momentum_base=0.9):
    """
    Nesterov with restart: if the score increases for several consecutive steps,
    reset momentum. Use backtracking to ensure progress.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    it = 0
    beta = beta_start
    beta_growth = 1.003
    lr = lr_start
    mu = momentum_base
    consecutive_worse = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, momentum_base + 0.00003 * it)
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            v = np.zeros(n)
            continue
        a_look = a_look / S_look
        
        # Compute gradient
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Update
        v = mu * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            v = np.zeros(n)
            continue
        a_new = a_new / S_new
        
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            consecutive_worse = 0
        elif sc_new < best_sc + 0.001 * best_sc:
            # Accept even if slightly worse (to escape local minima)
            a = a_new
            consecutive_worse += 1
        else:
            a = a_new  # Still take the step but note it's bad
            consecutive_worse += 1
        
        # Restart momentum if stuck

            consecutive_worse = 0
            lr *= 0.5

def targeted_peak_reduction(a, n_iters=200, n_top_coords=30):
    """
    Identify the peak lag of the convolution, then iteratively reduce
    the contributing elements at that lag.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        peak_k = int(np.argmax(c))
        
        # Find which (i, j) pairs contribute to c[peak_k]
        # c[peak_k] = sum_{i} a[i] * a[peak_k - i] for valid i
        i_start = max(0, peak_k - n + 1)
        i_end = min(peak_k, n - 1)
        
        # Contribution of a[i] to c[peak_k] is a[peak_k - i] for i in [i_start, i_end]
        # To reduce c[peak_k], we should reduce the product a[i]*a[peak_k-i]
        # Strategy: find the pair (i, j) with i+j=peak_k that maximizes a[i]*a[j],
        # and reduce the larger of the two.
        
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = peak_k - i_vals
        products = a[i_vals] * a[j_vals]
        
        # Find top contributing pairs
        n_top = min(n_top_coords, len(products))
        top_idx = np.argsort(products)[::-1][:n_top]
        
        improved = False
        for idx in top_idx:
            i = int(i_vals[idx])
            j = int(j_vals[idx])
            a_i = a[i]
            a_j = a[j]
            
            # Try reducing the larger element
            if a_i >= a_j:
                target = i
                other = a_j
            else:
                target = j
                other = a_i
            
            if a[target] < 1e-10 or other < 1e-10:
                continue
            
            # Try reducing by various factors
            for factor in [0.9, 0.95, 0.98, 1.02, 1.05]:
                a_new = a.copy()
                a_new[target] = a_new[target] * factor
                S_new = float(a_new.sum())
                if S_new < 1e-12:
                    continue
                a_new = a_new / S_new
                sc = compute_score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
        
        if not improved and it > 50:
            # Try a random perturbation to escape
            idx = np.random.randint(n)
            a[idx] *= 0.95
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S > 1e-12:
                a = a / S
    
    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def build_init(n, mode='cosine'):
    if mode == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif mode == 'flat':
        a = np.ones(n)
    elif mode == 'chirp':
        x = np.arange(n, dtype=np.float64) / n
        a = np.abs(np.sin(2 * np.pi * 2.0 * x * x)) + 0.1
    else:
        a = np.ones(n)
    return a / a.sum()


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)

"""
Spectral compression + Nesterov restart + targeted peak reduction optimizer.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovations:
1. Full-spectrum multiplicative compression (smooth, no hard threshold)
2. Nesterov with momentum restart and adaptive backtracking
3. Targeted peak-lag coordinate descent
4. Multi-scale warm start with progressive gamma scheduling
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient using FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def spectral_compression(a, n_iters=500, gamma_start=0.3, gamma_end=0.9,
                         momentum_init=0.0, momentum_end=0.5,
                         boost_cap=50.0):
    """
    Full-spectrum multiplicative compression.
    |A_new| = |A| * (P/P_mean)^(-gamma/2) for ALL bins.
    Flattens the power spectrum, reducing time-domain convolution peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * t
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        if P_mean < 1e-30:
            break

        ratio = P / (P_mean + 1e-30)
        factor = np.power(ratio, -gamma / 2.0)
        factor = np.clip(factor, 0.01, boost_cap)

        A_new = A * factor
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        delta = a_new - a
        v = momentum * v + delta
        a_candidate = a + v

        a_candidate = np.maximum(a_candidate, 0.0)
        S_new = float(a_candidate.sum())
        if S_new < 1e-12:
            a_candidate = a.copy()
        else:
            a_candidate = a_candidate / S_new

        sc = compute_score(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()

        a = a_candidate

    return best_a


def nesterov_restart(a, max_time=100.0, beta_start=20.0, beta_end=10000.0,
                     lr_start=0.02, momentum_base=0.9, beta_growth=1.004,
                     lr_decay=0.9995):
    """
    Nesterov accelerated gradient with momentum restart.
    Resets momentum when progress stalls. Adaptive beta schedule.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    lr = lr_start
    mu = momentum_base
    consecutive_worse = 0
    last_improve_it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, momentum_base + 0.00003 * it)

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            v = np.zeros(n)
            continue
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size: limit step relative to current gap
        sc_est = compute_score(a) if it % 50 == 1 else best_sc
        lb = 2.0 * n / (2.0 * n - 1.0)
        max_step = max(0.005, (sc_est - lb) * 0.3)
        effective_lr = min(lr, max_step / (gnorm + 1e-10))

        v = mu * v - effective_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            v = np.zeros(n)
            continue
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            consecutive_worse = 0
            last_improve_it = it
        else:
            a = a_new
            consecutive_worse += 1

        # Restart momentum if stuck for too long
        if consecutive_worse > 20 or (it - last_improve_it > 200 and it > 100):
            v = np.zeros(n)
            consecutive_worse = 0
            lr *= 0.7

        lr *= lr_decay
        beta = min(beta * beta_growth, beta_end)

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final

    return best_a


def targeted_peak_reduction(a, n_iters=300, n_top_coords=40):
    """
    Coordinate descent targeting the peak lag of the full convolution.
    Reduces the largest contributing products at the peak lag.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        peak_k = int(np.argmax(c))

        i_start = max(0, peak_k - n + 1)
        i_end = min(peak_k, n - 1)
        if i_end < i_start:
            break

        i_vals = np.arange(i_start, i_end + 1)
        j_vals = peak_k - i_vals
        products = a[i_vals] * a[j_vals]

        n_top = min(n_top_coords, len(products))
        top_idx = np.argsort(products)[::-1][:n_top]

        improved = False
        for idx in top_idx:
            i = int(i_vals[idx])
            j = int(j_vals[idx])
            a_i = a[i]
            a_j = a[j]

            if a_i < 1e-12 or a_j < 1e-12:
                continue

            # Reduce the larger element (or try both)
            targets = []
            if a_i >= a_j:
                targets = [i, j]
            else:
                targets = [j, i]

            for target in targets:
                if a[target] < 1e-12:
                    continue
                for factor in [0.85, 0.92, 0.96, 1.04, 1.1]:
                    a_new = a.copy()
                    a_new[target] *= factor
                    S_new = float(a_new.sum())
                    if S_new < 1e-12:
                        continue
                    a_new = a_new / S_new
                    sc = compute_score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        break
                if improved:
                    break

        if not improved and it > 100:
            # Random perturbation to escape
            idx = np.random.randint(n)
            a[idx] *= 0.9
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S > 1e-12:
                a = a / S

    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_init(n, mode='cosine'):
    """Build various initialization patterns."""
    if mode == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif mode == 'flat':
        a = np.ones(n)
    elif mode == 'chirp':
        x = np.arange(n, dtype=np.float64) / n
        a = np.abs(np.sin(2 * np.pi * 2.0 * x * x)) + 0.1
    elif mode == 'hamming':
        a = np.hamming(n)
        a = np.maximum(a, 0.01)
    elif mode == 'kaiser':
        from scipy.signal import \u200bkaiser
        a = kaiser(n, 6.0)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses spectral compression + Nesterov restart + targeted peak reduction + L-BFGS.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent if available
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 6962

    best_score = np.inf
    best_a = np.ones(n_target) / n_target

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic() - t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Multi-scale warm start (12% of time)
    # ============================================================
    phase1_end = t_start + 0.12 * (total_budget - 30.0)
    print(f"=== Phase 1: Multi-scale warm start ===")

    current_a = None
    scales = [64, 128, 256, 512, 1024, 2048, 4096]
    scales = [s for s in scales if s < n_target] + [n_target]
    scales = list(dict.fromkeys(scales))

    for n in scales:
        if time.monotonic() > phase1_end:
            break
        t_s = time.monotonic()
        budget_s = phase1_end - t_s
        if budget_s < 5:
            break

        print(f"  Scale n={n}")

        if current_a is None:
            a = build_init(n, 'cosine')
        else:
            a = upsample(current_a, n)

        # Spectral compression at this scale
        n_iters = max(50, min(300, int(budget_s * 5)))
        a = spectral_compression(a, n_iters=n_iters, gamma_start=0.3, gamma_end=0.7,
                                 momentum_init=0.0, momentum_end=0.3)
        current_a = a
        sc = compute_score(a)
        print(f"    Score: {sc:.8f}")

    if current_a is not None:
        if len(current_a) != n_target:
            current_a = upsample(current_a, n_target)
        update_best(compute_score(current_a), current_a, "multi-scale")

    # Start from parent
    if parent is not None:
        sc_p = compute_score(parent)
        update_best(sc_p, parent, "parent")
        if time.monotonic() < phase1_end - 10:
            a_pc = spectral_compression(parent, n_iters=150, gamma_start=0.2, gamma_end=0.5,
                                         momentum_init=0.0, momentum_end=0.2)
            update_best(compute_score(a_pc), a_pc, "parent+spec")

    # Also try diverse inits
    for mode in ['gaussian', 'hamming']:
        if time.monotonic() > phase1_end:
            break
        a_div = build_init(n_target, mode)
        a_div = spectral_compression(a_div, n_iters=100, gamma_start=0.3, gamma_end=0.6,
                                     momentum_init=0.0, momentum_end=0.2)
        update_best(compute_score(a_div), a_div, f"init-{mode}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Spectral compression main loop (30% of time)
    # ============================================================
    phase2_end = t_start + 0.42 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Spectral compression ===")

    spec_configs = [
        {'gamma_start': 0.3, 'gamma_end': 0.8, 'momentum_init': 0.0, 'momentum_end': 0.4, 'boost_cap': 50.0},
        {'gamma_start': 0.5, 'gamma_end': 1.0, 'momentum_init': 0.2, 'momentum_end': 0.5, 'boost_cap': 30.0},
        {'gamma_start': 0.1, 'gamma_end': 0.4, 'momentum_init': 0.1, 'momentum_end': 0.3, 'boost_cap': 100.0},
    ]

    for i, cfg in enumerate(spec_configs):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        n_iters = max(200, int(time_left * 2))
        n_iters = min(n_iters, 3000)
        print(f"  Spec {i}: gamma=[{cfg['gamma_start']},{cfg['gamma_end']}], iters={n_iters}")
        a_sc = spectral_compression(best_a, n_iters=n_iters, **cfg)
        update_best(compute_score(a_sc), a_sc, f"spec-{i}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Nesterov restart (28% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Nesterov restart ===")

    nesterov_configs = [
        {'beta_start': 10.0, 'beta_end': 5000.0, 'lr_start': 0.02, 'momentum_base': 0.9},
        {'beta_start': 50.0, 'beta_end': 10000.0, 'lr_start': 0.01, 'momentum_base': 0.92},
        {'beta_start': 100.0, 'beta_end': 20000.0, 'lr_start': 0.005, 'momentum_base': 0.93},
    ]

    for i, cfg in enumerate(nesterov_configs):
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 30:
            break
        print(f"  Nesterov {i}: beta=[{cfg['beta_start']},{cfg['beta_end']}], time={time_left:.0f}s")
        a_nl = nesterov_restart(best_a, max_time=time_left * 0.9, **cfg)
        update_best(compute_score(a_nl), a_nl, f"nesterov-{i}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Targeted peak reduction (12% of time)
    # ============================================================
    phase4_end = t_start + 0.82 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Targeted peak reduction ===")

    if time.monotonic() < phase4_end - 20:
        time_left = phase4_end - time.monotonic()
        n_iters = max(100, int(time_left * 1.5))
        a_tr = targeted_peak_reduction(best_a, n_iters=n_iters, n_top_coords=50)
        update_best(compute_score(a_tr), a_tr, "peak-red")

        # Follow up with Nesterov after peak reduction
        if time.monotonic() < phase4_end - 10:
            a_tr2 = nesterov_restart(best_a, max_time=30.0, beta_start=200.0,
                                     beta_end=10000.0, lr_start=0.005,
                                     momentum_base=0.93)
            update_best(compute_score(a_tr2), a_tr2, "peak-red+nesterov")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish (10% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: L-BFGS polish ===")

    lbfgs_betas = [500, 1000, 2000, 5000, 10000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase5_end - 10:
            break
        print(f"  L-BFGS beta={beta}")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=2000, maxfun=40000)
        update_best(sc, a_opt, f"lbfgs-b={beta}")

    print(f"After Phase 5: best = {best_score:.8f}")

    # ============================================================
    # Final: one more combined pass if time allows
    # ============================================================
    if time.monotonic() < deadline - 20:
        a_fin = spectral_compression(best_a, n_iters=100, gamma_start=0.5, gamma_end=0.9,
                                     momentum_init=0.3, momentum_end=0.5, boost_cap=20.0)
        update_best(compute_score(a_fin), a_fin, "final-spec")

        a_fin2 = nesterov_restart(best_a, max_time=20.0, beta_start=1000.0,
                                   beta_end=15000.0, lr_start=0.002, momentum_base=0.94)
        update_best(compute_score(a_fin2), a_fin2, "final-nest")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
