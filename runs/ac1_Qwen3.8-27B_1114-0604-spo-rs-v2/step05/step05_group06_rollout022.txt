Let me think about FISTA more carefully. FISTA is for problems of the form:
minimize f(x) + g(x)

where f is smooth and g is convex (but not necessarily smooth). In our case:
- f(x) = LSE-smoothed max-convolution (smooth)
- g(x) = indicator of {x ≥ 0} (convex, non-smooth)

The proximal operator of g is just clipping to [0, ∞).

FISTA iteration:
- y_k = x_k + (t_{k-1} - 1) / (t_k + 1) * (x_k - x_{k-1})
- x_{k+1} = prox_g(1/L) (y_k - (1/L) ∇f(y_k))
- t_{k+1} = (1 + sqrt(1 + 4*t_k²)) / 2

where L is the Lipschitz constant of ∇f.

For the LSE objective, L is hard to compute exactly, but we can estimate it or use backtracking.

def _fista_lse(a0, beta, k, time_limit, max_iters=10000):
    """FISTA on LSE-smoothed objective with simplex constraint."""
    a = _norm(a0)
    n = len(a)
    t = 1.0
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        
        # FISTA extrapolation
        t_prev = t
        t = (1 + math.sqrt(1 + 4*t*t)) / 2
        y = a + (t_prev - 1) / (t + 1) * (a - a_prev)
        y = np.clip(y, 0, None)  # project to non-negative
        
        # Compute gradient of LSE objective at y
        val, grad = _lse_obj_grad(y, beta, k)
        
        # Estimate step size (use a conservative one)
        # The gradient is O(n/S²) in magnitude, so step size ~ S²/(n * max|grad|)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        eta = 0.5 / gnorm  # backtracking-ish
        
        # Update
        x_new = y - eta * grad
        x_new = np.clip(x_new, 0, None)
        x_new = _norm(x_new)  # renormalize
        
        a_prev = a
        a = x_new
        
        # Check improvement
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_sc, best_a

def _next_pow2(m):
    m = int(max(1, m))

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size

    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _anneal_restart(a, rng, strength=0.05):
    """Apply a smooth random perturbation (annealing step)."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 100))
    if k % 2 == 0:
        k += 1
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Multi-phase optimization combining LSE-smoothed L-BFGS-B,
Nesterov-accelerated subgradient descent, block coordinate descent,
and simulated annealing restarts. Uses adaptive K scheduling and
multi-scale gradients for both exploration and exploitation.
"""
import numpy as np
import time
import math
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, 1.0 / a.size)
    return a / S


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + math.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.005 * max(0.5, float(n) / 4096.0)
    mf = int(0.80 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 5000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def _obj_fn(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            _obj_fn, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30,
                     'iprint': -1}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _nesterov_subgrad(a0, deadline, k_start=10, k_end=800, base_eta=0.005):
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    y = a.copy()
    t_val = 1.0
    it = 0
    no_improve = 0
    check_interval = max(100, n // 10)

    while time.monotonic() < deadline:
        frac = min(1.0, it / 20000.0)
        k_use = int(k_start + (k_end - k_start) * frac)
        k_use = max(1, k_use)
        M = 2 * n - 1
        k_use = min(k_use, M)

        t_prev = t_val
        t_val = (1.0 + math.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_prev - 1.0) / (t_val + 1.0)
        y_new = a + momentum * (a - y)
        y_new = np.clip(y_new, 0.0, None)
        S = float(y_new.sum())
        if S < 1e-12:
            y_new = a
            S = float(a.sum())

        c = _self_conv(y_new)
        d = float(n) * c / (S * S)
        idx = np.argpartition(d, M - k_use)[M - k_use:]
        d_act = d[idx]
        dmax = float(d_act.max())
        if dmax <= 0:
            break
        beta_sm = max(20.0, 2.0 * k_use)
        z = beta_sm * (d_act - dmax)
        e = np.exp(z)
        se = float(e.sum())
        if se < 1e-300:
            w_act = np.ones(k_use) / k_use
        else:
            w_act = e / se
        w_full = np.zeros(M)
        w_full[idx] = w_act
        corr = _cross_corr(y_new, w_full)
        LDS = float(w_act @ d_act)
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        eta = base_eta / (1.0 + 0.05 * it)
        x_new = y_new - eta * grad
        x_new = np.clip(x_new, 0.0, None)
        Sq = float(x_new.sum())
        if Sq < 1e-12:
            x_new = a
        else:
            x_new = x_new / Sq
        a = x_new
        y = y_new
        it += 1

        if it % check_interval == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 5 and it > 5000:
                break

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_sc, best_a


def _block_cd(a0, deadline, n_rounds=100):
    a = _norm(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    block_size = max(8, n // 500)
    n_blocks = n // block_size
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for _ in range(5):
            if time.monotonic() > deadline:
                break
            blk = rng.integers(0, n_blocks)
            start = blk * block_size
            end = start + block_size
            idx = np.arange(start, end)
            S = float(a.sum())
            c = _self_conv(a)
            d = float(n) * c / (S * S)
            dmax = float(d.max())
            thresh = dmax - 2e-4 * max(dmax, 1.0)
            active = d >= thresh
            if not active.any():
                active[np.argmax(d)] = True
            n_act = int(active.sum())
            w = np.zeros_like(d)
            w[active] = 1.0 / n_act
            corr = _cross_corr(a, w)
            LDS = float(w[active] @ d[active])
            grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
            g_block = grad[idx]
            gnorm_b = float(np.linalg.norm(g_block))
            if gnorm_b < 1e-13:
                continue
            best_local = None
            best_local_sc = best_sc
            for scale in (0.001, 0.005, 0.02, 0.06, 0.15):
                eta = scale / gnorm_b
                new_vals = a[idx] - eta * g_block
                new_vals = np.clip(new_vals, 0.0, None)
                a_test = a.copy()
                a_test[idx] = new_vals
                S_test = float(a_test.sum())
                if S_test < 1e-10:
                    continue
                a_test = a_test / S_test
                sc = _score(a_test)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = a_test.copy()
            if best_local is not None:
                a = best_local
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()

    return best_sc, best_a


def _subgrad_polish(a0, max_iters=60, deadline=None):
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active = d >= dmax - 5e-6 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0001, 0.0004, 0.0015, 0.005, 0.02, 0.06, 0.15, 0.4):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    return best_sc, best_a


def _smooth_perturb(a, rng, strength=0.1):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(51, max(5, n // 80))
    if k % 2 == 0:
        k += 1
    c = k // 2
    sig = max(1.0, c / 2.5)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)
