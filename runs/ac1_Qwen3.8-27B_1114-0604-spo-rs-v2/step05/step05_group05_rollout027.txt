"""
Peak-directed optimization with multi-method ensemble for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. Peak-directed gradient: directly targets the top-k autocorrelation peaks
2. Adaptive Fienup with spectrum-shaping inspired by BPSK sequences
3. Multi-start with non-symmetric bias patterns
4. Alternating between LSE-smoothed and hard-max subgradient steps
5. Final L-BFGS polish with very high beta
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * a / S3)
    return score, grad


def topk_peak_grad(a, k_top=10):
    """
    Compute gradient targeting the top-k peaks of the autocorrelation.
    This is a hard-constraint approach: minimize max(c) by pushing down top peaks.
    Returns (score, gradient) where score is the actual max-based score.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    # Get top-k peak indices
    k_top = min(k_top, len(c))
    peak_indices = np.argsort(c)[::-1][:k_top]

    # Weight peaks by how close they are to the max
    c_max = c[peak_indices[0]]
    c_range = c_max - c[peak_indices[-1]] + 1e-15
    weights = 1.0 / (1.0 + 10.0 * (c_max - c[peak_indices]) / c_range)
    weights = weights / weights.sum()

    # Compute gradient: weighted sum of gradients of c[k] for peak k
    # dc[k]/da[i] = a[k-i] + a[k-(-i)]... use FFT approach
    # Build weighted p vector
    p = np.zeros(len(c))
    for idx, k in enumerate(peak_indices):
        p[k] = weights[idx]
    p = p / p.sum()

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_c = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * c_max / S2
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * c_max * a / S3)
    return score, grad


def fienup_shaped(a, n_iters=200, thresh_start=2.5, thresh_end=1.05,
                  clip_start=0.3, clip_end=0.85, momentum_end=0.4,
                  spectral_shape='flat'):
    """
    Fienup with shaped spectrum target.
    spectral_shape: 'flat' (uniform target), 'tilted' (favors low frequencies)
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    # Precompute spectral shape target
    freqs = np.fft.rfftfreq(M_fft) * 2  # normalize to [0, 2]
    if spectral_shape == 'tilted':
        # Slightly favor low frequencies (helps with non-symmetric sequences)
        target_shape = 1.0 + 0.3 * np.exp(-freqs * 5.0)
        target_shape = target_shape / target_shape.mean()
    else:
        target_shape = np.ones_like(freqs)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        momentum = momentum_end * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target spectrum shaped
        P_target = P_mean * target_shape

        # Identify bins significantly above target
        above = P > P_target * thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(P_target[above])
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

            # Also boost bins that are too low (optional, helps convergence)
            below = P < P_target * (1.0 / thresh)
            if below.any() and t > 0.5:
                boost_factor = 1.0 + 0.1 * clip
                scale[below] = boost_factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Momentum update
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v

        # Project
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        if it % 20 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def peak_directed_descent(a, max_time=100.0, k_top=5, lr_base=0.01):
    """
    Directly minimize the maximum autocorrelation peak using top-k peak gradients.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    v = np.zeros(n)
    mu = 0.9
    lr = lr_base
    lr_decay = 0.998

    it = 0
    while time.monotonic() - t_start < max_time:
        it += 1
        lr *= lr_decay
        mu = min(0.95, 0.85 + 0.0001 * it)

        # Compute top-k peak gradient
        sc, grad = topk_peak_grad(a, k_top=k_top)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Nesterov-style update
        a_look = np.maximum(a + mu * v, 0.0)
        S_look = a_look.sum()
        if S_look > 1e-12:
            a_look = a_look / S_look
            _, grad = topk_peak_grad(a_look, k_top=k_top)
            gnorm = float(np.linalg.norm(grad))

        # Adaptive step size
        effective_lr = min(lr, 0.01 / (gnorm + 1e-10))
        v = mu * v - effective_lr * grad
        a = a + v

        # Project
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if it % 30 == 0 or it == 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def nesterov_lse_time(a, max_time=100.0, beta_start=50.0, beta_end=20000.0):
    """Time-adaptive Nesterov LSE optimizer."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    beta = beta_start
    lr = 0.01
    mu = 0.9
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.88 + 0.00003 * it)
        lr *= 0.9995
        beta = min(beta * 1.003, beta_end)

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        effective_lr = min(lr, 0.005 / (gnorm + 1e-10))
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def alternating_optimize(a, max_time=200.0):
    """
    Alternate between Fienup (frequency domain) and peak-directed descent (time domain).
    Each method helps escape the other's local minima.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    cycle = 0

    while time.monotonic() - t_start < max_time:
        time_left = max_time - (time.monotonic() - t_start)
        if time_left < 20:
            break

        # Fienup step
        t_f = min(60, time_left * 0.4)
        a_f = fienup_shaped(best_a, n_iters=int(100 * t_f / 60),
                            thresh_start=2.0, thresh_end=1.05,
                            clip_start=0.3, clip_end=0.8,
                            momentum_end=0.3 + 0.1 * min(cycle, 3))
        sc_f = compute_score(a_f)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_f.copy()

        # Peak-directed step
        if time.monotonic() - t_start < max_time - 20:
            t_p = min(60, max_time - (time.monotonic() - t_start))
            a_p = peak_directed_descent(best_a, max_time=t_p,
                                        k_top=5 + cycle, lr_base=0.008)
            sc_p = compute_score(a_p)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()

        cycle += 1

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=None):
    """Run L-BFGS-B on LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    t_start = time.monotonic()

    def objective(x):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            return (1e10, np.zeros(n))
        return lse_score_and_grad(x, beta)

    res = minimize(objective, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent_peaks(a, max_time=60.0, n_top=20):
    """
    Coordinate descent: for each top peak, try scaling individual elements
    that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    sweep = 0
    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Find top contributors to this peak
            contrib = np.array([a[k - j] * a[j] for j in range(j_start, j_end + 1)])
            n_local = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue

                # Try multiplicative adjustments
                for factor in [0.7, 0.85, 0.95, 1.05, 1.15]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_tmp = a / S_tmp
                    sc = compute_score(a_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_tmp.copy()
                        a = a_tmp
                        break
                else:
                    a[j] = old_val

        sweep += 1
        if sweep % 3 == 0:
            a = best_a.copy()

    return best_a


def build_non_symmetric_init(n, rng, style='exponential'):
    """Build non-symmetric initializations that bias mass to one side."""
    if style == 'exponential':
        rate = rng.uniform(0.3, 1.5)
        a = np.exp(-rate * np.arange(n) / n)
    elif style == 'linear_ramp':
        a = np.linspace(rng.uniform(0.1, 0.5), 1.0, n)
    elif style == 'half_gaussian':
        center = rng.uniform(n * 0.2, n * 0.5)
        sigma = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif style == 'bimodal':
        a = np.zeros(n)
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.6, 0.9)
        sigma = n * 0.08
        a += np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2)
        a += 0.5 * np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-method ensemble: Fienup + peak-directed descent + LSE + L-BFGS + CD.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(a, label):
        nonlocal best_score, best_a
        a = np.clip(a.astype(np.float64), 0.0, None)
        S = a.sum()
        if S < 1e-12:
            return False
        a = a / S
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 0: Initialize from parent + non-symmetric alternatives
    # ============================================================
    print(f"=== Phase 0: Initialization ===")
    candidates = []

    if parent is not None:
        candidates.append(("parent", parent.copy()))
        # Slightly perturbed parent
        p_pert = parent * (1.0 + 0.01 * rng.standard_normal(len(parent)))
        p_pert = np.clip(p_pert, 0, None)
        candidates.append(("parent_pert", p_pert))

    # Non-symmetric initializations at target length
    n_target = len(parent) if parent is not None else 6962
    for style in ['exponential', 'linear_ramp', 'half_gaussian', 'bimodal']:
        a_init = build_non_symmetric_init(n_target, rng, style)
        candidates.append((f"init_{style}", a_init))

    # Evaluate all candidates
    for label, a_cand in candidates:
        update_best(a_cand, label)

    if best_a is None:
        best_a = np.ones(6962) / 6962
        best_score = compute_score(best_a)

    print(f"  Best after init: {best_score:.8f}")

    # ============================================================
    # Phase 1: Multi-scale warm start from small n (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 30)
    print(f"\n=== Phase 1: Multi-scale warm start (t < {phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end:
        scales = [128, 256, 512, 1024, 2048, 4096]
        scales = [s for s in scales if s < n_target]
        current_small = None

        for n in scales:
            if time.monotonic() > phase1_end - 5:
                break
            if current_small is None:
                a = build_non_symmetric_init(n, rng, 'exponential')
            else:
                a = upsample(current_small, n)

            # Quick optimization at this scale
            a = fienup_shaped(a, n_iters=50, thresh_start=2.0, thresh_end=1.1,
                              clip_start=0.3, clip_end=0.7, momentum_end=0.2)
            sc_small = compute_score(a)

            current_small = a
            print(f"  n={n}: score={sc_small:.6f}")

        # Upsample to target
        if current_small is not None:
            a_up = upsample(current_small, n_target)
            update_best(a_up, "multi-scale-up")

    # ============================================================
    # Phase 2: Fienup refinement (15% time)
    # ============================================================
    phase2_end = t_start + 0.25 * (total_budget - 30)
    print(f"\n=== Phase 2: Fienup refinement (t < {phase2_end - t_start:.0f}s) ===")

    fienup_configs = [
        {'n_iters': 300, 'thresh_start': 2.5, 'thresh_end': 1.05,
         'clip_start': 0.3, 'clip_end': 0.85, 'momentum_end': 0.4,
         'spectral_shape': 'flat'},
        {'n_iters': 300, 'thresh_start': 2.0, 'thresh_end': 1.05,
         'clip_start': 0.4, 'clip_end': 0.9, 'momentum_end': 0.5,
         'spectral_shape': 'tilted'},
        {'n_iters': 400, 'thresh_start': 3.0, 'thresh_end': 1.1,
         'clip_start': 0.2, 'clip_end': 0.7, 'momentum_end': 0.3,
         'spectral_shape': 'flat'},
    ]

    for i, cfg in enumerate(fienup_configs):
        if time.monotonic() > phase2_end - 10:
            break
        a_f = fienup_shaped(best_a, **cfg)
        update_best(a_f, f"fienup-{i}")

    # ============================================================
    # Phase 3: Peak-directed descent (20% time)
    # ============================================================
    phase3_end = t_start + 0.45 * (total_budget - 30)
    print(f"\n=== Phase 3: Peak-directed descent (t < {phase3_end - t_start:.0f}s) ===")

    # Multiple runs with different k_top
    for k_top in [3, 5, 8]:
        if time.monotonic() > phase3_end - 30:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 30:
            break
        a_pd = peak_directed_descent(best_a, max_time=min(time_left * 0.9, 180),
                                     k_top=k_top, lr_base=0.008)
        update_best(a_pd, f"peak-kd-k{k_top}")

    # ============================================================
    # Phase 4: Nesterov LSE (20% time)
    # ============================================================
    phase4_end = t_start + 0.65 * (total_budget - 30)
    print(f"\n=== Phase 4: Nesterov LSE (t < {phase4_end - t_start:.0f}s) ===")

    for beta_s, beta_e in [(20, 5000), (100, 20000), (500, 50000)]:
        if time.monotonic() > phase4_end - 30:
            break
        time_left = phase4_end - time.monotonic()
        if time_left < 30:
            break
        a_nl = nesterov_lse_time(best_a, max_time=min(time_left * 0.9, 200),
                                  beta_start=beta_s, beta_end=beta_e)
        update_best(a_nl, f"nesterov-b{beta_s}-{beta_e}")

    # ============================================================
    # Phase 5: Alternating Fienup + Peak (15% time)
    # ============================================================
    phase5_end = t_start + 0.80 * (total_budget - 30)
    print(f"\n=== Phase 5: Alternating optimization (t < {phase5_end - t_start:.0f}s) ===")

    if time.monotonic() < phase5_end - 30:
        time_left = phase5_end - time.monotonic()
        a_alt = alternating_optimize(best_a, max_time=min(time_left * 0.9, 300))
        update_best(a_alt, "alternating")

    # ============================================================
    # Phase 6: L-BFGS polish (10% time)
    # ============================================================
    phase6_end = t_start + 0.90 * (total_budget - 30)
    print(f"\n=== Phase 6: L-BFGS polish (t < {phase6_end - t_start:.0f}s) ===")

    for beta in [500, 2000, 5000, 10000, 30000]:
        if time.monotonic() > phase6_end - 10:
            break
        time_left = phase6_end - time.monotonic()
        if time_left < 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000,
                              time_limit=time_left - 5)
        update_best(a_opt, f"lbfgs-b{beta}")

    # ============================================================
    # Phase 7: Coordinate descent on peaks (5% time)
    # ============================================================
    phase7_end = deadline - 5
    print(f"\n=== Phase 7: Coordinate descent (t < {phase7_end - t_start:.0f}s) ===")

    if time.monotonic() < phase7_end - 10:
        time_left = phase7_end - time.monotonic()
        a_cd = coordinate_descent_peaks(best_a, max_time=min(time_left * 0.7, 120),
                                        n_top=30)
        update_best(a_cd, "coord-descent")

        # Final LSE after CD
        if time.monotonic() < phase7_end - 10:
            time_left2 = phase7_end - time.monotonic()
            a_final = nesterov_lse_time(best_a, max_time=min(time_left2 * 0.8, 60),
                                         beta_start=5000, beta_end=50000)
            update_best(a_final, "final-lse")

    # ============================================================
    # Final verification and output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
