def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = n + m - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)

    n = len(a)
    M = 2 * n - 1

    A = np.fft.rfft(a, N)

def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)

    if S < 1e-12:

    if S < 1e-12:
        return np.inf

    return 2.0 * n * c.max() / (S * S)

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed gradient descent with backtracking, coordinate peak-shaving,
adaptive beta continuation, and multi-length exploration.
Key improvements over previous: backtracking line search, coordinate descent
peak shaving, equalization-aware subgradient, and smarter restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    return a * (target / S)


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _exact_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def _lse_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = _fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax < 0:
        dmax = 0.0

    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    # Compute H_j = sum_k w_k * (a_{j+k} + a_{j-k})
    # r1_j = sum_k w_k * a_{j+k} = conv_full(a, w_rev)[j + 2n-2]
    # r2_j = sum_k w_k * a_{j-k} = conv_full(a, w)[j]
    N = _next_pow2(4 * n + 8)
    A = np.fft.rfft(a, N)
    wr = w[::-1]
    WR = np.fft.rfft(wr, N)
    conv1 = np.fft.irfft(A * WR, N)
    r1 = conv1[2 * n - 2: 3 * n - 2]

    W = np.fft.rfft(w, N)
    conv2 = np.fft.irfft(A * W, N)
    r2 = conv2[0:n]

    H = r1 + r2
    Wdot = float(np.dot(w, c))

    grad = (float(n) / (S * S)) * H - (2.0 * float(n) * Wdot) / (S * S * S)

    # Add normalization penalty gradient (push sum toward 1)
    pen = 1e-3
    grad = grad + 2.0 * pen * (S - 1.0)
    score = score + pen * (S - 1.0) ** 2

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def _fun_lse(x, beta):
    return _lse_grad(x, beta)


def _backtrack_lbfgs_step(a, grad, eta0, n_steps=8):
    """Backtracking line search from a gradient direction."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S0 = float(a.sum())
    score0 = _exact_score(a)

    # Gradient direction (steepest descent in log-space for non-negativity)
    # Use multiplicative: a * exp(-eta * grad_normalized)
    gmax = float(np.max(np.abs(grad)))
    if gmax < 1e-12:
        return a, score0

    grad_norm = grad / gmax

    best_a = a.copy()
    best_score = score0

    eta = eta0
    for _ in range(n_steps):
        if eta < 1e-8:
            break
        q = a * np.exp(-eta * grad_norm)
        q = np.maximum(q, 0.0)
        q = _normalize(q)
        sc = _exact_score(q)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = q.copy()
        eta *= 0.5

    return best_a, best_score


def _peak_shave_coordinate(a, n_iter=500, deadline=None):
    """Coordinate descent: repeatedly reduce the element contributing most to the peak."""
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _exact_score(a)
    no_improve = 0

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        if no_improve > 50:
            break

        c = _fast_self_conv(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        if peak_val < 1e-15:
            break

        # Sensitivity: s_j = a_{j+k*} + a_{j-k*}
        k = peak_idx
        s = np.zeros(n, dtype=np.float64)
        # s[j] = a[j+k] (if valid) + a[j-k] (if valid)
        idx_pos = np.arange(n) + k
        valid_pos = idx_pos < n
        s[valid_pos] += a[idx_pos[valid_pos]]
        idx_neg = np.arange(n) - k
        valid_neg = idx_neg >= 0
        s[valid_neg] += a[idx_neg[valid_neg]]

        # Contribution: s_j * a_j (how much a_j contributes to the peak)
        contrib = s * a

        # Find the element with maximum contribution
        j = int(np.argmax(contrib))
        if contrib[j] < 1e-15:
            break

        # Reduce a[j], distribute to elements with low sensitivity
        # Target: reduce peak by moving mass from high-sensitivity to low-sensitivity
        delta = 0.1 * a[j]  # reduce by 10%
        a[j] -= delta

        # Distribute to elements with lowest s value (least impact on peak)
        # Sort by sensitivity (ascending)
        order = np.argsort(s)
        # Give to the lowest-sensitivity elements
        n_distribute = max(1, n // 10)
        targets = order[:n_distribute]
        per_target = delta / float(len(targets))
        a[targets] += per_target

        a = np.maximum(a, 0.0)
        a = _normalize(a)

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert if no improvement
            a = best_a.copy()

    return float(best_sc), best_a


def _lse_gradient_descent(a, deadline, beta_start=50.0, beta_end=50000.0,
                          n_total_steps=2000, eta_start=0.5):
    """LSE gradient descent with adaptive beta and backtracking."""
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _exact_score(a)

    total_steps = n_total_steps
    for step in range(total_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Adaptive beta: exponential ramp
        frac = float(step) / float(total_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)

        sc_lse, grad = _lse_grad(a, beta)

        # Backtracking line search
        a_new, sc_new = _backtrack_lbfgs_step(a, grad, eta_start, n_steps=6)

        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            # Occasionally do coordinate peak shaving
            if step % 100 == 0 and step > 0:
                if deadline is not None and time.monotonic() < deadline - 5.0:
                    sc_cv, a_cv = _peak_shave_coordinate(a_new, n_iter=200, deadline=deadline)
                    if np.isfinite(sc_cv) and sc_cv < best_sc:
                        best_sc = sc_cv
                        best_a = a_cv.copy()
                    a = best_a.copy()
                else:
                    a = a_new.copy()
            else:
                a = a_new.copy()
        else:
            a = a_new.copy()

    return float(best_sc), best_a


def _run_lbfgs(a0, beta, deadline, maxfun=2000):
    """Run L-BFGS-B optimization."""
    a = _normalize(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return _exact_score(a), a

    bounds = [(0.0, 5.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            _fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return _exact_score(a), a

    a_out = _normalize(res.x) if hasattr(res, 'x') and res.x is not None else a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def _add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        arr = np.maximum(arr, 0.0)
        if arr.sum() > 1e-10:
            inits.append(_normalize(arr))

    if parent is not None and parent.size > 0:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    _add(np.ones(n))

    # V-shapes (high at edges)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # U-shapes (low in middle, more extreme)
    for alpha in (1.0, 2.0, 3.0, 5.0):
        _add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Cosine
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            _add(1.0 + alpha * np.cos(np.pi * x))
            _add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Exponential taper
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        _add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))
        _add(np.exp(-rate * x))

    # Bimodal
    for width in (0.08, 0.12, 0.15, 0.2):
        _add(np.exp(-(((x - 0.15) / width) ** 2)) + np.exp(-(((x - 0.85) / width) ** 2)))
        _add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)))

    # Sine-based
    for f in (1, 2, 3, 4, 5):
        _add(1.0 + 0.4 * np.sin(np.pi * f * x))
        _add(1.0 + 0.4 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.3, 1.5))
        _add(1.0 + amp * noise)

    # Random trigonometric
    for _ in range(5):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 8))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        _add(vals)

    # Power-law shapes
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        _add(x ** p + (1.0 - x) ** p)
        _add(x ** p)

    return inits


def _perturb(a, rng, strength=None):

        s = s / (float(np.max(s)) + 1e-12)
        q = p * np.exp(-strength * s)
    elif mode == 4:
        # Smooth noise in log-space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(5, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape modulation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(7, max(9, n // 80)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Edge boost
        x = np.linspace(0.0, 1.0, n)
        edge = np.exp(-((x - 0.0) / 0.1) ** 2) + np.exp(-((x - 1.0) / 0.1) ** 2)
        q = p * (1.0 + strength * edge)
    elif mode == 8:
        # Swap mass between high and low sensitivity regions
        c = _fast_self_conv(p)
        peak_idx = int(np.argmax(c))
        k = peak_idx
        s = np.zeros(n)
        idx_pos = np.arange(n) + k
        valid = idx_pos < n
        s[valid] += p[idx_pos[valid]]
        idx_neg = np.arange(n) - k
        valid = idx_neg >= 0
        s[valid] += p[idx_neg[valid]]
        # Reduce high-s, increase low-s
        thresh = np.median(s)
        mask_high = s > thresh
        mask_low = s <= thresh
        total_high = p[mask_high].sum()
        q = p.copy()
        q[mask_high] *= (1.0 - strength * 0.5)
        if mask_low.sum() > 0:
            q[mask_low] += total_high * strength * 0.5 / mask_low.sum()
    else:
        # Random block modification
        n_blocks = int(rng.integers(2, 8))
        block_size = max(1, n // n_blocks)
        idx = int(rng.integers(0, n_blocks))
        start = idx * block_size
        end = min(start + block_size, n)
        q = p.copy()
        q[start:end] *= float(rng.uniform(0.5, 1.5))

    return _normalize(np.maximum(q, 0.0))


def _optimize_single(a_init, budget_s, deadline):
    """Optimize a single starting point with time budget."""
    a = _normalize(a_init)
    best_a = a.copy()
    best_sc = _exact_score(a)
    n = len(a)

    # Phase 1: LSE gradient descent (main workhorse)
    rem = float(deadline - time.monotonic())
    if rem > 30:
        sc, seq = _lse_gradient_descent(a, deadline - 60.0,
                                         beta_start=50.0, beta_end=20000.0,
                                         n_total_steps=3000, eta_start=0.5)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase 2: Coordinate peak shaving
    rem = float(deadline - time.monotonic())
    if rem > 20:
        sc, seq = _peak_shave_coordinate(best_a, n_iter=2000, deadline=deadline - 30.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase 3: L-BFGS-B polish with increasing beta
    for beta in (500, 2000, 10000, 50000):
        rem = float(deadline - time.monotonic())
        if rem < 10:
            break
        maxfun = min(3000, max(100, int(rem * 0.4 * 100)))
        sc, seq = _run_lbfgs(best_a, beta, deadline, maxfun=maxfun)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase 4: Final coordinate shaving
    rem = float(deadline - time.monotonic())
    if rem > 10:
        sc, seq = _peak_shave_coordinate(best_a, n_iter=1000, deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: finds optimal non-negative sequence minimizing 2*n*max(conv)/sum^2."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine lengths to try
    lengths = [10240]
    if parent is not None and parent.size != 10240:
        lengths.insert(0, min(parent.size, 16384))
    lengths.extend([8192, 12288, 6144])

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Quick scan of initializations
    all_candidates = []  # (score, sequence, length)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        inits = _generate_inits(n, rng, parent)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_candidates.append((sc, arr, n))

    if not all_candidates:
        n0 = lengths[0]
        all_candidates.append((2.0, np.ones(n0) / n0, n0))

    all_candidates.sort(key=lambda x: x[0])
    print("Best init: {:.8f} (n={})".format(all_candidates[0][0], all_candidates[0][2]))

    # Select top starts (max 3, prefer different lengths)
    starts = []
    used_lens = set()
    for sc, arr, n in all_candidates:
        if len(starts) >= 3:
            break
        if n not in used_lens:
            starts.append(arr.copy())
            used_lens.add(n)

    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Ensure parent is included if available and good
    if parent is not None:
        p_len = min(parent.size, 16384)
        p_resampled = _resample(parent, p_len)
        p_sc = _exact_score(p_resampled)
        if p_sc < min(s[0] for s in [( _exact_score(x), x, 0) for x in starts]):
            starts.insert(0, p_resampled)
        elif p_len not in used_lens:
            starts.append(p_resampled)
        else:
            # Blend parent with best start
            best_start = starts[0]
            if len(best_start) != p_len:
                best_start_r = _resample(best_start, p_len)
            else:
                best_start_r = best_start
            blended = _normalize(0.3 * p_resampled + 0.7 * best_start_r)
            if _exact_score(blended) < _exact_score(starts[0]):
                starts[0] = blended

    best_seq = starts[0].copy()
    best_score = _exact_score(best_seq)
    for i in range(1, len(starts)):
        sc = _exact_score(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (n={})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization
    main_deadline = deadline - 200.0
    n_starts = len(starts)
    time_per_start = max(60.0, (main_deadline - time.monotonic()) / float(n_starts))

    # Sort starts by quality
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))

    for rank, si in enumerate(starts_sorted):
        if time.monotonic() > main_deadline - 30:
            break
        sub_deadline = min(t_start + (rank + 1) * time_per_start + 60, main_deadline)
        sub_deadline = float(sub_deadline)

        a_start = starts[si].copy()
        # For non-best starts, blend with current best
        if rank > 0 and np.isfinite(best_score):
            if len(a_start) != len(best_seq):
                best_r = _resample(best_seq, len(a_start))
            else:
                best_r = best_seq
            t_blend = 0.4 + 0.2 * rank
            a_start = _normalize((1.0 - t_blend) * a_start + t_blend * best_r)

        print("[start {}] optimizing n={} budget={:.0f}s".format(
            rank, len(a_start), sub_deadline - time.monotonic()))

        sc, seq = _optimize_single(a_start, time_per_start, sub_deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("[start {}] improved to {:.8f}".format(rank, best_score))

    best_seq = _normalize(best_seq)
    print("After main optimization: {:.8f}".format(best_score))

    # Phase 3: Restart loop
    restart_deadline = deadline - 80.0
    n_restarts = 0
    while time.monotonic() < restart_deadline and n_restarts < 8:
        rem = float(restart_deadline - time.monotonic())
        if rem < 60:
            break
        per_restart = min(180.0, rem / (8 - n_restarts))

        p = _perturb(best_seq, rng, strength=float(rng.uniform(0.1, 0.4)))
        sc, seq = _optimize_single(p, per_restart, time.monotonic() + per_restart)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {}: {:.8f}".format(n_restarts, best_score))
        n_restarts += 1

    # Phase 4: Final intensive polish
    rem = float(deadline - time.monotonic())
    if rem > 20:
        # LSE gradient descent with very high beta
        sc, seq = _lse_gradient_descent(best_seq, deadline - 20.0,
                                         beta_start=10000.0, beta_end=100000.0,
                                         n_total_steps=2000, eta_start=0.3)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final LSE: {:.8f}".format(best_score))

        # Coordinate peak shaving
        sc, seq = _peak_shave_coordinate(best_seq, n_iter=3000, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final coord: {:.8f}".format(best_score))

        # L-BFGS-B with very high beta
        rem = float(deadline - time.monotonic())
        if rem > 10:
            sc, seq = _run_lbfgs(best_seq, 80000.0, deadline, maxfun=2000)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final L-BFGS: {:.8f}".format(best_score))

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
