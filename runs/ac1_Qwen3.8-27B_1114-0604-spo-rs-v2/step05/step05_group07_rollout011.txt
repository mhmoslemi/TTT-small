def fast_conv(a):

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a. Assumes sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # dL/da[i]: need conv of p with reversed a, shifted
    rev_a = a[::-1]
    M2 = len(p) + n - 1

    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -50, 50))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        p = np.ones(2*n-1) / (2*n-1)
        L = c_max
    else:
        p = exp_scaled / exp_sum
        L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

    n = len(a0)
    x = a0.astype(np.float64).copy()

    if S < 1e-12:
        x[:] = 1.0/n
        S = 1.0

    if lr is None:
        # Estimate Lipschitz constant using power iteration on Hessian
        # For LSE, the Hessian is bounded by 2*beta*max(p)*2*max(a) * something
        # Use backtracking instead
        lr = 1.0 / (2.0 * beta * float(a0.max()) * 4.0 * n)
        lr = min(lr, 0.1)

        sc_y, grad_y = lse_score_and_grad(y, beta)
        
        # Backtracking line search
        step = lr
        x_trial = np.maximum(y - step * grad_y, 0.0)
        S_trial = x_trial.sum()
        if S_trial < 1e-12:
            x_trial = y - step * 0.5 * grad_y
            x_trial = np.maximum(x_trial, 0.0)
            S_trial = x_trial.sum()
            if S_trial < 1e-12:
                break
            x_trial = x_trial / S_trial
        else:
            x_trial = x_trial / S_trial
        
        sc_trial = compute_score(x_trial)
        
        # Simple backtracking: if not decreasing, reduce step
        if sc_trial > sc_y + 1e-10:
            step *= 0.5
            x_trial = np.maximum(y - step * grad_y, 0.0)
            S_trial = x_trial.sum()
            if S_trial < 1e-12:
                break
            x_trial = x_trial / S_trial
            sc_trial = compute_score(x_trial)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        y = x_trial + theta * (x_trial - x)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy < 1e-12:

        y = y / Sy
        x = x_trial
        t = t_new

        if sc_x < best_sc:

            best_x = x.copy()

    S = x.sum()
    if S < 1e-12:
        x[:] = 1.0/n

    v = np.zeros(n)

    _G['beta'] = beta
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    else:
        a0 = a0 / S
    
    bounds = [(1e-15, 1000.0)] * n
    try:
        res = minimize(
            _lbfgs_obj, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return compute_score(a0), a0
    
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out

def fista_optimize(a0, beta, n_iters=5000):
    """FISTA on LSE-smoothed objective."""
    n = len(a0)
    x = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = x.sum()
    if S < 1e-12:
        x[:] = 1.0 / n
        return x, compute_score(x)
    x = x / S
    y = x.copy()
    t_mom = 1.0
    best_x = x.copy()
    best_sc = compute_score(x)
    lr = 1.0 / (4.0 * beta + 1.0)
    
    for it in range(n_iters):
        _, grad_y = lse_score_and_grad(y, beta)
        
        # Gradient step with projection
        x_new = y - lr * grad_y
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = y * 0.99
            x_new = np.maximum(x_new, 0.0)
            S_new = x_new.sum()
            if S_new < 1e-12:
                break
            x_new = x_new / S_new
        else:
            x_new = x_new / S_new
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        theta = (t_mom - 1.0) / t_new
        y = x_new + theta * (x_new - x)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy < 1e-12:
            break
        y = y / Sy
        
        x = x_new
        t_mom = t_new
        
        sc = compute_score(x)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()
    
    return best_x, best_sc

def adam_optimize(a0, beta, n_iters=10000, lr=0.01):
    """Adam on LSE-smoothed objective."""
    n = len(a0)
    x = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = x.sum()
    if S < 1e-12:
        x[:] = 1.0 / n
        return x, compute_score(x)
    x = x / S
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy()
    best_sc = compute_score(x)
    t_count = 0
    
    for it in range(n_iters):
        t_count += 1
        _, grad = lse_score_and_grad(x, beta)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t_count)
        v_hat = v / (1 - b2**t_count)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.maximum(x, 0.0)
        S = x.sum()
        if S < 1e-12:
            break
        x = x / S
        sc = compute_score(x)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()
    
    return best_x, best_sc

def make_perturbation(a, rng, scale):
    """Diverse perturbation strategies."""
    n = len(a)
    strat = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    
    if strat == 0:
        p = a * (1.0 + scale * rng.standard_normal(n))
    elif strat == 1:
        f = 1.0 + 30.0 * rng.random()
        ph = 2.0 * np.pi * rng.random()
        p = a * (1.0 + scale * np.cos(2.0 * np.pi * f * x + ph))
    elif strat == 2:
        nb = 2 + rng.integers(0, 12)
        bs = max(1, n // nb)
        p = a.copy()
        for b in range(min(nb, n // bs)):
            s = b * bs
            e = min(s + bs, n)
            p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strat == 3:
        # Peak reduction
        c = fast_conv(a)
        ks = int(np.argmax(c))
        js = max(0, ks - n + 1)
        je = min(ks, n - 1)
        p = a.copy()
        seg = p[js:je+1]
        contrib = seg[::-1]
        tc = max(3, n // 80)
        top = np.argsort(contrib)[::-1][:tc]
        for li in top:
            p[js + li] *= (1.0 - scale * 0.5 * rng.random())
        S_old = a.sum()
        S_new = p.sum()
        if S_new < S_old - 1e-15:
            bot = np.argsort(contrib)[:tc]
            per = (S_old - S_new) / len(bot)
            for li in bot:
                p[js + li] += per
    elif strat == 4:
        p = a * np.exp(scale * rng.standard_normal(n))
    elif strat == 5:
        ns = max(1, n // 8)
        sel = rng.choice(n, size=ns, replace=False)
        p = a.copy()
        p[sel] *= (1.0 + scale * 3.0 * (rng.random(ns) - 0.5))
    elif strat == 6:
        ks = max(3, n // 30)
        kern = rng.standard_normal(ks)
        pert = np.convolve(rng.standard_normal(n), kern / np.linalg.norm(kern), 'same')
        pert = (pert - pert.mean()) / (pert.std() + 1e-10)
        p = a * (1.0 + scale * pert)
    elif strat == 7:
        # Symmetric
        h = n // 2
        ph = scale * rng.standard_normal(h)
        p = a.copy()
        p[:h] *= (1.0 + ph)
        p[h:] *= (1.0 + ph[:n-h][::-1])
    elif strat == 8:
        # Low-rank perturbation
        r = rng.integers(1, 6)
        U = rng.standard_normal((n, r))
        V = rng.standard_normal((n, r))
        pert = U @ V.T
        pert = (pert - pert.mean()) / (pert.std() + 1e-10)
        p = a * (1.0 + scale * pert)
    else:
        # Two-scale perturbation
        p = a * (1.0 + scale * rng.standard_normal(n))
        p = a * (1.0 + 0.5 * scale * rng.standard_normal(n))
    
    p = np.maximum(p, 0.0)
    S = p.sum()
    if S > 1e-12:
        p = p / S
    return p

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

    """
    Multi-phase optimizer: FISTA + Adam + L-BFGS with extreme LSE,
    aggressive ILS with diverse perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False
    
    if parent is not None:
        parent_norm = normalize(parent)
        sc_parent = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(sc_parent, parent_norm, "parent")
        n_work = len(parent)
    else:
        n_work = 8192
        parent_norm = np.ones(n_work) / n_work
        update_best(compute_score(parent_norm), parent_norm, "uniform")
    
    print(f"Starting optimization with n={n_work}")
    
    # Phase 1: FISTA warmup (20%)
    p1_end = t_start + 0.20 * total_budget
    print(f"\n--- Phase 1: FISTA (target: {p1_end - t_start:.0f}s) ---")
    for beta in [50, 200, 1000, 5000, 20000]:
        if time.monotonic() > p1_end:
            break
        rem = p1_end - time.monotonic()
        if rem < 20:
            break
        ni = int(min(8000, max(1000, rem * 4)))
        x, sc = fista_optimize(best_a, beta, n_iters=ni)
        update_best(sc, x, f"FISTA b={beta}")
    print(f"  Phase 1 done, best={best_score:.10f}")
    
    # Phase 2
