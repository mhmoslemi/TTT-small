```python
"""
Peak-directed water-filling optimization for minimum-peak autocorrelation sequences.
Novel approach: directly targets top-K convolution peaks with gradient-derived updates,
combined with LSE-smoothed Nesterov for continuous refinement and spectral A^2 shaping.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def nesterov_lse(a, max_time, beta_start, beta_end):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    it = 0
    beta = beta_start
    beta_growth = (beta_end / beta_start) ** (1.0 / max(1, int(max_time / 0.05)))
    lr = 0.02
    mu = 0.9
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.0001 * it)
        lr *= 0.9995
        
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        beta = min(beta * beta_growth, beta_end)
        
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    
    return best_a

def peak_water_filling(a, max_time, K_init=10, K_end=50, temp_init=1.0, temp_end=0.1):
    """
    Peak-directed water-filling: directly reduce top-K peaks of conv(a,a).
    Uses a temperature parameter to control the "softness" of peak targeting.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    t_end = t_start + max_time
    
    v = np.zeros(n)  # momentum
    it = 0
    
    while time.monotonic() < t_end:
        it += 1
        t_frac = (time.monotonic() - t_start) / max_time
        K = int(K_init + (K_end - K_init) * t_frac)
        temp = temp_init + (temp_end - temp_init) * t_frac
        lr = 0.5 * temp / (1.0 + 0.1 * it)
        
        c = fast_conv(a)
        c_mean = float(c.mean())
        
        # Find top K peaks
        # Use "excess" over mean as the target
        excess = np.maximum(c - c_mean, 0)
        
        # Weight by excess (softmax-like)
        scaled = excess / temp
        weights = scaled / (scaled.sum() + 1e-15)

        # This is a weighted convolution: g = 2 * conv(weights_rev, a) shifted
        g = np.zeros(n)
        for k in range(2*n-1):
            if weights[k] < 1e-10:
                continue
            # a[k-i] is valid for i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                g[i_start:i_end+1] += weights[k] * 2.0 * a[k-i_end:k-i_start+1][::-1]
        
        # Apply update with momentum
        v = 0.9 * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a_new = a_new / S_new
        
        if it % 50 == 0:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new
        else:
            a = a_new
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a
```

```python
"""
Peak-directed water-filling + LSE Nesterov + spectral A^2 shaping for minimum-peak
self-convolution sequences. Novel approach combines direct top-K peak targeting with
smooth LSE refinement and multi-scale warm start from ramp-based inits.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def nesterov_lse(a, max_time, beta_start, beta_end, lr_init=0.02, mu_init=0.90, mu_max=0.97):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    lr = lr_init
    mu = mu_init
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(mu_max, mu_init + 0.0002 * it)
        lr *= 0.9997
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        beta = min(beta * 1.003, beta_end)
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def peak_water_filling(a, max_time, K_init=20, K_end=200, lr_init=0.08, momentum=0.80):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    v = np.zeros(n)
    it = 0
    lr = lr_init
    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = min(1.0, (time.monotonic() - t_start) / max_time)
        K = int(K_init + (K_end - K_init) * t_frac)
        lr *= 0.9995
        mu = momentum + 0.05 * t_frac

        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:K]
        c_vals = c[peak_idx].astype(np.float64)
        c_spread = max(1e-12, c_vals[0] - c_vals[-1])

def fienup_a2(a, n_iters=200, thresh=1.5, clip=0.5):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64))
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        C = A * A
        c = np.fft.irfft(C, M_fft)[:2 * n - 1]
        c_max = float(c.max())
        c_mean = float(c.mean())
        if c_max < 1e-15:
            break
        ratio = c_max / (c_mean + 1e-15)
        if ratio < thresh:
            break
        excess = np.maximum(c - c_mean * thresh, 0.0)
        excess_sum = float(excess.sum())
        if excess_sum < 1e-15:
            break
        scale = 1.0 - clip * excess / c_max
        C_new = C * scale
        A_new = np.sqrt(C_new)
        A_new = np.where(C_new.real >= 0, A_new, A * np.abs(A_new) / (np.abs(A) + 1e-15))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_standard(a, n_iters=300, thresh_init=2.0, thresh_end=1.1, clip_init=0.3, clip_end=0.8, momentum_end=0.4):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64))
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t
        momentum = momentum_end * 0.5 * (1 - np.cos(np.pi * t))
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1e6).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1e6)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
                                'maxfun': maxfun, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1e6)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_ramp_init(n, p=1.0):
    x = np.arange(n, dtype=np.float64)
    a = (x + 1.0) ** p
    return a / a.sum()


def build_bump_init(n, center_frac=0.6, sigma_frac=0.25):
    x = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_two_bump_init(n, c1=0.3, c2=0.7, s=0.15):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - c1 * n) / (s * n)) ** 2) + np.exp(-0.5 * ((x - c2 * n) / (s * n)) ** 2)
    return a / a.sum()


def build_cosine_weighted_ramp(n, p=0.5):
    x = np.arange(n, dtype=np.float64) / n
    a = (x + 0.01) ** p * (0.5 + 0.5 * np.cos(np.pi * x))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def peak_coord_descent(a, n_sweeps=3, n_top=20):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        for k_idx in peak_indices:
            k = int(k_idx)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local
            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for factor in [0.6, 0.75, 0.85, 1.15, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1e6)
    except Exception:
        pass

    n_target = 6962
    if parent is not None and parent.size == n_target:
        n_target = parent.size

    best_score = np.inf
    best_a = np.ones(n_target)

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1e6)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"=== Starting search (budget={total:.0f}s, n_target={n_target}) ===")

    # Phase 1: Multi-init rapid screening (15% time)
    phase1_end = t_start + 0.15 * (total - 30)
    print(f"\n--- Phase 1: Multi-init screening ({phase1_end - t_start:.0f}s) ---")

    inits = {}
    if parent is not None:
        inits['parent'] = parent.astype(np.float64).copy()
    inits['ramp1'] = build_ramp_init(n_target, p=1.0)
    inits['ramp05'] = build_ramp_init(n_target, p=0.5)
    inits['bump'] = build_bump_init(n_target, 0.55, 0.3)
    inits['two_bump'] = build_two_bump_init(n_target, 0.3, 0.7, 0.2)
    inits['cos_ramp'] = build_cosine_weighted_ramp(n_target, 0.7)
    inits['uniform'] = np.ones(n_target)
    inits['ramp2'] = build_ramp_init(n_target, p=2.0)

    candidates = []
    for name, a_init in inits.items():
        if time.monotonic() > phase1_end - 30:
            break
        a_init = normalize_nonneg(a_init)
        sc_init = compute_score(a_init)
        # Quick Fienup
        a_f = fienup_standard(a_init, n_iters=80, thresh_init=2.0, thresh_end=1.2,
                              clip_init=0.3, clip_end=0.7, momentum_end=0.3)
        sc_f = compute_score(a_f)
        # Quick water filling
        a_w = peak_water_filling(a_f, max_time=15.0, K_init=20, K_end=80, lr_init=0.05)
        sc_w = compute_score(a_w)
        best_local = min(sc_init, sc_f, sc_w)
        a_best_local = a_init if best_local == sc_init else (a_f if best_local == sc_f else a_w)
        candidates.append((best_local, a_best_local, name))
        print(f"  {name}: init={sc_init:.6f}, fienup={sc_f:.6f}, wf={sc_w:.6f}")

    candidates.sort(key=lambda x: x[0])
    update_best(candidates[0][0], candidates[0][1], candidates[0][2])
    print(f"Phase 1 best: {best_score:.8f} (from {candidates[0][2]})")

    # Phase 2: Peak water-filling + LSE alternation (35% time)
    phase2_end = t_start + 0.50 * (total - 30)
    print(f"\n--- Phase 2: Water-filling + LSE alternation ({phase2_end - t_start:.0f}s) ---")

    a_curr = best_a.copy()
    stall_count = 0
    stall_threshold = 300
    last_improve = time.monotonic()

    while time.monotonic() < phase2_end - 20:
        seg_budget = min(60.0, phase2_end - time.monotonic() - 20)
        if seg_budget < 10:
            break

        # Alternate: water-filling then LSE
        a_wf = peak_water_filling(a_curr, max_time=seg_budget * 0.5,
                                  K_init=30, K_end=150, lr_init=0.06)
        sc_wf = compute_score(a_wf)
        if sc_wf < best_score:
            update_best(sc_wf, a_wf, "wf")
            a_curr = a_wf
            last_improve = time.monotonic()
            stall_count = 0
        else:
            stall_count += 1

        if time.monotonic() < phase2_end - 25:
            seg2 = min(40.0, phase2_end - time.monotonic() - 25)
            if seg2 > 10:
                a_nl = nesterov_lse(a_curr, max_time=seg2, beta_start=50.0, beta_end=2000.0,
                                    lr_init=0.015, mu_init=0.92, mu_max=0.96)
                sc_nl = compute_score(a_nl)
                if sc_nl < best_score:
                    update_best(sc_nl, a_nl, "lse")
                    a_curr = a_nl
                    last_improve = time.monotonic()
                    stall_count = 0
                else:
                    stall_count += 1

        # Restart if stalled
        if stall_count >= stall_threshold or (time.monotonic() - last_improve > 120):
            print(f"  [restart] Stalled for {stall_count} iters, perturbing...")
            a_pert = a_curr * (1.0 + 0.05 * rng.standard_normal(n_target))
            a_curr = normalize_nonneg(a_pert)
            stall_count = 0
            last_improve = time.monotonic()

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Extended LSE Nesterov with multiple configs (25% time)
    phase3_end = t_start + 0.75 * (total - 30)
    print(f"\n--- Phase 3: Extended LSE ({phase3_end - t_start:.0f}s) ---")

    lse_configs = [
        {'beta_start': 10.0, 'beta_end': 5000.0, 'lr_init': 0.02, 'mu_init': 0.90, 'mu_max': 0.97},
        {'beta_start': 30.0, 'beta_end': 15000.0, 'lr_init': 0.01, 'mu_init': 0.93, 'mu_max': 0.97},
        {'beta_start': 5.0, 'beta_end': 3000.0, 'lr_init': 0.03, 'mu_init': 0.88, 'mu_max': 0.95},
    ]

    for cfg in lse_configs:
        if time.monotonic() > phase3_end - 30:
            break
        seg = min(180.0, phase3_end - time.monotonic() - 30)
        if seg < 20:
            break
        a_nl = nesterov_lse(best_a, max_time=seg, **cfg)
        update_best(compute_score(a_nl), a_nl, "lse-ext")

    # Also from a perturbed best
    if time.monotonic() < phase3_end - 60:
        a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_target))
        a_pert = normalize_nonneg(a_pert)
        seg = min(120.0, phase3_end - time.monotonic() - 30)
        if seg > 20:
            a_nl2 = nesterov_lse(a_pert, max_time=seg, beta_start=20.0, beta_end=8000.0,
                                 lr_init=0.015, mu_init=0.91, mu_max=0.96)
            update_best(compute_score(a_nl2), a_nl2, "lse-pert")

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Fienup + A^2 spectral (10% time)
    phase4_end = t_start + 0.85 * (total - 30)
    print(f"\n--- Phase 4: Spectral refinement ({phase4_end - t_start:.0f}s) ---")

    a_f1 = fienup_standard(best_a, n_iters=400, thresh_init=1.8, thresh_end=1.05,
                           clip_init=0.4, clip_end=0.9, momentum_end=0.5)
    update_best(compute_score(a_f1), a_f1, "fienup-fin")

    if time.monotonic() < phase4_end - 30:
        a_f2 = fienup_a2(best_a, n_iters=150, thresh=1.3, clip=0.4)
        update_best(compute_score(a_f2), a_f2, "fienup-a2")

    # Water-filling with high K
    if time.monotonic() < phase4_end - 20:
        a_wf2 = peak_water_filling(best_a, max_time=60.0, K_init=100, K_end=500, lr_init=0.03)
        update_best(compute_score(a_wf2), a_wf2, "wf-highK")

    print(f"After Phase 4: best = {best_score:.8f}")

    # Phase 5: L-BFGS polish (8% time)
    phase5_end = t_start + 0.93 * (total - 30)
    print(f"\n--- Phase 5: L-BFGS polish ({phase5_end - t_start:.0f}s) ---")

    for beta in [500, 2000, 5000, 10000]:
        if time.monotonic() > phase5_end - 15:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, f"lbfgs-{beta}")

    print(f"After Phase 5: best = {best_score:.8f}")

    # Phase 6: Final coordinate descent + micro LSE
    print(f"\n--- Phase 6: Final CD ---")
    a_cd = peak_coord_descent(best_a, n_sweeps=3, n_top=20)
    update_best(compute_score(a_cd), a_cd, "cd")

    if time.monotonic() < deadline - 30:
        a_final = nesterov_lse(best_a, max_time=min(60.0, deadline - time.monotonic() - 20),
                               beta_start=2000.0, beta_end=50000.0,
                               lr_init=0.005, mu_init=0.95, mu_max=0.98)
        update_best(compute_score(a_final), a_final, "micro-lse")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
```
