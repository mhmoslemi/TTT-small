def smooth_perturb(a, sigma, rng):
    n=len(a)
    noise=rng.standard_normal(n)
    N=next_pow2(n)
    freq=np.fft.rfftfreq(N)
    scale = 0.15
    window = 1.0/(1.0+(freq/scale)**4)
    # set window[0]=0 to remove mean
    if len(window)>0: window[0]=0.0
    noise_spec=np.fft.rfft(noise,N)
    noise_spec *= window
    h=np.fft.irfft(noise_spec,N)[:n]
    mean=float(np.mean(h)); std=float(np.std(h))
    if std < 1e-300: return normalize_pos(a)
    h=(h-mean)/std*sigma
    u=h
    u -= u.max()
    new=a*np.exp(u)
    return normalize_pos(new)
```

def interpolate_seq(parent,n):
   p=np.asarray(parent,dtype=np.float64).ravel()
   p=np.maximum(p,0.0)
   if p.size == 0:
       return np.ones(n)/n
   if p.size == 1:
       return np.array([p[0]]*n,dtype=np.float64)
   if p.size == n:
       out=p.copy()
   else:
       x_old=np.linspace(0.0,1.0,p.size, endpoint=True)
       x_new=np.linspace(0.0,1.0,n, endpoint=True)
       out=np.interp(x_new, x_old, p)
   out=np.maximum(out,0.0)
   s=out.sum()
   if s<1e-300: out=np.ones(n)/n
   else: out/=s
   return out
```

def lp_solve_fista(a, M, n_iters, beta, x_init=None):
    n=len(a)
    m=2*n-1
    N=next_pow2(m)
    A=np.fft.rfft(a,N)
    if x_init is not None:
        x=np.asarray(x_init,dtype=np.float64).copy()
        x=np.maximum(x,0.0)
        s=x.sum()
        if s<1e-300: x=np.ones(n)/n
        else: x=x/s
    else:
        x=np.ones(n)/n
    u=x.copy()
    t_mom=1.0
    for it in range(n_iters):
        X=np.fft.rfft(x,N)
        c=np.fft.irfft(A*X,N)[:m]
        z=beta*(c-M)
        zmax=float(np.max(z))
        expz=np.exp(z-zmax)
        s=float(np.sum(expz))
        if s<=1e-300 or not np.isfinite(s): p=np.ones(m)/m
        else: p=expz/s
        P=np.fft.rfft(p,N)
        grad=np.fft.irfft(P*np.conjugate(A),N)[:n] - 1.0
        grad=np.where(np.isfinite(grad), grad, 0.0)
        v=u - 0.8*grad
        x_new=np.maximum(v,0.0)
        t_new=(1.0+np.sqrt(1.0+4.0*t_mom*t_mom))*0.5
        u=x_new + ((t_mom-1.0)/t_mom)*(x_new - x)
        u=np.maximum(u,0.0)
        x=x_new
        t_mom=t_new
        sx=float(np.sum(x))
        if not np.isfinite(sx) or sx>1e100:
            break
    return x

"""
High-budget non-negative sequence optimizer for minimizing
    2*n*max(conv(a,a)) / (sum(a))^2

Main ideas:
- Fast FFT-based exact scoring and structured gradient directions.
- Power-mean / L-inf smoothing: for q = p-1, use direction proportional to
  sum_k (c_k/M)^q * (a*c)_k, and do multiplicative line-search steps.
- Cheap alternating LP directions with corrected FISTA gradients.
- Multi-length trial pool, smooth perturbations to escape plateaus,
  and final polish on the best candidate.

The final returned list is always non-negative and normalized to sum 1.
"""

import numpy as np
import time

_EPS_FLOOR = 1e-16


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def self_conv_fft(a):
    n = len(a)
    m = 2 * n - 1
    N = next_pow2(m)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:m]


def compute_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-300:
        return np.inf
    c = self_conv_fft(a)
    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M) / (S * S)


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-300:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a = a / s
    # Keep a tiny positive floor so multiplicative moves can recover zeros.
    a = a + _EPS_FLOOR
    a = a / float(np.sum(a))
    return a


def score_dir_power(a, q):
    """
    Score and direction for power-smoothed peak reduction.

    For q = p-1, this is proportional to the gradient of
    (sum_k c_k^p)^(1/p) under current sum=1 normalization.
    The factor of 2 in the identity gradient is included but later
    renormalized away by the step.
    """
    n = len(a)
    m = 2 * n - 1
    N = next_pow2(m)

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:m]

    S = float(np.sum(a))
    if S < 1e-300:
        return np.inf, None

    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    score = float(2.0 * n * M) / (S * S)

    if M <= 1e-300:
        d = np.zeros(n, dtype=np.float64)
        return score, d

    ratio = c / M
    np.clip(ratio, 0.0, 1.0, out=ratio)

    if q == 1:
        w = ratio.copy()
    else:
        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            w = ratio ** float(q)
        w = np.where(np.isfinite(w), w, 0.0)

    ws = float(np.sum(w))
    if ws <= 1e-300 or not np.isfinite(ws):
        w = np.ones(m, dtype=np.float64) / m
    else:
        w = w / ws

    P = np.fft.rfft(w, N)
    d = np.fft.irfft(P * np.conjugate(A), N)[:n]
    d = d * 2.0
    d = np.where(np.isfinite(d), d, 0.0)
    return score, d


def mul_step_best(a, h, cur_score, eta):
    """
    Multiplicative line search on the simplex:
        a_new = normalize( a * exp(-eta*h) )
    h should have been centered and scaled so max|h| ~= 1.
    """
    best_score = cur_score
    best_a = None
    best_eta = eta

    cands = []
    for mult in (0.5, 1.0, 2.0):
        et = float(eta) * float(mult)
        if et > 0.0:
            found = False
            for z in cands:
                if abs(z - et) < 1e-15:
                    found = True
                    break
            if not found:
                cands.append(et)

    if not cands:
        return None, cur_score, max(1e-6, eta * 0.5)

    for et in cands:
        u = -et * h
        u -= float(np.max(u))
        u = np.where(np.isfinite(u), u, 0.0)
        # No overflow because max(u) == 0; still guard underflow robustly.
        u = np.maximum(u, -700.0)

        new = a * np.exp(u)
        new = np.maximum(new, 0.0)
        new = new + _EPS_FLOOR
        s = float(np.sum(new))
        if s < 1e-300:
            continue
        new = new / s
        sc = compute_score(new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = new
            best_eta = et

    if best_a is None:
        eta2 = max(1e-6, eta * 0.5)
        return None, cur_score, eta2

    # Mild adaptive eta.
    if best_eta >= eta * 1.5:
        eta2 = min(1.0, eta * 1.15)
    elif best_eta <= eta * 0.6:
        eta2 = max(1e-6, eta * 0.75)
    else:
        eta2 = eta

    return best_a, best_score, eta2


def smooth_perturb(a, sigma, rng):
    """Add a low-frequency positive multiplicative perturbation."""
    n = len(a)
    noise = rng.standard_normal(n)
    N = next_pow2(n)
    freq = np.fft.rfftfreq(N)
    scale = 0.06
    window = 1.0 / (1.0 + (freq / scale) ** 4)
    if len(window) > 0:
        window[0] = 0.0

    F = np.fft.rfft(noise, N)
    F = F * window
    h = np.fft.irfft(F, N)[:n]

    mh = float(np.mean(h))
    sh = float(np.std(h))
    if sh < 1e-300:
        return normalize_pos(a)

    h = (h - mh) / sh * sigma
    h = np.where(np.isfinite(h), h, 0.0)

    u = h
    u -= float(np.max(u))
    u = np.maximum(u, -700.0)

    new = a * np.exp(u)
    new = np.maximum(new, 0.0)
    new = new + 1e-14
    return normalize_pos(new)


def optimize_power(a, deadline, rng, q_cycle=None, eta_init=0.12, verbose=False):
    """
    Multiplicative gradient-style local optimizer.
    Uses power-mean directions q = p-1 for p large.
    """
    if q_cycle is None:
        q_cycle = (3, 7, 15, 31, 63, 15, 31, 63, 127, 15)

    a = normalize_pos(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_score = best_sc
    eta = float(eta_init)

    start = time.monotonic()
    last_improve_t = start
    last_perturb_t = start - 1000.0
    last_print_sc = best_sc
    last_print_t = start
    perturbs = 0
    max_perturbs = 12

    local_limit = 35

    while time.monotonic() < deadline - 1.0:
        if cur_score > best_sc + 1e-9:
            a = best_a.copy()
            cur_score = best_sc

        for q in q_cycle:
            if time.monotonic() > deadline - 1.0:
                break

            local_no = 0
            while local_no < local_limit and time.monotonic() < deadline - 1.0:
                sc, d = score_dir_power(a, int(q))
                if d is None or not np.isfinite(sc):
                    local_no = local_limit + 1
                    break

                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    last_improve_t = time.monotonic()
                    local_no = 0
                    if verbose:
                        now = time.monotonic()
                        if best_sc < last_print_sc - 1e-9 and (now - last_print_t) > 30.0:
                            print("    power best: %.9f  (n=%d, t=%.0fs)" % (best_sc, len(a), now - start))
                            last_print_sc = best_sc
                            last_print_t = now

                h = d - float(np.dot(a, d))
                h = np.where(np.isfinite(h), h, 0.0)
                maxh = float(np.max(np.abs(h)))
                if maxh < 1e-300:
                    local_no += 1
                    if local_no >= local_limit:
                        break
                    continue

                h = h / maxh
                new_a, new_sc, eta = mul_step_best(a, h, cur_score, eta)
                if new_a is not None and new_sc < cur_score:
                    a = new_a
                    cur_score = new_sc
                    local_no = 0

                    if new_sc < best_sc:
                        best_sc = new_sc
                        best_a = a.copy()
                        last_improve_t = time.monotonic()
                        if verbose:
                            now = time.monotonic()
                            if best_sc < last_print_sc - 1e-9 and (now - last_print_t) > 30.0:
                                print("    power best: %.9f  (n=%d, t=%.0fs)" % (best_sc, len(a), now - start))
                                last_print_sc = best_sc
                                last_print_t = now
                else:
                    local_no += 1

        now = time.monotonic()
        if now > deadline - 1.0:
            break

        if now - last_improve_t > 200.0 and now - last_perturb_t > 120.0 and perturbs < max_perturbs:
            sig = 0.02 * (1.0 + 0.25 * float(perturbs))
            a = smooth_perturb(best_a, sig, rng)
            cur_score = compute_score(a)
            if cur_score < best_sc:
                best_sc = cur_score
                best_a = a.copy()
                last_improve_t = now
            perturbs += 1
            last_perturb_t = now
        elif now - last_improve_t > 450.0 and perturbs >= 8:
            # Hard stall: stop so outer phases can try other candidates.
            break

    return best_a, best_sc


def lp_solve_fista(a, M, n_iters, beta, x_init=None):
    """
    Solve approximately:
        max  sum(x)
        s.t. conv(a,x)_k <= M for all k, x >= 0

    using FISTA on a strict saddle/smooth surrogate
        f(x) = (1/beta)*logsumexp(beta(conv(a,x)-M)) - sum(x).

    The gradient is B^T p - 1, where p is the softmax over lags.
    """
    n = len(a)
    m = 2 * n - 1
    N = next_pow2(m)
    A = np.fft.rfft(a, N)

    if x_init is not None:
        x = np.asarray(x_init, dtype=np.float64).copy()
        x = np.maximum(x, 0.0)
        sx = float(np.sum(x))
        if sx < 1e-300 or not np.isfinite(sx):
            x = np.ones(n, dtype=np.float64) / n
        else:
            x = x / sx
    else:
        x = np.ones(n, dtype=np.float64) / n

    u = x.copy()
    t_old = 1.0

    for _ in range(int(n_iters)):
        X = np.fft.rfft(x, N)
        c = np.fft.irfft(A * X, N)[:m]

        z = float(beta) * (c - M)
        zmax = float(np.max(z))
        expz = np.exp(z - zmax)
        s = float(np.sum(expz))
        if s <= 1e-300 or not np.isfinite(s):
            p = np.ones(m, dtype=np.float64) / m
        else:
            p = expz / s

        P = np.fft.rfft(p, N)
        grad = np.fft.irfft(P * np.conjugate(A), N)[:n]
        grad = grad - 1.0
        grad = np.where(np.isfinite(grad), grad, 0.0)

        # The Hessian of the smooth part has operator norm at most about 1
        # when a is nonnegative and sum(a)=1, so a slightly conservative step.
        step = 0.8
        v = u - step * grad
        x_new = np.maximum(v, 0.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        u = x_new + ((t_old - 1.0) / t_new) * (x_new - x)
        u = np.maximum(u, 0.0)

        x = x_new
        t_old = t_new

        sx = float(np.sum(x))
        if not np.isfinite(sx) or sx > 1e100:
            break

    return x


def lp_optimize(a, deadline, rng, beta_schedule=None, fista_iters=None):
    """
    Alternating LP-direction optimizer:
    1. Fix current a, find x maximizing sum(x) under conv(a,x) <= M.
    2. Normalize x and line-search on mixing (1-t)a + t x.
    """
    if beta_schedule is None:
        beta_schedule = (10.0, 25.0, 50.0, 100.0)

    a = normalize_pos(a)
    if fista_iters is None:
        fista_iters = 350 if len(a) <= 12000 else 220

    best_a = a.copy()
    best_sc = compute_score(a)
    x_warm = None
    no_improve = 0

    t_list = (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        for _ in range(4):
            if time.monotonic() > deadline - 5.0:
                break

            c = self_conv_fft(a)
            M = float(np.max(c))
            if M <= 1e-300 or not np.isfinite(M):
                break

            x = lp_solve_fista(a, M, fista_iters, float(beta), x_init=x_warm)
            x_warm = x
            sx = float(np.sum(x))
            if sx <= 1e-300 or not np.isfinite(sx):
                break

            x_norm = x / sx
            cur_sc = compute_score(a)

            best_mix = None
            best_mix_sc = cur_sc
            for t in t_list:
                b = (1.0 - t) * a + t * x_norm
                b = normalize_pos(b)
                sc = compute_score(b)
                if np.isfinite(sc) and sc < best_mix_sc:
                    best_mix_sc = sc
                    best_mix = b

            if best_mix is not None and best_mix_sc < cur_sc - 1e-13:
                a = best_mix
                if a is not None and best_mix_sc < best_sc:
                    best_sc = best_mix_sc
                    best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 3:
                    break
                a = smooth_perturb(best_a, 0.01, rng)

    return best_a, best_sc


def interpolate_seq(parent, n):
    p = np.asarray(parent, dtype=np.float64).ravel()
    if p.size == 0:
        out = np.ones(n, dtype=np.float64) / float(n)
        return normalize_pos(out)
    p = np.maximum(p, 0.0)

    if p.size == 1:
        out = np.full(n, p[0], dtype=np.float64)
        return normalize_pos(out)

    if p.size == n:
        out = p.copy()
    else:
        x_old = np.linspace(0.0, 1.0, p.size, endpoint=True)
        x_new = np.linspace(0.0, 1.0, n, endpoint=True)
        out = np.interp(x_new, x_old, p)

    out = np.maximum(out, 0.0)
    out = out + 1e-12
    out = out / float(np.sum(out))
    return out


def default_sequence(n, seed=1):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.55 * np.sin(2.0 * np.pi * x + 0.35)
    a = a + 0.20 * np.cos(4.0 * np.pi * x + 1.10)
    a = np.maximum(a, 1e-8)
    a = smooth_perturb(a, 0.05, rng)
    return normalize_pos(a)


def harmonic_sequence(n, amp, phase, seed=2):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * x + phase)
    a = a + 0.30 * amp * np.cos(4.0 * np.pi * x + 0.5 * phase)
    a = a + 0.15 * amp * np.sin(6.0 * np.pi * x + 0.2 * phase)
    a = np.maximum(a, 1e-8)
    a = smooth_perturb(a, 0.03, rng)
    return normalize_pos(a)

        s2 = compute_score(a2)
        if np.isfinite(s2) and s2 < best_sc:
            best_a = a2.copy()
            best_sc = s2
        print("After LP phase: best=%.10f n=%d" % (best_sc, len(best_a)))

    # Optimize a second candidate (often a different length) if time remains.
    cand2_deadline = t_start + 0.82 * usable
    if L >= 2 and time.monotonic() < cand2_deadline - 60.0:
        a3, s3 = optimize_power(quick_results[1][1], cand2_deadline, rng, verbose=False)
        s3 = compute_score(a3)
        if np.isfinite(s3) and s3 < best_sc:
            best_a = a3.copy()
            best_sc = s3
            print("Second candidate improved: best=%.10f n=%d" % (best_sc, len(best_a)))

    # Try a third candidate if available and still ample time.
    cand3_deadline = t_start + 0.92 * usable
    if L >= 3 and time.monotonic() < cand3_deadline - 40.0:
        a4, s4 = optimize_power(quick_results[2][1], cand3_deadline, rng, verbose=False)
        s4 = compute_score(a4)
        if np.isfinite(s4) and s4 < best_sc:
            best_a = a4.copy()
            best_sc = s4
            print("Third candidate improved: best=%.10f n=%d" % (best_sc, len(best_a)))

    # Final polish on the current global best.
    if time.monotonic() < safe_deadline - 10.0:
        a5, s5 = optimize_power(best_a, safe_deadline, rng, verbose=True)
        s5 = compute_score(a5)
        if np.isfinite(s5) and s5 < best_sc:
            best_a = a5.copy()
            best_sc = s5

    print("Recommended internal best: %.10f n=%d" % (best_sc, len(best_a)))

    # Safe output conversion.
    out = []
    for x in best_a:
        y = float(x)
        if not np.isfinite(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        out.append(y)

    # Normalize one last time to make the sum invariant robust.
    ssum = float(sum(out))
    if ssum < 1e-12:
        out = [0.5] * 64
    else:
        out = [max(0.0, min(1000.0, x / ssum)) for x in out]

    # If an external evaluator is available, verify the sequence.
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = np.inf

    # Fall back if verification fails dramatically.
    if not np.isfinite(verified):
        fallback = parent if parent_exists else default_sequence(8192, 7)
        out2 = []
        for x in fallback:
            y = float(x)
            if not np.isfinite(y):
                y = 0.0
            if y < 0.0:
                y = 0.0
            if y > 1000.0:
                y = 1000.0
            out2.append(y)
        ssum2 = float(sum(out2))
        if ssum2 > 1e-12:
            out = [x / ssum2 for x in out2]
        else:
            out = [0.5] * 64
        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = np.inf

    elapsed = time.monotonic() - t_start
    print("Final verified score: %.10f, length=%d, elapsed=%.1fs" % (verified, len(out), elapsed))
    return out
