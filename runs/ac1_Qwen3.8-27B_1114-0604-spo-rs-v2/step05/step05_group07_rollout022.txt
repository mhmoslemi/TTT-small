"""
LP-based optimizer for non-negative step function minimization.
Uses the linear programming "best response" iteration as the primary engine,
combined with L-BFGS and subgradient polishing at full resolution.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix_sparse(a):
    """Build the sparse (2n-1) x n matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For column j, valid k values are: k = j + i where i ranges over valid a indices
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        # Also 0 <= k < 2n-1
        k_start = j
        k_end = min(j + n, 2 * n - 1) - 1  # inclusive
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

"""
LP-driven optimizer with full-resolution polishing.
Primary engine: iterative LP best-response on a decimated signal (using HiGHS).
Polishing: L-BFGS with LSE smoothing + subgradient refinement + targeted peak flattening.
Key innovation: the LP step computes the exact best response b maximizing sum(b)
subject to (a*b)[k] <= M for all k, which is a convex LP and provides a much
stronger direction than single-peak subgradient methods.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def decimate_signal(a, n_target):
    """Interpolate signal a to length n_target, preserving non-negativity."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, a)
    return np.clip(out, 0.0, 1e6)


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix A where A[k,j] = a[k-j] for valid (k,j).
    Each column j has n non-zeros at rows j..j+n-1 with values a[0..n-1]."""
    n = len(a)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.tile(np.arange(n, dtype=np.int32), n) + np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(a.astype(np.float64), n)
    return sparse.csc_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_lp_best_response(a, M):
    """Solve max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns optimal b or None on failure."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            b = np.clip(res.x, 0.0, None)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None


def lp_iteration(a, M=None, max_ls=8):
    """One LP-based iteration: solve best response, line search, update.
    Returns (new_a, improved)."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a, False
    a_norm = a / S

    c = fast_conv(a_norm)
    if M is None:
        M = float(np.max(c))
    if M < 1e-15:
        return a, False

    b = solve_lp_best_response(a_norm, M)
    if b is None:
        return a, False

    b_norm = b / b.sum()
    sc_current = compute_score(a_norm)

    for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
        a_new = (1.0 - t) * a_norm + t * b_norm
        a_new = np.clip(a_new, 0.0, 1e6)
        Snew = a_new.sum()
        if Snew < 1e-12:
            continue
        a_new = a_new / Snew
        sc_new = compute_score(a_new)
        if sc_new < sc_current:
            return a_new, True

    return a, False


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (on normalized simplex)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def mirror_descent(a0, beta=100, n_iters=5000, eta0=0.05, decay=0.9998):
    """Mirror descent with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    y = a.copy()
    t = 1.0

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def targeted_peak_flatting(a, n_sweeps=15, n_top=80):
    """Coordinate descent: reduce peak contributors, redistribute to low contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        top_count = min(n_top, max(5, total_len // 4))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]
        bottom_global = j_start + bottom_local

        for ji in range(min(len(top_global), 30)):
            j = top_global[ji]
            a_old = a[j]
            if a_old < 1e-12:
                continue
            best_val = a_old
            best_local = best_sc
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 0.99]:
                new_val = a_old * frac
                reduction = a_old - new_val
                if reduction > 1e-15:
                    a[j] = new_val
                    per = reduction / len(bottom_global)
                    a[bottom_global] += per
                else:
                    a[j] = new_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = new_val
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=5000, f_lb=1.0, top_k=5):
    """Subgradient using top-k peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.15

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(c))
        for rank, idx in enumerate(top_idx):
            weights[idx] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)

        Mak_weighted = float(np.sum(weights * c))
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        S2 = S_a * S_a
        S3 = S2 * S_a
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]

        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (0.9995 ** it)
        step = min(step, 2.0 * max(sc - f_lb, 1e-10) / (gnorm + 1e-30))
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        mid = len(seg) // 2
        a_p[j_start + mid:j_start + mid + max(1, n // 20)] *= (1.0 - scale)
        S = a_p.sum()
        if S > 1e-12:
            a_p = a_p / S

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

    print(f"{'='*50}")

    a_ns = nesterov_subgrad(best_a, n_iters=5000, f_lb=1.0)
    report(compute_score(a_ns), a_ns, "nesterov")

    a_mp = multi_peak_subgrad(best_a, n_iters=3000, f_lb=1.0, top_k=5)
    report(compute_score(a_mp), a_mp, "multi_peak")

    a_tpf = targeted_peak_flatting(best_a, n_sweeps=20, n_top=100)
    report(compute_score(a_tpf), a_tpf, "peak_flat")

    # Another round on the best
    if time.monotonic() < phase3_deadline - 60:
        a_ns2 = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
        report(compute_score(a_ns2), a_ns2, "nesterov2")

        a_tpf2 = targeted_peak_flatting(best_a, n_sweeps=15, n_top=60)
        report(compute_score(a_tpf2), a_tpf2, "peak_flat2")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Final high-beta L-BFGS + LP micro-iterations
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*50}")

    for beta in [200000, 500000]:
        if time.monotonic() > deadline - 30:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=20000, maxfun=80000)
        report(sc, a_opt, f"final LBFGS b={beta}")

    # A few more LP iterations at the best resolution
    if time.monotonic() < deadline - 60:
        n_lp_final = min(2500, len(best_a))
        a_lp = decimate_signal(best_a, n_lp_final)
        S = a_lp.sum()
        if S > 1e-12:
            a_lp = a_lp / S
        for it in range(30):
            if time.monotonic() > deadline - 30:
                break
            a_lp, _ = lp_iteration(a_lp)
            sc = compute_score(a_lp)
            report(sc, a_lp, f"final LP it={it}")
            # Also check at full length
            a_full_check = decimate_signal(a_lp, len(best_a))
            sc_full = compute_score(a_full_check)
            report(sc_full, a_full_check, f"final LP full it={it}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*50}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*50}")
    return out
