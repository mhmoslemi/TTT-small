    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    beta_growth = 1.008
    lr = 0.01
    lr_decay = 0.9995
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.97, 0.9 + 0.00003 * it)
        lr *= lr_decay

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute LSE gradient
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step
        effective_lr = min(lr, 0.05 / (gnorm + 1e-10))

        # Update
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Increase beta
        beta = min(beta * beta_growth, beta_end)

        # Track best periodically
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def subgradient_refine(a, max_time, n_top_peaks=3, alpha_start=0.05, alpha_end=0.001):
    """Exact subgradient refinement on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        # Cosine-annealed step size
        t_frac = min(1.0, (time.monotonic() - t_start) / max_time)
        alpha = alpha_start + (alpha_end - alpha_start) * 0.5 * (1 - np.cos(np.pi * t_frac))

        c = fast_conv(a)
        # Get top peaks
        top_k = min(n_top_peaks, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        # Weighted subgradient: weight peaks by how much they exceed mean
        mean_c = 1.0 / (2 * n - 1)
        weights = np.zeros(np.max(peak_indices) + 1)
        for k in peak_indices:
            k = int(k)
            excess = max(0.0, c[k] - mean_c)
            weights[k] = 0.5 * (1.0 + excess / (c[peak_indices[0]] + 1e-15))

        # Compute combined subgradient
        g = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            if weights[k] > 0:
                g += weights[k] * get_subgradient(a, k)

        # Normalize subgradient
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm

        # Step and project
        a_new = a - alpha * g
        a_new = project_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # If no improvement, try smaller step
            a_new2 = a - alpha * 0.3 * g
            a_new2 = project_simplex(a_new2)
            sc2 = compute_score(a_new2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_new2.copy()
                a = a_new2

    return best_a


def lbfgs_polish(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(
            objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coordinate_descent(a, max_time, n_top=50):
    """Coordinate descent targeting top peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    factors = [0.3, 0.5, 0.7, 0.85, 1.1, 1.3, 1.5]

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        improved = False

        for k in peak_indices:
            if time.monotonic() - t_start > max_time:
                break
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Find top contributors to this peak
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                for factor in factors:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_scaled.copy()
                        a = a_scaled
                        improved = True
                        break
                if not improved:
                    a[j] = old_val
                else:
                    a = best_a.copy()

    return best_a


def fienup_refine(a, n_iters=300, thresh=1.3, clip=0.5, momentum=0.3):
    """Fienup-style spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        current_thresh = 2.0 + (thresh - 2.0) * t
        current_clip = 0.3 + (clip - 0.3) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * current_thresh
        if above.any():
            scale = np.ones_like(P)
            target_amp = np.sqrt(P_mean * current_thresh)
            current_amp = np.sqrt(P[above])
            factor = 1.0 - current_clip + current_clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new

        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def build_init_variant(n, rng, variant=0):
    """Build different initialization variants."""
    if variant == 0:  # Symmetric cosine
        half = n // 2
        x = np.linspace(0, 1, half + 1)
        a_half = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a_half = np.maximum(a_half, 0.01)
        if n % 2 == 0:
            a = np.concatenate([a_half[-1:][::-1], a_half[1:]])
        else:
            a = np.concatenate([a_half[-1:][::-1], a_half])
        a = a[:n]
    elif variant == 1:  # Raised cosine with noise
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
        a += 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    elif variant == 2:  # Double-bump
        x = np.arange(n, dtype=np.float64) / n
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2) + 0.05
    elif variant == 3:  # Triangular
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 0.01)
    else:  # Uniform with noise
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    return a / a.sum()


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses hybrid LSE + subgradient + L-BFGS approach.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n_target = len(parent) if parent is not None else 6962

    best_score = np.inf
    best_a = np.ones(n_target) * (1.0 / n_target)

    def update_best(a, label):
        nonlocal best_score, best_a
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Initialization (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 30.0)
    print(f"=== Phase 1: Initialization (t < {phase1_end - t_start:.0f}s) ===")

    # Start from parent if available
    if parent is not None:
        a_parent = parent.copy()
        a_parent = np.maximum(a_parent, 0.0)
        S = a_parent.sum()
        if S > 1e-12:
            a_parent = a_parent / S
        update_best(a_parent, "parent")

    # Try a few alternative initializations at target length
    for var in range(4):
        if time.monotonic() > phase1_end:
            break
        a_init = build_init_variant(n_target, rng, variant=var)
        # Quick Fienup to improve
        a_quick = fienup_refine(a_init, n_iters=50, thresh=1.5, clip=0.5, momentum=0.2)
        update_best(a_quick, f"init-var{var}")

    # Multi-scale warm start from best
    if time.monotonic() < phase1_end - 10:
        scales = [1024, 2048, 4096]
        current = best_a.copy()
        for n in scales:
            if n < n_target:
                a_small = upsample(current, n)
                a_small = nesterov_lse_full(a_small, max_time=5.0, beta_start=10, beta_end=5000)
                current = upsample(a_small, n_target)
                update_best(current, f"multiscale-{n}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Main LSE optimization (35% time)
    # ============================================================
    phase2_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LSE Optimization (t < {phase2_end - t_start:.0f}s) ===")

    # Multiple LSE runs with different parameters
    lse_configs = [
        {'beta_start': 5.0, 'beta_end': 30000.0},
        {'beta_start': 20.0, 'beta_end': 50000.0},
        {'beta_start': 10.0, 'beta_end': 20000.0},
    ]

    for i, cfg in enumerate(lse_configs):
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        alloc = min(time_left * 0.85, 600.0)
        print(f"  LSE run {i}: beta [{cfg['beta_start']}, {cfg['beta_end']}], alloc={alloc:.0f}s")
        a_nl = nesterov_lse_full(best_a, max_time=alloc,
                                  beta_start=cfg['beta_start'],
                                  beta_end=cfg['beta_end'])
        update_best(a_nl, f"lse-{i}")

    # Try from perturbed version
    if time.monotonic() < phase2_end - 30:
        a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_target))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        time_left = phase2_end - time.monotonic()
        a_nl2 = nesterov_lse_full(a_pert, max_time=time_left * 0.8,
                                   beta_start=10.0, beta_end=30000.0)
        update_best(a_nl2, "lse-pert")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient refinement (25% time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Subgradient Refinement (t < {phase3_end - t_start:.0f}s) ===")

    # Multiple subgradient runs with different peak counts
    for n_peaks in [1, 3, 5]:
        if time.monotonic() > phase3_end:
            break
        time_left = phase3_end - time.monotonic()
        if time_left < 20:
            break
        alloc = min(time_left * 0.7, 300.0)
        print(f"  Subgradient: top-{n_peaks} peaks, alloc={alloc:.0f}s")
        a_sub = subgradient_refine(best_a, max_time=alloc, n_top_peaks=n_peaks,
                                    alpha_start=0.03, alpha_end=0.001)
        update_best(a_sub, f"subgrad-{n_peaks}")

    # Fienup after subgradient
    if time.monotonic() < phase3_end - 20:
        a_fien = fienup_refine(best_a, n_iters=200, thresh=1.2, clip=0.6, momentum=0.3)
        update_best(a_fien, "fienup-phase3")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS polish (20% time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: L-BFGS Polish (t < {phase4_end - t_start:.0f}s) ===")

    lbfgs_betas = [1000, 3000, 8000, 15000, 30000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase4_end:
            break
        time_left = phase4_end - time.monotonic()
        if time_left < 30:
            break
        sc, a_opt = lbfgs_polish(best_a, beta=beta, maxiter=3000, maxfun=50000)
        update_best(a_opt, f"lbfgs-beta{beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final coordinate descent (10% time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Coordinate Descent (t < {phase5_end - t_start:.0f}s) ===")

    if time.monotonic() < phase5_end - 5:
        time_left = phase5_end - time.monotonic()
        a_cd = peak_coordinate_descent(best_a, max_time=time_left * 0.6, n_top=60)
        update_best(a_cd, "coord-desc")

        # Final LSE micro-step
        if time.monotonic() < phase5_end - 10:
            a_final = nesterov_lse_full(best_a, max_time=min(60.0, phase5_end - time.monotonic()),
                                         beta_start=10000, beta_end=50000)
            update_best(a_final, "final-lse")

    # One more L-BFGS with very high beta if time permits
    if time.monotonic() < deadline - 10:
        sc, a_opt = lbfgs_polish(best_a, beta=50000, maxiter=2000, maxfun=30000)
        update_best(a_opt, "final-lbfgs-50k")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
