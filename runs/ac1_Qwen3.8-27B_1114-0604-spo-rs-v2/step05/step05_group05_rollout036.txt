Optimized peak-reduction strategy for non-negative sequence design.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2.

Strategy:
1. Start from parent solution
2. Fienup spectral shaping (coarse peak reduction)
3. Nesterov-accelerated projected gradient on top-peak-weighted objective (fine)
4. L-BFGS polish on high-beta LSE
5. Coordinate descent on top peak contributors
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = c_max + float(np.log(exp_shifted.sum())) / beta
    
    # Gradient via convolution with p
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def topk_weighted_score_and_grad(a, K=3, beta=100.0):
    """
    Score and gradient focusing on top-K peaks.
    Uses LSE on the top K values for smoothness.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top K indices
    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        # Partial sort for efficiency
        top_idx = np.argpartition(c, -K)[-K:]
    
    # LSE on top K values
    c_top = c[top_idx]
    c_top_max = c_top.max()
    shifted = beta * (c_top - c_top_max)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    L = c_top_max + float(np.log(exp_vals.sum())) / beta
    
    # Gradient: dL/da[i] = sum_k w_k * 2*a[k-i]
    # This is a weighted sum of shifted versions of a
    grad_L = np.zeros(n)
    for k, wk in zip(top_idx, w):
        k = int(k)
        # d c[k]/da[i] = 2*a[k-i] for valid k-i
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            grad_L[j] += 2.0 * wk * a[k - j]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

Peak-targeted iterative optimization for non-negative sequence design.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2.

Strategy:
1. Start from parent or good init
2. Fienup spectral shaping for coarse peak reduction
3. Nesterov-accelerated projected gradient with top-K peak tracking (exact subgradient)
4. L-BFGS polish on very high beta LSE

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient via FFT."""

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = c_max + float(np.log(exp_shifted.sum())) / beta

    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def topk_grad(a, top_indices, weights):
    """Compute gradient of sum_k w_k * c[k] w.r.t. a. O(K*n)."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        k = int(k)
        if w < 1e-15:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_end < i_start:
            continue
        idx_start = k - i_end
        idx_end = k - i_start
        grad[i_start:i_end+1] += 2.0 * w * a[idx_start:idx_end+1][::-1]
    return grad


def topk_objective_and_grad(a, K=3, beta=50.0):
    """
    Top-K weighted objective and gradient.
    Focuses optimization on the K largest convolution peaks.
    """

def project_simplex(v, S_target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = S_target}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()

    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S_target))[0][-1]
    theta = (cssv[rho] - S_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    t_start = time.monotonic()
    lr = lr_init
    it = 0
    stall_count = 0
    best_stall_sc = best_sc

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = momentum_init + (momentum_end - momentum_init) * min(1.0, it / 2000.0)
        lr *= 0.9998

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = topk_objective_and_grad(a_look, K=K, beta=beta)

        v = mu * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new, 1.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall_count = 0
                best_stall_sc = sc
            else:
                stall_count += 1

        if stall_count > 300:
            stall_count = 0
            a = best_a * (1.0 + 0.01 * np.random.standard_normal(n))
            a = np.maximum(a, 0.0)
            S_a = float(a.sum())
            if S_a < 1e-12:
                a = best_a.copy()
            else:
                a = a / S_a
            v = np.zeros(n)
            lr = lr_init * 0.3

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def nesterov_lse_fast(a, beta=500.0, max_time=60.0, lr_init=0.008):
    """Nesterov on LSE objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    lr = lr_init
    mu = 0.9
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.9 + 0.00005 * it)
        lr *= 0.9998

        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = lse_score_and_grad(a_look, beta)
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def peak_coord_descent(a, n_sweeps=3, n_top_peaks=20, n_top_local=15, max_time=30.0):
    """Coordinate descent on top peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > max_time:
            break
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]

        for k in peak_indices:
            if time.monotonic() - t_start > max_time:
                break
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_end < i_start:
                continue

            contrib = np.array([a[k - j] for j in range(i_start, i_end + 1)])
            n_tl = min(n_top_local, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_tl]
            top_global = i_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for factor in [0.6, 0.75, 0.88, 1.12, 1.25]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_tmp = a / S_tmp
                    sc = compute_score(a_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_tmp.copy()
                        a = a_tmp.copy()
                        break
                else:
                    a[j] = old_val

    return best_a


def fw_peak_reduction(a, n_iters=500, max_time=30.0):
    """Frank-Wolfe step targeting the maximum peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > max_time:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Gradient of c[k_star] w.r.t. a
        i_start = max(0, k_star - (n - 1))
        i_end = min(k_star, n - 1)
        if i_end < i_start:
            break
        idx_start = k_star - i_end
        idx_end = k_star - i_start
        grad = np.zeros(n)
        grad[i_start:i_end+1] = 2.0 * a[idx_start:idx_end+1][::-1]

        # F-W: move toward vertex e_{argmin(grad)}
        i_min = int(np.argmin(grad))

        # Step size: 2/(it+2) for F-W, or use backtracking
        t_fw = 2.0 / (it + 2)

        a_new = (1.0 - t_fw) * a + t_fw * np.zeros(n)
        a_new[i_min] = t_fw

        # Project to ensure sum=1 (F-W step already preserves this if a is on simplex)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        a = a_new
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
