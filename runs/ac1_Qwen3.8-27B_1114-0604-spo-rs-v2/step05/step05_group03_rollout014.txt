from scipy.optimize import linprog

def solve_lp_exact(a, M):
    """Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using HiGHS"""
    n = len(a)
    # Build constraint matrix: (Ax)_k = sum_j a_j * x_{k-j} <= M
    # For k = 0, ..., 2n-2:
    # (Ax)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * x_{k-j}
    # = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} a_j * x_{k-j}
    
    # c = -1 (we minimize -sum(x))
    c = -np.ones(n)
    
    # A_ub: (2n-1) x n matrix
    # A_ub[k, j] = a_j if 0 <= k-j < n, else 0
    # i.e., A_ub[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    # Equivalently: A_ub[k, j] = a[k-j] if j <= k < j+n, else 0
    
    # This is a banded matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A_ub[k, j_start:j_end+1] = a[j_end:j_start-1:-1]  # a[k-j_end]...a[k-j_start]
    
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun, res.x
    return None, None

"""
Multi-track optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses improved LP-based alternating method with adaptive FISTA, frequency-domain error reduction,

import numpy as np
from scipy.optimize import minimize

def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def fista_lp(a, M, n_iters=20000, beta=100.0, x_init=None):
    """
    Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using FISTA.
    Returns (objective_value, x).
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.dot(a, a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    # Lipschitz constant for the smoothed objective
    # The gradient involves A^T p where A is the convolution matrix
    # ||A^T p|| <= ||A|| * ||p|| <= ||A||_2 * ||p||_2
    # ||A||_2 = max singular value of the convolution matrix
    # For a Toeplitz-like matrix, this is approximately max of |FFT(a)|
    A_fft = np.fft.rfft(a, 2 * n)
    L = float(np.max(np.abs(A_fft))) ** 2  # This is the max eigenvalue of A^T A
    # More precisely, the Lipschitz constant of the gradient of the LSE of (conv(a,x) - M)^+
    # is roughly beta * ||A||_2^2 = beta * max|FFT(a)|^2
    # But let's use a simpler bound: beta * n * max(a)^2 as a safe upper bound
    # Actually, ||A x||_inf <= ||a||_1 ||x||_inf, so ||(Ax - M)||_inf <= ||a||_1 ||x||_inf + M
    # The gradient of the LSE is A^T p where p sums to 1, so ||grad||_2 <= ||A^T||_2 = ||A||_2
    # So L = beta * ||A||_2^2
    L = beta * float(np.sum(a * a))  # conservative: ||A||_2^2 <= trace(A^T A) = n * ||a||_2^2... hmm
    
    # Actually, let's use: the gradient of f(x) = (1/beta) * log(sum(exp(beta*(c-M)))) 
    # where c = A x. The Hessian is (1/beta) * A^T * (diag(exp(beta*(c-M))/sum(...))) * A
    # So the Lipschitz constant of the gradient is (1/beta) * beta^2 * ||A^T D A|| / (sum exp(...)) 
    # which is at most beta * ||A||_2^2
    # ||A||_2 for the convolution matrix is max |FFT(a)|
    L = beta * float(np.max(np.abs(A_fft)) ** 2)
    if L < 1e-15:
        L = 1.0
    step = 1.0 / L
    
    for it in range(n_iters):
        # Compute c = A x = conv(a, x)
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        # Gradient: A^T p - 1
        # (A^T p)_j = sum_k A[k,j] p[k] = sum_k a[k-j] p[k] (where 0<=k-j<n)
        # = sliding window sum of p weighted by a
        # More precisely: (A^T p)_j = sum_{k=j}^{j+n-1} a[k-j] p[k]
        # Let's compute this efficiently
        # (A^T p)_j = sum_{i=0}^{n-1} a[i] p[i+j]
        # This is the cross-correlation of a and p
        
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]  # sliding window sum
        # But we need weighted sum, not just sum!
        # (A^T p)_j = sum_{i=0}^{n-1} a[i] p[i+j]
        # This is cross_conv(a, p) but only the first n elements
        
        # Actually cross_conv(a, p) gives:
        # (cross_conv(a,p))[k] = sum_j a[j] p[k-j]
        # We want (A^T p)_j = sum_i a[i] p[j+i] = sum_i a[i] p[j+i]
        # Let k = j+i, so i = k-j: sum_k a[k-j] p[k] (for valid k)
        # This is the "uncrossed" correlation, which is:
        # (A^T p)_j = cross_conv(a_rev, p)[j + n - 1] where a_rev = a[::-1]
        # Or equivalently: cross_conv(p, a)[j]... let me just use FFT directly
        
        # Simpler: use the FFT approach
        M2 = len(a) + len(p) - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        A_fft2 = np.fft.rfft(a[::-1], M2_fft)
        P_fft2 = np.fft.rfft(p, M2_fft)
        ATp = np.fft.irfft(A_fft2 * P_fft2, M2_fft)[:n]
        
        grad = ATp - 1.0
        
        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return float(x.sum()), x

def lse_score_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    log_sum_exp = z.max() - z_shift.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum_exp / beta
    score = 2.0 * n * L_val / (S * S)
    
    # Gradient of L = c_max + (1/beta)*log(sum(exp(beta*(c_k - c_max))))
    # dL/da_j = (1/beta) * sum_k exp(beta*(c_k-c_max))/sum(exp(...)) * dc_k/da_j
    # dc_k/da_j = a[k-j] (if 0<=k-j<n) [for the convolution c = a*a]
    # So dL/da = 2 * cross_corr(a, p) where p_k = exp(beta*(c_k-c_max))/sum(...)
    
    p = exp_z / exp_z.sum()
    # cross_corr(a, p) at the right offsets
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def error_reduction_v2(a, n_iters=100, target_frac=0.8):
    """
    Frequency-domain error reduction: reduce peaks in the convolution
    by modifying the amplitude spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        v = A * A  # This is the "power spectrum" (real and non-negative at DC, but complex in general)
        # Actually v = |A|^2 * exp(2i*phase(A))... no.
        # For real a, A is Hermitian: A[k] = conj(A[M_fft-k])
        # v = A * A (element-wise, NOT |A|^2)
        # The IFT of v gives the convolution a*a
        conv = np.fft.irfft(v, M_fft)[:2*n-1]
        
        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        
        if conv_max - conv_mean < 1e-15 * conv_max:
            break
        
        # Reduce the peaks
        # Target: make all values closer to the mean
        thresh = conv_mean + (1.0 - target_frac) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        
        if excess.max() < 1e-15:
            break
        
        # Modify the spectrum to reduce excess
        conv_mod = conv.copy()
        conv_mod = conv_mod - excess
        # Make sure conv_mod is a valid autocorrelation (its FFT must be non-negative real)
        # Actually, we just need the IFT of the modified conv to give a valid amplitude spectrum
        
        # Error in time domain
        err = conv_mod - conv  # = -excess
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2*n-1] = err
        
        err_spec = np.fft.rfft(err_pad)
        
        # Newton step: A_new = A + err_spec / (2*A)
        # But A can be zero at some frequencies
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = err_spec / (2.0 * A)
        # Zero out where A is too small
        dA[np.abs(A) < 1e-10 * np.max(np.abs(A))] = 0.0
        
        # Damp
        damp = 0.3 + 0.5 * min(1.0, it / 20.0)
        A_new = A + damp * dA
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_averaging(a, n_iters=500, alpha=0.1):
    """
    Subgradient method with Nesterov averaging for the exact (non-smooth) objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    x_averaged = a.copy()
    t = 1.0
    step_scale = 1.0  # will decay
    
    for it in range(n_iters):
        c = self_conv(a)
        S = float(a.sum())
        S2 = S * S
        S3 = S2 * S
        
        # Subgradient of max(conv(a,a)) w.r.t. a
        # If the max is achieved at multiple positions, use their average
        c_max = float(c.max())
        # Find all positions where c is close to max
        close_to_max = c >= c_max - 1e-10 * c_max
        # Subgradient direction: for each peak position k, the subgradient
        # of c[k] w.r.t. a is: (dc_k/da_j) = a[k-j] (for valid j)
        # So subgradient of max is the average of subgradients at peak positions
        
        # Compute subgradient at all "close to max" positions
        # This is expensive if there are many, so let's just use the top few
        top_k = np.argsort(c)[::-1][:min(10, n)]
        weights = np.maximum(c[top_k] - c_max + 0.1 * (c_max - float(c.min())) + 1e-15, 0.0)
        weights /= weights.sum()
        
        # Subgradient = weighted sum of subgradients at those positions
        subgrad = np.zeros(n)
        for idx, ki in enumerate(top_k):
            if weights[idx] < 1e-15:
                continue
            # dc_ki/da_j = a[ki-j] for j in [max(0, ki-n+1), min(ki, n-1)]
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                subgrad[j_start:j_end+1] += weights[idx] * a[j_end:j_start-1:-1]
        
        # Full gradient of f(a) = 2n * max_c / S^2
        # df/da = 2n * (subgrad_max / S^2 - 2 * max_c * 1 / S^3)
        # where subgrad_max is the subgradient of max_c w.r.t. a
        # and 1 is the gradient of S = sum(a) w.r.t. a
        grad = 2.0 * n * (subgrad / S2 - 2.0 * c_max / S3)
        
        # Projected subgradient step
        step = step_scale / np.sqrt(t)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Nesterov averaging
        x_averaged = x_averaged + (a_new - x_averaged) / (t + 1)
        t += 1
        step_scale *= 0.999  # slow decay
        
        sc = compute_score(x_averaged)
        if sc < best_sc:
            best_sc = sc
            best_a = x_averaged.copy()
        
        a = a_new
    
    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64)
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x / n), 0.01))
    # Gaussian bump
    center = n / 2
    inits.append(np.maximum(0.1 + 0.9 * np.exp(-0.5 * ((x - center) / (n/6))**2), 0.01))
    # Random smooth
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 20))
    inits.append(np.clip(0.5 + 0.4 * smooth, 0.01, 10.0))
    # Bimodal
    inits.append(np.maximum(0.1 + 0.9 * (np.cos(4.0 * np.pi * x / n) ** 2), 0.01))
    # Sawtooth (smoothed)
    saw = np.mod(x / n, 1.0)
    inits.append(np.clip(0.3 + 0.7 * saw, 0.01, 10.0))
    # Chirp-like
    inits.append(np.clip(0.3 + 0.5 * np.abs(np.sin(np.pi * x / n + 0.1 * (x/n)**2 * n)), 0.01, 10.0))
    
    return inits


def propose_candidate(budget_s=2400):
    """
    Multi-track search for best non-negative sequence minimizing the evaluation function.
    """
    rng = np.random.default_rng(123)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)", flush=True)
            return True
        return False
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={ps:.8f}", flush=True)
        report(ps, parent, "parent")
    
    # Determine lengths
    lengths = [6962, 8192, 10240, 12288]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths: {lengths}", flush=True)
    
    # ============================================
    # Phase 1: Quick exploration (10% time)
    # ============================================
    p1_end = t_start + 0.10 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 1: Quick Exploration (t < {p1_end - t_start:.0f}s) ===", flush=True)
    
    for n in lengths[:3]:
        if time.monotonic() > p1_end - 10:
            break
        inits = build_inits(n, rng, parent)
        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > p1_end - 5:
                break
            a0 = np.maximum(a0, 0.01)
            a_er = error_reduction_v2(a0, n_iters=50, target_frac=0.8)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")
    
    print(f"After Phase 1: best = {best_score:.8f}", flush=True)
    
    # ============================================
    # Phase 2: LP alternating (45% time) - MAIN
    # ============================================
    p2_end = t_start + 0.55 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 2: LP Alternating (main) ===", flush=True)
    
    # Work with the best sequence so far
    n_main = len(best_a)
    a_work = best_a.copy()

    configs = [
        (15000, 100, 25),   # iters, beta, max_outer
        (20000, 200, 30),
        (25000, 400, 35),
        (30000, 600, 40),
        (35000, 800, 50),
    ]
    
    for fi, (f_iters, beta, max_out) in enumerate(configs):
        if time.monotonic() > p2_end - 30:
            break
        remaining = p2_end - time.monotonic()
        # Estimate: each LP solve takes roughly f_iters * 4 * n / 1e9 seconds
        # (4 FFT operations per iteration, each O(n log n))
        est_time_per = f_iters * n_main * np.log2(n_main) * 4e-8
        est_outer = int(remaining / max(1.0, est_time_per))
        actual_outer = min(max_out, max(3, est_outer))
        
        print(f"  LP config {fi}: iters={f_iters}, beta={beta}, outer={actual_outer}, remaining={remaining:.0f}s", flush=True)

        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP{fi}")
        a_work = a_out.copy()
    
    print(f"After Phase 2: best = {best_score:.8f}", flush=True)
    
    # ============================================
    # Phase 3: L-BFGS polish (20% time)
    # ============================================
