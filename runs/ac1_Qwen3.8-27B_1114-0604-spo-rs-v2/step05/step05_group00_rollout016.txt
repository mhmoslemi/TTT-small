```python
    """Solve LP using active set method."""
    n = len(f)
    active = [0]  # Start with DC component (k=0)
    
    while True:
        # Build constraint matrix for active set
        # A is |active| x n, b = M * ones(|active|)
        # Objective: maximize sum(g) => minimize -sum(g)
        
        A = np.zeros((len(active), n))
        for row, k in enumerate(active):
            for j in range(max(0, k-n+1), min(k+1, n)):
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_lp_active(f, M, active_set, max_new=20):
    """Solve LP with given active set, return solution and new active set."""
    n = len(f)
    k_list = list(active_set)
    m = len(k_list)
    
    # Build constraint matrix A (m x n)
    A = np.zeros((m, n))
    for i, k in enumerate(k_list):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[i, j] = f[k - j]
    
    # LP: min -sum(g) s.t. A@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, active_set
    
    g = res.x
    
    # Check all constraints
    fg = np.convolve(f, g)
    # fg has length 2n-1
    violations = fg - M
    
    # Find violated constraints not in active set
    new_violated = []
    for k in range(len(fg)):
        if k not in active_set and violations[k] > 1e-8 * M:
            new_violated.append((k, violations[k]))
    
    # Add worst violators
    if new_violated:
        new_violated.sort(key=lambda x: -x[1])
        for k, _ in new_violated[:max_new]:
            active_set.add(k)
    
    return g, active_set
```

```python
LP-based iterative optimizer for minimum autocorrelation peak step function.
Uses active-set linear programming to find certified descent directions,
combined with line search, L-BFGS polishing, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def convolve_ab(f, g):
    """Linear convolution of two arrays via FFT."""
    n = len(f)
    m = len(g)
    L = 1 << (n + m - 2).bit_length()
    F = np.fft.rfft(f, L)
    G = np.fft.rfft(g, L)
    return np.fft.irfft(F * G, L)[:n + m - 1]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.infinity
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_constraint_rows(f, k_list):
    """Build the constraint matrix rows for given lags. Returns (m, n) matrix."""
    n = len(f)
    m = len(k_list)
    A = np.zeros((m, n))
    for i, k in enumerate(k_list):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            # A[i, lo:hi+1] = f[k-lo : k-hi] reversed
            # A[i, j] = f[k - j] for j in lo..hi
            cols = np.arange(lo, hi + 1)
            A[i, cols] = f[k - cols]
    return A

    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0.
    Uses active set method: solve, check violations, add constraints, repeat.
    Returns (g, active_set, converged).
    """

    g = None

        k_list = sorted(active_set)
        m = len(k_list)

        # Build constraint matrix
        A = build_constraint_rows(f, k_list)

        # Solve LP: min -sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': time_per_lp})
        except Exception:
            break

        if res.status not in [0, 3]:  # 0=optimal, 3=time limit
            break

        if res.x is None:
            break

        g = res.x
        if g is None or not np.all(np.isfinite(g)):
            break

        # Check all 2n-1 constraints
        fg = convolve_ab(f, g)
        violations = fg - M
        tol = 1e-8 * max(M, 1e-10)

        # Find violated constraints not in active set
        violated_mask = violations > tol
        violated_idx = np.where(violated_mask)[0]
        new_violated = [k for k in violated_idx if k not in active_set]

        if not new_violated:
            return g, active_set, True

        # Add worst violators to active set
        new_violated.sort(key=lambda k: -violations[k])
        n_add = min(20, max(1, (max_active - len(active_set)) // 2))
        for k in new_violated[:n_add]:
            active_set.add(k)

        if len(active_set) >= max_active:
            # Can't add more, return current solution (may violate constraints)
            return g, active_set, False

    return g, active_set, False


def lp_optimize(f, max_time=1500.0):
    """
    Main LP optimization loop.
    f should be normalized so sum(f) = sqrt(2n).
    Returns (best_f, best_M).
    """
    t_start = time.monotonic()
    n = len(f)
    target_sum = np.sqrt(2.0 * n)

    M = float(np.max(fast_conv(f)))
    active_set = {n - 1}
    best_f = f.copy()
    best_M = M
    stagnant = 0

    for it in range(2000):
        if time.monotonic() - t_start > max_time:
            break

        t0 = time.monotonic()
        g, active_set, converged = solve_lp_for_direction(
            f, M, active_set, max_active=600, max_inner=25,
            time_per_lp=min(10.0, max(1.0, max_time - (time.monotonic() - t_start) * 0.01))
        )
        t_lp = time.monotonic() - t0

        if g is None:
            active_set = {n - 1}
            stagnant += 1
            if stagnant > 5:
                break
            continue

        sum_g = float(g.sum())

        if sum_g <= target_sum * (1.0 + 1e-7):
            # No improvement from this LP
            stagnant += 1
            if stagnant > 3:
                # Perturb and restart
                rng = np.random.default_rng(it)
                f_pert = f * (1.0 + 0.01 * rng.standard_normal(n))
                f_pert = np.maximum(f_pert, 0.0)
                S = f_pert.sum()
                if S > 1e-12:
                    f_pert = f_pert * target_sum / S
                    M_pert = float(np.max(fast_conv(f_pert)))
                    if M_pert < M * 1.001:
                        f = f_pert
                        M = min(M, M_pert)
                active_set = {n - 1}
                stagnant = 0
                continue
            break

        stagnant = 0

        # Line search: find max t in [0,1] such that max(((1-t)f + t*g_norm)^2) <= M
        g_norm = g * target_sum / sum_g

        # Quick check t=1
        M_at_1 = float(np.max(fast_conv(g_norm)))
        if M_at_1 <= M * (1.0 + 1e-10):
            t_best = 1.0
        else:
            # Bisection
            t_lo, t_hi = 0.0, 1.0
            t_best = 0.0
            for _ in range(40):
                t_mid = (t_lo + t_hi) / 2.0
                f_test = (1.0 - t_mid) * f + t_mid * g_norm
                f_test = np.maximum(f_test, 0.0)
                S_test = f_test.sum()
                if S_test < 1e-12:
                    t_hi = t_mid
                    continue
                f_test = f_test * target_sum / S_test
                M_test = float(np.max(fast_conv(f_test)))
                if M_test <= M * (1.0 + 1e-9):
                    t_lo = t_mid
                    t_best = t_mid
                else:
                    t_hi = t_mid

        if t_best < 1e-12:
            active_set = {n - 1}
            stagnant += 1
            if stagnant > 10:
                break
            continue

        # Take the step
        f_new = (1.0 - t_best) * f + t_best * g_norm
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            break
        f_new = f_new * target_sum / S_new

        M_new = float(np.max(fast_conv(f_new)))

        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            if it % 20 == 0:
                elapsed = time.monotonic() - t_start
                print(f"  LP iter {it}: M={M_new:.8f}, t={t_best:.4f}, "
                      f"|active|={len(active_set)}, lp_time={t_lp:.2f}s, elapsed={elapsed:.0f}s")

        f = f_new
        M = M_new

        # Periodically reset active set to avoid getting stuck
        if it % 50 == 0 and it > 0:
            active_set = {n - 1, 0, 2 * n - 2}

    return best_f, best_M


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=120.0):
    """Run L-BFGS-B on LSE-smoothed objective with time monitoring."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S

    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()

    def time_check():
        return time.monotonic() - t0 < time_limit

    res = minimize(
        lse_objective_func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )

    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent_peaks(a, n_sweeps=5, n_top=50):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            for factor in [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5]:
                cv = a_old_j * factor
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < cur_sc - 1e-12:
                    cur_sc = sc
                    a_old_j = cv
                a[j] = a_old_j
            a[j] = a_old_j

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave_subgrad(a, n_iters=500, top_k=10):
    """Subgradient peak shaving with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.01

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * len(top_idx) / (len(top_idx) * S3))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.05 / (gn + 1e-10))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.99

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-based iterative search for minimum autocorrelation peak sequence.
    Phases: LP iterations -> L-BFGS polish -> coord descent -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        parent = np.ones(n0) * 0.5
        sc_p = compute_score(parent)
        print(f"No parent, using uniform n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "init")

    print(f"Starting best: {best_score:.8f}")
    n = len(best_a)

    # ==================================================================
    # Phase 1: LP-based iteration (60% of time)
    # ==================================================================
    phase1_time = 0.60 * (total_budget - 60.0)
    print(f"\n=== Phase 1: LP optimization ({phase1_time:.0f}s budget) ===")

    # Normalize to sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    f_lp = best_a.copy()
    S = f_lp.sum()
    f_lp = f_lp * target_sum / S
    M_init = float(np.max(fast_conv(f_lp)))
    print(f"  Initial M (normalized): {M_init:.8f}")

    t_lp_start = time.monotonic()
    f_lp, M_final = lp_optimize(f_lp, max_time=phase1_time)
    t_lp_elapsed = time.monotonic() - t_lp_start

    # Convert back to score
    sc_lp = M_final  # Since sum(f_lp) = sqrt(2n), score = M
    # But let's verify
    sc_lp_verified = compute_score(f_lp / target_sum)
    print(f"  LP result: M={M_final:.8f}, score={sc_lp_verified:.8f} ({t_lp_elapsed:.0f}s)")
    update(sc_lp_verified, f_lp / target_sum, "LP")

    # ==================================================================
    # Phase 2: L-BFGS polishing (20% of time)
    # ==================================================================
    phase2_time = 0.20 * (total_budget - 60.0)
    print(f"\n=== Phase 2: L-BFGS polishing ({phase2_time:.0f}s budget) ===")

    a_work = best_a.copy()
    betas = [200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if time.monotonic() > deadline - phase2_time * 0.3:
            break
        remaining = deadline - phase2_time * 0.3 - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 15))
        maxfun = min(50000, int(remaining * 150))
        tl = min(120.0, remaining * 0.8)
        print(f"  L-BFGS beta={beta} (maxiter={maxiter}, tl={tl:.0f}s)...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun, time_limit=tl)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        if update(sc, a_opt, f"LBFGS b={beta}"):
            a_work = a_opt.copy()

    # ==================================================================
    # Phase 3: Coordinate descent (10% of time)
    # ==================================================================
    phase3_time = 0.10 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Coord descent ({phase3_time:.0f}s budget) ===")

    if time.monotonic() < deadline - phase3_time * 0.5:
        a_cd = coord_descent_peaks(best_a, n_sweeps=10, n_top=80)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD")

    # Peak shaving
    if time.monotonic() < deadline - 30:
        a_ps = peak_shave_subgrad(best_a, n_iters=300, top_k=10)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")

    # ==================================================================
    # Phase 4: Final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Final polish ===")

    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        if remaining > 60:
            sc, a_final = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=30000,
                                     time_limit=remaining * 0.7)
            update(sc, a_final, "FINAL-LBFGS")

    if time.monotonic() < deadline - 20:
        a_cd2 = coord_descent_peaks(best_a, n_sweeps=5, n_top=100)
        sc = compute_score(a_cd2)
        update(sc, a_cd2, "FINAL-CD")

    # One more LP round if time permits
    if time.monotonic() < deadline - 60:
        remaining_lp = deadline - 60 - time.monotonic()
        if remaining_lp > 120:
            print(f"\n=== Bonus LP round ({remaining_lp:.0f}s) ===")
            f2 = best_a.copy()
            S2 = f2.sum()
            f2 = f2 * target_sum / S2
            f2, M2 = lp_optimize(f2, max_time=remaining_lp)
            sc2 = compute_score(f2 / target_sum)
            update(sc2, f2 / target_sum, "BONUS-LP")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
