- Adaptive top fraction: start small (10%), grow to large (50%).
- Partial compression: `P_new = min(P, thresh * (1 + alpha))` where alpha decreases over time.
- Over-relaxation with adaptive alpha.
- Hard clip to non-negative (simplest and most robust).

Actually, for the Fienup, let me use a slightly different approach that I've found works well in practice:

**Gershberg-Preaton (ERR) variant:**
1. Compute `A = FFT(a)`.
2. Set `P = |A|^2`.
3. Find threshold `T = quantile(P, 1 - top_frac)`.
4. For bins above threshold: `P_new = T` (or `P_new = T * P / (P - T + T) = T` for hard capping).
5. Actually, a softer version: `P_new = T + (P - T) * exp(-(P - T) / sigma)` for bins above T.
6. `A_new = sqrt(P_new) * A / |A|`.
7. `a_new = IFFT(A_new)`.
8. `a_new = max(a_new, 0)`.
9. Optionally: `a_new = a + alpha * (a_new - a)` (over-relaxation).
10. Normalize: `a_new = a_new / sum(a_new)`.

Let me also add a "kick" step: if the score stops improving for a while, perturb the sequence to escape local minima.

Now let me write the complete code:

```python
"""
Hybrid Fienup + L-BFGS optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses over-relaxed Fienup with adaptive spectral compression, followed by
L-BFGS with log-sum-exp smoothed objective, and subgradient refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_or(a, n_iters, gamma_init=0.7, gamma_end=0.3, top_frac_init=0.10, top_frac_end=0.50, or_param=1.5, normalize=True):
    """Over-relaxed Fienup with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    if normalize:
        a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    patience = 30

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Adaptive parameters
        gamma = gamma_init + (gamma_end - gamma_init) * progress
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * progress
        # Over-relaxation parameter anneals down
        or_p = or_param * (1.0 - 0.5 * progress)
        
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(ratio ** gamma)
        
        A_mod = A * scale
        a_raw = np.fft.irfft(A_mod, M_fft)[:n]
        a_proj = np.maximum(a_raw, 0.0)
        
        # Over-relaxation
        a_new = a + or_p * (a_proj - a)
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_proj
            S_new = float(a_new.sum())
        if normalize:
            a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Restart with perturbation
            a = best_a * (1.0 + 0.05 * np.random.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient
    p = exp_scaled / exp_sum
    # dL/dx_j = sum_k p[k] * dc[k]/dx[j] = sum_k p[k] * 2*x[k-j] (if valid)
    # = 2 * conv(p, x)[n-1 + j] ... let me think about this.
    # 
    # dc[k]/dx[j] = 2*x[k-j] if 0 <= k-j < n, else 0.
    # dL/dx[j] = sum_k p[k] * 2*x[k-j] = 2 * sum_k p[k] * x[k-j]
    # Let m = k-j, so k = j+m: sum_{m=0}^{n-1} p[j+m] * x[m]
    # But p has length 2n-1, so this is like a cross-correlation.
    #
    # Actually: sum_k p[k] * x[k-j] where k ranges over valid indices of c (0 to 2n-2)
    # and x[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1.
    # This is: (p * reverse(x))[n-1+j] ... hmm, let me think differently.
    #
    # sum_k p[k] * x[k-j] = cross-correlation of p and x evaluated at j.
    # = (IFFT(FFT(p) * conj(FFT(x))))[j] ... or we can use the convolution formula.
    #
    # Let's define: for each j, dL/dx[j] = 2 * sum_{k=j}^{j+n-1} p[k] * x[k-j]
    # (where we set x[i] = 0 for i outside [0, n-1])
    # = 2 * sum_{m=0}^{n-1} p[j+m] * x[m]
    #
    # This is like the "anti-diagonal" of the outer product p[:, None] * x[None, :].
    # Alternatively: it's the cross-correlation (p ⋆ x)[j] where ⋆ denotes cross-correlation.
    #
    # In terms of convolution: cross_corr(p, x)[j] = sum_m p[j+m] * x[m]
    # = sum_m p[j+m] * x[m] = (reverse(p) * x)[len(p)-1-j] ... 
    #
    # Actually, let's just use FFT:
    # cross_corr(p, x) = IFFT(FFT(p) * conj(FFT(x)))
    # But this gives sum_m p[m] * x[m-j] (with appropriate indexing).
    #
    # Let me just use the direct formula with FFT:
    # We want g[j] = sum_{m=0}^{n-1} p[j+m] * x[m] for j = 0, ..., n-1
    # (where p is extended to length 2n-1, and we only use p[j] for j up to 2n-2)
    #
    # This is equivalent to: take p (length 2n-1), and for each j, compute the dot product
    # of p[j:j+n] with x.
    #
    # Using FFT: this is the same as the "uncentered" cross-correlation.
    # g = cross-correlation(p, x) = IFFT(FFT(p, N) * conj(FFT(x, N)))[:n]
    # where N >= 2n-1 + n - 1 = 3n-2.
    
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    dp = np.fft.rfft(p, M2_fft)
    dx = np.fft.rfft(x, M2_fft)
    # cross-correlation: IFFT(dp * conj(dx))
    # This gives sum_m p[m] * x[m-j] for j = 0, 1, ...
    # Hmm, that's not quite what we want.
    #
    # What we want: g[j] = sum_m p[j+m] * x[m] = sum_m p[m'] * x[m'-j] where m' = j+m
    # = sum_{m'} p[m'] * x[m'-j]
    # So g[j] = (p ⋆ x)[j] where ⋆ is cross-correlation defined as (p⋆x)[j] = sum_m p[m] x[m-j]
    #
    # Using FFT: (p⋆x)[j] = IFFT(FFT(p) * conj(FFT(x)))[j]
    # (with zero-padding to avoid circular effects)
    
    cross = np.fft.irfft(dp * np.conj(dx), M2_fft)
    # cross[j] = sum_m p[m] * x[m-j] for valid j
    # But we want g[j] = sum_m p[j+m] * x[m] = sum_{m'} p[m'] * x[m'-j] = cross[j]
    # Wait, that's the same thing! So g = cross[:n]
    
    g_L = 2.0 * cross[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=100000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                 'maxcor': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_refine(a, n_iters=5000, step_init=0.02, decay=0.9999):
    """Subgradient method for the max-conv objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        if it % 500 == 0 and it > 0:
            # Check time optionally (caller handles this)
            pass
        
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        # Find all peaks (within tolerance of max)
        peak_threshold = M_val * 0.999
        peak_idx = np.where(c >= peak_threshold)[0]
        
        # Average subgradient over peaks
        sg = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                segment = a[k - j_end : k - j_start + 1][::-1]
                # dc[k]/da[j] for j in [j_start, j_end] is 2*a[k-j]
                # a[k-j] for j=j_start is a[k-j_start], for j=j_end is a[k-j_end]
                # So the vector is [a[k-j_start], a[k-j_start-1], ..., a[k-j_end]]
                # = a[k-j_start : k-j_end-1 : -1] ... let me just index directly
                for j in range(j_start, j_end + 1):
                    sg[j] += 2.0 * a[k - j]
        
        # Normalize by number of peaks
        sg /= max(1, len(peak_idx))
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Barzilai-Borwein step size
        step = min(step, 0.03)
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Sine
    inits.append(np.maximum(0.5 + 0.4 * np.sin(np.pi * x), 0.01))
    # Gaussian
    inits.append(np.exp(-2.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    # Random
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    # Two-scale random
    inits.append(0.5 + 0.3 * np.sin(3.0 * np.pi * x + rng.random() * 2 * np.pi) + 0.2 * rng.standard_normal(n))
    # Triangular
    inits.append(1.0 - np.abs(2.0 * x - 1.0))
    
    return inits
```

Wait, I need to reconsider the subgradient method. The loop that computes `sg[j] += 2.0 * a[k-j]` for each peak is O(n * num_peaks) which could be slow if there are many peaks. Let me optimize this using the convolution approach.

Actually, for the subgradient, if I use just the single maximum peak (or a few peaks), it should be fast enough. Let me limit to the top 5 peaks.

Also, I realize the `subgradient_refine` function with the direct loop over peaks will be slow for large n. Let me use the FFT-based approach instead.

```python
"""
Hybrid Over-Relaxed Fienup + L-BFGS (LSE) + FFT Subgradient optimizer.
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements: over-relaxed Fienup with adaptive spectral compression,
aggressive L-BFGS with increasing LSE temperature, FFT-accelerated subgradient,
and alternating multi-phase search strategy.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_or(a, n_iters=400, gamma_init=0.7, gamma_end=0.3,
              top_frac_init=0.10, top_frac_end=0.50, or_param=1.5,
              patience=40, rng=None):
    """Over-relaxed Fienup with adaptive parameters and restart on stagnation."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        gamma = gamma_init + (gamma_end - gamma_init) * progress
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * progress
        or_p = or_param * (1.0 - 0.4 * progress)

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 0.001) ** gamma)

        A_mod = A * scale
        a_raw = np.fft.irfft(A_mod, M_fft)[:n]
        a_proj = np.maximum(a_raw, 0.0)

        a_new = a + or_p * (a_proj - a)
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_proj
            S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            a = best_a * (1.0 + 0.04 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            Sa = a.sum()
            if Sa < 1e-12:
                a = best_a.copy()
            else:
                a = a / Sa
            no_improve = 0

    return best_a


def fienup_flat(a, n_iters=300, alpha_init=0.3, alpha_end=0.85):
    """Spectral flattening: relax power spectrum towards uniform."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy.minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    dp = np.fft.rfft(p, M2_fft)
    da = np.fft.rfft(x, M2_fft)
    cross = np.fft.irfft(dp * np.conj(da), M2_fft)
    g_L = 2.0 * cross[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-17, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                     'maxcor': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fft_subgradient(a, n_iters=8000, step_init=0.015, decay=0.99995, n_peaks=5):
    """FFT-accelerated subgradient method for the max-conv objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    M_fft = 1 << (2 * n - 1).bit_length()
    M2_fft = 1 << (3 * n - 2).bit_length()
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        n_p = min(n_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:n_p]

        p = np.zeros(2 * n - 1)
        for k in peak_idx:
            p[int(k)] = 1.0 / n_p

        dp = np.fft.rfft(p, M2_fft)
        da = np.fft.rfft(a, M2_fft)
        cross = np.fft.irfft(dp * np.conj(da), M2_fft)
        sg = 2.0 * cross[:n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step_candidate = max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10)
        step = min(step, step_candidate, 0.025)

        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, np.clip(parent, 0, 1000)).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.sin(np.pi * x), 0.01))
    inits.append(np.exp(-2.5 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    inits.append(0.5 + 0.3 * np.sin(4.0 * np.pi * x + rng.random() * 6.28) + 0.2 * np.abs(rng.standard_normal(n)))
    inits.append(0.3 + 0.7 * np.abs(1.0 - 2.0 * x))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid Fienup + L-BFGS + Subgradient search.
    Minimizes 2n*max(convolve(a,a))/(sum(a))^2 for non-negative a of length 6962.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    n = 6962

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            old = best_score
            best_score = sc
            a_clipped = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
            best_a = a_clipped
            print(f"  [{label}] NEW BEST: {sc:.8f} (was {old:.8f}, n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        p_use = parent[:n] if len(parent) >= n else np.pad(parent, (0, n - len(parent)))
        sc_parent = compute_score(p_use)
        print(f"Parent score: {sc_parent:.8f} (len={len(parent)})")
        update_best(sc_parent, p_use, "PARENT")

    print(f"Initial best: {best_score:.8f}, n={n}")

    # Phase 1: Fienup from multiple inits (25% time)
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (deadline={phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.clip(a0, 0.0, 1000.0).copy()
        S0 = a0.sum()
        if S0 < 1e-12:
            a0 = np.ones(n) * 0.5
        a0 = a0 / S0
        print(f"  Init {i}: score={compute_score(a0):.8f}")

        a_f = fienup_or(a0, n_iters=350, gamma_init=0.7, gamma_end=0.3,
                        top_frac_init=0.10, top_frac_end=0.55, or_param=1.5, rng=rng)
        sc = compute_score(a_f)
        update_best(sc, a_f, f"FIENUP i={i}")

        if time.monotonic() < phase1_end - 30:
            a_f2 = fienup_or(a_f, n_iters=250, gamma_init=0.5, gamma_end=0.2,
                             top_frac_init=0.15, top_frac_end=0.65, or_param=1.3, rng=rng)
            sc2 = compute_score(a_f2)
            update_best(sc2, a_f2, f"FIENUP2 i={i}")

        if time.monotonic() < phase1_end - 20:
            a_f3 = fienup_flat(a_f, n_iters=200, alpha_init=0.3, alpha_end=0.8)
            sc3 = compute_score(a_f3)
            update_best(sc3, a_f3, f"FLAT i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: L-BFGS with increasing beta (50% time)
    phase2_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS (deadline={phase2_end - t_start:.0f}s) ===")

    betas = [80, 150, 300, 600, 1200, 2500, 5000, 10000, 20000, 40000]
    a_cur = best_a.copy()

    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break

        maxiter = min(15000, max(500, int(remaining * 80)))
        maxfun = maxiter * 25
        print(f"  LBFGS beta={beta}, maxiter={maxiter}, rem={remaining:.0f}s", end="")

        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        print(f" -> {sc:.8f}")
        if improved:
            a_cur = a_opt.copy()

        # Quick Fienup between LBFGS calls to escape local minima
        if improved and time.monotonic() < phase2_end - 60:
            a_mix = fienup_or(a_opt, n_iters=100, gamma_init=0.6, gamma_end=0.3,
                              top_frac_init=0.1, top_frac_end=0.5, or_param=1.4, rng=rng)
            sc_mix = compute_score(a_mix)
            if update_best(sc_mix, a_mix, f"FIENUP-after b={beta}"):
                a_cur = a_mix.copy()

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Subgradient refinement (15% time)
    phase3_end = deadline - 60.0
    print(f"\n=== Phase 3: Subgradient (deadline={phase3_end - t_start:.0f}s) ===")

    for n_p in [3, 5, 10]:
        if time.monotonic() > phase3_end:
            break
        a_sg = fft_subgradient(best_a, n_iters=8000, step_init=0.012,
                               decay=0.99995, n_peaks=n_p)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG peaks={n_p}")

    # Final Fienup + LBFGS combo
    if time.monotonic() < phase3_end - 40:
        a_f = fienup_or(best_a, n_iters=200, gamma_init=0.5, gamma_end=0.2,
                        top_frac_init=0.1, top_frac_end=0.5, or_param=1.3, rng=rng)
        sc = compute_score(a_f)
        update_best(sc, a_f, "Final FIENUP")

        if time.monotonic() < phase3_end:
            sc2, a_opt = run_lbfgs(a_f, 8000, maxiter=5000, maxfun=100000)
            update_best(sc2, a_opt, "Final LBFGS b=8000")

    # Phase 4: Perturbation restarts (remaining time)
    print(f"\n=== Phase 4: Restarts ===")
    trial = 0
    while time.monotonic() < deadline - 10:
        noise_scale = 0.02 * (1.0 + 0.4 * trial)
        a_p = best_a * (1.0 + noise_scale * rng.standard_normal(n))
        a_p = np.maximum(a_p, 0.0)
        Sp = a_p.sum()
        if Sp < 1e-12:
            trial += 1
            continue
        a_p = a_p / Sp

        a_r = fienup_or(a_p, n_iters=120, gamma_init=0.6, gamma_end=0.3,
                        top_frac_init=0.1, top_frac_end=0.5, or_param=1.4, rng=rng)
        sc = compute_score(a_r)
        update_best(sc, a_r, f"R{trial}")

        if time.monotonic() < deadline - 15:
            sc2, a_opt = run_lbfgs(a_r, 3000, maxiter=2000, maxfun=40000)
            update_best(sc2, a_opt, f"R{trial}-LB")

        trial += 1

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = evaluate_sequence(out)
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
