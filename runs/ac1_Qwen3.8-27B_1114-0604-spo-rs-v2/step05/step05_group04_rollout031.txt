"""

2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_momentum(a, n_iters=500, gamma=0.8, top_frac_init=0.1, top_frac_end=0.5,
                    momentum=0.5, relax=0.7, soft_alpha=30.0):
    """Fienup with momentum and relaxation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    v = np.zeros(n)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        # Decay momentum
        mom = momentum * (1.0 - 0.5 * progress)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        # Soft non-negativity
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-np.clip(soft_alpha * a_new, -500, 500)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Momentum update
        v = mom * v + relax * (a_new - a)
        a_candidate = a + v
        a_candidate = np.maximum(a_candidate, 0.0)
        S_cand = float(a_candidate.sum())
        if S_cand < 1e-12:
            a = a_new
        else:
            a = a_candidate / S_cand

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        # Target: make spectrum as flat as possible

        sc = compute_score(a)

            best_a = a.copy()

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient
    p = exp_scaled / exp_sum  # weights

    # dL/da[i] = 2 * sum_k p[k] * a[k-i] (with a out of range = 0)
    # This is 2 * cross_correlation(p, a)[i]
    # Using FFT: cross_corr(p, a)[i] = sum_k p[k] * a[k-i]
    # = ifft(fft(p) * conj(fft(a)))[i] (with proper zero-padding)
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P_fft = np.fft.rfft(p, M2_fft)
    A_fft = np.fft.rfft(x, M2_fft)
    # cross correlation: ifft(fft(p) * conj(fft(a)))
    conv_res = np.fft.irfft(P_fft * np.conj(A_fft), M2_fft)
    # The cross-correlation c[i] = sum_k p[k] * a[k-i]
    # With zero-padding, this appears at index i in the result
    # But we need to be careful about the indexing
    # Actually: ifft(fft(p) * conj(fft(a)))[i] = sum_k p[k] * conj(a[k-i mod N])
    # For real signals and proper zero-padding, this gives the cross-correlation
    g_L = 2.0 * conv_res[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_grad_v2(x, beta):
    """Alternative LSE gradient using direct convolution approach."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Direct approach: g[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # Use the rev_a * p convolution approach from the original
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)
    # conv_res[k] = sum_j rev_a[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]
    # Let i = n-1-j: conv_res[k] = sum_i a[i] * p[k-n+1+i]
    # We want g[i] = sum_k p[k] * a[k-i]
    # Let s = k-i: g[i] = sum_s p[s+i] * a[s]
    # conv_res[n-1+i] = sum_s a[s] * p[n-1+i-n+1+s] = sum_s a[s] * p[i+s]
    # So g[i] = 2 * conv_res[n-1+i]
    g_L = 2.0 * conv_res[n-1:2*n-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000, use_v2=True):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    func = lse_objective_grad_v2 if use_v2 else lse_objective_grad
    res = minimize(
        func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_subgradient(a, n_iters=3000, top_m=5, step_init=0.05, decay=0.9999):
    """
    Subgradient method focused on the top peaks.
    Uses adaptive step size based on the gap to the lower bound.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peak positions
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Compute subgradient (average over top peaks for smoothing)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i <= n-1
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # a[k-i] for i = j_start..j_end means k-i = k-j_start..k-j_end
                # = (k-j_start) .. (k-j_end)
                seg = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * seg

        # Average gradient over top peaks
        sg /= len(top_idx)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.08)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coordinate_refinement(a, n_rounds=500, n_coords=20):
    """
    Coordinate descent: pick the coordinates that most influence the peak
    and do 1D optimization on them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_max = int(np.argmax(c))

        # Find coordinates most influential for the peak at k_max
        # dc[k_max]/da[i] = 2*a[k_max-i] (if valid)
        # So coordinates i where k_max-i is in [0,n-1] and a[k_max-i] is large
        valid_i = np.arange(max(0, k_max - n + 1), min(k_max, n - 1) + 1)
        if len(valid_i) == 0:
            break
        influence = np.abs(a[np.clip(k_max - valid_i, 0, n-1)])
        # Pick top n_coords
        n_pick = min(n_coords, len(valid_i))
        top_local = np.argsort(influence)[::-1][:n_pick]
        coords = valid_i[top_local]

        # For each selected coordinate, try to optimize its value
        for i in coords:
            i = int(i)
            # Try a few values
            best_val = a[i]
            best_local_sc = best_sc
            current = a[i]
            if current < 1e-15:
                # Try small positive values
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.1]
            else:
                candidates = [0.0, current * 0.5, current * 0.8, current, current * 1.2, current * 1.5, current * 2.0]
            
            for val in candidates:
                a[i] = val
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = val
            a[i] = best_val

        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))

    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(np.maximum(0.3 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(8.0 * np.pi * x), 0.01))
    return inits

"""
Optimized Fienup + High-beta L-BFGS + Peak-directed subgradient for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous: momentum Fienup, very high-beta LSE L-BFGS,
adaptive peak-directed subgradient, and strategic multi-start with warm restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_momentum(a, n_iters=500, gamma=0.8, top_frac_init=0.1,
                    top_frac_end=0.5, momentum=0.4, relax=0.8, soft_alpha=25.0):
    """Fienup with momentum and relaxation for faster convergence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1.0 - np.cos(np.pi * progress))
        mom = momentum * (1.0 - 0.5 * progress)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-np.clip(soft_alpha * a_new, -500, 500)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        v = mom * v + relax * (a_new - a)
        a_candidate = a + v
        a_candidate = np.maximum(a_candidate, 0.0)
        S_cand = float(a_candidate.sum())
        if S_cand < 1e-12:
            a = a_new.copy()
        else:
            a = a_candidate / S_cand

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat_target(a, n_iters=400):
    """Fienup with adaptive flat-spectrum target."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        target = P_mean * (1.0 + 0.5 * (1.0 - progress))

        scale = np.ones_like(P)
        above = P > target
        if above.any():
            ratio = target / P[above]
            scale[above] = 0.5 + 0.5 * np.sqrt(np.maximum(ratio, 1e-10))

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy.minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_subgradient(a, n_iters=3000, top_m=5, step_init=0.05, decay=0.99995):
    """Subgradient method focused on the top autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * seg

        sg /= len(top_idx)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.08)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a

    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LBFGS increasing beta ===")

    a_work = best_a.copy() / best_a.sum()
    betas = [500, 2000, 5000, 20000, 50000, 100000]

    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        est_time_per_iter = 0.0004 * (len(a_work) / 6962.0)
        maxiter = min(80000, int(remaining * 0.7 / max(est_time_per_iter, 0.0001)))
        maxfun = maxiter * 15
        print(f"  LBFGS beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"P2 b={beta}")
        if improved:
            a_work = a_opt.copy() / a_opt.sum()

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Peak-directed subgradient (17% time)
    # ==================================================================
    phase3_end = t_start + 0.77 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak-directed subgradient ===")

    for top_m in [3, 5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break
        n_iters_sg = int(remaining * 80)
        n_iters_sg = max(500, min(n_iters_sg, 8000))
        a_sg = peak_directed_subgradient(best_a, n_iters=n_iters_sg, top_m=top_m,
                                         step_init=0.04, decay=0.99997)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"P3-SG m={top_m}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Warm restarts + final L-BFGS polish (18% time)
    # ==================================================================
    print(f"\n=== Phase 4: Warm restarts + final polish ===")

    a_work = best_a.copy() / best_a.sum()

    # Perturbation + Fienup + L-BFGS cycle
    for trial in range(4):
        if time.monotonic() > deadline - 60:
            break
        noise_scale = 0.01 * (1 + trial * 0.5)
        a_p = a_work * (1.0 + noise_scale * rng.standard_normal(len(a_work)))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S_p = a_p.sum()
        if S_p < 1e-12:
            break
        a_p = a_p / S_p

        # Short Fienup
        a_f = fienup_momentum(a_p, n_iters=100, gamma=0.7, top_frac_init=0.05,
                             top_frac_end=0.35, momentum=0.3, relax=0.9, soft_alpha=20.0)
        sc = compute_score(a_f)
        update_best(sc, a_f, f"P4-RESTART-{trial}")

        # Short L-BFGS from perturbed
        if time.monotonic() < deadline - 40:
            a_base = best_a.copy() / best_a.sum()
            sc2, a_opt = run_lbfgs(a_base, 100000, maxiter=8000, maxfun=120000)
            update_best(sc2, a_opt, f"P4-LBFGS-{trial}")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 30:
        a_base = best_a.copy() / best_a.sum()
        sc, a_opt = run_lbfgs(a_base, 200000, maxiter=10000, maxfun=150000)
        update_best(sc, a_opt, "P4-FINAL-200K")

    if time.monotonic() < deadline - 15:
        a_base = best_a.copy() / best_a.sum()
        sc, a_opt = run_lbfgs(a_base, 500000, maxiter=5000, maxfun=75000)
        update_best(sc, a_opt, "P4-FINAL-500K")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
