```python
"""
Adaptive p-norm gradient descent + soft Fienup + coordinate descent for MPSL.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def pnorn_score_and_grad(a, p):
    """
    Compute the p-norm of the autocorrelation and its gradient.
    f_p = (sum_k b[k]^p)^(1/p)
    Returns (f_p, gradient).
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    # Use absolute values to handle potential tiny negative values from FFT
    c_abs = np.abs(c)
    c_p = c_p_sum = c_abs ** p
    S_p = float(np.sum(c_p))
    if S_p < 1e-30:
        return 0.0, np.zeros(n)
    
    f_p = S_p ** (1.0 / p)
    # C = (sum b^p)^(1/p - 1) = S_p^(1/p - 1)
    C = S_p ** (1.0 / p - 1.0)
    
    # w[k] = c[k]^(p-1)
    w = c_abs ** (p - 1.0)
    
    # corr[i] = sum_k w[k] * a[k-i] (cross-correlation)
    # = linear_conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1].copy()
    conv_len = len(w) + n - 1  # (2n-1) + n - 1 = 3n-2
    M_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_res = np.fft.irfft(W * AR, M_fft)[:conv_len]
    corr = conv_res[n-1 : 2*n-1]  # length n
    
    # df_p/da[i] = 2 * C * corr[i]
    # (factor of 2 because b[k] = sum_i a[i]*a[k-i], so db/da[i] = 2*a[k-i])
    grad = 2.0 * C * corr
    
    return f_p, grad


def pnorn_grad_descent(a, p_start=4.0, p_end=50.0, max_time=60.0,
                       lr_start=0.01, lr_min=1e-6):
    """
    Gradient descent on the p-norm with adaptive p.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S  # normalize
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t_start = time.monotonic()
    lr = lr_start
    p = p_start
    p_growth = 1.01
    lr_decay = 0.998
    momentum = 0.5
    v = np.zeros(n)
    
    it = 0
    while time.monotonic() - t_start < max_time:
        it += 1
        p = min(p * p_growth, p_end)
        lr *= lr_decay
        if lr < lr_min:
            break
        
        f, grad = pnorn_score_and_grad(a, p)
        
        # Normalize gradient by its norm for stable steps
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Adaptive step: limit step to not overshoot
        step = lr / (gnorm + 1e-10)
        
        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a = a_new = best_a.copy()
            break
        a_new = a_new / S_new
        
        # Check improvement
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Rejected: dampen momentum
            v *= 0.5
            a = a  # keep current
        
        # Periodically re-normalize momentum
        if it % 100 == 0:
            momentum = min(0.8, 0.5 + 0.002 * it)
    
    return best_a


def soft_fienup(a, n_iters=500, thresh_start=3.0, thresh_end=1.05,
                softness=2.0, max_time=60.0):
    """
    Fienup with soft (tanh) thresholding instead of hard thresholding.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t_start = time.monotonic()
    it = 0
    while it < n_iters and time.monotonic() - t_start < max_time:
        it += 1
        t = it / n_iters
        thresh = thresh_start + (thresh_end - thresh_start) * t
        # Softness decreases over time (harder thresholding)
        s = max(softness * (1 - t * 0.7), 0.5)
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        # Soft squelching: reduce bins above threshold using tanh
        excess = (P - P_mean * thresh) / (P_mean * thresh * s)
        scale = np.ones_like(P)
        above = P > P_mean * thresh
        if above.any():
            # tanh-based soft reduction
            reduction = np.tanh(excess[above])
            scale[above] = 1.0 - 0.8 * reduction  # reduce by up to 80%
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def peak_coordinate_descent(a, n_sweeps=10, n_top=20, max_time=30.0):
    """
    Coordinate descent targeting the top peak contributors.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t_start = time.monotonic()
    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > max_time:
            break
        
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            if time.monotonic() - t_start > max_time:
                break
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contribution of each a[j] to c[k] is a[k-j]
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_top_local = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top_local]
            top_global = j_start + top_local
            
            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                
                for factor in [0.6, 0.8, 1.2, 1.4]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_scaled.copy()
                        a = a_scaled
                        break
                else:
                    a[j] = old_val
    
    return best_a


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-300:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_sum
    
    # Gradient of L w.r.t. a[i]:
    # dL/da[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * cross_correlation(a, p)[i]
    a_rev = a[::-1].copy()
    conv_len = len(p) + n - 1
    M_fft = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_res = np.fft.irfft(P * AR, M_fft)[:conv_len]
    corr = conv_res[n-1 : 2*n-1]
    grad_L = 2.0 * corr
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    """Nesterov LSE with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    beta = beta_start
    beta_growth = 1.008
    lr = 0.02
    lr_decay = 0.9985
    mu = 0.9
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.9 + 0.00003 * it)
        lr *= lr_decay
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        effective_lr = min(lr, 0.05 / (gnorm + 1e-10))
        
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        beta = min(beta * beta_growth, beta_end)

                best_sc = sc
                best_a = a.copy()

    if sc < best_sc:

        best_sc = sc
    
    return best_a
```

    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_jac(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        obj_and_jac, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_lse_adaptive(a, max_time=60.0, beta_start=20.0, beta_end=10000.0):
    """Nesterov LSE with adaptive parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    it = 0
    beta = beta_start
    beta_growth = 1.008
    lr = 0.02
    lr_decay = 0.9985
    mu = 0.9
    
    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.9 + 0.00003 * it)
        lr *= lr_decay
        
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        effective_lr = min(lr, 0.05 / (gnorm + 1e-10))
        
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        beta = min(beta * beta_growth, beta_end)
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a
