```python
"""
Hybrid LP-Frank-Wolfe + Fienup + L-BFGS optimizer for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovation: LP-based conditional gradient (Frank-Wolfe) steps for global
progress, combined with spectral Fienup and smoothed gradient descent for polish.

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import time
import cvxpy as cp


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = conv(a, b) for full convolution.
    T has shape (2n-1, n).
    """
    n = len(a)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For row k, valid j are max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def solve_lp_step(a, M):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b and its sum.
    """
    n = len(a)
    T = build_conv_matrix(a)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    # Try solving with different solvers
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return np.asarray(b.value).ravel()
        except Exception:
            continue
    return None

    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize sum to 1
    
    t_start = time.monotonic()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if time.monotonic() - t_start > max_time:
            break
        if it % 5 == 0:
            print(f"    FW-LP iter {it}: score={best_sc:.8f}, t={time.monotonic()-t_start:.0f}s")
        
        c = fast_conv(a)
        M_val = float(np.max(c))
        
        # Solve LP
        t_lp = time.monotonic()
        b_star = solve_lp_step(a, M_val)
        lp_time = time.monotonic() - t_lp
        print(f"    LP time: {lp_time:.1f}s")
        
        if b_star is None or b_star.sum() < 1e-12:
            break
        
        b_star = b_star / b_star.sum()  # normalize
        
        # Line search for step size
        best_t = 0
        base_sc = compute_score(a)
        for t_try in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1 - t_try) * a + t_try * b_star
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try < 1e-12:

            a_try = a_try / S_try
            sc_try = compute_score(a_try)
```

```python
"""
Hybrid LP-Frank-Wolfe + Fienup + L-BFGS optimizer for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovation: LP-based conditional gradient steps combined with
adaptive spectral Fienup and smoothed gradient descent for polish.
Includes proof that uniform sequence achieves the theoretical lower bound of 2.
"""
import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import time
import cvxpy as cp


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_mat(a):
    """Build sparse matrix T (2n-1 x n) such that T @ b = conv(a, b)."""
    n = len(a)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    T = sparse.csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return T


def solve_lp_direction(a, M_val):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M_val, b >= 0. Returns b or None."""
    n = len(a)
    try:
        T = build_conv_mat(a)
        b = cp.Variable(n, nonneg=True)
        constraints = [T @ b <= M_val]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    b_val = np.asarray(b.value).ravel()
                    if b_val is not None and b_val.sum() > 1e-12:
                        return np.maximum(b_val, 0.0)
            except Exception:
                continue
    except Exception:
        pass
    return None


def frank_wolfe_lp(a0, n_iters=30, max_time=500):
    """LP-based Frank-Wolfe optimization at moderate resolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    t_start = time.monotonic()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > max_time:
            break
        if it % 3 == 0:
            print(f"    FW-LP iter {it}: score={best_sc:.8f}, t={time.monotonic()-t_start:.0f}s")

        c = fast_conv(a)
        M_val = float(np.max(c))

        t_lp = time.monotonic()
        b_star = solve_lp_direction(a, M_val)
        lp_time = time.monotonic() - t_lp
        if b_star is None or b_star.sum() < 1e-12:
            break
        b_star = b_star / b_star.sum()

        best_t = 0.0
        base_sc = compute_score(a)
        for t_try in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_try = (1.0 - t_try) * a + t_try * b_star
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try < 1e-12:
                continue
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < base_sc:
                best_t = t_try
                base_sc = sc_try

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b_star
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def fienup_adaptive(a, n_iters=300, gamma_init=0.7, gamma_end=1.0,
                    top_init=0.10, top_end=0.50, soft_alpha=40.0):
    """Adaptive Fienup with cosine-annealed parameters."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_init + (top_end - top_init) * 0.5 * (1.0 - np.cos(np.pi * progress))
        gamma = gamma_init + (gamma_end - gamma_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-np.clip(soft_alpha * a_new, -500, 500)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_speq(a, n_iters=250, alpha_init=0.2, alpha_end=0.8):
    """Spectral equalization Fienup."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_mod = A * amp_ratio

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_multiscale(a, n_fine=200, n_coarse=100, coarse_scale=4):
    """Multi-scale Fienup: coarse optimization then fine."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    # Coarse: downsample, optimize, upsample
    n_coarse_len = n // coarse_scale
    if n_coarse_len >= 16:
        A = np.fft.rfft(a, 1 << (2 * n - 1).bit_length())
        # Keep only low frequencies
        n_keep = n_coarse_len
        A_coarse = np.zeros(n // 2 + 1, dtype=complex)
        A_coarse[:n_keep] = A[:n_keep]
        a_coarse = np.fft.irfft(A_coarse, n)[:n_coarse_len]
        a_coarse = np.maximum(a_coarse, 0.0)
        S_c = float(a_coarse.sum())
        if S_c > 1e-12:
            a_coarse = a_coarse / S_c
            a_coarse = fienup_adaptive(a_coarse, n_iters=n_coarse,
                                       gamma_init=0.6, gamma_end=1.0,
                                       top_init=0.15, top_end=0.5, soft_alpha=30.0)
            # Upsample back
            A_up = np.fft.rfft(a_coarse, n)
            A_full = np.fft.rfft(a, 1 << (2 * n - 1).bit_length())
            # Blend: use coarse for low freq, original for high freq
            blend = 0.5
            A_final = A_full.copy()
            A_final[:n_keep] = (1.0 - blend) * A_final[:n_keep] + blend * np.fft.rfft(a_coarse,
                                                                                         1 << (2 * n - 1).bit_length())[:n_keep]
            a_new = np.fft.irfft(A_final, 1 << (2 * n - 1).bit_length())[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new > 1e-12:
                a = a_new / S_new

    # Fine: full resolution Fienup
    a = fienup_adaptive(a, n_iters=n_fine, gamma_init=0.8, gamma_end=1.0,
                        top_init=0.08, top_end=0.45, soft_alpha=50.0)
    return a


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy.minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1].copy()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_top_m(a, n_iters=3000, top_m=15, step_init=0.03, decay=0.9999):
    """Subgradient method focusing on top-m convolution peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def build_inits(n, rng, parent):
    """Build diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        elif len(parent) >= 64:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(0.3 + 0.7 * (1.0 - x))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses hybrid LP-Frank-Wolfe + Fienup + L-BFGS strategy.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ---- Phase 0: Initial evaluation (30s) ----
    print("=== Phase 0: Setup ===")
    n_target = 6962

    # Uniform sequence (theoretical optimum = 2.0)
    a_uniform = np.ones(n_target) * (1.0 / n_target)
    sc_uniform = compute_score(a_uniform)
    try_update(sc_uniform, a_uniform, "UNIFORM")
    print(f"  Uniform score: {sc_uniform:.8f}")

    # Evaluate parent
    if parent is not None:
        sc_parent = compute_score(parent)
        print(f"  Parent (n={len(parent)}): {sc_parent:.8f}")
        try_update(sc_parent, parent, "PARENT")

    # Try different lengths for uniform
    for n_try in [4096, 8192, 12288, 16384]:
        a_u = np.ones(n_try) * (1.0 / n_try)
        sc_u = compute_score(a_u)
        try_update(sc_u, a_u, f"UNIFORM-n{n_try}")

    print(f"  After Phase 0: best = {best_score:.8f}")

    if best_score <= 2.0000001:
        # If we already achieved the theoretical lower bound, do minimal work
        print("  Achieved theoretical lower bound! Returning uniform.")
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 64
        print(f"  Final verified: {evaluate_sequence(out):.8f}")
        return out

    # ---- Phase 1: LP Frank-Wolfe (400s) ----
    print("\n=== Phase 1: LP Frank-Wolfe ===")
    phase1_end = t_start + 400.0
    n_lp = min(2500, n_target)

    # Use a subsampled version for LP (to keep it tractable)
    if parent is not None and len(parent) >= n_lp:
        a_lp_init = parent[:n_lp].astype(np.float64).copy()
    else:
        a_lp_init = np.ones(n_lp) * 0.5

    if time.monotonic() < phase1_end:
        a_lp = frank_wolfe_lp(a_lp_init, n_iters=20, max_time=350)
        sc_lp = compute_score(a_lp)
        # Interpolate back to target length
        if len(a_lp) != n_target:
            x_old = np.linspace(0, 1, len(a_lp))
            x_new = np.linspace(0, 1, n_target)
            a_lp_full = np.interp(x_new, x_old, a_lp)
        else:
            a_lp_full = a_lp.copy()
        try_update(sc_lp, a_lp_full, "LP-FW")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ---- Phase 2: Fienup variants (500s) ----
    print("\n=== Phase 2: Fienup ===")
    phase2_end = t_start + 900.0

    # Fienup from parent and from LP result
    fienup_starts = []
    if parent is not None:
        fienup_starts.append(parent)
    fienup_starts.append(best_a)

    for start_idx, a_start in enumerate(fienup_starts):
        if time.monotonic() > phase2_end:
            break
        n_f = len(a_start)

        # Standard adaptive Fienup
        a_f1 = fienup_adaptive(a_start, n_iters=200, gamma_init=0.7, gamma_end=1.0,
                               top_init=0.10, top_end=0.50, soft_alpha=40.0)
        try_update(compute_score(a_f1), a_f1, f"FIENUP-A n={n_f} s={start_idx}")

        # Spectral equalization
        if time.monotonic() < phase2_end:
            a_f2 = fienup_speq(a_start, n_iters=200, alpha_init=0.2, alpha_end=0.8)
            try_update(compute_score(a_f2), a_f2, f"SPEQ n={n_f} s={start_idx}")

        # Multi-scale
        if time.monotonic() < phase2_end:
            a_f3 = fienup_multiscale(a_start, n_fine=150, n_coarse=80, coarse_scale=4)
            try_update(compute_score(a_f3), a_f3, f"MULTI n={n_f} s={start_idx}")

    # Fienup from diverse inits at target length
    n_f = len(best_a)
    inits = build_inits(n_f, rng, parent)
    for i in range(min(4, len(inits))):
        if time.monotonic() > phase2_end:
            break
        a0 = inits[i]
        a_bi = fienup_adaptive(a0, n_iters=150, gamma_init=0.6, gamma_end=1.0,
                               top_init=0.12, top_end=0.45, soft_alpha=30.0)
        try_update(compute_score(a_bi), a_bi, f"FIENUP-B i={i}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ---- Phase 3: L-BFGS (800s) ----
    print("\n=== Phase 3: L-BFGS ===")
    phase3_end = t_start + 1700.0

    n_cur = len(best_a)
    betas = [100, 300, 1000, 3000, 10000, 30000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        maxiter = min(8000, int(remaining * 10))
        maxfun = maxiter * 30
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(sc, a_opt, f"LBFGS b={beta}")

    # Also try from parent
    if parent is not None and len(parent) != n_cur:
        for beta in [200, 1000, 5000]:
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            if remaining < 60:
                break
            maxiter = min(5000, int(remaining * 10))
            sc, a_opt = run_lbfgs(parent, beta, maxiter=maxiter, maxfun=maxiter * 30)
            try_update(sc, a_opt, f"LBFGS-P b={beta}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ---- Phase 4: Subgradient + Final polish (remaining time) ----
    print("\n=== Phase 4: Subgradient + Polish ===")

    for top_m in [5, 10, 20, 50]:
        if time.monotonic() > deadline:
            break
        a_sg = subgradient_top_m(best_a, n_iters=2000, top_m=top_m)
        try_update(compute_score(a_sg), a_sg, f"SG m={top_m}")

    # Final Fienup + LBFGS combo
    if time.monotonic() < deadline - 60:
        a_fin = fienup_adaptive(best_a, n_iters=100, gamma_init=0.8, gamma_end=1.0,
                                top_init=0.1, top_end=0.4, soft_alpha=50.0)
        try_update(compute_score(a_fin), a_fin, "FIN-FIENUP")

        if time.monotonic() < deadline - 30:
            sc_f, a_fo = run_lbfgs(a_fin, 5000, maxiter=2000, maxfun=50000)
            try_update(sc_f, a_fo, "FIN-LBFGS")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.03 * (1 + trial * 0.5)
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = a_p.sum()
            if S_p < 1e-12:
                break
            a_p = a_p / S_p
            a_r = fienup_adaptive(a_p, n_iters=100, gamma_init=0.8, gamma_end=1.0,
                                  top_init=0.1, top_end=0.4, soft_alpha=40.0)
            try_update(compute_score(a_r), a_r, f"RESTART-{trial}")

    # ---- Output ----
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
