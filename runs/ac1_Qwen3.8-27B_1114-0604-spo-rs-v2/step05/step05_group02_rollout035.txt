import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum=1, then rescale
    v = v * (target_sum / (target_sum))  # identity, just for clarity
    # Standard algorithm: sort in decreasing order, find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gs_fienup(a, n_iters=400, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 1 - 1).bit_length()
    
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def build_inits(n, rng, parent=None):
    """Build diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Cosine window
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (x - 0.5) * 2), 0.01))
    
    # Gaussian window
    center = (n - 1) / 2.0
    sigma = n / 4.0
    inits.append(np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2))
    
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    
    # Multi-harmonic
    inits.append(np.maximum(0.5 + 0.2*np.cos(2*np.pi*x) + 0.15*np.cos(4*np.pi*x) + 0.1*np.cos(6*np.pi*x), 0.01))
    
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    
    # Sine window
    inits.append(np.maximum(np.sin(np.pi * x), 0.01))
    
    # Triangular
    inits.append(1.0 - 2.0 * np.abs(x - 0.5))
    
    # Parent interpolation
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.01, 1000.0).copy())
        else:
            inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent).copy())
    
    return inits

def lbfgs_refine(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 1.0/n
        S = 1.0
    a0 = a0 / S
    
    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        S_x = float(x.sum())
        if S_x < 1e-12:
            return 1e10, np.zeros(n)
        x_norm = x / S_x
        sc, grad = lse_objective_and_grad(x_norm, beta)
        return sc, grad
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'ftol': 1e-15, 'gtol': 1e-12,
                                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = float(a_out.sum())
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

"""
Multi-scale FISTA + adaptive LSE annealing + peak-directed local search
for minimizing 2n*max(conv(a,a))/sum(a)^2 with non-negative sequences.
Key innovations over previous GS-Fienup approach:
1. FISTA (Nesterov acceleration) on LSE-smoothed objective on the simplex
2. Adaptive beta annealing: smooth-to-sharp landscape traversal
3. Multi-scale initialization: optimize small n, upscale, refine
4. Peak-directed coordinate descent targeting top convolution lags
5. Random restarts with structure-preserving perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho_indices = np.where(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) * (target_sum / n)
    rho = rho_indices[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * (1/beta) * log(sum(exp(beta*c)))
    Grad = 4n * corr where corr[k] = sum_m a_m * p[m+k]
    p = softmax(beta * c)
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L

    # Gradient: 4n * (a_rev * p)[n-1 : 2n-1]
    M2 = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], M2)
    P = np.fft.rfft(p, M2)
    conv_result = np.fft.irfft(A_rev * P, M2)
    grad = 4.0 * n * conv_result[n - 1:2 * n - 1]

    return score, grad


def fista_lse(a0, beta, n_iters=3000, step_scale=0.8, verbose=False):
    """FISTA on LSE-smoothed objective with simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Conservative Lipschitz constant
    L = 16.0 * n / beta
    step = step_scale / L

    y = a.copy()
    a_prev = a.copy()
    t = 1.0

    for k in range(n_iters):
        sc, grad = lse_score_grad(y, beta)

        # Gradient step with projection
        x_new = project_simplex(y - step * grad, 1.0)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = float(y.sum())
        if y_sum < 1e-15:
            break
        y = y / y_sum

        a_prev = a
        a = x_new
        t = t_new

        # Track best by exact score (less frequent for speed)
        if k % 50 == 0 or k == n_iters - 1:
            exact = compute_score(a)
            if exact < best_sc:
                best_sc = exact
                best_a = a.copy()
            if verbose and k % 500 == 0:
                print(f"      FISTA b={beta} iter={k}: LSE={sc:.6f}, exact={exact:.6f}")

    # Final check
    exact = compute_score(a)
    if exact < best_sc:
        best_sc = exact
        best_a = a.copy()

    return best_a


def gs_fienup(a, n_iters=400, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with flat convolution target (magnitude+phase)."""
    n = len(a)
    M = 1 << (2 * n - 1 - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def build_inits(n, rng, parent=None):

    center = (n - 1) / 2.0
    sigma = n / 4.0
    inits.append(np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.maximum(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(np.sin(np.pi * x), 0.01))
    inits.append(1.0 - 2.0 * np.abs(x - 0.5))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * x + 0.3), 0.01))

        else:

def lbfgs_refine(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        S_x = float(x.sum())
        if S_x < 1e-12:
            return 1e10, np.zeros(n)
        x_norm = x / S_x
        sc, grad = lse_score_grad(x_norm, beta)
        return sc, grad

    bounds = [(0.0, 1000.0)] * n
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = float(a_out.sum())
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_descent(a, n_sweeps=3, n_top=30):
    """
    Coordinate descent targeting elements that contribute to peak convolution lags.
    For each peak lag k*, the contributing elements are a[max(0,k*-n+1):min(k*,n-1)+1].
    We try scaling those elements to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Get top peaks
        top_lags = np.argsort(c)[::-1][:n_top]

        for k_star in top_lags:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue

            seg_len = j_end - j_start + 1
            # Contribution of each element to this lag
            contrib = a[j_start:j_end + 1][::-1].copy()

            # Try reducing the top contributors
            n_try = min(10, seg_len)
            order = np.argsort(contrib)[::-1][:n_try]

            for idx in order:
                j = j_start + idx
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                cur_sc = compute_score(a)
                best_val = old_val
                for scale in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5]:
                    a[j] = old_val * scale
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc_tmp = compute_score(a_scaled)
                    if sc_tmp < cur_sc:
                        cur_sc = sc_tmp
                        best_val = old_val * scale
                    a[j] = old_val
                a[j] = best_val

        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_scale_init(n_target, rng, parent=None):
    """
    Multi-scale initialization: optimize small sequences, then upscale.
    """
    small_lengths = [64, 128, 256, 512]
    best_small = None
    best_small_sc = np.inf

    for n_small in small_lengths:
        inits = build_inits(n_small, rng, None)
        for a0 in inits[:4]:
            a_gs = gs_fienup(a0, n_iters=200, alpha_init=0.7, alpha_final=0.2)
            sc = compute_score(a_gs)
            if sc < best_small_sc:
                best_small_sc = sc
                best_small = a_gs

    if best_small is None:
        return np.ones(n_target) / n_target

    # Upscale to target length
    a_up = np.interp(np.linspace(0, 1, n_target),
                     np.linspace(0, 1, len(best_small)),
                     best_small)
    a_up = np.maximum(a_up, 0.01)
    return a_up


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: GS-Fienup -> FISTA annealed -> L-BFGS -> local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Target lengths
    lengths = [4096, 6144, 6962, 8192, 10240]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:5]

    print(f"Target lengths: {lengths}")
    print(f"Parent available: {parent is not None}, n={len(parent) if parent is not None else 0}")
    print(f"Budget: {budget_s}s, deadline at +{deadline - t_start:.0f}s")

    # ============================================================
    # Phase 1: GS-Fienup exploration (15% of time)
    # ============================================================
    phase1_end = t_start + 0.15 * (budget_s - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 1: GS-Fienup exploration (t < {phase1_end - t_start:.0f}s)")
    print(f"{'='*50}")

    for n in lengths[:3]:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}:")
        inits = build_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_gs = gs_fienup(a0, n_iters=300, alpha_init=0.7, alpha_final=0.1)
            sc = compute_score(a_gs)
            try_update(sc, a_gs, f"GS n={n} i={i}")

            a_gsm = gs_magnitude_only(a_gs, n_iters=200, alpha=0.4)
            sc_m = compute_score(a_gsm)
            try_update(sc_m, a_gsm, f"GS-mag n={n} i={i}")

    # Multi-scale init
    if time.monotonic() < phase1_end:
        print("  Multi-scale init:")
        for n in [6962, 8192]:
            if time.monotonic() > phase1_end:
                break
            a_ms = multi_scale_init(n, rng)
            a_gs = gs_fienup(a_ms, n_iters=300, alpha_init=0.6, alpha_final=0.15)
            sc = compute_score(a_gs)
            try_update(sc, a_gs, f"MS-GS n={n}")

    print(f"  Phase 1 best: {best_score:.8f}")

    if best_a is None:
        best_a = np.ones(6962) / 6962
        best_score = compute_score(best_a)
        print(f"  Fallback to uniform, score={best_score:.8f}")

    # ============================================================
    # Phase 2: FISTA with adaptive beta (35% of time)
    # ============================================================
    phase2_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 2: FISTA adaptive beta (t < {phase2_end - t_start:.0f}s)")
    print(f"{'='*50}")

    n_best = len(best_a)
    # Beta annealing schedule: start smooth, end sharp
    beta_schedule = [30, 60, 100, 200, 400, 800, 1500, 3000, 6000, 12000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        t_beta = time.monotonic()
        # Scale iterations based on remaining time
        remaining = phase2_end - time.monotonic()
        n_betmas = len([b for b in beta_schedule if b >= beta])
        n_iters = max(500, int(3000 * remaining / (n_betmas * (budget_s * 0.35 / len(beta_schedule))) + 500))
        n_iters = min(n_iters, 5000)

        a_fista = fista_lse(best_a, beta, n_iters=n_iters, step_scale=0.7)
        sc = compute_score(a_fista)
        try_update(sc, a_fista, f"FISTA b={beta}")
        print(f"    beta={beta:5d}: score={sc:.8f}, time={time.monotonic()-t_beta:.1f}s, iters={n_iters}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS refinement (25% of time)
    # ============================================================
    phase3_end = t_start + 0.75 * (budget_s - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS refinement (t < {phase3_end - t_start:.0f}s)")
    print(f"{'='*50}")

    lbfgs_betas = [500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase3_end:
            break
        t_beta = time.monotonic()
        remaining_frac = (phase3_end - time.monotonic()) / max(1.0, phase3_end - t_start)
        maxiter = max(2000, int(5000 * remaining_frac))
        maxfun = max(5000, int(100000 * remaining_frac))

        sc, a_opt = lbfgs_refine(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(sc, a_opt, f"LBFGS b={beta}")
        print(f"    beta={beta:5d}: score={sc:.8f}, time={time.monotonic()-t_beta:.1f}s")

    print(f"  Phase 3 best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak-directed local search + restarts (20% of time)
    # ============================================================
    phase4_end = deadline
    print(f"\n{'='*50}")
    print(f"Phase 4: Local search + restarts (t < {phase4_end - t_start:.0f}s)")
    print(f"{'='*50}")

    n_best = len(best_a)

    # Peak-directed coordinate descent
    if time.monotonic() < phase4_end - 60:
        a_pd = peak_directed_descent(best_a, n_sweeps=5, n_top=40)
        sc = compute_score(a_pd)
        try_update(sc, a_pd, "peak-descent")
        print(f"  Peak-directed: score={sc:.8f}")

    # FISTA refinement after peak descent
    if time.monotonic() < phase4_end - 60:
        a_f2 = fista_lse(best_a, 5000, n_iters=2000, step_scale=0.5)
        sc = compute_score(a_f2)
        try_update(sc, a_f2, "FISTA-polish b=5000")

    # Random restarts
    n_restarts = 0
    while time.monotonic() < phase4_end - 60:
        n_restarts += 1
        # Structure-preserving perturbation
        pert_type = n_restarts % 3
        if pert_type == 0:
            # Gaussian noise
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
        elif pert_type == 1:
            # Block swap
            block_size = max(1, n_best // 20)
            i = rng.integers(0, n_best - 2 * block_size)
            j = rng.integers(0, n_best - 2 * block_size)
            a_pert = best_a.copy()
            tmp = a_pert[i:i + block_size].copy()
            a_pert[i:i + block_size] = a_pert[j:j + block_size]
            a_pert[j:j + block_size] = tmp
        else:
            # Random spike
            a_pert = best_a.copy()
            k = rng.integers(0, n_best)
            a_pert[k] *= rng.uniform(0.1, 3.0)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = float(a_pert.sum())
        if S < 1e-6:
            continue
        a_pert = a_pert / S

        # Quick FISTA
        a_r = fista_lse(a_pert, 1000, n_iters=1000, step_scale=0.5)
        sc = compute_score(a_r)
        try_update(sc, a_r, f"restart {n_restarts}")
        print(f"  Restart {n_restarts} (type={pert_type}): score={sc:.8f}")

    # ============================================================
    # Phase 5: Final polish
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 5: Final polish")
    print(f"{'='*50}")

    if time.monotonic() < deadline - 10:
        a_final_gs = gs_fienup(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.05)
        sc = compute_score(a_final_gs)
        try_update(sc, a_final_gs, "FINAL-GS")

    if time.monotonic() < deadline - 5:
        sc, a_opt = lbfgs_refine(best_a, 20000, maxiter=3000, maxfun=50000)
        try_update(sc, a_opt, "FINAL-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
