```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Multi-phase optimizer with improved LP FISTA (backtracking), 
direct max-subgradient, and spectral reweighting for 
flat-autocorrelation non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
```

```python
"""

Improves upon previous approach with: backtracking FISTA for LP subproblem,

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)

        return 1e10, np.zeros(n)

    S3 = S2 * S

def max_subgrad(a):
    """Exact subgradient of the max-conv objective (normalized)."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 0.0, np.zeros(n)
    k_star = int(np.argmax(c))
    c_max = float(c[k_star])
    # Subgradient of max_k conv[k] w.r.t. a:
    # d(conv[k])/d(a[m]) = a[m-k]_{valid} + a[k-m]_{valid}
    # For the peak at k_star: direction[m] = a[m-k_star] + a[k_star-m] (with bounds)
    direction = np.zeros(n)
    for m in range(n):
        idx1 = m - k_star
        idx2 = k_star - m
        if 0 <= idx1 < n:
            direction[m] += a[idx1]
        if 0 <= idx2 < n:
            direction[m] += a[idx2]
        if k_star == 2 * m:
            direction[m] -= a[m]  # correct double counting on diagonal
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (direction / S2 - 2.0 * c_max / S3)
    return c_max, grad


def lse_lp_fasta(a, M, n_iters, beta, x_init=None, backtrack=True):
    """
    FISTA for: min (1/2)||x||^2 + (1/beta)*log(sum exp(beta*(conv(a,x) - M)))
    which approximates max sum(x) s.t. conv(a,x) <= M, x >= 0.
    Uses backtracking for step size.
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    # Initial step size estimate
    sum_a2 = float(np.sum(a * a))
    L = beta * n * sum_a2 + 1e-12
    step = 1.0 / L

    def compute_grad(x_curr):
        c = cross_conv(a, x_curr)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        cum = np.concatenate([[0.0], np.cumsum(p)])
        len_c = len(p)
        sw = cum[n:n + len_c - 1] - cum[:len_c - 1]
        eff_len = min(n, len_c)
        grad = a[:eff_len] * sw[:eff_len] - 1.0
        if len(grad) < n:
            grad = np.concatenate([grad, np.ones(n - len(grad))])
        return grad, p, c

    for it in range(n_iters):
        grad, p, c = compute_grad(x)

        if backtrack:
            # Backtracking line search
            step_try = step
            v_try = u - step_try * grad
            x_try = np.maximum(v_try, 0.0)
            # Check sufficient decrease (simplified: just halve if x_try is all zero)
            if x_try.sum() < 1e-15:
                step_try *= 0.5
                v_try = u - step_try * grad
                x_try = np.maximum(v_try, 0.0)
            step = step_try
        else:
            step = 1.0 / L

        v = u - step * grad
        x_new = np.maximum(v, 0.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        # Occasionally increase step (if we're making progress)
        if it % 50 == 0 and it > 0:
            step *= 1.05

    return x

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on LSE-smoothed objective."""

def max_subgradient_descent(a, n_iters=1000, time_limit=None):
    """Direct subgradient descent on the exact max objective (normalized)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:

    best_a = a.copy()
    best_sc = compute_score(a)

    # Polyaak-like step size: step_t = (f(a_t) - f_lb) / ||g_t||^2
    # f_lb = 1.0 (theoretical lower bound ~ 2n/(2n-1) ≈ 1)
    f_lb = 1.0
    step_scale = 1.0

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        _, grad = max_subgrad(a)
        gn2 = float(np.dot(grad, grad))
        if gn2 < 1e-30:
            break

        # Polyak step (with cap)
        current_sc = compute_score(a)
        step = min(0.05, step_scale * (current_sc - f_lb) / gn2)
        step = max(step, 1e-6 / (it + 1))

        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            step *= 0.1
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())

            best_a = a_try.copy()
            a = a_try
            step_scale = min(step_scale * 1.1, 2.0)
        else:
            step_scale *= 0.8
            step_scale = max(step_scale, 0.01)

def spectral_reweight(a, n_iters=50, alpha=0.5):
    """
    Spectral reweighting: modify the power spectrum to flatten autocorrelation.
    Works in frequency domain to reduce peak concentration.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    a_pad = np.zeros(M, dtype=np.complex128)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)

    for it in range(n_iters):
        v = A * np.conj(A).real  # |A|^2 = power spectrum
        conv = np.fft.ifft(np.abs(A)**2).real[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.sum()) / len(conv)
        if conv_max - conv_mean < 1e-15:
            break

        # Weight frequencies: damp frequencies that contribute to peak
        k_star = int(np.argmax(conv))
        # The peak at lag k* is influenced by all frequencies
        # Reduce power at frequencies where |A|^2 is large relative to average
        target = conv_mean * (1.0 + alpha)
        excess = np.maximum(np.abs(A)**2 - target, 0.0)
        scale = 1.0 - 0.3 * excess / (target + 1e-10)
        scale = np.maximum(scale, 0.1)

        # Apply to amplitude (preserve phase)
        phase = np.angle(A)
        mag = np.abs(A) * np.sqrt(scale)
        A_new = mag * np.exp(1j * phase)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            A = np.fft.fft(np.concatenate([best_a, np.zeros(M - n)]))
        else:
            # Revert and reduce alpha
            alpha *= 0.7

    return best_a


def peak_local_search(a, n_steps=200):
    """Targeted local search reducing the convolution peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.02, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.5)
        improved = False
        for mult in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try adding small random perturbation to escape
            rng = np.random.default_rng(it)
            a = a + 0.001 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x)**2)
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    # Smoother noise
    from scipy.ndimage import uniform_filter1d
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    # Bimodal
    inits.append(np.clip(0.3 + 0.7 * np.cos(2.0 * np.pi * x)**4, 0.01, 1000.0))
    return inits


def error_reduction(a, n_iters=80, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: quick LP, main LP+L-BFGS alternation, subgradient polish,
    spectral reweighting, and final local search.
    """
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine length - try parent length and a few alternatives
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 8192

    # Use power-of-2 length for FFT efficiency if close
    lengths_to_try = [n_main]
    # Add a slightly different length for diversity
    if n_main != 8192:
        lengths_to_try.append(8192)
    lengths_to_try = sorted(set(lengths_to_try))[:3]

    print(f"Main length: {n_main}, exploring: {lengths_to_try}")

    # ==================================================================
    # Phase 1: Quick LP alternating for improvement (15% time)
    # ==================================================================
    p1_deadline = t_start + 0.15 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Quick LP Alternating ---")

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break

        inits = build_initializations(n, rng, parent)
        max_inits = min(len(inits), 3)

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > p1_deadline:
                break
            S = float(a0.sum())
            if S > 1e-12:
                a0 = a0 / S
            # Quick LP: 2-3 outer iters with moderate FISTA
            a_out, sc_out, _ = lp_alternating_improved(
                a0, max_outer_iters=3, fista_iters=5000,
                time_limit=p1_deadline - 10, beta_start=20, beta_end=200
            )
            sc = compute_score(a_out)
            report(sc, a_out, f"Q-LP n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Main optimization - alternating LP and L-BFGS (50% time)
    # ==================================================================
    p2_deadline = t_start + 0.65 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Main LP + L-BFGS Alternation ---")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S
    n_work = len(a_work)

    x_warm = None
    # Progressive refinement: increasing FISTA iters and LP outer iters
    rounds = [
        {'fista': 8000, 'outer': 8, 'lbfgs_beta': 100, 'lbfgs_iter': 3000},
        {'fista': 12000, 'outer': 10, 'lbfgs_beta': 300, 'lbfgs_iter': 5000},
        {'fista': 18000, 'outer': 12, 'lbfgs_beta': 800, 'lbfgs_iter': 5000},
        {'fista': 25000, 'outer': 15, 'lbfgs_beta': 2000, 'lbfgs_iter': 8000},
        {'fista': 30000, 'outer': 15, 'lbfgs_beta': 5000, 'lbfgs_iter': 10000},
    ]

    for ri, r in enumerate(rounds):
        if time.monotonic() > p2_deadline - 30:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  Round {ri}: fista={r['fista']}, outer={r['outer']}, beta={r['lbfgs_beta']}")

        # LP phase
        a_lp, sc_lp, x_warm = lp_alternating_improved(
            a_work, max_outer_iters=r['outer'], fista_iters=r['fista'],
            time_limit=min(p2_deadline - 60, time.monotonic() + remaining * 0.5),
            beta_start=20.0 + ri * 20, beta_end=500.0 + ri * 300
        )
        sc = compute_score(a_lp)
        report(sc, a_lp, f"LP r{ri}")
        a_work = a_lp

        # L-BFGS polish
        if time.monotonic() < p2_deadline - 30:
            S = float(a_work.sum())
            if S > 1e-12:
                a_work = a_work / S
            sc, a_opt = run_lbfgs(a_work, r['lbfgs_beta'],
                                  maxiter=r['lbfgs_iter'],
                                  maxfun=r['lbfgs_iter'] * 4)
            report(sc, a_opt, f"LBFGS r{ri} b={r['lbfgs_beta']}")
            a_work = a_opt
            S = float(a_work.sum())
            if S > 1e-12:
                a_work = a_work / S

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Subgradient + spectral refinement (20% time)
    # ==================================================================
    p3_deadline = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: Subgradient + Spectral ---")

    a_sub = best_a.copy()
    S = float(a_sub.sum())
    if S > 1e-12:
        a_sub = a_sub / S

    # Max subgradient descent
    a_sub_out = max_subgradient_descent(a_sub, n_iters=500, time_limit=p3_deadline - 60)
    sc = compute_score(a_sub_out)
    report(sc, a_sub_out, "subgrad")

    # Spectral reweighting
    if time.monotonic() < p3_deadline - 30:
        a_spec = spectral_reweight(best_a, n_iters=100, alpha=0.5)
        sc = compute_score(a_spec)
        report(sc, a_spec, "spectral")

    # Another round of LP with the improved solution
    if time.monotonic() < p3_deadline - 15:
        a_work = best_a.copy()
        S = float(a_work.sum())
        if S > 1e-12:
            a_work = a_work / S
        a_lp, sc_lp, _ = lp_alternating_improved(
            a_work, max_outer_iters=10, fista_iters=20000,
            time_limit=p3_deadline - 10, beta_start=100, beta_end=1000
        )
        sc = compute_score(a_lp)
        report(sc, a_lp, "LP p3")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (remaining time)
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 60:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        for b in [5000, 10000]:
            if time.monotonic() > deadline - 40:
                break
            print(f"  Final L-BFGS beta={b}...", end=" ", flush=True)
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(a_norm, b, maxiter=10000, maxfun=40000)
            print(f"score={sc:.8f}, t={time.monotonic()-t0:.1f}s")
            report(sc, a_opt, f"final LBFGS b={b}")
            a_norm = a_opt
            S = float(a_norm.sum())
            if S > 1e-12:
                a_norm = a_norm / S

    # Peak local search
    if time.monotonic() < deadline - 30:
        a_peak = peak_local_search(best_a, n_steps=300)
        sc = compute_score(a_peak)
        report(sc, a_peak, "peak search")

    # Final subgradient
    if time.monotonic() < deadline - 15:
        a_sg = max_subgradient_descent(best_a, n_iters=200, time_limit=deadline - 10)
        sc = compute_score(a_sg)
        report(sc, a_sg, "final subgrad")

    # Try a different length one more time
    alt_n = 8192 if len(best_a) != 8192 else 10000
    if time.monotonic() < deadline - 60:
        print(f"  Trying alt length n={alt_n}...")
        a_alt = np.interp(np.linspace(0, 1, alt_n),
                          np.linspace(0, 1, len(best_a)), best_a)
        a_alt = np.maximum(a_alt, 0.0)
        S = float(a_alt.sum())
        if S > 1e-12:
            a_alt = a_alt / S
        a_out, sc_out, _ = lp_alternating_improved(
            a_alt, max_outer_iters=8, fista_iters=15000,
            time_limit=deadline - 15, beta_start=50, beta_end=500
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"alt n={alt_n}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
```
