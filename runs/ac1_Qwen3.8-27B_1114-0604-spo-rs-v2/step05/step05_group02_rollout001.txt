"""
Hybrid optimization for minimizing normalized max self-convolution of non-negative sequences.
Combines: GS-Fienup (spectral), Nesterov-accelerated subgradient, LP-based alternating,
and LSE+L-BFGS polishing. Uses adaptive time allocation and multi-start strategy.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_fienup(a, n_iters=600, alpha_init=0.7, alpha_final=0.05, M=None, momentum=None):
    """
    GS-Fienup with correct flat-convolution target and optional momentum.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Momentum state
    a_mom = a.copy()

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        alpha = max(alpha, 0.05)

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        # Soft update combining current and target
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Momentum
        if momentum is not None:
            a_next = momentum * a_new + (1 - momentum) * a_mom
            a_mom = a_new
        else:
            a_next = a_new

        a = a_next

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_subgradient(a, n_iters=5000, top_k_schedule=None, step_init=0.02,
                         momentum_init=0.9, momentum_end=0.999):
    """
    Nesterov-accelerated projected subgradient method.
    Project onto {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    f_star_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    # Nesterov momentum
    a_prev = a.copy()
    beta = 1.0
    step = step_init

    if top_k_schedule is None:
        top_k_schedule = lambda t: min(10 + int(50 * t / n_iters), 200)

    for it in range(n_iters):
        # Nesterov extrapolation
        beta_next = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        theta = (1.0 - beta) / beta_next
        a_y = a + theta * (a - a_prev)
        a_y = np.maximum(a_y, 0.0)
        S_y = float(a_y.sum())
        if S_y < 1e-12:
            break
        a_y = a_y / S_y

        # Compute subgradient
        c = fast_conv(a_y)
        top_k = top_k_schedule(it)
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            # Use top-k peaks weighted by proximity to max
            c_sorted = np.sort(c)[::-1]
            threshold = c_sorted[min(top_k, len(c_sorted)) - 1] * 0.99
            top_idx = np.where(c >= threshold)[0]
            if len(top_idx) > top_k:
                top_idx = np.argsort(c)[::-1][:top_k]

        M_val = float(c.max())
        S_y = float(a_y.sum())

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # dc_k/da_j = a_{k-j} + a_j (for valid indices in [0, n-1])
                rev_seg = a_y[k - j_end : k - j_start + 1][::-1]
                fwd_seg = a_y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += (rev_seg + fwd_seg)

        # Normalize subgradient by number of active peaks
        if len(top_idx) > 0:
            sg /= len(top_idx)

        # Score at y
        sc_y = 2.0 * n * M_val / (S_y * S_y)

        # Polyak step size
        gap = max(sc_y - f_star_lb, 1e-10)
        gn_sq = float(np.dot(sg, sg))
        step = min(gap / (gn_sq + 1e-30), step_init)
        step = max(step, 1e-10)

        # Gradient step
        a_new = a_y - step * sg
        # Project onto {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        a_prev = a
        a = a_new
        beta = beta_next

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_alternating(a, n_iters=50, alpha=0.5, time_limit=3600):
    """
    ADMM-style alternating: fix b=a, solve LP min t s.t. conv(a_new, b) <= t, sum=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    # Precompute sparse structure for the Hankel-like constraint matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows)
    cols = np.array(cols)
    nnz = len(rows)
    print(f"  LP alternating: building matrix with {nnz} non-zeros")

    # Update equation constraint
    A_eq_sparse = sparse.lil_matrix((1, n + 1))
    A_eq_sparse[0, :n] = 1.0
    A_eq = A_eq_sparse.tocsr()
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    b_ub = np.zeros(2 * n - 1)
    bounds = [(0, None)] * (n + 1)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        b = a.copy()
        vals = np.zeros(nnz)
        for idx in range(nnz):
            k = rows[idx]
            j = cols[idx]
            vals[idx] = b[k - j]

        A_sparse = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        # Add -identity column for t
        A_full = sparse.hstack([A_sparse, -sparse.eye(2 * n - 1)], format='csr')

        res = linprog(c_obj, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})

        if not res.success:
            break

        a_new = np.maximum(res.x[:n], 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Damped update
        a = (1 - alpha) * a + alpha * a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"    LP iter {it}: score={best_sc:.8f}")

    return best_a


def annealed_perturb(a, sc, n_perturbs=5, noise_scale=0.01):
    """Try random perturbations and keep improvements."""
    n = len(a)
    best_a = a.copy()
    best_sc = sc
    rng = np.random.default_rng(12345)

    for _ in range(n_perturbs):
        a_pert = a + noise_scale * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        sc_pert = compute_score(a_pert)
        if sc_pert < best_sc:
            best_sc = sc_pert
            best_a = a_pert.copy()

    return best_a, best_sc


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4 * np.pi * x) * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2 * np.pi * 2 * x), 0.01))
    d = np.abs(2.0 * x - 1.0)
    inits.append(np.maximum(0.3 + 0.6 * d, 0.01))
    inits.append(np.maximum(0.4 - 0.3 * d, 0.01))
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * 3 * x) * np.cos(2.0 * np.pi * x))
    return inits
