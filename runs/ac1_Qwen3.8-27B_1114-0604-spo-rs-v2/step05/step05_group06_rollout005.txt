"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-phase optimization: LSE smoothing -> exact subgradient -> mirror descent.
Key innovations: multiplicative weight updates for non-negative constraint,
adaptive peak tracking, and systematic length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=100, deadline=None):
    """Exact subgradient descent with extensive line search."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within tolerance of max
        tol = 1e-4 * max(dmax, 1.0)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        # Extensive line search
        for scale in (0.0001, 0.0002, 0.0005, 0.001, 0.002, 0.005,
                      0.01, 0.02, 0.05, 0.1, 0.2, 0.5):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _mirror_descent(a, n_iters=50, deadline=None):
    """Multiplicative weight (mirror descent) optimization for non-negative variables."""
    a = _norm(a).copy()
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    S = float(a.sum())

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set
        tol = 1e-4 * max(dmax, 1.0)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        # Center the gradient (remove mean for scale invariance)
        grad_centered = grad - grad.mean()

        # Adaptive step size: decrease over iterations
        eta_base = 0.1 / (1.0 + 0.1 * it)
        gnorm = float(np.linalg.norm(grad_centered))
        if gnorm < 1e-14:
            break
        eta = eta_base / gnorm

        # Multiplicative update
        log_update = -eta * grad_centered
        # Clip to prevent overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.clip(a_new, 0.0, None)

        Sn = a_new.sum()
        if Sn < 1e-8:
            break
        a_new = a_new / Sn

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # If it got worse, try smaller step
        if sc > best_sc * 1.001:
            for factor in (0.5, 0.25, 0.1):
                log_update_small = -eta * factor * grad_centered
                log_update_small = np.clip(log_update_small, -10, 10)
                a_try = a * np.exp(log_update_small)
                a_try = np.clip(a_try, 0.0, None)
                Stry = a_try.sum()
                if Stry < 1e-8:
                    continue
                a_try = a_try / Stry
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, try even smaller
                a = _norm(a * np.exp(-0.01 / gnorm * grad_centered))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-8:
                    a = a / a.sum()
                else:
                    break

    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _peak_equalize(a, n_iters=20, deadline=None):
    """Try to equalize the top peaks of the convolution."""
    a = _norm(a).copy()
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())

        # Get top K peaks
        K = min(50, 2 * n - 1)
        top_idx = np.argsort(d)[-K:]
        top_vals = d[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()

        if top_std < 1e-8:
            break

        # Weight peaks by how much they exceed the mean (prioritize reducing tall peaks)
        weights = np.maximum(top_vals - top_mean + top_std, 0)
        weights = weights / weights.sum()

        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = weights

        corr = _cross_corr(a, w_full)
        LDS = float(weights @ top_vals)
        grad = 4.0 * n * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Try multiple step sizes
        best_new_sc = best_sc
        best_new_a = a
        for scale in (0.001, 0.005, 0.01, 0.02, 0.05):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            if q.sum() < 1e-8:
                continue
            q = q / q.sum()
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = q

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
            a = best_new_a
        else:
            break

    return best_sc, best_a

Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-phase optimization combining LSE smoothing, exact subgradient with

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _lbfgs_wrapper(x, beta, k, n):
    v, g = _lse_obj_grad(x, beta, k)
    Sx = float(x.sum())
    PEN = 1e-3
    v += PEN * (Sx - 1.0) ** 2
    g = g + 2.0 * PEN * (Sx - 1.0)
    return v, g


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)

    try:
        res = minimize(
            _lbfgs_wrapper, a, args=(beta, k, n),
            method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out

            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        # Center gradient for scale invariance
        grad_c = grad - grad.mean()
        gnorm = float(np.linalg.norm(grad_c))
        if gnorm < 1e-14:
            break

        # Geometrically decreasing step size
        eta = 0.15 / ((1.0 + 0.05 * it) * gnorm)

        # Try multiple step sizes
        best_new_sc = best_sc
        best_new_a = a
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05):
            log_up = np.clip(-eta * factor * grad_c, -8.0, 8.0)
            a_try = a * np.exp(log_up)
            a_try = np.clip(a_try, 0.0, None)
            St = a_try.sum()
            if St < 1e-8:
                continue
            a_try = a_try / St
            sc = _score(a_try)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = a_try

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
            a = best_new_a
        else:
            break

    return best_sc, best_a


def _peak_equalize(a, n_iters, deadline):
    """Weight peaks proportionally to their excess over mean to flatten them."""
    a = _norm(a).copy()
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    M = 2 * n - 1

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())

        # Top K peaks weighted by excess
        K = min(100, M)
        top_idx = np.argpartition(d, M - K)[M - K:]
        top_vals = d[top_idx]
        top_mean = top_vals[top_vals > 0].mean() if (top_vals > 0).any() else dmax
        weights = np.maximum(top_vals - top_mean, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = weights / weights.sum()

        w_full = np.zeros(M)
        w_full[top_idx] = weights

        corr = _cross_corr(a, w_full)
        LDS = float(weights @ top_vals)
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_new_sc = best_sc
        best_new_a = a
        for scale in (0.001, 0.003, 0.01, 0.03, 0.1):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            if q.sum() < 1e-8:
                continue
            q = q / q.sum()
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = q

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
            a = best_new_a
        else:
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _jitter_perturb(a, rng, strength):
    """Add structured periodic jitter."""
    n = len(a)
    freq = rng.integers(2, max(3, n // 100))
    phase = rng.uniform(0, 2 * np.pi)
    jitter = strength * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    q = a + jitter
    return _norm(q)


def _multilevel_optimize(a, deadline):
    """Optimize at reduced resolution then upscale."""
    n = len(a)
    if n <= 2048:
        return _norm(a)

    # Work at n/2
    n_small = n // 2
    a_small = _upsample(a, n_small)
    phases = [(5, 50), (10, 200), (30, 800), (80, 3000)]
    n_phases = len(phases)

    for pi, (k_p, beta_p) in enumerate(phases):
        remaining = deadline - time.monotonic() - 20.0
        if remaining < 10:
            break
        tb = remaining / (n_phases - pi)
        tb = max(5.0, min(tb, 30.0))
        sc, a_new = _run_lbfgs(a_small, beta_p, k_p, tb, 300)
        a_small = _norm(a_new)

    # Upscale back
    a_up = _norm(_upsample(a_small, n))
    return a_up


def _optimize_one(a_init, deadline, label):
    """Multi-phase optimization of a single candidate."""
    a = _norm(a_init)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"    [{label}] start: score={best_sc:.8f}, n={n}")

    # Phase 0: Multilevel warm-start (if large)
    if n > 4096 and time.monotonic() < deadline - 120:
        a_ml = _multilevel_optimize(a, deadline - 60)
        sc_ml = _score(a_ml)
        if sc_ml < best_sc:
            best_sc = sc_ml
            best_a = a_ml.copy()
            a = a_ml
            print(f"    [{label}] multilevel: {best_sc:.8f}")

    # Phase 1: Progressive LSE
    lse_phases = [
        (5, 50, 0.08),
        (10, 150, 0.10),
        (20, 400, 0.12),
        (50, 1200, 0.14),
        (100, 3000, 0.16),
        (200, 8000, 0.18),
        (400, 20000, 0.22),
    ]
    total_frac = sum(f for _, _, f in lse_phases)
    lse_deadline = t0 + 0.65 * (deadline - t0)

    cum = 0.0
    for k_p, beta_p, frac in lse_phases:
        cum += frac / total_frac
        phase_end = t0 + cum * 0.65 * (deadline - t0)
        phase_end = min(phase_end, lse_deadline)
        remaining = phase_end - time.monotonic()
        if remaining < 5:
            break

        sc_p, a_p = _run_lbfgs(a, beta_p, k_p, remaining, 500)
        if sc_p < best_sc:
            best_sc = sc_p
            best_a = a_p.copy()
            print(f"    [{label}] LSE K={k_p},b={beta_p}: {best_sc:.8f}")
        a = _norm(a_p)

    # Phase 2: Subgradient polishing
    sg_deadline = min(deadline - 30, t0 + 0.85 * (deadline - t0))
    if time.monotonic() < sg_deadline:
        sc_sg, a_sg = _subgrad_polish(a, max_iters=80, deadline=sg_deadline)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")
        a = _norm(a) if a_sg is None else a_sg

    # Phase 3: Mirror descent
    md_deadline = min(deadline - 20, t0 + 0.93 * (deadline - t0))
    if time.monotonic() < md_deadline:
        sc_md, a_md = _mirror_descent(a, n_iters=40, deadline=md_deadline)
        if sc_md < best_sc:
            best_sc = sc_md
            best_a = a_md.copy()
            print(f"    [{label}] mirror: {best_sc:.8f}")
        a = _norm(a) if a_md is None else a_md

    # Phase 4: Peak equalization
    pe_deadline = min(deadline - 10, t0 + 0.97 * (deadline - t0))
    if time.monotonic() < pe_deadline:
        sc_pe, a_pe = _peak_equalize(a, n_iters=30, deadline=pe_deadline)
        if sc_pe < best_sc:
            best_sc = sc_pe
            best_a = a_pe.copy()
            print(f"    [{label}] peak_eq: {best_sc:.8f}")

    # Phase 5: Final subgradient
    if time.monotonic() < deadline - 5:
        sc_fin, a_fin = _subgrad_polish(best_a, max_iters=30, deadline=deadline - 5)
        if sc_fin < best_sc:
            best_sc = sc_fin
            best_a = a_fin.copy()
            print(f"    [{label}] final_sg: {best_sc:.8f}")

    print(f"    [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _sort_first(arr):
    return arr[0]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: multilevel LSE -> subgradient -> mirror descent -> peak equalization.
    Explores multiple lengths and diverse initializations with adaptive time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    # Determine lengths to explore
    if parent is not None:
        base_n = len(parent)
        lengths = [base_n]
        for dn in (0, 32, 64, 128, -32, -64, 256, 512):
            ln = base_n + dn
            if 2048 <= ln <= 16384 and ln not in lengths:
                lengths.append(ln)
        lengths.sort()
    else:
        lengths = [4096, 8192]

    print(f"Parent: {parent is not None} (n={len(parent) if parent is not None else 0})")
    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")

    best_score = np.inf
    best_seq = None

    # Generate candidates for each length
    all_candidates = []

    for L in lengths:
        inits_L = []

        if parent is not None:
            p_resampled = _upsample(parent, L)
            inits_L.append(('parent', p_resampled))
            inits_L.append(('parent_rev', _upsample(parent[::-1], L)))
            inits_L.append(('pn05', _smooth_perturb(p_resampled, rng, 0.05)))
            inits_L.append(('pn10', _smooth_perturb(p_resampled, rng, 0.10)))
            inits_L.append(('jn', _jitter_perturb(p_resampled, rng, 0.08)))

        # Spectral inits
        for i in range(3):
            inits_L.append((f'spec{i}', _spectral_init(L, rng)))

        # Cosine modulations
        x = np.linspace(0, 1, L)
        for f_c in (1, 2, 3, 5):
            a_c = 1.0 + 0.3 * np.cos(2.0 * np.pi * f_c * x)
            a_c = np.clip(a_c, 0, None)
            if a_c.sum() > 0:
                inits_L.append((f'cos{f_c}', a_c / a_c.sum()))

        # Evaluate
        for name_L, arr_L in inits_L:
            arr_L = _norm(arr_L)
            sc_L = _score(arr_L)
            if np.isfinite(sc_L):
                all_candidates.append((sc_L, name_L, L, arr_L))

    # Sort by score
    all_candidates.sort(key=_sort_first)
    print(f"\nTotal candidates: {len(all_candidates)}")
    print(f"Top 10: {[(f'{s:.6f}', n, L) for s, n, L, _ in all_candidates[:10]]}")

    # Select diverse candidates (different lengths preferred)
    selected = []
    used_lengths = set()
    for sc_c, name_c, L_c, arr_c in all_candidates:
        if L_c in used_lengths and len(selected) >= 3:
            continue
        if any(name_c == s[1] and L_c == s[2] for s in selected):
            continue
        selected.append((sc_c, name_c, L_c, arr_c))
        used_lengths.add(L_c)
        if len(selected) >= 6:
            break

    if not selected:
        L_def = lengths[0]
        selected = [(2.0, 'uniform', L_def, _norm(np.ones(L_def)))]

    # Time allocation: more time to top candidates
    n_sel = len(selected)
    # Give 50% to best, 25% to second, 15% to third, rest split
    if n_sel == 1:
        fracs = [1.0]
    elif n_sel == 2:
        fracs = [0.60, 0.40]
    elif n_sel == 3:
        fracs = [0.50, 0.30, 0.20]
    elif n_sel == 4:
        fracs = [0.45, 0.28, 0.17, 0.10]
    else:
        fracs = [0.40, 0.22, 0.14, 0.10, 0.08, 0.06]

    main_deadline = deadline - 250.0  # Reserve 250s for final polish
    print(f"\n=== Main optimization: {n_sel} candidates, main deadline in {main_deadline - t_start:.0f}s ===")

    cum_frac = 0.0
    for ci, (sc_c, name_c, L_c, arr_c) in enumerate(selected):
        if time.monotonic() > main_deadline:
            break
        sub_deadline = t_start + (cum_frac + fracs[ci]) * (main_deadline - t_start)
        sub_deadline = min(sub_deadline, main_deadline)
        cum_frac += fracs[ci]

        time_budget = sub_deadline - time.monotonic()
        if time_budget < 20:
            continue

        print(f"\n--- Candidate {ci+1}/{n_sel}: '{name_c}', n={L_c}, init_score={sc_c:.8f}, budget={time_budget:.0f}s ---")

        sc_opt, seq_opt = _optimize_one(arr_c, sub_deadline, name_c)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"  *** Global best updated: {best_score:.8f} ***")

    # === Final polish on the global best ===
    print(f"\n{'='*60}")
    print(f"=== FINAL POLISH: current best = {best_score:.8f} ===")
    print(f"{'='*60}")

    # Additional final optimization rounds
    polish_deadline = deadline - 15.0

    if best_seq is not None:
        a_best = _norm(best_seq)
        n_best = len(a_best)

        # Round 1: High-beta LSE
        if time.monotonic() < polish_deadline - 120:
            for k_f, beta_f in [(200, 10000), (500, 30000)]:
                remaining = polish_deadline - time.monotonic() - 120
                if remaining < 15:
                    break
                sc_f, a_f = _run_lbfgs(a_best, beta_f, k_f, min(remaining * 0.4, 60), 600)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_f.copy()
                    a_best = a_f
                    print(f"  Final LSE K={k_f},b={beta_f}: {best_score:.8f}")
                else:
                    a_best = _norm(a_f)

        # Round 2: Subgradient
        if time.monotonic() < polish_deadline - 60:
            sc_sg, a_sg = _subgrad_polish(a_best, max_iters=60, deadline=polish_deadline - 60)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                a_best = a_sg
                print(f"  Final subgrad: {best_score:.8f}")
            else:
                a_best = _norm(a_best)

        # Round 3: Mirror descent
        if time.monotonic() < polish_deadline - 30:
            sc_md, a_md = _mirror_descent(a_best, n_iters=40, deadline=polish_deadline - 30)
            if sc_md < best_score:
                best_score = sc_md
                best_seq = a_md.copy()
                a_best = a_md
                print(f"  Final mirror: {best_score:.8f}")
            else:
                a_best = _norm(a_best)

        # Round 4: Peak equalization
        if time.monotonic() < polish_deadline - 15:
            sc_pe, a_pe = _peak_equalize(a_best, n_iters=30, deadline=polish_deadline - 15)
            if sc_pe < best_score:
                best_score = sc_pe
                best_seq = a_pe.copy()
                a_best = a_pe
                print(f"  Final peak_eq: {best_score:.8f}")
            else:
                a_best = _norm(a_best)

        # Round 5: Final subgradient (most important)
        if time.monotonic() < polish_deadline - 5:
            sc_fin, a_fin = _subgrad_polish(a_best, max_iters=50, deadline=polish_deadline - 5)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  Ultimate subgrad: {best_score:.8f}")
            else:
                a_best = _norm(a_best)

        # Round 6: Tiny mirror descent steps
        if time.monotonic() < polish_deadline:
            sc_tiny, a_tiny = _mirror_descent(a_best, n_iters=20, deadline=polish_deadline)
            if sc_tiny < best_score:
                best_score = sc_tiny
                best_seq = a_tiny.copy()
                print(f"  Tiny mirror: {best_score:.8f}")

    # Prepare final output
    if best_seq is None:
        best_seq = _norm(np.ones(4096))

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    print(f"{'='*60}")
    return result
