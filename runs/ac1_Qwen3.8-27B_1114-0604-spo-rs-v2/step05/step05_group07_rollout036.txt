    B = b[1:].copy()
    B -= aj * C[1:]
    C_use = C[1:].copy()  # Only k >= 1 (linear terms)
    B0 = b[0] - aj * aj  # = sum_{i != j} a[i]^2

    # Golden section search
    lo = 0.0
    hi = max(aj * 4.0 + 0.02, 0.05)

    # Expand bracket if needed
    fh_hi = h_eval_1d(hi, B, C_use, B0, S, aj, n)
    fh_mid = h_eval_1d(hi * 0.5, B, C_use, B0, S, aj, n)
    while fh_hi < fh_mid and hi < 5.0:
        hi *= 2.0
        fh_mid = fh_hi
        fh_hi = h_eval_1d(hi, B, C_use, B0, S, aj, n)

    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - invphi * (hi - lo)
    d_pt = lo + invphi * (hi - lo)
    fc = h_eval_1d(c_pt, B, C_use, B0, S, aj, n)
    fd = h_eval_1d(d_pt, B, C_use, B0, S, aj, n)

    for _ in range(50):
        if fc <= fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - invphi * (hi - lo)
            fc = h_eval_1d(c_pt, B, C_use, B0, S, aj, n)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + invphi * (hi - lo)
            fd = h_eval_1d(d_pt, B, C_use, B0, S, aj, n)
        if hi - lo < 1e-12:
            break

    x_opt = (lo + hi) * 0.5
    return max(0.0, x_opt)


def run_coord_descent(a, n_sweeps, rng, verbose=False):
    """Exact 1D coordinate descent with random coordinate ordering."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        b = fast_conv(a)[:n]
        S = float(a.sum())
        order = rng.permutation(n)

        for idx in range(n):
            j = int(order[idx])
            x_opt = coord_descent_1d(a, j, b, S, n)
            if abs(x_opt - a[j]) > 1e-14:
                a[j] = x_opt
                b = fast_conv(a)[:n]
                S = float(a.sum())

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}/{n_sweeps}: score={sc:.10f}")

    return best_a, best_sc


def greedy_peak_shave(a, n_iters=500, rng=None):
    """Greedy peak shaving: reduce elements contributing to the peak,
    redistribute to elements contributing least."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        b = fast_conv(a)[:n]
        k_star = int(np.argmax(b))
        # Identify contributing elements for lag k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end + 1]
        # Contribution of each element to b[k_star]
        contrib = np.zeros(j_end - j_start + 1)
        if k_star == 0:
            contrib = seg ** 2
        else:
            # a[i] * a[i+k_star] for i in [j_start, j_end]
            if j_start + k_star < n:
                partner = a[j_start + k_star:j_end + 1 + k_star]
                contrib = seg * partner
            else:
                partner = a[j_start - k_star:j_end + 1 - k_star]
                contrib = seg * partner

        # Find top contributors and reduce them
        n_reduce = max(3, len(contrib) // 10)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        bottom_idx = np.argsort(contrib)[:n_reduce]

        total_reduction = 0.0
        for li in top_idx:
            gi = j_start + li
            reduction = a[gi] * 0.1
            a[gi] -= reduction
            total_reduction += reduction

        # Redistribute to bottom contributors
        n_dist = len(bottom_idx)
        if n_dist > 0 and total_reduction > 0:
            per = total_reduction / n_dist
            for li in bottom_idx:
                gi = j_start + li
                a[gi] += per

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Early stop if no improvement in 50 iters
        if it > 50 and sc >= best_sc * 1.001:
            if it > 100:
                break

    return best_a, best_sc


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    # Uniform
    inits.append(np.ones(n) / n)
    # Raised cosine
    inits.append(np.maximum(0.01, 0.5 * (1 + np.cos(2 * np.pi * x))))
    # Gaussian
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    # Triangular
    inits.append(np.maximum(0.01, 1.0 - 2.0 * np.abs(x - 0.5)))
    # Bimodal
    inits.append(np.exp(-30.0 * (x - 0.3) ** 2) + np.exp(-30.0 * (x - 0.7) ** 2))
    # Random smooth
    freq = 3.0 + 5.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    inits.append(np.maximum(0.01, 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + phase)))
    # Superflat-ish: slowly varying
    inits.append(np.maximum(0.01, 0.5 + 0.4 * np.cos(np.pi * x) ** 4))

    for ini in inits:
        S = ini.sum()
        if S > 1e-12:
            ini = ini / S

    return inits


def perturb(a, rng, scale=0.03):
    """Perturb a for iterated local search."""
    n = len(a)
    strategy = rng.integers(0, 4)
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        x = np.arange(n, dtype=np.float64) / n
        freq = 2.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 5 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for bidx in range(min(n_blocks, n // block_size)):
            s = bidx * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.maximum(a_p, 0.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Multi-phase: diverse init -> exact coordinate descent -> LSE L-BFGS ->
    peak shaving -> iterated local search with coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024, dtype=np.float64) / 1024.0

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None and len(parent) >= 1024:
        n_work = len(parent)
        print(f"Using parent length: {n_work}")
    else:
        n_work = 8192
        print(f"Using default length: {n_work}")

    # Also try a power-of-2 length for FFT efficiency
    n_p2 = 1 << (n_work - 1).bit_length()
    if n_p2 != n_work and n_p2 <= n_work * 1.25:
        pass  # Will try both

    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-start initialization + quick CD (20% time)")
    print(f"{'='*60}")

    phase1_end = t_start + 0.20 * (total_budget - 15.0)
    inits = build_initializations(n_work, rng, parent)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S
        sc = compute_score(a_init)
        report(sc, a_init, f"init_{i}")

        # Quick coordinate descent (2 sweeps)
        a_cd, sc_cd = run_coord_descent(a_init, n_sweeps=2, rng=rng)
        report(sc_cd, a_cd, f"cd_init_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # =========================================================================
    # Phase 2: Extended coordinate descent (30% time) - THE KEY NOVEL PHASE
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Exact Coordinate Descent (30% time)")
    print(f"{'='*60}")

    phase2_end = t_start + 0.50 * (total_budget - 15.0)

    # Run many sweeps of coordinate descent on the best so far
    a_cd, sc_cd = run_coord_descent(best_a, n_sweeps=200, rng=rng, verbose=True)
    report(sc_cd, a_cd, "CD_200sweeps")

    # Continue if time allows
    if time.monotonic() < phase2_end - 30:
        remaining = phase2_end - time.monotonic()
        est_sweep_time = (time.monotonic() - t_start) / 210.0  # rough
        extra_sweeps = int(remaining / max(est_sweep_time, 0.5))
        extra_sweeps = min(extra_sweeps, 300)
        if extra_sweeps > 10:
            a_cd2, sc_cd2 = run_coord_descent(a_cd, n_sweeps=extra_sweeps, rng=rng, verbose=True)
            report(sc_cd2, a_cd2, f"CD_{200+extra_sweeps}sweeps")
            if sc_cd2 < sc_cd:
                a_cd = a_cd2

    print(f"After Phase 2: best = {best_score:.10f}")

    # =========================================================================
    # Phase 3: LSE L-BFGS refinement (25% time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LSE L-BFGS Refinement (25% time)")
    print(f"{'='*60}")

    phase3_end = t_start + 0.75 * (total_budget - 15.0)
    beta_schedule = [100, 300, 1000, 3000, 10000, 30000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining_s = phase3_end - time.monotonic()
        if remaining_s < 20:
            break
        maxiter = int(min(30000, max(2000, remaining_s * 30)))
        maxfun = int(min(100000, max(5000, remaining_s * 100)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS_b={beta}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # =========================================================================
    # Phase 4: Greedy peak shaving + more CD (15% time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak Shaving + CD (15% time)")
    print(f"{'='*60}")

    phase4_end = t_start + 0.90 * (total_budget - 15.0)

    # Greedy peak shaving
    a_ps, sc_ps = greedy_peak_shave(best_a, n_iters=200, rng=rng)
    report(sc_ps, a_ps, "peak_shave")

    # More coordinate descent from the shaved result
    if time.monotonic() < phase4_end - 30:
        a_cd3, sc_cd3 = run_coord_descent(best_a, n_sweeps=100, rng=rng, verbose=True)
        report(sc_cd3, a_cd3, "CD_phase4")

    print(f"After Phase 4: best = {best_score:.10f}")

    # =========================================================================
    # Phase 5: Final refinement (10% time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Refinement (10% time)")
    print(f"{'='*60}")

    # High-beta L-BFGS
    for beta in [50000, 100000]:
        if time.monotonic() > deadline - 20:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=20000, maxfun=60000)
        report(sc, a_opt, f"final_b={beta}")

    # Final coordinate descent polish
    if time.monotonic() < deadline - 30:
        a_final, sc_final = run_coord_descent(best_a, n_sweeps=50, rng=rng)
        report(sc_final, a_final, "CD_final")

    # =================================================================================
    # Final verification and output
    # =================================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
