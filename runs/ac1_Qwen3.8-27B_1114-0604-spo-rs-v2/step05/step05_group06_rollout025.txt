    But is `T` symmetric? `T[i,m] = 2*w[i+m]` and `T[m,i] = 2*w[m+i] = 2*w[i+m]`. Yes! `T` is symmetric!
    
    Is it positive definite? Not necessarily (depends on `w`). So we can't use CG directly. But we can use GMRES or use a regularization.
    
    Actually, for the specific structure here, let me think about whether `T` is positive definite.
    
    `T = 2 * H` where `H` is the Hankel matrix with `H[i,m] = w[i+m]`.
    
    The quadratic form: `a^T T a = 2 * sum_{i,m} a[i] * w[i+m] * a[m] = 2 * sum_k w[k] * (sum_{i+m=k} a[i]*a[m]) = 2 * sum_k w[k] * c[k]`.
    
    Where `c = a*a` is the self-convolution! So `a^T T a = 2 * w^T c = 2 * sum_k w[k] * c[k]`.
    
    This is NOT necessarily positive. It depends on `w` and `c`. Since `c >= 0` (because `a >= 0`) and `w >= 0`, we have `a^T T a >= 0`. So `T` is positive semi-definite! (As long as `w >= 0`.)
    
    Great, so we can use conjugate gradient (or LSQR if not SPD) for the linear system.
    
    Actually, the full system is:
    ```
    [T    1  ] [a]   [0]
    [1^T  0  ] [λ] = [1]
    ```
    
    This is a saddle-point system. We can eliminate `λ`: from the first equation, `T a = λ 1`, so `λ = (1^T T a) / (1^T 1)` if `T` is invertible... no, that's not right. From `T a = λ 1`, we get `λ = (1^T T a) / (1^T 1)` only if `1` is in the range of `T`.
    
    Alternatively, from `T a = λ 1`, multiply by `1^T`: `1^T T a = λ * n`. And from `1^T a = 1`.
    
    We can use block elimination. The Schur complement approach:
    - From `T a + λ 1 = 0`: `a = -T^{-1} λ 1` (if `T` is invertible).
    - Substituting into `1^T a = 1`: `-λ * 1^T T^{-1} 1 = 1`, so `λ = -1 / (1^T T^{-1} 1)`.
    - Then `a = T^{-1} 1 / (1^T T^{-1} 1)`.
    
    So we just need to solve `T x = 1` and then `a = x / (1^T x)`.
    
    This is great! We just need to solve one linear system `T x = 1` where `T` is an `n x n` symmetric PSD matrix with known structure (Hankel).
    
    We can solve this using:
    - CG (if SPD) with matrix-free products using FFT.
    - Or, for moderate `n` (up to a few thousand), just build the matrix and use a direct solver.
    
    For `n = 8192`, building the full matrix would take 8192^2 * 8 bytes = ~512 MB, which is feasible but the solve would be O(n^3) ≈ 5.5 * 10^11 operations, which is too slow.
    
    So we need to use the structure. The matrix-vector product `T @ v = 2 * H @ v` where `H[i,m] = w[i+m]` can be computed in O(n log n) using FFT.
    
    Specifically, `(H @ v)[i] = sum_m w[i+m] * v[m]`. This is a cross-correlation of `w` and `v`, which can be computed via FFT.

```python
    # cross(u, v)[i] = sum_m u[m+i] * v[m]
    # Using FFT: cross(u,v) = ifft(fft(u) * conj(fft(v))) ... 
    # Actually, the standard formula: (f * g)[n] = ifft(FFT(f) * FFT(g))
    # And cross-correlation: (f ⊛ g)[n] = sum_k f[k+n] * g[k] = (f * g_flipped) ...
```

Let `u = w` (length `2n-1` or more, with `u[k] = w[k]` for valid `k` and 0 otherwise).

`(Hv)[i] = sum_{m=0}^{n-1} u[i+m] * v[m]`.

```python
def hankel_mv(w, v):
    n = len(v)
    # w has length >= 2n-1 (indices 0 to 2n-2)
    # (Hv)[i] = sum_{m=0}^{n-1} w[i+m] * v[m] for i = 0,...,n-1
    # Pad v to length 2n-1
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(w, N)  # w padded to N
    V = np.fft.rfft(v, N)  # v padded to N
    # The full convolution of w and v has length len(w) + len(v) - 1
    # But we want cross-correlation, not convolution.
    # Let's define: full_conv = ifft(FFT(w) * FFT(v))
    # full_conv[k] = sum_m w[m] * v[k-m]
    # We want: sum_m w[i+m] * v[m] = sum_j w[j] * v[i-(j-m)]... 
    
    # Alternative: use the fact that cross-correlation = convolution with conjugate/reverse
    # For real signals: cross_corr(u, v)[d] = (u * v_rev)[d + len(v) - 1]
    # where v_rev = v[::-1]
    
    V_rev = np.fft.rfft(v[::-1], N)
    full = np.fft.irfft(W * V_rev, N)
    # full[k] = sum_m w[m] * v_rev[k-m] = sum_m w[m] * v[len(v)-1-(k-m)]
    # = sum_m w[m] * v[n-1-k+m]
    # We want sum_m w[i+m] * v[m]. Let j = i+m, so m = j-i:
    # = sum_j w[j] * v[j-i]
    # From full: sum_j w[j] * v[n-1-(k-j)] = sum_j w[j] * v[n-1-k+j]
    # Set n-1-k+j = j-i => n-1-k = -i => k = n-1+i
    # So (Hv)[i] = full[n-1+i]
    
    return full[n-1:2*n-1]
```

```python
"""
Active-set linear system + LSE hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovation: uses the KKT condition to set up a linear system (Hankel matrix)
that can be solved efficiently with CG and FFT-based matrix products.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return out

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))

def _hankel_mv(w, v):
    """Compute H @ v where H[i,m] = w[i+m]."""
    n = len(v)
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(w, N)
    V_rev = np.fft.rfft(v[::-1], N)
    full = np.fft.irfft(W * V_rev, N)
    return full[n-1:2*n-1]

def _solve_active_set(w, n, tol=1e-10, max_iter=2000, reg=1e-10):
    """
    Solve T x = 1 where T[i,m] = 2*w[i+m] + reg*delta_{i,m}.
    Returns x, info.
    """
    r = np.ones(n)  # residual
    p = r.copy()  # initial search direction
    rp = r @ p
    x = np.zeros(n)
    
    for it in range(max_iter):
        Tp = 2.0 * _hankel_mv(w, p) + reg * p
        alpha = rp / (p @ Tp + 1e-30)
        x += alpha * p
        r -= alpha * Tp
        rnorm = np.linalg.norm(r)
        if rnorm < tol:
            break
        rp_new = r @ r
        if rp_new < 1e-30:
            break
        beta = rp_new / rp
        p = r + beta * p
        rp = rp_new
    
    return x, it + 1

def _active_set_step(a, K, reg=1e-10):
    """
    One active-set step: identify top-K positions, solve linear system, return new a.
    """
    n = len(a)
    S = a.sum()
    c = _self_conv(a)
    # Normalize c to get the "height" values
    d = c / (S * S)  # d[j] = (a*a)[j] / S^2
    
    # Find top K positions
    M = 2 * n - 1
    K = min(K, M)
    idx = np.argpartition(d, M - K)[M - K:]
    
    # Weights: inverse of the values (to equalize them)
    d_act = d[idx]
    d_min = d_act.min()
    if d_min < 1e-30:
        w_act = np.ones(K) / K
    else:
        # Use weights proportional to 1/d_act to push down the larger ones
        w_act = 1.0 / d_act
        w_act /= w_act.sum()
    
    # Build full w vector
    w = np.zeros(M)
    w[idx] = w_act
    
    # Solve T x = 1 where T[i,m] = 2*w[i+m]
    x, iters = _solve_active_set(w, n, reg=reg)
    
    # Normalize and clip
    x = np.clip(x, 0, None)
    Sx = x.sum()
    if Sx < 1e-12:
        return a, False
    a_new = x / Sx
    
    return a_new, True

def _active_set_polish(a, deadline, K_list=None, reg=1e-10):
    """
    Polish using active-set linear system with multiple K values.
    """
    if K_list is None:
        K_list = [5, 10, 20, 50, 100, 200, 500]
    
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    for K in K_list:
        if time.monotonic() > deadline:
            break
        K = min(K, 2 * n - 1)
        
        # Multiple iterations for each K
        a_curr = best_a.copy()
        for it in range(20):
            if time.monotonic() > deadline:
                break
            a_new, success = _active_set_step(a_curr, K, reg=reg)
            if not success:
                break
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            if sc < _score(a_curr) - 1e-12:
                a_curr = a_new
            else:
                # No improvement, try with current best
                a_curr = best_a.copy()
                # If no improvement in 2 consecutive iterations, break
                if it > 0 and sc >= _score(a_curr):
                    break
        
    return best_sc, best_a

def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective."""
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    val = 2.0 * L
    return float(val), grad

def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1"""
    n = a.size
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=400):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3
    
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out

def _subgrad_polish(a, max_iters=30, deadline=None):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        best_step = None
        best_new_sc = best_sc
        for scale in (0.001, 0.005, 0.02, 0.08, 0.2, 0.5, 1.0):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        if best_step is None:
            break
```
