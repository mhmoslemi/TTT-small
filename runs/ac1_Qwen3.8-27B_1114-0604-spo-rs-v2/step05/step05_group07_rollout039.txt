```python
"""
Optimizer for min_a  2*n*max(conv(a,a))/sum(a)^2.

This version uses:
  1. Short analytic parameter scans (truncated central-binomial / power-law shapes).
  2. Alternating fixed-weight quadratic smoothing:
       Q_p(a) = sum_k p_k * (a*a)_k,
     where p is re-estimated from the current coefficient distribution.
     L-BFGS minimizes Q_p/S(a)^2 with non-negativity bounds.
  3. Exact active-peak group mass-transfer polishing.
"""

import time
import numpy as np
from scipy.optimize import minimize

_GLOBAL_BEST_SCORE = np.inf
_GLOBAL_BEST_A = None
_GLOBAL_START = 0.0
_GLOBAL_DEAD = 0.0
_GLOBAL_RNG = None

_FIXED_N = 0
_FIXED_P = None
_FIXED_PFT = None
_FIXED_M = 0

_LAST_FIXED_MS = 0.0020


def normalize(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if a.size == 0:
        return np.ones(1, dtype=float)
    if s <= 1e-12:
        return np.ones(a.size, dtype=float) / float(a.size)
    return a / s


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=float)
    m = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]


def score_unit(a):
    """a must be normalized to sum ~= 1."""
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.inf
    c = fast_conv(a)
    if c.size == 0:
        return np.inf
    return 2.0 * float(n) * float(np.max(c))


def exact_score(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    a = a / s
    return score_unit(a)


def update_global_best(sc, a):
    global _GLOBAL_BEST_SCORE, _GLOBAL_BEST_A
    if not np.isfinite(sc):
        return False
    if sc >= _GLOBAL_BEST_SCORE:
        return False
    _GLOBAL_BEST_SCORE = sc
    _GLOBAL_BEST_A = np.asarray(a, dtype=float).ravel().copy()
    _GLOBAL_BEST_A = np.clip(_GLOBAL_BEST_A, 0.0, 1000.0)
    bad = ~np.isfinite(_GLOBAL_BEST_A)
    if np.any(bad):
        _GLOBAL_BEST_A[bad] = 0.0
    if float(_GLOBAL_BEST_A.sum()) < 0.01:
        _GLOBAL_BEST_A = np.ones(_GLOBAL_BEST_A.size, dtype=float)
    elapsed = time.monotonic() - _GLOBAL_START
    print(f"  NEW BEST {sc:.10f} n={len(_GLOBAL_BEST_A)} t={elapsed:.1f}s")
    return True


def interpolate_sequence(a, n):
    a = np.asarray(a, dtype=float).ravel()
    a = np.clip(a, 0.0, 1000.0)
    m = a.size
    if m == int(n):
        return normalize(a)
    if m == 1:
        return normalize(np.full(int(n), float(a[0]), dtype=float))
    x_old = np.linspace(0.0, 1.0, m, dtype=float)
    x_new = np.linspace(0.0, 1.0, int(n), dtype=float)
    out = np.interp(x_new, x_old, a)
    return normalize(out)


def central_power_seq(n, taper=0.0, power=0.0):
    """Truncated (1-x)^(-1/2) coefficient sequence, with mild parameterization."""
    n = int(n)
    b = np.ones(n, dtype=float)
    for i in range(n - 1):
        b[i + 1] = b[i] * (2 * i + 1) / (2 * i + 2)
    if taper != 0.0:
        x = np.arange(n, dtype=float) / max(1, n - 1)
        base = 1.0 - float(taper) * x
        base = np.maximum(base, 1e-8)
        b = b * base
    if power != 0.0:
        x = np.arange(1, n + 1, dtype=float)
        mult = np.power(x, -float(power))
        b = b * mult
    return normalize(b)


def make_weights_from_conv(c, gamma=1.0, floor_frac=0.5):
    c = np.asarray(c, dtype=float).ravel()
    L = c.size
    if L == 0:
        return np.ones(1, dtype=float)
    mx = float(np.max(c))
    if mx <= 1e-300:
        return np.ones(L, dtype=float) / float(L)
    s = c / mx
    floor = float(floor_frac)
    w = np.zeros(L, dtype=float)
    if floor <= 1e-12:
        magic = np.maximum(c, 0.0)
        if gamma == 1.0:
            w = magic
        else:
            w = np.power(magic, float(gamma))
    else:
        mask = s > floor
        if np.any(mask):
            w[mask] = np.power(s[mask] - floor, float(gamma))
        else:
            w[int(np.argmax(s))] = 1.0
    sw = float(w.sum())
    if sw <= 1e-300:
        k = max(1, min(L, 64))
        if k >= L:
            w = np.ones(L, dtype=float)
        else:
            idx = np.argpartition(-s, k - 1)[:k]
            w[idx] = 1.0
            sw = w.sum()
    return w / sw


def set_fixed_p(p, n):
    global _FIXED_P, _FIXED_PFT, _FIXED_M, _FIXED_N
    n = int(n)
    L = 2 * n - 1
    p = np.asarray(p, dtype=float).ravel()
    if p.size != L:
        tmp = np.zeros(L, dtype=float)
        m = min(L, p.size)
        tmp[:m] = p[:m]
        p = tmp
    s = float(p.sum())
    if s <= 1e-300:
        p = np.ones(L, dtype=float) / float(L)
    else:
        p = p / s
    _FIXED_N = n
    _FIXED_P = p
    _FIXED_M = next_pow2(3 * n - 2)
    _FIXED_PFT = np.fft.rfft(p, _FIXED_M)


def fixed_p_fun(x):
    a = np.asarray(x, dtype=float).ravel()
    n = a.size
    if n != _FIXED_N or _FIXED_M <= 0 or _FIXED_PFT is None:
        return 1e100, np.zeros(n, dtype=float)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return 1e100, np.zeros(n, dtype=float)

    ar = np.fft.rfft(a[::-1], _FIXED_M)
    tf = np.fft.irfft(ar * _FIXED_PFT, _FIXED_M)
    t = tf[n - 1:n - 1 + n]
    t = np.maximum(t, 0.0)

    q = float(np.dot(a, t))
    if q < 0.0:
        q = 0.0

    s2 = s * s
    s3 = s2 * s
    val = 2.0 * float(n) * q / s2
    dev = s - 1.0
    val += 1e-10 * dev * dev

    grad = (4.0 * float(n) * t) / s2 - (4.0 * float(n) * q) * a / s3
    grad += (2e-10 * dev) * np.ones(n, dtype=float)
    return float(val), grad


def run_fixed_lbfgs(a0, p, maxfun, maxiter):
    global _LAST_FIXED_MS
    a0 = normalize(a0)
    n = a0.size
    set_fixed_p(p, n)
    bounds = [(0.0, 1000.0)] * n
    options = {
        "ftol": 1e-20,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 20,
        "maxcor": 50,
    }
    try:
        t0 = time.monotonic()
        res = minimize(
            fixed_p_fun,
            a0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        elapsed = time.monotonic() - t0
        nfev = int(getattr(res, "nfev", maxfun))
        if nfev > 0:
            ms = elapsed / float(nfev)
            if ms > 0.0:
                _LAST_FIXED_MS = 0.7 * _LAST_FIXED_MS + 0.3 * ms
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()

    a_out = normalize(a_out)
    return exact_score(a_out), a_out


def perturbed_restart(a, rng, scale=0.03):
    a = np.asarray(a, dtype=float).ravel().copy()
    n = a.size
    x = np.arange(n, dtype=float) / max(1, n)
    strat = int(rng.integers(0, 4))
    if strat == 0:
        ap = a * (1.0 + 2.0 * float(scale) * rng.standard_normal(n))
    elif strat == 1:
        freq = 1.0 + 8.0 * float(rng.random())
        phase = 2.0 * np.pi * float(rng.random())
        amp = float(scale) * (1.0 + float(rng.random()))
        ap = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strat == 2:
        ap = a * np.exp(2.0 * float(scale) * rng.standard_normal(n))
    else:
        ns = max(1, n // 4)
        sel = rng.choice(n, size=ns, replace=False)
        ap = a.copy()
        ap[sel] *= (1.0 + 3.0 * float(scale) * (rng.random(ns) - 0.5))
    ap = np.clip(ap, 0.0, 1000.0)
    s = float(ap.sum())
    if s <= 1e-12:
        return a.copy()
    return ap / s


def group_polish(a, budget_s=30.0, max_sweeps=500):
    a = normalize(a)
    n = a.size
    best_sc = score_unit(a)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    fractions = (0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.60)

    for _ in range(int(max_sweeps)):
        if time.monotonic() - t0 > float(budget_s):
            break

        c = fast_conv(a)
        sc = score_unit(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            update_global_best(best_sc, best_a)

        L = c.size
        if L <= 1:
            break

        ktop = min(20, L)
        if ktop == L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(-c, ktop - 1)[:ktop]

        ct = c[idx]
        cmax = float(np.max(ct))
        w = np.exp(20.0 * (ct - cmax))
        ws = float(w.sum())
        if ws <= 1e-300:
            w = np.ones_like(ct)
            ws = float(w.sum())
        w = w / ws

        sens = np.zeros(n, dtype=float)
        for ii in range(ktop):
            wv = float(w[ii])
            k = int(idx[ii])
            j0 = max(0, k - n + 1)
            j1 = min(k, n - 1)
            if j1 >= j0:
                js = np.arange(j0, j1 + 1)
                sens[js] += wv * 2.0 * a[k - js]

        if n < 8:
            D = max(1, n // 2)
            R = max(1, n // 2)
        else:
            D = min(256, max(16, n // 32))
            R = min(256, max(16, n // 32))

        donors_all = np.argpartition(-sens, D - 1)[:D]
        donor_options = [donors_all]
        if D > 32 and n > 32:
            donor_options.append(np.argpartition(-sens, 31)[:32])

        for donors in donor_options:
            dmask = np.zeros(n, dtype=bool)
            dmask[donors] = True
            dS = float(a[donors].sum())
            if dS <= 1e-12:
                continue

            # low-sensitivity recipients
            rank = np.argsort(sens)
            recip_low = []
            for ii in range(n):
                jj = int(rank[ii])
                if not dmask[jj]:
                    recip_low.append(jj)
                    if len(recip_low) >= R:
                        break
            recip_low = np.array(recip_low, dtype=int)

            recip_sets = []
            if recip_low.size > 0:
                recip_sets.append(recip_low)
            recip_all = np.flatnonzero(~dmask)
            if recip_all.size > 0:
                recip_sets.append(recip_all)

            for recip in recip_sets:
                if recip is None or recip.size == 0:
                    continue
                rS = float(a[recip].sum())
                if rS <= 1e-12:
                    continue
                base_vec = a[recip] / rS
                for f in fractions:
                    new = a.copy()
                    new[donors] *= (1.0 - f)
                    new[recip] += f * dS * base_vec
                    new = np.clip(new, 0.0, 1000.0)
                    ss = float(new.sum())
                    if ss <= 1e-12:
                        continue
                    new /= ss
                    val = score_unit(new)
                    if val < best_sc - 1e-12:
                        best_sc = val
                        best_a = new.copy()
                        no_improve = 0
                        update_global_best(best_sc, best_a)
                        a = new.copy()
                        break

        if no_improve >= 35:
            break
        else:
            no_improve += 1

    if best_sc < _GLOBAL_BEST_SCORE:
        update_global_best(best_sc, best_a)
    return best_a


def optimize_sequence(a, budget_s, label):
    global _GLOBAL_BEST_SCORE, _GLOBAL_BEST_A, _LAST_FIXED_MS, _GLOBAL_RNG
    if _GLOBAL_RNG is None:
        _GLOBAL_RNG = np.random.default_rng(0)

    a = normalize(a)
    n = a.size
    end = time.monotonic() + max(30.0, float(budget_s))
    if end > _GLOBAL_DEAD:
        end = _GLOBAL_DEAD

    params = [
        (0.0, 1.0),
        (0.0, 2.0),
        (0.1, 1.0),
        (0.2, 1.0),
        (0.3, 2.0),
        (0.4, 1.0),
        (0.5, 2.0),
        (0.6, 1.0),
        (0.7, 2.0),
        (0.8, 2.0),
        (0.9, 3.0),
    ]

    p_prev = None
    it = 0
    no_improve = 0

    while True:
        rem = end - time.monotonic()
        if rem < 45.0:
            break

        c = fast_conv(a)
        sc = score_unit(a)
        update_global_best(sc, a)

        idx = it % len(params)
        if _GLOBAL_RNG.random() < 0.35:
            idx = int(_GLOBAL_RNG.integers(0, len(params)))
        floor_frac, gamma = params[idx]

        p = make_weights_from_conv(c, gamma=gamma, floor_frac=floor_frac)
        if p_prev is not None:
            p = 0.60 * p + 0.40 * p_prev
            ps = float(p.sum())
            if ps > 0.0:
                p = p / ps

        slice_t = min(60.0, max(12.0, rem * 0.35))
        ms = _LAST_FIXED_MS
        if ms <= 1e-6:
            ms = 0.002
        maxfun = int(min(50000, max(2000, slice_t / (ms * 0.8))))
        maxiter = int(min(100000, max(1000, maxfun)))

        sc2, a2 = run_fixed_lbfgs(a, p, maxfun=maxfun, maxiter=maxiter)

        rem2 = end - time.monotonic()
        if rem2 > 70.0:
            gp_budget = min(20.0, rem2 * 0.25)
            if gp_budget >= 10.0:
                a2 = group_polish(a2, budget_s=gp_budget, max_sweeps=400)

        sc2 = exact_score(a2)
        update_global_best(sc2, a2)

        if sc2 < sc - 1e-12:
            a = normalize(a2)
            no_improve = 0
        else:
            blend = 0.30 * a + 0.70 * a2
            sb = exact_score(blend)
            if sb < sc - 1e-14:
                a = normalize(blend)
                no_improve = 0
            else:
                if no_improve > 6:
                    a = perturbed_restart(a, _GLOBAL_RNG, scale=0.02)
                no_improve += 1

        p_prev = p
        it += 1

        if no_improve > 25:
            p_prev = None
            a = perturbed_restart(a, _GLOBAL_RNG, scale=0.05)
            no_improve = 0

    rem = end - time.monotonic()
    if rem > 60.0:
        a = group_polish(a, budget_s=min(90.0, rem), max_sweeps=1000)
        update_global_best(exact_score(a), a)

    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _GLOBAL_BEST_SCORE, _GLOBAL_BEST_A
    global _GLOBAL_START, _GLOBAL_DEAD, _GLOBAL_RNG, _LAST_FIXED_MS

    start = time.monotonic()
    budget_s = max(30.0, float(budget_s))
    deadline = start + budget_s - 20.0

    _GLOBAL_START = start
    _GLOBAL_DEAD = deadline
    _GLOBAL_RNG = np.random.default_rng(int(seed))
    _GLOBAL_BEST_SCORE = np.inf
    _GLOBAL_BEST_A = None
    _LAST_FIXED_MS = 0.0020

    parent = None
    try:
        p = height_sequence_1
        if p is not None:
            arr = np.asarray(p, dtype=float).ravel()
            if arr.size >= 32:
                arr = np.clip(arr, 0.0, 1000.0)
                if float(arr.sum()) > 0.01:
                    parent = arr
    except Exception:
        parent = None

    n_main = 8192
    starts = []

    if parent is not None:
        if parent.size <= 12000:
            a = normalize(parent.copy())
            starts.append(a)
        a8 = interpolate_sequence(parent, n_main)
        starts.append(a8192 if False else a8)

    # Scan low-dimensional parametric family.
    taper_grid = [-0.60, -0.35, -0.20, -0.10, 0.0, 0.10, 0.20, 0.35, 0.60]
    power_grid = [-0.20, -0.12, -0.05, 0.0, 0.05, 0.12, 0.20]
    for tp in taper_grid:
        for pw in power_grid:
            starts.append(central_power_seq(n_main, taper=tp, power=pw))

    # Add a few blend / reverse variants based on the best parametric sequence.
    best_param_sc = np.inf
    best_param_a = None
    for a in starts:
        if len(a) == n_main:
            sc = exact_score(a)
            update_global_best(sc, a)
            if sc < best_param_sc:
                best_param_sc = sc
                best_param_a = a.copy()

    if best_param_a is not None:
        rev = best_param_a[::-1].copy()
        for alpha in [0.20, 0.35, 0.50, 0.65, 0.80]:
            starts.append(normalize((1.0 - alpha) * best_param_a + alpha * rev))

    # Pick best and second-best main-length starts manually (no lambda).
    main_first_sc = np.inf
    main_first_a = None
    main_second_sc = np.inf
    main_second_a = None

    for a in starts:
        if len(a) == n_main:
            sc = exact_score(a)
            update_global_best(sc, a)
            if sc < main_first_sc:
                main_second_sc = main_first_sc
                main_second_a = main_first_a
                main_first_sc = sc
                main_first_a = a.copy()
            elif sc < main_second_sc:
                main_second_sc = sc
                main_second_a = a.copy()

    if main_first_a is None:
        main_first_a = central_power_seq(n_main, taper=0.0, power=0.0)
        main_first_sc = exact_score(main_first_a)
        update_global_best(main_first_sc, main_first_a)

    other_first_sc = np.inf
    other_first_a = None
    for a in starts:
        if len(a) != n_main:
            sc = exact_score(a)
            update_global_best(sc, a)
            if sc < other_first_sc:
                other_first_sc = sc
                other_first_a = a.copy()

    total_time = max(120.0, budget_s - 20.0)

    # Main optimization on the 8192-length sequence.
    main_budget = min(0.55 * total_time, max(120.0, deadline - time.monotonic() - 120.0))
    if time.monotonic() < deadline - 30.0:
        main_first_a = optimize_sequence(main_first_a, main_budget, "main8192")

    # Second 8192 start, if available.
    if main_second_a is not None and time.monotonic() < deadline - 120.0:
        second_budget = min(0.25 * total_time, max(60.0, (deadline - time.monotonic()) * 0.45))
        main_second_a = optimize_sequence(main_second_a, second_budget, "second8192")

    # If the best so far is the other (parent) length, optimize it too.
    if other_first_a is not None and time.monotonic() < deadline - 90.0:
        if (_GLOBAL_BEST_A is not None and len(_GLOBAL_BEST_A) == len(other_first_a)):
            other_budget = min(0.28 * total_time, max(90.0, (deadline - time.monotonic()) * 0.60))
        else:
            other_budget = min(0.15 * total_time, max(60.0, (deadline - time.monotonic()) * 0.45))
        other_first_a = optimize_sequence(other_first_a, other_budget, "other")

    # Final exact polishing run on the global best.
    if _GLOBAL_BEST_A is not None and time.monotonic() < deadline - 25.0:
        final_budget = max(30.0, deadline - time.monotonic() - 10.0)
        final_budget = min(180.0, final_budget)
        a_final = group_polish(_GLOBAL_BEST_A, budget_s=final_budget, max_sweeps=2000)
        update_global_best(exact_score(a_final), a_final)

    if _GLOBAL_BEST_A is None:
        _GLOBAL_BEST_A = central_power_seq(n_main, taper=0.0, power=0.0)

    out_arr = np.asarray(_GLOBAL_BEST_A, dtype=float).ravel()
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    out_arr[~np.isfinite(out_arr)] = 0.0
    if float(out_arr.sum()) < 0.01:
        out_arr = np.ones(64, dtype=float)

    out = [float(x) for x in out_arr]
    try:
        verified = float(evaluate_sequence(out))
        print(f"FINAL verified evaluate_sequence = {verified:.10f}")
    except Exception as e:
        print(f"FINAL internal score = {_GLOBAL_BEST_SCORE:.10f} (evaluate_sequence check failed: {e})")
        print(f"FINAL length = {len(out)} time = {time.monotonic() - start:.1f}s")

    return out
```
