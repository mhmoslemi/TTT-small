    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_full(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def block_obj_grad(x, a_template, free_idx, beta):
    a = a_template.copy()
    a[free_idx] = np.maximum(x, 0.0)
    score, grad_full = lse_obj_grad(a, beta)
    return score, grad_full[free_idx]


def run_block_cd(a0, beta, block_size, time_limit):
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    n_blocks = (n + block_size - 1) // block_size
    sweep = 0

    while time.monotonic() < time_limit:
        for block_i in range(n_blocks):
            if time.monotonic() > time_limit:
                break
            start = block_i * block_size
            end = min(start + block_size, n)
            free_idx = np.arange(start, end)
            block_len = end - start

            if block_len < 2:
                continue

            x0 = a[free_idx].copy()
            bounds = [(0.0, 1000.0)] * block_len
            maxiter_block = min(40, max(10, block_len // 3))
            maxfun_block = maxiter_block * 4

            res = minimize(
                block_obj_grad, x0, args=(a, free_idx, beta),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-14, 'gtol': 1e-11,
                         'maxiter': maxiter_block, 'maxfun': maxfun_block}
            )
            a[free_idx] = np.clip(res.x, 0.0, 1000.0)

        sweep += 1
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if sweep % 5 == 0:
                print(f"    Block CD sweep {sweep}: score={sc:.8f}")

    return best_sc, best_a


def peak_pair_descent(a0, n_iters=2000, time_limit=None):
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.inf, a0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        peak_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        partners = k_star - js
        contribs = a[js] * a[partners]
        total = float(contribs.sum())
        if total < 1e-15:
            break

        top_idx = int(np.argmax(contribs))
        j_best = js[top_idx]
        partner_best = partners[top_idx]

        a_j = a[j_best]
        a_p = a[partner_best]

        if a_j < 1e-14 and a_p < 1e-14:
            break

        step_frac = 0.05 + 0.05 * (1.0 - float(it) / max(1, n_iters))
        candidates_frac = [step_frac, step_frac * 0.5, step_frac * 1.5]

        improved = False
        for frac in candidates_frac:
            if a_j >= a_p:
                a[j_best] = max(0.0, a_j * (1.0 - frac))
            else:
                a[partner_best] = max(0.0, a_p * (1.0 - frac))

            S_tmp = a.sum()
            if S_tmp < 1e-12:
                if a_j >= a_p:
                    a[j_best] = a_j
                else:
                    a[partner_best] = a_p
                continue
            a_tmp = a / S_tmp
            sc_tmp = compute_score(a_tmp)
            if sc_tmp < best_sc:
                best_sc = sc_tmp
                best_a = a_tmp.copy()
                a = a_tmp
                improved = True
                break
            else:
                if a_j >= a_p:
                    a[j_best] = a_j
                else:
                    a[partner_best] = a_p

        if not improved and it > 100:
            # Try increasing the smaller element of the worst pair
            if a_j < a_p and a_j > 1e-14:
                for frac in [0.1, 0.2, 0.5]:
                    a[j_best] = a_j * (1.0 + frac)
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j_best] = a_j
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < best_sc:
                        best_sc = sc_tmp
                        best_a = a_tmp.copy()
                        a = a_tmp
                        improved = True
                        break
                    a[j_best] = a_j
            elif a_p < a_j and a_p > 1e-14:
                for frac in [0.1, 0.2, 0.5]:
                    a[partner_best] = a_p * (1.0 + frac)
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[partner_best] = a_p
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < best_sc:
                        best_sc = sc_tmp
                        best_a = a_tmp.copy()
                        a = a_tmp
                        improved = True
                        break
                    a[partner_best] = a_p

    return best_sc, best_a


def fienup_flat(a, n_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = 0.1 + 0.5 * progress
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_target = (1.0 - alpha) * P + alpha * P_mean
        P_target = np.maximum(P_target, 1e-30)
        ratio = (P_target / (P + 1e-30)) ** alpha
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def make_init_uniform(n):
    return np.ones(n) / n


def make_init_cosine(n):
    x = np.linspace(0, n - 1, n)
    a = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_init_hann(n):
    x = np.linspace(0, n - 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1 + 1e-10)))
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_init_beta_dist(n, alpha_val, beta_val):
    x = np.linspace(0.01, 0.99, n)
    a = x ** (alpha_val - 1) * (1.0 - x) ** (beta_val - 1)
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_init_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_init_bimodal(n, center_sep=0.3, width=0.15):
    x = np.linspace(0, 1, n)
    c1 = 0.5 - center_sep
    c2 = 0.5 + center_sep
    a = np.exp(-0.5 * ((x - c1) / width) ** 2) + np.exp(-0.5 * ((x - c2) / width) ** 2)
    a = np.maximum(a, 1e-10)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def lse_anneal(a0, beta_schedule, time_limit):
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in beta_schedule:
        if time.monotonic() > time_limit:
            break
        remaining = time_limit - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 20))
        maxfun = min(100000, maxiter * 20)
        if maxiter < 100:
            maxiter = 100
            maxfun = 2000

        sc, a_opt = run_lbfgs_full(a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      LSE beta={beta}: score={sc:.8f}")
        a = a_opt

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 128:
            parent = np.clip(p, 0.0, 1000.0)
            sc_p = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
            try_update(sc_p, parent, "parent")
    except Exception:
        pass

    # Determine working length
    if parent is not None and 512 <= len(parent) <= 20000:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: n={n_work}")

    # Generate initializations
    inits = []
    if parent is not None:
        inits.append(("parent", parent.copy()))
    inits.append(("uniform", make_init_uniform(n_work)))
    inits.append(("cosine", make_init_cosine(n_work)))
    inits.append(("hann", make_init_hann(n_work)))
    inits.append(("beta_2_2", make_init_beta_dist(n_work, 2.0, 2.0)))
    inits.append(("beta_0.5_0.5", make_init_beta_dist(n_work, 0.5, 0.5)))
    inits.append(("gauss", make_init_gaussian(n_work, 0.4)))
    inits.append(("bimodal", make_init_bimodal(n_work, 0.3, 0.2)))
    # Random
    a_rand = np.abs(rng.standard_normal(n_work)) + 0.1
    inits.append(("random", a_rand / a_rand.sum()))

    # Sort by initial score (best first)
    scored_inits = []
    for name, a_init in inits:
        sc = compute_score(a_init)
        scored_inits.append((sc, name, a_init))
    scored_inits.sort(key=lambda x: x[0])
    print(f"Initializations sorted: {[f'{n}={s:.4f}' for s, n, _ in scored_inits]}")

    # Phase 1: LSE annealing on top initializations
    # Time allocation: 40% for LSE annealing
    phase1_end = t_start + 0.40 * (total_budget - 60.0)
    print(f"\n=== Phase 1: LSE Annealing (budget to t={phase1_end - t_start:.0f}s) ===")

    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    max_inits_to_refine = 4

    refined_candidates = []

    for i, (sc_init, name, a_init) in enumerate(scored_inits[:max_inits_to_refine]):
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  Annealing '{name}' (init score={sc_init:.6f}, remaining={remaining:.0f}s)...")

        # Fienup pre-processing for non-parent inits
        if name != "parent":
            a_fienup = fienup_flat(a_init, n_iters=100)
            sc_f = compute_score(a_fienup)
            a_working = a_fienup if sc_f < sc_init else a_init
            sc_init = min(sc_init, sc_f)
        else:
            a_working = a_init.copy()

        sc_anneal, a_anneal = lse_anneal(a_working, beta_schedule, time_limit=phase1_end - 30)
        try_update(sc_anneal, a_anneal, f"anneal-{name}")
        refined_candidates.append((sc_anneal, name, a_anneal))

    # Also try the current best through annealing if not already done
    if best_a is not None and time.monotonic() < phase1_end - 30:
        sc_anneal, a_anneal = lse_anneal(best_a, [2000, 5000, 10000], time_limit=phase1_end - 10)
        try_update(sc_anneal, a_anneal, "anneal-best")
        refined_candidates.append((sc_anneal, "best", a_anneal))

    # Phase 2: Block coordinate descent (50% of time)
    phase2_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Block CD (budget to t={phase2_end - t_start:.0f}s) ===")

    # Sort refined candidates by score
    refined_candidates.sort(key=lambda x: x[0])

    for sc_cand, name_cand, a_cand in refined_candidates[:3]:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 120:
            break
        print(f"  Block CD on '{name_cand}' (score={sc_cand:.6f}, remaining={remaining:.0f}s)")

        # Try different block sizes
        for bs in [100, 50]:
            if time.monotonic() > phase2_end - 60:
                break
            sc_bc, a_bc = run_block_cd(a_cand, beta=3000, block_size=bs,
                                       time_limit=phase2_end - 30)
            try_update(sc_bc, a_bc, f"blockCD-{name_cand}-bs{bs}")
            if sc_bc < sc_cand:
                a_cand = a_bc
                sc_cand = sc_bc

    # Phase 3: Final polish (10% of time)
    print(f"\n=== Phase 3: Final Polish ===")

    if best_a is not None:
        # Peak pair descent
        if time.monotonic() < deadline - 120:
            sc_pp, a_pp = peak_pair_descent(best_a, n_iters=3000,
                                            time_limit=deadline - 120)
            try_update(sc_pp, a_pp, "peak-pair")
            if sc_pp < best_score:
                best_a = a_pp

        # Final high-beta L-BFGS
        if time.monotonic() < deadline - 60:
            sc_final, a_final = run_lbfgs_full(best_a, 20000, maxiter=3000, maxfun=60000)
            try_update(sc_final, a_final, "final-LBFGS-20k")
            if sc_final < best_score:
                best_a = a_final

        # One more block CD pass on the best
        if time.monotonic() < deadline - 30:
            sc_final2, a_final2 = run_block_cd(best_a, beta=10000, block_size=80,
                                               time_limit=deadline - 20)
            try_update(sc_final2, a_final2, "final-blockCD")

        # Final peak pair pass
        if time.monotonic() < deadline - 10:
            sc_final3, a_final3 = peak_pair_descent(best_a, n_iters=1000,
                                                    time_limit=deadline - 5)
            try_update(sc_final3, a_final3, "final-peak-pair")

    # Output
    if best_a is None:
        out = [0.5] * 128
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
