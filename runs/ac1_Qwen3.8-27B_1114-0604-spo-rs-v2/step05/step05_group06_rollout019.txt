One approach: use the log-sum-exp (LSE) smoothing to make the problem smooth, then apply Nesterov acceleration to the smoothed problem. As the smoothing parameter β → ∞, the smoothed problem approaches the original.

The LSE objective is:
`L_β(a) = (1/β) * log(sum_m exp(β * c[m]))`

where `c[m] = (a*a)[m]`.

For large β, `L_β ≈ max(c)`.

The gradient is:
`∇L_β[j] = (1/β) * sum_m exp(β*c[m]) / sum_m' exp(β*c[m']) * ∇c[m][j]`
`= sum_m softmax(β*c)[m] * ∇c[m][j]`
`= sum_m w[m] * 2*a[m-j]` (with boundary)

where `w[m] = softmax(β*c)[m]`.

This is a convex smooth function, so we can apply any smooth convex optimization method (L-BFGS, Nesterov, etc.).

But the issue is that for large β, the function becomes very steep in some directions and flat in others, making it hard for gradient-based methods.

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def _grad_from_weights(a, w):
    """Compute gradient: grad[j] = 2 * sum_{k=0}^{n-1} a[k] * w[j+k] for j=0..n-1."""
    n = a.size
    M = w.size  # should be 2n-1

    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(U * W, N)[:n + M - 1]
    return 2.0 * conv[n-1 : 2*n-1]

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _project_simplex(v, target=1.0):
    """Project v onto {x : sum(x) = target, x >= 0}."""
    n = v.size
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target) / rho > 0
    rho_hat = rho[cond][-1]
    theta = (cssv[cond][-1] - target) / rho_hat
    return np.maximum(v - theta, 0.0)

def _lse_obj_grad(a, beta, top_k=None):
    """LSE-smoothed objective and gradient."""
    n = a.size
    c = _self_conv(a)
    S = a.sum()
    # Normalize: we're optimizing with sum(a) = 1
    d = c  # since S = 1, the objective is 2n * max(c)
    
    if top_k is not None and top_k < len(c):
        # Use top-k approximation
        idx = np.argpartition(d, -top_k)[-top_k:]
        d_work = d[idx]
    else:
        d_work = d
        idx = np.arange(len(d))
    
    dmax = d_work.max()
    z = beta * (d_work - dmax)
    e = np.exp(np.minimum(z, 0))  # safe: z <= 0
    se = e.sum()
    w = e / se
    
    # LSE value
    L = dmax + np.log(se) / beta
    
    # Gradient
    w_full = np.zeros(2*n-1)
    w_full[idx] = w
    grad = _grad_from_weights(a, w_full)
    
    # Objective value: 2n * L (since S=1)
    val = 2.0 * n * L
    return val, grad

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. LSE-smoothed optimization with increasing beta (L-BFGS-B)
2. FISTA (accelerated proximal gradient) on the unsmoothed max-conv objective

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def _grad_from_weights(a, w):
    """grad[j] = 2 * sum_{k=0}^{n-1} a[k] * w[j+k] for j=0..n-1.
    Uses FFT: result[j] = (a_rev * w)[j+n-1]"""
    n = a.size
    M = w.size
    N = _next_pow2(n + M - 1)
    U = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(U * W, N)[:n + M - 1]
    return 2.0 * conv[n-1:2*n-1]


def _score(a):
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _project_simplex(v, target=1.0):
    """Project v onto simplex: {x : sum(x)=target, x>=0}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target) / rho > 0
    if not cond.any():
        return np.full(n, target / n)
    rho_hat = rho[cond][-1]
    theta = (cssv[cond][-1] - target) / rho_hat
    return np.maximum(v - theta, 0.0)


def _lse_obj_grad(a, beta, top_k=None):
    """LSE-smoothed 2*n*max(c) objective and gradient (assuming sum(a)=1)."""
    n = a.size
    c = _self_conv(a)
    M = 2 * n - 1
    
    if top_k is not None and top_k < M:
        idx = np.argpartition(c, M - top_k)[M - top_k:]
        c_act = c[idx]
    else:
        idx = np.arange(M)
        c_act = c
    
    cmax = c_act.max()
    z = beta * (c_act - cmax)
    z = np.minimum(z, 0.0)  # for numerical stability
    e = np.exp(z)
    se = e.sum()
    if se < 1e-300:
        w = np.ones(len(c_act)) / len(c_act)
        se = 1.0
    else:
        w = e / se
    
    L = cmax + np.log(se) / beta
    val = 2.0 * n * L
    
    w_full = np.zeros(M)
    w_full[idx] = w
    grad = _grad_from_weights(a, w_full)
    
    return float(val), grad


def _run_lbfgs(a0, beta, top_k, time_limit, maxfun=2000):
    """Run L-BFGS-B on LSE objective with simplex constraint (via projection in penalty)."""
    a = _norm(a0)
    n = a.size
    if time_limit < 2.0:
        return _score(a), a
    
    est_t = 0.003 * max(0.5, n / 4096.0)
    mf = int(0.8 * time_limit / max(est_t, 1e-4))
    mf = max(20, min(mf, maxfun, 10000))
    
    def obj(x):
        v, g = _lse_obj_grad(x, beta, top_k)
        # Add penalty for deviation from sum=1
        S = x.sum()
        pen = 1e-2 * (S - 1.0) ** 2
        g = g + 2e-2 * (S - 1.0)
        return v + pen, g
    
    try:
        res = minimize(obj, a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': min(mf, 5000), 'maxfun': mf,
                                'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 30})
        out = _norm(res.x)
    except Exception:
        out = a
    return _score(out), out


def _fista(a0, max_iters, deadline=None, beta=100.0, verbose=False):
    """
    FISTA (Fast Iterative Shrinkage-Threshold Algorithm) for minimizing
    f(a) = max_m (a*a)[m] over the simplex {a: sum(a)=1, a>=0}.
    Uses LSE smoothing with given beta for the gradient.
    """
    a = _norm(a0)
    n = a.size
    x = a.copy()  # FISTA iterate
    y = a.copy()  # "lookahead" point
    t = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    
    # Initial Lipschitz constant estimate
    # For the LSE-smoothed objective, the Hessian is bounded by ~ beta * (something)
    # Let's use a backtracking approach
    L = 1.0  # initial step size parameter
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        val, grad = _lse_obj_grad(y, beta)
        grad = np.nan_to_num(grad, nan=0.0)
        
        # Backtracking line search
        while True:
            x_new = _project_simplex(y - grad / L)
            # Check descent
            val_new, _ = _lse_obj_grad(x_new, beta, top_k=50)
            if val_new <= val + 0.5 / L * np.sum((x_new - y) ** 2):
                break
            L *= 2.0
            if L > 1e10:
                x_new = _project_simplex(y - grad / L)
                break
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _norm(y)
        x = x_new
        t = t_new
        
        # Track best
        sc = _score(x)
        if sc < best_sc:
            best_sc = sc
            best_a = x.copy()
            if verbose and (it % 100 == 0):
                print(f"    FISTA it={it}: {best_sc:.8f}")
    
    return best_sc, best_a


def _subgrad_polish(a0, max_iters=200, deadline=None):
    """
    Exact subgradient descent with adaptive step size and momentum.
    Uses the true (non-smoothed) subgradient.
    """
    a = _norm(a0)
    n = a.size
    best_a = a.copy()
    best_sc = _score(a)
    D = np.sqrt(2.0)  # diameter of simplex
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _self_conv(a)
        cmax = c.max()
        if cmax <= 0:
            break
        
        # Active set: positions within 0.01% of max
        threshold = cmax * (1.0 - 1e-5)
        active = c >= threshold
        if not active.any():
            active[np.argmax(c)] = True
        n_act = active.sum()
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        
        grad = _grad_from_weights(a, w)
        grad = np.nan_to_num(grad)
        
        # Subtract mean to make it orthogonal to 1 (tangent to simplex)
        grad = grad - grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size: try several
        best_step_a = None
        best_step_sc = best_sc
        eta0 = D / gnorm
        for scale in (0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0):
            eta = eta0 * scale
            q = _project_simplex(a - eta * grad)
            sc = _score(q)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step_a = q
        
        if best_step_a is None:
            break
        
        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = best_step_a.copy()
        
        a = best_step_a
    
    return best_sc, best_a


def _upsample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a)


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)
