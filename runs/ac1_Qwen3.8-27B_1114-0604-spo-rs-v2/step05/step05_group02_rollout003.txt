"""
Hybrid optimization for flat convolution:
1. GS-Fienup spectral method for coarse optimization
2. LSE-smoothed Adam optimizer for fine-tuning
3. Peak-specific coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_smooth(a, beta):
    """LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    # Gradient of L w.r.t. a: convolve p with a (reversed)
    # dc_k/da_j = a_{k-j}, so dL/da_j = sum_k p_k * a_{k-j}
    # This is the cross-correlation of p and a
    # p has length 2n-1, a has length n
    # result[j] = sum_k p_k * a_{k-j} for valid k-j
    # = cross-correlation
    rev_a = a[::-1]
    M2 = len(p) + n - 1  # = 3n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    # Full gradient of score
    score = 2.0 * n * L / (S * S)
    grad_L = conv_res  # dL/da_j
    grad = (2.0 * n) * (grad_L / (S*S) - 2.0 * L / (S**3))
    return score, grad

def adam_optimize(a, beta, n_steps, lr, deadline):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:

        score, grad = lse_smooth(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2:
- GS-Fienup spectral method with adaptive annealing
- LSE-smoothed Adam optimizer for fine-tuning
- Multi-phase search with time management
- Peak-aware local optimization
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (unnormalized)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + (max_scaled + np.log(exp_scaled.sum())) / beta
    # dL/da_j = 2 * sum_k p_k * a_{k-j}
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL = 2.0 * conv_res
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n * dL) / (S * S) - (4.0 * n * L) / (S ** 3)
    return score, grad


def adam_lse(a0, beta, n_steps, lr_init, lr_final, deadline):
    """Adam optimizer on LSE-smoothed objective with normalization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0

    best_a = a.copy()
    best_sc = compute_score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        progress = step / max(1, n_steps)
        lr = lr_init + (lr_final - lr_init) * progress

        score, grad = lse_grad(a, beta)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_fienup(a0, n_iters, alpha_init, alpha_final, deadline=None):
    """GS-Fienup with correct flat-convolution target."""
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_magnitude_only(a0, n_iters, alpha, deadline=None):
    """GS variant that only updates magnitude, preserving phase."""
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_peaks(a0, n_iters, top_m, step_init, deadline):
    """Subgradient method focusing on top-M peaks with momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    momentum = np.zeros(n)
    beta_m = 0.9

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        momentum = beta_m * momentum + (1 - beta_m) * grad
        effective_grad = momentum

        step = min(step, 0.8 * (sc - 2.0 * n / (2.0 * n - 1.0)) / (gn * gn + 1e-30))
        step = min(step, 0.05)

        a = a - step * effective_grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def lse_objective_and_grad(x, beta):
    """Objective and gradient for scipy minimize."""
    return lse_grad(x, beta)


def run_lbfgs(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0_c = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0_c.sum() < 1e-6:
        a0_c[:] = 0.5
    a0_c = a0_c / a0_c.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_and_grad, a0_c, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0_c
    except Exception:
        a_out = a0_c
    return compute_score(a_out), a_out


def peak_local_opt(a0, n_rounds, n_top, deadline):
    """Local optimization on elements contributing to peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_k = np.argsort(c)[::-1][:min(5, len(c))]
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_local = np.argsort(contrib)[::-1][:n_top]
            cur_sc = best_sc

            for j in top_local:
                j_global = j_start + int(j)
                a_old = a[j_global]
                if a_old < 1e-10:
                    continue
                factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3]
                best_val = a_old
                for f in factors:
                    a[j_global] = a_old * f
                    s_tmp = a.sum()
                    if s_tmp < 1e-12:
                        a[j_global] = a_old
                        continue
                    sc_trial = compute_score(a / s_tmp)
                    if sc_trial < cur_sc:
                        cur_sc = sc_trial
                        best_val = a_old * f
                a[j_global] = best_val

            S = a.sum()
            if S > 1e-12:
                a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits
