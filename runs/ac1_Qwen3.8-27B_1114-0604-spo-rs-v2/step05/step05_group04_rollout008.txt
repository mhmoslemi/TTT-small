Advanced optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: LSE-smoothed gradient descent with Nesterov momentum, modified Fienup spectral shaping,

import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad(a, beta, top_M=None):
    """Return (L, grad) where L = LSE-smoothed max of conv(a,a), grad = dL/da."""
    n = len(a)
    c = fast_conv(a)
    if top_M is not None and top_M < len(c):
        top_idx = np.argsort(c)[::-1][:top_M]
        c_m = np.full_like(c, -1e300)
        c_m[top_idx] = c[top_idx]
    else:

    c_max = c_m.max()
    scaled = beta * (c_m - c_max)
    scaled = np.clip(scaled, -700, 700)
    log_sum = np.log(np.sum(np.exp(scaled)))
    L = c_max + log_sum / beta
    p = np.exp(scaled - log_sum)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return L, np.zeros(n)
    p = p / p_sum
    # grad via FFT
    rev_a = a[::-1]
    M_conv = 3 * n - 2
    M_fft = 1 << (M_conv - 1).bit_length()
    conv_res = np.fft.irfft(np.fft.rfft(p, M_fft) * np.fft.rfft(rev_a, M_fft), M_fft)
    g = 2.0 * conv_res[n-1:2*n-1]
    return L, g


def nesterov_lse_descent(a0, beta=200, top_M=50, n_iters=5000, max_time=None, t_start=None):
    """Nesterov-accelerated LSE-smoothed gradient descent with projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    a_prev = a.copy()
    m = 0.0  # momentum
    best_score = compute_score(a)
    best_a = a.copy()
    
    step_init = 1.0 / (beta * 2.0 * n)
    
    for it in range(n_iters):
        if max_time is not None:
            if time.monotonic() - t_start > max_time:
                break
        
        # Nesterov update
        m_new = (1.0 + np.sqrt(1.0 + 2.0 * (it + 1))) / (2.0 + np.sqrt(1.0 + 2.0 * (it + 1)))
        y = a + m * (a - a_prev)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-12:
            y = a.copy()
        else:
            y = y / S_y
        
        L, grad = lse_grad(y, beta, top_M)
        
        # Full objective gradient (including the (sum a)^2 in denominator)
        # f = 2n * L / S^2, df/da = 2n * (grad/S^2 - 2*L*a/S^3)
        S_y = y.sum()
        S2 = S_y * S_y
        S3 = S2 * S_y
        full_grad = 2.0 * n * (grad / S2 - 2.0 * L / S3)
        
        # Line search or fixed step
        grad_norm = np.linalg.norm(full_grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step: Polyak-like
        step = step_init / (1.0 + 0.01 * it)
        # Clip step to avoid overshooting
        step = min(step, 0.1 / (grad_norm + 1e-10))
        
        a_new = y - m_new * step * full_grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = y.copy()
        else:
            a_new = a_new / S_new
        
        a_prev = a.copy()
        a = a_new
        m = m_new
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def fienup_adaptive(a0, n_iters=2000, 
                    top_frac_init=0.1, top_frac_end=0.4,
                    compress_power=0.5,
                    floor_frac=0.01):
    """
    Adaptive Fienup with:
    - Progressive top-fraction
    - Power compression
    - Floor boost for low-frequency bins
    - Peak-aware: uses time-domain peaks to guide spectral modifications
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * (1 - np.cos(np.pi * progress * 0.5))
        
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        
        # Identify top bins
        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if above.any():
            ratio = np.ones_like(P)
            ratio[above] = (thresh / P[above]) ** compress_power
            P *= ratio
        
        # Boost low bins (floor)
        low_thresh = np.quantile(P, floor_frac)
        below = P < low_thresh
        if below.any():
            P[below] = low_thresh
        
        # Reconstruct
        amp = np.sqrt(P / (A.real**2 + A.imag**2 + 1e-30))
        A_mod = A * amp
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def active_set_subgradient(a0, n_iters=3000, top_M=10, step_init=0.01, decay=0.9999):
    """
    Subgradient method using top-M peaks.
    Step size: min(Polyak, fixed)
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1)  # theoretical LB (unachievable but useful)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = a.sum()
        M_val = c.max()
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-M peaks
        top_idx = np.argsort(c)[::-1][:top_M]
        
        # Compute subgradient as weighted combination
        # Weights: proportional to how much each peak exceeds the average
        c_avg = c.mean()
        weights = np.maximum(c[top_idx] - c_avg, 0.0)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(top_M) / top_M
        else:
            weights = weights / w_sum
        
        # Compute gradient: g_j = 2 * sum_k w_k * a[k_j - j]
        rev_a = a[::-1]
        # Build the weighted peak function and convolve
        p = np.zeros(2 * n - 1)
        for k, w in zip(top_idx, weights):
            p[k] += w
        
        M_conv = 3 * n - 2
        M_fft = 1 << (M_conv - 1).bit_length()
        conv_res = np.fft.irfft(np.fft.rfft(p, M_fft) * np.fft.rfft(rev_a, M_fft), M_fft)
        g = 2.0 * conv_res[n-1:2*n-1]
        
        # Full gradient including normalization
        S2 = S * S
        S3 = S2 * S
        full_grad = (2.0 * n) * (g / S2 - 2.0 * M_val / S3)
        
        gn = np.linalg.norm(full_grad)
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = min(step, (sc - lower_bound) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        
        a_new = a - step * full_grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            step *= 0.5
            continue
        a = a_new / S_new
        step *= decay
    
    return best_a, best_sc


def lbfgs_lse(a0, beta=500, maxiter=5000, maxfun=100000):
    """L-BFGS-B optimization of the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S0
    
    def objective_gradient(x):
        x = np.maximum(x, 0.0)
        L, g_L = lse_grad(x, beta)
        x = np.maximum(x, 0.0)
        S = x.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        S2 = S * S
        S3 = S2 * S
        val = 2.0 * n * L / S2
        grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
        return val, grad
    
    res = minimize(
        objective_gradient, a0,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    sc = compute_score(a_out)
    return sc, a_out


def perturbed_fienup_lbfgs_chain(a0, rng, budget_s=300):
    """Chain: perturbation -> Fienup -> LBFGS, repeated."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_sc = compute_score(a0)
    
    for trial in range(10):
        if time.monotonic() - t0 > budget_s:
            break
        # Perturb
        noise = 0.03 * (1 + trial * 0.3)
        a_p = best_a * (1.0 + noise * rng.standard_normal(n))
        a_p = np.maximum(a_p, 0.0)
        S = a_p.sum()
        if S < 1e-12:
            continue
        a_p = a_p / S
        
        # Fienup
        a_f, sc_f = fienup_adaptive(a_p, n_iters=100, top_frac_init=0.08, top_frac_end=0.3)
        
        # LBFGS
        sc_l, a_l = lbfgs_lse(a_f, beta=300, maxiter=2000, maxfun=50000)
        
        if sc_l < best_sc:
            best_sc = sc_l
            best_a = a_l.copy()
            print(f"    Chain trial {trial}: {best_sc:.8f}")
    
    return best_a, best_sc
