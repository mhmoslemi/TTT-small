```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative
sequences. Uses LP direction finding with reduced active sets as the main driver,
combined with mirror descent (multiplicative weight) updates, frequency-domain
reshaping, and LSE-smoothed L-BFGS-B polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def normalize(a, target=1.0):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / max(1, a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    # corr_pair: for each j, compute sum_k w[k] * a[k-j] (valid range)
    # This is the correlation of a with w (length 2n-1)
    # corr[j] = sum_k w[k] * a[k-j] where 0<=k<2n-1 and 0<=k-j<n
    # = sum_j' w[j'+j] * a[j'] where j' = k-j
    # So corr = ifft(rfft(a) * rfft(w_reversed)) ... let me use the direct method
    n_w = len(w)  # 2n-1
    N = next_power_of_two(n_w + n - 1)
    A_rev = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A_rev * W, N)[:n_w + n - 1]
    h = full[n - 1:2 * n - 1]  # length n

    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a, beta, deadline, maxfun=None):
    a = normalize(a)
    n = len(a)
    rem = float(deadline - time.monotonic())
    if rem < 4.0:
        return exact_score(a), a

    if maxfun is None:
        est_time = 0.02 * max(0.5, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
        maxfun = int((0.8 * rem) / max(est_time, 1e-4))
        maxfun = max(64, min(2000, maxfun))

    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize(res.x)
    else:
        a_out = a

    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def lp_direction(a, M, n_active=50, timeout_s=10.0):
    """
    Solve LP: max sum(b) s.t. conv(a,b)_k <= M for top-n_active peaks, b >= 0.
    Returns the LP solution b (unscaled), or None on failure.
    """
    n = len(a)
    c = fast_self_conv(a)
    if c.size == 0:
        return None

    # Find active set: positions where c is closest to M
    # Use all positions within a small fraction of the max, up to n_active
    c_sorted = np.sort(c)[::-1]
    cmax = c_sorted[0]
    if cmax <= 0:
        return None

    # Get top n_active positions
    if n_active >= c.size:
        active_indices = np.arange(c.size)
    else:
        # Use argpartition for efficiency
        active_indices = np.argpartition(c, -n_active)[-n_active:]

    K = len(active_indices)
    if K == 0:
        return None

    # Build constraint matrix C_S (K x n)
    # C_S[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0
    C_S = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(active_indices):
        k = int(k)
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            C_S[i, indices] = a[k - indices]

    # Solve LP: max sum(b) s.t. C_S @ b <= M * ones(K), b >= 0
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=C_S, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout_s, 'presolve': True})
    except Exception:
        return None

    if time.monotonic() - t0 > timeout_s * 1.5:
        return None

    if res.status != 0 or res.x is None:
        return None

    b = np.asarray(res.x, dtype=np.float64)
    b = np.clip(b, 0.0, None)
    if b.sum() < 1e-10:
        return None

    return b


def mirror_step(a, c, n_active=10, eta=0.3):
    """Multiplicative weight update based on active peaks."""
    n = len(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return a

    # Find active positions
    if n_active >= c.size:
        active_mask = np.ones(c.size, dtype=bool)
    else:
        active_indices = np.argpartition(c, -n_active)[-n_active:]
        active_mask = np.zeros(c.size, dtype=bool)
        active_mask[active_indices] = True

    # Compute correlation h[j] = sum_k active_mask[k] * a[k-j]
    w = active_mask.astype(np.float64)
    # h[j] = sum_k w[k] * a[k-j] for valid j
    N = next_power_of_two(c.size + n - 1)
    A_rev = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(np.zeros(c.size), N)  # placeholder
    # Actually use the correlation directly
    w_full = np.zeros(c.size, dtype=np.float64)
    w_full[active_mask] = 1.0
    W = np.fft.rfft(w_full, N)
    full = np.fft.irfft(A_rev * W, N)[:c.size + n - 1]
    h = full[n - 1:2 * n - 1]

    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a

    return a * np.exp(-eta * h / hmax)


def freq_reshape(a, alpha=1.0):
    """Frequency-domain power reshaping to flatten autocorrelation."""
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    Pmax = float(np.max(P))
    if Pmax < 1e-12:
        return a
    # Compress dynamic range
    P_new = P ** (1.0 / (1.0 + alpha))
    P_new = P_new / float(np.max(P_new)) * Pmax
    # Keep the phase
    A_new = A / (np.abs(A) + 1e-300) * np.sqrt(P_new)
    a_new = np.fft.irfft(A_new, N)[:n]
    return np.clip(a_new, 0.0, None)


def search_lp(a, deadline, verbose=True, n_active_min=20, n_active_max=100):
    """
    Main LP-based search loop.
    Iteratively finds LP directions and blends with current sequence.
    """
    a = normalize(a)
    best_a = a.copy()
    best_score = exact_score(a)
    n = len(a)

    if verbose:
        print("  LP search start: score={:.8f}, n={}".format(best_score, n))

    t = 0.3  # initial blend parameter
    n_active = n_active_min
    stagnant = 0
    iter_count = 0

    while time.monotonic() < deadline - 10.0:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        c = fast_self_conv(a)
        M = float(np.max(c))

        # Adaptive active set size: increase as we converge
        progress = 1.0 - (best_score - 1.0) / max(0.01, 2.0 - 1.0)
        n_active_cur = int(n_active_min + progress * (n_active_max - n_active_min))
        n_active_cur = max(n_active_min, min(n_active_max, n_active_cur))

        # Timeout for LP decreases as time runs out
        lp_timeout = min(10.0, max(2.0, rem * 0.05))

        b = lp_direction(a, M, n_active=n_active_cur, timeout_s=lp_timeout)

        if b is None or b.sum() < 1e-8:
            # Fallback: mirror step
            a_new = mirror_step(a, c, n_active=min(n_active_cur, 20), eta=0.2)
            a_new = normalize(a_new)
            sc = exact_score(a_new)
            if np.isfinite(sc) and sc < best_score * 1.001:
                a = a_new
            stagnant += 1
        else:
            b = normalize(b)
            # Try different blend parameters
            improved = False
            for t_try in [t, t * 0.5, t * 0.25, 0.1, 0.05]:
                a_new = normalize((1.0 - t_try) * a + t_try * b)
                sc = exact_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    stagnant = 0
                    # Adapt step size
                    if sc < best_score * 0.999:
                        t = min(0.8, t * 1.1)
                    else:
                        t = max(0.02, t * 0.9)
                    break

            if not improved:
                # Accept the LP direction anyway with a small step
                a_new = normalize((1.0 - 0.05) * a + 0.05 * b)
                sc = exact_score(a_new)
                if np.isfinite(sc) and sc < exact_score(a) * 1.01:
                    a = a_new
                stagnant += 1

        iter_count += 1

        # Periodic mirror step diversification
        if iter_count % 20 == 0 and stagnant < 10:
            a_m = mirror_step(a, c, n_active=10, eta=0.1)
            a_m = normalize(a_m)
            sc_m = exact_score(a_m)
            if np.isfinite(sc_m) and sc_m < best_score:
                best_score = sc_m
                best_a = a_m.copy()
                a = a_m

        # Check for stagnation
        if stagnant > 30:
            # Perturb to escape
            rng = np.random.default_rng(iter_count * 7 + 3)
            a = normalize(a * np.exp(0.1 * rng.standard_normal(n)))
            stagnant = 0

        if verbose and iter_count % 50 == 0:
            print("    LP iter {}: score={:.8f}, n_active={}, t={:.3f}, stagnant={}".format(
                iter_count, best_score, n_active_cur, t, stagnant))

    if verbose:
        print("  LP search done: best={:.8f}, iters={}".format(best_score, iter_count))

    return best_score, best_a


def quick_optimize(a, deadline):
    """Quick multi-beta LSE optimization."""
    a = normalize(a)
    best_a = a.copy()
    best_score = exact_score(a)

    for beta in [300, 1000, 3000, 10000, 30000]:
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = normalize(a2)

    return float(best_score), best_a


def perturb_a(a, rng, strength=0.3):
    """Perturb a sequence for diversification."""
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.3))
        q = (1 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        # V-shape perturbation
        x = np.linspace(0, 1, n)
        v = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = a * v
    elif mode == 3:
        # Frequency reshape
        q = freq_reshape(a, alpha=float(rng.uniform(0.5, 2.0)))
    elif mode == 4:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    else:
        # Block perturbation
        nb = int(rng.integers(4, 32))
        per = max(1, n // nb)
        block_mult = 0.5 + rng.random(nb)
        q = np.repeat(block_mult, nb * per)[:n] * a

    return normalize(np.clip(q, 0.0, None))


def generate_inits(n, rng, parent=None, cap=15):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        inits.append(normalize(arr))

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    # V-shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # U-shapes (quadratic)
    for alpha in [1.0, 2.0, 5.0]:
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Cosine
    for f in [1, 2, 3]:
        for alpha in [0.5, 1.0, 2.0]:
            add(1.0 + alpha * np.cos(np.pi * x))
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Exponential
    for rate in [1.0, 3.0, 5.0, 10.0]:
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Sine power
    for p in [2, 4, 8]:
        add(np.abs(np.sin(np.pi * x)) ** p)
        add(1.0 - 0.5 * np.abs(np.sin(np.pi * x)) ** p)

    # Random smooth
    Nf = next_power_of_two(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + 0.5 * noise)

    # Random
    add(0.1 + 0.9 * rng.random(n))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns the best sequence found."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Get parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 128:
                parent = p
    except Exception:
        pass

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                        parent.size if parent is not None else 0))

    # Determine sequence lengths to try
    if parent is not None:
        n_parent = min(parent.size, 16384)
        if n_parent < 2048:
            n_parent = 2048
        lengths = [n_parent, 8192, 12288]
    else:
        lengths = [8192, 10240, 6144]

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:3]
    print("Lengths: {}".format(lengths))

    # Phase 1: Find good starting points
    print("Phase 1: Finding starting points...")
    best_start = None
    best_start_score = np.inf
    best_start_n = 0

    for n in lengths:
        if time.monotonic() > deadline - 600.0:
            break
        inits = generate_inits(n, rng, parent if parent is not None and parent.size > 100 else None,
                               cap=20)
        for arr in inits:
            sc = exact_score(arr)
            if np.isfinite(sc) and sc < best_start_score:
                best_start_score = sc
                best_start = arr.copy()
                best_start_n = n

    if best_start is None:
        n0 = lengths[0]
        best_start = np.ones(n0) / n0
        best_start_score = exact_score(best_start)
        best_start_n = n0

    print("Best start: score={:.8f}, n={}".format(best_start_score, best_start_n))

    best_seq = best_start.copy()
    best_score = best_start_score

    # Phase 2: LP-based optimization on the best start
    print("Phase 2: LP optimization...")
    n0 = len(best_seq)
    a0 = best_start.copy()

    # Time allocation for LP search
    lp_deadline = t_start + 800.0
    lp_deadline = min(lp_deadline, deadline - 400.0)
    lp_deadline = max(lp_deadline, t_start + 100.0)

    sc, seq = search_lp(a0, lp_deadline, verbose=True, n_active_min=20, n_active_max=80)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print("After LP: {:.8f}".format(best_score))

    # Phase 3: LSE-LBFGS polish
    print("Phase 3: LSE polish...")
    polish_deadline = deadline - 300.0
    sc, seq = quick_optimize(best_seq, polish_deadline)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print("After polish: {:.8f}".format(best_score))

    # Phase 4: Try different lengths / restarts
    print("Phase 4: Restarts...")
    n_restarts = 3
    restart_budget_each = max(60.0, (deadline - time.monotonic() - 200.0) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 200.0:
            break
        chunk_deadline = time.monotonic() + restart_budget_each

        # Create perturbed version
        p = perturb_a(best_seq, rng, strength=float(rng.uniform(0.15, 0.5)))

        # Quick LP search
        sc_r, seq_r = search_lp(p, chunk_deadline, verbose=False, n_active_min=20, n_active_max=50)
        if np.isfinite(sc_r) and sc_r < best_score:
            best_score = sc_r
            best_seq = seq_r.copy()
            print("  Restart {}: {:.8f}".format(r + 1, best_score))

    # Phase 5: Final high-beta LSE polish
    print("Phase 5: Final polish...")
    if time.monotonic() < deadline - 50.0:
        rem = float(deadline - time.monotonic() - 10.0)
        if rem > 20.0:
            sc, seq = run_lbfgs(best_seq, 50000.0, deadline - 10.0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("  Final LSE: {:.8f}".format(best_score))

        if time.monotonic() < deadline - 30.0:
            sc, seq = run_lbfgs(best_seq, 100000.0, deadline - 10.0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("  Very high LSE: {:.8f}".format(best_score))

    # Phase 6: Final mirror descent cleanup
    if time.monotonic() < deadline - 15.0:
        a = normalize(best_seq)
        for _ in range(50):
            if time.monotonic() > deadline - 15.0:
                break
            c = fast_self_conv(a)
            a_new = mirror_step(a, c, n_active=5, eta=0.1)
            a_new = normalize(a_new)
            sc = exact_score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
            else:
                break

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        # Normalize so sum is reasonable (the evaluate function handles normalization)
        # But keep values in [0, 1000]
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
```
