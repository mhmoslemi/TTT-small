"""
Adam-softmax optimization for flat-convolution sequence design.
Uses softmax parameterization (a = softmax(b)) to automatically satisfy
non-negativity and normalization. Combines LSE smoothing with adaptive
learning rates for robust convergence.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient w.r.t. a.
    Score = 2n * L_beta / S^2 where L_beta ≈ max(c).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum
    
    # dL/da_j = cross-correlation of a and p, evaluated at lags 0..n-1
    # r[j] = sum_m a_m * p[m+j]
    M_pad = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.fft(a, M_pad)
    P_fft = np.fft.fft(p, M_pad)
    dL_da = np.fft.ifft(np.conj(A_fft) * P_fft).real[:n]
    
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n / (S ** 3)) * (S * dL_da - 2.0 * L)
    
    return score, grad


def softmax_from_b(b):
    """Convert unconstrained b to a = softmax(b)."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()


def adam_softmax_optimize(b0, n_iters, beta_schedule, lr=1e-3, 
                          beta1=0.9, beta2=0.999, eps=1e-8,
                          check_interval=100, verbose=False):
    """
    Adam optimizer on softmax parameterization with LSE-smoothed objective.
    b0: initial unconstrained parameters
    beta_schedule: function mapping progress [0,1] to beta value
    """
    n = len(b0)
    b = b0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    
    a = softmax_from_b(b)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = beta_schedule(progress)
        
        sc, grad_a = lse_score_grad(a, beta)
        
        if it % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
        
        # Transform gradient to b-space
        dot = float(np.dot(grad_a, a))
        grad_b = a * (grad_a - dot)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * grad_b ** 2
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_from_b(b)
    
    # Final check
    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()
    
    return best_a, best_sc


def gs_fienup(a, n_iters=300, alpha_init=0.7, alpha_final=0.1):
    """
    GS-Fienup iteration with flat convolution target.
    """
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def subgradient_refine(a0, n_iters=5000, n_top=20, verbose=False):
    """
    Subgradient method with momentum on top peaks.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        if n_top >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / (S2 * S))
        
        # Polyak step size
        gn = float(np.dot(grad, grad))
        if gn < 1e-30:
            break
        step = (sc - sc_lb) / (gn + 1e-30)
        step = min(step, 0.05)
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def lbfgs_polish(a0, beta=5000, maxiter=3000):
    """L-BFGS-B polish on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    
    def obj(x):
        x = np.maximum(x, 0.0)
        sc, g = lse_score_grad(x, beta)
        return sc, g
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1000.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10,
                            'ftol': 1e-18, 'gtol': 1e-14})
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    return compute_score(a_out), a_out

"""
Adam-softmax optimizer with LSE annealing for flat-convolution sequence design.
Key innovations over previous approach:
- Softmax parameterization eliminates constraint handling
- Adam with adaptive learning rates escapes local minima better than L-BFGS
- Annealed LSE beta schedule provides multi-scale optimization
- Momentum-based subgradient for peak refinement

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Score = 2n * L_beta / S^2 where L_beta approximates max(c).
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum

    # dL/da_j = cross-correlation of a and p at lags 0..n-1
    # r[j] = sum_m a_m * p[m+j]
    M_pad = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.fft(a, M_pad)
    P_fft = np.fft.fft(p, M_pad)
    dL_da = np.fft.ifft(np.conj(A_fft) * P_fft).real[:n]

    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n / (S * S * S)) * (S * dL_da - 2.0 * L)

    return score, grad


def softmax_from_b(b):
    """Convert unconstrained b to a = softmax(b). Numerically stable."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()


def beta_schedule_linear(progress):
    """Linear beta annealing: 50 -> 5000."""
    return 50.0 + progress * 4950.0


def beta_schedule_ramp(progress):
    """Ramp beta: 2000 -> 10000."""
    return 2000.0 + progress * 8000.0


def beta_schedule_high(progress):
    """High beta: 5000 -> 20000."""
    return 5000.0 + progress * 15000.0


def beta_schedule_const_high(progress):
    """Constant high beta: 10000."""
    return 10000.0


def beta_schedule_const_med(progress):
    """Constant medium beta: 3000."""
    return 3000.0


def adam_softmax_optimize(b0, n_iters, schedule_fn, lr, beta1=0.9, beta2=0.999, eps=1e-8, check_interval=500):
    """
    Adam optimizer in softmax parameterization with LSE-smoothed objective.
    b0: initial unconstrained parameter vector
    schedule_fn: function(progress) -> beta
    Returns (best_a, best_score).
    """
    n = len(b0)
    b = b0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)

    a = softmax_from_b(b)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = schedule_fn(progress)

        sc, grad_a = lse_score_grad(a, beta)

        if (it + 1) % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        # Transform gradient to b-space: grad_b = a * (grad_a - dot(grad_a, a))
        dot_val = float(np.dot(grad_a, a))
        grad_b = a * (grad_a - dot_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * grad_b * grad_b
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_from_b(b)

    # Final evaluation
    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc


def adam_direct_optimize(a0, n_iters, schedule_fn, lr, beta1=0.9, beta2=0.999, eps=1e-8, check_interval=500):
    """
    Adam optimizer directly on a (with projection to simplex).
    a0: initial non-negative sequence (will be normalized)
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = schedule_fn(progress)

        sc, grad = lse_score_grad(a, beta)

        if (it + 1) % check_interval == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = best_a.copy()
        else:
            a_new = a_new / S_new
        a = a_new

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()

    return best_a, best_sc


def gs_fienup(a0, n_iters=300, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup iteration with flat convolution target."""
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_momentum(a0, n_iters=5000, n_top=20, mu=0.85):
    """Subgradient method with momentum on top peaks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    momentum = np.zeros(n)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if n_top >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:n_top]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / (S2 * S))

        gn = float(np.dot(grad, grad))
        if gn < 1e-30:
            break
        step = (sc - sc_lb) / (gn + 1e-30)
        step = min(step, 0.03)

        momentum = mu * momentum + (1.0 - mu) * grad
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def lbfgs_polish(a0, beta, maxiter=2000):
    """L-BFGS-B polish on LSE-smoothed objective."""
    n = len(a0)
    a_init = np.maximum(a0, 0.0).copy()
    S = a_init.sum()
    if S < 1e-6:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / S

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(obj_and_grad, a_init, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1000.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 15,
                            'ftol': 1e-18, 'gtol': 1e-14})
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    return compute_score(a_out), a_out


def perturb_sequence(a0, rng, frac=0.05, scale=0.3):
    """Randomly perturb a fraction of elements for diversity."""
    n = len(a0)
    a = a0.copy()
    n_perturb = max(1, int(frac * n))
    idx = rng.choice(n, size=n_perturb, replace=False)
    a[idx] = a[idx] * (1.0 + scale * rng.standard_normal(n_perturb))
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    Phase 1: GS-Fienup initialization (fast, ~60s)
    Phase 2: Adam-softmax with annealed LSE (main, ~1200s)
    Phase 3: Direct Adam with restarts (~400s)
    Phase 4: Subgradient with momentum (~300s)
    Phase 5: L-BFGS polish (~150s)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    N = 6962

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(N) / N

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | t={elapsed:.0f}s")
            return True
        return False

    print(f"Starting | Parent: {'yes(n=' + str(len(parent)) + ')' if parent is not None else 'no'} | N={N}")

    # ============================================================
    # Phase 1: GS-Fienup Initialization
    # ============================================================
    print("\n=== Phase 1: GS-Fienup ===")
    t_phase1_end = t_start + 80.0

    if parent is not None and len(parent) == N:
        a0 = parent / parent.sum()
    else:
        a0 = np.ones(N) / N

    a_gs = gs_fienup(a0, n_iters=300, alpha_init=0.7, alpha_final=0.1)
    try_update(compute_score(a_gs), a_gs, "GS-parent")

    # Additional GS from diverse inits
    x = np.arange(N, dtype=np.float64) / N
    diverse_inits = [
        np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * x), 0.01),
        0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2,
        np.clip(0.5 + 0.2 * rng.standard_normal(N), 0.01, None),
        np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x + 0.5), 0.01),
    ]
    for i, a_init in enumerate(diverse_inits):
        if time.monotonic() > t_phase1_end:
            break
        a_gs_i = gs_fienup(a_init, n_iters=200, alpha_init=0.6, alpha_final=0.05)
        try_update(compute_score(a_gs_i), a_gs_i, f"GS-diverse-{i}")

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: Adam-Softmax with Annealing (main optimization)
    # ============================================================
    print("\n=== Phase 2: Adam-Softmax ===")
    t_phase2_start = time.monotonic()
    t_phase2_budget = 1200.0

    # Estimate iterations: ~12ms per iter
    ms_per_iter = 12.0
    n_iters_p2 = int(t_phase2_budget * 1000.0 / ms_per_iter * 0.85)
    n_iters_p2 = max(1000, min(n_iters_p2, 100000))
    print(f"  Estimated iterations: {n_iters_p2}")

    # Initialize b from best_a
    a_start = best_a.copy()
    S = a_start.sum()
    a_start = a_start / S
    a_start = np.maximum(a_start, 1e-15)
    b0 = np.log(a_start / a_start.mean())

    a_p2, sc_p2 = adam_softmax_optimize(
        b0, n_iters=n_iters_p2,
        schedule_fn=beta_schedule_linear,
        lr=0.01, beta1=0.9, beta2=0.999,
        check_interval=1000
    )
    try_update(sc_p2, a_p2, "Adam-SF-linear")

    # Continue with higher beta if time remains
    if time.monotonic() < t_phase2_start + t_phase2_budget:
        a_cur = best_a.copy()
        S = a_cur.sum()
        a_cur = a_cur / S
        a_cur = np.maximum(a_cur, 1e-15)
        b_cur = np.log(a_cur / a_cur.mean())

        remaining_s = t_phase2_start + t_phase2_budget - time.monotonic()
        n_iters_p2b = int(remaining_s * 1000.0 / ms_per_iter * 0.85)
        n_iters_p2b = max(1000, min(n_iters_p2b, 80000))

        a_p2b, sc_p2b = adam_softmax_optimize(
            b_cur, n_iters=n_iters_p2b,
            schedule_fn=beta_schedule_ramp,
            lr=0.005, beta1=0.9, beta2=0.999,
            check_interval=1000
        )
        try_update(sc_p2b, a_p2b, "Adam-SF-ramp")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Direct Adam with restarts
    # ============================================================
    print("\n=== Phase 3: Direct Adam ===")
    t_phase3_start = time.monotonic()
    t_phase3_budget = 400.0

    # Run several short Adam optimizations from perturbed starts
    for restart in range(5):
        if time.monotonic() > t_phase3_start + t_phase3_budget:
            break

        if restart == 0:
            a_r = best_a.copy()
        else:
            a_r = perturb_sequence(best_a, rng, frac=0.03, scale=0.2)

        n_iters_r = int(50.0 * 1000.0 / ms_per_iter * 0.8)
        n_iters_r = max(500, min(n_iters_r, 30000))

        if restart < 2:
            a_r_opt, sc_r = adam_direct_optimize(
                a_r, n_iters=n_iters_r,
                schedule_fn=beta_schedule_high,
                lr=0.001, beta1=0.9, beta2=0.999,
                check_interval=500
            )
        else:
            a_r_opt, sc_r = adam_direct_optimize(
                a_r, n_iters=n_iters_r,
                schedule_fn=beta_schedule_const_med,
                lr=0.002, beta1=0.9, beta2=0.999,
                check_interval=500
            )
        try_update(sc_r, a_r_opt, f"Adam-direct-r{restart}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Subgradient with momentum
    # ============================================================
    print("\n=== Phase 4: Subgradient ===")
    t_phase4_start = time.monotonic()
    t_phase4_budget = 300.0

    for n_top in [10, 30, 50, 100]:
        if time.monotonic() > t_phase4_start + t_phase4_budget:
            break
        n_iters_sg = 3000
        a_sg = subgradient_momentum(best_a, n_iters=n_iters_sg, n_top=n_top, mu=0.85)
        try_update(compute_score(a_sg), a_sg, f"SG-mom-top{n_top}")

    # Try from perturbed starts
    for pert in range(3):
        if time.monotonic() > t_phase4_start + t_phase4_budget:
            break
        a_p = perturb_sequence(best_a, rng, frac=0.05, scale=0.15)
        a_sg = subgradient_momentum(a_p, n_iters=2000, n_top=30, mu=0.9)
        try_update(compute_score(a_sg), a_sg, f"SG-mom-pert{pert}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS Polish
    # ============================================================
    print("\n=== Phase 5: L-BFGS Polish ===")
    t_phase5_start = time.monotonic()
    t_phase5_budget = max(60.0, deadline - time.monotonic())

    for beta in [5000, 20000, 50000]:
        if time.monotonic() > deadline - 5:
            break
        sc, a_opt = lbfgs_polish(best_a, beta=beta, maxiter=1500)
        try_update(sc, a_opt, f"LBFGS-b{beta}")

    # Final Adam polish
    if time.monotonic() < deadline - 10:
        a_cur = best_a.copy()
        S = a_cur.sum()
        a_cur = a_cur / S
        a_cur = np.maximum(a_cur, 1e-15)
        b_cur = np.log(a_cur / a_cur.mean())
        remaining = deadline - time.monotonic() - 5.0
        n_iters_final = int(remaining * 1000.0 / ms_per_iter * 0.8)
        n_iters_final = max(500, min(n_iters_final, 30000))
        a_final, sc_final = adam_softmax_optimize(
            b_cur, n_iters=n_iters_final,
            schedule_fn=beta_schedule_const_high,
            lr=0.001, beta1=0.9, beta2=0.999,
            check_interval=200
        )
        try_update(sc_final, a_final, "Final-Adam-HF")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
