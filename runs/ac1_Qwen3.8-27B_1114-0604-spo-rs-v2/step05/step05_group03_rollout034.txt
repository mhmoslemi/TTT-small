def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

"""
Multi-phase optimizer for flat-autocorrelation non-negative sequences.
Uses exact LP (HiGHS) with constraint generation for the alternating subproblem,
combined with L-BFGS LSE-smoothing and Nesterov-accelerated subgradient methods.
Key improvements: exact LP solves, adaptive constraint sets, golden-section line search,
and diversified restarts.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_sparse_rows(a, row_indices, n):
    """Build sparse data for rows row_indices of the conv matrix."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in row_indices:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(js), k, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(a[k - js])
    if not rows_list:
        return None, None, None
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return rows, cols, vals


def lp_subproblem_cg(a, M, active_frac=0.4, max_cg=15, time_per_lp=30.0):
    """
    Solve LP: max sum(x) s.t. cross_conv(a,x)[k] <= M for all k, x >= 0
    using constraint generation with HiGHS.
    Returns x (unnormalized) or None.
    """
    n = len(a)
    m = 2 * n - 1
    conv_aa = self_conv(a)

    # Start with top active_frac fraction of constraints
    threshold = np.percentile(conv_aa, (1.0 - active_frac) * 100.0)
    active_idx = np.where(conv_aa >= threshold)[0]
    if len(active_idx) < 50:
        active_idx = np.argsort(conv_aa)[::-1][:200]

    best_x = None
    for cg_iter in range(max_cg):
        rows, cols, vals = build_sparse_rows(a, active_idx, n)
        if rows is None:
            break
        A_sub = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))

        c_obj = -np.ones(n)
        b_sub = M * np.ones(len(active_idx))

        try:
            res = linprog(c=c_obj, A_ub=A_sub, b_ub=b_sub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': time_per_lp,
                                   'dual_feasibility_tolerance': 1e-9,
                                   'primal_feasibility_tolerance': 1e-9})
        except Exception:
            try:
                res = linprog(c=c_obj, A_ub=A_sub, b_ub=b_sub,
                              bounds=[(0.0, None)] * n, method='highs',
                              options={'presolve': True, 'time_limit': time_per_lp})
            except Exception:
                break

        if not res.success or res.x is None:
            break

        x = np.maximum(res.x, 0.0)
        best_x = x.copy()

        # Check all constraints
        all_conv = cross_conv(a, x)
        violated = (all_conv > M * (1.0 + 1e-6))
        n_violated = int(violated.sum())

        if n_violated == 0:
            return x

        # Add violated constraints
        viol_idx = np.where(violated)[0]
        active_idx = np.unique(np.concatenate([active_idx, viol_idx]))

        # Cap the number of constraints to avoid slowdown
        if len(active_idx) > 8000:
            # Keep the most violated ones
            cvals = all_conv[active_idx]
            order = np.argsort(-cvals)
            active_idx = active_idx[order[:8000]]

    return best_x


def lp_subproblem_fista(a, M, n_iters=10000, beta_start=10.0, beta_end=300.0):
    """Fallback FISTA-based LP solver (when HiGHS is too slow)."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_iters):
        if it < int(n_iters * 0.2):
            frac = it / max(1, int(n_iters * 0.2))
            beta = beta_start + (beta_end - beta_start) * frac
        else:
            beta = beta_end

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer=60, time_limit=None, use_hg=True,
                   active_frac=0.4, max_cg=15, fista_iters=10000):
    """
    LP-based alternating method with golden-section line search.
    a should be normalized to sum=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    rng = np.random.default_rng(123)

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        conv = self_conv(a)
        M = float(conv.max())

        # Solve LP
        if use_hg:
            x = lp_subproblem_cg(a, M, active_frac=active_frac, max_cg=max_cg)
        else:
            x = lp_subproblem_fista(a, M, n_iters=fista_iters)

        if x is None or float(x.sum()) < 1e-12:
            x = lp_subproblem_fista(a, M, n_iters=fista_iters)
            if x is None or float(x.sum()) < 1e-12:
                break

        Sx = float(x.sum())
        x = x / Sx

        # Golden-section line search on t
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c1 = lo + (1.0 - gr) * (hi - lo)
        c2 = lo + gr * (hi - lo)

        a_c1 = np.maximum((1.0 - c1) * a + c1 * x, 0.0)
        a_c2 = np.maximum((1.0 - c2) * a + c2 * x, 0.0)
        sc_c1 = compute_score(a_c1 / a_c1.sum()) if a_c1.sum() > 1e-12 else np.inf
        sc_c2 = compute_score(a_c2 / a_c2.sum()) if a_c2.sum() > 1e-12 else np.inf

        for _ in range(20):
            if sc_c1 < sc_c2:
                hi = c2
                c2 = c1
                sc_c2 = sc_c1
                c1 = lo + (1.0 - gr) * (hi - lo)
                a_c1 = np.maximum((1.0 - c1) * a + c1 * x, 0.0)
                sc_c1 = compute_score(a_c1 / a_c1.sum()) if a_c1.sum() > 1e-12 else np.inf
            else:
                lo = c1
                c1 = c2
                sc_c1 = sc_c2
                c2 = lo + gr * (hi - lo)
                a_c2 = np.maximum((1.0 - c2) * a + c2 * x, 0.0)
                sc_c2 = compute_score(a_c2 / a_c2.sum()) if a_c2.sum() > 1e-12 else np.inf

        t_best = (lo + hi) / 2.0
        a_new = np.maximum((1.0 - t_best) * a + t_best * x, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn
        sc = compute_score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                # Diversification
                a = best_a.copy()
                a = a * (1.0 + 0.03 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0

        a = a_new

    return best_a, best_sc


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max) - beta * c_max
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = 2.0 * c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val * p.sum() / S3 * 0 + 2.0 * L_val / S3 * S)
    # Simplify: d/da [2n * L / S^2] = 2n * (grad_L / S^2 - 2*L/S^3)
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):

        return s, g

    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,

                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a, n_iters=500, time_limit=None):
    """
    Nesterov-accelerated subgradient method on the exact (non-smooth) objective.
    Objective: f(a) = 2n * max_k (a*a)[k] / (sum a)^2, with a normalized to sum=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    y = a.copy()
    t_k = 1.0
    step0 = 0.5
    decay = 0.9995

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        conv_y = self_conv(y)
        k_star = int(np.argmax(conv_y))
        M_val = float(conv_y[k_star])
        S_y = float(y.sum())

        # Subgradient of max_k (a*a)[k] at the peak
        # (a*a)[k] = sum_i a[i]*a[k-i], gradient w.r.t. a[j] = a[k-j] + (if j != k-j) a[j]... 
        # Actually for (a*a)[k] = sum_i a[i] a[k-i], d/d a[j] = a[k-j] + a[j] (if 2j != k)
        # More carefully: (a*a)[k] = sum_i a_i a_{k-i}. d/d a_j = a_{k-j} + (if j != k-j) a_j * 0 + ...
        # Let me just use: d/da_j [(a*a)[k]] = a[k-j] (treating it as bilinear and ignoring the
        # fact that a appears twice -- this is a valid subgradient direction)
        # Actually for the full gradient: (a*a)[k] = sum_i a[i] a[k-i]
        # d/da[j] = a[k-j] + sum_i a[i] * d/d a[j] a[k-i] = a[k-j] + a[k-j] = 2*a[k-j] (if j != k-j)
        # or = a[k-j] + a[j] = 2*a[j] (if j = k-j, i.e., 2j = k)
        # So the gradient is: 2 * (a reversed and shifted)
        j_valid_start = max(0, k_star - n + 1)
        j_valid_end = min(k_star, n - 1)
        grad_peak = np.zeros(n)
        for j in range(j_valid_start, j_valid_end + 1):
            if j == k_star - j:
                grad_peak[j] = 2.0 * a[j]
            else:
                grad_peak[j] = 2.0 * a[k_star - j]

        # Gradient of full score: 2n * (grad_peak / S^2 - 2*M_val / S^3)
        step = step0 * 0.1 / (it + 10)
        a_new = a - step * (grad_peak / (S_y * S_y) - 2.0 * M_val / (S_y ** 3))
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k - 1.0) / t_new * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_k = t_new

    return best_a


def error_reduction_fft(a, n_iters=100, alpha=0.7):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: error reduction, LP alternating (HiGHS + FISTA), L-BFGS,
    Nesterov subgradient, and peak local search.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def report(sc, a_arr, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_arr.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a_arr)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Choose lengths to try
    lengths = [6962]
    if parent is not None and len(parent) not in lengths:
        lengths.append(len(parent))
    lengths = sorted(set(lengths))
    n_main = 6962

    # ==================================================================
    # Phase 1: Quick initialization (8% time)
    # ==================================================================
    phase1_deadline = t_start + 0.08 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Initialization (t_limit={phase1_deadline - t_start:.0f}s) ---")

    for n in [n_main]:
        if time.monotonic() > phase1_deadline:
            break
        inits = build_initializations(n, rng, parent)
        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            S = float(a0.sum())
            if S < 1e-12:
                continue
            a0 = a0 / S
            a_er = error_reduction_fft(a0, n_iters=60, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP alternating with HiGHS (50% time) - MAIN PHASE
    # ==================================================================
    phase2_deadline = t_start + 0.58 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LP Alternating with HiGHS (t_limit={phase2_deadline - t_start:.0f}s) ---")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP alternating
    rounds = [
        {'active_frac': 0.3, 'max_cg': 10, 'max_outer': 40, 'time_per_lp': 20.0},
        {'active_frac': 0.4, 'max_cg': 12, 'max_outer': 40, 'time_per_lp': 25.0},
        {'active_frac': 0.5, 'max_cg': 15, 'max_outer': 40, 'time_per_lp': 30.0},
        {'active_frac': 0.6, 'max_cg': 20, 'max_outer': 50, 'time_per_lp': 35.0},
    ]

    for ri, params in enumerate(rounds):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP round {ri}: active_frac={params['active_frac']}, "
              f"max_outer={params['max_outer']}, remaining={remaining:.0f}s")

        a_out, sc_out = lp_alternating(
            a_work,
            max_outer=params['max_outer'],
            time_limit=phase2_deadline - 20,
            use_hg=True,
            active_frac=params['active_frac'],
            max_cg=params['max_cg'],
            fista_iters=10000
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{ri}")
        a_work = a_out.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    # Try a slightly different length if time permits
    if time.monotonic() < phase2_deadline - 120:
        n_alt = 8192
        print(f"  LP alt length: n={n_alt}")
        a_alt = np.interp(np.linspace(0, 1, n_alt),
                          np.linspace(0, 1, len(best_a)), best_a)
        a_alt = np.maximum(a_alt, 0.0)
        S_alt = float(a_alt.sum())
        if S_alt > 1e-12:
            a_alt = a_alt / S_alt
        a_out_alt, sc_out_alt = lp_alternating(
            a_alt, max_outer=20, time_limit=phase2_deadline - 30,
            use_hg=True, active_frac=0.3, max_cg=10
        )
        sc = compute_score(a_out_alt)
        report(sc, a_out_alt, f"LP n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.78 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: L-BFGS Refinement (t_limit={phase3_deadline - t_start:.0f}s) ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [50, 150, 400, 800, 1500, 3000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=5000, maxfun=150000)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Nesterov subgradient + peak search + final polish (remaining)
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # Nesterov subgradient
    if time.monotonic() < deadline - 120:
        a_nest = nesterov_subgrad(best_a, n_iters=400, time_limit=deadline - 100)
        sc = compute_score(a_nest)
        report(sc, a_nest, "Nesterov")

    # Peak local search
    if time.monotonic() < deadline - 60:
        a_peak = peak_local_search(best_a, n_steps=500, rng=rng)
        sc = compute_score(a_peak)
        report(sc, a_peak, "Peak search")

    # Another round of LP with tighter parameters
    if time.monotonic() < deadline - 120:
        print("  Final LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating(
            a_final_in, max_outer=20, time_limit=deadline - 60,
            use_hg=True, active_frac=0.6, max_cg=20
        )
        sc = compute_score(a_final)
        report(sc, a_final, "Final LP")
        a_work = a_final.copy()

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 40:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 5000, maxiter=8000, maxfun=400000)
        report(sc, a_opt, "Final LBFGS b=5000")

    # Final peak search
    if time.monotonic() < deadline - 10:
        a_final = peak_local_search(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "Final peak")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
