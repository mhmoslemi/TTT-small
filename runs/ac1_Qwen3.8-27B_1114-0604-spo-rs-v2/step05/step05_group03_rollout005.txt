"""
Multi-phase optimizer using multiplicative gradient descent (mirror descent),
adaptive peak-focused updates, and LSE-smoothed L-BFGS polishing.
Designed to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def lse_gradient(a, beta):
    """
    Compute gradient of LSE-smoothed convolution max.
    Returns (L_val, grad) where L_val = LSE(beta * conv) and
    grad[j] = dL/d a[j].
    """
    n = len(a)
    c = self_conv(a)
    z = beta * c
    z_max = float(z.max())
    exp_z = np.exp(z - z_max)
    exp_sum = float(exp_z.sum())
    p = exp_z / exp_sum
    L_val = (z_max + np.log(exp_sum)) / beta  # = c_max + log(exp_sum)/beta
    
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # = conv(p, a_flipped)[n-1+j] + conv(p, a)[j]
    a_flip = a[::-1]
    M1 = len(p) + n - 1  # = 3n - 2
    M1_fft = 1 << (M1 - 1).bit_length()
    P = np.fft.rfft(p, M1_fft)
    PA = np.fft.rfft(a, M1_fft)
    PAF = np.fft.rfft(a_flip, M1_fft)
    
    conv_pa = np.fft.irfft(P * PA, M1_fft)[:M1]
    conv_paf = np.fft.irfft(P * PAF, M1_fft)[:M1]
    
    grad = conv_paf[n-1:2*n-1] + conv_pa[0:n]
    return L_val, grad


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def multiplicative_descend(a0, n_iters, eta0, beta_schedule, time_limit=None):
    """
    Multiplicative (mirror) descent on the LSE-smoothed objective.
    a is maintained with sum=1 throughout.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        beta = beta_schedule[min(it, len(beta_schedule) - 1)]
        L_val, grad = lse_gradient(a, beta)
        
        # Multiplicative update
        # Scale gradient for numerical stability
        g_scale = float(grad.max()) + 1e-10
        grad_normed = grad / g_scale
        
        a_new = a * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            # Try to increase eta slightly
            eta = min(eta * 1.05, 2.0)
        else:
            # Reduce eta
            eta = max(eta * 0.7, 0.01)
            # Don't update a
    
    return best_a, best_sc

    Focus on reducing only the top peaks in the convolution.
    Uses multiplicative updates targeted at peak positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.05
    n_iter = 0
    
    while n_iter < n_iters:
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        c = self_conv(a)
        c_mean = float(c.mean())
        c_max = float(c.max())
        
        if c_max - c_mean < 1e-12:
            break
        
        # Select peaks above threshold
        thresh = c_mean + 0.5 * (c_max - c_mean)
        hot = c > thresh
        
        if not hot.any():
            break
        
        # Weighted gradient focused on hot peaks
        # grad[j] = sum_{k: hot[k]} (c[k] - thresh) * (a[k-j] + a[j-k]) / (c_max - c_mean)
        weights = np.where(hot, (c - thresh) / (c_max - c_mean), 0.0)
        
        a_flip = a[::-1]
        M1 = len(weights) + n - 1
        M1_fft = 1 << (M1 - 1).bit_length()
        W = np.fft.rfft(weights, M1_fft)
        PA = np.fft.rfft(a, M1_fft)
        PAF = np.fft.rfft(a_flip, M1_fft)
        
        conv_wa = np.fft.irfft(W * PA, M1_fft)[:M1]
        conv_waf = np.fft.irfft(W * PAF, M1_fft)[:M1]
        
        grad = conv_waf[n-1:2*n-1] + conv_wa[0:n]
        grad = np.maximum(grad, 0)  # Only push down, never push up
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad / (float(grad.max()) + 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta = max(eta * 0.8, 0.01)
        
        n_iter += 1
    
    return best_a, best_sc
