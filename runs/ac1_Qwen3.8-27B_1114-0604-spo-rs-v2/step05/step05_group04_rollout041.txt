"""
A hybrid search for short-peak self-convolution sequences.

Main components:
1. Spectral square-root shaping:
   - Work on the polynomial square coefficients c = a * a.
   - Flatten/compress c with cooking-style maps such as
     c -> c/(1 + mu*c), c -> c**gamma, or blends to a smooth profile.
   - Rebuild a spectral factor A_new from the modified square, keep the
     current phase as closely as possible, inverse-FFT, and project to
     nonnegative sequences.

2. Square-residual L-BFGS:
   - Parameterize a_i = x_i^2 / sum(x^2).
   - Minimize a smooth log-sum-exp approximation of max(a*a), with exact
     score tracking.

3. Time-limited Adam-style polynomial-square-root polishing:
   - Unconstrained x variables, unit-norm rescaling.
   - Weights from top-k active convolution peaks or log-sum-exp softmax.

No IO, no networks, no lambdas, all helpers are top-level.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def get_fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()


def get_grad_fft_len(n):
    total = 3 * n - 2
    return 1 << (total - 1).bit_length()


def conv_with_m(a, m_len):
    a = np.asarray(a, dtype=np.float64)
    A = np.fft.rfft(a, m_len)
    return np.fft.irfft(A * A, m_len)[:2 * a.size - 1]


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S <= 1e-12:
        return np.inf
    a = a / S
    n = a.size
    m = get_fft_len(n)
    c = conv_with_m(a, m)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if not np.isfinite(cmax):
        return np.inf
    return 2.0 * float(n) * cmax


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def interp_to_length(a, n):
    a = normalize_nonnegative(a)
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(x_new, x_old, a, period=None)
    return normalize_nonnegative(out)


def make_profile(n, kind):
    L = 2 * n - 1
    i = np.arange(L, dtype=np.float64)
    if kind == "uniform":
        w = np.ones(L, dtype=np.float64)
    elif kind == "cosine":
        w = 0.5 * (1.0 - np.cos(np.pi * (i + 0.5) / float(L)))
        w = np.maximum(w, 1e-12)
    elif kind == "triangle":
        mid = 0.5 * (L - 1)
        w = 1.0 - np.abs(i - mid) / (0.5 * (L - 1) + 1e-12)
        w = np.maximum(w, 1e-12)
    else:
        w = np.ones(L, dtype=np.float64)
    s = float(w.sum())
    if s <= 0:
        w = np.ones(L, dtype=np.float64)
        s = float(L)
    return w / s


def initial_sequence(n, rng, parent, kind):
    if kind == "parent":
        if parent is not None and parent.size > 0:
            return interp_to_length(parent, n)
        kind = "uniform"

    if kind == "uniform":
        return np.ones(n, dtype=np.float64) / float(n)

    if kind == "random":
        a = 0.5 + 0.3 * rng.standard_normal(n)
        a = np.clip(a, 1e-3, 10.0)
        return normalize_nonnegative(a)

    if kind == "shape":
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        a = (
            0.5
            + 0.30 * np.cos(2.0 * np.pi * (x - 0.5))
            + 0.20 * np.cos(4.0 * np.pi * x)
            + 0.10 * np.cos(6.0 * np.pi * x)
        )
        a = np.maximum(a, 1e-3)
        return normalize_nonnegative(a)

    return np.ones(n, dtype=np.float64) / float(n)


def grad_weighted_len(a, p, m_len=None):
    """
    Gradient of sum_k p_k * (a*a)_k with respect to a.
    Returns 2 * cross-correlation vector b, where
        b_i = sum_k p_k a_{k-i}.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size

    if m_len is None:
        m_len = get_grad_fft_len(n)

    ra = np.fft.rfft(a[::-1], m_len)
    rp = np.fft.rfft(p, m_len)
    out = np.fft.irfft(ra * rp, m_len)[n - 1:n + n]
    return 2.0 * out


def shape_spectrum(a, mode, param_end, n_iters, alpha, profile=None, deadline=None):
    """
    Iterative spectral factor shaping of a*a.

    mode:
      'cap'     : target c = c/(1 + mu*c), mu ramp to param_end
      'power'   : target c = c**gamma, gamma ramps from 1 to param_end
      'profile' : target c = (1-lam)c + lam*profile, lam ramps to param_end
    """
    a = normalize_nonnegative(a)
    n = a.size
    L = 2 * n - 1
    m_len = get_fft_len(n)

    best = a.copy()
    c0 = conv_with_m(a, m_len)
    best_sc = 2.0 * float(n) * float(np.max(c0)) if c0.size > 0 else np.inf

    alpha_cur = float(alpha)
    stall = 0

    for it in range(max(0, int(n_iters))):
        if deadline is not None and time.monotonic() >= deadline:
            break

        r = it / float(max(1, n_iters - 1))
        ramp = 0.5 * (1.0 - np.cos(np.pi * r))

        c = conv_with_m(a, m_len)
        if c.size == 0:
            break

        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            break

        sc = 2.0 * float(n) * cmax
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                alpha_cur = max(1e-4, 0.8 * alpha_cur)
                stall = 0
                if alpha_cur < 0.1 * float(alpha):
                    break

        csum = float(c.sum())
        if csum <= 1e-30 or not np.isfinite(csum):
            break

        c = c / csum

        # Tiny lift to avoid permanent zero coefficients in shaped targets.
        c = c + (1e-12 / float(L))
        csum = float(c.sum())
        if csum <= 1e-30:
            break
        c = c / csum

        if mode == "cap":
            mu = float(param_end) * ramp
            if mu <= 1e-12:
                z = c.copy()
            else:
                z = c / (1.0 + mu * c)
        elif mode == "power":
            g_end = float(param_end)
            if g_end <= 0.0:
                g_end = 0.01
            gamma = g_end + (1.0 - g_end) * (1.0 - ramp)
            if gamma <= 1e-6:
                gamma = 1e-6
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                z = np.power(np.maximum(c, 1e-300), gamma)
        elif mode == "profile":
            lam = float(param_end) * ramp
            if profile is None:
                profile = np.ones(L, dtype=np.float64) / float(L)
            z = (1.0 - lam) * c + lam * profile
        else:
            z = c.copy()

        z = np.maximum(z, 0.0)
        zsum = float(z.sum())
        if zsum <= 1e-30 or not np.isfinite(zsum):
            break
        z = z / zsum

        C = np.fft.rfft(z, m_len).astype(np.complex128)
        if C.size > 0:
            C[0] = max(float(C[0].real), 0.0) + 0.0j
        if C.size > 1:
            C[-1] = max(float(C[-1].real), 0.0) + 0.0j

        A = np.fft.rfft(a, m_len)

        with np.errstate(invalid="ignore", over="ignore"):
            A_t = np.sqrt(C)

        if not np.all(np.isfinite(A_t)):
            bad = ~np.isfinite(A_t)
            A_t[bad] = A[bad]

        # Choose the valid square-root branch closest to the current A locally.
        diff = A_t * np.conj(A)
        scaleA = max(1.0, float(np.max(np.abs(A)))) if A.size > 0 else 1.0
        signs = np.where(diff.real >= 0.0, 1.0, -1.0)
        small = np.abs(A) < (1e-10 * scaleA)
        signs[small] = 1.0
        A_t = A_t * signs

        A_mix = (1.0 - alpha_cur) * A + alpha_cur * A_t
        a_raw = np.fft.irfft(A_mix, m_len)[:n]

        a_pos = np.maximum(a_raw, 0.0)
        spos = float(a_pos.sum())
        if spos <= 1e-12 or not np.isfinite(spos):
            a = a.copy()
        else:
            a = normalize_nonnegative(a_pos / spos)

    return best


def square_soft_objective(x, beta):
    """
    Smooth objective with square parameterization a = x^2 / sum(x^2).
    Returns (LSE(beta*q)/beta, gradient wrt x).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size

    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=np.float64)

    x = np.clip(x, -1e100, 1e100)
    T = float(np.dot(x, x))
    if T <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    a = (x * x) / T
    a = np.maximum(a, 0.0)
    a_sum = float(a.sum())
    if a_sum <= 1e-30:
        return 1e100, np.zeros(n, dtype=np.float64)
    a = a / a_sum

    m_len = get_fft_len(n)
    cg_len = get_grad_fft_len(n)

    q = conv_with_m(a, m_len)
    q = np.maximum(q, 0.0)
    qmax = float(np.max(q)) if q.size > 0 else 0.0

    if qmax <= 0.0 or not np.isfinite(qmax):
        return 1e100, np.zeros(n, dtype=np.float64)

    scaled = float(beta) * (q - qmax)
    scaled = np.clip(scaled, -700.0, 0.0)
    exps = np.exp(scaled)
    ess = float(exps.sum())

    if ess <= 1e-30 or not np.isfinite(ess):
        p = np.full(q.size, 1.0 / max(1, q.size), dtype=np.float64)
        obj = qmax / float(beta)
    else:
        p = exps / ess
        obj = (qmax + float(np.log(ess))) / float(beta)

    if not np.isfinite(obj):
        obj = 1e100

    g = grad_weighted_len(a, p, cg_len)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)

    A = float(np.dot(g, a))
    sqrtT = np.sqrt(T)
    gx = 2.0 * (x / sqrtT) * (g - A)

    if not np.all(np.isfinite(gx)):
        gx = np.zeros(n, dtype=np.float64)

    return float(obj), gx


def run_lbfgs_square(a0, beta, maxiter, maxfun):
    """L-BFGS polish in the square parameterization."""
    a0 = normalize_nonnegative(a0)
    x0 = np.sqrt(a0 + 1e-15)

    if minimize is None:
        return compute_score(a0), a0.copy()

    try:
        res = minimize(
            square_soft_objective,
            x0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            options={
                "ftol": 1e-18,
                "gtol": 1e-10,
                "maxiter": int(max(1, maxiter)),
                "maxfun": int(max(100, maxfun)),
                "maxls": 50,
                "m": 20,
            },
        )
        x = np.asarray(res.x if hasattr(res, "x") else x0, dtype=np.float64)
    except Exception:
        x = x0

    x = np.clip(x, -1e100, 1e100)
    T = float(np.dot(x, x))
    if T <= 1e-300:
        a_out = a0.copy()
    else:
        a_out = (x * x) / T

    a_out = normalize_nonnegative(a_out)
    sc = compute_score(a_out)
    return sc, a_out


def lbfgs_maxiter_for(n, seconds):
    seconds = max(1.0, float(seconds))
    if n <= 4096:
        rate = 18.0
    elif n <= 8192:
        rate = 11.0
    elif n <= 16384:
        rate = 5.5
    elif n <= 32768:
        rate = 2.8
    else:
        rate = 1.5
    return max(50, int(0.8 * seconds * rate))


def square_gradient_polish(a0, time_limit, beta=20000.0, top_k=0, lr=1e-3, deadline=None):
    """
    Time-limited Adam-style polish in square variables x, where a = x^2.
    x is renormalized every step; this does not change a.
    """
    a = normalize_nonnegative(a0)
    n = a.size
    start = time.monotonic()
    limit = start + float(time_limit)

    x = np.sqrt(a + 1e-15)
    norm = float(np.sqrt(np.dot(x, x)))
    if norm > 1e-12:
        x = x / norm

    m_len = get_fft_len(n)
    cg_len = get_grad_fft_len(n)

    c = conv_with_m(x * x, m_len)
    best_sc = 2.0 * float(n) * float(np.max(c)) if c.size > 0 else np.inf
    best = (x * x).copy()
    best = normalize_nonnegative(best)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 1
    b1 = 0.90
    b2 = 0.999
    lr_cur = float(lr)
    stall = 0

    for it in range(10000000):
        if time.monotonic() >= limit:
            break
        if deadline is not None and time.monotonic() >= deadline:
            break

        a = x * x
        T = float(np.dot(x, x))
        if T <= 1e-30:
            break
        a = a / T

        c = conv_with_m(a, m_len)
        cmax = float(np.max(c)) if c.size > 0 else 0.0
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

        sc = 2.0 * float(n) * cmax
        if sc < best_sc - 1e-15:
            best_sc = sc
            best = normalize_nonnegative(a)
            best = best.copy()
            stall = 0
            lr_cur = min(0.05, 1.03 * lr_cur)
        else:
            stall += 1
            if stall > 30:
                lr_cur = max(1e-6, 0.6 * lr_cur)
                stall = 0
                if sc > 1.05 * best_sc:
                    x = np.sqrt(best + 1e-15)
                    norm = float(np.sqrt(np.dot(x, x)))
                    if norm > 1e-12:
                        x = x / norm
                    else:
                        x = np.sqrt(best + 1e-15)
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 1
                    lr_cur = 1e-4

        if int(top_k) > 0:
            if int(top_k) >= c.size:
                p = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
            else:
                p = np.zeros(c.size, dtype=np.float64)
                idx = np.argpartition(c, -int(top_k))[-int(top_k):]
                p[idx] = 1.0 / float(int(top_k))
        else:
            scaled = float(beta) * (c - cmax)
            scaled = np.clip(scaled, -700.0, 0.0)
            exps = np.exp(scaled)
            ess = float(exps.sum())
            if ess <= 1e-30 or not np.isfinite(ess):
                p = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
            else:
                p = exps / ess

        g = grad_weighted_len(a, p, cg_len)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        A = float(np.dot(g, a))
        sqrtT = np.sqrt(T)
        gx = 2.0 * (x / sqrtT) * (g - A)
        if not np.all(np.isfinite(gx)):
            continue

        m = b1 * m + (1.0 - b1) * gx
        v = b2 * v + (1.0 - b2) * (gx * gx)

        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        denom = np.sqrt(vhat) + 1e-12

        x_new = x - lr_cur * mhat / denom
        x_new = np.maximum(x_new, 1e-8)

        S = float(np.dot(x_new, x_new))
        if S <= 1e-30 or not np.isfinite(S):
            x_new = x.copy()
        else:
            x_new = x_new / np.sqrt(S)
            x = x_new

        t += 1

        if it % 250 == 0:
            print(f"    polish n={n} it={it} best={best_sc:.10f} elapsed={time.monotonic()-start:.0f}s")

    return best, best_sc


def record_candidate(length_best, sc, a):
    """Record best candidate per length. Updated dict in-place."""
    if not np.isfinite(sc):
        return False, np.inf

    n = int(len(a))
    sc = float(sc)

    if n not in length_best or sc < length_best[n][0] - 1e-12:
        length_best[n] = (sc, np.array(a, dtype=np.float64, copy=True))
        return True, sc

    return False, length_best[n][0]

            tasks.append((n, "parent", "profile", 0.26, 0.12, 550, int(50 * priority), "cosine"))
            tasks.append((n, "random", "cap", 0.18, 0.15, 500, int(35 * priority), ""))
            tasks.append((n, "shape", "cap", 0.14, 0.12, 400, int(30 * priority), ""))
        else:
            tasks.append((n, "parent", "cap", 0.065, 0.09, 1000, int(55 * priority), ""))
            tasks.append((n, "parent", "cap", 0.210, 0.13, 800, int(75 * priority), ""))
            if priority >= 0.6:
                tasks.append((n, "parent", "power", 0.84, 0.11, 600, int(55 * priority), ""))
                tasks.append((n, "parent", "profile", 0.16, 0.11, 450, int(40 * priority), "uniform"))

            if parent is None:
                tasks.append((n, "uniform", "cap", 0.16, 0.13, 550, int(45 * priority), ""))
                tasks.append((n, "random", "cap", 0.20, 0.16, 450, int(35 * priority), ""))

    print(f"Selected lengths: {lengths}")
    print(f"Spectral tasks: {len(tasks)}")

    for task in tasks:
        if time.monotonic() >= deadline:
            break

        rem = deadline - time.monotonic()
        if rem < 12.0:
            break

        n, init_kind, mode, pval, alpha, iters, sec, pkind = task
        if int(sec) <= 8:
            continue

        limit = min(deadline, time.monotonic() + float(min(int(sec), 3600)))
        if limit - time.monotonic() < 8.0:
            continue

        a0 = initial_sequence(n, rng, parent, init_kind)
        L = 2 * int(n) - 1

        if mode == "cap":
            param = float(pval) * float(L)
            profile = None
        else:
            param = float(pval)
            profile = make_profile(n, pkind) if mode == "profile" else None

        print(
            f"  spectral n={n} init={init_kind} mode={mode} p={pval:.4f} "
            f"alpha={alpha:.3f} iters<= {iters} budget={int(limit - time.monotonic())}s"
        )
        ar = shape_spectrum(
            a0,
            mode=mode,
            param_end=param,
            n_iters=iters,
            alpha=alpha,
            profile=profile,
            deadline=limit,
        )

        sc = compute_score(ar)
        improved, _ = record_candidate(length_best, sc, ar)
        if improved and sc < best_score:
            best_score = sc
            best_a = ar.copy()
            print(f"    NEW BEST after spectral: {sc:.10f} (n={n})")

    # ---------------------------------------------------------------
    # L-BFGS polish phase.
    # ---------------------------------------------------------------
    lb_items = []
    counter = 0
    for n in length_best.keys():
        sc, arr = length_best[n]
        if np.isfinite(sc):
            lb_items.append((float(sc), int(n), counter, arr))
            counter += 1
    lb_items.sort()

    beta_list = [2000.0, 8000.0, 30000.0, 100000.0]

    for item in lb_items[:4]:
        if time.monotonic() >= deadline:
            break

        rem = deadline - time.monotonic()
        if rem < 55.0:
            break

        sc0, n, _, arr0 = item
        if n > 32768:
            continue

        a_cur = normalize_nonnegative(arr0)
        sc_cur = sc0

        for beta in beta_list:
            if time.monotonic() >= deadline:
                break

            rem = deadline - time.monotonic()
            if rem < 45.0:
                break

            if n <= 8192:
                sec_cap = 140.0
            elif n <= 16384:
                sec_cap = 95.0
            else:
                sec_cap = 55.0

            sec = max(15.0, min(rem - 35.0, sec_cap))
            maxiter = lbfgs_maxiter_for(n, sec)
            maxfun = max(1000, int(maxiter * 25))

            print(
                f"  LBFGS n={n} beta={beta:.0f} maxiter={maxiter} "
                f"budget~{int(min(sec, rem - 30.0))}s"
            )

            sc, ar = run_lbfgs_square(a_cur, beta, maxiter, maxfun)
            improved, _ = record_candidate(length_best, sc, ar)

            if sc < best_score:
                best_score = sc
                best_a = ar.copy()
                print(f"    NEW BEST after LBFGS: {sc:.10f} (n={n})")

            if sc < sc_cur - 1e-12:
                a_cur = normalize_nonnegative(ar)
                sc_cur = sc
            else:
                # If the smooth objective moved away from exact score,
                # keep the better exact point as starting point for next beta.
                pass

    # ---------------------------------------------------------------
    # Final exact-ish topical polish phase.
    # ---------------------------------------------------------------
    pol_items = []
    counter = 0
    for n in length_best.keys():
        sc, arr = length_best[n]
        if np.isfinite(sc):
            pol_items.append((float(sc), int(n), counter, arr))
            counter += 1
    pol_items.sort()

    polish_settings = [
        # (seconds, beta, top_k, lr)
        (50.0, 60000.0, 0, 1.0e-3),
        (40.0, 0.0, 16, 1.0e-3),
        (30.0, 0.0, 8, 6.0e-4),
        (20.0, 250000.0, 0, 5.0e-4),
    ]

    for idx, item in enumerate(pol_items[:3]):
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break

        sc0, n, _, arr0 = item
        if idx == 0:
            a_cur = best_a.copy()
        else:
            a_cur = normalize_nonnegative(arr0)
            if len(best_a) == n and best_score < sc0:
                a_cur = best_a.copy()

        print(f"Final polish candidate idx={idx} n={n} score={sc0:.10f}")

        for sec, beta, top_k, lr in polish_settings:
            rem = deadline - time.monotonic()
            if rem < 12.0:
                break

            use_sec = max(8.0, min(float(sec), rem - 8.0))
            if use_sec <= 5.0:
                break

            ar, sc = square_gradient_polish(
                a_cur,
                time_limit=use_sec,
                beta=float(beta),
                top_k=int(top_k),
                lr=float(lr),
                deadline=deadline,
            )

            improved, _ = record_candidate(length_best, sc, ar)
            if sc < best_score:
                best_score = sc
                best_a = ar.copy()
                print(f"    NEW BEST after polish: {sc:.10f} (n={n})")

            if sc < sc0:
                a_cur = ar.copy()
                sc0 = sc

    # Final sweep over per-length bests.
    for n in length_best.keys():
        sc, arr = length_best[n]
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = arr.copy()

    out_a = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    S = float(out_a.sum())
    if S <= 0.01:
        out_a = np.ones(64, dtype=np.float64) / 64.0
    else:
        out_a = out_a / S

    out = [float(x) for x in out_a]

    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    elapsed = time.monotonic() - t_start
    print("=" * 70)
    print(f"Final internal best: {best_score:.12f}")
    print(f"Final validated score: {verified:.12f}")
    print(f"Length: {len(out)}")
    print(f"Elapsed: {elapsed:.1f}s / {total:.1f}s")
    print("=" * 70)

    return out
