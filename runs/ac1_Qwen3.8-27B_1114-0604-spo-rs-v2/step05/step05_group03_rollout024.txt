```python
from scipy.optimize import linprog
import numpy as np

def solve_lp(a, M, n):
    """Solve: max sum(x) s.t. conv(a,x) <= M, x >= 0"""
    # Build sparse matrix A where A[k, j] = a[j] if valid
    # conv(a,x)[k] = sum_j a[j] * x[k-j] for valid j
    # So A[k, j] = a[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # and j in [0, n-1]
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # max sum(x) = min -sum(x)
    # s.t. A x <= M
    #      x >= 0  (handled by bounds)
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sparse, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def build_lp_structure(n):
    """Precompute sparsity structure for convolution LP."""
    rows = np.empty(n * n, dtype=np.int32)  # approximately n^2 nonzeros
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        count = i_end - i_start + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(i_start, i_end + 1)
        idx += count
    return rows[:idx], cols[:idx]

def solve_lp_convolve(a, M, rows, cols):
    """Solve: max sum(x) s.t. conv(a,x) <= M, x >= 0 using HiGHS."""
    n = len(a)

    if res.success:

    return np.ones(n) / n  # fallback
```

        i_end = min(k, n - 1)
        rows_list.append(np.full(i_end - i_start + 1, k, dtype=np.int32))
        cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def solve_lp(a, M, rows, cols):
    n = len(a)
    ki = rows - cols
    vals = a[ki]
    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_sp, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_alternating(a, max_iters, rows, cols, time_limit):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > time_limit:
            break
        
        c_curr = self_conv(a)
        M = float(c_curr.max())
        
        x = solve_lp(a, M, rows, cols)
        if x is None:
            break
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        best_t = 0.3
        best_t_sc = best_sc
        for t in [0.15, 0.25, 0.35, 0.5, 0.65, 0.8]:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t
        t = best_t
        
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                rng = np.random.default_rng(42 + it)
                a_new = a_new * (1.0 + 0.03 * rng.standard_normal(n))
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                no_improve = 0
        
        a = a_new
    
    return best_a, best_sc


def error_reduction(a, n_iters=100, alpha=0.5):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        # Weighted error reduction: focus on peaks
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        conv_mod = conv.copy()
        conv_mod[excess > 0] -= alpha * excess[excess > 0]
        
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0
        
        A_new = A + d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc


def gs_algorithm(a, n_iters=200, gamma=0.5):
    """Gerchberg-Saxton / HIO variant for flat spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Target: flat magnitude (sqrt of mean power)
        target_mag = np.sqrt(np.mean(mag ** 2))
        
        A_new = target_mag * phase
        a_raw = np.fft.ifft(A_new).real[:n]
        
        # HIO: combine with original
        a_proj = np.maximum(a_raw, 0.0)
        a_new = (1.0 - gamma) * a_proj + gamma * a
        
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    
    # Gradient: dL/da = 2 * cross_corr(a, p) at the center
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    # Actually, the gradient of L = logsumexp(beta * (conv(a,a) - cmax)) / beta + cmax
    # dL/da[j] = (1/beta) * sum_k p[k] * d(conv(a,a)[k])/da[j]
    # d(conv(a,a)[k])/da[j] = a[k-j] + a[j-k] (for valid indices)
    # = 2 * (a convolved with reversed a, windowed)
    # More precisely: d(conv(a,a)[k])/da[j] = a[k-j] if 0<=k-j<n, else 0
    #                      + a[j-k] if 0<=j-k<n, else 0
    # So dL/da[j] = (1/beta) * sum_k p[k] * (a[k-j] + a[j-k] for valid)
    # = (1/beta) * 2 * sum_k p[k] * a[|k-j|] (for valid |k-j|)
    # This is 2/beta * (p * a_reversed) convolved, evaluated at j
    
    # Simpler: use the formula grad_L[j] = 2 * sum_k p[k] * a[(k-j) mod ...]
    # Actually let me just compute it directly via the convolution
    # d(conv(a,a)[k])/da[j] = a[k-j] (if valid) + a[j-k] (if valid, and j!=k-j)
    # For the sum: sum_k p[k] * a[k-j] = (p convolved with a_reversed)[j]
    
    rev_a = a[::-1]
    M3 = n + len(p) - 1  # length of conv(p, a)
    M3_fft = 1 << (M3 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M3_fft)
    P = np.fft.rfft(p, M3_fft)
    conv_pa = np.fft.irfft(RA * P, M3_fft)[:M3]
    
    # conv_pa[j] = sum_k p[k] * a[j-k] (for valid j-k)
    # This gives us sum_k p[k] * a[j-k] which is one part of the gradient
    # The other part is sum_k p[k] * a[k-j] = sum_k p[k] * a[(n-1)-(n-1-k+j)] ...
    # Actually, sum_k p[k] * a[k-j] = (a convolved with p_reversed)[j]
    # = same as conv_pa if p is symmetric... but p is not necessarily symmetric
    
    # Let me just use: grad_L[j] = 2 * (a * p_convolved)[j] where p_convolved[j] = sum_k p[k]*a[k-j]
    # which is the cross-correlation of a and p
    
    # cross_corr(a, p)[j] = sum_k a[k] * p[k-j] = sum_k a[k+j] * p[k] (shifting p)
    # This is equivalent to: (a * flip(p))[j] where * is convolution
    # = conv(a, p[::-1])[j + len(p) - 1 - (len(a)-1)]... this is getting complicated
    
    # Let me just use the direct formula:
    # grad_score[j] = 2n/S^2 * grad_L[j] - 4n*L/S^3
    # where grad_L[j] = (2/beta) * sum_k p[k] * a[k-j] (approximately, for the peak term)
    
    # Actually, let me use a simpler approach: finite differences or the chain rule
    # L = cmax + (1/beta) * log(sum_k exp(beta*(c[k]-cmax)))
    # dL/dc[k] = p[k] = exp(beta*(c[k]-cmax)) / sum_j exp(beta*(c[j]-cmax))
    # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j≠k-j)
    # For j = k-j (i.e., 2j=k): dc[k]/da[j] = a[j] (only one term)
    
    # So grad_L[j] = sum_k p[k] * (a[k-j] + a[j-k] for valid)
    # = sum_k p[k] * a[|k-j|] (approximately, where valid)
    
    # This is 2 * cross_correlation(a, p) at position j (with appropriate windowing)
    
    # Let me compute cross_correlation(a, p) properly
    # xcorr(a, p)[j] = sum_k a[k] * p[k-j] (for k-j in range of p)
    # = sum_i a[i+j] * p[i] (substituting i=k-j)
    
    # This is the cross-correlation. I can compute it via FFT:
    # xcorr(a, p) = ifft(fft(a) * conj(fft(p)))
    # But I need to be careful about the length
    
    # For now, let me use a simpler (but correct) gradient:
    # grad_L[j] = 2 * sum_{k: 0<=k-j<2n-1} p[k] * a[k-j] (this is conv(p, a_rev) at position j)
    
    # conv(p, a_rev)[j] = sum_k p[k] * a_rev[j-k] = sum_k p[k] * a[n-1-(j-k)] = sum_k p[k] * a[n-1-j+k]
    # Hmm, this is getting confusing. Let me just use the direct computation.
    
    # Actually, the simplest correct formula:
    # d(conv(a,a)[k])/da[j] = a[k-j] if 0 <= k-j < n, else 0
    #                       + a[j-k] if 0 <= j-k < n, else 0
    # Note: if k-j = j-k (i.e., k=2j), these two terms are the same, so we only count once.
    # But the derivative of a[j]^2 is 2*a[j], so actually:
    # d(conv(a,a)[k])/da[j] = a[k-j] (if valid) + a[j-k] (if valid and j≠k-j)
    # = 2*a[k-j] if j≠k-j and both valid
    # = a[j] if k=2j
    # = a[k-j] if only one is valid
    
    # For the gradient:
    # grad_L[j] = sum_k p[k] * d(conv(a,a)[k])/da[j]
    # = sum_k p[k] * (a[k-j] if 0<=k-j<n) + sum_k p[k] * (a[j-k] if 0<=j-k<n, k≠2j)
    
    # The first sum is: sum_{k=j}^{j+n-1} p[k] * a[k-j] (for k in range(0, 2n-1))
    # = cross-correlation of p and a at lag j
    # The second sum is: sum_{k=j-n+1}^{j} p[k] * a[j-k] (for k in range(0, 2n-1))
    # = cross-correlation of a and p at lag j
    
    # Both sums are related by symmetry. If p is symmetric (p[k] = p[2n-2-k]), then both sums are equal.
    # In general, grad_L[j] = xcorr(p, a)[j] + xcorr(a, p)[j] = 2 * xcorr(p, a)[j] (if p is symmetric)
    
    # For the log-sum-exp, p is NOT symmetric in general. But for the peak (which is typically at k=n-1),
    # p is concentrated near k=n-1, so p[k] ≈ p[2n-2-k], making the two sums approximately equal.
    
    # Let me just compute it as: grad_L[j] = 2 * (p * a)[j] where (p * a) is the cross-correlation
    # For simplicity and speed, I'll use: grad_L ≈ 2 * conv(p, a)[n-1 + j - (n-1)] ... 
    
    # OK I'm overcomplicating this. Let me just use a numerical gradient check for debugging,
    # and for the actual computation, use the formula:
    # grad_L[j] = 2 * sum_i p[i] * a[i - (n-1)] shifted appropriately
    
    # The cleanest way: 
    # d(conv(a,a)[k])/da[j] = a[(k-j) % n] ... no, that's circular convolution, not linear.
    
    # Let me just do it the slow way for correctness:
    # grad_L = zeros(n)
    # for j in range(n):
    #     for k in range(max(0, j), min(2*n-1, j+n)):
    #         if k - j < n:
    #             grad_L[j] += p[k] * a[k-j]
    #     for k in range(max(0, j-n+1), min(2*n-1, j+1)):
    #         if j - k < n and k != 2*j:  # avoid double counting
    #             grad_L[j] += p[k] * a[j-k]
    #     if 2*j < 2*n-1:  # the case k=2j
    #             # already counted once in first sum, need to check
    #             pass
    
    # This is O(n^2) which is too slow for n=7000.
    
    # Let me use the FFT-based approach:
    # sum_k p[k] * a[k-j] for k in [j, j+n-1] (intersected with [0, 2n-2])
    # This is the cross-correlation of p (length 2n-1) and a (length n) at lag j.
    # xcorr(p, a)[j] = sum_k p[k] * a[k-j] where a[m]=0 for m notin [0,n-1]
    # = sum_{m=0}^{n-1} p[m+j] * a[m]
    
    # I can compute this via:
    # xcorr(p, a) = ifft(conj(fft(p, N)) * fft(a, N)) for N >= 2n-1+n-1 = 3n-2
    
    # Let me use N = next power of 2 >= 3n-2
    
    N = 1 << (3 * n - 2).bit_length()
    P_fft = np.fft.rfft(p, N)
    A_fft = np.fft.rfft(a, N)
    # xcorr[p, a][j] = ifft(conj(P) * A)[j] = ifft(P * conj(A))[N-j] ... 
    # Actually: ifft(conj(P) * A)[j] = sum_k p[k] * a[k-j] (circular)
    # For linear: I need N >= len(p) + len(a) - 1 = 3n-2, which I have.
    xcorr = np.fft.irfft(np.conj(P_fft) * A_fft, N)[:2*n-1+n-1]
    # xcorr[j] = sum_k p[k] * a[k-j] for j = 0, ..., N-1
    # But I only need j = 0, ..., n-1
    
    # The gradient is approximately 2 * xcorr[0:n] (symmetric case)
    # More precisely, it's xcorr[0:n] + xcorr_reversed[0:n]
    # where xcorr_reversed[j] = sum_k p[k] * a[j-k] = xcorr_conj[j]
    
    # For now, use the approximation:
    grad_L = 2.0 * xcorr[:n].copy()
    # This is approximately correct when p is roughly symmetric around n-1
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_and_grad, a0,
        args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_local_search(a, n_steps=300, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.02, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.5)
        for mult in [0.3
