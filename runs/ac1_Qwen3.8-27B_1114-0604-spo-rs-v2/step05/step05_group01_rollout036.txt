"""

Uses LSE-smoothed L-BFGS-B with adaptive beta, weighted subgradient descent,

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_conv(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    L = n + m - 1
    N = next_power_of_two(L)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:L]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr_response(a, w):
    """
    Compute R[i] = sum_k w[k]*a[k-i] (valid indices only).
    This is the cross-correlation of w with a.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    nw = w.size
    if n == 0 or nw == 0:
        return np.empty(n, dtype=np.float64)
    L = n + nw - 1
    N = next_power_of_two(L)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:L]
    return full[n - 1:2 * n - 1]


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e4, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / S)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_obj_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient.
    Objective: 2*n*max(conv)/S^2 approximated by 2*n*LSE(conv, beta)/S^2
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = fast_conv(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (c - cmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = cmax + np.log(sum_e) / beta
    # R[i] = sum_k w[k]*a[k-i]
    R = cross_corr_response(a, w)
    LDS = float(np.dot(w, c))

    # f = 2*n*L/S^2, but for gradient use the normalized version
    # Actually let's use f = 2*n*L / S^2
    # df/da[i] = 2*n/S^2 * 2*R[i] - 2*n*L/S^3 * 1
    #          = 4*n*R[i]/S^2 - 2*n*L/S^3
    score = float(2.0 * n * L / (S * S))
    grad = (4.0 * n * R) / (S * S) - 2.0 * n * L / (S ** 3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return score, grad


def run_lbfgs(a, beta, deadline, maxfun=500):
    a = normalize(a)
    n = a.size
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score(a), a
        n_est = max(1.0, float(n) / 4096.0)
        est = 0.015 * n_est * max(1.0, np.log2(max(16, n)) / 12.0)
        cap = int((0.85 * rem) / max(est, 1e-4))
        maxfun = int(min(maxfun, max(32, cap)))

    maxfun = max(32, int(maxfun))
    maxiter = max(30, maxfun)
    bounds = [(0.0, None)] * n

    def fun(x):
        x = np.clip(x, 0.0, None)
        sc, g = lse_obj_grad(x, beta)
        return sc, g

    try:
        res = minimize(
            fun, a, method="L-BFGS-B", jac=True,
            bounds=bounds,
            options={"ftol": 1e-18, "gtol": 1e-14, "maxiter": maxiter,
                     "maxfun": maxfun, "maxls": 40, "iprint": -1}
        )
    except Exception:
        return exact_score(a), a

    a_out = np.clip(res.x, 0.0, None) if res.x is not None else a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), normalize(a_out)


def subgrad_step_weighted(a, eta, temperature=1.0):
    """Subgradient step with exponential weighting over convolution lags."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_conv(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return a
    # Exponential weights
    z = c / (temperature * cmax) if cmax > 0 else c
    e = np.exp(z)
    swe = float(e.sum())
    if swe <= 0:
        w = np.ones(c.size) / c.size
    else:
        w = e / swe
    R = cross_corr_response(a, w)
    Rmax = float(np.max(R))
    if Rmax < 1e-12:
        return a
    a_new = a * np.exp(-eta * R / Rmax)
    return a_new


def subgrad_polish(a, max_iters=40, deadline=None):
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    improved_last = True

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        if not improved_last:
            break
        improved_last = False
        cur_sc = best_sc
        cur_best = None

        for k_active in (1, 2, 3, 5, 8, 12):
            c = fast_conv(a)
            L = len(c)
            # Top-k weights
            if k_active >= L:
                w = np.ones(L) / L
            else:
                idx = np.argpartition(c, L - k_active)[L - k_active:]
                w = np.zeros(L)
                w[idx] = 1.0 / k_active
            R = cross_corr_response(a, w)
            Rmax = float(np.max(R))
            if Rmax < 1e-12:
                continue
            for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = normalize(a * np.exp(-eta * R / Rmax))
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved_last = True
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        # Also try exponential temperature weighting
        for temp_mult in (0.5, 1.0, 2.0, 5.0):
            c = fast_conv(a)
            cmax = float(np.max(c))
            if cmax < 1e-12:
                break
            for eta in (0.01, 0.03, 0.08, 0.2):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step_weighted(a, eta, temperature=temp_mult * cmax)
                q = normalize(q)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved_last = True

        if cur_best is not None:
            a = cur_best
        elif improved_last:
            a = best_a.copy()
        else:
            a = best_a

    return float(best_sc), best_a


def resample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def freq_domain_init(n, rng):
    """
    Generate a sequence with approximately flat power spectrum.
    Choose random phases, compute IFFT, rectify to non-negative.
    """
    N = next_power_of_two(n)
    # Flat magnitude spectrum
    freqs = np.fft.rfftfreq(N)
    power = np.ones(N // 2 + 1, dtype=np.float64)
    # Zero out DC to allow non-trivial shape
    power[0] = 0.0
    # Optionally taper edges
    taper_start = max(1, n // 4)
    taper_end = N // 2 + 1 - n // 4
    if taper_start < taper_end:
        ramp_len = max(1, N // 8)
        for i in range(taper_start, min(taper_start + ramp_len, N // 2 + 1)):
            power[i] *= (i - taper_start) / ramp_len
        for i in range(max(taper_end - ramp_len, 0), taper_end):
            power[i] *= (taper_end - 1 - i) / ramp_len

    # Random phases (conjugate symmetric)
    phase = rng.uniform(-np.pi, np.pi, size=N // 2 + 1)
    spec = power * np.exp(1j * phase)
    spec[0] = 0.0

    # IFFT to get time-domain signal
    sig = np.fft.irfft(spec, N)[:n]
    # Rectify: shift to non-negative
    mn = float(np.min(sig))
    sig = sig - mn + 0.1 * float(np.max(sig) - np.min(sig)) + 1e-6
    # Add small random component for diversity
    sig += 0.05 * rng.random(n)
    return sig


def structured_init(n, rng, parent=None, cap=25):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e4, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05 * float(np.max(arr) - np.min(arr))
        if float(arr.max()) < 1e-10:
            arr = np.ones(n)
        p = normalize(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample(parent, n)
        add(pr)
        add(pr[::-1])

    # Uniform
    add(np.ones(n))

    # V-shapes (U-shaped: high at ends)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0) ** 1.0)
        add(1.0 + alpha * np.abs(2.0 * x - 1.0) ** 2.0)

    # Inverted V (peak in middle)
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * np.cos(np.pi * x))

    # W-shape
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * np.abs(np.cos(np.pi * x)))

    # Bimodal
    for width in (0.1, 0.15, 0.2, 0.3):
        add(np.exp(-(((x - 0.2) / width) ** 2)) + np.exp(-(((x - 0.8) / width) ** 2)))
        add(np.exp(-(((x - 0.3) / width) ** 2)) + np.exp(-(((x - 0.7) / width) ** 2)))

    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.1)

    # Sine-based
    for f in (1, 2, 3, 4, 6, 8):
        add(1.0 + 0.8 * np.sin(np.pi * f * x) ** 2)
        add(1.0 + 0.5 * np.sin(2.0 * np.pi * f * x))

    # Random smooth
    Nf = max(2, next_power_of_two(n))
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(2.0, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.5, 1.5)) * noise)

    # Frequency domain init
    for _ in range(3):
        add(freq_domain_init(n, rng))

    # Linear ramps
    for alpha in (0.5, 1.0, 2.0):
        add(np.abs(x - 0.5) * alpha + 1.0)

    return inits


def perturb(a, rng, strength=None):
    """Perturb a sequence for restart exploration."""
    a = normalize(a)
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * a + t * (1.0 / n)
    elif mode == 2:
        q = a[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        # Subgradient-based perturbation
        c = fast_conv(a)
        cmax = float(np.max(c))
        if cmax > 1e-12:
            idx = np.argmax(c)
            L = len(c)
            w = np.zeros(L)
            w[max(0, idx - 1):idx + 2] = 1.0 / 3.0
            R = cross_corr_response(a, w)
            Rmax = float(np.max(R))
            if Rmax > 1e-12:
                q = a * np.exp(-strength * R / Rmax)
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 4:
        # Smooth perturbation in log-space
        loga = np.log(a + 1e-300)
        k = min(31, max(3, n // 200))
        if k % 2 == 0:
            k += 1
        c_k = k // 2
        sig = max(1.0, c_k / 2.0)
        kern = np.exp(-((np.arange(k) - c_k) ** 2) / (2.0 * sig ** 2))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        loga += strength * 0.5 * noise
        q = np.exp(loga - loga.max())
    elif mode == 5:
        # V-shape modulation
        x = np.linspace(0, 1, n)
        q = a * (1.0 + strength * np.abs(2 * x - 1))
    elif mode == 6:
        # Periodic perturbation
        f = int(rng.integers(1, max(2, n // 50)))
        amp = float(rng.uniform(0.05, 0.2))
        ph = float(rng.uniform(0, 2 * np.pi))
        x = np.linspace(0, 1, n)
        q = a * (1.0 + amp * np.sin(2 * np.pi * f * x + ph))
    elif mode == 7:
        # Block perturbation
        nb = int(rng.integers(4, max(5, n // 500)))
        per = n // nb
        if per < 2:
            per = 2
            nb = n // per
        blocks = 1.0 + strength * (rng.random(nb) - 0.5)
        q = a * np.repeat(blocks, per)
        if len(q) < n:
            q = np.concatenate([q, np.ones(n - len(q))]) * a
            q[:len(q)//2] = a[:len(q)//2] * np.repeat(blocks[:len(q)//2//per+1], per)[:len(q)//2]
        q = a[:n] * np.tile(blocks, n // nb + 1)[:n]
    elif mode == 8:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c_k = k // 2
        sig = max(1.0, c_k / 2.0)
        kern = np.exp(-((np.arange(k) - c_k) ** 2) / (2.0 * sig ** 2))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode="same")
    else:
        # Mirror + noise
        q = 0.5 * a + 0.5 * a[::-1]
        q *= np.exp(0.05 * rng.standard_normal(n))

    return normalize(np.clip(q, 0.0, None))


def optimize_with_schedule(a, deadline, label="opt"):
    """
    Run LSE-LBFGS with adaptive beta schedule and interspersed subgradient steps.
    """
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    t0 = time.monotonic()
    print("[{}] start: score={:.8f} n={}".format(label, best_sc, n))

    # Beta schedule: start moderate, increase aggressively
    betas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]
    total_time = max(10.0, float(deadline - t0))

    # Allocate time proportionally (more time to higher betas)
    fracs = [0.04, 0.06, 0.08, 0.10, 0.12, 0.18, 
             max(0.10, 0.7 - 0.58), max(0, 1.0 - 0.7 - 0.12)]
    total_frac = sum(fracs)
    fracs = [f / total_frac for f in fracs]

    prev_sc = best_sc

    for i, (beta, frac) in enumerate(zip(betas, fracs)):
        if time.monotonic() >= deadline - 10:
            break
        sub_dead = t0 + (fracs[:i+1].sum()) * total_time
        sub_dead = min(sub_dead, deadline - 15.0)

        # Multiple LSE passes at this beta
        progress = True
        while progress and time.monotonic() < sub_dead - 5:
            rem = sub_dead - time.monotonic()
            if rem < 10:
                break
            est = 0.015 * max(1.0, n / 4096.0) * max(1.0, np.log2(max(16, n)) / 12.0)
            mf = int(min(1000, max(32, (rem * 0.8) / max(est, 1e-4))))
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                improvement = best_sc - sc
                best_sc = sc
                best_a = a2.copy()
                print("[{}] beta={:.0f} score={:.8f} improvement={:.2e}".format(
                    label, beta, best_sc, improvement))
            a = normalize(a2)
            # Check for convergence
            if np.isfinite(sc) and abs(sc - prev_sc) < 1e-10:
                progress = False
            prev_sc = sc

        # Subgradient polish between stages
        if time.monotonic() < sub_dead - 3:
            sg_dead = min(sub_dead, deadline - 15)
            if sg_dead - time.monotonic() > 5:
                sc_sg, a_sg = subgrad_polish(a, max_iters=15, deadline=sg_dead)
                if np.isfinite(sc_sg) and sc_sg < best_sc:
                    best_sc = sc_sg
                    best_a = a_sg.copy()
                    print("[{}] subgrad after beta={:.0f}: {:.8f}".format(label, beta, best_sc))
                a = normalize(a_sg)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 256 and float(p.sum()) > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("found" if parent is not None else "none",
                                        parent.size if parent is not None else 0))

    # Determine target length(s)
    if parent is not None:
        target_len = min(parent.size, 16384)
        if target_len < 2048:
            target_len = 4096
    else:
        target_len = 8192

    print("Target length: {}".format(target_len))

    # Phase 1: Generate and evaluate initializations (30s)
    phase1_dead = t_start + 30.0
    inits = structured_init(target_len, rng, parent, cap=30)

    # Also try parent at different lengths
    if parent is not None and parent.size != target_len:
        inits.append(normalize(resample(parent, target_len)))

    scores = []
    for init in inits:
        if time.monotonic() > phase1_dead:
            break
        sc = exact_score(init)
        scores.append((sc, init))

    scores.sort(key=lambda x: x[0])
    print("Top 5 inits:")
    for i, (sc, _) in enumerate(scores[:5]):
        print("  init[{}]: {:.8f}".format(i, sc))

    best_seq = scores[0][1].copy()
    best_score = scores[0][0]

    # Select top 2 starting points (diverse if possible)
    starts = [scores[0][1].copy()]
    if len(scores) > 1:
        # Pick the best one that's from a different "family"
        for sc, init in scores[1:]:
            # Check if it's sufficiently different
            diff = float(np.max(np.abs(init - best_seq)))
            if diff > 0.1 or len(starts) < 2:
                starts.append(init.copy())
                break
        if len(starts) < 2:
            starts.append(scores[1][1].copy() if len(scores) > 1 else perturb(best_seq, rng))

    # Phase 2: Main optimization on best start (majority of time)
    phase2_dead = deadline - 300.0
    main_budget = max(60.0, phase2_dead - t_start)

    # Allocate 70% to best start, 30% to second
    if len(starts) == 1:
        fracs = [1.0]
    else:
        fracs = [0.70, 0.30]

    cum = 0.0
    for rank, start in enumerate(starts):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, phase2_dead)

        if time.monotonic() > sub_dead - 20:
            continue

        # For secondary start, blend with best
        a_work = start.copy()
        if rank > 0:
            blend = 0.4
            a_work = normalize((1 - blend) * a_work + blend * best_seq)

        sc, seq = optimize_with_schedule(a_work, sub_dead, label="main{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("New best after main{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize(best_seq)

    # Phase 3: Extended high-beta polish on best (120s)
    phase3_dead = min(deadline - 120.0, phase2_dead + 120.0)
    if time.monotonic() < phase3_dead - 10:
        sc, seq = optimize_with_schedule(best_seq, phase3_dead, label="polish1")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Polish1: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts (120s)
    phase4_dead = deadline - 100.0
    n_restarts = 4
    if time.monotonic() < phase4_dead:
        remaining = phase4_dead - time.monotonic()
        per_restart = remaining / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > phase4_dead - 10:
                break
            p = perturb(best_seq, rng, strength=float(rng.uniform(0.05, 0.25)))
            chunk_dead = time.monotonic() + per_restart
            sc, seq = optimize_with_schedule(p, chunk_dead, label="restart{}".format(r + 1))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Restart {}: {:.8f}".format(r + 1, best_score))

    # Phase 5: Final aggressive subgradient (60s)
    phase5_dead = deadline - 40.0
    if time.monotonic() < phase5_dead:
        sc, seq = subgrad_polish(best_seq, max_iters=50, deadline=phase5_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Phase 6: Final LSE with very high beta (40s)
    phase6_dead = deadline - 15.0
    if time.monotonic() < phase6_dead:
        for beta in (200000, 1000000):
            if time.monotonic() > phase6_dead - 5:
                break
            rem = phase6_dead - time.monotonic() - 5
            if rem < 10:
                break
            est = 0.015 * max(1.0, len(best_seq) / 4096.0)
            mf = max(32, int(min(500, rem * 0.8 / max(est, 1e-4))))
            sc, seq = run_lbfgs(best_seq, float(beta), phase6_dead, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
            best_seq = normalize(seq if seq is not None else best_seq)

    # Final verification
    out = np.asarray(best_seq, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    out_list = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out_list), time.monotonic() - t_start))
    return out_list
