    score = 2.0 * n * L / (S * S)
    g = compute_g(x, w)
    dL_dx = 2.0 * g
    grad = (2.0 * n / (S * S)) * dL_dx - (4.0 * n * L / (S ** 3))
    return score, grad


def variance_obj_grad(x):
    """Variance of convolution: sum(c_k - mean(c))^2, and its gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    mu = S * S / (2 * n - 1)
    d = c - mu
    F = float(np.sum(d * d))
    g = compute_g(x, d)
    grad = 4.0 * g
    return F, grad


def run_lbfgs_var(a0, maxiter, maxfun):
    """L-BFGS-B on the variance objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        variance_obj_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)


def run_lbfgs_lse(a0, beta, maxiter, maxfun):
    """L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)
