"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated exact subgradient with adaptive active set,
combined with LSE-smoothed L-BFGS polishing and frequency-domain diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # We need cross_corr: h[j] = sum_k w[k] * a[k-j]
    # This is the same as: h = conv(w_rev, a) shifted appropriately
    # Or: h[j] = sum_k w[k] * a[k-j]
    # Let's use: flip a, convolve with w, then flip and shift
    # Actually: sum_k w[k]*a[k-j] = (w * a_flip)[j + something]
    # Let me think again.
    # a_flip[i] = a[n-1-i]
    # conv(w, a_flip)[m] = sum_i w[i]*a_flip[m-i] = sum_i w[i]*a[n-1-(m-i)] = sum_i w[i]*a[n-1-m+i]
    # We want h[j] = sum_k w[k]*a[k-j]
    # Let k-j = t, so k = t+j: h[j] = sum_t w[t+j]*a[t] = sum_t w[j+t]*a[t]
    # This is cross_correlation(w, a)[j] = sum_t w[t+j]*a[t]
    # In terms of convolution: cross_correlation(w,a)[j] = conv(w_flip, a)[j + len(w) - 1]
    # where w_flip[i] = w[len(w)-1-i]
    
    # Simpler: use the fact that sum_k w[k]*a[k-j] with a[k]=0 for k<0 or k>=n
    # = sum_{k=j}^{n-1} w[k]*a[k-j] (since a[k-j] requires k-j < n, i.e., k < j+n, and k-j >= 0, i.e., k >= j)
    # Also w[k] is defined for k in [0, 2n-2], so we need k in [j, min(2n-2, j+n-1)] = [j, j+n-1] (since j <= n-1)
    
    # Let's just do it directly with FFT:
    # h[j] = sum_k w[k]*a[k-j] 
    # = sum_k w[k+j]*a[k] (substituting k->k+j)
    # This is the cross-correlation of w and a at lag j.
    # Using FFT: cross_corr(w,a)[j] = ifft(fft(w) * conj(fft(a)))[j] (circular)
    # For linear: pad both to length >= len(w) + len(a) - 1
    
    N2 = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, N2)
    Af = np.fft.rfft(a, N2)
    # cross_correlation in linear sense:
    # We want h[j] = sum_k w[k+j]*a[k] for j = 0, ..., n-1
    # This is: h = ifft(conj(fft(w)) * fft(a))[0:n] (circular correlation, but we need linear)
    # Actually for the "cross-correlation" defined as (w ⋆ a)[j] = sum_k w[k+j]*a[k]:
    # This equals conv(w_flip, a)[j + len(w) - 1]
    
    w_flip = w[::-1]
    full = np.fft.irfft(np.fft.rfft(w_flip, N2) * Af, N2)[:len(w) + n - 1]
    return full[len(w) - 1:len(w) - 1 + n]


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _project_simplex(a, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a
    # Simple projection: first clip to 0, then scale
    # More exact: water-filling
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.full(n, target / n)
    return a * (target / S)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _nesterov_subgrad(a, max_iters, deadline, tol=1e-10):
    """
    Nesterov-accelerated subgradient method for exact max objective.
    Step size: Polyak-like, using current best as upper bound.
    """
    a = _norm(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    f_star = best_score  # Lower bound estimate (updated optimistically)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    momentum_max = 0.8
    patience = 0
    max_patience = 50
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        S = float(y.sum())
        if S < 1e-12:
            break
        
        c = _self_conv(y)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        current_score = 2.0 * dmax
        
        # Active set: positions within relative tolerance of max
        rel_tol = max(1e-6, 0.01 * dmax)
        active = d >= dmax - rel_tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(y, w)
        avg_d = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * avg_d / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        # Polyak step size
        # f(y) - f* where f* is our best (optimistic lower bound = 2*n*max_possible)
        # Theoretical lower bound: 2 (from Cauchy-Schwarz on peak)
        # But we can do better: use 2 * n * (sum_a^2 / n) / sum_a^2 = 2 as loose bound
        # Better: use best_score * 0.99 as a soft lower bound
        lb = max(2.0 * (float(n) * (1.0 / n)) / 1.0, best_score * 0.95)
        # Actually the true lower bound is more like 2 * n * (sum a_i^2) / (sum a_i)^2
        # which for uniform is exactly 2. Let's just use a conservative bound.
        gap = current_score - lb
        if gap < 1e-15:
            gap = 1e-10
            
        # Adaptive step: combine Polyak with decreasing step
        step_polyak = gap / (gnorm * gnorm + 1e-15)
        step_decay = 1.0 / (0.1 + it * 0.05)
        step = min(step_polyak, step_decay) * 0.5
        step = max(step, 1e-6)
        
        # Nesterov update
        momentum = min(momentum + 0.01, momentum_max)
        new_y = y - step * grad
        new_y = _project_simplex(new_y)
        
        # Evaluate
        new_score = _score(new_y)
        
        if new_score < best_score:
            best_score = new_score
            best_a = new_y.copy()
            patience = 0
        else:
            patience += 1
            # Reduce momentum if stuck
            if patience > max_patience:
                momentum = max(0.0, momentum - 0.1)
                step = step * 0.5
        
        y = new_y
        a = y  # For next Nesterov step (simplified: no true Nesterov lookahead)
        
        if patience > 200:
            break
    
    return best_score, best_a


def _subgrad_polish(a, max_iters=50, deadline=None):
    """Direct subgradient descent with backtracking line search on exact objective."""
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    it = 0
    
    while it < max_iters:
        if deadline and time.monotonic() > deadline:
            break
        
        S = float(a.sum())
        if S < 1e-12:
            break
        
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        
        # Active set: positions within tight tolerance of max
        active = d >= dmax - 1e-4 * max(dmax, 1e-10)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        best_step = None
        best_new_sc = best_sc
        eta = 1.0 / (gnorm * np.sqrt(it + 1))
        
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001):
            q = a - scale * eta * grad
            q = _project_simplex(q)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
            if sc < best_sc * (1 - 1e-10):
                break
        
        if best_step is None:
            break
        
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        
        a = best_step
        it += 1
    
    return best_sc, best_a


def _lse_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_eval_t, 1e-4))
    mf = max(20, min(mf, maxfun_cap, 6000))
    maxiter = max(15, mf // 4)
    PEN = 1e-3

    def obj(x):
        x = np.maximum(x, 0.0)  # Enforce non-negativity in objective
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0) * np.ones_like(x)
        # Zero out gradient for negative components (boundary)
        g[x < 0] = 0
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _freq_perturb(a, rng, strength=0.1):
    """Apply smooth perturbation in frequency domain, project to non-negative."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    freqs = np.fft.rfftfreq(N) * N
    
    # Low-pass filter the perturbation
    cutoff = rng.uniform(0.02, 0.15) * N / 2
    lowpass = 1.0 / (1.0 + (np.abs(freqs) / max(cutoff, 1.0)) ** 4)
    lowpass[0] = 0.0  # Don't change DC
    
    phase = rng.uniform(-np.pi, np.pi, size=len(A))
    A_pert = A * (1.0 + strength * lowpass * np.exp(1j * phase))
    sig = np.fft.irfft(A_pert, N)[:n]
    sig = np.maximum(sig, 0.0)
    S = sig.sum()
    if S < 1e-12:
        return a.copy()
    return sig / S


def _time_perturb(a, rng, strength=0.1):
    """Apply smooth time-domain perturbation."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(61, max(5, n // 100))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _project_simplex(q)

def _run_trajectory(a0, deadline, label, rng, strategy="combined"):
    """Run a combined optimization trajectory."""

    n = len(a)

    t1 = t0 + 0.3 * (deadline - t0)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = _norm(a_new)

            break

            best_a = a_new.copy()

        a = _norm(a_new)
    
    # Phase 3: Subgradient polish (exact)
    if time.monotonic() < deadline - 10.0:
        sc, a_sg = _subgrad_polish(a, max_iters=100, deadline=deadline - 5.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad polish: {best_sc:.8f}")
    
    # Phase 4: Diversification + re-optimize (if time)

            if pert % 2 == 0:

            else:

            chunk_dl = min(deadline - 30.0, time.monotonic() + 20.0)
            sc_p, a_p = _lse_lbfgs(a_p, 5000, 500, max(2, chunk_dl - time.monotonic()), 200)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()
                print(f"    [{label}] pert{pert}: {best_sc:.8f}")
    
    # Final exact subgradient
    if time.monotonic() < deadline - 5.0:
        sc_f, a_f = _subgrad_polish(best_a, max_iters=50, deadline=deadline - 2.0)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_f.copy()
            print(f"    [{label}] final subgrad: {best_sc:.8f}")
    
    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a
