import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500, 500)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient of L w.r.t. x
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(x[::-1], M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / (S2 * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter, maxfun):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0, 0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'ftol': 1e-15, 'gtol': 1e-12,
                                           'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out

def fienup_iter(a, n_iters, mode='flatten'):
    """Fienup iteration with mode selection."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        
        if mode == 'flatten':
            alpha = 0.3 + 0.5 * (it / n_iters)
            P_blend = (1 - alpha) * P + alpha * P_mean
        else:  # 'quantile'
            q = 0.8 + 0.15 * (it / n_iters)
            thresh = np.quantile(P, q)
            P_blend = P.copy()
            P_blend[P > thresh] = thresh
        
        P_blend = np.maximum(P_blend, 1e-20)
        ratio = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_subgradient(a, n_iters, top_k=10):
    """Subgradient method targeting top peaks."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / M_val
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w * a[j_start:j_end+1][::-1]
        
        S2 = S * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / (S2 * S))
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        step = min(0.05, (sc - 2.0*n/(2.0*n-1)) / (gn*gn + 1e-30))
        step = max(step, 1e-8)
        
        a -= step * grad
        a = np.clip(a, 0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

    n = len(a)

        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc

                if S_tmp < 1e-12:

        sc = compute_score(a)
        if sc < best_sc:

def frequency_phase_randomize(a, rng, n_tries=20):
    """Randomize phase in frequency domain to escape local minima."""
    n = len(a)

        S = a_new.sum()
        if S < 1e-12:

        a_new = a_new / S
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
    
    return best_a

def interpolate_to(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, a)

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500, 500)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(x[::-1], M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def flat_target_score_grad(x, beta=None):
    """
    L2 distance from flat autocorrelation target, smoothed by LSE on the squared error.
    Minimizes sum_k (r_k - target)^2 approximately, which encourages uniform autocorrelation.
    """
    n = len(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = x / S
    r = fast_conv(a)
    target = 1.0 / (2 * n - 1)
    e = r - target
    # Score: proportional to ||e||_2^2 (use LSE for smoothness of the max of e^2)
    e2 = e * e
    e2_max = float(e2.max())
    if beta is None:
        beta = 50.0
    scaled = np.clip(beta * (e2 - e2_max), -500, 500)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = e2_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L  # proportional to original score when sum=1
    # Gradient: 4 * sum_k p_k * e_k * dc_k/da_j
    # dc_k/da_j = 2 * a_{k-j}
    # So grad_j = 4 * sum_k p_k * e_k * a_{k-j} = 4 * (cross_corr of pe with a at lag j)
    pe = p * e
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PE_A = np.fft.rfft(pe, M2_fft)
    A_REV = np.fft.rfft(a[::-1], M2_fft)
    conv_res = np.fft.irfft(PE_A * A_REV, M2_fft)[:M2]
    g = 4.0 * conv_res[n - 1:2 * n - 1]
    # Account for the normalization: x = S * a, so dx_j = S * da_j + a_j * dS
    # For simplicity, use the gradient w.r.t. a and scale
    S2 = S * S
    # d(score)/dx_j = d(score)/da_j * da_j/dx_j + d(score)/dS * dS/dx_j
    # da_j/dx_j = 1/S - a_j/S^2 (from a = x/S)
    # dS/dx_j = 1
    # This is complex; use a simpler approximation:
    grad = g / S - np.sum(g * a) / S
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, method='L-BFGS-B'):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(obj_and_grad, a0, method=method, jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out
