```python
"""
CVXPY-based LP active set optimizer for flat-autocorrelation non-negative sequences.
Uses iterative LP with row generation (active set) to solve the linearized subproblem,
combined with LSE-based L-BFGS polishing and peak-targeted local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import minimize
import cvxpy as cp
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_rows(a0, indices):
    """Build sparse rows of convolution matrix for given row indices.
    A[k,j] = a0[k-j] if 0 <= k-j < n, else 0.
    Returns (rows, cols, vals) for CSR.
    """
    n = len(a0)
    row_list = []
    col_list = []
    val_list = []
    for k in indices:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_list.append(k)
            col_list.append(j)
            val_list.append(a0[k - j])
    return row_list, col_list, val_list


def solve_lp_subproblem(a0, active_indices, solver='PDLP', verbose=False):
    """Solve: min t s.t. conv(a0, x)[k] <= t for k in active_indices, sum(x)=1, x>=0"""
    n = len(a0)
    x = cp.Variable(n)
    t = cp.Variable()
    
    rows, cols, vals = build_conv_rows(a0, active_indices)
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    
    constraints = [
        A @ x <= t,
        cp.sum(x) == 1,
        x >= 0
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, verbose=verbose)
    except Exception:
        try:
            prob.solve(solver='CLARABEL', verbose=verbose)
        except Exception:
            try:
                prob.solve(solver='GLPK', verbose=verbose)
            except Exception:
                return None, None
    
    if x.value is None:
        return None, None
    
    x_val = np.clip(x.value, 0, 1e10)
    S = float(x_val.sum())
    if S < 1e-12:
        return None, None
    x_val = x_val / S
    return x_val, float(t.value) if t.value is not None else float('inf')


def lp_active_set_iteration(a, max_active=200, solver='PDLP', time_limit=None, t_start=None):
    """One iteration of LP active set method.
    Returns (new_a, new_score, improved).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, compute_score(a), False
    a = a / S
    
    # Compute current max
    conv_curr = self_conv(a)
    M_curr = float(conv_curr.max())
    
    # Initial active set: top constraints
    sorted_idx = np.argsort(conv_curr)[::-1]
    active = sorted_idx[:max_active]
    
    x_best = None
    t_best = None
    
    for gen in range(30):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > time_limit:
                break
        
        x, t_val = solve_lp_subproblem(a, active, solver=solver)
        if x is None:
            break
        
        # Check if solution violates any non-active constraints
        conv_check = cross_conv(a, x)
        max_active_val = float(conv_check[active].max())
        max_all_val = float(conv_check.max())
        
        if max_all_val > t_val + 1e-6:
            # Find violated constraints
            violated = np.where(conv_check > t_val + 1e-6)[0]
            # Merge into active set
            active = np.unique(np.concatenate([active, violated]))
            # Keep top max_active
            if len(active) > max_active:
                vals_at_active = conv_check[active]
                keep_order = np.argsort(vals_at_active)[::-1][:max_active]
                active = active[keep_order]
            continue
        
        # Solution is feasible for all constraints
        if x_best is None or t_val < t_best:
            x_best = x
            t_best = t_val
        break
    
    if x_best is None:
        return a, compute_score(a), False
    
    # Try blending
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
        a_try = (1.0 - alpha) * a + alpha * x_best
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            continue
        a_try = a_try / St
        sc = compute_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try
    
    return best_a, best_sc, best_sc < compute_score(a)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def peak_local_search(a, n_steps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.03, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.5)
        for mult in [0.3, 0.5, 0.7, 1.0, 1.5]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

        if it % 20 == 0 and rng.random() < 0.3:
            top_k = len(c) // 10
            candidates = np.argsort(c)[::-1][:top_k]
            k_star = candidates[rng.integers(len(candidates))]

    return best_a


def error_reduction(a, n_iters=100, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a
```

```python
"""
CVXPY LP active-set optimizer for flat-autocorrelation non-negative sequences.
Main innovation: uses a real LP solver with row generation (active set) to solve
the linearized subproblem exactly, rather than FISTA approximation. Combined with
LSE-based L-BFGS polishing and peak-targeted local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import minimize
import cvxpy as cp
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix_sparse(a0, indices):
    n = len(a0)
    k_arr = np.asarray(indices, dtype=np.int64)
    total_nnz = 0
    for k in k_arr:
        cs = max(0, int(k) - n + 1)
        ce = min(int(k), n - 1)
        total_nnz += ce - cs + 1
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    vals = np.empty(total_nnz, dtype=np.float64)
    offset = 0
    for k in k_arr:
        ki = int(k)
        cs = max(0, ki - n + 1)
        ce = min(ki, n - 1)
        ln = ce - cs + 1
        rows[offset:offset + ln] = ki
        cols[offset:offset + ln] = np.arange(cs, ce + 1)
        vals[offset:offset + ln] = a0[ki - ce:ki - cs + 1][::-1]
        offset += ln
    return csr_matrix((vals, (rows, cols)), shape=(len(k_arr), n))


def solve_lp_subproblem(a0, active_indices, solver_name, t_upper=None):
    n = len(a0)
    x = cp.Variable(n)
    t = cp.Variable()
    A = build_conv_matrix_sparse(a0, active_indices)
    constraints = [
        A @ x <= t,
        cp.sum(x) == 1,
        x >= 0
    ]
    if t_upper is not None:
        constraints.append(t <= t_upper)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        try:
            prob.solve(solver='CLARABEL', verbose=False)
        except Exception:
            try:
                prob.solve(solver='SCS', verbose=False)
            except Exception:
                return None, None
    if x.value is None:
        return None, None
    x_val = np.clip(x.value.ravel(), 0, 1e10)
    S = float(x_val.sum())
    if S < 1e-12:
        return None, None
    x_val = x_val / S
    t_val = float(t.value) if t.value is not None else float('inf')
    return x_val, t_val


def lp_active_set_iteration(a, max_active=200, solver_name='CLARABEL', time_limit=None, t_start=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, compute_score(a), False
    a = a / S
    conv_curr = self_conv(a)
    M_curr = float(conv_curr.max())
    sorted_idx = np.argsort(conv_curr)[::-1]
    active = sorted_idx[:max_active].copy()
    x_best = None
    t_best = None
    for gen in range(40):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > time_limit:
                break
        t_upper = M_curr * 1.5
        x, t_val = solve_lp_subproblem(a, active, solver_name, t_upper=t_upper)
        if x is None:
            break
        conv_check = cross_conv(a, x)
        max_all_val = float(conv_check.max())
        if max_all_val > t_val + 1e-6:
            violated = np.where(conv_check > t_val + 1e-6)[0]
            active = np.unique(np.concatenate([active, violated]))
            if len(active) > max_active:
                vals_at = conv_check[active]
                keep = np.argsort(vals_at)[::-1][:max_active]
                active = active[keep]
            continue
        if x_best is None or t_val < t_best:
            x_best = x
            t_best = t_val
        break
    if x_best is None:
        return a, compute_score(a), False
    best_a = a.copy()
    best_sc = compute_score(a)
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        a_try = (1.0 - alpha) * a + alpha * x_best
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            continue
        a_try = a_try / St
        sc = compute_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
    sc_orig = compute_score(a)
    return best_a, best_sc, best_sc < sc_orig - 1e-10


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def peak_local_search(a, n_steps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        base_step = min(0.03, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.5)
        for mult in [0.3, 0.5, 0.7, 1.0, 1.5]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break
        if it % 20 == 0 and rng.random() < 0.3:
            top_k = max(10, len(c) // 10)
            candidates = np.argsort(c)[::-1][:top_k]
            k_star = candidates[rng.integers(len(candidates))]
    return best_a


def error_reduction(a, n_iters=80, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a

    return 'CLARABEL'
```
