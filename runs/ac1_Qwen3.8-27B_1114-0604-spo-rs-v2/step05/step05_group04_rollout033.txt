def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup(a, n_iters=300, gamma=0.8, top_frac_init=0.1, top_frac_end=0.5,
           gate_alpha=30.0, momentum=0.3):
    """Fienup with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Momentum buffer
    a_prev = a.copy()

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        
        # Soft non-negativity
        if gate_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-gate_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Momentum

            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())

                a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

"""
Convex-structure-aware optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Exploits the convexity of max(conv(a,a)) over the simplex to use:
1. Fienup spectral shaping for diverse initializations
2. L-BFGS with LSE smoothing (adaptive beta scheduling)
3. Nesterov-accelerated projected gradient (exploits convexity)
4. Multi-restart with structured perturbations
"""
import numpy as np
from scipy.optimize import minimize
import time

# Global deadline for safety
_global_deadline = 0


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def fienup_shaped(a, n_iters=300, gamma=0.8, tf_init=0.1, tf_end=0.5,
                  gate_alpha=30.0, momentum=0.0):
    """Fienup with spectral compression and optional momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    a_mom = a.copy()

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        tf = tf_init + (tf_end - tf_init) * 0.5 * (1.0 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - tf)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 0.001)) * gamma
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        if gate_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-np.clip(gate_alpha * a_new, -500, 500)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        if momentum > 0.0:
            a_new = (1.0 - momentum) * a_new + momentum * a_mom
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new > 1e-12:
                a_new = a_new / S_new
        a_mom = a_new.copy()
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat(a, n_iters=200, alpha_init=0.2, alpha_end=0.8):
    """Spectral equalization: relax towards flat spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_init + (alpha_end - alpha_init) * (it / max(1, n_iters))
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    np.clip(scaled, -700, 700, out=scaled)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    func_calls = [0]
    call_limit = maxfun

    def obj_wrapper(x, b):
        func_calls[0] += 1
        if deadline is not None and time.monotonic() > deadline:
            raise InterruptedError("timed out")
        return lse_objective_and_grad(x, b)

    try:
        res = minimize(
            lambda x: obj_wrapper(x, beta), a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': call_limit, 'maxls': 40}
        )
    except (InterruptedError, Exception):
        res = None

    if res is not None and hasattr(res, 'x'):
        a_out = np.clip(res.x, 0.0, 1000.0)
    else:
        a_out = a0.copy()

    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    return compute_score(a_out), a_out


def nesterov_projected(a, n_iters=10000, beta=200, lr=0.005,
                       lr_decay=0.9999, deadline=None):
    """Nesterov-accelerated projected gradient with LSE smoothing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_mom = 1.0
    step = lr

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        _, grad_y = lse_objective_and_grad(a, beta)
        a_new = a - step * grad_y
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        if it > 0:
            a_new = a_new + (t_mom - 1.0) / t_new * (a_new - a)
            a_new = project_simplex(a_new)
        t_mom = t_new

        a = a_new
        step *= lr_decay

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def subgradient_topm(a, n_iters=5000, top_m=20, step_init=0.02,
                     deadline=None):
    """Top-m subgradient with Adaptive step (Polyak-like)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.04)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999

    return best_a


def build_inits(n, rng, parent):
    """Build diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.05, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-6.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 50.0))
    # Tapered with oscillation
    taper = np.sin(np.pi * x)
    osc = 1.0 + 0.3 * np.cos(4.0 * np.pi * x)
    inits.append(np.maximum(taper * osc + 0.01, 0.001))
    # Bell-shaped with ringing
    bell = np.exp(-8.0 * (2.0 * x - 1.0) ** 2)
    ring = 1.0 + 0.2 * np.cos(6.0 * np.pi * x) * np.exp(-2.0 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(bell * ring + 0.02, 0.001))
    return inits

                update_best(sc_alt2, a_alt2, f"ALTLEN-LBFGS n={alt_n}")

    # ==================================================================
    # Final validation
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
