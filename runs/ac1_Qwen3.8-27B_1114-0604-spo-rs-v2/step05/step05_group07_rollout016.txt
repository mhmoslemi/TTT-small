Improved multi-phase optimizer combining:
1. Adaptive LSE-smoothed L-BFGS with dynamic beta scheduling
2. Active-set-aware subgradient with exact line search
3. Spectral-domain perturbation for diversity
4. High-precision final polishing

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 1000.0

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta, active_weights=None):
    """LSE-smoothed score with optional active-set weighting."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.maximum(scaled, -50)
    exp_scaled = np.exp(scaled)
    
    if active_weights is not None:
        exp_scaled = exp_scaled * active_weights
    
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # dL/da[i] = sum_k p_k * d(conv)_k/da_i = sum_k p_k * a[k-i] (reversed)
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000, active_weights=None):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

    return compute_score(a_out), a_out


def energy_minimize(a0, n_iters=10000):
    """Minimize sum(c_k^2) using gradient descent with backtracking line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = 0.5 / n
    for it in range(n_iters):
        grad = energy_grad(a)
        grad = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        step = lr
        accepted = False
        for _ in range(30):
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            sc_old = compute_score(a)
            if sc_new < sc_old:
                a = a_new
                accepted = True
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                break
            step *= 0.5
        if not accepted:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            lr = min(lr * 1.05, 1.0 / n)
    return best_a


def spectral_flaten(a, n_iters=200):
    """Iteratively flatten the magnitude spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-20)
        phase = A / mag
        target = mag.max() / (1.0 + 0.5 * np.exp(-0.1 * (np.arange(len(mag)) / len(mag))))
        alpha = 0.005 + 0.015 * (it / max(1, n_iters))
        new_mag = (1.0 - alpha) * mag + alpha * target
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still accept for diversity, but track best
    return best_a


def active_subgrad(a, n_iters=5000, n_active_budget=None):
    """FFT-accelerated subgradient with adaptive activation and exact line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.05
    for it in range(n_iters):
        c = fast_conv(a)
        Sa = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (Sa * Sa)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Adaptive activation: increasing selectivity over time
        beta_act = 5.0 + 80.0 * (it / n_iters)
        w = np.exp(np.maximum(beta_act * (c - c_max), -60))
        wsum = float(w.sum())
        if wsum < 1e-30:
            break
        w = w / wsum
        # Subgradient via FFT
        rev_a = a[::-1]
        M2 = len(w) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        W = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_result = np.fft.irfft(W * RA, M2_fft)[:M2]
        sg = 2.0 * conv_result[n - 1:n - 1 + n]
        S2 = Sa * Sa
        S3 = S2 * Sa
        L = float(np.dot(w, c))
        grad = (2.0 * n) * (sg / S2 - 2.0 * L / S3)
        grad = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        # Exact line search (bisection on log scale)
        best_alpha = 0.0
        best_val = sc
        lo, hi = 0.0, step * 50.0 / (gnorm + 1e-30)
        for _ in range(25):
            mid = (lo + hi) / 2.0
            a_trial = a - mid * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            St = float(a_trial.sum())
            if St < 1e-12:
                hi = mid
                continue
            a_trial = a_trial / St
            val = compute_score(a_trial)
            if val < best_val:
                best_val = val
                best_alpha = mid
            # Check left and right
            a_left = a - (mid * 0.8) * grad
            a_left = np.clip(a_left, 0.0, 1000.0)
            Sl = float(a_left.sum())
            a_right = a - (mid * 1.25) * grad
            a_right = np.clip(a_right, 0.0, 1000.0)
            Sr = float(a_right.sum())
            if Sl > 1e-12 and Sr > 1e-12:
                vl = compute_score(a_left / Sl)
                vr = compute_score(a_right / Sr)
                if vl < best_val:
                    hi = mid
                elif vr < best_val:
                    lo = mid
                else:
                    break
            else:
                hi = mid
        if best_alpha > 0:
            a = a - best_alpha * grad
            a = np.clip(a, 0.0, 1000.0)
            S_new = float(a.sum())
            if S_new > 1e-12:
                a = a / S_new
        step *= 0.998
    return best_a


def nesterov_accelerated(a, n_iters=3000):
    """Nesterov-accelerated subgradient on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        # Subgradient of the active constraint
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * y[k_star - idx]
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)
        grad = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        step = (sc - 1.0) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = float(a.sum())
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    return best_a


def coordinate_descent_peaks(a, n_rounds=20):
    """Targeted coordinate descent: reduce peak-contributing elements and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total = len(contrib)
        n_top = min(100, max(5, total // 5))
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local
        bottom_local = np.argsort(contrib)[:max(1, total - n_top)]
        bottom_global = j_start + bottom_local
        improved = False
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            best_frac = 1.0
            best_local = best_sc
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 0.95]:
                new_val = a_old_j * frac
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    a[j] = new_val
                    per = reduction / len(bottom_global)
                    a[bottom_global] += per
                    S_t = float(a.sum())
                    if S_t > 1e-12:
                        a = a / S_t
                    sc = compute_score(a)
                    if sc < best_local:
                        best_local = sc
                        best_frac = frac
                    a[bottom_global] -= per
                a[j] = a_old_j
            if best_frac < 1.0:
                reduction = a_old_j * (1.0 - best_frac)
                a[j] = a_old_j * best_frac
                per = reduction / len(bottom_global)
                a[bottom_global] += per
                S_t = float(a.sum())
                if S_t > 1e-12:
                    a = a / S_t
                improved = True
        if not improved:
            break
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def perturbed(a, rng, scale=0.03):
    """Generate a perturbed variant for ILS."""
    n = len(a)
    strat = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n
    if strat == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strat == 1:
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strat == 2:
        n_blocks = 5 + int(rng.integers(0, 10))
        bs = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // bs)):
            s = b * bs
            e = min(s + bs, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strat == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 3)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = float(a_p.sum())
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: energy warm-start -> adaptive LSE L-BFGS -> active subgradient -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    # Start from parent
    if parent is not None:
        pn = parent / parent.sum()
        sc_p = compute_score(pn)
        print(f"Parent: n={len(pn)}, score={sc_p:.10f}")
        report(sc_p, pn, "parent")
        n_work = len(parent)
    else:
        n_work = 4096
        best_a = np.ones(n_work) * 0.5
        print(f"No parent found, using n={n_work}")

    print(f"Working length: {n_work}")

    # ============ Phase 1: Energy minimization (15% time) ============
    p1_deadline = t_start + 0.15 * (total_budget - 15.0)
    print(f"\n--- Phase 1: Energy Minimization ---")
    a_energy = energy_minimize(best_a, n_iters=5000)
    sc = compute_score(a_energy)
    report(sc, a_energy, "energy")

    # Spectral flattening
    if time.monotonic() < p1_deadline:
        a_spec = spectral_flaten(best_a, n_iters=100)
        sc = compute_score(a_spec)
        report(sc, a_spec, "spectral")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============ Phase 2: Adaptive LSE L-BFGS (45% time) ============
    p2_deadline = t_start + 0.60 * (total_budget - 15.0)
    print(f"\n--- Phase 2: LSE L-BFGS ---")

    beta_schedule = [30, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    prev_sc = best_score
    for beta in beta_schedule:
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 20:
            break
        maxiter = int(min(50000, max(2000, remaining * 30)))
        maxfun = int(min(200000, max(5000, remaining * 100)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        # If improvement stalled, try perturbed restart
        if sc > prev_sc * 0.9999 and beta >= 200:
            a_pert = perturbed(best_a, rng, scale=0.01)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")
        prev_sc = best_score

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============ Phase 3: Active-set subgradient + Nesterov (25% time) ============
    p3_deadline = t_start + 0.85 * (total_budget - 15.0)
    print(f"\n--- Phase 3: Subgradient Methods ---")

    # Active-set subgradient
    if time.monotonic() < p3_deadline:
        a_as = active_subgrad(best_a, n_iters=5000)
        sc = compute_score(a_as)
        report(sc, a_as, "active_sg")

    # Nesterov
    if time.monotonic() < p3_deadline:
        a_nes = nesterov_accelerated(best_a, n_iters=3000)
        sc = compute_score(a_nes)
        report(sc, a_nes, "nesterov")

    # Coordinate descent on peaks
    if time.monotonic() < p3_deadline:
        a_cd = coordinate_descent_peaks(best_a, n_rounds=15)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_descent")

    # ILS: perturb + quick L-BFGS
    n_trials = 8
    for trial in range(n_trials):
        if time.monotonic() > p3_deadline:
            break
        remaining = p3_deadline - time.monotonic()
        if remaining < 30:
            break
        scale = 0.01 + 0.03 * rng.random()
        a_p = perturbed(best_a, rng, scale=scale)
        beta = 500 + int(2000 * rng.random())
        maxiter = int(min(20000, max(1000, remaining * 10)))
        maxfun = int(min(50000, max(2000, remaining * 40)))
        sc, a_opt = run_lbfgs(a_p, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} b={beta}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============ Phase 4: Final polish (15% time) ============
    p4_deadline = deadline
    print(f"\n--- Phase 4: Final Polish ---")

    # Very high beta L-BFGS
    for beta in [50000, 100000, 200000, 500000]:
        if time.monotonic() > p4_deadline - 10:
            break
        remaining = p4_deadline - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(30000, max(2000, remaining * 20)))
        maxfun = int(min(100000, max(5000, remaining * 60)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final b={beta}")

    # Final coordinate descent
    if time.monotonic() < p4_deadline - 10:
        a_cd2 = coordinate_descent_peaks(best_a, n_rounds=20)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "final_coord")

    # Final Nesterov
    if time.monotonic() < p4_deadline - 5:
        a_nes2 = nesterov_accelerated(best_a, n_iters=2000)
        sc = compute_score(a_nes2)
        report(sc, a_nes2, "final_nesterov")

    # ============ Final output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
