def fast_conv(a):
    """FFT-based fast linear convolution."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M.bit_length() - 1)
    if M_fft < M + 1:
        M_fft *= 2
    A = np.fft.rfft(a, M_fft)
    conv = np.fft.irfft(A ** 2, M_fft)[:M]
    return conv
