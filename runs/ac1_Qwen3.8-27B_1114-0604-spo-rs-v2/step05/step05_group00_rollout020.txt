    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    return compute_score(a_out), a_out

def lp_direction(f, M, active_idx, n_max_constraints=500):
    """
    Solve the LP to find direction g that maximizes sum(g)
    subject to (f*g)[k] <= M for k in active_idx, g >= 0.
    
    Returns the optimal g (unscaled).
    """
    n = len(f)
    
    # Build constraint matrix A_ub where A_ub @ g <= M * ones
    # A_ub[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # i.e., (f*g)[k] = sum_j f[j] * g[k-j] = sum_j A_ub[k,j] * g[j]
    
    K = len(active_idx)
    if K > n_max_constraints:
        # Take the most "active" (highest conv value) constraints
        K = n_max_constraints
    
    # Build the matrix row by row
    A_ub = np.zeros((K, n))
    for i, k in enumerate(active_idx[:K]):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # A_ub[i, j] = f[k-j] for j in [j_start, j_end]
            js = np.arange(j_start, j_end + 1)
            A_ub[i, js] = f[k - js]
    
    b_ub = np.full(K, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x, 0)
        return g
    else:
        return None

def lp_iteration(f, M, t=0.1, n_active_frac=0.1):
    """
    One LP-based iteration:
    1. Find active constraints (top fraction of conv values).
    2. Solve LP for direction g.
    3. Update f <- (1-t)*f + t*g (both normalized to same sum).
    """
    n = len(f)
    conv = fast_conv(f)
    S = float(f.sum())
    
    # Find active constraints
    n_active = max(10, int(n_active_frac * (2 * n - 1)))
    active_idx = np.argsort(conv)[::-1][:n_active]
    
    g = lp_direction(f, M, active_idx, n_max_constraints=200)
    if g is None or g.sum() < 1e-12:
        return f, False
    
    # Normalize g to have same sum as f
    S_g = float(g.sum())
    g = g * (S / S_g)
    
    # Update
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    S_new = f_new.sum()
    if S_new < 1e-12:
        return f, False
    f_new = f_new / S_new
    
    return f_new, True

def fienup_flatten(a, n_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.2 + 0.6 * progress
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def peak_shave_grad(a, n_iters=3000, top_k=10, step_init=0.01):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        avg_w = float(np.mean([c[int(k)] / (M_val + 1e-30) for k in top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, max((sc - 1.001) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)
        step = max(step, 1e-10)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9995
    return best_a

def coord_descent(a, n_sweeps=5, n_top=50):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.85, 0.9, 1.1, 1.3, 1.5]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def interpolate_sequence(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)

to find principled descent directions for the minimax objective, which avoids the
smoothing approximation error inherent in LSE-based gradients. Combined with
LSE-polished L-BFGS and targeted coordinate descent for final squeezing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    return compute_score(a_out), a_out


def build_lp_matrix(f, active_idx, n_max):
    """Build the LP constraint matrix for active convolution constraints."""
    n = len(f)
    K = min(len(active_idx), n_max)
    A_ub = np.zeros((K, n))
    for i in range(K):
        k = int(active_idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            A_ub[i, js] = f[k - js]
    return A_ub


def solve_lp_direction(f, M, active_idx, n_max_constraints=150):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for k in active_idx, g >= 0.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    A_ub = build_lp_matrix(f, active_idx, n_max_constraints)
    b_ub = np.full(A_ub.shape[0], M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_full(f, M, n_max_constraints=500):
    """Solve LP using top n_max_constraints most active convolution positions."""
    n = len(f)
    conv = fast_conv(f)
    active_idx = np.argsort(conv)[::-1][:n_max_constraints]
    return solve_lp_direction(f, M, active_idx, n_max_constraints)


def lp_iteration_step(f, t_step, n_active=100, n_max_constraints=150):
    """
    One LP-based iteration. Returns (new_f, success, score).
    f should be normalized to sum=1.
    """
    n = len(f)
    conv = fast_conv(f)
    M = float(conv.max())
    sc = 2.0 * n * M  # since sum=1

    # Find active constraints
    K = min(n_max_constraints, 2 * n - 1)
    active_idx = np.argsort(conv)[::-1][:K]

    g = solve_lp_direction(f, M, active_idx, n_max_constraints)
    if g is None:
        return f, False, sc

    S_g = float(g.sum())
    if S_g < 1e-12:
        return f, False, sc

    # Normalize g to sum 1
    g = g / S_g

    # Blend
    f_new = (1.0 - t_step) * f + t_step * g
    f_new = np.maximum(f_new, 0.0)
    S_new = float(f_new.sum())
    if S_new < 1e-12:
        return f, False, sc
    f_new = f_new / S_new

    return f_new, True, sc


def fienup_refine(a, n_iters=150):
    """Fienup-style spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        # Gradually increase flattening pressure
        alpha = 0.3 + 0.5 * (it / n_iters)
        P_target = P * (1.0 - alpha) + P_mean * alpha
        P_target = np.maximum(P_target, 1e-20)
        ratio = np.sqrt(P_target / (P + 1e-30))
        ratio = np.clip(ratio, 0.5, 2.0)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def subgradient_peak_reduce(a, n_iters=2000, top_k=8, step_init=0.008):
    """Subgradient descent targeting top peaks with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Top-K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        avg_w = float(np.mean(c[top_idx] / (M_val + 1e-30)))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step
        step = min(step, max((sc - 1.001) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.02)
        step = max(step, 1e-10)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9997
    return best_a


def coord_descent_peak(a, n_sweeps=8, n_top=60):
    """Coordinate descent on elements contributing to the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        if contribs.sum() < 1e-15:
            break
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            factors = [0.3, 0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.5]
            best_val = a_old_j
            best_local = cur_sc
            for f in factors:
                cv = max(0.0, min(1000.0, a_old_j * f))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-guided search combining active-constraint LP directions,
    LSE-smoothed L-BFGS, subgradient peak reduction, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    global best_score, best_a
    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(np.asarray(a, dtype=np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
        a_init = parent.astype(np.float64).copy()
    else:
        n0 = 4096
        a_init = np.ones(n0) / n0
        sc_p = compute_score(a_init)
        update(sc_p, a_init, "uniform")
        print(f"No parent. Uniform n={n0}, score={sc_p:.8f}")

    # Normalize working copy
    a_work = a_init.copy()
    S = a_work.sum()
    if S < 1e-12:
        a_work = np.ones(len(a_init)) / len(a_init)
        S = a_work.sum()
    a_work = a_work / S

    # ==================================================================
    # Phase 1: LP-based iterations (45% of time)
    # ==================================================================
    phase1_end = t_start + 0.45 * (total_budget - 40.0)
    print(f"\n=== Phase 1: LP iterations (until t={phase1_end - t_start:.0f}s) ===")

    t_step = 0.03
    n_iter = 0
    stagnation = 0

    while time.monotonic() < phase1_end - 30:
        t0 = time.monotonic()
        conv = fast_conv(a_work)
        M = float(conv.max())
        sc_cur = 2.0 * len(a_work) * M  # sum=1

        # Get active constraints (top 150)
        n_act = min(150, 2 * len(a_work) - 1)
        active_idx = np.argsort(conv)[::-1][:n_act]

        g = solve_lp_direction(a_work, M, active_idx, n_max_constraints=100)
        lp_time = time.monotonic() - t0

        if g is None or g.sum() < 1e-12:
            stagnation += 1
            if stagnation > 10:
                # Perturb and reset
                a_work = a_work * (1.0 + 0.005 * rng.standard_normal(len(a_work)))
                a_work = np.maximum(a_work, 0)
                a_work = a_work / a_work.sum()
                stagnation = 0
            continue

        S_g = float(g.sum())
        g_norm = g / S_g

        f_new = (1.0 - t_step) * a_work + t_step * g_norm
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            break
        f_new = f_new / S_new

        sc_new = compute_score(f_new)
        n_iter += 1

        if n_iter % 10 == 0 or sc_new < best_score:
            print(f"  LP[{n_iter}]: sc={sc_new:.8f}, t={t_step:.4f}, "
                  f"lp_time={lp_time:.1f}s, M={M:.4f}")

        if sc_new < sc_cur:
            a_work = f_new
            update(sc_new, a_work, f"LP-{n_iter}")
            stagnation = 0
            # Gradually increase step
            t_step = min(0.08, t_step * 1.05)
        else:
            stagnation += 1
            t_step = max(0.005, t_step * 0.7)
            if stagnation > 15:
                # Escape: perturb + Fienup
                a_work = a_work * (1.0 + 0.01 * rng.standard_normal(len(a_work)))
                a_work = np.maximum(a_work, 0)
                a_work = a_work / a_work.sum()
                a_f = fienup_refine(a_work, n_iters=30)
                sc_f = compute_score(a_f)
                if sc_f < compute_score(a_work):
                    a_work = a_f
                t_step = 0.02
                stagnation = 0

    # ==================================================================
    # Phase 2: LSE-smoothed L-BFGS refinement (30% of time)
    # ==================================================================
    phase2_end = t_start + 0.75 * (total_budget - 40.0)
    print(f"\n=== Phase 2: L-BFGS refinement ===")

    a_work = best_a.astype(np.float64).copy()
    betas = [100, 300, 1000, 3000, 8000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > phase2_end - 20:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        maxiter = min(8000, max(200, int(remaining * 15)))
        maxfun = min(80000, max(1000, int(remaining * 300)))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        if update(sc, a_opt, f"LBFGS b={beta}"):
            a_work = a_opt.copy()

    # ==================================================================
    # Phase 3: Subgradient + LP refinement (10% of time)
    # ==================================================================
    phase3_end = t_start + 0.88 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Subgradient + LP ===")

    if time.monotonic() < phase3_end - 60:
        a_sg = subgradient_peak_reduce(best_a, n_iters=2000, top_k=8, step_init=0.005)
        sc = compute_score(a_sg)
        update(sc, a_sg, "SUBGRAD")

    # Another round of LP from the best
    if time.monotonic() < phase3_end - 30:
        a_work = best_a.astype(np.float64).copy()
        S = a_work.sum()
        a_work = a_work / S
        n_iter2 = 0
        while time.monotonic() < phase3_end - 10:
            conv = fast_conv(a_work)
            M = float(conv.max())
            active_idx = np.argsort(conv)[::-1][:100]
            g = solve_lp_direction(a_work, M, active_idx, n_max_constraints=80)
            if g is None or g.sum() < 1e-12:
                break
            g = g / g.sum()
            t_s = 0.01
            f_new = (1 - t_s) * a_work + t_s * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            sc_new = compute_score(f_new)
            if sc_new < compute_score(a_work):
                a_work = f_new
                update(sc_new, a_work, f"LP2-{n_iter2}")
            n_iter2 += 1
            if n_iter2 > 50:
                break

    # ==================================================================
    # Phase 4: Coordinate descent polish (5% of time)
    # ==================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Coordinate descent polish ===")

    if time.monotonic() < phase4_end - 30:
        a_cd = coord_descent_peak(best_a, n_sweeps=10, n_top=50)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD")

    if time.monotonic() < phase4_end - 20:
        # One more L-BFGS pass with high beta
        remaining = phase4_end - 20 - time.monotonic()
        if remaining > 15:
            maxiter = int(remaining * 15)
            sc, a_opt = run_lbfgs(best_a, 30000, maxiter=maxiter, maxfun=maxiter * 10)
            update(sc, a_opt, "FINAL-LBFGS")

    if time.monotonic() < phase4_end - 10:
        a_cd = coord_descent_peak(best_a, n_sweeps=5, n_top=80)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
