"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) GS-Fienup for initialization, (2) Top-K LSE with L-BFGS,
(3) Linearized Programming (active-set LP) for structural improvements,
(4) Peak surgery for final local optimization.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def gs_fienup(a, n_iters=500, alpha_init=0.7, alpha_final=0.1):

    M = 1 << (2 * n - 1).bit_length()

    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)

        a = a_new
        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    return best_a

    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(c))
    c_top = c[top_idx]
    c_max = float(c_top.max())
    scaled = beta * (c_top - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via cross-correlation
    q = np.zeros(2*n-1)
    q[top_idx] = p

    corr = np.fft.irfft(Q * np.conj(Af), N)[:n]
    dL_da = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)

def run_topk_lbfgs(a0, beta, top_k, maxiter=5000, maxfun=100000):

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    bounds = [(0.0, 1000.0)] * n

        topk_lse_grad, a0, args=(beta, top_k),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}

    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:

    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def subgradient_linesearch(a, top_k=10, n_trials=20):
    """Subgradient step followed by line search."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    c_max = float(c.max())
    
    # Subgradient of max(c) (averaged over top-k for smoothness)
    weights = np.exp(100 * (c[top_idx] - c_max))
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for m in range(max(0, k - n + 1), min(k, n - 1) + 1):
            sg[m] += 2.0 * w * a[k - m]
    
    # Full gradient of score
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
    
    # Line search: try different step sizes
    best_sc = compute_score(a)
    best_alpha = 0.0
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a
    
    # Armijo-style line search
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    base_step = (best_sc - sc_lb) / (grad_norm * grad_norm + 1e-30)
    base_step = min(base_step, 0.5)
    
    for i in range(n_trials):
        alpha = base_step * (0.5 ** i)
        if alpha < 1e-10:
            break
        a_trial = a - alpha * grad
        a_trial = np.clip(a_trial, 0.0, None)
        if a_trial.sum() < 1e-6:
            continue
        a_trial = a_trial / a_trial.sum()
        sc_trial = compute_score(a_trial)
        if sc_trial < best_sc:
            best_sc = sc_trial
            best_alpha = alpha
    
    if best_alpha > 0:
        a_new = a - best_alpha * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() > 1e-6:
            a_new = a_new / a_new.sum()
            return a_new
    return a

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines: GS-Fienup spectral method, Top-K LSE L-BFGS, Active-Set Linearized
Programming, and targeted Peak Surgery. Key innovations over prior work:
- Multi-peak LSE gradient (top-K aware) for better L-BFGS direction
- Linearized LP (SAP-style) for structural improvements
- Systematic peak surgery using "safe position" transfer
- Adaptive scheduling based on observed progress
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to unit sum, handling edge cases."""
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def gs_fienup(a, n_iters=500, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with correct flat-convolution target."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    b_target = np.zeros(M)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def topk_lse_grad(x, beta, top_k):
    """Top-K LSE smoothed objective and gradient for L-BFGS."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(c))
    c_top = c[top_idx]
    c_max = float(c_top.max())
    scaled = beta * (c_top - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient via cross-correlation
    q = np.zeros(2 * n - 1)
    q[top_idx] = p
    N = 1 << (3 * n - 2).bit_length()
    Q = np.fft.rfft(q, N)
    Af = np.fft.rfft(x, N)
    corr = np.fft.irfft(Q * np.conj(Af), N)[:n]
    dL_da = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_topk_lbfgs(a0, beta, top_k, maxiter=5000, maxfun=80000):
    """Run L-BFGS-B on Top-K LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a0 = normalize(a0)
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        topk_lse_grad, a0, args=(beta, top_k),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def active_set_lp(a, top_k=10, max_blends=5):
    """
    Linearized Programming: solve LP with linearized peak constraints.
    Returns best point along the LP direction found by line search.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    top_idx = np.argpartition(c, -top_k)[-top_k:]
    c_max = float(c.max())

    # Build LP: minimize t
    # s.t. 2*sum_m a0[k-m]*a[m] - t <= c[k] for k in top_idx
    #      sum(a) = 1, a >= 0
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    a0 = a

    for i, k in enumerate(top_idx):
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            A_ub[i, m_start:m_end + 1] = 2.0 * a0[k - m_end:k - m_start + 1][::-1]
        A_ub[i, n] = -1.0
        b_ub[i] = c[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10})
    except Exception:
        return a

    if not res.success:
        return a

    a_lp = np.maximum(res.x[:n], 0.0)
    if a_lp.sum() < 1e-10:
        return a
    a_lp = a_lp / a_lp.sum()

    # Line search: blend between a and a_lp
    sc_old = compute_score(a)
    best_sc = sc_old
    best_a = a

    for blend in [1.0, 0.7, 0.5, 0.3, 0.1]:
        a_trial = (1.0 - blend) * a + blend * a_lp
        a_trial = np.maximum(a_trial, 0.0)
        if a_trial.sum() < 1e-10:
            continue
        a_trial = normalize(a_trial)
        sc_trial = compute_score(a_trial)
        if sc_trial < best_sc:
            best_sc = sc_trial
            best_a = a_trial

    return best_a


def peak_surgery(a, n_iters=100, top_k=5, rng=None):
    """
    Targeted peak reduction: identify top peaks, find contributing pairs,
    and move mass to 'safe' positions (non-contributing to any top peak).
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argpartition(c, -top_k)[-top_k:]

        # Identify "unsafe" positions: those contributing to any top peak
        unsafe = np.zeros(n, dtype=bool)
        for k in top_idx:
            k = int(k)
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                if a[j] * a[k - j] > 0.05 * c_max / max(1, n // 10):
                    unsafe[j] = True
                    unsafe[k - j] = True

        safe_positions = np.where(~unsafe)[0]
        if len(safe_positions) < n // 4:
            safe_positions = np.where(a < np.percentile(a, 25))[0]
        if len(safe_positions) == 0:
            safe_positions = np.arange(n)

        # For the worst peak, try targeted moves
        k_star = int(np.argmax(c))
        contribs = []
        for j in range(max(0, k_star - n + 1), min(k_star, n - 1) + 1):
            val = a[j] * a[k_star - j]
            if val > 1e-15:
                contribs.append((val, j, k_star - j))
        contribs.sort(reverse=True)

        moved = False
        for val, j1, j2 in contribs[:5]:
            n_tries = 10
            for _ in range(n_tries):
                j_target = safe_positions[rng.integers(0, len(safe_positions))]
                if j_target == j1 or j_target == j2:
                    continue
                # Try moving a fraction of a[j1] to j_target
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = a[j1] * frac
                    a_trial = a.copy()
                    a_trial[j1] -= delta
                    a_trial[j_target] += delta
                    if a_trial[j1] < 0:
                        break
                    a_trial = normalize(a_trial)
                    sc_trial = compute_score(a_trial)
                    if sc_trial < sc:
                        a = a_trial
                        sc = sc_trial
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break

        if not moved:
            # Try random perturbation
            j_rand = rng.integers(0, n)
            j_target = safe_positions[rng.integers(0, len(safe_positions))]
            if j_rand != j_target and a[j_rand] > 1e-8:
                delta = a[j_rand] * 0.1
                a_trial = a.copy()
                a_trial[j_rand] -= delta
                a_trial[j_target] += delta
                a_trial = normalize(a_trial)
                sc_trial = compute_score(a_trial)
                if sc_trial < sc:
                    a = a_trial
                    sc = sc_trial

    return best_a


def subgradient_linesearch(a, top_k=10, n_trials=15):
    """Subgradient step with line search for robustness."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    top_idx = np.argpartition(c, -top_k)[-top_k:]
    c_max = float(c.max())

    # Weighted subgradient (softmax over top-k)
    scaled = 50.0 * (c[top_idx] - c_max)
    weights = np.exp(scaled)
    weights /= weights.sum()

    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            sg[m_start:m_end + 1] += 2.0 * w * a[k - m_end:k - m_start + 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)

    best_sc = compute_score(a)
    best_a = a.copy()
    grad_norm = float(np.linalg.norm(grad))
    if grad_norm < 1e-15:
        return a

    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    base_step = (best_sc - sc_lb) / (grad_norm * grad_norm + 1e-30)
    base_step = min(base_step, 0.5)

    for i in range(n_trials):
        alpha = base_step * (0.6 ** i)
        if alpha < 1e-12:
            break
        a_trial = a - alpha * grad
        a_trial = np.clip(a_trial, 0.0, None)
        if a_trial.sum() < 1e-6:
            continue
        a_trial = normalize(a_trial)
        sc_trial = compute_score(a_trial)
        if sc_trial < best_sc:
            best_sc = sc_trial
            best_a = a_trial

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    return inits
