```python
    run_phases = [
        # (phase_label, length_adjustment, decay_factor, 
        ("ER", 1.4, 0.003, 1.2),
        ("LP", 0.1, 0.02, 60),
        ("Cons", 0.7, 0.01, 35)
    ]

    sorted_results = []

    for (phase, l_adj, decay, max_time) in run_phases:
        # Prepare sequences for this phase
        if phase in ["LP", "Cons"]:
            time_window = max_time * (1.0 - 0.1) ** decay  # time decay adjustment
            new_seqs = generate_sequences(best_seq, decay_factor=decay, count=l_adj*size)
            
        # Execute phase optimization
        if phase == "ER":
            candidates = [best_seq]
            new_seqs = candidates
        elif phase == "LP":
            candidates = lbfgs_optimize(best_seq) + [error_reduction(best_seq)]
            new_seqs = candidates * l_adj + [peak_shrink(best_seq, rate=decay)]
        elif phase == "Cons":
            candidates = sample_cosine_patterns(best_seq) + [uniform_low_freq(best_seq)]
            # Use constraint-aware optimization
            times = [time_window] * 3
            constraint_times = optimize_constrained_sequences(candidates)
            
            for i, candidate in enumerate(candidates):
                time_used["cons"] += times[i]
                score = evaluate_sequence(candidate)
                if score < best_score:
                    best_seq = candidate
                    best_score = score
                    break
        
        # Memory management
        sorted_results = cleanup_old_sequences(sorted_results, candidates, keep_count=10)

    # Phase d: Fallback generation
    fallback = create_uniform_candidate(rng, n_parent)
    fallback_score = evaluate_sequence(fallback)
    
    # Verify best candidate
    if fallback_score < best_score:
        return fallback.tolist()
    elif best_score > 0.01:
        return best_seq.tolist()
    else:
        # Create combined fallback
        cleanup = np.array([0.5]*fallback_length + [1.0]*(default_size*2))
        return cleanup.tolist()
```

Each phase includes:
- **Error reduction (ER)**: Focuses on minimization using exponential smoothing
- **LP-based optimization**: Employs large-step length search
- **Constraint (Cons)**: Handles general constraints and synthesizes valid sequences
