def _norm(a, target=1.0):
    """Normalize sequence to have target sum, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    """Resample sequence to length n via linear interpolation."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _multiplicative_peak_shaving(a, max_iters=200, deadline=None, verbose=True):
    """
    Multiplicative update algorithm for peak shaving.
    At each step:
    1. Compute autocorrelation and identify peak positions
    2. Compute the "pressure" on each a[i] from its contribution to peaks
    3. Apply multiplicative update: a[i] *= (1 - eta * pressure[i] / mean(pressure))
    4. Project onto non-negative orthant and renormalize
    """
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    # Adaptive step size schedule
    eta_start = 0.5
    eta_end = 0.001
    # Adaptive active set threshold
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        S = float(a.sum())
        c = _self_conv(a)
        c_max = float(c.max())
        if c_max <= 0:
            break
        
        # Active set: positions where c is within fraction of max
        # Start tight, loosen as we converge
        frac = 0.01 + 0.1 * (it / max(1, max_iters))
        active_thresh = c_max * (1.0 - frac)
        active = c >= active_thresh
        if not active.any():
            idx = np.argmax(c)
            active[idx] = True
        
        n_active = int(active.sum())
        
        # Compute pressure via cross-correlation
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        corr = _cross_corr(a, w)  # This is h/2 where h is the gradient
        
        # Multiplicative update
        # Schedule: eta decreases over iterations
        progress = it / max(1, max_iters)
        eta = eta_start * (eta_end / eta_start) ** progress
        
        # pressure = corr (proportional to how much a[i] contributes to peaks)
        mean_p = float(corr.mean())
        if mean_p < 1e-15:
            break
        
        # Update: reduce positions with high pressure, increase those with low
        ratio = 1.0 - eta * (corr - mean_p) / mean_p
        ratio = np.clip(ratio, 0.1, 5.0)  # Bound the multiplicative factor
        a_new = a * ratio
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-8:
            break
        a_new = a_new / S_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Accept if improvement, or if very early (to escape plateaus)
        if sc_new < best_sc or it < 5:
            a = a_new
        else:
            # Reject this step, try smaller eta
            eta /= 3.0
            ratio = 1.0 - eta * (corr - mean_p) / mean_p
            ratio = np.clip(ratio, 0.5, 2.0)
            a_new = a * ratio
            a_new = np.clip(a_new, 0.0, None)
            S_new = a_new.sum()
            if S_new < 1e-8:
                break
            a_new = a_new / S_new
            sc_new = _score(a_new)
            if sc_new <= _score(a):
                a = a_new
            else:
                # Stuck, try even smaller
                break
        
        if verbose and it % 20 == 0:
            print(f"    [peak_shave] it={it}, score={best_sc:.8f}, eta={eta:.4f}, n_active={n_active}")
    
    return best_sc, best_a


def _subgradient_polish_v2(a, max_iters=100, deadline=None):
    """
    Improved subgradient polish with:
    - Backtracking line search
    - Adaptive active set that tightens over time
    - Momentum (Nesterov)
    """
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    # Momentum state (for Nesterov)
    y = a.copy()  # extrapolated point
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        S = float(y.sum())
        if S < 1e-12:
            break
        
        c = _self_conv(y)
        c_max = float(c.max())
        if c_max <= 0:
            break
        
        # Active set: tighten over iterations
        # Start with 1% of max, tighten to 0.01%
        tol_frac = max(0.0001, 0.01 * (1 - it / max_iters))
        active = c >= c_max * (1.0 - tol_frac)
        if not active.any():
            active[np.argmax(c)] = True
        n_act = int(active.sum())
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(y, w)  # = h/2
        # Gradient of the objective (with sum=1 normalization):
        # grad = 2*n*corr/S^2 (for the max term)
        # But we also need to account for the normalization constraint.
        # Using the Lagrangian: L = 2*n*max(c) + lambda*(sum(a)-1)
        # grad_L[i] = 2*n*corr[i]/S^2 + lambda
        # At optimum, lambda = -2*n*sum(corr)/S^2 (from sum of grads = 0)
        # Actually, let's just use the projected subgradient:
        # The subgradient of 2*n*max(c)/S^2 at a (with the normalization) is:
        # We work in the space where sum(a) = 1, so we project the gradient.
        
        # Simple approach: compute the "raw" gradient and project
        grad = 2.0 * float(n) * corr / (S * S)
        # Project: subtract the mean to ensure sum(grad) = 0 (tangential to the simplex)
        # Actually, for the constrained problem min f(a) s.t. a >= 0, sum(a) = 1:
        # The KKT conditions give: grad_f[i] + mu = 0 for a[i] > 0
        # where mu is the Lagrange multiplier for the equality constraint.
        # mu = -mean of grad_f over active coordinates (where a[i] > 0)
        # The projected gradient is: proj_grad[i] = grad[i] - mu (for a[i] > 0)
        # = grad[i] - mean(grad over {j: a[j] > 0})
        
        support = a > 1e-15
        if support.sum() > 0:
            mu = float(grad[support].mean())
            proj_grad = grad - mu
        else:
            proj_grad = grad
        
        # Set proj_grad = 0 where a = 0 (normal cone of non-negativity)
        proj_grad[~support] = 0.0
        
        gnorm = float(np.sqrt(np.sum(proj_grad ** 2)))
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        best_step_val = None
        best_step_a = None
        best_step_sc = best_sc
        base_step = 1.0 / gnorm
        
        for scale in (1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            step = scale * base_step
            q = a - step * proj_grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step_a = q.copy()
                best_step_val = step
        
        if best_step_a is None:
            # Try Nesterov extrapolation
            momentum = min(0.9, (it + 1.0) / (it + 2.0))
            y_new = a + momentum * (a - y)
            y_new = np.clip(y_new, 0.0, None)
            Sy = y_new.sum()
            if Sy > 1e-8:
                y_new = y_new / Sy
                y = y_new
            break
        
        # Check improvement over global best
        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = best_step_a.copy()
        
        # Nesterov momentum
        momentum = min(0.9, (it + 1.0) / (it + 2.0))
        y_new = best_step_a + momentum * (best_step_a - y)
        y_new = np.clip(y_new, 0.0, None)
        Sy = y_new.sum()
        if Sy > 1e-8:
            y_new = y_new / Sy
        else:
            y_new = best_step_a.copy()
        y = y_new
        a = best_step_a.copy()
    
    return best_sc, best_a


def _lse_optimize(a, beta, k, time_limit, maxfun_cap=600):
    """
    LSE-smoothed optimization via L-BFGS-B.
    Minimizes 2 * LSE_beta(top_k of d = n*conv/S^2).
    """
    a = _norm(a)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 6000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        Sx = float(x.sum())
        if Sx < 1e-8:
            return 1e10, np.zeros_like(x)
        
        c = _self_conv(x)
        d = float(n) * c / (Sx * Sx)
        M = 2 * n - 1
        kk = max(1, min(k, M))
        
        idx = np.argpartition(d, M - kk)[M - kk:]
        d_act = d[idx]
        dmax = float(d_act.max())
        if dmax <= 0 or not np.isfinite(dmax):
            return 1e10, np.zeros_like(x)
        
        z = beta * (d_act - dmax)
        e = np.exp(z)
        se = float(e.sum())
        if se <= 0 or not np.isfinite(se):
            e = np.ones(kk) / kk
            se = 1.0
        w_act = e / se
        L = dmax + np.log(se) / beta
        
        w_full = np.zeros(M)
        w_full[idx] = w_act
        
        corr = _cross_corr(x, w_full)
        LDS = float(w_act @ d_act)
        
        grad = 4.0 * float(n) * corr / (Sx * Sx) - 4.0 * LDS / Sx
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        # Add sum penalty
        v = 2.0 * L + PEN * (Sx - 1.0) ** 2
        g = grad + 2.0 * PEN * (Sx - 1.0)
        return float(v), g

    try:
        res = minimize(
            obj, a.copy(), method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)
