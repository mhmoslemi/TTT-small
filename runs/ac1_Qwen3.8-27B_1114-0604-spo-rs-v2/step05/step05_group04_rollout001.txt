def fast_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_sum1(a):
    """Normalize a to sum to 1, clip non-negative."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a / S


def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} via water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n + 1)) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fienup_optimize(a, n_iters=300, top_frac_init=0.1, top_frac_end=0.5,
                    gamma=0.7, soft_alpha=30.0):
    """Spectral shaping via Fienup with adaptive top-fraction."""
    n = len(a)
    a = normalize_sum1(a.astype(np.float64))
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 0.001)) ** gamma
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * np.clip(a_new, -50, 50)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_flat(a, n_iters=200, alpha_init=0.3, alpha_end=0.8):
    """Spectral equalization: relax power spectrum towards flat."""
    n = len(a)
    a = normalize_sum1(a.astype(np.float64))
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        A_mod = A * np.sqrt(P_new / P)
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    return best_a

    M2_fft = 1 << (M2 - 1).bit_length()
    PX = np.fft.rfft(p, M2_fft)
    RX = np.fft.rfft(rev_x, M2_fft)
    conv_res = np.fft.irfft(PX * RX, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    a0 = normalize_sum1(a0)
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    a_out = normalize_sum1(a_out)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def exact_subgradient_step(a, n_iters=5000, momentum=0.9, step_init=0.001,
                           step_decay=0.9999, step_min=1e-6):
    """
    Projected subgradient method with momentum for min max_k (a*a)_k s.t. a>=0, sum(a)=1.
    Gradient at argmax k*: g_i = 2 * a_{k*-i} (when 0 <= k*-i <= n-1)
    """
    n = len(a)
    a = normalize_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    v = np.zeros(n)  # momentum
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find argmax (all positions achieving max, for a valid subgradient use one)
        k_star = int(np.argmax(c))

        # Gradient: g_i = 2 * a_{k_star - i} where 0 <= k_star - i <= n-1
        g = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            # g[i] = 2*a[k_star - i] for i in [k_star-(n-1), k_star] clipped to [0,n-1]
            # a[k_star - i] for i=0..n-1: valid when 0 <= k_star-i <= n-1
            # i.e., k_star-(n-1) <= i <= k_star
            vals = a[j_start:j_end + 1][::-1]  # a[k_star - i] for i = j_start..j_end
            g[j_start:j_end + 1] = 2.0 * vals

        # Momentum update
        v = momentum * v - step * g
        a_new = a + v

        # Project onto {a >= 0, sum(a) = 1}
        a_new = project_simplex(a_new)
        a = a_new

        # Adaptive step size
        step *= step_decay
        step = max(step, step_min)

        # Occasionally restart momentum
        if it % 500 == 0:
            v *= 0.5

    return best_a


def spectral_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain (random phase noise on amplitude)."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = np.abs(A)
    phase = np.angle(A)
    phase += strength * rng.standard_normal(len(phase))
    A_new = amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence of length 6962 minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Fienup warmup + high-beta L-BFGS + exact subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    n = 6962

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size == n:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(n) * (1.0 / n)

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    print(f"n={n}, deadline={total_budget-20:.0f}s")

    # Initialize from parent
    if parent is not None:
        sc_p = compute_score(parent)
        update_best(sc_p, parent, "PARENT")
        a_init = normalize_sum1(parent)
    else:
        a_init = np.ones(n) / n
        print("No parent found, starting from uniform.")

    # ==================================================================
    # Phase 1: Fienup warmup (10% time ~ 228s)
    # ==================================================================
    phase1_end = t_start + 0.10 * (total_budget - 20.0)
    print(f"\n=== Phase 1: Fienup warmup (until t={phase1_end-t_start:.0f}s) ===")

    # Run multiple Fienup variants
    a_cur = a_init.copy()

    # Variant 1: Top-fraction compression
    a_f1 = fienup_optimize(a_cur, n_iters=200, top_frac_init=0.08, top_frac_end=0.5,
                           gamma=0.6, soft_alpha=20.0)
    sc = compute_score(a_f1)
    update_best(sc, a_f1, "FIENUP1")
    if sc < best_score:
        a_cur = a_f1.copy()

    # Variant 2: Spectral flattening
    a_f2 = fienup_flat(a_cur, n_iters=200, alpha_init=0.3, alpha_end=0.8)
    sc = compute_score(a_f2)
    update_best(sc, a_f2, "FLAT1")
    if sc < best_score:
        a_cur = a_f2.copy()

    # Variant 3: More aggressive compression
    a_f3 = fienup_optimize(a_cur, n_iters=200, top_frac_init=0.15, top_frac_end=0.6,
                           gamma=0.4, soft_alpha=10.0)
    sc = compute_score(a_f3)
    update_best(sc, a_f3, "FIENUP2")
    if sc < best_score:
        a_cur = a_f3.copy()

    # Variant 4: Moderate flattening
    a_f4 = fienup_flat(a_cur, n_iters=150, alpha_init=0.4, alpha_end=0.9)
    sc = compute_score(a_f4)
    update_best(sc, a_f4, "FLAT2")
    if sc < best_score:
        a_cur = a_f4.copy()

    print(f"  After Fienup: best={best_score:.8f}")

    # ==================================================================
    # Phase 2: Two-phase L-BFGS (60% time ~ 1368s)
    # ==================================================================
    phase2_end = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n=== Phase 2: L-BFGS (until t={phase2_end-t_start:.0f}s) ===")

    # Phase 2a: Moderate beta to get into basin
    beta_list = [500, 2000, 10000, 50000, 100000]
    a_lbfgs_start = a_cur.copy()

    elapsed_p2 = 0
    for i, beta in enumerate(beta_list):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break

        # Allocate time proportional to available
        maxiter = min(15000, int(remaining * 0.8 / 0.005))  # ~5ms per iteration estimate
        maxfun = min(200000, maxiter * 15)
        # Cap for very high beta (converges slower)
        if beta > 10000:
            maxiter = min(maxiter, 8000)
            maxfun = min(maxfun, 100000)

        print(f"  L-BFGS beta={beta}, maxiter={maxiter}, maxfun={maxfun}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_lbfgs_start, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f}, time={dt:.1f}s")
        update_best(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_lbfgs_start = a_opt.copy()
        elapsed_p2 += dt

    # Try from a spectrally perturbed version if time allows
    if time.monotonic() < phase2_end - 60:
        a_pert = spectral_perturb(best_a, rng, strength=0.05)
        a_pert = normalize_sum1(a_pert)
        # Quick Fienup on perturbed
        a_pert2 = fienup_optimize(a_pert, n_iters=100, top_frac_init=0.1, top_frac_end=0.4)
        sc = compute_score(a_pert2)
        update_best(sc, a_pert2, "PERT-FIENUP")
        a_try = a_pert2 if sc < best_score else best_a
        a_try = normalize_sum1(a_try)
        sc, a_opt = run_lbfgs(a_try, 50000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "PERT-LBFGS")

    print(f"  After L-BFGS: best={best_score:.8f}")

    # ==================================================================
    # Phase 3: Exact subgradient with momentum (20% time ~ 456s)
    # ==================================================================
    phase3_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Exact subgradient (until t={phase3_end-t_start:.0f}s) ===")

    # Multiple subgradient runs with different parameters
    sg_configs = [
        {'n_iters': 3000, 'momentum': 0.9, 'step_init': 0.002, 'step_decay': 0.9999},
        {'n_iters': 3000, 'momentum': 0.95, 'step_init': 0.001, 'step_decay': 0.9998},
        {'n_iters': 2000, 'momentum': 0.85, 'step_init': 0.005, 'step_decay': 0.9997},
        {'n_iters': 2000, 'momentum': 0.9, 'step_init': 0.003, 'step_decay': 0.9999},
    ]

    for i, cfg in enumerate(sg_configs):
        if time.monotonic() > phase3_end:
            break
        # Estimate time per iteration (~2ms for FFT + projection)
        max_iters = min(cfg['n_iters'], int((phase3_end - time.monotonic()) / 0.003))
        if max_iters < 100:
            break
        cfg['n_iters'] = max_iters
        print(f"  Subgradient run {i}: {cfg}")
        a_sg = exact_subgradient_step(best_a, **cfg)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG-{i}")

    print(f"  After subgradient: best={best_score:.8f}")

    # ==================================================================
    # Phase 4: Final L-BFGS polish (remaining time)
    # ==================================================================
    print(f"\n=== Phase 4: Final polish ===")

    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            sc, a_opt = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=min(80000, int(remaining * 150)))
            update_best(sc, a_opt, "FINAL-LBFGS-100k")

    # One more subgradient pass if time allows
    if time.monotonic() < deadline - 10:
        remaining_iter = int((deadline - time.monotonic() - 5) / 0.003)
        if remaining_iter > 200:
            a_sg = exact_subgradient_step(best_a, n_iters=remaining_iter,
                                          momentum=0.95, step_init=0.001, step_decay=0.99995)
            sc = compute_score(a_sg)
            update_best(sc, a_sg, "FINAL-SG")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
