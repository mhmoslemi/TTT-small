"""
Hybrid optimizer for minimum peak autocorrelation: GS-Fienup init,
LSE continuation (L-BFGS), reweighted peak-targeted LSE, Nesterov
accelerated subgradient, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_m=None):
    """
    LSE-smoothed score and optional gradient.
    If top_m is given, only the top-m peaks contribute to the LSE.
    Returns (score, grad) or (score, None) if want_grad=False.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(np.max(c))
    if top_m is not None and top_m < len(c):
        top_idx = np.argsort(c)[::-1][:top_m]
        c_active = c[top_idx]
        scaled = beta * (c_active - c_max)
        scaled = np.clip(scaled, -700, 700)
        exp_scaled = np.exp(scaled)
        log_sum = float(np.log(np.sum(exp_scaled)))
        L = c_max + log_sum / beta
        p_active = exp_scaled / np.sum(exp_scaled)
    else:
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -700, 700)
        exp_scaled = np.exp(scaled)
        log_sum = float(np.log(np.sum(exp_scaled)))
        L = c_max + log_sum / beta
        p_active = exp_scaled / np.sum(exp_scaled)
        top_idx = None

    score = 2.0 * n * L / (S * S)

    # Build full p vector
    p = np.zeros(len(c), dtype=np.float64)
    if top_idx is not None:
        p[top_idx] = p_active
    else:
        p = p_active

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def gs_fienup(a, n_iters=3000, alpha_init=0.8, alpha_final=0.05, M=None):
    """
    GS-Fienup with correct flat-convolution frequency target.
    Uses adaptive alpha annealing.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_magnitude_only(a, n_iters=2000, alpha=0.4, M=None):
    """GS that only updates magnitude, preserving phase. More stable."""
    n = len(a)

def run_lbfgs(a0, beta, maxiter=20000, top_m=None):
    """Run L-BFGS-B on the LSE-smoothed objective with optional top-m reweighting."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n

    def obj_func(x):
        sc, gr = lse_score_grad(x, beta, top_m=top_m)
        return sc, gr

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxiter * 50, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_subgradient(a, n_iters=5000, top_m=20, step_init=0.03, decay=0.9999):
    """
    Nesterov-accelerated subgradient method targeting top-m peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    # Nesterov momentum
    y = a.copy()
    momentum = 1.0

    for it in range(n_iters):
        # Update momentum
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        theta = (1.0 + momentum) / (momentum + 1.0)
        # Actually standard Nesterov: theta_k = (1 - alpha_k) / (1 + alpha_k) where alpha_k = 2/(k+2)
        # Let me use a simpler scheme
        alpha_k = 2.0 / (it + 2.0)
        theta_k = (1.0 - alpha_k) / (1.0 + alpha_k)

        c = fast_conv(y)
        S_y = float(np.sum(y))
        if S_y < 1e-12:
            break
        M_val = float(c.max())
        sc_y = 2.0 * n * M_val / (S_y * S_y)

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Subgradient at y
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Step size with safety
        step = min(step, 0.5 * (sc_y - 1.0) / (gn * gn + 1e-30))
        step = min(step, 0.05 / alpha_k)

        a_new = y - alpha_k * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = y
        a_new = a_new / S_new

        # Nesterov update: y = a_new + theta_k * (a_new - a)
        y = a_new + theta_k * (a_new - a)
        y = np.clip(y, 0.0, 1000.0)
        S_y = float(y.sum())
        if S_y < 1e-12:
            y = a_new
        y = y / S_y
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_descent_peaks(a, n_sweeps=10, n_top=200):
    """
    Coordinate descent: for each of the top-contributing elements,
    do a fine 1D search to minimize the score.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Find elements that contribute most to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        contrib = np.zeros(n)
        seg = a[j_start:j_end + 1]
        contrib[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Get top contributing indices
        top_count = min(n_top, n)
        top_global = np.argsort(contrib)[::-1][:top_count]

        for j in top_global:
            j = int(j)
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue

            # Try a range of multipliers
            cur_sc = compute_score(a)
            best_val = a_old_j
            best_local_sc = cur_sc

            for mult in [0.2, 0.4, 0.6, 0.75, 0.85, 0.9, 1.1, 1.25, 1.5, 2.0]:
                cv = a_old_j * mult
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
                a[j] = a_old_j

            a[j] = best_val

        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Early stop if no improvement
        if best_sc >= best_a.__class__ and abs(sc - best_sc) < 1e-15:
            pass

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.2 * np.sin(10.0 * np.pi * x) * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * x + 0.5) + 0.2 * np.cos(4.0 * np.pi * x - 0.3), 0.01))
    # Slightly randomized structured inits
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(np.maximum(0.4 + 0.4 * np.cos(2.0 * np.pi * freq * x + phase), 0.01))

    return inits

        sc, a_opt = run_lbfgs(best_a, 200000, maxiter=2000)
        update_best(sc, a_opt, "FINAL-LBFGS-200k")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
