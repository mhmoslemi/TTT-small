```python
"""
Min-norm active-set subgradient descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. Min-norm subgradient (exact steepest descent direction) via efficient QP
2. Gram matrix computed from convolution (O(1) per entry)
3. Backtracking line search with Armijo condition
4. Adaptive active set management
5. L-BFGS-B refinement on LSE-smoothed objective
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0)

def _norm1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0), 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, 1.0/a.size)
    return a / s

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (s*s))

def _grad_conv_j(a, j):
    """Gradient of conv(a,a)[j] w.r.t. a. Result: 2*a[j-k] for valid j-k."""
    n = a.size
    g = np.zeros(n)
    k = np.arange(n)
    jk = j - k
    valid = (jk >= 0) & (jk < n)
    g[valid] = 2.0 * a[jk[valid]]
    return g

def _min_norm_subgrad(weights_grads, gram_matrix):
    """Solve min ||sum w_i g_i||^2 s.t. sum w = 1, w >= 0.
    Equivalent to: min w^T G w s.t. 1^T w = 1, w >= 0."""
    m = len(weights_grads)
    if m == 1:
        return np.array([1.0])
    
    G = gram_matrix  # m x m
    
    # Try unconstrained solution with equality constraint
    # min w^T G w s.t. 1^T w = 1
    # KKT: 2Gw = lambda*1 -> w = lambda/2 * G^{-1}*1
    # Constraint: 1^T w = 1 -> lambda/2 * 1^T G^{-1} 1 = 1 -> lambda = 2/(1^T G^{-1} 1)
    try:
        G_inv_1 = np.linalg.solve(G, np.ones(m))
        denom = G_inv_1.sum()
        if denom > 1e-15:
            w = G_inv_1 / denom
            if w.min() >= -1e-10:
                w = np.maximum(w, 0)
                w /= w.sum()
                return w
    except:
        pass
    
    # Fallback: use scipy
    from scipy.optimize import minimize as sp_min
    def qf(w):
        return w @ G @ w
    def qg(w):
        return 2 * G @ w
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1, 'jac': lambda w: np.ones(m)}]
    bounds = [(0, None)] * m
    w0 = np.ones(m) / m
    res = sp_min(qf, w0, jac=qg, method='SLSQP', bounds=bounds, constraints=cons,
                 options={'maxiter': 200, 'ftol': 1e-14})
    if res.success:
        w = np.maximum(res.x, 0)
        w /= w.sum()
        return w
    return w0

def _sse_optimize(a0, deadline, label):
    """Min-norm subgradient descent with backtracking."""
    a = _norm1(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    
    tol = 1e-8
    eta0 = 0.01
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = _self_conv(a)
        c_max = c.max()
        score = 2.0 * n * c_max / (a.sum()**2)
        
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"  [{label}] iter {it}: score={score:.8f}")
        
        # Active set
        active_idx = np.where(c >= c_max - tol * max(1.0, c_max))[0]
        m = len(active_idx)
        
        if m == 0:
            active_idx = np.array([np.argmax(c)])
            m = 1
        
        # Compute min-norm subgradient
        if m == 1:
            j = active_idx[0]
            g = _grad_conv_j(a, j)
        else:
            # Build gram matrix from convolution
            # G[i][r] = 4 * c[j_r - j_i] (approximately, ignoring boundary effects)
            # More precisely: G[i][r] = g_{j_i}^T g_{j_r}
            # We can compute this as 4 * sum_k a[j_i - k] * a[j_r - k]
            # For efficiency, use the convolution approximation
            G = np.zeros((m, m))
            for i in range(m):
                for r in range(m):
                    diff = active_idx[r] - active_idx[i]
                    if abs(diff) < 2*n - 1:
                        G[i, r] = 4.0 * c[abs(diff)]  # Use convolution value
                    # Diagonal: G[i,i] = 4 * c[0] = 4 * sum(a^2)
                if G[i,i] <= 0:
                    G[i,i] = 4.0 * np.sum(a*a)
            # Ensure symmetry and PSD
            G = (G + G.T) / 2.0
            # Add small regularization
            G += 1e-10 * np.eye(m)
            
            w = _min_norm_subgrad(None, G)
            # Compute gradient as weighted sum
            g = np.zeros(n)
            for i in range(m):
                g += w[i] * _grad_conv_j(a, active_idx[i])
        
        # Normalize gradient
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Project onto sum(a)=1 manifold: subtract mean component
        # The effective gradient on the constraint manifold is g - (g.mean())*1
        g_eff = g - g.mean()
        
        # Backtracking line search
        f_cur = c_max  # Current max convolution value
        eta = eta0
        improved = False
        for ls_iter in range(20):
            a_trial = a - eta * g_eff
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                eta *= 0.5
                continue
            a_trial = a_trial / s
            c_trial = _self_conv(a_trial)
            f_trial = c_trial.max()
            
            # Armijo condition
            if f_trial < f_cur - 0.01 * eta * np.sum(g_eff * (a - a.mean())):
                improved = True
                a = a_trial
                break
            eta *= 0.5
        
        if not improved:
            # Try even smaller step
            eta = 1e-6
            a_trial = a - eta * g_eff
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-12:
                a_trial = a_trial / s
                a = a_trial
            else:
                break
        
        # Adaptive step size
        if improved:
            eta0 = min(eta * 1.5, 0.1)
        else:
            eta0 = max(eta, 1e-8)
        
        # Update tolerance periodically
        if it % 200 == 0:
            tol = max(1e-10, 5e-5 * c_max / (1 + it * 0.001))
    
    return best_sc, best_a

def _lse_refine(a0, deadline, label):
    """L-BFGS-B refinement on LSE-smoothed objective."""
    a = _norm1(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    def lse_obj(a, beta, k):
        c = _self_conv(a)
        s = a.sum()
        d = n * c / (s*s)
        M = 2*n - 1
        k = max(1, min(k, M))
        idx = np.argpartition(d, M - k)[M - k:]
        d_act = d[idx]
        dmax = d_act.max()
        if dmax <= 0:
            return 2.0, np.zeros(n)
        z = beta * (d_act - dmax)
        e = np.exp(z)
        se = e.sum()
        if se <= 0:
            e = np.ones(k)/k
            se = 1.0
        w_act = e / se
        L = dmax + np.log(se)/beta
        
        # Gradient
        w_full = np.zeros(M)
        w_full[idx] = w_act
        # Cross-correlation
        Nf = _next_pow2(M + n - 1)
        Ar = np.fft.rfft(a[::-1], Nf)
        W = np.fft.rfft(np.concatenate([w_full, np.zeros(Nf - M)]), Nf)
        corr_full = np.fft.irfft(Ar * W, Nf)
        corr = corr_full[n-1:2*n-1]
        
        LDS = w_act @ d_act
        grad = 4.0*n*corr/(s*s) - 4.0*LDS/s
        grad = np.nan_to_num(grad, nan=0.0)
        return 2.0*L, grad
    
    # Run L-BFGS-B with increasing k
    for k, beta in [(20, 100), (50, 300), (100, 1000), (300, 3000), (500, 10000)]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        def obj(x):
            v, g = lse_obj(x, beta, k)
            return v, g
        
        try:
            res = minimize(obj, a, method='L-BFGS-B', jac=True,
                          bounds=[(0.0, None)]*n,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'ftol': 1e-15, 'gtol': 1e-12})
            a_new = _norm1(res.x) if res.x is not None else a
        except:
            a_new = a
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"  [{label}] LSE k={k}, beta={beta}: {best_sc:.8f}")
        a = a_new
    
    return best_sc, best_a

def _spectral_init(n, rng):
    Nf = _next_pow2(2*n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs))**exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n)/n
    else:
        sig = sig / sig.sum()
    return sig
```

```python
def _project_onto_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find tau using the standard algorithm
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    tau = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - tau, 0)
```

```python
"""
Min-norm active-set subgradient descent with simplex projection for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses exact min-norm subgradient (steepest descent
direction for the max), water-filling projection, adaptive step sizes with
momentum, and LSE refinement via L-BFGS-B. Multiple diverse initializations
with progressive optimization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _norm_simplex(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (s * s))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via water-filling."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    tau = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - tau, 0.0)


def _grad_conv_j(a, j):
    """Gradient of conv(a,a)[j] w.r.t. a: result[k] = 2*a[j-k] if 0<=j-k<n else 0."""
    n = a.size
    g = np.zeros(n)
    k = np.arange(n)
    jk = j - k
    valid = (jk >= 0) & (jk < n)
    g[valid] = 2.0 * a[jk[valid]]
    return g


def _solve_min_norm_qp(G):
    """Solve min w^T G w s.t. sum(w)=1, w>=0. G is m x m symmetric PSD."""
    m = G.shape[0]
    if m == 1:
        return np.array([1.0])
    if m == 2:
        # Analytic: w1 = (G22 - G12) / (G11 + G22 - 2*G12), w2 = 1-w1
        denom = G[0, 0] + G[1, 1] - 2.0 * G[0, 1]
        if denom < 1e-15:
            return np.array([0.5, 0.5])
        w1 = (G[1, 1] - G[0, 1]) / denom
        w1 = float(np.clip(w1, 0.0, 1.0))
        return np.array([w1, 1.0 - w1])
    # General case: use scipy SLSQP
    def qf(w):
        return float(w @ G @ w)
    def qg(w):
        return 2.0 * (G @ w)
    cons = [{'type': 'eq', 'fun': lambda w: float(w.sum() - 1.0),
             'jac': lambda w: np.ones(m)}]
    bounds = [(0.0, None)] * m
    w0 = np.ones(m) / m
    try:
        res = minimize(qf, w0, jac=qg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
        if res.success:
            w = np.maximum(res.x, 0.0)
            sw = w.sum()
            if sw > 1e-15:
                return w / sw
    except Exception:
        pass
    # Fallback: uniform
    return w0


def _sse_optimize(a0, deadline, label, use_momentum=True):
    """Min-norm subgradient descent with simplex projection and backtracking."""
    a = _norm_simplex(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    momentum = 0.0
    v = np.zeros(n)

    eta = 0.05
    min_eta = 1e-7
    tol_frac = 1e-6
    iter_count = 0
    no_improve = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        c_max = float(c.max())
        s = float(a.sum())
        score = 2.0 * n * c_max / (s * s)

        if score < best_sc - 1e-12:
            best_sc = score
            best_a = a.copy()
            no_improve = 0
            if iter_count % 1000 == 0:
                print(f"  [{label}] it={it}: score={score:.8f}, cmax={c_max:.6f}")
        else:
            no_improve += 1

        #.Active set
        threshold = c_max - tol_frac * max(1.0, c_max)
        active_idx = np.where(c >= threshold)[0]
        m = len(active_idx)

        if m == 0:
            active_idx = np.array([int(np.argmax(c))])
            m = 1

        # Compute min-norm subgradient
        if m == 1:
            g = _grad_conv_j(a, int(active_idx[0]))
        else:
            # Build gradient vectors
            grads = np.zeros((m, n))
            for i in range(m):
                grads[i] = _grad_conv_j(a, int(active_idx[i]))
            # Gram matrix
            G = grads @ grads.T
            G = (G + G.T) / 2.0
            G += 1e-12 * np.eye(m)
            # Solve QP
            if m <= 100:
                w = _solve_min_norm_qp(G)
            else:
                w = np.ones(m) / m
            g = w @ grads

        # Apply momentum
        if use_momentum:
            v = momentum * v + (1.0 - momentum) * g
            direction = v
        else:
            direction = g

        # Project gradient onto sum(a)=1 tangent space
        direction = direction - direction.mean()

        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-14:
            break

        # Backtracking line search
        improved = False
        for ls in range(15):
            a_trial = a - eta * direction
            a_trial = _project_simplex(a_trial)
            c_trial = _self_conv(a_trial)
            f_trial = float(c_trial.max())

            # Sufficient decrease: f_trial < f_cur - 0.001 * eta * ||dir||^2
            if f_trial < c_max - 1e-5 * eta * dnorm * dnorm:
                a = a_trial
                improved = True
                break
            eta *= 0.5
            if eta < min_eta:
                break

        if not improved:
            # Try tiny step
            eta_trial = min_eta
            a_trial = a - eta_trial * direction
            a_trial = _project_simplex(a_trial)
            c_trial = _self_conv(a_trial)
            if c_trial.max() < c_max:
                a = a_trial
                improved = True

        if not improved and no_improve > 2000:
            # Restart with perturbation
            noise = np.random.default_rng(it).standard_normal(n) * 0.01
            a = _project_simplex(a + noise)
            momentum = 0.0
            v = np.zeros(n)
            eta = 0.05
            no_improve = 0

        # Adaptive step size and momentum
        if improved:
            eta = min(eta * 1.3, 0.2)
            momentum = min(momentum + 0.001, 0.6)
        else:
            eta = max(eta * 0.7, min_eta)

        iter_count += 1

    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_sc, best_a


def _lse_refine(a0, deadline, label):

    a = _norm_simplex(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

        (10, 50.0, 0.15),
        (30, 150.0, 0.15),

        (120, 1000.0, 0.15),
        (250, 3000.0, 0.15),
        (500, 8000.0, 0.25),
    ]

    total_time = deadline - time.monotonic()
    if total_time < 10:
        return best_sc, best_a

    cum_frac = 0.0
    for k, beta, frac in stages:
        cum_frac += frac
        stage_end = time.monotonic() + min(frac, max(0.0, 1.0 - cum_frac + frac)) * total_time
        stage_end = min(stage_end, deadline)
        if time.monotonic() > stage_end - 2:
            continue

        def obj_lse(x):
            x = np.maximum(x, 0.0)
            sx = float(x.sum())
            if sx < 1e-8:
                return 1e10, np.zeros(len(x))
            c = _self_conv(x)
            d = float(len(x)) * c / (sx * sx)
            M = 2 * len(x) - 1
            kk = max(1, min(k, M))
            idx = np.argpartition(d, M - kk)[M - kk:]
            d_act = d[idx]
            dmax = float(d_act.max())
            if dmax <= 0:
                return 2.0, np.zeros(len(x))
            z = beta * (d_act - dmax)
            e = np.exp(z)
            se = float(e.sum())
            if se <= 0 or not np.isfinite(se):
                e = np.ones(kk) / kk
                se = 1.0
            w_act = e / se
            L = dmax + np.log(se) / beta

            w_full = np.zeros(M)
            w_full[idx] = w_act
            Nf = _next_pow2(M + len(x) - 1)
            Ar = np.fft.rfft(x[::-1], Nf)
            W = np.fft.rfft(w_full, Nf)
            corr_full = np.fft.irfft(Ar * W, Nf)
            corr = corr_full[len(x) - 1:2 * len(x) - 1]

            LDS = float(w_act @ d_act)
            grad = 4.0 * len(x) * corr / (sx * sx) - 4.0 * LDS / sx
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
            return 2.0 * L, grad

        try:
            maxfun = int(max(50, 0.8 * (stage_end - time.monotonic()) / 0.003))
            maxfun = min(maxfun, 2000)
            res = minimize(obj_lse, a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max(20, maxfun // 3),
                                    'maxfun': maxfun, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxls': 30})
            a_new = _norm_simplex(res.x) if res.x is not None else a
        except Exception:
            a_new = a

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"  [{label}] LSE k={k}, beta={beta:.0f}: {best_sc:.8f}")
        a = a_new

    return best_sc, best_a


def _subgrad_polish(a0, max_iters, deadline):
    """Fine subgradient polishing with very small steps."""
    a = _norm_simplex(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _self_conv(a)
        c_max = float(c.max())
        s = float(a.sum())
        score = 2.0 * n * c_max / (s * s)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()

        active_idx = np.where(c >= c_max - 1e-8)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        g = np.zeros(n)
        for j in active_idx:
            g += _grad_conv_j(a, int(j))
        g /= len(active_idx)
        g = g - g.mean()

        for eta in (1e-4, 1e-5, 1e-6, 1e-7):
            a_trial = _project_simplex(a - eta * g)
            sc_trial = _score(a_trial)
            if sc_trial < best_sc:
                best_sc = sc_trial
                best_a = a_trial.copy()
                a = a_trial
                break

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with controlled power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _upsample(a, n):
    m = a.size
    if m == n:
        return a.copy()
    if m == 0:
        return np.ones(n) / n
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _smooth_perturb(a, rng, strength):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(61, max(5, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 3.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    return _norm_simplex(a + strength * noise)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: min-norm SSE + LSE refinement + subgrad polish."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm_simplex(p)
    except Exception:
        pass

    # Target length
    if parent is not None:
        target_n = max(len(parent), 8192)
        target_n = 1 << (target_n - 1).bit_length()
        target_n = min(target_n, 32768)
    else:
        target_n = 16384

    print(f"Parent: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    # Generate diverse initializations
    inits = []
    if parent is not None:
        p_resampled = _upsample(parent, target_n)
        inits.append(('parent', p_resampled))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))
        inits.append(('perturb_01', _smooth_perturb(p_resampled, rng, 0.1)))
        inits.append(('perturb_03', _smooth_perturb(p_resampled, rng, 0.3)))

    for i in range(5):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # Power-law inits
    x = np.linspace(0, 1, target_n)
    for alpha in (0.3, 0.7, 1.3):
        a = (1.0 - x) ** alpha + 0.01
        inits.append((f'pow_{alpha}', _norm_simplex(a)))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm_simplex(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))
    scored.sort(key=lambda x: x[0])

    print(f"Top inits: " + ", ".join(f"{n}={s:.6f}" for s, n, _ in scored[:5]))

    best_score = scored[0][0] if scored else 2.0
    best_seq = scored[0][2].copy() if scored else _norm_simplex(np.ones(target_n))

    # Time allocation
    total = deadline - t_start
    phase1_end = t_start + 0.55 * total
    phase2_end = t_start + 0.80 * total
    phase3_end = deadline

    # Phase 1: SSE on top 2-3 initializations
    n_inits = min(3, len(scored))
    per_init = (phase1_end - t_start) / max(1, n_inits)

    for ii in range(n_inits):
        sub_end = t_start + (ii + 1) * per_init
        sub_end = min(sub_end, phase1_end)
        name, arr = scored[ii][1], scored[ii][2]
        print(f"\n=== SSE [{name}] init_score={scored[ii][0]:.8f}, n={target_n} ===")

        sc, seq = _sse_optimize(arr, sub_end, name, use_momentum=True)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"  *** Global best updated: {best_score:.8f} ***")
        print(f"  SSE [{name}] final: {sc:.8f}")

    # Phase 2: LSE refinement on best
    print(f"\n=== LSE refinement, current best={best_score:.8f} ===")
    sc, seq = _lse_refine(best_seq, phase2_end, 'LSE')
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  *** Global best updated: {best_score:.8f} ***")

    # Phase 3: Final SSE + subgrad polish
    print(f"\n=== Final polish, current best={best_score:.8f} ===")
    mid_deadline = phase3_end - (phase3_end - phase2_end) * 0.5
    sc, seq = _sse_optimize(best_seq, mid_deadline, 'final_sse', use_momentum=False)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Final SSE: {best_score:.8f}")

    sc, seq = _subgrad_polish(best_seq, 50, phase3_end)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"  Final subgrad: {best_score:.8f}")

    # Prepare output
    out = _norm_simplex(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
```
