    h = np.fft.irfft(AA_rev * PA, M_fft)[:2*n-1][n-1:2*n-1]

    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Novel approaches: adaptive LSE-beta scheduling, multiplicative weight updates
for peak targeting, symmetry exploitation, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: g_m = 2n * (h_m / S^2 - 2L / S^3)
    # where h_m = sum_j a_j * p_{m+j} (the "cross" term)
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_annealed(a, n_iters=500):
    """Fienup with annealed flattening: aggressive early, gentle late."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Annealed parameter: start aggressive (high alpha), end gentle
        alpha = 0.8 * (1.0 - progress) ** 2 + 0.05
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def peak_targeted_cd(a, n_sweeps=20, n_top=200):
    """
    Coordinate descent targeting top peaks with finer granularity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        S = float(a.sum())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all peaks above 99% of max
        c_max = c.max()
        active_peaks = np.where(c >= 0.99 * c_max)[0]

        # For each active peak, adjust the top-contributing elements
        for k in active_peaks[:10]:  # Limit to 10 peaks
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            total = contribs.sum()
            if total < 1e-15:
                continue

            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global[:50]:  # Further limit
                a_old = a[j]
                if a_old < 1e-12:
                    continue
                # Try several multipliers
                best_val = a_old
                best_local = sc
                for f in [0.6, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2]:
                    cv = a_old * f
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < best_local:
                        best_local = sc_tmp
                        best_val = cv
                a[j] = best_val

        S = a.sum()
        if S > 1e-12:

def alternating_projection(a, n_iters=2000, decay=0.999):

    With "repair" smoothing in time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    strength = 0.6

    for it in range(n_iters):
        # Frequency domain step: flatten spectrum
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_max = P.max()
        P_min = P[P > 1e-30].min() if (P > 1e-30).any() else 1e-30

        # Adaptive flattening: stronger for high peaks, preserve low values
        threshold = P_max * (0.5 + 0.3 * (it / n_iters))
        P_mod = P.copy()
        above = P > threshold
        if above.any():
            P_mod[above] = threshold * (1.0 + 0.5 * (P[above] - threshold) / (P_max - threshold + 1e-30))
        P_mod = np.maximum(P_mod, 1e-20)

        scale = np.sqrt(P_mod / (P + 1e-30))
        A_new = A * scale ** strength
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Time domain step: non-negativity + smoothing + normalization
        a_new = np.maximum(a_new, 0.0)
        # Light Gaussian smoothing to repair artifacts
        if n > 8:
            kernel = np.exp(-0.5 * (np.arange(-3, 4) / 2.0) ** 2)
            kernel /= kernel.sum()
            a_smooth = np.convolve(a_new, kernel, mode='same')
            a_new = 0.7 * a_new + 0.3 * a_smooth
            a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Reduce strength if not improving
            strength *= 0.995

        strength = max(strength, 0.05)

    return best_a


def try_different_length(parent, target_n, rng):
    """Interpolate parent to a different length and re-optimize briefly."""
    if len(parent) == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def adaptive_lbfgs_schedule(a, deadline, t_start, total_budget):
    """
    Adaptive L-BFGS: start with moderate beta, increase as we get closer
    to the optimum of each smoothing level.
    """
    best_a = a.copy()
    best_sc = compute_score(a)
    time_budget = 0.35 * (total_budget - 60.0)

    # Beta schedule: increasing
    betas = [100, 300, 800, 2000, 5000, 10000, 20000, 50000, 100000]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate more iterations to later (higher beta) runs
        frac = (i + 1) / len(betas) ** 1.5
        maxiter = int(min(30000, remaining * 20 * frac + 1000))
        maxiter = max(maxiter, 2000)

        print(f"    L-BFGS beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s)")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 25)
        dt = time.monotonic() - t0
        print(f"      score={sc:.8f} ({dt:.1f}s)")

        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


def symmetry_exploit(a):
    """
    If the sequence is close to symmetric, enforce symmetry and optimize
    only the first half. This can help escape local minima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    # Check symmetry
    a_sym = 0.5 * (a + a[::-1])
    sym_error = np.linalg.norm(a - a_sym) / (np.linalg.norm(a) + 1e-30)

    if sym_error < 0.3:  # Close enough to symmetric
        # Enforce symmetry and optimize the half
        half = (n + 1) // 2
        x = a_sym[:half].copy()
        x = np.maximum(x, 0.0)
        x = x / x.sum()

        best_x = x.copy()
        best_sc = compute_score(_expand_symmetric(x, n))

        # Gradient descent on x
        for it in range(2000):
            a_full = _expand_symmetric(x, n)
            S = a_full.sum()
            if S < 1e-12:
                break
            a_full = a_full / S
            c = fast_conv(a_full)
            c_max = c.max()
            sc = 2.0 * n * c_max / (S * S)

            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

            # LSE gradient for the full sequence
            beta = 500
            scaled = beta * (c - c_max)
            scaled = np.clip(scaled, -500, 500)
            exp_s = np.exp(scaled)
            p = exp_s / exp_s.sum()
            L = c_max + np.log(exp_s.sum()) / beta

            # Gradient w.r.t. full a
            S2 = S * S
            S3 = S2 * S
            a_rev = a_full[::-1]
            M_fft = 1 << (2 * n - 1).bit_length()
            PA = np.fft.rfft(p, M_fft)
            AA = np.fft.rfft(a_rev, M_fft)
            h = np.fft.irfft(AA * PA, M_fft)[:2 * n - 1][n - 1:2 * n - 1]
            grad_full = 2.0 * n * (h / S2 - 2.0 * L / S3)

            # Project gradient onto symmetric subspace
            grad_sym = 0.5 * (grad_full[:half] + grad_full[:half][::-1])

            # Step
            step = 0.01 / (np.linalg.norm(grad_sym) + 1e-10)
            x_new = x - step * grad_sym
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new

        best_a = _expand_symmetric(best_x, n)
        best_a = best_a / best_a.sum()
        return best_a
    else:
        return a


def _expand_symmetric(x, n):
    """Expand a symmetric half to full sequence."""
    half = (n + 1) // 2
    a = np.zeros(n)
    a[:half] = x
    a[half:] = x[:n - half][::-1]
    return a


def stochastic_peek(a, rng, n_restarts=10, deadline=None):
    """
    Random perturbations + quick L-BFGS to escape local minima.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(n_restarts):
        if deadline and time.monotonic() > deadline:
            break

        # Various perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Gaussian noise
            a_pert = a * (1.0 + 0.03 * rng.standard_normal(n))
        elif strategy == 1:
            # Peak targeting: increase elements that contribute to peaks
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            contribs = a_pert[j_start:j_end + 1] * a_pert[j_start + (k_star - j_start) - (k_star - j_start - (j_end - j_start)):j_end + 1][::-1]
            # Just add noise concentrated at peak-contributing indices
            noise = rng.standard_normal(n) * 0.02
            noise[j_start:j_end + 1] *= 2.0
            a_pert = a * (1.0 + noise)
        elif strategy == 2:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.05 * np.cos(2 * np.pi * freq * x + phase))
        else:
            # Block perturbation
            block_size = max(1, n // 20)
            block = rng.integers(0, n - block_size)
            a_pert = a.copy()
            a_pert[block:block + block_size] *= (1.0 + 0.1 * rng.standard_normal(block_size))

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Stochastic restart {r}: {sc:.8f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization with novel strategies:
    adaptive L-BFGS, multiplicative peak reduction, alternating projections
    with repair, symmetry exploitation, and stochastic restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform")

    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Fienup variants (10% of time)
    # ==================================================================
    p1_end = t_start + 0.10 * (total_budget - 60.0)
    print(f"\n=== Phase 1: Fienup (until {p1_end - t_start:.0f}s) ===")

    t_check = time.monotonic()
    a_f = fienup_annealed(best_a, n_iters=400)
    sc = compute_score(a_f)
    update(sc, a_f, "FIENUP-annealed")

    if time.monotonic() < p1_end:
        a_f2 = fienup_spectral_shape(best_a, n_iters=300)
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-shape")

    # ==================================================================
    # Phase 2: Adaptive L-BFGS (35% of time) - main workhorse
    # ==================================================================
    p2_end = t_start + 0.45 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Adaptive L-BFGS (until {p2_end - t_start:.0f}s) ===")

    a_lb, sc_lb = adaptive_lbfgs_schedule(best_a, p2_end, t_start, total_budget)
    update(sc_lb, a_lb, "ADAPTIVE-LBFGS")

    # ==================================================================
    # Phase 3: Multiplicative peak reduction + Alternating projection (20%)
    # ==================================================================
    p3_end = t_start + 0.65 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Multiplicative + Alt Proj (until {p3_end - t_start:.0f}s) ===")

    if time.monotonic() < p3_end:
        a_mp = multiplicative_peak_reduction(best_a, n_iters=3000, top_k=3, eta_init=0.03)
        sc = compute_score(a_mp)
        update(sc, a_mp, "MULTI-PEAK")

    if time.monotonic() < p3_end:
        a_ap = alternating_projection(best_a, n_iters=2000)
        sc = compute_score(a_ap)
        update(sc, a_ap, "ALT-PROJ")

    # ==================================================================
    # Phase 4: Coordinate descent + polish (15%)
    # ==================================================================
    p4_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Coord descent (until {p4_end - t_start:.0f}s) ===")

    if time.monotonic() < p4_end:
        a_cd = peak_targeted_cd(best_a, n_sweeps=20, n_top=150)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")

    # L-BFGS polish after CD
    if sc < best_score and time.monotonic() < p4_end:
        sc2, a2 = run_lbfgs(a_cd, 10000, maxiter=15000, maxfun=300000)
        update(sc2, a2, "CD+POLISH")

    # ==================================================================
    # Phase 5: Stochastic restarts + symmetry (10%)
    # ==================================================================
    p5_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Restart + Symmetry (until {p5_end - t_start:.0f}s) ===")

    a_sr, sc_sr = stochastic_peek(best_a, rng, n_restarts=12, deadline=p5_end)
    update(sc_sr, a_sr, "STOCHASTIC")

    if time.monotonic() < p5_end:
        a_sym = symmetry_exploit(best_a)
        sc_sym = compute_score(a_sym)
        update(sc_sym, a_sym, "SYMMETRY")

    # ==================================================================
    # Phase 6: Final high-beta L-BFGS + CD (5%)
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = int(min(20000, remaining * 15))
        sc, a_fin = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_fin, "FINAL-LBFGS-50k")

    if time.monotonic() < deadline - 20:
        sc, a_fin = run_lbfgs(best_a, 200000, maxiter=10000, maxfun=200000)
        update(sc, a_fin, "FINAL-LBFGS-200k")

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_cd = peak_targeted_cd(best_a, n_sweeps=10, n_top=200)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
