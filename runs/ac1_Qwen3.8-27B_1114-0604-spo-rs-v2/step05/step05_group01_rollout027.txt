"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using active-set exponentiated gradient with adaptive step sizes,
multi-grid initialization, and diverse perturbations.
"""

import numpy as np
import time
import math

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute (a*w) correlation for the overlapping part."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    # We want h_i = sum_k w_k * a_{k-i} for i = 0,...,n-1
    # This is the cross-correlation
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)  # reverse a for correlation
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n-1:2*n-1]

def normalize_a(a, target=1.0):

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using adaptive active-set exponentiated gradient with backtracking line search,
variance-reduction equalization, LSE-LBFGS refinement, and stagnation-based
diversification. Key differences from prior approach: two-timescale active set
management, geometric step-size search, convolution equalization via top-m
variance minimization, and periodic annealing to escape local flat spots.

import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_active_weights_conv(conv, top_k=10, eps_frac=0.01):
    """Compute active weights: uniform over top-k or positions above threshold."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    # Cap the active set size to avoid dilution
    max_count = max(min_count, int(0.05 * conv.size))
    if active.sum() > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active[idx] = True
        # Re-apply: only keep top max_count
        active2 = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active2[top_idx] = True
        active = active2
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_update(a, conv, top_k, eps_frac, eta):
    """Perform one exponentiated gradient step."""
    w = compute_active_weights_conv(conv, top_k=top_k, eps_frac=eps_frac)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def variance_reduction_update(a, conv, top_m, eta):
    """
    Equalize the top-m convolution values by applying pressure proportional
    to (conv[k] - mean_top_m).
    """
    d = np.sort(conv)[::-1][:top_m]
    mean_top = float(np.mean(d))
    # Weight: how much each position exceeds the mean of top-m
    excess = np.maximum(conv - mean_top, 0.0)
    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def active_set_optimize(a, deadline, verbose=True, tag=""):
    """
    Main active-set exponentiated gradient optimizer with adaptive parameters.
    Two-timescale: outer loop adjusts top_k and eps, inner loop does updates.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    # Adaptive parameters
    top_k = 5
    eps_frac = 0.05
    eta = 0.3
    stagnant = 0
    max_stagnant = 30

    for phase in range(20):
        if time.monotonic() > deadline:
            break
        phase_dead = min(deadline, t0 + (deadline - t0) * (phase + 1) / 20.0 + 5.0)

        for it in range(200):
            if time.monotonic() > phase_dead - 2.0:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))  # since sum(a)=1

            # Try multiple step sizes
            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            etas = [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0]
            for eta_try in etas:
                if eta_try < 0.001:
                    continue
                new_a = exp_grad_update(a, conv, top_k, eps_frac, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            # Also try variance reduction
            vr_a = variance_reduction_update(a, conv, min(20, top_k + 10), best_eta * 0.5)
            vr_a = normalize_a(vr_a)
            vr_sc = exact_score(vr_a)
            if vr_sc < best_sc_iter:
                best_sc_iter = vr_sc
                best_a_iter = vr_a

            if best_sc_iter < best_sc - 1e-10:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter
                eta = best_eta
                stagnant = 0
                if verbose and it % 50 == 0:
                    print("  [{} p{} i{}] score={:.9f} top_k={} eps={:.3f} eta={:.3f}".format(
                        tag, phase, it, best_sc, top_k, eps_frac, eta))
            else:
                stagnant += 1
                eta *= 0.85
                if stagnant >= max_stagnant:
                    break

        # Adapt active set parameters
        if stagnant >= max_stagnant:
            # Increase active set to escape
            top_k = min(int(top_k * 1.5) + 2, 100)
            eps_frac = min(eps_frac * 1.3, 0.15)
            eta = max(0.1, eta * 0.7)
            stagnant = 0
            if verbose:
                print("  [{}] expand: top_k={}, eps={:.3f}".format(tag, top_k, eps_frac))
        else:
            # Gradually tighten
            top_k = max(3, int(top_k * 0.95))
            eps_frac = max(0.005, eps_frac * 0.95)

    return float(best_sc), best_a


def lse_score_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxfun=500):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent, cap=25):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    # V-shapes (high at center)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    # U-shapes (high at ends)
    for alpha in (0.3, 0.5, 0.8, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    # Cosine
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    # Sine absolute
    for f in (1, 2, 3):
        for alpha in (0.5, 1.0, 1.5):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(x - 0.5)) + 0.01)
    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2):
        for center in (0.2, 0.3, 0.4):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Triangular
    for shift in (0.0, 0.1, -0.1):
        add(np.maximum(0.0, 1.0 - np.abs(2.0 * x + shift - 1.0)) + 0.01)
    # Power shapes
    for p in (0.5, 1.0, 2.0, 3.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    # Random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Active-set guided perturbation
        conv = fast_self_conv(p)
        w = compute_active_weights_conv(conv, top_k=10, eps_frac=0.02)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        # Smooth noise in log space
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        # V-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        # U-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        # Block perturbation
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        # Multiplicative random
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def subgrad_fine_polish(a, deadline, verbose=False, tag=""):

    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
