One more idea: **Use the DFT to directly control the power spectrum.**

Since (a*a) = IFT(|DFT(a)|²), we can think of this in the frequency domain. We want |DFT(a)|² to be as flat as possible (to make the IFT as flat as possible). But a is real and non-negative, which constrains the DFT.

def fast_conv(a):
    """Linear self-convolution via FFT. Length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]

    """Exact score: 2n * max(conv) / sum^2."""
    n = len(a)

    return 2.0 * n * c.max() / (S * S)

def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()

        return 1e10, np.zeros(n)

    p = e / e.sum()
    
    # Gradient: dL/da[i] = 2 * sum_k p[k] * a[k-i] (with a[j]=0 for j out of [0,n))

    """Perturb by targeting the peak lags of the autocorrelation."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    
    # Find top peak lags
    top_idx = np.argsort(c)[::-1][:n_top_lags]
    
    a_p = a.copy()
    for k in top_idx:
        # Find coordinates that most contribute to lag k
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        idx = np.arange(j_start, j_end + 1)
        contrib = a[idx] * a[k - idx]

        mid = len(idx) // 2

        order = np.argsort(contrib)[::-1]
        top = order[:top_n]
        bot = order[-top_n:]

        a_p[idx[bot]] += reduce_amt * len(top) / len(bot)

    S = a_p.sum()
