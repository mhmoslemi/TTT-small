    sequence = [min(1000.0, x) for x in sequence]
    
    conv = np.convolve(sequence, sequence, mode='full')
    max_conv_val = np.max(conv)
    sum_val = np.sum(sequence)
    
    # Avoid division by zero or near overflow scenarios
    if sum_val < 0.01 or max_conv_val < 0.01 or np.isnan(sum_val) or np.isinf(sum_val):
        return np.inf
        
    return float(2.0 * n * max_conv_val / (sum_val ** 2))

def linear_convolution(x):
    """Linear convolution of a sequence for efficient computation."""
    n = len(x)
    M = 2*n-1
    if M > 0:
        # Find the next power of two for FFT
        M_pad = 1 << (M.bit_length())
        X = np.asarray(x, dtype=np.float64)
        # Compute the FFT and multiply
        FX = rfft(X, M_pad)
        c = irfft(FX * FX, M_pad)[:M]
    else:
        c = np.array([])
        
    return c

def fienup_step(a, n_iters=100, mode='flatten'):
    """Implements the Fienup method with either 'flatten' or 'quantile' approach."""
    n_len = len(a)
    a_work = np.clip(a.astype(np.float64), 0, 1000).copy()
    sum_work = np.sum(a_work)
    
    if sum_work < 1e-6:
        return a_work
        
    # Normalize to sum=1 if not possible in the optimal mode
    a_work = a_work / sum_work
    
    # FFT size for the convolution
    M = 2*n_len-1
    M_fft = 1 << np.ceil(np.log2(M+1)).astype(int)
    
    for it in range(n_iters):
        Ft = rfft(a_work, M_fft)
        P = Ft.real**2 + Ft.imag**2
        target = np.mean(P) if mode == 'flatten' else np.quantile(P, 0.8)
        scale = (np.log(target)) ** 0.5
        
        # Update based on the Fienup idea
        a_work = a_work * scale
        
    # Avoid clipping issues with bounds
    a_work = np.clip(a_work, 0, 1000) if any(a_work < 0) else None
    return np.divide(a_work, np.sum(a_work)/1e6)  # Scale to keep average reasonable
