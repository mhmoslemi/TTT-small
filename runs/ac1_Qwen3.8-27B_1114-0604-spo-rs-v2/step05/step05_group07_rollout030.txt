Subject to: `sum(g) = 1`, `g >= 0`, `conv(g) <= M * 1`

This is NOT an LP (because `conv(g)` is quadratic in `g`), but it IS a non-convex QCQP. However, if we fix `f` and optimize over `g` with the constraint `conv(f,g) <= M`, that IS an LP (linear in `g`).

So the approach is:
1. Fix `f`, find best `g` via LP: `max sum(g) s.t. conv(f,g) <= M, g >= 0`
2. Update `f` towards `g`
3. Repeat

This is essentially a **Zeno's paradox** type iteration (or a fixed-point iteration in a nonlinear setting). The key question is convergence.

The original paper (Busch et al.) shows that this iteration has non-increasing `||f*f||_infty`, which is great for our purposes.

"""
Active-set LP iteration combined with LSE-smoothed L-BFGS refinement
for minimizing 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fast_cross_conv(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # dL/da[i] = 2 * (p * rev_a)[n-1+i]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def build_convolution_row(f, k, n):
    """Build the row of the convolution matrix for lag k.
    Returns (coeffs, indices) such that sum_j f[k-j]*g[j] = coeffs @ g[indices]."""
    j_min = max(0, k - (n - 1))
    j_max = min(k, n - 1)
    if j_max < j_min:
        return None, None
    js = np.arange(j_min, j_max + 1)
    coeffs = f[k - js]
    return coeffs, js

def active_set_lp(f, M, active_lags):
    """Solve the active-set LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for k in active_lags.
    Returns the optimal g."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    constraints = []
    # Build the constraint matrix sparsely
    row_data = []
    col_data = []
    for k in active_lags:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_max < j_min:
            continue
        js = np.arange(j_min, j_max + 1)
        coeffs = f[k - js]
        # Add non-zero entries
        mask = coeffs > 1e-15
        row_data.extend([len(constraints)] * mask.sum())
        col_data.extend(js[mask].tolist())
        # Store for later
        constraints.append(None)  # placeholder
    
    # Actually, let's build a sparse matrix
    from scipy.sparse import lil_matrix
    n_rows = len(active_lags)
    C = lil_matrix((n_rows, n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            coeffs = f[k - js]
            C[i, js] = coeffs
    
    C = C.tocsr()
    constraint = C @ g <= M * np.ones(n_rows)
    constraints = [constraint]
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        # or try ECOS, CLARABEL
        if g.value is not None:
            return np.clip(g.value, 0, None)
    except Exception:
        pass
    
    # Fallback: try with CLARABEL
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            return np.clip(g.value, 0, None)
    except Exception:
        pass
    
    return None

def line_search_lp_update(f, g, alpha_min=1e-6):
    """Find optimal t in [0,1] minimizing score of (1-t)*f + t*g (normalized)."""
    n = len(f)
    S_f = f.sum()
    S_g = g.sum()
    
    def score_at_t(t):
        h = (1 - t) * f + t * g
        S = h.sum()
        if S < 1e-12:
            return np.inf
        c = fast_conv(h)
        return 2.0 * n * float(c.max()) / (S * S)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = score_at_t(c)
    fd = score_at_t(d)
    
    for _ in range(50):
        if b - a < alpha_min:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = score_at_t(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = score_at_t(d)
    
    t_opt = (a + b) / 2
    return t_opt, score_at_t(t_opt)

    S_g = g.sum()

    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Cross-convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


_global_beta = 1000.0


def _lse_objective(x):
    """Wrapper for L-BFGS callback using global beta."""
    return lse_score_and_grad(x, _global_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _global_beta
    _global_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        _lse_objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_active_lp(f, M, active_lags, timeout=30):
    """Solve active-set LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for k in active_lags.
    Returns optimal g or None on failure."""
    n = len(f)
    K = len(active_lags)

    # Build sparse constraint matrix
    C = lil_matrix((K, n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            coeffs = f[k - js]
            C[i, js] = coeffs

    C = C.tocsr()

    g = cp.Variable(n, nonneg=True)
    constraint = C @ g <= M * np.ones(K)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, [constraint])

    t0 = time.monotonic()
    solvers_to_try = [cp.ECOS, cp.CLARABEL, cp.SCS, cp.GLPK, cp.CBC]
    for solver in solvers_to_try:
        try:
            if solver == cp.CLARABEL and not hasattr(cp, 'CLARABEL'):
                continue
            if solver == cp.SCS and not hasattr(cp, 'SCS'):
                continue
            if solver == cp.GLPK and not hasattr(cp, 'GLPK'):
                continue
            if solver == cp.CBC and not hasattr(cp, 'CBC'):
                continue
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                result = np.clip(g.value, 0, None)
                if result.sum() > 1e-12:
                    return result
        except Exception:
            continue
        if time.monotonic() - t0 > timeout:
            break

    return None


def line_search_update(f, g):
    """Find optimal t in [0,1] minimizing score of (1-t)*f + t*g."""
    n = len(f)
    cf = fast_conv(f)
    cfg = fast_cross_conv(f, g)
    cg = fast_conv(g)

    S_f = float(f.sum())
    S_g = float(g.sum())

    def score_at_t(t):
        c = (1.0 - t) ** 2 * cf + 2.0 * t * (1.0 - t) * cfg + t ** 2 * cg
        S = (1.0 - t) * S_f + t * S_g
        if S < 1e-12:
            return np.inf
        return 2.0 * n * float(np.max(c)) / (S * S)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = 0.0
    b = 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = score_at_t(c_pt)
    fd = score_at_t(d_pt)

    for _ in range(60):
        if b - a < 1e-9:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = score_at_t(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = score_at_t(d_pt)

    t_opt = (a + b) / 2.0
    return t_opt, score_at_t(t_opt)


def lp_iteration(f, K=30, M_margin=1.01, max_iters=200, time_budget=600):
    """Active-set LP iteration to reduce max convolution.
    Returns improved sequence."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S

    best_f = f.copy()
    best_sc = compute_score(f)
    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t_start > time_budget:
            break

        c = fast_conv(f)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max / (f.sum() ** 2)

        if current_sc < best_sc:
            best_sc = current_sc
            best_f = f.copy()

        # Select active set: top K peaks
        sorted_idx = np.argsort(c)[::-1]
        active_lags = sorted_idx[:K].tolist()

        # Solve LP with margin
        M = c_max * M_margin
        g = solve_active_lp(f, M, active_lags, timeout=20)

        if g is None:
            # Fallback: use gradient-based step
            break

        g = np.clip(g, 0, None)
        S_g = g.sum()
        if S_g < 1e-12:
            break
        g = g * (S / S_g)

        # Line search
        t_opt, sc_new = line_search_update(f, g)

        if sc_new >= current_sc:
            # Try smaller step
            t_try = t_opt * 0.5
            f_new = (1 - t_try) * f + t_try * g
            sc_try = compute_score(f_new)
            if sc_try < current_sc:
                f = f_new / f_new.sum()
            else:
                # No improvement, try reducing K
                K = max(10, K // 2)
                continue
        else:
            f = (1 - t_opt) * f + t_opt * g
            f = f / f.sum()

        # Periodic progress check
        if it % 20 == 0:
            print(f"    LP iter {it}: score={compute_score(f):.10f}, best={best_sc:.10f}, K={K}")

    return best_f


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=3000, f_lb=1.0, top_k=5):
    """Subgradient using top-k peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.1

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(c))
        for rank in range(top_k):
            weights[top_idx[rank]] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)

        Mak_weighted = float(np.sum(weights * c))
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        S2 = S_a * S_a
        S3 = S2 * S_a
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]

        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (0.999 ** it)
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def targeted_peak_flattening(a, n_sweeps=10, n_top=50):
    """Coordinate descent: reduce peak contributors and redistribute mass."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        top_count = min(n_top, max(1, total_len // 3))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 0.9, 0.95]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    per = reduction / max(1, len(bottom_local))
                    for li in bottom_local:
                        a[j_start + li] += per
                S_curr = a.sum()
                if S_curr > 1e-12:
                    a = a / S_curr
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S_curr = a.sum()
            if S_curr > 1e-12:
                a = a / S_curr

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.9995):
    """Mirror descent with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = int(rng.integers(0, 6))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.05, np.cos(np.pi * x)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.exp(-3.0 * (x - 0.5) ** 2))
    # Asymmetric triangular
    inits.append(np.maximum(0.1, 1.0 - np.abs(2.0 * x - 0.3)))
    # Two-peak
    inits.append(0.3 + 0.7 * (np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: LP-based iteration -> LSE L-BFGS refinement -> subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}, score: {compute_score(parent/parent.sum()):.10f}")
        parent_norm = parent / parent.sum()
        report(compute_score(parent_norm), parent_norm, "parent")
    else:
        n_work = 8192

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: Multi-start with quick LSE optimization (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Multi-Start Initialization ---")

    inits = build_initializations(n_work, rng, parent if parent is not None and len(parent) == n_work else None)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S

        sc_init = compute_score(a_init)
        report(sc_init, a_init, f"init_{i}")

        # Quick LSE optimization
        if time.monotonic() < phase1_deadline - 10:
            sc, a_opt, _ = run_lbfgs(a_init, beta=100, maxiter=5000, maxfun=20000)
            report(sc, a_opt, f"lse100_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: Active-set LP iteration (35% time)
    # ==================================================================
    phase2_deadline = t_start + 0.50 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Active-Set LP Iteration ---")

    # Try different K values
    K_schedule = [20, 30, 50, 80, 120, 200]
    for K in K_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        time_for_lp = min(remaining - 5, 120)
        a_lp = lp_iteration(best_a, K=K, M_margin=1.01,
                            max_iters=500, time_budget=time_for_lp)
        sc = compute_score(a_lp)
        report(sc, a_lp, f"LP_K={K}")

        # Quick LSE polish after LP
        if time.monotonic() < phase2_deadline - 20:
            sc2, a_opt2, _ = run_lbfgs(a_lp, beta=500, maxiter=10000, maxfun=50000)
            report(sc2, a_opt2, f"LP+LSE_K={K}")

    # Additional LP passes with best so far
    if time.monotonic() < phase2_deadline:
        for K in [50, 100]:
            if time.monotonic() > phase2_deadline - 10:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 20:
                break
            a_lp = lp_iteration(best_a, K=K, M_margin=1.005,
                                max_iters=500, time_budget=remaining - 5)
            sc = compute_score(a_lp)
            report(sc, a_lp, f"LP2_K={K}")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: LSE + L-BFGS with increasing beta (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: LSE L-BFGS Refinement ---")

    beta_schedule = [200, 500, 1000, 3000, 10000, 30000, 100000, 300000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(80000, max(5000, remaining * 50)))
        maxfun = int(min(300000, max(20000, remaining * 200)))

        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LSE b={beta}")

        # Perturbed restart for higher betas
        if time.monotonic() < phase3_deadline - 60 and beta >= 1000:
            for pert_i in range(2):
                if time.monotonic() > phase3_deadline - 30:
                    break
                scale = 0.01 + 0.02 * rng.random()
                a_pert = perturbed_restart(best_a, rng, scale=scale)
                maxiter2 = int(min(30000, max(2000, remaining * 15)))
                maxfun2 = int(min(100000, max(5000, remaining * 60)))
                sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter2, maxfun=maxfun2)
                report(sc2, a_opt2, f"LSE b={beta} p{pert_i}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated local search (10% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 8
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 60:
            break

        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Mirror descent for exploration
        a_md = mirror_descent(a_pert, beta=500, n_iters=2000, eta0=0.03)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"ILS#{trial} md")

        # L-BFGS refinement
        beta = 1000 + int(5000 * rng.random())
        maxiter = int(min(30000, max(3000, remaining * 10)))
        maxfun = int(min(100000, max(10000, remaining * 40)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-10), beta,
                                  maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} lbfgs b={beta}")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final polishing (10% time)
    # ==================================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final Polishing ---")

    # Nesterov subgradient
    if time.monotonic() < phase5_deadline - 30:
        a_ns = nesterov_subgrad(best_a, n_iters=5000, f_lb=1.0)
        sc = compute_score(a_ns)
        report(sc, a_ns, "nesterov")

    # Multi-peak subgradient
    if time.monotonic() < phase5_deadline - 30:
        a_mp = multi_peak_subgrad(best_a, n_iters=3000, f_lb=1.0, top_k=8)
        sc = compute_score(a_mp)
        report(sc, a_mp, "multi_peak")

    # Targeted peak flattening
    if time.monotonic() < phase5_deadline - 30:
        a_tpf = targeted_peak_flattening(best_a, n_sweeps=15, n_top=80)
        sc = compute_score(a_tpf)
        report(sc, a_tpf, "peak_flat")

    # Another Nesterov on updated best
    if time.monotonic() < phase5_deadline - 30:
        a_ns2 = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "nesterov2")

    # Final high-beta L-BFGS
    if time.monotonic() < phase5_deadline - 20:
        for beta in [500000, 1000000]:
            if time.monotonic() > phase5_deadline - 15:
                break
            maxiter = int(min(50000, max(5000, (phase5_deadline - time.monotonic()) * 30)))
            maxfun = int(min(200000, max(20000, (phase5_deadline - time.monotonic()) * 100)))
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            report(sc, a_opt, f"final b={beta}")

    # One more peak flattening
    if time.monotonic() < phase5_deadline - 15:
        a_tpf2 = targeted_peak_flattening(best_a, n_sweeps=10, n_top=100)
        sc = compute_score(a_tpf2)
        report(sc, a_tpf2, "peak_flat_final")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
