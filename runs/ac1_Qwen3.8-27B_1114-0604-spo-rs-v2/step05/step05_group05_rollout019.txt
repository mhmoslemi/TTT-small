Optimized subgradient method with BB step size, Nesterov acceleration,
and adaptive top-K smoothing for minimizing the peak autocorrelation
of non-negative sequences.

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def top_k_gradient(a, c, K, S, c_max):
    """
    Compute gradient of the score using top-K peak approximation.
    Returns the gradient of score = 2n * max(c) / S^2.
    For top-K: score_K = 2n * (sum of top-K c values) / (K * S^2)
    Gradient: 4n/(K*S^2) * sum_{k in topK} d(c_k)/d(a) - 4n*sum_topK/(K*S^3) * ones
    """

    # Get top-K indices
    if K >= len(c):
        top_indices = np.arange(len(c))
    else:

    c_top_sum = float(c[top_indices].sum())
   
    # For each top index k, d(c_k)/d(a_i) = 2*a_{k-i} for valid i

    score_grad = (2.0 * n) / (K * S**3) * (S * grad - 2.0 * c_top_sum)
    return score_grad

                lr_bb = np.clip(lr_bb, 1e-6, 0.1)
                lr = 0.5 * lr + 0.5 * lr_bb

                lr = 0.5 * lr + 0.5 * lr_bb

        if S_look < 1e-12:

        # Update
        v = mu * v - lr * score_grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a_new = a_new / S_new
        
        prev_a = a.copy()
        prev_grad = score_grad.copy()
        a = a_new

        lr *= lr_decay
        
        # Track best

                best_a = a.copy()

"""
Hybrid optimizer: Nesterov top-K subgradient + BB step + LP polish + LSE L-BFGS.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 for non-negative sequences.
Key innovations: (1) exact subgradient with Barzilai-Borwein step, (2) periodic
linear programming polish on the active peak set, (3) Fourier perturbation init.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def subgrad_topk(a, c, K, S):
    """Compute gradient of 2n*(sum top-K c)/S^2 with respect to a."""
    n = len(a)
    if K >= len(c):
        top_idx = np.arange(len(c))
        c_top_sum = float(c.sum())
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top_sum = float(c[top_idx].sum())
    
    # d(c_top_sum)/d(a_i) = sum_{k in topK} 2*a_{k-i} (for valid k-i)
    grad_c = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            j_start = k - i_end
            j_end = k - i_start
            grad_c[i_start:i_end+1] += 2.0 * a[j_start:j_end+1][::-1]
    
    # score = 2n * c_top_sum / S^2
    # d(score)/d(a_i) = 2n * (S * grad_c[i] - 2*c_top_sum) / S^3
    score_grad = (2.0 * n / S**3) * (S * grad_c - 2.0 * c_top_sum)
    return score_grad, c_top_sum


def lse_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient of L with respect to a
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lp_polish(a, c, K=10, alpha=0.3):
    """Solve LP to minimize max of top-K peaks (linearized). Returns improved a."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    
    # Work with normalized a (sum=1)
    a_norm = a / S
    
    top_idx = np.argpartition(c, -K)[-K:]
    c_top = c[top_idx]
    
    # Constraints: sum_i 2*a_norm[k-i] * x_i <= t + c_k(a) for k in topK
    # sum(x) = 1, x >= 0
    # Variables: x_0,...,x_{n-1}, t (total n+1 variables)
    
    # Build A matrix (K rows, n+1 cols)
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for idx, k in enumerate(top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            j_start = k - i_end
            j_end = k - i_start
            A_ub[idx, i_start:i_end+1] = 2.0 * a_norm[j_start:j_end+1][::-1]
        A_ub[idx, -1] = -1.0  # coefficient of t
        b_ub[idx] = float(c[int(k)])
    
    # Equality: sum x = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x_opt = res.x[:n]
            x_opt = np.maximum(x_opt, 0.0)
            S_opt = float(x_opt.sum())
            if S_opt > 1e-12:
                x_opt = x_opt / S_opt
                # Blend
                a_new = (1 - alpha) * a_norm + alpha * x_opt
                a_new = np.maximum(a_new, 0.0)
                S_new = float(a_new.sum())
                if S_new > 1e-12:
                    return a_new / S_new
    except Exception:
        pass
    
    return a.copy()


def fienup_quick(a, n_iters=50, thresh=1.5, clip=0.5):
    """Quick Fienup iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        above = P > P_mean * thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def fourier_init(n, M=20, seed=42):
    """Initialize with uniform + Fourier perturbation."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n) / n
    
    # Add small sinusoidal perturbations
    eps = 0.003
    for m in range(M):
        freq = rng.integers(1, n//2 + 1)
        phase = rng.uniform(0, 2*np.pi)
        amp = eps / (m + 1) * rng.uniform(0.5, 1.0)
        a += amp * np.cos(2*np.pi * freq * i / n + phase)
    
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def nesterov_main(a, max_time, K_start=3, K_end=200, 
                  mu_start=0.85, mu_end=0.97,
                  lr_init=0.01, lr_decay=0.9997,
                  lp_every=500, lp_K=10, lp_alpha=0.3,
                  restart_patience=3000):
    """Main Nesterov optimization loop."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    
    mu = mu_start
    lr = lr_init
    prev_a = a.copy()
    prev_grad = None
    it = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = (time.monotonic() - t_start) / max_time
        
        # Adaptive K
        K = int(K_start + (K_end - K_start) * t_frac)
        K = max(1, K)
        
        # Compute convolution
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        
        # Compute gradient
        score_grad, c_top_sum = subgrad_topk(a, c, K, S)
        
        # BB step size
        if prev_grad is not None and it > 1:
            s = a - prev_a
            y = score_grad - prev_grad
            sy = float(np.dot(s, y))
            ss = float(np.dot(s, s))
            yy = float(np.dot(y, y))
            if yy > 1e-15 and sy > 1e-15:
                lr_bb = sy / yy
                lr_bb = float(np.clip(lr_bb, 1e-6, 0.05))
                lr = 0.7 * lr + 0.3 * lr_bb
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            a = best_a.copy()
            break
        a_look_norm = a_look / S_look
        
        # Use gradient at lookahead (every other iteration for speed)
        if it % 2 == 0:
            c_look = fast_conv(a_look_norm)
            score_grad, _ = subgrad_topk(a_look_norm, c_look, K, 1.0)
        
        # Update
        v = mu * v - lr * score_grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a_new = a_new / S_new
        
        prev_a = a.copy()
        prev_grad = score_grad.copy()
        a = a_new
        
        # Schedule
        mu = mu_start + (mu_end - mu_start) * t_frac
        lr *= lr_decay
        lr = max(lr, 1e-5)
        
        # LP polish periodically
        if it % lp_every == 0:
            c_current = fast_conv(a)
            a_lp = lp_polish(a, c_current, K=lp_K, alpha=lp_alpha)
            sc_lp = compute_score(a_lp)
            if sc_lp < best_sc:
                best_sc = sc_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                v = np.zeros(n)  # Reset momentum
                no_improve = 0
            else:
                no_improve += lp_every
        
        # Track best
        if it % 1000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1000
        
        # Restart if stuck
        if no_improve > restart_patience:
            rng = np.random.default_rng(it)
            a = a * (1.0 + 0.005 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = lr_init
            mu = mu_start
            no_improve = 0
            print(f"    [restart] it={it}, best={best_sc:.8f}")
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    
    return best_a


def lse_lbfgs_polish(a, beta, maxiter=2000):
    """L-BFGS polish on LSE-smoothed objective."""
    n = len(a)
    a0 = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    def objective(x):
        # LSE score and gradient (with normalization constraint handled by projection)
        x = np.maximum(x, 0.0)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(len(x))
        x = x / S
        return lse_grad(x, beta)
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
                            'maxfun': 50000, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = float(a_out.sum())
    if S > 1e-6:
        a_out = a_out / S
    return compute_score(a_out), a_out


def coord_descent_peak(a, n_sweeps=5, n_top=20, factors=(0.5, 0.7, 0.85, 1.15, 1.3)):
    """Coordinate descent on peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end < i_start:
                continue
            
            contrib = np.array([a[k - j] for j in range(i_start, i_end + 1)])
            n_local = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = i_start + top_local
            
            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for factor in factors:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    sc = compute_score(a / S_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = (a / S_tmp).copy()
                        a = best_a.copy()
                        improved = True
                        break
                else:
                    a[j] = old_val
        
        if not improved:
            break
    
    return best_a
