def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

        return sc, g

        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,

    )

    return a_out

def peak_targeted_refinement(a0, n_rounds=50, top_k=20):

def random_perturb(a0, rng, strength=0.05):

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses iterative reweighting, LSE-LBFGS, GS, and peak-targeted refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 45.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None and abs(len(parent) - n) < 100:
        n = len(parent)
    
    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting optimization with n={n}")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: GS warmup on diverse initializations (10% of time)
    # ==================================================================
    phase1_end = t_start + 0.10 * (total_budget - 45.0)
    print(f"\n=== Phase 1: GS warmup (deadline +{phase1_end - t_start:.0f}s) ===")

    inits = build_diverse_inits(n, rng, parent if len(parent) == n else None)
    
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / max(a0.sum(), 1e-12)
        
        a1 = gs_flat_target(a0, n_iters=300, alpha_init=0.5, alpha_final=0.1)
        sc = compute_score(a1)
        update_best(sc, a1)
        
        if time.monotonic() < phase1_end:
            a2 = gs_adaptive_target(a1, n_iters=200)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE-LBFGS with multiple beta values (30% of time)
    # ==================================================================
    phase2_end = t_start + 0.40 * (total_budget - 45.0)
    print(f"\n=== Phase 2: LSE-LBFGS (deadline +{phase2_end - t_start:.0f}s) ===")

    betas = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        t_beta = time.monotonic()
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=8000, maxfun=30000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Iterative Reweighting (30% of time)
    # ==================================================================
    phase3_end = t_start + 0.70 * (total_budget - 45.0)
    print(f"\n=== Phase 3: Iterative Reweighting (deadline +{phase3_end - t_start:.0f}s) ===")

    # Run iterative reweighting from current best
    a_ir = iterative_reweighting(best_a, n_outer=30, n_inner=2000, maxfun_per_inner=3000)
    sc = compute_score(a_ir)
    update_best(sc, a_ir)
    
    # Also try from a perturbed version
    if time.monotonic() < phase3_end:
        a_pert = random_perturb(best_a, rng, strength=0.03)
        a_ir2 = iterative_reweighting(a_pert, n_outer=20, n_inner=1500, maxfun_per_inner=2000)
        sc2 = compute_score(a_ir2)
        update_best(sc2, a_ir2)

    # Continue with LSE refinement after IR
    if time.monotonic() < phase3_end:
        for beta in [10000, 30000, 80000]:
            if time.monotonic() > phase3_end:
                break
            a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=5000, maxfun=20000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak-targeted refinement + Subgradient (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.85 * (total_budget - 45.0)
    print(f"\n=== Phase 4: Peak refinement + Subgradient (deadline +{phase4_end - t_start:.0f}s) ===")

    # Peak targeted refinement
    a_ptr = peak_targeted_refinement(best_a, n_rounds=30, top_k=30)
    sc = compute_score(a_ptr)
    update_best(sc, a_ptr)
    
    # Then LSE on the refined version
    if time.monotonic() < phase4_end:
        a_opt = run_lbfgs(best_a, lse_score_grad, (20000,), maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
    
    # Subgradient with multiple peak settings
    if time.monotonic() < phase4_end:
        for n_peaks in [20, 50, 100]:
            if time.monotonic() > phase4_end:
                break
            a_sg = subgradient_multi_peak(best_a, n_iters=3000, n_peaks=n_peaks)
            sc = compute_score(a_sg)
            update_best(sc, a_sg)

    # Nesterov momentum
    if time.monotonic() < phase4_end:
        a_nm = nesterov_momentum(best_a, beta=5000, n_iters=10000, lr_init=0.005, momentum=0.9)
        sc = compute_score(a_nm)
        update_best(sc, a_nm)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish - interleaved GS + LSE + IR (15% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish (deadline +{deadline - t_start:.0f}s) ===")

    polish_rounds = 5
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break
        
        # GS refinement
        a_gs = gs_flat_target(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.02)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)
        
        # LSE fine-tune
        if time.monotonic() > deadline:
            break
        beta_final = 50000 + 50000 * pr
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta_final,), maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        
        # Quick IR
        if time.monotonic() > deadline:
            break
        a_ir = iterative_reweighting(best_a, n_outer=5, n_inner=1000, maxfun_per_inner=2000)
        sc = compute_score(a_ir)
        update_best(sc, a_ir)
        
        print(f"  Polish round {pr}: best = {best_score:.8f}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
