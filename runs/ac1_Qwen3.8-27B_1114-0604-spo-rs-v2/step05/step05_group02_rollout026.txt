- Minimize `t` subject to `a >= 0, sum(a) = 1, a^T K_k a <= t` for all `k`
- Where `K_k` is the matrix such that `c[k] = a^T K_k a`

Each constraint `a^T K_k a <= t` is a **second-order cone constraint** (if `K_k` is positive semi-definite) or a **non-convex quadratic constraint** (if `K_k` has negative eigenvalues).

For `k = n-1` (the center), `K_{n-1}` is the reversal matrix, which has eigenvalues ±1, so it's indefinite. This means the problem is non-convex.

For `k = 0`, `K_0` is the matrix with a 1 in position `(0,0)` and 0 elsewhere (since `c[0] = a[0]^2`). This is positive semi-definite.

So the problem is a mixed convex/non-convex quadratic optimization. L-BFGS is a reasonable choice.

**Alternative: Semidefinite programming (SDP) relaxation**

We can relax the problem by replacing `a a^T` with a matrix variable `X >= aa^T` (i.e., `X - aa^T` is positive semi-definite... wait, that's not right either).

Actually, the standard SDP relaxation would be:
- Let `X = aa^T` (rank-1, PSD)
- `c[k] = trace(K_k * X)` (linear in X)
- Minimize `max_k trace(K_k * X)` subject to `trace(X) + sum_{i≠j} X[i,j] = 1` (from `sum(a) = 1`)
- `X >= 0`, `rank(X) = 1`

Dropping the rank constraint gives an SDP relaxation. But this is a large SDP (size `n x n` with `n ~ 7000`), which would be intractable.

So let's stick with the gradient-based approach.

3. Peak-directed refinement:

   - Do coordinate descent on the elements that contribute to these peaks
   - Use a "push-pull" strategy: reduce peaks, increase valleys

6. Return the best

"""
Multi-beta LSE continuation with L-BFGS + peak-directed coordinate descent.
Improves upon GS-Fienup by using a principled continuation method on the
log-sum-exp smoothing parameter, combined with aggressive peak-directed
refinement to flatten the convolution.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    Returns (score, gradient) where gradient is w.r.t. a (unnormalized).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        exp_scaled = np.ones(len(c)) / len(c)
        exp_sum = 1.0
    p = exp_scaled / exp_sum
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a
    # dL/da[i] = sum_k p[k] * (a[k-i] + a[i-k]) (valid indices)
    # = 2 * convolve(p, a)[i] adjusted for boundaries
    # Using FFT: convolve(rev_a, p)[n-1 + i] gives sum_m a[m] * p[i+m]
    # and convolve(a, rev_p)[n-1 + i] gives sum_m a[m] * p[m-i] (by symmetry)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    # Full gradient of score = 2n*L/S^2
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_and_grad(x, beta):
    """Callable for scipy.minimize (L-BFGS-B)."""
    return lse_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective. Returns (score, a_opt)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_refinement(a, n_iters=5000, top_m=20, step_init=0.01):
    """
    Subgradient method focused on top-M peaks with adaptive step size.
    Uses the exact subgradient of the max (not LSE approximation).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    patience = 0
    max_patience = 500

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
            # Weight by how close to max
            c_top = c[top_idx]
            weights = (c_top / (c_top + M_val))  # All close to 1
            
        # Compute subgradient
        sg = np.zeros(n)
        for ki in range(len(top_idx)):
            k = int(top_idx[ki])
            w = weights[ki]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        if step < 1e-10:
            break

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999

    return best_a


def nesterov_lse(a0, beta, n_iters=3000, momentum=0.9):
    """
    Nesterov accelerated gradient descent on LSE objective.
    Operates on the simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Adaptive step size
    c = fast_conv(a)
    c_spread = float(c.max() - c.min())
    if c_spread < 1e-12:
        c_spread = 1e-6
    lr = 0.1 / (beta * c_spread + 1.0)
    lr = min(lr, 0.05)

    for it in range(n_iters):
        # Nesterov lookahead
        y = (1 + momentum) * a - momentum * a_prev
        
        # Compute gradient at y
        sc_y, grad_y = lse_grad(y, beta)
        
        # Gradient step
        a_new = y - lr * grad_y
        a_new = np.clip(a_new, 0.0, None)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a_prev = a
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive learning rate
        if it % 100 == 0:
            c = fast_conv(a)
            c_spread = float(c.max() - c.min())
            if c_spread < 1e-12:
                c_spread = 1e-6
            lr = 0.1 / (beta * c_spread + 1.0)
            lr = min(lr, 0.05)
            lr *= (0.995 ** (it // 100))

    return best_a


def coordinate_descent_peaks(a, n_sweeps=3, n_top=50):
    """
    Coordinate descent focusing on elements that contribute to peak convolution values.
    For each peak, try reducing the contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            j = int(j)
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j
            
            # Try multiplicative reductions
            for f in [0.5, 0.7, 0.85, 0.9]:
                cv = a_old_j * f
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = cv
                a[j] = a_old_j
            
            # Try small increases (to redistribute)
            for f in [1.2, 1.5]:
                cv = a_old_j * f
                if cv > 1000:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = cv
                a[j] = a_old_j
            
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

"""
Multi-beta LSE continuation with L-BFGS + Nesterov mirror descent + peak-directed refinement.
Uses a principled continuation method on the log-sum-exp smoothing parameter,
combined with mirror descent on the simplex and aggressive peak-directed
coordinate descent to flatten the auto-convolution.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        exp_scaled = np.ones(len(c)) / len(c)
        exp_sum = 1.0
    p = exp_scaled / exp_sum
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_and_grad(x, beta):
    """Callable for scipy.minimize L-BFGS-B."""
    return lse_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (exact_score, a_opt)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def mirror_descent_lse(a0, beta, n_iters=2000, lr_init=0.02):
    """
    Multiplicative weights (mirror descent) on the simplex for LSE objective.
    Update: a_new[i] prop to a[i] * exp(-lr * grad[i])
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        sc, grad = lse_grad(a, beta)
        # Scale gradient for stability
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        scaled_lr = lr / (g_norm + 1e-10)
        # Multiplicative update
        log_update = -scaled_lr * grad
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = a * np.exp(log_update)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        # Decay learning rate
        lr *= 0.998

    return best_a


def peak_directed_refinement(a, n_iters=5000, top_m=20, step_init=0.008):
    """
    Subgradient on top-M peaks with adaptive step size and patience.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    patience = 0
    max_patience = 800

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - sc_lb + 1e-10) / (gn * gn + 1e-30))
        step = min(step, 0.04)
        if step < 1e-12:
            break

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9992

    return best_a


def coordinate_descent_peaks(a, n_sweeps=3, n_top=40):
    """Coordinate descent on peak-contributing elements with multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            j = int(j)
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j

            for f in [0.4, 0.6, 0.8, 0.9, 1.1, 1.3]:
                cv = a_old_j * f
                if cv > 1000.0:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = cv
                a[j] = a_old_j

            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def two_swap_search(a, n_attempts=200, rng=None):
    """
    Random 2-swap: move mass from one position to another.
    Accept if score improves.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_attempts):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        a_old_i = a[i]
        a_old_j = a[j]
        if a_old_i < 1e-12:
            continue
        # Move a fraction of mass from i to j
        for frac in [0.1, 0.3, 0.5]:
            delta = a_old_i * frac
            a[i] = a_old_i - delta
            a[j] = a_old_j + delta
            S_tmp = a.sum()
            if S_tmp < 1e-12:
                a[i] = a_old_i
                a[j] = a_old_j
                continue
            a_scaled = a / S_tmp
            sc = compute_score(a_scaled)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_scaled.copy()
            a[i] = a_old_i
            a[j] = a_old_j

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.exp(-5.0 * (x - 0.25) ** 2) + np.exp(-5.0 * (x - 0.75) ** 2) + 0.1)
    inits.append(np.maximum(0.1 + 0.9 * np.exp(-3.0 * (x - 0.5) ** 2), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(2.0 * np.pi * 3 * x + 0.7), 0.01))

    normalized = []
    for init in inits:
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-12:
            normalized.append(init / s)
        else:
            normalized.append(np.ones(n) / n)
    return normalized

        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  FinalLBFGS: NEW BEST {sc:.8f}")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
