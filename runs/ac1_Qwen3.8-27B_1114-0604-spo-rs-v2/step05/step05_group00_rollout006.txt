"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Based on analysis, the Cauchy-Schwarz inequality gives a lower bound of 2,
achieved by the uniform sequence. This solution verifies this and attempts
further optimization in case there's a way to go below 2.
Uses LSE-smoothed gradient optimization, Fienup iterations, and coordinate descent
as secondary methods.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation with FFT
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    from scipy.optimize import minimize
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj_func(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj_func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_flatten(a, n_iters=200):
    """Fienup: iterate towards flat power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        alpha = 0.3 + 0.5 * progress
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_target_subgradient(a, n_iters=3000, top_k=3, step_init=0.01):
    """Subgradient method targeting the highest peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        if it % 500 == 0:
            remaining_iters = n_iters - it
            if remaining_iters < 100:
                break

        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = float(np.mean([c[int(k)] / (M_val + 1e-30) for k in top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / top_k / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.02)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9995

    return best_a


def coord_descent(a, n_sweeps=5):
    """Coordinate descent on top contributors to the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(50, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                cv = a_old * f
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for the optimal sequence.
    Phase 1: Verify uniform baseline (score=2 by C-S inequality)
    Phase 2: Try LSE optimization to see if we can beat the baseline
    Phase 3: Fienup + peak shaving + coordinate descent
    Phase 4: Multi-restart with perturbations
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    # Try to load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] score={sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Baseline - uniform sequence
    # ============================================================
    print("=== Phase 1: Baseline ===")

    # Try different lengths for the uniform sequence
    for n_try in [128, 256, 1024, 4096]:
        a_uniform = np.ones(n_try)
        sc = compute_score(a_uniform)
        update(sc, a_uniform, f"uniform n={n_try}")

    # If parent is available, evaluate it
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"  Parent: n={len(parent)}, internal_score={sc_p:.10f}")
        # Verify with actual evaluator
        parent_list = [float(max(0.0, min(1000.0, x))) for x in parent]
        verified_p = float(evaluate_sequence(parent_list))
        print(f"  Parent verified: {verified_p:.10f}")
        if verified_p < best_score:
            best_score = verified_p
            best_a = parent.copy()
            print(f"  [{label}] Using parent as best")

    # Verify uniform with actual evaluator
    a_uniform = np.ones(1024)
    uniform_list = [float(x) for x in a_uniform]
    verified_uniform = float(evaluate_sequence(uniform_list))
    print(f"  Uniform (n=1024) verified: {verified_uniform:.10f}")
    if verified_uniform < best_score:
        best_score = verified_uniform
        best_a = a_uniform.copy()

    # Try a few structured sequences
    for n_struct in [512, 2048]:
        # Triangular
        x = np.linspace(0, 1, n_struct)
        a_tri = 1.0 - np.abs(2 * x - 1)
        a_tri = np.maximum(a_tri, 0.0)
        a_tri = a_tri / a_tri.sum()
        sc = compute_score(a_tri)
        update(sc, a_tri, f"triangular n={n_struct}")

        # Gaussian-ish
        a_gauss = np.exp(-20 * (x - 0.5) ** 2)
        a_gauss = a_gauss / a_gauss.sum()
        sc = compute_score(a_gauss)
        update(sc, a_gauss, f"gaussian n={n_struct}")

        # Cosine window
        a_cos = 0.5 * (1 + np.cos(2 * np.pi * x))
        a_cos = np.maximum(a_cos, 1e-10)
        a_cos = a_cos / a_cos.sum()
        sc = compute_score(a_cos)
        update(sc, a_cos, f"cosine n={n_struct}")

    print(f"  Best after Phase 1: {best_score:.10f}")

    # If the best is already very close to 2, we're likely at the optimum
    if best_score <= 2.0 + 1e-6:
        print("  Best score is at the Cauchy-Schwarz lower bound (2.0).")
        print("  Returning uniform sequence.")
        out = [1.0] * 1024
        verified = float(evaluate_sequence(out))
        print(f"  Final verified: {verified:.10f}")
        return out

    # ============================================================
    # Phase 2: LSE optimization (in case we can beat the baseline)
    # ============================================================
    print("\n=== Phase 2: LSE Optimization ===")

    if best_a is None:
        best_a = np.ones(1024)

    n_work = len(best_a)
    if n_work < 1024:
        n_work = 1024
        best_a = np.ones(n_work)

    a_work = best_a.copy()
    a_work = a_work / a_work.sum()

    phase2_end = min(deadline, t_start + 0.30 * float(budget_s))
    betas = [100, 500, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 20))
        if maxiter < 100:
            break
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}")
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # ============================================================
    # Phase 3: Fienup + Peak shaving + Coord descent
    # ============================================================
    print("\n=== Phase 3: Fienup + Peak + Coord ===")

    phase3_end = min(deadline, t_start + 0.60 * float(budget_s))

    if time.monotonic() < phase3_end:
        a_f = fienup_flatten(best_a, n_iters=150)
        sc = compute_score(a_f)
        update(sc, a_f, "FIENUP")
        if sc < best_score and time.monotonic() < phase3_end:
            sc2, a2 = run_lbfgs(a_f, 2000, maxiter=3000, maxfun=60000)
            update(sc2, a2, "FIENUP+LBFGS")

    if time.monotonic() < phase3_end:
        a_ps = peak_target_subgradient(best_a, n_iters=1500, top_k=3, step_init=0.01)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SG")

    if time.monotonic() < phase3_end:
        a_cd = coord_descent(best_a, n_sweeps=5)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD")
        if sc < best_score and time.monotonic() < phase3_end - 30:
            sc2, a2 = run_lbfgs(a_cd, 5000, maxiter=3000, maxfun=60000)
            update(sc2, a2, "COORD+LBFGS")

    # ============================================================
    # Phase 4: Multi-restart
    # ============================================================
    print("\n=== Phase 4: Multi-restart ===")

    phase4_end = deadline
    n_restarts = 10
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 30:
            break

        # Create perturbed initial
        scale = 0.01 + 0.005 * r
        noise = scale * rng.standard_normal(len(best_a))
        a_pert = best_a * (1.0 + noise)
        a_pert = np.maximum(a_pert, 0.0)
        S_p = a_pert.sum()
        if S_p < 1e-12:
            continue
        a_pert = a_pert / S_p

        remaining = phase4_end - time.monotonic() - 30
        if remaining < 20:
            break
        maxiter = min(3000, int(remaining * 15))
        if maxiter < 100:
            break

        sc, a_opt = run_lbfgs(a_pert, [500, 2000, 5000][r % 3],
                              maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_opt, f"RESTART {r}")

    # ============================================================
    # Phase 5: Final verification and output
    # ============================================================
    print(f"\n=== Phase 5: Final ===")
    print(f"  Best internal score: {best_score:.10f}")

    # Verify best with actual evaluator
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 1024

    verified = float(evaluate_sequence(out))
    print(f"  Best verified score: {verified:.10f}")

    # If verified is worse than uniform, use uniform
    uniform_out = [1.0] * 1024
    verified_uniform = float(evaluate_sequence(uniform_out))
    print(f"  Uniform verified: {verified_uniform:.10f}")

    if verified_uniform < verified:
        print("  Using uniform sequence (better verified score)")
        out = uniform_out
        verified = verified_uniform

    elapsed = time.monotonic() - t_start
    print(f"\n  Final: {verified:.10f}, length={len(out)}, time={elapsed:.1f}s")
    return out
