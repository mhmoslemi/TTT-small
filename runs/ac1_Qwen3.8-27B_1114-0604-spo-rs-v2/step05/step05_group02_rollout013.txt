import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, c
    return 2.0 * n * float(np.max(c)) / (S * S), c

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L with respect to a:
    # dL/da_j = sum_k p_k * db_k/da_j = sum_k p_k * 2*a_{k-j}
    # = 2 * sum_k p_k * a_{k-j} = 2 * (p * a)_j (circular-ish)
    rev_a = a[::-1]
    M2 = 3 * n - 2  # length of conv(p, rev_a)
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc, _ = compute_score(a_out)
    return exact_sc, a_out

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2. Returns (score, convolution)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, c
    return 2.0 * n * float(np.max(c)) / (S * S), c


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def swap_search(a0, n_trials=5000, rng=None):
    """Random swap-based local search: move mass between pairs."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc, _ = compute_score(a)

    for trial in range(n_trials):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        a_i = a[i]
        if a_i < 1e-10:
            continue
        # Try moving fraction of a[i] to a[j]
        for frac in [0.1, 0.3, 0.5, 0.7]:
            da = a_i * frac
            a[i] = a_i - da
            a[j] += da
            S_tmp = a.sum()
            if S_tmp < 1e-12:
                a[i] = a_i
                a[j] -= da
                continue
            sc_tmp, _ = compute_score(a / S_tmp)
            if sc_tmp < best_sc:
                best_sc = sc_tmp
                best_a = (a / S_tmp).copy()
            a[i] = a_i
            a[j] -= da

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.2 * np.sin(2.0 * np.pi * 5 * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x + 0.5) + 0.2 * np.cos(4.0 * np.pi * x), 0.01))
    # Ramp
    inits.append(0.1 + 0.9 * x)
    inits.append(0.1 + 0.9 * (1 - x))
    # Two-bump
    inits.append(np.maximum(0.1 + 0.6 * np.exp(-((x - 0.25) ** 2) / 0.01) + 0.6 * np.exp(-((x - 0.75) ** 2) / 0.01), 0.01))
    return inits


def refine_sequence(a, deadline, label_prefix=""):
    """
    Refine a sequence using multiple methods in sequence.
    Returns the best (score, sequence) found.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, a
    a = a / S

    best_sc, _ = compute_score(a)
    best_a = a.copy()

    # GS for quick improvement
    if time.monotonic() < deadline - 60:
        a_gs = gs_adaptive(best_a, n_iters=300)
        sc, _ = compute_score(a_gs)
        if sc < best_sc:
            best_sc, best_a = sc, a_gs.copy()
            print(f"  [{label_prefix}GS] {sc:.8f}")

    # Adaptive LSE L-BFGS: sweep betas
    if time.monotonic() < deadline - 60:
        betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 25000, 50000, 100000]
        for beta in betas:
            if time.monotonic() > deadline - 30:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
            elapsed = time.monotonic() - t0
            if sc < best_sc:
                best_sc, best_a = sc, a_opt.copy()
                print(f"  [{label_prefix}LBFGS b={beta}] {sc:.8f} ({elapsed:.1f}s)")
            else:
                print(f"  [{label_prefix}LBFGS b={beta}] {sc:.8f} no improvement ({elapsed:.1f}s)")

    # Peak subgradient with momentum
    if time.monotonic() < deadline - 30:
        for top_m in [10, 30, 50]:
            if time.monotonic() > deadline - 15:
                break
            t0 = time.monotonic()
            a_sg = peak_subgrad_momentum(best_a, n_iters=3000, momentum=0.9, top_m=top_m)
            sc, _ = compute_score(a_sg)
            elapsed = time.monotonic() - t0
            if sc < best_sc:
                best_sc, best_a = sc, a_sg.copy()
                print(f"  [{label_prefix}SG m={top_m}] {sc:.8f} ({elapsed:.1f}s)")
            else:
                print(f"  [{label_prefix}SG m={top_m}] {sc:.8f} no improvement ({elapsed:.1f}s)")

    # Coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd = coord_descent_peak(best_a, n_sweeps=10, n_top=80)
        sc, _ = compute_score(a_cd)
        if sc < best_sc:
            best_sc, best_a = sc, a_cd.copy()
            print(f"  [{label_prefix}CD] {sc:.8f}")

    # Final high-beta L-BFGS polish
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc, best_a = sc, a_opt.copy()
            print(f"  [{label_prefix}FINAL-LBFGS] {sc:.8f}")

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses GS, adaptive LSE L-BFGS, Nesterov peak subgradient, coordinate descent,
    and swap search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    # Determine lengths to try
    lengths = [6962, 4096, 8192, 10240, 5348]
    if parent is not None:
        p_len = len(parent)
        if p_len not in lengths:
            lengths.insert(0, p_len)
        else:
            lengths.remove(p_len)
            lengths.insert(0, p_len)
    lengths = sorted(set(lengths))
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'n=' + str(len(parent)) if parent is not None else 'none'}")
    print(f"Time budget: {total_budget}s, deadline at t={deadline - t_start:.0f}s")

    # Phase 1: Quick GS on all lengths to find good basins (30% of time)
    phase1_end = t_start + 0.30 * (total_budget - 30.0)
    print(f"\n=== Phase 1: GS initialization (until t={phase1_end - t_start:.0f}s) ===")

    gs_results = {}
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        best_gs_sc = np.inf
        best_gs_a = np.ones(n) * 0.5

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_gs = gs_adaptive(a0, n_iters=300)
            sc, _ = compute_score(a_gs)
            if sc < best_gs_sc:
                best_gs_sc = sc
                best_gs_a = a_gs.copy()

        gs_results[n] = (best_gs_sc, best_gs_a)
        if best_gs_sc < best_score:
            best_score = best_gs_sc
            best_a = best_gs_a.copy()
            print(f"    NEW BEST after GS: {best_score:.8f} (n={n})")
        else:
            print(f"    best GS score: {best_gs_sc:.8f}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: Full refinement on best length (40% of time)
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Full refinement (until t={phase2_end - t_start:.0f}s) ===")

    n_main = len(best_a)
    sc, a_refined = refine_sequence(best_a, phase2_end, label_prefix=f"main n={n_main} ")
    if sc < best_score:
        best_score = sc
        best_a = a_refined.copy()
    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Try other lengths (20% of time)
    phase3_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Other lengths (until t={phase3_end - t_start:.0f}s) ===")

    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase3_end:
            break
        # Interpolate best to this length
        a_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_main), best_a)
        a_interp = np.maximum(a_interp, 0.01)
        gs_results[n] = ref_sc, ref_a = refine_sequence(a_interp, phase3_end, label_prefix=f"alt n={n} ")
        if ref_sc < best_score:
            best_score = ref_sc
            best_a = ref_a.copy()
            print(f"  NEW BEST from alt length: {best_score:.8f} (n={n})")

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Final polish on the best (10% of time)
    print(f"\n=== Phase 4: Final polish ===")

    # More subgradient iterations
    if time.monotonic() < deadline - 20:
        a_sg = peak_subgrad_momentum(best_a, n_iters=5000, momentum=0.95, top_m=30, step_base=0.02)
        sc, _ = compute_score(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()
            print(f"  [FINAL-SG] {sc:.8f}")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [FINAL-LBFGS b=100000] {sc:.8f}")

    # Coordinate descent polish
    if time.monotonic() < deadline - 5:
        a_cd = coord_descent_peak(best_a, n_sweeps=5, n_top=100)
        sc, _ = compute_score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
            print(f"  [FINAL-CD] {sc:.8f}")

    # Swap search on best
    if time.monotonic() < deadline - 3:
        a_swap = swap_search(best_a, n_trials=2000, rng=rng)
        sc, _ = compute_score(a_swap)
        if sc < best_score:
            best_score = sc
            best_a = a_swap.copy()
            print(f"  [FINAL-SWAP] {sc:.8f}")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
