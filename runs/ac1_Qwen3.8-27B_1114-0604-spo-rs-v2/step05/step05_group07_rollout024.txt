"""
Multi-phase optimizer: IRLS + LSE-LBFGS + Peak-aware ILS + Coordinate Descent.
Minimizes 2*n*max(a*a)/(sum a)^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 1000.0

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fft_matvec(w, x):
    """Compute (Wx)[i] = sum_j w[i+j]*x[j]. w has length 2n-1, x has length n."""
    n = len(x)
    b = w[::-1]
    M = len(b) + n - 1
    M_fft = 1 << (M - 1).bit_length()
    B = np.fft.rfft(b, M_fft)
    X = np.fft.rfft(x, M_fft)
    result = np.fft.irfft(B * X, M_fft)[:M]
    return result[2*n-2 : n-2 : -1]

def irls_solve(w, a_init, n_cg=200, tol=1e-12):
    """Solve min a^T W a s.t. sum(a)=1 using projected CG.
    W[i,j] = w[i+j]. Uses projection onto a>=0 after each step."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    # Use projected gradient (simpler and more robust than CG for indefinite W)
    b = np.ones(n)
    r = b - fft_matvec(w, a)
    p = r.copy()
    rs = float(r @ r)
    
    for k in range(n_cg):
        Wp = fft_matvec(w, p)
        pWp = float(p @ Wp)
        if abs(pWp) < 1e-30:
            break
        alpha = rs / pWp
        if alpha < -0.5 or alpha > 0.5:
            # Damped step
            alpha = np.sign(alpha) * 0.5
        a_trial = a + alpha * p
        a_trial = np.clip(a_trial, 0, None)
        S_trial = a_trial.sum()
        if S_trial < 1e-12:
            break
        a_trial = a_trial / S_trial
        r_new = b - fft_matvec(w, a_trial)
        rs_new = float(r_new @ r_new)
        if rs_new < rs * (1 + 1e-6):
            a = a_trial
            r = r_new
            rs = rs_new
            p = r + (rs_new / rs) * p if rs > 0 else r
        else:
            # Line search
            for frac in [0.5, 0.25, 0.1]:
                a_trial = a + alpha * frac * p
                a_trial = np.clip(a_trial, 0, None)
                S_trial = a_trial.sum()
                if S_trial < 1e-12:
                    break
                a_trial = a_trial / S_trial
                r_new = b - fft_matvec(w, a_trial)
                rs_new = float(r_new @ r_new)
                if rs_new < rs:
                    a = a_trial
                    r = r_new
                    rs = rs_new
                    p = r + (rs_new / max(rs, 1e-30)) * p
                    break
            else:
                break
    return a

def irls_iteration(a, beta, n_cg=200):
    """One IRLS iteration: compute weights, solve subproblem."""
    n = len(a)
    r = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a_norm = a / S
    r_norm = fast_conv(a_norm)
    
    r_max = float(r_norm.max())
    scaled = beta * (r_norm - r_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    w = exp_scaled / exp_scaled.sum()
    
    a_new = irls_solve(w, a_norm, n_cg=n_cg)
    return a_new

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def gradient_descent_irls(a, beta, n_iters=5000, eta0=0.01, decay=0.999):
    """Gradient descent using IRLS-style gradient (2*W*a)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    
    for it in range(n_iters):
        r = fast_conv(a)
        r_max = float(r.max())
        scaled = beta * (r - r_max)
        exp_scaled = np.exp(np.maximum(scaled, -50))
        w = exp_scaled / exp_scaled.sum()
        
        # Gradient of sum_k w[k] * (a*a)[k] = 2 * W * a
        grad_w = 2.0 * fft_matvec(w, a)
        # Plus the constraint gradient (sum a = 1): subtract mean
        grad = grad_w - np.mean(grad_w)
        
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        eta *= decay
    
    return best_a

def peak_aware_perturb(a, rng, scale=0.03):
    """Perturb a in a way that's aware of the peak structure."""
    n = len(a)
    r = fast_conv(a)
    k_star = int(np.argmax(r))
    
    # Identify peak contributors
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    seg = a[j_start:j_end + 1]
    contrib = seg[::-1].copy()
    
    a_p = a.copy()
    strategy = rng.integers(0, 5)
    
    if strategy == 0:
        # Reduce top contributors, add to bottom
        top_k = max(5, n // 20)
        top_local = np.argsort(contrib)[::-1][:top_k]
        bot_local = np.argsort(contrib)[:top_k]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - scale * rng.random())
        reduction = a.sum() - a_p.sum()
        for li in bot_local:
            a_p[j_start + li] += reduction / top_k
    elif strategy == 1:
        # Smooth: blend with a shifted version
        shift = rng.integers(1, max(2, n // 100))
        a_smooth = (np.roll(a, shift) + np.roll(a, -shift)) / 2
        a_p = (1 - scale) * a + scale * a_smooth
    elif strategy == 2:
        # Multiplicative noise
        a_p = a * np.exp(scale * rng.standard_normal(n))
    elif strategy == 3:
        # Block perturbation
        n_blocks = 5 + rng.integers(0, 10)
        block_size = max(1, n // n_blocks)
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2 * (rng.random() - 0.5))
    else:
        # Gaussian smooth
        kernel_size = max(3, n // 50)
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2)**2 / (kernel_size//4)**2)
        kernel /= kernel.sum()
        a_p = np.convolve(a, kernel, mode='same')
    
    a_p = np.clip(a_p, 0, None)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

        contrib = seg[::-1].copy()

        top_local = np.argsort(contrib)[::-1][:top_count]

                    forbidden.add(j)
        
        # Candidate positions for redistribution (not in forbidden set)
        candidates = [j for j in range(n) if j not in forbidden and a[j] < np.median(a)]
        if not candidates:

            if S > 1e-12:
