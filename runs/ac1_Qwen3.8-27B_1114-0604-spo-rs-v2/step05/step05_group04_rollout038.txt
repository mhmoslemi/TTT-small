    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Softmax weights
    p = exp_scaled / exp_sum
    
    # Compute dL/dx = 2 * cross_correlate(p, x)
    # cross_correlate(p, x)[i] = sum_{j=0}^{n-1} p[i+j] * x[j]
    M2_fft = 1 << (3 * n - 2 - 1).bit_length()  # need size >= 3n-2
    Q_pad = np.zeros(M2_fft)
    Q_pad[:2*n-1] = p
    X_pad = np.zeros(M2_fft)
    X_pad[:n] = x
    
    FQ = np.fft.fft(Q_pad)
    FX = np.fft.fft(X_pad)
    R = np.fft.ifft(np.conj(FX) * FQ).real[:n]
    g_L = 2.0 * R
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    
    return score, grad

def fienup_flat_spectrum(a, n_iters=500, alpha=0.3):
    """
    Fienup iteration: flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return ideal_init(n)
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Relax towards flat spectrum
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def fienup_peak_reduce(a, n_iters=500, top_frac=0.1):
    """
    Fienup: reduce the top spectral bins to flatten the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return ideal_init(n)
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Compress bins above threshold
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 0.01))
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

```python
"""
Optimization of non-negative sequence to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses combinatorial initialization a_k = C(2k,k)/4^k (gives flat autocorrelation
at lags 0..n-1) as starting point, refined by L-BFGS with LSE smoothing,
top-k subgradient methods, and Fienup spectral iterations.
"""
import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the evaluation score 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def ideal_init(n):
    """Generate ideal initialization a_k = C(2k,k)/4^k normalized to sum=1.
    This gives exactly flat autocorrelation at lags 0..n-1."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    a = a / a.sum()
    return a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = 1.0 - u / (cssv - 1.0 / arange)
    rho = int(np.argmax(cond > 0))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for the score function.
    Objective: 2n * LSE_beta(conv(x,x)) / (sum(x))^2"""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Gradient: dL/dx_i = 2 * sum_j p[i+j] * x[j]
    # Computed via FFT cross-correlation
    M2_fft = 1 << (3 * n - 2 - 1).bit_length()
    Q_pad = np.zeros(M2_fft)
    Q_pad[:2 * n - 1] = p
    X_pad = np.zeros(M2_fft)
    X_pad[:n] = x

    FQ = np.fft.fft(Q_pad)
    FX = np.fft.fft(X_pad)
    R = np.fft.ifft(np.conj(FX) * FQ).real[:n]
    g_L = 2.0 * R

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    a0 = a0 / S

    bounds = [(1e-10, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def topk_subgradient(a, n_iters=5000, top_k=10, step_init=0.01, decay=0.99995):
    """Top-k subgradient method with simplex projection.
    Minimizes (1/k)*sum of top-k autocorrelation values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return ideal_init(n)
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k indices
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Compose subgradient: g_i = (2/k) * sum_{j in top_k} x_{j-i}
        g = np.zeros(n)
        for j in top_idx:
            j = int(j)
            j_start = max(0, j - n + 1)
            j_end = min(j, n - 1)
            if j_end >= j_start:
                # dc_j/dx_i = 2*x_{j-i} for i in [j_start, j_end]
                seg = a[j - j_end:j - j_start + 1][::-1]
                g[j_start:j_end + 1] += seg

        g *= 2.0 / len(top_idx)

        # Remove mean to maintain sum constraint (projected gradient)
        g_centered = g - g.mean()
        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-15:
            break

        a -= (step / gn) * g_centered
        a = project_simplex(a)
        step *= decay

    return best_a


def fienup_spectral(a, n_iters=500, mode='flat', alpha=0.3, top_frac=0.1):
    """Fienup spectral iteration to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return ideal_init(n)
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        if mode == 'flat':
            P_mean = float(P.mean())
            P_new = (1.0 - alpha) * P + alpha * P_mean
            P_new = np.maximum(P_new, 1e-30)
        elif mode == 'peak':
            thresh = np.quantile(P, 1.0 - top_frac)
            P_new = P.copy()
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                P_new[above] = P[above] * (1.0 - alpha) + thresh * alpha
        else:
            P_new = P.copy()

        amp = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
