```python
def project_nonneg_bleedback(a, n_passes=3):
    a = a.copy()
    for _ in range(n_passes):
        neg = a < 0
        if not neg.any():
            break
        # Distribute negative mass to neighbors
        i = np.where(neg)[0]
        for idx in i:
            excess = -a[idx]
            a[idx] = 0
            # Distribute to neighbors
            left = max(0, idx-1)
            right = min(len(a)-1, idx+1)
            # Give half to each (or proportional to neighbor values)
            a[left] += excess / 2
            a[right] += excess / 2
    return a
```

```python
    return 2.0 * n * float(np.max(c)) / (S * S)

def project_nonneg(a, bleed=1):
    """Project to non-negative with optional bleed-back to neighbors."""
    a = a.copy()
    for _ in range(bleed):
        neg_mask = a < 0
        if not neg_mask.any():
            break
        neg_idx = np.where(neg_mask)[0]
        for idx in neg_idx:
            excess = -a[idx]
            a[idx] = 0.0
            n = len(a)
            left = idx - 1 if idx > 0 else 0
            right = idx + 1 if idx < n - 1 else n - 1
            a[left] += 0.5 * excess
            a[right] += 0.5 * excess
    return a

def fienup_hybrid(a, n_iters=500, top_frac_init=0.1, top_frac_end=0.5,
                  gamma=0.8, use_er=True, er_alpha=0.5):
    """
    Hybrid Fienup alternating between Direct Method (DM) and Error Reduction (ER).
    DM: compress spectrum amplitude, IFFT, project non-negative.
    ER: FFT the negative error, subtract from spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * progress
        
        if it % 2 == 0 or not use_er:
            # DM step: modify amplitude
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            n_bins = len(P)
            thresh = np.quantile(P, 1.0 - top_frac)
            scale = np.ones_like(P)
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                scale[above] = np.sqrt(ratio) * gamma
            A_mod = A * scale
            a_new = np.fft.irfft(A_mod, M_fft)[:n]
        else:
            # ER step: modify based on time-domain error
            A = np.fft.rfft(a, M_fft)
            a_trial = np.fft.irfft(A, M_fft)[:n]
            # The error is the negative part
            err = np.minimum(a_trial, 0.0)
            if np.abs(err).sum() < 1e-15:
                a_new = a_trial
            else:
                E = np.fft.rfft(err, M_fft)
                A_mod = A - er_alpha * E
                a_new = np.fft.irfft(A_mod, M_fft)[:n]
            
        # Non-negativity projection with bleed-back
        a_new = project_nonneg(a_new, bleed=2)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_objective_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out

def momentum_quadratic(a, n_iters=5000, top_m=20, momentum=0.9, step_init=0.03):
    """
    Momentum-based projected gradient on the quadratic defined by top-m peaks.
    Objective: (1/|S|) * sum_{k in S} conv(a,a)_k / (sum a)^2
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    v = np.zeros(n)  # momentum

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # Reduce step size on improvement
            step *= 0.95

        # Get top-m peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Gradient of (1/m) * sum_{k in top} conv(a,a)_k / S^2
        # = (1/m) * sum_k 2 * a_reversed_at_k / S^2 - 2 * (1/m)*sum_k conv(a,a)_k / S^3
        sg = np.zeros(n)
        m = len(top_idx)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= m

        sum_c_top = float(c[top_idx].sum()) / m
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * sum_c_top / S3)

        # Momentum update
        v = momentum * v + grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        # Decay step size slowly
        step *= 0.99995
        step = max(step, 1e-6)

    return best_a
```

```python
                scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma
            A_mod = A * scale
            a_new = np.fft.irfft(A_mod, M_fft)[:n]
        else:
            # ER step: subtract FFT of negative error
            a_trial = np.fft.irfft(A, M_fft)[:n]
            err = np.minimum(a_trial, 0.0)
            err_sum = float(np.abs(err).sum())
            if err_sum < 1e-14:
                a_new = a_trial
            else:
                E = np.fft.rfft(err, M_fft)
                A_mod = A - er_alpha * E
                a_new = np.fft.irfft(A_mod, M_fft)[:n]

        a_new = project_nonneg_bleed(a_new, n_passes=bleed_passes)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_waterfill(a, n_iters=300, n_levels=5, strength_init=0.3, strength_end=0.7):
    """Multi-level water-filling Fienup."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        strength = strength_init + (strength_end - strength_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        scale = np.ones_like(P)
        for lv in range(n_levels):
            frac = 0.08 * (lv + 1) / n_levels
            thresh = np.quantile(P, 1.0 - frac)
            above = P > thresh
            if above.any():
                ratio = thresh / P[above]
                scale[above] *= 1.0 - strength + strength * np.sqrt(np.maximum(ratio, 0.001))

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = project_nonneg_bleed(a_new, n_passes=2)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def momentum_subgradient(a, n_iters=5000, top_m=20, momentum=0.90,
                         step_init=0.025, step_decay=0.99998, step_min=1e-7):
    """
    Nesterov-momentum subgradient on convex combination of top-m peaks.
    Minimizes (2n/S^2) * mean_{k in top-m} conv(a,a)_k as a surrogate for max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    v = np.zeros(n)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Adaptive top_m: start small, grow over time
        if it < n_iters * 0.3:
            m_eff = max(3, top_m // 2)
        elif it < n_iters * 0.7:
            m_eff = top_m
        else:
            m_eff = top_m * 2

        if m_eff >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:m_eff]

        # Gradient of mean of top-m conv values
        sg = np.zeros(n)
        m = len(top_idx)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= m

        sum_top = float(c[top_idx].sum()) / m
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * sum_top / S3)

        # Nesterov momentum
        v = momentum * v + grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        # Adaptive step: shrink if no improvement recently
        step = max(step * step_decay, step_min)

    return best_a


def perturbation_restart(a, rng, scale=0.03):
    """Perturb a sequence and renormalize."""
    a_p = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = float(a_p.sum())
    if S < 1e-12:
        return a.copy()
    return a_p / S


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Combines Fienup hybrid, momentum subgradient,
    and L-BFGS with LSE smoothing to minimize the evaluation score.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None and len(parent) > 100:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    if parent is not None:
        sc = compute_score(parent)
        best_score = sc
        best_a = parent.copy()
    else:
        best_score = compute_score(best_a)

    print(f"n={n}, initial best={best_score:.8f}")

    # ================================================================
    # Phase 1: Fienup variants (35% of time)
    # ================================================================
    p1_end = t_start + 0.35 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Fienup (deadline t={p1_end - t_start:.0f}s) ===")

    fienup_configs = [
        dict(n_iters=250, top_frac_init=0.08, top_frac_end=0.50, gamma=0.7, use_er=True, er_alpha=0.5),
        dict(n_iters=250, top_frac_init=0.12, top_frac_end=0.55, gamma=0.9, use_er=True, er_alpha=0.3),
        dict(n_iters=250, top_frac_init=0.10, top_frac_end=0.45, gamma=0.6, use_er=False, er_alpha=0.0),
    ]

    for ci, cfg in enumerate(fienup_configs):
        if time.monotonic() > p1_end:
            break
        t0 = time.monotonic()
        a_f = fienup_hybrid(best_a, **cfg)
        sc = compute_score(a_f)
        if sc < best_score:
            best_score = sc
            best_a = a_f.copy()
            print(f"  [F-H {ci}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  [F-H {ci}] score={sc:.8f} (no improvement, t={time.monotonic()-t0:.0f}s)")

    # Waterfill variant
    if time.monotonic() < p1_end:
        a_wf = fienup_waterfill(best_a, n_iters=200, n_levels=6, strength_init=0.3, strength_end=0.7)
        sc = compute_score(a_wf)
        if sc < best_score:
            best_score = sc
            best_a = a_wf.copy()
            print(f"  [WATER] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  [WATER] score={sc:.8f}")

    # Chained Fienup: run again from best with different params
    if time.monotonic() < p1_end - 60:
        a_f2 = fienup_hybrid(best_a, n_iters=200, top_frac_init=0.05, top_frac_end=0.35,
                             gamma=0.85, use_er=True, er_alpha=0.4)
        sc = compute_score(a_f2)
        if sc < best_score:
            best_score = sc
            best_a = a_f2.copy()
            print(f"  [F-H2] NEW BEST: {sc:.8f}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Momentum subgradient (30% of time)
    # ================================================================
    p2_end = t_start + 0.65 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Momentum Subgradient (deadline t={p2_end - t_start:.0f}s) ===")

    sg_configs = [
        dict(n_iters=4000, top_m=15, momentum=0.90, step_init=0.025),
        dict(n_iters=4000, top_m=25, momentum=0.85, step_init=0.020),
        dict(n_iters=4000, top_m=40, momentum=0.92, step_init=0.015),
    ]

    for ci, cfg in enumerate(sg_configs):
        if time.monotonic() > p2_end:
            break
        t0 = time.monotonic()
        a_m = momentum_subgradient(best_a, **cfg)
        sc = compute_score(a_m)
        if sc < best_score:
            best_score = sc
            best_a = a_m.copy()
            print(f"  [SG {ci}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  [SG {ci}] score={sc:.8f} (no improvement, t={time.monotonic()-t0:.0f}s)")

    # Perturbation + re-run
    if time.monotonic() < p2_end - 120:
        a_p = perturbation_restart(best_a, rng, scale=0.04)
        a_m2 = momentum_subgradient(a_p, n_iters=3000, top_m=20, momentum=0.90, step_init=0.02)
        sc = compute_score(a_m2)
        if sc < best_score:
            best_score = sc
            best_a = a_m2.copy()
            print(f"  [SG-PERT] NEW BEST: {sc:.8f}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: L-BFGS with LSE (25% of time)
    # ================================================================
    p3_end = t_start + 0.90 * (budget_s - 30.0)
    print(f"\n=== Phase 3: L-BFGS (deadline t={p3_end - t_start:.0f}s) ===")

    betas = [200, 500, 1000, 2000, 5000]
    for beta in betas:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 60:
            break
        # Scale maxiter based on remaining time and problem size
        time_per_iter_est = len(best_a) / 5000.0 + 0.1
        maxiter = max(500, int(remaining * 0.8 / time_per_iter_est * 0.5))
        maxiter = min(maxiter, 15000)
        maxfun = maxiter * 15

        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [LBFGS b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  [LBFGS b={beta}] score={sc:.8f} (t={time.monotonic()-t0:.0f}s)")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish (remaining time)
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    if time.monotonic() < deadline - 60:
        # Quick Fienup
        a_f = fienup_hybrid(best_a, n_iters=100, top_frac_init=0.05, top_frac_end=0.25,
                            gamma=0.85, use_er=True, er_alpha=0.3)
        sc = compute_score(a_f)
        if sc < best_score:
            best_score = sc
            best_a = a_f.copy()
            print(f"  [FINAL-FIENUP] NEW BEST: {sc:.8f}")

    if time.monotonic() < deadline - 30:
        # Quick momentum
        a_m = momentum_subgradient(best_a, n_iters=2000, top_m=10, momentum=0.93, step_init=0.01)
        sc = compute_score(a_m)
        if sc < best_score:
            best_score = sc
            best_a = a_m.copy()
            print(f"  [FINAL-MOM] NEW BEST: {sc:.8f}")

    # One more perturbation restart if time allows
    if time.monotonic() < deadline - 20:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            a_p = perturbation_restart(best_a, rng, scale=0.015 * (trial + 1))
            a_r = fienup_hybrid(a_p, n_iters=80, top_frac_init=0.05, top_frac_end=0.20,
                                gamma=0.8, use_er=True, er_alpha=0.3)
            sc = compute_score(a_r)
            if sc < best_score:
                best_score = sc
                best_a = a_r.copy()
                print(f"  [RESTART {trial}] NEW BEST: {sc:.8f}")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
