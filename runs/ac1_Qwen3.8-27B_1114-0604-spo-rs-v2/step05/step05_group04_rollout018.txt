def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

        return 1e10, np.zeros(n)

    if a0.sum() < 1e-6:

    b0 = np.sqrt(a0)
    
    res = minimize(
        lse_b_objective_grad, b0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    b_out = res.x
    a_out = b_out * b_out
    a_out = np.clip(a_out, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def top_k_subgradient(a, n_iters=3000, top_k=20, step_init=0.02):

    step = step_init
    decay = 0.9999

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top k indices
        if top_k >= M:
            top_idx = np.arange(M)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        lb = 2.0 * n / M
        step = min(step, max((sc - lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    x_flip = x[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    XF = np.fft.rfft(x_flip, M2_fft)
    corr = np.fft.irfft(P * XF, M2_fft)[n - 1:2 * n - 1]
    g_L = (2.0 / beta) * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L * x / S3)
    return score, grad


def run_lbfgs_b(a0, beta, maxiter=10000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    b0 = np.sqrt(a0)
    res = minimize(
        lse_b_objective_grad, b0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    b_out = res.x
    a_out = b_out * b_out
    a_out = np.clip(a_out, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def run_lbfgs_direct(a0, beta, maxiter=10000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_direct_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_flat(a, n_iters=300, alpha_start=0.1, alpha_end=0.6):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * amp_ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_thresh(a, n_iters=300, top_frac=0.15, strength_start=0.3, strength_end=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        strength = strength_start + (strength_end - strength_start) * progress
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = 1.0 - strength + strength * np.sqrt(np.maximum(ratio, 0.001))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_multiscale(a, n_iters=400):
    """Multi-level threshold Fienup with growing top fraction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        n_levels = 3 + int(3 * progress)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        scale = np.ones_like(P)
        for lv in range(1, n_levels + 1):
            frac = min(0.5, 0.08 * lv)
            thresh = np.quantile(P, 1.0 - frac)
            above = P > thresh
            if above.any():
                s = 0.2 + 0.4 * progress
                ratio = thresh / P[above]
                scale[above] *= 1.0 - s + s * np.sqrt(np.maximum(ratio, 0.001))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def subgradient_topk(a, n_iters=5000, top_k=20, step_init=0.03):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    decay = 0.99995
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if top_k >= M:
            top_idx = np.arange(M)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        lb = 2.0 * n / M
        step = min(step, max((sc - lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.04)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    return best_a


def active_set_lbfgs(a, beta=500, n_iters=3000):
    """L-BFGS with forced zeros on inactive coordinates."""
    n = len(a)
    threshold = 1e-8
    active = a > threshold
    n_active = int(active.sum())
    if n_active < 20 or n_active == n:
        return run_lbfgs_direct(a, beta, maxiter=n_iters, maxfun=n_iters * 15)
    bounds = []
    for i in range(n):
        if active[i]:
            bounds.append((0.0, 1000.0))
        else:
            bounds.append((0.0, 0.0))
    a_norm = a.copy()
    a_norm = a_norm / a_norm.sum()
    res = minimize(
        lse_direct_objective_grad, a_norm, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-13,
                 'maxiter': n_iters, 'maxfun': n_iters * 15, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a.copy()
    a_out = a_out / a_out.sum()
    return a_out


def build_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses b=sqrt(a) parameterization for L-BFGS, spectral Fienup, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        print(f"Parent loaded: n={len(parent)}, score={sc:.8f}")
    else:
        print("No parent available.")

    # Determine working lengths
    if parent is not None:
        n_parent = len(parent)
        lengths = [n_parent, 8192, 12288]
        lengths = sorted(set(lengths))[:3]
    else:
        lengths = [8192, 12288]

    print(f"Working lengths: {lengths}")
    effective_budget = total_budget - 35.0

    # ==================================================================
    # Phase 1: Quick Fienup + L-BFGS to get a good baseline (15% time)
    # ==================================================================
    phase1_end = t_start + 0.15 * effective_budget
    print(f"\n=== Phase 1: Quick setup (t<{(phase1_end-t_start):.0f}s) ===")

    start_a = best_a.copy() if parent is not None else np.ones(8192) * 0.5
    n_main = len(start_a)

    # Quick spectral shaping
    if time.monotonic() < phase1_end:
        a_sp = fienup_flat(start_a, n_iters=100, alpha_start=0.1, alpha_end=0.4)
        update_best(compute_score(a_sp), a_sp, "P1-FIENUP-FLAT")

    # Quick L-BFGS with b-param
    if time.monotonic() < phase1_end:
        sc, a_opt = run_lbfgs_b(best_a, beta=200, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "P1-LBFGS-B b=200")

    print(f"Phase 1 done: best={best_score:.8f}")

    # ==================================================================
    # Phase 2: Main L-BFGS beta sweep with b-parameterization (45% time)
    # ==================================================================
    phase2_end = t_start + 0.60 * effective_budget
    print(f"\n=== Phase 2: L-BFGS beta sweep (t<{(phase2_end-t_start):.0f}s) ===")

    n_main = len(best_a)
    betas_schedule = [100, 300, 1000, 3000, 10000, 30000]

    for beta in betas_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        # Allocate iterations based on remaining time and problem size
        est_time_per_fun = n_main * 0.0001
        maxfun = max(5000, int(remaining / (est_time_per_fun * 8)))
        maxfun = min(maxfun, 200000)
        maxiter = max(2000, maxfun // 15)

        print(f"  L-BFGS b-param beta={beta}, maxfun={maxfun}")
        sc, a_opt = run_lbfgs_b(best_a, beta=beta, maxiter=maxiter, maxfun=maxfun)
        updated = update_best(sc, a_opt, f"P2-LBFGS-B b={beta}")

        # Also try direct L-BFGS if b-param improved significantly
        if updated and time.monotonic() < phase2_end - 30:
            sc2, a_opt2 = run_lbfgs_direct(best_a, beta=beta, maxiter=3000, maxfun=40000)
            update_best(sc2, a_opt2, f"P2-LBFGS-D b={beta}")

    # Try different lengths
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_end:
            break
        if time.monotonic() > phase2_end - 60:
            break
        print(f"  Trying n={n}")
        a_interp = best_a.copy()
        if len(a_interp) != n:
            x_old = np.linspace(0, 1, len(a_interp))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, a_interp)
        a_interp = np.maximum(a_interp, 0)
        a_interp = a_interp / a_interp.sum()
        sc, a_opt = run_lbfgs_b(a_interp, beta=1000, maxiter=3000, maxfun=40000)
        update_best(sc, a_opt, f"P2-n={n}")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ==================================================================
    # Phase 3: Alternating spectral + L-BFGS (20% time)
    # ==================================================================
    phase3_end = t_start + 0.80 * effective_budget
    print(f"\n=== Phase 3: Spectral + L-BFGS alternation (t<{(phase3_end-t_start):.0f}s) ===")

    n_main = len(best_a)
    a_cur = best_a.copy() / best_a.sum()

    for round_idx in range(6):
        if time.monotonic() > phase3_end:
            break

        # Spectral step
        sf = 0.1 + 0.7 * round_idx / 6
        a_sp = fienup_thresh(a_cur, n_iters=150, top_frac=0.1 + 0.1 * sf,
                             strength_start=0.3, strength_end=0.6 + 0.3 * sf)
        sc = compute_score(a_sp)
        if sc < best_score:
            a_cur = a_sp.copy() / a_sp.sum()
        update_best(sc, a_sp, f"P3-R{round_idx}-THRESH")

        # Multi-scale spectral
        if time.monotonic() < phase3_end - 30:
            a_ms = fienup_multiscale(a_cur, n_iters=150)
            sc_ms = compute_score(a_ms)
            if sc_ms < best_score:
                a_cur = a_ms.copy() / a_ms.sum()
            update_best(sc_ms, a_ms, f"P3-R{round_idx}-MULTI")

        # L-BFGS refinement
        if time.monotonic() < phase3_end - 30:
            beta = 1000 + 5000 * round_idx
            sc_lb, a_lb = run_lbfgs_b(a_cur, beta=beta, maxiter=3000, maxfun=50000)
            if sc_lb < best_score:
                a_cur = a_lb.copy() / a_lb.sum()
            update_best(sc_lb, a_lb, f"P3-R{round_idx}-LBFGS b={beta}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ==================================================================
    # Phase 4: Active set + subgradient polishing (10% time)
    # ==================================================================
    phase4_end = t_start + 0.90 * effective_budget
    print(f"\n=== Phase 4: Active set + Subgradient (t<{(phase4_end-t_start):.0f}s) ===")

    # Active set L-BFGS
    if time.monotonic() < phase4_end:
        a_act = active_set_lbfgs(best_a, beta=5000, n_iters=3000)
        update_best(compute_score(a_act), a_act, "P4-ACTIVE")

    # Subgradient with different top-k
    for top_k in [10, 20, 50]:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 10:
            break
        iters = min(5000, int(remaining * 300))
        a_sg = subgradient_topk(best_a, n_iters=iters, top_k=top_k, step_init=0.02)
        update_best(compute_score(a_sg), a_sg, f"P4-SG k={top_k}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polishing (remaining time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish ===")

    # Final high-beta L-BFGS on b-parameterization
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        maxfun = max(5000, int(remaining * 50))
        sc, a_opt = run_lbfgs_b(best_a, beta=20000, maxiter=5000, maxfun=maxfun)
        update_best(sc, a_opt, "P5-FINAL-B b=20000")

    # Final direct L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs_direct(best_a, beta=10000, maxiter=3000, maxfun=40000)
        update_best(sc, a_opt, "P5-FINAL-D b=10000")

    # One more spectral + LBFGS combo
    if time.monotonic() < deadline - 10:
        a_sp = fienup_flat(best_a, n_iters=80, alpha_start=0.2, alpha_end=0.5)
        sc_sp = compute_score(a_sp)
        update_best(sc_sp, a_sp, "P5-SPEC")
        if time.monotonic() < deadline - 5:
            sc2, a2 = run_lbfgs_b(a_sp, beta=10000, maxiter=2000, maxfun=30000)
            update_best(sc2, a2, "P5-SPEC+LBFGS")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
