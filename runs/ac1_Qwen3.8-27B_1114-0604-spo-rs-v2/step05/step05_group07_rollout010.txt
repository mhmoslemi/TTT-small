```python
def solve_lp(a, M):
    """Solve max sum(g) s.t. g >= 0, conv(a,g) <= M."""
    from scipy.optimize import linprog
    n = len(a)
    # Build the constraint matrix A such that A @ g <= M
    # A[k, i] = a[i] if 0 <= k-i < n, else 0
    # For efficiency, use sparse matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A[k, i] = a[i]
    A = A.tocsr()
    b = np.ones(2*n-1) * M
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
Optimized step function search using:
1. Mirror descent with adaptive step size and momentum
2. L-BFGS with adaptive LSE smoothing

from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
from scipy.special import softmax
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

_current_beta = 1000.0

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_callback, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-15, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 200})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res
```

```python
"""

LP-based direction finding for local minimum escape, L-BFGS with adaptive

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

_current_beta = 1000.0

def lse_callback(x):
    """Callback for L-BFGS."""
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_callback, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-15,
                            'maxiter': maxiter, 'maxfun': maxfun,
                            'maxls': 200})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out, res

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.argmax(cond)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_mirror_descent(a0, beta, n_iters=5000, eta0=0.1, beta1=0.9, beta2=0.999):
    """Mirror descent with Adam-style per-coordinate adaptation."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        # Center gradient
        g_centered = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_centered
        v = beta2 * v + (1 - beta2) * g_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive step size
        eta = eta0 / np.sqrt(t)
        exponent = -eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        exponent = np.clip(exponent, -50, 50)
        
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
    
    return best_a

def lp_direction(a, M_mult=1.0):
    """Find a direction g via LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns the normalized direction or None if LP fails."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return None
    
    c = fast_conv(a)
    M = float(np.max(c)) * M_mult
    
    # Build constraint matrix: A @ g <= M
    # A[k, i] = a[i] if 0 <= k-i < n
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            A[k, i_start:i_end + 1] = a[i_start:i_end + 1]
    A = A.tocsr()
    
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            Sg = g.sum()
            if Sg > 1e-12:
                return g / Sg
    except Exception:
        pass
    return None

def subgradient_backtrack(a, n_iters=5000, f_lb=1.0):
    """Projected subgradient descent with backtracking line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.5
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        grad = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            grad[idx] = 2.0 * a[k_star - idx]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = 2.0 * n * (grad / S2 - 2.0 * Mak / S3)
        
        # Center gradient for simplex projection
        grad_centered = grad - np.mean(grad)

        a_trial = a - step * grad_centered
        a_proj = project_simplex(a_trial)
        sc_trial = compute_score(a_proj)

            step = min(step * 1.05, 2.0)
```
