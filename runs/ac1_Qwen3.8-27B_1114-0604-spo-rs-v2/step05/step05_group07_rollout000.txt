def fast_conv(a):

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

"""
LP-based iterative refinement combined with L-BFGS for minimum peak convolution.
Key innovation: Uses the paper's LP direction (maximize sum(g) s.t. conv(f,g)<=M)
with constraint reduction (only top-k active constraints), combined with 
LSE-smoothed L-BFGS refinement and adaptive stepping.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


_current_beta = 1000.0


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_reduced_lp(f, M, k_positions):
    """
    Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for k in k_positions, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    k = len(k_positions)

    # Build constraint matrix (k x n)
    A = np.zeros((k, n))
    for i, kp in enumerate(k_positions):
        j_start = max(0, kp - n + 1)
        j_end = min(kp, n - 1)
        js = np.arange(j_start, j_end + 1)
        A[i, js] = f[kp - js]

    c_obj = -np.ones(n)
    b_ub = np.full(k, M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})

    if res.success:
        return res.x
    return None


def lp_iteration(f, n_constraints=200, relax=1.0):
    """
    One LP-based iteration: find direction g, return (g, M).
    f should be normalized (sum=1).
    """
    n = len(f)
    c = fast_conv(f)
    M = float(np.max(c)) * relax

    # Select top-k constraint positions
    if n_constraints >= 2 * n - 1:
        k_positions = list(range(2 * n - 1))
    else:
        # Use positions where c is in the top n_constraints
        threshold_idx = np.argsort(c)[::-1][:n_constraints]
        k_positions = sorted(threshold_idx.tolist())

    g = solve_reduced_lp(f, M, k_positions)
    if g is None:
        return None, M
    return g, M


def lp_step(f, n_constraints=200, relax=1.0, max_attempts=5):
    """
    Full LP step: find g, rescale, take convex combination step.
    Returns (improved_f, improved_score) or (f, current_score) if no improvement.
    """
    n = len(f)
    current_score = compute_score(f)
    S = f.sum()

    g, M = lp_iteration(f, n_constraints, relax)
    if g is None:
        return f, current_score

    # Rescale g to same sum as f
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f, current_score
    g = g / g_sum * S

    # Try different step sizes
    for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, 1000.0)
        sc = compute_score(f_new)
        if sc < current_score - 1e-12:
            return f_new, sc
    return f, current_score


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.999):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def targeted_peak_flatting(a, n_sweeps=5, n_top=50):
    """Targeted coordinate descent: reduce peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        top_count = min(n_top, max(1, total_len // 3))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.3, 0.5, 0.7, 0.85, 0.95]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
