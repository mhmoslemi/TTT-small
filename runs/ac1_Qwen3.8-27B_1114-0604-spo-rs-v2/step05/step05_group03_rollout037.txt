"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: Corrected FISTA gradient (FFT-based cross-correlation instead of approximate
sliding window), progressive L-BFGS LSE as main driver, adaptive beta scheduling,
backtracking line search, and peak-targeted subgradient polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

        # Progressive beta within outer loop
        frac = outer / max(1, max_outer - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        beta = min(beta, 3000.0)

        x = lp_solve_fista(a, M, n_fista_iters=fista_iters,
                           beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Backtracking line search
        best_t_sc = best_sc
        best_t_val = 0.0
        best_t_a = None
        t_candidates = [0.3, 0.2, 0.15, 0.1, 0.05, 0.4, 0.5, 0.25]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t_val = t
                best_t_a = a_try.copy()

        if best_t_a is not None:
            a = best_t_a
            if best_sc > sc_try:
                best_sc = sc_try
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        if no_improve_count >= 15:
            # Perturb to escape
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve_count = 0

    return best_a, best_sc, x_warm


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_local_search(a, n_steps=300, rng=None):
    """Targeted local search reducing the convolution peak."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.05, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3)
        improved = False
        for mult in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 30 == 0 and rng.random() < 0.4:
            top_k = max(5, len(c) // 20)
            candidates = np.argsort(c)[::-1][:top_k]
            k_trial = candidates[rng.integers(len(candidates))]
            j_s = max(0, k_trial - n + 1)
            j_e = min(k_trial, n - 1)
            if j_e >= j_s:
                direction2 = np.zeros(n)
                direction2[j_s:j_e + 1] = 2.0 * a[j_s:j_e + 1][::-1]
                grad2 = 2.0 * n * (direction2 / S2 - 2.0 * float(c[k_trial]) / S3)
                step2 = 0.02 / (np.linalg.norm(grad2) + 1e-15)
                a_try = a - step2 * grad2
                a_try = np.clip(a_try, 0.0, 1000.0)
                St = float(a_try.sum())
                if St > 1e-12:
                    a_try = a_try / St
                    sc_try = compute_score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    return inits

            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf

        nonlocal best_score, best_a
        if sc < best_score:

            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

        ps = compute_score(parent)
