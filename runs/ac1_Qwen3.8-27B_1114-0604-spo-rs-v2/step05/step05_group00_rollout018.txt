- Identify the peak location k*
- The peak value is c[k*] = sum_i a_i * a_{k*-i}
- To reduce c[k*], we need to reduce the sum of products a_i * a_{k*-i}
- But reducing a_i increases the relative weight of other peaks

4. Transfer a small amount of mass from this pair to reduce the peak
5. Repeat

Let me try a different approach: **use the explicit structure of the problem**.

For a = [a_0, ..., a_{n-1}], the autocorrelation at lag k (centered) is:
c_k = sum_{i=max(0,k-n+1)}^{min(k, n-1)} a_i * a_{k-i}

The peak is at k = n-1 (center) for most reasonable sequences. So:
c_{n-1} = sum_{i=0}^{n-1} a_i * a_{n-1-i}

This is the "anti-diagonal" sum. To minimize this, we want the sequence to be anti-symmetric (a_i = 0 when a_{n-1-i} is large and vice versa). But we also need a >= 0.

The optimal anti-diagonal sum for a >= 0, sum(a) = 1 is achieved by putting all mass on one side: a = [1, 0, 0, ..., 0], giving c_{n-1} = 0. But then c_0 = 1, so the max is 1, and the score is 2n. That's terrible.

"""
Continuation-based LSE optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses adaptive beta scheduling, Fienup warm-up, peak-directed subgradient,
coordinate descent, and multi-restart polishing. All helpers are top-level.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(np.sum(exp_scaled))) / beta
    score = 2.0 * n * L / (S * S)
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x, beta):
    return lse_obj_grad(x, beta)


def fienup_run(a, n_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P = np.maximum(P, 1e-30)
        alpha = 0.15 + 0.6 * progress
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-30)
        A_new = A * np.sqrt(P_blend / P)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_quantile(a, n_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P = np.maximum(P, 1e-30)
        q = 0.8 + 0.15 * (it / n_iters)
        thresh = float(np.quantile(P, q))
        scale = np.ones_like(P)
        above = P > thresh
        scale[above] = np.sqrt(thresh / P[above])
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def peak_subgrad_step(a, n_peaks, step_size, n_iters, decay):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(np.max(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        for idx in range(len(top_idx)):
            k = int(top_idx[idx])
            weight = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += weight * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * len(top_idx) / (len(top_idx) * S3))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        a = a - (step_size / gn) * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step_size *= decay
    return best_a


def coord_descent_run(a, n_sweeps, n_top):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        cur_sc = compute_score(a)
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            fractions = [0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.5]
            best_val = a_old
            best_local = cur_sc
            for fr in fractions:
                cv = a_old * fr
                cv = max(0.0, min(1000.0, cv))
                a[j] = cv
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old
            a[j] = best_val
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def run_lbfgs_single(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return a_out


def structured_perturb_run(a, rng, n_modes, scale):
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = int(rng.integers(1, 20))
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = float(a_new.sum())
    if S > 1e-12:
        a_new = a_new / S
    else:
        a_new = a.copy()
    return a_new


def noise_perturb_run(a, rng, scale):
    n = len(a)
    a_new = a * (1.0 + scale * rng.standard_normal(n))
    a_new = np.maximum(a_new, 0.0)
    S = float(a_new.sum())
    if S > 1e-12:
        a_new = a_new / S
    else:
        a_new = a.copy()
    return a_new
