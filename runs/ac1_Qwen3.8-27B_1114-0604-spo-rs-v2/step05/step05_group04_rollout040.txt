def lse_natural_objective(x, beta):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    a = np.maximum(x,0.0)
    S = float(a.sum())
    if S <= 1e-300:
        return 1e100, np.zeros(n)
    u = a/S
    c = full_convolve(u)
    c = np.maximum(c,0.0)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <=0: return 0, zeros
    z = float(beta)*(c-cmax)
    # To avoid over/under, z max0. But if beta inf? finite.
    exp_z = np.exp(z)  # z<=0 safe; but z maybe nan if c nan
    zsum = float(exp_z.sum())

        # fallback p at max
        p = np.zeros_like(c); max_idx=np.argmax(c); p[max_idx]=1; zsum=1; L=cmax
    else:
        L = cmax + float(np.log(zsum))/float(beta)
        p = exp_z/zsum
    score = 2.0*n*L

    if not np.all(np.isfinite(grad)): grad = np.zeros(n)
    return float(score), grad

```python
"""
Search algorithm for nonnegative sequences minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Method summary:
1. Build several initializations: parent (if available), central-binomial one-sided sequences,
   weighted/mixed variants, power-law sequences, and random textures.
2. Use a correct "square-spectrum" Fienup-style iteration:
   - compute c = conv(a,a),
   - build a flattened target coefficient vector c_t,
   - modify the Fourier spectrum A of a by easing |A| or complex phase/magnitude toward sqrt(FFT(c_t)),
   - backproject, enforce nonnegativity, normalize, and accept only improvements.
3. Polish with L-BFGS-B on a log-sum-exp smoothed maximum-convolution objective.
   The objective is optimized on the real nonnegative variables, but evaluations normalize
   internally so the scale-invariance of the score is handled exactly in the gradient.
4. Final exact-peak active-set subgradient polish and a few high-beta polishes.

The final returned sequence is scaled to sum 1, nonnegative, finite, and clipped to [0, 1000].
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def full_convolve(a):
    """Linear convolution a * a via zero-padded FFT."""
    n = int(len(a))
    K = 2 * n - 1
    N = next_pow2(K)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:K]


def score_seq(a):
    """Internal objective, scale invariant."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    c = full_convolve(a)
    M = float(np.max(c))
    if not np.isfinite(M):
        return np.inf
    return float(2.0 * a.size * M / (S * S))


def normalize_nonneg(a):
    """Clip to nonnegative finite values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S <= 1e-300:
        return np.full(a.shape, 1.0 / float(a.size), dtype=np.float64)
    return a / S


def central_binomial(n):
    """
    Central binomial probabilities:
        c_k = binom(2k,k) / 4^k.
    Its infinite square has constant convolution coefficients.
    """
    n = max(1, int(n))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (4.0 * i - 2.0) / (4.0 * i)
    return a


def weighted_central(n, w, r):
    """Boost the right tail of the central-binomial sequence."""
    n = max(1, int(n))
    c = central_binomial(n)
    if n == 1:
        return normalize_nonneg(c)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    boost = np.power(np.maximum(x, 1e-15), float(max(0.01, r)))
    a = c * (1.0 + float(w) * boost)
    return normalize_nonneg(a)


def mixed_central(n, w):
    """Mix a one-sided central-binomial sequence with its reverse."""
    n = max(1, int(n))
    c = central_binomial(n)
    cr = c[::-1].copy()
    a = float(w) * c + (1.0 - float(w)) * cr
    return normalize_nonneg(a)


def power_seq(n, alpha, beta):
    """
    Power-law amplitude:
        (i+1)^(-alpha) * (n-i)^(-beta).
    """
    n = max(1, int(n))
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(np.maximum(x, 1e-15), -float(alpha))
    b = np.power(np.maximum(float(n + 1) - x, 1e-15), -float(beta))
    a = a * b
    a = np.maximum(a, 1e-300)
    return normalize_nonneg(a)


def resample_sequence(a, n):
    """Linearly resample a sequence to length n on [0,1]."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = max(1, int(n))
    m = int(a.size)
    if m == n:
        return a.copy()
    if m == 0:
        return np.ones(n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.interp(x_new, x_old, a)


def weighted_corr(w, u):
    """
    Compute h_j = sum_k w_k u_{k-j} for j=0..n(u)-1.
    This is the quadratic gradient of weighted convolution coefficients.
    """
    n = int(len(u))
    M2 = next_pow2(3 * n - 2)
    Wf = np.fft.rfft(w, M2)
    Uf = np.fft.rfft(u[::-1], M2)
    return np.fft.irfft(Wf * Uf, M2)[n - 1:2 * n - 1]


def spectral_square_fienup(a,
                           n_iters=100,
                           target_mode=0,
                           p0=0.2,
                           p1=0.7,
                           mix0=0.15,
                           mix1=0.5,
                           phase=0,
                           stop_time=None):
    """
    Square-spectrum Fienup iteration.

    target_mode:
        0: C_t = (1-rho)c + rho*mean(c)
        1: C_t = c**q normalized
        2: water-filling cap on c

    phase:
        0: preserve phase of A, mix amplitudes toward sqrt(|FFT(C_t)|)
        1: mix complex spectral ratio toward sqrt(FFT(C_t))
    """
    a = normalize_nonneg(a)
    n = int(len(a))
    K = 2 * n - 1
    N = next_pow2(K)
    avg = 1.0 / K

    cur = a.copy()
    cur_sc = score_seq(cur)
    best = cur.copy()
    best_sc = cur_sc
    no_improve = 0

    n_iters = max(1, int(n_iters))

    for it in range(n_iters):
        if stop_time is not None and time.monotonic() > stop_time:
            break

        c = full_convolve(cur)
        c = np.maximum(c, 0.0)
        csum = float(c.sum())
        if csum <= 1e-300:
            break
        c = c / csum

        progress = 0.0 if n_iters == 1 else float(it) / float(n_iters - 1)
        mixA = float(mix0) + (float(mix1) - float(mix0)) * progress

        # Build target coefficient profile.
        if target_mode == 0:
            rho = float(p0) + (float(p1) - float(p0)) * progress
            rho = min(1.0, max(0.0, rho))
            Ct = (1.0 - rho) * c + rho * avg
        elif target_mode == 1:
            q = float(p0) + (float(p1) - float(p0)) * progress
            if q < 0.05:
                q = 0.05
            Ct = np.power(np.maximum(c, 0.0), q)
            s = float(Ct.sum())
            if s <= 1e-300:
                Ct = c.copy()
            else:
                Ct = Ct / s
        else:
            tf = float(p0) + (float(p1) - float(p0)) * progress
            tf = min(1.0, max(0.0, tf))
            tau = float(np.quantile(c, tf))
            cap = np.minimum(c, tau)
            deficit = 1.0 - float(cap.sum())
            if deficit < 0.0:
                deficit = 0.0
            Ct = cap + deficit / K

        Ct = np.maximum(Ct, 0.0)
        s = float(Ct.sum())
        if s <= 1e-300:
            continue
        Ct = Ct * (1.0 / s)

        D = np.fft.rfft(Ct, N)
        D[-1] = complex(float(np.real(D[-1])), 0.0)

        A = np.fft.rfft(cur, N)
        A[-1] = complex(float(np.real(A[-1])), 0.0)

        if phase == 0:
            Aamp = np.abs(A)
            T_amp = np.sqrt(np.abs(D))
            amax = float(np.max(Aamp)) if Aamp.size else 1.0
            tmax = float(np.max(T_amp)) if T_amp.size else 1.0
            floorA = max(1e-300, 1e-15 * max(amax, 1.0))
            floorT = max(1e-300, 1e-15 * max(tmax, 1.0))

            Asafe = np.maximum(Aamp, floorA)
            Tsafe = np.maximum(T_amp, floorT)
            new_amp = np.exp((1.0 - mixA) * np.log(Asafe) + mixA * np.log(Tsafe))

            phase_arr = np.ones_like(A, dtype=np.complex128)
            maskA = Aamp > floorA
            if maskA.any():
                phase_arr[maskA] = A[maskA] / Aamp[maskA]

            # For frequencies where old A is too small, use target phase.
            Tph = np.zeros_like(A, dtype=np.complex128)
            maskT = T_amp > floorT
            if maskT.any():
                sqrtD = np.sqrt(D[maskT])
                Tph[maskT] = sqrtD / T_amp[maskT]
            if (~maskA).any():
                phase_arr[~maskA] = Tph[~maskA]

            A_new = phase_arr * new_amp

        else:
            Cfft = A * A
            cmaxf = float(np.max(np.abs(Cfft))) if Cfft.size else 1.0
            eps = max(1e-300, 1e-12 * max(cmaxf, 1.0))
            mask = np.abs(Cfft) > eps
            A_new = np.zeros_like(A, dtype=np.complex128)

            if mask.any():
                Dm = D[mask]
                C2m = Cfft[mask]
                Am = A[mask]
                zeroD = np.abs(Dm) <= eps
                vals = np.zeros_like(Am, dtype=np.complex128)

                if (~zeroD).any():
                    ratio = Dm[~zeroD] / C2m[~zeroD]
                    absr = np.abs(ratio)
                    ratio2 = ratio.copy()
                    small = absr < 1e-12
                    large = absr > 1e12
                    if small.any():
                        ratio2[small] = ratio[small] * (1e-12 / absr[small])
                    if large.any():
                        ratio2[large] = ratio[large] * (1e12 / absr[large])
                    vals[~zeroD] = Am[~zeroD] * np.exp(0.5 * mixA * np.log(ratio2))

                if zeroD.any():
                    vals[zeroD] = 0.0
                A_new[mask] = vals

            if (~mask).any():
                A_new[~mask] = np.sqrt(D[~mask] + 0.0j)

        A_new[-1] = complex(float(np.real(A_new[-1])), 0.0)

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new <= 1e-300:
            no_improve += 1
            if no_improve > 40:
                break
            continue
        a_new = a_new / S_new

        sc = score_seq(a_new)
        if np.isfinite(sc) and sc < cur_sc - 1e-14:
            cur = a_new.copy()
            cur_sc = sc
            no_improve = 0
        else:
            accepted = False
            for t in (0.6, 0.3, 0.12, 0.05):
                cand = (1.0 - t) * cur + t * a_new
                scnd = float(cand.sum())
                if scnd <= 1e-300:
                    continue
                cand = cand / scnd
                sc2 = score_seq(cand)
                if np.isfinite(sc2) and sc2 < cur_sc - 1e-14:
                    cur = cand
                    cur_sc = sc2
                    accepted = True
                    no_improve = 0
                    break
            if not accepted:
                no_improve += 1
                if no_improve > 60:
                    break

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = cur.copy()

    return best


def lse_natural_objective(x, beta):
    """
    Log-sum-exp smoothed score with exact scale-invariant gradient.
    x is the raw nonnegative variable. Internally u = x/sum(x).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(x.size)
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)

    a = np.maximum(x, 0.0)
    S = float(a.sum())
    if S <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    u = a / S
    c = full_convolve(u)
    c = np.maximum(c, 0.0)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if beta <= 1.0:
        p = np.full_like(c, 1.0 / float(c.size))
        L = float(np.mean(c))
    else:
        z = beta * (c - cmax)
        exp_z = np.exp(z)
        zsum = float(exp_z.sum())
        if not np.isfinite(zsum) or zsum <= 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
            zsum = 1.0
            L = cmax
        else:
            L = cmax + float(np.log(zsum)) / beta
            p = exp_z / zsum

    score = float(2.0 * n * L)

    corr = weighted_corr(p, u)
    g_u = 2.0 * corr
    grad = (2.0 * n / S) * (g_u - 2.0 * L)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)

    return score, grad


def run_lbfgs_natural(a0, beta, maxfun=500):
    """Run L-BFGS-B on the natural nonnegative variables."""
    a0 = normalize_nonneg(a0)
    n = int(len(a0))
    if n < 2:
        return score_seq(a0), a0.copy()

    base_sc = score_seq(a0)
    if not np.isfinite(base_sc):
        return base_sc, a0.copy()

    bounds = [(0.0, None)] * n
    maxfun = max(20, int(maxfun))

    try:
        res = minimize(
            lse_natural_objective,
            a0.copy(),
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1e-18,
                "gtol": 1e-14,
                "maxfun": maxfun,
                "maxiter": max(fun if (fun := maxfun) else 20, 20),
                "maxls": 40
            }
        )
        a_out = np.asarray(res.x, dtype=np.float64).ravel()
    except Exception:
        return base_sc, a0.copy()

    if not a_out.shape or not np.all(np.isfinite(a_out)):
        return base_sc, a0.copy()

    a_out = np.maximum(a_out, 0.0)
    S = float(a_out.sum())
    if S <= 1e-300:
        return base_sc, a0.copy()

    a_out = a_out / S
    sc_out = score_seq(a_out)

    if not np.isfinite(sc_out) or sc_out >= base_sc:
        return base_sc, a0.copy()

    return sc_out, a_out


def peak_gradient_polish(a, n_iters=200, stop_time=None):
    """
    Active-set-like projected subgradient polish on the maximum convolution coefficient.
    Uses a soft active set and backtracking, accepting only real improvements.
    """
    a = normalize_nonneg(a)
    n = int(len(a))
    K = 2 * n - 1
    avg = 1.0 / K

    cur = a.copy()
    cur_sc = score_seq(cur)
    best = cur.copy()
    best_sc = cur_sc

    step = 1e-3
    no_improve = 0
    n_iters = max(1, int(n_iters))

    for it in range(n_iters):
        if stop_time is not None and time.monotonic() > stop_time:
            break

        c = full_convolve(cur)
        c = np.maximum(c, 0.0)
        csum = float(c.sum())
        if csum <= 1e-300:
            break
        c = c / csum

        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        # Soft active set: all coefficients near the peak.
        thr = M - 0.25 * max(0.0, M - avg)
        w = (c >= thr).astype(np.float64)
        if float(w.sum()) <= 0.0:
            w[int(np.argmax(c))] = 1.0
        w = w / float(w.sum())

        corr = weighted_corr(w, cur)
        g = 2.0 * corr
        # Tangent to the simplex (constant-sum directions only).
        g = g - float(np.mean(g))

        maxg = float(np.max(np.abs(g)))
        if not np.isfinite(maxg) or maxg <= 1e-20:
            break

        step = min(1e-2, 0.02 / maxg)

        # Ensure nonnegativity in the descent direction a - step*g.
        pos = g > 0.0
        if pos.any():
            denom = g[pos]
            ms = float(np.min(cur[pos] / denom))
            if np.isfinite(ms) and ms > 0.0:
                step = min(step, 0.5 * ms)

        if step <= 1e-14:
            break

        accepted = False
        for frac in (1.0, 0.5, 0.2, 0.05):
            cand = cur - step * frac * g
            cand = np.maximum(cand, 0.0)
            S = float(cand.sum())
            if S <= 1e-300:
                continue
            cand = cand / S
            sc = score_seq(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best = cand.copy()
                cur = cand
                accepted = True
                no_improve = 0
                break

        if accepted:
            # Keep the step when successful.
            pass
        else:
            step *= 0.5
            no_improve += 1
            if no_improve > 40:
                break

    return best


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search entry point.
    """
    t0 = time.monotonic()
    total = max(600.0, float(budget_s))
    # Leave a hard reserve for final verification and return.
    final_end = t0 + total - 150.0
    spec_end = t0 + 0.36 * (total - 150.0)
    lbfgs_end = t0 + 0.78 * (total - 150.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        p_raw = height_sequence_1
        p = np.asarray(p_raw, dtype=np.float64).ravel()
        if p.size >= 64:
            MAX_OUT_N = 12288
            p = np.maximum(p, 0.0)
            p = np.where(np.isfinite(p), p, 0.0)
            if p.size > MAX_OUT_N:
                p = resample_sequence(p, MAX_OUT_N)
            if p.sum() > 1e-12:
                parent = normalize_nonneg(p)
    except Exception:
        parent = None

    MAX_OUT_N = 12288
    cand_lengths = {MAX_OUT_N, 10240, 8192, 6962, 4096}
    if parent is not None:
        cand_lengths.add(int(len(parent)))
    lengths = sorted([l for l in cand_lengths if 512 <= l <= MAX_OUT_N])

    best_a = None
    best_sc = np.inf

    scored = []
    idx = 0

    # ------------------------------------------------------------------
    # Phase 0: diverse initializations.
    # ------------------------------------------------------------------
    for n in lengths:
        inits = []

        if parent is not None:
            if int(len(parent)) == n:
                inits.append(parent.copy())
            else:
                inits.append(resample_sequence(parent, n))

        inits.append(central_binomial(n))
        inits.append(weighted_central(n, 0.15, 2.0))
        inits.append(weighted_central(n, 0.30, 1.0))
        inits.append(weighted_central(n, 0.45, 0.6))
        inits.append(mixed_central(n, 0.8))
        inits.append(mixed_central(n, 0.6))
        inits.append(power_seq(n, 0.50, 0.00))
        inits.append(power_seq(n, 0.45, 0.10))
        inits.append(power_seq(n, 0.55, 0.00))
        inits.append(np.maximum(0.0, rng.standard_normal(n)))
        inits.append(rng.exponential(scale=1.0, size=n) + 1e-3)

        for a0 in inits:
            a = normalize_nonneg(a0)
            sc = score_seq(a)
            if not np.isfinite(sc):
                continue
            scored.append((sc, idx, a))
            idx += 1
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    if best_a is None:
        best_a = np.ones(4096, dtype=np.float64)
        best_a = normalize_nonneg(best_a)
        best_sc = score_seq(best_a)
        scored.append((best_sc, 0, best_a.copy()))
        idx = 1

    scored.sort()

    print(f"[init] lengths={lengths}")
    print(f"[init] parent={'yes' if parent is not None else 'no'} best={best_sc:.10f}")

    # Build base list for spectral refinement: include best and a few distinct-length elites.
    spectral_bases = [(best_sc, best_a.copy())]
    seen_len = {int(len(best_a))}
    for sc, i, a in scored:
        if len(spectral_bases) >= 4:
            break
        ln = int(len(a))
        if ln in seen_len:
            continue
        if not np.isfinite(sc) or sc > best_sc + 0.04:
            continue
        seen_len.add(ln)
        spectral_bases.append((sc, a.copy()))

    # ------------------------------------------------------------------
    # Phase 1: square-spectrum Fienup.
    # ------------------------------------------------------------------
    variants = [
        # (target_mode, p0, p1, mix0, mix1, phase)
        (0, 0.10, 0.45, 0.05, 0.25, 0),
        (0, 0.15, 0.65, 0.10, 0.40, 0),
        (0, 0.20, 0.80, 0.15, 0.55, 0),
        (0, 0.25, 0.70, 0.20, 0.55, 1),
        (1, 0.90, 0.50, 0.10, 0.40, 0),
        (1, 0.85, 0.35, 0.15, 0.55, 1),
        (2, 0.70, 0.95, 0.15, 0.55, 0),
        (2, 0.85, 0.99, 0.10, 0.40, 0),
    ]

    for bi in range(len(spectral_bases)):
        if time.monotonic() >= spec_end - 30:
            break
        base_sc, base_a = spectral_bases[bi]
        base_a = normalize_nonneg(base_a)

        for mode, p0, p1, mix0, mix1, phase in variants:
            if time.monotonic() >= spec_end - 30:
                break
            rem = spec_end - time.monotonic()
            if rem < 25.0:
                break

            stop = time.monotonic() + min(120.0, max(20.0, rem - 20.0))
            nit = 100 if int(len(base_a)) <= 8192 else 80

            a_out = spectral_square_fienup(
                base_a,
                n_iters=nit,
                target_mode=mode,
                p0=p0,
                p1=p1,
                mix0=mix0,
                mix1=mix1,
                phase=phase,
                stop_time=stop
            )
            sc = score_seq(a_out)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_out.copy()
                spectral_bases[0] = (best_sc, best_a.copy())
                if bi == 0:
                    base_a = best_a.copy()

    print(f"[spec] best={best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # ------------------------------------------------------------------
    # Phase 2: L-BFGS with log-sum-exp smoothing.
    # ------------------------------------------------------------------
    lbfgs_bases = []
    seen_l = {int(len(best_a))}
    lbfgs_bases.append(best_a.copy())

    if parent is not None:
        ln = int(len(parent))
        if ln not in seen_l:
            seen_l.add(ln)
            lbfgs_bases.append(parent.copy())

    for sc, i, a in scored:
        if len(lbfgs_bases) >= 3:
            break
        ln = int(len(a))
        if ln in seen_l:
            continue
        if np.isfinite(sc) and sc < best_sc + 0.03:
            seen_l.add(ln)
            lbfgs_bases.append(a.copy())

    betas = [2000, 8000, 30000, 80000]

    for base in lbfgs_bases:
        if time.monotonic() >= lbfgs_end - 30:
            break
        base = normalize_nonneg(base)
        nbase = int(len(base))

        for beta_base in betas:
            if time.monotonic() >= lbfgs_end - 30:
                break
            rem = lbfgs_end - time.monotonic()
            if rem < 60.0:
                break

            # Scale beta with length because coefficient values scale like 1/n.
            beta_eff = max(100, int(beta_base * (nbase / 6962.0)))

            if nbase <= 8192:
                maxfun = 700
            elif nbase <= 12288:
                maxfun = 450
            else:
                maxfun = 300
            if rem < 180.0:
                maxfun = min(maxfun, 250)
            if rem < 90.0:
                maxfun = min(maxfun, 120)

            sc, a_out = run_lbfgs_natural(base, beta_eff, maxfun=maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_out.copy()
                base = a_out.copy()

    print(f"[lbfgs] best={best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # ------------------------------------------------------------------
    # Phase 3: final polishing: active-set peak subgradient + high-beta LBFGS.
    # ------------------------------------------------------------------
    TARGET_INTERNAL = 1.50245

    if time.monotonic() < final_end - 60:
        rem = final_end - 40 - time.monotonic()
        if rem > 30:
            stop = time.monotonic() + min(100.0, rem)
            a_out = peak_gradient_polish(best_a, n_iters=200, stop_time=stop)
            sc = score_seq(a_out)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_out.copy()

    if time.monotonic() < final_end - 60:
        rem = final_end - 40 - time.monotonic()
        if rem > 30:
            nbase = int(len(best_a))
            beta_eff = max(40000, int(40000 * (nbase / 6962.0)))
            maxfun = 250 if nbase <= 8192 else 160
            sc, a_out = run_lbfgs_natural(best_a, beta_eff, maxfun=maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_out.copy()

    # Tiny perturbation restart if still above the safety threshold and time permits.
    if best_sc > TARGET_INTERNAL and time.monotonic() < final_end - 45:
        rem = final_end - 30 - time.monotonic()
        if rem > 30:
            for eps in (0.003, 0.008, 0.015):
                if time.monotonic() >= final_end - 30:
                    break
                nbest = int(len(best_a))
                a_pert = best_a * (1.0 + eps * rng.standard_normal(nbest))
                a_pert = normalize_nonneg(a_pert)
                stop = time.monotonic() + min(45.0, max(15.0, final_end - 35 - time.monotonic()))
                a_out = spectral_square_fienup(
                    a_pert,
                    n_iters=60,
                    target_mode=0,
                    p0=0.20,
                    p1=0.60,
                    mix0=0.10,
                    mix1=0.35,
                    phase=0,
                    stop_time=stop
                )
                sc = score_seq(a_out)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_out.copy()

    print(f"[final] best internal={best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # ------------------------------------------------------------------
    # Prepare final output.
    # ------------------------------------------------------------------
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)

    S = float(out.sum())
    if S <= 0.0:
        out = np.full(64, 0.5, dtype=np.float64)
    else:
        out = out / S
        out = np.clip(out, 0.0, 1000.0)

    out_list = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(out_list))
        print(f"[verify] evaluate_sequence={verified:.10f}")
    except Exception as e:
        print(f"[verify] failed: {e}")

    return out_list
```
