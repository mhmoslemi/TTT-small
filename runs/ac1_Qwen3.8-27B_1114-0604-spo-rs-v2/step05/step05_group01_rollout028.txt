"""

using LSE-smoothed L-BFGS-B, LP-based balancing (Gerchberg-Saxton style),
anti-palindromic spectral initialization, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize
import scipy.sparse as sp


def _next_pot(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pot(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _convolve_pair(a, b):
    """Full linear convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na, nb = int(a.size), int(b.size)
    if na == 0 or nb == 0:
        return np.empty(0, dtype=np.float64)
    M = na + nb - 1
    N = _next_pot(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _correlate_for_grad(a, w):
    """Compute the gradient-relevant cross-correlation slice.
    Given convolution c = conv(a, a), d/dc_j/da_i involves w.
    Specifically, if L = sum_j w_j * c_j, then dL/da_i = 2 * (w_conv_a)_i
    where (w_conv_a)_i = sum_j w_j * a_{j-i+2n-2} ... 
    Actually: c_k = sum_i a_i * a_{k-i}, so dc_k/da_i = a_{k-i}.
    dL/da_i = sum_k w_k * a_{k-i} = (w * a_reversed)_i (appropriate slice).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or int(w.size) == 0:
        return np.zeros(n, dtype=np.float64)
    # We need: for each i, sum_k w[k] * a[k-i] where 0<=k-i<n
    # This is the cross-correlation of w and a, sliced to length n.
    # Equivalently: conv(a_reversed, w)[n-1 : 2n-1]
    ar = a[::-1]
    full = _convolve_pair(ar, w)
    # full has length n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    # We want indices [n-1, n-1+n) = [n-1, 2n-1)
    result = full[n - 1:2 * n - 1]
    return result


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def _resample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / int(n)
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

    # sum_j w_j * dd_j/da_i = n/S^2 * sum_j w_j * a_{j-i} - 2n*a_i/S^3 * sum_j w_j * c_j
    # = n/S^2 * h_i - 2n*a_i/S^3 * (S^2/n) * sum_j w_j * d_j
    # = n/S^2 * h_i - 2*a_i/S * LDS
    
    # The objective is 2*L, so gradient is 2 * dL/da:
    # grad_i = 2*n/S^2 * h_i - 4*a_i/S * LDS
    
    grad = (2.0 * float(n) / (S * S)) * h - (4.0 * LDS / S) * a

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    score = 2.0 * L
    return float(score), grad


def _fun_lse(a, beta):
    return _lse_score_grad(a, float(beta))


def _run_lbfgs(a0, beta, deadline, maxfun=None):
    a = _normalize(a0)
    n = len(a)
    if n == 0:
        return np.inf, a

    rem = float(deadline - time.monotonic())
    if rem < 5.0:
        return _exact_score(a), a

    if maxfun is None:
        maxfun = int(min(2000, max(64, int(rem * 20))))
    maxfun = max(64, int(maxfun))
    maxiter = max(30, maxfun // 2)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            _fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None and res.x is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def _subgrad_polish(a, max_iters=30, deadline=None):
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _exact_score(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc

        for k in (1, 2, 3, 5, 10):
            for mult in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                c = _fast_self_conv(a)
                d = c / (float(c.max()) + 1e-30)
                w = _active_weights(d, k)
                h = _correlate_for_grad(a, w)
                hmax = float(np.max(np.abs(h)))
                if hmax < 1e-15:
                    continue
                q = a * np.exp(-mult * h / hmax)
                q = _normalize(q)
                sc = _exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def _lp_balance(f, M, deadline):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b)<=M.
    Only called for small n. Returns b (or None if infeasible/too slow).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    if n < 64 or n > 4096:
        return None

    # Add small epsilon to avoid unboundedness
    f_pos = f + 1e-10
    f_pos = np.maximum(f_pos, 1e-10)

    # Build sparse convolution matrix C of shape (2n-1, n)
    # C[k, j] = f_pos[k-j] if 0 <= k-j < n, else 0
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        idxs = k - js  # these are the indices into f_pos
        rows_list.extend([k] * len(js))
        cols_list.extend(js.tolist())
        vals_list.extend(f_pos[idxs].tolist())

    if len(vals_list) == 0:
        return None

    C = sp.csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            return None

    if time.monotonic() - t0 > float(deadline - time.monotonic()):
        return None

    if b.value is None:
        return None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.nan_to_num(b_val, nan=0.0, posinf=0.0, neginf=0.0)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-10:
        return None
    return b_val


def _generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(arr.min())
        if mn < 0.0:
            arr = arr - mn + 0.01
        p = _normalize(arr)
        inits.append(p)

    if parent is not None:
        pr = _resample(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n, dtype=np.float64))

    # Anti-palindromic: high on one side, low on the other
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(1.0 + alpha * x)
        add(1.0 + alpha * (1.0 - x))

    # V-shape (high at ends)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Inverted V (high in middle)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(1.0 + alpha * (1.0 - np.abs(2.0 * x - 1.0)))

    # Cosine shapes
    for f in (1, 2, 3, 4):
        for alpha in (0.5, 1.0, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x + 0.5))
            add(1.0 + alpha * np.cos(np.pi * (2.0 * f * x - 0.5)))

    # Exponential tapers (asymmetric)
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(0.1 + np.exp(-rate * x))
        add(0.1 + np.exp(-rate * (1.0 - x)))

    # Two-peak (asymmetric positions)
    for w1 in (0.08, 0.15, 0.25):
        for c1 in (0.2, 0.3, 0.4):
            for c2 in (0.6, 0.7, 0.8):
                add(np.exp(-(((x - c1) / w1) ** 2)) + 0.5 * np.exp(-(((x - c2) / w1) ** 2)))

    # Power law
    for p in (0.5, 1.0, 2.0, 3.0):
        add(np.power(x + 0.01, p))
        add(np.power(1.0 - x + 0.01, p))

    # Sinusoidal with asymmetry
    for f in (1, 2, 3):
        add(1.0 + 0.8 * np.sin(np.pi * x + 0.3))
        add(1.0 + 0.8 * np.sin(2.0 * np.pi * f * x))

    # Spectral: inverse FFT of flat-ish magnitude
    N = _next_pot(2 * n - 1)
    for _ in range(4):
        mag = np.ones(N // 2 + 1, dtype=np.float64)
        # Add some randomness to magnitude
        mag *= 1.0 + rng.uniform(-0.3, 0.3, size=mag.size)
        mag = np.maximum(mag, 0.1)
        phase = rng.uniform(-np.pi, np.pi, size=mag.size)
        # Ensure conjugate symmetry for real output
        spec = mag * np.exp(1j * phase)
        full_spec = np.zeros(N, dtype=np.complex128)
        full_spec[:N // 2 + 1] = spec
        for k in range(1, N // 2):
            full_spec[N - k] = np.conj(full_spec[k])
        sig = np.fft.ifft(full_spec).real[:n]
        add(sig)

    # Random smooth (low-pass filtered noise)
    Nf = _next_pot(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.min()) + 0.5
        add(noise)

    # Block patterns
    for nb in (4, 8, 16, 32, 64):
        if nb > n // 2:
            continue
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = n // nb
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def _perturb(a, rng, strength=None):
    """Generate a perturbed version of a."""
    p = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.2, 0.7))
    mode = int(rng.integers(0, 8))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.15 * rng.standard_normal(n))
    elif mode == 3:
        # Subgradient-style
        c = _fast_self_conv(p)
        d = c / (float(c.max()) + 1e-30)
        w = _active_weights(d, int(rng.integers(1, 8)))
        h = _correlate_for_grad(p, w)
        hmax = float(np.max(np.abs(h)))
        if hmax > 1e-15:
            q = p * np.exp(-strength * h / hmax)
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed Gaussian noise in log domain
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 200))
        if k % 2 == 0:
            k += 1
        if 3 <= k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp += strength * 0.5 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Add V-shape
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + 0.5 * strength * np.abs(2.0 * x - 1.0))
    elif mode == 6:
        # Multiply by asymmetric ramp
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.3 * (x - 0.5))
    else:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize(a, deadline):
    """Run a quick multi-beta LSE optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _exact_score(a)

    for beta in (100, 300, 1000, 3000, 10000, 30000):
        rem = float(deadline - time.monotonic())
        if rem < 8.0:
            break
        mf = int(min(1500, max(64, int(rem * 10))))
        sc, a2 = _run_lbfgs(a, float(beta), deadline,.maxfun=mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = _normalize(a2)

    return float(best_sc), best_a


def _optimize_trajectory(a_start, deadline, rng, label=""):
    """Run LSE optimization with continuation on beta, plus subgradient polish."""
    a = _normalize(a_start)
    best_a = a.copy()
    best_sc = _exact_score(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    n = len(a)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    stages = [
        (50.0, 0.04),
        (100.0, 0.05),
        (200.0, 0.07),
        (400.0, 0.10),
        (800.0, 0.12),
        (1500.0, 0.14),
        (3000.0, 0.14),
        (6000.0, 0.12),
        (12000.0, 0.10),
        (25000.0, 0.08),
        (50000.0, 0.06),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 4.0:
                break
            mf = int(min(2000, max(64, int(rem * 15))))
            sc, a2 = _run_lbfgs(a, float(beta), deadline, maxfun=mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                elapsed = time.monotonic() - t0
                print("[{}] new best {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, elapsed))
            a = _normalize(a2)

        # Subgradient polish between stages
        if time.monotonic() < sub_dead - 4.0:
            sc_sg, a_sg = _subgrad_polish(a, max_iters=8, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = _normalize(a_sg)

    return float(best_sc), best_a


def _lp_incorporate(a, deadline, label=""):
    """Try LP-based balancing if n is small enough."""
    a = _normalize(a)
    n = len(a)
    if n < 128 or n > 4096:
        return _exact_score(a), a

    best_a = a.copy()
    best_sc = _exact_score(a)

    c = _fast_self_conv(a)
    S = float(a.sum())
    M = float(c.max()) * n / (S * S)  # This is max(d) where d = n*c/S^2
    # Actually M for the LP should be the actual max convolution value
    M_actual = float(c.max())

    for _ in range(3):
        if time.monotonic() > float(deadline):
            break
        b = _lp_balance(a, M_actual, deadline)
        if b is None:
            break
        b_norm = _normalize(b)
        for t in (0.1, 0.2, 0.4, 0.6, 0.8):
            q = _normalize((1.0 - t) * a + t * b_norm)
            sc = _exact_score(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                print("[{}] LP blend t={:.1f}: {:.8f}".format(label, t, best_sc))
        a = best_a.copy()
        c = _fast_self_conv(a)
        M_actual = float(c.max())

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Try to get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        pass

    print("Parent: {}, length={}".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Choose lengths to explore
    lengths = []
    if parent is not None:
        pl = int(min(parent.size, 16384))
        if pl < 2048:
            pl = 2048
        lengths.append(pl)
    for L in [8192, 10240, 6144, 12288, 4096, 16384]:
        if L not in lengths:
            lengths.append(L)
    lengths = lengths[:5]
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate and score initializations
    all_scores = []
    all_seqs = []

    for n in lengths:
        inits = _generate_inits(n, rng, parent)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(arr)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init: {:.8f}".format(all_scores[int(order[0])]))

    # Select top 3-4 starts (diverse lengths)
    starts = []
    used = set()
    if parent is not None:
        ps = _normalize(_resample(parent, int(min(parent.size, 16384))))
        starts.append(ps)
        used.add(len(ps))

    for idx in order:
        if len(starts) >= 4:
            break
        n = len(all_seqs[int(idx)])
        if n not in used:
            starts.append(all_seqs[int(idx)].copy())
            used.add(n)

    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Initial best
    best_seq = starts[0].copy()
    best_score = _exact_score(best_seq)
    for i in range(1, len(starts)):
        sc = _exact_score(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (n={})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization trajectories
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))
    n_starts = len(starts_sorted)

    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.6, 0.4]
    elif n_starts == 3:
        fracs = [0.5, 0.3, 0.2]
    else:
        fracs = [0.45, 0.25, 0.18, 0.12]

    main_budget = max(60.0, float(deadline - t_start - 300.0))
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(deadline - 300.0))

        if time.monotonic() > sub_dead - 10.0:
            continue

        sq = starts[si].copy()
        if rank > 0:
            base = best_seq
            if len(base) != len(sq):
                base = _resample(base, len(sq))
            base = _normalize(base)
            t_blend = 0.4 if rank == 1 else 0.5
            sq = _normalize((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = _optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: LP-based balancing (for smaller sequences)
    if time.monotonic() < deadline - 200.0:
        lp_dead = min(deadline - 200.0, time.monotonic() + 180.0)
        if len(best_seq) <= 4096:
            sc_lp, seq_lp = _lp_incorporate(best_seq, lp_dead, "lp")
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = seq_lp.copy()
        else:
            # Try on a resampled smaller version
            small_n = 2048
            small = _normalize(_resample(best_seq, small_n))
            sc_small, small_opt = _lp_incorporate(small, lp_dead, "lp_small")
            if np.isfinite(sc_small) and sc_small < best_score:
                best_score = sc_small
                best_seq = _normalize(_resample(small_opt, len(best_seq)))
            # Also try direct optimization on small
            sc_s2, small_opt2 = _optimize_trajectory(
                _normalize(_resample(best_seq, small_n)), lp_dead, rng, "small")
            if np.isfinite(sc_s2) and sc_s2 < best_score:
                best_score = sc_s2
                best_seq = _normalize(_resample(small_opt2, len(best_seq)))

    # Phase 4: Subgradient polish
    if time.monotonic() < deadline - 150.0:
        sg_dead = min(deadline - 150.0, time.monotonic() + 120.0)
        sc_sg, seq_sg = _subgrad_polish(best_seq, max_iters=30, deadline=sg_dead)
        if np.isfinite(sc_sg) and sc_sg < best_score:
            best_score = sc_sg
            best_seq = seq_sg.copy()
            print("Subgrad: {:.8f}".format(best_score))

    # Phase 5: Perturbation restarts
    perturb_count = 0
    while time.monotonic() < deadline - 120.0 and perturb_count < 5:
        left = float(deadline - 120.0 - time.monotonic())
        if left < 40.0:
            break
        n_rem = 5 - perturb_count
        chunk_end = time.monotonic() + left / n_rem
        p = _perturb(best_seq, rng)
        sc_p, seq_p = _quick_optimize(p, chunk_end)
        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = seq_p.copy()
            print("Perturb {}: {:.8f}".format(perturb_count, best_score))
        perturb_count += 1

    # Phase 6: Final high-beta LSE
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 10.0:
            mf = int(min(3000, max(64, int(rem * 20))))
            sc_f, seq_f = _run_lbfgs(best_seq, 50000.0, deadline, maxfun=mf)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = seq_f.copy()
                print("High beta: {:.8f}".format(best_score))

    if time.monotonic() < deadline - 15.0:
        rem = float(deadline - time.monotonic() - 3.0)
        if rem > 10.0:
            mf = int(min(2000, max(64, int(rem * 20))))
            sc_f2, seq_f2 = _run_lbfgs(best_seq, 100000.0, deadline, maxfun=mf)
            if np.isfinite(sc_f2) and sc_f2 < best_score:
                best_score = sc_f2
                best_seq = seq_f2.copy()
                print("Very high beta: {:.8f}".format(best_score))

    # Phase 7: Final subgradient
    if time.monotonic() < deadline - 8.0:
        sc_final, seq_final = _subgrad_polish(best_seq, max_iters=10, deadline=deadline - 5.0)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = seq_final.copy()

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = float(out_arr.sum())
    # Scale so that sum is in a reasonable range (avoiding clipping issues)
    out_arr = out_arr / S
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
