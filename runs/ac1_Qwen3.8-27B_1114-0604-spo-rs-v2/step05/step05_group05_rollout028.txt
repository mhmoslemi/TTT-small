```python
def project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    # Use the standard algorithm
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / (1.0 / n) > 0)  # Actually: find rho where 1 + rho*u[rho] - cssv[rho] > 0
    # More standard:
    # Find largest rho such that 1 + rho * u[rho] - cssv[rho] > 0
    rho_candidates = np.arange(1, n+1)
    vals = 1.0 + rho_candidates * u[rho_candidates - 1] - cssv[rho_candidates - 1]
    rho = np.sum(vals > 0)
    tau = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(x - tau, 0.0)
```

```python
Convex-optimized non-negative sequence minimizer using top-K subgradient Nesterov acceleration.

Key innovations:
1. Exploits convexity: min max_k (a*a)_k s.t. a>=0, sum(a)=1 is convex
2. Top-K subgradient averaging with adaptive K schedule (large K -> small K)
3. Nesterov acceleration with analytically optimal step size (1/L where L=2)
4. Exact simplex projection (O(n) via sorting)
5. Multiple restarts with different K schedules and initializations
6. LSE post-polish for final convergence

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(x):
    """Euclidean projection onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 1:
        return np.array([1.0])
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    # Find rho: largest index such that 1 + rho * u[rho-1] - cssv[rho-1] > 0
    rho_candidates = np.arange(1, n + 1)
    vals = 1.0 + rho_candidates.astype(np.float64) * u - cssv
    rho = int(np.sum(vals > 0))
    if rho == 0:
        rho = 1
    tau = (cssv[rho - 1] - 1.0) / rho
    result = np.maximum(x - tau, 0.0)
    # Ensure exact sum
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def topk_subgradient(a, K):
    """
    Compute (2/K) * sum_{k in topK} a[k-i] (the top-K averaged subgradient of max c_k).
    Returns the gradient vector of length n.
    """
    n = len(a)
    c = fast_conv(a)
    if K >= len(c):
        top_k = np.arange(len(c))
    else:
        top_k = np.argpartition(c, -K)[-K:]
    top_k = top_k.astype(int)
    grad = np.zeros(n)
    for k in top_k:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            indices = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * a[indices]
    grad /= K
    return grad, c


def nesterov_topk(a, K_start=20, K_end=1, n_iters=50000, step=0.5,
                   momentum_schedule='nesterov', time_limit=60.0):
    """
    Nesterov-accelerated top-K subgradient method on the simplex.
    Minimizes max_k c_k(a) where c_k = (a*a)[k], subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    a_prev = a.copy()
    a_prev2 = a.copy()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t_start = time.monotonic()
    it = 0
    
    while it < n_iters:
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        it += 1
        
        # Adaptive K schedule
        t = min(1.0, it / n_iters)
        K = max(1, int(K_start * (1 - t)**1.5 + K_end * t**1.5))
        
        # Step size schedule (for subgradient, use 1/sqrt(k) or constant with decay)
        # Since each c_k is 2-smooth, use step = 1/2 for the smooth part
        # But for subgradient, use a decaying step
        eta = step / (1 + 0.01 * t)
        
        # Nesterov momentum
        if momentum_schedule == 'nesterov':
            beta = it / (it + 2)
            y = a + beta * (a - a_prev)
        else:
            beta = 0.9 + 0.05 * t
            y = a + beta * (a - a_prev)
        
        # Project y onto simplex (to keep it feasible before gradient step)
        y = project_simplex(y)
        
        # Compute subgradient at y
        grad, c = topk_subgradient(y, K)
        
        # Gradient step + projection
        a_new = project_simplex(y - eta * grad)
        
        # Update
        a_prev2 = a_prev
        a_prev = a
        a = a_new
        
        # Track best (check every 100 iterations)
        if it % 100 == 0 or it == n_iters:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient (for the normalized problem sum(a)=1)."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum
    
    # Gradient of LSE: p weighted average of 2*a shifted
    grad_L = np.zeros(n)
    # Use FFT for efficiency
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
    
    score = 2.0 * n * L  # since sum(a) = 1
    grad = 2.0 * n * grad_L
    return score, grad


def nesterov_lse_simplex(a, beta_start=50.0, beta_end=500.0, n_iters=10000,
                          step=0.3, time_limit=60.0):
    """Nesterov LSE on simplex (sum(a)=1 constraint)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    
    while it < n_iters:
        if time.monotonic() - t_start > time_limit:
            break
        it += 1
        
        t = min(1.0, it / n_iters)
        beta = beta_start * (beta_end / beta_start) ** t
        eta = step * (1.0 - 0.5 * t)
        mu = 0.9 + 0.05 * t
        
        # Nesterov lookahead
        y = a + mu * (a - a_prev)
        y = project_simplex(y)
        
        _, grad = lse_score_and_grad(y, beta)
        
        a_new = project_simplex(y - eta * grad)
        a_prev = a
        a = a_new
        
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def fienup_refined(a, n_iters=200, thresh_start=2.0, thresh_end=1.05,
                   clip_start=0.3, clip_end=0.8, time_limit=30.0):
    """Refined Fienup with gradual threshold annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        bin_thresh = P_mean * thresh
        above = P > bin_thresh
        
        if not above.any():
            continue
        
        scale = np.ones_like(P)
        target_amp = np.sqrt(bin_thresh)
        current_amp = np.sqrt(P[above])
        factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
        scale[above] = np.clip(factor, 0.01, 1.0)
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def lbfgs_polish(a, beta, maxiter=2000, time_limit=60.0):
    """L-BFGS-B polish on LSE-smoothed objective with simplex constraint."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    bounds = [(0.0, None)] * n
    
    def objective(x):
        x = np.maximum(x, 0.0)
        S = x.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        x = x / S
        score, grad = lse_score_and_grad(x, beta)
        return score, grad
    
    t_start = time.monotonic()
    
    def callback(xk):
        return time.monotonic() - t_start > time_limit
    
    res = minimize(objective, a, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                            'maxfun': 100000})
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a.copy()
    else:
        a_out = a_out / S
    return a_out


def coordinate_descent_peaks(a, n_sweeps=3, n_top=20):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = np.argpartition(c, -n_top)[-n_top:]
        
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            
            for j_idx in top_local:
                j = j_start + int(j_idx)
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                for factor in [0.7, 0.85, 1.15, 1.3]:
                    a[j] = old_val * factor
                    S = a.sum()
                    if S < 1e-12:
                        a[j] = old_val
                        break
                    a_norm = a / S
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
    
    return best_a


def build_flat_spectrum_init(n):
    """Initialize by trying to create a flat power spectrum with non-negative time-domain signal."""
    rng = np.random.default_rng(123)
    # Start with random non-negative
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    # Run a few Fienup iterations to flatten spectrum
    M_fft = 1 << (2 * n - 1).bit_length()
    for _ in range(20):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        target = np.ones(len(P)) * P[0]  # target: flat except DC
        target[1:] = P.mean()  # Actually let's target constant power
        scale = np.sqrt(target / np.maximum(P, 1e-20))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a_new, 0.0)
        a = a / a.sum()
    return a


def build_mixed_init(n, seed=42):
    """Mixed initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Combine multiple shapes
    a1 = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    a2 = np.exp(-0.5 * ((x - 0.5) / 0.2)**2)
    a3 = np.abs(np.sin(2 * np.pi * 3 * x)) + 0.1
    a = 0.4 * np.maximum(a1, 0.01) + 0.3 * a2 + 0.3 * a3
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a
```

```python
"""
Convex-optimized non-negative sequence minimizer.
Uses top-K subgradient Nesterov acceleration exploiting the convexity of
min max_k (a*a)_k s.t. a>=0, sum(a)=1. Each c_k is 2-smooth (Hessian spectral norm = 2),
enabling analytically optimal step sizes. Adaptive K schedule provides smooth-to-sharp
transition. Combined with LSE polish, Fienup, and L-BFGS for final convergence.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(x):
    """Euclidean projection onto {a >= 0, sum(a) = 1} in O(n log n)."""
    n = len(x)
    if n == 1:
        return np.array([1.0])
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 + rho_candidates * u - cssv
    rho = int(np.sum(vals > 0))
    if rho < 1:
        rho = 1
    tau = (cssv[rho - 1] - 1.0) / rho
    result = np.maximum(x - tau, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def topk_subgradient(a, K):
    """
    Compute (2/K) * sum_{k in topK} a[k-i] for valid i.
    This is the subgradient of L_K(a) = (1/K)*sum_{k in topK} (a*a)[k].
    Returns (gradient, full_convolution).
    """
    n = len(a)
    c = fast_conv(a)
    if K >= len(c):
        top_k = np.arange(len(c))
    else:
        top_k = np.argpartition(c, -K)[-K:]
    top_k = top_k.astype(int)
    grad = np.zeros(n)
    for k in top_k:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            indices = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * a[indices]
    grad /= K
    return grad, c


def nesterov_topk(a, K_start=20, K_end=1, n_iters=100000, step=0.5,
                  time_limit=120.0, momentum='nesterov'):
    """
    Nesterov-accelerated top-K subgradient on simplex.
    Minimizes max_k (a*a)[k] s.t. a>=0, sum(a)=1.
    Each c_k is 2-smooth, so optimal step = 1/L = 0.5.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    t_nest_prev = 1.0

    while it < n_iters:
        if time.monotonic() - t_start > time_limit:
            break
        it += 1

        # Adaptive K: smooth early, sharp late
        t = min(1.0, it / n_iters)
        if t < 0.3:
            K = K_start
        elif t < 0.7:
            frac = (t - 0.3) / 0.4
            K = max(1, int(K_start * (1 - frac) + K_end * frac))
        else:
            K = max(1, K_end)

        # Nesterov coefficient
        if momentum == 'nesterov':
            t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest_prev * t_nest_prev)) / 2.0
            beta = (t_nest_prev - 1.0) / t_nest
            eta = 2.0 / (2.0 * (t_nest + 1.0))  # 1/(L*(t_k+1)/2) = 2/(L*(t_k+1)), L=2
            t_nest_prev = t_nest
        else:
            beta = 0.9 + 0.05 * t
            eta = step * (1.0 - 0.3 * t)

        # Nesterov preview point
        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        # Subgradient at y
        grad, _ = topk_subgradient(y, K)

        # Gradient step + projection
        a_new = project_simplex(y - eta * grad)

        a_prev = a
        a = a_new

        # Periodic score check
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def nesterov_topk_fixed_step(a, K_start=10, K_end=1, n_iters=50000,
                              step=0.35, time_limit=120.0):
    """Fixed-step variant with simpler momentum for stability."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0

    while it < n_iters:
        if time.monotonic() - t_start > time_limit:
            break
        it += 1

        t = min(1.0, it / n_iters)
        K = max(1, int(K_start * (1 - t) ** 2 + K_end * t ** 2))
        mu = 0.9 + 0.05 * t
        eta = step * (1.0 + 0.5 * (1 - t))

        # Nesterov lookahead
        y = a + mu * v
        y_p = project_simplex(y)

        # Gradient
        grad, _ = topk_subgradient(y_p, K)

        # Update
        a_new = project_simplex(y_p - eta * grad)
        v = a_new - a
        a = a_new

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def lse_score_grad_normalized(a, beta):
    """
    LSE-smoothed objective and gradient for the normalized problem (sum a = 1).
    Objective: 2n * LSE_beta(c) where c = conv(a,a).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_shifted / exp_sum

    # Gradient via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    score = 2.0 * n * L
    grad = 2.0 * n * grad_L
    return score, grad


def nesterov_lse_simplex(a, beta_start=30.0, beta_end=500.0, n_iters=20000,
                          step=0.25, time_limit=120.0):
    """Nesterov LSE on simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-15)

    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0

    while it < n_iters:
        if time.monotonic() - t_start > time_limit:
            break
        it += 1
        t = min(1.0, it / n_iters)
        beta = beta_start * (beta_end / beta_start) ** t
        mu = 0.9 + 0.05 * t
        eta = step * (1.0 - 0.4 * t)

        y = a + mu * (a - a_prev)
        y = project_simplex(y)
        _, grad = lse_score_grad_normalized(y, beta)
        a_new = project_simplex(y - eta * grad)
        a_prev = a
        a = a_new

        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_annealed(a, n_iters=300, time_limit=30.0):
    """Fienup with gradual threshold and clip annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-15)
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        t = it / max(1, n_iters - 1)
        thresh = 2.0 + (1.05 - 2.0) * t
        clip = 0.3 + (0.85 - 0.3) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        above = P > P_mean * thresh
        if not above.any():
            continue

        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        curr = np.sqrt(P[above])
        factor = 1.0 - clip + clip * (target / np.maximum(curr, 1e-10))
        scale[above] = np.clip(factor, 0.01, 1.0)

        a_new = np.fft.irfft(A * scale, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S

        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def lbfgs_polish(a, beta, time_limit=60.0):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-15)

    def obj(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        x = x / s
        return lse_score_grad_normalized(x, beta)

    bounds = [(0.0, None)] * n
    t0 = time.monotonic()

    def cb(xk):
        return time.monotonic() - t0 > time_limit

    try:
        res = minimize(obj, a, method='L-BFGS-B', jac=True,
                       bounds=bounds, callback=cb,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        a_out = np.maximum(res.x, 0.0)
        s = a_out.sum()
        if s < 1e-15:
            return a.copy()
        return a_out / s
    except Exception:
        return a.copy()


def coord_desc_peaks(a, n_sweeps=3, n_top=15):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_sweeps):
        c = fast_conv(a)
        top_idx = np.argpartition(c, -n_top)[-n_top:]
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je < js:
                continue
            contrib = [a[k - j] for j in range(js, je + 1)]
            top_local = np.argsort(contrib)[::-1][:min(8, len(contrib))]
            for li in top_local:
                j = js + int(li)
                ov = a[j]
                if ov < 1e-10:
                    continue
                for f in [0.7, 0.85, 1.2]:
                    a[j] = ov * f
                    s = a.sum()
                    if s < 1e-12:
                        a[j] = ov
                        break
                    an = a / s
                    sc = compute_score(an)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = an.copy()
                        a = an
                        break
                else:
                    a[j] = ov
    return best_a


def build_init_flat(n, seed=0):
    """Init by flattening power spectrum via few Fienup iterations."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    M_fft = 1 << (2 * n - 1).bit_length()
    for _ in range(30):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        target = np.full(len(P), P.mean())
        target[0] = P[0]
        scale = np.sqrt(np.maximum(target, 1e-10) / np.maximum(P, 1e-20))
        a = np.maximum(np.fft.irfft(A * scale, M_fft)[:n], 0.0)
        a = a / max(a.sum(), 1e-15)
    return a


def build_init_cosine(n, freq=1.0):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_init_multi(n, seed=42):
    """Multi-frequency cosine mixture."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.1

        f = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.2 * np.maximum(np.cos(2 * np.pi * f * x + phase), 0)
    return a / a.sum()

                                        n_iters=50000, step=0.45,
                                        time_limit=tl * 0.4)
        update_best(a_r, "fienup+topk")

    print(f"  After Phase 3: {best_score:.8f}")

    # ===== Phase 4: L-BFGS polish (15% time) =====
    p4_end = t_start + 0.83 * (budget_s - 25)
    print("Phase 4: L-BFGS polish")

    for beta in [200, 500, 1500, 5000]:
        if time.monotonic() > p4_end:
            break
        tl = p4_end - time.monotonic()
        if tl < 10:
            break
        a_r = lbfgs_polish(best_a, beta, time_limit=tl * 0.7)
        update_best(a_r, f"lbfgs-{beta}")
        # Quick topK after
        if time.monotonic() < p4_end - 5:
            a_r2 = nesterov_topk_fixed_step(a_r, K_start=3, K_end=1,
                                             n_iters=30000, step=0.45,
                                             time_limit=p4_end - time.monotonic() - 3)
            update_best(a_r2, f"lbfgs{beta}+tk")

    print(f"  After Phase 4: {best_score:.8f}")

    # ===== Phase 5: Final micro-optimization (7% time) =====
    print("Phase 5: Final polish")

    if time.monotonic() < deadline - 15:
        a_cd = coord_desc_peaks(best_a, n_sweeps=3, n_top=12)
        update_best(a_cd, "coord")

    if time.monotonic() < deadline - 10:
        tl = deadline - time.monotonic() - 5
        a_r = nesterov_topk(best_a, K_start=3, K_end=1,
                             n_iters=100000, step=0.4, time_limit=tl * 0.5)
        update_best(a_r, "final-tk")

    if time.monotonic() < deadline - 5:
        tl = deadline - time.monotonic() - 2
        a_f = fienup_annealed(best_a, n_iters=150, time_limit=tl * 0.6)
        update_best(a_f, "final-fienup")

    # Final topK squeeze
    if time.monotonic() < deadline - 3:
        a_r = nesterov_topk_fixed_step(best_a, K_start=2, K_end=1,
                                        n_iters=50000, step=0.3,
                                        time_limit=deadline - time.monotonic() - 1)
        update_best(a_r, "final-squeeze")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
