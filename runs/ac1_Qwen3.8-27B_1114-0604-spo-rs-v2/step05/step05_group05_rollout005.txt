"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines: (1) Fienup spectral shaping, (2) Top-K subgradient with Nesterov momentum,
(3) Active-set refinement targeting specific peaks, (4) LP-based Frank-Wolfe moves,
(5) Coordinate descent on peak contributors. Uses multi-start with diverse inits
and periodic restarts to avoid local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_subgrad_step(a, K=5, lr=0.01, momentum=0.9):
    """
    One step of subgradient descent on the sum of top-K peaks.
    Returns (new_a, score).
    """
    n = len(a)
    b = fast_conv(a)
    b_sorted = np.sort(b)[::-1]
    threshold = b_sorted[min(K, len(b_sorted)) - 1]
    active = b >= threshold
    
    # Gradient: sum over active peaks of 2*a[k-j] for valid j
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end+1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    
    return grad, b, active


def nesterov_topk(a, max_time=60.0, K_start=3, K_end=15, lr_start=0.005,
                  lr_end=0.0005, momentum_start=0.85, momentum_end=0.95):
    """
    Nesterov-accelerated subgradient on top-K peaks with cosine-annealed parameters.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t_start < max_time:
        it += 1
        t = min(1.0, it / max(1, int(max_time * 30)))  # rough progress estimate
        K = int(K_start + (K_end - K_start) * t)
        lr = lr_start * (lr_end / lr_start) ** t
        mu = momentum_start + (momentum_end - momentum_start) * t
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        # Compute subgradient at lookahead
        grad, b, _ = topk_subgrad_step(a_look, K=K)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Update velocity
        v = mu * v - lr * grad / gn
        
        # Step
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        # Track best
        if it % 30 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        
        if patience >= max_patience:
            break
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def fienup_adaptive(a, max_time=60.0, thresh_start=2.5, thresh_end=1.05,
                    clip_start=0.2, clip_end=0.9):
    """
    Adaptive Fienup: reduces high-amplitude frequency bins.
    The threshold and clip are annealed over time.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        t = min(1.0, (time.monotonic() - t_start) / max_time)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        clip = clip_start + (clip_end - clip_start) * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        current = np.sqrt(P[above])
        factor = 1.0 - clip + clip * (target / np.maximum(current, 1e-10))
        factor = np.clip(factor, 0.01, 1.0)
        scale[above] = factor
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
    
    return best_a


def fienup_momentum(a, n_iters=200, momentum=0.3, thresh=1.5, clip=0.6):
    """Fienup with momentum buffer."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh_t = thresh + (1.05 - thresh) * t
        clip_t = clip + (0.9 - clip) * t
        mom_t = momentum * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh_t
        if above.any():
            scale = np.ones_like(P)
            target = np.sqrt(P_mean * thresh_t)
            current = np.sqrt(P[above])
            factor = 1.0 - clip_t + clip_t * (target / np.maximum(current, 1e-10))
            scale[above] = np.clip(factor, 0.01, 1.0)
            A_new = A * scale
        else:
            A_new = A
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        v = mom_t * v + (a_new - a)
        a = a_new + v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def frank_wolfe_move(a, M=None, max_time=30.0):
    """
    Frank-Wolfe (conditional gradient) step using LP.
    Find g >= 0 that maximizes sum(g) subject to (a*g)[k] <= M for all k.
    Then return the step direction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    a = a / S
    
    if M is None:
        b = fast_conv(a)
        M = float(b.max())
    
    # LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0
    # (a*g)[k] = sum_i a_i * g_{k-i} for valid i
    # Variables: g_0, ..., g_{n-1}
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_{j: valid} a[k-j] * g_j  (where valid means 0<=k-j<n and 0<=j<n)
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[k-j] * g_j
    
    A_ub_rows = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        row = np.zeros(n)
        row[j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
        A_ub_rows.append(row)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.full(len(A_ub_rows), M)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = float(g.sum())
            if g_sum > 1e-12:
                g = g / g_sum
                # Blend
                return g
    except Exception:
        pass
    
    return a.copy()


def peak_coordinate_descent(a, n_sweeps=5, n_top=20):
    """Coordinate descent on peak-contributing coordinates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Find top contributors to this peak
            contrib = np.array([a[k - j] * a[j] for j in range(j_start, j_end + 1)])
            n_loc = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_loc]
            top_global = j_start + top_local
            
            for j in top_global:
                j = int(j)
                old = a[j]
                if old < 1e-12:
                    continue
                for factor in [0.7, 0.85, 0.9, 1.1, 1.3]:
                    a[j] = old * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old
                        continue
                    sc = compute_score(a / S_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = (a / S_tmp).copy()
                        a = best_a.copy()
                        break
                else:
                    a[j] = old
    
    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    
    # Gradient: dL/da_i = 2 * sum_m a[m] * p[m+i]
    # = 2 * (a * p_rev)[2n-2-i] where * is linear conv
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def nesterov_lse(a, max_time=60.0, beta_start=50.0, beta_end=20000.0,
                 lr=0.008, mu=0.92):
    """Nesterov LSE with adaptive beta."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    beta = beta_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        beta *= 1.003
        if beta > beta_end:
            beta = beta_end
        lr_t = lr * (0.9995 ** it)
        
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        _, grad = lse_score_and_grad(a_look, beta)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        v = mu * v - lr_t * grad / gn
        a = a + v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 10:
                break
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def build_init_type1(n):
    """Raised cosine."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_init_type2(n):
    """Gaussian."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_init_type3(n):
    """Chirp (non-negative)."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.abs(np.sin(2 * np.pi * x * x * 1.5)) + 0.2
    return a / a.sum()


def build_init_type4(n, rng):
    """Random smooth."""
    a = rng.random(n)
    # Smooth with a box filter
    k = max(1, n // 200)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_init_type5(n):
    """Triangular (tent)."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(x - center) / center
    a = np.maximum(a, 0.01)
    return a / a.sum()


def upsample(a, new_n):
    """Linear interpolation upsampling."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase strategy:
    Phase 1: Multi-start initialization + Fienup (20% time)
    Phase 2: Nesterov Top-K subgradient (30% time)
    Phase 3: Frank-Wolfe LP moves + refinement (20% time)
    Phase 4: High-beta LSE polish (15% time)
    Phase 5: Coordinate descent + final polish (15% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    
    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    n_target = len(parent) if parent is not None else 6962
    best_score = np.inf
    best_a = np.ones(n_target) * 0.5
    
    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # ============================================================
    # Phase 1: Multi-start + Fienup (20% of time)
    # ============================================================
    phase1_end = t_start + 0.20 * (total_budget - 20.0)
    print(f"=== Phase 1: Multi-start + Fienup ===")
    
    if parent is not None:
        sc_p = compute_score(parent)
        update_best(sc_p, parent, "parent")
    
    # Try multiple initializations
    inits = []
    if parent is not None:
        inits.append(("parent", parent.copy()))
    
    init_builders = [build_init_type1, build_init_type2, build_init_type3, build_init_type5]
    for i, builder in enumerate(init_builders):
        if i == 3:
            a_init = builder(n_target, rng)
        else:
            a_init = builder(n_target)
        inits.append((f"init{i}", a_init))
    
    # Also a random smooth init
    inits.append(("rand", build_init_type4(n_target, rng)))
    
    # Apply Fienup to each
    candidates = []
    for label, a_c in inits:
        if time.monotonic() > phase1_end - 10:
            break
        a_f = fienup_adaptive(a_c, max_time=min(30.0, phase1_end - time.monotonic() - 5))
        sc = compute_score(a_f)
        update_best(sc, a_f, f"fienup-{label}")
        candidates.append(a_f)
    
    # Also try Fienup with momentum on the current best
    if time.monotonic() < phase1_end - 10:
        a_fm = fienup_momentum(best_a, n_iters=200, momentum=0.4, thresh=1.8, clip=0.5)
        update_best(compute_score(a_fm), a_fm, "fienup-mom-best")
    
    print(f"After Phase 1: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 2: Nesterov Top-K subgradient (30% of time)
    # ============================================================
    phase2_end = t_start + 0.50 * (total_budget - 20.0)
    print(f"\n=== Phase 2: Nesterov Top-K ===")
    
    # Run with different K and lr settings
    topk_configs = [
        {'K_start': 3, 'K_end': 10, 'lr_start': 0.003, 'lr_end': 0.0003},
        {'K_start': 5, 'K_end': 20, 'lr_start': 0.005, 'lr_end': 0.0005},
        {'K_start': 3, 'K_end': 5, 'lr_start': 0.002, 'lr_end': 0.0002},
    ]
    
    for i, cfg in enumerate(topk_configs):
        if time.monotonic() > phase2_end - 15:
            break
        time_left = phase2_end - time.monotonic() - 10
        if time_left < 10:
            break
        print(f"  Top-K config {i}, time={time_left:.0f}s")
        a_tk = nesterov_topk(best_a, max_time=min(120.0, time_left), **cfg)
        update_best(compute_score(a_tk), a_tk, f"topk-{i}")
    
    # Try from a perturbed best
    if time.monotonic() < phase2_end - 30:
        a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_target))
        a_pert = np.clip(a_pert, 0.0, None)
        S_p = a_pert.sum()
        if S_p > 1e-12:
            a_pert = a_pert / S_p
            time_left = phase2_end - time.monotonic() - 10
            a_tk2 = nesterov_topk(a_pert, max_time=min(60.0, time_left),
                                   K_start=5, K_end=15, lr_start=0.004, lr_end=0.0004)
            update_best(compute_score(a_tk2), a_tk2, "topk-pert")
    
    print(f"After Phase 2: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 3: Frank-Wolfe + Fienup interleaved (20% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Frank-Wolfe + Fienup ===")
    
    # Frank-Wolfe moves
    for i in range(3):
        if time.monotonic() > phase3_end - 15:
            break
        time_left = phase3_end - time.monotonic() - 10
        if time_left < 5:
            break
        g = frank_wolfe_move(best_a, max_time=min(20.0, time_left * 0.3))
        # Blend
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_blend = (1 - alpha) * best_a + alpha * g
            a_blend = np.maximum(a_blend, 0.0)
            S_b = a_blend.sum()
            if S_b < 1e-12:
                continue
            a_blend = a_blend / S_b
            sc = compute_score(a_blend)
            if update_best(sc, a_blend, f"fw-alpha={alpha}"):
                break
    
    # Fienup refinement after FW
    if time.monotonic() < phase3_end - 20:
        time_left = phase3_end - time.monotonic() - 10
        a_f3 = fienup_adaptive(best_a, max_time=min(60.0, time_left),
                               thresh_start=1.8, thresh_end=1.05)
        update_best(compute_score(a_f3), a_f3, "fienup-p3")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 4: High-beta LSE polish (15% of time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n=== Phase 4: LSE Polish ===")
    
    lse_configs = [
        {'beta_start': 200.0, 'beta_end': 50000.0, 'lr': 0.005, 'mu': 0.93},
        {'beta_start': 500.0, 'beta_end': 100000.0, 'lr': 0.003, 'mu': 0.95},
        {'beta_start': 100.0, 'beta_end': 30000.0, 'lr': 0.008, 'mu': 0.90},
    ]
    
    for i, cfg in enumerate(lse_configs):
        if time.monotonic() > phase4_end - 15:
            break
        time_left = phase4_end - time.monotonic() - 10
        if time_left < 10:
            break
        print(f"  LSE config {i}, time={time_left:.0f}s")
        a_l = nesterov_lse(best_a, max_time=min(90.0, time_left), **cfg)
        update_best(compute_score(a_l), a_l, f"lse-{i}")
    
    print(f"After Phase 4: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 5: Coordinate descent + final polish (15% of time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: CD + Final Polish ===")
    
    if time.monotonic() < phase5_end - 30:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=6, n_top=25)
        update_best(compute_score(a_cd), a_cd, "cd")
    
    # One more round of Top-K after CD
    if time.monotonic() < phase5_end - 30:
        time_left = phase5_end - time.monotonic() - 20
        a_final_tk = nesterov_topk(best_a, max_time=min(60.0, time_left),
                                    K_start=3, K_end=8, lr_start=0.002, lr_end=0.0002)
        update_best(compute_score(a_final_tk), a_final_tk, "final-topk")
    
    # Final LSE with very high beta
    if time.monotonic() < phase5_end - 20:
        time_left = phase5_end - time.monotonic() - 15
        a_final_lse = nesterov_lse(best_a, max_time=min(60.0, time_left),
                                    beta_start=5000.0, beta_end=200000.0,
                                    lr=0.002, mu=0.96)
        update_best(compute_score(a_final_lse), a_final_lse, "final-lse")
    
    # Final coordinate descent
    if time.monotonic() < phase5_end - 15:
        a_cd2 = peak_coordinate_descent(best_a, n_sweeps=4, n_top=15)
        update_best(compute_score(a_cd2), a_cd2, "cd-final")
    
    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
