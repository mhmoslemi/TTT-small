import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import solve
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_active_set(n, K, S_indices, tol=1e-10):
    """
    Solve the KKT linear system for given active set K and support S.
    Returns (a, C, success).
    """
    m = len(S_indices)
    r = len(K)
    
    # Build the (m+1) x (m+1) system
    # Unknowns: a[S_indices] (m values) and C (1 value)
    # Total: m+1 unknowns
    
    A_mat = np.zeros((m+1, m+1))
    b_vec = np.zeros(m+1)
    
    # Row 0: normalization
    A_mat[0, :m] = 1.0
    b_vec[0] = 1.0
    
    # Rows 1 to m: KKT conditions
    # For each j (index in S), equation: (1/r) * sum_{l} a[K[l] - S[j]] = C
    S_arr = np.array(S_indices)
    K_arr = np.array(K)
    
    for j in range(m):
        s_j = S_arr[j]
        for l in range(r):
            k_l = K_arr[l]
            target = k_l - s_j
            if 0 <= target < n:
                # Find if target is in S
                idx = np.searchsorted(S_arr, target)
                if idx < m and S_arr[idx] == target:
                    A_mat[j+1, idx] += 1.0 / r
        A_mat[j+1, m] = -1.0
        b_vec[j+1] = 0.0
    
    # Solve
    try:
        sol = solve(A_mat, b_vec, assume_a='gen')
    except np.linalg.LinAlgError:
        try:
            sol, *_ = np.linalg.lstsq(A_mat, b_vec, rcond=tol)
        except:
            return None, None, False
    
    a = np.zeros(n)
    for j in range(m):
        a[S_arr[j]] = sol[j]
    C = sol[m]
    
    return a, C, True

def check_validity(a, K, M_target, tol=1e-6):
    """Check if the solution satisfies all KKT conditions."""
    b = fast_conv(a)
    S = a.sum()
    if abs(S - 1.0) > 0.01:
        return False
    
    # Check non-negativity
    if np.any(a < -tol):
        return False
    
    # Check that active peaks are at or near the target
    for k in K:
        if abs(b[k] - M_target) > tol * M_target:
            return False
    
    # Check that no inactive peak exceeds the target
    max_inactive = 0
    for k in range(len(b)):
        if k not in K:
            max_inactive = max(max_inactive, b[k])
    
    if max_inactive > M_target * (1 + tol):
        return False
    
    return True
