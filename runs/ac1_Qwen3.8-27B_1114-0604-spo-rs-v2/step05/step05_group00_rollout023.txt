def run_mirror_descent(a0, beta, n_iters, eta_init, eta_decay, momentum=True):
    """
    Nesterov-accelerated mirror descent (multiplicative weights) on LSE-smoothed objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    eta = eta_init
    t_k = 1.0
    a_prev = a.copy()
    
    for it in range(n_iters):
        # Nesterov lookahead
        if momentum:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
            y = np.maximum(y, 1e-30)  # Keep positive for log
            y = y / y.sum()
        else:
            y = a.copy()
        
        # Compute gradient at y
        f_y, g_y = lse_obj_grad(y, beta)
        
        # Mirror descent step: a_new = y * exp(-eta * g_y) / Z
        # For numerical stability, subtract max
        g_shifted = g_y - g_y.max()
        weights = y * np.exp(-eta * g_shifted)
        Z = weights.sum()
        if Z < 1e-30:
            break
        a_new = weights / Z
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / S_new
        
        # Check improvement (optional, for tracking)
        if it % 200 == 0:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        if momentum:
            t_k = t_new
        
        eta *= eta_decay
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a

def fienup_shaped(a, n_iters=300, shape_param=0.1):
    """
    Fienup with shaped target spectrum.
    Target: P_target[m] ∝ 1 / (1 + shape_param * (m/n)^2)
    This de-emphasizes the DC component slightly, which helps reduce the
    autocorrelation peak at the center.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Precompute target spectrum
    m = np.arange(N // 2 + 1)
    # Shape: slightly reduce DC and high frequencies
    target = 1.0 / (1.0 + shape_param * (m / n) ** 2)
    target = target / target.mean()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        
        # Blend towards target
        alpha = 0.3 + 0.5 * (it / n_iters)
        P_blend = (1 - alpha) * P + alpha * (P.mean() * target)
        P_blend = np.maximum(P_blend, 1e-30)
        
        # Update amplitude, keep phase
        scale = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: shaped-spectrum Fienup, multiplicative weights (mirror descent),
high-beta L-BFGS, peak-pair coordinate descent, multi-scale init, and multi-restart.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/dx[j] = 2 * sum_k p[k]*x[k-j] (valid indices)
    p_flipped = p[::-1]
    M_corr = 2 * n - 1
    M_corr_fft = 1 << (M_corr - 1).bit_length()
    PF = np.fft.rfft(p_flipped, M_corr_fft)
    XF = np.fft.rfft(x, M_corr_fft)
    conv_res = np.fft.irfft(PF * XF, M_corr_fft)[:M_corr]
    dL_dx = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_dx / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(x, beta):
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def run_mirror_descent(a0, beta, n_iters=3000, eta_init=0.02, eta_decay=0.9995, momentum=True):
    """Nesterov-accelerated mirror descent (multiplicative weights)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    t_k = 1.0
    a_prev = a.copy()
    
    for it in range(n_iters):
        if momentum:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y = y / y.sum()
        else:
            t_new = t_k
            y = a.copy()
        
        f_y, g_y = lse_obj_grad(y, beta)
        
        g_shifted = g_y - g_y.max()
        weights = y * np.exp(-eta * g_shifted)
        Z = weights.sum()
        if Z < 1e-30:
            break
        a_new = weights / Z
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / S_new
        
        if it % 300 == 0:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        t_k = t_new
        eta *= eta_decay
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_shaped(a, n_iters=300, shape_param=0.05):
    """Fienup with shaped target spectrum to reduce center peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.arange(N // 2 + 1)
    target = 1.0 / (1.0 + shape_param * (m / n) ** 2)
    target = target / target.mean()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.25 + 0.5 * (it / n_iters)
        P_blend = (1 - alpha) * P + alpha * (P.mean() * target)
        P_blend = np.maximum(P_blend, 1e-30)
        scale = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a


def fienup_flat(a, n_iters=200):
    """Standard Fienup targeting flat spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.2 + 0.5 * (it / n_iters)
        P_target = np.full_like(P, P.mean())
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)
        scale = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a


def peak_pair_cd(a, n_sweeps=5, n_top=200, n_factors=7):
    """
    Peak-pair coordinate descent: for peaks, try multiplicative factor pairs
    on the two most-contributing coordinates to reduce the peak while
    maintaining sum normalization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    factors = [0.5, 0.7, 0.85, 0.9, 1.1, 1.3, 1.5]
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Get top peaks
        top_k = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            if contribs.sum() < 1e-15:
                continue
            # Top 2 contributors
            top2 = np.argsort(contribs)[::-1][:2]
            j1 = j_start + top2[0]
            j2 = j_start + top2[1] if len(top2) > 1 else j1
            
            cur_sc = compute_score(a)
            for f1 in factors:
                for f2 in factors:
                    old1, old2 = a[j1], a[j2]
                    new1 = old1 * f1
                    new2 = old2 * f2
                    if new1 < 0 or new2 < 0:
                        continue
                    if j1 == j2:
                        a[j1] = new1
                    else:
                        a[j1] = new1
                        a[j2] = new2
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j1], a[j2] = old1, old2
                        continue
                    sc = compute_score(a / S_tmp)
                    if sc < cur_sc:
                        cur_sc = sc
                        # Keep this change
                        if j1 != j2:
                            a = a / S_tmp
                        else:
                            a[j1] = new1
                            a = a / a.sum()
                    else:
                        a[j1] = old1
                        if j1 != j2:
                            a[j2] = old2
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def single_coord_cd(a, n_sweeps=5, n_top=100):
    """Single coordinate descent on top-contributing elements at peak lags."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    candidates = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.7, 2.5]
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        cur_sc = compute_score(a)
        for j in top_global:
            old_j = a[j]
            if old_j < 1e-14:
                continue
            best_val = old_j
            best_local = cur_sc
            for cv_factor in candidates:
                cv = old_j * cv_factor
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = old_j
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def structured_perturb(a, rng, n_modes=10, scale=0.03):
    """Low-frequency structured perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):
    """Gaussian noise perturbation."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def interpolated_to_length(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def multi_scale_init(n_target, n_small=512, n_iters_small=100, rng=None):
    """
    Initialize at smaller length using Fienup, then interpolate to target length.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a_small = np.ones(n_small) / n_small
    # Add random perturbation
    a_small = a_small * (1.0 + 0.1 * rng.standard_normal(n_small))
    a_small = np.maximum(a_small, 0.0)
    a_small = a_small / a_small.sum()
    # Run Fienup at small length
    a_small = fienup_flat(a_small, n_iters=n_iters_small)
    # Interpolate to target
    a_target = interpolated_to_length(a_small, n_target)
    a_target = np.maximum(a_target, 0.0)
    S = a_target.sum()
    if S > 1e-12:
        a_target = a_target / S
    return a_target

        maxiter = min(8000, int(remaining * 8))
        if maxiter > 200:
            sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 10)
            update(sc, a_final, "FINAL-LBFGS-50k")
    
    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=3000, maxfun=60000)
        update(sc, a_final, "FINAL-LBFGS-100k")
    
    # Final coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd = single_coord_cd(best_a, n_sweeps=5, n_top=150)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")
    
    # Final peak-pair
    if time.monotonic() < deadline - 10:
        a_cd2 = peak_pair_cd(best_a, n_sweeps=2, n_top=100)
        sc = compute_score(a_cd2)
        update(sc, a_cd2, "FINAL-PEAK-PAIR")
    
    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
