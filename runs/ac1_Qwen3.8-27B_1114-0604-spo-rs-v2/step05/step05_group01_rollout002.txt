"""

Uses a hybrid approach: LSE-smoothed gradient with adaptive step size,

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Compute cross-correlation: result[i] = sum_j a[j]*w[i-j+n-1] for i in 0..n-1"""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)

    w = e / sum_e

    LDS = float(np.dot(w, d))

    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

        count = int(mask.sum())
        if count > 0 and count < m:
            w = np.zeros(m)
            w[mask] = 1.0
            return w / count
    if k <= 0 or k >= m:
        return np.ones(m) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m)
    w[idx] = 1.0
    return w / k


def subgrad_step_direct(a, step, k_active=5, eps_frac=0.0):
    """Direct subgradient step on the max of convolution."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a.copy()
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if dmax <= 0:
        return a.copy()
    w = active_set_weights(d, k_active, eps_frac)
    h = cross_corr(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    direction = -h / hmax
    # Step with projection
    a_new = a + step * direction * S
    a_new = np.clip(a_new, 0.0, None)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a.copy()
    return a_new * (S / S_new)


def subgrad_with_line_search(a, base_step, k_active=5, eps_frac=0.0, max_backtrack=8):
    """Subgradient step with backtracking line search."""
    S = float(np.sum(a))
    cur_score = exact_score(a)
    step = base_step
    for _ in range(max_backtrack):
        a_new = subgrad_step_direct(a, step, k_active, eps_frac)
        sc = exact_score(a_new)
        if np.isfinite(sc) and sc < cur_score:
            return a_new, sc
        step *= 0.5
    return a, cur_score


def lp_direction_approx(a, k_active=10, max_iter=200):
    """
    Approximate LP-based direction finding.
    Find g >= 0 such that cross_conv(f, g) <= M (current max) at all lags,
    maximizing sum(g). We approximate by iterative relaxation.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a.copy()
    c = fast_self_conv(a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()

    # Normalize a to sum=1 for the LP
    a_norm = a / S

    # Start with g = uniform
    g = np.ones(n) / n

    for it in range(max_iter):
        # Compute cross_conv(a_norm, g)
        fc = cross_corr(a_norm, g)
        # Extend to full convolution-like: we need all lags
        # Actually cross_corr gives us the "valid" part. For the full constraint,
        # we need conv(a_norm, g) at all lags 0..2n-2
        full_conv = fast_cross_conv_full(a_norm, g)
        # Find violated constraints
        violation = float(np.max(full_conv)) - M / S
        if violation < 1e-8:
            break
        # Reduce g at positions that contribute most to violated lags
        # Find which lag is most violated
        worst_lag = int(np.argmax(full_conv))
        # Find which g indices contribute to that lag
        # full_conv[k] = sum_j a_norm[j] * g[k-j]
        # For a given lag k, contributing g indices are k-j for valid j
        contrib = np.zeros(n)
        for j in range(n):
            gi = worst_lag - j
            if 0 <= gi < n:
                contrib[gi] += a_norm[j]
        # Reduce g where contribution is high
        max_contrib = float(np.max(contrib))
        if max_contrib > 1e-15:
            g = g - 0.1 * contrib / max_contrib
            g = np.clip(g, 0.0, None)
        else:
            g = g * 0.9

    g = np.clip(g, 0.0, None)
    Sg = float(np.sum(g))
    if Sg < 1e-12:
        return a.copy()
    return g * (S / Sg)


def fast_cross_conv_full(a, b):
    """Full cross-correlation a * b (linear convolution)."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = int(a.size)
    m = int(b.size)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def run_lbfgs_lse(a0, beta, deadline, maxfun=None):
    a = normalize_a(a0)
    n = len(a)
    t0 = time.monotonic()

    def time_check():
        return time.monotonic() < float(deadline)

    if maxfun is None:
        maxfun = int(max(100, min(3000, (float(deadline) - t0 - 10) * 30)))

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": int(max(30, maxfun // 2)),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(
            fun_lse, a.copy(), method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
        if res.x is not None:
            a_out = normalize_a(res.x)
            sc = exact_score(a_out)
            if np.isfinite(sc) and sc < 1e6:
                return sc, a_out
    except Exception:
        pass
    return exact_score(a), a


def optimize_with_continuation(a_start, deadline):
    """Main optimization: LSE continuation + subgradient + LP direction."""
    a = normalize_a(a_start.copy())
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    t0 = time.monotonic()
    total_time = max(10.0, float(deadline) - t0)

    print("  Continuing optimization: n={}, initial score={:.8f}".format(n, best_sc))

    # Stage 1: LSE with increasing beta
    betas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]
    fracs = [0.05, 0.08, 0.10, 0.12, 0.15, 0.18, 0.20, 0.12]

    for i, (beta, frac) in enumerate(zip(betas, fracs)):
        rem = float(deadline) - time.monotonic()
        if rem < 15.0:
            break
        sub_dead = t0 + (sum(fracs[:i+1]) * total_time)
        sub_dead = min(sub_dead, float(deadline) - 10.0)

        while time.monotonic() < sub_dead - 3.0:
            rem2 = float(sub_dead) - time.monotonic()
            if rem2 < 8.0:
                break
            mf = int(max(50, min(2000, rem2 * 20)))
            sc, a2 = run_lbfgs_lse(a, beta, sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("    LSE beta={:.0f}: {:.8f} (t={:.0f}s)".format(
                    beta, best_sc, time.monotonic() - t0))
            a = normalize_a(a2)
            if time.monotonic() > sub_dead - 5.0:
                break

        # Subgradient polish after each LSE stage
        if time.monotonic() < sub_dead - 5.0:
            for k in (3, 5, 10):
                for st in (0.01, 0.03, 0.05, 0.08):
                    if time.monotonic() > sub_dead - 3.0:
                        break
                    a2, sc = subgrad_with_line_search(a, st, k_active=k)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a2.copy()
                    a = normalize_a(a_new if sc < best_sc else a)
            a = normalize_a(a)

    # Stage 2: Aggressive subgradient with various active set sizes
    rem = float(deadline) - time.monotonic()
    if rem > 20.0:
        sub_dead = float(deadline) - 50.0
        print("  Subgradient phase: {:.0f}s remaining".format(rem))
        for iteration in range(200):
            if time.monotonic() > sub_dead:
                break
            improved = False
            for k in (1, 2, 3, 5, 8, 15):
                for st in (0.005, 0.01, 0.02, 0.04, 0.06, 0.1):
                    if time.monotonic() > sub_dead:
                        break
                    a2, sc = subgrad_with_line_search(a, st, k_active=k)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a2.copy()
                        a = a2.copy()
                        improved = True
            if not improved and iteration > 50:
                # Try eps-based active set
                for eps in (0.001, 0.01, 0.05, 0.1):
                    if time.monotonic() > sub_dead:
                        break
                    for st in (0.005, 0.01, 0.02, 0.05):
                        a2 = subgrad_step_direct(a, st, k_active=5, eps_frac=eps)
                        a2 = normalize_a(a2)
                        sc = exact_score(a2)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a2.copy()
                            a = a2.copy()
                            improved = True
            if not improved:
                break

    # Stage 3: Final high-beta LSE polish
    rem = float(deadline) - time.monotonic()
    if rem > 15.0:
        sc, a2 = run_lbfgs_lse(best_a, 200000.0, float(deadline) - 5.0, maxfun=500)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  Final LSE: {:.8f}".format(best_sc))

    # Final subgradient cleanup
    if time.monotonic() < float(deadline) - 10.0:
        for _ in range(50):
            if time.monotonic() > float(deadline) - 8.0:
                break
            for k in (3, 5):
                a2, sc = subgrad_with_line_search(best_a, 0.01, k_active=k)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a2.copy()

    return best_sc, best_a


def generate_diverse_inits(n, rng, parent=None, cap=25):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        S = float(arr.sum())
        if S < 1e-12:
            return
        arr = arr / S
        inits.append(arr)

    if parent is not None and parent.size > 0:
        # Resample parent to n
        m = parent.size
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, n)
        pr = np.interp(x_new, x_old, parent)
        add(pr)
        add(pr[::-1])

    # Uniform
    add(np.ones(n))

    # V-shapes
    for alpha in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
        add(1.0 + alpha * (np.abs(2.0 * x - 1.0) ** 2))

    # Inverted V (peak in middle)
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * np.sin(np.pi * x))

    # Cosine
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.7, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Exponential taper
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * np.abs(x - 0.5) * 2))

    # Power law taper
    for p in (0.5, 1.0, 2.0, 3.0):
        add((np.abs(2.0 * x - 1.0) + 0.1) ** p)

    # Two-peak (bimodal)
    for width in (0.05, 0.1, 0.15, 0.2):
        for centers in [(0.2, 0.8), (0.15, 0.85), (0.25, 0.75)]:
            v = np.zeros(n)
            for c in centers:
                v += np.exp(-(((x - c) / width) ** 2))
            add(v + 0.01)

    # Smooth random (low-pass)
    Nf = next_power_of_two(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(2.0, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        base = 1.0 + rng.uniform(0.5, 2.0) * np.abs(noise)
        add(base)

    return inits


def perturb_for_restart(a, rng, strength=0.3):
    """Perturb a sequence for restart diversity."""
    n = len(a)
    a = normalize_a(a)
    mode = int(rng.integers(0, 6))

    if mode == 0:
        # Multiplicative noise
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * a + t * np.ones(n) / n
    elif mode == 2:
        # V-shape twist
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = a * vshape
    elif mode == 3:
        # Gaussian smoothing
        k = int(rng.integers(7, max(9, n // 500)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        c = k // 2
        sig = max(1.0, c / 2.5)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode="same")
    elif mode == 4:
        # Active-set-based: reduce peak contributors
        S = float(np.sum(a))
        cc = fast_self_conv(a)
        d = cc / (S * S)
        dmax = float(np.max(d))
        if dmax > 0:
            idx = np.argpartition(d, -5)[-5:]
            w = np.zeros(len(d))
            w[idx] = 1.0
            w /= 5.0
            h = cross_corr(a, w)
            hmax = float(np.max(h))
            if hmax > 1e-15:
                q = a * np.exp(-strength * 0.5 * h / hmax)
            else:
                q = a.copy()
        else:
            q = a.copy()
    else:
        # Frequency-domain perturbation
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        mask = rng.random(len(A)) > 0.3
        phase_shift = strength * 0.1 * rng.standard_normal(len(A)) * mask
        A = A * np.exp(1j * phase_shift)
        q = np.fft.irfft(A, N)[:n]

    return normalize_a(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 128:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = int(parent.size)
        lengths = [parent_len]
        for L in [4096, 6144, 8192, 10240, 12288, 16384, 2048]:
            if L != parent_len and L not in lengths:
                lengths.append(L)
    else:
        lengths = [8192, 10240, 6144, 4096, 12288]

    lengths = [L for L in lengths if 1024 <= L <= 20000]
    lengths = lengths[:4]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Quick evaluation of inits
    best_score = np.inf
    best_seq = None

    for n in lengths:
        if time.monotonic() > deadline - 600.0:
            break
        inits = generate_diverse_inits(n, rng, parent, cap=15)
        for arr in inits:
            sc = exact_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = arr.copy()
    
    if best_seq is None:
        n0 = lengths[0]
        best_seq = np.ones(n0) / n0
        best_score = 2.0

    print("Phase 1 best: {:.8f} (len={})".format(best_score, len(best_seq)))

    # Phase 2: Optimize the best from Phase 1
    # Allocate time: 70% to primary, 30% to secondary
    primary_dead = t_start + 0.65 * (total_budget - 100)
    primary_dead = min(primary_dead, deadline - 200.0)

    sc, seq = optimize_with_continuation(best_seq, primary_dead)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print("Phase 2 primary done: {:.8f}".format(best_score))

    # Phase 3: Try a secondary length or perturbation
    rem = float(deadline) - time.monotonic()
    if rem > 300.0:
        # Perturbation restart
        p = perturb_for_restart(best_seq, rng, strength=0.4)
        sec_dead = time.monotonic() + min(rem * 0.4, 400.0)
        sc2, seq2 = optimize_with_continuation(p, sec_dead)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = seq2.copy()
            print("Phase 3 perturb restart: {:.8f}".format(best_score))

    # Phase 4: Another perturbation with different mode
    rem = float(deadline) - time.monotonic()
    if rem > 200.0:
        p = perturb_for_restart(best_seq, rng, strength=0.2)
        # Also try a different structure blend
        x = np.linspace(0.0, 1.0, len(p))
        vshape = 1.0 + 1.5 * np.abs(2.0 * x - 1.0)
        p = normalize_a(p * vshape)
        sec_dead = time.monotonic() + min(rem * 0.4, 300.0)
        sc3, seq3 = optimize_with_continuation(p, sec_dead)
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_seq = seq3.copy()
            print("Phase 4 vshape blend: {:.8f}".format(best_score))

    # Phase 5: Final intensive subgradient polish
    rem = float(deadline) - time.monotonic()
    if rem > 30.0:
        a = normalize_a(best_seq)
        for _ in range(200):
            if time.monotonic() > deadline - 10.0:
                break
            improved = False
            for k in (1, 2, 3, 5, 8):
                for st in (0.002, 0.005, 0.01, 0.02):
                    if time.monotonic() > deadline - 10.0:
                        break
                    a2, sc = subgrad_with_line_search(a, st, k_active=k)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a2.copy()
                        a = a2.copy()
                        improved = True
            if not improved:
                # Try eps-based
                for eps in (0.005, 0.02, 0.05):
                    if time.monotonic() > deadline - 10.0:
                        break
                    for st in (0.002, 0.005, 0.01):
                        a2 = subgrad_step_direct(a, st, k_active=5, eps_frac=eps)
                        a2 = normalize_a(a2)
                        sc = exact_score(a2)
                        if sc < best_score:
                            best_score = sc
                            best_seq = a2.copy()
                            a = a2.copy()
                            improved = True
            if not improved:
                break
        print("Phase 5 final subgrad: {:.8f}".format(best_score))

    # Phase 6: Very final high-beta LSE
    rem = float(deadline) - time.monotonic()
    if rem > 20.0:
        sc, a2 = run_lbfgs_lse(best_seq, 500000.0, float(deadline) - 5.0, maxfun=200)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Phase 6 final LSE: {:.8f}".format(best_score))

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = float(out_arr.sum())
    out_arr = out_arr / S

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
