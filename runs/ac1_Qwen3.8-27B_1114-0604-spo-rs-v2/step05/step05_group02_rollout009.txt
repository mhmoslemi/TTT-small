def fast_conv(a):

    n = len(a)

    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))

    p = exp_scaled / exp_scaled.sum()

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

    )

    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_subgradient_bt(a, n_iters=3000, top_m=10, step_init=0.03, patience=500):
    """Subgradient with backtracking on top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    no_improve = 0
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Backtracking line search
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        try_step = step
        accepted = False
        for _ in range(20):
            a_trial = a - try_step * grad
            a_trial = np.clip(a_trial, 0.0, None)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                try_step *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial < sc:
                a = a_trial
                S = S_trial
                accepted = True
                break
            try_step *= 0.5
        if not accepted:
            a = a - step * 0.01 * grad
            a = np.clip(a, 0.0, None)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
        step *= 0.999
    return best_a


def directed_peak_reduction(a, n_rounds=200, n_peaks=5):
    """Directly reduce top peaks by adjusting contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for rnd in range(n_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        reduced = False
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg_len = j_end - j_start + 1
            contributions = np.zeros(seg_len)
            for i in range(seg_len):
                j = j_start + i
                m = k - j
                contributions[i] = a[j] * a[m]
            # Find the element with largest contribution that we can reduce
            order = np.argsort(contributions)[::-1]
            for idx in order[:min(10, len(order))]:
                j = j_start + idx
                m = k - j
                if j == m:
                    # Diagonal: a[j]^2, reduce a[j]
                    a[j] *= 0.95
                else:
                    # Off-diagonal: a[j]*a[m], reduce the larger one
                    if a[j] >= a[m]:
                        a[j] *= 0.97
                    else:
                        a[m] *= 0.97
                # Quick score check
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    break
                a_tmp = a / S_tmp
                c_tmp = fast_conv(a_tmp)
                sc_tmp = 2.0 * n * float(c_tmp.max()) / (S_tmp * S_tmp)
                if sc_tmp > sc + 0.01:
                    # Revert
                    a = best_a.copy()
                    break
                reduced = True
                break
        if not reduced:
            break
        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    return best_a

    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing the evaluation function.
    Multi-phase: LSE (adaptive beta) + GS + subgradient + directed peak reduction.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent and set initial best
    if parent is not None:
        p_norm = parent / parent.sum()
        sc_parent = compute_score(p_norm)
        update_best(sc_parent, p_norm, "parent")
        print(f"Parent score: {sc_parent:.8f}, n={len(parent)}")

    n_main = len(best_a)
    print(f"Main length: {n_main}, starting score: {best_score:.8f}")

    # ==============================================================
    # PHASE 1: LSE optimization with adaptive beta schedule (50% time)
    # ==============================================================
    phase1_end = t_start + 0.50 * (total_budget - 40.0)
    print(f"\n=== Phase 1: LSE adaptive beta (t={0:.0f}s to t={phase1_end - t_start:.0f}s) ===")

    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    a_work = best_a.copy()

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t_beta = time.monotonic()
        maxiter = max(500, min(8000, int(120.0 / ((time.monotonic() - t_start + 1) / total_budget * 2))))
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 12)
        dt = time.monotonic() - t_beta
        if sc < best_score:
            update_best(sc, a_opt, f"LSE beta={beta}")
            a_work = a_opt.copy()
        else:
            # Try GS to escape, then re-run LSE
            a_gs = gs_correct_target(a_work, n_iters=200, alpha=0.3)
            sc_gs = compute_score(a_gs)
            if sc_gs < best_score:
                update_best(sc_gs, a_gs, f"GS-escape beta={beta}")
                a_work = a_gs.copy()
            sc2, a_opt2 = run_lbfgs(a_work, beta, maxiter=maxiter // 2, maxfun=maxiter * 6)
            if sc2 < best_score:
                update_best(sc2, a_opt2, f"LSE2 beta={beta}")
                a_work = a_opt2.copy()
        a_work = best_a.copy()
        print(f"  beta={beta}: score={best_score:.8f}, dt={dt:.1f}s")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==============================================================
    # PHASE 2: GS refinement + perturbation restarts (25% time)
    # ==============================================================
    phase2_end = t_start + 0.75 * (total_budget - 40.0)
    print(f"\n=== Phase 2: GS + restarts (t={phase1_end - t_start:.0f}s to t={phase2_end - t_start:.0f}s) ===")

    # Multiple GS passes with different alphas
    for alpha in [0.3, 0.2, 0.15, 0.1, 0.05]:
        if time.monotonic() > phase2_end:
            break
        a_gs = gs_adaptive(best_a, n_iters=400)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"GS alpha={alpha}")

    # Perturbation restarts
    for trial in range(8):
        if time.monotonic() > phase2_end:
            break
        noise_scale = 0.001 * (1 + trial * 0.5)
        a_pert = best_a + noise_scale * rng.standard_normal(len(best_a))
        a_pert = np.clip(a_pert, 0.0, None)
        a_pert = a_pert / a_pert.sum()
        # Quick LSE from perturbation
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=30000)
        update_best(sc, a_opt, f"perturb-{trial}")
        # Also try GS from perturbation
        a_gs_p = gs_correct_target(a_pert, n_iters=200, alpha=0.2)
        sc_gs = compute_score(a_gs_p)
        update_best(sc_gs, a_gs_p, f"perturb-GS-{trial}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==============================================================
    # PHASE 3: Peak-directed search + subgradient (15% time)
    # ==============================================================
    phase3_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Peak-directed + subgradient ===")

    # Subgradient with different parameters
    for top_m in [5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        a_sg = peak_subgradient_bt(best_a, n_iters=3000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    # Directed peak reduction
    if time.monotonic() < phase3_end:
        a_dpr = directed_peak_reduction(best_a, n_rounds=300, n_peaks=3)
        sc = compute_score(a_dpr)
        update_best(sc, a_dpr, "peak-reduction")

    # Final LSE polish
    if time.monotonic() < phase3_end:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=50000)
        update_best(sc, a_opt, "final-LSE-hi")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==============================================================
    # PHASE 4: Length exploration (remaining time)
    # ==============================================================
    print(f"\n=== Phase 4: Length exploration ===")

    for n_alt in [n_main - 500, n_main + 200, n_main + 500, n_main + 1000]:
        if time.monotonic() > deadline:
            break
        if n_alt < 1000:
            continue
        # Interpolate
        a_interp = np.interp(np.linspace(0, 1, n_alt),
                             np.linspace(0, 1, len(best_a)), best_a)
        a_interp = np.maximum(a_interp, 0.01)
        a_interp = a_interp / a_interp.sum()
        # LSE at this length
        sc, a_opt = run_lbfgs(a_interp, 2000, maxiter=3000, maxfun=30000)
        update_best(sc, a_opt, f"length {n_alt}")
        # GS
        a_gs_alt = gs_correct_target(a_opt, n_iters=300, alpha=0.3)
        sc_gs = compute_score(a_gs_alt)
        update_best(sc_gs, a_gs_alt, f"GS length {n_alt}")

    # ==============================================================
    # PHASE 5: Final ultra-high-beta LSE polish
    # ==============================================================
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 5: Final polish ===")
        sc, a_opt = run_lbfgs(best_a, 200000, maxiter=3000, maxfun=30000)
        update_best(sc, a_opt, "ultra-LSE")
        sc, a_opt2 = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=50000)
        update_best(sc, a_opt2, "ultra-LSE2")

    # ==============================================================
    # Final output
    # ==============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
