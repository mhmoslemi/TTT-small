```
def sanitize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    if n == 0:
```

```python
"""
Active-peak gradient + spectral flat-convolution refinement.

The search starts from the supplied construction (if present) and works on a
few nearby lengths.  The main engine is a projected subgradient method that
optimizes the sum of the K largest self-convolution coefficients.  A line
search accepts only actual improvements of this active-peak sum or of the
final maximum coefficient.  Spectral GS/Fienup-type iterations with dynamic
peak-capped targets are used to escape local basins, and a log-sum-exp stage
polishes with a smooth approximation of the maximum.  The final sequence is
returned normalized to sum 1, which makes the evaluation scale-invariant and
keeps it well inside the evaluator's validity bounds.
"""

import numpy as np
import time


def fft_len_for_n(n):
    """Smallest power of two at least 2*n-1 (and at least 2)."""
    m = 2 * int(n) - 1
    size = 1
    while size < m:
        size <<= 1
    return max(size, 2)


def sanitize_seq(a):
    """Clip to non-negative finite values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(64, dtype=np.float64)
    a = a.copy()
    bad = ~np.isfinite(a)
    if np.any(bad):
        a[bad] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if S <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a /= S
    return a


def resample_seq(a, n):
    """Interpolate a normalized sequence to length n."""
    n = int(n)
    if n <= 0:
        return np.ones(64, dtype=np.float64)
    a = sanitize_seq(a)
    if a.size == n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, n)
    out = np.interp(new_x, old_x, a)
    return sanitize_seq(out)


def fast_conv(a, M=None):
    """Linear self-convolution a*a by zero-padded FFT."""
    n = int(a.size)
    if M is None:
        M = fft_len_for_n(n)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def score_seq(a):
    """Exact evaluation objective for an arbitrary non-negative sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float('inf')
    S = float(np.sum(a))
    if S < 1e-12:
        return float('inf')
    n = a.size
    M = fft_len_for_n(n)
    b = fast_conv(a, M)
    m = float(np.max(b))
    return float(2.0 * n * m / (S * S))


def top_indices(b, K):
    """Indices of the K largest values (not necessarily sorted)."""
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    K = int(K)
    if K <= 0:
        return np.empty(0, dtype=np.int64)
    if K >= L:
        return np.arange(L, dtype=np.int64)
    kth = L - K
    return np.argpartition(b, kth)[kth:]


def cap_fill_target(b, cap_frac):
    """
    Feasible-shaped target near b: cap coefficients at cap_frac*max(b),
    then add the lost L1 mass uniformly.
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    total = float(np.sum(b))
    if total <= 1e-300 or not np.isfinite(total):
        return np.full(b.size, 1.0 / max(1, b.size))
    b = b / total
    cap_frac = float(np.clip(cap_frac, 0.05, 1.0))
    m = float(np.max(b))
    cap = m * cap_frac
    q = np.minimum(b, cap)
    s = float(np.sum(q))
    L = q.size
    if s < 1.0:
        q = q + (1.0 - s) / float(L)
    elif s > 1.0:
        q = q / s
    q = np.maximum(q, 0.0)
    ss = float(np.sum(q))
    if ss <= 1e-300 or not np.isfinite(ss):
        return b
    return q / ss


def flat_target(n):
    """Uniform target over all linear-convolution lags."""
    L = max(1, 2 * int(n) - 1)
    return np.full(L, 1.0 / float(L))


def bell_target(n, amp):
    """Bell-shaped target: higher in the middle, lower near edges."""
    L = max(1, 2 * int(n) - 1)
    if L == 1:
        return np.ones(1)
    t = np.arange(L, dtype=np.float64) / float(L - 1)
    q = 1.0 + float(amp) * np.cos(2.0 * np.pi * (t - 0.5))
    q = np.maximum(q, 1e-6)
    s = float(np.sum(q))
    if s <= 1e-300 or not np.isfinite(s):
        return flat_target(n)
    return q / s


def blend_spectra(a, target, M, alpha):
    """
    GS/Fienup-style spectral blend:
    A_new^2 ~ A^2^(1-alpha) * target_spectrum^alpha.
    """
    A = np.fft.rfft(a, M)
    T = np.fft.rfft(target, M)

    am = np.maximum(np.abs(A), 1e-300)
    tm = np.maximum(np.abs(T), 1e-300)
    ap = np.angle(A)
    tp = np.angle(T)

    logmag = (1.0 - alpha) * np.log(am) + 0.5 * alpha * np.log(tm)
    mag = np.exp(logmag)
    phase = (1.0 - alpha) * ap + 0.5 * alpha * tp
    A_new = mag * (np.cos(phase) + 1j * np.sin(phase))
    out = np.fft.irfft(A_new, M)[:a.size]
    return out


def gs_run(a_in,
           n_iters,
           M,
           alpha_start=0.45,
           alpha_end=0.10,
           target_kind='cap',
           cap_start=0.82,
           cap_end=0.995,
           amp=0.6,
           deadline=None):
    """
    Spectral GS iteration with a chosen target convolution shape.
    Returns the best sequence and exact score seen during the run.
    """
    a = sanitize_seq(a_in)
    n = a.size
    if M is None:
        M = fft_len_for_n(n)
    best = a.copy()
    best_sc = score_seq(a)

    for it in range(int(n_iters)):
        if deadline is not None and time.monotonic() >= deadline:
            break

        A = np.fft.rfft(a, M)
        b = np.fft.irfft(A * A, M)[:2 * n - 1]
        m = float(np.max(b))
        cur = float(2.0 * n * m)
        if cur < best_sc:
            best_sc = cur
            best = a.copy()

        prog = 0.0 if n_iters <= 1 else float(it) / float(max(1, n_iters - 1))
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        if target_kind == 'cap':
            cap = cap_start + (cap_end - cap_start) * prog
            target = cap_fill_target(b, cap)
        elif target_kind == 'flat':
            target = flat_target(n)
        elif target_kind == 'bell':
            target = bell_target(n, amp)
        else:
            target = flat_target(n)

        anew = blend_spectra(a, target, M, alpha)
        anew = sanitize_seq(anew)
        a = anew

    b = fast_conv(a, M)
    sc = float(2.0 * n * np.max(b))
    if sc < best_sc:
        best_sc = sc
        best = a.copy()
    return best, best_sc


def conv_corr_weights_a(weights, A, M, n):
    """
    Gradient subvector for sum_k weights[k] * (a*a)[k] with respect to a,
    using the FFT of a already computed as A.
    Returns 2 * cross-correlation, length n.
    """
    W = np.fft.rfft(weights, M)
    return 2.0 * np.fft.irfft(np.conjugate(W) * A, M)[:n]


def lse_value_from_b(b, beta):
    """Smooth max approximation L = max(b) + log(sum exp(beta*(b-max)))/beta."""
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return float('inf'), 0.0
    scaled = float(beta) * (b - m)
    ew = np.exp(scaled)
    psum = float(np.sum(ew))
    if psum <= 1e-300 or not np.isfinite(psum):
        return m, m
    L = m + float(np.log(psum)) / float(beta)
    return L, m


def topk_stage(a_in,
               K,
               deadline,
               M=None,
               max_iters=200000,
               momentum=0.75,
               verbose=False):
    """
    Projected gradient descent on the sum of K largest convolution coeffs.
    Uses exact line search on the active set and optional GS escapes.
    Returns (best_a, best_score, current_a, current_score).
    """
    a = sanitize_seq(a_in)
    n = a.size
    if M is None:
        M = fft_len_for_n(n)
    L = 2 * n - 1

    A = np.fft.rfft(a, M)
    b = np.fft.irfft(A * A, M)[:L]
    best = a.copy()
    best_sc = float(2.0 * n * np.max(b))
    current_sc = best_sc

    v = np.zeros(n, dtype=np.float64)
    step = max(1e-7, min(1e-3, 0.5 * float(np.max(b))))
    w = np.zeros(M, dtype=np.float64)
    fail = 0
    gs_count = 0
    last_best_it = 0
    it = 0

    if verbose:
        print(f"    TOPK K={K} start={best_sc:.10f}")

    while it < int(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break

        A = np.fft.rfft(a, M)
        b = np.fft.irfft(A * A, M)[:L]
        m = float(np.max(b))
        cur_sc = float(2.0 * n * m)
        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best = a.copy()
            last_best_it = it

        part = top_indices(b, K)
        part = np.asarray(part, dtype=np.int64)
        if part.size == 0:
            break

        bP = b[part]
        Ocur = float(np.sum(bP))

        w.fill(0.0)
        Kc = float(part.size)
        bmin = float(np.min(bP))
        excess = bP - bmin
        se = float(np.sum(excess))
        if se > 1e-30 * max(1.0, Ocur):
            wP = 0.5 / Kc + 0.5 * (excess / se)
            sw = float(np.sum(wP))
            if sw > 0.0 and np.isfinite(sw):
                wP = wP / sw
            else:
                wP = np.full(part.size, 1.0 / Kc)
        else:
            wP = np.full(part.size, 1.0 / Kc)
        w[part] = wP

        g = conv_corr_weights_a(w, A, M, n)
        h = g - float(np.mean(g))
        h_norm2 = float(np.sum(h * h))
        if h_norm2 <= 1e-30:
            break

        v = float(momentum) * v + h
        v = v - float(np.mean(v))
        v_norm2 = float(np.sum(v * v))
        if v_norm2 <= 1e-30:
            v = h.copy()
            v_norm2 = h_norm2

        d = -v / float(np.sqrt(v_norm2))

        tau = min(5e-4, max(1e-9, step))
        acc = False
        acc_a = None
        acc_sc = cur_sc

        for _ in range(5):
            trial = a + tau * d
            trial = sanitize_seq(trial)

            bt = fast_conv(trial, M)
            mt = float(np.max(bt))
            sc_t = float(2.0 * n * mt)
            if not np.isfinite(sc_t):
                tau *= 0.45
                continue

            if sc_t < best_sc - 1e-15:
                best_sc = sc_t
                best = trial.copy()
                last_best_it = it

            Ot = float(np.sum(bt[part]))
            tol = max(1e-18, 1e-12 * max(Ocur, 1e-12))
            if Ot < Ocur - tol or sc_t < cur_sc - 1e-14:
                acc = True
                acc_a = trial
                acc_sc = sc_t
                break
            tau *= 0.45
            if tau <= 1e-14:
                break

        if acc:
            a = acc_a
            current_sc = acc_sc
            step = min(5e-4, max(1e-12, step * 1.10))
            fail = 0
        else:
            step *= 0.45
            fail += 1
            if fail == 18:
                v *= 0.25
            if fail >= 32 and gs_count < 12:
                gs_dead = min(deadline, time.monotonic() + 25.0) if deadline is not None else None
                a_gs, sc_gs = gs_run(
                    a,
                    70,
                    M,
                    alpha_start=0.35,
                    alpha_end=0.08,
                    target_kind='cap',
                    cap_start=0.78,
                    cap_end=0.995,
                    amp=0.5,
                    deadline=gs_dead
                )
                if np.isfinite(sc_gs) and sc_gs < best_sc:
                    best_sc = sc_gs
                    best = a_gs.copy()
                    last_best_it = it
                a = a_gs.copy()
                current_sc = score_seq(a)
                v.fill(0.0)
                fail = 0
                step = max(1e-7, min(5e-4, 0.5 * float(np.max(fast_conv(a, M)))))
                gs_count += 1
            if fail >= 80 and gs_count >= 2:
                break

        if it > 0 and it % 500 == 0 and gs_count < 24:
            gs_dead = min(deadline, time.monotonic() + 12.0) if deadline is not None else None
            a_gs, sc_gs = gs_run(
                a,
                35,
                M,
                alpha_start=0.30,
                alpha_end=0.08,
                target_kind='cap',
                cap_start=0.86,
                cap_end=0.997,
                amp=0.5,
                deadline=gs_dead
            )
            if np.isfinite(sc_gs) and sc_gs < best_sc:
                best_sc = sc_gs
                best = a_gs.copy()
                last_best_it = it
            # Only replace current trajectory if it is not visibly worse.
            if np.isfinite(sc_gs) and sc_gs <= best_sc + 1e-8:
                a = a_gs.copy()
                current_sc = sc_gs
            v.fill(0.0)
            gs_count += 1

        it += 1

    b = fast_conv(a, M)
    current_sc = float(2.0 * n * np.max(b))
    if current_sc < best_sc:
        best_sc = current_sc
        best = a.copy()

    if verbose:
        print(f"    TOPK K={K} end best={best_sc:.10f} cur={current_sc:.10f} iters={it}")

    return best, best_sc, a.copy(), current_sc


def lse_stage(a_in,
              beta,
              deadline,
              M=None,
              max_iters=50000,
              momentum=0.70,
              verbose=False):
    """
    Smooth max polishing via log-sum-exp of the convolution coefficients.
    Returns (best_a, best_score, current_a, current_score).
    """
    a = sanitize_seq(a_in)
    n = a.size
    if M is None:
        M = fft_len_for_n(n)
    L = 2 * n - 1

    A = np.fft.rfft(a, M)
    b = np.fft.irfft(A * A, M)[:L]
    best = a.copy()
    best_sc = float(2.0 * n * np.max(b))
    current_sc = best_sc

    v = np.zeros(n, dtype=np.float64)
    step = max(1e-8, min(1e-3, 0.5 * float(np.max(b))))
    fail = 0
    gs_count = 0
    it = 0

    if verbose:
        print(f"    LSE beta={beta:.1f} start={best_sc:.10f}")

    while it < int(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break

        A = np.fft.rfft(a, M)
        b = np.fft.irfft(A * A, M)[:L]
        m = float(np.max(b))
        cur_sc = float(2.0 * n * m)
        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best = a.copy()

        Lcur, _ = lse_value_from_b(b, beta)
        if not np.isfinite(Lcur):
            break

        scaled = float(beta) * (b - m)
        ew = np.exp(scaled)
        psum = float(np.sum(ew))
        if psum <= 1e-300 or not np.isfinite(psum):
            break
        p = ew / psum

        g = conv_corr_weights_a(p, A, M, n)
        h = g - float(np.mean(g))
        h_norm2 = float(np.sum(h * h))
        if h_norm2 <= 1e-30:
            break

        v = float(momentum) * v + h
        v = v - float(np.mean(v))
        v_norm2 = float(np.sum(v * v))
        if v_norm2 <= 1e-30:
            v = h.copy()
            v_norm2 = h_norm2
        d = -v / float(np.sqrt(v_norm2))

        tau = min(5e-4, max(1e-9, step))
        acc = False
        acc_a = None
        acc_sc = cur_sc

        for _ in range(5):
            trial = sanitize_seq(a + tau * d)
            bt = fast_conv(trial, M)
            mt = float(np.max(bt))
            sc_t = float(2.0 * n * mt)
            if not np.isfinite(sc_t):
                tau *= 0.45
                continue

            if sc_t < best_sc - 1e-15:
                best_sc = sc_t
                best = trial.copy()

            Lt, _ = lse_value_from_b(bt, beta)
            tol = max(1e-18, 1e-12 * max(abs(Lcur), 1e-12))
            if Lt < Lcur - tol or sc_t < cur_sc - 1e-14:
                acc = True
                acc_a = trial
                acc_sc = sc_t
                break
            tau *= 0.45
            if tau <= 1e-14:
                break

        if acc:
            a = acc_a
            current_sc = acc_sc
            step = min(5e-4, max(1e-12, step * 1.10))
            fail = 0
        else:
            step *= 0.45
            fail += 1
            if fail == 18:
                v *= 0.25
            if fail >= 32 and gs_count < 10:
                gs_dead = min(deadline, time.monotonic() + 25.0) if deadline is not None else None
                a_gs, sc_gs = gs_run(
                    a,
                    70,
                    M,
                    alpha_start=0.32,
                    alpha_end=0.08,
                    target_kind='cap',
                    cap_start=0.80,
                    cap_end=0.996,
                    amp=0.5,
                    deadline=gs_dead
                )
                if np.isfinite(sc_gs) and sc_gs < best_sc:
                    best_sc = sc_gs
                    best = a_gs.copy()
                if np.isfinite(sc_gs) and sc_gs <= best_sc + 1e-8:
                    a = a_gs.copy()
                    current_sc = sc_gs
                v.fill(0.0)
                fail = 0
                step = max(1e-8, min(1e-3, 0.5 * float(np.max(fast_conv(a, M)))))
                gs_count += 1
            if fail >= 80 and gs_count >= 2:
                break

        it += 1

    b = fast_conv(a, M)
    current_sc = float(2.0 * n * np.max(b))
    if current_sc < best_sc:
        best_sc = current_sc
        best = a.copy()

    if verbose:
        print(f"    LSE beta={beta:.1f} end best={best_sc:.10f} cur={current_sc:.10f} iters={it}")

    return best, best_sc, a.copy(), current_sc


def perturb_seq(a, rel, rng):
    """Small relative Gaussian perturbation, still non-negative after sanitize."""
    a = sanitize_seq(a)
    if a.size == 0:
        return a
    mean = float(np.mean(a))
    noise = rng.standard_normal(a.size) * (float(rel) * mean)
    return sanitize_seq(a + noise)


def propose_candidate(*args, **kwargs):
    seed = 42
    budget = 2400.0
    if len(args) > 0 and args[0] is not None:
        seed = int(args[0])
    if len(args) > 1 and args[1] is not None:
        budget = float(args[1])
    if 'seed' in kwargs and kwargs['seed'] is not None:
        seed = int(kwargs['seed'])
    if 'budget_s' in kwargs and kwargs['budget_s'] is not None:
        budget = float(kwargs['budget_s'])

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    global_deadline = t0 + float(budget) - 45.0

    # ---------------- Start from previous construction if available ---------------
    parent = None
    try:
        p_obj = globals().get('height_sequence_1', None)
        if p_obj is not None:
            p_arr = np.asarray(p_obj, dtype=np.float64).ravel()
            if p_arr.size >= 8:
                parent = sanitize_seq(p_arr)
    except Exception:
        parent = None

    if parent is None:
        parent = sanitize_seq(np.ones(6962, dtype=np.float64))

    best_a = parent.copy()
    best_sc = score_seq(best_a)
    print(f"Initial parent: n={best_a.size}, score={best_sc:.10f}, t={time.monotonic()-t0:.1f}s")

    # ---------------- Quick multi-length scouting ---------------------------------
    base_len = int(best_a.size)
    raw_lengths = [
        base_len,
        8192,
        10240,
        int(base_len * 1.5),
        int(base_len * 2),
        16384,
        20480,
        24576,
        32768,
    ]
    cand_lengths = []
    for L in raw_lengths:
        try:
            Li = int(L)
        except Exception:
            continue
        if Li <= 0:
            continue
        if Li > 40000:
            continue
        if Li not in cand_lengths:
            cand_lengths.append(Li)
    if base_len not in cand_lengths:
        cand_lengths.insert(0, base_len)

    larger = [L for L in cand_lengths if L > base_len]
    larger.sort()
    chosen_lengths = [base_len] + larger[-3:]
    # Keep them mostly unique and limited.
    unique_chosen = []
    for L in chosen_lengths:
        if L not in unique_chosen:
            unique_chosen.append(L)
    chosen_lengths = unique_chosen[:4]

    quick_end = min(global_deadline, t0 + 75.0)
    quick_seg = 75.0 / max(1, len(chosen_lengths))
    quick_best_a = best_a.copy()
    quick_best_sc = best_sc

    print("Quick length scouting:", chosen_lengths, "end ~", quick_end - t0)

    for L in chosen_lengths:
        stage_dead = min(quick_end, time.monotonic() + quick_seg)
        Mq = fft_len_for_n(L)

        if L == len(best_a):
            a_cur = best_a.copy()
        else:
            a_cur = resample_seq(best_a, L)
        sc_cur = score_seq(a_cur)

        a_new, sc_new = gs_run(
            a_cur, 70, Mq,
            alpha_start=0.45, alpha_end=0.12,
            target_kind='cap', cap_start=0.82, cap_end=0.995, amp=0.6,
            deadline=stage_dead
        )
        if np.isfinite(sc_new) and sc_new < sc_cur:
            a_cur, sc_cur = a_new, sc_new

        a_new, sc_new = gs_run(
            a_cur, 45, Mq,
            alpha_start=0.40, alpha_end=0.08,
            target_kind='flat', cap_start=1.0, cap_end=1.0, amp=0.0,
            deadline=stage_dead
        )
        if np.isfinite(sc_new) and sc_new < sc_cur:
            a_cur, sc_cur = a_new, sc_new

        a_best_st, sc_best_st, a_cur_st, sc_cur_st = topk_stage(
            a_cur, 5, stage_dead, Mq,
            max_iters=120, momentum=0.60, verbose=False
        )
        if np.isfinite(sc_best_st) and sc_best_st < sc_cur:
            a_cur, sc_cur = a_best_st, sc_best_st
        elif np.isfinite(sc_cur_st) and sc_cur_st <= sc_cur + 1e-8:
            a_cur, sc_cur = a_cur_st, sc_cur_st

        if np.isfinite(sc_cur) and sc_cur < quick_best_sc:
            quick_best_sc = sc_cur
            quick_best_a = a_cur.copy()
        print(f"  quick n={L}: {sc_cur:.10f} t={time.monotonic()-t0:.1f}s")

    best_a = quick_best_a.copy()
    best_sc = quick_best_sc
    n_best = len(best_a)
    M = fft_len_for_n(n_best)
    print(f"After quick selection: n={n_best}, best={best_sc:.10f}")

    # ---------------- Main active-peak gradient ------------------------------------
    top_end = min(global_deadline, t0 + 0.68 * float(budget))
    K_schedule = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 80, 120]
    K_weights = [6, 5, 4, 3, 3, 2, 2, 1.5, 1.4, 1.2, 1.0, 0.8, 0.6]
    rem_w = float(sum(K_weights))

    print(f"Main TOPK phase end ~ {top_end - t0:.0f}s")
    for idx, K in enumerate(K_schedule):
        now = time.monotonic()
        if now >= top_end or rem_w <= 0.0:
            break
        frac = float(K_weights[idx]) / float(rem_w)
        rem_w -= float(K_weights[idx])
        stage_dead = min(top_end, now + max(25.0, (top_end - now) * 0.92 * frac))

        a_best_st, sc_best_st, a_cur_st, sc_cur_st = topk_stage(
            best_a, K, stage_dead, M,
            max_iters=200000, momentum=0.75, verbose=False
        )
        if np.isfinite(sc_best_st) and sc_best_st < best_sc:
            best_sc = sc_best_st
            best_a = a_best_st.copy()
            n_best = len(best_a)
            M = fft_len_for_n(n_best)

        # Continue from current trajectory if it is not visibly worse; this can
        # preserve lower active-sum improvements even when exact max has not yet dropped.
        if np.isfinite(sc_cur_st) and sc_cur_st <= best_sc + 5e-7:
            a_work = a_cur_st.copy()
        else:
            a_work = best_a.copy()

        # Keep a_work in sync if its length would differ (should not, but be safe).
        if len(a_work) != n_best:
            a_work = resample_seq(a_work, n_best)
            M = fft_len_for_n(n_best)

        print(f"  TOPK K={K}: best={best_sc:.10f} cur={sc_cur_st:.10f} t={time.monotonic()-t0:.0f}s")
        best_a = a_work

    # ---------------- Spectral refinement / escape attempts ------------------------
    gs_end = min(global_deadline, t0 + 0.86 * float(budget))
    gs_targets = [
        ('cap', 0.80, 0.995, 0.35, 0.08, 0.6),
        ('cap', 0.65, 0.950, 0.30, 0.06, 0.5),
        ('flat', 1.0, 1.0, 0.45, 0.05, 0.0),
        ('bell', 1.0, 1.0, 0.40, 0.07, 0.8),
        ('cap', 0.75, 0.990, 0.25, 0.04, 0.5),
    ]
    n_gs = len(gs_targets)
    gs_seg = (gs_end - time.monotonic()) / float(n_gs) if gs_end > time.monotonic() else 0.0

    print(f"GS phase end ~ {gs_end - t0:.0f}s")
    for i, (kind, cs, ce, as_, ae, amp) in enumerate(gs_targets):
        now = time.monotonic()
        if now >= gs_end:
            break
        stage_dead = min(gs_end, now + max(20.0, gs_seg))
        start_a = best_a.copy()
        if i == 3:
            start_a = perturb_seq(start_a, 0.025, rng)
        n_iter = max(150, int((gs_seg + 10.0) * 80.0))
        a_gs, sc_gs = gs_run(
            start_a,
            n_iter,
            M,
            alpha_start=as_,
            alpha_end=ae,
            target_kind=kind,
            cap_start=cs,
            cap_end=ce,
            amp=amp,
            deadline=stage_dead
        )
        if np.isfinite(sc_gs) and sc_gs < best_sc:
            best_sc = sc_gs
            best_a = a_gs.copy()
            M = fft_len_for_n(len(best_a))
        if np.isfinite(sc_gs) and sc_gs <= best_sc + 5e-7:
            best_a = a_gs.copy()
        print(f"  GS {kind}: best={best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # ---------------- Log-sum-exp polishing ----------------------------------------
    lse_end = min(global_deadline, t0 + 0.95 * float(budget))
    betas = [1e5, 1e6, 3e6, 1e7]
    rem_b = float(len(betas))

    print(f"LSE phase end ~ {lse_end - t0:.0f}s")
    for beta in betas:
        now = time.monotonic()
        if now >= lse_end or rem_b <= 0:
            break
        stage_dead = min(lse_end, now + max(20.0, (lse_end - now) / rem_b))
        rem_b -= 1.0
        a_best_st, sc_best_st, a_cur_st, sc_cur_st = lse_stage(
            best_a, beta, stage_dead, M,
            max_iters=50000, momentum=0.70, verbose=False
        )
        if np.isfinite(sc_best_st) and sc_best_st < best_sc:
            best_sc = sc_best_st
            best_a = a_best_st.copy()
            M = fft_len_for_n(len(best_a))
        if np.isfinite(sc_cur_st) and sc_cur_st <= best_sc + 5e-7:
            best_a = a_cur_st.copy()
        print(f"  LSE beta={beta:.1f}: best={best_sc:.10f} t={time.monotonic()-t0:.0f}s")

    # ---------------- Final tiny active-peak polish ---------------------------------
    final_dead = global_deadline
    if time.monotonic() < final_dead - 20.0:
        final_seg = (final_dead - time.monotonic()) / 3.0
        for K in [1, 3, 5]:
            now = time.monotonic()
            if now >= final_dead:
                break
            stage_dead = min(final_dead, now + max(10.0, final_seg))
            a_best_st, sc_best_st, a_cur_st, sc_cur_st = topk_stage(
                best_a, K, stage_dead, M,
                max_iters=20000, momentum=0.70, verbose=False
            )
            if np.isfinite(sc_best_st) and sc_best_st < best_sc:
                best_sc = sc_best_st
                best_a = a_best_st.copy()
                M = fft_len_for_n(len(best_a))
            if np.isfinite(sc_cur_st) and sc_cur_st <= best_sc + 5e-7:
                best_a = a_cur_st.copy()

    # ---------------- Final valid output -------------------------------------------
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(np.sum(out_arr))
    if S < 0.01:
        out_arr = np.ones(64, dtype=np.float64)
    else:
        out_arr = out_arr / S

    out_list = [float(x) for x in out_arr]
    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = float('inf')

    print("=" * 70)
    print(f"FINAL internal={best_sc:.10f}, verified={verified:.10f}")
    print(f"Length={len(out_list)}, elapsed={time.monotonic()-t0:.1f}s / {budget:.0f}s")
    print("=" * 70)

    return out_list
```
