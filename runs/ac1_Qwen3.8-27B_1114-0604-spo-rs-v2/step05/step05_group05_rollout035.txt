4. Line-search-based coordinate descent on peak contributors
5. Multi-start with diverse initialization templates
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient using FFT.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum

    # Gradient: dL/da_i = 2 * sum_m a[m] * p[m+i] (over valid m)
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2 * n - 1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def nesterov_lse_polyak(a, beta_start, beta_end, max_time,
                         lr_start=0.02, lr_decay=0.9995,
                         mu_start=0.85, mu_end=0.97,
                         beta_growth=1.003):
    """
    Nesterov-accelerated LSE with Polyak averaging.
    Geometric beta increase from beta_start to beta_end.
    Tracks both best single iterate and Polyak average.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    # Polyak averaging
    avg_a = a.copy()
    avg_count = 0
    avg_sc = best_sc
    best_avg_a = a.copy()

    it = 0
    beta = beta_start
    lr = lr_start
    mu = mu_start

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = mu_start + (mu_end - mu_start) * min(1.0, it / 5000.0)
        lr = max(0.0005, lr_start * (lr_decay ** it))

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute LSE gradient
        _, grad = lse_score_and_grad(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        # Cap step size
        max_lr = 0.05 / (gnorm + 1e-10)
        eff_lr = min(lr, max_lr)

        # Update
        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        # Increase beta geometrically
        beta = min(beta * beta_growth, beta_end)

        # Polyak averaging (weight recent iterates more)
        w = 1.0 / (avg_count + 1)
        avg_a = (1 - w) * avg_a + w * a
        avg_count += 1

        # Periodic evaluation
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            # Check average
            sc_avg = compute_score(avg_a)
            if sc_avg < avg_sc:
                avg_sc = sc_avg
                best_avg_a = avg_a.copy()
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = avg_a.copy()

    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    sc_avg = compute_score(avg_a)
    if sc_avg < best_sc:
        best_a = avg_a.copy()
        best_sc = sc_avg

    return best_a


def active_set_optimize(a, max_time, n_top_bands=5, step_init=0.01):
    """
    Subgradient optimization targeting the top convolution bins.
    Uses the exact subgradient (indicator of active set) with backtracking.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    step = step_init

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find active set: bins within 0.5% of max
        active_mask = c >= c_max * 0.995
        active_indices = np.where(active_mask)[0]

        if len(active_indices) == 0:
            break

        # Compute combined subgradient
        # g_i = (1/|K|) * sum_{k in K} a[k-i] for valid (k-i)
        g = np.zeros(n)
        for k in active_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                g[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        g /= len(active_indices)

        # Backtracking line search
        a_trial = a.copy()
        accepted = False
        for trial in range(8):
            lr = step * (0.5 ** trial)
            a_trial = a - lr * g
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial < sc * (1 - 1e-8):
                a = a_trial
                accepted = True
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a.copy()
                break

        if not accepted:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            step = min(step * 1.1, 0.05)

    return best_a


def coordinate_descent_ls(a, max_time, n_top_peaks=20, n_top_contrib=15):
    """
    Coordinate descent with 1D line search on peak contributors.
    For each top peak, perturb the top contributing elements individually.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    while time.monotonic() - t_start < max_time:
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(np.max(c)) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peaks
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contributions: a[j] contributes a[k-j] to c[k]
            js = np.arange(j_start, j_end + 1)
            contribs = np.array([a[k - j] for j in js])
            top_local = np.argsort(contribs)[::-1][:n_top_contrib]
            top_global = js[top_local]

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue

                # 1D line search: try scaling factors
                best_factor = 1.0
                best_local_sc = sc
                for factor in np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]):
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc_trial = compute_score(a_norm)
                    if sc_trial < best_local_sc:
                        best_local_sc = sc_trial
                        best_factor = factor

                if best_factor != 1.0:
                    a[j] = old_val * best_factor
                    S = float(a.sum())
                    a = a / S
                    sc = best_local_sc
                    improved = True
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                else:
                    a[j] = old_val

        if not improved:
            break

    return best_a


def build_alternating_init(n, ratio=1.707):
    """Build an alternating h/l initialization with optimal ratio."""
    a = np.empty(n)
    a[0::2] = ratio
    a[1::2] = 1.0
    return a / a.sum()


def build_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_chirp_init(n, cycles=2.0):
    """Absolute chirp initialization (non-negative)."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.abs(np.sin(2 * np.pi * cycles * x * x)) + 0.1
    return a / a.sum()


def build_multi_harmonic_init(n, freqs=(1, 2, 3), amps=None):
    """Multi-harmonic non-negative initialization."""
    x = np.arange(n, dtype=np.float64) / n
    if amps is None:
        amps = np.array([1.0, 0.5, 0.25][:len(freqs)])
    a = np.ones(n)
    for i, f in enumerate(freqs):
        a += amps[i] * np.cos(2 * np.pi * f * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def harmonic_restart(a, n_harmonics=5, amp=0.02, seed=0):
    """
    Create perturbed versions of a by adding small harmonic components.
    Returns a list of perturbed sequences.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    results = []
    for h in range(1, n_harmonics + 1):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation = amp * np.cos(2 * np.pi * freq * x + phase)
        a_new = a + perturbation
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S > 1e-12:
            a_new = a_new / S
            results.append(a_new)
    return results


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def run_lbfgs_safe(a0, beta, maxiter=2000, maxfun=30000):
    """Run L-BFGS-B on LSE-smoothed objective with safety checks."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = float(a_out.sum())
    if S < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    Multi-phase optimization: Nesterov LSE with Polyak averaging,
    active-set subgradient, coordinate descent, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) / 4096.0

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 0: Initialization (5% of time)
    # ============================================================
    phase0_end = t_start + 0.05 * total_budget
    print(f"=== Phase 0: Initialization ===")

    n_target = len(parent) if parent is not None else 6962
    candidates = []

    if parent is not None:
        candidates.append(('parent', parent.copy()))
        sc_p = compute_score(parent)
        print(f"  Parent score: {sc_p:.8f}")
        update_best(sc_p, parent, "parent")

    # Build diverse initializations at target scale
    inits = [
        ('alternating', build_alternating_init(n_target, 1.707)),
        ('gaussian', build_gaussian_init(n_target, 0.3)),
        ('chirp', build_chirp_init(n_target, 2.0)),
        ('multi_harm', build_multi_harmonic_init(n_target, (1, 2, 3, 5), [1.0, 0.5, 0.3, 0.15])),
        ('multi_harm2', build_multi_harmonic_init(n_target, (2, 3, 7, 11), [0.8, 0.5, 0.3, 0.2])),
    ]

    for name, a_init in inits:
        if time.monotonic() > phase0_end:
            break
        sc = compute_score(a_init)
        print(f"  {name}: {sc:.8f}")
        update_best(sc, a_init, name)
        candidates.append((name, a_init))

    # Quick Nesterov on best init to get a warm start
    if time.monotonic() < phase0_end - 5:
        warm = nesterov_lse_polyak(best_a, beta_start=20, beta_end=200,
                                    max_time=phase0_end - time.monotonic() - 2,
                                    lr_start=0.03, lr_decay=0.999,
                                    mu_start=0.8, mu_end=0.95,
                                    beta_growth=1.01)
        update_best(compute_score(warm), warm, "warm-start")

    print(f"After Phase 0: best = {best_score:.8f}")

    # ============================================================
    # Phase 1: Multi-beta Nesterov LSE with Polyak (40% of time)
    # ============================================================
    phase1_end = t_start + 0.45 * total_budget
    print(f"\n=== Phase 1: Multi-beta Nesterov LSE (t < {phase1_end - t_start:.0f}s) ===")

    # Geometric schedule of betas
    beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]

    # Time allocation: more time at higher betas
    time_per_beta = []
    remaining = phase1_end - time.monotonic()
    # Allocate proportionally to log-spaced intervals
    weights = np.array([1, 1, 1.5, 1.5, 2, 2, 2.5, 2.5, 3, 3, 3, 3])
    weights = weights / weights.sum()
    time_per_beta = weights * remaining * 0.9

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_end:
            break
        t_b = max(5.0, time_per_beta[i])
        t_available = min(t_b, phase1_end - time.monotonic())
        if t_available < 5:
            break

        print(f"  Beta={beta}, time={t_available:.0f}s")
        a_out = nesterov_lse_polyak(best_a, beta_start=beta, beta_end=beta * 5,
                                     max_time=t_available,
                                     lr_start=0.015 / np.sqrt(beta / 10.0),
                                     lr_decay=0.9997,
                                     mu_start=0.85, mu_end=0.96,
                                     beta_growth=1.005)
        update_best(compute_score(a_out), a_out, f"nesterov-b{beta}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Active-set subgradient (20% of time)
    # ============================================================
    phase2_end = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: Active-set optimization (t < {phase2_end - t_start:.0f}s) ===")

    t_avail = phase2_end - time.monotonic()
    if t_avail > 10:
        a_as = active_set_optimize(best_a, max_time=t_avail * 0.7,
                                    n_top_bands=5, step_init=0.005)
        update_best(compute_score(a_as), a_as, "active-set")
        # Follow up with Nesterov
        if time.monotonic() < phase2_end - 5:
            a_as2 = nesterov_lse_polyak(a_as, beta_start=500, beta_end=5000,
                                         max_time=phase2_end - time.monotonic() - 2,
                                         lr_start=0.005, lr_decay=0.9998,
                                         mu_start=0.9, mu_end=0.96,
                                         beta_growth=1.005)
            update_best(compute_score(a_as2), a_as2, "active-set+nest")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Coordinate descent with line search (20% of time)
    # ============================================================
    phase3_end = t_start + 0.85 * total_budget
    print(f"\n=== Phase 3: Coordinate descent (t < {phase3_end - t_start:.0f}s) ===")

    t_avail = phase3_end - time.monotonic()
    if t_avail > 10:
        a_cd = coordinate_descent_ls(best_a, max_time=t_avail * 0.6,
                                      n_top_peaks=30, n_top_contrib=20)
        update_best(compute_score(a_cd), a_cd, "coord-desc")

        # Quick Nesterov after CD
        if time.monotonic() < phase3_end - 5:
            a_cd2 = nesterov_lse_polyak(a_cd, beta_start=1000, beta_end=10000,
                                         max_time=phase3_end - time.monotonic() - 2,
                                         lr_start=0.003, lr_decay=0.9999,
                                         mu_start=0.92, mu_end=0.97,
                                         beta_growth=1.003)
            update_best(compute_score(a_cd2), a_cd2, "cd+nest")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Harmonic restarts + re-optimization (10% of time)
    # ============================================================
    phase4_end = t_start + 0.95 * total_budget
    print(f"\n=== Phase 4: Harmonic restarts (t < {phase4_end - t_start:.0f}s) ===")

    t_avail = phase4_end - time.monotonic()
    if t_avail > 20:
        # Generate harmonic perturbations
        perturbed = harmonic_restart(best_a, n_harmonics=8, amp=0.01, seed=seed)
        n_pert = len(perturbed)
        time_per_pert = t_avail * 0.8 / n_pert

        for i, a_p in enumerate(perturbed):
            if time.monotonic() > phase4_end - 5:
                break
            t_p = min(time_per_pert, phase4_end - time.monotonic() - 5)
            if t_p < 3:
                break
            a_opt = nesterov_lse_polyak(a_p, beta_start=100, beta_end=5000,
                                         max_time=t_p,
                                         lr_start=0.01, lr_decay=0.9995,
                                         mu_start=0.85, mu_end=0.96,
                                         beta_growth=1.01)
            update_best(compute_score(a_opt), a_opt, f"restart-{i}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final L-BFGS polish (5% of time)
    # ============================================================
    print(f"\n=== Phase 5: Final L-BFGS polish ===")

    final_time = deadline - time.monotonic()
    if final_time > 10:
        for beta in [2000, 5000, 10000]:
            if time.monotonic() > deadline - 5:
                break
            t_avail_final = min(final_time / 3.0, deadline - time.monotonic() - 3)
            if t_avail_final < 5:
                break
            # L-BFGS with time-limited iterations
            maxiter = max(100, int(t_avail_final * 200))
            maxfun = max(500, int(t_avail_final * 1000))
            sc, a_opt = run_lbfgs_safe(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            update_best(sc, a_opt, f"lbfgs-b{beta}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
