"""
Multi-pronged non-negative sequence optimizer:
- GS/Fienup-style flat-convolution projection
- Active-constraint linear-programming steps based on the current peaks
- Low-dimensional DCT/softmax parameterization for global shape search
- Full-space log-softmax L-BFGS with annealed log-sum-exact peak surrogate
- Exact 1-D coordinate polishing and log-space peak descent
All helpers are top-level and avoid closures/lambdas.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

OPT_DEADLINE = 0.0


def opt_callback(xk):
    return time.monotonic() >= OPT_DEADLINE


def next_power_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    mf = next_power_two(m)
    A = np.fft.rfft(a, mf)
    c = np.fft.irfft(A * A, mf)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-12:
        return float("inf")
    c = fast_conv(a)
    cm = float(np.max(c)) if c.size else 0.0
    return float(2.0 * a.size * cm / (S * S))


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 0.5
        S = 0.5 * a.size
    return a / S


def softmax_from_log(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.zeros(0, dtype=np.float64)
    z = y - float(np.max(y))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return e / s


def convolution_derivative(a, p):
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 3 * n - 2
    mf = next_power_two(m)
    ra = np.asarray(a[::-1], dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    R = np.fft.rfft(ra, mf)
    P = np.fft.rfft(p, mf)
    h = np.fft.irfft(R * P, mf)[n - 1:2 * n - 1]
    return 2.0 * h


def lse_components_normalized(a, beta):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if n == 0:
        return 0.0, np.zeros(0, dtype=np.float64), 0.0
    c = fast_conv(a)
    cmax = float(np.max(c)) if c.size else 0.0
    if beta <= 0.0:
        p = np.full(c.size, 1.0 / c.size if c.size > 0 else 1.0)
        lse_val = float(np.mean(c)) if c.size else 0.0
    else:
        shifted = beta * (c - cmax)
        np.minimum(shifted, 0.0, out=shifted)
        ev = np.exp(shifted)
        sum_exp = float(ev.sum())
        if not np.isfinite(sum_exp) or sum_exp <= 1e-300:
            p = np.full(c.size, 1.0 / c.size if c.size > 0 else 1.0)
            lse_val = cmax
        else:
            p = ev / sum_exp
            lse_val = cmax + float(np.log(sum_exp)) / float(beta)
    h = convolution_derivative(a, p)
    dot = float(np.dot(h, a))
    return float(lse_val), h, dot


def objective_log_full(y, beta):
    n = len(y)
    a = softmax_from_log(y)
    lse_val, h, dot = lse_components_normalized(a, beta)
    val = 2.0 * n * lse_val
    vec = a * (h - dot)
    grad = 2.0 * n * vec
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def build_dct_basis(n, nfreq):
    n = int(n)
    nfreq = int(min(nfreq, max(1, n // 4)))
    idx = np.arange(n, dtype=np.float64) / float(n)
    cols = []
    for f in range(1, nfreq + 1):
        arg = 2.0 * np.pi * f * idx
        cols.append(np.cos(arg))
        if 2 * f < n:
            cols.append(np.sin(arg))
    if not cols:
        return np.zeros((n, 0), dtype=np.float64)
    return np.column_stack(cols).astype(np.float64, copy=False)


def fit_theta(a, basis):
    if basis.size == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64).ravel()
    f0 = np.log(np.maximum(a, 1e-300))
    f0 = f0 - float(np.mean(f0))
    colnorm = np.mean(basis * basis, axis=0)
    colnorm = np.maximum(colnorm, 1e-12)
    theta = (basis.T @ f0) / colnorm
    if not np.all(np.isfinite(theta)):
        theta = np.zeros(basis.shape[1], dtype=np.float64)
    return theta


def objective_theta(theta, beta, basis):
    n = basis.shape[0]
    f = basis @ theta
    a = softmax_from_log(f)
    lse_val, h, dot = lse_components_normalized(a, beta)
    val = 2.0 * n * lse_val
    vec = a * (h - dot)
    grad = 2.0 * basis.T @ vec
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(theta.size, dtype=np.float64)
    return float(val), grad


def get_beta_ladder(n):

def estimate_maxfun(remaining_seconds, n, low_dim=False):
    remaining_seconds = float(max(0.0, remaining_seconds))
    if remaining_seconds <= 0.0:
        return 0
    n = int(max(1, n))
    scale = n / 6962.0
    if scale < 1.0:
        scale = 1.0
    if low_dim:
        cost = 0.012 * scale
        lo, hi = 800, 25000
    else:
        cost = 0.022 * scale
        lo, hi = 350, 16000
    mf = int(remaining_seconds / max(cost, 1e-6) * 0.55)
    return max(lo, min(hi, mf))


def gs_fienup(a, iters=5000, alpha_init=0.7, alpha_final=0.1, deadline=None):
    a = normalize_seq(a)
    n = a.size
    if n == 0:
        return a
    m = 2 * n - 1
    mf = next_power_two(m)
    target = np.zeros(mf, dtype=np.float64)
    target[:m] = 1.0 / m
    T = np.fft.fft(target)
    T_mag = np.maximum(np.abs(T), 1e-300)
    T_phase = np.angle(T)

    best_a = a.copy()
    best_sc = compute_score(a)
    if deadline is None:
        deadline = float("inf")

    for it in range(int(iters)):
        if time.monotonic() >= float(deadline):
            break
        alpha = alpha_init + (alpha_final - alpha_init) * (it / max(1, int(iters)))
        A = np.fft.fft(a, mf)
        A_mag = np.maximum(np.abs(A), 1e-300)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * (T_mag ** (0.5 * alpha))
        new_phase = (1.0 - alpha) * A_phase + 0.5 * alpha * T_phase
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, mf).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def make_variants_from_parent(parent, rng, n):

        ]
        return [normalize_seq(z) for z in inits]

    if len(parent) != n:
        xo = np.linspace(0.0, 1.0, len(parent), dtype=np.float64)

    else:

    p = normalize_seq(p)
    out = [p]

    k = 31
    ker = np.ones(k, dtype=np.float64) / float(k)
    out.append(normalize_seq(np.convolve(p, ker, mode="same")))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out.append(normalize_seq(np.maximum(p * (1.0 + 0.35 * np.cos(np.pi * x)), 1e-8)))

    f = np.log(np.maximum(p, 1e-12)) + 0.10 * rng.standard_normal(n)
    out.append(normalize_seq(np.exp(f - float(np.max(f)))))
    return out


def interpolate_seq(a, n):
    a = normalize_seq(a)
    n = int(n)
    if a.size == n:
        return a.copy()
    xo = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    xn = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(xn, xo, a)
    b = np.maximum(b, 1e-8)
    return normalize_seq(b)


def active_lp_blend_once(a, top_m, M_bound, t_values=(0.05, 0.1, 0.25, 0.5, 0.8, 1.0)):
    a = normalize_seq(a)
    n = a.size
    if n == 0:
        return None, 0.0
    c = fast_conv(a)
    Mcur = float(np.max(c)) if c.size else 0.0
    if Mcur <= 0.0:
        return None, Mcur
    M_bound = min(float(M_bound), Mcur)
    if M_bound <= 1e-15:
        return None, Mcur

    top_m = int(max(1, min(top_m, c.size)))
    idx_all = np.argsort(c)[::-1]
    thresh = max(M_bound, 0.45 * Mcur)
    candidates = idx_all[idx_all >= thresh]
    if candidates.size < top_m:
        candidates = idx_all[:top_m]
    else:
        candidates = candidates[:top_m]
    if candidates.size == 0:
        candidates = idx_all[:top_m]

    A = np.zeros((candidates.size, n), dtype=np.float64)
    for r, k in enumerate(candidates):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            partners = np.arange(k - js, k - je - 1, -1)
            A[r, js:je + 1] = a[partners]

    obj = -np.ones(n, dtype=np.float64)
    b = np.full(candidates.size, M_bound, dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            disp=False,
            options={"presolve": True},
        )
    except Exception:
        return None, Mcur

    if not getattr(res, "success", False):
        return None, Mcur
    g = getattr(res, "x", None)
    if g is None:
        return None, Mcur
    g = np.asarray(g, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return None, Mcur
    g = np.clip(g, 0.0, 1.0)
    Sg = float(g.sum())
    if Sg <= 1e-10:
        return None, Mcur
    g = g / Sg

    best_cand = None
    best_sc = float("inf")
    for t in t_values:
        if t <= 0.0:
            cand = a.copy()
        elif t >= 1.0:
            cand = g.copy()
        else:
            cand = (1.0 - t) * a + t * g
        cand = normalize_seq(cand)
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_cand = cand.copy()
    return best_cand, Mcur


def active_lp_search(a0, deadline, top_m_list=(6, 10, 16), reduce_facs=(0.99, 0.97, 0.94), target_score=1.49):
    a = normalize_seq(a0)
    if a.size == 0:
        return a, float("inf")
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    attempt = 0
    stagnation = 0

    while time.monotonic() < float(deadline) and attempt < 100 and stagnation < 18:
        c = fast_conv(a)
        Mcur = float(np.max(c)) if c.size else 0.0
        if Mcur <= 0.0:
            break
        top_m = int(top_m_list[attempt % len(top_m_list)])
        if (attempt % 4 == 0) and target_score is not None:
            Mtry = min(Mcur, float(target_score) / (2.0 * n))
        else:
            ri = (attempt // len(top_m_list)) % len(reduce_facs)
            Mtry = Mcur * float(reduce_facs[ri])
            if target_score is not None and (attempt % 8 == 3):
                Mtry = min(Mtry, float(target_score) / (2.0 * n))
        Mtry = min(max(Mtry, 0.0), Mcur)

        cand, _ = active_lp_blend_once(a, top_m, Mtry)
        if cand is None:
            stagnation += 1
        else:
            sc = compute_score(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-9:
                best_a = cand.copy()
                best_sc = sc
                a = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        attempt += 1

    return best_a, best_sc

        p[idx] = 1.0 / max(1, q)
        h = convolution_derivative(a, p)
        dot = float(np.dot(h, a))
        d = a * (h - dot)
        dmax = float(np.max(np.abs(d))) if d.size else 0.0
        if dmax < 1e-14:
            break
        direction = d / dmax

        accepted = False
        e = eta
        for _ in range(10):
            z = loga - e * direction
            z -= float(np.max(z))
            trial = np.exp(z)
            s = float(trial.sum())
            if s < 1e-12:
                e *= 0.5
                continue
            trial /= s
            sc = compute_score(trial)
            if np.isfinite(sc) and sc < best_sc - 1e-9 and sc < current_sc * (1.0 - 1e-7):
                loga = z
                a = trial
                current_sc = sc
                no_improve = 0
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                eta = min(2.0, eta * 1.15)
                accepted = True
                break
            e *= 0.5
            if e < 1e-4:
                break

        if not accepted:
            no_improve += 1
            if no_improve > 25:
                break
            eta *= 0.5
            if eta < 1e-4:
                break
        qi += 1
        if qi % 80 == 0:
            loga -= float(np.median(loga))

    return best_a, best_sc


def random_log_descent(a0, deadline, rng, sigma=0.06):
    a = normalize_seq(a0)
    if a.size == 0:
        return a, float("inf")
    best_a = a.copy()
    best_sc = compute_score(a)
    loga = np.log(a + 1e-300)
    sigma = float(sigma)
    misses = 0

    while time.monotonic() < float(deadline):
        delta = rng.normal(0.0, sigma, a.size)
        z = loga + delta
        z -= float(np.max(z))
        trial = np.exp(z)
        s = float(trial.sum())
        if s < 1e-12:
            sigma *= 0.6
            continue
        trial /= s
        sc = compute_score(trial)
        if np.isfinite(sc) and sc < best_sc - 1e-9:
            loga = z
            a = trial
            best_a = a.copy()
            best_sc = sc
            misses = 0
            sigma = min(0.35, sigma * 1.05)
        else:
            misses += 1
            if misses > 40:
                sigma *= 0.7
                misses = 0
                if sigma < 1e-4:
                    break
    return best_a, best_sc


def coordinate_peak_descent(a0, deadline, K=160, top_peaks=32, ratios=None):
    if ratios is None:
        ratios = (0.25, 0.45, 0.65, 0.85, 1.15, 1.4, 1.8)
    a = normalize_seq(a0)
    if a.size == 0:
        return a, float("inf")
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    no_sweep = 0
    sweep = 0
    K_cur = int(K)
    peaks_cur = int(top_peaks)

    while time.monotonic() < float(deadline) and no_sweep < 5 and sweep < 60:
        c = fast_conv(a)
        cur_sc = 2.0 * n * float(np.max(c)) if c.size else 0.0
        q = min(peaks_cur, c.size)
        idx = np.argsort(c)[::-1][:q]
        p = np.zeros(c.size, dtype=np.float64)
        p[idx] = 1.0 / max(1, q)
        h = convolution_derivative(a, p)
        dot = float(np.dot(h, a))
        vec = a * (h - dot)
        order = np.argsort(np.abs(vec))[::-1][:K_cur]
        improved = False

        for j in order:
            if time.monotonic() >= float(deadline):
                break
            j = int(j)
            old = float(a[j])
            if old < 1e-14:
                continue
            seg = np.zeros(c.size, dtype=np.float64)
            seg[j:j + n] = 2.0 * a
            best_lam = 1.0
            best_cur = cur_sc
            for lam in ratios:
                lam = float(lam)
                if lam <= 0.0 or abs(lam - 1.0) < 1e-12:
                    continue
                delta = (lam - 1.0) * old
                S_trial = 1.0 + delta
                if S_trial < 1e-8:
                    continue
                c_trial = c + delta * seg
                d2 = 2 * j
                if d2 < c.size:
                    c_trial[d2] += (lam - 1.0) * (lam - 1.0) * old * old
                mc = float(np.max(c_trial)) if c_trial.size else 0.0
                sc_trial = 2.0 * n * mc / (S_trial * S_trial)
                if np.isfinite(sc_trial) and sc_trial < best_cur - 1e-10:
                    best_cur = sc_trial
                    best_lam = lam

            if best_lam != 1.0:
                a[j] = best_lam * old
                a = normalize_seq(a)
                c = fast_conv(a)
                cur_sc = 2.0 * n * float(np.max(c)) if c.size else 0.0
                improved = True
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_a = a.copy()

        sweep += 1
        if improved:
            no_sweep = 0
        else:
            no_sweep += 1
            K_cur = max(50, K_cur // 2)
            peaks_cur = max(8, peaks_cur * 2)

    return best_a, best_sc


def sanitize_output(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64) * 0.5
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if S < 0.01:
        a[:] = 0.5
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global OPT_DEADLINE

    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    end = max(t_start + 15.0, t_start + total_budget - 45.0)
    rng = np.random.default_rng(seed)

    print(f"Starting optimizer, budget={total_budget:.0f}s, end_offset={end - t_start:.0f}s")

    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            arr = np.asarray(p, dtype=np.float64).ravel()
            if arr.size >= 32:
                parent = normalize_seq(arr)
    except Exception:
        parent = None

    if parent is None:
        n_main = 6962
        base = np.ones(n_main, dtype=np.float64) * 0.5
    else:
        n_main = parent.size
        base = parent.copy()
        print(f"Using parent length={n_main}")

    variants = make_variants_from_parent(parent, rng, n_main)
    best_a = normalize_seq(base)
    best_score = compute_score(best_a)
    print(f"Initial parent/uniform length={n_main} score={best_score:.8f}")

    p1_end = t_start + 0.35 * max(30.0, total_budget - 45.0)
    p2_end = t_start + 0.72 * max(30.0, total_budget - 45.0)
    p3_end = t_start + 0.90 * max(30.0, total_budget - 45.0)

    # -----------------------------------------------------------------
    # Phase 1: GS projections, active LP, and low-dimensional shape search
    # -----------------------------------------------------------------
    print("Phase 1: GS + active LP + DCT theta search")

    gs_limit = min(p1_end, t_start + 0.10 * total_budget)
    for a0 in variants[:4]:
        if time.monotonic() >= gs_limit:
            break
        a1 = gs_fienup(a0, 6000, 0.7, 0.1, deadline=gs_limit)
        sc = compute_score(a1)
        if np.isfinite(sc) and sc < best_score:
            best_a = a1.copy()
            best_score = sc
            print(f"  GS improve: {best_score:.8f} (n={n_main})")
        if time.monotonic() < gs_limit:
            a2 = gs_fienup(a1, 3500, 0.5, 0.05, deadline=gs_limit)
            sc2 = compute_score(a2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_a = a2.copy()
                best_score = sc2
                print(f"  GS2 improve: {best_score:.8f} (n={n_main})")

    if time.monotonic() < p1_end:
        lp_deadline = min(p1_end, time.monotonic() + min(260.0, 0.16 * total_budget))
        lp_tops = (4, 8) if n_main > 10000 else (6, 10, 16)
        a_lp, sc_lp = active_lp_search(
            best_a,
            deadline=lp_deadline,
            top_m_list=lp_tops,
            reduce_facs=(0.99, 0.97, 0.94),
            target_score=1.49,
        )
        if sc_lp < best_score:
            best_a = a_lp.copy()
            best_score = sc_lp
            print(f"  Active LP improve: {best_score:.8f}")

    if time.monotonic() < p1_end:
        a_gs_after_lp = gs_fienup(best_a, 5000, 0.55, 0.08, deadline=p1_end)
        sc = compute_score(a_gs_after_lp)
        if np.isfinite(sc) and sc < best_score:
            best_a = a_gs_after_lp.copy()
            best_score = sc
            print(f"  GS after LP improve: {best_score:.8f}")

    if time.monotonic() < p1_end:
        if n_main > 12000:
            nfreqs = [48, 96]
        elif n_main > 8000:
            nfreqs = [64, 128]
        else:
            nfreqs = [64, 128, 256]

        theta_starts = [best_a] + variants[:2]
        for nfreq in nfreqs:
            if time.monotonic() >= p1_end:
                break
            basis = build_dct_basis(n_main, nfreq)
            if basis.size == 0:
                continue
            betas_theta = [
                max(2000, 4 * n_main),
                max(6000, 12 * n_main),
            ]
            for a_start in theta_starts:
                if time.monotonic() >= p1_end:
                    break
                sub_deadline = min(p1_end, time.monotonic() + min(180.0, 0.09 * total_budget))
                a_th, sc_th = run_theta_lbfgs_ladder(
                    a_start, betas_theta, basis, sub_deadline, label=f"theta{n_main}_{nfreq}"
                )
                if sc_th < best_score:
                    best_a = a_th.copy()
                    best_score = sc_th
                    print(f"  Theta improve: {best_score:.8f}")

    # -----------------------------------------------------------------
    # Phase 2: full-space log-softmax L-BFGS with annealed LSE
    # -----------------------------------------------------------------
    print("Phase 2: full log-softmax L-BFGS")
    if time.monotonic() < p2_end:
        betas_full = get_beta_ladder(n_main)
        a_fl, sc_fl = run_full_lbfgs_ladder(best_a, betas_full, p2_end, label=f"full{n_main}")
        if sc_fl < best_score:
            best_a = a_fl.copy()
            best_score = sc_fl
            print(f"  Full LBFGS improve: {best_score:.8f}")

    if time.monotonic() < p2_end - 50:
        high_betas = get_beta_ladder(n_main)[-2:]
        a_fl2, sc_fl2 = run_full_lbfgs_ladder(best_a, high_betas, p2_end, label=f"full-hi{n_main}")
        if sc_fl2 < best_score:
            best_a = a_fl2.copy()
            best_score = sc_fl2
            print(f"  High-beta full improve: {best_score:.8f}")

    # -----------------------------------------------------------------
    # Phase 3: exact coordinate polishing and log peak descent
    # -----------------------------------------------------------------
    print("Phase 3: coordinate / log peak polishing")
    if time.monotonic() < p3_end:
        coord_deadline = min(p3_end, time.monotonic() + min(160.0, 0.10 * total_budget))
        a_cd, sc_cd = coordinate_peak_descent(best_a, deadline=coord_deadline, K=160, top_peaks=32)
        if sc_cd < best_score:
            best_a = a_cd.copy()
            best_score = sc_cd
            print(f"  Coordinate improve: {best_score:.8f}")

    if time.monotonic() < p3_end:
        log_deadline = min(p3_end, time.monotonic() + min(120.0, 0.07 * total_budget))
        a_lg, sc_lg = log_peak_descent(best_a, deadline=log_deadline)
        if sc_lg < best_score:
            best_a = a_lg.copy()
            best_score = sc_lg
            print(f"  Log peak improve: {best_score:.8f}")

    if time.monotonic() < p3_end:
        a_rd, sc_rd = random_log_descent(best_a, deadline=p3_end, rng=rng, sigma=0.045)
        if sc_rd < best_score:
            best_a = a_rd.copy()
            best_score = sc_rd
            print(f"  Random log improve: {best_score:.8f}")

    # -----------------------------------------------------------------
    # Phase 4: alternate-length exploration if time remains
    # -----------------------------------------------------------------
    if time.monotonic() < end - 160:
        if len(best_a) == 8192:
            alt_n = 16384
        elif len(best_a) == 16384:
            alt_n = 8192
        else:
            alt_n = 8192
        alt_a = interpolate_seq(best_a, alt_n)

        a_alt1 = gs_fienup(alt_a, 5000, 0.6, 0.1, deadline=min(end, time.monotonic() + 70))
        sc = compute_score(a_alt1)
        if np.isfinite(sc) and sc < best_score:
            best_a = a_alt1.copy()
            best_score = sc
            print(f"  Alt n={alt_n} GS improve: {best_score:.8f}")

        if time.monotonic() < end - 100:
            a_alt2, sc_alt2 = active_lp_search(
                a_alt1,
                deadline=min(end, time.monotonic() + 90),
                top_m_list=(4, 8),
                reduce_facs=(0.98, 0.96),
                target_score=1.49,
            )
            if sc_alt2 < best_score:
                best_a = a_alt2.copy()
                best_score = sc_alt2
                print(f"  Alt n={alt_n} LP improve: {best_score:.8f}")

        if time.monotonic() < end - 80:
            betas_alt = get_beta_ladder(alt_n)[:3]
            a_alt3, sc_alt3 = run_full_lbfgs_ladder(
                a_alt1, betas_alt, min(end, time.monotonic() + 120), label=f"alt-full{alt_n}"
            )
            if sc_alt3 < best_score:
                best_a = a_alt3.copy()
                best_score = sc_alt3
                print(f"  Alt n={alt_n} full improve: {best_score:.8f}")

    # -----------------------------------------------------------------
    # Final polish before verification
    # -----------------------------------------------------------------
    if time.monotonic() < end - 70:
        high_beta = max(get_beta_ladder(len(best_a)))
        a_final, sc_final = run_full_lbfgs_ladder(
            best_a, [high_beta], min(end, time.monotonic() + 70), label="final-full"
        )
        if sc_final < best_score:
            best_a = a_final.copy()
            best_score = sc_final
            print(f"  Final full improve: {best_score:.8f}")

    if not np.isfinite(best_score):
        best_a = normalize_seq(np.ones(max(32, n_main), dtype=np.float64) * 0.5)
        best_score = compute_score(best_a)

    out = sanitize_output(best_a)

    verified = best_score
    final_pool = [best_a]
    if parent is not None and parent.size != best_a.size:
        final_pool.append(parent)

    if time.monotonic() < end and len(out) <= 14000:
        best_verified = np.inf
        best_out = None
        for cand in final_pool[:2]:
            cand_out = sanitize_output(cand)
            try:
                v = float(evaluate_sequence(cand_out))
            except Exception:
                v = compute_score(np.asarray(cand, dtype=np.float64))
            if np.isfinite(v) and v < best_verified:
                best_verified = v
                best_out = cand_out
        if best_out is not None:
            out = best_out
            verified = best_verified
        else:
            try:
                v = float(evaluate_sequence(out))
                if np.isfinite(v):
                    verified = v
            except Exception:
                verified = compute_score(np.asarray(out, dtype=np.float64))

    elapsed = time.monotonic() - t_start
    print("=" * 70)
    print(f"FINAL internal score: {best_score:.8f}")
    print(f"FINAL verified score: {verified:.8f}" if np.isfinite(verified) else "FINAL verified score: inf")
    print(f"Length: {len(out)}, elapsed: {elapsed:.1f}s / {total_budget:.0f}s")
    print("=" * 70)

    return out
