"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS homotopy with increasing smoothness parameter (beta),
top-k focused LSE for sharp peak targeting, and Nesterov-accelerated
subgradient as a complementary method.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def make_nonneg_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def fienup_compress(a, n_iters=300, top_frac=0.3, gamma=0.7):
    """Spectral Fienup: compress top-frequency bins."""
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Adaptive top fraction
        tf = top_frac * (0.5 + 0.5 * progress)
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - tf)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = np.sqrt(thresh / P[above])
            scale[above] = ratio ** gamma
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def fienup_flat(a, n_iters=300, alpha=0.5):
    """Spectral flattening: relax towards flat spectrum."""
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        a_eff = alpha * (0.5 + 0.5 * progress)
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - a_eff) * P + a_eff * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_mod = A * amp
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_objective(x, beta):
    """Standard LSE-smoothed max of convolution."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_v = np.exp(scaled)
    log_sum = c_max + np.log(exp_v.sum()) / beta
    score = 2.0 * n * log_sum / (S * S)
    
    # Gradient
    p = exp_v / exp_v.sum()
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    p_fft = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(rx_fft * p_fft, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * log_sum / S3)
    
    return score, grad


def topk_lse_objective(x, beta, k):
    """Top-k focused LSE: weights the top-k convolution values more heavily."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Sort c in descending order
    c_sorted = np.sort(c)[::-1]
    c_top = float(c_sorted[0])
    k_eff = min(k, M)
    
    # LSE over top-k values
    top_vals = c_sorted[:k_eff]
    scaled = beta * (top_vals - c_top)
    scaled = np.clip(scaled, -500, 500)
    exp_v = np.exp(scaled)
    L = c_top + np.log(exp_v.sum()) / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Weights on the top-k values
    w = exp_v / exp_v.sum()
    
    # Map back: c_sorted[i] corresponds to c[idx_sorted[i]]
    idx_sorted = np.argsort(c)[::-1]
    top_idx = idx_sorted[:k_eff]
    
    # Gradient: sum over top-k of w[i] * d(c[k_i])/dx
    # d(c[k])/dx_i = 2 * x[k-i] (for valid i)
    rev_x = x[::-1]
    p = np.zeros(M)
    for i, idx in enumerate(top_idx):
        p[int(idx)] = w[i]
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    p_fft = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(rx_fft * p_fft, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    
    return score, grad


def run_lbfgs(a0, obj_func, maxiter=20000, maxfun=500000):
    """Run L-BFGS-B with the given objective function."""
    n = len(a0)
    a0 = make_nonneg_sum1(a0.copy())
    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            obj_func, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.maximum(res.x, 0.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    
    a_out = make_nonneg_sum1(a_out)
    return compute_score(a_out), a_out


def nesterov_subgrad(a0, n_iters=5000, top_k=10, lr_init=0.01, momentum=0.9):
    """Nesterov-accelerated subgradient method on top-k peaks."""
    n = len(a0)
    a = make_nonneg_sum1(a0.copy())
    v = np.zeros(n)  # velocity
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        # Nesterov lookahead
        a_look = np.maximum(a - momentum * v, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

def pnorm_objective(x, p):
    """p-norm of convolution as smooth approximation to max."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    c = np.maximum(c, 0.0)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_p = float(np.sum(c ** p)) ** (1.0 / p)
    score = 2.0 * n * c_p / (S * S)
    
    # Gradient of c_p w.r.t. x_i
    # dc_p/dx_i = (1/p) * (sum c^p)^(1/p - 1) * p * c^(p-1) * dc/dx_i
    # = (sum c^p)^(1/p - 1) * c^(p-1) * dc/dx_i
    # dc/dx_i = 2 * conv(x, delta_i) = 2 * (x reversed at appropriate offset)
    
    if p == 1:
        w = np.ones_like(c)
    else:
        c_safe = np.maximum(c, 1e-20)
        w = c_safe ** (p - 1)
        w = w * (float(np.sum(c ** p)) ** (1.0 / p - 1.0))
    
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    w_fft = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(rx_fft * w_fft, M2_fft)[n - 1:2 * n - 1]
    g_p = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_p / S2 - 2.0 * c_p / S3)
    
    return score, grad

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def make_nonneg_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def fienup_compress(a, n_iters, top_frac, gamma):
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        tf = top_frac * (0.5 + 0.5 * progress)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - tf)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = np.sqrt(thresh / P[above])
            scale[above] = ratio ** gamma
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

        return 1e10, np.zeros(n)

    scaled = np.clip(scaled, -500, 500)
    exp_v = np.exp(scaled)
    log_sum = c_max + np.log(exp_v.sum()) / beta

        )

        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    a_out = make_nonneg_sum1(a_out)
    return compute_score(a_out), a_out

        a_new = make_nonneg_sum1(a_new)
        a = a_new
        momentum = max(0.5, momentum * 0.9999)
    
    return best_a

"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS homotopy with increasing smoothness, top-k focused LSE
for sharp peak targeting, Nesterov-accelerated subgradient, and p-norm
objectives. Key improvements: longer L-BFGS runs at high beta, adaptive
top-k scheduling, and momentum-based subgradient with Polyak step sizes.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based full self-convolution."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute evaluation score 2n*max(c)/S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def make_nonneg_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def fienup_compress(a, n_iters, top_frac_init, top_frac_end, gamma):
    """Spectral Fienup: compress top-frequency bins with power law."""
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        tf = top_frac_init + (top_frac_end - top_frac_init) * progress
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - tf)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = np.sqrt(max(thresh, 1e-20) / P[above])
            scale[above] = ratio ** gamma
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_flat(a, n_iters, alpha_init, alpha_end):
    """Spectral flattening via amplitude relaxation."""
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        a_eff = alpha_init + (alpha_end - alpha_init) * progress
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - a_eff) * P + a_eff * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_mod = A * amp
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_multilevel(a, n_iters, n_levels, strength):
    """Multi-level water-filling spectral compression."""
    n = len(a)
    a = make_nonneg_sum1(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        scale = np.ones_like(P)
        for lv in range(n_levels):
            frac = 0.08 * (lv + 1) / n_levels
            thresh = np.quantile(P, 1.0 - frac)
            above = P > thresh
            if above.any():
                s = strength * (0.3 + 0.7 * progress)
                ratio = np.sqrt(max(thresh, 1e-20) / P[above])
                scale[above] *= (1.0 - s + s * np.maximum(ratio, 0.01))
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def _lse_obj(x, beta):
    """LSE-smoothed max of self-convolution, with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_v = np.exp(scaled)
    exp_sum = float(exp_v.sum())
    log_sum = c_max + float(np.log(max(exp_sum, 1e-300))) / beta
    score = 2.0 * n * log_sum / (S * S)
    p = exp_v / max(exp_sum, 1e-300)
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    p_fft = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(rx_fft * p_fft, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * log_sum / S3)
    return score, grad


def _topk_lse_obj(x, beta, k):
    """Top-k focused LSE: only smooth over the top-k convolution values."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    idx_sorted = np.argsort(c)[::-1]
    k_eff = min(k, M)
    c_top = float(c[idx_sorted[0]])
    top_vals = c[idx_sorted[:k_eff]]
    scaled = beta * (top_vals - c_top)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_v = np.exp(scaled)
    exp_sum = float(exp_v.sum())
    L = c_top + float(np.log(max(exp_sum, 1e-300))) / beta
    score = 2.0 * n * L / (S * S)
    w = exp_v / max(exp_sum, 1e-300)
    p = np.zeros(M)
    for i in range(k_eff):
        p[int(idx_sorted[i])] = w[i]
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    p_fft = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(rx_fft * p_fft, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def _pnorm_obj(x, p):
    """p-norm of positive part of convolution, with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    c = np.maximum(c, 0.0)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_safe = np.maximum(c, 1e-20)
    sp = float(np.sum(c_safe ** p))
    c_p = sp ** (1.0 / p)
    score = 2.0 * n * c_p / (S * S)
    w = c_safe ** (p - 1) * (sp ** (1.0 / p - 1.0))
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rx_fft = np.fft.rfft(rev_x, M2_fft)
    w_fft = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(rx_fft * w_fft, M2_fft)[n - 1:2 * n - 1]
    g_p = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_p / S2 - 2.0 * c_p / S3)
    return score, grad


def run_lbfgs(a0, obj_func, args, maxiter, maxfun):
    """Run L-BFGS-B with given objective function and arguments."""
    n = len(a0)
    a0 = make_nonneg_sum1(a0.copy())
    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            obj_func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.maximum(res.x, 0.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    a_out = make_nonneg_sum1(a_out)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a0, n_iters, top_k, lr_init, momentum_init):
    """Nesterov-accelerated subgradient on top-k convolution peaks."""
    n = len(a0)
    a = make_nonneg_sum1(a0.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    momentum = momentum_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        a_look = np.maximum(a - momentum * v, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        c = fast_conv(a_look)
        S = float(a_look.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a_look.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.ones(top_k) / top_k

        sg = np.zeros(n)
        for i in range(top_k):
            k = int(top_idx[i])
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a_look[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        grad_norm_sq = float(np.dot(grad, grad))
        lr = min(lr, max(sc - sc_lb, 1e-10) / (grad_norm_sq + 1e-10))
        lr = min(lr, 0.05)

        v = momentum * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        a_new = make_nonneg_sum1(a_new)
        a = a_new
        momentum = max(0.5, momentum * 0.99995)

    return best_a

    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))

    return inits

    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            est_per_iter = n / 3000.0 + 0.004
            maxiter = max(50, int(remaining * 0.6 / est_per_iter))
            maxfun = maxiter * 4
            for p_val in [20, 40]:
                if time.monotonic() > deadline - 10:
                    break
                sc, a_opt = run_lbfgs(a_cur, _pnorm_obj, (p_val,), maxiter, maxfun)
                if sc < best_score:
                    best_score = sc
                    best_a = a_opt.copy()
                    a_cur = a_opt.copy()
                    print(f"    P-norm p={p_val}: NEW BEST {sc:.8f}")

    # One more topk with k=2 (almost exact max)
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            est_per_iter = n / 2500.0 + 0.004
            maxiter = max(50, int(remaining * 0.7 / est_per_iter))
            maxfun = maxiter * 4
            sc, a_opt = run_lbfgs(a_cur, _topk_lse_obj, (10000, 2), maxiter, maxfun)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"    TopK k=2 b=10000: NEW BEST {sc:.8f}")

    # ==============================================================
    # Output
    # ==============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
