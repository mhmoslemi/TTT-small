This is a QP! And it's convex (minimize a convex function subject to linear constraints). We can solve it with scipy or CVXPY.

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a multi-phase optimization: LSE-smoothed gradient descent, exact subgradient
with multiplicative updates, and a novel "balanced descent" QP-based polish that
finds simultaneous descent directions for all active convolution peaks.
Key improvements over prior approach: full-position LSE (no top-K truncation),
balanced descent via KKT-based active-set QP, multiplicative mirror-descent
updates, and power-of-2 length for FFT efficiency.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _conv_gradient(a, w):
    """
    Compute grad[j] = sum_k w[k] * d(conv(a,a)[k])/d(a[j])
    = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices.
    Uses FFT for O(n log n) computation.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    M = 2 * n - 1
    if n == 0:
        return np.empty(0)
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # term1[j] = sum_k w[k] * a[k-j] = (w conv a)[j] shifted
    # This is cross-correlation: (w * a_reversed)
    A_rev = np.fft.rfft(a[::-1], N)
    cross1 = np.fft.irfft(W * A_rev, N)
    # term1[j] = cross1[n-1+j] for j=0..n-1
    term1 = cross1[n - 1: 2 * n - 1]
    # term2[j] = sum_k w[k] * a[j-k] = (a conv w)[j] shifted
    cross2 = np.fft.irfft(W * A, N)
    # term2[j] = cross2[n-1-j] for j=0..n-1... let me think
    # Actually: sum_k w[k]*a[j-k] for k=0..M-1, j=0..n-1
    # This is the convolution (a * w)[j] where we sum over k: a[j-k]*w[k]
    # = (a conv w)[j] but only for k where both are valid
    # In full conv: (a conv w)[j] = sum_i a[i]*w[j-i] = sum_k a[j-k]*w[k]
    # So term2[j] = (a conv w)[j]
    term2 = cross2[0:n]
    return term1 + term2


def _lse_obj_grad(a, beta):
    """
    Full-position LSE-smoothed objective: 2 * n * LSE(c) / S^2
    where LSE(c) = c_max + log(sum(exp(beta*(c - c_max))))/beta
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    M = 2 * n - 1
    c_max = float(c.max())
    if c_max <= 0 or not np.isfinite(c_max):
        return 1e10, np.zeros(n)

    z = beta * (c - c_max)
    e = np.exp(np.minimum(z, 0.0))
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(M) / M
        se = 1.0
    w = e / se
    L = c_max + np.log(se) / beta

    # Gradient of 2n*L/S^2 w.r.t. a
    # = 2n/S^2 * grad(L)
    # grad(L) = sum_k w[k] * grad_c[k]
    grad_L = _conv_gradient(a, w)

    # Also need to account for S in denominator:
    # d(2nL/S^2)/da = 2n*(grad_L * S^2 - L * 2*S * 1) / S^4
    # = 2n*grad_L/S^2 - 4n*L/(S^3)
    # Wait: d/d(a[j]) of (2n*L/S^2) = 2n*(dL/da[j])/S^2 + 2n*L*d(1/S^2)/da[j]
    # = 2n*grad_L[j]/S^2 + 2n*L*(-2/S^3)*(1/S)
    # = 2n*grad_L[j]/S^2 - 4n*L/S^4... no
    # d(1/S^2)/da[j] = -2/S^3 * dS/da[j] = -2/S^3
    # So: 2n*grad_L[j]/S^2 + 2n*L*(-2/S^3) = 2n*grad_L[j]/S^2 - 4nL/S^3

    # Actually let me just use: objective = 2n * L / S^2
    # grad = 2n * (grad_L * S^2 - L * 2S) / S^4 = 2n*grad_L/S^2 - 4nL/S^3

    val_obj = 2.0 * n * L / (S * S)
    grad = 2.0 * n * grad_L / (S * S) - 4.0 * n * L / (S * S * S)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val_obj), grad


def _run_lbfgs(a0, beta, time_limit, maxfun_cap=3000):
    """Run L-BFGS-B on full-position LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.005 * max(0.5, float(n) / 4096.0)
    mf = int(0.8 * time_limit / max(est_eval_t, 1e-4))
    mf = max(50, min(mf, maxfun_cap, 6000))
    maxiter = max(30, mf // 2)

    def obj(x):
        v, g = _lse_obj_grad(x, beta)
        Sx = float(x.sum())
        v += 1e-4 * (Sx - 1.0) ** 2
        g = g + 2e-4 * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=50, deadline=None):
    """Exact subgradient with backtracking on the max-convolution objective."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.5% of max
        threshold = dmax - 0.005 * max(dmax, 1e-10)
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _conv_gradient(a, w)
        LDS = float(w[active] @ d[active])
        grad = 2.0 * n * corr / (S * S) - 4.0 * n * LDS / (S * S * S)
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search
        best_step = None
        best_new_sc = best_sc
        step = 1.0 / gnorm
        for attempt in range(8):
            q = a - step * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                step *= 0.5
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_sc:
                best_new_sc = sc
                best_step = q.copy()
                break
            step *= 0.5

        if best_step is None:
            break
        best_sc = best_new_sc
        best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _mult_subgrad(a, max_iters=50, deadline=None):
    """Multiplicative (mirror descent) subgradient - maintains non-negativity naturally."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):

        c = _self_conv(a)

        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _conv_gradient(a, w)
        LDS = float(w[active] @ d[active])
        grad = 2.0 * n * corr / (S * S) - 4.0 * n * LDS / (S * S * S)
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Multiplicative update: a_new = a * exp(-eta * grad), then normalize
        best_step = None
        best_new_sc = best_sc
        for scale in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0):
            eta = scale / gnorm
            log_step = -eta * grad
            log_step = np.clip(log_step, -5.0, 5.0)  # numerical safety
            q = a * np.exp(log_step)
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        best_sc = best_new_sc
        best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _balanced_descent(a, max_iters=500, deadline=None):
    """
    Novel polish: find simultaneous descent directions for all active peaks
    using KKT-based QP. At each iteration, solve for the minimum-norm
    direction d that decreases all active convolution positions.
    """
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        cmax = float(c.max())
        M = len(c)

        # Active set: top-K positions (K adapts)
        K = min(50, M)
        idx = np.argpartition(c, M - K)[M - K:]
        # Refine: include all within 1% of max
        threshold = cmax - 0.01 * max(cmax, 1e-10)
        active_mask = c >= threshold
        if active_mask.sum() > 200:
            active_mask = np.zeros(M, dtype=bool)
            active_mask[idx] = True
        elif active_mask.sum() == 0:
            active_mask = np.zeros(M, dtype=bool)
            active_mask[np.argmax(c)] = True
        active_idx = np.where(active_mask)[0]
        m = len(active_idx)

        # Compute gradients for active positions
        # grad_k[j] = a[k-j] + a[j-k] for valid indices
        # We compute G (m x n) using FFT
        w = np.zeros(M)
        w[active_idx] = 1.0
        # But we need individual gradients, not just the sum
        # For the QP we need G (m x n), so let's compute each row
        # For efficiency, we compute all at once using the cross-correlation structure

        # G[i, j] = a[active_idx[i] - j] + a[j - active_idx[i]]
        # Let's compute this via the FFT cross-correlation for each active position
        # Actually, for m <= 200 and n <= 16384, direct computation might be OK
        # But let's use a smarter approach:
        # G[i, j] = a[k_i - j] (if 0 <= k_i - j < n) + a[j - k_i] (if 0 <= j - k_i < n)

        # Vectorized: for each j, G[:, j] = a[k_i - j] + a[j - k_i]
        # Let's compute a shifted version
        # For position k, the gradient w.r.t. a[j] is:
        # a[k-j] (valid when j in [k-n+1, k]) + a[j-k] (valid when j in [k, k+n-1])

        # Efficient computation using FFT:
        # cross_corr[k, j] = a[k - j] for valid j
        # This is the cross-correlation of a with a delta at position k

        # Alternative: just compute G directly using numpy broadcasting for small m
        if m * n < 50_000_000:
            # Direct computation (fast enough for m*n < 50M)
            ks = active_idx.astype(np.int64)  # shape (m,)
            js = np.arange(n)  # shape (n,)
            # ks[:, None] - js[None, :] gives (m, n) array of (k_i - j)
            diff1 = ks[:, None] - js[None, :]  # (m, n)
            diff2 = js[None, :] - ks[:, None]  # (m, n)
            G = np.zeros((m, n), dtype=np.float64)
            mask1 = (diff1 >= 0) & (diff1 < n)
            mask2 = (diff2 >= 0) & (diff2 < n)
            # For mask1: G[i,j] += a[diff1[i,j]]
            idx_flat1 = diff1[mask1]
            G[mask1] = a[idx_flat1]
            idx_flat2 = diff2[mask2]
            G[mask2] += a[idx_flat2]
        else:
            # Fallback: compute one row at a time
            G = np.zeros((m, n), dtype=np.float64)
            for ii in range(m):
                k = int(active_idx[ii])
                # a[k-j] valid for j in max(0,k-n+1)..min(k, n-1)
                j_start1 = max(0, k - n + 1)
                j_end1 = min(k, n - 1)
                if j_start1 <= j_end1:
                    G[ii, j_start1:j_end1 + 1] = a[k - j_end1:k - j_start1 + 1][::-1]
                # a[j-k] valid for j in max(k,0)..min(k+n-1, n-1)
                j_start2 = max(k, 0)
                j_end2 = min(k + n - 1, n - 1)
                if j_start2 <= j_end2:
                    G[ii, j_start2:j_end2 + 1] += a[j_start2 - k:j_end2 - k + 1]

        # Solve QP: min ||d||^2 s.t. G*d <= -1, sum(d) = 0
        # KKT: d = -G^T*lambda - mu*1
        # Active set method: start with all constraints active
        # G*d = -1, 1^T d = 0
        # => -G*G^T*lambda - mu*(G*1) = -1  =>  G*G^T*lambda + mu*(G*1) = 1

        G1 = G.sum(axis=1)  # G * ones(n), shape (m,)
        GGT = G @ G.T  # (m, m)

        A_sys = np.zeros((m + 1, m + 1))
        A_sys[:m, :m] = GGT
        A_sys[:m, m] = G1
        A_sys[m, :m] = G1
        A_sys[m, m] = float(n)
        b_sys = np.zeros(m + 1)
        b_sys[:m] = 1.0

        try:
            sol = np.linalg.solve(A_sys, b_sys)
            lam = sol[:m]
            mu = sol[m]
        except np.linalg.LinAlgError:

            try:
                sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
                lam = sol[:m]
                mu = sol[m]
            except Exception:

        if np.min(lam) < -1e-8:
            # Some constraints inactive - use only active ones
            # Simple approach: just use the gradient of the single max

                j_s = max(0, k - n + 1)
                j_e = min(k, n - 1)

                j_s2 = max(k, 0)
                j_e2 = min(k + n - 1, n - 1)

                    d = -grad_single / gnorm
                    # Project to sum=0

                    d = d / (np.linalg.norm(d) + 1e-15)
                else:

            else:

        t_max = min(t_max * 0.9, 1.0)  # safety margin
        if t_max < 1e-12:
            break

        # Try the step
        a_new = a + t_max * d
        a_new = np.clip(a_new, 0.0, None)
        a_new = _norm(a_new)
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in (0.5, 0.2, 0.1):
                a_try = _norm(np.clip(a + t_max * frac * d, 0.0, None))
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, try reducing active set
                # Remove the constraint with smallest lambda
                if m > 1:
                    min_lam_idx = int(np.argmin(lam))
                    new_active = np.delete(active_idx, min_lam_idx)
                    # Re-solve with smaller active set (just use the max position gradient)
                    max_pos = int(np.argmax(c))
                    k = max_pos
                    grad_single = np.zeros(n)
                    j_s = max(0, k - n + 1)
                    j_e = min(k, n - 1)
                    if j_s <= j_e:
                        grad_single[j_s:j_e + 1] = a[k - j_e:k - j_s + 1][::-1]
                    j_s2 = max(k, 0)
                    j_e2 = min(k + n - 1, n - 1)
                    if j_s2 <= j_e2:
                        grad_single[j_s2:j_e2 + 1] += a[j_s2 - k:j_e2 - k + 1]
                    gnorm = np.linalg.norm(grad_single)
                    if gnorm > 1e-14:
                        d2 = -grad_single / gnorm
                        d2 -= d2.mean()
                        d2n = np.linalg.norm(d2)
                        if d2n > 1e-14:
                            d2 = d2 / d2n
                            t2 = 0.1
                            a_try = _norm(np.clip(a + t2 * d2, 0.0, None))
                            sc_try = _score(a_try)
                            if sc_try < best_sc:
                                best_sc = sc_try
                                best_a = a_try.copy()
                                a = a_try
                                continue
                break

    return best_sc, best_a


def _spectral_init(n, rng, shape='flat'):
    """Generate non-negative sequence from a specified power spectrum shape."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    if shape == 'flat':
        power = np.ones(len(freqs))
        power[0] = 0.0
    elif shape == 'lowpass':
        cutoff = rng.uniform(0.3, 0.7)
        power = np.exp(-freqs ** 2 / (2 * (cutoff * Nf / 2) ** 2))
        power[0] = 0.0
    elif shape == 'bandpass':
        center = rng.uniform(0.2, 0.5)
        bandwidth = rng.uniform(0.1, 0.3)
        power = np.exp(-(freqs - center * Nf / 2) ** 2 / (2 * (bandwidth * Nf / 2) ** 2))
        power[0] = 0.0
    else:
        power = 1.0 / (1.0 + np.abs(freqs) ** rng.uniform(0.5, 1.5))
        power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _two_hump_init(n, rng):
    """Initialize with mass concentrated in two bumps."""
    center1 = int(rng.uniform(0.1, 0.4) * n)
    center2 = int(rng.uniform(0.6, 0.9) * n)
    width = int(rng.uniform(0.05, 0.15) * n)
    x = np.arange(n)
    a = np.exp(-(x - center1) ** 2 / (2 * width ** 2)) + np.exp(-(x - center2) ** 2 / (2 * width ** 2))
    a = a + 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0, None)
    return _norm(a)


def _hann_init(n, rng):
    """Hann window with random modulation."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    mod_freq = rng.uniform(1, 5)
    a *= (1 + 0.3 * np.sin(2 * np.pi * mod_freq * x / (n - 1)))
    a = np.clip(a, 0, None)
    return _norm(a)


def _optimize_candidate(a0, time_budget, label, rng):
    """Full optimization pipeline for a single candidate."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}, budget={time_budget:.0f}s")

    # Phase 1: Coarse LSE (low beta)
    t_phase1 = t0 + 0.15 * time_budget
    if time.monotonic() < t_phase1:
        sc, a = _run_lbfgs(a, beta=50, time_limit=t_phase1 - time.monotonic(), maxfun_cap=500)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"    [{label}] coarse LSE: {best_sc:.8f}")

    # Phase 2: Medium LSE
    t_phase2 = t0 + 0.35 * time_budget
    if time.monotonic() < t_phase2:
        sc, a = _run_lbfgs(a, beta=200, time_limit=t_phase2 - time.monotonic(), maxfun_cap=800)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"    [{label}] medium LSE: {best_sc:.8f}")

    # Phase 3: Fine LSE (high beta)
    t_phase3 = t0 + 0.55 * time_budget
    if time.monotonic() < t_phase3:
        sc, a = _run_lbfgs(a, beta=1000, time_limit=t_phase3 - time.monotonic(), maxfun_cap=1000)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"    [{label}] fine LSE: {best_sc:.8f}")

    # Phase 4: Subgradient polish
    t_phase4 = t0 + 0.70 * time_budget
    if time.monotonic() < t_phase4:
        sc, a_sg = _subgrad_polish(a, max_iters=40, deadline=t_phase4)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    # Phase 5: Multiplicative subgradient
    t_phase5 = t0 + 0.80 * time_budget
    if time.monotonic() < t_phase5:
        sc, a_ms = _mult_subgrad(a, max_iters=40, deadline=t_phase5)
        if sc < best_sc:
            best_sc = sc
            best_a = a_ms.copy()
            print(f"    [{label}] mult subgrad: {best_sc:.8f}")

    # Phase 6: Balanced descent polish
    t_phase6 = t0 + time_budget - 10.0
    if time.monotonic() < t_phase6:
        sc, a_bd = _balanced_descent(a, max_iters=300, deadline=t_phase6)
        if sc < best_sc:
            best_sc = sc
            best_a = a_bd.copy()
            print(f"    [{label}] balanced descent: {best_sc:.8f}")

    # Phase 7: Final subgradient
    if time.monotonic() < t0 + time_budget - 3.0:
        sc, a_final = _subgrad_polish(a, max_iters=20, deadline=t0 + time_budget - 3.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_final.copy()
            print(f"    [{label}] final subgrad: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization with LSE, subgradient, multiplicative
    mirror descent, and balanced descent QP polish.
    Uses power-of-2 length for FFT efficiency and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
                print(f"Loaded parent: length={len(parent)}, score={_score(parent):.8f}")
    except Exception as e:
        print(f"Warning: could not load parent: {e}")

    # Choose target length (power of 2 for FFT)
    if parent is not None:
        pparent = len(parent)
        # Use next power of 2 at or above parent length, but not too large
        if pparent <= 4096:
            target_n = 4096
        elif pparent <= 8192:
            target_n = 8192
        elif pparent <= 16384:
            target_n = 16384
        else:
            target_n = 16384
    else:
        target_n = 8192

    print(f"Target n = {target_n}")

    # Generate diverse candidates
    inits = []

    if parent is not None:
        p = _upsample(parent, target_n)
        inits.append(('parent', p))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))
        for amp in (0.05, 0.1, 0.2):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    # Spectral inits
    for shape in ('flat', 'lowpass', 'bandpass'):
        inits.append((f'spec_{shape}', _spectral_init(target_n, rng, shape)))

    # Structural inits
    for i in range(3):
        inits.append((f'twOHump{i}', _two_hump_init(target_n, rng)))
    for i in range(2):
        inits.append((f'hann{i}', _hann_init(target_n, rng)))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    top_scores = [f'{s:.6f}({n})' for s, n, _ in scored[:8]]
    print(f"Generated {len(scored)} inits. Top: {top_scores}")

    if not scored:
        scored = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = scored[0][0]
    best_seq = scored[0][2].copy()

    # Select top candidates for optimization (focus on best 2-3)
    n_candidates = 3
    candidates = scored[:n_candidates]

    # Allocate time
    time_used = time.monotonic() - t_start
    remaining = deadline - t_start - 60.0  # 60s for final polish
    remaining = max(remaining, 100.0)

    # Give more time to better candidates
    if len(candidates) == 3:
        fractions = [0.45, 0.30, 0.25]
    elif len(candidates) == 2:
        fractions = [0.60, 0.40]
    else:
        fractions = [1.0]

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        time_budget = fractions[ci] * remaining
        sub_deadline = t_start + time_used + sum(fractions[:ci + 1]) * remaining + 60.0
        sub_deadline = min(sub_deadline, deadline - 60.0)

        if time.monotonic() > sub_deadline - time_budget * 0.1:
            continue

        print(f"\n=== Optimizing candidate {ci+1}: '{name}', init_score={sc_init:.8f} ===")
        sc_traj, seq_traj = _optimize_candidate(arr_init, time_budget, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** GLOBAL BEST UPDATED: {best_score:.8f} ***")

    # === Final global polish on the best sequence ===
    print(f"\n=== Final global polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 15.0
    a_best = _norm(best_seq)

    # One more LSE at very high beta
    if time.monotonic() < final_deadline - 60:
        t_lse = min(45.0, final_deadline - time.monotonic() - 60)
        if t_lse > 10:
            sc_f, a_f = _run_lbfgs(a_best, beta=5000, time_limit=t_lse, maxfun_cap=500)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                a_best = a_f.copy()
                print(f"  Final high-beta LSE: {best_score:.8f}")

    # Balanced descent on the best
    if time.monotonic() < final_deadline - 30:
        t_bd = min(60.0, final_deadline - time.monotonic() - 30)
        if t_bd > 5:
            sc_bd, a_bd = _balanced_descent(a_best, max_iters=200, deadline=time.monotonic() + t_bd)
            if sc_bd < best_score:
                best_score = sc_bd
                best_seq = a_bd.copy()
                a_best = a_bd.copy()
                print(f"  Final balanced descent: {best_score:.8f}")

    # Subgradient
    if time.monotonic() < final_deadline - 10:
        sc_sg, a_sg = _subgrad_polish(a_best, max_iters=30, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Multiplicative subgradient
    if time.monotonic() < final_deadline - 5:
        sc_ms, a_ms = _mult_subgrad(a_best, max_iters=20, deadline=final_deadline - 5)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()
            print(f"  Final mult subgrad: {best_score:.8f}")

    # Random restarts if time permits
    if time.monotonic() < final_deadline - 30:
        for ri in range(4):
            if time.monotonic() > final_deadline - 30:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05)
            t_r = min(20.0, final_deadline - time.monotonic() - 30)
            if t_r < 5:
                break
            sc_r, a_r = _run_lbfgs(p, beta=2000, time_limit=t_r, maxfun_cap=200)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {ri}: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
