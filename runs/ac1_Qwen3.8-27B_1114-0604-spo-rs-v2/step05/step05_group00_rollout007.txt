For the uniform sequence (all 1/n), the convolution is triangular: c_k = (n-|k-(n-1)|)/n^2. The peak is at the center with value 1/n, and the score is 2.

To reduce the score below 2, we need to reduce the center peak. But we can't increase other peaks too much.

Actually, for a = uniform + small perturbation, the convolution is:
(a+e)*(a+e) = a*a + a*e + e*a + e*e

The cross terms a*e and e*a are approximately the "smeared" version of e (since a is roughly uniform). So for small e, the change in the convolution is approximately (1/n) * (smeared e) * 2 + e*e.

To reduce the center peak, we want e to have a "notch" at the center (negative contribution). But e must keep a >= 0, so e_j >= -a_j = -1/n at each point.

This suggests that the optimal perturbation is a small "dip" in the middle of the sequence, balanced by slight increases elsewhere.

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key innovations over previous approach:
1. Parametric shape exploration (low-dim Fourier/spline parameterization) to find good basins
2. Improved Fienup with adaptive anti-peak spectral targeting

4. Peak-equalization via top-K weighted subgradient with annealing
5. Diverse structured restarts (power transforms, windowing, sinusoidal modulation)
6. Length exploration (try different n values via interpolation)

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L = sum_k p_k * c_k with respect to x
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    """Top-level wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 0.5
        S0 = a0.sum()
    a0 = a0 / S0
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_adaptive(a, n_iters=400, gamma_init=0.5, gamma_end=3.0):
    """
    Fienup with adaptive anti-peak spectral targeting.
    The target spectrum emphasizes reducing power where it's already high.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        gamma = gamma_init + (gamma_end - gamma_init) * progress
        alpha = 0.3 + 0.5 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_safe = np.maximum(P, 1e-30)

        # Anti-peak target: reduce where P is high
        ratio = P_safe / P_mean
        P_target = P_mean / (1.0 + gamma * (ratio - 1.0) ** 2)
        P_target = np.maximum(P_target, 1e-30)

        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        A_new = A * (np.sqrt(P_blend / P_safe) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=300):
    """Fienup with quantile-based thresholding."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        q = 0.7 + 0.25 * (it / n_iters)
        thresh = np.quantile(P, q) * (1.0 + 0.5 * (1.0 - it / n_iters))

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave_adaptive(a, n_iters=5000, top_k=8, step_init=0.015, decay=0.9995):
    """
    Adaptive subgradient method targeting top-K peaks with cooling schedule.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    patience = 0
    max_patience = 2000

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        # Top-K peaks with softmax weighting
        top_count = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:top_count]

        # Softmax weights (emphasize the max more)
        temps = c[top_idx] / (M_val + 1e-30)
        exp_t = np.exp(10.0 * (temps - temps.max()))
        weights = exp_t / exp_t.sum()

        # Subgradient from weighted peaks
        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step with cooling
        progress = it / n_iters
        step_scale = 1.0 - 0.5 * progress
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30) * 0.5)
        step = min(step, 0.04 * step_scale)
        step = max(step, 1e-9)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step = max(step * decay, 1e-10)

    return best_a


def parametric_search(a_base, n_fourier=12, n_polynomial=6, rng=None):
    """
    Parametric optimization: represent a as base * (1 + sum of basis functions * params).
    Optimize the small number of parameters with L-BFGS.
    """
    n = len(a_base)
    x = np.arange(n, dtype=np.float64) / n
    if rng is None:
        rng = np.random.default_rng(42)

    # Build basis functions
    bases = []
    # Fourier (real)
    for k in range(1, n_fourier + 1):
        bases.append(np.cos(2.0 * np.pi * k * x))
        bases.append(np.sin(2.0 * np.pi * k * x))
    # Polynomial (Legendre-like)
    for k in range(1, n_polynomial + 1):
        bases.append((2.0 * x - 1.0) ** k)

    n_params = len(bases)
    B = np.array(bases)  # shape (n_params, n)

    def param_obj(w):
        pert = B @ w
        a_new = a_base * (1.0 + 0.3 * pert)
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            return 1e10
        a_new = a_new / S
        return compute_score(a_new)

    w0 = np.zeros(n_params)
    res = minimize(param_obj, w0, method='Nelder-Mead',
                   options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
    w_opt = res.x

    pert = B @ w_opt
    a_new = a_base * (1.0 + 0.3 * pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a_base.copy()
    a_new = a_new / S
    return a_new

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(new_n) / new_n
    a = a / S
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new / S_new
    return a_new


def hill_climb_local(a, n_steps=200, rng=None):
    """Simple random hill climbing with multiplicative perturbations."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for step in range(n_steps):
        # Perturb a random subset
        n_perturb = max(1, n // 50)
        idx = rng.integers(0, n, size=n_perturb)
        factors = 1.0 + 0.1 * rng.standard_normal(n_perturb)
        factors = np.maximum(factors, 0.1)
        a[idx] *= factors
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        update(sc, a_sm, "SINUSOIDAL")

    # Parametric search
    if time.monotonic() < phase1_end - 30:
        a_param = parametric_search(best_a, n_fourier=8, n_polynomial=4, rng=rng)
        sc = compute_score(a_param)
        update(sc, a_param, "PARAMETRIC")

    # Hill climbing from best
    if time.monotonic() < phase1_end - 10:
        a_hc = hill_climb_local(best_a, n_steps=300, rng=rng)
        sc = compute_score(a_hc)
        update(sc, a_hc, "HILL-CLIMB")

    # ==================================================================
    # Phase 2: L-BFGS multi-beta sweep (37% of time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 60.0)
    print(f"\n=== Phase 2: L-BFGS sweep (until t={phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas = [100, 300, 800, 2000, 5000, 10000, 20000, 40000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(20000, int(remaining * 25))
        maxfun = min(400000, int(remaining * 500))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # Second pass on best with different bets
    if time.monotonic() < phase2_end - 180:
        print("  L-BFGS second pass on best...")
        for beta in [3000, 10000, 30000]:
            if time.monotonic() > phase2_end - 60:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=15000, maxfun=300000)
            update(sc, a_opt, f"LBFGS-2 b={beta}")
            if sc < best_score:
                a_work = a_opt.copy()

    # ==================================================================
    # Phase 3: Peak shaving with adaptive cooling (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.70 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Peak shaving (until t={phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        n_iters = min(8000, int(remaining * 5))
        a_ps = peak_shave_adaptive(best_a, n_iters=n_iters, top_k=8,
                                    step_init=0.012, decay=0.9997)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")

        # L-BFGS polish after peak shave
        if time.monotonic() < phase3_end - 60:
            sc2, a2 = run_lbfgs(a_ps, 5000, maxiter=8000, maxfun=160000)
            update(sc2, a2, "PS+LBFGS")

    # Peak shave with different parameters
    if time.monotonic() < phase3_end - 60:
        a_ps2 = peak_shave_adaptive(best_a, n_iters=3000, top_k=4,
                                     step_init=0.02, decay=0.999)
        sc = compute_score(a_ps2)
        update(sc, a_ps2, "PEAK-SHAVE-4")

    # ==================================================================
    # Phase 4: Diverse multi-restart (20% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Multi-restart (until t={phase4_end - t_start:.0f}s) ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 30:
            break

        # Different perturbation strategies for each restart
        strategy = r % 6
        if strategy == 0:
            # Power transform + noise
            a_pert = best_a ** (1.0 + 0.1 * rng.random())
            a_pert = a_pert * (1.0 + 0.05 * rng.standard_normal(len(a_pert)))
        elif strategy == 1:
            # Gaussian window
            x = np.arange(len(best_a)) / len(best_a)
            sigma = 0.2 + 0.3 * rng.random()
            w = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
            a_pert = best_a * w
        elif strategy == 2:
            # Sinusoidal
            x = np.arange(len(best_a)) / len(best_a)
            freq = rng.integers(1, 8)
            phase = 2 * np.pi * rng.random()
            amp = 0.1 + 0.2 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Structured low-freq perturbation
            x = np.arange(len(best_a), dtype=np.float64) / len(best_a)
            pert = np.zeros(len(best_a))
            for m in range(1, 6):
                freq = rng.integers(1, 12)
                ph = 2.0 * np.pi * rng.uniform(0, 1)
                amp = 0.03 / np.sqrt(m)
                pert += amp * np.cos(2.0 * np.pi * freq * x + ph)
            a_pert = best_a * (1.0 + pert)
        elif strategy == 4:
            # Noise
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(len(best_a)))
        else:
            # "Shuffle-like": perturb blocks
            a_pert = best_a.copy()
            n_blocks = rng.integers(3, 10)
            block_size = len(a_pert) // n_blocks
            for b in range(n_blocks):
                start = b * block_size
                end = start + block_size
                factor = 1.0 + 0.15 * rng.standard_normal()
                a_pert[start:end] *= factor

        a_pert = np.maximum(a_pert, 0.0)
        S_pert = a_pert.sum()
        if S_pert < 1e-12:
            continue
        a_pert = a_pert / S_pert

        # Quick optimization from perturbed start
        remaining = phase4_end - time.monotonic() - 30
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 15))
        if maxiter < 500:
            break

        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"RESTART-{r} (strat={strategy})")

    # ==================================================================
    # Phase 5: Final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish ===")

    # High-beta L-BFGS
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = int(remaining * 20)
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_final, "FINAL-LBFGS-50000")

    # Very high beta for final squeeze
    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=100000)
        update(sc, a_final, "FINAL-LBFGS-100000")

    # Quick peak shave
    if time.monotonic() < deadline - 15:
        a_ps = peak_shave_adaptive(best_a, n_iters=2000, top_k=5,
                                    step_init=0.005, decay=0.999)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSH")

    # Final L-BFGS on the very best
    if time.monotonic() < deadline - 10:
        sc, a_final = run_lbfgs(best_a, 20000, maxiter=3000, maxfun=60000)
        update(sc, a_final, "FINAL-LBFGS-20000")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
