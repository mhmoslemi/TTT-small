def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize so sum = 1."""
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def adam_log_space(w0, p_schedule, lr0, n_iters, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer in log-space. w parameterizes a = exp(w) / sum(exp(w)).
    p_schedule: list of p values to use over the iterations.
    Returns best a found.
    """
    n = len(w0)
    w = w0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = normalize(np.exp(w - np.max(w)))
    best_score = compute_score(best_a)
    
    n_schedules = len(p_schedule)
    iters_per_schedule = n_iters // n_schedules
    
    for s in range(n_schedules):
        p = p_schedule[s]
        start_it = s * iters_per_schedule
        end_it = (s + 1) * iters_per_schedule if s < n_schedules - 1 else n_iters
        
        for it in range(start_it, end_it):
            a = normalize(np.exp(w - np.max(w)))
            sc, grad = power_norm_score_and_grad(a, p)
            
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            # Chain rule: d score / d w[j] = (d score / d a[j]) * a[j]
            grad_w = grad * a
            
            t = it + 1
            lr = lr0 / (1.0 + 0.01 * t)
            
            m = beta1 * m + (1 - beta1) * grad_w
            v = beta2 * v + (1 - beta2) * grad_w ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            w -= lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Keep w bounded
            w = np.clip(w, -20, 20)
    
    return best_a, best_score


def lbfgs_callback(x):
    """Callback for L-BFGS."""
    return lse_score_and_grad(x, _lse_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _lse_beta
    _lse_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-15, 1000.0).copy()
    S = a0.sum()
    if S < 1e-12:
        a0[:] = 0.5
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lbfgs_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def iterative_reweighting(a0, n_iters=500, target_ratio=None):
    """
    Iterative reweighting: reduce elements that contribute most to peak lags,
    increase elements that contribute to low lags.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.1
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_a * S_a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Target: flat = S_a^2 / (2n-1)
        target = S_a * S_a / (2 * n - 1)
        
        # For each a[j], compute weighted excess
        # a[j] contributes to conv[k] for k in [j, j+n-1] (clamped)
        # Excess contribution = sum over those k of (c[k] - target) * a[k-j] / S_a
        # Use FFT: excess_contrib[j] = sum_k (c[k] - target) * a[k-j] * I(0<=k-j<n)
        
        c_shifted = c - target
        a_flipped = a[::-1]
        M = len(c_shifted) + n - 1
        M_fft = 1 << (M - 1).bit_length()
        C = np.fft.rfft(c_shifted, M_fft)
        AF = np.fft.rfft(a_flipped, M_fft)
        conv_res = np.fft.irfft(C * AF, M_fft)[:M]
        excess = conv_res[n - 1:n - 1 + n]
        
        # Update: a[j] *= exp(-eta * excess[j] / c_max)
        scale = eta * excess / (c_max + 1e-10)
        scale = np.clip(scale, -5, 5)
        a = a * np.exp(-scale)
        a = np.maximum(a, 1e-15)
        a = normalize(a)
        
        eta *= 0.999
    
    return best_a, best_sc


def targeted_deflation(a0, n_steps=200, n_top_lags=10):
    """
    Targeted deflation: explicitly reduce the top contributors to peak lags
    and redistribute to non-peak lags.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for step in range(n_steps):
        c = fast_conv(a)
        S_a = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_a * S_a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top lag
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        
        # Reduce top contributors
        top_count = max(3, len(contrib) // 10)
        top_local = np.argsort(contrib)[::-1][:top_count]
        
        total_reduction = 0.0
        for li in top_local:
            reduce_frac = 0.05 + 0.05 * rng_refill[0]  # Will use deterministic
            a[j_start + li] *= (1.0 - reduce_frac)
            total_reduction += a[j_start + li] * reduce_frac
        
        # Redistribute to bottom contributors
        bottom_local = np.argsort(contrib)[:top_count]
        per = total_reduction / len(bottom_local)
        for li in bottom_local:
            a[j_start + li] += per
        
        a = np.maximum(a, 1e-15)
        a = normalize(a)
    
    return best_a, best_sc


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + 2.0 * scale * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + 2.0 * scale * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 1e-10, 1000.0)
    return normalize(a_p)


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Edge-enhanced (U-shaped)
    u = 0.5 * (2.0 * x - 1.0) ** 2 + 0.1
    inits.append(u / u.sum())
    
    # Strongly edge-enhanced
    u2 = (2.0 * x - 1.0) ** 6 + 0.01
    inits.append(u2 / u2.sum())
    
    # Pseudorandom non-negative
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    
    # Cosine-based
    inits.append(np.maximum(0.01, np.abs(np.cos(np.pi * x))))
    
    # Multi-peak
    n_peaks = 3
    peak_pos = rng.uniform(0, 1, n_peaks)
    widths = rng.uniform(0.02, 0.1, n_peaks)
    multi = np.zeros(n)
    for pp, ww in zip(peak_pos, widths):
        multi += np.exp(-((x - pp) ** 2) / (2 * ww ** 2))
    inits.append(multi + 0.01)
    
    # Whittaker-Shannon interpolation of random Fourier coefficients
    n_freqs = min(n, 50)
    freqs = rng.integers(1, n // 2, size=n_freqs)
    amps = rng.uniform(0.5, 2.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    fourier = 0.5 * np.ones(n)
    for f, a_f, p_f in zip(freqs, amps, phases):
        fourier += a_f * np.cos(2 * np.pi * f * x + p_f)
    inits.append(np.abs(fourier) + 0.01)
    
    return inits

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: Adam in log-space with power-norm -> L-BFGS refinement ->
    Iterative reweighting -> Targeted deflation -> ILS.
    """
    global _lse_beta, rng_refill
    
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0  # Safety margin
    
    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}, score: {compute_score(normalize(parent)):.10f}")
    else:
        n_work = 8192
        print(f"No parent, using n={n_work}")
    
    parent_norm = normalize(parent) if parent is not None else None
    if parent_norm is not None:
        report(compute_score(parent_norm), parent_norm, "parent")
    
    print(f"Working length: n={n_work}")
    
    # ==================================================================
    # Phase 1: Multi-start Adam in log-space (25% time)
    # ==================================================================
    phase1_deadline = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n--- Phase 1: Adam Log-Space Multi-Start ---")
    
    inits = build_initializations(n_work, rng, parent_norm)
    
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 1e-10, 1000.0)
        a_init = normalize(a_init)
        sc_init = compute_score(a_init)
        report(sc_init, a_init, f"init_{i}")
        
        # Log-space Adam with power norm schedule
        w0 = np.log(a_init + 1e-15)
        p_schedule = [10, 20, 50, 100]
        lr0 = 0.05
        n_iters = 3000
        a_out, sc_out = adam_log_space(w0, p_schedule, lr0, n_iters)
        report(sc_out, a_out, f"adam_init{i}")
    
    print(f"After Phase 1: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 2: Refined Adam with increasing p (40% time)
    # ==================================================================
    phase2_deadline = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n--- Phase 2: Refined Adam (increasing p) ---")
    
    w = np.log(np.maximum(best_a, 1e-15))
    
    # Schedule of (p, lr, n_iters)
    schedules = [
        (20, 0.03, 2000),
        (50, 0.02, 2000),
        (100, 0.015, 2000),
        (200, 0.01, 2000),
        (500, 0.008, 2000),
        (1000, 0.005, 2000),
        (3000, 0.003, 2000),
        (10000, 0.002, 2000),
    ]
    
    for si, (p, lr, n_it) in enumerate(schedules):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Adjust iterations based on time
        n_it_adj = min(n_it, max(500, int(remaining * 2)))
        
        a_out, sc_out = adam_log_space(w, [p], lr, n_it_adj)
        report(sc_out, a_out, f"adam_p{p}")
        
        # Update w from best
        w = np.log(np.maximum(best_a, 1e-15))
        
        # Occasionally run L-BFGS for a short burst
        if p >= 100 and remaining > 60:
            _lse_beta = float(p)
            maxiter = int(min(10000, max(1000, remaining * 5)))
            sc_lb, a_lb = run_lbfgs(best_a, float(p), maxiter=maxiter, maxfun=maxiter * 2)
            report(sc_lb, a_lb, f"lbfgs_b{int(p)}")
            w = np.log(np.maximum(best_a, 1e-15))
    
    print(f"After Phase 2: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 3: Iterative reweighting + targeted deflation (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n--- Phase 3: Reweighting + Deflation ---")
    
    # Iterative reweighting
    a_rw, sc_rw = iterative_reweighting(best_a, n_iters=300)
    report(sc_rw, a_rw, "reweight")
    
    # More Adam from reweighted
    w = np.log(np.maximum(best_a, 1e-15))
    a_out, sc_out = adam_log_space(w, [200, 500], 0.01, 3000)
    report(sc_out, a_out, "adam_after_rw")
    
    # L-BFGS polish
    if time.monotonic() < phase3_deadline - 30:
        _lse_beta = 1000.0
        sc_lb, a_lb = run_lbfgs(best_a, 1000.0, maxiter=20000, maxfun=50000)
        report(sc_lb, a_lb, "lbfgs_1000")
    
    # Another reweighting pass
    if time.monotonic() < phase3_deadline - 30:
        a_rw2, sc_rw2 = iterative_reweighting(best_a, n_iters=200)
        report(sc_rw2, a_rw2, "reweight2")
    
    print(f"After Phase 3: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 4: Iterated local search (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.95 * (total_budget - 30.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")
    
    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 60:
            break
        
        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)
        
        # Quick Adam refinement
        w = np.log(np.maximum(a_pert, 1e-15))
        p_val = 100 + int(400 * rng.random())
        n_it = int(min(2000, max(500, remaining * 1.5)))
        lr = 0.01 + 0.01 * rng.random()
        a_out, sc_out = adam_log_space(w, [p_val], lr, n_it)
        report(sc_out, a_out, f"ILS{trial}_p{p_val}")
        
        # L-BFGS if we have time
        if time.monotonic() < phase4_deadline - 40:
            _lse_beta = float(p_val * 3)
            sc_lb, a_lb = run_lbfgs(best_a, _lse_beta, maxiter=5000, maxfun=15000)
            report(sc_lb, a_lb, f"ILS{trial}_lbfgs")
    
    print(f"After Phase 4: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 5: Final high-precision refinement (5% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Refinement ---")
    
    # Very high p Adam
    w = np.log(np.maximum(best_a, 1e-15))
    a_out, sc_out = adam_log_space(w, [5000, 20000], 0.003, 3000)
    report(sc_out, a_out, "final_adam_high_p")
    
    # L-BFGS with very high beta
    for beta in [5000, 20000, 50000]:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(20000, max(2000, remaining * 3)))
        sc_lb, a_lb = run_lbfgs(best_a, float(beta), maxiter=maxiter, maxfun=maxiter * 2)
        report(sc_lb, a_lb, f"final_lbfgs_b{beta}")
    
    # One more reweighting
    if time.monotonic() < deadline - 20:
        a_rw3, sc_rw3 = iterative_reweighting(best_a, n_iters=200)
        report(sc_rw3, a_rw3, "final_reweight")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
