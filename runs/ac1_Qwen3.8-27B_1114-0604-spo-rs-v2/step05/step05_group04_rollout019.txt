        # For position i, its contribution to violation at lag k is f_norm[k-i] * g[i]

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses peak-directed Fienup, LP-projection approximation, and L-BFGS with LSE smoothing.
"""
import numpy as np
from scipy.optimize import minimize
import time

N_TARGET = 6962

def fast_conv(a, b=None):
    """FFT-based convolution. If b is None, computes a*a (autocorrelation)."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_peak_directed(a, n_iters=300, top_frac_init=0.1, top_frac_end=0.5,
                          peak_weight=2.0, soft_alpha=30.0):
    """
    Fienup iteration with peak-directed spectral compression.
    Instead of uniformly compressing top bins, weight the compression
    by how much each frequency contributes to the current spatial peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        A2 = A * A  # This is the spectrum of conv(a,a)
        P = np.abs(A2)  # Magnitude spectrum of autocorrelation

        # Find the peak position in spatial domain
        c = np.fft.irfft(A2, M_fft)[:2*n-1]
        peak_pos = int(np.argmax(c))
        peak_val = c[peak_pos]
        
        # Compute how much each frequency bin contributes to the peak
        # c[peak_pos] = (1/M_fft) * sum_k A2[k] * exp(2πi k * peak_pos / M_fft)
        # The contribution of bin k to the peak is Re(A2[k] * exp(2πi k * peak_pos / M_fft))
        phases = np.exp(-2j * np.pi * np.arange(len(A2)) * peak_pos / M_fft)
        contributions = np.real(A2 * phases)  # Contribution of each bin to the peak
        
        # We want to reduce bins with positive contribution (they add to the peak)
        # and can increase bins with negative contribution (they subtract from the peak)
        
        thresh = np.quantile(np.abs(A2), 1.0 - top_frac)
        above = np.abs(A2) > thresh
        
        scale = np.ones_like(A, dtype=np.complex128)
        if above.any():
            # For bins above threshold, compress them
            # Weight the compression by how much they contribute to the peak
            contrib_above = contributions[above]
            max_pos_contrib = max(contrib_above.max(), 1e-10)
            
            # Scale: reduce more for bins that contribute positively to the peak
            reduction = np.sqrt(thresh / np.abs(A2[above]))
            # Apply peak-directed weighting
            weight = 1.0 - peak_weight * 0.1 * (contrib_above / max_pos_contrib)
            weight = np.clip(weight, 0.3, 1.5)
            
            scale[above] = reduction * weight

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        
        # Non-negativity
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * np.clip(a_new, -50, 50)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def fienup_standard(a, n_iters=300, top_frac_init=0.1, top_frac_end=0.5, gamma=0.8):
    """Standard Fienup with power-law compression."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2

        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh

        scale = np.ones_like(P)
        if above.any():
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma

        A_mod = A * np.sqrt(scale)
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def fienup_spectral_eq(a, n_iters=300, alpha_init=0.2, alpha_end=0.8):
    """Spectral equalization: relax |A|^2 towards flat."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_mod = A * amp_ratio

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_score_grad(x, beta):
    """LSE-smoothed objective and its gradient."""
    n = len(x)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    A2 = A * A
    c = np.fft.irfft(A2, M_fft)[:2*n-1]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    # LSE approximation of max
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient computation
    # dL/dx[i] = sum_k p_k * d(c[k])/d(x[i])
    # where p_k = exp(beta*(c[k]-c_max)) / sum(exp(...))
    # d(c[k])/d(x[i]) = x[k-i] + x[i-k] (for valid indices) = 2*x[k-i] when only one is valid
    # More precisely: c[k] = sum_j x[j]*x[k-j]
    # d(c[k])/d(x[i]) = x[k-i] (if 0<=k-i<n) + x[i-k] (if 0<=i-k<n)
    
    p = exp_scaled / exp_sum
    
    # Vectorized gradient:
    # dL/dx[i] = sum_k p[k] * (x[k-i] + x[i-k]) for valid indices

```python
def lp_projection(a, n_iters=100, M=None):
    """Iteratively find b >= 0, sum(b)=sum(a) that minimizes max(conv(a,b))."""
    n = len(a)
    a = a.astype(np.float64)
    S = float(a.sum())
    if M is None:
        c_a = fast_conv(a)
        M = float(c_a.max())
    
    M_fft = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    
    # Initialize b = a
    b = a.copy()
    
    for it in range(n_iters):
        B = np.fft.rfft(b, M_fft)
        # Compute conv(a,b) = IDFT(A*B)
        h = np.fft.irfft(A * B, M_fft)[:2*n-1]
        # Clip h to M
        h_clipped = np.minimum(h, M)
        # Update B: B_new = DFT(h_clipped) / A
        H = np.fft.rfft(h_clipped, M_fft)
        # Avoid division by zero
        A-safe = np.where(np.abs(A) > 1e-10, A, 1e-10)
        B_new = H / A_safe
        b_new = np.fft.irfft(B_new, M_fft)[:n]
        b_new = np.maximum(b_new, 0.0)
        S_new = float(b_new.sum())
        if S_new < 1e-12:
            break
        b_new = b_new / S_new * S
        b = b_new
    
    return b
```
