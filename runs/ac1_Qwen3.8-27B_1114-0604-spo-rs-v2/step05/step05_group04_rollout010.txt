def fast_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)

    A = np.fft.rfft(a, M_fft)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - a_alpha) * P + a_alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / P)
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def topk_sg_descent(a, n_iters=3000, top_k=50, momentum=0.85):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        w = 1.0 / len(top_idx)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        gap = sc - sc_lb
        step = min(gap / (gn * gn + 1e-30), 0.04) / (1.0 + it * 0.0005)
        step = max(step, 1e-10)
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
    return best_a


def build_inits(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    n = 6962
    print(f"Target n={n}")

    if parent is not None:
        if len(parent) == n:
            print(f"Parent available (n={n}), score={compute_score(parent):.10f}")
            sc = compute_score(parent)
            update_best(sc, parent, "PARENT")
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            parent_n = np.interp(x_new, x_old, parent)
            sc = compute_score(parent_n)
            update_best(sc, parent_n, "PARENT-RESAMPLED")

    # Phase 1: Fienup initialization
    p1_end = t_start + 0.12 * (total_budget - 20.0)
    print(f"\n=== Phase 1: Fienup init (t<{p1_end - t_start:.0f}s) ===")

    a_work = best_a.copy() / best_a.sum()

    a_f = fienup_multi(a_work, n_iters=200, n_levels=6, base_frac=0.05)
    sc = compute_score(a_f)
    update_best(sc, a_f, "P1-FIENUP-MULTI")

    if time.monotonic() < p1_end:
        a_f2 = fienup_flat(a_work, n_iters=150, alpha=0.4)
        sc = compute_score(a_f2)
        update_best(sc, a_f2, "P1-FIENUP-FLAT")

    a_cur = best_a.copy() / best_a.sum()

    # Phase 2: LSE continuation with L-BFGS
    p2_end = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n=== Phase 2: LSE Continuation (t<{p2_end - t_start:.0f}s) ===")

    betas = [10, 30, 100, 300, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    for bi, beta in enumerate(betas):
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 30:
            break
        # Allocate time based on position in schedule
        frac_time = 0.8 if beta >= 5000 else 0.4
        phase_time = min(remaining * frac_time, 120.0)
        maxiter = max(500, int(phase_time * 500))
        maxfun = maxiter * 10

        print(f"  Beta={beta} (iters={maxiter}, time_budget={phase_time:.0f}s)")
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun, m=50)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            a_cur = a_opt.copy() / a_opt.sum()
        else:
            # Subgradient kick before next beta
            if time.monotonic() < p2_end and bi < len(betas) - 1:
                a_kick = topk_sg_descent(a_cur, n_iters=200, top_k=30, momentum=0.8)
                sc_k = compute_score(a_kick)
                if sc_k < best_score:
                    a_cur = a_kick.copy() / a_kick.sum()

    print(f"After Phase 2: best = {best_score:.10f}")

    # Phase 3: Subgradient refinement
    p3_end = t_start + 0.88 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Subgradient refinement (t<{p3_end - t_start:.0f}s) ===")

    for top_k in [20, 50, 100, 200]:
        if time.monotonic() > p3_end:
            break
        a_sg = topk_sg_descent(best_a, n_iters=5000, top_k=top_k, momentum=0.9)
        sc = compute_score(a_sg)
        improved = update_best(sc, a_sg, f"SG k={top_k}")
        if improved:
            a_cur = a_sg.copy() / a_sg.sum()

    print(f"After Phase 3: best = {best_score:.10f}")

    # Phase 4: Final L-BFGS polish at highest beta
    print(f"\n=== Phase 4: Final L-BFGS polish ===")
    a_cur = best_a.copy() / best_a.sum()
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - 10 - time.monotonic()
        if remaining < 20:
            break
        maxiter = max(2000, int(remaining * 300))
        maxfun = maxiter * 10
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun, m=70)
        improved = update_best(sc, a_opt, f"FINAL b={beta}")
        if improved:
            a_cur = a_opt.copy() / a_opt.sum()

    # Phase 5: Perturbation restarts
    print(f"\n=== Phase 5: Perturbation restarts ===")
    for trial in range(5):
        if time.monotonic() > deadline - 10:
            break
        noise_scale = 0.01 + 0.01 * trial
        a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S = a_p.sum()
        if S < 1e-12:
            continue
        a_p = a_p / S
        # Quick Fienup + L-BFGS
        a_r = fienup_multi(a_p, n_iters=50, n_levels=4, base_frac=0.08)
        sc = compute_score(a_r)
        if sc < best_score:
            a_r = a_r / a_r.sum()
            sc2, a_opt = run_lbfgs(a_r, 50000, maxiter=5000, maxfun=50000, m=50)
            update_best(sc2, a_opt, f"RESTART-{trial}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
