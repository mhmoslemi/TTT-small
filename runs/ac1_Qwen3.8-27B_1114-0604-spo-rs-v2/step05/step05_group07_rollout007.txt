= a[j] * dF/da[j] - a[j] * sum_i (dF/da[i]) * a[i]
= a[j] * (dF/da[j] - sum_i a[i] * dF/da[i])

So the gradient in x-space is: grad_x[j] = a[j] * (grad_a[j] - dot(a, grad_a))

This is the "natural gradient" for the exponential family parameterization. It's equivalent to the mirror descent gradient!

So mirror descent and the softmax-parameterized gradient descent are the same thing. The previous code already uses mirror descent, which is good.

The advantage of the x-parameterization is that we can use L-BFGS directly in x-space without needing to handle the non-negativity constraint (since exp is always positive). The only constraint is sum(a) = 1, which is automatically satisfied.

def lse_score_and_grad_x(x, beta):

    n = len(x)
    # Compute a from x (softmax-like)
    x_shifted = x - x.max()  # for numerical stability
    exp_x = np.exp(x_shifted)
    Z = exp_x.sum()
    a = exp_x / Z
    
    # Compute convolution and LSE
    c = fast_conv(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L  # since sum(a) = 1, S^2 = 1
    
    # Gradient of L w.r.t. a
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)

    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    grad_a = 2.0 * n * dL_da
    
    # Gradient w.r.t. x
    grad_x = a * (grad_a - np.dot(a, grad_a))
    
    return score, grad_x

"""
Optimized step function coefficient search using log-parameterized L-BFGS with
adaptive LSE smoothing, spectral initialization, and targeted peak shaving.
Key innovations: (1) optimize in log-space (softmax parameterization) to naturally
handle non-negativity, (2) aggressive beta schedule for high-precision max
approximation, (3) iterative peak deflation polish, (4) multi-scale exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def a_to_x(a, eps=1e-10):
    """Convert a-space to x-space (log parameterization)."""
    a = np.maximum(a, eps)
    return np.log(a)


def x_to_a(x):
    """Convert x-space to a-space (softmax normalization)."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    Z = exp_x.sum()
    a = exp_x / Z
    return a


def lse_score_grad_x(x, beta):
    """LSE-smoothed score and gradient w.r.t. x (log-parameterization, sum(a)=1)."""
    n = len(x)
    a = x_to_a(x)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -60))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L

    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    grad_a = 2.0 * n * dL_da
    grad_x = a * (grad_a - np.dot(a, grad_a))
    return score, grad_x


def run_lbfgs_x(x0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on LSE objective in x-space."""
    n = len(x0)

    def obj(x):
        return lse_score_grad_x(x, beta)

    res = minimize(
        obj, x0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = x_to_a(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, float(res.fun), res


def run_lbfgs_a(a0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on LSE objective in a-space with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-12:
        a0[:] = 1.0 / n
        return compute_score(a0), a0, np.inf, None
    a0 = a0 / S

    def obj(x):
        x = np.maximum(x, 1e-15)
        c = fast_conv(x)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(np.maximum(scaled, -60))
        log_sum = float(np.log(exp_scaled.sum()))
        L = c_max + log_sum / beta
        score = 2.0 * n * L / (S * S)

        p = exp_scaled / exp_scaled.sum()
        rev_a = x[::-1]
        M2 = len(p) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        P = np.fft.rfft(p, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
        dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
        return score, grad

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, float(res.fun), res


def mirror_descent_x(x0, beta, n_iters=5000, eta0=0.1, decay=0.999):
    """Mirror descent in x-space (equivalent to multiplicative updates in a-space)."""
    n = len(x0)
    x = x0.astype(np.float64).copy()
    a = x_to_a(x)
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_grad_x(x, beta)
        a_curr = x_to_a(x)
        exact_sc = compute_score(a_curr)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a_curr.copy()

        x = x - eta * grad
        eta *= decay

    return best_a


def peak_shaving(a, n_sweeps=50, top_frac=0.3):
    """Aggressive peak shaving via coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Find all peaks above 95% of max
        threshold = 0.95 * c[k_star]
        peak_indices = np.where(c >= threshold)[0]

        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            total_complement = n - len(contrib)
            if total_complement <= 0:
                continue

            top_count = max(1, int(len(contrib) * top_frac))
            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[len(contrib) - max(1, total_complement):]

            for frac in [0.7, 0.8, 0.9]:
                a_try = a.copy()
                reduction = 0.0
                for li in top_local:
                    old_val = a_try[j_start + li]
                    new_val = old_val * frac
                    a_try[j_start + li] = new_val
                    reduction += old_val - new_val
                if reduction > 1e-15:
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a_try[j_start + li] += per
                a_try = np.clip(a_try, 0.0, 1000.0)
                S_try = a_try.sum()
                if S_try > 1e-12:
                    a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()

    return best_a


def subgradient_polish(a, n_iters=5000, gamma=1.0):
    """Nesterov subgradient with adaptive step on exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = gamma * (sc - 1.0) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def perturbation(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 1.5 * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(2.0 * scale * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    # Hann window
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * x)))
    # Raised cosine variant
    inits.append(0.5 * (1.0 + np.cos(np.pi * (2 * x - 1))))
    # Gaussian
    inits.append(np.exp(-3.0 * (x - 0.5) ** 2))
    # Inverted (bathtub)
    inits.append(0.3 + 0.7 * (2.0 * x - 1.0) ** 2)
    # Two-peak
    inits.append(np.maximum(0.1, np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)))
    # Random
    inits.append(np.clip(rng.standard_normal(n), 0.01, 5.0))
    return inits

    # Mirror descent with high beta
    x_best = a_to_x(best_a)
    a_md = mirror_descent_x(x_best, beta=10000, n_iters=5000, eta0=0.03, decay=0.9995)
    sc = compute_score(a_md)
    report(sc, a_md, "mirror_hi_beta")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Final high-precision L-BFGS (10% time)
    # ============================================================
    print(f"\n--- Phase 5: Final High-Precision L-BFGS ---")

    x_best = a_to_x(best_a)
    for beta in [200000, 500000, 1000000]:
        if time.monotonic() > deadline - 10:
            break
        remaining_s = deadline - time.monotonic() - 5
        if remaining_s < 10:
            break
        maxiter = int(min(50000, max(1000, remaining_s * 40)))
        maxfun = int(min(150000, max(3000, remaining_s * 150)))
        sc, a_opt, _, _ = run_lbfgs_x(x_best, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final b={beta}")
        x_best = a_to_x(a_opt)

    # Final peak shave
    a_final = peak_shaving(best_a, n_sweeps=20, top_frac=0.25)
    sc = compute_score(a_final)
    report(sc, a_final, "final_shave")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
