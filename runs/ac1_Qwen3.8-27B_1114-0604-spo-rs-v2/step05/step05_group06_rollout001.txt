With sum(a) = 1 and a >= 0:
- The feasible set is the standard simplex
- The objective is `2*n*max_k (a*a)[k]`
- The maximum possible value of (a*a)[k] is 1 (when a is a delta, but that gives (a*a)[0] = 1 and the score = 2n)
- We want to minimize this

The subgradient of max_k (a*a)[k] at a point where the max is achieved at positions in set S is:

where w_k are any weights summing to 1 (for the subdifferential of max).

Using the uniform weights over the active set gives the "most conservative" subgradient.

a_i <- a_i * exp(-eta * grad_i) / Z

where Z = sum_j a_j * exp(-eta * grad_j) normalizes.

This is equivalent to the "exponential weighting" or "Hedge" algorithm. It naturally keeps a_i > 0.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. LSE-smoothed optimization with L-BFGS-B (progressive K growth)
2. Multiplicative weight (mirror descent) polishing
3. Exact subgradient descent with adaptive line search
Key improvements: adaptive active set, multiplicative updates, longer polishing.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed max(conv) objective with top-k active set."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 50}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _multiplicative_weight_polish(a, max_iters=500, deadline=None):
    """
    Mirror descent with entropy (multiplicative weight updates).
    Updates: a_i <- a_i * exp(-eta * grad_i) / Z
    """
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter {it}: score={best_sc:.8f}, elapsed={elapsed:.1f}s")
        
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        
        # Active set: positions within 0.5% of max
        active = d >= dmax - 0.005 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        # Try several step sizes with multiplicative update
        best_new_sc = best_sc
        best_new_a = None
        for eta in (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2):
            updated = a * np.exp(-eta * grad)
            updated = np.clip(updated, 1e-15, None)
            updated = updated / updated.sum()
            sc = _score(updated)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = updated.copy()
        
        if best_new_a is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
        a = best_new_a
    
    return best_sc, best_a


def _subgrad_polish(a, max_iters=100, deadline=None):
    """Exact subgradient descent with adaptive step size."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        
        # Active set: positions within 0.1% of max (tight for exact subgradient)
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        best_step = None
        best_new_sc = best_sc
        for scale in (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3):

            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    
    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _structured_init(n, rng):
    """Create a structured initialization based on low-autocorrelation sequence ideas."""
    #趨向 flat spectrum: use a windowed version of a pseudorandom binary sequence
    # mapped to non-negative values
    x = np.linspace(0, 1, n)
    # Base: gentle cosine modulation
    base = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * x)
    # Add structured noise
    k = rng.integers(3, 15, size=n)
    angles = k * 2 * np.pi / 16
    structured = np.cos(angles)
    # Smooth it
    from scipy.ndimage import gaussian_filter1d
    structured = gaussian_filter1d(structured, sigma=n/50.0)
    a = base + 0.2 * structured
    a = np.clip(a, 0.01, None)
    return a / a.sum()
