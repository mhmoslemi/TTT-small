"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def _cross_corr(a, w):
    """Compute corr[m] = sum_i a[i]*w[m+i] for m in [0, n-1]."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _thue_morse(n, rng):
    """Generate a Thue-Morse inspired non-negative sequence of length n."""
    # Thue-Morse: t(k) = popcount(k) mod 2
    # We create a smooth non-negative version
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=np.float64)
    # Smooth it
    k = max(5, n // 50)
    if k % 2 == 0:
        k += 1
    sig = max(1.0, k / 3.0)
    kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    smooth = np.convolve(tm, kern, mode='same')
    # Make non-negative and add some random perturbation
    smooth = smooth - smooth.min() + 0.1
    smooth = smooth + 0.05 * rng.standard_normal(n)
    smooth = np.clip(smooth, 0.0, None)
    return smooth / smooth.sum()

def _thue_morse_freq(n, rng):

    # Generate Thue-Morse

def _spectral_init(n, rng, exponent=None):
    """Generate sequence with shaped power spectrum."""
    if exponent is None:
        exponent = rng.uniform(0.5, 2.0)
    Nf = _next_pow2(2 * n - 1)
    freqs = np.abs(np.fft.rfftfreq(Nf) * Nf)
    freqs[0] = 1.0  # avoid division by zero
    power = 1.0 / (1.0 + freqs) ** exponent
    power[0] = 0.0  # remove DC
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig

def _mirror_descent_step(a, grad, eta):
    """Exponential family (mirror) descent step with renormalization."""
    log_a = np.full_like(a, -50.0)  # log of very small number
    mask = a > 1e-15
    log_a[mask] = np.log(a[mask])
    new_log = log_a - eta * grad
    new_log -= new_log.max()  # for numerical stability
    new_a = np.exp(new_log)
    S = new_a.sum()
    if S < 1e-15:
        return a.copy()
    return new_a / S

def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=500):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3
    
    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out
