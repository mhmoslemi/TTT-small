"""
Optimized step-function search: minimizes 2n*max(conv(a,a))/(sum(a))^2
using Fienup + adaptive LSE-LBFGS + Nesterov subgradient.
Single-length focus (n=6962) with multi-phase time allocation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based circular-free convolution of a with itself."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective and its gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # Shift for numerical stability
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. x
    p = exp_scaled / exp_sum  # softmax weights on conv values
    
    # dL/dx[j] = 2 * sum_m x[m] * p[m+j] for j=0,...,n-1
    # = 2 * conv(rev_x, p)[n-1 : 2n-1]
    rev_x = x[::-1]
    M2 = 3 * n - 2  # need at least n + (2n-1) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, maxls=30):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = 0.5 * n
    a0 = a0 / S  # normalize to sum=1
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': maxls}
    )
    a_out = np.clip(res.x, 0.0, 1.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_compress(a, n_iters=300, gamma=0.85, top_frac=0.2, soft_alpha=30.0):
    """Fienup spectral compression with soft non-negativity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def nesterov_subgradient(a0, n_iters=5000, top_m=20, step_scale=0.01, decay=0.99999):

    Nesterov-accelerated projected subgradient on the true (non-smooth) objective.
    Uses top-m subdtype for a better subgradient approximation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    beta_nesterov = 1.0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Initialize step size
    c = fast_conv(y)
    M_val = float(c.max())
    S_y = float(y.sum())
    sc_init = 2.0 * n * M_val / (S_y * S_y)
    step = step_scale / (beta_nesterov + 1)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        # Compute subgradient at y
        c = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M_val = float(c.max())
        
        # Top-m indices
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient: average of gradients at top-m points
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                seg = y[j_lo:j_hi + 1]
                sg[j_lo:j_hi + 1] += 2.0 * seg[::-1]
        sg /= len(top_idx)
        
        # Full gradient of F = 2n*M/S^2
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_nesterov * beta_nesterov)) / 2.0
        momentum_new = (beta_nesterov - 1.0) / t_new
        beta_nesterov = t_new
        
        # Gradient step
        y_new = y - step * grad
        
        # Project onto {x >= 0, sum(x) = 1}
        y_new = np.maximum(y_new, 0.0)
        S_new = float(y_new.sum())
        if S_new < 1e-12:
            y_new = y.copy()
            S_new = S_y
        else:
            y_new = y_new / S_new
        
        # Nesterov extrapolation for next y
        a_next = y_new + momentum_new * (y_new - y)
        a_next = np.maximum(a_next, 0.0)
        S_next = float(a_next.sum())
        if S_next < 1e-12:
            a_next = y_new.copy()
        else:
            a_next = a_next / S_next
        
        # Update
        y = y_new
        a = a_next
        
        # Update step size (diminishing)
        step *= decay
        
        # Track best
        sc = compute_score(y)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
    
    return best_a


def top_m_subgradient_plain(a0, n_iters=3000, top_m=20, step_init=0.005, decay=0.9999):
    """Plain projected subgradient (no Nesterov) for stability."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        if S_a < 1e-12:
            break
        M_val = float(c.max())
        
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                seg = a[j_lo:j_hi + 1]
                sg[j_lo:j_hi + 1] += 2.0 * seg[::-1]
        sg /= len(top_idx)
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        step *= decay
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Optimized step-function sequence search minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Fienup warm-up -> Adaptive LSE-LBFGS (multi-beta) -> Nesterov subgradient polishing.
Focused on n=6962 with careful time management.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based linear autocorrelation of a with itself."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective and gradient for L-BFGS."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # dL/dx[j] = 2 * sum_m x[m] * p[m+j] = 2*conv(rev_x, p)[n-1:2n-1]
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, maxls=30):
    """Run L-BFGS-B on the LSE-smoothed objective with normalization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S

    bounds = [(1e-10, 1.0)] * n
    try:
        res = minimize(
            lse_objective_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': maxls}
        )
        a_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        a_out = a0.copy()

    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
        S_out = S
    a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_compress(a, n_iters=300, gamma=0.85, top_frac=0.15):
    """Fienup spectral compression: reduce peak of |FFT|^2."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat_spec(a, n_iters=300, alpha=0.4):
    """Relax amplitude spectrum towards flat."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)
        amp_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_subgrad(a0, n_iters=5000, top_m=30, step_scale=0.005, decay=0.99999):
    """Nesterov-accelerated projected subgradient on the true objective."""
    n = len(a0)
    y = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(y.sum())
    if S < 1e-12:
        return np.ones(n) / n
    y = y / S

    alpha = y.copy()
    t_k = 1.0
    best_a = y.copy()
    best_sc = compute_score(y)
    step = step_scale

    for it in range(n_iters):
        c = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M_val = float(c.max())

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                seg = y[j_lo:j_hi + 1]
                sg[j_lo:j_hi + 1] += 2.0 * seg[::-1]
        sg /= float(len(top_idx))

        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Nesterov
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        t_k = t_new

        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        S_new = float(y_new.sum())
        if S_new < 1e-12:
            y_new = y.copy()
        else:
            y_new = y_new / S_new

        alpha_new = y_new + momentum * (y_new - y)
        alpha_new = np.maximum(alpha_new, 0.0)
        S_alpha = float(alpha_new.sum())
        if S_alpha < 1e-12:
            alpha_new = y_new.copy()
        else:
            alpha_new = alpha_new / S_alpha

        y = y_new
        alpha = alpha_new
        step *= decay

        sc = compute_score(y)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

    return best_a


def plain_subgrad(a0, n_iters=3000, top_m=50, step_init=0.003, decay=0.99995):
    """Plain projected subgradient method."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        if S_a < 1e-12:
            break
        M_val = float(c.max())

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                seg = a[j_lo:j_hi + 1]
                sg[j_lo:j_hi + 1] += 2.0 * seg[::-1]
        sg /= float(len(top_idx))

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        step *= decay

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_adaptive(a, n_iters=500, gamma_init=0.7, gamma_end=1.0,
                    tf_init=0.1, tf_end=0.5):
    """Fienup with adaptive gamma and top-fraction via cosine schedule."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        gamma = gamma_init + (gamma_end - gamma_init) * 0.5 * (1 - np.cos(np.pi * progress))
        top_frac = tf_init + (tf_end - tf_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Fienup + adaptive LSE-LBFGS + Nesterov subgradient.
    Returns best sequence found within time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 90.0

    n_target = 6962

    best_score = np.inf
    best_a = np.ones(n_target) / n_target

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0).copy()
    except Exception:
        pass

    if parent is not None:
        if len(parent) != n_target:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_target)
            parent = np.interp(x_new, x_old, parent).astype(np.float64)
        parent = np.maximum(parent, 0.0)
        S_p = parent.sum()
        if S_p < 1e-12:
            parent = np.ones(n_target) / n_target
        else:
            parent = parent / S_p
        sc_p = compute_score(parent)
        if sc_p < best_score:
            best_score = sc_p
            best_a = parent.copy()
        print(f"Parent score: {sc_p:.8f} (n={n_target})")
    else:
        print("No parent available, using uniform init.")

    print(f"Initial best: {best_score:.8f}")
    print(f"Deadline: t={deadline - t_start:.0f}s")

    # ==========================================================
    # Phase 1: Fienup warm-up (90s)
    # ==========================================================
    phase1_end = t_start + 90.0
    print(f"\n{'='*50}")
    print(f"Phase 1: Fienup warm-up")
    print(f"{'='*50}")

    # Adaptive Fienup from parent
    a_f1 = fienup_adaptive(best_a, n_iters=400, gamma_init=0.7, gamma_end=1.0,
                           tf_init=0.08, tf_end=0.5)
    sc = compute_score(a_f1)
    if sc < best_score:
        best_score = sc
        best_a = a_f1.copy()
        print(f"  [F1-ADAPT] NEW BEST: {sc:.8f}")

    # Flat spectrum relaxation
    a_f2 = fienup_flat_spec(best_a, n_iters=200, alpha=0.35)
    sc = compute_score(a_f2)
    if sc < best_score:
        best_score = sc
        best_a = a_f2.copy()
        print(f"  [F1-FLAT] NEW BEST: {sc:.8f}")

    # From a perturbed version
    a_pert = best_a * (1.0 + 0.08 * rng.standard_normal(n_target))
    a_pert = np.clip(a_pert, 0.0, None)
    S_pert = a_pert.sum()
    if S_pert > 1e-12:
        a_pert = a_pert / S_pert
        a_f3 = fienup_adaptive(a_pert, n_iters=300, gamma_init=0.6, gamma_end=1.0,
                               tf_init=0.12, tf_end=0.55)
        sc = compute_score(a_f3)
        if sc < best_score:
            best_score = sc
            best_a = a_f3.copy()
            print(f"  [F1-PERT] NEW BEST: {sc:.8f}")

    print(f"After Phase 1: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ==========================================================
    # Phase 2: L-BFGS with adaptive LSE (1500s)
    # ==========================================================
    phase2_end = t_start + 1590.0
    print(f"\n{'='*50}")
    print(f"Phase 2: L-BFGS adaptive LSE")
    print(f"{'='*50}")

    beta_schedule = [
        (100, 150),
        (300, 200),
        (800, 250),
        (2000, 300),
        (5000, 350),
        (15000, 250),
    ]

    cur_a = best_a.copy()
    for beta, t_alloc in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        t_alloc = min(t_alloc, remaining)

        maxiter = min(30000, max(500, int(t_alloc * 50)))
        maxfun = maxiter * 30
        maxls = min(50, max(15, maxfun // max(1, maxiter)))

        print(f"  beta={beta}, alloc={t_alloc:.0f}s, maxiter={maxiter}")
        t0 = time.monotonic()
        cur_a = np.clip(cur_a, 0.0, 1.0)
        sc_cur = cur_a.sum()
        if sc_cur < 1e-6:
            cur_a = best_a.copy()
        cur_a = cur_a / cur_a.sum()

        sc, a_opt = run_lbfgs(cur_a, beta, maxiter=maxiter, maxfun=maxfun, maxls=maxls)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            cur_a = a_opt.copy()
            print(f"  [LBFGS b={beta}] NEW BEST: {sc:.8f} ({elapsed:.0f}s)")
        else:
            print(f"  [LBFGS b={beta}] score={sc:.8f} (no improvement, {elapsed:.0f}s)")

    print(f"After Phase 2: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ==========================================================
    # Phase 3: Subgradient refinement (400s)
    # ==========================================================
    phase3_end = min(deadline, t_start + 1990.0)
    print(f"\n{'='*50}")
    print(f"Phase 3: Subgradient refinement")
    print(f"{'='*50}")

    # Nesterov with various top_m
    for top_m in [15, 40, 80]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        n_iters = int(min(remaining * 60, 30000))
        t0 = time.monotonic()
        a_sg = nesterov_subgrad(best_a, n_iters=n_iters, top_m=top_m,
                                step_scale=0.004, decay=0.99999)
        sc = compute_score(a_sg)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()
            print(f"  [NEST m={top_m}] NEW BEST: {sc:.8f} ({elapsed:.0f}s)")
        else:
            print(f"  [NEST m={top_m}] score={sc:.8f} ({elapsed:.0f}s)")

    # Plain subgradient
    if time.monotonic() < phase3_end - 30:
        remaining = phase3_end - time.monotonic()
        n_iters = int(min(remaining * 60, 20000))
        t0 = time.monotonic()
        a_sg = plain_subgrad(best_a, n_iters=n_iters, top_m=100,
                             step_init=0.002, decay=0.99997)
        sc = compute_score(a_sg)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()
            print(f"  [PLAIN-SG] NEW BEST: {sc:.8f} ({elapsed:.0f}s)")
        else:
            print(f"  [PLAIN-SG] score={sc:.8f} ({elapsed:.0f}s}")

    print(f"After Phase 3: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ==========================================================
    # Phase 4: Final polish (remaining time)
    # ==========================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Final polish")
    print(f"{'='*50}")

    # Quick Fienup polish
    if time.monotonic() < deadline - 60:
        a_pf = fienup_adaptive(best_a, n_iters=200, gamma_init=0.85, gamma_end=1.0,
                               tf_init=0.08, tf_end=0.35)
        sc = compute_score(a_pf)
        if sc < best_score:
            best_score = sc
            best_a = a_pf.copy()
            print(f"  [PF-FIENUP] NEW BEST: {sc:.8f}")

    # High-beta L-BFGS final
    if time.monotonic() < deadline - 90:
        remaining = deadline - time.monotonic() - 60
        maxiter = min(20000, max(1000, int(remaining * 20)))
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=maxiter,
                              maxfun=maxiter * 25, maxls=30)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [PF-LBFGS b=20000] NEW BEST: {sc:.8f}")

    # One more Nesterov pass
    if time.monotonic() < deadline - 30:
        a_sg = nesterov_subgrad(best_a, n_iters=2000, top_m=50,
                                step_scale=0.002, decay=0.99999)
        sc = compute_score(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg.copy()
            print(f"  [PF-NEST] NEW BEST: {sc:.8f}")

    # ==========================================================
    # Output
    # ==========================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    total_sum = sum(out)
    if total_sum < 0.01:
        out = [0.5] * n_target

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
