    # Use golden section search

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set LSE optimization with optimal-step subgradient polishing,
peak deflation, and multilevel length scaling.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Precompute power-of-2 helper
def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def _max_conv(a):
    c = _self_conv(a)
    return float(c.max())

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))

def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad

def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out

            else:
                lo = m1
        
        t_star = (lo + hi) / 2
        q = a - t_star * grad
        q = np.clip(q, 0.0, None)
        Sq = q.sum()
        if Sq < 1e-8:
            break
        q = q / Sq
        sc = _score(q)
        
        if sc < best_sc:
            best_sc = sc
            best_a = q.copy()
            if (it % 10 == 0):
                print(f"    subgrad_opt: iter={it}, score={best_sc:.8f}, t*={t_star:.6f}")
        a = q
    
    return best_sc, best_a

def _subgrad_polish(a, max_iters=40, deadline=None):
    """Subgradient descent with multiple step sizes and adaptive selection."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        best_step = None
        best_new_sc = best_sc
        for scale in (0.0001, 0.0005, 0.002, 0.008, 0.02, 0.08, 0.2, 0.5, 1.0, 2.0):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()
        
        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step
    
    return best_sc, best_a

def _peak_deflation(a, rng, n_peaks=5, strength=0.05):
    """Apply targeted perturbation to reduce the top peaks in the convolution."""
    a = _norm(a)
    n = len(a)
    c = _self_conv(a)
    d = c  # just use the raw convolution values for ranking
    
    # Get top peak positions
    top_idx = np.argsort(d)[-n_peaks:]
    
    # For each peak position j, the gradient direction to reduce it is -(a[j-*] + a[*-j])
    # Combined direction:
    direction = np.zeros(n)
    for j in top_idx:
        for i in range(n):
            j_minus_i = j - i
            i_minus_j = i - j
            if 0 <= j_minus_i < n:
                direction[i] -= a[j_minus_i]
            if 0 <= i_minus_j < n:
                direction[i] -= a[i_minus_j]
    
    # Normalize direction
    dnorm = np.linalg.norm(direction)
    if dnorm < 1e-12:
        return a
    
    # Apply with random sign flip for diversity (sometimes pushing "up" helps)
    q = a + strength * direction / dnorm
    q = np.clip(q, 0.0, None)
    return _norm(q)

def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig

    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
