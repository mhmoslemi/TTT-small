def solve_lp_direction(a, M, top_k=50):
    """Solve LP to find direction g that maximizes sum(g) subject to peak constraints."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A (top_k x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((top_k, n))
    for i, k in enumerate(top_idx):
        k = int(k)
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A[i, j] = a[k - j]
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # linprog minimizes, so: min -sum(g) s.t. A @ g <= M, g >= 0
    from scipy.optimize import linprog
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(top_k, M), bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None

"""
LP-guided multi-strategy optimizer for minimizing 2n*max(conv(a,a))/S^2.
Combines: (1) LP-based direction finding with adaptive constraint sets,
(2) L-BFGS on LSE-smoothed objective with multi-beta sweep,
(3) peak-targeted coordinate descent with mass-transfer moves,
(4) Fienup spectral initialization. Time-managed to fit 2400s budget.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_func(x, beta):
    return lse_obj_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 0.5
        S0 = a0.sum()
    a0 = a0 / S0
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_func, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def solve_lp_direction(a, M, top_idx):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for k in top_idx, g >= 0.
    Returns g or None if infeasible.
    """
    n = len(a)
    K = len(top_idx)
    if K == 0:
        return None

    # Build constraint matrix A (K x n): A[i, j] = a[k_i - j] if 0 <= k_i - j < n
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A[i, j_start:j_end + 1] = a[k - j_start:k - j_end + 1][::-1]

    # LP: min -sum(g) s.t. A @ g <= M, g >= 0
    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(K, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 30}
        )
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_direction_optimize(a, n_iters=5000, max_top_k=100, time_limit=60.0):
    """
    Iterative LP-based optimization with adaptive constraint set.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    no_improve_count = 0

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        S = a.sum()

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 200:
                break

        # Adaptive top-K: more constraints when close to optimum
        k_frac = min(1.0, 0.02 + 0.03 * (it / max(1, n_iters)))
        K = max(10, min(max_top_k, int(k_frac * len(c))))
        top_idx = np.argsort(c)[::-1][:K]

        # Solve LP for direction
        g = solve_lp_direction(a, M, top_idx)
        if g is None:
            # LP failed, try subgradient step
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg = np.zeros(n)
            sg[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
            a -= 0.01 * sg / (S * S)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
            continue

        # Rescale g to match sum of a
        g_sum = g.sum()
        if g_sum < 1e-12:
            continue
        g = g * (S / g_sum)

        # Line search along blend direction
        best_t = 0.0
        best_blend_sc = sc
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_blend = (1 - t) * a + t * g
            a_blend = np.maximum(a_blend, 0.0)
            S_blend = a_blend.sum()
            if S_blend < 1e-12:
                continue
            a_blend = a_blend / S_blend
            c_blend = fast_conv(a_blend)
            M_blend = float(c_blend.max())
            sc_blend = 2.0 * n * M_blend / (S_blend * S_blend)
            if sc_blend < best_blend_sc:
                best_blend_sc = sc_blend
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def peak_shave_advanced(a, n_iters=3000, time_limit=60.0):
    """
    Advanced peak shaving with mass transfer moves.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end < j_start:
            break

        # Contributions to peak
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Find top contributors
        n_top = min(50, len(contribs))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_global = j_start + top_local

        # Try mass transfer: reduce a[j] and add to a[j']
        moved = False
        for j in top_global:
            if a[j] < 1e-10:
                continue
            # Reduce a[j] by small fraction
            delta = a[j] * 0.1
            a[j] -= delta
            S_tmp = a.sum()

            # Try adding to various positions
            # Choose positions that are "far" from the peak in a useful way
            candidates = np.random.default_rng(it + j).choice(n, size=min(20, n), replace=False)
            for jp in candidates:
                a[jp] += delta
                S2 = a.sum()
                if S2 < 1e-12:
                    a[jp] -= delta
                    continue
                a_scaled = a / S2
                c_new = fast_conv(a_scaled)
                M_new = float(c_new.max())
                sc_new = 2.0 * n * M_new / (S2 * S2)
                if sc_new < sc - 1e-12:
                    a = a_scaled
                    S = S2
                    sc = sc_new
                    moved = True
                    break
                a[jp] -= delta

            if moved:
                break

            a[j] += delta  # undo

        if not moved:
            # Try simple subgradient step
            sg = np.zeros(n)
            sg[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
            step = min(0.05, (sc - 1.0) / (np.linalg.norm(sg) ** 2 + 1e-10))
            a -= step * sg / (S * S)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def fienup_flatten(a, n_iters=300):
    """Fienup: flatten power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.15 + 0.7 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        ratio = np.clip(ratio, 0.5, 2.0) ** alpha
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def structured_perturb(a, rng, n_modes=10, scale=0.03):
    """Add low-frequency structured perturbations."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):
    """Add multiplicative Gaussian noise."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new

def coord_descent(a, n_sweeps=5, n_top=50, time_limit=30.0):
    """
    Coordinate descent: try multiplicative perturbations on top peak contributors.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    factors = [0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            best_val = a_old_j
            best_local = cur_sc
            for f in factors:
                cv = max(0.0, a_old_j * f)
                if abs(cv - a_old_j) < 1e-14:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_s = a / S_tmp
                sc = compute_score(a_s)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase LP-guided optimizer.
    Phase 1: Fienup initialization (5%)
    Phase 2: LP-based direction finding (40%)
    Phase 3: L-BFGS multi-beta (30%)
    Phase 4: Peak shaving + coordinate descent (15%)
    Phase 5: Final polish (10%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = compute_score(best_a)
        print(f"Parent: n={len(best_a)}, score={best_score:.8f}")
    else:
        n0 = 4096
        best_a = np.ones(n0) * 0.5
        best_score = compute_score(best_a)
        print(f"No parent, using uniform n={n0}, score={best_score:.8f}")

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Starting best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Fienup initialization (5% of time)
    # ==================================================================
    p1_end = t_start + 0.05 * (total_budget - 15.0)
    print(f"\n=== Phase 1: Fienup (until t={p1_end - t_start:.0f}s) ===")

    a_f1 = fienup_flatten(best_a, n_iters=200)
    sc = compute_score(a_f1)
    update(sc, a_f1, "FIENUP")

    # ==================================================================
    # Phase 2: LP-based direction finding (40% of time)
    # ==================================================================
    p2_end = t_start + 0.45 * (total_budget - 15.0)
    print(f"\n=== Phase 2: LP-based optimization (until t={p2_end - t_start:.0f}s) ===")

    time_for_lp = max(30.0, p2_end - time.monotonic())
    a_lp = lp_direction_optimize(best_a, n_iters=5000, max_top_k=80, time_limit=time_for_lp)
    sc = compute_score(a_lp)
    update(sc, a_lp, "LP")

    # Multiple LP passes with different K settings
    for round_k in [30, 50, 100]:
        if time.monotonic() > p2_end - 10:
            break
        remaining = min(60.0, p2_end - time.monotonic())
        if remaining < 10:
            break
        print(f"  LP round K={round_k}, time_limit={remaining:.0f}s")
        a_lp2 = lp_direction_optimize(best_a, n_iters=2000, max_top_k=round_k, time_limit=remaining)
        sc = compute_score(a_lp2)
        update(sc, a_lp2, f"LP K={round_k}")

    # ==================================================================
    # Phase 3: L-BFGS multi-beta (30% of time)
    # ==================================================================
    p3_end = t_start + 0.75 * (total_budget - 15.0)
    print(f"\n=== Phase 3: L-BFGS multi-beta (until t={p3_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas = [200, 500, 1000, 2000, 4000, 8000, 15000, 30000]
    for beta in betas:
        if time.monotonic() > p3_end - 15:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 20:
            break
        maxiter = min(15000, int(remaining * 20))
        maxfun = min(300000, int(remaining * 400))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter}, ~{remaining:.0f}s avail)...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # Second pass with high betas on best
    if time.monotonic() < p3_end - 30:
        print("  L-BFGS second pass (high beta)...")
        for beta in [10000, 25000, 50000]:
            if time.monotonic() > p3_end - 10:
                break
            remaining = p3_end - time.monotonic()
            if remaining < 15:
                break
            maxiter = min(10000, int(remaining * 15))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 30)
            print(f"    beta={beta}: score={sc:.8f}")
            update(sc, a_opt, f"LBFGS2 b={beta}")

    # ==================================================================
    # Phase 4: Peak shaving + coordinate descent (15% of time)
    # ==================================================================
    p4_end = t_start + 0.90 * (total_budget - 15.0)
    print(f"\n=== Phase 4: Peak shaving + CD (until t={p4_end - t_start:.0f}s) ===")

    # Peak shaving
    if time.monotonic() < p4_end - 30:
        remaining = min(60.0, p4_end - time.monotonic() - 10)
        a_ps = peak_shave_advanced(best_a, n_iters=5000, time_limit=remaining)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAPE")

    # Coordinate descent
    if time.monotonic() < p4_end - 15:
        remaining = min(45.0, p4_end - time.monotonic() - 5)
        a_cd = coord_descent(best_a, n_sweeps=10, n_top=80, time_limit=remaining)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESC")

        # L-BFGS polish after CD
        if sc <= best_score and time.monotonic() < p4_end - 10:
            sc2, a2 = run_lbfgs(a_cd, 5000, maxiter=5000, maxfun=100000)
            update(sc2, a2, "CD+LBFGS")

    # Multi-restart with perturbations
    for r in range(6):
        if time.monotonic() > p4_end - 10:
            break
        if r % 2 == 0:
            a_pert = structured_perturb(best_a, rng, n_modes=10, scale=0.015 + 0.005 * r)
        else:
            a_pert = noise_perturb(best_a, rng, scale=0.01 + 0.003 * r)
        remaining = p4_end - time.monotonic() - 5
        if remaining < 15:
            break
        maxiter = max(1000, int(remaining * 10))
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=maxiter, maxfun=maxiter * 20)
        print(f"  Restart {r}: score={sc:.8f}")
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 5: Final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish ===")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        if remaining > 10:
            maxiter = int(remaining * 15)
            sc, a_final = run_lbfgs(best_a, 8000, maxiter=maxiter, maxfun=maxiter * 25)
            update(sc, a_final, "FINAL-LBFGS-8000")

    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        if remaining > 10:
            maxiter = int(remaining * 10)
            sc, a_final = run_lbfgs(best_a, 20000, maxiter=maxiter, maxfun=maxiter * 20)
            update(sc, a_final, "FINAL-LBFGS-20000")

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_cd = coord_descent(best_a, n_sweeps=5, n_top=100, time_limit=max(5.0, deadline - 10 - time.monotonic()))
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # Final LP polish
    if time.monotonic() < deadline - 8:
        a_lp = lp_direction_optimize(best_a, n_iters=500, max_top_k=50,
                                     time_limit=max(5.0, deadline - 8 - time.monotonic()))
        sc = compute_score(a_lp)
        update(sc, a_lp, "FINAL-LP")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
