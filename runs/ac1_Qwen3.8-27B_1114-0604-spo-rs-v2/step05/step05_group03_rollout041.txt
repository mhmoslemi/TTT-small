```
def spectral_update(f, F, L, R_norm, M_target):
    n=len(f)
    g = cap_fill_target(R_norm, M_target, n)
    G = np.fft.rfft(g, L)
    target_spec = float(n) * G
    den = F*F
    maxden = float(np.max(np.abs(den))) if den.size else 1.0
    threshold = 1e-12 * max(1.0, maxden)
    ratio = np.ones(len(den), dtype=np.complex128)
    mask = np.abs(den) > threshold
    if np.any(mask):
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio[mask] = target_spec[mask] / den[mask]
    ratio = np.where(np.isfinite(ratio), ratio, 1.0+0j)
    sqrt_ratio = np.sqrt(ratio)
    # For real output, force last (Nyquist) real
    if sqrt_ratio.size:
        sqrt_ratio[-1] = abs(sqrt_ratio[-1])
    F_new = F * sqrt_ratio
    raw = np.fft.irfft(F_new, L)
    raw = raw[:len(f)]
    raw = np.maximum(raw, 0.0)
    raw = np.nan_to_num(raw, nan=0.0, posinf=0.0, neginf=0.0)
```

```python
"""
Search for a non-negative sequence minimizing
    2 * n * max(conv(a, a)) / (sum(a))^2

The code works internally with density sequences f = n*a / sum(a), so the
objective is 2 * max(conv(f, f)) / n.  It combines:
  - spectral peak-shaving / convolutional square-root iterations,
  - exponentiated-gradient subgradient / LSE local search,
  - symmetrization and multi-length candidate resampling.
The best sequence found is returned as a non-negative list.
"""

import numpy as np
import time

try:
    np.set_num_threads(1)
except Exception:
    pass

GLOBAL_START_TIME = 0.0
GLOBAL_DEADLINE = 0.0
GLOBAL_BEST_F = None
GLOBAL_BEST_SCORE = np.inf
GLOBAL_RNG = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def sanitize_real(x):
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1:
        x = x.ravel()
    if x.size == 0:
        return x
    return np.where(np.isfinite(x), x, 0.0)


def normalize_density(x):
    x = sanitize_real(x)
    n = x.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64)
    return x * (float(n) / s)


def square_fft(a):
    n = len(a)
    m = 2 * n - 1
    L = next_pow2(m)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    if c.size > m:
        c = c[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c, A, L


def density_score(f):
    f = normalize_density(f)
    n = f.size
    if n == 0:
        return np.inf
    c, _, _ = square_fft(f)
    return 2.0 * float(np.max(c)) / float(n)


def report_best(f, score, label):
    global GLOBAL_BEST_F, GLOBAL_BEST_SCORE
    if f is None:
        return
    if not np.isfinite(score):
        return
    f = normalize_density(f)
    if f.size == 0:
        return
    if GLOBAL_BEST_F is None or score < GLOBAL_BEST_SCORE - 1e-9:
        GLOBAL_BEST_F = f.copy()
        GLOBAL_BEST_SCORE = float(score)
        print(
            f"[{label}] new best score={score:.8f} len={len(f)} "
            f"t={time.monotonic() - GLOBAL_START_TIME:.1f}s"
        )


def load_parent_density():
    try:
        p = globals().get("height_sequence_1")
        if p is None:
            return None
        arr = np.asarray(p, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = sanitize_real(arr)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s <= 1e-16:
            return None
        a = arr / s
        n = arr.size
        f = normalize_density(float(n) * a)
        return f
    except Exception:
        return None


def resample_density(f, n):
    f = normalize_density(f)
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    if len(f) == n:
        return f.copy()
    if len(f) == 0:
        return np.ones(n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, len(f))
    x_new = np.linspace(0.0, 1.0, n)
    vals = np.interp(x_new, x_old, f)
    vals = np.maximum(vals, 1e-12)
    return normalize_density(vals)


def cap_fill_target(R, M, n):
    R = np.asarray(R, dtype=np.float64).copy()
    m = R.size
    if m == 0:
        return R
    M = float(M)
    if M <= 0.0:
        M = float(np.mean(R))
    over = np.maximum(R - M, 0.0)
    E = float(np.sum(over))
    if E <= 1e-14:
        return R
    cap = np.maximum(M - R, 0.0)
    cap_sum = float(np.sum(cap))
    if cap_sum <= 1e-300:
        return R
    if cap_sum < E:
        factor = cap_sum / E
        over = over * factor
        E = float(np.sum(over))
        if E <= 1e-14:
            return R
    g = R.copy()
    g -= over
    g += E * (cap / cap_sum)
    g = np.maximum(g, 0.0)
    gm = float(np.max(g))
    if gm > M * (1.0 + 1e-9) and gm > M + 1e-12:
        g = np.minimum(g, M)
    s = float(np.sum(g))
    if s > 1e-300:
        g *= (float(n) / s)
    return g


def spectral_update(f, F, L, R_norm, M_target):
    n = len(f)
    if n == 0:
        return f
    m = 2 * n - 1
    target = cap_fill_target(R_norm[:m], M_target, n)
    G = np.fft.rfft(target, L)
    target_spec = float(n) * G
    den = F * F
    maxden = float(np.max(np.abs(den))) if den.size else 1.0
    threshold = 1e-12 * max(1.0, maxden)
    ratio = np.ones(len(den), dtype=np.complex128)
    mask = np.abs(den) > threshold
    if np.any(mask):
        with np.errstate(divide="ignore", invalid="ignore"):
            ratio[mask] = target_spec[mask] / den[mask]
    ratio = np.where(np.isfinite(ratio), ratio, 1.0 + 0.0j)
    sqrt_ratio = np.sqrt(ratio)
    if sqrt_ratio.size:
        sqrt_ratio[-1] = float(abs(sqrt_ratio[-1]))
    F_new = F * sqrt_ratio
    raw = np.fft.irfft(F_new, L)
    raw = raw[:n]
    raw = sanitize_real(raw)
    raw = np.maximum(raw, 0.0)
    return normalize_density(raw)


def spectral_peak_shave(f0, deadline, label, max_iters=20000):
    f = normalize_density(f0)
    n = f.size
    if n == 0:
        return f, np.inf
    best_f = f.copy()
    best_sc = density_score(f)
    report_best(best_f, best_sc, label + " init")

    m = 2 * n - 1
    floor = float(n) / float(m) + 1e-10

    c, F, L = square_fft(f)
    R = c / float(n)
    Rmax = float(np.max(R))
    if Rmax <= floor + 1e-9:
        return best_f, best_sc

    step = max(1e-5, min(0.012, 0.05 * (Rmax - floor) + 1e-4))
    min_step = 5e-5
    alphas = (0.25, 0.45, 0.70, 1.00)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        if Rmax <= floor + 1e-9 or step <= min_step:
            break

        M_target = max(floor, Rmax - step)

        f_raw = spectral_update(f, F, L, R, M_target)

        best_cand = None
        best_cand_sc = density_score(f)

        for alpha in alphas:
            cand = f + alpha * (f_raw - f)
            cand = normalize_density(cand)
            sc = density_score(cand)
            if sc < best_cand_sc - 1e-12:
                best_cand_sc = sc
                best_cand = cand

        if best_cand is not None:
            fsym = normalize_density(0.5 * (best_cand + best_cand[::-1]))
            scsym = density_score(fsym)
            if scsym < best_cand_sc - 1e-12:
                best_cand = fsym
                best_cand_sc = scsym

        if best_cand is not None and best_cand_sc < best_sc - 1e-12:
            f = best_cand
            best_f = f.copy()
            best_sc = float(best_cand_sc)
            report_best(f, best_sc, label + " spec")
            no_improve = 0
            step = min(0.02, step * 1.03)
            Rmax = best_sc / 2.0
            c, F, L = square_fft(f)
            R = c / float(n)
        else:
            no_improve += 1
            step *= 0.70
            if no_improve >= 14:
                no_improve = 0
                if step <= min_step * 1.25:
                    break

    return best_f, best_sc


def full_weights(R, beta):
    m = R.size
    if m == 0:
        return np.ones(0, dtype=np.float64)
    if beta <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    M = float(np.max(R))
    z = beta * (R - M)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300:
        p = np.zeros(m, dtype=np.float64)
        idx = np.where(R >= M - 1e-12)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(R))], dtype=int)
        p[idx] = 1.0 / float(idx.size)
        return p
    return e / s


def top_weights(R, K, beta=8.0):
    m = R.size
    if m == 0:
        return np.ones(0, dtype=np.float64)
    K = int(max(1, min(int(K), m)))
    if K == m:
        return full_weights(R, beta)
    idx = np.argpartition(R, -K)[-K:]
    p = np.zeros(m, dtype=np.float64)
    vals = R[idx]
    M = float(np.max(vals))
    if beta <= 0.0:
        p[idx] = 1.0 / float(K)
    else:
        z = beta * (vals - M)
        e = np.exp(z)
        s = float(np.sum(e))
        if s <= 1e-300:
            p[idx] = 1.0 / float(K)
        else:
            p[idx] = e / s
    return p


def self_gradient_fft(f, p, n):
    L = next_pow2(max(1, 3 * n - 2))
    rev = f[::-1]
    A = np.fft.rfft(rev, L)
    P = np.fft.rfft(p, L)
    out = np.fft.irfft(A * P, L)
    if len(out) >= 2 * n - 1:
        return out[n - 1:2 * n - 1]
    return out


def gradient_search(f0, deadline, label, mode="lse", beta=100.0, topk=512):
    f = normalize_density(f0)
    n = f.size
    if n == 0:
        return f
    m = 2 * n - 1

    best_f = f.copy()
    best_sc = density_score(f)
    report_best(best_f, best_sc, label + " init")

    lf = np.log(np.maximum(f, 1e-12))
    v = np.zeros(n, dtype=np.float64)
    lr = 0.08
    mu = 0.75
    no_improve = 0
    it = 0

    if mode == "top":
        K_local = int(max(16, min(topk, m)))
        beta_local = 8.0
    else:
        K_local = m
        beta_local = float(beta)

    while time.monotonic() < deadline and it < 20000:
        it += 1

        lf_shift = lf - float(np.max(lf))
        f = normalize_density(np.exp(lf_shift))

        c, _, _ = square_fft(f)
        R = c / float(n)
        Rmax = float(np.max(R))
        sc = 2.0 * Rmax
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            report_best(best_f, best_sc, label)

        if it % 30 == 0:
            fsym = normalize_density(0.5 * (f + f[::-1]))
            scsym = density_score(fsym)
            if scsym < sc - 1e-12:
                f = fsym
                lf = np.log(np.maximum(f, 1e-12))
                v *= 0.3
                if scsym < best_sc - 1e-12:
                    best_sc = scsym
                    best_f = f.copy()
                    report_best(best_f, best_sc, label + " sym")
                continue

        if mode == "top":
            p = top_weights(R, K_local, beta_local)
        else:
            p = full_weights(R, beta_local)

        h = self_gradient_fft(f, p, n)
        g = h - float(np.mean(h))
        std = float(np.std(g))
        if std < 1e-14:
            lf += 0.05 * GLOBAL_RNG.standard_normal(n)
            v *= 0.5
            continue

        sg = g / std
        v = mu * v + (1.0 - mu) * sg
        nv = float(np.linalg.norm(v))
        if nv > 12.0:
            v *= 12.0 / nv

        current_sc = sc
        best_lf = None
        best_try_sc = current_sc

        for mult in (0.5, 1.0, 2.0):
            lf_try = lf - lr * mult * v
            f_try = normalize_density(np.exp(lf_try - float(np.max(lf_try))))
            sc_try = density_score(f_try)
            if sc_try < best_try_sc - 1e-12:
                best_try_sc = sc_try
                best_lf = lf_try

        if best_lf is not None:
            lf = best_lf
            f_try = normalize_density(np.exp(lf - float(np.max(lf))))
            fsym = normalize_density(0.5 * (f_try + f_try[::-1]))
            scsym = density_score(fsym)
            if scsym < best_try_sc - 1e-12:
                lf = np.log(np.maximum(fsym, 1e-12))
                best_try_sc = scsym
            v *= 0.9
            no_improve = 0
            lr = min(0.45, lr * 1.02)
        else:
            no_improve += 1
            lr *= 0.75
            if no_improve >= 25:
                v *= 0.4
                no_improve = 0
                if mode == "top":
                    K_local = int(min(m, max(16, int(K_local * 1.5) + 16)))
                    beta_local = min(60.0, beta_local * 1.5)
                else:
                    beta_local = min(2000.0, beta_local * 1.7)

        if it % 80 == 0:
            print(
                f"[{label}] it={it} curr={sc:.8f} best_local={best_sc:.8f} "
                f"lr={lr:.5f} K={K_local}"
            )

    return best_f


def try_symmetrize(f, label):
    if f is None or len(f) < 2:
        return f
    f = normalize_density(f)
    fs = normalize_density(0.5 * (f + f[::-1]))
    sc_fs = density_score(fs)
    sc_f = density_score(f)
    if sc_fs < sc_f - 1e-12:
        report_best(fs, sc_fs, label + " sym")
        return fs
    return f


def make_basic_density(n, kind=0):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        f = np.ones(n, dtype=np.float64)
    elif kind == 1:
        f = 1.0 + 0.6 * np.cos(2.0 * np.pi * x)
    elif kind == 2:
        f = 1.0 + 0.35 * np.cos(4.0 * np.pi * x) + 0.20 * np.cos(6.0 * np.pi * x)
    elif kind == 3:
        f = 0.4 + 0.6 * (2.0 * x - 1.0) ** 2
    else:
        f = 1.0 + 0.25 * np.sin(2.0 * np.pi * 3.0 * x + 0.7) \
            + 0.15 * np.sin(2.0 * np.pi * 5.0 * x + 1.9)
    f = np.maximum(f, 1e-3)
    return normalize_density(f)


def make_random_density(n):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    f = np.ones(n, dtype=np.float64)
    for _ in range(4):
        freq = float(GLOBAL_RNG.uniform(1.0, 9.0))
        phase = float(GLOBAL_RNG.uniform(0.0, 2.0 * np.pi))
        amp = float(GLOBAL_RNG.uniform(0.05, 0.30))
        f += amp * np.sin(2.0 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-3)
    return normalize_density(f)


def improve_sequence(f0, deadline, label):
    f = normalize_density(f0)
    report_best(f, density_score(f), label + " start")

    t0 = time.monotonic()
    rem = max(30.0, deadline - t0)

    d1 = min(deadline - 10.0, t0 + 0.35 * rem)
    if d1 > t0 + 15.0:
        f, _ = spectral_peak_shave(f, d1, label + ":spec1")

    f = try_symmetrize(f, label + ":1")

    d2 = min(deadline - 10.0, t0 + 0.58 * rem)
    if time.monotonic() < d2 - 15.0:
        f = gradient_search(f, d2, label + ":grad-top", mode="top", beta=8.0, topk=320)
        f = try_symmetrize(f, label + ":2")

    d3 = min(deadline - 10.0, t0 + 0.82 * rem)
    if time.monotonic() < d3 - 15.0:
        f = gradient_search(f, d3, label + ":grad-lse", mode="lse", beta=120.0)
        f = try_symmetrize(f, label + ":3")

    d4 = max(time.monotonic() + 15.0, min(deadline - 5.0, t0 + rem))
    if time.monotonic() < d4 - 10.0:
        f, _ = spectral_peak_shave(f, d4, label + ":spec2")

    return f


def choose_tasks(parent_available, base_n):
    max_work_n = 20000
    if not parent_available:
        opts = [8192, 12288, 16384]
        base_n = 8192
    else:
        base_n = int(max(1024, min(int(base_n), max_work_n)))
        opts = [
            base_n,
            int(base_n * 1.4) + 1,
            int(base_n * 1.8) + 1,
        ]

    unique = []
    for nn in opts:
        nn = int(max(1024, min(nn, max_work_n)))
        if nn not in unique:
            unique.append(nn)
    if int(base_n) not in unique:
        unique.append(int(base_n))

    fracs = [0.40, 0.30, 0.20, 0.10]
    tasks = []
    for i, nn in enumerate(unique):
        tasks.append((nn, fracs[i] if i < len(fracs) else 0.08, False))
    tasks.append((int(base_n), 0.12, True))
    return tasks


def propose_candidate(budget_s=2400):
    global GLOBAL_START_TIME, GLOBAL_DEADLINE, GLOBAL_RNG, GLOBAL_BEST_F, GLOBAL_BEST_SCORE

    try:
        budget = float(budget_s) if budget_s is not None else 2400.0
    except Exception:
        budget = 2400.0
    budget = max(120.0, min(budget, 3600.0))

    t0 = time.monotonic()
    GLOBAL_START_TIME = t0
    GLOBAL_DEADLINE = t0 + max(90.0, budget - 30.0)
    GLOBAL_RNG = np.random.default_rng(20260101)
    GLOBAL_BEST_F = None
    GLOBAL_BEST_SCORE = np.inf

    print(f"budget={budget:.0f}s, internal deadline in {GLOBAL_DEADLINE - t0:.1f}s")

    parent = load_parent_density()
    parent_available = parent is not None
    if parent_available:
        if len(parent) > 25000:
            parent = resample_density(parent, 20000)
        report_best(parent, density_score(parent), "parent")

    if GLOBAL_BEST_F is None:
        base_n = 8192
        report_best(make_basic_density(base_n, 0), density_score(make_basic_density(base_n, 0)), "default")
    else:
        base_n = len(GLOBAL_BEST_F)

    tasks = choose_tasks(parent_available, base_n)
    final_reserve = 70.0
    total_work = max(120.0, (GLOBAL_DEADLINE - t0) - final_reserve)

    for n_task, frac, random_start in tasks:
        if time.monotonic() >= GLOBAL_DEADLINE - final_reserve - 30.0:
            break
        task_deadline = min(GLOBAL_DEADLINE - 30.0, time.monotonic() + frac * total_work)
        if task_deadline <= time.monotonic() + 30.0:
            continue

        if random_start:
            f_start = make_random_density(n_task)
            tag = f"rand-n{n_task}"
        else:
            if GLOBAL_BEST_F is None:
                f_start = make_basic_density(n_task, 0)
            else:
                f_start = resample_density(GLOBAL_BEST_F, n_task)
            tag = f"n{n_task}"

        try:
            improve_sequence(f_start, task_deadline, tag)
        except Exception as exc:
            print(f"exception in {tag}: {exc}")

    if GLOBAL_BEST_F is None:
        GLOBAL_BEST_F = make_basic_density(2048, 0)
        GLOBAL_BEST_SCORE = density_score(GLOBAL_BEST_F)

    if time.monotonic() < GLOBAL_DEADLINE - 15.0:
        try:
            GLOBAL_BEST_F, _ = spectral_peak_shave(
                GLOBAL_BEST_F,
                GLOBAL_DEADLINE - 10.0,
                "final-polish"
            )
        except Exception as exc:
            print(f"final polish exception: {exc}")

    best = normalize_density(GLOBAL_BEST_F)
    n = len(best)
    a = best / float(n)
    eps = 1e-9
    a = (1.0 - eps) * a + eps * (1.0 / float(n))
    a = np.clip(a, 0.0, 1000.0)
    out = [float(x) for x in a]

    if sum(out) < 0.01:
        out = [0.5]

    print(f"FINAL internal best={GLOBAL_BEST_SCORE:.8f}, len={len(out)}, "
          f"time={time.monotonic() - t0:.1f}s")

    if "evaluate_sequence" in globals():
        try:
            if len(out) <= 12000 and time.monotonic() < GLOBAL_DEADLINE - 5.0:
                verified = evaluate_sequence(out)
                print(f"verified evaluate_sequence={verified:.8f}")
        except Exception as exc:
            print(f"verification skipped/failed: {exc}")

    return out
```
