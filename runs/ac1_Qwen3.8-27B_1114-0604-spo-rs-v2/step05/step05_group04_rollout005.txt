def fast_conv(a):

def compute_score(a):

    c = fast_conv(a)

    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Compute dL/dx_i = 2 * sum_j x[j] * p[i+j] for i=0..n-1
    # Using convolution: G_i = conv(x_flipped, p)[i + n - 1]

    conv_res = np.fft.irfft(XF * PF, N_conv)
    G = conv_res[n - 1:2 * n - 1]

    dL_dx = 2.0 * G
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_dx / S2 - 2.0 * L / S3)

    return score, grad

        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_inits(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 using adaptive Fienup spectral shaping
    combined with LSE-smoothed L-BFGS optimization and subgradient refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        n_work = len(parent)
    else:
        n_work = 6962

    print(f"Length: {n_work}, Initial best: {best_score:.8f}")

    # === Phase 1: Fienup from parent and diverse inits (25% time) ===
    p1_end = t_start + 0.25 * (budget_s - 30)
    print(f"\n=== Phase 1: Fienup (t < {p1_end - t_start:.0f}s) ===")

    modes = ['compress_top', 'raise_floor', 'exponent', 'hybrid', 'waterfill']
    a_cur = best_a.copy()

    for mode in modes:
        if time.monotonic() > p1_end:
            break
        a_out = fienup_modes(a_cur, 300, mode, gamma=0.8, top_frac=0.3, floor_frac=0.15)
        sc = compute_score(a_out)
        if update_best(sc, a_out, f"F1-{mode}"):
            a_cur = a_out.copy()

    # Second pass with different gamma
    for mode in modes[:3]:
        if time.monotonic() > p1_end:
            break
        a_out = fienup_modes(a_cur, 200, mode, gamma=0.5, top_frac=0.2, floor_frac=0.1)
        sc = compute_score(a_out)
        if update_best(sc, a_out, f"F1b-{mode}"):
            a_cur = a_out.copy()

    # Try from diverse initializations
    inits = build_inits(n_work, rng, parent)
    for idx, a0 in enumerate(inits[1:4]):
        if time.monotonic() > p1_end:
            break
        for mode in ['compress_top', 'hybrid']:
            if time.monotonic() > p1_end:
                break
            a_out = fienup_modes(a0, 200, mode, gamma=0.8)
            sc = compute_score(a_out)
            update_best(sc, a_out, f"F1-init{idx}-{mode}")

    print(f"After Phase 1: {best_score:.8f}")

    # === Phase 2: LSE L-BFGS with annealing (35% time) ===
    p2_end = t_start + 0.60 * (budget_s - 30)
    print(f"\n=== Phase 2: LSE L-BFGS (t < {p2_end - t_start:.0f}s) ===")

    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 20:
            break
        maxiter = min(8000, max(500, int(remaining * 2)))
        maxfun = maxiter * 8
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"L2-b{beta}")

    print(f"After Phase 2: {best_score:.8f}")

    # === Phase 3: Iterative refinement (30% time) ===
    p3_end = t_start + 0.90 * (budget_s - 30)
    print(f"\n=== Phase 3: Iterative Refinement (t < {p3_end - t_start:.0f}s) ===")

    modes_cycle = ['compress_top', 'raise_floor', 'hybrid', 'exponent', 'waterfill']
    m_idx = 0

    while time.monotonic() < p3_end:
        # Fienup step
        mode = modes_cycle[m_idx % len(modes_cycle)]
        m_idx += 1
        n_it = 150 if time.monotonic() < p3_end - 300 else 80
        a_out = fienup_modes(best_a, n_it, mode, gamma=0.7 + 0.3 * rng.random())
        sc = compute_score(a_out)
        improved = update_best(sc, a_out, f"P3-{mode}")

        # L-BFGS step
        if time.monotonic() < p3_end - 120:
            beta = [100, 300, 500, 1000, 2000][int(rng.integers(5))]
            maxiter = min(5000, max(1000, int((p3_end - time.monotonic()) * 1.5)))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 6)
            update_best(sc, a_opt, f"P3-LBFGS-b{beta}")

        # Subgradient step
        if time.monotonic() < p3_end - 100:
            top_m = [5, 10, 20][int(rng.integers(3))]
            a_sg = subgradient_step(best_a, n_iters=500, top_m=top_m, step_init=0.015)
            sc = compute_score(a_sg)
            update_best(sc, a_sg, f"P3-SG-m{top_m}")

        # Perturbation restart
        if time.monotonic() < p3_end - 200 and m_idx % 3 == 0:
            noise_scale = 0.01 + 0.02 * rng.random()
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(n_work))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = a_p.sum()
            if S_p > 1e-6:
                a_p = a_p / S_p
                a_out = fienup_modes(a_p, 100, 'hybrid', gamma=0.8)
                sc = compute_score(a_out)
                update_best(sc, a_out, f"P3-PERT-{m_idx}")

    print(f"After Phase 3: {best_score:.8f}")

    # === Phase 4: Final polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")

    # Best beta L-BFGS
    if time.monotonic() < deadline - 60:
        for beta in [5000, 10000, 20000]:
            if time.monotonic() > deadline - 60:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=40000)
            update_best(sc, a_opt, f"P4-b{beta}")

    # Final Fienup
    if time.monotonic() < deadline - 30:
        a_out = fienup_modes(best_a, 100, 'hybrid', gamma=0.6)
        sc = compute_score(a_out)
        update_best(sc, a_out, "P4-FIENUP")

    # Very fine subgradient
    if time.monotonic() < deadline - 20:
        a_sg = subgradient_step(best_a, n_iters=1000, top_m=5, step_init=0.005)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "P4-SG-fine")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
