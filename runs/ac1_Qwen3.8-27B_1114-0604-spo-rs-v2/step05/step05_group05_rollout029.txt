```python
"""
Softmax-parameterized L-BFGS with continuation for minimum peak autocorrelation.
Key innovation: exponential parameterization eliminates non-negativity projection,
allowing unconstrained optimization that avoids the kink at zero.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_and_grad_c(c, beta):
    """Compute LSE of c and its gradient wrt c."""
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    return L, p


def lse_score_grad_a(a, beta, S):
    """
    Compute LSE-smoothed objective and gradient wrt a.
    Objective: 2n * LSE_beta(c) / S^2 where S = sum(a)
    Returns (objective, gradient_wrta)
    """
    n = len(a)
    c = fast_conv(a)
    L, p = lse_and_grad_c(c, beta)
    
    # grad_L[i] = 2 * sum_m a[m] * p[m+i] for i=0,...,n-1
    # Computed via FFT
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev))[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    obj = 2.0 * n * L / S2
    grad_a = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return obj, grad_a


def softmax_obj_grad(b, beta, S_target):
    """
    Objective and gradient in b-space (softmax parameterization).
    a = S_target * exp(b) / sum(exp(b))
    Objective: 2n * max(c) (approximately, via LSE)
    """
    # Compute softmax (numerically stable)
    b_shifted = b - b.max()
    u = np.exp(b_shifted)
    S_u = float(u.sum())
    a = S_target * u / S_u
    S = S_target  # by construction
    
    obj, grad_a = lse_score_grad_a(a, beta, S)
    
    # Chain rule: grad_b[i] = a[i] * grad_a[i] - (a[i]/S) * sum_j(a[j]*grad_a[j])
    weighted = float(np.dot(a, grad_a)) / S
    grad_b = a * (grad_a - weighted)
    
    return obj, grad_b


def run_lbfgs_softmax(b0, beta, maxiter=3000, maxfun=50000, S_target=None):
    """Run L-BFGS in b-space."""
    n = len(b0)
    if S_target is None:
        S_target = np.sqrt(2.0 * n)
    
    def fun_and_grad(b):
        return softmax_obj_grad(b, beta, S_target)
    
    res = minimize(
        fun_and_grad, b0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-20, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    
    # Convert back to a
    b_out = res.x
    b_shifted = b_out - b_out.max()
    u = np.exp(b_shifted)
    a = S_target * u / u.sum()
    
    return a, res


def fienup_refined(a, max_time=60.0):
    """
    Refined Fienup/GERCOR with:
    - Adaptive threshold (starts high, linearly decreases)
    - Momentum
    - Power-based scaling (not amplitude)
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    
    t_start = time.monotonic()
    n_total_iters = 200  # estimate
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        t = it / max(1, n_total_iters)
        
        # Adaptive parameters
        momentum = 0.3 * t
        thresh = 3.0 - 1.9 * t  # 3.0 -> 1.1
        power = 0.5 + 0.4 * t   # 0.5 -> 0.9
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        bin_thresh = P_mean * thresh
        above = P > bin_thresh
        
        if above.any():
            scale = np.ones_like(P)
            # Scale power: P_new = P_mean * thresh for bins above threshold
            target_P = P_mean * thresh
            factor = (target_P / P[above]) ** power
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor
        else:
            # No bins above threshold, try to increase flatness
            # Slightly boost low bins
            below = P < P_mean * 0.8
            if below.any():
                scale = np.ones_like(P)
                scale[below] = 1.0 + 0.01 * (1.0 - t)
            else:
                scale = np.ones_like(P)
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Momentum
        update = a_new - a
        v = momentum * v + update
        a = a + v
        
        # Project and normalize
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a / best_a.sum()
            break
        a = a / S_new
        
        # Track best
        if it % 20 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def peak_reduced_optimize(a, max_time=120.0):
    """
    Targeted optimization of top peaks.
    For each top peak, identify contributing coordinates and optimize them jointly.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for sweep in range(10):
        if time.monotonic() - t_start > max_time:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        # Top peaks
        n_top = min(8, 2*n-1)
        peak_indices = np.argsort(c)[::-1][:n_top]
        
        for k in peak_indices:
            if time.monotonic() - t_start > max_time:
                break
            k = int(k)
            
            # Find contributing coordinates
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            
            # Contribution of a[j] to c[k]: a[j] * a[k-j]
            # To reduce c[k], we can reduce the products a[j]*a[k-j]
            # For each pair (j, k-j), the product contributes to c[k]
            
            # Simple approach: for each j in range, try reducing a[j] slightly
            # and see if the score improves
            n_try = min(10, j_end - j_start + 1)
            
            # Get top contributors
            contribs = np.array([a[j] * a[k-j] for j in range(j_start, j_end+1)])
            top_local = np.argsort(contribs)[::-1][:n_try]
            top_global = j_start + top_local
            
            for j in top_global:
                if time.monotonic() - t_start > max_time:
                    break
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                
                # Try several reductions
                for factor in [0.7, 0.8, 0.9, 1.1, 1.2]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
                    continue
                # Renormalize a to match best_a scale
                S_cur = float(a.sum())
                if S_cur > 0:
                    a = a / S_cur
    
    return best_a


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)
```

```python
"""
Softmax-parameterized L-BFGS with beta continuation for minimum peak autocorrelation.
Key innovation: exponential parameterization (a = S*softmax(b)) eliminates non-negativity
projection, enabling unconstrained smooth optimization. Combined with multi-beta
continuation, refined Fienup, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_a(a, beta):
    """
    Compute LSE-smoothed objective and gradient wrt a.
    Objective: 2n * LSE_beta(c(a)) / S^2 where c = a*a (linear conv)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    # grad_L[i] = 2 * sum_m a[m] * p[m+i] for i=0,...,n-1
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev))[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]

    S2 = S * S
    S3 = S2 * S
    obj = 2.0 * n * L / S2
    grad_a = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return obj, grad_a


def softmax_obj_grad(b, beta, S_target):
    """
    Objective and gradient in b-space using softmax parameterization.
    a = S_target * softmax(b), so a >= 0 and sum(a) = S_target automatically.
    """
    b_shifted = b - b.max()
    u = np.exp(b_shifted)
    Z = float(u.sum())
    a = S_target * u / Z

    obj, grad_a = lse_score_grad_a(a, beta)

    # Chain rule for softmax: da_j/db_i = a_j*delta_ji - a_j*a_i/S
    # df/db_i = a_i * grad_a[i] - (a_i/S) * sum_j(a_j * grad_j)
    weighted = float(np.dot(a, grad_a)) / S_target
    grad_b = a * (grad_a - weighted)

    return obj, grad_b


def run_lbfgs_softmax(b0, beta, S_target, maxiter=2000, maxfun=20000):
    """Run L-BFGS-B in b-space (unconstrained)."""
    n = len(b0)

    def fun_jac(b):
        return softmax_obj_grad(b, beta, S_target)

    try:
        res = minimize(
            fun_jac, b0,
            method='L-BFGS-B', jac=True,
            options={'ftol': 1e-20, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
    except Exception:
        res = None

    if res is None or not hasattr(res, 'x'):
        return b0, 0, 0

    b_out = res.x
    # Convert to a
    b_shifted = b_out - b_out.max()
    u = np.exp(b_shifted)
    a = S_target * u / u.sum()

    return a, res.nit, res.nfev


def fienup_refined(a, max_time=60.0):
    """Refined GERCOR with adaptive threshold, momentum, and power scaling."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    t_start = time.monotonic()
    it = 0
    max_iters = int(max_time * 8)  # estimate ~8 iters/sec

    while time.monotonic() - t_start < max_time and it < max_iters:
        it += 1
        t = min(1.0, it / max(1, max_iters))

        momentum = 0.2 + 0.4 * t
        thresh = 3.0 - 1.95 * t  # 3.0 -> 1.05
        power = 0.3 + 0.6 * t    # 0.3 -> 0.9

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        scale = np.ones_like(P)
        if above.any():
            target_P = P_mean * thresh
            factor = (target_P / P[above]) ** power
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        update = a_new - a
        v = momentum * v + update
        a = a + v

        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        if it % 25 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def peak_targeted_descent(a, max_time=120.0):
    """
    Targeted coordinate descent on top peaks.
    For each peak, identify top contributing coordinates and try local reductions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    n_sweeps_max = 20

    for sweep in range(n_sweeps_max):
        if time.monotonic() - t_start > max_time:
            break

        c = fast_conv(a)
        n_top = min(6, 2 * n - 1)
        peak_indices = np.argsort(c)[::-1][:n_top]

        improved = False
        for k in peak_indices:
            if time.monotonic() - t_start > max_time:
                break
            k = int(k)

            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            n_contrib = j_end - j_start + 1
            if n_contrib <= 0:
                continue

            # Get contributions
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * (a[np.clip(k - js, 0, n - 1)])

            n_try = min(8, n_contrib)
            top_local = np.argsort(contribs)[::-1][:n_try]
            top_global = js[top_local]

            for j in top_global:
                if time.monotonic() - t_start > max_time:
                    break
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue

                for factor in [0.6, 0.75, 0.9, 1.1, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        break
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        improved = True
                        break
                else:
                    a[j] = old_val
                    continue
                S_cur = float(a.sum())
                if S_cur > 1e-12:
                    a = a / S_cur

        if not improved:
            # Try a perturbation to escape local minimum
            a = a * (1.0 + 0.01 * np.random.default_rng(sweep).standard_normal(n))
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S > 1e-12:
                a = a / S

    return best_a


def nesterov_lse_direct(a, max_time=60.0, beta_start=50.0, beta_end=5000.0):
    """
    Nesterov-accelerated projected gradient on LSE objective (in a-space).
    Fallback when softmax L-BFGS is not effective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    beta = beta_start
    lr = 0.01
    mu = 0.9
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = min(1.0, (time.monotonic() - t_start) / max_time)
        mu = 0.9 + 0.05 * t_frac
        lr *= 0.9995

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        _, grad = lse_score_grad_a(a_look, beta)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        effective_lr = min(lr, 0.05 / (gnorm + 1e-10))
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Increase beta
        beta = min(beta * 1.003, beta_end)

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def upsample(a, new_n):
    """Upsample a sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)
```
