Let me now think about the "peak-pushing spectral method" in more detail.

**Peak-Pushing Spectral Method:**

Goal: Reduce the peak of `conv(a,a)` by modifying the power spectrum.

The convolution `c = IFFT(P)` where `P = |FFT(a, M)|^2` and M = padded length.

If we want to reduce `c[k0]` (the peak at position k0), we need to reduce the contribution of frequency bins that are "aligned" at position k0. The contribution of bin j to c[k0] is proportional to `P[j] * cos(2π*j*k0/M)` (for real c).

So the "sensitivity" of c[k0] to P[j] is `cos(2π*j*k0/M)` (times a constant).

To reduce the peak, we should decrease P[j] for bins j where `cos(2π*j*k0/M)` is large (positive), and we can increase P[j] where the cosine is negative (to maintain the total energy if needed).

But we need to be careful: we can't arbitrarily change P because it must correspond to a valid non-negative sequence.

**Algorithm:**
1. Compute `a`, `A = FFT(a, M)`, `P = |A|^2`, `c = IFFT(P)`.
2. Find the top-m peaks in c: positions `k_1, ..., k_m` with values `c[k_1], ..., c[k_m]`.
3. For each peak position k_i, compute the "sensitivity vector" `s_i[j] = cos(2π*j*k_i/M)` for j = 0, ..., M/2.
4. Compute the combined sensitivity: `s[j] = sum_i w_i * s_i[j]` where `w_i` are weights (e.g., proportional to how much c[k_i] exceeds the average).
5. Update P: `P_new[j] = P[j] * (1 - eta * s[j])` (for positive s) and `P_new[j] = P[j] * (1 + eta * |s[j]|)` (for negative s, to maintain energy).
   Or more simply: `P_new = P - eta * s` (clipped to non-negative).
6. Recover a from P_new using iterative Fourier method:
   - `a_curr = current a`
   - Repeat: `A = FFT(a_curr, M)`, `A_new = sqrt(P_new) * exp(i*angle(A))`, `a_new = IFFT(A_new)[:n]`, `a_new = max(a_new, 0)`, `a_curr = a_new`.
7. Check if the score improved. If yes, accept. If no, reduce eta and try again.

This is essentially a subgradient method in the spectral domain with non-negativity recovery.

The advantage over the time-domain subgradient is that in the spectral domain, we can more directly target specific peaks.

        sensitivity = np.zeros(M_fft // 2 + 1)

            weight = max(0, c[int(k)] - c_avg)  # Only target peaks above average
            if weight > 0:

        # Don't modify DC (P[0])

        P_mod = np.maximum(P_mod, 0.0)

"""
Optimized LSE-Gradient + Spectral Peak-Pushing + Subgradient for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations:
1. Efficient LSE gradient via FFT cross-correlation (correct derivation)
2. Spectral peak-pushing: directly reduce peak contributions in frequency domain
3. Adaptive beta continuation for LSE approximation
4. Multi-method ensemble with time-adaptive allocation
5. Symmetric subspace optimization to reduce effective dimensionality
6. Momentum subgradient for fine-tuning top peaks

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        L = c_max
        p = np.zeros(M)
        p[int(np.argmax(c))] = 1.0
    else:
        L = c_max + float(np.log(exp_sum)) / beta
        p = exp_scaled / exp_sum
    score = 2.0 * n * L / (S * S)
    N = 3 * n - 2
    N_fft = 1 << (N - 1).bit_length()
    P_fft = np.fft.rfft(p, N_fft)
    X_fft = np.fft.rfft(x, N_fft)
    g = np.fft.irfft(P_fft * np.conj(X_fft), N_fft)[:n]
    S2 = S * S
    S3 = S2 * S
    grad = 4.0 * n * g / S2 - 4.0 * n * L / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 0.5
        S0 = n * 0.5
    a0 = a0 / S0
    a0 = a0 * np.sqrt(2.0 * n)
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_peak_push(a, n_iters=200, eta_init=0.05, eta_decay=0.995,
                       top_m=30, n_recovers=8, relax=0.5):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    n_half = M_fft // 2 + 1
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    j_arr = np.arange(n_half, dtype=np.float64)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = np.fft.irfft(A * A, M_fft)[:M]

        A_curr = A

def top_m_subgradient_momentum(a, n_iters=3000, top_m=20, step_init=0.02,
                               decay=0.9999, momentum=0.9):

    step = step_init
    vel = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):

        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if top_m >= M:
            top_idx = np.arange(M)
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        step_adapt = max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12)
        step = min(step, step_adapt)
        step = min(step, 0.05)
        vel = momentum * vel - step * grad
        a_new = a + vel
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:

            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= decay
    return best_a

            gz += g[n - 1: n - 1 - n_right: -1][::-1][:half]
        return sc, gz

    bounds = [(0.0, 1000.0)] * half
    res = minimize(
        sym_obj_grad, a0_sym,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    z = np.clip(res.x, 0.0, 1000.0)
    a_out = np.zeros(n)
    a_out[:half] = z
    a_out[half:] = z[half - 1: 0 - 1: -1][:n - half]
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = 0.5 * np.ones(n)
    sc = compute_score(a_out)
    return sc, a_out


def multi_scale_refine(a, n_iters_outer=5, down_factor=2):
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for o in range(n_iters_outer):
        n_small = max(64, n // (down_factor ** (o + 1)))
        a_small = np.interp(
            np.linspace(0, 1, n_small),
            np.linspace(0, 1, n),
            a
        )
        a_small = np.maximum(a_small, 0.0)
        sc_small, a_small_opt = run_lbfgs(a_small, 500, maxiter=2000, maxfun=30000)
        a_up = np.interp(
            np.linspace(0, 1, n),
            np.linspace(0, 1, n_small),
            a_small_opt
        )
        a_up = np.maximum(a_up, 0.0)
        sc_up = compute_score(a_up)
        if sc_up < best_sc:
            best_sc = sc_up
            best_a = a_up.copy()
            a = a_up.copy()
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        print(f"Parent score: {sc:.8f}, length: {len(parent)}")
    else:
        print("No parent available, starting from scratch")
        n = 6962
        best_a = np.ones(n) * 0.5
        best_score = compute_score(best_a)

    n_main = len(best_a)
    print(f"Main length: {n_main}")

    # Phase 0: Quick LBFGS from parent
    phase0_end = t_start + 0.05 * (total_budget - 30.0)
    print(f"\n=== Phase 0: Quick LBFGS (5%) ===")
    if parent is not None and time.monotonic() < phase0_end:
        for beta in [50, 100]:
            if time.monotonic() > phase0_end:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"P0-LBFGS b={beta}")

    # Phase 1: LBFGS continuation
    phase1_end = t_start + 0.30 * (total_budget - 30.0)
    print(f"\n=== Phase 1: LBFGS Continuation (30%) ===")
    betas = [100, 200, 500, 1000, 2000, 5000]
    for beta in betas:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(10000, int(remaining * 2))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        update_best(sc, a_opt, f"P1-LBFGS b={beta}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: Spectral peak pushing
    phase2_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Spectral Peak Pushing (55%) ===")
    a_spec = spectral_peak_push(best_a, n_iters=300, eta_init=0.03,
                                eta_decay=0.998, top_m=40, n_recovers=6, relax=0.4)
    sc = compute_score(a_spec)
    update_best(sc, a_spec, "P2-SPEC")

    if time.monotonic() < phase2_end:
        a_spec2 = spectral_peak_push(best_a, n_iters=200, eta_init=0.08,
                                     eta_decay=0.995, top_m=15, n_recovers=10, relax=0.7)
        sc = compute_score(a_spec2)
        update_best(sc, a_spec2, "P2-SPEC2")

    # LBFGS after spectral
    if time.monotonic() < phase2_end:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "P2-POLISH")

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Symmetric optimization
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Symmetric LBFGS (70%) ===")
    if time.monotonic() < phase3_end:
        for beta in [200, 1000, 3000]:
            if time.monotonic() > phase3_end:
                break
            sc, a_opt = symmetric_lbfgs(best_a, beta, maxiter=4000, maxfun=60000)
            update_best(sc, a_opt, f"P3-SYM b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Subgradient with momentum
    phase4_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Subgradient Momentum (85%) ===")
    for top_m in [10, 20, 50]:
        if time.monotonic() > phase4_end:
            break
        a_sg = top_m_subgradient_momentum(best_a, n_iters=3000, top_m=top_m,
                                          step_init=0.015, decay=0.9998, momentum=0.85)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"P4-SG m={top_m}")

    # LBFGS polish after subgradient
    if time.monotonic() < phase4_end:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "P4-POLISH")

    print(f"After Phase 4: best = {best_score:.8f}")

    # Phase 5: Final refinement
    phase5_end = deadline
    print(f"\n=== Phase 5: Final Refinement ===")

    # Try perturbation + re-optimize
    for trial in range(3):
        if time.monotonic() > phase5_end - 30:
            break
        noise_scales = [0.01, 0.03, 0.05]
        a_p = best_a * (1.0 + noise_scales[trial] * rng.standard_normal(n_main))
        a_p = np.clip(a_p, 0.0, 1000.0)
        S_p = a_p.sum()
        if S_p < 1e-12:
            break
        a_p = a_p / S_p * np.sqrt(2.0 * n_main)
        sc, a_opt = run_lbfgs(a_p, 5000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, f"P5-PERT-{trial}")

    # Final spectral push
    if time.monotonic() < phase5_end - 60:
        a_final_spec = spectral_peak_push(best_a, n_iters=200, eta_init=0.02,
                                          eta_decay=0.997, top_m=60,
                                          n_recovers=8, relax=0.3)
        sc = compute_score(a_final_spec)
        update_best(sc, a_final_spec, "P5-SPEC-FINAL")
        if time.monotonic() < phase5_end - 30:
            sc, a_opt = run_lbfgs(a_final_spec, 20000, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, "P5-FINAL-LBFGS")

    # Try a different length
    if time.monotonic() < phase5_end - 90 and parent is not None:
        n_alt = 8192
        if n_alt != n_main:
            print(f"\n  Trying alternative length n={n_alt}")
            a_alt = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n_main), best_a)
            a_alt = np.maximum(a_alt, 0.0)
            sc, a_opt = run_lbfgs(a_alt, 500, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"ALT n={n_alt}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
