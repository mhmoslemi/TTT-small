def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr_wa(w, a):
    """Compute cross_corr(w, a)[j] = sum_k w[k+j]*a[k] for j=0..n-1."""
    n = int(a.size)
    N = next_power_of_two(2 * n)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:n] = w
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    out = np.fft.irfft(W * np.conjugate(A), N)[:n]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def d_from_a(a):

def exact_score(a):

        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.clip(np.interp(x_new, x_old, a), 0.0, None).astype(np.float64)


PEN_SUM = 1e-3


def lse_score_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    wcdot_d = float(np.dot(w, d))

    h = cross_corr_wa(w, a)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * wcdot_d / S) * a
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def solve_lp_direction(a, M, max_iters_cd=50, tol=1e-8):
    """
    Solve the LP dual:
      minimize M * sum(y)
      subject to (L_a) y >= 1, y >= 0
    where (L_a) is lower-triangular Toeplitz with a on the diagonal pattern:
      (L_a y)_i = sum_{k=0}^{i} a[i-k] * y_k
    
    Returns the primal solution g (the optimal direction).
    Uses column generation / cutting plane approach for efficiency.
    """
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64), 0.0

    # Start with uniform y
    y = np.ones(n, dtype=np.float64) / float(n)
    
    best_obj = np.inf
    best_y = y.copy()

    for it in range(max_iters_cd):
        # Check which constraints are violated
        # (L_a y)_i = conv(a, y)[i] for i=0..n-1 (lower-triangular part)
        # conv(a, y)[i] = sum_{k=0}^{i} a[k] * y[i-k]
        N = next_power_of_two(2 * n)
        a_pad = np.zeros(N, dtype=np.float64)
        a_pad[:n] = a
        y_pad = np.zeros(N, dtype=np.float64)
        y_pad[:n] = y
        conv_val = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(y_pad), N)[:n]
        
        # Find violated constraints: conv_val[i] < 1 - tol
        violated = conv_val < 1.0 - tol
        if not np.any(violated):
            break
        
        # Find the most violated constraint
        worst_idx = int(np.argmin(conv_val))
        
        # Update y to satisfy the worst constraint
        # Simple approach: scale y up in a targeted way
        # y_i needs to increase at positions that affect constraint worst_idx
        # (L_a y)[worst_idx] = sum_{k=0}^{worst_idx} a[worst_idx-k] * y_k
        # To increase this, we should increase y_k for k where a[worst_idx-k] > 0
        
        deficit = 1.0 - conv_val[worst_idx]
        if deficit <= 0:
            break
        
        # Distribute the deficit to y indices that have large a[worst_idx-k]
        weights = np.zeros(n, dtype=np.float64)
        max_k = min(worst_idx, n - 1)
        for k in range(max_k + 1):
            weights[k] = a[worst_idx - k]
        
        wsum = float(weights.sum())
        if wsum < 1e-15:
            # Fallback: uniform update
            y += deficit / n
        else:
            # Proportional update
            y += deficit * weights / wsum
        
        y = np.clip(y, 0.0, None)
        obj = M * float(y.sum())
        if obj < best_obj:
            best_obj = obj
            best_y = y.copy()
    
    # Recover primal from complementary slackness (approximate)
    # g_i > 0 only where the dual constraint (L_a^T y)_i is tight
    # (L_a^T y)_i = sum_{k=i}^{n-1} a[k-i] * y_k = cross_corr(a, y)[i]
    N = next_power_of_two(2 * n)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    y_pad = np.zeros(N, dtype=np.float64)
    y_pad[:n] = best_y
    cross_val = np.fft.irfft(np.fft.rfft(a_pad) * np.conjugate(np.fft.rfft(y_pad)), N)[:n]
    
    # Primal solution: g_i = 1 where dual constraint is tight, 0 otherwise
    # (This is the extreme point solution)
    tol_cs = 1e-6 * max(1.0, float(np.max(cross_val)))
    g = (np.abs(cross_val - 1.0) <= tol_cs).astype(np.float64)
    
    # Also include indices where y_i > 0 (strong duality hint)
    # Actually for the LP, the extreme point is determined by active constraints.
    # Let's use a simpler approach: g proportional to where dual is active
    g = np.clip(cross_val - 0.9 * np.max(cross_val), 0.0, None)
    if g.sum() < 1e-12:
        g = cross_val.copy()
    g = g / g.sum()
    
    return g, M * float(best_y.sum())


def multiplicative_step(a, h, eta):
    """Apply multiplicative update: a_i <- a_i * exp(-eta * h_i)."""
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= np.log(log_a.max() + 1e-300)
    new_a = np.exp(log_a)
    return new_a


def active_set_weights(d, threshold_frac=0.01, top_k=20):
    """Get weights for active lags (near the max)."""
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return np.ones(len(d)) / len(d)
    
    # Lags where d is within threshold_frac of max
    threshold = dmax * (1.0 - threshold_frac)
    active = d >= threshold
    
    # Also include top_k
    if int(active.sum()) < top_k:
        idx = np.argpartition(d, -top_k)[-top_k:]
        active[idx] = True
    
    w = np.zeros(len(d), dtype=np.float64)
    w[active] = d[active]  # Weight proportional to d value
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.ones(len(d)) / len(d)
    else:
        w /= wsum
    return w


def subgrad_polish(a, max_iters=30, deadline=None):
    """Polish using multiplicative subgradient steps."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc

        d, S = d_from_a(a)
        if d.size == 0:
            break

        # Try different active set sizes and step sizes
        for k_frac in (0.005, 0.01, 0.02, 0.05):
            for eta in (0.01, 0.03, 0.08, 0.15, 0.30):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                w = active_set_weights(d, threshold_frac=k_frac, top_k=max(5, int(k_frac * len(d))))
                h = cross_corr_wa(w, a)
                q = multiplicative_step(a, h, eta)
                q = normalize_a(q)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if n == 0:
        return np.inf, a
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = 0.02 * max(0.5, n / 8192.0)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))

    maxfun = max(64, maxfun)
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def exact_score_a(a):
    return exact_score(a)


def lp_guided_iteration(a, deadline, max_lp_iters=3):
    """
    Perform LP-guided iterations:
    1. Compute active set from current convolution peaks
    2. Solve LP dual to find optimal direction
    3. Blend with multiplicative update
    4. Normalize and evaluate
    """
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for it in range(max_lp_iters):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        d, S = d_from_a(a)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax < 1e-15:
            break
        
        # Get active set weights
        w_active = active_set_weights(d, threshold_frac=0.02, top_k=30)
        
        # Try LP-based direction
        # For efficiency, solve LP on a reduced set of variables
        # Use the active set structure to limit the LP size
        g_lp, _ = solve_lp_direction(a, dmax, max_iters_cd=20, tol=1e-6)
        
        if g_lp.size == n and g_lp.sum() > 1e-12:
            # Blend LP direction with current
            for t in (0.1, 0.2, 0.3, 0.5):
                q = normalize_a((1.0 - t) * a + t * g_lp)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
        
        # Also try multiplicative update with active set
        h = cross_corr_wa(w_active, a)
        for eta in (0.02, 0.05, 0.1, 0.2):
            q = multiplicative_step(a, h, eta)
            q = normalize_a(q)
            sc = exact_score(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
        
        a = best_a.copy()
    
    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=25):
    """Generate diverse initial sequences."""
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # U-shapes (diverging at edges)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        arr = 1.0 + alpha / (0.05 + np.abs(2.0 * x - 1.0))
        add(arr)

    # Cosine shapes
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Sine shapes
    for f in (1, 2, 3, 4):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    # Exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Bimodal
    for width in (0.1, 0.15, 0.25):
        add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)))

    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    # Pure random
    for _ in range(2):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):
    """Perturb the best sequence for restarts."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.60))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.40))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        d, S = d_from_a(p)
        if d.size > 0:
            w = active_set_weights(d, 0.02, 20)
            h = cross_corr_wa(w, p)
            q = multiplicative_step(p, h, strength)
        else:
            q = p.copy()
    elif mode == 3:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.4 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 4:
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    else:
        k = int(rng.integers(5, max(7, min(n, 101))))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")

    return normalize_a(np.clip(q, 0.0, None))


def optimize_trajectory(a_start, deadline, rng, label):
    """Main optimization loop with continuation."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 10.0))
    n = len(a)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Continuation schedule: increasing beta (sharper max approximation)
    stages = [
        (50.0, 0.05),
        (100.0, 0.07),
        (200.0, 0.08),
        (400.0, 0.08),
        (800.0, 0.10),
        (1500.0, 0.10),
        (3000.0, 0.12),
        (6000.0, 0.12),
        (12000.0, 0.12),
        (25000.0, 0.10),
        (50000.0, 0.08),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est = 0.02 * max(0.5, n / 8192.0)
            mf = int((0.7 * rem) / max(est, 1e-4))
            mf = max(64, min(2000, mf))
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} elapsed={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0
                ))
            a = normalize_a(a2)

        # LP-guided step between stages
        if time.monotonic() < sub_dead - 5.0:
            sc_lp, a_lp = lp_guided_iteration(a, sub_dead, max_lp_iters=2)
            if np.isfinite(sc_lp) and sc_lp < best_sc:
                best_sc = sc_lp
                best_a = a_lp.copy()
                print("[{}] LP best {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_lp)

        # Subgradient polish
        if time.monotonic() < sub_dead - 3.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=8, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad best {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 256:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine target lengths
    lengths = []
    parent_len = 0
    if parent is not None:
        parent_len = int(min(parent.size, 16384))
        if parent_len < 2048:
            parent_len = 2048
        lengths.append(parent_len)
        for L in [4096, 6144, 8192, 10240, 12288, 16384]:
            if L != parent_len:
                lengths.append(L)
    else:
        lengths = [8192, 10240, 6144, 12288, 4096]

    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:5]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Quick evaluation of diverse inits
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=18 if n > 8000 else 25)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))

    if not all_scores:
        n0 = lengths[0] if lengths else 8192
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.8f} (from length {})".format(
        all_scores[int(order[0])], all_lens[int(order[0])]))

    # Select top starts
    starts = []
    used_lens = set()

    if parent is not None and parent_len > 0:
        ps = normalize_a(resample_sequence(parent, parent_len))
        starts.append(ps)
        used_lens.add(parent_len)

    for idx in order:
        if len(starts) >= 3:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) ==
