def lp_peak_equalization(a, M, active_indices, max_b_sum):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for k in active_indices, b >= 0
    
    Where (a*b)_k = sum_j a_j * b_{k-j} (linear convolution)
    """
    import cvxpy as cp
    
    n = len(a)
    b = cp.Variable(n, non_neg=True)
    
    # Compute convolution a * b for each active lag
    # (a * b)_k = sum_j a_j * b_{k-j} where 0 <= j < n and 0 <= k-j < n
    # So j ranges from max(0, k-n+1) to min(k, n-1)
    
    constraints = []
    for k in active_indices:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            # (a*b)_k = sum_{j=j_min}^{j_max} a_j * b_{k-j}
            # Note: b_{k-j} where k-j ranges from k-j_max to k-j_min
            coeffs = np.zeros(n)
            for j in range(j_min, j_max + 1):
                coeffs[k - j] = a[j]
            constraints.append(coeffs @ b <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)  # or another
    
    return b.value

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Compute dL/da_i = 2 * sum_j a_j * p[j+i] for i=0,...,n-1
    # Using FFT: IFFT(A * conj(P))[k] = sum_j a_j * p[j+k] (for valid k)
    L_fft = 1 << (n + len(p) - 1).bit_length()
    A = np.fft.rfft(a, L_fft)
    P = np.fft.rfft(p, L_fft)
    r_full = np.fft.irfft(A * np.conj(P), L_fft)
    r = r_full[:n]

    dL_da = 2.0 * r

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * dL_da - (4.0 * n * L / S3) * a

    return score, grad


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000, time_limit=300):
    """Run L-BFGS-B on the LSE-smoothed objective with time limit."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()

    def time_check():
        return time.monotonic() - t0 > time_limit

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=5000, eta0=0.05, decay=0.999):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        g_centered = grad - np.mean(grad)
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay

    return best_a


def spectral_init(n, rng):
    """Initialize with random flat power spectrum, take absolute value."""
    # Generate random phases with conjugate symmetry for real signal
    k = n // 2 + 1
    phases = 2.0 * np.pi * rng.random(k)
    Z = np.exp(1j * phases)
    # Enforce conjugate symmetry for real IFFT
    Z_full = np.zeros(n, dtype=complex)
    Z_full[:k] = Z
    for i in range(1, min(n - i, k)):
        Z_full[n - i] = np.conj(Z[i])
    a_raw = np.fft.irfft(Z_full, n)
    a = np.abs(a_raw)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def spectral_init_v2(n, rng):
    """Initialize with sum of random flat-spectrum signals for smoother result."""
    a = np.zeros(n)
    nsum = 4
    for _ in range(nsum):
        a += spectral_init(n, rng)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def lp_peak_direction(a, max_time=30.0):
    """
    Solve LP: max sum(b) s.t. (a*b)_k <= M for active k, b >= 0.
    Returns direction b (or None if LP fails).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    c = fast_conv(a)
    M = float(c.max())

    # Find active constraints: lags where c_k > 0.95 * M
    active_mask = c > 0.95 * M
    active_indices = np.where(active_mask)[0]

    if len(active_indices) == 0:
        return None

    # Limit to top 200 active constraints for efficiency
    if len(active_indices) > 200:
        top_c = c[active_indices]
        sorted_idx = np.argsort(top_c)[::-1][:200]
        active_indices = active_indices[sorted_idx]

    b = cp.Variable(n, non_neg=True)
    constraints = []
    t0 = time.monotonic()

    for k in active_indices:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max < j_min:
            continue
        # (a*b)_k = sum_{j=j_min}^{j_max} a_j * b_{k-j}
        # b index: k-j for j in [j_min, j_max] => b indices in [k-j_max, k-j_min]
        b_idx_start = k - j_max
        b_idx_end = k - j_min
        coeff = a[j_min:j_max + 1]
        constraints.append(coeff @ b[b_idx_start:b_idx_end + 1] <= M)

        if time.monotonic() - t0 > max_time * 0.5:
            break

    if not constraints:
        return None

    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None

    if b.value is None:
        return None

    b_val = np.clip(np.array(b.value).flatten(), 0.0, None)
    if b_val.sum() < 1e-12:
        return None
    return b_val


def lp_step(a, step_size=0.1, max_time=30.0):
    """Take an LP-based step: move in direction b found by LP."""
    b = lp_peak_direction(a, max_time=max_time)
    if b is None:
        return a, compute_score(a)

    n = len(a)
    S_a = float(a.sum())
    S_b = float(b.sum())
    if S_b < 1e-12:
        return a, compute_score(a)
    b = b / S_b  # normalize

    # Try different step sizes
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
        a_new = a + t * step_size * (b - a)  # move towards b direction
        a_new = np.clip(a_new, 0.0, None)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a, best_sc


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 6))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for bidx in range(min(n_blocks, n // block_size)):
            s = bidx * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 4:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    else:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.clip(np.interp(x_new, x_old, parent), 0.0, 10.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init_v2(n, rng))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.05, np.cos(np.pi * x)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.exp(-3.0 * (x - 0.5) ** 2))
    return inits


def normalize_a(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / len(a)
    else:
        a = a / S
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    Multi-phase: spectral init -> mirror descent -> L-BFGS -> LP polish -> restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Report parent
    if parent is not None:
        parent_norm = normalize_a(parent)
        sc_parent = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        sc_parent = np.inf
        print("No parent available.")

    # Determine working length: try parent length and a larger one
    if parent is not None and 4096 <= len(parent) <= 32768:
        n_work = len(parent)
    else:
        n_work = 16384

    print(f"Working length: {n_work}")

    # ================================================================
    # Phase 1: Multi-start exploration (25% time)
    # ================================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 1: Multi-start (deadline t={phase1_end - t_start:.0f}s)")
    print(f"{'='*50}")

    inits = build_initializations(n_work, rng, parent)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a_init = normalize_a(a_init)
        sc = compute_score(a_init)
        report(sc, a_init, f"init_{i}")

        # Mirror descent for exploration
        t_md_start = time.monotonic()
        a_md = mirror_descent(a_init, beta=50, n_iters=3000, eta0=0.08, decay=0.999)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"mirror_{i}")
        print(f"    init_{i} mirror: {sc_md:.10f} ({time.monotonic()-t_md_start:.1f}s)")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ================================================================
    # Phase 2: L-BFGS with increasing beta (40% time)
    # ================================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 2: L-BFGS Refinement (deadline t={phase2_end - t_start:.0f}s)")
    print(f"{'='*50}")

    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 60:
            break

        maxiter = int(min(50000, max(5000, remaining * 30)))
        maxfun = int(min(200000, max(10000, remaining * 100)))
        tl = min(remaining * 0.8, 300.0)

        print(f"  L-BFGS beta={beta}, time_limit={tl:.0f}s...")
        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun, time_limit=tl)
        report(sc, a_opt, f"LBFGS b={beta}")

        # Perturbed second pass for higher betas
        if beta >= 1000 and time.monotonic() < phase2_end - 90:
            a_pert = perturbed_restart(best_a, rng, scale=0.015)
            tl2 = min(phase2_end - time.monotonic() - 10, 120.0)
            if tl2 > 30:
                sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter//2,
                                           maxfun=maxfun//2, time_limit=tl2)
                report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ================================================================
    # Phase 3: LP-based peak equalization (20% time)
    # ================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n{'='*50}")
    print(f"Phase 3: LP Peak Equalization (deadline t={phase3_end - t_start:.0f}s)")
    print(f"{'='*50}")

    n_lp_rounds = 5
    for rnd in range(n_lp_rounds):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break

        # LP step
        t_lp = time.monotonic()
        a_lp, sc_lp = lp_step(best_a, step_size=0.1, max_time=min(remaining * 0.4, 60.0))
        sc_lp = compute_score(a_lp)
        report(sc_lp, a_lp, f"LP#{rnd+1}")
        print(f"    LP round {rnd+1}: {sc_lp:.10f} ({time.monotonic()-t_lp:.1f}s)")

        # Follow up with L-BFGS
        if time.monotonic() < phase3_end - 60:
            beta = 5000 + 5000 * rnd
            tl = min(remaining * 0.4, 120.0)
            if tl > 20:
                sc, a_opt, _ = run_lbfgs(a_lp, beta, maxiter=20000, maxfun=80000, time_limit=tl)
                report(sc, a_opt, f"LBFGS post-LP#{rnd+1} b={beta}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ================================================================
    # Phase 4: Iterated local search + final polish (15% time)
    # ================================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: ILS + Final Polish")
    print(f"{'='*50}")

    n_trials = 8
    for trial in range(n_trials):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 120:
            break

        scale = 0.01 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Quick mirror descent
        a_md = mirror_descent(a_pert, beta=200, n_iters=1000, eta0=0.03, decay=0.999)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"ILS#{trial} md")

        # L-BFGS refinement
        beta = 3000 + int(10000 * rng.random())
        tl = min(remaining * 0.5, 150.0)
        if tl > 30:
            sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-10), beta,
                                     maxiter=20000, maxfun=80000, time_limit=tl)
            report(sc, a_opt, f"ILS#{trial} b={beta}")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 40:
        for beta in [50000, 100000]:
            if time.monotonic() > deadline - 30:
                break
            tl = min(deadline - 30 - time.monotonic(), 180.0)
            if tl > 20:
                sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000, time_limit=tl)
                report(sc, a_opt, f"FINAL b={beta}")

    # One more LP polish at the very end
    if time.monotonic() < deadline - 20:
        a_lp, sc_lp = lp_step(best_a, step_size=0.05, max_time=30.0)
        sc_lp = compute_score(a_lp)
        report(sc_lp, a_lp, "final_LP")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
