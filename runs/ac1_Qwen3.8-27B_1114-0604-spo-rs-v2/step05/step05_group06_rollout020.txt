    n = len(a)
    S = a.sum()
    c = _self_conv(a)
    
    # Prefix max and suffix max of c
    c_max = c.max()
    # For each j, C_other = max(c[0:j], c[j+n:2n-1])
    # We can precompute prefix_max[i] = max(c[0:i]) and suffix_max[i] = max(c[i:2n-1])
    prefix_max = np.maximum.accumulate(c)
    suffix_max = np.maximum.accumulate(c[::-1])[::-1]
    
    for j in range(n):
        # k range: [j, min(2n-2, j+n-1)]
        k_start = j
        k_end = min(2*n-1, j+n)  # exclusive

        if k_start > 0:

        if k_end < 2*n-1:
            C_other = max(C_other, suffix_max[k_end])

        if a[j] > 1e-10:

def _obj_1d(x, a, j, k_start, k_end, C_other, S_rem, c):
    """Evaluate the 1D objective for a[j] = x."""
    delta = x - a[j]
    if S_rem + x < 1e-12:
        return np.inf
    
    # Compute max(c[k] + delta * a[k-j] for k in [k_start, k_end))

def _mass_transfer(a, n_iters=1000):
    """Reduce peaks by transferring mass from peak contributors to safe positions."""
    n = len(a)
    S = a.sum()
    for it in range(n_iters):
        c = _self_conv(a)
        cmax = c.max()
        
        # Top K peak positions
        K = 10
        top_k = np.argsort(c)[-K:]
        
        # For each peak position k*, find the top contributor j
        # c[k*] = sum_j a[j] * a[k*-j]
        # The contribution of j to c[k*] is a[j] * a[k*-j]
        
        # Build a matrix of contributions: contrib[j, i] = a[j] * a[top_k[i] - j] (if valid)
        # For each j, total_contribution[j] = sum_i contrib[j, i]
        # Transfer mass from j with highest total_contribution to j' with lowest
        
        # For efficiency, compute total contribution vector:
        total_contrib = np.zeros(n)
        for k_star in top_k:
            # Contribution of a[j] to c[k_star] is a[j] * a[k_star - j] for 0 <= k_star - j < n
            j_min = max(0, k_star - n + 1)
            j_max = min(n, k_star + 1)
            for j in range(j_min, j_max):
                total_contrib[j] += a[j] * a[k_star - j]
        
        # Donor: j with highest total_contrib and a[j] > threshold
        donor = np.argmax(total_contrib * (a > S/n * 0.5))
        
        # Acceptors: j' with low total_contrib and low a[j']
        # We want a[j'] such that a[j'] * a[k* - j'] is small for all top_k
        accept_scores = np.zeros(n)
        for k_star in top_k:
            j_min = max(0, k_star - n + 1)
            j_max = min(n, k_star + 1)
            # For j' not in [j_min, j_max), contribution is 0
            # For j' in [j_min, j_max), contribution is a[j'] * a[k_star - j']
            for j in range(j_min, j_max):
                accept_scores[j] += a[j] * a[k_star - j]
        
        # Choose acceptor with lowest accept_score and a[j'] < S/n
        # Actually, we want to ADD mass to j', so we want j' where adding mass won't create new peaks
        # A safe position is one where a[k* - j'] is small for all top_k
        
        # For each j', the "risk" of adding mass is max over k* of a[k* - j']
        risk = np.zeros(n)
        for k_star in top_k:
            # risk[j'] += a[k_star - j'] for valid j'
            j_min = max(0, k_star - n + 1)
            j_max = min(n, k_star + 1)
            for j in range(j_min, j_max):
                risk[j] += a[k_star - j]
        
        # Best acceptor: low risk, and a[j'] is small (so we have room to add)
        # Score: risk[j'] + a[j'] * n (penalize large a[j'])
        score = risk + a * n * 0.1
        acceptor = np.argmin(score)
        
        if donor == acceptor:
            continue
        
        # Transfer a small amount
        amount = 0.01 * a[donor]
        a[donor] -= amount
        a[acceptor] += amount
        
    return a

        amount = min(0.05 * a[donor], 0.01 * S)
        a[donor] -= amount
        a[acceptor] += amount
    
    return a / a.sum()

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using a multi-phase approach:
Phase 1: Diverse initialization (spectral, window, parent-based)
Phase 2: Progressive LSE optimization with aggressive beta schedule
Phase 3: Mass redistribution heuristic for peak reduction
Phase 4: Subgradient polishing with adaptive step size
Phase 5: Final very-high-beta LSE polish

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 2.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 2)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50,
                     'm': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=50, deadline=None):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break
        active_thresh = max(1e-6 * dmax, dmax - 1e-4)
        active = d >= active_thresh
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        found_improvement = False
        for scale in (0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q.copy()
                found_improvement = True
                break
        if not found_improvement:
            break
    return best_sc, best_a


def _mass_redistribution(a, n_iters=200, deadline=None):
    n = len(a)
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        c = _self_conv(a)
        S = a.sum()
        K = min(15, 2 * n - 1)
        top_k = np.argsort(c)[-K:]

        risk = np.zeros(n)
        contrib = np.zeros(n)
        for k_star in top_k:
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n, k_star + 1)
            valid_j = np.arange(j_lo, j_hi)
            a_vals = a[k_star - valid_j]
            risk[valid_j] += a_vals
            contrib[valid_j] += a[valid_j] * a_vals

        donor_mask = a > (S / n) * 0.2
        if not donor_mask.any():
            break
        donor_scores = contrib.copy()
        donor_scores[~donor_mask] = -1
        donor = int(np.argmax(donor_scores))
        if donor_scores[donor] <= 0:
            break

        acceptor_mask = ~donor_mask
        if not acceptor_mask.any():
            acceptor_mask = np.ones(n, dtype=bool)
            acceptor_mask[donor] = False
        acceptor_scores = risk.copy()
        acceptor_scores[~acceptor_mask] = np.inf
        acceptor = int(np.argmin(acceptor_scores))

        if donor == acceptor:
            break

        amount = 0.02 * a[donor]
        amount = min(amount, 0.005 * S)
        a[donor] -= amount
        a[acceptor] += amount

    return _norm(a)


def _spectral_init_flat(n, rng):
    Nf = _next_pow2(2 * n - 1)
    phase = rng.uniform(-np.pi, np.pi, size=Nf // 2 + 1)
    spec = np.ones(Nf // 2 + 1, dtype=np.complex128) * np.exp(1j * phase)
    spec[0] = 0.0
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _spectral_init_decaying(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.5, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _window_init(n, rng):
    x = np.linspace(0, 1, n)
    method = rng.integers(0, 4)
    if method == 0:
        alpha = rng.uniform(2.0, 12.0)
        a = np.exp(-alpha * (x - 0.5) ** 2)
    elif method == 1:
        beta = rng.uniform(2.0, 15.0)
        a = np.sinc(2 * beta * (x - 0.5)) ** 2
        a = np.abs(a)
    elif method == 2:
        gamma = rng.uniform(0.3, 1.5)
        a = (1 - np.abs(2 * x - 1)) ** gamma
    else:
        f = rng.uniform(1.0, 5.0)
        amp = rng.uniform(0.2, 0.8)
        a = 1.0 + amp * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 0.01, None)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def _smooth_perturb(a, rng, strength=0.2):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)

            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, remaining * 0.45, 800)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_f}, beta={beta_f}: {best_score:.8f}")

    # Mass redistribution on final best
    if time.monotonic() < final_deadline - 60:
        mt_deadline = final_deadline - 50.0
        a_mt = _mass_redistribution(best_seq, n_iters=400, deadline=mt_deadline)
        sc_mt = _score(a_mt)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = a_mt.copy()
            print(f"  Final mass_redist: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < final_deadline - 20:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=40, deadline=final_deadline - 15)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Try small perturbations and re-optimize
    if time.monotonic() < final_deadline - 50:
        for restart in range(4):
            if time.monotonic() > final_deadline - 30:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05 + 0.03 * restart)
            chunk_dl = min(final_deadline - 30, time.monotonic() + 20)
            sc_r, a_r = _run_lbfgs(p, 30000, 500, max(3, chunk_dl - time.monotonic()), 300)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=20, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
