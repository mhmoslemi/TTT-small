def waterfill_project(c, cap):
    c=np.asarray(c,dtype=np.float64)
    L=c.size
    s=float(c.sum())
    if s<=0 or not np.isfinite(s): return np.full(L,1.0/L)
    c=c/s
    avg=1.0/L
    if cap <= avg*(1+1e-10):
       return np.full(L, avg)
    cc=np.minimum(c, cap)
    deficit=1.0 - float(cc.sum())
    if deficit <= 1e-12:
       return cc
    mask=cc < cap - 1e-12
    if not np.any(mask):
       cdf=np.full(L, cap)
       rem=1.0-float(cdf.sum())
       if rem>0: cdf += rem/L
       return cdf
    idx=np.where(mask)[0]
    vals=cc[idx]
    order=np.argsort(vals)[::-1]
    vals=vals[order]
    idx=idx[order]
    prefix=np.cumsum(vals)
    p=np.arange(1, vals.size+1, dtype=np.float64)
    w=(deficit+prefix)/p
    # conditions:
    cond1 = (w <= vals + 1e-12)
    if vals.size > 1:
       cond2 = np.concatenate(([True], w[:-1] >= vals[1:] - 1e-12))
    else: cond2=np.array([True])
    good=cond1 & cond2
    if not np.any(good):
       # choose p where w closest to vals?
       p_idx=int(np.argmin(np.abs(w-vals)))
    else:
       # choose largest p? maybe the last good? The projection level condition should have exactly one. Choose last good to cover more? 
       p_idx=int(np.where(good)[0][-1])
    p_int=p_idx+1
    # If w maybe >cap due rounding, cap
    w_val=float(np.min([w[p_idx], cap]))
    # If w_val < vals[p_int]? but condition. 
    # To ensure sum, compute actual deficit if setting prefix to w_val; adjust maybe use water proportional if mismatch.
    new_cc=cc.copy()

"""
Search for a non-negative sequence minimizing
    2*n*max(conv(a,a)) / (sum(a)**2)

The optimizer combines:
  1) direct spectral projection on the linear self-convolution domain,
     using A(z)^2 = FFT(a*a) and water-filled convolution targets;
  2) an LSE-smoothed L-BFGS refinement in log-softmax parameters;
  3) active-peak subgradient polishing;
  4) length resizing / multi-length exploration.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def fft_len(L):
    """Smallest power-of-two >= L."""
    L = int(max(1, L))
    return 1 << (L - 1).bit_length()


def normalize_nonnegative(a):
    """Clip to [0, inf), normalize to sum 1. Fallback is uniform."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-14 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def conv_fft(a):
    """Linear self-convolution via FFT. Length is 2n-1."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = fft_len(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.real(c)
    # Remove tiny negative numerical noise.
    c = np.where(c > 0.0, c, 0.0)
    return c


def seq_score(a):
    """Internal score matching evaluate_sequence (without exact np.convolve)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-14 or not np.isfinite(s):
        return float("inf")
    c = conv_fft(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if m < 0 or not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def waterfill_project(c, cap):
    """
    Project nonnegative c with sum 1 onto {0 <= x <= cap, sum 1}
    using an approximately L2-best increase of the largest feasible coordinates.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return c
    s = float(c.sum())
    if s <= 1e-14 or not np.isfinite(s):
        return np.full(L, 1.0 / L, dtype=np.float64)
    c = c / s

    avg = 1.0 / L
    if cap <= avg * (1.0 + 1e-12):
        return np.full(L, avg, dtype=np.float64)

    cc = np.minimum(c, cap)
    deficit = 1.0 - float(cc.sum())
    if deficit <= 1e-12:
        return cc

    mask = cc < cap - 1e-12
    if not np.any(mask):
        out = np.full(L, cap, dtype=np.float64)
        rem = 1.0 - float(out.sum())
        if rem > 0:
            out += rem / L
        return np.clip(out, 0.0, None)

    idx = np.where(mask)[0]
    vals = cc[idx]
    order = np.argsort(vals)[::-1]
    vals = vals[order]
    idx = idx[order]

    prefix = np.cumsum(vals)
    p = np.arange(1, vals.size + 1, dtype=np.float64)
    w = (deficit + prefix) / p

    cond1 = (w <= vals + 1e-11)
    if vals.size > 1:
        cond2 = np.empty(vals.size, dtype=bool)
        cond2[0] = True
        cond2[1:] = w[:-1] >= vals[1:] - 1e-11
    else:
        cond2 = np.array([True], dtype=bool)

    good = cond1 & cond2
    if np.any(good):
        # The projection level is unique; use a stable choice.
        p_idx = int(np.where(good)[0][-1])
    else:
        p_idx = int(np.argmin(np.abs(w - vals)))

    p_int = p_idx + 1
    w_val = float(np.clip(w[p_idx], 0.0, cap))

    new_cc = cc.copy()
    new_cc[idx[:p_int]] = w_val

    ss = float(new_cc.sum())
    if not np.isfinite(ss) or abs(ss - 1.0) > 1e-7:
        # Fallback proportional fill within box.
        head = cap - new_cc
        rem = 1.0 - ss
        if rem > 0:
            th = float(head.sum())
            if th > 1e-30:
                new_cc += head * (rem / th)
        else:
            new_cc += rem / L

    new_cc = np.clip(new_cc, 0.0, None)
    ss = float(new_cc.sum())
    if ss > 1e-14:
        new_cc = new_cc / ss
    return new_cc


def spectral_step_ratio(a, ct, N, alpha):
    """
    Move A toward sqrt(target FFT) multiplicatively:
        A_new = A * exp(0.5*alpha*(log(Ct)-log(C))), C=A^2.
    """
    A = np.fft.rfft(a, N)
    C = A * A
    Ct = np.fft.rfft(ct, N).astype(np.complex128)

    scale = max(float(np.max(np.abs(C))) if C.size else 1.0,
                float(np.max(np.abs(Ct))) if Ct.size else 1.0,
                1e-30)
    eps = scale * 1e-12

    zC = np.abs(C) < eps
    zT = np.abs(Ct) < eps

    Csafe = C.copy().astype(np.complex128)
    Ctsafe = Ct.copy().astype(np.complex128)
    Csafe[zC] = eps
    Ctsafe[zT] = eps

    mag_ratio = np.abs(Ctsafe) / np.abs(Csafe)
    phi = np.angle(Ctsafe / Csafe)
    log_ratio = np.log(np.clip(mag_ratio, 1e-12, 1e12)) + 1j * phi
    log_ratio = np.clip(log_ratio.real, -6.0, 6.0) + 1j * np.clip(log_ratio.imag, -4.0, 4.0)

    Anew = A.astype(np.complex128) * np.exp(0.5 * alpha * log_ratio)

    # If current spectrum was zero, jump directly to the target square root there.
    if np.any(zC):
        Anew[zC] = np.sqrt(Ctsafe[zC], where=True)

    if Anew.size:
        Anew[0] = abs(Anew[0])
    if N % 2 == 0 and Anew.size:
        if not np.isfinite(Anew[-1]):
            Anew[-1] = 0.0
        else:
            Anew[-1] = Anew[-1].real

    out = np.fft.irfft(Anew, N)[:a.size]
    return np.real(out)


def spectral_step_blend(a, ct, N, alpha):
    """
    Blend the squared spectrum C=A^2 with target spectrum Ct, then square-root.
    """
    A = np.fft.rfft(a, N)
    C = A * A
    Ct = np.fft.rfft(ct, N).astype(np.complex128)

    Cb = (1.0 - alpha) * C.astype(np.complex128) + alpha * Ct

    if Cb.size:
        Cb[0] = max(0.0, float(np.real(Cb[0])))
    if N % 2 == 0 and Cb.size:
        val = float(np.real(Cb[-1]))
        Cb[-1] = max(0.0, val)

    Anew = np.sqrt(Cb.astype(np.complex128))

    if Anew.size:
        if not np.isfinite(Anew[0]):
            Anew[0] = 0.0
        else:
            Anew[0] = abs(Anew[0])
    if N % 2 == 0 and Anew.size:
        if not np.isfinite(Anew[-1]):
            Anew[-1] = 0.0
        else:
            Anew[-1] = Anew[-1].real

    out = np.fft.irfft(Anew, N)[:a.size]
    return np.real(out)


def spectral_optimize(a, target_score, max_time, mode="ratio",
                      reduce_score=0.0003,
                      alphas=(0.2, 0.45, 0.7),
                      t_blends=(0.4, 0.7, 1.0)):
    """
    Direct spectral water-filling optimizer.
    a should be normalized. target_score is an absolute score target.
    """
    if max_time <= 1e-3:
        return normalize_nonnegative(a)

    a = normalize_nonnegative(a)
    n = a.size
    if n == 0:
        return a

    L = 2 * n - 1
    N = fft_len(L)
    avgM = 1.0 / L

    best = a.copy()
    best_sc = seq_score(a)

    # Ensure target max is not below the trivial average.
    goalM = target_score / (2.0 * n)
    if goalM < avgM * 1.0005:
        goalM = avgM * 1.0005

    t_start = time.monotonic()
    stall = 0

    while (time.monotonic() - t_start) < max_time:
        c = conv_fft(a)
        cs = float(c.sum())
        if cs <= 1e-14:
            break
        c = c / cs
        cmax = float(np.max(c))
        cur_sc = float(2.0 * n * cmax)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()

        curM = cur_sc / (2.0 * n)
        cap = curM - reduce_score / (2.0 * n)
        if cap < goalM:
            cap = goalM
        if cap <= avgM:
            cap = avgM * 1.0001
        # Keep a small enforced decrease so iteration moves.
        if curM - cap < max(1e-6 / (2.0 * n), 1e-12):
            cap = curM - max(1e-6 / (2.0 * n), 1e-12)

        ct = waterfill_project(c, cap)

        improved = False

        for alpha in alphas:
            if mode == "blend":
                raw = spectral_step_blend(a, ct, N, alpha)
            else:
                raw = spectral_step_ratio(a, ct, N, alpha)

            if not np.all(np.isfinite(raw)):
                raw = np.where(np.isfinite(raw), raw, 0.0)

            # Candidate 1: hard nonnegativity.
            raw1 = np.clip(raw, 0.0, None)
            s1 = float(raw1.sum())

            # Candidate 2: shift-to-nonnegative, useful if spectrum has an offset.
            raw2_shift = raw - np.min(raw) if raw.size else raw
            raw2 = np.clip(raw2_shift, 0.0, None)
            s2 = float(raw2.sum())

            if s1 < 1e-10 and s2 < 1e-10:
                continue

            for rawv, sv in ((raw1, s1), (raw2, s2)):
                if sv <= 1e-10:
                    continue
                rawv = rawv / sv
                for t in t_blends:
                    cand = (1.0 - t) * best + t * rawv
                    cs_ = float(cand.sum())
                    if cs_ < 1e-10:
                        continue
                    cand = cand / cs_
                    scv = seq_score(cand)
                    if scv < best_sc:
                        best_sc = scv
                        best = cand.copy()
                        improved = True
                # Also use the component itself, which is often strongest.
                sc_raw = seq_score(rawv)
                if sc_raw < best_sc:
                    best_sc = sc_raw
                    best = rawv.copy()
                    improved = True

        a = best

        if improved:
            stall = 0
        else:
            stall += 1
            reduce_score *= 0.80
            if reduce_score < 1e-7 or cap <= avgM * 1.001:
                break
            if stall > 50:
                break

    if seq_score(a) > best_sc:
        a = best
    return a


def softmax_stable(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.zeros_like(x)
    x = x - float(np.max(x))
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return ex / s


def lse_obj_a(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a (unconstrained a).
    """
    n = a.size
    if n == 0:
        return 1e10, np.empty(0)
    S = float(a.sum())
    if S < 1e-14 or not np.isfinite(S):
        return 1e10, np.zeros(n, dtype=np.float64)

    c = conv_fft(a)
    if c.size == 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    cmax = float(np.max(c))
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        p = np.full(c.size, 1.0 / c.size, dtype=np.float64)
        Lval = cmax
    else:
        p = e / se
        Lval = cmax + float(np.log(se)) / beta

    score = 2.0 * n * Lval / (S * S)
    if not np.isfinite(score):
        return 1e10, np.zeros(n, dtype=np.float64)

    a_rev = a[::-1]
    M = 2 * n - 1
    N = fft_len(M)
    PA = np.fft.rfft(p, N)
    RA = np.fft.rfft(a_rev, N)
    convp = np.fft.irfft(PA * RA, N)[:M]
    gL = 2.0 * convp[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    g = (2.0 * n) * (gL / S2 - 2.0 * Lval / S3)

    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return float(score), g


def lse_s_fun(logit, beta):
    """Objective in log-softmax parameters."""
    a = softmax_stable(logit)
    f, ga = lse_obj_a(a, beta)
    if not np.isfinite(f):
        return 1e10, np.zeros_like(logit, dtype=np.float64)

    dot = float(np.dot(ga, a))
    gs = a * (ga - dot)

    # Tiny ridge to fix the constant-offset redundancy in softmax.
    reg = 1e-12
    mean = float(np.mean(logit)) if logit.size else 0.0
    lm = logit - mean
    f += reg * float(np.dot(lm, lm))
    gs = gs + 2.0 * reg * lm

    if not np.all(np.isfinite(gs)):
        gs = np.zeros_like(logit, dtype=np.float64)

    return float(f), gs


def run_lbfgs_logit(a, beta, maxiter=1000, maxfun=5000):
    """L-BFGS-B polish on log-softmax parameters."""
    if scipy_minimize is None:
        a = normalize_nonnegative(a)
        return seq_score(a), a

    a = normalize_nonnegative(a)
    n = a.size
    if n == 0:
        return float("inf"), np.ones(1)

    a_pos = np.maximum(a, 1e-300)
    logit0 = np.log(a_pos / max(1e-300, float(a_pos.mean())))
    logit0 = logit0 - float(np.mean(logit0))

    try:
        res = scipy_minimize(
            lse_s_fun,
            logit0,
            args=(beta,),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": int(max(100, maxiter)),
                "maxfun": int(max(200, maxfun)),
                "ftol": 1e-14,
                "gtol": 1e-11,
                "maxls": 20,
            }
        )
        logit = np.asarray(res.x, dtype=np.float64)
        if not np.all(np.isfinite(logit)):
            logit = logit0
        a_new = softmax_stable(logit)
    except Exception:
        a_new = a

    a_new = normalize_nonnegative(a_new)
    return seq_score(a_new), a_new


def choose_active_indices(c, m_max, m_min, rel_band=0.03):
    """Choose active convolution peaks: near-max band, limited by max count."""
    M = float(np.max(c))
    thresh = M * (1.0 - rel_band)
    idx = np.where(c >= thresh)[0]
    if idx.size < m_min:
        idx = np.argsort(c)[::-1][:m_min]
    if idx.size > m_max:
        # Keep the largest among the near-max band.
        vals = c[idx]
        order = np.argsort(vals)[::-1][:m_max]
        idx = idx[order]
    return idx


def active_peak_descent(a, max_time, top_mins=(10, 30, 80, 200, 400),
                        alphas=(0.0004, 0.0009, 0.0018, 0.0035, 0.007)):
    """
    Active-peak subgradient polishing:
        v_j = sum_{active k} a_{k-j}
        direction d = -alpha * v_centered
    """
    if max_time <= 1.0:
        return normalize_nonnegative(a)

    a = normalize_nonnegative(a)
    n = a.size
    best = a.copy()
    best_sc = seq_score(a)

    t_start = time.monotonic()
    stall = 0

    while (time.monotonic() - t_start) < max_time:
        c = conv_fft(a)
        M = float(np.max(c))
        if M <= 0 or not np.isfinite(M):
            break

        improved = False

        for m in top_mins:
            m_use = int(min(max(1, m), max(1, c.size)))
            idx = choose_active_indices(c, m_use, min(m_use, 5), rel_band=0.03)
            if idx.size == 0:
                continue

            v = np.zeros(n, dtype=np.float64)
            for k in idx:
                k = int(k)
                start = max(0, k - n + 1)
                end = min(k, n - 1)
                if end < start:
                    continue
                # v[j] += a[k-j]
                v[start:end + 1] += a[k - end:k - start + 1][::-1]

            v -= float(np.mean(v))
            nv = float(np.linalg.norm(v))
            if nv < 1e-14:
                continue

            scale = float(np.linalg.norm(a)) / nv

            for alpha in alphas:
                d = -alpha * scale * v
                neg = d < -best
                if np.any(neg):
                    # Keep a safety factor.
                    ratio = np.min((~best)[neg] / (-d[neg])) if ((~best)[neg].size) else 1.0
                    ratio = max(0.0, float(ratio)) * 0.999
                    d *= ratio

                cand = best + d
                cs = float(cand.sum())
                if cs < 1e-12:
                    continue
                cand = cand / cs
                sc = seq_score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best = cand.copy()
                    improved = True

        a = best

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break

    return best


def resize_sequence(old, new_n):
    """Linear resampling of a normalized nonnegative sequence."""
    old = np.asarray(old, dtype=np.float64).ravel()
    old_n = old.size
    new_n = int(max(1, new_n))
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == new_n:
        return old.copy()

    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return normalize_nonnegative(new)

    # ------------------------------------------------------------------
    # Phase 1: direct spectral water-filling on the best current length.
    # ------------------------------------------------------------------
    phase1_end = t0 + 0.48 * (deadline - t0)
    if phase1_end > t0:
        rem1 = phase1_end - time.monotonic()
        if rem1 > 30:
            target1 = max(1.5000, best_sc - 0.0045)
            a1 = spectral_optimize(
                best,
                target_score=target1,
                max_time=rem1 - 10.0,
                mode="ratio",
                reduce_score=0.00035,
                alphas=(0.18, 0.40, 0.70),
                t_blends=(0.35, 0.65, 1.0)
            )
            sc1 = seq_score(a1)
            if sc1 < best_sc:
                best_sc = sc1
                best = a1
                print(f"Spectral ratio improved: {sc1:.10f}, n={best.size}")
            else:
                print(f"Spectral ratio final: {sc1:.10f}, n={a1.size}")

    phase1b_end = t0 + 0.62 * (deadline - t0)
    if phase1b_end > time.monotonic():
        rem1b = phase1b_end - time.monotonic()
        if rem1b > 30:
            target1b = max(1.5000, best_sc - 0.0035)
            a1b = spectral_optimize(
                best,
                target_score=target1b,
                max_time=rem1b - 10.0,
                mode="blend",
                reduce_score=0.00025,
                alphas=(0.20, 0.50),
                t_blends=(0.50, 1.0)
            )
            sc1b = seq_score(a1b)
            if sc1b < best_sc:
                best_sc = sc1b
                best = a1b
                print(f"Spectral blend improved: {sc1b:.10f}, n={best.size}")
            else:
                print(f"Spectral blend final: {sc1b:.10f}, n={a1b.size}")

    # ------------------------------------------------------------------
    # Phase 2: LSE L-BFGS polish.
    # ------------------------------------------------------------------
    phase2_end = t0 + 0.84 * (deadline - t0)
    if phase2_end > time.monotonic() and scipy_minimize is not None:
        for beta in [2500, 8000, 25000, 75000]:
            rem = phase2_end - time.monotonic()
            if rem < 120:
                break
            maxiter = int(min(2500, max(300, rem * 1.6)))
            maxfun = int(min(12000, max(1000, rem * 8)))
            sc, a2 = run_lbfgs_logit(best, beta, maxiter=maxiter, maxfun=maxfun)
            if sc < best_sc:
                best_sc = sc
                best = a2
                print(f"LBFGS beta={beta} improved: {sc:.10f}, n={best.size}")
            else:
                print(f"LBFGS beta={beta} final: {sc:.10f}")
            if time.monotonic() > phase2_end:
                break

    # ------------------------------------------------------------------
    # Phase 3: active-peak subgradient polishing.
    # ------------------------------------------------------------------
    phase3_end = t0 + 0.93 * (deadline - t0)
    if phase3_end > time.monotonic():
        a3 = active_peak_descent(best, max_time=phase3_end - time.monotonic() - 15.0)
        sc3 = seq_score(a3)
        if sc3 < best_sc:
            best_sc = sc3
            best = a3
            print(f"Active-peak improved: {sc3:.10f}, n={best.size}")
        else:
            print(f"Active-peak final: {sc3:.10f}, n={a3.size}")

        # A short L-BFGS after active polish can squeeze a little more.
        rem = deadline - time.monotonic() - 90.0
        if rem > 120 and scipy_minimize is not None:
            sc, a4 = run_lbfgs_logit(best, 20000, maxiter=800, maxfun=4000)
            if sc < best_sc:
                best_sc = sc
                best = a4
                print(f"Final LBFGS improved: {sc:.10f}, n={best.size}")

    # ------------------------------------------------------------------
    # Phase 4: multi-length escape (optional, bounded by remaining time).
    # ------------------------------------------------------------------
    rem = deadline - time.monotonic() - 35.0
    if rem > 180 and best.size <= 20000:
        big_n = min(2 * best.size, 32768)
        if big_n > best.size:
            a_big = resize_sequence(best, big_n)
            sc_big = seq_score(a_big)
            if sc_big < best_sc:
                best_sc = sc_big
                best = a_big
            target_big = max(1.5000, best_sc - 0.0025)
            a_big = spectral_optimize(
                best,
                target_score=target_big,
                max_time=rem - 25.0,
                mode="ratio",
                reduce_score=0.00025,
                alphas=(0.25, 0.55),
                t_blends=(0.50, 1.0)
            )
            sc_big = seq_score(a_big)
            if sc_big < best_sc:
                best_sc = sc_big
                best = a_big
                print(f"Big-length spectral improved: {sc_big:.10f}, n={best.size}")
            else:
                print(f"Big-length spectral final: {sc_big:.10f}, n={a_big.size}")

    # ------------------------------------------------------------------
    # Final safety and output.
    # ------------------------------------------------------------------
    best = np.asarray(best, dtype=np.float64).ravel()
    best = np.clip(best, 0.0, 1000.0)
    ss = float(best.sum())
    if ss < 0.05 or not np.isfinite(ss):
        n_out = max(1024, best.size if np.isfinite(best.size) else 1024)
        best = np.ones(n_out, dtype=np.float64) / n_out
    else:
        # Return normalized; score is scale-invariant and all values are small.
        best = best / ss

    out = [float(max(0.0, min(1000.0, x))) for x in best]

    internal = seq_score(np.asarray(out, dtype=np.float64))
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float("inf")

    elapsed = time.monotonic() - t0
    print("=" * 70)
    print(f"FINAL internal score: {internal:.12f}")
    print(f"FINAL verified score: {verified:.12f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget:.0f}s")
    print("=" * 70)

    return out
