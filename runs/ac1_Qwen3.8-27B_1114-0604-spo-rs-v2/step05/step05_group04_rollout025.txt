def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def spectral_flatten_soft(a, alpha, iterations=100):
    """
    Soft spectral flattening using quadratic compression.
    P_new = P - alpha * (P - P_mean) * max(0, P/P_mean - 1)
    This compresses bins above mean more aggressively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(iterations):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Quadratic compression of high bins
        ratio = P / P_mean
        excess = np.maximum(ratio - 1.0, 0.0)
        # P_new = P - alpha * P_mean * excess^2 (compresses quadratically)
        P_new = P - alpha * P_mean * excess * excess
        P_new = np.maximum(P_new, 1e-20)
        
        # Also apply linear compression for very high bins
        excess2 = np.maximum(ratio - 2.0, 0.0)
        P_new = P_new - alpha * 0.5 * P_mean * excess2 * excess2
        
        amp_ratio = np.sqrt(P_new / P)
        A_mod = A * amp_ratio
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())

        a_new = a_new / S_new
        a = a_new

def lse_objective_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

    """
    Spectral optimization with momentum in the time domain.
    The momentum helps escape flat regions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
        S = 1.0
    a = a / S

    v = np.zeros(n)  # momentum
    
    for it in range(n_iters):

        gamma = gamma_init + (gamma_end - gamma_init) * progress
        
        A = np.fft.rfft(a, M_fft)
        P = np.maximum(A.real**2 + A.imag**2, 1e-30)

        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    
    return best_a

    """

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break

    """Objective in log-space: a = exp(b) / sum(exp(b)) = softmax(b)"""
    b_max = b.max()
    log_a = b - b_max
    a = np.exp(log_a)
    a = a / a.sum()
    
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    # sum(a) = 1, so score = 2n * max(c)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_vals = np.exp(scaled)
    exp_sum = exp_vals.sum()

    score = 2.0 * n * L  # since sum(a) = 1
    
    # Gradient w.r.t. b
    # da/db = a * (I - outer(a, 1)) / sum(a) but since sum(a)=1 already via normalization...
    # Actually softmax: a_i = exp(b_i) / sum_j exp(b_j)
    # da_i/db_j = a_i * (delta_ij - a_j)

    g_a = 2.0 * conv_res  # dL/da_i
    
    # score = 2n * L (since S=1)
    # d(score)/da_i = 2n * g_a_i
    # d(score)/db_j = sum_i d(score)/da_i * da_i/db_j
    # da_i/db_j = a_i * (delta_ij - a_j)

    # = 2n * (g_a_j * a_j - a_j * sum_i g_a_i * a_i)
    
    ga_a = float(np.sum(g_a * a))
    grad_b = 2.0 * n * (g_a * a - a * ga_a)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_a = 2.0 * conv_res

    ga_a = float(np.dot(g_a, a))
    grad_b = 2.0 * n * (g_a * a - a * ga_a)

    return score, grad_b


def run_lbfgs_std(a0, beta, maxiter=5000):
    """L-BFGS with standard parameterization and bounds."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_grad_std, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def run_lbfgs_log(a0, beta, maxiter=5000):
    """L-BFGS with log-space (softmax) parameterization - no bounds needed."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
        S = n
    a_norm = a0 / S
    b0 = np.log(a_norm) - np.log(a_norm).mean()  # center for numerical stability

    res = minimize(
        lse_grad_log, b0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 40}
    )
    b_out = res.x
    b_shifted = b_out - b_out.max()
    a_out = np.exp(b_shifted)
    a_out = a_out / a_out.sum()
    a_out = np.clip(a_out, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def spectral_anneal_perturb(a, n_iters=500, gamma_start=0.95, gamma_end=0.2,
                            perturb_every=100, perturb_scale=0.05, rng=None):
    """
    Spectral annealing with periodic phase perturbations to escape local minima.
    Uses power-law: P_new = P^gamma * P_mean^(1-gamma)
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    stall_count = 0
    stall_threshold = 200

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        gamma = gamma_end + (gamma_start - gamma_end) * 0.5 * (1 + np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = np.maximum(A.real ** 2 + A.imag ** 2, 1e-30)
        P_mean = float(P.mean())

        P_new = np.exp(gamma * np.log(P) + (1 - gamma) * np.log(P_mean))

        amp = np.sqrt(P_new / P)
        A_mod = A * amp

        # Periodic phase perturbation
        if (it + 1) % perturb_every == 0:
            phase_perturb = rng.standard_normal(len(A_mod)) * perturb_scale
            A_mod = A_mod * np.exp(1j * phase_perturb)

        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart if stalled
        if stall_count >= stall_threshold:
            a = best_a * (1.0 + 0.03 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall_count = 0

    return best_a


def subgradient_nesterov(a, n_iters=5000, top_m=20, mu=0.9):
    """
    Subgradient method with Nesterov momentum and Polyak step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    v = np.zeros(n)
    step_base = 0.015

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step_base, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.04)

        v = mu * v + step * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

    return best_a


def alternating_spectral_subgrad(a, n_rounds=50, spec_iters=20, sg_iters=50,
                                 gamma=0.6, top_m=20):
    """
    Alternates between spectral flattening and subgradient steps.
    This can escape local minima that each method alone cannot.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for rnd in range(n_rounds):
        # Spectral step
        for _ in range(spec_iters):
            A = np.fft.rfft(a, M_fft)
            P = np.maximum(A.real ** 2 + A.imag ** 2, 1e-30)
            P_mean = float(P.mean())
            P_new = np.exp(gamma * np.log(P) + (1 - gamma) * np.log(P_mean))
            amp = np.sqrt(P_new / P)
            A_mod = A * amp
            a_new = np.fft.irfft(A_mod, M_fft)[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a = a_new / S_new

        # Subgradient steps
        v = np.zeros(n)
        for _ in range(sg_iters):
            c = fast_conv(a)
            S = float(a.sum())
            M_val = float(c.max())
            sc = 2.0 * n * M_val / (S * S)

            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            if top_m >= len(c):
                top_idx = np.arange(len(c))
            else:
                top_idx = np.argsort(c)[::-1][:top_m]

            sg = np.zeros(n)
            for k in top_idx:
                k = int(k)
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                if j_end >= j_start:
                    seg = a[j_start:j_end + 1]
                    sg[j_start:j_end + 1] += 2.0 * seg[::-1]

            S2 = S * S
            S3 = S2 * S
            grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break

            step = min(0.01, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
            step = min(step, 0.03)
            v = 0.85 * v + step * grad
            a_new = a - v
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a = a_new / S_new

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage optimization combining spectral annealing, log-space LSE,
    Nesterov subgradient, and alternating spectral-subgradient method.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        n_best = len(parent)
    else:
        n_best = 6962

    print(f"Initial best: {best_score:.8f}, n={n_best}")

    # ==================================================================
    # Phase 1: Extensive spectral annealing with perturbations (25%)
    # ==================================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Spectral annealing + perturb (t={phase1_end - t_start:.0f}s) ===")

    n = n_best
    a_start = parent if parent is not None else best_a

    # Run multiple spectral annealing variants
    configs = [
        {"gamma_start": 0.95, "gamma_end": 0.3, "perturb_every": 80, "perturb_scale": 0.06, "n_iters": 600},
        {"gamma_start": 0.9, "gamma_end": 0.2, "perturb_every": 120, "perturb_scale": 0.04, "n_iters": 800},
        {"gamma_start": 0.98, "gamma_end": 0.4, "perturb_every": 60, "perturb_scale": 0.08, "n_iters": 500},
    ]

    for idx, cfg in enumerate(configs):
        if time.monotonic() > phase1_end:
            break
        a_out = spectral_anneal_perturb(a_start, rng=np.random.default_rng(seed + idx), **cfg)
        sc = compute_score(a_out)
        update_best(sc, a_out, f"P1-spec-{idx}")
        # Chain: use output as input for next
        a_start = a_out

    print(f"After Phase 1: {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE refinement with adaptive beta (30%)
    # ==================================================================
    phase2_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LSE adaptive beta (t={phase2_end - t_start:.0f}s) ===")

    # Standard LSE with increasing beta
    beta_schedule = [30, 100, 300, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(8000, int(remaining / 30))
        sc, a_opt = run_lbfgs_std(best_a, beta, maxiter=maxiter)
        update_best(sc, a_opt, f"P2-LSE-std b={beta}")

    # Log-space LSE (no bounds - potentially better conditioned)
    for beta in [100, 500, 2000]:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(6000, int(remaining / 30))
        sc, a_opt = run_lbfgs_log(best_a, beta, maxiter=maxiter)
        update_best(sc, a_opt, f"P2-LSE-log b={beta}")

    print(f"After Phase 2: {best_score:.8f}")

    # ==================================================================
    # Phase 3: Nesterov subgradient + alternating (25%)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Nesterov subgradient (t={phase3_end - t_start:.0f}s) ===")

    for top_m in [10, 20, 40]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        n_iters = min(5000, int(remaining / 15))
        a_sg = subgradient_nesterov(best_a, n_iters=n_iters, top_m=top_m, mu=0.9)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"P3-SG m={top_m}")

    # Alternating spectral-subgradient
    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        n_rounds = max(10, int(remaining / 20))
        a_alt = alternating_spectral_subgrad(best_a, n_rounds=n_rounds,
                                             spec_iters=15, sg_iters=40,
                                             gamma=0.5, top_m=20)
        sc = compute_score(a_alt)
        update_best(sc, a_alt, "P3-ALT")

    print(f"After Phase 3: {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final high-beta LSE + polish (20%)
    # ==================================================================
    print(f"\n=== Phase 4: Final refinement ===")

    for beta in [5000, 10000, 20000]:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic() - 30
        if remaining < 60:
            break
        maxiter = min(10000, int(remaining / 30))
        sc, a_opt = run_lbfgs_std(best_a, beta, maxiter=maxiter)
        update_best(sc, a_opt, f"P4-LSE b={beta}")

    # Final log-space with very high beta
    if time.monotonic() < deadline - 60:
        sc, a_opt = run_lbfgs_log(best_a, 5000, maxiter=4000)
        update_best(sc, a_opt, "P4-LSE-log-5k")

    # Quick spectral polish
    if time.monotonic() < deadline - 30:
        a_fin = spectral_anneal_perturb(best_a, n_iters=100,
                                        gamma_start=0.5, gamma_end=0.3,
                                        perturb_every=50, perturb_scale=0.02,
                                        rng=np.random.default_rng(seed + 99))
        sc = compute_score(a_fin)
        update_best(sc, a_fin, "P4-spec-polish")

        if time.monotonic() < deadline - 20:
            sc2, a_opt2 = run_lbfgs_std(a_fin, 10000, maxiter=3000)
            update_best(sc2, a_opt2, "P4-final")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
