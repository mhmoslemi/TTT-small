"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines convolution-domain Fienup, spectral-domain Fienup, and L-BFGS refinement.
Key innovations:
1. Direct convolution-domain Fienup (operates on conv(a,a) directly)
2. Adaptive multi-level compression
3. Phase-preserving spectral reconstruction
4. Careful L-BFGS beta scheduling with warm-starting
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def conv_domain_fienup(a, n_iters=500, top_frac=0.15, gamma=0.5, mix_rate=1.0):
    """
    Fienup iterations that directly modify the convolution.
    - Forward: a -> c = conv(a,a)
    - Modify: compress the top of c
    - Inverse: c -> a via A = sqrt(FFT(c)), a = IFFT(A)
    - Constraint: a >= 0, normalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Forward transform: compute convolution
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        
        # Modify convolution: compress peaks
        thresh = np.quantile(c, 1.0 - top_frac)
        c_mod = c.copy()
        above = c_mod > thresh
        if above.any():
            c_mod[above] = thresh + (c_mod[above] - thresh) * gamma
        
        # Inverse: compute new a from modified convolution
        # C = FFT(c_mod), A_new = sqrt(C), a_new = IFFT(A_new)
        C = np.fft.fft(c_mod, M_fft)
        # Complex square root
        A_new = np.sqrt(C)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Mixing
        if mix_rate < 1.0:
            a = (1.0 - mix_rate) * a + mix_rate * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def spectral_fienup(a, n_iters=500, top_frac_init=0.1, top_frac_end=0.5, gamma=0.8):
    """
    Spectral-domain Fienup: modify |A| to flatten A^2.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))
        
        A = np.fft.rfft(a, M_fft)
        # |A|^2 determines the power spectrum
        # We want A^2 to be flat, which means |A| should be flat
        P = A.real**2 + A.imag**2  # |A|^2
        
        thresh = np.quantile(P, 1.0 - top_frac)
        P_mod = P.copy()
        above = P_mod > thresh
        if above.any():
            P_mod[above] = thresh + (P_mod[above] - thresh) * gamma
        
        # Update amplitude, preserve phase
        scale = np.sqrt(P_mod / np.maximum(P, 1e-30))
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. x
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def proportional_fienup(a, n_iters=500, target_peak_frac=0.3):
    """
    Fienup that targets a specific peak/mean ratio for the convolution.
    The peak of conv should be target_peak_frac of the mean (times (2n-1)).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        
        c_mean = float(c.mean())
        target_peak = c_mean / target_peak_frac  # We want max(c) / mean(c) <= 1/target_peak_frac
        # Actually, we want max(c) to be as close to c_mean as possible
        # So target: max(c) should be at most some multiple of c_mean
        
        # Use a "soft" threshold: values above target are compressed
        above = c > target_peak
        if above.any():
            c_mod = c.copy()
            # Compress: c_new = target + (c - target)^0.5 * scale
            # Or simply: c_new = target * (c/target)^0.5
            c_mod[above] = target_peak * np.sqrt(c[above] / target_peak)
        else:
            c_mod = c
        
        # Inverse
        C = np.fft.fft(c_mod, M_fft)
        A_new = np.sqrt(C)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    # Standard inits
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.exp(-2.0 * (2.0 * x - 1.0)**2) + 0.05)
    inits.append(0.5 + 0.3 * rng.standard_normal(n))

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def conv_domain_fienup(a, n_iters=500, top_frac=0.15, gamma=0.4, mix_rate=1.0, adaptive=True):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    stagnant = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]

        # Adaptive top fraction: starts narrow, widens over time
        if adaptive:
            progress = it / max(1, n_iters)
            tf = top_frac * (0.5 + 0.5 * progress)
        else:
            tf = top_frac

        thresh = np.quantile(c, 1.0 - tf)
        c_mod = c.copy()
        above = c_mod > thresh
        if above.any():
            c_mod[above] = thresh + (c_mod[above] - thresh) * gamma

        # Inverse: complex FFT, square root, IFFT
        C = np.fft.fft(c_mod, M_fft)
        A_new = np.sqrt(C)
        a_new = np.fft.ifft(A_new).real[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        if mix_rate < 1.0:
            a = (1.0 - mix_rate) * a + mix_rate * a_new
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                break
            a = a / s
        else:
            a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break

    return best_a


def spectral_fienup(a, n_iters=500, top_frac_init=0.08, top_frac_end=0.6, gamma=0.7,
                    soft_alpha=30.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    stagnant = 0

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1.0 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2

        thresh = np.quantile(P, 1.0 - top_frac)
        P_mod = P.copy()
        above = P_mod > thresh
        if above.any():
            P_mod[above] = thresh + (P_mod[above] - thresh) * gamma

        scale = np.sqrt(P_mod / np.maximum(P, 1e-30))
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        # Soft non-negativity via sigmoid gate
        if soft_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break

    return best_a


def proportional_fienup(a, n_iters=500, target_ratio=0.4, power=0.5):
    """Target: max(c)/mean(c) <= 1/target_ratio."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    stagnant = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]

        c_mean = float(c.mean())
        target_peak = c_mean / target_ratio

        c_mod = c.copy()
        above = c_mod > target_peak
        if above.any():
            c_mod[above] = target_peak * (c[above] / target_peak) ** power

        C = np.fft.fft(c_mod, M_fft)
        A_new = np.sqrt(C)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break

    return best_a


def lse_objective_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

            for scale in [0.005, 0.02, 0.05, 0.1]:
                pert = parent * (1.0 + scale * rng.standard_normal(n))
                pert = np.clip(pert, 0.0, 1000.0)
                if pert.sum() > 1e-6:
                    inits.append(pert / pert.sum())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, parent)
            if interp.sum() > 1e-6:
                inits.append(interp / interp.sum())

    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0)**2) + 0.05)
    inits.append(x**0.3 + 0.1)
    inits.append((1.0 - x)**0.3 + 0.1)
    inits.append(np.maximum(0.5 + 0.3 * rng.standard_normal(n), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * x)

    cleaned = []
    for a in inits:
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() > 1e-6:
            cleaned.append(a / a.sum())
    return cleaned
