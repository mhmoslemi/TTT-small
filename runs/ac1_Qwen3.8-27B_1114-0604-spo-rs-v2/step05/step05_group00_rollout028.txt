"""
Optimized step function generator using adaptive Fienup, LSE-smoothed L-BFGS,
and peak-targeted coordinate descent. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_smoothed_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dc_k/dx_j involves a_{k-j} and a_{j-k}
    # Use FFT for efficiency
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    """L-BFGS-B on LSE-smoothed objective."""

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    
    # Time-aware callback
    stop_flag = [False]
    def callback(xk):
        if time_limit is not None:

                stop_flag[0] = True
        return stop_flag[0]

    best_a = a.copy()

            if a_j >= a_p:
                # Try reducing a_j
                for factor in [0.7, 0.8, 0.9]:
                    new_val = a_j * factor
                    a[j] = new_val
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_j
                        continue
                    sc = compute_score(a / S_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = (a / S_tmp).copy()
                        a = a / S_tmp
                        improved = True
                        break

            else:

def multi_length_search(best_a_start, budget_remaining, t_start, deadline):
    """
    Try using the best sequence as a template for different lengths.
    Interpolate to new lengths and optimize.
    """
    results = []
    lengths_to_try = [8000, 10000, 12000]
    
    for new_n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        # Interpolate
        x_old = np.linspace(0, 1, len(best_a_start))
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, best_a_start)
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S
        
        # Quick Fienup
        a_f = fienup_adaptive(a_new, n_iters=200, time_limit=deadline - time.monotonic(), t_start=t_start)
        sc = compute_score(a_f)
        print(f"  Length {new_n}: score={sc:.8f}")
        results.append((sc, a_f))
        
        # Quick L-BFGS if time
        if time.monotonic() < deadline - 120 and len(results) < 3:
            remaining = min(120, deadline - time.monotonic() - 30)
            sc2, a_opt = run_lbfgs(a_f, 1000, maxiter=3000, time_limit=remaining, t_start=t_start)
            print(f"  Length {new_n} + LBFGS: score={sc2:.8f}")
            if sc2 < sc:
                results[-1] = (sc2, a_opt)
    
    if results:
        results.sort(key=lambda x: x[0])
        return results[0][1]
    return None


def spectral_perturb(a, rng, n_modes=10, scale=0.05):
    """Perturb in the frequency domain."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    
    # Add perturbation to amplitude and phase
    for m in range(n_modes):
        k0 = rng.integers(0, len(A))
        dk = rng.integers(-3, 4)
        k1 = min(k0 + dk, len(A) - 1)
        max_k = max(k0, k1)
        min_k = min(k0, k1)
        
        amp_scale = 1.0 + scale * rng.standard_normal()
        phase_shift = scale * rng.standard_normal()
        
        A[min_k:max_k+1] *= amp_scale
        A[min_k:max_k+1] *= np.exp(1j * phase_shift)
    
    a_new = np.fft.irfft(A, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.03):
    """Multiplicative Gaussian noise."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: DC-aware Fienup, adaptive LSE-smoothed L-BFGS with temperature scheduling,
peak-targeted coordinate descent, multi-length exploration, and spectral restarts.
Key innovation: treats the DC spectral component explicitly and uses
annealing-style beta scheduling for the LSE approximation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via FFT: dLSE/dx_j = 2 * sum_k p_k * x_{k-j} (cross-correlation)
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """L-BFGS-B on LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    stop = [False]

    def _cb(xk):
        if deadline is not None and time.monotonic() > deadline:
            stop[0] = True
        return stop[0]

    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50},
        callback=_cb
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def fienup_dc_aware(a, n_iters=500, deadline=None):
    """
    Fienup with DC-aware target spectrum.
    Keeps X(0) fixed and flattens the rest with adaptive relaxation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_safe = np.maximum(P, 1e-30)

        progress = it / max(1, n_iters)
        alpha = 0.05 + 0.85 * (1.0 - np.exp(-3.0 * progress))

        # DC-aware target: keep X(0) large, flatten the rest
        P_mean_nonDC = float(P[1:].mean()) if len(P) > 1 else P[0]
        P_target = np.full_like(P, max(P_mean_nonDC, 1e-10))
        P_target[0] = P[0]  # Keep DC fixed

        P_new = (1.0 - alpha) * P + alpha * P_new_target(P, P_target, n)
        P_new[0] = P[0]  # Always keep DC
        P_new = np.maximum(P_new, 1e-30)

        scale = np.sqrt(P_new / P_safe)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def P_new_target(P, P_target, n):
    """Smooth target that tapered at high frequencies slightly."""
    k = np.arange(len(P))
    taper = 1.0 - 0.1 * (k / max(len(P) - 1, 1)) ** 2
    return P_target * taper


def fienup_aggressive(a, n_iters=400, deadline=None):
    """
    Aggressive Fienup: hard-threshold the power spectrum at each iteration.
    Strongly suppresses high-power spectral components.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_safe = np.maximum(P, 1e-30)

        # Progressive threshold: start loose, get tighter
        progress = it / max(1, n_iters)
        P_mean = float(P[1:].mean()) if len(P) > 1 else float(P[0])
        thresh_mult = 2.5 - 1.5 * progress  # From 2.5x to 1.0x the mean
        thresh = max(P_mean * thresh_mult, P[0] * 0.1)

        P_new = P.copy()
        above = P > thresh
        above[0] = False  # Never threshold DC
        soft = 0.3 + 0.5 * (1.0 - progress)
        P_new[above] = thresh + (P[above] - thresh) * (1.0 - soft)
        P_new = np.maximum(P_new, 1e-30)
        P_new[0] = P[0]

        scale = np.sqrt(P_new / P_safe)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_targeted_cd(a, n_sweeps=30, n_top=80, deadline=None):
    """
    Coordinate descent: identify top contributors to the peak lag and
    try multiplicative adjustments to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        top_count = min(n_top, len(contribs))
        order = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + order

        cur_sc = best_sc
        for j in top_global:
            j = int(j)
            partner = k_star - j
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue

            candidates = [a_old_j * f for f in [0.5, 0.65, 0.75, 0.85, 0.92, 1.1, 1.25]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            if best_val != a_old_j:
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                    cur_sc = best_local

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if best_sc <= 2.0 * n / (2.0 * n - 1.0) + 1e-6:
            break

    return best_a


def spectral_perturb(a, rng, n_bands=6, scale=0.04):
    """Perturb in frequency domain: random amplitude/phase shifts in bands."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    band_size = max(1, len(A) // (n_bands * 2))

    for _ in range(n_bands):
        k0 = int(rng.integers(1, len(A) - band_size))
        k1 = min(k0 + band_size, len(A))
        amp = 1.0 + scale * rng.standard_normal()
        ph = scale * rng.standard_normal()
        A[k0:k1] *= amp * np.exp(1j * ph)

    a_new = np.fft.irfft(A, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):

def struct_perturb(a, rng, n_harmonics=5):
    """Add low-frequency harmonic structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)

        freq = int(rng.integers(1, 12))

        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)

    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    if len(a) < 2:
        return np.full(new_n, float(a[0]) if len(a) == 1 else 0.5)
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def anneal_lbfgs(a0, deadline):
    """
    Run L-BFGS with annealing beta: start low (smooth landscape)
    and increase (sharper max approximation). This mimics simulated
    annealing and helps escape shallow local minima.
    """
    n = len(a0)
    a_cur = a0.copy()
    best_a = a0.copy()
    best_sc = compute_score(a0)

    # Beta schedule: geometric progression
    betas = [50, 150, 400, 1000, 3000, 8000, 20000, 50000, 100000]
    # Allocate time proportionally (later bets get more time)
    total_weight = sum(np.sqrt(b) for b in betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        weight = np.sqrt(beta) / total_weight
        time_alloc = weight * (deadline - time.monotonic())
        time_alloc = max(10.0, min(time_alloc, 300.0))
        sub_deadline = time.monotonic() + time_alloc
        maxiter = int(max(500, time_alloc * 20))
        maxfun = maxiter * 15

        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun,
                              deadline=sub_deadline)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    anneal beta={beta}: {sc:.8f}")
        a_cur = a_opt.copy()

    return best_a, best_sc

    n_candidates = []
    for delta in [-0.1, 0.1, 0.2, -0.2, 0.3]:
        new_n = int(current_n * (1.0 + delta))
        new_n = max(1024, min(20000, new_n))
        if new_n != current_n and new_n not in n_candidates:
            n_candidates.append(new_n)
    n_candidates = n_candidates[:3]  # Limit to 3 extra lengths

    for new_n in n_candidates:
        if time.monotonic() > phase4_end - 60:
            break
        a_interp = interpolate_to_length(best_a, new_n)
        a_interp = np.maximum(a_interp, 0.0)
        S = a_interp.sum()
        if S < 1e-12:
            continue
        a_interp = a_interp / S

        # Quick Fienup on new length
        a_f = fienup_dc_aware(a_interp, n_iters=200,
                              deadline=min(phase4_end - 30, time.monotonic() + 60))
        sc = compute_score(a_f)
        print(f"  n={new_n}: Fienup score={sc:.8f}")

        # Brief L-BFGS
        if time.monotonic() < phase4_end - 40 and sc < best_score + 0.01:
            sc2, a_opt2 = run_lbfgs(a_f, 2000, maxiter=2000, maxfun=30000,
                                    deadline=phase4_end - 5)
            print(f"  n={new_n}: +LBFGS score={sc2:.8f}")
            if sc2 < sc:
                sc = sc2
                a_f = a_opt2

        update(sc, a_f, f"LEN-{new_n}")

    # ==================================================================
    # Phase 5: Diversified restarts (13% of time)
    # ==================================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Restarts ===")

    n_restarts = 15
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 25:
            break

        scale = 0.015 + 0.004 * r
        if r % 4 == 0:
            a_pert = spectral_perturb(best_a, rng, n_bands=4, scale=scale)
        elif r % 4 == 1:
            a_pert = noise_perturb(best_a, rng, scale=scale)
        elif r % 4 == 2:
            a_pert = struct_perturb(best_a, rng, n_harmonics=4)
        else:
            a_pert = noise_perturb(best_a, rng, scale=scale * 0.5)
            a_pert = spectral_perturb(a_pert, rng, n_bands=3, scale=scale * 0.5)

        remaining = phase5_end - time.monotonic() - 25
        if remaining < 20:
            break
        maxiter = min(4000, int(remaining * 15))
        if maxiter < 500:
            maxiter = 500
        betas_to_try = [1000, 5000, 20000]
        beta_use = betas_to_try[r % len(betas_to_try)]

        sc, a_opt = run_lbfgs(a_pert, beta_use, maxiter=maxiter,
                              maxfun=maxiter * 15, deadline=phase5_end - 2)
        label = f"R{r}-b{beta_use}"
        update(sc, a_opt, label)

    # ==================================================================
    # Phase 6: Final deterministic polish
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 15:
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=50000,
                                deadline=deadline - 10)
        update(sc, a_final, "FINAL-50K")

    if time.monotonic() < deadline - 8:
        sc, a_final = run_lbfgs(best_a, 150000, maxiter=2000, maxfun=30000,
                                deadline=deadline - 3)
        update(sc, a_final, "FINAL-150K")

    if time.monotonic() < deadline - 3:
        a_cd_final = peak_targeted_cd(best_a, n_sweeps=8, n_top=120, deadline=deadline - 1)
        sc = compute_score(a_cd_final)
        update(sc, a_cd_final, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")

    return out
