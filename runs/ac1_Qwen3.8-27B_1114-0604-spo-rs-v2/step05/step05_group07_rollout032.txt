import time

# Global state for callbacks

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta, top_k=1):

        return 1e10, np.zeros(n)

    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # dL/da[i] = 2 * (p * rev_a)[n-1+i]

    M2_fft = 1 << (M2 - 1).bit_length()

    S3 = S2 * S

def run_lbfgs(a0, beta, top_k=1, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B with configurable smoothing."""
    _state['beta'] = beta
    _state['top_k'] = top_k
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        objective_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def freq_domain_perturb(a, rng, n_modes=5):
    """Perturb in frequency domain to explore different shapes."""
    n = len(a)
    a_p = a.astype(np.float64).copy()
    
    # Pad to FFT length
    M = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M)
    
    # Perturb random frequency bins (keep DC to preserve sum)
    n_freq = len(A)
    non_dc = np.arange(1, n_freq)
    sel = rng.choice(non_dc, size=min(n_modes, len(non_dc)), replace=False)
    
    A_p = A.copy()
    for f in sel:
        delta_mag = (rng.random() - 0.5) * 0.02 * np.abs(A[f])
        delta_phase = (rng.random() - 0.5) * 0.1
        A_p[f] = (np.abs(A[f]) + delta_mag) * np.exp(1j * (np.angle(A[f]) + delta_phase))
        # Mirror for real signal
        if n_freq - 1 - f >= 0 and n_freq - 1 - f != f:
            A_p[n_freq - 1 - f] = np.conj(A_p[f])
    
    a_p = np.fft.irfft(A_p, M)[:n]
    a_p = np.maximum(a_p, 0.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

def mirror_descent_mlt(a0, beta, n_iters=5000, eta0=0.05, decay=0.9995, top_k=1):
    """Mirror descent with multi-lag tracking."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    
    for it in range(n_iters):
        if top_k == 1:
            sc, grad = lse_grad(a, beta)
        else:
            sc, grad = multi_lag_score_grad(a, top_k=top_k, temperature=1.0/beta)
        
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        # Center gradient
        g_centered = grad - np.mean(grad)
        g_norm = np.linalg.norm(g_centered)
        if g_norm < 1e-15:
            break
        
        # Normalize and apply
        g_normed = g_centered / (g_norm * n)
        exponent = -eta * g_normed
        exponent = np.clip(exponent, -20, 20)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        eta *= decay
    
    return best_a

def active_set_method(a0, n_iters=5000, n_active=20, f_lb=1.4):
    """
    Active set method: maintain a set of "active" lags (those near the maximum)
    and use their weighted gradient for descent.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.05
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        c_max = float(np.max(c))
        sc = 2.0 * n * c_max / (S_a * S_a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Active set: lags with c[k] >= c_max - delta
        delta = 0.01 * c_max
        active = np.where(c >= c_max - delta)[0]
        active = active[:n_active]  # Cap at n_active
        
        # Weighted subgradient
        # For each active lag k, the gradient of c[k] w.r.t. a is 2*a[k-:] (shifted)
        # Weight by how close to max
        weights = c[active] / sum(c[active])
        
        sg = np.zeros(n)
        for k, w in zip(active, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                for j in range(j_start, j_end + 1):
                    sg[j] += w * 2.0 * a[k - j]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = min(0.3, 2.0 * max(sc - f_lb, 0.01) / (gnorm + 1e-30))
        
        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    
    return best_a

"""
Optimizer using binomial-sequence initialization (flat autocorrelation) combined 
with high-precision L-BFGS, multi-lag gradient, and targeted peak deflation.
Key insight: The sequence a[i] = C(2i,i)/4^i has perfectly flat self-convolution,
providing an excellent starting point for further optimization.
"""

import numpy as np
from scipy.optimize import minimize
import time

_state = {'beta': 1000.0}

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def binomial_init(n):
    """Generate the flat-convolution sequence a[i] = C(2i,i) / 4^i."""
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n-1:n-1+n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def objective_fn(x):
    return lse_score_grad(x, _state['beta'])

def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    _state['beta'] = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective_fn, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / max(a0.sum(), 1e-12)
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out

def subgradient_step(a, n_active=50, step_scale=0.1):
    """One step of subgradient descent using active lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(np.max(c))
    
    # Active lags: those within 1% of max
    threshold = c_max * 0.99
    active = np.where(c >= threshold)[0][:n_active]
    
    if len(active) == 0:
        return a
    
    # Compute subgradient (average of active lag gradients)
    sg = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            sg[j] += 2.0 * a[k - j]
    sg /= len(active)
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-15:
        return a
    
    step = step_scale / (gnorm + 1e-10)
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    Sa = a_new.sum()
    if Sa > 1e-12:
        a_new = a_new / Sa
    return a_new

def mirror_descent(a0, beta, n_iters=3000, eta0=0.05, decay=0.9998):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    
    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        
        g_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break
        g_normed = g_centered / (gnorm * n)
        exponent = np.clip(-eta * g_normed, -20, 20)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay
    
    return best_a

def targeted_redistribution(a, n_rounds=100):
    """Identify peak contributors and redistribute mass to below-peak regions."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for r in range(n_rounds):
        c = fast_conv(a)
        c_max = float(np.max(c))
        sc = 2.0 * n * c_max / (float(a.sum())**2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg_len = j_end - j_start + 1
        
        # Compute contributions to peak
        contrib = np.zeros(seg_len)
        for j in range(j_start, j_end + 1):
            contrib[j - j_start] = a[j] * a[k_star - j]
        
        total = float(np.sum(contrib))
        if total < 1e-15:
            break
        
        # Top 25% contributors
        n_top = max(1, seg_len // 4)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:n_top]
        
        # Reduce top, increase bottom
        frac = 0.05
        reduction = 0.0
        for li in top_idx:
            gi = j_start + li
            if a[gi] > 1e-10:
                a[gi] *= (1.0 - frac)
                reduction += a[gi] * frac / (1.0 - frac)  # approximate
        
        # Actually let me redo this more carefully
        reduction = 0.0
        for li in top_idx:
            gi = j_start + li
            old = a[gi]
            new = old * (1.0 - frac)
            reduction += old - new
            a[gi] = new
        
        if reduction > 1e-15:
            per = reduction / len(bot_idx)
            for li in bot_idx:
                gi = j_start + li
                a[gi] += per
        
        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa > 1e-12:
            a = a / Sa
    
    return best_a
