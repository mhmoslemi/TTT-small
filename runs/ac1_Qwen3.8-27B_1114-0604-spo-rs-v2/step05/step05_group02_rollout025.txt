```python
def simplex_project(y):
    """Project y onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(y)
    if n == 0:
        return y
    # Algorithm from Duchi et al. (2008)
    if np.all(y <= 0):
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)
```

```python
    Q = np.fft.rfft(q, M)
```

```python
    """FISTA on the simplex for the LSE-smoothed objective."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    y = a.copy()
    t = 1.0
    L = 4.0 * beta  # Lipschitz constant
    
    t_start = time.monotonic()
    for k in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(y)
        c_max = c.max()
        scaled = beta * (c - c_max)

        p = exp_scaled / exp_scaled.sum()
        
        # Gradient of LSE
        g = lse_grad_fft(y, p, n)

        t = t_new
    
    return a
```

```python
"""
Convex optimization approach to minimize max autocorrelation of non-negative sequence.
Key insight: fixing sum(a)=1, the problem min max_k conv(a,a)[k] s.t. a>=0, sum(a)=1
is CONVEX. We solve it via FISTA (accelerated proximal gradient) with LSE smoothing
and beta annealing, followed by L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def simplex_project(y):
    """Project y onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(y)
    if n == 0:
        return y
    if np.all(y <= 0):
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def lse_grad_fft(a, p, n):
    """Compute gradient of LSE_beta(conv(a,a)) wrt a using FFT.
    p is the probability vector (softmax weights) of length 2n-1.
    """
    q = p[::-1]
    M = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    P = np.fft.rfft(p, M)
    Q = np.fft.rfft(q, M)
    conv_pa = np.fft.irfft(A * P, M)
    conv_qa = np.fft.irfft(A * Q, M)
    g = conv_qa[n - 1:2 * n - 1][::-1] + conv_pa[:n]
    return g


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and gradient for the full score function.
    Score = 2n * LSE_beta(c) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    g = lse_grad_fft(a, p, n)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def fista_lse(a0, beta, max_iters=5000, time_limit=30.0, verbose=False):
    """FISTA on the simplex for LSE-smoothed max-convolution objective.
    Minimizes LSE_beta(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a0
    a = a / S

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    L = 4.0 * beta  # Lipschitz constant bound

    t_start = time.monotonic()
    for k in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        # Compute gradient at y
        c = fast_conv(y)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        g = lse_grad_fft(y, p, n)

        # Adaptive Lipschitz constant (backtracking)
        L_trial = L
        for _ in range(20):
            a_trial = simplex_project(y - g / L_trial)
            c_trial = fast_conv(a_trial)
            c_max_t = float(c_trial.max())
            scaled_t = beta * (c_trial - c_max_t)
            scaled_t -= scaled_t.max()
            exp_t = np.exp(scaled_t)
            L_trial_val = c_max_t + float(np.log(exp_t.sum())) / beta
            L_curr_val = c_max + float(np.log(exp_scaled.sum())) / beta
            descent = float(g.dot(y - a_trial))
            if L_trial_val <= L_curr_val - descent + 0.5 / L_trial * float(g.dot(g)) + 1e-12:
                break
            L_trial *= 2.0
        L = L_trial

        a_new = simplex_project(y - g / L)

        if np.linalg.norm(a_new - a) < 1e-14:
            a = a_new
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        a_prev = a_new
        t = t_new

        if verbose and k % 500 == 0:
            sc = compute_score(a)
            print(f"    FISTA beta={beta} iter={k} score={sc:.8f}")

    return a


def fista_lse_no_backtrack(a0, beta, max_iters=5000, time_limit=30.0):
    """FISTA without backtracking (faster, uses fixed L)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a0
    a = a / S

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    L = 4.0 * beta

    t_start = time.monotonic()
    for k in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(y)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        g = lse_grad_fft(y, p, n)

        a_new = simplex_project(y - g / L)

        if np.linalg.norm(a_new - a) < 1e-14:

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        a_prev = a
        a = a_new
        t = t_new

    return a


def subgradient_multi_peak(a, n_iters=5000, time_limit=30.0, top_k=3):
    """Subgradient method using top-k peaks with decreasing step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    t_start = time.monotonic()
    for k in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Compute weighted subgradient
        weights = c[top_idx]
        weights = weights / weights.sum()

        g = np.zeros(n)
        for w, k_star in zip(weights, top_idx):
            k_star = int(k_star)
            for i in range(n):
                j1 = k_star - i
                j2 = i - k_star
                if 0 <= j1 < n:
                    g[i] += w * a[j1]
                if 0 <= j2 < n:
                    g[i] += w * a[j2]

        # Step size: diminish as 1/sqrt(k)
        eta = 1.0 / np.sqrt(k + 1.0)
        eta = min(eta, 0.5 / (float(np.linalg.norm(g)) + 1e-10))

        a_new = simplex_project(a - eta * g)
        a = a_new

    return best_a


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000, time_limit=60.0):
    """Run L-BFGS-B on the LSE-smoothed full score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
        S = n * 0.5
    a0 = a0 / S

    def obj_grad(x):
        x = np.maximum(x, 0.0)
        sc, gr = lse_score_grad(x, beta)
        return sc, gr

    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        S_out = a_out.sum()
        if S_out < 1e-6:
            a_out = a0
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return compute_score(a0), a0

    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
