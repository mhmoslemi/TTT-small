def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def water_fill_project(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    S = v.sum()
    if S < 1e-30:
        return np.full(n, target_sum / n)
    # Sort in decreasing order
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find k such that (cumsum[k] - target_sum)/(k+1) < sorted_v[k]
    k = n - 1
    for i in range(n - 1, -1, -1):
        t_candidate = (cumsum[i] - target_sum) / (i + 1)
        if t_candidate < sorted_v[i]:
            k = i
            break
    else:
        k = n - 1
    t = (cumsum[k] - target_sum) / (k + 1)
    t = max(t, 0.0)
    result = np.maximum(v - t, 0.0)
    # Ensure exact sum
    S2 = result.sum()
    if S2 > 1e-30:

    return result

    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -50.0, 50.0)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L
    p = exp_scaled / exp_scaled.sum()

    # Gradient: g_j = 2 * sum_m a_m * p_{j+m}
    # Use FFT cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad = 2.0 * conv_res
    return score, grad


def power_score_grad(a, p):
    """
    p-norm smoothed score and gradient, assuming sum(a)=1, a>=0.
    f(a) = 2n * (sum b_k^p)^(1/p)
    """
    n = len(a)
    c = fast_conv(a)
    cp = c ** p
    sum_cp = float(cp.sum())
    if sum_cp < 1e-300:
        return 2.0 * n * float(c.max()), np.zeros(n)
    norm_val = sum_cp ** (1.0 / p)
    score = 2.0 * n * norm_val
    # df/da_j = 2n * (1/p) * (sum_cp)^(1/p - 1) * p * sum_k c_k^(p-1) * dc_k/da_j
    #         = 2n * (sum_cp)^(1/p - 1) * sum_k c_k^(p-1) * dc_k/da_j
    weight = (sum_cp) ** (1.0 / p - 1.0)
    w = c ** (p - 1)
    # g_j = 2 * sum_m a_m * w_{j+m} (same structure as LSE gradient)
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    grad = 2.0 * n * weight * 2.0 * conv_res
    return score, grad


def gs_fienup(a, n_iters=500, alpha_init=0.7, alpha_final=0.1, M=None):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def nesterov_lse(a0, beta_start, beta_end, n_steps, lr_start, lr_end,
                 momentum=0.9, gs_every=200, gs_iters=50, deadline=None):
    """
    Nesterov-accelerated LSE gradient descent with water-filling projection
    and periodic GS-Fienup refresh.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)

    M_fft = 1 << (2 * n - 1).bit_length()

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        progress = step / max(1, n_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        mu = momentum * (1.0 - 0.01 * progress)

        # Nesterov extrapolation
        y = a + mu * v

        # Compute gradient at y
        y_pos = np.maximum(y, 0.0)
        Sy = float(y_pos.sum())
        if Sy < 1e-12:
            y_pos = np.ones(n) / n
            Sy = 1.0
        y_norm = y_pos / Sy

        score_y, grad_y = lse_score_grad_constrained(y_norm, beta)

        # Rescale gradient back to original scale
        # Actually, since the objective is scale-invariant and we work on sum=1,
        # the gradient is already on the sum=1 manifold.
        grad = grad_y

        # Update velocity
        v = mu * v - lr * grad

        # New point
        a_new = y + v

        # Project onto {a >= 0, sum(a) = 1}
        a_new = water_fill_project(a_new, 1.0)

        # Compute score
        sc = compute_score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # Periodic GS refresh
        if gs_every > 0 and (step + 1) % gs_every == 0:
            a_gs = gs_fienup(a, n_iters=gs_iters, alpha_init=0.5, alpha_final=0.1, M=M_fft)
            sc_gs = compute_score(a_gs)
            if sc_gs < best_sc:
                best_sc = sc_gs
                best_a = a_gs.copy()
                a = a_gs.copy()
                v = np.zeros(n)

    return best_a, best_sc


def nesterov_power(a0, p, n_steps, lr_start, lr_end,
                   momentum=0.9, gs_every=200, gs_iters=50, deadline=None):
    """Nesterov-accelerated p-norm gradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    M_fft = 1 << (2 * n - 1).bit_length()

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        progress = step / max(1, n_steps)
        lr = lr_start * (lr_end / lr_start) ** progress
        mu = momentum

        y = a + mu * v
        y_pos = np.maximum(y, 0.0)
        Sy = float(y_pos.sum())
        if Sy < 1e-12:
            y_pos = np.ones(n) / n
            Sy = 1.0
        y_norm = y_pos / Sy

        score_y, grad_y = power_score_grad(y_norm, p)
        grad = grad_y

        v = mu * v - lr * grad
        a_new = y + v
        a_new = water_fill_project(a_new, 1.0)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        if gs_every > 0 and (step + 1) % gs_every == 0:
            a_gs = gs_fienup(a, n_iters=gs_iters, alpha_init=0.5, alpha_final=0.1, M=M_fft)
            sc_gs = compute_score(a_gs)
            if sc_gs < best_sc:
                best_sc = sc_gs
                best_a = a_gs.copy()
                a = a_gs.copy()
                v = np.zeros(n)

    return best_a, best_sc


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        Sx = x.sum()
        if Sx < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / Sx
        sc, g = lse_score_grad_constrained(xn, beta)
        # Chain rule: d/dx [f(x/sum(x))] = (1/sum(x)) * (g - f'(sum_dir))
        # f'(sum_dir) = g dot ones / n ... actually:
        # If h(x) = f(x/S) where S = sum(x), then:
        # dh/dx_i = (1/S) * df/dx_i|_{x=S*xn} - (f*(x_i/S^2)) ... 
        # Actually simpler: on the simplex, the gradient projected to the ambient space
        # is just g/S (since the constraint sum=1 means we can ignore the 1/S^2 factor
        # for L-BFGS purposes, as it's a constant on the constraint set).
        # For L-BFGS with bounds [0, inf], we just use g directly.
        return sc, g / S

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def run_lbfgs_power(a0, p, maxiter=5000, maxfun=100000):
    """L-BFGS-B on p-norm objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

def peak_pair_cd(a, n_iters=2000, top_m=20, deadline=None):
    """
    Peak-pair coordinate descent: directly reduce elements contributing to top peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # For each peak, find the element with largest contribution and reduce it
        reduced = False
        for k in top_idx[:min(top_m, 10)]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            # Contributions: a_j * a_{k-j} for j in [j_start, j_end]
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k - js]
            # Find the largest contributor
            j_max = js[np.argmax(contrib)]
            # Reduce it
            a_old = a[j_max]
            if a_old < 1e-10:
                continue
            # Try reducing by various factors
            best_factor = 1.0
            cur_sc = sc
            for factor in [0.7, 0.8, 0.85, 0.9, 0.95]:
                a[j_max] = a_old * factor
                a_norm = a / a.sum()
                sc_try = compute_score(a_norm)
                if sc_try < cur_sc:
                    cur_sc = sc_try
                    best_factor = factor
            a[j_max] = a_old * best_factor
            if best_factor < 1.0:
                reduced = True

        a = a / a.sum()

        # Periodic GS refresh
        if (it + 1) % 200 == 0:
            a_gs = gs_fienup(a, n_iters=30, alpha_init=0.4, alpha_final=0.1, M=M_fft)
            sc_gs = compute_score(a_gs)
            if sc_gs < best_sc:
                best_sc = sc_gs
                best_a = a_gs.copy()
                a = a_gs.copy()

    return best_a


def top_m_subgradient(a, n_iters=3000, top_m=10, step_init=0.03, decay=0.9998, deadline=None):
    """Subgradient method using top-M peaks with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

        # Periodic GS
        if (it + 1) % 500 == 0:
            a_gs = gs_fienup(a, n_iters=30, alpha_init=0.4, alpha_final=0.1, M=M_fft)
            sc_gs = compute_score(a_gs)
            if sc_gs < best_sc:
                best_sc = sc_gs
                best_a = a_gs.copy()
                a = a_gs.copy()

    return best_a


def multires_init(n_target, parent, deadline=None):
    """Multi-resolution initialization: solve at increasing sizes and upscale."""
    sizes = [64, 128, 256, 512, 1024, 2048, 4096]
    sizes = [s for s in sizes if s < n_target]
    if n_target not in sizes:
        sizes.append(n_target)

    if parent is not None:
        a = parent.copy()
    else:
        # Start with a structured initialization
        x = np.arange(64, dtype=np.float64) / 64
        a = np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01)
        a = a / a.sum()

    for target_n in sizes:
        if deadline is not None and time.monotonic() > deadline:
            break

        # Upsample to target_n
        if len(a) != target_n:
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, target_n)
            a = np.interp(x_new, x_old, a)
            a = np.maximum(a, 0.0)

        # Run GS + Nesterov at this size
        a_gs = gs_fienup(a, n_iters=100, alpha_init=0.6, alpha_final=0.1)
        sc_gs = compute_score(a_gs)

        a_ns, sc_ns = nesterov_lse(a_gs, beta_start=200, beta_end=2000,
                                    n_steps=500, lr_start=0.01, lr_end=0.001,
                                    gs_every=100, gs_iters=20)

        a = a_ns if sc_ns < sc_gs else a_gs

    return a


def diversity_init(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Asymmetric ramps (important: we established symmetric can't beat 2)
    inits.append(np.maximum(1.0 + 0.5 * (2 * x - 1), 0.01))
    inits.append(np.maximum(1.0 - 0.5 * (2 * x - 1), 0.01))
    inits.append(np.exp(1.5 * (2 * x - 1)))
    inits.append(np.exp(-1.5 * (2 * x - 1)))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.sin(4.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.3 + 0.7 * x, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - x), 0.01))

    for a in inits:
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            inits[-1] = a / S
        else:
            inits[-1] = np.ones(n) / n
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov-accelerated LSE/power-norm gradient with water-filling projection,
    GS-Fienup refresh, L-BFGS, and peak-pair coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    best_score = np.inf
    best_a = np.ones(n) / n

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"=== Starting search, n={n}, budget={total_budget}s ===")
    print(f"Parent available: {parent is not None}")

    # ==================================================================
    # Phase 1: Initialization (8% of time)
    # ==================================================================
    p1_end = t_start + 0.08 * (total_budget - 30.0)
    print(f"\n--- Phase 1: Initialization (deadline {p1_end - t_start:.0f}s) ---")

    candidates = []

    # Multi-resolution from parent or structured init
    if parent is not None and time.monotonic() < p1_end:
        a_mr = multires_init(n, parent, deadline=p1_end)
        sc = compute_score(a_mr)
        update_best(sc, a_mr, "multires")
        candidates.append(a_mr)

    # Diversity inits
    if time.monotonic() < p1_end:
        inits = diversity_init(n, rng)
        for i, a0 in enumerate(inits[:5]):  # Limit to 5 for time
            if time.monotonic() > p1_end:
                break
            a_gs = gs_fienup(a0, n_iters=100, alpha_init=0.7, alpha_final=0.2)
            sc = compute_score(a_gs)
            update_best(sc, a_gs, f"init{i}-gs")
            candidates.append(a_gs)

    # Use best as starting point
    a_start = best_a.copy()
    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ==================================================================
    # Phase 2: Nesterov LSE gradient (35% of time)
    # ==================================================================
    p2_end = t_start + 0.43 * (total_budget - 30.0)
    print(f"\n--- Phase 2: Nesterov LSE (deadline {p2_end - t_start:.0f}s) ---")

    # Long Nesterov run with annealing temperature
    a_ns, sc_ns = nesterov_lse(
        a_start,
        beta_start=100, beta_end=10000,
        n_steps=500000,
        lr_start=0.005, lr_end=0.0001,
        momentum=0.9,
        gs_every=500, gs_iters=30,
        deadline=p2_end
    )
    update_best(sc_ns, a_ns, "nesterov-lse")

    # Power-norm variant
    if time.monotonic() < p2_end:
        a_pn, sc_pn = nesterov_power(
            a_ns,
            p=4,
            n_steps=200000,
            lr_start=0.002, lr_end=0.00005,
            momentum=0.9,
            gs_every=500, gs_iters=20,
            deadline=p2_end
        )
        update_best(sc_pn, a_pn, "nesterov-power4")

    # More aggressive LSE with higher beta
    if time.monotonic() < p2_end:
        a_ns2, sc_ns2 = nesterov_lse(
            a_ns,
            beta_start=500, beta_end=50000,
            n_steps=200000,
            lr_start=0.002, lr_end=0.00005,
            momentum=0.95,
            gs_every=300, gs_iters=20,
            deadline=p2_end
        )
        update_best(sc_ns2, a_ns2, "nesterov-lse-high-beta")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% of time)
    # ==================================================================
    p3_end = t_start + 0.68 * (total_budget - 30.0)
    print(f"\n--- Phase 3: L-BFGS (deadline {p3_end - t_start:.0f}s) ---")

    betas_to_try = [200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas_to_try:
        if time.monotonic() > p3_end:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, f"LBFGS-beta{beta}")
        print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Power-norm L-BFGS
    if time.monotonic() < p3_end:
        for p in [4, 8]:
            if time.monotonic() > p3_end:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs_power(best_a, p, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"LBFGS-p{p}")
            print(f"    p={p}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak-pair CD + Subgradient (20% of time)
    # ==================================================================
    p4_end = t_start + 0.88 * (total_budget - 30.0)
    print(f"\n--- Phase 4: Peak-pair CD + Subgradient (deadline {p4_end - t_start:.0f}s) ---")

    for top_m in [5, 10, 20, 40]:
        if time.monotonic() > p4_end:
            break
        a_cd = peak_pair_cd(best_a, n_iters=2000, top_m=top_m, deadline=p4_end)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, f"peakcd-top{top_m}")

    for top_m in [5, 10, 20]:
        if time.monotonic() > p4_end:
            break
        a_sg = top_m_subgradient(best_a, n_iters=3000, top_m=top_m, deadline=p4_end)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"subgrad-top{top_m}")

    print(f"Phase 4 done. Best: {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish (7% of time)
    # ==================================================================
    p5_end = deadline
    print(f"\n--- Phase 5: Final Polish (deadline {p5_end - t_start:.0f}s) ---")

    # One more Nesterov pass on the best
    if time.monotonic() < p5_end - 60:
        a_final, sc_final = nesterov_lse(
            best_a,
            beta_start=5000, beta_end=50000,
            n_steps=100000,
            lr_start=0.001, lr_end=0.00001,
            momentum=0.95,
            gs_every=200, gs_iters=20,
            deadline=p5_end - 60
        )
        update_best(sc_final, a_final, "final-nesterov")

    # Final L-BFGS
    if time.monotonic() < p5_end - 30:
        sc, a_opt = run_lbfgs_lse(best_a, 50000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "final-lbfgs")

    # Final GS
    if time.monotonic() < p5_end - 10:
        a_gs = gs_fienup(best_a, n_iters=200, alpha_init=0.3, alpha_final=0.05)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, "final-gs")

    # ==================================================================
    # Final output
    # ==================================================================
    # Ensure the output is valid for evaluate_sequence
    out_arr = np.clip(best_a, 0.0, 1000.0)
    if out_arr.sum() < 0.01:
        out_arr = np.full(n, 0.5)

    out = [float(x) for x in out_arr]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")

    # Use the verified score to decide
    if verified < best_score:
        pass  # verified is our true score
    else:
        # Something went wrong with verification, use internal
        print(f"WARNING: verified ({verified:.6f}) > internal ({best_score:.6f})")

    return out
