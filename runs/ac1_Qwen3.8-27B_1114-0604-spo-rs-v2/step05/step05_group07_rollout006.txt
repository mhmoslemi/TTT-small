Multi-strategy optimizer for minimizing 2*n*max(autocorrelation)/(sum)^2.
Key innovations over previous approach:
1. Nesterov-accelerated subgradient with backtracking line search

# Core computation functions

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):

    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

    #   - As a[j]: when j = i, term is a[i] * a[k-i]
    #   - As a[k-j]: when k-j = i, i.e., j = k-i, term is a[k-i] * a[i]

    # dL/da[i] = sum_k weights[k] * d(c[k])/da[i]
    # = 2 * sum_k weights[k] * a[k-i] * 1_{0<=k-i<n}  (approximately, ignoring the diagonal)

    # dL/da[i] ≈ 2 * (weights * a_reversed)[n-1+i]

    # = 2 * sum_k weights[k] * a[k-i] where 0 <= k-i < n

    # 
    # Actually, let's think of it differently.
    # Let r = n-1. a_reversed[j] = a[r-j].
    # convolve(weights, a_reversed)[m] = sum_k weights[k] * a_reversed[m-k] = sum_k weights[k] * a[r-(m-k)]

    M2 = len(weights) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    W = np.fft.rfft(weights, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)

    dL_da = 2.0 * conv_res[n - 1:n - 1 + n]
    
    # Full gradient of score = 2n * L / S^2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad

def nesterov_subgrad_ls(a, n_iters, f_lb=1.0, top_k=5, use_ls=True):
    """
    Nesterov-accelerated subgradient descent with optional line search.
    Uses multi-peak (top-K) weighting for smoother gradients.
    """
    n = len(a)
    a = normalize(a.copy())
    y = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t = 1.0
    step_base = 0.5

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        
        # Compute multi-peak subgradient
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]

        sorted_tops = np.sort(top_vals)[::-1]
        weights = np.zeros(len(c))
        for rank, idx in enumerate(top_idx):
            weights[idx] = (k - rank) / (k * (k + 1) / 2.0)
        
        # Compute gradient
        Mak_weighted = float(np.sum(weights * c[top_idx]))
        S2 = Sy * Sy
        S3 = S2 * Sy
        
        sg = np.zeros(n)
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * y[idx - ii]
        
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)
        
        # Center the gradient (remove mean for sum=1 constraint)
        grad -= np.mean(grad)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        sc = 2.0 * n * float(np.max(c)) / S2
        if sc < best_sc:

            step = step_base / (gnorm + 1e-10)

            best_step = step

        else:
            step = step_base / (gnorm * np.sqrt(it + 1) + 1e-10)

        Sa = a.sum()

def peak_peeling(a, n_sweeps=20, n_top=100, n_redistribute=50):
    """
    Directly identify top contributors to the peak autocorrelation value
    and reduce them, redistributing mass to low-contribution positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find the segment that contributes to position k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        # Contribution of position j_start + i to c[k_star] is seg[i] * a[k_star - (j_start + i)]
        # = seg[i] * seg[n-1-i] (reversed)
        contrib = seg * seg[::-1]
        
        # Reduce top contributors
        total_len = len(contrib)
        top_count = min(n_top, max(5, total_len // 5))
        bot_count = min(n_redistribute, max(5, total_len - top_count))
        
        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]
        
        for j_idx in top_local:
            global_j = j_start + j_idx
            old_val = a[global_j]
            if old_val < 1e-10:
                continue
            
            # Try reducing this element by various fractions
            best_val = old_val
            best_local_sc = best_sc
            
            for frac in [0.5, 0.3, 0.1, 0.0]:
                a[global_j] = old_val * frac
                reduction = old_val - a[global_j]
                
                # Redistribute to bottom contributors
                if reduction > 1e-15 and len(bot_local) > 0:
                    per = reduction / len(bot_local)
                    for bj in bot_local:
                        a[j_start + bj] += per
                
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = a[global_j]
            
            a[global_j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def perturb(a, rng, scale=0.05):
    """Generate a perturbed version of a."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 8)
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 5 + int(rng.integers(0, 20))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top contributors to max peak
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg * seg[::-1]
        total_len = len(contrib)
        top_count = max(10, total_len // 10)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.2 * rng.random())
    elif strategy == 4:
        # Multiplicative log-normal
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    elif strategy == 5:
        # Random subset rescale
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))
    elif strategy == 6:
        # Smooth perturbation (low-frequency)
        n_freq = 3
        pert = np.zeros(n)
        for f in range(n_freq):
            freq = (f + 1) * (1.0 + 5.0 * rng.random())
            phase = 2.0 * np.pi * rng.random()
            amp = scale / (f + 1)
            pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
        a_p = a * (1.0 + pert)
    else:
        # Sparse perturbation
        n_pert = max(5, n // 20)
        idx = rng.choice(n, size=n_pert, replace=False)
        a_p = a.copy()
        a_p[idx] += scale * a.mean() * rng.standard_normal(n_pert)
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    return normalize(a_p)

def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Gaussian centered
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    
    # Two bumps (left and right) - helps with non-symmetric autocorrelation
    inits.append(np.exp(-8.0 * (x - 0.2) ** 2) + np.exp(-8.0 * (x - 0.8) ** 2))
    
    # Cosine window
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    
    # Asymmetric ramp
    inits.append(0.1 + 0.9 * x)
    
    # Asymmetric inverse ramp
    inits.append(0.1 + 0.9 * (1.0 - x))
    
    # Periodic structure
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x))
    
    # Raised cosine (Hann)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    
    # Flat top with tapered edges
    taper = 0.15
    inits.append(np.where((x < taper) | (x > 1 - taper),
                         0.5 * (1 + np.cos(np.pi * np.minimum(x / taper, (1-x) / taper))),
                         1.0))
    
    return inits

def optimize_from_start(a_init, n_iters_sg, n_iters_lse, beta_start, top_k, rng):

    if sc < best_sc:
        best_sc = sc
        best_a = a_sg2.copy()
    
    return best_sc, best_a

# Main entry point

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: diverse init -> subgradient+LS -> LSE+LBFGS -> peeling -> ILS.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0
    
    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                parent = normalize(parent)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Determine working length
    if parent is not None:
        n_parent = len(parent)
        sc_parent = compute_score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        report(sc_parent, parent, "parent")
        # We'll work at parent length primarily
        n_work = n_parent
    else:
        n_work = 8192
        print(f"No parent found, using n={n_work}")
    
    print(f"Working length: {n_work}")
    
    # ==================================================================
    # Phase 1: Multi-start with subgradient (30% time)
    # ==================================================================
    phase1_deadline = t_start + 0.30 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Multi-start Subgradient (n={n_work}) ---")
    
    inits = build_inits(n_work, rng, parent if parent is not None and len(parent) == n_work else None)
    
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        
        a_init = normalize(a_init)
        sc_init = compute_score(a_init)
        report(sc_init, a_init, f"init_{i}")
        
        # Run subgradient from this start
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break
        n_iters = max(500, int(remaining * 30))
        
        a_opt = nesterov_subgrad_ls(a_init, n_iters=n_iters, f_lb=1.0,
                                     top_k=5, use_ls=True)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"sg_init_{i}")
    
    print(f"After Phase 1: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 2: LSE + L-BFGS with adaptive beta (30% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LSE + L-BFGS Refinement ---")
    
    beta_schedule = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
    top_k_schedule = [20, 10, 10, 5, 5, 5, 3, 3]
    
    for beta, tk in zip(beta_schedule, top_k_schedule):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        
        maxiter = int(min(30000, max(2000, remaining * 20)))
        maxfun = int(min(100000, max(5000, remaining * 80)))
        
        sc, a_opt = run_lbfgs_refine(best_a, beta=beta, top_k=tk,
                                      maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta} k={tk}")
        
        # Try from a perturbed start too
        if time.monotonic() < phase2_deadline - 60 and beta >= 100:
            a_pert = perturb(best_a, rng, scale=0.02)
            sc2, a_opt2 = run_lbfgs_refine(a_pert, beta=beta, top_k=tk,
                                            maxiter=maxiter // 3, maxfun=maxfun // 3)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")
    
    print(f"After Phase 2: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 3: Peak peeling + subgradient polish (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Peak Peeling + Polish ---")
    
    a_peel = peak_peeling(best_a, n_sweeps=20, n_top=100, n_redistribute=50)
    sc = compute_score(a_peel)
    report(sc, a_peel, "peeling")
    
    # Subgradient polish after peeling
    if time.monotonic() < phase3_deadline - 30:
        a_polish = nesterov_subgrad_ls(best_a, n_iters=3000, f_lb=1.0,
                                        top_k=3, use_ls=True)
        sc = compute_score(a_polish)
        report(sc, a_polish, "sg_polish")
    
    # Another peeling pass
    if time.monotonic() < phase3_deadline - 20:
        a_peel2 = peak_peeling(best_a, n_sweeps=15, n_top=50, n_redistribute=30)
        sc = compute_score(a_peel2)
        report(sc, a_peel2, "peeling2")
    
    print(f"After Phase 3: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 4: Iterated Local Search (20% time)
    # ==================================================================
    phase4_deadline = t_start + 0.95 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")
    
    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 60:
            break
        
        # Adaptive perturbation scale (decrease over time)
        scale = 0.03 + 0.05 * (1.0 - trial / max(1, n_trials)) * rng.random()
        a_pert = perturb(best_a, rng, scale=scale)
        
        # Subgradient from perturbed start
        n_iters = max(500, int(remaining * 10))
        a_sg = nesterov_subgrad_ls(a_pert, n_iters=min(n_iters, 3000),
                                    f_lb=1.0, top_k=5, use_ls=True)
        sc = compute_score(a_sg)
        report(sc, a_sg, f"ILS#{trial} sg")
        
        # LSE refinement if we improved
        if sc < best_score and time.monotonic() < phase4_deadline - 30:
            beta = 50 + int(200 * rng.random())
            sc2, a_opt = run_lbfgs_refine(a_sg, beta=beta, top_k=5,
                                           maxiter=5000, maxfun=20000)
            report(sc2, a_opt, f"ILS#{trial} lbfgs")
    
    print(f"After Phase 4: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 5: Final high-precision refinement (5% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Refinement ---")
    
    if time.monotonic() < deadline - 20:
        for beta in [50000, 100000, 300000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt = run_lbfgs_refine(best_a, beta=beta, top_k=2,
                                          maxiter=20000, maxfun=80000)
            report(sc, a_opt, f"final b={beta}")
    
    # Final subgradient with very few top peaks (essentially pure subgradient at max)
    if time.monotonic() < deadline - 10:
        a_final = nesterov_subgrad_ls(best_a, n_iters=2000, f_lb=1.0,
                                       top_k=1, use_ls=True)
        sc = compute_score(a_final)
        report(sc, a_final, "final_sg")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
