"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft active-set LSE with exponential (multiplicative) updates,
Nesterov-accelerated subgradient polishing, and jitter-based restarts
to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_soft_obj_grad(a, beta):
    """
    LSE-smoothed objective using ALL positions with exponential weighting.
    L = dmax + log(sum(exp(beta*(d_k - dmax))))/beta
    This uses all positions but effectively focuses on the top ones.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = len(d)
    dmax = float(d.max())
    if dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)  # z <= 0
    # For numerical stability, cap z
    z = np.clip(z, -50, 0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(M) / M
        se = 1.0
    w = e / se
    L = dmax + np.log(se) / beta

    corr = _cross_corr(a, w)
    LDS = float(w @ d)

    # grad of 2*L w.r.t. a:
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs_soft(a0, beta, time_limit, maxfun_cap=2000):
    """Run L-BFGS-B on soft LSE objective."""
    a = _norm(a0)
    n = len(a)

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses continuation-based soft LSE optimization with adaptive beta scheduling,
Nesterov-accelerated subgradient polishing, multiplicative updates for
non-negativity, and jitter-reoptimize cycles to escape local minima.
Explores multiple sequence lengths and diverse initializations.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_k_obj_grad(a, beta, top_k):
    """LSE-smoothed objective using top-k active set with full weighting."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = len(d)
    k = max(1, min(top_k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    z = np.clip(z, -60, 0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs_k(a0, beta, top_k, time_limit, maxfun_cap=3000):

    a = _norm(a0)
    n = len(a)

    return _score(out), out


def _nesterov_subgrad(a, max_iters=50, deadline=None):
    """Nesterov-accelerated subgradient descent on the max-convolution objective."""
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    velocity = np.zeros(n)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.5% of max
        threshold = dmax - 0.005 * max(dmax, 1e-10)
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Nesterov: look-ahead step
        # Compute gradient at the Nesterov point
        momentum = 2.0 / (it + 3.0)
        a_nest = a + momentum * velocity
        a_nest = np.clip(a_nest, 0.0, None)
        Sn = float(a_nest.sum())
        if Sn < 1e-8:
            a_nest = _norm(a)
            Sn = 1.0
        a_nest = a_nest / Sn

        c_n = _self_conv(a_nest)
        d_n = float(n) * c_n / (Sn * Sn)
        dmax_n = float(d_n.max())
        threshold_n = dmax_n - 0.005 * max(dmax_n, 1e-10)
        active_n = d_n >= threshold_n
        if not active_n.any():
            active_n[np.argmax(d_n)] = True
        n_act_n = int(active_n.sum())
        w_n = np.zeros_like(d_n)
        w_n[active_n] = 1.0 / n_act_n
        corr_n = _cross_corr(a_nest, w_n)
        LDS_n = float(w_n[active_n] @ d_n[active_n])
        grad_n = 4.0 * float(n) * corr_n / (Sn * Sn) - 4.0 * LDS_n / Sn
        grad_n = np.nan_to_num(grad_n, nan=0.0)

        # Adaptive step size
        gnorm_n = float(np.linalg.norm(grad_n))
        if gnorm_n < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            eta = scale / gnorm_n
            q = a - eta * grad_n
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break

        velocity = best_step - a
        a = best_step

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()

    return best_sc, best_a


def _multiplicative_polish(a, max_iters=100, deadline=None):
    """Multiplicative (exponential) updates that naturally preserve non-negativity."""
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S = float(a.sum())
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Soft active set with exponential weighting
        beta = max(50.0, 50.0 + it * 5.0)
        z = beta * (d - dmax)
        z = np.clip(z, -50, 0)
        w = np.exp(z)
        se = float(w.sum())
        if se < 1e-30:
            break
        w = w / se

        corr = _cross_corr(a, w)
        LDS = float(w @ d)
        load = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
        load = np.nan_to_num(load, nan=0.0)

        # Multiplicative update with adaptive step
        improved = False
        for scale in (0.002, 0.005, 0.01, 0.02, 0.05):
            q = a * np.exp(-scale * load)
            q = np.clip(q, 1e-15, None)
            Sq = float(q.sum())
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q
                improved = True
                break

        if not improved:
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _structured_init(n, rng, kind='bimodal'):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'bimodal':
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        s = rng.uniform(0.02, 0.08)
        a = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s ** 2))
    elif kind == 'exponential':
        rate = rng.uniform(1.0, 5.0)
        center = rng.uniform(0.3, 0.7)
        a = np.exp(-rate * np.abs(x - center))
    elif kind == 'cosine_taper':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** rng.uniform(0.5, 3.0)
    elif kind == 'plateau':
        w = rng.uniform(0.1, 0.3)
        c = 0.5
        a = np.ones(n)
        a[x < c - w] = rng.uniform(0.1, 0.5)
        a[x > c + w] = rng.uniform(0.1, 0.5)
    else:
        a = np.ones(n)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

        (100, 3000.0),
        (150, 8000.0),
        (250, 20000.0),
        (400, 50000.0),
    ]

    # Allocate time: more time for higher beta stages
    weights = [1, 1, 1, 1, 1.5, 1.5, 2, 2, 2, 2]
    total_w = sum(weights)
    avail = max(10.0, deadline - t0 - 30.0)

    cum = 0.0
    for si, (k, beta) in enumerate(stages):
        frac = weights[si] / total_w
        cum += frac
        stage_end = t0 + cum * avail
        stage_end = min(stage_end, deadline - 30.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        # Run L-BFGS-B
        sc, a_new = _run_lbfgs_k(a, beta, k, stage_budget, maxfun_cap=500)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    [{label}] beta={beta:.0f}, K={k}: {best_sc:.8f}")
        a = _norm(a_new)

    # Multiplicative polish
    if time.monotonic() < deadline - 15.0:
        sc, a_mp = _multiplicative_polish(a, max_iters=80, deadline=deadline - 5.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_mp.copy()
            print(f"    [{label}] multiplicative: {best_sc:.8f}")

    # Nesterov subgradient polish
    if time.monotonic() < deadline - 10.0:
        sc, a_ns = _nesterov_subgrad(a, max_iters=40, deadline=deadline - 5.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_ns.copy()
            print(f"    [{label}] nesterov: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _jitter_reoptimize(a, deadline, label, rng):
    """Perturb the sequence and re-optimize to escape local minima."""
    a = _norm(a)
    n = len(a)

    # Generate a smooth perturbation
    noise = rng.standard_normal(n)
    k = min(51, max(5, n // 100))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')

    # Try several perturbation strengths
    best_a = a.copy()
    best_sc = _score(a)

    for strength in (0.05, 0.1, 0.15, 0.2, 0.3):
        if time.monotonic() > deadline - 30.0:
            break
        q = a + strength * noise
        q = _norm(q)

        # Quick LSE optimization on the perturbed sequence
        chunk_deadline = min(deadline - 30.0, time.monotonic() + 60.0)
        sc, q_opt = _run_lbfgs_k(q, 2000, 200, chunk_deadline - time.monotonic(), 300)
        if sc < best_sc:
            best_sc = sc
            best_a = q_opt.copy()
            print(f"    [{label}] jitter s={strength}: {best_sc:.8f}")

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: continuation LSE with Nesterov subgradient and jitter."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
                print(f"Parent loaded: length={len(parent)}, score={_score(parent):.8f}")
    except Exception:
        pass

    # --- Phase 1: Explore different lengths ---
    # Try n=4096 and n=8192
    lengths_to_try = [4096, 8192]

    if parent is not None:
        pn = len(parent)
        if pn <= 4096 and 4096 not in lengths_to_try:
            lengths_to_try.insert(0, pn)
        elif pn <= 8192 and pn > 4096 and 8192 not in lengths_to_try:
            pass  # 8192 already in list

    best_overall_score = np.inf
    best_overall_seq = None

    # Allocate time per length: more time for 8192 (more robust)
    time_alloc = {}
    total_len_w = sum(l for l in lengths_to_try)
    for ln in lengths_to_try:
        time_alloc[ln] = 0.7 * (ln / total_len_w) * (deadline - t_start - 100.0)

    for target_n in lengths_to_try:
        n_deadline = t_start + time_alloc[target_n] + 50.0
        n_deadline = min(n_deadline, deadline - 60.0)

        if time.monotonic() > n_deadline - 60.0:
            continue

        print(f"\n{'='*60}")
        print(f"Target n={target_n}, time budget={n_deadline - t_start:.0f}s")
        print(f"{'='*60}")

        # Generate initializations for this length
        inits = []

        # 1. Parent resampled
        if parent is not None:
            inits.append(('parent', _upsample(parent, target_n)))
            inits.append(('parent_rev', _upsample(parent[::-1], target_n)))

        # 2. Spectral inits
        for i in range(4):
            inits.append((f'spec{i}', _spectral_init(target_n, rng)))

        # 3. Structured inits
        for kind in ('bimodal', 'exponential', 'cosine_taper', 'plateau'):
            inits.append((f'struct_{kind}', _structured_init(target_n, rng, kind)))

        # 4. Parent + noise
        if parent is not None:
            p = _upsample(parent, target_n)
            for amp in (0.05, 0.15, 0.3):
                noise = rng.standard_normal(target_n)
                k = max(5, target_n // 200)
                if k % 2 == 0:
                    k += 1
                kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4) ** 2))
                kern /= kern.sum()
                noise = np.convolve(noise, kern, mode='same')
                inits.append((f'pnoise_{amp}', _norm(p + amp * noise)))

        # Evaluate and rank
        scored = []
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr))

        scored.sort(key=lambda x: x[0])
        top_scores = [f'{s:.6f}' for s, _, _ in scored[:5]]
        print(f"  {len(scored)} inits. Top: {top_scores}")

        # Select top 2-3 diverse candidates
        candidates = []
        used_names = set()
        for sc, name, arr in scored:
            if len(candidates) >= 3:
                break
            # Ensure diversity: skip if too similar to already selected
            if used_names:
                too_similar = False
                for _, _, prev_arr in candidates:
                    cos_sim = float(np.dot(arr, prev_arr) / (np.linalg.norm(arr) * np.linalg.norm(prev_arr) + 1e-15))
                    if cos_sim > 0.99:
                        too_similar = True
                        break
                if too_similar:
                    continue
            candidates.append((sc, name, arr))
            used_names.add(name)

        if not candidates:
            candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

        # Optimize each candidate
        n_cands = len(candidates)
        frac_list = [0.5, 0.3, 0.2][:n_cands]

        cum_frac = 0.0
        for ci, (sc_init, name, arr_init) in enumerate(candidates):
            cand_deadline = t_start + (cum_frac + frac_list[ci]) * (n_deadline - t_start - 20.0) + 10.0
            cand_deadline = min(cand_deadline, n_deadline - 20.0)
            cum_frac += frac_list[ci]

            if time.monotonic() > cand_deadline - 10.0:
                continue

            # Multilevel: optimize at n/4 first
            a_work = _norm(arr_init)
            if target_n > 2048 and time.monotonic() < cand_deadline - 60.0:
                n_small = max(256, target_n // 4)
                a_small = _upsample(a_work, n_small)
                # Quick LSE at small level
                time_small = min(30.0, (cand_deadline - time.monotonic()) * 0.2)
                for k_s, beta_s in [(5, 50), (15, 300), (40, 2000)]:
                    if time.monotonic() > cand_deadline - 0.8 * time_small:
                        break
                    sc_s, a_s = _run_lbfgs_k(a_small, beta_s, k_s, 10.0, 200)
                    if sc_s < _score(a_small):
                        a_small = a_s
                a_up = _norm(_upsample(a_small, target_n))
                if _score(a_up) < _score(a_work):
                    a_work = a_up

            # Main continuation optimization
            sc_traj, seq_traj = _continuation_optimize(a_work, cand_deadline, f'n{target_n}_{name}')
            if sc_traj < best_overall_score:
                best_overall_score = sc_traj
                best_overall_seq = seq_traj.copy()
                print(f"  *** Global best: n={target_n}, {sc_traj:.8f} ***")

    # --- Phase 2: Jitter + re-optimize on the best ---
    if best_overall_seq is not None and time.monotonic() < deadline - 120.0:
        print(f"\n=== Jitter phase (score={best_overall_score:.8f}) ===")
        jitter_deadline = deadline - 60.0
        n = len(best_overall_seq)

        # Try multiple jitter rounds
        for jround in range(4):
            if time.monotonic() > jitter_deadline - 30.0:
                break
            chunk_dl = min(jitter_deadline - 30.0, time.monotonic() + 150.0)
            sc_j, seq_j = _jitter_reoptimize(best_overall_seq, chunk_dl, f'jit{jround}', rng)
            if sc_j < best_overall_score:
                best_overall_score = sc_j
                best_overall_seq = seq_j.copy()
                print(f"  Jitter round {jround}: improved to {best_overall_score:.8f}")
            else:
                print(f"  Jitter round {jround}: no improvement ({sc_j:.8f})")

    # --- Phase 3: Final intensive polishing ---
    if best_overall_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\n=== Final polish (score={best_overall_score:.8f}) ===")
        final_deadline = deadline - 15.0
        a = best_overall_seq.copy()

        # Very high beta LSE
        for k_f, beta_f in [(300, 30000), (500, 80000), (1000, 200000)]:
            if time.monotonic() > final_deadline - 20.0:
                break
            remaining = final_deadline - time.monotonic() - 15.0
            if remaining < 10.0:
                break
            sc_f, a_f = _run_lbfgs_k(a, beta_f, k_f, remaining * 0.5, 400)
            if sc_f < best_overall_score:
                best_overall_score = sc_f
                best_overall_seq = a_f.copy()
                a = a_f
                print(f"  Final LSE K={k_f}, beta={beta_f}: {best_overall_score:.8f}")

        # Nesterov polish
        if time.monotonic() < final_deadline - 15.0:
            sc_ns, a_ns = _nesterov_subgrad(a, max_iters=50, deadline=final_deadline - 10.0)
            if sc_ns < best_overall_score:
                best_overall_score = sc_ns
                best_overall_seq = a_ns.copy()
                print(f"  Final Nesterov: {best_overall_score:.8f}")

        # Multiplicative polish
        if time.monotonic() < final_deadline - 10.0:
            sc_mp, a_mp = _multiplicative_polish(a, max_iters=100, deadline=final_deadline - 5.0)
            if sc_mp < best_overall_score:
                best_overall_score = sc_mp
                best_overall_seq = a_mp.copy()
                print(f"  Final multiplicative: {best_overall_score:.8f}")

    # --- Prepare output ---
    if best_overall_seq is None:
        # Fallback
        best_overall_seq = _norm(np.ones(1024))
        best_overall_score = _score(best_overall_seq)

    out = _norm(best_overall_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_overall_score

    print(f"\nFINAL: internal={best_overall_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
