import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_constraint_matrix(f, n):
    """Build the (2n-1) x n constraint matrix A where A[k,i] = f[k-i] if valid."""
    # A is banded Toeplitz
    # Row k has non-zeros at columns i where 0 <= k-i < n
    # i.e., max(0, k-n+1) <= i <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            idx = np.arange(i_start, i_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

"""
LP-direction iterative optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses linear programming to find descent directions that reduce the autocorrelation peak,
interleaved with spectral shaping and L-BFGS polishing. Explores multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via real FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_toeplitz_constraints(f, n):
    """Build sparse constraint matrix A where A[k,i] = f[k-i] if 0<=k-i<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            idx = np.arange(i_start, i_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M, n, timeout=60):
    """Solve LP: max sum(b) s.t. A*b <= M, b >= 0. Returns optimal b or None on failure."""
    try:
        A = build_toeplitz_constraints(f, n)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
        elif res.status == 3:  # timeout
            if res.x is not None:
                return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"  LP error: {e}")
    return None

    S_b = float(b.sum())
    if S_b < 1e-12:

    g = b / S_b
    
    # Check if direction is essentially the same as f

    if t_grid is None:

def fienup_fast(a, n_iters=100):
    """Quick Fienup spectral flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        alpha = 0.3 + 0.5 * (it / n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)
        
        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):

    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:

        a0 = a0 / S

        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)

    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out

    return exact_sc, a_out


def peak_shave_quick(a, n_iters=500, top_k=3):

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

        return np.ones(n) / n
    a = a / S

    step = 0.01
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        sc = 2.0 * n * M_val
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        avg_w = float(c[top_idx].sum()) / (top_k * M_val + 1e-30)
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, (sc - 2.0 * n / (2.0 * n - 1)) / (gn * gn + 1e-30) * 0.5)
        step = min(step, 0.03)
        step = max(step, 1e-8)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999
    
    return best_a


def coord_descent_quick(a, n_sweeps=5, n_top=50):
    """Quick coordinate descent on peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        cur_sc = best_sc
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            for f in [0.7, 0.85, 0.9, 1.1, 1.15, 1.3]:
                cv = a_old_j * f
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_tmp = a / S_tmp
                sc = compute_score(a_tmp)
                if sc < cur_sc:
                    cur_sc = sc
                    best_sc = sc
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    break
                a[j] = a_old_j
            else:
                a[j] = a_old_j
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    return best_a


def interpolate_length(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        parent = np.random.exponential(1.0, n0)
        sc_p = compute_score(parent)
        update(sc_p, parent, "random-init")
        print(f"Random init: n={n0}, score={sc_p:.10f}")

    # =================================================================
    # Phase 1: Quick Fienup to improve starting point (5% time)
    # =================================================================
    phase1_end = t_start + 0.05 * budget_s
    print(f"\n=== Phase 1: Fienup (until {phase1_end - t_start:.0f}s) ===")
    
    a_f = fienup_fast(best_a, n_iters=150)
    sc = compute_score(a_f)
    update(sc, a_f, "FIENUP")

    # =================================================================
    # Phase 2: LP-direction iterations (55% time)
    # =================================================================
    phase2_end = t_start + 0.60 * budget_s
    print(f"\n=== Phase 2: LP-direction iterations (until {phase2_end - t_start:.0f}s) ===")
    
    a_work = best_a.copy()
    n_work = len(a_work)
    a_work = a_work / a_work.sum()
    
    n_lp_iters = 0
    n_failures = 0
    consec_no_improve = 0
    
    while time.monotonic() < phase2_end - 30:
        elapsed = time.monotonic() - t_start
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break
        
        n_lp_iters += 1
        t0 = time.monotonic()
        new_a, new_sc, direction_found = lp_step(a_work, n_work)
        dt = time.monotonic() - t0
        
        if direction_found and new_sc < compute_score(a_work):
            a_work = new_a
            a_work = a_work / a_work.sum()
            consec_no_improve = 0
            if n_lp_iters % 5 == 1:
                print(f"  LP iter {n_lp_iters}: score={new_sc:.10f} ({dt:.1f}s)")
            update(new_sc, a_work, f"LP-{n_lp_iters}")
        else:
            consec_no_improve += 1
            if n_lp_iters % 10 == 0:
                print(f"  LP iter {n_lp_iters}: no improvement ({dt:.1f}s), streak={consec_no_improve}")
            
            # If stuck, try Fienup to escape
            if consec_no_improve >= 5:
                print(f"  Escaping with Fienup after {consec_no_improve} stuck iterations...")
                a_esc = fienup_fast(a_work, n_iters=50)
                sc_esc = compute_score(a_esc)
                if sc_esc < compute_score(a_work):
                    a_work = a_esc / a_esc.sum()
                    update(sc_esc, a_work, "FLENUP-ESCAPE")
                    consec_no_improve = 0
                else:
                    # Add small perturbation
                    a_work = a_work * (1.0 + 0.01 * rng.standard_normal(n_work))
                    a_work = np.maximum(a_work, 0)
                    a_work = a_work / a_work.sum()
                    consec_no_improve = 0
        
        n_failures += 1 if not direction_found else 0
    
    print(f"  Phase 2 complete: {n_lp_iters} LP iterations")
    
    # =================================================================
    # Phase 3: L-BFGS polishing (20% time)
    # =================================================================
    phase3_end = t_start + 0.80 * budget_s
    print(f"\n=== Phase 3: L-BFGS polish (until {phase3_end - t_start:.0f}s) ===")
    
    betas_to_try = [500, 2000, 5000, 10000, 20000]
    for beta in betas_to_try:
        if time.monotonic() > phase3_end - 20:
            break
        remaining = phase3_end - time.monotonic() - 10
        if remaining < 10:
            break
        maxiter = min(8000, int(remaining * 40))
        maxfun = min(150000, int(remaining * 800))
        if maxiter < 200:
            break
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.10f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
    
    # Peak shave on the best
    if time.monotonic() < phase3_end - 30:
        a_ps = peak_shave_quick(best_a, n_iters=1000, top_k=3)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")
        
        # L-BFGS after peak shave
        if sc < best_score and time.monotonic() < phase3_end - 20:
            sc2, a2 = run_lbfgs(a_ps, 5000, maxiter=3000, maxfun=60000)
            update(sc2, a2, "SHA+LBFGS")

    # =================================================================
    # Phase 4: Try different sequence lengths (10% time)
    # =================================================================
    phase4_end = t_start + 0.90 * budget_s
    print(f"\n=== Phase 4: Multi-length exploration (until {phase4_end - t_start:.0f}s) ===")
    
    current_n = len(best_a)
    candidate_lengths = []
    for dl in [-1024, -512, -256, 256, 512, 1024, 2048]:
        nl = current_n + dl
        if 512 <= nl <= 20000:
            candidate_lengths.append(nl)
    
    # Always try some round numbers
    for nl in [2048, 4096, 6144, 8192, 10240]:
        if nl not in candidate_lengths and 512 <= nl <= 20000:
            candidate_lengths.append(nl)
    
    # Remove duplicates and sort
    candidate_lengths = sorted(set(candidate_lengths))
    
    for nl in candidate_lengths:
        if time.monotonic() > phase4_end - 30:
            break
        remaining = phase4_end - time.monotonic() - 10
        if remaining < 20:
            break
        
        a_interp = interpolate_length(best_a, nl)
        S = a_interp.sum()
        if S < 1e-12:
            continue
        a_interp = a_interp / S
        
        # Quick Fienup + L-BFGS on this length
        a_new = fienup_fast(a_interp, n_iters=50)
        sc = compute_score(a_new)
        
        if remaining > 30:
            maxiter = min(3000, int(remaining * 20))
            sc2, a_opt = run_lbfgs(a_new, 3000, maxiter=maxiter, maxfun=maxiter * 20)
            sc = min(sc, sc2)
            a_new = a_opt if sc2 < sc else a_new
        
        update(sc, a_new, f"LEN-{nl}")
    
    # =================================================================
    # Phase 5: Final polish (10% time)
    # =================================================================
    print(f"\n=== Phase 5: Final polish ===")
    
    # Coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd = coord_descent_quick(best_a, n_sweeps=10, n_top=80)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")
    
    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 5
        maxiter = min(5000, int(remaining * 30))
        if maxiter > 500:
            sc, a_final = run_lbfgs(best_a, 10000, maxiter=maxiter, maxfun=maxiter * 30)
            update(sc, a_final, "FINAL-LBFGS")
    
    # Another coordinate descent pass
    if time.monotonic() < deadline - 10:
        a_cd = coord_descent_quick(best_a, n_sweeps=5, n_top=100)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")
    
    # Final small LP iterations if there's time
    if time.monotonic() < deadline - 15 and len(best_a) <= 8000:
        a_tmp = best_a / best_a.sum()
        for _ in range(5):
            if time.monotonic() > deadline - 10:
                break
            new_a, new_sc, found = lp_step(a_tmp, len(a_tmp))
            if found and new_sc < compute_score(a_tmp):
                a_tmp = new_a / new_a.sum()
                update(new_sc, a_tmp, "FINAL-LP")
    
    # =================================================================
    # Output
    # =================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
