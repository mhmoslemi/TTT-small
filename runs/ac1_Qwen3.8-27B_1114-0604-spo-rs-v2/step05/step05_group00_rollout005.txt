    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    # First project onto sum<=target_sum, x>=0
    # Then scale if needed
    # Actually, project onto {x: sum(x)=target_sum, x>=0}
    # This is the same as project onto {x: sum(x)<=target_sum, x>=0} then scale

    # u = sort(v) in decreasing order
    # Find rho such that sum_{i: u_i + lambda > 0} (u_i + lambda) = 1
    # lambda = (1 - sum_{i=1..rho} u_i) / rho
    
    v_sorted = np.sort(v)[::-1]

    proj = np.maximum(v - threshold, 0.0)
    # Scale to target_sum
    proj_sum = proj.sum()
    if proj_sum > 0:
        proj = proj * target_sum / proj_sum
    return proj

                  options={'time_limit': time_limit})
    if res.status == 0:
        g = res.x
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None

def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_flatten(a, n_iters=200):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.2 + 0.6 * progress
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def coord_descent_peaks(a, n_sweeps=10, n_top=100):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3]]
            best_val = a_old_j
            best_local = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
