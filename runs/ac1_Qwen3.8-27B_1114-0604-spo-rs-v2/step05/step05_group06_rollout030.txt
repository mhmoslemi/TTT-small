"""
Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses progressive LSE optimization with improved subgradient (BB step size),
asymmetric ramp initializations, multi-scale optimization, and focused
active-set polishing. Key insight: asymmetry breaks the C-S barrier at 2.0.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_m a[m] * w[m+j] for j=0..n-1 (linear cross-correlation)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    N = _next_pow2(n + w.size - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cc = np.fft.irfft(np.conj(A) * W, N)
    return cc[:n]


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, K):
    """LSE-smoothed objective and gradient. Returns (value, gradient)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    K = max(1, min(K, M))

    if K >= M:
        idx = np.arange(M)
    else:
        idx = np.argpartition(d, M - K)[M - K:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(len(idx)) / len(idx)
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = L
    return float(val), grad


def _run_lbfgs(a0, beta, K, time_limit, maxfun_cap):
    """Run L-BFGS-B on LSE objective. Returns (score, optimized_array)."""
    a = _normalize(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a

    t_end = time.monotonic() + time_limit
    est_eval_t = 0.005 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(50, min(mf, maxfun_cap, 10000))
    maxiter = max(30, mf // 2)
    PEN = 5e-4

    def obj(x):
        v, g = _lse_obj_grad(x, beta, K)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g.copy()
        g += 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
        )
        out = _normalize(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_bb(a, max_iters=80, deadline=None, tol=1e-14):
    """Subgradient descent with Barzilai-Borwein step size and non-monotone acceptance."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    x_prev = a.copy()
    g_prev = None
    s_prev = None
    y_prev = None
    bb_counter = 0
    history = []
    NM_WINDOW = 10

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        active = d >= dmax - 1e-6 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True

        w = np.zeros_like(d)
        w[active] = 1.0 / float(active.sum())

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < tol:
            break

        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_prev, y_prev))
            if sy > 1e-16:
                byy = float(np.dot(y_prev, y_prev))
                s2 = float(np.dot(s_prev, s_prev))
                if bb_counter % 2 == 0:
                    eta = sy / byy if byy > 1e-16 else 1.0 / gnorm
                else:
                    eta = s2 / sy
                eta = max(1e-10, min(eta, 10.0 / gnorm))
            else:
                eta = 1.0 / gnorm
        else:
            eta = 1.0 / gnorm
        bb_counter += 1

        # Non-monotone line search
        accepted = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            step = eta * scale
            x_new = a - step * grad
            x_new = np.clip(x_new, 0.0, None)
            Sn = x_new.sum()
            if Sn < 1e-8:
                continue
            x_new = x_new / Sn
            sc_new = _score(x_new)

            # Non-monotone acceptance
            if len(history) < NM_WINDOW:
                threshold = sc_new < best_sc
            else:
                threshold = sc_new <= max(history[-NM_WINDOW:]) + 1e-12

            if threshold or (sc_new < best_sc * 1.001):
                # Update BB quantities
                g_new = grad  # approximate (recomputing is expensive)
                s_cur = x_new - a
                y_cur = g_new - g_prev if g_prev is not None else None
                if y_cur is not None:
                    s_prev = s_cur
                    y_prev = y_cur
                g_prev = g_new
                a = x_new
                accepted = True

                history.append(sc_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = x_new.copy()
                break

        if not accepted:
            # Try a smaller uniform step
            step = 1e-4 / gnorm
            x_new = np.clip(a - step * grad, 0.0, None)
            Sn = x_new.sum()
            if Sn > 1e-8:
                x_new = x_new / Sn
                sc_new = _score(x_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = x_new.copy()
            break

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with target spectral shape."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.2, 2.0)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _ramp_init(n, rng, asymmetric=True):
    """Generate a ramp-like initialization that breaks symmetry."""
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.5, 3.0)
    beta = rng.uniform(0.3, 2.0)
    a = x ** alpha * (1 - x) ** beta
    if not asymmetric:
        a = a + 0.3 * (1 - a)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _smooth_perturb(a, rng, strength=0.2):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(61, max(5, n // 100))
    if k % 2 == 0:
        k += 1
    c = k // 2
    sig = max(1.0, c / 2.5)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _normalize(q)


def _active_set_focused(a, K_active, max_iters=50, deadline=None):
    """Focused optimization using only top K_active lags in the subgradient."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())

        # Focus on top K_active lags only
        idx = np.argpartition(d, -(K_active))[-K_active:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / K_active

        corr = _cross_corr(a, w)
        LDS = float(w[idx] @ d[idx])
        grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Try multiple step sizes
        best_step = None
        best_new_sc = best_sc
        for scale in (1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005):
            eta = scale / gnorm
            q = np.clip(a - eta * grad, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        a = best_step
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        elif best_new_sc > best_sc * 1.0001:
            a = best_a.copy()

    return best_sc, best_a


def _optimize_trajectory(a0, deadline, label, rng):

    a = _normalize(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Progressive LSE stages: (K, beta, time_fraction)
    total_time = max(20.0, deadline - t0 - 30.0)
    stages = [
        (3, 30.0, 0.05),
        (5, 80.0, 0.07),
        (8, 200.0, 0.08),
        (15, 500.0, 0.10),

        (120, 8000.0, 0.17),
        (250, 20000.0, 0.18),
        (500, 50000.0, 0.18),
    ]

    cum_frac = 0.0
    for k, beta, frac in stages:
        cum_frac += frac
        stage_end = t0 + cum_frac * total_time

                best_a = a_new.copy()

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a

            continue

        n = len(arr_init)
        print(f"\n=== Candidate {ci+1}/{n_cands}: '{name}', n={n}, init_score={sc_init:.8f} ===")

        # Multi-level warm start
        a_work = _normalize(arr_init)

        if n > 4096 and time.monotonic() < sub_deadline - 80.0:
            # Optimize at n/4
            n_small = max(512, n // 4)
            a_small = _upsample(a_work, n_small)
            sc_small = _score(a_small)

            t_phase_end = time.monotonic() + min(60.0, (sub_deadline - 80.0 - time.monotonic()) * 0.3)
            for k_s, beta_s in [(5, 100), (20, 500), (60, 2000)]:
                if time.monotonic() > t_phase_end:
                    break
                sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, 15.0, 200)
                if sc_s < sc_small:
                    sc_small = sc_s
                    a_small = a_s

            # Upscale
            a_up = _normalize(_upsample(a_small, n))
            sc_up = _score(a_up)
            if sc_up < _score(a_work):
                a_work = a_up
                print(f"  Multi-level warm start improved: {sc_up:.8f}")

        # Main trajectory
        sc_traj, seq_traj = _optimize_trajectory(a_work, sub_deadline, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** Global best: {best_score:.8f} ***")

    # --- Final global polish ---
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 15.0

    # Very high beta LSE
    for k_f, beta_f in [(300, 30000), (600, 80000), (1200, 200000)]:
        if time.monotonic() > final_deadline - 30:
            break
        remaining = final_deadline - time.monotonic() - 30
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_f, k_f, remaining * 0.35, 800)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_f}, beta={beta_f}: {best_score:.8f}")

    # Subgradient BB
    if time.monotonic() < final_deadline - 20:
        sc_sg, a_sg = _subgrad_bb(best_seq, max_iters=60, deadline=final_deadline - 15)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad_BB: {best_score:.8f}")

    # Active-set focused
    for Ka in (2, 3, 5, 8):
        if time.monotonic() > final_deadline - 15:
            break
        sc_af, a_af = _active_set_focused(best_seq, Ka, max_iters=30, deadline=final_deadline - 10)
        if sc_af < best_score:
            best_score = sc_af
            best_seq = a_af.copy()
            print(f"  Final focused K={Ka}: {best_score:.8f}")

    # Random restart perturbations
    if time.monotonic() < final_deadline - 40:
        for restart in range(4):
            if time.monotonic() > final_deadline - 20:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.05 + 0.05 * restart)
            chunk_end = min(final_deadline - 20, time.monotonic() + 20)
            sc_r, a_r = _run_lbfgs(p, 50000, 500, max(3, chunk_end - time.monotonic()), 300)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")
            # Quick subgradient
            sc_r2, a_r2 = _subgrad_bb(a_r, max_iters=15, deadline=final_deadline - 15)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_seq = a_r2.copy()
                print(f"  Restart {restart} sg: {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_final, a_final = _subgrad_bb(best_seq, max_iters=30, deadline=final_deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Prepare output
    out = _normalize(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
