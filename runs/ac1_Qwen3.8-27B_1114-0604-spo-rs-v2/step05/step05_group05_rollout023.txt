The objective is score = 2n * max(conv(a,a)) / (sum(a))^2.

def weighted_grad(a, p):
    """
    Compute gradient of (sum p[k]*conv[k]) / S^2 * 2n.
    p is a weight vector of length 2n-1.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S

def topk_gradient_step(a, K, lr, momentum_v, mu):
    """
    One step of top-K gradient descent with momentum.
    Returns (new_a, new_v, score).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a, momentum_v, np.inf

    score = 2.0 * n * c_max / (S * S)

    Prft = np.fft.rfft(p_rev)

    S2 = S * S
    S3 = S2 * S

    if S_new < 1e-12:
        return a, momentum_v, score
    a_new = a_new / S_new
    
    return a_new, momentum_v, score


def fienup_refresh(a, alpha=0.7, iters=5):
    """
    Fienup refresh: compress the dynamic range of the spectrum.
    alpha < 1 compresses (reduces peaks in frequency domain).
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for _ in range(iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Compress: replace sqrt(P) with P^(alpha/2) * P_mean^((1-alpha)/2)
        # This makes the spectrum flatter
        mag = np.sqrt(P)
        mag_new = P_mean ** ((1.0 - alpha) / 2.0) * mag**alpha
        # Renormalize to keep total power the same
        mag_new_ratio = mag.sum() / (mag_new.sum() + 1e-30)
        mag_new = mag_new * mag_new_ratio
        
        # Reconstruct
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
    
    return a


def fienup_thresh(a, thresh=1.2, clip=0.5, iters=10):
    """
    Threshold-based Fienup: reduce bins above threshold * mean.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for _ in range(iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above])**clip
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
    
    return a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    def objective(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def diverse_init(n, rng, method=0):

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
