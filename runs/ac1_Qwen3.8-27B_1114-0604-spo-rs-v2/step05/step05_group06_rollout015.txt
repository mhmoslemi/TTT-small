```python
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines mirror descent (exponential weights), LSE-LBFGS, and convolution
equalization for diverse optimization trajectories.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out
```

```python
"""
Multi-strategy search for non-negative sequence minimizing 2n*max(conv(a,a))/S^2.
Combines adaptive mirror descent (exponential weights), LSE-LBFGS polishing,
iterative variance equalization, and multilevel coarse-to-fine optimization.
Key innovations over prior approach: mirror descent as primary engine (different
basin exploration), variance-equalization pre-conditioning, and multi-length search.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global state for L-BFGS objective (to avoid nested function definitions)
_lse_beta = 100.0
_lse_k = 50
_lse_pen = 1e-3


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    w_full = np.zeros(M)
    w_full[idx] = w_act
    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    val = 2.0 * L
    return float(val), grad


def _lbfgs_objective(x):
    global _lse_beta, _lse_k, _lse_pen
    v, g = _lse_obj_grad(x, _lse_beta, _lse_k)
    Sx = float(x.sum())
    v += _lse_pen * (Sx - 1.0) ** 2
    g = g + 2.0 * _lse_pen * (Sx - 1.0)
    return v, g


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap=400):
    global _lse_beta, _lse_k, _lse_pen
    _lse_beta = float(beta)
    _lse_k = int(k)
    _lse_pen = 1e-3
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a
    est_eval_t = 0.003 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    try:
        res = minimize(
            _lbfgs_objective, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _mirror_descent(a0, deadline, eta0=0.02, beta_start=5.0, beta_end=500.0,
                    min_iters=500):
    """
    Adaptive mirror descent (exponential weights) on the simplex.
    Uses softmax-weighted active set with increasing sharpness (beta).
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    it = 0
    total_time = max(5.0, deadline - t0)

    while time.monotonic() < deadline:
        it += 1
        if it < min_iters and time.monotonic() - t0 < 2.0:
            pass  # ensure minimum iterations even for very short budgets

        # Progress-based beta (sharpness of softmax weights)
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = _self_conv(a)
        S = float(a.sum())
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0 or not np.isfinite(dmax):
            break

        # Softmax weights over all positions (or active subset for efficiency)
        M = 2 * n - 1
        if M <= 4096:
            z = beta * (d - dmax)
            e = np.exp(z)
            se = float(e.sum())
            if se < 1e-30:
                break
            w = e / se
        else:
            # Use top-k for large M
            k_use = min(M, max(100, int(M * 0.05)))
            idx = np.argpartition(d, M - k_use)[M - k_use:]
            d_act = d[idx]
            z = beta * (d_act - dmax)
            e = np.exp(z)
            se = float(e.sum())
            if se < 1e-30:
                break
            w = np.zeros(M)
            w[idx] = e / se

        corr = _cross_corr(a, w)
        corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)

        # Adaptive eta: start small, grow with progress
        eta = eta0 * (1.0 + 5.0 * progress)
        # Scaling by typical magnitude of corr for stability
        corr_scale = float(np.mean(np.abs(corr))) + 1e-12
        eta = eta * (1.0 / corr_scale) * 0.1

        # Mirror descent step: multiplicative update
        log_update = -eta * corr
        log_update = np.clip(log_update, -50.0, 50.0)
        update = np.exp(log_update)
        update = np.clip(update, 1e-30, None)
        a_new = a * update
        S_new = float(a_new.sum())
        if S_new < 1e-12 or not np.isfinite(S_new):
            break
        a_new = a_new / S_new

        # Check improvement
        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        # Periodic print
        if it % 50000 == 0:
            print(f"    MD it={it}, beta={beta:.1f}, score={best_sc:.8f}")

    return best_sc, best_a


def _iter_equalize(a0, deadline, max_iters=50000):
    """
    Iterative variance equalization: reduce the spread of convolution values.
    Uses broader weighting than mirror descent to "precondition" the solution.
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S = float(a.sum())
        d = float(n) * c / (S * S)
        dmean = float(d.mean())
        dmax = float(d.max())

        # Weight by excess above mean (broad, not just peak)
        excess = np.maximum(d - dmean, 0.0)
        if excess.max() < 1e-15:
            break
        w = excess / float(excess.sum())

        corr = _cross_corr(a, w)
        corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)

        # Adaptive step
        corr_scale = float(np.mean(np.abs(corr))) + 1e-12
        progress = min(1.0, (time.monotonic() - t0) / max(1.0, deadline - t0))
        eta = 0.05 * (1.0 / corr_scale) * (1.0 + 3.0 * progress)

        log_update = -eta * corr
        log_update = np.clip(log_update, -30.0, 30.0)
        a_new = a * np.exp(log_update)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 10000 == 0:
            print(f"    EQ it={it+1}, dmax={dmax:.6f}, dmean={dmean:.6f}, score={best_sc:.8f}")

    return best_sc, best_a


def _subgrad_polish(a, max_iters=50, deadline=None):
    """Exact subgradient descent with adaptive step size."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.05% of max
        active = d >= dmax - 5e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0002, 0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.3):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence with specified spectral shape."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.2, 1.8)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _gen_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []

    # Parent resampled
    if parent is not None:
        inits.append(('parent', _upsample(parent, n)))
        inits.append(('parent_rev', _upsample(parent[::-1], n)))
        for amp in (0.05, 0.1, 0.2, 0.4):
            p = _upsample(parent, n)
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    # Spectral
    for i in range(8):
        inits.append((f'spec{i}', _spectral_init(n, rng)))

    # Cosine modulations
    x = np.linspace(0, 1, n)
    for f in (1, 2, 3, 5):
        for amp in (0.1, 0.2, 0.4, 0.7):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'cos{f}_{amp}', a / a.sum()))

    # Smoothed random noise
    for i in range(4):
        noise = rng.standard_normal(n)
        k = max(5, n // 80)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * max(1, k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'noise{i}', a / a.sum()))

    # Exponential decay variants
    for r in (0.5, 0.8, 1.2, 2.0):
        a = np.exp(-r * np.linspace(0, 1, n))
        inits.append((f'exp{r}', a / a.sum()))

    return inits


def _optimize_single(a0, deadline, label, rng):
    """Full optimization pipeline for a single candidate at fixed length."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()
    total = max(10.0, deadline - t0)

    print(f"  [{label}] n={n}, start score={best_sc:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_sc, best_a

    # Phase 1: Iterative equalization (broad, to find good basin)
    eq_deadline = t0 + 0.25 * total
    eq_deadline = min(eq_deadline, deadline - 0.5 * total)
    if remaining > 0.3 * total:
        sc_eq, a_eq = _iter_equalize(a, eq_deadline, max_iters=30000)
        if sc_eq < best_sc:
            best_sc = sc_eq
            best_a = a_eq.copy()
            print(f"    [{label}] after EQ: {best_sc:.8f}")
        a = a_eq

    # Phase 2: Mirror descent (progressive sharpness)
    md_deadline = t0 + 0.55 * total
    md_deadline = min(md_deadline, deadline - 0.35 * total)
    if time.monotonic() < md_deadline:
        sc_md, a_md = _mirror_descent(a, md_deadline, eta0=0.03,
                                       beta_start=3.0, beta_end=800.0)
        if sc_md < best_sc:
            best_sc = sc_md
            best_a = a_md.copy()
            print(f"    [{label}] after MD: {best_sc:.8f}")
        a = a_md

    # Phase 3: LSE + L-BFGS-B (progressive K)
    lse_stages = [
        (5, 30.0, 0.06),
        (10, 80.0, 0.08),
        (20, 200.0, 0.10),
        (40, 500.0, 0.12),
        (80, 1200.0, 0.14),
        (160, 3000.0, 0.16),
        (320, 8000.0, 0.18),
    ]

    lse_start = max(time.monotonic(), t0 + 0.55 * total)
    lse_end = deadline - 0.1 * total

    for k, beta, frac in lse_stages:
        stage_end = lse_start + frac * (lse_end - lse_start)
        stage_end = min(stage_end, deadline - 0.1 * total)
        if time.monotonic() > stage_end - 3.0:
            continue
        budget = stage_end - time.monotonic()
        if budget < 5.0:
            continue
        sc_lse, a_lse = _run_lbfgs(a, beta, k, budget * 0.9, 300)
        if sc_lse < best_sc:
            best_sc = sc_lse
            best_a = a_lse.copy()
            print(f"    [{label}] LSE K={k}, beta={beta:.0f}: {best_sc:.8f}")
        a = _norm(a_lse)

    # Phase 4: Subgradient polish
    if time.monotonic() < deadline - 5.0:
        sc_sg, a_sg = _subgrad_polish(a, max_iters=20, deadline=deadline - 2.0)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    elapsed = time.monotonic() - t0
    print(f"  [{label}] done: score={best_sc:.8f}, time={elapsed:.0f}s")
    return best_sc, best_a


def _multilevel_optimize(a0, n, deadline, label, rng):
    """Coarse-to-fine: optimize at n/4, then n/2, then n."""
    a = _norm(a0)
    t0 = time.monotonic()
    total = max(10.0, deadline - t0)
    best_sc = _score(a)
    best_a = a.copy()

    # Level 1: n/4 (quick)
    if n > 1024:
        n1 = max(256, n // 4)
        a1 = _upsample(a, n1)
        d1 = t0 + 0.15 * total
        d1 = min(d1, deadline - 0.6 * total)
        if d1 - t0 > 10.0:
            sc1, a1 = _optimize_single(a1, d1, label + "_L1", rng)
            a2 = _norm(_upsample(a1, n))
            sc2 = _score(a2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
                a = a2
                print(f"    [{label}] multilevel L1->full: {best_sc:.8f}")

    # Level 2: full n
    d2 = deadline - 5.0
    if time.monotonic() < d2 - 10.0:
        sc2, a2 = _optimize_single(a, d2, label + "_full", rng)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: multi-strategy, multi-length, with diverse inits."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"Parent length: {len(parent)}, parent score: {_score(parent):.8f}")

    # Determine lengths to try
    if parent is not None:
        p_len = len(parent)
        lengths = sorted(set([4096, 8192, 16384, p_len]))
    else:
        lengths = [8192, 16384]

    best_score = np.inf
    best_seq = None
    overall_deadline = deadline

    # Allocate time across lengths
    n_lengths = len(lengths)
    # Give more time to lengths around parent length
    time_fracs = []
    for i, ln in enumerate(lengths):
        if parent is not None and ln == p_len:
            time_fracs.append(0.4)
        elif parent is not None:
            time_fracs.append(0.15)
        else:
            time_fracs.append(1.0 / n_lengths)
    # Normalize
    s = sum(time_fracs)
    time_fracs = [f / s for f in time_fracs]

    for li, n in enumerate(lengths):
        if time.monotonic() > overall_deadline - 60:
            break

        # Sub-deadline for this length
        cum_frac = sum(time_fracs[:li + 1])
        sub_deadline = t_start + cum_frac * (overall_deadline - t_start - 60) + 30
        sub_deadline = min(sub_deadline, overall_deadline - 60)

        if sub_deadline - time.monotonic() < 30:
            continue

        print(f"\n{'='*60}")
        print(f"Length n={n}, time budget={sub_deadline - time.monotonic():.0f}s")
        print(f"{'='*60}")

        # Generate inits
        inits = _gen_inits(n, rng, parent)
        scored = []
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr))

        scored.sort(key=lambda x: x[0])
        print(f"Generated {len(scored)} inits. Top 5: "
              f"{[(n2, f'{s:.6f}') for s, n2, _ in scored[:5]]}")

        # Select top diverse candidates (at most 4)
        candidates = []
        used_names = set()
        for sc, name, arr in scored:
            # Diversity: skip if too similar to already selected
            if len(candidates) >= 4:
                break
            candidates.append((sc, name, arr))
            used_names.add(name)

        if not candidates:
            candidates = [(2.0, 'uniform', _norm(np.ones(n)))]

        # Optimize each candidate with time allocation
        n_cands = len(candidates)
        cand_time = (sub_deadline - time.monotonic() - 20) / n_cands

        for ci, (sc_init, name, arr_init) in enumerate(candidates):
            cand_deadline = time.monotonic() + cand_time - 5
            cand_deadline = min(cand_deadline, sub_deadline - 10)
            if time.monotonic() > cand_deadline:
                continue

            print(f"\n  --- Candidate {ci+1}: '{name}', init_score={sc_init:.8f} ---")

            sc_cand, seq_cand = _optimize_single(arr_init, cand_deadline, name, rng)

            if sc_cand < best_score:
                best_score = sc_cand
                best_seq = seq_cand.copy()
                print(f"  *** NEW BEST: {best_score:.8f} (n={n}, init='{name}') ***")

    # --- Final global polish on the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}")
        print(f"Final polish: score={best_score:.8f}, n={len(best_seq)}")
        print(f"{'='*60}")

        a = _norm(best_seq)
        fin_deadline = deadline - 10

        # High-beta LSE
        for k_f, beta_f in [(500, 15000), (1000, 40000), (2000, 100000)]:
            if time.monotonic() > fin_deadline - 20:
                break
            remaining = fin_deadline - time.monotonic() - 20
            if remaining < 8:
                break
            sc_f, a_f = _run_lbfgs(a, beta_f, k_f, remaining * 0.35, 800)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                a = a_f
                print(f"  Final LSE K={k_f}, beta={beta_f}: {best_score:.8f}")

        # Subgradient
        if time.monotonic() < fin_deadline - 10:
            sc_sg, a_sg = _subgrad_polish(a, max_iters=30, deadline=fin_deadline - 5)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                a = a_sg
                print(f"  Final subgrad: {best_score:.8f}")

        # Quick mirror descent polish
        if time.monotonic() < fin_deadline - 15:
            sc_md, a_md = _mirror_descent(a, fin_deadline - 5, eta0=0.01,
                                           beta_start=100.0, beta_end=2000.0,
                                           min_iters=100)
            if sc_md < best_score:
                best_score = sc_md
                best_seq = a_md.copy()
                print(f"  Final MD: {best_score:.8f}")

    # Prepare output
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
```
