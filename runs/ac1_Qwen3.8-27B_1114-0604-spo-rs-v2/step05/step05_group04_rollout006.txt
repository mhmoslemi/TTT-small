"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full auto-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_adaptive(a, n_iters=300, gamma_init=0.7, gamma_end=1.0,
                    top_frac_init=0.1, top_frac_end=0.5,
                    soft_alpha=30.0, restart_period=60):
    """Fienup with adaptive parameters and periodic restarts."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * progress
        gamma = gamma_init + (gamma_end - gamma_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) * gamma

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        gate = 1.0 / (1.0 + np.exp(-soft_alpha * a_new))
        a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= restart_period:
            a = best_a * (1.0 + 0.03 * np.random.default_rng(it).standard_normal(n))
            a = np.clip(a, 0.0, None)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
            no_improve = 0

    return best_a


def nesterov_subgrad(a, n_iters=5000, top_k=20, step_init=0.02,
                     mu_init=0.9, mu_end=0.999):
    """Nesterov-accelerated subgradient with top-K peak averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    y = a.copy()
    t = 1.0
    step = step_init

    for it in range(n_iters):
        c = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break

        k = len(c)
        K = min(top_k, k)
        top_idx = np.argsort(c)[::-1][:K]
        avg_peak = float(c[top_idx].mean())
        sc = 2.0 * n * float(c[top_idx].max()) / (S_y * S_y)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute subgradient: average over top-K peaks
        g = np.zeros(n)
        for idx in top_idx:
            k_pos = int(idx)
            j_start = max(0, k_pos - n + 1)
            j_end = min(k_pos, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                g[j_start:j_end + 1] += 2.0 * seg[::-1]
        g /= K

        # Full subgradient of objective
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (g / S2 - 2.0 * avg_peak / S3)

        # Nesterov update
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        step = 2.0 / (t * t * grad_norm + 1e-10)
        step = min(step, 0.05)

        x_new = y - step * grad * grad_norm
        x_new = np.clip(x_new, 0.0, None)
        S_new = x_new.sum()
        if S_new < 1e-12:
            break
        x_new = x_new / S_new

        mu = mu_init + (mu_end - mu_init) * (it / max(1, n_iters))
        y = x_new + mu * (x_new - y)
        y = np.clip(y, 0.0, None)
        S_y2 = y.sum()
        if S_y2 < 1e-12:
            y = x_new.copy()
            S_y2 = S_new
        y = y / S_y2

        a = x_new
        t = t_new

    return best_a


def surgical_mass_transfer(a, top_frac=0.1, amount=0.02, n_transfers=5):
    """Surgical perturbation: transfer mass from high-blame to low-blame positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_transfers):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Compute blame vector
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        blame = np.zeros(n)
        if j_end >= j_start:
            blame[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Identify transfer fractions
        n_transfer = max(1, int(n * top_frac))
        sorted_idx = np.argsort(blame)
        sources = sorted_idx[n - n_transfer:]  # high blame
        targets = sorted_idx[:n_transfer]  # low blame

        # Perform transfer
        source_vals = a[sources].copy()
        transfer_amount = amount * source_vals
        total_transfer = float(transfer_amount.sum())
        if total_transfer < 1e-15:
            continue

        a[sources] -= transfer_amount
        target_weights = np.ones(n_transfer) / n_transfer
        a[targets] += total_transfer * target_weights

        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def coordinate_descent_peaks(a, n_iters=2000, n_peaks=5):
    """Coordinate descent: iteratively reduce the top peaks by adjusting individual elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # For each peak, find the worst contributor and reduce it
        for k_pos in top_idx:
            k_pos = int(k_pos)
            j_start = max(0, k_pos - n + 1)
            j_end = min(k_pos, n - 1)
            if j_end < j_start:
                continue
            # Products a[i]*a[k-i] for i in [j_start, j_end]
            i_range = np.arange(j_start, j_end + 1)
            products = a[i_range] * a[k_pos - i_range]
            worst = i_range[np.argmax(products)]
            # Reduce this element slightly
            reduction = 0.001 * a[worst]
            a[worst] -= reduction
            # Distribute to a low-contribution position
            # Find position with minimum blame across all peaks
            blame = np.zeros(n)
            for kp in top_idx:
                kp = int(kp)
                js = max(0, kp - n + 1)
                je = min(kp, n - 1)
                if je >= js:
                    blame[js:je+1] += a[js:je+1][::-1]
            # Add to position with least blame (and nonzero potential)
            target_candidates = np.where(blame < np.median(blame))[0]
            if len(target_candidates) > 0:
                target = target_candidates[np.argmax(a[target_candidates] + 1.0/n)]
                a[target] += reduction

        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        if it % 200 == 0 and it > 0:
            print(f"    CD iter {it}: score={sc:.8f}")

    return best_a

        sc = compute_score(a_out)
        improved = update_best(sc, a_out, f"P3-{name}")
        if improved:
            a_cur = a_out.copy() / a_out.sum()

    # L-BFGS polish after Fienup
    if time.monotonic() < phase3_end - 60:
        for beta in [2000, 10000]:
            if time.monotonic() > phase3_end:
                break
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=5000, maxfun=50000)
            update_best(sc, a_opt, f"P3-LBFGS b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Surgical mass transfer + re-optimize (15% time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 4: Surgical perturbations (deadline {phase4_end - t_start:.0f}s) ===")

    for trial in range(5):
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 60:
            break

        # Different amounts of mass transfer
        fracs = [0.05, 0.1, 0.15, 0.2, 0.3]
        amounts = [0.01, 0.02, 0.03, 0.05, 0.08]
        frac = fracs[trial % len(fracs)]
        amt = amounts[trial % len(amounts)]

        a_surg = surgical_mass_transfer(best_a, top_frac=frac, amount=amt, n_transfers=8)
        sc = compute_score(a_surg)
        update_best(sc, a_surg, f"URGICAL f={frac} a={amt}")

        # Quick L-BFGS polish from surgical result
        if time.monotonic() < phase4_end - 30:
            sc2, a_opt = run_lbfgs(a_surg, 5000, maxiter=3000, maxfun=30000)
            update_best(sc2, a_opt, f"URGICAL+LBFGS f={frac}")

    # Coordinate descent on peaks
    if time.monotonic() < phase4_end - 60:
        a_cd = coordinate_descent_peaks(best_a, n_iters=1000, n_peaks=5)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD-DESCENT")
        if sc < best_score:
            # Polish
            sc2, a_opt = run_lbfgs(a_cd, 10000, maxiter=5000, maxfun=50000)
            update_best(sc2, a_opt, "CD+LBFGS")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final aggressive polish (10% time)
    # ============================================================
    print(f"\n=== Phase 5: Final polish (deadline {deadline - t_start:.0f}s) ===")

    # Highest beta L-BFGS
    if time.monotonic() < deadline - 30:
        for beta in [50000, 100000, 200000]:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic() - 20
            if remaining < 20:
                break
            maxiter = min(8000, max(200, int(remaining * 10)))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter*10)
            update_best(sc, a_opt, f"FINAL b={beta}")

    # One more Fienup cycle
    if time.monotonic() < deadline - 30:
        a_final = fienup_adaptive(best_a, n_iters=150, gamma_init=0.6, gamma_end=0.9,
                                  top_frac_init=0.1, top_frac_end=0.4, soft_alpha=25.0)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-FIENUP")
        if time.monotonic() < deadline - 20:
            sc2, a_opt = run_lbfgs(a_final, 50000, maxiter=3000, maxfun=30000)
            update_best(sc2, a_opt, "FINAL-COMBO")

    # Explore alternative length
    if time.monotonic() < deadline - 60 and n_alt != n_primary:
        print(f"  Trying n={n_alt}...")
        if parent is not None:
            a_alt = np.interp(np.linspace(0, 1, n_alt),
                              np.linspace(0, 1, len(parent)), parent)
        else:
            a_alt = np.ones(n_alt) * 0.5
        a_alt = np.maximum(a_alt, 0.0)
        a_alt = a_alt / a_alt.sum()
        sc, a_opt = run_lbfgs(a_alt, 500, maxiter=5000, maxfun=50000)
        update_best(sc, a_opt, f"ALT-n={n_alt}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
