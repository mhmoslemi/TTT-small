"""
Adaptive LSE-LBFGS with GS-Fienup initialization and peak-shaving refinement.
Key improvements: much higher beta values (up to 10^6), adaptive beta annealing,
and a peak-shaving heuristic that directly targets the maximum convolution values.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: subtract max before exp
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] * I(0 <= k-i < n)
    # = 2 * cross_correlation(p, a)[i]
    # Computed via convolution: rev_a * p, extract positions [n-1, 2n-2]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    # Full gradient of score = 2n * L / S^2
    # d(score)/da[i] = 2n * (g[i]/S^2 - 2*L/S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize (returns value and gradient)."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S  # Normalize to sum=1
    bounds = [(0.0, 1.0)] * n  # Since we normalize, elements in [0,1]
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def gs_fienup(a, n_iters=300, alpha_init=0.7, alpha_final=0.1):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    
    # Target: flat convolution
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-30)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-30)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def subgradient_polish(a, n_iters=500, top_m=10):
    """Subgradient method with Polyak step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Lower bound: score >= 2n/(2n-1) (theoretical, not tight)
    # Better lower bound: use the uniform sequence score = 2
    # Even better: use the current best as a proxy
    sc_lb = 1.0  # Theoretical lower bound (flat convolution)
    
    step = 0.01
    decay = 0.999
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # dc[k]/da[i] = 2*a[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                # a[k-i] for i in [i_start, i_end]
                vals = a[k - i_end:k - i_start + 1][::-1]
                sg[i_start:i_end + 1] += 2.0 * vals
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a


def peak_shave(a, n_trials=200):
    """Try to reduce peaks by scaling down contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Elements contributing to c[k_star]:
        # c[k_star] = sum_i a[i]*a[k_star-i] for valid i
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        
        if i_end < i_start:
            continue
        
        # Contribution of each element to the peak
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            contrib[i] = a[k_star - i]
        
        # Try reducing the top contributors
        top_count = min(10, int(np.sum(contrib > 1e-10)))
        if top_count == 0:
            continue
        top_locals = np.argsort(contrib)[::-1][:top_count]
        
        # Pick a random subset to reduce
        n_reduce = max(1, top_count // 3)
        chosen = rng.choice(top_count, size=min(n_reduce, top_count), replace=False)
        
        a_trial = a.copy()
        for idx in chosen:
            i = int(top_locals[idx])
            a_trial[i] *= rng.uniform(0.3, 0.8)
        
        S_trial = a_trial.sum()
        if S_trial < 1e-12:
            continue
        a_trial = a_trial / S_trial
        sc = compute_score(a_trial)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a
