"""
Hybrid optimizer combining LP-based direction finding, adaptive mirror descent,
L-BFGS with LSE smoothing, and exact coordinate descent for minimizing the
self-convolution peak ratio score 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: (1) LP-based direction search inspired by SOTA, (2) symmetric
constraint enforcement, (3) anti-Hann and diverse initializations, (4) adaptive
step-size mirror descent with exact line search.
"""

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=10000, eta0=0.05, decay=0.9995):
    """Mirror descent with LSE-smoothed objective and adaptive step."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        g_centered = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(g_centered))
        if gnorm < 1e-15:
            break
        # Adaptive step: scale by score gap
        sc_gap = max(exact_sc - 1.0, 0.01)
        eta_eff = eta * sc_gap / (gnorm + 1e-10)
        eta_eff = min(eta_eff, 1.0)
        exponent = -eta_eff * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        eta *= decay
    return best_a


def lp_direction(a, M=None, timeout=60):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b (normalized) or None on failure.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a = np.clip(a, 0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return None
    a_norm = a / S
    if M is None:
        conv_aa = fast_conv(a_norm)
        M = float(conv_aa.max())

    b = cp.Variable(n)
    # Build cross-convolution constraints: (a_norm * b)[k] <= M for k=0..2n-2
    # (a_norm * b)[k] = sum_j a_norm[j] * b[k-j] (valid j)
    # This is a linear function of b for each k.
    # Use a matrix formulation: constraint_matrix @ b <= M * ones(2n-1)
    # constraint_matrix[k, j] = a_norm[j] if 0 <= k-j <= n-1, else 0
    # This is equivalent to: for each k, sum over j of a_norm[j]*b[k-j]
    # which is the k-th element of np.convolve(a_norm, b)

    # Build the sparse constraint matrix
    # For k=0..2n-2, constraint is: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_norm[j] * b[k-j] <= M
    # Let's build it as: C @ b <= M * 1
    # C[k, j] = a_norm[k - j] if 0 <= k - j <= n - 1, else 0
    # This is a Toeplitz-like matrix.

    from scipy.sparse import csr_matrix, diags

    # Build C as a sparse matrix
    # C[k,j] = a_norm[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_norm[k - j])

    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    objective = cp.Maximize(cp.sum(b))
    constraints = [C @ b <= M * np.ones(2 * n - 1), b >= 0]

    prob = cp.Problem(objective, constraints)

    # Try solvers in order
    solvers_to_try = ['ECOS', 'CLARABEL', 'SCS', 'GLPK', 'CBC']
    result = None
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if b.value is not None:
                result = b.value.copy()
                break
        except Exception:
            continue
        try:
            prob.solve()
            if b.value is not None:
                result = b.value.copy()
                break
        except Exception:
            continue

    if result is None:
        return None
    result = np.clip(result, 0, 1000.0)
    S_b = result.sum()
    if S_b < 1e-12:
        return None
    return result / S_b


def lp_iteration(a, t=0.1, M=None):
    """One LP-based iteration: find LP direction and blend."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    a_norm = a / S

    conv_aa = fast_conv(a_norm)
    M = float(conv_aa.max())

    b = lp_direction(a_norm, M, timeout=60)
    if b is None:
        return a

    # Blend
    a_new = (1 - t) * a_norm + t * b
    a_new = np.clip(a_new, 0, 1000.0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a
    a_new = a_new / S_new
    return a_new


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    return best_a


def multi_peak_subgrad(a, n_iters=3000, f_lb=1.0, top_k=5):
    """Subgradient using top-k peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.1
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(c))
        for rank, idx in enumerate(top_idx):
            weights[idx] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)
        Mak_weighted = float(np.sum(weights * c))
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        sg = np.zeros(n)
        S2 = S_a * S_a
        S3 = S2 * S_a
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        step = step0 * (0.999 ** it)
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))
        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    return best_a


def targeted_peak_flatting(a, n_sweeps=10, n_top=50):
    """Targeted coordinate descent: reduce peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        top_count = min(n_top, max(1, total_len // 3))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 0.9, 0.95]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S_curr = a.sum()
                if S_curr > 1e-12:
                    a = a / S_curr
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S_curr = a.sum()
            if S_curr > 1e-12:
                a = a / S_curr
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def greedy_peak_reduction(a, n_iters=500):
    """
    Greedy: at each step, find the coordinate that most contributes to the peak,
    reduce it, and redistribute to coordinates that least contribute.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Find which coordinate contributes most to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()

        # Reduce top contributor
        top_idx = int(np.argmax(contrib))
        top_global = j_start + top_idx
        a_old = a[top_global]
        if a_old < 1e-10:
            break

        # Try reducing by various amounts
        best_reduction = 0
        best_score = best_sc
        best_a_copy = a.copy()

        for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
            reduction = a_old * frac
            a_trial = a.copy()
            a_trial[top_global] -= reduction
            # Redistribute to bottom contributors
            bottom_count = max(1, total_len - top_count) if 'total_len' in dir() else max(1, n // 10)
            bottom_count = max(1, len(contrib) // 5)
            bottom_local = np.argsort(contrib)[:bottom_count]
            per = reduction / bottom_count
            for li in bottom_local:
                a_trial[j_start + li] += per
            a_trial = np.clip(a_trial, 0, 1000.0)
            S_t = a_trial.sum()
            if S_t < 1e-12:
                continue
            a_trial = a_trial / S_t
            sc_t = compute_score(a_trial)
            if sc_t < best_score:
                best_score = sc_t
                best_reduction = reduction
                best_a_copy = a_trial.copy()

        if best_score < best_sc:
            a = best_a_copy
            best_sc = best_score
            best_a = a.copy()
        else:
            # No improvement; try a smaller reduction
            break

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Hann window (sin^2)
    inits.append(np.sin(np.pi * (x + 0.5 / n)) ** 2)
    # Anti-Hann (cos^2 - large at edges)
    inits.append(np.cos(np.pi * (x - 0.5)) ** 2)
    # Raised cosine (Hann)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5))))
    # Quadratic edge-favored
    inits.append((2.0 * x - 1.0) ** 2)
    # Inverse square root (edge-favored, smoother)
    inits.append(1.0 / np.sqrt(np.maximum(x * (1.0 - x), 0.01) + 0.1))
    # Bimodal
    inits.append(np.exp(-8.0 * (x - 0.15) ** 2) + np.exp(-8.0 * (x - 0.85) ** 2))
    # Flat with edge boost
    inits.append(1.0 + 0.5 * ((2.0 * x - 1.0) ** 2))
    # Gaussian center
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2))
    # Random
    inits.append(0.1 + 0.9 * rng.random(n))
    # S-curve
    inits.append(1.0 / (1.0 + np.exp(-10.0 * (x - 0.5))))
    return [np.clip(a, 0.01, 10.0) for a in inits]


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 6))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / max(1, len(bottom_local))
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p

        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        n_work = 8192

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: Diverse multi-start with mirror descent (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: Multi-Start Mirror Descent (n={n_work}) ---")

    p_for_init = parent if (parent is not None and len(parent) == n_work) else None
    inits = build_initializations(n_work, rng, p_for_init)

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S
        sc_init = compute_score(a_init)
        report(sc_init, a_init, f"init_{i}")

        # Quick mirror descent
        a_md = mirror_descent(a_init, beta=80, n_iters=2000, eta0=0.06, decay=0.999)
        sc = compute_score(a_md)
        report(sc, a_md, f"mirror_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: LP-based refinement (30% time)
    # ==================================================================
    phase2_deadline = t_start + 0.45 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: LP-Based Refinement ---")

    n_lp_iters = 12
    for it in range(n_lp_iters):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        t_blend = 0.05 + 0.05 * (it / n_lp_iters)  # Gradually increase blend
        a_lp = lp_iteration(best_a, t=t_blend)
        sc = compute_score(a_lp)
        improved = report(sc, a_lp, f"LP iter {it}")

        if not improved:
            # Try with different t
            for t_try in [0.02, 0.1, 0.15]:
                a_try = lp_iteration(best_a, t=t_try)
                sc_try = compute_score(a_try)
                report(sc_try, a_try, f"LP iter {it} t={t_try}")
                if sc_try < best_score:
                    break

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement with adaptive beta (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        remaining_s = phase3_deadline - time.monotonic()
        if remaining_s < 30:
            break
        maxiter = int(min(50000, max(3000, remaining_s * 30)))
        maxfun = int(min(200000, max(10000, remaining_s * 100)))

        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")

        # If time, try perturbed restart with same beta
        if time.monotonic() < phase3_deadline - 60 and beta >= 1000:
            a_pert = perturbed_restart(best_a, rng, scale=0.01)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS beta={beta} pert")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated local search (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 10
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining_s = phase4_deadline - time.monotonic()
        if remaining_s < 60:
            break

        scale = 0.02 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Mirror descent
        a_md = mirror_descent(a_pert, beta=200, n_iters=1500, eta0=0.04, decay=0.999)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"ILS#{trial} mirror")

        # L-BFGS
        beta = 2000 + int(5000 * rng.random())
        maxiter = int(min(20000, max(2000, remaining_s * 10)))
        maxfun = int(min(80000, max(5000, remaining_s * 40)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-10), beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} lbfgs b={beta}")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Subgradient + Peak Flattening + Final Polish (10% time)
    # ==================================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final Polish ---")

    # Nesterov subgradient
    a_ns = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
    sc = compute_score(a_ns)
    report(sc, a_ns, "nesterov")

    # Multi-peak subgradient
    a_mp = multi_peak_subgrad(best_a, n_iters=2000, f_lb=1.0, top_k=5)
    sc = compute_score(a_mp)
    report(sc, a_mp, "multi_peak_sg")

    # Targeted peak flattening
    a_tpf = targeted_peak_flatting(best_a, n_sweeps=10, n_top=50)
    sc = compute_score(a_tpf)
    report(sc, a_tpf, "peak_flat")

    # Greedy peak reduction
    if time.monotonic() < phase5_deadline - 60:
        a_gr = greedy_peak_reduction(best_a, n_iters=300)
        sc = compute_score(a_gr)
        report(sc, a_gr, "greedy_pr")

    # Another Nesterov pass
    if time.monotonic() < phase5_deadline - 60:
        a_ns2 = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "nesterov2")

    # Final high-beta L-BFGS
    for beta in [100000, 500000]:
        if time.monotonic() > phase5_deadline - 30:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000)
        report(sc, a_opt, f"final beta={beta}")

    # Final mirror descent at high precision
    if time.monotonic() < phase5_deadline - 20:
        a_md_final = mirror_descent(best_a, beta=10000, n_iters=2000, eta0=0.01, decay=0.999)
        sc = compute_score(a_md_final)
        report(sc, a_md_final, "mirror_final")

    # One more targeted peak flattening
    if time.monotonic() < phase5_deadline - 15:
        a_tpf2 = targeted_peak_flatting(best_a, n_sweeps=15, n_top=30)
        sc = compute_score(a_tpf2)
        report(sc, a_tpf2, "peak_flat2")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
