"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using progressive LSE optimization with

Key improvements: interleaved LSE/subgradient phases, longer sequences,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[m] = sum_k w[k] * a[k-m] for m=0..n-1, w has length >= 1."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    nw = w.size
    if n == 0 or nw == 0:
        return np.empty(n, dtype=np.float64)
    M = n + nw - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u > 0) & (rho >= 1)
    # Find largest rho such that u[rho-1] > (cssv[rho-1] - target_sum)/rho
    valid = u > 0
    if not valid.any():
        return np.full(n, target_sum / n)
    rv = rho[valid]
    uv = u[valid]
    csv = cssv[valid]
    thresh = (csv - target_sum) / rv
    good = uv > thresh
    if not good.any():
        rho_val = 1
    else:
        rho_val = int(rv[good][-1])
    theta = (cssv[rho_val - 1] - target_sum) / rho_val
    return np.maximum(v - theta, 0.0)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed objective and gradient for top-k active positions."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 2.0:
        return _score(a), a

    est_eval_t = 0.005 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(20, min(mf, maxfun_cap, 6000))
    maxiter = max(15, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 30}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=200, deadline=None, verbose=False):
    """Projected subgradient descent with adaptive step size."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within adaptive tolerance of max
        tol = max(1e-10, 0.005 * dmax)
        active = d >= dmax - tol
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())
        n_act = min(n_act, 200)  # Cap active set size for efficiency
        # Keep only the top n_act if too many
        if n_act > 200:
            idx_top = np.argpartition(d, -200)[-200:]
            active = np.zeros_like(d)
            active[idx_top] = True
            n_act = 200

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        mean_d = float(np.sum(d[active])) / n_act
        grad = (4.0 * n) * (corr / (S * S) - mean_d / S)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size: try multiple scales
        best_new_sc = best_sc
        best_new_a = None
        for scale in (0.002, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4):
            eta = scale / gnorm
            q = a - eta * grad
            q = _project_simplex(q, 1.0)
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new_a = q

        if best_new_a is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_new_a.copy()
            if verbose and it % 50 == 0:
                print(f"    SG iter {it}: {best_sc:.8f}")
        a = best_new_a

        # Early stop if no progress for a while
        if it > 10 and best_new_sc >= best_sc + 1e-12:
            # Try larger steps
            pass

    return best_sc, best_a


def _adam_lse(a0, beta, k, time_limit, lr=0.01, momentum=0.9):
    """Custom Adam-like optimizer for LSE objective with projection."""
    a = _norm(a0).astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()
    t_end = t0 + time_limit

    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    eps = 1e-8
    b1, b2 = 0.9, 0.999

    while time.monotonic() < t_end:
        t += 1
        val, grad = _lse_obj_grad(a, beta, k)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        # Adaptive learning rate that decreases over time
        lr_t = lr / (1.0 + 0.01 * t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new, 1.0)
        sc = _score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # Periodic check
        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    Adam iter {t}: score={best_sc:.8f}, elapsed={elapsed:.0f}s")

    return best_sc, best_a


def _gradient_step(a, beta=1000, k=100):
    """Single gradient step with LSE, returns new a."""
    val, grad = _lse_obj_grad(a, beta, k)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a
    return _project_simplex(a - 0.01 / gnorm * grad, 1.0)


def _spectral_init(n, rng):
    """Generate a non-negative sequence with shaped power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.2):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _velvet_init(n, rng):
    """Initialize using a 'velvet' sequence: smooth with gentle modulation."""
    x = np.linspace(0, 1, n)
    # Sum of several smooth bumps
    a = np.ones(n) * 0.3
    for _ in range(rng.integers(3, 7)):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: progressive LSE + intensive subgradient with Adam optimizer."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    # Target length: use 8192 (power of 2 for FFT efficiency)
    target_n = 8192

    print(f"Parent available: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    # --- Generate initializations ---
    inits = []

    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))
        for amp in (0.05, 0.1, 0.2):
            inits.append((f'pn{amp}', _smooth_perturb(_upsample(parent, target_n), rng, amp)))

    # Spectral inits with varied parameters
    for i in range(8):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # Velvet inits
    for i in range(3):
        inits.append((f'velvet{i}', _velvet_init(target_n, rng)))

    # Cosine modulations
    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3, 5):
        for amp in (0.2, 0.4, 0.6):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
            a = np.clip(a, 0.01, None)
            inits.append((f'cos{f}_{amp}', a / a.sum()))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top 5: {[f'{s:.6f}({nm})' for s, nm, _ in scored[:5]]}")

    # Select top candidates for optimization (diverse)
    best_score = scored[0][0] if scored else 2.0
    best_seq = scored[0][2].copy() if scored else _norm(np.ones(target_n))

    # Use top 2 candidates
    candidates = scored[:2] if len(scored) >= 2 else [(best_score, 'top', best_seq)]

    # --- Main optimization: alternating LSE and subgradient ---
    n_cands = len(candidates)
    time_per_cand = (deadline - t_start - 200.0) / n_cands

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        cand_start = time.monotonic()
        cand_deadline = min(cand_start + time_per_cand, deadline - 200.0)

        print(f"\n=== Candidate {ci+1}: '{name}', n={target_n}, init_score={sc_init:.8f} ===")

        a = _norm(arr_init)
        cur_best_sc = _score(a)
        cur_best_a = a.copy()

        # Progressive stages: (K, beta, method, time_fraction)
        stages = [
            (3, 30, 'lbfgs', 0.08),
            (5, 60, 'lbfgs', 0.08),
            (10, 120, 'lbfgs', 0.08),
            (10, 120, 'subgrad', 0.06),
            (20, 250, 'lbfgs', 0.10),
            (20, 250, 'subgrad', 0.06),
            (40, 500, 'lbfgs', 0.10),
            (40, 500, 'subgrad', 0.08),
            (80, 1000, 'lbfgs', 0.10),
            (80, 1000, 'subgrad', 0.06),
            (150, 2000, 'lbfgs', 0.10),
            (150, 2000, 'subgrad', 0.06),
            (300, 4000, 'lbfgs', 0.08),
            (300, 4000, 'subgrad', 0.06),
        ]

        total_frac = sum(s[3] for s in stages)
        avail_time = max(10.0, cand_deadline - cand_start - 10.0)

        cum_frac = 0.0
        for stage_idx, (k, beta, method, frac) in enumerate(stages):
            cum_frac += frac
            stage_deadline = cand_start + (cum_frac / total_frac) * avail_time + 5.0
            stage_deadline = min(stage_deadline, cand_deadline - 5.0)
            remaining = stage_deadline - time.monotonic()
            if remaining < 5.0:
                break

            print(f"  Stage {stage_idx}: K={k}, beta={beta}, method={method}, time={remaining:.0f}s")

            if method == 'lbfgs':
                # Split into sub-calls for robustness
                n_sub = min(3, max(1, int(remaining / 30)))
                sub_time = remaining / n_sub
                for si in range(n_sub):
                    if time.monotonic() > stage_deadline - 2.0:
                        break
                    sc, a_new = _run_lbfgs(a, beta, k, sub_time, 500)
                    if sc < cur_best_sc:
                        cur_best_sc = sc
                        cur_best_a = a_new.copy()
                        print(f"    LSE sub{si}: {cur_best_sc:.8f}")
                    a = _norm(a_new)
            elif method == 'subgrad':
                sc, a_new = _subgrad_polish(a, max_iters=500, deadline=stage_deadline)
                if sc < cur_best_sc:
                    cur_best_sc = sc
                    cur_best_a = a_new.copy()
                    print(f"    Subgrad: {cur_best_sc:.8f}")
                a = _norm(a_new)

            # Track global best
            if cur_best_sc < best_score:
                best_score = cur_best_sc
                best_seq = cur_best_a.copy()
                print(f"  *** Global best: {best_score:.8f} ***")

        print(f"  Cand {ci+1} done: {cur_best_sc:.8f}, elapsed={time.monotonic()-cand_start:.0f}s")

    # --- Final intensive polishing phase ---
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0
    a = _norm(best_seq)

    # High-beta LSE
    for k_f, beta_f in [(500, 6000), (1000, 12000), (2000, 25000)]:
        if time.monotonic() > final_deadline - 60:
            break
        rem = min(120, final_deadline - time.monotonic() - 60)
        if rem < 10:
            break
        sc, a_new = _run_lbfgs(a, beta_f, k_f, rem, 800)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            a = _norm(a_new)
            print(f"  Final LSE K={k_f}: {best_score:.8f}")
        else:
            a = _norm(a_new)

    # Intensive subgradient
    if time.monotonic() < final_deadline - 30:
        sc, a_sg = _subgrad_polish(a, max_iters=2000, deadline=final_deadline - 30)
        if sc < best_score:
            best_score = sc
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Adam optimizer on best
    if time.monotonic() < final_deadline - 30:
        rem = min(120, final_deadline - time.monotonic() - 30)
        if rem > 20:
            sc, a_adam = _adam_lse(_norm(best_seq), beta=8000, k=500, time_limit=rem, lr=0.005)
            if sc < best_score:
                best_score = sc
                best_seq = a_adam.copy()
                print(f"  Adam: {best_score:.8f}")

    # Random perturbations + quick re-optimize
    if time.monotonic() < final_deadline - 60:
        for restart in range(4):
            if time.monotonic() > final_deadline - 30:
                break
            p = _smooth_perturb(_norm(best_seq), rng, strength=0.05)
            # Quick subgradient
            sc_r, a_r = _subgrad_polish(p, max_iters=200, deadline=final_deadline - 30)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_f, a_f = _subgrad_polish(_norm(best_seq), max_iters=500, deadline=final_deadline)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final final subgrad: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
