- Compute `P = |FFT(a)|^2`
- Apply `P_new = P / (P + lambda)` (normalization) - this compresses large values
- Or: `P_new = max(P, floor)` (raise the floor)
- Or: `P_new = P^alpha` for alpha < 1 (power compression)

Actually, the most effective approach I've seen in the literature for this type of problem is:

**Algorithm: Proximal Gradient with LSE**
1. Compute the LSE-smoothed max: `f(x) = (1/beta) * log(sum(exp(beta * c_k)))`
2. Compute the gradient
3. Take a gradient step
4. Project onto non-negative orthant (clip at 0)
5. Normalize (optional, to keep sum constant)

The key parameter is `beta`. High beta → closer to the true max (sharper, more local). Low beta → smoother (more global).

A good strategy is to use an "annealing" schedule: start with low beta for global exploration, then increase beta for local refinement.

Let me also add a **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)** variant with momentum for accelerated convergence.

def lbfgs_lse(a0, beta, maxiter=5000, ftol=1e-18):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)

        score = 2.0 * n * L / (S * S)
        # d(score)/d(a[i]) = 2n * (grad_L[i] / S^2 - 2*L / S^3)

        return score, grad

                   bounds=[(0, 1000)] * n,

    a_out = np.clip(res.x, 0, 1000)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

"""
Optimized sequence search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Aggressive L-BFGS with adaptive LSE smoothing + Fienup spectral shaping
+ FISTA acceleration + multi-scale perturbation restarts.
Focuses on precise local optimization from the parent, with targeted exploration
to escape local minima via spectral perturbation and sub-sampling.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(x, beta):
    """Compute LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -600, 0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_scaled / exp_sum

    # Gradient via FFT: grad_L[i] = 2 * conv(p, rev_a)[n-1+i]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = x[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1: 2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L * x / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    def func(x):
        s, g = lse_score_grad(x, beta)
        return s, g

    res = minimize(
        func, a0, method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-20, 'gtol': 1e-16, 'maxiter': maxiter,
                 'maxfun': maxiter * 50, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def fienup_multi(a_init, n_iters=300, top_frac_init=0.15, top_frac_end=0.5,
                 gamma=0.8, power_mode=False):
    """Fienup with adaptive top-fraction and power compression."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = n * 0.5
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2

        scale = np.ones_like(P)
        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if above.any() and thresh > 1e-30:
            if power_mode:
                # Power-law: P^alpha where alpha < 1
                alpha = 0.5 + 0.3 * progress
                P_max = float(P.max())
                P_ratio = P[above] / P_max
                scale[above] = (thresh / P_max) ** (1.0 - alpha) * P_ratio ** alpha / P_ratio
                scale[above] = np.sqrt(np.maximum(thresh / P[above], 1e-10)) ** gamma
            else:
                ratio = thresh / P[above]
                scale[above] = np.sqrt(np.maximum(ratio, 1e-10)) ** gamma

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_soft_compress(a_init, n_iters=300, strength_init=2.0, strength_end=0.5):
    """Fienup with soft spectral compression: P/(P+lambda)^0.5 type."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = n * 0.5
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        strength = strength_init + (strength_end - strength_init) * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())

        # Soft compression: reduce peaks by factor that depends on P/P_mean
        lam = P_mean * strength
        # P_new = P / (1 + (P-lam)/lam)^0.5 for P > lam, else P
        P_new = P.copy()
        above = P > lam
        if above.any():
            excess = (P[above] - lam) / lam
            P_new[above] = P[above] / np.sqrt(1.0 + excess)

        # Preserve total power
        P_new = P_new * (P.sum() / max(float(P_new.sum()), 1e-30))

        A_amp = np.sqrt(P) + 1e-30
        A_new = A * (np.sqrt(P_new) / A_amp)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a_init, n_iters=5000, step_init=0.02, top_k=1):
    """Polish using exact subgradient of max(conv(a,a))."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S * S  # keep as is, scale invariant anyway

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find peak lags
        if top_k >= len(c):
            peak_mask = np.ones(len(c), dtype=bool)
        else:
            peak_mask = np.zeros(len(c), dtype=bool)
            peak_mask[np.argsort(c)[::-1][:top_k]] = True

        # Compute subgradient using FFT
        # grad[i] = 2 * sum_{k: peak} (a[k-i] + a[i-k]) for valid indices
        # This equals 4 * ifft(p * fft(a))[n-1-i] type expression
        # More precisely: sum_k p[k] * a[k-i] = cross-correlation
        # Use: grad = 2 * convolve(peak_indicator, a_reversed) shifted
        p = peak_mask.astype(np.float64) / max(peak_mask.sum(), 1)
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        rev_a = a[::-1]
        RA = np.fft.rfft(rev_a, M2_fft)
        PP = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * PP, M2_fft)[:M2]
        grad = 2.0 * conv_res[n - 1: 2 * n - 1]

        S2 = S * S
        S3 = S2 * S
        grad_obj = 2.0 * n * (grad / S2 - 2.0 * M_val * a / S3)

        gn = float(np.linalg.norm(grad_obj))
        if gn < 1e-18:
            break

        # Adaptive step
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.05)

        a_new = a - step * grad_obj
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new * S
        step *= 0.9995

    return best_a


def fista_optimize(a0, beta, n_iters=10000):
    """FISTA (accelerated proximal gradient) for LSE objective."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    y = a.copy()
    t_mom = 1.0

    # Initial step size estimation
    S = float(a.sum())
    if S < 1e-12:
        S = 1.0
    step = (S * S) / (4.0 * n * (float(a.max()) + 1e-10) + 1e-10)
    step = min(step, 1.0)

    best_a = a.copy()
    best_sc = compute_score(a)
    stalled = 0

    for it in range(n_iters):
        # Gradient at y
        _, grad_y = lse_score_grad(y, beta)

        # Gradient step + projection
        a_new = y - step * grad_y
        a_new = np.maximum(a_new, 0.0)  # project onto non-negative
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new  # normalize

        # Backtracking if function increased
        sc_new = compute_score(a_new)
        if sc_new > compute_score(y) * 1.01 + 1e-10:
            step *= 0.5
            a_new = y - step * grad_y
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_mom = t_new

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 500:
                step *= 0.9

    return best_a


def spectral_perturb(a, rng, strength=0.05):
    """Add random phase perturbation in frequency domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = np.abs(A)
    phase = np.angle(A)
    # Add small random phase noise

    A_new = amp * np.exp(1j * (phase + phase_noise))
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S * float(a.sum())


def subsample_refine(a, factor, rng, n_fine_iters=2000):
    """Subsample, optimize, then upsample and continue."""
    n = len(a)
    # Subsample

    nc = len(coarse)

    x_old = np.linspace(0, 1, nc)

    a_blend = 0.5 * a_up + 0.5 * a
    a_blend = np.maximum(a_blend, 0.0)
    S = a_blend.sum()
    if S < 1e-12:

        # Polish with LSE

            a_pol = run_lbfgs(a_f, 5000, maxiter=min(3000, int((p2_end - time.monotonic()) * 20)))
            try_update(a_pol, f"FIENUP+LSE-{name}")

    print(f"After Phase 2: {best_sc:.10f}")

    # ============================================================
    # Phase 3: Subgradient + FISTA polish (15% time)
    # ============================================================
    p3_end = t0 + 0.85 * (total - 25.0)
    print(f"\n=== Phase 3: Subgrad+FISTA (until t={p3_end-t0:.0f}s) ===")

    # Subgradient with different top_k
    for top_k in [1, 2, 5]:
        if time.monotonic() > p3_end - 20:
            break
        a_sg = subgradient_polish(best_a, n_iters=5000, step_init=0.01, top_k=top_k)
        try_update(a_sg, f"SUBGRAD k={top_k}")

    # FISTA with various betas
    for beta in [1000, 5000, 20000]:
        if time.monotonic() > p3_end - 20:
            break
        remaining = p3_end - time.monotonic()
        n_it = int(min(8000, max(500, remaining * 15.0)))
        a_f = fista_optimize(best_a, beta, n_iters=n_it)
        try_update(a_f, f"FISTA2 b={beta}")

    print(f"After Phase 3: {best_sc:.10f}")

    # ============================================================
    # Phase 4: Perturbation + re-optimization (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Perturb+Optimize (until t={deadline-t0:.0f}s) ===")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Different perturbation strategies
        strat = trial % 4
        if strat == 0:
            # Small additive noise
            a_p = best_a + rng.standard_normal(n) * 0.01 * best_a.mean()
        elif strat == 1:
            # Spectral phase perturbation
            a_p = spectral_perturb(best_a, rng, strength=0.03 + 0.02 * trial / n_trials)
        elif strat == 2:
            # Multiplicative noise
            a_p = best_a * np.exp(rng.standard_normal(n) * 0.02)
        else:
            # Subsample-refine
            a_p = subsample_refine(best_a, factor=2, rng=rng)

        a_p = np.clip(a_p, 0.0, 1000.0)
        S_p = a_p.sum()
        if S_p < 1e-6:
            continue

        # Quick LSE optimization from perturbed start
        maxiter = int(min(3000, max(500, remaining * 10.0 / (1 + trial * 0.1))))
        a_opt = run_lbfgs(a_p, 10000, maxiter=maxiter)
        try_update(a_opt, f"PERT-{trial} s={strat}")

        # If improved, do one more round of high-beta LSE
        if compute_score(a_opt) < best_sc + 1e-10 and time.monotonic() < deadline - 30:
            a_opt2 = run_lbfgs(a_opt, 50000, maxiter=2000)
            try_update(a_opt2, f"PERT-{trial} s={strat} h")

    # ============================================================
    # Final: one more LSE pass with very high beta
    # ============================================================
    if time.monotonic() < deadline - 5:
        a_final = run_lbfgs(best_a, 100000, maxiter=2000)
        try_update(a_final, "FINAL-HB")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_sc:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'='*60}")
    return out
