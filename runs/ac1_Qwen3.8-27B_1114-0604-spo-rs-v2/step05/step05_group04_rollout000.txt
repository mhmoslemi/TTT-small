"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed L-BFGS, adaptive Fienup iterations, spectral peak suppression,
and top-M subgradient descent. Key innovations: FFT-based LSE gradient,
progressive spectral flattening, and peak-aware Fienup with non-negativity-aware
amplitude targets.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """
    LSE-smoothed objective and gradient.
    Score = 2n * L / S^2 where L = LSE_soft_max(conv(x,x))
    Uses FFT circular convolution trick for efficient gradient.
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: subtract max already done
    exp_scaled = np.exp(np.maximum(scaled, -500.0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient via circular convolution trick
    p = exp_scaled / exp_sum  # length M
    # Pad p to M_fft length
    p_pad = np.zeros(M_fft // 2 + 1)
    p_pad[:M] = p
    # Need full FFT length for circular conv
    p_full = np.zeros(M_fft)
    p_full[:M] = p
    # For IFFT we need the full complex spectrum
    P_full = np.fft.rfft(p, M_fft)
    grad_pad = 2.0 * np.fft.irfft(P_full * A, M_fft)
    grad_L = grad_pad[:n]  # dL/dx for first n elements

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x):
        return lse_objective_and_grad(x, beta)[0]

    def jac_wrapper(x):
        return lse_objective_and_grad(x, beta)[1]

    res = minimize(
        obj_wrapper, a0, jac=jac_wrapper,
        method='L-BFGS-B', bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = float(a_out.sum())
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_adaptive(a, n_iters=300, gamma=0.75, top_frac_init=0.10,
                    top_frac_end=0.50, soft_gate=40.0, sum_preserve=True,
                    freq_floor=0.01):
    """
    Adaptive Fienup with:
    - Progressive top-fraction growth
    - Non-negativity-aware amplitude target (slight DC boost)
    - Soft sigmoid gate for non-negativity
    - Frequency floor to avoid zeroing out bins
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.1
        S = n * 0.1
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * (1 - np.cos(np.pi * progress)) * 0.5

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)

        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            ratio = thresh / P[above]
            # Compress above-threshold bins towards threshold
            scale[above] = (ratio ** 0.5) * gamma
            # Floor to avoid zeroing
            scale[above] = np.maximum(scale[above], freq_floor)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        # Soft non-negativity via sigmoid
        if soft_gate > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_gate * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break

        if sum_preserve:
            a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_peak_aware(a, n_iters=300, n_peaks=10, strength=0.5):
    """
    Peak-aware Fienup: identify the specific time-domain peaks in the convolution
    and find which frequency bins contribute most to them, then suppress those bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    M = 2 * n - 1
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = fast_conv(a)

        # Find top peaks in convolution
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # For each peak at lag k, the contribution from frequency bin b is
        # proportional to P[b] * cos(2*pi*b*k/M_fft + phase_c_k)
        # Simplified: bins with high P contribute more to all peaks
        # Weighted suppression: bins that correlate with peak lags get more suppression
        suppress = np.zeros(len(P))
        for k in top_idx:
            k = int(k)
            # Frequency response at lag k
            resp = np.cos(2.0 * np.pi * np.arange(len(P)) * k / M_fft)
            suppress += np.abs(resp) * P

        # Normalize and apply
        suppress_max = suppress.max()
        if suppress_max < 1e-12:
            break
        suppress = suppress / suppress_max

        # Progressive strength
        s = strength * (0.5 + 0.5 * progress)
        scale = 1.0 - s * suppress
        scale = np.maximum(scale, 0.05)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def top_m_subgradient(a, n_iters=3000, top_m=20, step_init=0.02, momentum=0.9):
    """
    Subgradient descent focusing on top-M convolution peaks.
    Uses momentum for faster convergence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    vel = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Momentum
        vel = momentum * vel - step * grad
        a = a + vel
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Adaptive step
        gap = sc - sc_lb
        if gap < 1e-8:
            step *= 0.9
        else:
            gn = float(np.linalg.norm(grad))
            step = min(step * 1.01, max(gap / (gn * gn + 1e-30), 1e-12))
            step = min(step, 0.05)

    return best_a


def spectral_phase_optimize(a, n_iters=200, n_trials=5, temperature=0.3):
    """
    Simulated annealing on the phase of the spectrum.
    Keep amplitude fixed (from current a), randomize phase, accept if better.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        temp = temperature * (0.95 ** it)
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)

        # Random phase perturbation
        delta = temp * rng.standard_normal(len(phase))
        phase_new = phase + delta

        A_new = amp * np.exp(1j * phase_new)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new < best_sc or (rng.random() < np.exp((best_sc - sc_new) / max(temp, 1e-10))):
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()

    return best_a

                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[i] = best_val
            a[i] = max(0.0, best_val)

        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, np.clip(parent, 0.0, 1000.0)).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.3 + 0.7 * x)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Strategy: LSE-L-BFGS + Adaptive Fienup + Peak-Aware Fienup + Subgradient + Phase SA.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 6962

    print(f"Working length: n={n_work}")
    print(f"Parent: {'yes (score=' + str(compute_score(parent)) + ')' if parent is not None else 'no'}")

    # Evaluate parent
    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # =================================================================
    # Stage 1: Quick LSE-L-BFGS (10% time)
    # =================================================================
    stage1_end = t_start + 0.10 * total_budget
    print(f"\n=== Stage 1: LSE-LBFGS (deadline {stage1_end - t_start:.0f}s) ===")

    if parent is not None:
        for beta in [50, 200]:
            if time.monotonic() > stage1_end:
                break
            maxiter = min(3000, int((stage1_end - time.monotonic()) // 0.2))
            if maxiter < 100:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=maxiter, maxfun=maxiter * 15)
            update_best(sc, a_opt, f"S1-LBFGS b={beta}")

    # Also try from diverse inits at working length
    if time.monotonic() < stage1_end:
        inits = build_initializations(n_work, rng, parent)
        for i, a0 in enumerate(inits[:3]):
            if time.monotonic() > stage1_end:
                break
            sc, a_opt = run_lbfgs(a0, 100, maxiter=1000, maxfun=15000)
            update_best(sc, a_opt, f"S1-INIT{i}")

    print(f"After Stage 1: best = {best_score:.10f}")

    # =================================================================
    # Stage 2: Adaptive Fienup variants (25% time)
    # =================================================================
    stage2_end = t_start + 0.35 * total_budget
    print(f"\n=== Stage 2: Fienup (deadline {stage2_end - t_start:.0f}s) ===")

    a_cur = best_a.astype(np.float64).copy()
    a_cur = a_cur / a_cur.sum()

    fienup_configs = [
        ("F-adapt", dict(n_iters=250, gamma=0.7, top_frac_init=0.08, top_frac_end=0.55, soft_gate=30.0)),
        ("F-adapt2", dict(n_iters=250, gamma=0.5, top_frac_init=0.12, top_frac_end=0.65, soft_gate=50.0)),
        ("F-peak", None),  # peak-aware
        ("F-adapt3", dict(n_iters=200, gamma=0.9, top_frac_init=0.05, top_frac_end=0.40, soft_gate=20.0)),
    ]

    for name, cfg in fienup_configs:
        if time.monotonic() > stage2_end:
            break
        if cfg is None:
            a_out = fienup_peak_aware(a_cur, n_iters=200, n_peaks=15, strength=0.4)
        else:
            a_out = fienup_adaptive(a_cur, **cfg)
        sc = compute_score(a_out)
        improved = update_best(sc, a_out, f"S2-{name}")
        if improved:
            a_cur = a_out / a_out.sum()

    # More Fienup iterations from best
    if time.monotonic() < stage2_end:
        for trial in range(3):
            if time.monotonic() > stage2_end:
                break
            gamma = 0.4 + 0.2 * trial
            a_out = fienup_adaptive(a_cur, n_iters=200, gamma=gamma,
                                    top_frac_init=0.1, top_frac_end=0.6, soft_gate=30.0)
            sc = compute_score(a_out)
            improved = update_best(sc, a_out, f"S2-F{trial}")
            if improved:
                a_cur = a_out / a_out.sum()

    print(f"After Stage 2: best = {best_score:.10f}")

    # =================================================================
    # Stage 3: LSE-L-BFGS with increasing beta (30% time)
    # =================================================================
    stage3_end = t_start + 0.65 * total_budget
    print(f"\n=== Stage 3: LBFGS increasing beta (deadline {stage3_end - t_start:.0f}s) ===")

    betas = [100, 300, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if time.monotonic() > stage3_end:
            break
        remaining = stage3_end - time.monotonic()
        if remaining < 30:
            break
        # Scale maxiter by remaining time and problem size
        n = len(best_a)
        cost_per_iter = n / 500.0  # rough estimate of seconds per iteration
        maxiter = int(min(20000, remaining / max(cost_per_iter, 0.1)))
        maxiter = max(maxiter, 500)
        maxfun = maxiter * 10

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"S3-LBFGS b={beta}")
        if not improved and beta >= 1000:
            # If no improvement at high beta, break
            if beta >= 5000:
                break

    print(f"After Stage 3: best = {best_score:.10f}")

    # =================================================================
    # Stage 4: Top-M subgradient (20% time)
    # =================================================================
    stage4_end = t_start + 0.85 * total_budget
    print(f"\n=== Stage 4: Subgradient (deadline {stage4_end - t_start:.0f}s) ===")

    for top_m in [10, 20, 40, 80]:
        if time.monotonic() > stage4_end:
            break
        remaining = stage4_end - time.monotonic()
        n_iters = int(min(5000, remaining * 10))
        if n_iters < 100:
            break
        a_sg = top_m_subgradient(best_a, n_iters=n_iters, top_m=top_m,
                                  step_init=0.015, momentum=0.85)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"S4-SG m={top_m}")

    # Peak-aware Fienup after subgradient
    if time.monotonic() < stage4_end - 30:
        a_pa = fienup_peak_aware(best_a, n_iters=200, n_peaks=20, strength=0.35)
        sc = compute_score(a_pa)
        update_best(sc, a_pa, "S4-PEAK")

    print(f"After Stage 4: best = {best_score:.10f}")

    # =================================================================
    # Stage 5: Final refinement (15% time)
    # =================================================================
    print(f"\n=== Stage 5: Final refinement ===")

    # LSE-LBFGS with very high beta
    if time.monotonic() < deadline - 60:
        for beta in [5000, 10000, 20000]:
            if time.monotonic() > deadline - 30:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=30000)
            update_best(sc, a_opt, f"S5-LBFGS b={beta}")

    # Spectral phase optimization
    if time.monotonic() < deadline - 30:
        a_sp = spectral_phase_optimize(best_a, n_iters=100, temperature=0.2)
        sc = compute_score(a_sp)
        update_best(sc, a_sp, "S5-PHASE")

    # Final Fienup + LBFGS chain
    if time.monotonic() < deadline - 20:
        a_f = fienup_adaptive(best_a, n_iters=100, gamma=0.6,
                              top_frac_init=0.1, top_frac_end=0.5, soft_gate=40.0)
        sc = compute_score(a_f)
        update_best(sc, a_f, "S5-FIENUP")

        if time.monotonic() < deadline - 10:
            sc2, a_opt = run_lbfgs(a_f, 10000, maxiter=2000, maxfun=20000)
            update_best(sc2, a_opt, "S5-FINAL")

    # Perturbation restarts
    if time.monotonic() < deadline - 10:
        for trial in range(5):
            if time.monotonic() > deadline - 5:
                break
            noise = 0.01 * (trial + 1)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = float(a_p.sum())
            if S < 1e-12:
                continue
            a_p = a_p / S
            a_r = fienup_adaptive(a_p, n_iters=100, gamma=0.7,
                                  top_frac_init=0.1, top_frac_end=0.5)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"S5-RESTART{trial}")

    # =================================================================
    # Output
    # =================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
