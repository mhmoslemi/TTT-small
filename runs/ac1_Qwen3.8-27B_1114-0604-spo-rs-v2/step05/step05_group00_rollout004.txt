LP-refinement based optimizer for minimizing 2n*max(conv(a,a))/S^2.
Combines LP-based "response" finding with L-BFGS polish and coordinate descent.

# ... helper functions ...

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows.append(np.full(len(js), k))
        cols.append(js)
        vals.append(f[k - js])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_refine(f, M=None):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Return normalized g."""
    n = len(f)
    if M is None:
        c = fast_conv(f)
        M = float(np.max(c))
    
    # Build constraint matrix
    A = build_conv_matrix(f)
    
    # LP
    g_var = cp.Variable(n, nonneg=True)
    constraints = [A @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return None
    
    if g_var.value is None:
        return None
    
    g = np.asarray(g_var.value).ravel()
    g = np.maximum(g, 0)
    S = g.sum()
    if S < 1e-12:
        return None
    return g / S * float(f.sum())  # Rescale to match f's sum

def lp_refine_active_set(f, M=None, max_rounds=50, tol=1e-8):
    """
    LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Uses active set / cutting plane method.
    """
    n = len(f)
    if M is None:
        c = fast_conv(f)
        M = float(np.max(c))
    
    # Start with the top constraints
    c_f = fast_conv(f)  # = conv(f, f), these are the "diagonal" constraints
    # Actually, the constraints are (f*g)_k <= M, not (f*f)_k <= M
    # At the initial point g = f, the tightest constraints are where (f*f)_k is largest
    
    top_k = 20
    active = np.argsort(c_f)[::-1][:top_k]
    
    for round in range(max_rounds):
        # Build LP with only active constraints
        # Constraint for row k: sum_j A[k,j] * g_j <= M
        # where A[k,j] = f[k-j] for valid j
        
        # For a small number of active constraints, we can build the matrix quickly
        rows_data = []
        cols_data = []
        vals_data = []
        rhs = np.zeros(len(active))
        
        for i, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            rows_data.append(np.full(len(js), i))
            cols_data.append(js)
            vals_data.append(f[k - js])
            rhs[i] = M
        
        rows_data = np.concatenate(rows_data)
        cols_data = np.concatenate(cols_data)
        vals_data = np.concatenate(vals_data)
        A_active = csc_matrix((vals_data, (rows_data, cols_data)), shape=(len(active), n))
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A_active @ g_var <= rhs]
        objective = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GLOP)
        
        if g_var.value is None:
            break
        
        g = np.asarray(g_var.value).ravel()
        g = np.maximum(g, 0)
        
        # Check if solution violates any inactive constraints
        c_g = np.zeros(2 * n - 1)
        # Compute (f*g)_k for all k
        # Use FFT
        M_fft = 1 << (2 * n - 1).bit_length()
        F = np.fft.rfft(f, M_fft)
        G = np.fft.rfft(g, M_fft)
        c_g = np.fft.irfft(F * G, M_fft)[:2*n-1]
        
        # Find most violated inactive constraint
        violations = c_g - M
        # Mask out active constraints
        for k in active:
            violations[k] = -np.inf
        
        if violations.max() < tol:
            break  # Feasible!
        
        # Add most violated constraint
        new_k = int(np.argmax(violations))
        active = np.append(active, new_k)
    
    # Return normalized g
    S = g.sum()
    if S < 1e-12:
        return None
    return g / S * float(f.sum())
