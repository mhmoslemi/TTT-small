    At each step, compute the subgradient of max(c) and take a step.

    a = a / S

        c = fast_conv(a)

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using progressive LSE-LBFGS 
with Fienup exploration and subgradient refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective for auto-convolution peak.
    Returns (score, gradient).
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -100.0, 0.0)  # Numerical stability
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Softmax weights on convolution values
    p = exp_scaled / exp_sum
    
    # dL/dx[j] = 2 * sum_m x[m] * p[m+j]
    # Computed as 2 * conv(x_rev, p)[n-1 : 2n-1]
    x_rev = x[::-1]
    conv_len = n + M - 1
    conv_fft = 1 << (conv_len - 1).bit_length()
    XR = np.fft.rfft(x_rev, conv_fft)
    PR = np.fft.rfft(p, conv_fft)
    conv_res = np.fft.irfft(XR * PR, conv_fft)
    grad_L = 2.0 * conv_res[n-1 : 2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True,
        bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out

def fienup_v3(a, n_iters=300, top_frac=0.3, gamma=0.7, soft_alpha=0.0):
    """
    Fienup targeting A^2 (auto-convolution spectrum).
    Modifies magnitude of A to compress dynamic range.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        
        A = np.fft.rfft(a, M_fft)
        abs_A = np.abs(A)
        abs_A_max = abs_A.max()
        
        if abs_A_max < 1e-20:
            break
        
        # Compress dynamic range of |A|
        # This makes |A|^2 (and hence A^2 magnitude) more uniform
        A_norm = abs_A / abs_A_max
        A_compressed = A_norm ** gamma
        ratio = A_compressed / (A_norm + 1e-30)

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        a_new = a_new / S_new
        a = a_new

def subgradient_peak(a, n_iters=3000, top_m=1, step_init=0.02, decay=0.9995):
    """
    Subgradient descent targeting top-m peaks of auto-convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top-m peaks
        if top_m <= 1:
            k_star = int(np.argmax(c))
            top_idx = [k_star]
            weights = [1.0]
        else:
            top_idx = np.argsort(c)[::-1][:top_m].astype(int).tolist()
            # Weights proportional to peak height
            peak_vals = np.array([c[k] for k in top_idx])
            weights = (peak_vals / peak_vals.sum()).tolist()
        
        # Compute subgradient
        sg = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k_off = k - np.arange(n)
            mask = (k_off >= 0) & (k_off < n)
            sg[mask] += w * 2.0 * a[k_off[mask]]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        
        # Adaptive step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, 0.05)
        # Bound step by objective gap
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        step_bound = max((sc - lower_bound) / (gn * gn + 1e-20), 1e-12)
        step_use = min(step, step_bound)
        
        a_new = a - step_use * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= decay
    
    return best_a
