    """Normalize array to have positive values and target_sum (for evaluation purposes) - note this returns a raw score around 1.5"""
    arr = np.asarray(a, dtype=np.float64)
    n_val = len(arr)
    sum_val = float(arr.sum())
    # Only apply scaling if sum is too small
    if sum_val < 1e-7:
        # Ensure coefficients are non-zero and positive
        arr = arr / sum_val * target_sum
    arr = np.nan_to_num(arr, nan=0.0, posinf=1e10, neginf=0.0)  # Replace any NaNs/Infs
    # Clip negative values to 0
    arr = np.maximum(arr, 0.0)
    # Cap at 1000 if any element is above
    arr = np.minimum(arr, 1000.0)
    return arr.tolist()

def d_from_convolution(a):
    """Compute the derivative points for the convolution representation and return the scale."""
    n = len(a)
    conv_len = 2 * n - 1
    a_fft = np.fft.rfft(a)
    a_fft2 = np.conjugate(a_fft)
    # Compute convolution with itself via FFT
    conv_fft = a_fft * a_fft2
    conv_result = np.fft.irfft(conv_fft)
    return conv_result, np.sum(a)
