    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500.0, 500.0)
    exp_s = np.exp(scaled)
    logsum = float(np.log(exp_s.sum()))
    L = c_max + logsum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/dx[m] = 2 * (p cross-correlate x)[m]
    M_fft = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    X = np.fft.rfft(x, M_fft)
    corr = np.fft.irfft(P * np.conj(X), M_fft)[:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * corr / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    x0 = np.maximum(a0.astype(np.float64), 0.0)
    S = x0.sum()
    if S < 1e-12:
        x0 = np.ones(n) / n
    else:
        x0 = x0 / S
    bounds = [(0.0, 1e6)] * n

    def obj_func(x):
        return lse_obj_grad(x, beta)

    res = minimize(obj_func, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100})
    a_out = np.clip(res.x, 0.0, 1e6)
    if a_out.sum() < 1e-12:
        a_out = x0.copy()
    return compute_score(a_out), a_out


def fienup_flat(a, n_iters=300, alpha_start=0.05, alpha_end=0.5):
    """Fienup: blend spectrum towards flat to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    return best


def fienup_threshold(a, n_iters=300):
    """Fienup with quantile thresholding on spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        q = 0.70 + 0.25 * progress
        thresh = float(np.quantile(P, q)) * (1.0 + 0.5 * (1.0 - progress))
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    return best


def spectral_equalize(a, n_iters=500):
    """
    Novel: directly target time-domain peak equalization via spectral
    amplitude adjustment. Identify over-represented time lags and
    suppress their spectral content.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        c = fast_conv(a)
        c_mean = float(c.mean())
        # Weight: peaks above mean get suppressed
        excess = np.maximum(c - c_mean, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break

        # Compute how much each frequency bin contributes to the excess
        # Ifft of |A|^2 gives c. We want to reduce c at peak lags.
        # Use iterative approach: compute a "correction" signal.
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2

        # Strategy: compute the DFT of the excess, and use it to
        # guide amplitude reduction
        E = np.fft.rfft(excess, M_fft)
        # Where E has high magnitude, reduce P
        E_mag = np.abs(E)
        E_max = E_mag.max()
        if E_max < 1e-15:
            break
        E_norm = E_mag / E_max

        # Reduce amplitude where it contributes to peaks
        strength = 0.05 + 0.15 * progress
        P_new = P * (1.0 - strength * E_norm)
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    return best


def subgrad_anneal(a, n_iters=5000, lr_init=0.02, lr_end=0.001):
    """
    Subgradient method with adaptive active set: targets the top peaks
    with soft weighting, using FFT-based gradient computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        lr = lr_init * (lr_end / lr_init) ** progress
        c = fast_conv(a)
        S = a.sum()
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

        # Soft active set: weight peaks by how far above mean
        c_mean = float(c.mean())
        weight = np.maximum(c - c_mean, 0.0)
        w_sum = weight.sum()
        if w_sum < 1e-15:
            break
        p = weight / w_sum

        # Gradient: 2 * (p cross-correlate a)
        P = np.fft.rfft(p, M_fft)
        X = np.fft.rfft(a, M_fft)
        corr = np.fft.irfft(P * np.conj(X), M_fft)[:n]
        grad = 2.0 * corr / (S * S)
        grad -= grad.mean()  # Remove component along 1-vector

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        gap = sc - sc_lb
        if gap > 1e-12:
            lr = min(lr, 0.5 * gap / (gn * gn + 1e-10))
        lr = min(lr, 0.05)

        a -= lr * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best


def fista_lse(a0, beta, n_iters=5000):
    """FISTA on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0)
    S = a0.sum()
    if S < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    x = a0.copy()
    z = a0.copy()
    t_k = 1.0
    _, g0 = lse_obj_grad(z, beta)
    gn0 = float(np.linalg.norm(g0))
    L = max(gn0 + 1.0, 50.0)
    best_x = a0.copy()
    best_sc = compute_score(a0)

    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()
        f_y, g_y = lse_obj_grad(z, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new
        f_new, _ = lse_obj_grad(x_new, beta)
        quad = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quad:
            L *= 1.5
            if L > 1e12:
                break
            continue
        else:
            L = max(0.7 * L, 1.0)
        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        Sz = z.sum()
        if Sz > 1e-12:
            z = z / Sz
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()
    return best_x


def coord_descent_peaks(a, n_sweeps=10, top_lags=3, top_per_lag=50):
    """Coordinate descent on top-contributing elements at peak lags."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_lag_indices = np.argsort(c)[::-1][:top_lags]
        for lag_val in top_lag_indices:
            lag = int(lag_val)
            j_min = max(0, lag - n + 1)
            j_max = min(lag, n - 1)
            if j_max < j_min:
                continue
            js = np.arange(j_min, j_max + 1)
            contribs = a[js] * a[lag - js]
            total = contribs.sum()
            if total < 1e-15:
                continue
            n_top = min(top_per_lag, len(contribs))
            top_idx = np.argsort(contribs)[::-1][:n_top]
            for idx in top_idx:
                j = int(js[idx])
                old_val = a[j]
                if old_val < 1e-14:
                    continue
                for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return best


def diverse_init(n, kind):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif kind == 'raised_cos':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 'triangle':
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif kind == 'exponential':
        a = np.exp(-3.0 * x)
    elif kind == 'bimodal':
        a = np.exp(-25.0 * (x - 0.3) ** 2) + np.exp(-25.0 * (x - 0.7) ** 2)
    elif kind == 'sinc':
        a = np.sinc(3.0 * (x - 0.5))
        a = np.maximum(a, 0.0)
    elif kind == 'chirp':
        a = 1.0 + 0.4 * np.cos(6.0 * np.pi * x ** 2)
    elif kind == 'platypus':
        a = (1.0 + 0.3 * np.cos(4.0 * np.pi * x)) * np.exp(-2.0 * (x - 0.5) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: Fienup + spectral equalize -> L-BFGS multi-beta ->
    Subgradient anneal -> FISTA -> Coord descent -> Diverse restarts ->
    Multi-length exploration -> Final high-beta polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 60.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label=""):
        nonlocal best_score, best_a
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64 and np.all(p >= 0) and np.all(np.isfinite(p)):
            parent_a = np.clip(p, 0, 1000.0)
            sc_p = compute_score(parent_a)
            print(f"Parent: n={len(parent_a)}, score={sc_p:.8f}")
            try_update(sc_p, parent_a, "parent")
    except Exception:
        pass

    if best_a is None:
        best_a = diverse_init(4096, 'uniform')
        try_update(compute_score(best_a), best_a, "init-uniform")

    n_base = len(best_a)
    print(f"Starting: score={best_score:.8f}, n={n_base}")

    # ============================================================
    # Phase 1: Spectral shaping (15% time)
    # ============================================================
    p1_end = t_start + 0.15 * (total - 60.0)
    print(f"\n=== Phase 1: Spectral shaping (t < {p1_end - t_start:.0f}s) ===")

    a_f1 = fienup_flat(best_a, n_iters=250, alpha_start=0.05, alpha_end=0.4)
    try_update(compute_score(a_f1), a_f1, "fienup-flat")

    a_f2 = fienup_threshold(best_a, n_iters=250)
    try_update(compute_score(a_f2), a_f2, "fienup-thresh")

    a_se = spectral_equalize(best_a, n_iters=400)
    try_update(compute_score(a_se), a_se, "spec-equalize")

    # Combine: run spectral equalize from best fienup result
    a_comb = spectral_equalize(best_a, n_iters=300)
    a_comb2 = fienup_flat(a_comb, n_iters=100, alpha_start=0.1, alpha_end=0.3)
    try_update(compute_score(a_comb2), a_comb2, "comb-seq")

    # ============================================================
    # Phase 2: L-BFGS multi-beta (30% time)
    # ============================================================
    p2_end = t_start + 0.45 * (total - 60.0)
    print(f"\n=== Phase 2: L-BFGS multi-beta (t < {p2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas_phase2 = [300, 800, 2000, 5000, 10000, 20000, 40000]
    for beta in betas_phase2:
        if time.monotonic() > p2_end - 30:
            break
        remaining = p2_end - 30 - time.monotonic()
        if remaining < 20:
            break
        maxiter = min(15000, int(remaining * 40))
        maxfun = min(300000, int(remaining * 800))
        print(f"    beta={beta}, maxiter={maxiter}...")
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        if try_update(sc, a_opt, f"lbfgs-b{beta}"):
            a_work = a_opt.copy()

    # Second pass on best
    if time.monotonic() < p2_end - 60:
        for beta in [10000, 30000]:
            if time.monotonic() > p2_end - 30:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=160000)
            try_update(sc, a_opt, f"lbfgs2-b{beta}")

    # ============================================================
    # Phase 3: Subgradient + FISTA (15% time)
    # ============================================================
    p3_end = t_start + 0.60 * (total - 60.0)
    print(f"\n=== Phase 3: Subgrad + FISTA (t < {p3_end - t_start:.0f}s) ===")

    a_sg = subgrad_anneal(best_a, n_iters=4000, lr_init=0.015, lr_end=0.0005)
    try_update(compute_score(a_sg), a_sg, "subgrad")

    a_fi = fista_lse(a_sg, beta=3000, n_iters=3000)
    try_update(compute_score(a_fi), a_fi, "fista-3000")

    if time.monotonic() < p3_end - 60:
        a_fi2 = fista_lse(best_a, beta=10000, n_iters=2000)
        try_update(compute_score(a_fi2), a_fi2, "fista-10000")

    # L-BFGS polish after subgrad
    if time.monotonic() < p3_end - 30:
        sc, a_p = run_lbfgs(a_sg, 8000, maxiter=8000, maxfun=160000)
        try_update(sc, a_p, "sg+lbfgs")

    # ============================================================
    # Phase 4: Coordinate descent (10% time)
    # ============================================================
    p4_end = t_start + 0.70 * (total - 60.0)
    print(f"\n=== Phase 4: Coord descent (t < {p4_end - t_start:.0f}s) ===")

    a_cd = coord_descent_peaks(best_a, n_sweeps=15, top_lags=3, top_per_lag=80)
    try_update(compute_score(a_cd), a_cd, "cd")

    if compute_score(a_cd) < best_score and time.monotonic() < p4_end - 30:
        sc, a_p = run_lbfgs(a_cd, 10000, maxiter=5000, maxfun=100000)
        try_update(sc, a_p, "cd+lbfgs")

    # ============================================================
    # Phase 5: Diverse restarts + length exploration (15% time)
    # ============================================================
    p5_end = t_start + 0.85 * (total - 60.0)
    print(f"\n=== Phase 5: Restarts + lengths (t < {p5_end - t_start:.0f}s) ===")

    # Different initializations at same length
    inits = ['gaussian', 'raised_cos', 'bimodal', 'chirp', 'platypus']
    for init_kind in inits:
        if time.monotonic() > p5_end - 60:
            break
        a_init = diverse_init(n_base, init_kind)
        a_f = fienup_flat(a_init, n_iters=100, alpha_start=0.1, alpha_end=0.4)
        sc_f = compute_score(a_f)
        try_update(sc_f, a_f, f"init-{init_kind}")
        if time.monotonic() < p5_end - 90:
            sc, a_opt = run_lbfgs(a_f, 2000, maxiter=5000, maxfun=100000)
            try_update(sc, a_opt, f"init-{init_kind}-lbfgs")

    # Different lengths
    for n_try in [2048, 4096, 8192, 16384]:
        if time.monotonic() > p5_end - 120:
            break
        if n_try == n_base:
            continue
        a_init = diverse_init(n_try, 'raised_cos')
        a_f = fienup_flat(a_init, n_iters=150, alpha_start=0.05, alpha_end=0.4)
        # Also try spectral equalize
        a_se = spectral_equalize(a_f, n_iters=200)
        sc_se = compute_score(a_se)
        try_update(sc_se, a_se, f"len-{n_try}-se")
        a_base = a_se if sc_se < compute_score(a_f) else a_f
        if time.monotonic() < p5_end - 150:
            sc, a_opt = run_lbfgs(a_base, 2000, maxiter=8000, maxfun=160000)
            try_update(sc, a_opt, f"len-{n_try}-lbfgs")

    # ============================================================
    # Phase 6: Final high-beta polish (10% time)
    # ============================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        if remaining > 180:
            maxiter = min(25000, int(remaining * 30))
            sc, a_final = run_lbfgs(best_a, 20000, maxiter=maxiter,
                                     maxfun=min(500000, int(remaining * 500)))
            try_update(sc, a_final, "final-20k")

    if time.monotonic() < deadline - 60:
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=10000, maxfun=200000)
        try_update(sc, a_final, "final-50k")

    if time.monotonic() < deadline - 30:
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=100000)
        try_update(sc, a_final, "final-100k")

    # Final coord descent polish
    if time.monotonic() < deadline - 15:
        a_cd_final = coord_descent_peaks(best_a, n_sweeps=5, top_lags=2, top_per_lag=100)
        try_update(compute_score(a_cd_final), a_cd_final, "final-cd")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
