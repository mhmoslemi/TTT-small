def fast_conv(a):

        # So grad_j / 2 = (w ⋆ a)[j] = ifft(fft(w) * conj(fft(a)))[j] for real sequences
        # = ifft(fft(w) * fft(a))[j] (since a is real, conj(fft(a)) = fft(conj(a)) = fft(a) ... no)
        # For real a: conj(FFT(a))[k] = FFT(a)[-k] = FFT(a)[M-k]
        # So: (w ⋆ a)[j] = ifft(FFT(w) * conj(FFT(a)))[j]
        
        M2 = 4 * n
        W_f = np.fft.rfft(w, M2)
        A_f = np.fft.fft(a, M2)  # full complex FFT for the conjugate
        # cross_corr = ifft(FFT(w) * conj(FFT(a)))
        # Using real FFT: we need the full result
        # Let's use full FFT for simplicity
        W_full = np.fft.fft(w, M2)
        A_full = np.fft.fft(a, M2)
        cross = np.fft.ifft(W_full * np.conj(A_full)).real[:n]
        grad = 2.0 * cross
        
        # Also account for the normalization gradient (we're optimizing on the simplex)
        # Since we normalize after each step, the gradient of the normalized problem is:
        # grad_normalized = grad - mean(grad) (projected onto the tangent space of the simplex)
        # But since we re-normalize, we can just use grad directly and clip+normalize
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        a = a_new
        step *= decay
        
        if it % 1000 == 0:
            print(f"    ER iter {it}: score={sc:.8f}, step={step:.6f}")
    
    return best_a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a_j: p = softmax weights, L = LSE(c)
    # dL/da_j = sum_k p_k * dc_k/da_j = sum_k p_k * a[k-j]
    # Same computation as before: (p ⋆ a)[j]
    M2 = 4 * n
    P_full = np.fft.fft(p, M2)
    A_full = np.fft.fft(a, M2)
    cross = np.fft.ifft(P_full * np.conj(A_full)).real[:n]
    dL_da = 2.0 * cross  # factor of 2 from the quadratic
    
    # Full gradient of score = 2n * L / S^2
    # d(score)/da_j = 2n * (dL/da_j * S^2 - L * 2*S * 1) / S^4
    # = 2n * (dL/da_j / S^2 - 2*L / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def lse_continuation(a0, betas, maxiter_per_beta=3000):
    """Continue through beta values, using previous solution as init."""
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    for beta in betas:
        sc, a = run_lbfgs(a, beta, maxiter=maxiter_per_beta, maxfun=maxiter_per_beta*20)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        print(f"    LSE beta={beta}: score={sc:.8f}")
    return best_a

def active_set_refinement(a, n_iters=5000, n_active=20):
    """Subgradient method on the active set (top lags)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.02
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top active lags
        top_idx = np.argsort(c)[::-1][:n_active]
        p = np.zeros(2*n-1)
        p[top_idx] = 1.0 / n_active
        
        # Gradient
        M2 = 4 * n
        P_full = np.fft.fft(p, M2)
        A_full = np.fft.fft(a, M2)
        cross = np.fft.ifft(P_full * np.conj(A_full)).real[:n]
        grad = 2.0 * cross
        
        # Project onto tangent of simplex (remove mean)
        grad = grad - grad.mean()
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, 1.0 / (gn * gn + 1e-10))
        step = min(step, 0.05)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = normalize(a)
        step *= 0.9999
        
        if it % 1000 == 0:
            print(f"    AS iter {it}: score={sc:.8f}, n_active={n_active}")
    
    return best_a

def multiplicative_update(a, n_iters=3000, alpha=0.5):
    """
    Multiplicative (BLA) update for minimizing max autocorrelation.
    Inspired by the "Multiplicative Weight Update" / "Belief Propagation" approach.
    a_j <- a_j * (target_j / current_j)^alpha
    where target is based on the desired flat autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Target: flat autocorrelation -> a_j should be proportional to (1/sqrt(sum of neighbors))
    # In frequency domain: |A|^2 should be flat -> |A_k| = 1/sqrt(2n-1) for all k
    # This means a should be "equispectral"
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute "pressure" on each element: how much it contributes to the peaks
        # pressure_j = sum_k (c_k / c_max) * |dc_k/da_j| / n
        # Simplified: pressure_j = sum_k c_k * a[k-j] / sum_k c_k
        M2 = 4 * n
        C_full = np.fft.fft(c, M2)
        A_full = np.fft.fft(a, M2)
        contrib = np.fft.ifft(C_full * np.conj(A_full)).real[:n]
        contrib = np.maximum(contrib, 1e-20)
        
        # Target: uniform contribution
        target = contrib.sum() / n
        
        # Multiplicative update
        ratio = target / contrib
        a = a * (ratio ** alpha)
        a = np.maximum(a, 1e-10)
        a = normalize(a)
    
    return best_a

The QP can be solved with scipy.optimize.minimize (using the analytical gradient and Hessian).

Let me implement this. The key question is whether Q(p) is PSD.

Q(p)[j,l] = p_{j+l} for j,l in {0,...,n-1}.

This is a Hankel matrix. A Hankel matrix H[j,l] = h_{j+l} is PSD if and only if the sequence h = (h_0, h_1, ...) is a "positive definite" sequence (in the sense of Bochner's theorem, for the infinite case; for the finite case, it's related to the finite moment problem).

For p being a probability vector on {0,...,2n-2}, the finite Hankel matrix H[j,l] = p_{j+l} for j,l in {0,...,n-1} is PSD if and only if p is the "autocorrelation" of some sequence of length n. But we already know that p = conv(a,a) for the current a, so in that case Q is PSD!

def alternating_qp(a0 n_iters=100, beta=100):
    n = len(a0)
    a = a0.copy()
    
    for it in range(n_iters):
        c = fast_conv(a)
        # Compute weights
        scaled = beta * (c - c.max())
        p = np.exp(scaled)
        p /= p.sum()
        
        # Form Q = Hankel matrix with Q[j,l] = p[j+l]
        # We don't need to form it explicitly; we just need Q @ a
        # (Q @ a)[j] = sum_l p[j+l] * a[l]
        # This is a "Hankel matrix-vector product" which can be done in O(n^2) 
        # or O(n log n) with FFT
        
        # Actually, (Q @ a)[j] = sum_{l=0}^{n-1} p[j+l] * a[l]
        # This is a cross-correlation of p and a (reversed)
        # = (p * a_rev)[j + n - 1] where a_rev[l] = a[n-1-l]
        # Or simply computed as a sliding dot product
        
        # For the QP, we need:
        # gradient: 2 * Q @ a
        # Hessian: 2 * Q
        # objective: a^T Q a
        
        # Let's solve with scipy.optimize.minimize using BFGS or L-BFGS-B
        
        # Define the QP objective and gradient
        def qp_obj(x):
            qx = hankel_matvec(p, x)
            return float(x @ qx)
        
        def qp_grad(x):
            return 2.0 * hankel_matvec(p, x)
        
        # Solve
        from scipy.optimize import minimize as scipy_minimize
        bounds = [(0, 1000)] * n
        # Add equality constraint: sum(x) = 1
        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0}]
        
        res = scipy_minimize(qp_obj, a, jac=qp_grad, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 200, 'ftol': 1e-10})
        
        a = np.maximum(res.x, 0)
        a = normalize(a)
        
        if it % 10 == 0:
            print(f"    Alt-QP iter {it}: score={compute_score(a):.8f}")
    
    return a

But forming the Hankel matrix-vector product efficiently is important. Let me think about this.

(Q @ a)[j] = sum_{l=0}^{n-1} p[j+l] * a[l] for j = 0, ..., n-1

This is: for each j, take the (n-length) window of p starting at position j, and dot it with a.

This is equivalent to the cross-correlation of p and a (reversed):
Let a_rev[l] = a[n-1-l]. Then:
(Q @ a)[j] = sum_l p[j+l] * a[l] = sum_l p[j+l] * a_rev[n-1-l]

Let m = n-1-l: = sum_m p[j+n-1-m] * a_rev[m]

This is (p_rev ⋆ a_rev)[j+n-1] where p_rev[k] = p[2n-2-k]... this is getting confusing.

Let me just use the direct formula: for the QP, I'll form Q explicitly (as an n x n matrix) for n up to ~7000. That's 49 million entries, which is about 400 MB as float64. That might be too much.

Alternative: use the Kronecker/sparse structure. Actually, for the QP solve, I can just compute the gradient (which is 2*Q@a) using the FFT-based cross-correlation, and use L-BFGS-B which only needs the gradient (not the full Hessian).

For the Hessian (if I want to use Newton's method), I'd need Q itself. But L-BFGS-B approximates the inverse Hessian, so I just need the gradient.

The gradient of the QP objective a^T Q a is 2*Q*a, where (Q*a)[j] = sum_l p[j+l] * a[l].

I can compute Q*a efficiently using FFT:
- Pad p to length 2n-1 (it already is)
- (Q*a)[j] = sum_{l=0}^{n-1} p[j+l] * a[l]
- This is the "cross-correlation" of the sequence p (viewed as length 2n-1) with a (length n), evaluated at lags 0 to n-1.

Using FFT:
- Compute FFT of p (length 2n-1, padded to next power of 2)
- Compute FFT of a (same length)
- The cross-correlation can be obtained from the product of FFTs

Actually, let me define it more carefully:
(Q*a)[j] = sum_{l=0}^{n-1} p[j+l] * a[l]

This is the "correlation" of p and a at lag j (where we slide a over p starting at position j).

In terms of standard convolution: let's define b[l] = a[n-1-l] (reversed a).
Then (Q*a)[j] = sum_l p[j+l] * b[n-1-l] = sum_l' p[j+n-1-l'] * b[l'] where l' = n-1-l
= (p * b)[j+n-1] in the convolution sense? Let's check:
(p * b)[k] = sum_l p[l] * b[k-l]
We want sum_l' p[j+n-1-l'] * b[l'] = sum_l' b[l'] * p[j+n-1-l'] = (b * p)[j+n-1]

So (Q*a)[j] = (b * p)[j+n-1] where b = reversed a.

This can be computed with FFT:
- Compute FFT of b (length n, padded)
- Compute FFT of p (length 2n-1, padded)
- conv_bp = ifft(FFT(b) * FFT(p))
- (Q*a)[j] = conv_bp[j+n-1] for j = 0, ..., n-1

This is O(n log n)! Great.

"""
Multi-phase optimization for minimizing max autocorrelation of non-negative sequences.
Combines: GS-Fienup, LSE continuation with L-BFGS, Alternating QP with Hankel structure,
and Nesterov-accelerated projected subgradient. Novel contribution is the alternating
QP scheme that exploits the Hankel matrix structure of the quadratic subproblems.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_seq(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def hankel_matvec(p, a):
    """Compute Q@a where Q[j,l] = p[j+l]. Uses FFT for O(n log n)."""
    n = len(a)
    # (Q*a)[j] = sum_l p[j+l] * a[l]
    # = (reversed_a * p)[j + n - 1]
    a_rev = a[::-1]
    M = 3 * n + 10
    M_fft = 1 << (M - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, M_fft)
    P = np.fft.rfft(p, M_fft)
    conv = np.fft.irfft(A_rev * P, M_fft)
    return conv[n - 1: 2 * n - 1]


def cross_corr_fft(w, a):
    """Compute (w cross-correlate a)[j] = sum_m w[j+m]*a[m] for j=0..n-1."""
    n = len(a)
    M_fft = 1 << (3 * n + 10 - 1).bit_length()
    W = np.fft.fft(w, M_fft)
    A_full = np.fft.fft(a, M_fft)
    result = np.fft.ifft(W * np.conj(A_full)).real[:n]
    return result


def gs_fienup(a, n_iters=300, alpha=0.5):
    """GS-Fienup with magnitude update targeting flat convolution spectrum."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize_seq(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new)
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da_j = 2 * sum_k p_k * a[k-j] = 2 * (p cross-corr a)[j]
    cross = cross_corr_fft(p, a)
    dL_da = 2.0 * cross

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_grad(x, beta)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lse_continuation(a0, betas, maxiter_per_beta=2000):
    """Continuation method through LSE beta values."""
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    for beta in betas:
        t0 = time.monotonic()
        sc, a = run_lbfgs(a, beta, maxiter=maxiter_per_beta, maxfun=maxiter_per_beta * 20)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        print(f"    LSE beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")
    return best_a


def alternating_qp(a0, n_iters=30, beta=200, maxiter_qp=100):
    """
    Alternating optimization:
    1. Given a, compute p = softmax(beta * conv(a,a))
    2. Solve QP: min a^T Q(p) a s.t. sum(a)=1, a>=0 where Q(p)[j,l]=p[j+l]
    Novel: exploits Hankel structure for efficient gradient computation.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        # Compute weights (softmax of convolution)
        scaled = beta * (c - c.max())
        p = np.exp(scaled)
        p /= p.sum()

        # Solve QP: min a^T Q a s.t. sum(a)=1, a>=0
        # Gradient: 2 * Q @ a where (Q@a)[j] = sum_l p[j+l]*a[l]
        def qp_obj(x):
            qx = hankel_matvec(p, x)
            return float(x @ qx)

        def qp_grad(x):
            return 2.0 * hankel_matvec(p, x)

        bounds = [(0.0, 100.0)] * n
        res = minimize(
            qp_obj, a, jac=qp_grad,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter_qp, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        a_new = np.clip(res.x, 0.0, 100.0)
        a_new = normalize_seq(a_new)
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if it % 5 == 0:
            print(f"    Alt-QP iter {it}: score={sc:.8f}, max_p={p.max():.4f}")

    return best_a


def nesterov_subgrad(a0, n_iters=10000, top_k=20, step_init=0.03):
    """
    Nesterov-accelerated projected subgradient method.
    Uses top-k active lags for the subgradient.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_kest = 1.0
    step = step_init

    for it in range(n_iters):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_kest * t_kest)) / 2.0
        # Nesterov momentum
        momentum = (t_kest - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        y = normalize_seq(y)

        c = fast_conv(y)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-k active lags
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / top_k

        # Subgradient
        cross = cross_corr_fft(p, a)
        grad = 2.0 * cross
        # Project onto tangent of simplex
        grad = grad - grad.mean()

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step = min(step, 1.0 / (gn * gn + 1e-10))
        step = min(step, 0.05)

        a_prev = a.copy()
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = normalize_seq(a)
        t_kest = t_new
        step *= 0.99995

        if it % 2000 == 0:
            print(f"    Nesterov iter {it}: score={sc:.8f}, t={t_kest:.1f}")

    return best_a


def peak_shave(a0, n_iters=5000, n_peaks=5):
    """
    Targeted peak shaving: iteratively reduce elements contributing most
    to the current maximum lags, with careful redistribution.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.02

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute total gradient from all top peaks
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / n_peaks
        cross = cross_corr_fft(p, a)
        grad = 2.0 * cross
        grad = grad - grad.mean()

        # Only update elements in the top decile of |grad|
        thresh = np.percentile(np.abs(grad), 90)
        mask = np.abs(grad) > thresh

        a[mask] = a[mask] - step * grad[mask]
        a = np.maximum(a, 0.0)
        a = normalize_seq(a)
        step *= 0.99998

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, np.maximum(parent, 0.0)).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.2 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.sin(np.pi * x) ** 2 + 0.05)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * 3 * x), 0.01))

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Uses multi-phase approach:
    Phase 1: GS-Fienup for quick initialization
    Phase 2: LSE continuation with L-BFGS (main workhorse)
    Phase 3: Alternating QP (novel method)
    Phase 4: Nesterov subgradient + peak shaving
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    # Use n=6962 as primary (matches parent length)
    n = 6962
    if parent is not None and len(parent) >= 1024:
        n = len(parent)

    print(f"n={n}, parent={'yes' if parent is not None else 'no'}")

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============ Phase 1: GS initialization (15% time) ============
    phase1_end = t_start + 0.15 * (budget_s - 30.0)
    print(f"\n=== Phase 1: GS (until t={phase1_end-t_start:.0f}s) ===")

    inits = build_initializations(n, rng, parent)

    for i, a0 in enumerate(inits[:5]):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.01)
        a0 = normalize_seq(a0)
        a_gs = gs_fienup(a0, n_iters=200, alpha=0.5)
        sc = compute_score(a_gs)
        try_update(sc, a_gs, f"GS{i}")

    # ============ Phase 2: LSE continuation (35% time) ============
    phase2_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n=== Phase 2: LSE Continuation (until t={phase2_end-t_start:.0f}s) ===")

    betas = [50, 200, 500, 2000, 10000, 30000, 80000]
    a_lse = lse_continuation(best_a, betas, maxiter_per_beta=1500)
    sc = compute_score(a_lse)
    try_update(sc, a_lse, "LSE-cont")

    # ============ Phase 3: Alternating QP (25% time) ============
    phase3_end = t_start + 0.75 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Alternating QP (until t={phase3_end-t_start:.0f}s) ===")

    a_qp = alternating_qp(best_a, n_iters=20, beta=200, maxiter_qp=80)
    sc = compute_score(a_qp)
    try_update(sc, a_qp, "Alt-QP")

    # Second round of Alt-QP with different beta
    if time.monotonic() < phase3_end - 60:
        a_qp2 = alternating_qp(best_a, n_iters=10, beta=500, maxiter_qp=60)
        sc = compute_score(a_qp2)
        try_update(sc, a_qp2, "Alt-QP-b500")

    # ============ Phase 4: Nesterov + Peak shave (15% time) ============
    phase4_end = t_start + 0.90 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Nesterov + Peak Shave (until t={phase4_end-t_start:.0f}s) ===")

    for top_k in [10, 30, 50]:
        if time.monotonic() > phase4_end:
            break
        a_ns = nesterov_subgrad(best_a, n_iters=5000, top_k=top_k)
        sc = compute_score(a_ns)
        try_update(sc, a_ns, f"Nesterov-{top_k}")

    if time.monotonic() < phase4_end - 30:
        a_ps = peak_shave(best_a, n_iters=3000, n_peaks=5)
        sc = compute_score(a_ps)
        try_update(sc, a_ps, "PeakShave")

    # ============ Phase 5: Final polish (10% time) ============
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 60:
        # High-beta LSE for final precision
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=2000, maxfun=40000)
        try_update(sc, a_fin, "Final-LSE-100k")

    if time.monotonic() < deadline - 30:
        # One more GS pass
        a_gs_final = gs_fienup(best_a, n_iters=100, alpha=0.2)
        sc = compute_score(a_gs_final)
        try_update(sc, a_gs_final, "Final-GS")

    if time.monotonic() < deadline - 15:
        sc, a_fin2 = run_lbfgs(best_a, 200000, maxiter=1000, maxfun=20000)
        try_update(sc, a_fin2, "Final-LSE-200k")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
