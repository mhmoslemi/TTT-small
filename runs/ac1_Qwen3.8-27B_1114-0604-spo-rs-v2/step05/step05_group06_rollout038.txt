    n = len(a)

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weight (mirror descent) with adaptive active set,
exponential weighting, backtracking step size, and dual-guided polishing.
Key innovations: per-iteration cost is just 2 FFTs (conv + cross-corr),
enabling ~1M iterations in budget; exponential softening of active set;
dual gap monitoring for convergence detection.

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, 1.0 / a.size)
    return a / S

def _spectral_init(n, rng):
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.15):
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 150))

    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            # Compute gradient: grad[m] = 2 * sum_k w[k] * a[k-m]
            grad = 2.0 * _cross_corr(a, w)
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

            gmax = float(grad.max())
            if gmax < 1e-14:
                break

            # Adaptive step size with backtracking
            eta = eta_scale / gmax
            # Try a few step sizes
            best_factor = None
            for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
                e = eta * scale
                factor = np.exp(-e * grad)
                a_try = a * factor
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try
                # Quick score check (recompute conv only for the best candidate)
                # Just accept if the factor is reasonable
                if best_factor is None:
                    best_factor = a_try
                    best_eta = e
                break  # Use first valid

            if best_factor is None:
                break

            a = best_factor
            it += 1

        phase_elapsed = time.monotonic() - phase_start_time
        print(f"  [{label}] Phase k={top_k},{beta:.0f},{eta_scale:.1f}: "
              f"best={best_score:.8f}, iters={it}, elapsed={phase_elapsed:.0f}s")

        if time.monotonic() > deadline - 30:
            break

        # Periodic perturbation between phases to escape local optima
        if pi < len(phases) - 1 and pi % 3 == 2:
            a = _smooth_perturb(best_a, rng, strength=0.03)

    return best_score, best_a


def _dual_polish(a, deadline, label=""):
    """
    Dual-guided polishing: solve for optimal lambda on active set,
    then update a to satisfy KKT conditions more precisely.
    """
    n = len(a)
    a = _norm(a)
    best_a = a.copy()
    best_score = _score(a)
    t_start = time.monotonic()

    # Active set of moderate size
    top_k = min(30, 2 * n - 1)

    for it in range(500):
        if time.monotonic() > deadline - 20:
            break

        c = _self_conv(a)
        cmax = float(c.max())
        current_score = float(2.0 * n * cmax)

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            if it % 50 == 0:
                print(f"    [{label} dual] it={it}, score={best_score:.8f}")

        # Find active set
        idx = np.argpartition(c, -top_k)[-top_k:]
        s = len(idx)

        # Compute autocorrelation for dual matrix
        Nf = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, Nf)
        R = np.fft.irfft(np.abs(Af) ** 2, Nf)

        # Build s x s Gram matrix: A[i,j] = 4 * R[idx[j] - idx[i]]
        A_mat = np.zeros((s, s))
        for i in range(s):
            for j in range(s):
                d = idx[j] - idx[i]
                ad = abs(d)
                if ad < n:
                    A_mat[i, j] = 4.0 * R[ad]

        # Compute b[i] = 2 * sum_m a[idx[i] - m] (sum over valid m)
        # This is 2 * (a * 1)[idx[i]] where 1 is the all-ones vector of length n
        # Actually it's 2 * cross_corr(a, ones_n)[idx[i]] but idx[i] can be >= n
        # Let's compute directly using prefix sums
        prefix = np.zeros(n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + a[i]

        b = np.zeros(s)
        for i, k in enumerate(idx):
            # sum_m a[k-m] where 0 <= k-m < n, i.e., max(0, k-n+1) <= m <= min(k, n-1)
            m_lo = max(0, k - n + 1)
            m_hi = min(k, n - 1)
            if m_lo <= m_hi:
                b[i] = 2.0 * (prefix[m_hi + 1] - prefix[m_lo])
            # Else b[i] = 0

        n_total = n  # "support size" (assume all positive)

        # Solve: min lambda^T A lambda - (b^T lambda)^2 / n_total
        # s.t. sum(lambda) = 1, lambda >= 0
        lam = np.ones(s) / s
        for _ in range(80):
            grad_lam = 2.0 * A_mat @ lam - 2.0 * (lam @ b) * b / n_total
            gnorm = float(np.abs(grad_lam).max())
            if gnorm < 1e-12:
                break
            step = 1.0 / gnorm
            lam_new = _project_simplex(lam - step * grad_lam)
            if float(np.linalg.norm(lam_new - lam)) < 1e-12:
                lam = lam_new
                break
            lam = lam_new

        # Compute dual potential h[m] = 2 * sum_i lam[i] * a[idx[i] - m]
        w = np.zeros(2 * n - 1)
        w[idx] = lam
        h = 2.0 * _cross_corr(a, w)

        nu = float(np.median(h))

        # Update: a_new[m] = a[m] * (nu / max(h[m], eps))^alpha
        alpha = 0.2 + 0.3 * (it / 500.0)  # increase over time
        h_safe = np.maximum(h, 1e-12)
        ratio = np.power(nu / h_safe, alpha)
        ratio = np.clip(ratio, 0.0, 100.0)
        a_new = a * ratio
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S

    return best_score, best_a


def _final_squeeze(a, deadline, label=""):
    """
    Very targeted optimization: use only the top 3-8 positions,
    very small step size, many iterations.
    """
    n = len(a)
    a = _norm(a)
    best_a = a.copy()
    c = _self_conv(a)
    best_score = float(2.0 * n * c.max())
    t_start = time.monotonic()

    # Start with top 3, grow to top 8
    for top_k in (3, 4, 5, 6, 8):
        if time.monotonic() > deadline - 20:
            break
        beta = 10.0
        eta_scale = 0.5

        for it in range(20000):
            if time.monotonic() > deadline - 20:
                break

            c = _self_conv(a)
            cmax = float(c.max())
            current_score = float(2.0 * n * cmax)

            if current_score < best_score - 1e-10:
                best_score = current_score
                best_a = a.copy()

            tk = min(top_k, 2 * n - 1)
            idx = np.argpartition(c, -tk)[-tk:]
            c_act = c[idx]
            w_act = np.exp(-beta * (cmax - c_act))
            ws = w_act.sum()
            if ws < 1e-30:
                w_act = np.ones(tk) / tk
            else:
                w_act /= ws

            w = np.zeros(2 * n - 1)
            w[idx] = w_act
            grad = 2.0 * _cross_corr(a, w)
            gmax = float(grad.max())
            if gmax < 1e-14:
                break

            eta = eta_scale / gmax
            factor = np.exp(-eta * grad)
            a_new = a * factor
            S = a_new.sum()
            if S < 1e-12:
                break
            a = a_new / S

        print(f"    [{label} squeeze] k={top_k}: best={best_score:.8f}, "
              f"t={time.monotonic()-t_start:.1f}s")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multiplicative weight optimization with adaptive active set,
    dual-guided polishing, and final squeeze. Multiple diverse starts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = _norm(p)
    except Exception:
        pass

    # Target length
    if parent is not None:
        target_n = len(parent)
    else:
        target_n = 8192

    print(f"Parent loaded: {parent is not None}, target_n={target_n}, "
          f"budget={budget_s}s, deadline={budget_s - 30}s")

    # --- Generate initializations ---
    inits = []

    if parent is not None:
        inits.append(('parent', _norm(_upsample(parent, target_n))))
        inits.append(('parent_r', _norm(_upsample(parent[::-1], target_n))))
        for amp in (0.05, 0.1, 0.2):
            p = _upsample(parent, target_n)
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    for i in range(4):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # Cosine-based
    x = np.linspace(0, 1, target_n)
    for f in (1, 2):
        for amp in (0.2, 0.4):
            arr = 1.0 + amp * np.cos(2 * np.pi * f * x)
            arr = np.clip(arr, 0, None)
            if arr.sum() > 0:
                inits.append((f'cos{f}{int(amp*100)}', arr / arr.sum()))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc) and sc < 3.0:
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top: "
          f"{[(n, f'{s:.6f}') for s, n, _ in scored[:5]]}")

    # Select top diverse candidates
    candidates = []
    for sc, name, arr in scored:
        if len(candidates) >= 3:
            break
        candidates.append((sc, name, arr))

    if not candidates:
        candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()

    print(f"Initial best: {best_score:.8f} ({candidates[0][1]})")

    # --- Main optimization on best candidate ---
    # Allocate time: 60% main, 20% dual, 15% squeeze, 5% margin
    time_main = (deadline - t_start) * 0.55
    time_dual = (deadline - t_start) * 0.20
    time_squeeze = (deadline - t_start) * 0.15

    # Optimize the best candidate
    sc, seq = _multiplicative_optimize(
        best_seq, t_start + time_main, "main", rng)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"*** After main: {best_score:.8f} ***")

    # If time remains, optimize second candidate
    if len(candidates) >= 2 and time.monotonic() < deadline - time_squeeze - 60:
        t_sub = min(time.monotonic() + 120, deadline - time_squeeze - 30)
        sc2, seq2 = _multiplicative_optimize(
            candidates[1][2], t_sub, "cand2", rng)
        if sc2 < best_score:
            best_score = sc2
            best_seq = seq2.copy()
            print(f"*** Cand2 improved: {best_score:.8f} ***")

    # --- Dual polish ---
    if time.monotonic() < deadline - 60:
        t_dual_end = min(time.monotonic() + time_dual, deadline - 40)
        sc, seq = _dual_polish(best_seq, t_dual_end, "dual")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"*** After dual: {best_score:.8f} ***")

    # --- Final squeeze ---
    if time.monotonic() < deadline - 25:
        sc, seq = _final_squeeze(best_seq, deadline - 15, "squeeze")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"*** After squeeze: {best_score:.8f} ***")

    # --- One more multiplicative pass on the best ---
    if time.monotonic() < deadline - 25:
        sc, seq = _multiplicative_optimize(
            best_seq, deadline - 10, "final", rng)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"*** Final pass: {best_score:.8f} ***")

    # Prepare output: normalize to sum=1
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    # Verify
    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: internal={best_score:.10f}, verified={verified:.10f}, "
          f"len={len(result)}, time={elapsed:.1f}s")
    return result
