Active-set gradient descent with peak equalization, LSE Nesterov, and
simulated annealing for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def gradient_max_peaks(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    w has length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    # w_flipped = w[::-1]
    # conv = a * w_flipped (linear convolution)
    # grad[i] = 2 * conv[2n-2-i]
    w_flipped = w[::-1]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    Wf = np.fft.rfft(w_flipped, M_fft)
    conv = np.fft.irfft(A * Wf, M_fft)[:conv_len]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    return grad

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = c_max + float(np.log(exp_shifted.sum())) / beta
    
    w = p  # weights

    S3 = S2 * S
    score = 2.0 * n * L / S2

def gradient_max_peaks(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    a has length n, w has length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    a_flipped = a[::-1]
    conv_len = 3 * n - 2  # length of linear conv of (2n-1) and n
    M_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    Af = np.fft.rfft(a_flipped, M_fft)
    conv = np.fft.irfft(W * Af, M_fft)[:conv_len]
    # grad[i] = 2 * conv[i + n - 1] for i = 0..n-1
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

"""
Active-set gradient descent with Nesterov momentum, peak equalization,
LSE-smooth optimization, and L-BFGS polish.

Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key strategies:
1. Active-set gradient: only top K peaks contribute to gradient (sharp, targeted)
2. Nesterov acceleration with adaptive step size
3. LSE-smooth Nesterov for broader gradient information
4. Peak shaving: targeted coordinate adjustments on peak contributors
5. L-BFGS polish on LSE objective for final refinement
6. Multiple K sweeps and restarts for exploration
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_gradient_active(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    a: length n, w: length 2n-1 (weights for each convolution bin).
    Returns gradient of length n.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    a_flipped = a[::-1]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    Af = np.fft.rfft(a_flipped, M_fft)
    conv = np.fft.irfft(W * Af, M_fft)[:conv_len]
    # grad[i] = 2 * conv[i + n - 1] for i = 0..n-1
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient.
    LSE_beta(c) = (1/beta)*log(sum(exp(beta*c))) approximates max(c).
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = c_max + float(np.log(exp_shifted.sum())) / beta

    grad_L = compute_gradient_active(a, p)

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def active_set_nesterov(a, max_time, K_init=10, K_end=60,
                        lr_init=0.008, lr_end=0.0005,
                        momentum_init=0.85, momentum_end=0.95,
                        eval_interval=30):
    """
    Active-set Nesterov-accelerated gradient descent.
    Only top K peaks contribute to the gradient.
    Works with sum(a)=1 normalization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = (time.monotonic() - t_start) / max_time
        K = int(K_init + (K_end - K_init) * min(1.0, t_frac * 3))
        lr = lr_init * (lr_end / lr_init) ** t_frac
        mu = momentum_init + (momentum_end - momentum_init) * t_frac

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            a = a + v
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S
            continue
        a_look = a_look / S_look

        # Compute convolution
        c = fast_conv(a_look)

        # Find top K peaks and build weight vector
        top_k_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(2 * n - 1)
        w[top_k_idx] = 1.0 / K

        # Compute gradient of average top-K peaks
        grad_c = compute_gradient_active(a_look, w)

        # Project onto tangent space (sum=0) for the constraint sum(a)=1
        grad = grad_c - grad_c.mean()

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive learning rate: cap step to avoid overshooting
        c_max = float(c.max())
        c_min_topk = float(c[top_k_idx].min())
        peak_spread = c_max - c_min_topk
        max_lr = max(1e-6, peak_spread / (gnorm * n + 1e-10))
        eff_lr = min(lr, max_lr)

        # Nesterov update
        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = a_look
            v = np.zeros(n)
            continue
        a = a / S

        # Evaluate periodically
        if it % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # Final evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def lse_nesterov(a, max_time, beta_start=50.0, beta_end=5000.0,
                 lr_start=0.01, lr_end=0.001,
                 momentum_init=0.90, momentum_end=0.96,
                 eval_interval=20):
    """
    LSE-smoothed Nesterov gradient descent with adaptive beta.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    beta = beta_start
    beta_growth = (beta_end / beta_start) ** (1.0 / max(1, int(max_time * 5)))

    while time.monotonic() - t_start < max_time:
        it += 1
        t_frac = (time.monotonic() - t_start) / max_time
        lr = lr_start * (lr_end / lr_start) ** t_frac
        mu = momentum_init + (momentum_end - momentum_init) * t_frac

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute LSE gradient
        _, grad = lse_score_and_grad(a_look, beta)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step
        eff_lr = min(lr, 0.1 / (gnorm + 1e-10))

        v = mu * v - eff_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Grow beta
        beta = min(beta * beta_growth, beta_end)

        if it % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def peak_shaving(a, max_time, n_top_peaks=5, n_top_contribs=30):
    """
    Targeted peak shaving: identify the top peaks and reduce their
    contributing coordinates via line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    it = 0
    while time.monotonic() - t_start < max_time:
        it += 1
        c = fast_conv(a)
        S_cur = float(a.sum())

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        M = float(c.max())

        for k in peak_indices:
            k = int(k)
            # Find contributing pairs
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution of a[j] to c[k] is a[k-j]
            contribs = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            # Find top contributors
            n_local = min(n_top_contribs, len(contribs))
            top_local = np.argsort(contribs)[::-1][:n_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try reducing this coordinate
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_norm = a / S_tmp
                    c_new = fast_conv(a_norm)
                    sc = 2.0 * n * float(c_new.max()) / (float(a_norm.sum()) ** 2)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] = old_val
                    continue
                # Re-normalize
                S = float(a.sum())
                if S > 1e-12:
                    a = a / S

        # Also try increasing low-contributing coordinates to maintain sum
        if it % 5 == 0:
            # Find coordinates with low contribution to top peaks
            c = fast_conv(a)
            peak_indices2 = np.argsort(c)[::-1][:3]
            low_contrib = np.zeros(n)
            for k in peak_indices2:
                k = int(k)
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                for j in range(j_start, j_end + 1):
                    low_contrib[j] += a[k - j]
            # Try increasing some low-contribution coordinates
            candidates = np.argsort(low_contrib)[:20]
            for j in candidates:
                j = int(j)
                for factor in [1.1, 1.2, 1.3]:
                    a[j] *= factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] /= factor
                        continue
                    a_norm = a / S_tmp
                    c_new = fast_conv(a_norm)
                    sc = 2.0 * n * float(c_new.max()) / (float(a_norm.sum()) ** 2)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_norm.copy()
                        a = a_norm
                        break
                else:
                    a[j] /= factor

    return best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    def objective(x):
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_quick(a, n_iters=100, thresh=1.5, clip=0.6):
    """Quick Fienup-style frequency domain optimization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def simulated_annealing(a, max_time, n_iter_per_step=500, temp_start=0.01, temp_end=0.001):
    """
    Simulated annealing with structured perturbations focused on peak contributors.
    """
    n = len(a)
    rng = np.random.default_rng(12345)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    total_iters = 0

    while time.monotonic() - t_start < max_time:
        batch = min(n_iter_per_step, max(100, int(max_time * 20)))
        for _ in range(batch):
            if time.monotonic() - t_start >= max_time:
                break
            total_iters += 1
            t_frac = min(1.0, (time.monotonic() - t_start) / max_time)
            temp = temp_start * (temp_end / temp_start) ** t_frac

            # Find current peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))

            # Pick a random contributor to the peak
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                j = int(rng.integers(j_start, j_end + 1))
                # Reduce a[j] slightly
                delta = temp * a[j] * (0.5 + rng.random())
                a[j] = max(0, a[j] - delta)
                # Distribute delta to random low-position (far from peak)
                # Pick a coordinate that doesn't contribute to top peaks
                far_region = list(range(0, n // 4)) + list(range(3 * n // 4, n))
                i = int(rng.choice(far_region)) if far_region else int(rng.integers(0, n))
                a[i] += delta

            S = float(a.sum())
            if S < 1e-12:
                continue
            a = a / S

            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            elif rng.random() > np.exp(-(sc - best_sc) / (temp * 0.01 + 1e-10)):
                # Reject: revert (we don't keep a full history, so approximate)
                # Just keep going; the best_a is preserved
                pass

    return best_a


def upsample_seq(a, new_n):
    """Upsample a sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)

        update_best(compute_score(a_lse_p), a_lse_p, "LSE-pert")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Peak shaving + coordinate descent (15% time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 45.0)
    print(f"\n=== Phase 4: Peak shaving (t < {phase4_end - t_start:.0f}s) ===")

    time_left = phase4_end - time.monotonic()
    if time_left > 30:
        a_ps = peak_shaving(best_a, max_time=time_left * 0.7,
                            n_top_peaks=5, n_top_contribs=40)
        update_best(compute_score(a_ps), a_ps, "peak-shave")

    # Short LSE after shaving
    if time.monotonic() < phase4_end - 20:
        time_left2 = phase4_end - time.monotonic()
        a_lse2 = lse_nesterov(best_a, max_time=time_left2 * 0.8,
                              beta_start=200.0, beta_end=5000.0,
                              lr_start=0.004, lr_end=0.0005,
                              momentum_init=0.93, momentum_end=0.96)
        update_best(compute_score(a_lse2), a_lse2, "LSE-post-shave")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS polish (10% time)
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: L-BFGS polish (t < {phase5_end - t_start:.0f}s) ===")

    lbfgs_betas = [500, 1000, 2000, 5000]
    for i, beta in enumerate(lbfgs_betas):
        if time.monotonic() > phase5_end:
            break
        time_left = phase5_end - time.monotonic()
        if time_left < 15:
            break
        # Limit L-BFGS calls by time
        maxfun = min(30000, int(time_left * 100))
        maxiter = min(2000, maxfun // 10)
        print(f"  L-BFGS beta={beta}, maxfun={maxfun}, time_left={time_left:.0f}s")
        try:
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            update_best(sc, a_opt, f"lbfgs-b{beta}")
        except Exception as e:
            print(f"  L-BFGS error: {e}")

    # Final active-set polish
    if time.monotonic() < phase5_end - 10:
        time_left = phase5_end - time.monotonic()
        a_final_as = active_set_nesterov(best_a, max_time=time_left * 0.8,
                                         K_init=5, K_end=15,
                                         lr_init=0.002, lr_end=0.0001,
                                         momentum_init=0.93, momentum_end=0.96,
                                         eval_interval=50)
        update_best(compute_score(a_final_as), a_final_as, "final-AS")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
