def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    """Gradient of c[k] w.r.t. a. Returns vector of length n."""
    # d(c[k])/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad

def pnorm_descent(a, n_iters=3000, p_start=4.0, p_end=1000.0, lr_start=0.02, lr_end=0.002):
    """
    Gradient descent on the smoothed p-norm of the convolution.
    As p increases, this approaches the max more closely.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Precompute FFT size
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        p = p_start * (p_end / p_start) ** t
        lr = lr_start + (lr_end - lr_start) * t
        
        c = fast_conv(a)
        S = float(a.sum())
        S2 = S * S
        S3 = S2 * S
        
        c_safe = np.maximum(c, 1e-30)
        cp = c_safe ** p
        cp_sum = float(cp.sum())
        norm_p = cp_sum ** (1.0 / p)
        
        # Weights for gradient: w[k] = p * cp[k]^(p-1) / cp_sum
        w = p * (c_safe ** (p - 1)) / cp_sum
        
        # Gradient of sum_k w[k]*c[k] w.r.t. a
        # = 2 * (a * w_rev) at appropriate indices
        w_rev = np.zeros(M_fft)
        w_rev[:2 * n - 1] = w[::-1]
        A_pad = np.zeros(M_fft)
        A_pad[:n] = a
        Aft = np.fft.rfft(A_pad)
        Wft = np.fft.rfft(w_rev)
        conv_res = np.fft.irfft(Aft * Wft)[:conv_len]
        grad_c = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
        
        # Full gradient of 2n*norm_p/S^2
        grad = (2.0 * n / S2) * grad_c - (4.0 * n * norm_p / S3) * a
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        a = a - lr * grad / gnorm
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def simultaneous_coord_search(a, n_peaks=10, n_top_per_peak=15, n_factors=7):
    """
    For each of the top peaks, try adjusting the top contributing coordinates
    by various factors. Accept changes that reduce the global max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc, c, S = score_and_peaks(a)
    
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
    
    for _ in range(3):  # 3 sweeps
        c = fast_conv(a)
        S = float(a.sum())
        S2 = S * S
        c_max = float(c.max())
        sc = 2.0 * n * c_max / S2
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            k = int(k)
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            if not valid.any():
                continue
            contrib = np.zeros(n)
            contrib[valid] = a[idx[valid]]
            top_coords = np.argsort(contrib)[::-1][:n_top_per_peak]
            
            for j in top_coords:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for f in factors:
                    a[j] = old_val * f
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < sc:
                        a = a_tmp
                        sc = sc_tmp
                        S = S_tmp
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        improved = True
                        break
                else:
                    a[j] = old_val
        
        if not improved:
            break
    
    return best_a


def fienup_aggressive(a, n_iters=500, thresh_start=3.0, thresh_end=1.05, clip=0.6):
    """Aggressive Fienup: flatten power spectrum more aggressively."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh = thresh_start + (thresh_end - thresh_start) * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        above = P > P_mean * thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        target = np.sqrt(P_mean * thresh)
        current = np.sqrt(P[above])
        scale[above] = np.clip(target / np.maximum(current, 1e-10), 0.0, 1.0) ** clip
        # Also slightly boost below-mean bins
        below = P < P_mean * 0.5
        if below.any():
            scale[below] = 1.0 + (1.0 - clip) * 0.1
        
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lse_descent(a, beta=500.0, n_iters=2000, lr=0.005):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    mu = 0.9
    
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    
    for it in range(n_iters):
        mu = min(0.97, 0.9 + 0.00002 * it)
        lr_i = lr * (1.0 + 0.5 * np.cos(np.pi * it / n_iters))
        
        a_look = np.maximum(a + mu * v, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        c = fast_conv(a_look)
        S_l = float(a_look.sum())
        S_l2 = S_l * S_l
        S_l3 = S_l2 * S_l
        
        c_max = float(c.max())
        shifted = beta * (c - c_max)
        exp_s = np.exp(shifted)
        exp_sum = float(exp_s.sum())
        L = c_max + np.log(exp_sum) / beta
        p = exp_s / exp_sum
        
        # Gradient of LSE
        w_rev = np.zeros(M_fft)
        w_rev[:2 * n - 1] = p[::-1]
        A_pad = np.zeros(M_fft)
        A_pad[:n] = a_look
        Aft = np.fft.rfft(A_pad)
        Wft = np.fft.rfft(w_rev)
        conv_res = np.fft.irfft(Aft * Wft)[:conv_len]
        grad_L = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
        
        grad = (2.0 * n / S_l2) * grad_L - (4.0 * n * L / S_l3) * a_look
        
        v = mu * v - lr_i * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Peak-steering optimization: directly targets the maximum convolution value
    using multi-peak subgradient descent with peak-equalization step sizes,
    p-norm smoothing, simultaneous coordinate search, and LSE polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    effective_budget = budget_s - 30.0
    
    # Get parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    n_target = len(parent) if parent is not None else 6962
    
    # Initialize from parent
    if parent is not None:
        a0 = parent.copy()
        a0 = np.maximum(a0, 0.0)
        sc0 = compute_score(a0)
        update(sc0, a0, "parent")
    else:
        x = np.linspace(0, 1, n_target)
        a0 = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        sc0 = compute_score(a0)
        update(sc0, a0, "init")
    
    print(f"Initial score: {best_score:.8f}, n={n_target}")
    
    # ============================================================
    # Phase 1: Peak steering descent (35% of time)
    # ============================================================
    t1_end = t_start + 0.35 * effective_budget
    print(f"\n=== Phase 1: Peak steering descent (budget {t1_end - t_start:.0f}s) ===")
    
    window_sizes = [0.05, 0.10, 0.03, 0.08, 0.15, 0.02]
    wi = 0
    while time.monotonic() < t1_end:
        remaining = t1_end - time.monotonic()
        n_iter = min(3000, max(200, int(remaining * 500)))
        a_out = peak_steering_descent(best_a, n_iters=n_iter,
                                       peak_window=window_sizes[wi % len(window_sizes)])
        sc_out = compute_score(a_out)
        if not update(sc_out, a_out, f"peak-steer-w{window_sizes[wi % len(window_sizes)]}"):
            # Try perturbed start
            pert = best_a * (1.0 + 0.01 * rng.standard_normal(len(best_a)))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            a_out2 = peak_steering_descent(pert, n_iters=n_iter // 2,
                                            peak_window=window_sizes[(wi + 1) % len(window_sizes)])
            sc_out2 = compute_score(a_out2)
            if not update(sc_out2, a_out2, f"peak-steer-pert{wi}"):
                break
        wi += 1
    
    print(f"After Phase 1: {best_score:.8f}")
    
    # ============================================================
    # Phase 2: P-norm descent (25% of time)
    # ============================================================
    t2_end = t_start + 0.60 * effective_budget
    print(f"\n=== Phase 2: P-norm descent (budget {t2_end - t_start:.0f}s) ===")
    
    pnorm_configs = [
        (4.0, 500.0, 0.02, 0.002),
        (8.0, 1000.0, 0.015, 0.001),
        (2.0, 200.0, 0.03, 0.003),
        (16.0, 2000.0, 0.01, 0.001),
    ]
    pi = 0
    while time.monotonic() < t2_end:
        remaining = t2_end - time.monotonic()
        if remaining < 10:
            break
        cfg = pnorm_configs[pi % len(pnorm_configs)]
        n_iter = min(3000, max(500, int(remaining * 400)))
        a_out = pnorm_descent(best_a, n_iters=n_iter, p_start=cfg[0],
                              p_end=cfg[1], lr_start=cfg[2], lr_end=cfg[3])
        sc_out = compute_score(a_out)
        if not update(sc_out, a_out, f"pnorm-{pi}"):
            # Try from perturbed
            pert = best_a + 0.005 * rng.standard_normal(len(best_a))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            a_out2 = pnorm_descent(pert, n_iters=n_iter // 2, p_start=cfg[0],
                                   p_end=cfg[1], lr_start=cfg[2], lr_end=cfg[3])
            sc_out2 = compute_score(a_out2)
            if not update(sc_out2, a_out2, f"pnorm-pert{pi}"):
                break
        pi += 1
    
    print(f"After Phase 2: {best_score:.8f}")
    
    # ============================================================
    # Phase 3: Simultaneous coordinate search (20% of time)
    # ============================================================
    t3_end = t_start + 0.80 * effective_budget
    print(f"\n=== Phase 3: Coordinate search (budget {t3_end - t_start:.0f}s) ===")
    
    cs_configs = [(10, 15, 7), (5, 20, 7), (20, 10, 5), (8, 25, 9)]
    ci = 0
    while time.monotonic() < t3_end:
        cfg = cs_configs[ci % len(cs_configs)]
        a_out = simultaneous_coord_search(best_a, n_peaks=cfg[0],
                                           n_top_per_peak=cfg[1], n_factors=cfg[2])
        sc_out = compute_score(a_out)
        if not update(sc_out, a_out, f"coord-{ci}"):
            break
        ci += 1
    
    print(f"After Phase 3: {best_score:.8f}")
    
    # ============================================================
    # Phase 4: Fienup + LSE polish (15% of time)
    # ============================================================
    t4_end = deadline
    print(f"\n=== Phase 4: Fienup + LSE polish (budget {t4_end - t_start:.0f}s) ===")
    
    # Fienup
    if time.monotonic() < t4_end - 30:
        a_f = fienup_aggressive(best_a, n_iters=500, thresh_start=2.5,
                                 thresh_end=1.05, clip=0.7)
        sc_f = compute_score(a_f)
        update(sc_f, a_f, "fienup")
    
    # LSE with various betas
    for beta in [200, 500, 1000, 2000, 5000]:
        if time.monotonic() > t4_end - 10:
            break
        remaining = t4_end - time.monotonic()
        if remaining < 10:
            break
        n_iter = min(3000, max(200, int(remaining * 300)))
        a_lse = lse_descent(best_a, beta=beta, n_iters=n_iter, lr=0.004)
        sc_lse = compute_score(a_lse)
        update(sc_lse, a_lse, f"lse-{beta}")
    
    # Final coordinate search
    if time.monotonic() < deadline - 5:
        a_final = simultaneous_coord_search(best_a, n_peaks=5,
                                             n_top_per_peak=20, n_factors=9)
        sc_final = compute_score(a_final)
        update(sc_final, a_final, "final-coord")
    
    # One more peak steering
    if time.monotonic() < deadline - 3:
        a_ps = peak_steering_descent(best_a, n_iters=500, peak_window=0.01)
        sc_ps = compute_score(a_ps)
        update(sc_ps, a_ps, "final-peak-steer")
    
    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
