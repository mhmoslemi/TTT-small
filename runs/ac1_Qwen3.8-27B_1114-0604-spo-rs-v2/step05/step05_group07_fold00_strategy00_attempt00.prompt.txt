import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS callback
_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # dL/da[i] = 2 * (p * rev_a)[n-1+i]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta

        a0[:] = 0.5

        lse_callback, a0,

        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=10000, eta0=0.05, decay=0.9995):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        # Center gradient for normalization
        g_centered = grad - np.mean(grad)

        # Exponential weights update
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        eta *= decay

    return best_a


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=3000, f_lb=1.0, top_k=5):
    """Subgradient using top-k peaks for a smoother descent direction."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.1

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        # Use top-k peaks weighted
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(c))
        for rank, idx in enumerate(top_idx):
            weights[idx] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)

        Mak_weighted = float(np.sum(weights * c))
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Weighted subgradient
        sg = np.zeros(n)
        S2 = S_a * S_a
        S3 = S2 * S_a
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]

        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (0.999 ** it)
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        # Multiplicative perturbation
        a_p = a * np.exp(scale * 2.0 * rng.standard_normal(n))
    else:
        # Random subset rescale
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def targeted_peak_flatting(a, n_sweeps=10, n_top=50):
    """Targeted coordinate descent: reduce peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        total_len = len(contrib)
        top_count = min(n_top, max(1, total_len // 3))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 0.9, 0.95]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.1, np.cos(np.pi * x)))
    inits.append(0.3 + 0.7 * np.cos(np.pi * np.clip((x - 0.3) / 0.4, -1, 1)) ** 2)
    # Added: raised cosine (Hann window)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Added: Gaussian
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2))
    return inits
