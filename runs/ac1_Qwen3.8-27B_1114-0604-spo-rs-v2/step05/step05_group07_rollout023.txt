"""
LP-based descent with L-BFGS polishing for minimizing peak self-convolution ratio.
Key innovation: Uses linear programming to find globally-informed descent directions,
combined with high-precision L-BFGS refinement and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

_current_beta = 1000.0

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A such that (A @ g)[k] = conv(f, g)[k]
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # Using sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            v = f[k - j]
            if v > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return np.clip(res.x, 0, None)
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def lp_based_descent(p, n_iters=50, tol=1e-8):
    """Iterative LP-based descent on probability vector p."""
    n = len(p)
    p = p / p.sum()
    best_p = p.copy()
    best_score = compute_score(p)
    
    for it in range(n_iters):
        c = fast_conv(p)
        M = float(c.max())
        
        # Solve LP
        g = solve_lp_direction(p, M)
        Sg = g.sum()
        if Sg < 1e-12:
            break
        g = g / Sg  # Normalize to probability
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            p_new = (1 - t) * p + t * g
            sc = compute_score(p_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.001, 0.005]:
                p_new = (1 - t) * p + t * g
                sc = compute_score(p_new)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        
        if best_t == 0.0:
            break
        
        p = (1 - best_t) * p + best_t * g
        p = p / p.sum()
        
        sc = compute_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
        
        print(f"    LP iter {it}: score={sc:.10f}, t={best_t:.4f}, M={M:.6f}")
    
    return best_p

"""
LP-based descent with L-BFGS polishing and spectral initialization for minimizing
peak self-convolution ratio. Key innovations: (1) LP-based globally-informed descent
directions, (2) spectral initialization via flat power spectrum design, (3) adaptive
coordinate descent polishing, (4) multi-scale iterated local search.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

_current_beta = 1000.0

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_callback(x):
    return lse_score_and_grad(x, _current_beta)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def solve_lp_direction(f, M, time_limit=8):
    n = len(f)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            v = f[k - j]
            if v > 1e-14:
                rows.append(k)
                cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = f[rows.astype(np.int64) - cols.astype(np.int64)]
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit})
    if res.success:
        return np.clip(res.x, 0, None)
    return None

def greedy_direction(f, M, n_steps=2000):
    n = len(f)
    g = np.zeros(n)
    c_fg = np.zeros(2 * n - 1)
    for step in range(n_steps):
        best_j = -1
        best_inc = 0.0
        for j in range(n):
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)
            k_range = np.arange(k_min, k_max + 1)
            f_vals = f[k_range - j]
            current = c_fg[k_range]
            slack = M - current
            valid = f_vals > 1e-14
            if not valid.any():
                inc = 1.0
            else:
                ratios = np.where(valid, slack / np.where(valid, f_vals, 1.0), np.inf)
                inc = ratios.min()
            if inc > 1e-10 and (best_j == -1 or inc * f[j] > best_inc):
                best_inc = inc * f[j]
                best_j = j
                best_inc_val = inc
        if best_j < 0 or best_inc_val < 1e-12:
            break
        g[best_j] += best_inc_val
        k_min = best_j
        k_max = min(best_j + n - 1, 2 * n - 2)
        c_fg[k_min:k_max+1] += f[k_min - best_j:k_max - best_j + 1] * best_inc_val
    Sg = g.sum()
    if Sg > 1e-12:
        return g / Sg
    return None

        if g is None or g.sum() < 1e-12:

def spectral_init(n, rng, n_attempts=20):
    best_p = None
    best_sc = np.inf
    L = 2 * n - 1
    L_fft = 1 << (L - 1).bit_length()

        A = np.exp(1j * phase)

        a = np.clip(a_pad[:n], 0, None)
        S = a.sum()
        if S < 1e-6:
            continue
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_p = a.copy()
    if best_p is None:
        best_p = np.ones(n) / n
    return best_p, best_sc

def perturbed_restart(a, rng, scale=0.03):

    elif strategy == 2:

        for b in range(min(n_blocks, n // block_size)):

            e = min(s + block_size, n)

    elif strategy == 3:

    elif strategy == 4:

        if time.monotonic() < phase3_deadline - 60 and beta >= 1000:
            a_pert = perturbed_restart(best_a, rng, scale=0.01)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            report(sc2, a_opt2, f"LBFGS b={beta} pert")

    print(f"After Phase 3: best = {best_score:.10f}")

    # Phase 4: Iterated local search (13% time)
    phase4_deadline = t_start + 0.95 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 12
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining_s = phase4_deadline - time.monotonic()
        if remaining_s < 40:
            break

        scale = 0.015 + 0.03 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Quick LP step
        remaining_sub = remaining_s - 30
        if remaining_sub > 20:
            a_lp_sub, sc_sub = lp_based_descent(a_pert, n_iters=int(min(8, remaining_sub/8)), time_limit=3)
            report(sc_sub, a_lp_sub, f"ILS{trial} LP")
            a_pert = a_lp_sub

        # L-BFGS
        beta = 500 + int(2000 * rng.random())
        maxiter = int(min(20000, max(2000, remaining_s * 10)))
        maxfun = int(min(80000, max(5000, remaining_s * 40)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_pert, 1e-10), beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS{trial} b={beta}")

    print(f"After Phase 4: best = {best_score:.10f}")

    # Phase 5: Final polishing (5% time)
    print(f"\n--- Phase 5: Final Polishing ---")

    # Coordinate descent
    a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=4, n_top=30)
    report(sc_cd, a_cd, "coord_desc")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 15:
        for beta in [200000, 500000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=20000, maxfun=80000)
            report(sc, a_opt, f"final b={beta}")

    # One more coordinate descent pass
    if time.monotonic() < deadline - 10:
        a_cd2, sc_cd2 = coordinate_descent(best_a, n_sweeps=3, n_top=20)
        report(sc_cd2, a_cd2, "coord_final")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
