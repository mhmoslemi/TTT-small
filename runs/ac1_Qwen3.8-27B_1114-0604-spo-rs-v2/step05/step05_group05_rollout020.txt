Improved optimizer combining FISTA-LSE, subgradient, spectral shaping,
and peak-targeted coordinate descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Precompute FFT sizes
def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = next_power_of_2(M)
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())

    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

        if 0 <= kj < n:
            grad[j] += 4.0 * n * a[kj] / S3

    a_new = np.maximum(a_new, 0.0)

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = next_power_of_2(M)
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient via FFT.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_sum

    # dL/da_j = 2 * sum_m p[j+m] * a[m] (cross-correlation of p and a)
    # Computed via padded FFT
    conv_len = 3 * n - 2
    M_fft = next_power_of_2(conv_len)
    p_pad = np.zeros(M_fft)
    p_pad[:2 * n - 1] = p
    a_rev = np.zeros(M_fft)
    a_rev[:n] = a[::-1]
    Aft = np.fft.rfft(p_pad)
    Art = np.fft.rfft(a_rev)
    cross = np.fft.irfft(Aft * Art)[:conv_len]
    # cross[n-1+j] = sum_m p[j+m] * a[m]
    grad_L = 2.0 * cross[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fista_lse(a, beta_start, beta_end, max_time, lr_init=0.005):
    """
    FISTA (Nesterov accelerated projected gradient) on LSE-smoothed objective.
    Normalizes to sum=1 throughout.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    t_momentum = 1.0
    z = a.copy()
    lr = lr_init
    beta = beta_start
    beta_growth = 1.004
    stalled = 0

    t_start = time.monotonic()

    while time.monotonic() - t_start < max_time:
        beta = min(beta * beta_growth, beta_end)

        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum)) / 2.0
        z_new = a + (t_momentum - 1.0) / t_new * (a - z)
        z_new = np.maximum(z_new, 0.0)
        Sz = float(z_new.sum())
        if Sz < 1e-12:
            break
        z_new = z_new / Sz

        # Gradient at extrapolated point
        _, grad = lse_score_and_grad(z_new, beta)

        # Step and project
        a_new = z_new - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            lr *= 0.5
            stalled += 1
            if stalled > 30:
                break
            continue
        a_new = a_new / S_new

        # Track best (check every 20 iters for efficiency)
        check_score = (int((time.monotonic() - t_start) * 100) % 20 == 0)
        if check_score:
            sc = compute_score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                stalled = 0
            else:
                stalled += 1

        z = a.copy()
        a = a_new
        t_momentum = t_new

        # Adaptive LR
        if stalled > 15:
            lr *= 0.7
            stalled = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def subgradient_momentum(a, max_time, lr_init=0.001, mu=0.85):
    """
    Subgradient descent with momentum on the exact max objective.
    Vectorized: the subgradient at peak k* is essentially 2*a reflected/shifted.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    lr = lr_init

    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        L = float(c[k_star])
        S = float(a.sum())
        S3 = S ** 3

        # Vectorized subgradient:
        # dF/da_j = 4*n*(a[k*-j]*S - L) / S^3  for j where 0<=k*-j<n
        # dF/da_j = -4*n*L / S^3  otherwise
        grad = np.full(n, -4.0 * n * L / S3)
        # The valid range for j: k*-j in [0, n-1] => j in [k*-n+1, k*]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi >= j_lo:
            # a[k*-j] for j in [j_lo, j_hi]: this is a[0..n-1] reversed and shifted
            idx = k_star - np.arange(j_lo, j_hi + 1)  # indices into a
            grad[j_lo:j_hi + 1] += 4.0 * n * a[idx] / S3

        # Momentum update
        v = mu * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            v *= 0.5
            lr *= 0.5
            continue
        a_new = a_new / S_new
        a = a_new
        lr *= 0.9998

        if it % 200 == 0 or it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def spectral_shaping(a, max_time, init_agg=0.25):
    """
    Iteratively compress the dynamic range of the power spectrum.
    Transforms to frequency, reduces peak-to-mean ratio, transforms back.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = next_power_of_2(2 * n - 1)

    t_start = time.monotonic()
    agg = init_agg
    it = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2 + 1e-30
        P_mean = float(P.mean())

        # Compress toward mean
        P_target = P_mean + (P - P_mean) * (1.0 - agg)
        P_target = np.maximum(P_target, 1e-30)

        A_new = A * np.sqrt(P_target / P)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            agg *= 0.9
            if agg < 0.01:
                break

    return best_a


def fienup_power_law(a, n_iters=300, thresh=1.4, power=0.4):
    """Fienup with power-law spectral clipping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = next_power_of_2(2 * n - 1)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2 + 1e-30
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (power * 0.5)

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a


def peak_coordinate_descent(a, max_time, n_top_peaks=25, n_top_contrib=40):
    """
    Aggressive coordinate descent targeting peak contributors.
    For each top peak, reduces the top contributing elements and checks improvement.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    t_start = time.monotonic()
    sweep = 0

    while time.monotonic() - t_start < max_time:
        sweep += 1
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]

        improved_this_sweep = False

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Compute contributions: contrib[j] = a[j]*a[k-j]
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k - js]

            n_local = min(n_top_contrib, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = js[top_local]

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try reducing
                for factor in [0.75, 0.85, 0.92]:
                    a[j] = old_val * factor
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        break
                    a_norm = a / S_tmp
                    sc = compute_score(a_norm)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        a = a_norm
                        improved_this_sweep = True
                        break
                    a[j] = old_val

                if time.monotonic() - t_start > max_time:
                    break

            if time.monotonic() - t_start > max_time:
                break

        # Also try increasing non-contributing elements (to spread energy)
        if sweep <= 3:
            # Pick random positions unlikely to be in peaks
            c = fast_conv(a)
            peak_set = set(np.argsort(c)[::-1][:10].tolist())
            candidate_positions = [j for j in range(n) if j not in peak_set and (n - 1 - j) not in peak_set]
            if candidate_positions:
                rng_local = np.random.default_rng(sweep)
                for _ in range(min(20, len(candidate_positions))):
                    j = int(rng_local.choice(candidate_positions))
                    old_val = a[j]
                    if old_val < 1e-8:
                        continue
                    for factor in [1.3, 1.5]:
                        a[j] = old_val * factor
                        S_tmp = float(a.sum())
                        if S_tmp < 1e-12:
                            a[j] = old_val
                            break
                        a_norm = a / S_tmp
                        sc = compute_score(a_norm)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                            a = a_norm
                            improved_this_sweep = True
                            break
                        a[j] = old_val

        if not improved_this_sweep and sweep > 10:
            break

    return best_a


def targeted_subgradient_refine(a, max_time, lr_init=0.0003):
    """
    Fine-grained subgradient steps with very small learning rate.
    Focuses on the single dominant peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    t_start = time.monotonic()
    it = 0
    last_peak = -1
    same_peak_count = 0

    while time.monotonic() - t_start < max_time:
        it += 1
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        if k_star == last_peak:
            same_peak_count += 1
        else:
            same_peak_count = 0
            last_peak = k_star

        # If stuck on same peak for too long, increase lr
        if same_peak_count > 500:
            lr *= 1.1

        L = float(c[k_star])
        S = float(a.sum())
        S3 = S ** 3

        # Subgradient
        grad = np.full(n, -4.0 * n * L / S3)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi >= j_lo:
            idx = k_star - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += 4.0 * n * a[idx] / S3

        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            lr *= 0.5
            continue
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            else:
                lr *= 0.95

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def build_raised_cosine(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def build_gaussian(n):
    """Gaussian initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_chirp_abs(n):
    """Absolute chirp (non-negative)."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.abs(np.sin(2 * np.pi * x * x * 3.0)) + 0.1
    return a / a.sum()


def build_mixed_fourier(n):
    """Mixed Fourier initialization."""
    x = np.linspace(0, 1, n)
    a = 0.4 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5) + 0.1 * np.cos(6 * np.pi * x + 1.0)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: FISTA-LSE, spectral shaping, subgradient, coordinate descent.
    Returns best sequence found within time budget.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    total = deadline - t_start

    # Load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine target length
    n_target = 6962
    if parent is not None and len(parent) > 200:
        n_target = len(parent)

    # Initialize
    a0 = None
    if parent is not None:
        a0 = build_init_from_parent(parent, n_target)
    else:
        a0 = build_raised_cosine(n_target)

    best_a = a0.copy()
    best_sc = compute_score(a0)
    print(f"Initial: score={best_sc:.8f}, n={n_target}")

    def try_update(a_new, label):
        nonlocal best_a, best_sc
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"  [{label}] {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        return best_a

    # ============================================================
    # Phase 1: FISTA-LSE multi-config (35% of time)
    # ============================================================
    p1_end = t_start + 0.35 * total
    print(f"\n=== Phase 1: FISTA-LSE (to t={p1_end-t_start:.0f}s) ===")

    configs = [
        (20.0, 3000.0, 0.006),
        (50.0, 8000.0, 0.003),
        (10.0, 1500.0, 0.009),
    ]
    for i, (bs, be, lr0) in enumerate(configs):
        if time.monotonic() >= p1_end:
            break
        rem = min((p1_end - time.monotonic()) / (len(configs) - i), p1_end - time.monotonic())
        if rem < 15:
            break
        print(f"  FISTA cfg{i}: beta=[{bs},{be}], lr={lr0}, time={rem:.0f}s")
        res = fista_lse(best_a, bs, be, max_time=rem, lr_init=lr0)
        try_update(res, f"fista-{i}")

    print(f"  Phase 1 done: {best_sc:.8f}")

    # ============================================================
    # Phase 2: Spectral shaping + Fienup (15% of time)
    # ============================================================
    p2_end = t_start + 0.50 * total
    if time.monotonic() < p2_end:
        print(f"\n=== Phase 2: Spectral/Fienup (to t={p2_end-t_start:.0f}s) ===")
        rem = p2_end - time.monotonic()

        if rem > 20:
            res = spectral_shaping(best_a, max_time=rem * 0.4, init_agg=0.2)
            try_update(res, "spectral-0.2")

        if time.monotonic() < p2_end - 10:
            rem2 = p2_end - time.monotonic() - 10
            if rem2 > 15:
                res = spectral_shaping(best_a, max_time=rem2, init_agg=0.12)
                try_update(res, "spectral-0.12")

        if time.monotonic() < p2_end - 10:
            res = fienup_power_law(best_a, n_iters=200, thresh=1.3, power=0.35)
            try_update(res, "fienup-pl")

        print(f"  Phase 2 done: {best_sc:.8f}")

    # ============================================================
    # Phase 3: Subgradient refinement (15% of time)
    # ============================================================
    p3_end = t_start + 0.65 * total
    if time.monotonic() < p3_end:
        print(f"\n=== Phase 3: Subgradient (to t={p3_end-t_start:.0f}s) ===")
        rem = p3_end - time.monotonic()

        if rem > 15:
            res = targeted_subgradient_refine(best_a, max_time=rem * 0.6, lr_init=0.0002)
            try_update(res, "subgrad-refine")

        if time.monotonic() < p3_end - 10:
            rem2 = p3_end - time.monotonic() - 10
            if rem2 > 10:
                res = subgradient_momentum(best_a, max_time=rem2, lr_init=0.0003, mu=0.8)
                try_update(res, "subgrad-mom")

        print(f"  Phase 3 done: {best_sc:.8f}")

    # ============================================================
    # Phase 4: Peak coordinate descent (20% of time)
    # ============================================================
    p4_end = t_start + 0.85 * total
    if time.monotonic() < p4_end:
        print(f"\n=== Phase 4: Peak CD (to t={p4_end-t_start:.0f}s) ===")
        rem = p4_end - time.monotonic()

        if rem > 20:
            res = peak_coordinate_descent(best_a, max_time=rem * 0.7, n_top_peaks=30, n_top_contrib=50)
            try_update(res, "peak-cd-1")

        if time.monotonic() < p4_end - 10:
            rem2 = p4_end - time.monotonic() - 10
            if rem2 > 10:
                res = peak_coordinate_descent(best_a, max_time=rem2, n_top_peaks=15, n_top_contrib=30)
                try_update(res, "peak-cd-2")

        print(f"  Phase 4 done: {best_sc:.8f}")

    # ============================================================
    # Phase 5: Final FISTA polish with high beta (10% of time)
    # ============================================================
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final polish ===")
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            res = fista_lse(best_a, 1000.0, 20000.0, max_time=rem, lr_init=0.001)
            try_update(res, "final-fista")

        # One more coordinate descent pass
        if time.monotonic() < deadline - 10:
            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10:
                res = peak_coordinate_descent(best_a, max_time=rem2, n_top_peaks=10, n_top_contrib=20)
                try_update(res, "final-cd")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_sc:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out


def build_init_from_parent(parent, n_target):
    """Create initial from parent, resampled if needed."""
    parent = np.clip(parent, 0.0, 1000.0)
    if len(parent) == n_target:
        S = parent.sum()
        if S < 1e-12:
            return build_raised_cosine(n_target)
        return parent / S
    return resample(parent, n_target) / resample(parent, n_target).sum()
