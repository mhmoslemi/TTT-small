def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Compute full autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def corr_at_lag(a, lag):
    """
    Compute the gradient of c(lag) with respect to a.
    c(lag) = sum_i a[i] * a[i+lag]
    dc(lag)/da[i] = a[i+lag] + a[i-lag] (with 0 for out of bounds)
    Returns vector of length n.
    """
    n = a.size
    g = np.zeros(n)
    if lag == 0:
        g[:] = 2.0 * a
    elif lag > 0:
        # a[i+lag] for i in [0, n-1-lag]
        g[:n-lag] += a[lag:]
        # a[i-lag] for i in [lag, n-1]
        g[lag:] += a[:n-lag]
    else:
        lag = -lag
        g[:n-lag] += a[lag:]
        g[lag:] += a[:n-lag]
    return g

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size

    x_old = np.linspace(0.0, 1.0, m)

        active = active[top_idx]

    # Average weight on active peaks
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    # R[i] = sum_k w_active[k] * a[k-i]  (a extended by 0)
    # Using the same convolution trick: R[i] = conv(a_rev, w_active)[i+n-1]
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """
    Nesterov accelerated subgradient method for minimizing 2*n*max(c)/S^2.
    a is maintained on the simplex (non-negative, sum=1).
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    # Nesterov parameters
    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    # Initial step size estimate
    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a

    # Scale step: we want the step to move ~1% of the range
    eta = 0.05 / g_norm

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * v

        # Evaluate subgradient at y
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break

        # Gradient step
        step = eta / (1.0 + 0.1 * it)
        x_new = y_norm - step * grad_y

        # Project to simplex (approximate: clip and normalize)
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        # Update momentum
        v = x_new - a
        a = x_new
        t = t_new

        # Track best
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive step size: if no improvement in last 50 iters, reduce eta
        if it > 0 and it % 50 == 0:
            if not (sc < best_score + 1e-10):
                eta *= 0.7

    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    if maxfun is None:
        maxfun = 500

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        # Chain rule for normalization: x = xn * S, so we optimize over unnormalized
        # Actually simpler: just optimize normalized directly with bounds
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _nesterov_lbfgs_hybrid(a0, deadline, label=""):
    """
    Hybrid optimization: alternating LSE-LBFGS and Nesterov subgradient.
    Progressive increase in beta for LSE.
    """
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total_time = max(10.0, float(deadline - t0 - 10.0))

    print("[{}] start: score={:.8f} n={}".format(label, best_score, a.size))

    # Stage 1: LSE with moderate beta
    betas_stage1 = [200, 1000, 5000]
    # Stage 2: LSE with high beta
    betas_stage2 = [20000, 80000, 200000, 500000]
    # Stage 3: Nesterov exact subgradient
    # Stage 4: Very high beta LSE + Nesterov

    # Time allocation: 30% stage1, 30% stage2, 25% stage3, 15% stage4
    stages = [
        (betas_stage1, 0.30, "lse"),
        (betas_stage2, 0.30, "lse"),
        (None, 0.25, "nesterov"),
        ([1000000, 2000000], 0.15, "mixed"),
    ]

    cum_frac = 0.0
    for betas, frac, method in stages:
        cum_frac += frac
        sub_dead = t0 + cum_frac * total_time
        sub_dead = min(sub_dead, float(deadline - 10.0))

        if time.monotonic() > sub_dead - 5.0:
            continue

        if method == "lse":
            for beta in betas:
                rem = sub_dead - time.monotonic()
                if rem < 10.0:
                    break
                # Estimate maxfun
                n = len(a)
                est_time_per = 0.01 * np.log2(max(16, n)) / np.log2(8192)
                maxfun = int(max(50, min(2000, (rem * 0.8) / max(est_time_per, 0.001))))
                sc, a2 = _lbfgs_lse(a, beta, sub_dead, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
                a = _normalize(a2)

        elif method == "nesterov":
            # Multiple Nesterov passes with different k_active
            n_iters_est = int(max(200, (sub_dead - time.monotonic() - 5.0) * 30))
            for k_act in [1, 2, 3, 5]:
                rem = sub_dead - time.monotonic()
                if rem < 10.0:
                    break
                n_it = max(200, int(rem * 20))
                sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=sub_dead,
                                            k_active=k_act, tol=1e-7)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] Nesterov k={} tol=1e-7: {:.8f}".format(label, k_act, best_score))
                a = _normalize(a2)

        elif method == "mixed":
            # Alternate between very high beta LSE and Nesterov
            for beta in betas:
                rem = sub_dead - time.monotonic()
                if rem < 10.0:
                    break
                maxfun = int(max(50, min(1000, (rem * 0.6) / 0.01)))
                sc, a2 = _lbfgs_lse(a, beta, sub_dead - 5.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] Mixed LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
                a = _normalize(a2)

                rem = sub_dead - time.monotonic()
                if rem < 10.0:
                    break
                sc, a2 = _nesterov_optimize(a, n_iters=int(max(100, rem * 15)),
                                            deadline=sub_dead, k_active=3, tol=5e-7)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] Mixed Nesterov: {:.8f}".format(label, best_score))
                a = _normalize(a2)

        # Quick Nesterov polish after each stage
        rem = sub_dead - time.monotonic()
        if rem > 10.0:
            sc, a2 = _nesterov_optimize(a, n_iters=int(max(100, rem * 20)),
                                        deadline=sub_dead, k_active=2, tol=1e-8)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] polish after stage: {:.8f}".format(label, best_score))
            a = _normalize(a2)

    return best_score, best_a


def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        # Multiplicative log-normal noise
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.05, 0.25))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Subgradient-based push
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.3 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth noise (low frequency)
        K = min(31, max(5, n // 200))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.5)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 4:
        # V-shape modulation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = a * vshape
    else:
        # Coordinate swap perturbation
        q = a.copy()
        n_swaps = int(rng.integers(3, max(4, n // 100)))
        for _ in range(n_swaps):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = a * np.exp(0.05 * rng.standard_normal(n))

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize(a, deadline):
    """Quick 3-phase optimization for a candidate."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    # Phase 1: Moderate LSE
    rem = deadline - time.monotonic()
    if rem > 15.0:
        for beta in [500, 5000, 50000]:
            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            maxfun = int(max(30, min(500, rem * 10)))
            sc, a2 = _lbfgs_lse(a, beta, deadline, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
            a = _normalize(a2)

    # Phase 2: Nesterov
    rem = deadline - time.monotonic()
    if rem > 10.0:
        n_it = int(max(100, rem * 15))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                    k_active=3, tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    return best_score, best_a


def _generate_structured_inits(n, rng, parent, cap=25):
    """Generate diverse initial sequences."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Uniform
    _add(np.ones(n))
    # V-shapes
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    # Inverted V
    for alpha in [0.3, 0.6, 1.0]:
        _add(1.0 + alpha * (1.0 - np.abs(2.0 * x - 1.0) ** 2))
    # Cosine
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 1.0]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
            _add(1.0 + amp * np.cos(np.pi * f * x))
    # Exponential
    for rate in [1.0, 2.0, 4.0, 8.0]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1.0 - x)) + 0.01)
        _add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    # Bimodal
    for w in [0.1, 0.15, 0.25, 0.4]:
        for ctr in [0.2, 0.3, 0.35, 0.4]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)
    # Sine absolute
    for f in [1, 2, 3, 4]:
        _add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns list of non-negative floats."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = min(parent.size, 16384)
        if parent_len < 2048:
            parent_len = 4096
        lengths = [parent_len]
        for L in [4096, 8192, 10240, 12288, 16384]:
            if L != parent_len:
                lengths.append(L)
    else:
        lengths = [8192, 10240, 6144, 4096]

    lengths = [L for L in lengths if 2048 <= L <= 16384][:5]
    print("Lengths: {}".format(lengths))

    # Phase 1: Quick init evaluation
    all_scores = []
    all_seqs = []

    for n in lengths:
        inits = _generate_structured_inits(n, rng, parent, cap=20)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)

    order = np.argsort(all_scores)
    print("Best init: {:.8f} (len={})".format(all_scores[order[0]], all_seqs[order[0]].size))

    # Select top starts from different lengths
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 3:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main optimization on selected starts
    # Sort starts by score
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))

    if len(starts_sorted) == 1:
        fracs = [1.0]
    elif len(starts_sorted) == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.55, 0.25, 0.20]

    main_end = deadline - 400.0
    main_budget = max(120.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 20.0:
            continue

        sq = starts[si].copy()

        # For secondary starts, blend with current best (if different length, resample)
        if rank > 0:
            if len(best_seq) != len(sq):
                best_resampled = _resample(best_seq, len(sq))
            else:
                best_resampled = best_seq.copy()
            t_blend = 0.4 + 0.2 * rank
            sq = _normalize((1.0 - t_blend) * _normalize(sq) + t_blend * _normalize(best_resampled))

        sc, seq = _nesterov_lbfgs_hybrid(sq, sub_dead, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: Perturbation restarts
    n_perturb = 5
    perturb_start = time.monotonic()
    perturb_end = deadline - 200.0
    if perturb_end > perturb_start + 60.0:
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 5.0:
                break

            p = _perturb(best_seq, rng)
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 4: Final intensive Nesterov + very high beta LSE
    final_dead = deadline - 15.0
    if time.monotonic() < final_dead - 30.0:
        # Very high beta LSE
        rem = final_dead - time.monotonic()
        if rem > 30.0:
            for beta in [500000, 1000000, 2000000]:
                rem = final_dead - time.monotonic()
                if rem < 15.0:
                    break
                maxfun = int(max(30, min(500, rem * 5)))
                sc, a2 = _lbfgs_lse(best_seq, beta, final_dead - 5.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
                best_seq = _normalize(a2)

        # Final Nesterov with very tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            n_it = int(max(200, rem * 25))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=2, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # One more Nesterov pass with k=1 (focus on single peak)
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            n_it = int(max(100, rem * 20))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=1, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov k=1: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
