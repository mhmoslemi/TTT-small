        scale[above] = (P_mean * thresh / P[above]) ** clip

"""
Peak-shaving optimizer: minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative a.
Key innovation: time-domain multiplicative update driven by soft-max peak weights 
(KKT-condition-motivated), combined with Fienup frequency-domain steps and L-BFGS polish.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    conv_res = np.fft.irfft(np.fft.rfft(A_pad) * np.fft.rfft(p_rev), M_fft)[:conv_len]
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def peak_shaving(a, T, alpha, a_pad_M=None, M=None):
    """
    One peak-shaving iteration.
    T: temperature for soft-max
    alpha: step size (0 < alpha <= 1)
    Returns updated a.
    """
    n = len(a)
    c = fast_conv(a)
    
    # Soft-max weights
    c_max = c.max()
    mu = np.exp((c - c_max) / T)
    mu_sum = mu.sum()
    if mu_sum < 1e-30:
        mu = np.exp(c / T)
        mu_sum = mu.sum()
    mu = mu / mu_sum
    
    # Cross-correlation: f_i = 2 * sum_k mu_k * a_{k-i}
    if M is None:
        M = 1 << (3 * n - 2).bit_length()
    if a_pad_M is None:
        a_pad_M = M
    
    mu_pad = np.zeros(M)
    mu_pad[:2*n-1] = mu
    a_pad = np.zeros(M)
    a_pad[:n] = a
    
    cross = np.fft.irfft(np.fft.rfft(mu_pad) * np.conj(np.fft.rfft(a_pad)), M)
    f = 2.0 * cross[:n]
    
    # Multiplicative update
    f_mean = f.mean()
    if f_mean < 1e-30:
        return a
    f_safe = np.maximum(f, 1e-15 * f_mean)
    a_new = a * (f_mean / f_safe) ** alpha
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a
    return a_new / S


def fienup_step(a, n_iters=30, thresh=1.5, clip=0.5):
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = a.copy()
    for _ in range(n_iters):
        A = np.fft.rfft(a, M)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        if P_mean < 1e-30:
            break
        above = P > P_mean * thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = (P_mean * thresh / P[above]) ** clip
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
    return a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_score_and_grad(x, beta)
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out

def peak_coord_descent(a, n_sweeps=5, n_top_peaks=30, n_top_coord=15):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(n_top_peaks, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(n_top_coord, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_local]
            top_global = j_start + top_local
            
            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for factor in [0.5, 0.7, 0.85, 1.2, 1.4]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_tmp = a / S_tmp
                    sc = compute_score(a_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_tmp.copy()
                        a = a_tmp
                        break
                else:
                    a[j] = old_val
    
    return best_a
