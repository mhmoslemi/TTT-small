def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """
    LSE-smoothed score and gradient. Returns (score, grad).
    Assumes sum(a) is normalized to 1 for simplicity.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    p = exp_shifted / exp_shifted.sum()

    # Wait: d/da[i] (sum_j a[j] * a[k-j]) = a[k-i] (from j=i term) + sum_j a[j] * d(a[k-j])/da[i]
    # = a[k-i] + a[k-i] (from the k-j=i term, i.e., j=k-i)
    # = 2*a[k-i] (if k-i is in [0,n-1])

    # = 2 * (cross-correlation of a and p, evaluated at offset i)
    
    # This is: 2 * sum_j a[j] * p[j+i] = 2 * (a * p_rev)[2n-2-i] ... hmm
    # Let me just compute it directly using the convolution result.
    
    # conv_res[k] = sum_j a[j] * p_rev[k-j] where p_rev[m] = p[conv_len-1-m]
    # = sum_j a[j] * p[conv_len-1-k+j]
    # For this to contribute to dL/da[i], we need conv_len-1-k+j to be a valid index in p,
    # and the relationship to i is: we want p[k] * a[k-i], so k-i = j means j = k-i, k = j+i
    # conv_res[j+i] = sum_m a[m] * p[conv_len-1-j-i+m]
    # If m = j: a[j] * p[conv_len-1-i]
    # Hmm, this is getting confusing. Let me just use a direct approach.
    
    # dL/da[i] = 2 * sum_{k=0}^{2n-2} p[k] * a[k-i] (with a[m]=0 if m not in [0,n-1])
    # = 2 * sum_{k=i}^{i+n-1} p[k] * a[k-i]  (since a[k-i] != 0 only when 0 <= k-i <= n-1)
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # = 2 * cross_correlation(a, p)[i]
    
    # cross_correlation(a, p)[i] = sum_j a[j] * p[i+j]
    # This can be computed as: (a * p_flip)[2n-2-i] where p_flip = p[::-1] and * is linear conv
    # linear_conv(a, p_flip)[k] = sum_j a[j] * p_flip[k-j] = sum_j a[j] * p[2n-2-(k-j)] = sum_j a[j] * p[2n-2-k+j]

    # So dL/da[i] = 2 * linear_conv(a, p_flip)[2n-2-i]
    
    # conv_res above computes linear_conv(a, p_rev) where p_rev = p[::-1] padded
    # So conv_res[2n-2-i] = sum_j a[j] * p[2n-2-(2n-2-i)+j] = sum_j a[j] * p[i+j] ✓

    # So the array is conv_res[2n-2], conv_res[2n-3], ..., conv_res[n-1] = conv_res[n-1:2n-1][::-1] ✓
    
    # Full score and gradient (including the S^2 normalization)
    # score = 2n * L / S^2
    # d(score)/da[i] = 2n * (grad_L[i] / S^2 - 2*L*a[i] / S^3)
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3 * a)
    return score, grad

def adam_lse(a, max_iters=5000, lr=0.01, beta1=0.9, beta2=0.999,

def peak_coord_descent(a, n_sweeps=10, n_top_peaks=10, n_top_coords=30):

    For each peak, try adjusting the coordinates that contribute most.
    """
    n = len(a)

        return a

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contribution of a[j] to c[k] is a[k-j]
            contribs = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            n_local = min(n_top_coords, len(contribs))
            top_local = np.argsort(contribs)[::-1][:n_local]
            
            for li in top_local:

                old_val = a[j]
                if old_val < 1e-12:

                        continue
                    a_n = a / S_tmp
                    sc = compute_score(a_n)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_n.copy()
                        a = a_n.copy()
                        improved = True
                        break
                if not improved:
                    a[j] = old_val
        
        if not improved:
            # Try smaller steps
            for j in range(0, n, max(1, n // 500)):
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                for factor in [0.9, 0.95, 1.05, 1.1]:

def hessian_vec_product(a, v, p, beta):
    """
    Compute Hessian-vector product for the LSE objective.
    H = (2n/S^2) * (2/beta) * sum_k p_k C_k - (4n*L/S^3) * (outer terms)
    We compute the first part (the main Hessian of the LSE of conv).
    """
    n = len(a)
    conv_len = 2 * n - 1
    M_fft = 1 << (conv_len - 1).bit_length()
    
    # (sum_k p_k C_k v)_i = sum_k p_k v_{k-i}
    # = cross-correlation of p and v at offset i
    # = (p * v_rev)[conv_len-1-i] ... let me verify
    # linear_conv(p, v_rev)[k] = sum_j p[j] * v_rev[k-j] = sum_j p[j] * v[conv_len-1-k+j]
    # For k = conv_len-1-i: sum_j p[j] * v[i+j] ... hmm that's not quite right
    
    # Let me just compute: (C_k v)_i = v_{k-i} (if 0 <= k-i <= n-1, else 0)
    # sum_k p_k (C_k v)_i = sum_k p_k v_{k-i} = sum_{m} p_{m+i} v_m (where m = k-i, k = m+i)
    # = sum_m v_m p_{m+i} = cross-correlation(v, p)[i]
    
    # cross_correlation(v, p)[i] = sum_m v[m] * p[m+i]
    # = linear_conv(v, p_rev)[2n-2-i] where p_rev = p padded and reversed
    # Actually let me use the same formula as before:
    # linear_conv(v_pad, p_rev_pad)[k] = sum_j v[j] * p_rev[k-j] = sum_j v[j] * p[2n-2-k+j]
    # For k = 2n-2-i: sum_j v[j] * p[i+j] ✓
    
    v_pad = np.zeros(M_fft)
    v_pad[:n] = v
    p_pad = np.zeros(M_fft)
    p_pad[:conv_len] = p
    p_rev_pad = np.zeros(M_fft)
    p_rev_pad[:conv_len] = p[::-1]
    
    cross = np.fft.irfft(np.fft.rfft(v_pad) * np.fft.rfft(p_rev_pad), M_fft)[:conv_len]

    hvp = cross[n-1:2*n-1][::-1]  # [cross[2n-2], cross[2n-3], ..., cross[n-1]]
    
    return hvp
