```python
def soca_peak_shave(a, n_iters=1000, tau_init=1.0, tau_final=0.0, M=None):
    """
    SOCA-like peak shaving.
    tau is the threshold as a fraction of the current peak.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S  # normalize to sum 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / n_iters
        tau = tau_init * (1 - progress) + tau_final * progress
        
        A = np.fft.rfft(a, M)
        B = np.abs(A) ** 2
        peak = B.max()
        threshold = peak * (1 - tau)
        
        # Form target
        B_target = np.minimum(B, threshold)
        A_target = np.sqrt(B_target) * np.exp(1j * np.angle(A))
        
        # Reconstruct
        a_new = np.fft.irfft(A_target, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
```

If `a` is monotone (say increasing), then `a[j]` is increasing in `j` and `a[n-1-j]` is decreasing in `j`, so by rearrangement, `b[n-1] <= sum_j a[j]^2` (opposite sorting gives minimum). But if `a` is "bell-shaped" (increasing then decreasing), then `a[j]` and `a[n-1-j]` are similarly sorted, and `b[n-1] >= sum_j a[j]^2`.

In any case, for our optimization, the peak of `b` could be at any lag.

```python
def greedy_peak_shave(a, n_iters=5000, top_k=5):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:

        contrib = a[j_start:j_end+1][::-1].copy()

        top_idx_global = j_start + top_idx_local

            a[j] *= 0.95

            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
```

```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines improved SOCA peak shaving, GS-Fienup with adaptive annealing,
L-BFGS with LSE smoothing (adaptive beta), greedy peak shaving,

multi-length strategy, adaptive LSE beta, and targeted perturbation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def soca_multi(a, n_iters=1500, M=None, schedule='cosine', n_peaks=1):
    """
    SOCA with adaptive peak shaving. Shaves the top n_peaks simultaneously.
    schedule controls how aggressively peaks are reduced over iterations.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    half_M = M // 2 + 1
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        if schedule == 'cosine':
            tau = 0.5 * (1.0 + np.cos(np.pi * progress))
        elif schedule == 'linear':
            tau = 1.0 - progress
        elif schedule == 'aggressive':
            tau = max(0.0, 1.0 - 2.0 * progress)
        else:
            tau = 0.5
        A = np.fft.rfft(a, M)
        B = np.abs(A) ** 2
        peak = float(B.max())
        target_floor = peak * tau
        B_target = np.maximum(B, target_floor)
        # Soft clip: smooth transition at the threshold
        B_target = B + (target_floor - B) * np.exp(-(B - target_floor) / (0.01 * peak + 1e-20))
        B_target = np.maximum(B_target, 0.0)
        A_target = np.sqrt(B_target) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_target, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def gs_flat_adaptive(a, n_iters=800, M=None, alpha_init=0.5, alpha_final=0.03):
    """GS-Fienup with flat convolution target and cosine alpha annealing."""
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()
    b_target = np.zeros(M)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.rfft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        # Cosine annealing

        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)

def lse_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # q[j] = sum_k p[k] * a[k-j] via cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_pad = np.zeros(M2_fft)
    A_pad[:n] = a
    P_pad = np.zeros(M2_fft)
    P_pad[:2*n-1] = p
    P_fft = np.fft.rfft(P_pad)
    A_fft = np.fft.rfft(A_pad)
    Q = np.fft.irfft(P_fft * np.conj(A_fft), M2_fft)
    q = Q[:n]
    grad_L = 2.0 * q
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x, beta):
    x_safe = np.maximum(x, 0.0)
    return lse_grad(x_safe, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=150000):

    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def greedy_peak_shave(a, n_iters=3000, top_k=8, decay=0.97):
    """Greedy: reduce elements contributing most to the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end+1]
        contrib = seg[::-1].copy()
        top_local = np.argsort(contrib)[::-1][:top_k]
        top_global = j_start + top_local
        for j in top_global:
            a[j] *= decay
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def subgradient_topm(a, n_iters=3000, top_m=15, step_init=0.06):
    """Subgradient method on top-M peaks with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += 2.0 * a[j_start:j_end+1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.08)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9997
    return best_a


def perturbed_reopt(a, rng, n_restarts=12, n_iters_gs=300):
    """Random perturbations followed by GS re-optimization."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for r in range(n_restarts):
        eps = rng.uniform(0.02, 0.15)
        noise = rng.standard_normal(n)
        a_pert = np.maximum(a + eps * noise * a, 0.0)
        S = float(a_pert.sum())
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        a_opt = gs_flat_adaptive(a_pert, n_iters=n_iters_gs, alpha_init=0.3, alpha_final=0.05)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent, 0.0, 1000.0).copy())
        else:
            inits.append(np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(parent)), parent))
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.3 + 0.7 * np.cos(2.0*np.pi*x)**2)
    inits.append(np.maximum(0.5 + 0.4*np.cos(2.0*np.pi*x), 0.01))
    inits.append(np.maximum(0.4 + 0.35*np.cos(4.0*np.pi*x) + 0.2*np.sin(2.0*np.pi*x), 0.01))
    inits.append(np.clip(0.5 + 0.3*rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.5 + 0.5*(2.0*x - 1.0)**2)
    inits.append(np.maximum(0.3 + 0.4*np.cos(6.0*np.pi*x) + 0.2*np.cos(2.0*np.pi*x), 0.01))
    inits.append(np.maximum(0.4 + 0.3*np.sin(2.0*np.pi*3.0*x)*np.cos(2.0*np.pi*x), 0.01))
    # Chirp-like initialization
    inits.append(0.5 + 0.3*np.cos(2.0*np.pi*x**2))
    return inits


def coarse_to_fine(a_coarse, n_target, n_gs_iters=400):
    """Upsample coarse solution to target length and refine with GS."""
    a_up = np.interp(np.linspace(0, 1, n_target),
                     np.linspace(0, 1, len(a_coarse)), a_coarse)
    a_up = np.maximum(a_up, 0.0)
    a_up = a_up / a_up.sum()
    a_fine = gs_flat_adaptive(a_up, n_iters=n_gs_iters, alpha_init=0.4, alpha_final=0.05)
    return a_fine


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-phase hybrid: GS + SOCA + L-BFGS + greedy + subgradient + perturbation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    total = float(budget_s) - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    n_parent = len(parent) if parent is not None else 6962
    lengths = [n_parent, 6962, 8192]
    lengths = sorted(set(l for l in lengths if l >= 1024))[:4]
    print(f"Lengths: {lengths}, parent={'yes' if parent is not None else 'no'} (n={n_parent})")

    # ============================================================
    # Phase 0: Quick coarse solve to get a shape (5% time)
    # ============================================================
    phase0_end = t_start + 0.05 * total
    print(f"\n=== Phase 0: Coarse solve (t={phase0_end-t_start:.0f}s) ===")
    n_coarse = 512
    for i in range(5):
        if time.monotonic() > phase0_end:
            break
        x = np.arange(n_coarse, dtype=np.float64) / n_coarse
        if i == 0:
            a_c = np.ones(n_coarse) * 0.5
        elif i == 1:
            a_c = 0.3 + 0.7 * np.cos(2*np.pi*x)**2
        elif i == 2:
            a_c = np.maximum(0.5 + 0.4*np.cos(4*np.pi*x), 0.01)
        elif i == 3:
            a_c = np.clip(0.5 + 0.3*rng.standard_normal(n_coarse), 0.01, 10.0)
        else:
            a_c = 0.5 + 0.5*(2*x-1)**2
        a_c = gs_flat_adaptive(a_c, n_iters=300, alpha_init=0.5, alpha_final=0.05)
        sc_c = compute_score(a_c)
        print(f"  Coarse init {i}: score={sc_c:.6f}")

    # ============================================================
    # Phase 1: GS + SOCA on main lengths (35% time)
    # ============================================================
    phase1_end = t_start + 0.40 * total
    print(f"\n=== Phase 1: GS+SOCA (t={phase1_end-t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Length n={n}")
        inits = build_inits(n, rng, parent)
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            # GS
            a1 = gs_flat_adaptive(a0, n_iters=600, alpha_init=0.5, alpha_final=0.03)
            update_best(compute_score(a1), a1, f"GS n={n} i={i}")
            # SOCA
            a2 = soca_multi(a1, n_iters=800, schedule='cosine')
            update_best(compute_score(a2), a2, f"SOCA n={n} i={i}")
            # Phase-only GS
            a3 = gs_phase_only(a2, n_iters=300, alpha=0.2)
            update_best(compute_score(a3), a3, f"GS-phase n={n} i={i}")

    print(f"After Phase 1: best={best_score:.8f}")

    # ============================================================
    # Phase 2: Subgradient + Greedy (10% time)
    # ============================================================
    phase2_end = t_start + 0.50 * total
    print(f"\n=== Phase 2: SG+Greedy (t={phase2_end-t_start:.0f}s) ===")

    n_best = len(best_a)
    for top_m in [8, 15, 30]:
        if time.monotonic() > phase2_end:
            break
        a_sg = subgradient_topm(best_a, n_iters=2000, top_m=top_m)
        update_best(compute_score(a_sg), a_sg, f"SG m={top_m}")

    if time.monotonic() < phase2_end:
        a_gr = greedy_peak_shave(best_a, n_iters=2000, top_k=8)
        update_best(compute_score(a_gr), a_gr, "greedy")

    # ============================================================
    # Phase 3: L-BFGS with multiple beta (30% time)
    # ============================================================
    phase3_end = t_start + 0.80 * total
    print(f"\n=== Phase 3: L-BFGS (t={phase3_end-t_start:.0f}s) ===")

    n_best = len(best_a)
    betas = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        update_best(sc, a_opt, f"LBFGS b={beta}")
        print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # Phase 4: Perturbation + re-optimize (12% time)
    # ============================================================
    phase4_end = t_start + 0.92 * total
    print(f"\n=== Phase 4: Perturbation (t={phase4_end-t_start:.0f}s) ===")

    if best_score < 2.0:
        a_pert = perturbed_reopt(best_a, rng, n_restarts=10, n_iters_gs=300)
        update_best(compute_score(a_pert), a_pert, "perturb")
        if time.monotonic() < phase4_end:
            sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=5000, maxfun=80000)
            update_best(sc, a_opt, "perturb+LBFGS")

    # Try at a different length
    n_alt = 8192 if n_best != 8192 else 9216
    if time.monotonic() < phase4_end - 30:
        a_alt = np.interp(np.linspace(0, 1, n_alt),
                          np.linspace(0, 1, n_best), best_a)
        a_alt = np.maximum(a_alt, 0.0)
        a_alt = a_alt / a_alt.sum()
        a_alt_gs = gs_flat_adaptive(a_alt, n_iters=400, alpha_init=0.4, alpha_final=0.05)
        a_alt_soc = soca_multi(a_alt_gs, n_iters=500, schedule='cosine')
        update_best(compute_score(a_alt_soc), a_alt_soc, f"alt n={n_alt}")

    # ============================================================
    # Phase 5: Final refinement (8% time)
    # ============================================================
    print(f"\n=== Phase 5: Final refinement ===")
    n_best = len(best_a)

    if time.monotonic() < deadline - 20:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "final-LBFGS-hi")

    if time.monotonic() < deadline - 10:
        a_final = soca_multi(best_a, n_iters=500, schedule='aggressive')
        update_best(compute_score(a_final), a_final, "final-SOCA-aggr")

    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 100, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "final-LBFGS-lo")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {float(budget_s)}s")
    print(f"{'='*60}")
    return out
```
