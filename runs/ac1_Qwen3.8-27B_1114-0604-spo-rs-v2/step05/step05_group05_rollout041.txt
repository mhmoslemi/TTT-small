```python
"""
Searches for non-negative sequences minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

Strategy:
  1. Use height_sequence_1 if available, otherwise build several dense
     asymmetric initial profiles.
  2. Normalize sequences to sum 1 and optimize with a mirror-descent /
     negative-entropy multiplicative update:
          log a_i <- log a_i - eta * g_i
     where g_i is a smooth (tempered max) subgradient of max(conv(a,a)).
  3. Tempered max is implemented by assigning larger weight to lags of
     the convolution that are close to the maximum. A compact active-lag
     path is used when only few lags carry weight; otherwise FFT gradient
     is used.
  4. Periodically restart from small log-space perturbations, and try
     slightly larger lengths by upsampling.
  5. A projected-gradient polish is included.
"""

import numpy as np
import time

FLOOR = 1e-30


def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _normalize(a):
    """Clip/normalize a sequence to non-negative sum-1 float64 array."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.clip(a, FLOOR, 1000.0)
    s = float(a.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    a /= s
    return a


def _self_conv(a, L):
    """Linear convolution a*a by FFT. Assumes sum(a)=1 mostly."""
    n = int(a.size)
    m = 2 * n - 1
    x = np.zeros(L, dtype=np.float64)
    x[:n] = a
    X = np.fft.rfft(x)
    X *= X
    c = np.fft.irfft(X, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_c(c, n):
    return 2.0 * float(n) * float(np.max(c))


def _score_std(a):
    n = int(a.size)
    L = _next_pow2(2 * n - 1)
    c = _self_conv(a, L)
    return _score_from_c(c, n)


def _grad_fft(w, a, L):
    """
    Gradient for M(a) approximated by weighted max:
        g_i = 2 * sum_k w_k a_{k-i}
    w length is 2n-1, a length n. Returns g length n.
    """
    n = int(a.size)
    x = np.zeros(L, dtype=np.float64)
    x[:n] = a
    y = np.zeros(L, dtype=np.float64)
    y[:2 * n - 1] = w[::-1]

    X = np.fft.rfft(x)
    Y = np.fft.rfft(y)
    Y *= X
    conv = np.fft.irfft(Y, L)

    # Need indices 2n-2, 2n-3, ..., n-1
    g = 2.0 * conv[n - 1:2 * n - 1]
    g = g[::-1].copy()
    np.maximum(g, 0.0, out=g)
    return g


def _grad_active(active_idx, active_w, a, out):
    """Sparse active-lag gradient. out is overwritten and returned."""
    n = int(a.size)
    out.fill(0.0)
    for t in range(active_idx.size):
        k = int(active_idx[t])
        ww = 2.0 * float(active_w[t])
        if ww == 0.0:
            continue

        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e > n - 1:
            e = n - 1
        if e < s:
            continue

        # For i=s..e, partner index is k-i, decreasing from k-s to k-e.
        seg = a[k - e:k - s + 1][::-1]
        out[s:e + 1] += ww * seg
    return out


def _grad_from_c(a, c, M, gamma, L2, max_top=128):
    """
    Compute a tempered-max gradient. gamma is dimensionless:
       weight_k = exp(-gamma * (M - c_k) / M).
    """
    n = int(a.size)
    if M <= 1e-300:
        return np.zeros(n, dtype=np.float64)

    # If only a few lags have significant weight, use sparse gradient.
    if gamma >= 8.0:
        cut = 1e-8
        max_delta = -np.log(cut) / float(gamma)
        if max_delta < 1.0:
            threshold = M * (1.0 - max_delta)
            idx = np.flatnonzero(c >= threshold)
            if 0 < idx.size <= max_top:
                vals = np.exp(-gamma * (M - c[idx]) / M)
                s = float(vals.sum())
                if np.isfinite(s) and s > 1e-300:
                    vals = np.asarray(vals, dtype=np.float64)
                    vals /= s
                    out = np.zeros(n, dtype=np.float64)
                    return _grad_active(idx, vals, a, out)

    # Full FFT gradient.
    shifted = gamma * (c - M) / M
    np.clip(shifted, -700.0, 0.0, out=shifted)
    w = np.exp(shifted)
    s = float(w.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        idx = np.asarray([int(np.argmax(c))], dtype=np.int64)
        vals = np.asarray([1.0], dtype=np.float64)
        out = np.zeros(n, dtype=np.float64)
        return _grad_active(idx, vals, a, out)

    w /= s
    return _grad_fft(w, a, L2)


def _log_perturb(a, seed, scale):
    """Add log-normal multiplicative noise."""
    a = np.asarray(a, dtype=np.float64)
    if scale <= 0.0:
        return a.copy()
    rng = np.random.default_rng(seed)
    loga = np.log(np.maximum(a, FLOOR))
    b = loga + float(scale) * rng.standard_normal(a.size)
    np.clip(b, -700.0, 700.0, out=b)
    c = np.exp(b)
    c = np.maximum(c, FLOOR)
    s = float(c.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return a.copy()
    c /= s
    return c


def _upsample(a, new_n):
    """Linearly interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    old_n = int(a.size)
    new_n = int(new_n)
    if old_n <= 0 or new_n <= old_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _build_inits(n, seed):
    """Fallback initial asymmetric profiles."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    out = []

    a = np.exp(rng.normal(0.0, 0.5, int(n)))
    out.append(_normalize(a))

    a = np.exp(-3.0 * x) + 0.08
    out.append(_normalize(a))

    a = np.exp(3.0 * x) + 0.08
    out.append(_normalize(a))

    a = np.sin(np.pi * x) ** 2 + 0.03
    out.append(_normalize(a))

    a = 1.0 / (1.0 + np.exp(-9.0 * (x - 0.5))) + 0.03
    out.append(_normalize(a))

    a = 0.10 + x
    out.append(_normalize(a))

    a = 0.10 + (1.0 - x)
    out.append(_normalize(a))

    a = np.exp(-2.0 * np.abs(x - 0.35)) + 0.08
    out.append(_normalize(a))

    a = np.exp(-2.0 * np.abs(x - 0.65)) + 0.08
    out.append(_normalize(a))

    return out


def multiplicative_refine(start, max_time, gamma_start=5.0, gamma_end=400.0,
                          eta_init=1.0, eta_min=0.02, eta_max=8.0,
                          max_top=128):
    """Mirror-descent / negative-entropy multiplicative refinement.

    Returns (best_a, best_score) where best_a has sum 1.
    """
    if max_time <= 0.0:
        a0 = _normalize(start)
        return a0, _score_std(a0)

    a = _normalize(start)
    n = int(a.size)
    if n <= 0:
        return a, np.inf

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n - 2)

    c = _self_conv(a, L1)
    M = float(c.max())
    sc = _score_from_c(c, n)

    best_a = a.copy()
    best_sc = sc

    end = time.monotonic() + float(max_time)
    gamma = int(gamma_start) if gamma_start == int(gamma_start) else float(gamma_start)
    eta = float(eta_init)
    no_accept = 0
    it = 0

    while time.monotonic() < end:
        it += 1
        if M <= 1e-300:
            break

        if it % 300 == 0:
            gamma = min(float(gamma_end), gamma * 1.003)

        g = _grad_from_c(a, c, M, float(gamma), L2, max_top=max_top)
        gmean = float(np.mean(g))
        gmin = float(np.min(g))
        gmax = float(np.max(g))
        range_g = gmax - gmin
        range_safe = max(range_g, 1e-8 * max(abs(gmean), 1e-8))

        if range_g <= 1e-16 * max(abs(gmean), 1e-8):
            no_accept += 1
            if no_accept > 8000:
                break
            continue

        accepted = False
        base_eta = eta

        for mult in (1.0, 0.5, 0.25):
            e = base_eta * mult
            if e < eta_min:
                break

            logch = -e * (g - gmean) / range_safe
            if not np.all(np.isfinite(logch)):
                logch = np.where(np.isfinite(logch), logch, 0.0)
            np.clip(logch, -15.0, 15.0, out=logch)

            if float(np.max(np.abs(logch))) < 1e-14:
                break

            cand = a * np.exp(logch)
            cand = np.maximum(cand, FLOOR)
            cs = float(cand.sum())
            if (not np.isfinite(cs)) or cs <= 1e-300:
                continue
            cand /= cs

            c2 = _self_conv(cand, L1)
            M2 = float(c2.max())
            sc2 = _score_from_c(c2, n)

            if sc2 < sc - 1e-13:
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = cand.copy()

                a = cand
                c = c2
                M = M2
                sc = sc2
                eta = min(float(eta_max), max(float(eta_min), e * 1.02))
                no_accept = 0
                accepted = True
                break

        if not accepted:
            eta = max(float(eta_min), base_eta * 0.6)
            no_accept += 1
            if no_accept % 2000 == 0:
                gamma = min(float(gamma_end), gamma * 1.35)
            if no_accept > 10000:
                break

    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def projected_refine(start, max_time, gamma_start=30.0, gamma_end=400.0,
                     lr_init=0.4, lr_min=1e-4, lr_max=8.0, max_top=128):
    """Projected gradient polish on the nonnegative simplex."""
    if max_time <= 0.0:
        a0 = _normalize(start)
        return a0, _score_std(a0)

    a = _normalize(start)
    n = int(a.size)
    if n <= 0:
        return a, np.inf

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n - 2)

    c = _self_conv(a, L1)
    M = float(c.max())
    sc = _score_from_c(c, n)

    best_a = a.copy()
    best_sc = sc

    end = time.monotonic() + float(max_time)
    gamma = float(gamma_start)
    lr = float(lr_init)
    no_accept = 0
    it = 0

    while time.monotonic() < end:
        it += 1
        if M <= 1e-300:
            break

        if it % 300 == 0:
            gamma = min(float(gamma_end), gamma * 1.003)

        g = _grad_from_c(a, c, M, float(gamma), L2, max_top=max_top)
        d = g - float(np.mean(g))

        maxd = float(np.max(np.abs(d)))
        if maxd <= 1e-15:
            no_accept += 1
            if no_accept > 8000:
                break
            continue

        accepted = False
        base_lr = lr

        for mult in (1.0, 0.5, 0.25):
            ell = base_lr * mult
            if ell < lr_min:
                break

            cand = a - ell * d
            cand = np.maximum(cand, FLOOR)
            cs = float(cand.sum())
            if (not np.isfinite(cs)) or cs <= 1e-300:
                continue
            cand /= cs

            c2 = _self_conv(cand, L1)
            M2 = float(c2.max())
            sc2 = _score_from_c(c2, n)

            if sc2 < sc - 1e-13:
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = cand.copy()

                a = cand
                c = c2
                M = M2
                sc = sc2
                lr = min(float(lr_max), max(float(lr_min), ell * 1.03))
                no_accept = 0
                accepted = True
                break

        if not accepted:
            lr = max(float(lr_min), base_lr * 0.6)
            no_accept += 1
            if no_accept % 2000 == 0:
                gamma = min(float(gamma_end), gamma * 1.35)
            if no_accept > 10000:
                break

    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best sequence found within the time budget."""
    t0 = time.monotonic()
    total_budget = float(budget_s)
    deadline = t0 + max(60.0, total_budget - 45.0)

    rng = np.random.default_rng(seed)
    best_a = None
    best_score = np.inf
    has_parent = False

    # ---------------- Load parent if available ----------------
    try:
        parent = globals().get("height_sequence_1", None)
        if parent is not None:
            arr = np.asarray(parent, dtype=np.float64).ravel()
            if arr.size > 0:
                bad = ~np.isfinite(arr)
                if bad.any():
                    arr = arr.copy()
                    arr[bad] = 0.0
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.size >= 128 and float(arr.sum()) > 0.01:
                    if arr.size > 20000:
                        arr = _upsample(arr, 16384)
                    best_a = _normalize(arr)
                    best_score = _score_std(best_a)
                    has_parent = True
                    print(f"Loaded parent length={len(best_a)} score={best_score:.8f}")
    except Exception:
        has_parent = False
        best_a = None

    # ---------------- Fallback initial profile ----------------
    if best_a is None:
        n = 6962
        cands = _build_inits(n, seed)
        for cand in cands:
            s = _score_std(cand)
            if s < best_score:
                best_score = s
                best_a = cand
        print(f"Fallback init length={len(best_a)} score={best_score:.8f}")

    def remaining():
        return max(0.0, deadline - time.monotonic())

    # ---------------- Quick deterministic start polish ----------------
    # Small direct perturbations / tilts, cheap but sometimes useful.
    if remaining() > 120.0:
        x = np.linspace(-1.0, 1.0, len(best_a), dtype=np.float64)
        tilts = [-0.45, -0.25, -0.12, 0.12, 0.25, 0.45]
        for ti in tilts:
            cand = best_a * np.exp(0.35 * ti * x)
            cand = _normalize(cand)
            s = _score_std(cand)
            if s < best_score:
                best_score = s
                best_a = cand
                print(f"tilt {ti:.2f}: {s:.8f}")

        for sc in (0.001, 0.003, 0.01):
            cand = _log_perturb(best_a, seed + 11 * int(1000 * sc), sc)
            s = _score_std(cand)
            if s < best_score:
                best_score = s
                best_a = cand
                print(f"quick perturb {sc}: {s:.8f}")

    # ---------------- Main multiplicative refinement ----------------
    main_budget = min(0.42 * (total_budget - 60.0), remaining() - 30.0)
    if main_budget > 120.0:
        first = min(650.0, 0.62 * main_budget)
        if has_parent:
            g0 = 10.0
        else:
            g0 = 3.0
        print(f"Main multiplicative phase budget={main_budget:.0f}s")
        ra, sa = multiplicative_refine(
            best_a, first,
            gamma_start=g0, gamma_end=220.0,
            eta_init=1.0, eta_min=0.02, eta_max=8.0
        )
        if sa < best_score:
            best_score = sa
            best_a = ra
            print(f"After main A: {best_score:.8f}")

        second = max(0.0, main_budget - first - 10.0)
        if second > 60.0:
            rb, sb = multiplicative_refine(
                best_a, second,
                gamma_start=80.0, gamma_end=500.0,
                eta_init=0.45, eta_min=0.01, eta_max=4.0
            )
            if sb < best_score:
                best_score = sb
                best_a = rb
                print(f"After main B: {best_score:.8f}")

    # ---------------- Projected gradient polish ----------------
    proj_budget = min(0.12 * (total_budget - 60.0), remaining() - 30.0)
    if proj_budget > 80.0:
        print(f"Projected phase budget={proj_budget:.0f}s")
        rc, sc = projected_refine(
            best_a, proj_budget,
            gamma_start=30.0, gamma_end=450.0,
            lr_init=0.4, lr_min=1e-4, lr_max=8.0
        )
        if sc < best_score:
            best_score = sc
            best_a = rc
            print(f"After projected: {best_score:.8f}")

    # ---------------- Perturbation restarts ----------------
    perturb_budget = min(0.22 * (total_budget - 60.0), remaining() - 30.0)
    if perturb_budget > 180.0:
        scales = [0.002, 0.005, 0.010, 0.020, 0.040, 0.080]
        print(f"Perturbation phase budget={perturb_budget:.0f}s")
        idx = 0
        while remaining() > 120.0 and idx < len(scales):
            scale = scales[idx]
            idx += 1
            t_alloc = min(180.0, remaining() - 40.0)
            pert = _log_perturb(best_a, seed + 9173 * (idx + 1), scale)
            spert = _score_std(pert)
            if spert < best_score:
                best_score = spert
                best_a = pert
                print(f"perturb {scale:g} direct: {best_score:.8f}")

            rd, sd = multiplicative_refine(
                pert, t_alloc,
                gamma_start=20.0, gamma_end=450.0,
                eta_init=0.9, eta_min=0.015, eta_max=5.0
            )
            if sd < best_score:
                best_score = sd
                best_a = rd
                print(f"perturb {scale:g} refined: {best_score:.8f}")
            elif remaining() < 120.0:
                break

    # ---------------- Larger length exploration ----------------
    if remaining() > 260.0:
        n0 = len(best_a)
        if n0 < 8192:
            n2 = 8192
        else:
            n2 = min(16384, max(8192, 2 * n0))

        if n2 != n0 and n2 <= 16384:
            print(f"Upsample exploration to n={n2}, remaining={remaining():.0f}s")
            cand = _upsample(best_a, n2)
            tc = min(330.0, remaining() - 40.0)
            re, se = multiplicative_refine(
                cand, tc,
                gamma_start=5.0 if not has_parent else 12.0,
                gamma_end=450.0,
                eta_init=0.9, eta_min=0.015, eta_max=4.5
            )
            if se < best_score:
                best_score = se
                best_a = re
                print(f"Upsampled n={n2}: {best_score:.8f}")
            else:
                print(f"Upsampled n={n2} best local={se:.8f}, global remains {best_score:.8f}")

            # One more bigger length if still useful and safe.
            n3 = 12288
            if remaining() > 320.0 and n3 > n2 and n3 <= 16384:
                print(f"Upsample exploration to n={n3}, remaining={remaining():.0f}s")
                cand3 = _upsample(best_a, n3)
                tc3 = min(300.0, remaining() - 40.0)
                rf, sf = multiplicative_refine(
                    cand3, tc3,
                    gamma_start=8.0, gamma_end=450.0,
                    eta_init=0.8, eta_min=0.015, eta_max=4.0
                )
                if sf < best_score:
                    best_score = sf
                    best_a = rf
                    print(f"Upsampled n={n3}: {best_score:.8f}")
                else:
                    print(f"Upsampled n={n3} best local={sf:.8f}, global remains {best_score:.8f}")

    # ---------------- Final high-temperature / high-beta polish ----------------
    final_budget = min(0.12 * (total_budget - 60.0), remaining() - 20.0)
    if final_budget > 60.0:
        print(f"Final polish budget={final_budget:.0f}s")
        final_a = min(0.6 * final_budget, 180.0)
        rg, sg = multiplicative_refine(
            best_a, final_a,
            gamma_start=110.0, gamma_end=520.0,
            eta_init=0.30, eta_min=0.005, eta_max=2.5
        )
        if sg < best_score:
            best_score = sg
            best_a = rg
            print(f"Final multiplicative: {best_score:.8f}")

        final_p = max(30.0, final_budget - final_a - 10.0)
        if final_p > 30.0:
            rh, sh = projected_refine(
                best_a, final_p,
                gamma_start=120.0, gamma_end=520.0,
                lr_init=0.08, lr_min=1e-5, lr_max=2.0
            )
            if sh < best_score:
                best_score = sh
                best_a = rh
                print(f"Final projected: {best_score:.8f}")

    # ---------------- Prepare valid output ----------------
    if best_a is None or best_a.size == 0:
        out = [0.5] * 64
        print("Fallback uniform output.")
        return out

    out = []
    for v in best_a:
        if v < 0.0:
            v = 0.0
        elif v > 1000.0:
            v = 1000.0
        out.append(float(v))

    s_out = sum(out)
    if (not np.isfinite(s_out)) or s_out < 0.01:
        out = [0.5] * 64

    # Final verification (evaluate_sequence is guaranteed to be present).
    try:
        verified = float(evaluate_sequence(out))
        if np.isfinite(verified):
            print(f"Final verified score: {verified:.8f}, length={len(out)}, time={time.monotonic()-t0:.1f}s")
        else:
            print("Final verified score was invalid; using safe fallback.")
            out = [0.5] * 64
    except Exception:
        print(f"Final internal score: {best_score:.8f}, length={len(out)}, time={time.monotonic()-t0:.1f}s")

    return out
```
