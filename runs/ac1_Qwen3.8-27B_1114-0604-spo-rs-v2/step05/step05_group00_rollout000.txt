"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: adaptive-beta LSE-smoothed L-BFGS, peak-targeted momentum subgradient,
exact coordinate descent, spectral Fienup shaping, and multi-length exploration.
Focuses on squeezing marginal improvements via aggressive beta annealing and
active-set peak identification.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_smooth(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of sum(p_k * c_k) w.r.t. a via FFT
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    a_rev = a[::-1]
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g_c = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_c / S2 - 2.0 * L * (2.0 * a) / S3) / 2.0
    grad = (2.0 * n) * (g_c / S2 - 2.0 * L / S3) - (2.0 * n) * (2.0 * L / S3) * 0
    # More carefully: d/da [2n*L/S^2] = 2n*(dL/da * S^2 - L * 2*S*dS/da) / S^4
    # = 2n*(dL/da/S^2 - 2L*dS/da/S^3)
    # dL/da = g_c (the weighted conv gradient)
    # dS/da = 1 (vector of ones)
    grad = (2.0 * n) * (g_c / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(x, beta):
    """Wrapper for scipy minimize."""
    return lse_smooth(x, beta)


def run_lbfgs_adaptive(a0, beta_start, beta_end, n_outer=5, n_inner_iters=3000):
    """L-BFGS with beta annealing from beta_start to beta_end."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
        S = a.sum()
    a = a / S
    bounds = [(0.0, 1000.0)] * n
    best_a = a.copy()
    best_sc = compute_score(a)

    for outer in range(n_outer):
        frac = outer / max(1, n_outer - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        res = minimize(
            lse_objective_wrapper, a, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': n_inner_iters, 'maxfun': n_inner_iters * 3, 'maxls': 30}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-6:
            continue
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new.copy()

    return best_sc, best_a


def peak_subgradient_momentum(a, n_iters, top_k=10, lr_init=0.01, momentum=0.9):
    """Subgradient descent on the exact max with momentum, targeting top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-k peak subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = c[top_idx] / c_max
        w_sum = float(weights.sum())

        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += (weights[i] / w_sum) * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        # Momentum
        v = momentum * v - lr * grad
        a = a + v
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        lr *= 0.9995

    return best_a


def coord_descent_exact(a, n_sweeps=5, n_top=200):
    """Coordinate descent with golden-section line search on top contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    golden = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break
        top_count = min(n_top, len(contribs))
        order = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + order
        cur_sc = compute_score(a)

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            # Golden section search in [0, 3*a_old]
            lo = 0.0
            hi = 3.0 * a_old
            for _ in range(8):
                m1 = hi - golden * (hi - lo)
                m2 = lo + golden * (hi - lo)
                a[j] = m1
                S1 = a.sum()
                if S1 > 1e-12:
                    sc1 = compute_score(a / S1)
                else:
                    sc1 = 1e10
                a[j] = m2
                S2v = a.sum()
                if S2v > 1e-12:
                    sc2 = compute_score(a / S2v)
                else:
                    sc2 = 1e10
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
            a[j] = (lo + hi) / 2.0
            S_tmp = a.sum()
            if S_tmp > 1e-12:
                a = a / S_tmp

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_spectral(a, n_iters=100, alpha_init=0.3, alpha_end=0.8):
    """Spectral-domain Fienup to flatten the power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters)
        alpha = alpha_init + (alpha_end - alpha_init) * frac
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        # Reduce peaks in spectrum
        P_target = P_mean
        above = P > P_target * 1.5
        if above.any():
            P[above] = P[above] * (1 - alpha) + P_target * alpha
        P_new = P * (1 - alpha) + P_target * alpha * 0.3
        P_new = np.maximum(P_new, 1e-30)
        A_new = A * np.sqrt(P_new / np.maximum(P, 1e-30))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)
