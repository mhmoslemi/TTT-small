- Given p, optimize a (this is a ratio of quadratic forms - can be solved via eigenvalue)
- Given a, optimize p (this is just p = argmax p^T c / sum p where c = a*a, so p is concentrated on the max)

The "optimize p" step is trivial (put all mass on the argmax), which gives us back the original problem. So this alternation doesn't directly help.

But the LSE smoothing is a practical relaxation:
max_k c_k ≈ (1/beta) * log(sum_k exp(beta * c_k))

And the gradient of this smoothed objective with respect to a is:
(2/beta) * sum_k p_k * (T_c)_{:k} where p_k = exp(beta*c_k)/sum exp(beta*c_j) and (T_c)_{:k} is the k-th row of the "convolution operator" matrix.

This is what the previous code does. The issue is that for the smoothing to be accurate, beta needs to be large, which makes the optimization ill-conditioned.

**Alternative smoothing: Hinge-based**

Instead of LSE, use a hinge-based approximation:
max_k c_k ≈ max_k c_k + (1/beta) * sum_k max(0, c_k - max_c - gamma)^2

This is a smooth approximation that's more locally accurate.

Or even better: use the **exact** subgradient of the max (which is a vector with 1 at the argmax and 0 elsewhere) and use a **proximal gradient method** (like FISTA) with the subgradient.

Actually, the previous code already does something similar with `peak_shave`. Let me think about how to improve it.

**Key insight for improvement: Variable step size and momentum**

The previous peak_shave uses a fixed decay for the step size. A better approach:
- Use backtracking line search
- Use Nesterov momentum
- Adaptively choose which peaks to focus on

**Another key insight: Normalization matters**

Since the score is scale-invariant, we should always normalize after each step. The previous code does this, but we should be careful about the interaction between normalization and the gradient step.

"""
Improved optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key improvements over previous:
1. LP-based direction finding (simplified with top-K constraints)
2. Nesterov momentum subgradient with backtracking
3. Full coordinate descent (all coordinates, not just peak contributors)
4. Adaptive Fienup with multiple target spectra
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time
import sys

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_aggressive(a, n_iters=500, n_targets=5):
    """
    Fienup with multiple target spectra for better mixing.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Generate multiple target spectra
    # Target 1: flat
    # Target 2: slightly peaked at DC
    # Target 3-5: random smooth spectra
    
    targets = []
    # Flat target
    targets.append(None)  # None means flat
    
    for target_iter in range(n_targets):
        progress = target_iter / n_targets
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        if target_iter == 0:
            P_target = np.full_like(P, P_mean)
        else:
            # Smoothed random target
            freq = np.arange(len(P))
            # Low-pass filter the current spectrum
            from scipy.ndimage import uniform_filter
            P_smooth = uniform_filter(P, size=max(2, len(P)//10))
            P_target = P_smooth + 0.1 * P_mean
        
        P_target = np.maximum(P_target, 1e-20)
        
        for it in range(n_iters // n_targets):
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P = np.maximum(P, 1e-30)
            
            # Blend towards target
            alpha = 0.3 + 0.5 * (it / max(1, n_iters // n_targets))
            P_blend = (1 - alpha) * P + alpha * P_target
            P_blend = np.maximum(P_blend, 1e-20)
            
            A_new = A * (np.sqrt(P_blend / P) ** alpha)
            a_new = np.fft.irfft(A_new, M_fft)[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            a = a_new
            
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a

def momentum_subgradient(a, n_iters=5000, top_k=10, momentum_init=0.9):
    """
    Subgradient method with Nesterov momentum and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    y = a.copy()
    t = 1.0
    momentum = momentum_init
    step = 0.01
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(y)
            if sc < best_sc:
                best_sc = sc
                best_a = y.copy()
        
        c = fast_conv(y)
        S_y = float(y.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S_y * S_y)
        
        # Find top-K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute subgradient (averaged over top-K peaks, weighted by peak value)
        sg = np.zeros(n)
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            weights[i] = c[k]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += weights[i] * y[j_start:j_end + 1][::-1]
        
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        
        # Gradient of score = 2n * (sg / S^2 - 2*M*w_mean/S^3)
        S2 = S_y * S_y
        S3 = S2 * S_y
        w_mean = w_sum / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * w_mean / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size with backtracking
        # Try to use (sc - sc_lb) as a guidance
        step_candidate = step
        while step_candidate > 1e-10:
            y_new = y - step_candidate * grad
            y_new = np.maximum(y_new, 0.0)
            S_new = float(y_new.sum())
            if S_new < 1e-12:
                step_candidate *= 0.5
                continue
            y_new = y_new / S_new
            sc_new = compute_score(y_new)
            if sc_new <= sc + 1e-12:
                break
            step_candidate *= 0.5
        
        step = step_candidate
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y_new if step_candidate > 0 else y
        a_new = y + ((t - 1.0) / t_new) * (y - a)
        a_new = np.maximum(a_new, 0.0)
        S_a = float(a_new.sum())
        if S_a > 1e-12:
            a_new = a_new / S_a
        a = a_new
        t = t_new
        
        # Decay momentum occasionally
        if it % 1000 == 0 and it > 0:
            momentum = max(0.5, momentum * 0.95)
    
    sc = compute_score(y)
    if sc < best_sc:
        best_a = y.copy()
    
    return best_a

def full_coord_descent(a, n_sweeps=5, step_factors=None):
    """
    Full coordinate descent: try perturbations on ALL coordinates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    if step_factors is None:
        step_factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7]
    
    for sweep in range(n_sweeps):
        cur_sc = compute_score(a)
        improved = False
        
        # Process coordinates in random order
        order = np.arange(n)
        np.random.shuffle(order)
        
        for i in order:
            a_old = a[i]
            if a_old < 1e-15:
                # Try to increase from zero
                candidates = [0.01 / n, 0.05 / n, 0.1 / n]
            else:
                candidates = [a_old * f for f in step_factors]
            
            best_val = a_old
            best_local = cur_sc
            
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old) < 1e-15:
                    continue
                a[i] = cv
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[i] = a_old
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[i] = a_old
            
            if best_val != a_old:
                a[i] = best_val
                improved = True
                cur_sc = best_local
        
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def lp_direction(a, M, top_k_constraints=50):
    """
    Find a direction g that maximizes sum(g) subject to (a*g)[l] <= M for top-K constraints.
    Uses scipy linprog.
    """
    n = len(a)
    ag = fast_cross_conv(a, a)  # placeholder, we need a*g not a*a
    # Actually we need the constraint matrix
    # (a*g)[l] = sum_i a[l-i]*g[i] for valid i
    # We want: (a*g)[l] <= M for the top-K most binding l
    
    # First, find which lags are most "binding" - use a uniform g=1/n as reference
    g_ref = np.ones(n) / n
    ag_ref = fast_cross_conv(a, g_ref)
    
    # Most binding: those with highest ag_ref / M
    binding_ratio = ag_ref / M
    top_lags = np.argsort(binding_ratio)[::-1][:top_k_constraints]
    
    # Build constraint matrix A_ub for linprog
    # For each lag l in top_lags: sum_i a[l-i]*g[i] <= M
    # Variables: g[0], ..., g[n-1]
    
    A_ub_rows = []
    b_ub = []
    
    for l in top_lags:
        row = np.zeros(n)
        i_start = max(0, l - n + 1)
        i_end = min(l, n - 1)
        for i in range(i_start, i_end + 1):
            row[i] = a[l - i]
        A_ub_rows.append(row)
        b_ub.append(M)
    
    A_ub = np.array(A_ub_rows)
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    bounds = [(0, 1000)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return np.ones(n) / n

def lse_lbfgs(a, beta, maxiter=5000):
    """L-BFGS on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    
    def obj_grad(x):
        c = fast_conv(x)
        S_x = float(x.sum())
        if S_x < 1e-12:
            return 1e10, np.zeros(n)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_scaled = np.exp(scaled)
        log_sum = float(np.log(exp_scaled.sum()))
        L = c_max + log_sum / beta
        score = 2.0 * n * L / (S_x * S_x)
        p = exp_scaled / exp_scaled.sum()
        
        # Gradient of L w.r.t. x via cross-conv with p
        a_rev = x[::-1]
        PA = np.fft.rfft(p, M2_fft)
        RA = np.fft.rfft(a_rev, M2_fft)
        conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
        g = 2.0 * conv_res[n - 1:2 * n - 1]
        
        S2 = S_x * S_x
        S3 = S2 * S_x
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
        return score, grad
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(obj_grad, a, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxiter * 20})
    
    a_out = np.clip(res.x, 0, 1000)
    S_out = float(a_out.sum())
    if S_out < 1e-12:
        a_out = a
    else:
        a_out = a_out / S_out
    return compute_score(a_out), a_out

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2.
New techniques: LP-based direction finding (HiGHS), 
Nesterov momentum subgradient, adaptive LSE L-BFGS, 
full coordinate descent, multi-target Fienup.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy import sparse
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_corr(a, p):
    """Compute (a * p_corr)[i] = sum_m a[m]*p[m+i] for i=0,...,2n-2 via FFT."""
    n = len(a)
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    a_rev = a[::-1]
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    return conv_res[n - 1: 2 * n - 1]  # This gives (a ⊛ p)[0:n]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    return a / S


def lse_obj_and_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    corr = fast_cross_corr(x, p)
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n * corr) / S2 - (4.0 * n * L) / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000):
    n = len(a0)
    a0 = normalize_nonneg(a0.astype(np.float64).copy())
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_obj_and_grad, a0, args=(beta,),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxiter * 20,
                            'maxls': 50})
    a_out = normalize_nonneg(res.x)
    return compute_score(a_out), a_out


def run_fista(a0, beta, n_iters=5000):
    n = len(a0)
    x0 = normalize_nonneg(a0.astype(np.float64).copy())
    x = x0.copy()
    z = x0.copy()
    t_k = 1.0
    
    f_y, g_y = lse_obj_and_grad(z, beta)
    gn = float(np.linalg.norm(g_y))
    L = max(gn + 1.0, 10.0)
    
    best_x = x0.copy()
    best_sc = compute_score(x0)
    
    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Key improvements: LP-based direction finding (HiGHS with top-K binding constraints),
Nesterov momentum subgradient with backtracking, full coordinate descent with FFT
score computation, multi-target Fienup, and adaptive LSE L-BFGS/FISTA.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy import sparse
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_corr(a, p):
    """Compute (a cross-corr p)[i] = sum_m a[m]*p[m+i] for i=0..n-1 via FFT."""
    n = len(a)
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    a_rev = a[::-1]
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    return conv_res[n - 1:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    return a / S


def lse_obj_and_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    corr = fast_cross_corr(x, p)
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n * corr) / S2 - (4.0 * n * L) / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000):
    n = len(a0)
    a0 = normalize_nonneg(a0.astype(np.float64).copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(lse_obj_and_grad, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxiter * 20,
                                'maxls': 50})
        a_out = normalize_nonneg(res.x)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def run_fista(a0, beta, n_iters=5000):
    n = len(a0)
    x0 = normalize_nonneg(a0.astype(np.float64).copy())
    x = x0.copy()
    z = x0.copy()
    t_k = 1.0
    f_y, g_y = lse_obj_and_grad(z, beta)
    gn = float(np.linalg.norm(g_y))
    L = max(gn + 1.0, 10.0)
    best_x = x0.copy()
    best_sc = compute_score(x0)
    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()
        f_y, g_y = lse_obj_and_grad(z, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)
        for _ in range(50):
            x_new = y - g_y / L
            x_new = normalize_nonneg(x_new)
            f_new, _ = lse_obj_and_grad(x_new, beta)
            quad = 0.5 * L * float(np.sum((x_new - y) ** 2))
            if f_new <= f_y + float(g_y @ (x_new - y)) + quad + 1e-12:
                break
            L *= 1.5
        else:
            break
        L = max(0.9 * L, 1.0)
        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = float(z.sum())
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new
    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()
    return best_x


def fienup_multi_target(a, n_iters=300, n_targets=4):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64).copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    iters_per_target = max(50, n_iters // n_targets)
    for target_idx in range(n_targets):
        progress = target_idx / max(1, n_targets)
        for it in range(iters_per_target):
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P = np.maximum(P, 1e-30)
            P_mean = float(P.mean())
            if target_idx == 0:
                P_target = np.full_like(P, P_mean)
            elif target_idx == 1:
                freq = np.arange(len(P)) / len(P)
                P_target = P_mean * (1.0 + 0.3 * np.exp(-((freq - 0.2) ** 2) / 0.01))
            elif target_idx == 2:
                freq = np.arange(len(P)) / len(P)
                P_target = P_mean * (1.0 + 0.2 * ((freq - 0.5) ** 2))
            else:
                kernel_size = max(2, len(P) // 20)
                kernel = np.ones(kernel_size) / kernel_size
                P_target = np.convolve(P, kernel, mode='same')
                P_target = np.maximum(P_target, 1e-30)
            alpha = 0.3 + 0.4 * (it / max(1, iters_per_target))
            P_blend = (1.0 - alpha) * P + alpha * P_target
            P_blend = np.maximum(P_blend, 1e-30)
            A_new = A * (np.sqrt(P_blend / P) ** alpha)
            a_new = np.fft.irfft(A_new, M_fft)[:n]
            a_new = normalize_nonneg(a_new)
            a = a_new
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a


def momentum_subgradient(a, n_iters=5000, top_k=10):
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    y = a.copy()
    prev = a.copy()
    t = 1.0
    decay = 0.5
    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(y)
            if sc < best_sc:
                best_sc = sc
                best_a = y.copy()
        c = fast_conv(y)
        S = float(y.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        agg = y.copy()
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            weights[i] = c[k]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                agg[j_start:j_end + 1] += weights[i] * y[j_start:j_end + 1][::-1]
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            break
        agg = agg / w_sum
        S2 = S * S
        S3 = S2 * S
        scale = 2.0 * M_val / S2 - 4.0 * M_val / S3
        grad = 2.0 * n * agg * scale
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        decay = 0.5 + 0.5 * (1 - it / n_iters)
        step = (sc - 1.0001) / (gn * gn + 1e-30) if sc > 1.0001 else 0.005
        step = min(step, 0.03 * decay)
        step = max(step, 1e-7)
        y_new = y - step * grad
        y_new = normalize_nonneg(y_new)
        sc_new = compute_score(y_new)
        if sc_new > sc + 1e-10:
            step = step * 0.5
            y_new = y - step * grad
            y_new = normalize_nonneg(y_new)
            sc_new = compute_score(y_new)
        if step > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            z = y_new + ((t - 1.0) / t_new) * (y_new - prev)
            z = normalize_nonneg(z)
            prev = y
            y = z
            t = t_new
    sc = compute_score(y)
    if sc < best_sc:
        best_a = y.copy()
    return best_a


def full_coord_descent(a, n_sweeps=3, factors=None):
    n = len(a)
    if factors is None:
        factors = [0.5, 0.75, 0.9, 1.1, 1.3, 1.6]
    a = normalize_nonneg(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)
    for sweep in range(n_sweeps):
        cur_sc = compute_score(a)
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            a_old = a[i]
            if a_old < 1e-15:
                candidates = [1.0 / n, 2.0 / n, 5.0 / n]
            else:
                candidates = [a_old * f for f in factors]
            best_val = a_old
            best_local = cur_sc
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old) < 1e-14:
                    continue
                a[i] = cv
                sc_trial = compute_score(a)
                if sc_trial < best_local:
                    best_local = sc_trial
                    best_val = cv
                a[i] = a_old
            if best_val != a_old:
                a[i] = best_val
                improved = True
                cur_sc = best_local
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved:
            break
    return best_a


def lp_direction(a, M, k_constraints=200):
    n = len(a)
    ag = fast_conv(a)
    binding_ratio = ag / max(M, 1e-30)
    top_lags = np.argsort(binding_ratio)[::-1][:k_constraints]
    rows = []
    for l in top_lags:
        col_indices = []
        values = []
        i_start = max(0, int(l) - n + 1)
        i_end = min(int(l), n - 1)
        for i in range(i_start, i_end + 1):
            val = a[i]
            if val > 1e-15:
                col_indices.append(i)
                values.append(val)
        rows.append((int(l), col_indices, values))
    all_rows = []
    all_cols = []
    all_vals = []
    row_ptr = [0]
    for lag, cols, vals in rows:
        all_rows.extend(cols)
        all_vals.extend(vals)
        all_cols.append(lag)
        row_ptr.append(len(all_rows))
    A_sp = sparse.csc_matrix((all_vals, (all_rows, all_cols)), shape=(n, k_constraints))
    c_obj = -np.ones(n)
    b_ub = np.full(k_constraints, M)
    bounds = [(0, 1000)] * n
    try:
        res = linprog(c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return np.ones(n) / n


def greedy_direction(a, M):
    n = len(a)
    g = np.zeros(n)
    c = np.zeros(2 * n - 1)
    for i in range(n):
        l_lo = i
        l_hi = min(i + n - 1, 2 * n - 2)
        m_max = l_hi - i
        a_seg = a[0:m_max + 1]
        c_seg = c[l_lo:l_hi + 1]
        pos = a_seg > 1e-15
        if pos.any():
            max_val = float(np.min((M - c_seg[pos]) / a_seg[pos]))
        else:
            max_val = 1000.0
        g[i] = max(0.0, min(max_val, 1000.0))
        c[l_lo:l_hi + 1] += a_seg * g[i]
    return g


def line_search_blend(a, g, sc_a):
    best_sc = sc_a
    best_t = 0.0
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        a_try = (1.0 - t) * a + t * g
        a_try = normalize_nonneg(a_try)
        sc_try = compute_score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_t = t
    if best_t == 0.0:
        return a.copy()
    return (1.0 - best_t) * a + best_t * g


def greedy_lp_loop(a, max_iters=50):
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    for it in range(max_iters):

        M = float(c.max())
        g = greedy_direction(a, M)
        S_g = float(g.sum())
        if S_g < 1e-12:

        g = g / S_g
        a_new = line_search_blend(a, g, best_sc)
        sc_new = compute_score(a_new)
        if sc_new < best_sc - 1e-10:
            a = a_new
            best_sc = sc_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:

def structured_perturb(a, rng, n_modes=8, scale=0.03):
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = int(rng.integers(1, 15))
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize_nonneg(a_new)


def noise_perturb(a, rng, scale=0.02):
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    return normalize_nonneg(a_new)


def interpolate_to_length(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

        if time.monotonic() > phase4_end - 30:
            break
        remaining = phase4_end - time.monotonic() - 10
        if remaining < 30:
            break
        maxiter = int(remaining * 10)
        maxiter = max(1000, min(maxiter, 30000))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter)
        update(sc, a_opt, f"LBFGS b={beta}")
        if time.monotonic() < phase4_end - 60:
            a_fi = run_fista(best_a, beta, n_iters=min(3000, int(remaining * 5)))
            sc = compute_score(a_fi)
            update(sc, a_fi, f"FISTA b={beta}")

    # === Phase 5: Coordinate Descent (10% time) ===
    print(f"\n=== Phase 5: Coordinate Descent ===", flush=True)
    if time.monotonic() < deadline - 30:
        n = len(best_a)
        est_per_sweep = n * 6 * 8e-5  # rough estimate
        possible_sweeps = int((deadline - time.monotonic() - 10) / max(1.0, est_per_sweep))
        possible_sweeps = max(1, min(possible_sweeps, 4))
        print(f"  n={n}, est_per_sweep={est_per_sweep:.2f}s, sweeps={possible_sweeps}", flush=True)
        a_cd = full_coord_descent(best_a, n_sweeps=possible_sweeps)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")

    # === Phase 6: Restarts with perturbation ===
    if time.monotonic() < deadline - 20:
        for r in range(4):
            if time.monotonic() > deadline - 15:
                break
            if r % 2 == 0:
                a_pert = structured_perturb(best_a, rng, n_modes=6, scale=0.015 + 0.005 * r)
            else:
                a_pert = noise_perturb(best_a, rng, scale=0.01 + 0.003 * r)
            a_pert2 = greedy_lp_loop(a_pert, max_iters=5)
            sc = compute_score(a_pert2)
            update(sc, a_pert2, f"RESTART-{r}")

    # === Final Polish ===
    if time.monotonic() < deadline - 10:
        sc, a_final = run_lbfgs(best_a, 10000, maxiter=5000)
        update(sc, a_final, "FINAL-POLISH")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return out
