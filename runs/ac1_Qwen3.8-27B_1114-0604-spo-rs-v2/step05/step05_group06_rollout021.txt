"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-scale optimization with
LSE smoothing, Nesterov-accelerated subgradient descent, and spectral
diversification. Uses parent sequence as warm start with frequency-domain
perturbations for diversity.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """Compute cross-correlation h[j] = sum_k w[k] * a[k-j] for j=0..n-1."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _downsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    if m < n:
        return _upsample(a, n)
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed max of d = n*conv(a,a)/S^2, using top-k positions."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _lbfgs_optimize(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on LSE objective."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _nesterov_subgrad(a, initial_lr=0.01, momentum=0.9, max_iters=500, deadline=None,
                      tol=1e-10, patience=50):
    """Nesterov-accelerated subgradient descent on the max-convolution objective."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    v = np.zeros(n)
    lr = initial_lr
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break

        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.5% of max
        threshold = dmax * 0.995
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < tol:
            break

        # Nesterov momentum
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-8:
            a_new = a
        else:
            a_new = a_new / S_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Adaptive learning rate: reduce if no improvement
            if no_improve % 10 == 0:
                lr *= 0.8
            if no_improve > patience and lr < 1e-10:
                break

        a = a_new

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence from flat-ish power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _freq_perturb(a, rng, n_targets=None, strength=0.2):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    Nf = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, Nf)
    # Add random phase perturbation to non-DC components
    pert = np.zeros_like(A)
    mask = np.ones(len(A), dtype=bool)
    mask[0] = False  # Don't perturb DC
    # Only perturb a fraction of frequencies
    if n_targets is not None:
        idx = rng.choice(len(A) - 1, size=min(n_targets, len(A)-1), replace=False) + 1
        mask[:len(A)-1][idx] = False  # Only modify selected
    mag = rng.exponential(strength / len(A), size=A.shape)
    phase = rng.uniform(-np.pi, np.pi, size=A.shape)
    pert[mask] = mag[mask] * np.exp(1j * phase[mask])
    A_new = A + pert
    sig = np.fft.irfft(A_new, Nf)[:n]
    sig = np.clip(sig, 0, None)
    if sig.sum() < 1e-8:
        return _norm(a)
    return _norm(sig)


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_with_trajectory(a0, deadline, label, rng):
    """Progressive LSE optimization followed by Nesterov subgradient."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Phase 1: Progressive LSE via L-BFGS-B
    total_time = max(10.0, deadline - t0 - 60.0)
    stages = [
        (3, 30.0, 0.08),
        (5, 60.0, 0.10),
        (10, 150.0, 0.12),
        (20, 400.0, 0.14),
        (50, 1000.0, 0.16),
        (100, 3000.0, 0.18),
        (200, 8000.0, 0.18),
        (400, 20000.0, 0.14),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + cum_time * total_time
        stage_end = min(stage_end, deadline - 60.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        n_calls = min(3, max(1, int(stage_budget / 15)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _lbfgs_optimize(a, beta, k, per_call, 500)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if ci % 2 == 0:
                    print(f"    [{label}] K={k}, beta={beta:.0f}: {best_sc:.8f}")
            a = _norm(a_new)

    # Phase 2: Nesterov subgradient
    if time.monotonic() < deadline - 30.0:
        sc, a_sg = _nesterov_subgrad(a, initial_lr=0.005, momentum=0.95,
                                     max_iters=200, deadline=deadline - 20.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] nesterov: {best_sc:.8f}")
        a = _norm(a_sg if sc < best_sc else a)

    # Phase 3: Final high-beta LSE
    if time.monotonic() < deadline - 20.0:
        for k_f, beta_f in [(500, 30000), (1000, 60000)]:
            if time.monotonic() > deadline - 20.0:
                break
            remaining = deadline - 20.0 - time.monotonic()
            if remaining < 5.0:
                break
            sc_f, a_f = _lbfgs_optimize(a, beta_f, k_f, remaining * 0.5, 400)
            if sc_f < best_sc:
                best_sc = sc_f
                best_a = a_f.copy()
                print(f"    [{label}] final LSE K={k_f}: {best_sc:.8f}")
            a = _norm(a_f)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-scale optimization with
LSE smoothing, Nesterov-accelerated subgradient descent, and frequency-domain
diversification. Explores multiple sequence lengths (up to 16384) with
progressive active-set growth and aggressive final polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    """LSE-smoothed 2*max(d) using top-k positions of d = n*conv(a,a)/S^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _lbfgs_optimize(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on LSE objective with sum penalty."""
    a = _norm(a0)
    n = len(a)
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 10000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _nesterov_subgrad(a, initial_lr=0.01, momentum=0.95, max_iters=500,
                      deadline=None, tol=1e-12, patience=80):
    """Nesterov-accelerated subgradient descent on max-convolution objective."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    v = np.zeros(n)
    lr = initial_lr
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break

        c = _self_conv(a)
        d = float(n) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.3% of max
        threshold = dmax * 0.997
        active = d >= threshold
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < tol:
            break

        # Nesterov momentum
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-8:
            a_new = a
        else:
            a_new = a_new / S_new

        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > patience and lr < 1e-12:
                break

        a = a_new

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate non-negative sequence from shaped power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _freq_perturb(a, rng, strength=0.2, n_freqs=None):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    Nf = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, Nf)
    pert = np.zeros_like(A)
    if n_freqs is None:
        n_freqs = max(10, n // 4)
    n_freqs = min(n_freqs, len(A) - 1)
    idx = rng.choice(len(A) - 1, size=n_freqs, replace=False) + 1
    mag = rng.exponential(strength / n_freqs, size=n_freqs)
    phase = rng.uniform(-np.pi, np.pi, size=n_freqs)
    pert[idx] = mag * np.exp(1j * phase)
    A_new = A + pert
    sig = np.fft.irfft(A_new, Nf)[:n]
    sig = np.clip(sig, 0, None)
    if sig.sum() < 1e-8:
        return _norm(a)
    return _norm(sig)


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_trajectory(a0, deadline, label, rng):
    """Progressive LSE optimization followed by Nesterov subgradient and final polish."""
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Phase 1: Progressive LSE via L-BFGS-B
    total_time = max(10.0, deadline - t0 - 120.0)
    stages = [
        (3, 30.0, 0.07),
        (5, 60.0, 0.09),
        (10, 150.0, 0.11),
        (20, 400.0, 0.13),
        (50, 1000.0, 0.15),
        (100, 3000.0, 0.16),
        (200, 8000.0, 0.17),
        (400, 20000.0, 0.17),
        (800, 50000.0, 0.15),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + cum_time * total_time
        stage_end = min(stage_end, deadline - 120.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        n_calls = min(4, max(1, int(stage_budget / 15)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _lbfgs_optimize(a, beta, k, per_call, 600)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if cum_time < 0.5:
                    print(f"    [{label}] K={k}, beta={beta:.0f}: {best_sc:.8f}")
            a = _norm(a_new)

    # Phase 2: Nesterov subgradient (multiple rounds with decaying LR)
    if time.monotonic() < deadline - 60.0:
        for round_n in range(3):
            if time.monotonic() > deadline - 60.0:
                break
            lr0 = 0.003 / (round_n + 1)
            sc, a_sg = _nesterov_subgrad(a, initial_lr=lr0, momentum=0.95,
                                         max_iters=300, deadline=deadline - 60.0)
            if sc < best_sc:
                best_sc = sc
                best_a = a_sg.copy()
                print(f"    [{label}] nesterov r{round_n}: {best_sc:.8f}")
            a = _norm(a_sg if sc < best_sc else a)

    # Phase 3: Final high-beta LSE polish
    if time.monotonic() < deadline - 30.0:
        for k_f, beta_f in [(800, 50000), (1500, 100000)]:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - 30.0 - time.monotonic()
            if remaining < 5.0:
                break
            sc_f, a_f = _lbfgs_optimize(a, beta_f, k_f, remaining * 0.45, 500)
            if sc_f < best_sc:
                best_sc = sc_f
                best_a = a_f.copy()
                print(f"    [{label}] final LSE K={k_f}: {best_sc:.8f}")
            a = _norm(a_f)

    # Phase 4: Final Nesterov
    if time.monotonic() < deadline - 10.0:
        sc_sg2, a_sg2 = _nesterov_subgrad(a, initial_lr=0.001, momentum=0.95,
                                          max_iters=200, deadline=deadline - 5.0)
        if sc_sg2 < best_sc:
            best_sc = sc_sg2
            best_a = a_sg2.copy()
            print(f"    [{label}] final nesterov: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def _multilevel_warmstart(a0, n_target, deadline, label, rng):
    """Optimize at n//4 then upscale to n_target."""
    n = len(a0)
    a = _norm(a0)

    if n > 4096 and deadline - time.monotonic() > 90:
        n_small = max(1024, n // 4)
        a_small = _upsample(a, n_small)
        sc_small = _score(a_small)
        print(f"  [{label}] multilevel: n_small={n_small}, score={sc_small:.8f}")

        t_budget = min(40.0, (deadline - time.monotonic()) * 0.12)
        t_smal_end = time.monotonic() + t_budget
        for k_s, beta_s in [(5, 100), (20, 500), (60, 2000)]:
            if time.monotonic() > t_smal_end:
                break
            per_call = max(3.0, (t_smal_end - time.monotonic()) / 2.0)
            sc_s, a_s = _lbfgs_optimize(a_small, beta_s, k_s, per_call, 200)
            if sc_s < sc_small:
                sc_small = sc_s
                a_small = a_s

        a_up = _norm(_upsample(a_small, n))
        sc_up = _score(a_up)
        if sc_up < _score(a):
            a = a_up
            print(f"  [{label}] multilevel improved: {sc_up:.8f}")

    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-scale search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses progressive LSE smoothing, Nesterov-accelerated subgradient,
    frequency-domain diversification, and multi-level warm starts.
    Explores lengths up to 16384 for improved solution quality.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = _norm(p)
    except Exception:
        pass

    print(f"Parent loaded: {parent is not None}, "
          f"parent_len={len(parent) if parent is not None else 0}, "
          f"budget={budget_s}s")

    if parent is not None:
        parent_score = _score(parent)
        print(f"Parent score: {parent_score:.8f}")

    # Target lengths to explore (ascending)
    target_lengths = []
    for L in [8192, 12288, 16384, 20480]:
        if parent is not None and L < len(parent) * 0.85:
            continue
        target_lengths.append(L)

    if not target_lengths:
        target_lengths = [max(4096, len(parent) if parent is not None else 4096)]

    print(f"Target lengths: {target_lengths}")

    best_score = np.inf
    best_seq = np.ones(1024) / 1024

    # Time allocation: give progressively more time to longer sequences
    n_lengths = len(target_lengths)
    if n_lengths == 1:
        fractions = [1.0]
    elif n_lengths == 2:
        fractions = [0.35, 0.65]
    elif n_lengths == 3:
        fractions = [0.20, 0.30, 0.50]
    else:
        fractions = [0.12, 0.20, 0.30, 0.38]

    cum_frac = 0.0
    for li, (target_n, frac) in enumerate(zip(target_lengths, fractions)):
        sub_start = t_start + cum_frac * (budget_s - 70.0)
        sub_deadline = t_start + (cum_frac + frac) * (budget_s - 70.0)
        sub_deadline = min(sub_deadline, deadline - 20.0)
        cum_frac += frac

        if time.monotonic() > sub_deadline:
            continue

        print(f"\n{'='*60}")
        print(f"Length {li+1}/{n_lengths}: n={target_n}, "
              f"budget={frac:.2f}, deadline at {sub_deadline - t_start:.0f}s")

        # Generate diverse initializations
        inits = []

        # 1. Parent upsampled (forward and reverse)
        if parent is not None:
            inits.append(('parent', _upsample(parent, target_n)))
            inits.append(('parent_rev', _upsample(parent[::-1], target_n)))

        # 2. Frequency-perturbed parent variants
        if parent is not None:
            p_up = _upsample(parent, target_n)
            for sp in (0.05, 0.15, 0.3, 0.5):
                inits.append((f'fpert_{sp}', _freq_perturb(p_up, rng, strength=sp)))

        # 3. Spectral inits with varied exponents
        for i in range(4):
            inits.append((f'spec{i}', _spectral_init(target_n, rng)))

        # 4. Cosine modulations
        x = np.linspace(0, 1, target_n)
        for f in (1, 2, 3, 5):
            for amp in (0.2, 0.4, 0.7):
                phase_off = rng.uniform(0, 2 * np.pi)
                a = 1.0 + amp * np.cos(2 * np.pi * f * x + phase_off)
                a = np.clip(a, 0, None)
                if a.sum() > 0:
                    inits.append((f'cos{f}_{amp}', a / a.sum()))

        # 5. Smoothed exponential noise
        for i in range(2):
            noise = rng.standard_normal(target_n)
            k = max(7, target_n // 80)
            if k % 2 == 0:
                k += 1
            kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4) ** 2))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode='same')
            a = np.exp(noise)
            inits.append((f'expnoise{i}', a / a.sum()))

        # Score and rank
        scored = []
        for name, arr in inits:
            arr = _norm(arr)
            sc = _score(arr)
            if np.isfinite(sc):
                scored.append((sc, name, arr))

        scored.sort(key=lambda x: x[0])
        top_str = ", ".join([f"{nm}:{s:.6f}" for s, nm, _ in scored[:6]])
        print(f"  Inits ({len(scored)}): {top_str}")

        # Select top 3 candidates
        candidates = []
        for sc, name, arr in scored:
            if len(candidates) >= 3:
                break
            candidates.append((sc, name, arr))

        if not candidates:
            candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

        # Allocate time among candidates
        n_cands = len(candidates)
        if n_cands == 1:
            cand_fracs = [1.0]
        elif n_cands == 2:
            cand_fracs = [0.6, 0.4]
        else:
            cand_fracs = [0.5, 0.3, 0.2]

        cum_cf = 0.0
        for ci, (sc_init, name, arr_init) in enumerate(candidates):
            cand_deadline = sub_start + (cum_cf + cand_fracs[ci]) * (sub_deadline - sub_start - 5.0)
            cand_deadline = min(cand_deadline, sub_deadline - 5.0)
            cum_cf += cand_fracs[ci]

            if time.monotonic() > cand_deadline:
                continue

            print(f"\n  --- Cand {ci+1}/{''.join([cand_fracs and ''])} '{name}', "
                  f"init={sc_init:.8f}, time={cand_deadline - time.monotonic():.0f}s ---")

            a_work = _norm(arr_init)

            # Multilevel warm start
            a_work = _multilevel_warmstart(a_work, target_n, cand_deadline, name, rng)

            # Main trajectory optimization
            sc_traj, seq_traj = _optimize_trajectory(a_work, cand_deadline, name, rng)
            if sc_traj < best_score:
                best_score = sc_traj
                best_seq = seq_traj.copy()
                print(f"  *** GLOBAL BEST: {best_score:.8f} (n={target_n}, {name}) ***")

    # === Final global polish ===
    print(f"\n{'='*60}")
    print(f"=== FINAL POLISH (score={best_score:.8f}) ===")
    final_deadline = deadline - 10.0

    if time.monotonic() < final_deadline - 40.0:
        # Ultra-high beta LSE
        for k_f, beta_f in [(1000, 60000), (2000, 120000)]:
            if time.monotonic() > final_deadline - 40.0:
                break
            remaining = final_deadline - time.monotonic() - 40
            if remaining < 5:
                break
            sc_f, a_f = _lbfgs_optimize(best_seq, beta_f, k_f, remaining * 0.4, 500)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final LSE K={k_f}: {best_score:.8f}")

    # Nesterov rounds
    if time.monotonic() < final_deadline - 20.0:
        for rnd in range(3):
            if time.monotonic() > final_deadline - 20.0:
                break
            lr0 = 0.002 / (rnd + 1)
            sc_sg, a_sg = _nesterov_subgrad(best_seq, initial_lr=lr0, momentum=0.95,
                                            max_iters=300, deadline=final_deadline - 20.0)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  Final nesterov r{rnd}: {best_score:.8f}")

    # Tiny perturbations + quick optimize
    if time.monotonic() < final_deadline - 15.0:
        for attempt in range(5):
            if time.monotonic() > final_deadline - 15.0:
                break
            p = _freq_perturb(best_seq, rng, strength=0.03)
            t_rem = final_deadline - time.monotonic() - 15
            if t_rem < 3:
                break
            sc_r, a_r = _lbfgs_optimize(p, 50000, 800, t_rem * 0.3, 200)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Perturb+opt {attempt}: {best_score:.8f}")

    # Very final Nesterov
    if time.monotonic() < final_deadline - 3.0:
        sc_last, a_last = _nesterov_subgrad(best_seq, initial_lr=0.0005, momentum=0.95,
                                            max_iters=100, deadline=final_deadline)
        if sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
            print(f"  Very final: {best_score:.8f}")

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
