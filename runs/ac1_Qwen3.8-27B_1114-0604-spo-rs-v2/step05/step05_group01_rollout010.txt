def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / S


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def lse_objective_grad(a, beta, top_k=None):
    """Compute LSE objective and gradient. Uses top-k lags if specified for sharper gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    M = len(d)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    if top_k is not None and top_k < M:
        # Use only top-k lags for the LSE
        idx = np.argpartition(d, M - top_k)[M - top_k:]
        d_sub = d[idx]
        dmax_sub = float(np.max(d_sub))
        z = beta * (d_sub - dmax_sub)
        z -= np.max(z)  # numerical stability
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0 or not np.isfinite(sum_e):
            return 1e10, np.zeros(n, dtype=np.float64)
        w_sub = e / sum_e
        L = dmax_sub + np.log(sum_e) / beta
        # Build full w vector
        w = np.zeros(M, dtype=np.float64)
        w[idx] = w_sub
    else:
        z = beta * (d - dmax)
        z -= np.max(z)
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0 or not np.isfinite(sum_e):
            return 1e10, np.zeros(n, dtype=np.float64)
        w = e / sum_e
        L = dmax + np.log(sum_e) / beta

    h = corr_pair(a, w)
    grad = 2.0 * h / (S * S) * float(n)
    # Project gradient to tangent space (sum=0) for the simplex constraint
    grad = grad - np.mean(grad)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e4, 1e4)
    return float(2.0 * L), grad


def make_lbfgs_fun(beta, top_k):
    def fun(x):
        x_clip = np.maximum(np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
        S = float(x_clip.sum())
        if S < 1e-12:
            return 1e10, np.zeros_like(x)
        x_norm = x_clip / S
        val, g = lse_objective_grad(x_norm, beta, top_k)
        # Transform gradient back to x-space (accounting for normalization)
        g = g - x_norm * np.dot(g, x_norm)  # project to tangent of normalization
        return val, g
    return fun


def run_lbfgs(af, beta, deadline, maxfun, top_k=None):
    a = normalize_a(af)
    n = len(a)
    if time.monotonic() > deadline - 10:
        return exact_score(a), a

    fun = make_lbfgs_fun(beta, top_k)
    bounds = [(0.0, 10.0)] * n
    maxiter = max(30, maxfun // 2)
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def peeling_step(a, eta, k_active=3):
    """Multiplicative update targeting the top-k_active autocorrelation lags."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    M = len(d)
    dmax = float(np.max(d))
    if dmax <= 0:
        return a

    # Get top-k_active lags
    if k_active >= M:
        w = np.ones(M, dtype=np.float64) / M
    else:
        idx = np.argpartition(d, M - k_active)[M - k_active:]
        w = np.zeros(M, dtype=np.float64)
        w[idx] = 1.0 / k_active

    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a
    log_a = np.log(a + 1e-300)
    log_a = log_a - eta * h / hmax
    a_new = np.exp(log_a - np.max(log_a))
    return normalize_a(a_new)


def peeling_schedule(af, deadline, max_iter=100, base_eta=0.3):
    """Run peeling with decreasing step size."""
    a = normalize_a(af)
    best_a = a.copy()
    best_sc = exact_score(a)
    no_improve = 0

    for it in range(int(max_iter)):
        if time.monotonic() > deadline:
            break
        # Adaptive eta: decrease when no improvement
        eta = base_eta * (0.95 ** no_improve)
        if eta < 0.001:
            break

        for k_act in (1, 2, 3, 5):
            a_try = peeling_step(a, eta, k_act)
            sc = exact_score(a_try)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
                no_improve = 0
                break
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return float(best_sc), best_a


def coordinate_descent(af, deadline, max_sweeps=5):
    """Try reducing individual elements that contribute to peak lags."""
    a = normalize_a(af)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for sweep in range(int(max_sweeps)):
        if time.monotonic() > deadline:
            break
        c = fast_self_conv(a)
        d = float(n) * c / (float(a.sum()) ** 2)
        dmax = float(np.max(d))
        active_idx = np.where(d > 0.95 * dmax)[0]

        # For each active lag, find the most contributing pair
        for k_act in active_idx:
            if time.monotonic() > deadline:
                break
            # Find j that maximizes a[j]*a[k_act-j]
            j_max = 0
            max_prod = 0
            for j in range(int(n)):
                k_j = int(k_act) - j
                if 0 <= k_j < int(n):
                    prod = float(a[j]) * float(a[k_j])
                    if prod > max_prod:
                        max_prod = prod
                        j_max = j

            if max_prod < 1e-15:
                continue

            # Try reducing a[j_max] or a[k_act - j_max]
            k_j = int(k_act) - j_max
            for frac in (0.95, 0.90, 0.80, 0.70):
                a_try = a.copy()
                # Reduce the larger of the pair
                if a_try[j_max] >= a_try[k_j]:
                    a_try[j_max] *= frac
                    a_try[k_j] += (a[j_max] * (1 - frac)) * 0.5  # redistribute half
                else:
                    a_try[k_j] *= frac
                    a_try[j_max] += (a[k_j] * (1 - frac)) * 0.5
                a_try = normalize_a(a_try)
                sc = exact_score(a_try)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                break  # only try first successful reduction

    return float(best_sc), best_a


def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n_new), dtype=np.float64) / float(max(1, n_new))
    if m == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, a).astype(np.float64)

    # Power-law taper
    for p in (1.5, 2.0, 3.0):

        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    for _ in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 7))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.05, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns
    for nb in (8, 16, 32, 64):

        add(arr[:n])

def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        # Multiplicative Gaussian noise
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        # Reverse + noise
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Lag-targeted perturbation
        c = fast_self_conv(p)
        dmax = float(np.max(c))
        if dmax > 0:
            idx = np.argpartition(c, len(c) - 3)[len(c) - 3:]
            w = np.zeros(len(c), dtype=np.float64)
            w[idx] = 1.0 / 3.0
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-15:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed Gaussian noise in log-space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(5, n // 300))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise -= float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - np.max(logp))
    else:
        # V-shape blend
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0) ** 2
        q = p * vshape

    return normalize_a(np.maximum(q, 0.0))


def estimate_maxfun(t_left, n):
    t_left = max(1.0, float(t_left))
    if t_left < 3.0:
        return 32
    base_time = 0.015 * max(0.5, float(n) / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    mf = int((0.8 * t_left) / max(base_time, 1e-4))
    return max(32, min(2000, mf))


def multi_beta_optimize(a, deadline, rng):
    """Run L-BFGS-B with increasing beta schedule, interleaved with peeling."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    beta_schedule = [
        (200, 1.0),
        (500, 2.0),
        (1000, 3.0),
        (2000, 5.0),
        (5000, 8.0),
        (10000, 10.0),
        (20000, 12.0),
        (50000, 15.0),
        (100000, 20.0),
    ]

    t0 = time.monotonic()
    total_time = max(10.0, float(deadline - t0))

    cum_frac = 0.0
    total_frac = sum(f for _, f in beta_schedule)

    for beta, frac in beta_schedule:
        cum_frac += frac / total_frac
        sub_deadline = t0 + min(1.0, cum_frac) * total_time
        sub_deadline = min(sub_deadline, float(deadline - 5.0))

        # Run L-BFGS-B multiple times at this beta
        while time.monotonic() < sub_deadline - 3.0:
            rem = float(sub_deadline - time.monotonic())
            if rem < 3.0:
                break
            mf = estimate_maxfun(rem, len(a))
            if mf < 32:
                break
            # Vary top_k for diversity
            top_k = None
            if beta >= 5000:
                top_k = int(rng.choice([3, 5, 10]))
            sc, a2 = run_lbfgs(a, float(beta), sub_deadline, mf, top_k)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  LBFGS beta={:.0f} sc={:.8f}".format(beta, best_sc))
            a = normalize_a(a2)

        # Peeling between beta stages
        if time.monotonic() < sub_deadline - 5.0:
            sc_p, a_p = peeling_schedule(a, sub_deadline, max_iter=20, base_eta=0.2)
            if np.isfinite(sc_p) and sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()
                print("  Peeling after beta={:.0f} sc={:.8f}".format(beta, best_sc))
            a = normalize_a(a_p)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e3, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        pass

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    n = 10240

    # Generate initializations
    inits = generate_inits(n, rng, parent)
    scores = []
    for arr in inits:
        sc = exact_score(arr)
        scores.append(sc)

    order = np.argsort(np.array(scores))
    print("Best init: {:.8f} out of {} inits".format(float(scores[order[0]]), len(inits)))

    # Select top starts
    selected = []
    used_types = set()
    for idx in order:
        if len(selected) >= 4:
            break
        selected.append(inits[idx])

    # Always include parent if available
    if parent is not None:
        pr = normalize_a(resample_sequence(parent, n))
        # Check if already included (first init is parent)
        if pr is not inits[0]:
            selected.insert(0, pr)
        selected = selected[:4]

    # Evaluate initial best
    best_score = np.inf
    best_seq = None
    for s in selected:
        sc = exact_score(s)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Starting best: {:.8f}".format(best_score))

    # Phase 1: Multi-beta optimization on top starts (55% time)
    phase1_end = t_start + 0.55 * (total_budget - 25)
    n_starts = min(3, len(selected))
    time_per_start = (phase1_end - t_start) / n_starts

    for i in range(n_starts):
        start_i = t_start + i * time_per_start
        end_i = min(start_i + time_per_start + 10, phase1_end)
        if time.monotonic() > end_i - 30:
            break

        a_start = selected[i].copy()
        # Blend with current best for secondary starts
        if i > 0 and best_seq is not None:
            blend = 0.3 + 0.2 * i
            a_start = normalize_a((1.0 - blend) * a_start + blend * best_seq)

        sc, seq = multi_beta_optimize(a_start, end_i, rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Phase1 start{}: {:.8f}".format(i, best_score))

    print("After Phase 1: {:.8f}".format(best_score))

    # Phase 2: Aggressive peeling (15% time)
    phase2_end = t_start + 0.70 * (total_budget - 25)
    if time.monotonic() < phase2_end - 10:
        sc, seq = peeling_schedule(best_seq, phase2_end, max_iter=200, base_eta=0.15)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
        print("After Phase 2: {:.8f}".format(best_score))

        # Coordinate descent
        if time.monotonic() < phase2_end - 10:
            sc, seq = coordinate_descent(best_seq, phase2_end, max_sweeps=3)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("  Coord desc: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts (20% time)
    phase3_end = t_start + 0.90 * (total_budget - 25)
    n_restarts = 5
    time_per_restart = (phase3_end - phase2_end) / n_restarts

    for r in range(n_restarts):
        restart_start = phase2_end + r * time_per_restart
        restart_end = min(restart_start + time_per_restart + 5, phase3_end)
        if time.monotonic() > restart_end - 20:
            break

        # Generate diverse perturbation
        p = perturb_best(best_seq, rng)
        # Sometimes try a different type of perturbation
        if r % 2 == 1:
            p = perturb_best(best_seq, rng)

        # Quick multi-beta optimization
        sc, seq = multi_beta_optimize(p, restart_end, rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {}: {:.8f}".format(r, best_score))

    print("After Phase 3: {:.8f}".format(best_score))

    # Phase 4: Final polish (10% time)
    if time.monotonic() < deadline - 10:
        # High-beta L-BFGS-B
        rem = float(deadline - time.monotonic() - 5)
        if rem > 20:
            mf = estimate_maxfun(rem * 0.7, n)
            sc, seq = run_lbfgs(best_seq, 200000.0, deadline, mf, top_k=3)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final LBFGS: {:.8f}".format(best_score))

        # Final peeling
        if time.monotonic() < deadline - 10:
            sc, seq = peeling_schedule(best_seq, deadline, max_iter=50, base_eta=0.05)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final peel: {:.8f}".format(best_score))

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 10240

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
