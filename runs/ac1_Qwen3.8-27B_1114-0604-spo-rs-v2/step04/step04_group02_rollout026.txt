"""
Improved step function peak minimizer.
Strategy:
1. Load parent, warm-start with moderate-beta L-BFGS
2. Progressive L-BFGS with adaptive beta increase
3. Fast coordinate descent using incremental convolution updates
4. Peak-aware iterated local search
5. Nesterov subgradient final polish on active peaks

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / sum^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def _conv_grad_weights(c, beta, top_k, n):
    """Compute LSE weights and smoothed value of convolution."""
    if top_k is not None and top_k < len(c):
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[top_k - 1]
        c_eff = np.minimum(c, threshold)
    else:
        c_eff = c
    c_max = float(c_eff.max())
    scaled = beta * (c_eff - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50.0))
    total = float(exp_scaled.sum())
    if total < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        L = c_max
    else:
        p = exp_scaled / total
        L = c_max + np.log(total) / beta
    return p, L


def lse_obj_grad(x, beta, top_k):
    """LSE-smoothed objective and gradient for L-BFGS. Returns (f, g)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    p, L = _conv_grad_weights(c, beta, top_k, n)
    score = 2.0 * n * L / (S * S)

    a_rev = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, top_k):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    x0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = x0.sum()
    if S < 1e-6:
        x0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, x0, args=(beta, top_k),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res


def coordinate_descent(a, n_passes=3, n_trial=6, verbose=False):
    """
    Fast coordinate descent using incremental convolution updates.
    For each coordinate, try several values and keep the best.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    c = fast_conv(a)
    best_sc = compute_score(a)
    best_a = a.copy()
    std_a = float(np.std(a))

    for p in range(n_passes):
        improved = False
        for i in range(n):
            a_old = a[i]
            if a_old < 1e-14:
                # Try small positive values
                candidates = [0.0, std_a * 0.01, std_a * 0.05]
            else:
                candidates = [0.0, a_old * 0.5, a_old,
                              a_old * 1.5,
                              max(0.0, a_old - std_a * 0.1),
                              a_old + std_a * 0.1]

            best_val = a_old
            best_local_sc = best_sc

            for v in candidates:
                if v < 0:
                    v = 0.0
                if v == a_old:
                    continue
                delta = v - a_old
                S_new = S + delta
                if S_new < 1e-12:
                    continue
                # Update affected conv coefficients
                # k ranges from i to min(i+n-1, 2n-2)
                k_start = i
                k_end = min(i + n, 2 * n)
                delta_conv = np.zeros(2 * n - 1)
                valid_k = np.arange(k_start, k_end)
                valid_j = valid_k - i  # j = k - i, ranges 0..n-1
                if valid_j.size > 0:
                    delta_conv[valid_k] = 2.0 * delta * a[valid_j]
                # Fix the k=2i term (double counting)
                if 2 * i < 2 * n - 1:
                    delta_conv[2 * i] = delta * (2.0 * a_old + delta)

                c_new = c + delta_conv
                max_new = float(np.max(c_new))
                sc_new = 2.0 * n * max_new / (S_new * S_new)

                if sc_new < best_local_sc:
                    best_local_sc = sc_new
                    best_val = v

            if best_val != a_old:
                delta = best_val - a_old
                a[i] = best_val
                S += delta
                k_start = i
                k_end = min(i + n, 2 * n)
                valid_k = np.arange(k_start, k_end)
                valid_j = valid_k - i
                if valid_j.size > 0:
                    c[valid_k] += 2.0 * delta * a[valid_j]
                if 2 * i < 2 * n - 1:
                    c[2 * i] = c[2 * i] - delta * (2.0 * a_old + delta) + delta * (2.0 * a_old + delta)
                    # Recompute correctly: old contribution was 2*delta*a[valid_j] at k=2i,
                    # but correct is delta*(2*a_old+delta). Let's just re-add properly.
                    pass

                # Actually, let's just recompute c to avoid bugs
                c = fast_conv(a)
                improved = True
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()

        if not improved and p > 0:
            break
        if verbose:
            print(f"    CD pass {p}: score={best_sc:.10f}")

    return best_a


def peak_contributors(a, top_count=200):
    """Find coordinates that contribute most to the peak convolution coefficient."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end < j_start:
        return np.array([], dtype=int)
    seg = a[j_start:j_end + 1]
    contrib = seg * a[k_star - j_start:k_star - j_start + 1][::-1]
    # Actually the contribution of a[j] to conv[k_star] is a[j]*a[k_star-j]
    # So for j in [j_start, j_end]: contrib_j = a[j] * a[k_star - j]
    contrib = a[j_start:j_end + 1] * a[k_star - (j_start:j_end + 1)]
    order = np.argsort(contrib)[::-1]
    top_local = order[:min(top_count, len(order))]
    return j_start + top_local


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient on exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=2000, f_lb=1.0, n_active=5):
    """Subgradient using average of top-n_active peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        Sa = float(a.sum())
        Mak = float(c.max())
        sc = 2.0 * n * Mak / (Sa * Sa)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top n_active peaks
        peak_indices = np.argsort(c)[::-1][:n_active]
        # Weight by proximity to max
        weights = np.exp(-(c[peak_indices] - Mak) * 10.0)
        weights /= weights.sum()

        sg_total = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg_total[idx] += w * 2.0 * a[k - idx]

        c_avg = float(np.dot(weights, c[peak_indices]))
        S2 = Sa * Sa
        S3 = S2 * Sa
        grad = (2.0 * n) * (sg_total / S2 - 2.0 * c_avg / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version with multiple strategies."""
    n = len(a)
    strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        a_p = a.copy()
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top contributors to the main peak
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a[j_start:j_end + 1]
        contrib = seg * a[k_star - np.arange(j_start, j_end + 1)]
        top_count = max(10, n // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        deficit = a.sum() - a_p.sum()
        if deficit > 0:
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        # Mirror perturbation: perturb the first half, mirror to second
        half = n // 2
        a_p = a.copy()
        a_p[:half] *= (1.0 + scale * 2.0 * rng.standard_normal(half))
        a_p[half:] = a_p[:half][::-1] if n % 2 == 0 else a_p[:half][::-1]
        # Actually just perturb both halves independently
        a_p = a.copy()
        a_p[:half] *= (1.0 + scale * 1.5 * rng.standard_normal(half))
        a_p[half:] *= (1.0 + scale * 1.5 * rng.standard_normal(n - half))
    else:
        # Sparse large perturbation
        a_p = a.copy()
        n_sel = max(10, n // 20)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p[sel] *= (1.0 + 0.2 * rng.standard_normal(n_sel))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy() / parent.sum())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p_interp = np.interp(x_new, x_old, parent / parent.sum())
            inits.append(p_interp.copy())

    x = np.arange(n, dtype=np.float64) / n
    # Uniform
    inits.append(np.ones(n) * (1.0 / n))
    # Random
    r = np.abs(rng.standard_normal(n))
    inits.append(r / r.sum())
    # U-shaped
    u = (2.0 * x - 1.0) ** 2
    inits.append(u / u.sum())
    # Cosine
    cos_v = np.maximum(0.01, np.cos(np.pi * x))
    inits.append(cos_v / cos_v.sum())
    # Two-peaked
    two = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    inits.append(two / two.sum())
    return inits

            if time.monotonic() > deadline - 15:
                break
            sc_k, a_k, _ = run_lbfgs(best_a, 200000, maxiter=20000,
                                      maxfun=60000, top_k=tk)
            report(sc_k, a_k, f"final tk={tk}")

    # Final coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd_final = coordinate_descent(best_a, n_passes=3, n_trial=6)
        sc = compute_score(a_cd_final)
        report(sc, a_cd_final, "final CD")

    # Final Nesterov
    if time.monotonic() < deadline - 10:
        a_ns_final = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns_final)
        report(sc, a_ns_final, "final Nesterov")

    # Multi-peak subgradient final
    if time.monotonic() < deadline - 5:
        a_mp_final = multi_peak_subgrad(best_a, n_iters=1500, f_lb=1.0, n_active=3)
        sc = compute_score(a_mp_final)
        report(sc, a_mp_final, "final multi-peak")

    # ==================================================================
    # Final verification and output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 256

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
