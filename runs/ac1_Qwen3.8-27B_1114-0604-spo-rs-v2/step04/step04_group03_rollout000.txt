"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses improved Fienup error reduction with soft thresholding,
LSE-smoothed L-BFGS, and peak-targeting coordinate descent.
Key improvements: adaptive soft-threshold Fienup, better beta scheduling,
multi-length exploration, and targeted peak elimination.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient computation via FFT cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def fienup_soft(a, n_iters=500, softness=20.0, alpha_start=0.5, alpha_end=0.95):
    """
    Improved Fienup error reduction with soft (tanh) thresholding.
    Instead of hard-clipping the convolution, uses a smooth projection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        # Softness increases over iterations (harder projection later)
        beta_soft = softness * (1 + 4 * t)
        
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M_fft)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break
        
        # Soft threshold: reduce values above mean by a tanh-scaled amount
        excess = np.maximum(conv - conv_mean, 0.0)
        # Apply soft reduction: new_conv = conv - alpha * tanh(beta_soft * excess / conv_mean) * conv_mean * (excess / conv_mean)
        # Simplified: new_conv = conv - alpha * excess * tanh(beta_soft * excess / (2 * conv_mean))
        scale = excess / (2.0 * conv_mean + 1e-30)
        reduction = alpha * excess * np.tanh(beta_soft * scale)
        conv_mod = conv - reduction
        
        # Compute error in frequency domain
        err = conv_mod - conv  # negative where we want to reduce
        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M_fft)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0
        
        A_new = A + d
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def fienup_reverse(a, n_iters=200):
    """
    Reverse Fienup: amplify valleys instead of reducing peaks.
    This can escape local minima that peak-reduction gets stuck in.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M_fft)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_min = float(conv.min())
        if conv_max - conv_min < 1e-16:  # This should be conv_max
            break
        
        # Amplify valleys (values below mean)

        conv_mod = conv + 0.5 * deficit

        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M_fft)

            d = err_spec / (2.0 * A + 1e-15)

        A_new = A + d
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

def lse_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_targeted_cd(a, n_sweeps=10, n_top=30):
    """
    Coordinate descent that targets the top peaks of the convolution.
    For each peak, identify the top contributors and reduce them,
    redistributing mass to low-contributor positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]
        
        for k_star in top_peaks:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            
            # Reduce top contributors, increase bottom contributors
            n_reduce = max(5, n_top)
            n_reduce = min(n_reduce, len(contrib) // 2)
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            bottom_local = np.argsort(contrib)[:n_reduce]
            
            for j in top_local:
                gi = j_start + j
                old_val = a[gi]
                # Try reducing
                for factor in [0.7, 0.8, 0.85, 0.9]:
                    a[gi] = old_val * factor
                    S_curr = a.sum()
                    if S_curr < 1e-12:
                        a[gi] = old_val
                        break
                    a_temp = a / S_curr
                    sc = compute_score(a_temp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_temp.copy()
                        a = a_temp
                    else:
                        a[gi] = old_val
                        return  # revert if worse
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

            break
        
        # Reduce the frequency components that contribute most to the peak
        k_peak = int(np.argmax(conv))
        # The peak at k_peak is dominated by the DC component and low frequencies
        # Dampen low-frequency components (other than DC) to flatten
        for f in range(1, min(len(A), M_fft // 4)):
            A_abs[f] *= (1.0 - 0.01)
        
        A_new = A_abs * np.exp(1j * A_angle)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    
    # Various structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * np.abs(2.0 * x - 1.0))
    
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Chirp-like
    freq = 1 + 5 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * (freq * x + 0.5 * freq * x * x)))
    
    return inits
