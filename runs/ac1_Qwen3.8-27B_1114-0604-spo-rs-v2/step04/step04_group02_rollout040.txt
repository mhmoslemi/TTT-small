```python
"""
Optimizer for the step-function peak-coefficient problem.

Main ideas:
- Work on the probability simplex using a softmax parameterization v -> a,
  which removes the scale invariance and enforces non-negativity automatically.
- Minimize a scaled beta-LSE approximation of the exact max-convolution objective.
  The beta values are scaled as 1 / current_peak so that the LSE width is
  meaningful for large sequences.
- Run L-BFGS-B on the softmax variable with several smoothed-to-sharp betas.
- Use a mirror / projected subgradient polishing step with exact line search.
- Perform iterated local search perturbations and length resampling.
- Always keep a globally best normalized sequence and return a safe list.
"""

import time
import numpy as np
from scipy.optimize import minimize

MAX_N = 15000
BEST_SCORE = float("inf")
BEST_A = None
FUNC_START = 0.0
FUNC_DEADLINE = 0.0
EST_COST = None
EST_COST_N = 0


def fft_pow2(m):
    m = int(m)
    if m < 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=float)
    m = 2 * n - 1
    M = fft_pow2(m)
    A = np.fft.rfft(a, M)
    out = np.fft.irfft(A * A, M)[:m]
    np.maximum(out, 0.0, out=out)
    return out


def corr_p_a(p, a):
    a = np.asarray(a, dtype=float).ravel()
    p = np.asarray(p, dtype=float).ravel()
    n = a.size
    L = 2 * n - 1
    if p.size != L:
        tmp = np.zeros(L, dtype=float)
        k = min(L, p.size)
        if k > 0:
            tmp[:k] = p[:k]
        p = tmp
    pr = p[::-1]
    M = fft_pow2(3 * n - 2)
    A = np.fft.rfft(a, M)
    P = np.fft.rfft(pr, M)
    return np.fft.irfft(A * P, M)[n - 1:2 * n - 1]


def exact_score(a):
    a = np.asarray(a, dtype=float).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    S = float(a.sum())
    if S <= 1e-12:
        return float("inf")
    a = a / S
    c = fast_conv(a)
    m = float(np.max(c))
    return float(2.0 * n * m)


def clean_sequence(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        a = np.full(64, 1.0 / 64.0)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size > MAX_N:
        a = resample_sequence(a, MAX_N)
    S = float(a.sum())
    if S <= 1e-12:
        a = np.ones(a.size, dtype=float) / max(1, a.size)
    else:
        a = a / S
    return a


def resample_sequence(a, n):
    a = np.asarray(a, dtype=float).ravel()
    n = int(n)
    if a.size == 0:
        a = np.full(64, 1.0 / 64.0)
    if n <= 0:
        n = 64
    if a.size == n:
        return a
    if a.size == 1:
        return np.full(n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return clean_sequence(out)


def record_sequence(x, label):
    global BEST_SCORE, BEST_A
    try:
        a = clean_sequence(np.asarray(x, dtype=float).ravel())
        sc = exact_score(a)
        if np.isfinite(sc) and sc < BEST_SCORE:
            BEST_SCORE = float(sc)
            BEST_A = a.copy()
            el = time.monotonic() - FUNC_START
            print(f"[{label}] new best score={sc:.12f} n={a.size} time={el:.1f}s")
    except Exception as exc:
        print(f"[{label}] record failed: {exc}")


def initial_v(a):
    a = clean_sequence(a)
    n = a.size
    if n <= 1:
        return np.empty(0, dtype=float)
    eps = 1e-18
    z = np.log(np.clip(a, eps, 1000.0))
    z = z - z[-1]
    return np.clip(z, -50.0, 50.0)


def v_to_a(v):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size + 1
    z = np.empty(n, dtype=float)
    if n > 1:
        z[:-1] = v
    if n > 0:
        z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=float) / float(n)
    m = float(np.max(z))
    if not np.isfinite(m):
        return np.ones(n, dtype=float) / float(n)
    z = z - m
    z = np.clip(z, -100.0, 0.0)
    a = np.exp(z)
    S = float(a.sum())
    if S <= 1e-300 or not np.isfinite(S):
        return np.ones(n, dtype=float) / float(n)
    return a / S


def lse_v_fun(v, beta):
    n = int(v.size) + 1
    if n <= 1:
        return 2.0, np.empty(0, dtype=float)

    a = v_to_a(v)
    M = fft_pow2(3 * n - 2)
    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

    cmax = float(np.max(c))
    beta = float(beta)
    if beta <= 0.0 or not np.isfinite(beta):
        beta = 1.0

    scaled = beta * (c - cmax)
    expv = np.exp(np.clip(scaled, -700.0, 0.0))
    U = float(expv.sum())

    if U <= 0.0 or not np.isfinite(U):
        p = np.full(c.size, 1.0 / c.size)
        L = cmax
    else:
        p = expv / U
        L = cmax + float(np.log(U)) / beta

    score = 2.0 * n * L

    pr = p[::-1]
    P = np.fft.rfft(pr, M)
    corr = np.fft.irfft(A * P, M)[n - 1:2 * n - 1]
    g = 2.0 * corr

    D = float(np.dot(g, a))
    grad_full = 2.0 * n * a * (g - D)
    if not np.all(np.isfinite(grad_full)):
        grad_full = np.zeros(n, dtype=float)

    if n > 1:
        return score, grad_full[:-1].copy()
    return score, np.empty(0, dtype=float)


def estimate_v_cost(a):
    a = clean_sequence(a)
    v = initial_v(a)
    if v.size == 0:
        return 1e-3
    n = v.size + 1
    sc = exact_score(a)
    m = sc / (2.0 * n)
    beta = min(1e6, max(1.0, 1.0 / max(m, 1e-12)))
    st = time.monotonic()
    for _ in range(4):
        lse_v_fun(v, beta)
    dt = (time.monotonic() - st) / 4.0
    return max(dt, 1e-5)


def run_lbfgs_v(v0, beta, time_limit):
    global EST_COST, EST_COST_N

    if time.monotonic() > FUNC_DEADLINE:
        return None, None, float("inf")

    v0 = np.asarray(v0, dtype=float).ravel()
    if v0.size == 0:
        a = np.ones(1, dtype=float)
        sc = 2.0
        return np.empty(0, dtype=float), a, sc

    v0 = np.clip(v0, -50.0, 50.0)
    a0 = v_to_a(v0)
    sc0 = exact_score(a0)
    best_v = v0.copy()
    best_a = a0.copy()
    best_sc = sc0

    if EST_COST is None or EST_COST_N != v0.size:
        EST_COST = estimate_v_cost(a0)
        EST_COST_N = v0.size

    cost = max(float(EST_COST), 1e-5)
    rem_global = max(1.0, FUNC_DEADLINE - time.monotonic())
    flim = min(max(1.0, float(time_limit)), 0.95 * rem_global)

    maxfun = int(min(80000, max(20, flim * 0.35 / cost)))
    maxiter = max(20, int(maxfun * 0.45))

    bounds = [(-50.0, 50.0)] * v0.size

    t0 = time.monotonic()
    try:
        res = minimize(
            lse_v_fun,
            v0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1e-16,
                "gtol": 1e-10,
                "maxls": 50,
                "maxcor": 20,
                "iprint": 0,
            },
        )
        a1 = v_to_a(res.x)
        sc1 = exact_score(a1)
        if sc1 < best_sc:
            best_a = a1
            best_sc = sc1
            best_v = np.asarray(res.x, dtype=float).copy()

        if res.nfev > 0:
            emp = (time.monotonic() - t0) / float(res.nfev)
            if emp > EST_COST * 1.25:
                EST_COST = emp
            elif emp < EST_COST * 0.65:
                EST_COST = 0.85 * EST_COST
    except Exception as exc:
        print("L-BFGS step failed:", exc)

    return best_v, best_a, best_sc


def run_lbfgs_a(a0, beta, time_limit):
    v0 = initial_v(a0)
    _, a_out, sc_out = run_lbfgs_v(v0, beta, time_limit)
    if a_out is None:
        return a0, exact_score(a0)
    return a_out, sc_out


def peak_of(a):
    a = clean_sequence(a)
    n = a.size
    sc = exact_score(a)
    if not np.isfinite(sc):
        return 1e-12
    return sc / (2.0 * n)


def mirror_polish(a, time_limit=45.0):
    a = clean_sequence(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = a.size
    if n < 2:
        return best_a

    start = time.monotonic()
    max_iter = int(min(50, max(3, time_limit / 1.7)))

    for _ in range(max_iter):
        if time.monotonic() - start > 0.85 * max(1.0, time_limit):
            break
        a = best_a
        c = fast_conv(a)
        M = float(np.max(c))
        if M <= 1e-300:
            break

        cut = M - max(0.22 * M, 1e-12)
        active = c >= cut
        cnt = int(active.sum())
        topk = min(16, c.size)
        if cnt < 8:
            active = np.zeros(c.size, dtype=bool)
            if topk >= c.size:
                idx = np.arange(c.size)
            else:
                idx = np.argpartition(c, c.size - topk)[-topk:]
            active[idx] = True

        w = np.zeros(c.size, dtype=float)
        vals = (c[active] - M) / M
        wact = np.exp(np.clip(20.0 * vals, -100.0, 0.0))
        w[active] = wact
        wsum = float(w.sum())
        if wsum <= 1e-300:
            p = np.zeros(c.size, dtype=float)
            p[int(np.argmax(c))] = 1.0
        else:
            p = w / wsum

        g = 2.0 * corr_p_a(p, a)
        D = float(np.dot(g, a))
        d = g - D
        maxd = float(np.max(np.abs(d)))
        if maxd <= 1e-18 or not np.isfinite(maxd):
            break

        improved = False
        for fac in (0.05, 0.1, 0.2, 0.5, 1.0):
            eta = fac / maxd

            cand = a - eta * d
            np.clip(cand, 0.0, 1000.0, out=cand)
            S = float(cand.sum())
            if S > 1e-12:
                cand = cand / S
                sc2 = exact_score(cand)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = cand.copy()
                    improved = True

            z = np.log(np.maximum(a, 1e-18)) - eta * d
            z = z - float(np.max(z))
            z = np.clip(z, -60.0, 0.0)
            cand = np.exp(z)
            S = float(cand.sum())
            if S > 1e-12:
                cand = cand / S
                sc2 = exact_score(cand)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = cand.copy()
                    improved = True

        if not improved:
            for fac in (0.01, 0.005):
                if time.monotonic() - start > 0.85 * max(1.0, time_limit):
                    break
                eta = fac / maxd
                cand = a - eta * d
                np.clip(cand, 0.0, 1000.0, out=cand)
                S = float(cand.sum())
                if S > 1e-12:
                    cand = cand / S
                    sc2 = exact_score(cand)
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_a = cand.copy()
                        improved = True
            break

    return best_a


def gen_base_candidates(n, rng, parent, current_best, max_count=5):
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    raw = []

    if parent is not None:
        raw.append(resample_sequence(parent, n))

    raw.append(np.ones(n, dtype=float))

    for s in (1.0, -1.0):
        aa = 1.0 + 0.6 * s * (2.0 * x - 1.0)
        if np.min(aa) > -1e-12:
            raw.append(np.clip(aa, 0.0, None))

    for m in (1, 2, 3):
        for alpha in (0.7, 0.9):
            phase = 2.0 * np.pi * float(rng.random())
            aa = 1.0 + alpha * np.cos(2.0 * np.pi * float(m) * x + phase)
            if np.min(aa) > -1e-8:
                raw.append(np.clip(aa, 0.0, None))

    for std in (0.5, 1.5):
        vv = np.clip(rng.standard_normal(n - 1) * std, -50.0, 50.0)
        raw.append(v_to_a(vv))

    raw.append(np.maximum(0.05, 1.0 + 0.8 * rng.standard_normal(n)))

    scored = []
    for i, aa in enumerate(raw):
        bb = clean_sequence(aa)
        sc = exact_score(bb)
        if np.isfinite(sc):
            scored.append((float(sc), i, bb))

    scored.sort()
    bases = []
    for sc, i, aa in scored[:max_count]:
        if sc <= current_best + 0.5:
            bases.append(aa)
    if not bases:
        bases.append(clean_sequence(np.ones(n, dtype=float)))
    return bases


def candidate_lengths(n):
    n = int(n)
    out = []
    for mult in (1.0, 1.25, 1.5, 2.0, 0.75, 0.5):
        m = int(round(n * mult))
        if 512 <= m <= MAX_N:
            if m not in out:
                out.append(m)
    for m in (1024, 4096, 8192, MAX_N):
        if m not in out:
            out.append(m)
    return out

    print(f"--- Length exploration ---")

    if BEST_A is not None and time.monotonic() < length_deadline:
        n_best = int(BEST_A.size)
        for n_try in candidate_lengths(n_best):
            if time.monotonic() >= length_deadline:
                break
            if n_try == n_best:
                continue
            a_try = resample_sequence(BEST_A, n_try)
            record_sequence(a_try, f"resample n={n_try}")

            if n_try > n_best:
                M_try = peak_of(a_try)
                if M_try <= 1e-12:
                    continue
                for mult in (1.0, 10.0):
                    if time.monotonic() >= length_deadline - 20:
                        break
                    tlim = min(60.0, max(8.0, 0.22 * (length_deadline - time.monotonic())))
                    beta = min(1e7, max(1.0, mult / M_try))
                    a_res, sc_res = run_lbfgs_a(
                        resample_sequence(BEST_A, n_try),
                        beta,
                        tlim,
                    )
                    if a_res is not None:
                        record_sequence(a_res, f"len n={n_try} mult={mult:g}")
                    if BEST_A is not None and BEST_A.size == n_try:
                        M_try = peak_of(BEST_A)

    print("Final polish.")

    if BEST_A is not None:
        a_sym_final = 0.5 * (BEST_A + BEST_A[::-1])
        record_sequence(a_sym_final, "final sym")

        if time.monotonic() < FUNC_DEADLINE - 30:
            M_final = peak_of(BEST_A)
            if M_final > 1e-12:
                beta_final = min(1e7, max(1.0, 30.0 / M_final))
                tlim = min(45.0, max(8.0, 0.45 * (FUNC_DEADLINE - time.monotonic())))
                a_res, sc_res = run_lbfgs_a(BEST_A, beta_final, tlim)
                if a_res is not None:
                    record_sequence(a_res, "final lbfgs30")

        if time.monotonic() < FUNC_DEADLINE - 20:
            a_m = mirror_polish(BEST_A, time_limit=max(8.0, 0.25 * (FUNC_DEADLINE - time.monotonic())))
            record_sequence(a_m, "final mirror")

    if BEST_A is None:
        out_a = np.full(64, 1.0 / 64.0, dtype=float)
    else:
        out_a = BEST_A.copy()
        if out_a.size > MAX_N:
            out_a = resample_sequence(out_a, MAX_N)
        out_a = clean_sequence(out_a)

    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if not out or sum(out) < 0.01:
        out = [1.0 / 64.0] * 64

    verified = float("inf")
    try:
        if time.monotonic() < FUNC_DEADLINE - 5:
            verified = float(evaluate_sequence(out))
    except Exception:
        verified = exact_score(out_a)

    internal = exact_score(out_a)
    print("=" * 70)
    print(f"FINAL internal_best={BEST_SCORE:.12f}")
    print(f"FINAL returned_score_fft={internal:.12f}")
    print(f"FINAL verified_by_evaluate={verified:.12f}")
    print(f"Length={len(out)} elapsed={time.monotonic() - FUNC_START:.1f}s / {budget_s:.1f}s")
    print("=" * 70)

    return out
```
