            scores = [evaluate_sequence(s) for s in sequences_to_test[:3]]
            best_sequence = sequences_to_test[scores == np.min(scores)]
            break
    
    print(f"Optimization completed. Final score: {best_score:.6g}")
    # Convert sequence to required format (list of floats)
    candidate = [float(x) for x in np.clip(best_sequence, 0, 1000) * 0.999 if x > 0]
    
    if sum(candidate) < 0.01:
        # Fallback to reasonable sequence if needed
        candidate = np.ones(len(candidate)) * (len(candidate) * 0.01) if len(candidate) > 0 else [0.5]
    
    # Ensure sequence validity
    retry_candidate = []
    for x in candidate:
        if np.isinf(x) or np.isnan(x):
            retry_candidate.append(0.5)
        else:
            retry_candidate.append(x)

    if not any(np.isnan(r) or np.isinf(r) for r in retry_candidate):
        final_candidate = retry_candidate
    else:
        final_candidate = [0.5] * len(candidate)
    
    return final_candidate

# Target evaluation function
def evaluate_sequence(sequence):
    "Evaluates the normalized sequence against the scoring metric"
    n = len(sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:  # Scale adjustment could be applied here
        return np.inf
    
    conv_result = np.convolve(sequence, sequence)
    max_b = np.max(conv_result)
    return 2 * n * max_b / (sum_a ** 2)

# Function to estimate the change in score before costly update
def compute_change_in_score(sequence):
    "Estimates how much the score would change by small perturbations"
    perturb = np.random.normal(size=len(sequence)) * 0.01
    new_seq = sequence + perturb
    new_seq = np.clip(new_seq, 0, 1000)
    
    if np.sum(new_seq) < 0.01:
        return evaluate_sequence(new_seq) - evaluate_sequence(sequence)
    
    current_score = evaluate_sequence(sequence)
    new_score = evaluate_sequence(new_seq)
    return new_score - current_score
