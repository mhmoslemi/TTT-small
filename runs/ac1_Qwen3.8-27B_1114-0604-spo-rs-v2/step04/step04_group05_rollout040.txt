```
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if n == 1: return a*a
```

```python
"""
Search for low-peak non-negative self-convolution sequences.

The objective is
    2*n*max_k((a*a)[k]) / (sum(a))**2
with a >= 0 and finite sequence length n.

This program combines several ideas:
- starts from the available parent sequence and explicit continuous-motivated
  baseline families (endpoint square-root and related smooth densities);
- optimizes a normalized nonnegative sequence in log/softmax coordinates
  using LSE-smoothed objectives and L-BFGS continuation;
- polish at multiple resolutions with fast exact 1D coordinate moves using an
  incremental convolution update;
- uses a small LP projection restart from the fixed parent autocorrelation,
  mirroring the previous state-of-the-art projection idea;
- finally refines the best full-length candidate with many exact coordinate
  sweeps and returns a scaled list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

GLOBAL_DEADLINE = None


def set_global_deadline(t):
    global GLOBAL_DEADLINE
    GLOBAL_DEADLINE = t


def fast_conv(a):
    """Linear self-convolution using a real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    f = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, f)
    return np.fft.irfft(A * A, f)[:m]


def normalize_nonneg(a):
    """Clip to nonnegative and rescale to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(64, dtype=np.float64)
    a = np.clip(a, 0.0, 1.0e6)
    s = float(a.sum())
    if s <= 1.0e-15 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def compute_score(a):
    """Exact internal evaluation, equivalent to evaluate_sequence for valid input."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1.0e-12:
        return np.inf
    c = fast_conv(a)
    if c.size == 0:
        return np.inf
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def log_resample(a, m):
    """Resample a nonnegative sequence logarithmically to length m."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(m, dtype=np.float64)
    a = np.clip(a, 0.0, 1.0e6)
    s = float(a.sum())
    if s <= 1.0e-15:
        return np.ones(m, dtype=np.float64)
    a = a / s
    if a.size == m:
        return a
    eps = max(1.0e-300, 1.0e-12 * float(a.max()))
    xa = np.arange(a.size, dtype=np.float64)
    xm = np.linspace(0.0, a.size - 1.0, m)
    ya = np.log(np.maximum(a, eps))
    ym = np.interp(xm, xa, ya)
    out = np.exp(ym)
    return normalize_nonneg(out)


def softmax_from_z(z):
    """Stable softmax, forced to sum 1."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1.0e-15 or not np.isfinite(s):
        e = np.zeros_like(z)
        e[0] = 1.0
        s = 1.0
    out = e / s
    ss = float(out.sum())
    if ss <= 0.0 or not np.isfinite(ss):
        out = np.ones_like(z) / z.size
    else:
        out = out / ss
    return out


def make_family_sequence(n):
    """Build a fairly good explicit baseline from several smooth families."""
    n = int(max(16, n))
    x = np.linspace(0.5 / n, 1.0 - 0.5 / n, n)
    t = np.arange(n, dtype=np.float64) / n
    best_a = np.ones(n, dtype=np.float64) / n
    best_score = np.inf

    pairs = [
        (0.50, 0.00), (0.48, 0.00), (0.45, 0.00),
        (0.45, 0.05), (0.45, 0.10), (0.40, 0.05),
        (0.40, 0.12), (0.35, 0.08), (0.30, 0.05),
        (0.30, 0.15), (0.50, 0.08), (0.50, 0.18),
    ]
    for alpha, beta in pairs:
        with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
            a = (x ** (-alpha)) * ((1.0 - x) ** beta)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1.0e6)
        a = normalize_nonneg(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a

    # Smooth endpoint bulges / dents on top of the 1/sqrt(x) endpoint profile.
    base = x ** (-0.5)
    for amp, center, wd in [
        (0.15, 0.90, 0.10), (0.30, 0.95, 0.08), (0.50, 1.00, 0.12),
        (0.20, 0.75, 0.15), (0.35, 0.85, 0.10),
        (-0.15, 0.95, 0.10), (-0.30, 1.00, 0.15),
    ]:
        with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
            a = base * (1.0 + amp * np.exp(-((t - center) / max(1.0e-6, wd)) ** 2))
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1.0e6)
        a = normalize_nonneg(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a

    return best_a, best_score


def best_sequence_from_list(arrays):
    """Choose the lowest-score sequence among a list of arrays/list/cloud of arrays."""
    best_a = None
    best_score = np.inf
    for a in arrays:
        if a is None:
            continue
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            continue
        if a.size < 16:
            continue
        a = normalize_nonneg(a)
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    if best_a is None:
        best_a = np.ones(128, dtype=np.float64) / 128
        best_score = compute_score(best_a)
    return best_score, best_a


def lse_score_grad_a(a, beta):
    """LSE-smoothed objective and gradient with respect to a (sum(a)=1 assumed)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    s = float(a.sum())
    if s <= 1.0e-15:
        return 1.0e100, np.zeros(n, dtype=np.float64)
    a = a / s
    c = fast_conv(a)
    cmax = float(c.max())
    if not np.isfinite(cmax):
        return 1.0e100, np.zeros(n, dtype=np.float64)
    scaled = float(beta) * (c - cmax)
    e = np.exp(scaled)
    total = float(e.sum())
    if total <= 1.0e-300 or not np.isfinite(total):
        p = np.zeros_like(c)
        if c.size:
            p[int(c.argmax())] = 1.0
        total = 1.0
        L = cmax
    else:
        p = e / total
        L = cmax + math.log(total) / float(beta)
    mconv = 3 * n - 2
    ffft = 1 << (mconv - 1).bit_length()
    RA = np.fft.rfft(a[::-1], ffft)
    P = np.fft.rfft(p, ffft)
    convp = np.fft.irfft(RA * P, ffft)[:mconv]
    dL = 2.0 * convp[n - 1:2 * n - 1]
    score = 2.0 * n * L
    grad = 2.0 * n * (dL - 2.0 * L)
    return float(score), grad


def z_objective(z, beta):
    """Objective and gradient in softmax/log coordinates."""
    if GLOBAL_DEADLINE is not None and time.monotonic() > GLOBAL_DEADLINE:
        raise RuntimeError("stage deadline exceeded")
    z = np.asarray(z, dtype=np.float64)
    n = int(z.size)
    if n == 0:
        return 1.0e100, np.zeros(0, dtype=np.float64)
    a = softmax_from_z(z)
    score, grad_a = lse_score_grad_a(a, beta)
    if not np.isfinite(score):
        return 1.0e100, np.zeros(n, dtype=np.float64)
    proj = float(np.dot(a, grad_a))
    g = a * (grad_a - proj)
    g = np.clip(g, -1.0e6, 1.0e6)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return float(score), g


def run_z_lbfgs(z, betas, phase_deadline, stage_time=40.0, maxfun=30000):
    """L-BFGS continuation on log/softmax coordinates with respect to phase deadline."""
    z = np.asarray(z, dtype=np.float64).copy()
    z = np.clip(z, -40.0, 40.0)
    n = int(z.size)
    if n == 0:
        return z
    bounds = [(-40.0, 40.0)] * n
    betas = list(betas)
    for beta in betas:
        now = time.monotonic()
        if now >= phase_deadline - 4.0:
            break
        remaining = phase_deadline - now
        per = max(5.0, remaining / max(1, len(betas)))
        stage = min(float(stage_time), per)
        if stage <= 2.0:
            break
        stage_deadline = now + stage
        set_global_deadline(stage_deadline)
        mf = max(300, int(maxfun * stage / 60.0))
        mi = max(100, int(mf * 0.7))
        try:
            res = minimize(
                z_objective,
                z,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                args=(float(beta),),
                options={
                    "ftol": 1.0e-12,
                    "gtol": 1.0e-7,
                    "maxiter": mi,
                    "maxfun": mf,
                    "maxls": 25,
                },
            )
            if res is not None and getattr(res, "x", None) is not None:
                cand = np.asarray(res.x, dtype=np.float64)
                if cand.size == n and np.all(np.isfinite(cand)):
                    z = np.clip(cand, -40.0, 40.0)
        except Exception:
            pass
        set_global_deadline(None)
        z = z - float(z.mean())
    set_global_deadline(None)
    return z


def adam_z(z, beta, phase_deadline, lr=0.025):
    """Adam-like optimizer in log/softmax coordinates."""
    set_global_deadline(None)
    z = np.asarray(z, dtype=np.float64).copy()
    z = np.clip(z - float(z.mean()), -40.0, 40.0)
    n = int(z.size)
    if n == 0:
        return z, np.inf
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1, b2 = 0.90, 0.999
    eps = 1.0e-8
    t = 0
    best_z = z.copy()
    best_score = np.inf
    while time.monotonic() < phase_deadline - 0.5:
        t += 1
        try:
            score, g = z_objective(z, float(beta))
        except Exception:
            break
        if not np.isfinite(score):
            break
        if score < best_score:
            best_score = float(score)
            best_z = z.copy()
        g = np.clip(g, -1.0e3, 1.0e3)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        lr_t = lr / (1.0 + 1.0e-5 * t)
        step = mh / (np.sqrt(vh) + eps)
        step = np.clip(step, -1.0, 1.0)
        z = z - lr_t * step
        z = z - float(z.mean())
        z = np.clip(z, -40.0, 40.0)
        if t % 5000 == 0 and phase_deadline - time.monotonic() <= 1.0:
            break
    set_global_deadline(None)
    return best_z, best_score


def coord_score_delta(c, b, S, sq, dx, n):
    """Exact score after changing one coordinate by dx, using precomputed b."""
    if abs(dx) <= 1.0e-18:
        return 2.0 * n * float(np.max(c))
    base = c + 2.0 * dx * b
    m = float(np.max(base))
    if 0 <= sq <= 2 * n - 2:
        vv = float(c[sq] + 2.0 * dx * b[sq] + dx * dx)
        if vv > m:
            m = vv
    Sx = S + dx
    if Sx <= 1.0e-15:
        return np.inf
    return 2.0 * n * m / (Sx * Sx)


def coord_descent(a, phase_deadline, top_k=None, grid_points=13, golden_iters=11,
                  recompute_every=8, rng=None):
    """
    Local minimization by exact one-dimensional coordinate search.

    The convolution is maintained incrementally.  For coordinate j, changing
    a[j] by dx updates c_k by 2*dx*a[k-j], plus dx^2 at k=2j.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size < 16:
        return compute_score(a), a
    a = np.clip(a, 0.0, 1000.0)
    s0 = float(a.sum())
    if s0 <= 1.0e-15:
        a = np.ones(a.size, dtype=np.float64) / a.size
    else:
        a = a / s0
    n = int(a.size)
    if top_k is None:
        top_k = int(max(16, min(80, n // 32)))
    if rng is None:
        rng = np.random.default_rng(12345)

    c = fast_conv(a)
    c = np.maximum(c, 0.0)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_score = cur_score
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    sweeps = 0

    while time.monotonic() < phase_deadline - 1.5:
        sweeps += 1
        if sweeps % recompute_every == 0:
            a = np.clip(a, 0.0, 1000.0)
            ss = float(a.sum())
            if ss <= 1.0e-15:
                break
            a = a / ss
            c = fast_conv(a)
            c = np.maximum(c, 0.0)
            S = 1.0

        k = int(c.argmax())
        M = float(c[k])
        cur_score = float(2.0 * n * M)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        L = 2 * n - 1
        if L > 8:
            top_lags = np.argpartition(c, -8)[-8:]
        else:
            top_lags = np.arange(L)

        grad = np.zeros(n, dtype=np.float64)
        for lag in top_lags:
            lag = int(lag)
            js = max(0, lag - n + 1)
            je = min(lag, n - 1)
            if je >= js:
                idx = lag - np.arange(js, je + 1)
                grad[js:je + 1] += a[idx] - c[lag]

        order1 = np.argsort(np.abs(grad))[::-1][:top_k]
        order2 = np.argsort(a)[::-1][:max(2, top_k // 10)]
        order3 = rng.integers(0, n, size=min(10, n))
        order = np.unique(np.concatenate((order1, order2, order3)))

        maxa = float(a.max())
        typ = max(1.0 / n, 1.0e-8)

        for jj in order:
            if time.monotonic() > phase_deadline - 1.2:
                break
            j = int(jj)
            old = float(a[j])

            b = np.zeros(2 * n - 1, dtype=np.float64)
            b[j:j + n] = a
            sq = 2 * j

            hi = max(4.0 * old + 1.0e-15, 8.0 * maxa + 1.0e-15, 12.0 * typ)
            hi = min(1000.0, hi)
            if hi <= 1.0e-14:
                hi = 10.0 * maxa + 1.0e-6

            if old > 1.0e-15:
                lo_g = min(old * 1.0e-2, hi)
                if lo_g <= 0.0:
                    lo_g = 1.0e-12
                geo = np.geomspace(lo_g, hi, grid_points)
                grid = np.unique(np.concatenate(([0.0, min(old, hi)], geo)))
            else:
                grid = np.linspace(0.0, hi, grid_points)
            grid = grid[(grid >= 0.0) & (grid <= hi)]
            if grid.size == 0:
                grid = np.array([0.0])

            deltas = grid - old
            if float(np.max(np.abs(deltas))) > 1.0e-16:
                cand = c[None, :] + 2.0 * deltas[:, None] * b[None, :]
                if 0 <= sq <= 2 * n - 2:
                    cand[:, sq] += deltas * deltas
                mx = cand.max(axis=1)
                Sx = S + deltas
                valid = Sx > 1.0e-12
                fvals = np.full(grid.shape, np.inf, dtype=np.float64)
                if np.any(valid):
                    fvals[valid] = (2.0 * n * mx[valid]) / (Sx[valid] * Sx[valid])
                p = int(np.argmin(fvals))
                best_x = float(grid[p])
                best_f = float(fvals[p])

                if p > 0 and p < grid.size - 1:
                    lo_b = float(grid[p - 1])
                    hi_b = float(grid[p + 1])
                elif p == 0:
                    lo_b = 0.0
                    hi_b = float(grid[1]) if grid.size > 1 else hi
                else:
                    lo_b = float(grid[-2]) if grid.size > 1 else 0.0
                    hi_b = float(grid[-1])

                if hi_b > lo_b + 1.0e-16:
                    c1 = hi_b - gr * (hi_b - lo_b)
                    c2 = lo_b + gr * (hi_b - lo_b)
                    f1 = coord_score_delta(c, b, S, sq, c1 - old, n)
                    f2 = coord_score_delta(c, b, S, sq, c2 - old, n)
                    for _ in range(golden_iters):
                        if f1 < f2:
                            hi_b = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi_b - gr * (hi_b - lo_b)
                            f1 = coord_score_delta(c, b, S, sq, c1 - old, n)
                        else:
                            lo_b = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo_b + gr * (hi_b - lo_b)
                            f2 = coord_score_delta(c, b, S, sq, c2 - old, n)
                    if f1 < f2:
                        best_x = c1
                        best_f = f1
                    else:
                        best_x = c2
                        best_f = f2

                if best_f < cur_score - 1.0e-15:
                    dx = best_x - old
                    if abs(dx) > 1.0e-16:
                        c += 2.0 * dx * b
                        if 0 <= sq <= 2 * n - 2:
                            c[sq] += dx * dx
                        a[j] = best_x
                        S += dx
                        cur_score = best_f

        if S > 1.0e-12:
            a = a / S
            c = c / (S * S)
            S = 1.0
        else:
            break

        val = float(2.0 * n * c.max())
        if val < best_score:
            best_score = val
            best_a = a.copy()

    final_val = compute_score(best_a)
    if final_val < best_score:
        best_score = final_val
    else:
        best_score = final_val
    return best_score, best_a


def spectral_perturb(a, rng, strength=0.18):
    """Perturb amplitudes/phases of the real FFT and rectify/normalize."""
    a = normalize_nonneg(a)
    n = int(a.size)
    m = 2 * n - 1
    f = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, f)
    amp = 1.0 + float(strength) * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.25, 2.5)
    ph = 0.55 * float(strength) * rng.standard_normal(len(A))
    An = A * amp * np.exp(1j * ph)
    out = np.fft.irfft(An, f)[:n]
    out = np.clip(out, 0.0, 1000.0)
    return normalize_nonneg(out)


def lp_projection(a, phase_deadline, max_iter=3):
    """
    A few steps of the fixed-parent convolutionbound projection:
    solve max sum(g) s.t. g>=0, f*g <= max(f*f), and blend.
    Applied only at small resolution for speed.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    if n < 64:
        a = log_resample(a, 128)
        n = 128
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iter):
        now = time.monotonic()
        if now >= phase_deadline - 10.0:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M <= 1.0e-15 or not np.isfinite(M):
            break
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a
        b_ub = np.ones(2 * n - 1, dtype=np.float64) * M
        cvec = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 20.0)] * n
        remaining = max(3.0, phase_deadline - now)
        tl = min(28.0, remaining / 2.0)
        ok = False
        try:
            res = linprog(
                cvec,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"time_limit": tl, "presolve": True},
            )
            ok = True
        except Exception:
            try:
                res = linprog(
                    cvec,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
                ok = True
            except Exception:
                res = None

        if not ok or res is None or getattr(res, "success", False) is False:
            break
        g = getattr(res, "x", None)
        if g is None:
            break
        g = np.asarray(g, dtype=np.float64)
        if g.size != n:
            break
        g = np.clip(g, 0.0, 1000.0)
        sg = float(g.sum())
        if sg <= 1.0e-12:
            break
        g = g / sg
        theta = 0.12 + 0.08 * it
        mix = (1.0 - theta) * a + theta * g
        mix = np.clip(mix, 0.0, 1000.0)
        mix = normalize_nonneg(mix)
        sc = compute_score(mix)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = mix.copy()
        # Keep moving from mix even if not improving, unless it is much worse.
        if sc > best_score * 1.25:
            break
        a = mix
    return best_score, best_a


def optimize_one(a, phase_deadline, rng):
    """
    Improve one candidate:
      1) short Adam/LSE in z-space,
      2) L-BFGS continuation in z-space,
      3) high-quality coordinate polish.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    now = time.monotonic()
    if now >= phase_deadline - 6.0 or n < 16:
        return compute_score(a), a.copy()

    best_a = a.copy()
    best_score = compute_score(a)

    a_safe = np.maximum(a, 1.0e-300)
    z = np.log(a_safe)
    if z.size:
        z = z - float(z.mean())
    z = np.clip(z, -40.0, 40.0)

    rem0 = phase_deadline - now
    adam_time = min(45.0, 0.18 * rem0)
    if adam_time > 4.0:
        try:
            z, sc_adam = adam_z(z, 20.0, now + adam_time, lr=0.028)
            if np.isfinite(sc_adam):
                a_adam = softmax_from_z(z)
                sc_adam2 = compute_score(a_adam)
                if sc_adam2 < best_score:
                    best_score = sc_adam2
                    best_a = a_adam.copy()
        except Exception:
            pass

    now = time.monotonic()
    if now < phase_deadline - 25.0:
        if n <= 128:
            betas = [8.0, 30.0, 120.0, 500.0, 2000.0]
            stage = 18.0
        elif n <= 256:
            betas = [12.0, 50.0, 200.0, 800.0, 3000.0]
            stage = 24.0
        elif n <= 512:
            betas = [25.0, 100.0, 400.0, 1600.0]
            stage = 32.0
        elif n <= 1024:
            betas = [40.0, 160.0, 700.0, 2800.0]
            stage = 48.0
        else:
            betas = [80.0, 320.0, 1300.0, 5200.0]
            stage = 70.0
        try:
            z = run_z_lbfgs(z, betas, phase_deadline, stage_time=stage, maxfun=30000)
            a_z = softmax_from_z(z)
            sc_z = compute_score(a_z)
            if np.isfinite(sc_z) and sc_z < best_score:
                best_score = sc_z
                best_a = a_z.copy()
        except Exception:
            pass

    now = time.monotonic()
    if now < phase_deadline - 8.0:
        remaining = max(1.0, phase_deadline - now)
        if n <= 256:
            ct = min(60.0, remaining - 3.0)
        elif n <= 512:
            ct = min(90.0, remaining - 3.0)
        elif n <= 1024:
            ct = min(150.0, remaining - 3.0)
        else:
            ct = min(320.0, remaining - 3.0)
        if ct > 8.0:
            cd = time.monotonic() + ct
            if cd > phase_deadline - 2.0:
                cd = phase_deadline - 2.0
            try:
                sc_cd, a_cd = coord_descent(
                    best_a,
                    cd,
                    top_k=None,
                    grid_points=12,
                    golden_iters=10,
                    recompute_every=8,
                    rng=rng,
                )
                if np.isfinite(sc_cd) and sc_cd < best_score:
                    best_score = sc_cd
                    best_a = a_cd.copy()
            except Exception:
                pass

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.

    Returns the best nonnegative float sequence found within the time budget.
    """
    global GLOBAL_DEADLINE
    set_global_deadline(None)

    try:
        rng = np.random.default_rng(int(seed) if seed is not None else 42)
    except Exception:
        rng = np.random.default_rng(42)

    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    eff = max(60.0, total - 45.0)
    hard = t0 + total - 8.0

    # ------------------------------------------------------------------
    # Load parent if available.
    # ------------------------------------------------------------------
    parent = None
    try:
        obj = globals().get("height_sequence_1", None)
        if obj is not None:
            p = np.asarray(obj, dtype=np.float64).ravel()
            if p.size >= 64:
                p = np.clip(p, 0.0, 1000.0)
                ps = float(p.sum())
                if ps > 1.0e-6:
                    parent = p / ps
    except Exception:
        parent = None

    if parent is not None:
        parent_len = int(parent.size)
        full_n = int(min(12000, max(4096, parent_len)))
        print(f"Parent loaded: length={parent.size}, score={compute_score(parent):.9f}")
    else:
        full_n = 8192
        print("No parent available; using synthetic initializations.")

    # ------------------------------------------------------------------
    # Initial full-length candidates.
    # ------------------------------------------------------------------
    full_cands = []
    if parent is not None:
        if int(parent.size) != full_n:
            full_cands.append(log_resample(parent, full_n))
        else:
            full_cands.append(parent.copy())
    try:
        fam_full, fam_score_full = make_family_sequence(full_n)
        full_cands.append(fam_full)
    except Exception:
        fam_full = None

    initial_score, initial_a = best_sequence_from_list(full_cands)
    best_score = initial_score
    best_a = initial_a.copy()
    print(f"Initial best: score={best_score:.9f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Quick full-length coordinate polish of the best start.
    # ------------------------------------------------------------------
    quick_deadline = t0 + min(240.0, 0.10 * eff)
    if quick_deadline > time.monotonic() + 25.0:
        try:
            sc, aa = coord_descent(
                best_a,
                quick_deadline,
                top_k=min(70, max(32, best_a.size // 64)),
                grid_points=12,
                golden_iters=10,
                recompute_every=8,
                rng=rng,
            )
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
                print(f"Quick full coord: score={best_score:.9f}, n={best_a.size}")
        except Exception as e:
            print("Quick full coord failed:", e)

    # ------------------------------------------------------------------
    # Low/medium-resolution global-ish search and refinement.
    # ------------------------------------------------------------------
    low_deadline = t0 + 0.55 * eff

    small_start = [best_a]
    if parent is not None:
        small_start.append(parent)
    try:
        small_start.append(make_family_sequence(128)[0])
    except Exception:
        pass
    small_score, current_small = best_sequence_from_list(small_start)

    low_sizes = [128, 256, 512, 1024, 2048]
    low_weights = [1.0, 1.25, 1.75, 2.75, 3.50]
    low_caps = [70.0, 110.0, 170.0, 270.0, 420.0]

    for idx, m in enumerate(low_sizes):
        now = time.monotonic()
        if now >= low_deadline - 35.0:
            break
        remaining = low_deadline - now
        if remaining <= 40.0:
            break
        weights_left = sum(low_weights[idx:])
        alloc = remaining * (low_weights[idx] / weights_left)
        alloc = min(alloc, low_caps[idx])
        if alloc <= 25.0:
            break

        cands = []
        try:
            cands.append(log_resample(current_small, m))
        except Exception:
            cands.append(np.ones(m, dtype=np.float64) / m)
        if parent is not None:
            try:
                cands.append(log_resample(parent, m))
            except Exception:
                pass
        try:
            fam_m, _ = make_family_sequence(m)
            cands.append(fam_m)
        except Exception:
            pass
        qs, qa = best_sequence_from_list(cands)

        # Main optimization of the most promising low-res candidate.
        deadline_main = now + alloc * 0.68
        if deadline_main > low_deadline - 20.0:
            deadline_main = low_deadline - 20.0
        if deadline_main > now + 20.0:
            try:
                sc1, a1 = optimize_one(qa, deadline_main, rng)
                if sc1 < best_score:
                    best_score = sc1
                    best_a = a1.copy()
                    print(f"Low-res n={m} main: score={best_score:.9f}")
                cands.append(a1)
            except Exception as e:
                print(f"Low-res n={m} main failed:", e)

            # If time remains, try the second-best start for diversity.
            now2 = time.monotonic()
            deadline_second = min(low_deadline - 15.0, now + alloc)
            if len(cands) >= 2 and deadline_second > now2 + 35.0:
                for aa in cands:
                    sc_try = compute_score(aa)
                    if not np.isfinite(sc_try):
                        continue
                    if sc_try < qs + 0.01 and np.allclose(aa, qa, rtol=0, atol=1e-12):
                        continue
                    try:
                        sc2, a2 = optimize_one(aa, deadline_second, rng)
                        if sc2 < best_score:
                            best_score = sc2
                            best_a = a2.copy()
                            print(f"Low-res n={m} second: score={best_score:.9f}")
                        cands.append(a2)
                    except Exception as e2:
                        print(f"Low-res n={m} second failed:", e2)
                    break

        # Update the evolving low-resolution champion.
        sc_small, cur_small2 = best_sequence_from_list(cands)
        if sc_small < small_score or cur_small2.size <= m + 3:
            small_score = sc_small
            current_small = cur_small2
            print(f"Low-res champion after n={m}: score={small_score:.9f}, n={current_small.size}")

    # ------------------------------------------------------------------
    # Diversity: LP projection + spectral restart.
    # ------------------------------------------------------------------
    div_deadline = t0 + 0.70 * eff
    if current_small is not None and time.monotonic() < div_deadline - 40.0:
        n_lp = int(min(256, max(128, current_small.size)))
        try:
            a_lp = log_resample(current_small, n_lp)
            deadline_lp = min(div_deadline - 25.0, time.monotonic() + 140.0)
            sc_lp, a_lp = lp_projection(a_lp, deadline_lp, max_iter=2)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
                print(f"LP restart: score={best_score:.9f}, n={best_a.size}")
        except Exception as e:
            print("LP restart failed:", e)

        if time.monotonic() < div_deadline - 40.0:
            try:
                a_sp = spectral_perturb(current_small, rng, strength=0.28)
                a_sp = log_resample(a_sp, 512)
                deadline_sp = min(div_deadline - 18.0, time.monotonic() + 160.0)
                sc_sp, a_sp = optimize_one(a_sp, deadline_sp, rng)
                if sc_sp < best_score:
                    best_score = sc_sp
                    best_a = a_sp.copy()
                    print(f"Spectral restart: score={best_score:.9f}, n={best_a.size}")
            except Exception as e:
                print("Spectral restart failed:", e)

    # ------------------------------------------------------------------
    # Full-length refinement.
    # ------------------------------------------------------------------
    final_full_deadline = t0 + 0.95 * eff
    full_starts = []
    if parent is not None:
        full_starts.append(parent)
    full_starts.append(best_a)
    if fam_full is not None:
        full_starts.append(fam_full)
    if current_small is not None:
        full_starts.append(current_small)

    scored_full = []
    for aa in full_starts:
        if aa is None:
            continue
        try:
            aa = np.asarray(aa, dtype=np.float64).ravel()
        except Exception:
            continue
        if aa.size < 64:
            continue
        if aa.size != full_n:
            aa = log_resample(aa, full_n)
        sc = compute_score(aa)
        if np.isfinite(sc):
            scored_full.append((sc, aa))
    scored_full.sort()

    if scored_full and time.monotonic() < final_full_deadline - 25.0:
        full0_score, full0_a = scored_full[0]
        if full0_score < best_score:
            best_score = full0_score
            best_a = full0_a.copy()
        try:
            sc_full, a_full = coord_descent(
                full0_a,
                final_full_deadline,
                top_k=min(80, max(36, full_n // 64)),
                grid_points=14,
                golden_iters=12,
                recompute_every=10,
                rng=rng,
            )
            if sc_full < best_score:
                best_score = sc_full
                best_a = a_full.copy()
                print(f"Full coord main: score={best_score:.9f}, n={best_a.size}")
        except Exception as e:
            print("Full coord main failed:", e)

        # Try a second full-length start if time remains and it is distinct.
        now = time.monotonic()
        if len(scored_full) > 1 and now < final_full_deadline - 150.0:
            second_score, second_a = scored_full[1]
            if second_score < best_score + 0.03 and not np.allclose(second_a, full0_a, rtol=0, atol=1e-12):
                sec_deadline = min(final_full_deadline, now + (final_full_deadline - now) * 0.75)
                if sec_deadline > now + 60.0:
                    try:
                        sc2, a2 = coord_descent(
                            second_a,
                            sec_deadline,
                            top_k=min(70, max(32, full_n // 72)),
                            grid_points=13,
                            golden_iters=11,
                            recompute_every=10,
                            rng=rng,
                        )
                        if sc2 < best_score:
                            best_score = sc2
                            best_a = a2.copy()
                            print(f"Full coord second: score={best_score:.9f}, n={best_a.size}")
                    except Exception as e2:
                        print("Full coord second failed:", e2)

    # ------------------------------------------------------------------
    # Final polish: one short high-beta LSE L-BFGS stage, then coordinate.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < hard - 40.0:
        finish_a = best_a
        if finish_a.size < max(2048, full_n // 2):
            finish_a = log_resample(finish_a, full_n)
        try:
            a_safe = np.maximum(finish_a, 1.0e-300)
            z = np.log(a_safe)
            z = z - float(z.mean())
            z = np.clip(z, -40.0, 40.0)
            z_deadline = min(hard - 30.0, now + 55.0)
            z = run_z_lbfgs(z, [8000.0, 25000.0], z_deadline, stage_time=25.0, maxfun=12000)
            a_fin = softmax_from_z(z)
            sc_fin = compute_score(a_fin)
            if sc_fin < best_score:
                best_score = sc_fin
                best_a = a_fin.copy()
        except Exception:
            pass

        now = time.monotonic()
        if now < hard - 8.0:
            try:
                sc_fin2, a_fin2 = coord_descent(
                    best_a,
                    hard - 3.0,
                    top_k=min(90, max(40, full_n // 56)),
                    grid_points=13,
                    golden_iters=11,
                    recompute_every=6,
                    rng=rng,
                )
                if sc_fin2 < best_score:
                    best_score = sc_fin2
                    best_a = a_fin2.copy()
                final_a = a_fin2
            except Exception:
                final_a = best_a
        else:
            final_a = best_a
    else:
        final_a = best_a

    # ------------------------------------------------------------------
    # Return a safely scaled list and verify with the provided evaluator.
    # ------------------------------------------------------------------
    out_arr = np.asarray(final_a if final_a is not None else best_a, dtype=np.float64).ravel()
    if out_arr.size < 16:
        out_arr = np.ones(64, dtype=np.float64)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    ss = float(out_arr.sum())
    if ss <= 1.0e-12:
        out_arr = np.ones(64, dtype=np.float64)
        ss = 64.0
    # Scale to sum 10 for robustness; objective is scale invariant.
    out_arr = out_arr / ss
    out_arr = out_arr * 10.0
    out_list = [float(max(0.0, min(1000.0, x))) for x in out_arr]

    verified = np.inf
    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = np.inf

    if not np.isfinite(verified):
        # Fallback to unscaled normalized sequence.
        out_arr = np.asarray(best_a if best_a is not None else out_arr, dtype=np.float64)
        out_arr = np.clip(out_arr, 0.0, 1000.0)
        ss = float(out_arr.sum())
        if ss <= 1.0e-12:
            out_arr = np.ones(64, dtype=np.float64)
        else:
            out_arr = out_arr / ss
        out_list = [float(x) for x in out_arr]
        try:
            verified = float(evaluate_sequence(out_list))
        except Exception:
            verified = np.inf

    if not np.isfinite(verified):
        # Last-resort valid sequence.
        out_list = [1.0 / 64.0] * 64
        try:
            verified = float(evaluate_sequence(out_list))
        except Exception:
            verified = np.inf

    elapsed = time.monotonic() - t0
    print(f"FINAL internal={best_score:.9f}, verified={verified:.9f}, n={len(out_list)}, t={elapsed:.1f}s")
    return out_list
```
