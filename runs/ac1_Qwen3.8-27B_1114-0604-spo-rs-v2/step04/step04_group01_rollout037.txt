"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: the ramp a[i] proportional to (i+1) gives score ~4/3 for large n,
far below the previous best of ~1.5068. We start from the ramp and use
LSE-smoothed gradient descent with continuation to find the local optimum,
then apply subgradient polish. Multiple power-ramp initializations at various
lengths provide diversity.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute h_i = sum_k w_k * a[k-i] for i=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    """
    LSE-smooth objective: 2 * LSE_beta(d) where d_k = n*c_k/S^2.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)

    score = 2.0 * L
    grad = 4.0 * float(n) * h / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    if S > 0:
        grad = grad / S
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun):
    """Run L-BFGS-B on the LSE objective."""
    a = normalize_a(a0)
    n = len(a)

    def obj(x):
        return lse_score_grad(x, beta)

    rem = float(deadline - time.monotonic())
    if rem < 5.0:
        return exact_score_a(a), a

    est_time = 0.02 * max(1.0, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    cap = int((0.85 * rem) / max(est_time, 0.005))
    maxfun = int(min(maxfun, max(64, cap)))
    maxiter = max(30, maxfun // 2)

    bounds = [(0.0, 100.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(
            obj, a, method="L-BFGS-B", jac=True,
            bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m)
    w[idx] = 1.0
    return w / k


def subgrad_polish(a, max_iters=30, deadline=None):
    """Exact subgradient method: push down the active peaks."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        n = len(a)
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        d = float(n) * c / (S * S)
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        best_step_sc = best_sc
        best_step_a = best_a

        for k in (1, 2, 3, 5, 8):
            w = active_weights(d, k)
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            for mult in (0.01, 0.03, 0.1, 0.3, 0.8):
                if deadline is not None and time.monotonic() > deadline:
                    break
                eta = mult / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < best_step_sc:
                    best_step_sc = sc
                    best_step_a = q.copy()

        if best_step_a is best_a and best_step_sc >= best_sc:
            break
        a = best_step_a

    return float(best_sc), best_a


def make_ramp(n, power=1.0):
    """Create a power-ramp sequence: a[i] = (i+1)^power, normalized to sum=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return normalize_a(a)


def make_variations(n, power):
    """Create a slight variation of the power-ramp to help escape local minima."""
    a = make_ramp(n, power)
    rng = np.random.default_rng(12345 + int(power * 100))
    # Add small smooth perturbation
    x = np.linspace(0, 1, n)
    pert = 0.05 * rng.uniform() * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2 * np.pi))
    a = a * (1.0 + pert)
    return normalize_a(a)


def quick_optimize_ramp(a, deadline):
    """Run LSE continuation on a ramp-like starting point."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    betas = [100, 300, 800, 2000, 5000, 12000, 25000]
    for beta in betas:
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        est = 0.02 * max(1.0, n / 8192.0)
        maxfun = int(max(64, (0.8 * rem) / max(est, 0.005)))
        maxfun = min(maxfun, 2000)

        sc, a2 = run_lbfgs(a, float(beta), deadline, maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    return float(best_sc), best_a

    # Try to load parent sequence

    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        pass

    # Define search configurations: (length, power, label)
    configs = [
        (8192, 1.0, "ramp_8192"),
        (8192, 1.5, "pow15_8192"),
        (16384, 1.0, "ramp_16384"),
        (4096, 1.0, "ramp_4096"),
        (8192, 0.75, "pow75_8192"),
        (16384, 1.2, "pow12_16384"),
    ]

    # If parent exists, resample it as an additional config
    if parent is not None:
        configs.append((min(len(parent), 8192), 1.0, "parent_resamp"))

    all_results = []  # (score, sequence, label)

    # Allocate time: main search gets 80%, polish gets 20%
    main_deadline = t_start + 0.78 * (float(budget_s) - 30.0)
    polish_deadline = deadline

    # Phase 1: Evaluate ramps quickly to find the best starting point
    print("=== Phase 1: Evaluate initial candidates ===")
    for n, power, label in configs[:4]:  # Quick eval of top 4
        if parent is not None and "parent" in label:
            a = normalize_a(resample_seq(parent, n))
        else:
            a = make_ramp(n, power)
        sc = exact_score_a(a)
        all_results.append((sc, a.copy(), label))
        print("  {} n={} power={}: score={:.8f}".format(label, n, power, sc))

    # Sort by score
    all_results.sort(key=lambda x: x[0])
    best_sc = all_results[0][0]
    best_seq = all_results[0][1].copy()
    print("Best initial: {} score={:.8f}".format(all_results[0][2], best_sc))

    # Phase 2: Optimize the top configurations
    print("=== Phase 2: LSE gradient optimization ===")
    # Allocate time proportional to number of configs
    n_configs = min(5, len(configs))
    time_per_config = (main_deadline - t_start) / n_configs

    for idx in range(n_configs):
        n, power, label = configs[idx]
        if "parent" in label and parent is not None:
            a = normalize_a(resample_seq(parent, n))
        else:
            a = make_ramp(n, power)

        # Start from the best found so far if it's similar length
        if best_sc < 1.4 and idx > 0:
            if abs(len(best_seq) - n) < n * 0.5:
                a = resample_seq(best_seq, n)
                a = normalize_a(a)
                # Add small perturbation for diversity
                a = a * (1.0 + 0.02 * rng.standard_normal(n))
                a = normalize_a(a)

        sub_dead = min(t_start + (idx + 1) * time_per_config, main_deadline)
        sc, seq = quick_optimize_ramp(a, sub_dead)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("  NEW BEST: {} score={:.8f}".format(label, best_sc))
        else:
            print("  {} score={:.8f} (best remains {:.8f})".format(label, sc, best_sc))

    # Phase 3: Subgradient polish
    print("=== Phase 3: Subgradient polish ===")
    polish_dead = min(deadline - 60.0, t_start + 0.92 * (float(budget_s) - 30.0))
    sc, seq = subgrad_polish(best_seq, max_iters=40, deadline=polish_dead)
    if sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("  After subgrad: {:.8f}".format(best_sc))

    # Phase 4: Final high-beta LSE polish
    print("=== Phase 4: Final LSE polish ===")
    rem = float(deadline - 10.0 - time.monotonic())
    if rem > 15.0:
        for beta in [30000, 50000]:
            rem = float(deadline - 10.0 - time.monotonic())
            if rem < 10.0:
                break
            est = 0.02 * max(1.0, len(best_seq) / 8192.0)
            maxfun = int(max(32, (0.7 * rem) / max(est, 0.005)))
            maxfun = min(maxfun, 500)
            sc, seq = run_lbfgs(best_seq, float(beta), deadline, maxfun)
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("  Final LSE beta={}: {:.8f}".format(beta, best_sc))

    # Phase 5: Final subgradient cleanup
    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=15, deadline=deadline - 15.0)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("  Final subgrad: {:.8f}".format(best_sc))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = make_ramp(1024, 1.0)
        out_arr = out_arr * 0.5  # keep values small to avoid issues
    else:
        # Scale so values stay in [0, 1000] but maintain the shape
        # Score is scale-invariant, so any scaling works
        out_arr = out_arr / S * 1.0  # normalize to sum=1

    out = [float(x) for x in out_arr]

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("=== FINAL ===")
    print("Internal best={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start
    ))
    return out


def resample_seq(a, n_new):
    """Resample a non-negative sequence to a new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(a.size)
    if m == 0:
        return np.ones(n_new, dtype=np.float64) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)
