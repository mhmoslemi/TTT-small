"""
Hybrid optimizer for non-negative flat-autocorrelation sequences.
Combines: (1) LP-alternating with improved FISTA, (2) exact LP via cvxpy for
moderate n, (3) LSE-smoothed L-BFGS polish, (4) coordinate descent refinement,
(5) simulated annealing for local minimum escape.

import numpy as np
from scipy.optimize import minimize
import time
import sys

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        L_val = c_max
        p = np.ones(len(c)) / len(c)
    else:
        L_val = c_max + (z.max() - z_shift.max() + np.log(exp_sum)) / beta
        p = exp_z / exp_sum
    score = 2.0 * n * L_val / (S * S)
    rev_a = a[::-1]
    M2 = n + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def lse_objective_wrapper(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def lp_solve_fista_improved(a, M, n_fista_iters=5000, beta_start=3.0, beta_end=80.0,
                            focus_top_k=100):
    """
    Improved FISTA for LP: max sum(x) s.t. conv(a,x) <= M, x >= 0.
    Uses adaptive beta and constraint focusing.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_fista_iters):
        frac = min(1.0, it / (n_fista_iters * 0.4))
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        c_max = float(c.max())

        if c_max <= M:
            # Feasible: try to increase sum
            z = beta * (c - M)
            z_max = float(z.max())
            exp_z = np.exp(z - z_max)
            exp_sum = float(exp_z.sum())
            if exp_sum < 1e-300:
                p = np.ones(len(c)) / len(c)
            else:
                p = exp_z / exp_sum
        else:
            # Infeasible: focus on most violated constraints
            z = beta * (c - M)
            z_max = float(z.max())
            exp_z = np.exp(z - z_max)
            exp_sum = float(exp_z.sum())
            if exp_sum < 1e-300:
                p = np.ones(len(c)) / len(c)
            else:
                p = exp_z / exp_sum
            # Boost weight on top violated constraints
            if focus_top_k > 0 and len(c) > focus_top_k:
                top_idx = np.argsort(c)[::-1][:focus_top_k]
                p_boost = p.copy()
                boost = 1.0 + 5.0 * (1.0 - frac)
                p_boost[top_idx] *= boost
                p_boost /= p_boost.sum()
                p = p_boost

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x

                #              = 0 otherwise.
                # But wait, this double-counts when i=j AND i=k-j, i.e., j=k-j, k=2j.
                # In that case, the term a[j]*a[j] appears only once in the sum (at i=j = k-j).
                # So dc[2j]/da[j] = 2*a[j] (the derivative of a[j]^2 with respect to a[j] is 2*a[j]).
                # And by the formula above: 2*a[2j-j] = 2*a[j]. Consistent!
                
                # So: contrib[k] = 2*a[k-j] if 0 <= k-j < n, else 0.
                # In other words: contrib[k] = 2*a[k-j] for k in [j, j+n-1].
                
                for k in range(j, min(j + n, 2 * n - 1)):
                    contrib[k] = 2.0 * a[k - j]
                
                c_base = c_full - aj * contrib
                
                # Minimize max_k (c_base[k] + t * contrib[k]) over t >= 0
                # Use bisection: find t* where the max is minimized.
                # The optimal t satisfies: at the optimum, the max of (c_base + t*contrib) 
                # is achieved at at least two points (or one point with zero slope).
                # Simple approach: try t=0, t=aj, and a few values in between.
                
                t_candidates = [0.0, aj, aj * 0.5, aj * 0.75, aj * 1.25, aj * 0.1]
                best_t = aj
                best_max = np.inf
                for t_cand in t_candidates:
                    if t_cand < 0:
                        t_cand = 0.0
                    max_val = float(np.max(c_base + t_cand * contrib))
                    if max_val < best_max:
                        best_max = max_val
                        best_t = t_cand
                
                a[j] = best_t
            
            # Update c_full
            c_full = self_conv(a)
            S = float(a.sum())
            if S < 1e-12:
                break
            sc = 2.0 * n * float(c_full.max()) / (S * S)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a

def simulated_annealing_step(a, temperature, rng):
    """Perturb the sequence and accept with probability based on score change."""
    n = len(a)
    a_pert = a.copy()
    
    strategy = rng.integers(4)
    if strategy == 0:
        # Gaussian perturbation
        scale = temperature * 0.1
        a_pert = a_pert * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        # Swap two segments
        seg_len = max(1, n // 50)
        i = rng.integers(0, n - seg_len)
        j = rng.integers(0, n - seg_len)
        if abs(i - j) > seg_len:
            a_pert[i:i+seg_len], a_pert[j:j+seg_len] = a_pert[j:j+seg_len].copy(), a_pert[i:i+seg_len].copy()
    elif strategy == 2:
        # Smooth perturbation
        freq = 1.0 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        x = np.arange(n) / n
        amp = temperature * 0.15
        a_pert = a_pert * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    else:
        # Targeted: reduce peak contributors
        c = self_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a_pert[j_start:j_end + 1].copy()
        contrib = seg[::-1]
        top_count = max(3, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        for li in top_idx:
            a_pert[j_start + li] *= (1.0 - temperature * 0.05)
        total_red = float(np.sum(a[j_start + top_idx]) - np.sum(a_pert[j_start + top_idx]))
        if total_red > 0:
            bottom_idx = np.argsort(contrib)[:top_count]
            a_pert[j_start + bottom_idx] += total_red / top_count
    
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = float(a_pert.sum())
    if S < 1e-12:
        return a
    return a_pert / S

def error_reduction(a, n_iters=100, alpha=0.7):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid search: LP-alternating + LSE-LBFGS + coordinate descent + SA.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)", flush=True)
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}", flush=True)

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 5000, 6962, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}", flush=True)

    # ================================================================
    # Phase 1: Quick exploration with error reduction (10% time)
    # ================================================================
    phase1_deadline = t_start + 0.10 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 1: Exploration (t_budget={phase1_deadline - t_start:.0f}s) ---", flush=True)

    x = None
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        xx = np.arange(n, dtype=np.float64) / n
        inits = [
            np.ones(n) * 0.5,
            0.3 + 0.7 * np.cos(np.pi * xx) ** 2,
            0.5 + 0.3 * np.cos(2 * np.pi * xx),
            np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
            0.2 + 0.8 * (2 * xx - 1) ** 2,
        ]
        if parent is not None:
            if len(parent) == n:
                inits.insert(0, parent.copy())
            else:
                inits.insert(0, np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent))

        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=80, alpha=0.75)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best={best_score:.8f}", flush=True)

    # ================================================================
    # Phase 2: LP-Alternating (55% time) - MAIN PHASE
    # ================================================================
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 2: LP-Alternating (t_budget={phase2_deadline - t_start:.0f}s) ---", flush=True)

    # Work on the best length
    n_main = len(best_a)
    
    # Try LP alternating on best sequence
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Progressive refinement
    configs = [
        (3000, 0.5, 0.05),
        (5000, 0.4, 0.03),
        (7000, 0.3, 0.02),
        (10000, 0.2, 0.01),
    ]

    for ci, (fista_it, t_s, t_m) in enumerate(configs):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP round {ci}: fista={fista_it}, t=[{t_s},{t_m}], remaining={remaining:.0f}s", flush=True)
        
        a_out, sc_out = lp_alternating_improved(
            a_work, max_outer_iters=100, fista_iters=fista_it,
            t_start=t_s, t_min=t_m, time_limit=phase2_deadline - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{ci}")
        a_work = a_out.copy()

    # Try other lengths
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_deadline - 60:
            break
        print(f"  LP alt: n={n}", flush=True)
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating_improved(
            a_test, max_outer_iters=40, fista_iters=4000,
            t_start=0.4, t_min=0.03, time_limit=phase2_deadline - 15
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best={best_score:.8f}", flush=True)

    # ================================================================
    # Phase 3: LSE L-BFGS Refinement (20% time)
    # ================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 3: LSE L-BFGS (t_budget={phase3_deadline - t_start:.0f}s) ---", flush=True)

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 400, 800]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 10:
            break
        print(f"  L-BFGS beta={beta}", flush=True)
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=4000, maxfun=150000)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best={best_score:.8f}", flush=True)

    # ================================================================
    # Phase 4: Coordinate Descent + SA + Final Polish (10% time)
    # ================================================================
    print(f"\n--- Phase 4: CD + SA + Final Polish ---", flush=True)

    n_best = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Coordinate descent
    if time.monotonic() < deadline - 120:
        print("  Coordinate descent...", flush=True)
        a_cd = coordinate_descent(a_work, n_iters=300, time_limit=deadline - 120)
        sc = compute_score(a_cd)
        report(sc, a_cd, "CD")

    # Simulated annealing
    if time.monotonic() < deadline - 90:
        print("  Simulated annealing...", flush=True)
        a_sa = best_a.copy()
        S = float(a_sa.sum())
        if S > 1e-12:
            a_sa = a_sa / S
        sa_best = a_sa.copy()
        sa_best_sc = compute_score(sa_best)
        temp = 1.0
        for sa_it in range(200):
            if time.monotonic() > deadline - 60:
                break
            a_try = simulated_annealing_step(sa_best, temp, rng)
            sc_try = compute_score(a_try)
            if sc_try < sa_best_sc:
                sa_best = a_try.copy()
                sa_best_sc = sc_try
                report(sc_try, a_try, f"SA it={sa_it}")
            else:
                # Accept with probability (always accept in first few iters)
                if sa_it < 10 or np.log(rng.random()) < (sa_best_sc - sc_try) / (temp + 1e-10):
                    sa_best = a_try.copy()
            temp *= 0.97

    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        print("  Final LBFGS beta=2000...", flush=True)
        sc, a_opt, _ = run_lbfgs(a_norm, 2000, maxiter=4000, maxfun=80000)
        report(sc, a_opt, "final LBFGS")

    # Final LP polish
    if time.monotonic() < deadline - 20:
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        print("  Final LP polish...", flush=True)
        a_final, sc_final = lp_alternating_improved(
            a_final_in, max_outer_iters=15, fista_iters=5000,
            t_start=0.1, t_min=0.01, time_limit=deadline - 5
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # One more coordinate descent pass
    if time.monotonic() < deadline - 10:
        a_cd2 = coordinate_descent(best_a, n_iters=100, time_limit=deadline - 5)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "final CD")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return out
