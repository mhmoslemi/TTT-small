def lps_bound(a):

    n = len(a)

    A = np.fft.rfft(a, M_fft)

def lp_solve_fista_bt(a, M, n_fista_iters=5000, beta_start=10.0, beta_end=100.0,
                      x0=None, time_limit=None):

    Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA with backtracking on the LSE-smoothed surrogate.

        S = x.sum()

    t_mom = 1.0

    cum_len = p_len + n  # for sliding window

    lps_quad = lps_bound(a)

    for it in range(n_fista_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        beta = beta_start + (beta_end - beta_start) * frac

        # Compute c = conv(a, x)
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(p_len) / p_len
        else:
            p = exp_z / exp_sum

        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

def minimax_multiplicative(a, n_rounds=30, p_power=8.0, n_inner=50, gamma=0.1,
                           time_limit=None):
    """
    Minimax multiplicative weight update.
    At each outer step:
    1. Compute c = conv(a,a)
    2. Set mu_k = c_k^p / sum(c_j^p)  (concentrate on peaks)
    3. Do several inner gradient steps on sum_k mu_k * c_k with multiplicative update
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = self_conv(a)
        c_pos = np.maximum(c, 1e-15)
        mu = c_pos ** p_power
        mu_sum = mu.sum()
        if mu_sum < 1e-300:
            mu = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            mu = mu / mu_sum

        # Inner loop: multiplicative gradient descent on sum_k mu_k * c_k
        for inner in range(n_inner):
            c = self_conv(a)
            # Gradient: G_i = 2 * (M_mu @ a)_i - a_i * mu_{2i}
            # (M_mu @ a)_i = sum_j a_j * mu_{i+j}
            rev_a = a[::-1]
            conv_res = np.convolve(rev_a, mu, mode='full')  # length 3n-2
            Ma = conv_res[n - 1:2 * n - 1]

            # Diagonal correction: for k=2i, the self-term is overcounted
            # mu_{2i} is defined for 2i in [0, 2n-2]
            diag_idx = 2 * np.arange(n)
            diag_val = np.zeros(n)
            valid = diag_idx < 2 * n - 1
            diag_val[valid] = mu[diag_idx[valid]]

            G = 2.0 * Ma - a * diag_val

            # Multiplicative update: a_i <- a_i * exp(-gamma * G_i / max(G))
            G_max = float(np.max(np.abs(G)))
            if G_max < 1e-15:
                break
            log_update = -gamma * G / G_max
            # Clip for numerical stability
            log_update = np.clip(log_update, -5.0, 5.0)
            a = a * np.exp(log_update)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                a = best_a.copy()
                break
            a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Increase p_power to focus more on peaks
        p_power *= 1.3

    return best_a


def error_reduction(a, n_iters=150, alpha=0.8):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        # Threshold and reduce peaks
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10 * np.max(np.abs(A))] = 0.0

        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 5))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=4000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

    phase1_deadline = t_start + 0.12 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 1: Exploration ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 20:
            break
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, int(remaining / 15.0))
        inits = inits[:max_inits]
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=80, alpha=0.8)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LP-based alternating (55% time) - MAIN
    # ============================================================
    phase2_deadline = t_start + 0.67 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 2: LP Alternating ---")

    # Focus on top 2 lengths
    n_main = len(best_a)
    if parent is not None and len(parent) in lengths:
        n_main = len(parent)

    a_work = best_a.copy()
    if len(a_work) != n_main:
        a_work = np.interp(np.linspace(0, 1, n_main),
                           np.linspace(0, 1, len(best_a)), best_a)
    a_work = np.maximum(a_work, 0.0)
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP alternating
    configs = [
        (3000, 0.5, 0.05, 15),
        (4000, 0.4, 0.03, 20),
        (6000, 0.3, 0.02, 25),
        (8000, 0.2, 0.01, 30),
    ]

    for fi, (fista_it, t_s, t_m, max_outer) in enumerate(configs):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - t_start
        if remaining < 60:
            break
        print(f"  LP round {fi}: n={n_main}, fista={fista_it}, t=[{t_s},{t_m}]")

        a_out, sc_out = lp_alternating(
            a_work, max_outer_iters=max_outer, fista_iters=fista_it,
            t_start=t_s, t_min=t_m, time_limit=phase2_deadline - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    # Try second-best length if time permits
    for n in lengths:
        if n == n_main or time.monotonic() > phase2_deadline - 90:
            continue
        print(f"  LP alt: n={n}")
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating(
            a_test, max_outer_iters=20, fista_iters=3000,
            t_start=0.3, t_min=0.03, time_limit=phase2_deadline - 15
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Minimax multiplicative refinement (12% time)
    # ============================================================
    phase3_deadline = t_start + 0.79 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 3: Minimax Multiplicative ---")

    a_mm_in = best_a.copy()
    S = float(a_mm_in.sum())
    if S > 1e-12:
        a_mm_in = a_mm_in / S

    a_mm = minimax_multiplicative(a_mm_in, n_rounds=20, p_power=6.0,
                                  n_inner=40, gamma=0.15,
                                  time_limit=phase3_deadline)
    sc = compute_score(a_mm)
    report(sc, a_mm, "minimax")

    # Also try on a perturbed version
    if time.monotonic() < phase3_deadline - 30:
        a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(len(best_a)))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = a_pert.sum()
        if S > 1e-12:
            a_pert = a_pert / S
        a_mm2 = minimax_multiplicative(a_pert, n_rounds=15, p_power=10.0,
                                       n_inner=30, gamma=0.1,
                                       time_limit=phase3_deadline)
        sc = compute_score(a_mm2)
        report(sc, a_mm2, "minimax pert")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: LSE-LBFGS polish (15% time)
    # ============================================================
    phase4_deadline = t_start + 0.94 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 4: LBFGS Polish ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [50, 150, 400, 1000, 2000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - t_start
        if remaining < 30:
            break
        maxit = min(4000, int(remaining / 3))
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()

    # Local perturbation + LBFGS
    if time.monotonic() < phase4_deadline - 20:
        for trial in range(3):
            if time.monotonic() > phase4_deadline:
                break
            scale = 0.005 + 0.01 * rng.random()
            a_p = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S > 1e-12:
                a_p = a_p / S
                sc, a_opt, _ = run_lbfgs(a_p, 500, maxiter=2000, maxfun=50000)
                report(sc, a_opt, f"local #{trial}")

    # Final LP polish
    if time.monotonic() < deadline - 60:
        print("  Final LP...")
        a_fin = best_a.copy()
        S = float(a_fin.sum())
        if S > 1e-12:
            a_fin = a_fin / S
        a_fin, sc_fin = lp_alternating(
            a_fin, max_outer_iters=15, fista_iters=5000,
            t_start=0.15, t_min=0.01, time_limit=deadline - 20
        )
        sc = compute_score(a_fin)
        report(sc, a_fin, "final LP")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
