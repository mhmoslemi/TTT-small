Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses LSE-smoothed Adam optimization with multiple initializations,
adaptive beta scheduling, and coordinate descent refinement.
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient: dL/da_j = (2/beta) * sum_k p[k] * a[k-j] * 1[0<=k-j<n]
    # = (2/beta) * r_j where r_j = sum_m a[m] * p[m+j]
    # Use FFT: r[:n] = irfft(rfft(p, L) * rfft(a, L))[:n]
    L_fft = 1 << (3 * n - 2).bit_length()

    A = np.fft.rfft(a, L_fft)
    r = np.fft.irfft(P * A, L_fft)[:n]
    dL_da = 2.0 * r / beta
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def adam_optimize(a_init, time_limit, t_start, beta_start=20, beta_end=500, 
                  lr=0.01, beta_schedule='linear', eval_interval=500):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    total_time = t0 - t_start + time_limit  # absolute deadline
    
    while time.monotonic() < total_time:
        step += 1
        # Beta schedule
        frac = min(1.0, step / 50000)  # ramp up over 50k steps
        if beta_schedule == 'linear':
            beta = beta_start + (beta_end - beta_start) * frac
        elif beta_schedule == 'log':
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end
        
        score, grad = lse_score_grad(a, beta)
        
        # Adam update
        lr_t = lr / (1.0 + 1e-5 * step)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** step)
        v_hat = v / (1 - b2 ** step)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: non-negative and normalize
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        else:
            a[:] = 1.0 / n
            break
        
        # Periodic exact evaluation
        if step % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a

"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses LSE-smoothed Adam optimization with adaptive beta scheduling,
multiple initializations, perturbation restarts, and coordinate descent.
Key insight: the score is scale-invariant, so we normalize to sum=1.
The LSE approximation of max(b) provides a smooth surrogate whose gradient
is computable via FFT-based cross-correlation.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # r_j = sum_m a[m] * p[m+j] for j=0..n-1
    # FFT correlation: r[:n] = irfft(rfft(p, L) * rfft(a, L))[:n]
    L_fft = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    r = np.fft.irfft(P * A, L_fft)[:n]
    dL_da = 2.0 * r / beta
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def exact_subgrad(a):
    """Exact score and subgradient (using argmax of convolution)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    # Subgradient: active at k_star
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse_optimize(a_init, time_limit, beta_start=20.0, beta_end=500.0,
                      lr=0.01, ramp_steps=30000, eval_every=2000):
    """
    Adam optimizer with LSE-smoothed gradient.
    Works in normalized space (sum a = 1).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + time_limit
    step = 0

    while time.monotonic() < deadline:
        step += 1
        frac = min(1.0, step / max(1, ramp_steps))
        beta = beta_start + (beta_end - beta_start) * frac

        score, grad = lse_score_grad(a, beta)

        lr_t = lr * (0.5 ** (step // 50000))  # halve lr every 50k steps
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** step)
        v_hat = v / (1.0 - b2 ** step)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        else:
            a[:] = 1.0 / n
            break

        if step % eval_every == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_score, best_a


def coord_descent_refine(a, time_limit, top_k=40):
    """
    Coordinate descent: reduce active coords (contributing to max),
    increase inactive coords.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    deadline = time.monotonic() + time_limit
    rng = np.random.default_rng(12345)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        c = fast_conv(a)
        S = a.sum()
        k_star = int(c.argmax())

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        active = list(range(js, je + 1))

        improved = False
        # Try reducing active coordinates
        for j in active[:top_k]:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-15:
                continue
            best_local = best_score
            best_val = old
            for factor in [0.3, 0.5, 0.7, 0.85, 0.95]:
                a[j] = old * factor
                s = a.sum()
                if s > 1e-12:
                    sc = compute_score(a / s)
                    if sc < best_local:
                        best_local = sc
                        best_val = old * factor
                        improved = True
            a[j] = best_val
            s = a.sum()
            if s > 1e-12:
                a = a / s
                if best_local < best_score:
                    best_score = best_local

        # Try increasing non-active coordinates
        if time.monotonic() < deadline:
            non_active = [j for j in range(n) if j < js or j > je]
            if non_active:
                # Pick a few random non-active coords to increase
                num_try = min(15, len(non_active))
                chosen = non_active[rng.choice(len(non_active), size=num_try, replace=False)]
                for j in chosen:
                    if time.monotonic() > deadline:
                        break
                    old = a[j]
                    best_local = best_score
                    best_val = old
                    for factor in [2.0, 3.0, 5.0, 10.0]:
                        a[j] = old * factor
                        s = a.sum()
                        if s > 1e-12:
                            sc = compute_score(a / s)
                            if sc < best_local:
                                best_local = sc
                                best_val = old * factor
                                improved = True
                    a[j] = best_val
                    s = a.sum()
                    if s > 1e-12:
                        a = a / s
                        if best_local < best_score:
                            best_score = best_local

        if not improved and iteration > 50:
            # Try a random perturbation to escape local minimum
            perturb_idx = rng.choice(n, size=min(100, n // 10), replace=False)
            scale = 0.1 + 0.5 * rng.random()
            a[perturb_idx] *= (1.0 + scale * rng.standard_normal(len(perturb_idx)))
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            else:
                break

    return best_score, a


def lse_fine_tune(a, beta, n_steps, lr):
    """Short LSE optimization at a fixed high beta for fine-tuning."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + n_steps * 0.003  # rough time estimate

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        score, grad = lse_score_grad(a, beta)
        lr_t = lr / (1.0 + 1e-4 * t)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        else:
            break
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-smoothed Adam with adaptive beta, multiple initializations,
    perturbation restarts, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = float(budget_s) - 45.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    # Phase 1: Evaluate starting points
    print("=" * 60)
    print("Phase 1: Evaluate starting points")
    print("=" * 60)

    if parent is not None:
        sc = compute_score(parent)
        print(f"  Parent: n={len(parent)}, score={sc:.10f}")
        try_update(sc, parent, "parent")

    for n_try in [4096, 8192]:
        a_u = np.ones(n_try) / n_try
        sc = compute_score(a_u)
        print(f"  Uniform n={n_try}: score={sc:.10f}")
        try_update(sc, a_u, f"uniform n={n_try}")

    # Phase 2: Main LSE-Adam optimization (coarse to fine)
    print("=" * 60)
    print("Phase 2: LSE-Adam coarse-to-fine optimization")
    print("=" * 60)

    start_a = best_a.copy()

    # Coarse phase
    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 100:
        alloc = min(200.0, remaining * 0.15)
        print(f"  Coarse: alloc={alloc:.0f}s")
        sc, a_opt = adam_lse_optimize(start_a, alloc,
                                       beta_start=10.0, beta_end=50.0,
                                       lr=0.02, ramp_steps=10000)
        try_update(sc, a_opt, "coarse")
        start_a = a_opt

    # Medium phase
    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 100:
        alloc = min(400.0, remaining * 0.2)
        print(f"  Medium: alloc={alloc:.0f}s")
        sc, a_opt = adam_lse_optimize(start_a, alloc,
                                       beta_start=50.0, beta_end=300.0,
                                       lr=0.008, ramp_steps=20000)
        try_update(sc, a_opt, "medium")
        start_a = a_opt

    # Fine phase
    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 100:
        alloc = min(700.0, remaining * 0.35)
        print(f"  Fine: alloc={alloc:.0f}s")
        sc, a_opt = adam_lse_optimize(start_a, alloc,
                                       beta_start=300.0, beta_end=2000.0,
                                       lr=0.003, ramp_steps=30000)
        try_update(sc, a_opt, "fine")
        start_a = a_opt

    # Ultra-fine phase
    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 60:
        alloc = min(400.0, remaining * 0.25)
        print(f"  Ultra-fine: alloc={alloc:.0f}s")
        sc, a_opt = adam_lse_optimize(start_a, alloc,
                                       beta_start=2000.0, beta_end=8000.0,
                                       lr=0.001, ramp_steps=20000)
        try_update(sc, a_opt, "ultra-fine")
        start_a = a_opt

    # Phase 3: Perturbation restarts
    print("=" * 60)
    print("Phase 3: Perturbation restarts")
    print("=" * 60)

    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    n_restarts = 0
    while remaining > 80 and n_restarts < 4:
        n_restarts += 1
        # Different perturbation strategies
        if n_restarts == 1:
            noise_scale = 0.03
            a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        elif n_restarts == 2:
            noise_scale = 0.1
            a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        elif n_restarts == 3:
            # Block perturbation
            n = len(best_a)
            block_size = max(64, n // 32)
            a_pert = best_a.copy()
            for _ in range(n // block_size):
                start = int(rng.integers(0, n - block_size))
                a_pert[start:start + block_size] *= rng.uniform(0.5, 2.0)
        else:
            # Large random restart from uniform
            n = len(best_a)
            a_pert = np.ones(n) / n + 0.05 * rng.standard_normal(n)
            a_pert = np.clip(a_pert, 0.0, None)

        a_pert = np.clip(a_pert, 0.0, None)
        s = a_pert.sum()
        if s > 1e-12:
            a_pert = a_pert / s
        else:
            break

        alloc = min(180.0, remaining - 60)
        if alloc < 20:
            break
        print(f"  Restart {n_restarts}: alloc={alloc:.0f}s")
        sc, a_opt = adam_lse_optimize(a_pert, alloc,
                                       beta_start=50.0, beta_end=1000.0,
                                       lr=0.005, ramp_steps=15000)
        try_update(sc, a_opt, f"restart {n_restarts}")
        remaining = effective_budget - (time.monotonic() - t_start)

    # Phase 4: Coordinate descent refinement
    print("=" * 60)
    print("Phase 4: Coordinate descent refinement")
    print("=" * 60)

    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 40 and best_a is not None:
        alloc = min(300.0, remaining - 30)
        print(f"  Coord descent: alloc={alloc:.0f}s")
        t0 = time.monotonic()
        sc, a_opt = coord_descent_refine(best_a, alloc, top_k=40)
        try_update(sc, a_opt, "coord descent")
        print(f"  Done: {sc:.10f} ({time.monotonic() - t0:.1f}s)")

    # Phase 5: Very fine LSE at the end
    elapsed = time.monotonic() - t_start
    remaining = effective_budget - elapsed
    if remaining > 30 and best_a is not None:
        alloc = min(120.0, remaining - 20)
        print(f"  Final fine-tune: alloc={alloc:.0f}s")
        sc, a_opt = lse_fine_tune(best_a, beta=5000, n_steps=100000, lr=0.0005)
        try_update(sc, a_opt, "final fine-tune")

    # Final output
    if best_a is None:
        best_a = np.ones(1024) / 1024

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print("=" * 60)
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print("=" * 60)

    return out
