import numpy as np

def solve_lp(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: (a*g)_k <= M for k=0,...,2n-2
    # (a*g)_k = sum_j a[j] * g[k-j]
    # This is a Toeplitz matrix multiplication
    
    objective = cp.Maximize(cp.sum(g))
    
    constraints = []
    for k in range(2*n - 1):
        # (a*g)_k = sum_j a[j] * g[k-j] where j in [0,n-1] and k-j in [0,n-1]

    return np.clip(g.value, 0, None)

"""
Minimize 2*n*max(self-convolution)/(sum)^2 using LP-based iterative refinement.
The LP-based approach guarantees non-increasing peak at each iteration.
Combined with multiple starting points and n values.

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    return A

def solve_lp_direction(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, 
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    
    if res.status != 0:
        return None
    return res.x

def lp_refine(a, n_iter=50, time_limit_total=600):
    """Iterative LP-based refinement."""
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    # Normalize to sum = 1
    a = a / S
    
    t_start = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        if time.monotonic() - t_start > time_limit_total:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_direction(a, M, time_limit=30)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-12:
            break
        
        # Normalize g to sum = 1
        g = g / g.sum()
        
        # Line search for step size t
        best_t = 0.0
        best_new_M = M
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            new_a = (1-t)*a + t*g
            new_c = fast_conv(new_a)
            new_M = float(new_c.max())
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.001, 0.005]:
                new_a = (1-t)*a + t*g
                new_c = fast_conv(new_a)
                new_M = float(new_c.max())
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1-best_t)*a + best_t*g
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  LP iter {it}: M={M:.6f} -> {best_new_M:.6f}, t={best_t:.3f}, score={sc:.8f} ({lp_time:.1f}s)")
        else:
            print(f"  LP iter {it}: M={M:.6f} -> {best_new_M:.6f}, t={best_t:.3f}")
    
    return best_a

def fast_conv(a):

This is a totally unimodular matrix (it's a network matrix), so the LP has an extreme point solution that is "basic" in a simple way.

The greedy algorithm for this type of problem:
- We want to minimize sum(y) subject to window lower bounds.
- Process constraints from j = 0 to j = n-1.
- For constraint j, if the current window sum is insufficient, add the deficit to the RIGHTMOST position in the window (position j+n-1).

Why rightmost? Because position j+n-1 is shared with the fewest future constraints. Specifically, position j+n-1 is in the window of constraint j' only if j' <= j+n-1 <= j'+n-1, i.e., j-n+2 <= j' <= j. Since we've already processed constraints 0 to j-1, position j+n-1 will only be in the windows of constraints j, j+1, ..., j+n-2 (future constraints). Wait, that's a lot.

Actually, position k is in the window of constraint j iff j <= k <= j+n-1, i.e., k-n+1 <= j <= k. So position k is in the windows of constraints max(0, k-n+1) through min(n-1, k).

For position j+n-1: it's in the windows of constraints max(0, j) through min(n-1, j+n-1) = j through min(n-1, j+n-1).

For j < n-1: position j+n-1 is in windows j, j+1, ..., n-1. That's n-j constraints.
For j = n-1: position 2n-2 is in window n-1 only. That's 1 constraint.

So putting mass at position j+n-1 for constraint j is "wasteful" for small j (it helps many future constraints) and "efficient" for large j (it helps few future constraints).

Actually, for this specific problem (minimizing sum subject to lower bounds on windows), the optimal strategy is:
- Process from RIGHT to LEFT (j = n-1, n-2, ..., 0).
- For each constraint j, if the window sum is insufficient, add the deficit to the LEFTMOST position in the window that hasn't been "claimed" yet.

Or equivalently:
- Process from LEFT to RIGHT (j = 0, 1, ..., n-1).
- For each constraint j, if the window sum is insufficient, add the deficit to the RIGHTMOST position in the window.

Both should give the same optimal solution (by symmetry of the problem).

```python
def solve_dual_greedy(a, M):
    n = len(a)
    if np.any(a <= 0):
        return np.inf, np.zeros(2*n-1)
    
    y = np.zeros(2*n-1)
    window_sum = 0.0  # This is sum(y[j:j+n]) for the current j
    
    for j in range(n):
        if j == 0:
            # window_sum(0) = y[0] + y[1] + ... + y[n-1]
            # At this point, all y are 0, so window_sum = 0
            window_sum = 0.0
        else:
            # window_sum(j) = window_sum(j-1) - y[j-1] + y[j+n-1]
            window_sum = window_sum - y[j-1] + y[j+n-1]
        
        needed = 1.0 / a[j]
        if window_sum < needed:
            deficit = needed - window_sum
            # Add to rightmost position: y[j+n-1]
            y[j+n-1] += deficit
            window_sum += deficit
    
    return M * y.sum(), y
```
