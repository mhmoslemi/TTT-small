def subgrad_score(a):
    """Exact score and a subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(np.argmax(c))
    score = 2.0 * n * M / (S * S)
    # Subgradient of max at k_star:
    # d(c_{k_star})/d(a_j) = a_{k_star - j} [0<=k_star-j<n] + a_{j-k_star} [0<=j-k_star<n]
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
        # Also add the "reflected" contribution
        js2 = max(0, n - 1 - k_star)  # j - k_star >= 0 => j >= k_star
        je2 = min(n - 1, k_star + n - 1)  # j - k_star <= n-1 => j <= k_star + n-1
        # Actually: d(c_k)/d(a_j) = a_{k-j} (if 0<=k-j<n) + a_{j-k} (if 0<=j-k<n)
        # First term: 0<=k-j<n => k-n+1<=j<=k => j in [max(0,k-n+1), min(n-1,k)]
        # Second term: 0<=j-k<n => k<=j<=k+n-1 => j in [max(0,k), min(n-1,k+n-1)]
        # These can overlap at j=k (if k in [0,n-1])
        js2 = max(0, k_star)
        je2 = min(n - 1, k_star + n - 1)
        if je2 >= js2:
            idx2 = np.arange(js2, je2 + 1) - k_star
            sg[js2:je2 + 1] += a[idx2]
    
    S2 = S * S
    S3 = S2 * S
    grad_score = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad_score


def adam_optimize(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start, pal=False):
    """Adam optimization with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            
            lr_t = lr / (1.0 + 1e-5 * t)
            if pal:
                score, grad = lse_grad_pal(a, beta)
            else:
                score, grad = lse_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)
            
            s = a.sum()
            if s > 1e-12:
                a = a / s
            
            if (bi * steps_per_beta + t) % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

    if pal:
        def obj(x):
            xc = np.clip(x, 0.0, 1e10)
            sc, gr = lse_grad_pal(xc, beta)
            return sc, gr
    else:
        def obj(x):
            xc = np.clip(x, 0.0, 1e10)
            sc, gr = lse_grad(xc, beta)
            return sc, gr

                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)

def peak_equalization(a, n_iters=100, lr=0.1):
    """
    Iteratively identify the top peaks in the convolution and try to equalize them.
    Uses a projected subgradient approach targeting the active set of peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        S = 1.0  # normalized
        score = 2.0 * n * M
        
        # Find all "active" peaks (within tolerance of max)
        tol = 0.01 * M
        active = np.where(c >= M - tol)[0]
        
        # Compute subgradient: average of subgradients at active peaks
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                sg[js:je + 1] += a[idx]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if je2 >= js2:
                idx2 = np.arange(js2, je2 + 1) - k
                sg[js2:je2 + 1] += a[idx2]
        
        sg /= len(active)
        
        # Gradient of score (w.r.t. a, normalized):
        # score = 2n * M, d(score)/d(a) = 2n * sg (since S=1, dM/da = sg)
        # But we also need to account for normalization constraint
        # Using Lagrange multiplier: grad = 2n * sg - lambda * ones
        # lambda = sum(2n * sg) / n = 2 * sum(sg) ... 
        # Actually for constrained optimization with sum(a)=1:
        # projected gradient = sg - mean(sg)
        grad_proj = sg - sg.mean()
        
        # Step
        a = a - lr * grad_proj
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a


def multiplicative_weight_peak(a, n_iters=200, lr=0.5):
    """
    Multiplicative weight update: reduce elements contributing to peaks,
    boost elements not contributing to peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        tol = 0.05 * M
        active = np.where(c >= M - tol)[0]
        
        # Compute total contribution of each element to active peaks
        contrib = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                contrib[js:je + 1] += a[idx]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if je2 >= js2:
                idx2 = np.arange(js2, je2 + 1) - k
                contrib[js2:je2 + 1] += a[idx2]
        
        # Normalized contribution
        contrib_norm = contrib / max(M, 1e-10)
        
        # Multiplicative update: reduce high-contributing elements
        a = a * np.exp(-lr * contrib_norm)
        a = np.clip(a, 1e-15, None)
        a = a / a.sum()
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def symmetrize(a):
    """Make sequence palindromic by symmetrization."""
    n = len(a)
    n_half = n // 2
    half = (a[:n_half] + a[::-1][:n_half]) / 2.0
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, [a[n_half]], half[::-1]])


def generate_inits(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian
    for sigma in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())
    
    # Sinusoidal
    for freq in [2, 4, 8]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())
    
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    k = max(3, n // 50)
    r_smooth = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r_smooth / r_smooth.sum())
    
    # Palindromic random
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.5
    k = max(3, half_n // 50)
    h_smooth = np.convolve(h, np.ones(k) / k, mode='same')
    pal = np.concatenate([h_smooth, h_smooth[::-1]])
    inits.append(pal / pal.sum())
    
    return [np.clip(x, 1e-10, 1000.0) for x in inits]

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    L = max(c) + (1/beta)*log(sum exp(beta*(c - max(c))))
    score = 2n*L/S^2
    grad_score_j = 2n * (gradL_j/S^2 - 2L/S^3)
    where gradL_j = 2 * sum_m a[m] * p[m+j] with p_k = softmax(beta*(c-max(c)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = np.log(exp_scaled.sum()) + max_scaled
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # gradL[j] = 2 * sum_m a[m] * p[m+j]
    # = 2 * (a_rev * p)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    M2 = 2 * n  # enough for a_rev (len n) convolved with p (len 2n-1)
    M2_fft = 1 << (n + (2 * n - 1) - 1 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:n + (2 * n - 1) - 1]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad_score = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad_score


def lse_grad_pal(half_a, beta):

    score, grad_full = lse_grad(a, beta)
    grad_half = grad_full[:n_half] + grad_full[::-1][:n_half]
    return score, grad_half


def subgrad_exact(a):
    """Exact score and subgradient at the maximizing lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(np.argmax(c))
    score = 2.0 * n * M / (S * S)
    # d(c_{k*})/d(a_j) = 2*a_{k*-j} if 0<=k*-j<n, else 0
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def subgrad_multi_peak(a, n_peaks=3):
    """Average subgradient over top n_peaks convolution lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    M = float(c.max())

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

def adam_optimize(x_init, beta_schedule, lr, n_steps, time_limit_s, t_start, pal=False):
    """Adam optimization with LSE smoothing."""
    n = len(x_init)
    x = x_init.astype(np.float64).copy()
    if x.sum() < 1e-12:
        x[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_x = x.copy()
    best_score = compute_score(np.concatenate([x, x[::-1]]) if pal else x)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                break
            lr_t = lr / (1.0 + 1e-5 * t)
            if pal:
                _, grad = lse_grad_pal(x, beta)
            else:
                _, grad = lse_grad(x, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            x = np.clip(x, 0.0, 1e10)
            
            s = x.sum()
            if s > 1e-12:
                x = x / s
        
        # Check score at end of each beta phase
        a_full = np.concatenate([x, x[::-1]]) if pal else x
        sc = compute_score(a_full)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    
    a_full = np.concatenate([x, x[::-1]]) if pal else x
    sc_final = compute_score(a_full)
    if sc_final < best_score:
        best_score = sc_final
        best_x = x.copy()
    
    return best_score, best_x


def lbfgs_optimize(x0, beta, maxiter=5000, maxfun=80000, pal=False):
    """L-BFGS-B with LSE smoothing."""
    n = len(x0)
    x0 = np.clip(x0.astype(np.float64), 0.0, 1000.0).copy()
    if x0.sum() < 1e-6:
        x0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    if pal:
        def obj_func(xx):
            xc = np.clip(xx, 0.0, 1e10)
            return lse_grad_pal(xc, beta)
    else:
        def obj_func(xx):
            xc = np.clip(xx, 0.0, 1e10)
            return lse_grad(xc, beta)
    
    res = minimize(obj_func, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    x_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([x_out, x_out[::-1]]) if pal else x_out
    exact_sc = compute_score(a_full)
    return exact_sc, x_out


def peak_equalization_fft(a, n_iters=500, lr=0.05, n_top=5):
    """
    FFT-accelerated peak equalization.
    Identifies top peaks and uses averaged subgradient to reduce them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    
    a_rev_full = np.zeros(2 * n - 1)
    a_rev_full[:n] = a[::-1]
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        # Find top peaks (within tolerance)
        tol = max(0.005 * M, 1e-10)
        active_mask = (c >= M - tol).astype(np.float64)
        # Weight by proximity to max
        weights = np.maximum(0, (c - (M - tol)) / tol)
        active_mask = active_mask * weights
        if active_mask.sum() == 0:
            active_mask[int(np.argmax(c))] = 1.0
        active_mask = active_mask / active_mask.sum()
        
        # Compute averaged subgradient via FFT:
        # sg[j] = 2 * sum_k active_mask[k] * a_{k-j} [0<=k-j<n]
        #       = 2 * (a_rev * active_mask)[n-1+j]
        M_fft = 1 << (n + (2 * n - 1) - 2).bit_length()
        AR = np.fft.rfft(a_rev_full, M_fft)
        PM = np.fft.rfft(active_mask, M_fft)
        conv_res = np.fft.irfft(AR * PM, M_fft)[:n + (2 * n - 1) - 1]
        sg = 2.0 * conv_res[n - 1:2 * n - 1]
        
        # Projected gradient (maintain sum=1 constraint)
        grad_proj = sg - sg.mean()
        
        # Adaptive step size
        step = lr / (1.0 + 0.01 * it)
        a = a - step * grad_proj
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def symmetrize(a):
    """Make sequence palindromic."""
    n = len(a)
    n_half = n // 2
    half = (a[:n_half] + a[::-1][:n_half]) / 2.0
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        mid = (a[n_half] + a[n_half]) / 2.0
        return np.concatenate([half, [mid], half[::-1]])

            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            update_count[0] += 1
            elapsed = time.monotonic() - t_start
            print(f"  [#{update_count[0]}] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    print(f"{'='*60}")
    print(f"Optimization started at {time.monotonic()-t_start:.1f}s")
    print(f"Deadline: {deadline - t_start:.1f}s")
    print(f"{'='*60}")
    
    # Determine working sequence
    if parent is not None:
        a_cur = parent.copy()
        n_work = len(a_cur)
        # Ensure even
        if n_work % 2 != 0:
            n_work += 1
            a_cur = resample_sequence(a_cur, n_work)
        print(f"Parent: n={n_work}, score={compute_score(a_cur):.8f}")
        try_update(compute_score(a_cur), a_cur, "parent")
    else:
        n_work = 4096
        a_cur = np.ones(n_work) / n_work
        print(f"No parent. Starting uniform n={n_work}, score={compute_score(a_cur):.8f}")
        try_update(compute_score(a_cur), a_cur, "uniform init")
    
    # Palindromize for easier optimization
    a_pal = symmetrize(a_cur)
    pal_score = compute_score(a_pal)
    try_update(pal_score, a_pal, "palindromized")
    
    n_half = n_work // 2
    half = (a_cur[:n_half] + a_cur[::-1][:n_half]) / 2.0
    half = half / half.sum()
    
    # === Phase 1: Quick Adam screening (10% time) ===
    print(f"\n--- Phase 1: Quick Adam (10%) ---")
    t1_budget = (deadline - t_start) * 0.10
    sc, h_opt = adam_optimize(half, [20, 100, 500], lr=0.01, n_steps=30000,
                               time_limit_s=t1_budget, t_start=time.monotonic(), pal=True)
    a_full = np.concatenate([h_opt, h_opt[::-1]])
    try_update(sc, a_full, "Adam-pal quick")
    half = h_opt
    
    # Also try non-pal
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sc, a_opt = adam_optimize(a_cur, [50, 300], lr=0.008, n_steps=20000,
                                   time_limit_s=min(30.0, remaining * 0.3),
                                   t_start=time.monotonic(), pal=False)
        try_update(sc, a_opt, "Adam-direct quick")
    
    # === Phase 2: Peak equalization (10% time) ===
    print(f"\n--- Phase 2: Peak equalization (10%) ---")
    remaining = deadline - time.monotonic()
    t2_budget = (deadline - t_start) * 0.10
    sc, a_pe = peak_equalization_fft(best_a, n_iters=2000, lr=0.05)
    try_update(sc, a_pe, "peak-eq 1")
    
    # Blend and re-equalize
    a_blend = 0.5 * best_a + 0.5 * a_pe
    a_blend = np.clip(a_blend, 0, None) / a_blend.sum()
    sc_b = compute_score(a_blend)
    if sc_b < best_score:
        try_update(sc_b, a_blend, "blend pe")
        sc, a_pe2 = peak_equalization_fft(a_blend, n_iters=1000, lr=0.03)
        try_update(sc, a_pe2, "peak-eq 2")
    
    # === Phase 3: MW update (10% time) ===
    print(f"\n--- Phase 3: Multiplicative weight (10%) ---")
    remaining = deadline - time.monotonic()
    sc, a_mw = multiplicative_weight_fft(best_a, n_iters=500, lr=0.3)
    try_update(sc, a_mw, "MW 1")
    
    for t_blend in [0.3, 0.5, 0.7]:
        a_b = (1 - t_blend) * best_a + t_blend * a_mw
        a_b = np.clip(a_b, 1e-10, None)
        a_b = a_b / a_b.sum()
        sc_b = compute_score(a_b)
        if sc_b < best_score:
            try_update(sc_b, a_b, f"MW blend={t_blend}")
    
    # === Phase 4: High-beta LSE Adam continuation (25% time) ===
    print(f"\n--- Phase 4: High-beta LSE Adam (25%) ---")
    n_half = len(best_a) // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()
    
    p4_budget = (deadline - t_start) * 0.25
    betas = [200, 1000, 5000, 20000, 50000, 100000, 200000]
    lrs = [0.003, 0.002, 0.001, 0.0007, 0.0004, 0.0003, 0.0002]
    
    per_beta = p4_budget / len(betas)
    for beta, lr in zip(betas, lrs):
        if time.monotonic() > deadline:
            break
        alloc = min(per_beta * 1.3, deadline - time.monotonic())
        if alloc < 3:
            break
        t0 = time.monotonic()
        sc, h_opt = adam_optimize(half_best, [beta], lr=lr, n_steps=200000,
                                   time_limit_s=alloc, t_start=time.monotonic(), pal=True)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"Adam-pal b={beta}")
        half_best = h_opt
        print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 5: L-BFGS polishing (20% time) ===
    print(f"\n--- Phase 5: L-BFGS polish (20%) ---")
    n_half = len(best_a) // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()
    
    p5_budget = (deadline - t_start) * 0.20
    per_lbfgs = p5_budget / 4
    for beta in [5000, 50000, 200000, 500000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        alloc = min(per_lbfgs * 1.5, deadline - time.monotonic())
        maxiter = min(5000, int(alloc * 50))
        maxfun = min(100000, int(alloc * 1000))
        sc, h_opt = lbfgs_optimize(half_best, beta, maxiter=maxiter, maxfun=maxfun, pal=True)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"LBFGS-pal b={beta}")
        half_best = h_opt
        print(f"    LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Also try non-palindrome L-BFGS
    for beta in [20000, 100000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        alloc = min(per_lbfgs, deadline - time.monotonic())
        sc, a_opt = lbfgs_optimize(best_a, beta, maxiter=3000, maxfun=50000, pal=False)
        try_update(sc, a_opt, f"LBFGS-direct b={beta}")
        print(f"    LBFGS-direct beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 6: Second round of peak equalization + refinement (15% time) ===
    print(f"\n--- Phase 6: Second round refinement (15%) ---")
    p6_budget = (deadline - t_start) * 0.15
    
    # Peak equalization
    sc, a_pe3 = peak_equalization_fft(best_a, n_iters=3000, lr=0.02)
    try_update(sc, a_pe3, "peak-eq 3")
    
    # MW
    sc, a_mw2 = multiplicative_weight_fft(best_a, n_iters=500, lr=0.1)
    try_update(sc, a_mw2, "MW 2")
    
    # Blend and polish
    best_blend = best_a.copy()
    for other in [a_pe3, a_mw2]:
        for t in [0.3, 0.5, 0.7]:
            a_b = (1 - t) * best_a + t * other
            a_b = np.clip(a_b, 1e-10, None) / a_b.sum()
            sc_b = compute_score(a_b)
            if sc_b < best_score:
                try_update(sc_b, a_b, f"blend6 t={t}")
                best_blend = a_b.copy()
    
    # Quick Adam on best blend
    if time.monotonic() < deadline - 10:
        n_half_b = len(best_blend) // 2
        half_b = (best_blend[:n_half_b] + best_blend[::-1][:n_half_b]) / 2.0
        half_b = half_b / half_b.sum()
        alloc = min(30.0, deadline - time.monotonic())
        sc, h_b = adam_optimize(half_b, [50000, 200000], lr=0.0005, n_steps=100000,
                                 time_limit_s=alloc, t_start=time.monotonic(), pal=True)
        a_full_b = np.concatenate([h_b, h_b[::-1]])
        try_update(sc, a_full_b, "Adam polish 6")
    
    # === Phase 7: Final squeeze (10% time) ===
    print(f"\n--- Phase 7: Final squeeze (10%) ---")
    p7_budget = (deadline - t_start) * 0.10
    
    # Very high beta L-BFGS
    n_half = len(best_a) // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()
    
    for beta in [200000, 500000, 1000000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        alloc = min(30.0, deadline - time.monotonic())
        sc, h_opt = lbfgs_optimize(half_best, beta, maxiter=8000, maxfun=150000, pal=True)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"final LBFGS b={beta}")
        half_best = h_opt
        print(f"    final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final peak equalization fine-tune
    if time.monotonic() < deadline - 5:
        sc, a_pe_f = peak_equalization_fft(best_a, n_iters=5000, lr=0.01)
        try_update(sc, a_pe_f, "final peak-eq")
    
    # Final Adam micro-step
    if time.monotonic() < deadline - 3:
        n_half = len(best_a) // 2
        half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
        half_best = half_best / half_best.sum()
        alloc = min(15.0, deadline - time.monotonic() - 3)
        sc, h_f = adam_optimize(half_best, [500000], lr=0.0001, n_steps=50000,
                                 time_limit_s=alloc, t_start=time.monotonic(), pal=True)
        a_full_f = np.concatenate([h_f, h_f[::-1]])
        try_update(sc, a_full_f, "final Adam")
    
    # === Output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Updates: {update_count[0]}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
