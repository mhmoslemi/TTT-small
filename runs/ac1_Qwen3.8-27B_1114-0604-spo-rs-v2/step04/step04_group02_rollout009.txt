"""
Multi-strategy optimizer combining peak-aware subgradient, 
adaptive LSE L-BFGS, and active-set QP refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_smooth_and_grad(a, beta):
    """LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled = np.maximum(scaled, -50)
    exp_scaled = np.exp(scaled)
    Z = float(exp_scaled.sum())
    p = exp_scaled / Z
    L = c_max + np.log(Z) / beta
    
    # Gradient of L w.r.t. a: p^T * (convolution with a)
    # dL/da_i = sum_k p_k * dc_k/da_i = sum_k p_k * 2 * a_{k-i} (for valid indices)
    # This is 2 * a_reversed convolved with p (restricted to valid range)
    rev_a = a[::-1]
    M2 = 3 * n - 2  # length of a_conv_p
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    # We need conv_res[i] for i in [n-1, 2n-2] which gives sum_k p_k * a_{k-i}
    g = 2.0 * conv_res[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def multi_peak_subgrad(a, top_k=5):
    """Compute a multi-peak-aware subgradient direction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n), 0.0
    c_max = float(c.max())
    sc = 2.0 * n * c_max / (S * S)
    
    # Find top-k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    # Weight them by how close they are to the max
    weights = np.exp(-(c_max - c[top_idx]) / (0.1 * c_max + 1e-10))
    weights = weights / weights.sum()
    
    # Combined subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    
    S2 = S * S
    S3 = S2 * S
    # Weighted max approximation
    c_weighted = float(np.sum(weights * c[top_idx]))
    grad = (2.0 * n) * (grad / S2 - 2.0 * c_weighted / S3)
    return grad, sc

def peak_direction_line_search(a, n_tries=20):
    """Line search along the peak-reducing direction."""
    n = len(a)
    S = a.sum()
    a_norm = a / S  # work with normalized for direction
    
    grad, sc = multi_peak_subgrad(a_norm, top_k=3)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a_norm, sc
    
    direction = -grad / gnorm
    best_a = a_norm.copy()
    best_sc = sc
    
    # Try various step sizes
    base_step = (sc - 1.0) / (gnorm + 1e-10)
    for t in base_step * np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]):
        a_try = a_norm + t * direction
        a_try = np.clip(a_try, 0, None)
        S_try = a_try.sum()
        if S_try < 1e-12:
            continue
        a_try = a_try / S_try
        sc_try = compute_score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc

def subgrad_with_line_search(a, n_iters=2000, top_k=5, verbose=False):
    """Subgradient descent with line search and momentum."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    a_work = a / S
    
    v = np.zeros(n)
    momentum = 0.8
    best_a = a_work.copy()
    best_sc = compute_score(a_work)
    
    for it in range(n_iters):
        grad, sc = multi_peak_subgrad(a_work, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        
        # Line search
        base_step = 0.05
        best_t = 0
        best_local_sc = sc
        for t in base_step * np.array([0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]):
            a_try = a_work + t * direction - momentum * v
            a_try = np.clip(a_try, 0, None)
            S_try = a_try.sum()
            if S_try < 1e-12:
                continue
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_t = t
                best_a_try = a_try.copy()
                best_local_sc = sc_try
        
        if best_t > 0:
            v = momentum * v + best_t * direction
            a_work = best_a_try
        else:
            v *= 0.9
            base_step *= 0.5
        
        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best_a = a_work.copy()
    
    return best_a

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_smooth_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def active_set_refinement(a, n_active=10, n_iters=50, verbose=False):
    """
    Active-set method: identify near-maximum convolution positions,
    then solve a QP to equalize them.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    a_work = a / S
    best_a = a_work.copy()
    best_sc = compute_score(a_work)
    
    for it in range(n_iters):
        c = fast_conv(a_work)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()
        
        # Find active set: positions within epsilon of max
        epsilon = 0.01 * c_max
        active = np.where(c >= c_max - epsilon)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:n_active]
        
        # Try to equalize active peaks
        # Direction: reduce the highest, increase the lowest among active
        active_c = c[active]
        active_mean = active_c.mean()
        
        # Compute gradient that pushes toward equalization
        grad_eq = np.zeros(n)
        for k in active:
            weight = (c[k] - active_mean) / (epsilon + 1e-10)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            idx = np.arange(j_start, j_end + 1)
            grad_eq[idx] += weight * 2.0 * a_work[k - idx]
        
        gnorm = np.linalg.norm(grad_eq)
        if gnorm < 1e-14:
            break
        
        # Small step
        step = 0.01 / gnorm
        a_try = a_work - step * grad_eq
        a_try = np.clip(a_try, 0, None)
        S_try = a_try.sum()
        if S_try < 1e-12:
            break
        a_work = a_try / S_try
    
    return best_a

def spectral_phase_correction(a, n_iters=100):
    """
    Try to flatten the convolution by adjusting in frequency domain.
    For convolution: fft(c) = fft(a)^2
    We want c to be flat, so fft(c) should have only DC component.
    This means fft(a)^2 should have only DC, i.e., fft(a) should be real and non-negative
    with only DC active. But that's impossible for non-trivial a.
    
    Instead, we use an iterative approach:
    1. Compute c = ifft(fft(a)^2)
    2. Replace c with min(c, threshold) (clip the peaks)
    3. Set fft(a)^2 = fft(c_clipped) → a = sqrt(ifft...) 
       This is problematic for real signals.
    
    Alternative: use the correlation formulation.
    For autocorrelation: r[k] = sum a_i * a_{i+k}, fft(r) = |fft(a)|^2
    But we have convolution, not correlation.
    
    For real symmetric a (a[i] = a[n-1-i]), convolution = correlation.
    Let's try enforcing symmetry.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    
    # Make symmetric
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.clip(a_sym, 0, None)
    S = a_sym.sum()
    if S < 1e-12:
        return a
    a_sym = a_sym / S
    
    best_a = a_sym.copy()
    best_sc = compute_score(a_sym)
    
    for it in range(n_iters):
        c = fast_conv(a_sym)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sym.copy()
        
        # Flatten: reduce peaks in c
        threshold = c_max * 0.95
        c_flat = np.minimum(c, threshold)
        # Adjust a_sym to produce c_flat
        # fft(c_flat) = fft(a_sym_padded)^2
        # a_sym_padded = sqrt(fft(c_flat)) -- but this needs to be real
        L = 1 << (2*n-2).bit_length()
        C_flat = np.fft.fft(c_flat, L)
        # We need A such that A^2 = C_flat, and ifft(A) is real and non-negative
        # This is generally impossible, so use a relaxed version
        A_target = np.sqrt(np.abs(C_flat)) * np.exp(0.5j * np.angle(C_flat))
        a_candidate = np.fft.ifft(A_target)[:n]
        a_candidate = np.abs(a_candidate)  # take magnitude
        a_candidate = np.clip(a_candidate, 0, None)
        Sc = a_candidate.sum()
        if Sc < 1e-12:
            break
        a_candidate = a_candidate / Sc
        
        # Blend

        a_sym = np.clip(a_sym, 0, None)
        S = a_sym.sum()
        if S < 1e-12:
            break
        a_sym = a_sym / S
    
    return best_a

"""
Multi-strategy optimizer combining:
1. Parameterized shape search (low-dim global optimization)
2. Spectral iterative method for symmetric sequences
3. Multi-peak subgradient with adaptive line search
4. LSE L-BFGS with continuation (increasing beta)
5. Smart ILS with peak-aware perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(x, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.maximum(scaled, -50)
    exp_scaled = np.exp(scaled)
    Z = float(exp_scaled.sum())
    p = exp_scaled / Z
    L = c_max + np.log(Z) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def multi_peak_subgrad_direction(a, top_k=5):
    """Compute multi-peak subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n), np.inf
    c_max = float(c.max())
    sc = 2.0 * n * c_max / (S * S)
    # Top-k peaks with exponential weights
    top_idx = np.argsort(c)[::-1][:top_k]
    diffs = c_max - c[top_idx]
    scale = max(diffs.max(), 1e-10)
    weights = np.exp(-diffs / (0.5 * scale))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    S2 = S * S
    S3 = S2 * S
    c_w = float(np.dot(weights, c[top_idx]))
    grad = (2.0 * n) * (grad / S2 - 2.0 * c_w / S3)
    return grad, sc

def subgrad_line_search(a, n_iters=3000, top_k=5):
    """Subgradient with combined line search."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return a
    a_w = a / S
    v = np.zeros(n)
    mom = 0.85
    best_a = a_w.copy()
    best_sc = compute_score(a_w)
    base_step = 0.05
    for it in range(n_iters):
        grad, sc = multi_peak_subgrad_direction(a_w, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        best_t = 0.0
        best_local_sc = sc
        best_local_a = None
        for mult in [0.1, 0.3, 0.5, 1.0, 2.0, 4.0, 8.0]:
            t = base_step * mult
            a_try = a_w + t * direction
            a_try = np.clip(a_try, 0, None)
            St = a_try.sum()
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_t = t
                best_local_a = a_try.copy()
        if best_t > 0 and best_local_a is not None:
            v = mom * v + best_t * direction
            a_w = best_local_a
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = a_w.copy()
        else:
            v *= 0.5
            base_step *= 0.7
    return best_a

def spectral_iteration_symmetric(a, n_iters=200, blend=0.5):
    """
    Gerchberg-Saxton-like iteration for symmetric sequences.
    For symmetric a, self-convolution = self-autocorrelation.
    FFT of autocorrelation = |FFT(a)|^2.
    We want to flatten the autocorrelation → flatten |FFT(a)|^2.
    """
    n = len(a)
    # Symmetrize
    a_w = (a + a[::-1]) / 2.0
    a_w = np.clip(a_w, 0, None)
    S = a_w.sum()
    if S < 1e-12:
        return a
    a_w = a_w / S
    
    best_a = a_w.copy()
    best_sc = compute_score(a_w)
    
    L = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        # Compute autocorrelation spectrum
        A = np.fft.rfft(a_w, L)
        A2 = A * A  # Since a is symmetric and real, A is real, so |A|^2 = A^2
        
        # Target: flatten A2 (make it as uniform as possible, except DC)
        A2_mean = A2[1:].mean() if len(A2) > 1 else 0
        # Blend toward uniform
        A2_target = A2.copy()
        A2_target[1:] = (1 - blend) * A2[1:] + blend * A2_mean
        A2_target = np.maximum(A2_target, 0)
        
        # Recover a from A2_target
        A_new = np.sqrt(A2_target)
        # Preserve sign of original A (for symmetric real, A should be real)
        signs = np.sign(A.real)
        signs[signs == 0] = 1
        A_new = A_new * signs
        
        # Inverse FFT
        a_candidate = np.fft.irfft(A_new, L)[:n]
        a_candidate = np.clip(a_candidate, 0, None)
        # Symmetrize
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        Sc = a_candidate.sum()
        if Sc < 1e-12:
            break
        a_candidate = a_candidate / Sc
        
        # Blend
        a_w = (1 - 0.3) * a_w + 0.3 * a_candidate
        a_w = np.clip(a_w, 0, None)
        S = a_w.sum()
        if S < 1e-12:
            break
        a_w = a_w / S
        
        sc = compute_score(a_w)
        if sc < best_sc:
            best_sc = sc
            best_a = a_w.copy()
    
    return best_a

def parameterized_shape(n, params):
    """Generate a sequence from parameterized shape."""
    x = np.linspace(0, 1, n)
    # params: [center_width, exponent, baseline]
    cw = params[0] if len(params) > 0 else 0.3
    exp_p = params[1] if len(params) > 1 else 2.0
    base = params[2] if len(params) > 2 else 0.1
    # Super-Gaussian centered at 0.5
    a = base + (1.0 - base) * np.exp(-((x - 0.5) / cw) ** exp_p)
    return np.maximum(a, 0)

def optimize_shape_params(n):
    """Optimize shape parameters in low dimension."""
    def shape_score(params):
        a = parameterized_shape(n, params)
        S = a.sum()
        if S < 1e-12:
            return 1e10
        a = a / S
        return compute_score(a)
    
    best_sc = np.inf
    best_params = None
    best_a = None
    
    # Grid + local refinement
    for cw in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5]:
        for exp_p in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0, 8.0]:
            for base in [0.0, 0.05, 0.1, 0.2, 0.3]:
                params = [cw, exp_p, base]
                sc = shape_score(params)
                if sc < best_sc:
                    best_sc = sc
                    best_params = params
                    best_a = parameterized_shape(n, params).copy()
    
    # Local refinement
    from scipy.optimize import minimize as scipym
    if best_params is not None:
        res = scipym(shape_score, best_params, method='Nelder-Mead',
                     options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12})
        if res.fun < best_sc:
            best_sc = res.fun
            best_params = list(res.x)
            best_a = parameterized_shape(n, best_params).copy()
    
    if best_a is not None:
        S = best_a.sum()
        if S > 0:
            best_a = best_a / S
    return best_sc, best_a

def peak_aware_perturbation(a, rng, strength=0.05):
    """Perturbation that specifically targets peak contributors."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    seg = a[j_start:j_end+1]
    contrib = seg[::-1].copy()
    
    a_p = a.copy()
    top_count = max(10, n // 30)
    top_local = np.argsort(contrib)[::-1][:top_count]
    bot_local = np.argsort(contrib)[:top_count]
    
    # Reduce top contributors
    for li in top_local:
        a_p[j_start + li] *= (1.0 - strength * rng.random())
    # Increase bottom contributors
    deficit = a.sum() - a_p.sum()
    if deficit > 0:
        per = deficit / len(bot_local)
        for li in bot_local:
            a_p[j_start + li] += per
    
    # Also add some general noise
    a_p += strength * 0.5 * rng.standard_normal(n) * a.mean()
    a_p = np.clip(a_p, 0, None)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p
