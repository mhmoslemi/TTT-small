Actually, let me think about a fundamentally different approach that might be more effective:

**Approach: Alternating Minimization on Lags**

Instead of optimizing all of a simultaneously, we can:
1. Fix the set of "active" lags K (where the convolution is at its maximum).
2. For a fixed K, the problem becomes: minimize max_{k in K} (a*a)_k subject to a >= 0, sum(a) = 1.
3. This is still non-convex (because of the max of quadratics), but we can use the following trick:
   - Introduce a variable t and minimize t subject to (a*a)_k <= t for all k in K, a >= 0, sum(a) = 1.
   - This is a QP (quadratically constrained program) with 2n-1 quadratic constraints (reduced to |K| constraints).
   - We can solve this using semi-definite programming or iterative methods.

But solving a QP with quadratic constraints is still hard. Let me think of another way.

**Alternative: Water-filling / Proportional Fairness**

At the optimum, we expect the active lags to all have the same value (by the "water-filling" intuition). So we want:
(a*a)_{k1} = (a*a)_{k2} = ... = (a*a)_{km} = t (the maximum)

This gives us m equations in n+m unknowns (n values of a + t + multipliers). Combined with the sum constraint and non-negativity, this determines the solution.

The iterative algorithm would be:
1. Start with some a.
2. Identify the active lags K.
3. Solve for the a that equalizes the active lag values (while maintaining sum = 1 and a >= 0).
4. Update K based on the new a.
5. Repeat.

Step 3 is the hard part. We can approximate it by:
- Computing the gradient of each active lag w.r.t. a
- Finding the direction that equalizes them (a "balancing" direction)
- Taking a step in that direction

This is essentially what the subgradient method does, but with the additional goal of equalizing rather than just reducing the maximum.

Let me also think about the "equalization" direction more carefully.

Suppose the top-k lags have values c_{k1} > c_{k2} > ... > c_{k,p} (the p that are close to max). We want to reduce c_{k1} the most, but not increase c_{k2}, ..., c_{k,p} too much.

The gradient of c_{ki} w.r.t. a is 2 * a[::-1] shifted by ki. So the "balanced" direction would be a weighted combination of these gradients, where the weights are chosen to equalize the values.

import numpy as np

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

    return lse_score_grad_tuple(a, beta)

def estimate_eval_time(n):

    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale

def estimate_maxfun(t_left, n):

    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))

def run_lbfgs_tuple(obj_func, a0, param, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64

    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    if maxfun < 64:
        maxfun = 64

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            obj_func,
            a,
            method="L-BFGS-B",
            jac=True,
            args=(param,),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)

def smooth_weights(d, k, temperature=0.05):
    """Smooth weights concentrated on top-k lags."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return np.ones(m, dtype=np.float64) / m
    z = (d - dmax) / (temperature * dmax)
    e = np.exp(z)
    return e / e.sum()

def coordinate_descent_polish(a, max_sweeps=10, deadline=None):
    """Coordinate descent: optimize one element at a time."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    
    for sweep in range(int(max_sweeps)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        
        # Randomized coordinate order for diversity
        order = np.random.permutation(n) if n > 100 else np.arange(n)
        
        for i in order:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            
            # Try several new values for a[i]
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Candidate values: scale by factors
            for factor in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5):
                new_val = old_val * factor
                a[i] = new_val
                a = normalize_a(a)
                sc = exact_score_a(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_val
                    a = normalize_a(a)
        
        if not improved:
            break
    
    return float(best_sc), best_a

def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.70))
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.50))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            k = int(rng.integers(1, 6))
            w = active_weights(d, k)
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smooth noise in log domain
        logp = np.log(p + 1e-300)
        sigma = strength * 0.3
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(3, n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + sigma * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # Block perturbation
        nb = int(rng.integers(4, 32))
        per = max(1, n // nb)
        block_mult = np.exp(rng.standard_normal(nb) * strength * 0.3)
        mult = np.repeat(block_mult, per)
        if mult.size < n:
            mult = np.concatenate([mult, np.repeat(block_mult, n - mult.size)])
        q = p * mult[:n]
    else:
        # Gaussian window perturbation
        center = float(rng.uniform(0.3, 0.7))
        width = float(rng.uniform(0.1, 0.4))
        x = np.linspace(0, 1, n)
        window = 1.0 + strength * 0.5 * np.exp(-((x - center) / width) ** 2)
        q = p * window

    return normalize_a(q)

def generate_diverse_inits(n, rng, parent, cap=30):
    """Generate diverse initial candidates."""
    inits = []
    n = int(n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        if float(np.min(arr)) < 0:
            arr = arr - float(np.min(arr)) + 0.05
        p = normalize_a(arr)
        inits.append(p)
    
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    x = np.linspace(0.0, 1.0, n)
    
    # Uniform
    add(np.ones(n, dtype=np.float64))
    
    # Power law (bump shapes)
    for alpha in (1.5, 2.0, 3.0, 4.0, 6.0):
        center = 0.5
        add(np.abs(x - center) ** (-alpha) + 0.1)
        add((1.0 + np.cos(2.0 * np.pi * x)) ** alpha + 0.1)
    
    # Trigonometric
    for f in range(1, 8):
        add(1.0 + 0.3 * np.cos(2.0 * np.pi * f * x))
        add(1.0 + 0.3 * np.sin(2.0 * np.pi * f * x + np.pi / 4))
    
    # Frequency domain random
    Nf = 1 << (n - 1).bit_length()
    for gamma in (1.0, 1.5, 2.0, 3.0):
        freqs = np.fft.rfftfreq(Nf) * Nf
