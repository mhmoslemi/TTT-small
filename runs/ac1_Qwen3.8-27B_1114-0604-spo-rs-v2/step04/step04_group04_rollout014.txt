```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M_fft = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:2*n-1]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp(a, M):
    """Solve: max sum(x) s.t. conv(a,x)_k <= M for all k, x >= 0.
    Returns optimal x."""
    n = len(a)
    # conv(a, x) has length 2n-1
    # (conv(a,x))_k = sum_i a[i] * x[k-i] (valid indices only)
    
    # The constraint matrix: for each lag k (0 to 2n-2),
    # (conv(a,x))_k = sum over valid (i,j) with i+j=k of a[i]*x[j]
    # = sum over j of a[k-j] * x[j] (where 0 <= k-j < n and 0 <= j < n)
    
    # So the A_ub matrix has shape (2n-1, n)
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Building this matrix explicitly would be (2n-1) x n = O(n^2) memory
    # For n=4096, that's about 8M entries, which is fine (32MB as float64)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    c_obj = -np.ones(n)  # minimize -sum(x) = maximize sum(x)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:
        return res.x
    return None
```

def build_constraint_matrix(a):
    """Build the sparse convolution matrix A where (A@x)[k] = (conv(a,x))[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    # For each output lag k (0 to 2n-2), and each input index j (0 to n-1):
    # A[k,j] = a[k-j] if 0 <= k-j < n
    for k in range(2*n-1):

        rows.extend([k] * len(j_range))
        cols.extend(j_range.tolist())
        vals.extend(a_vals.tolist())

    return A

    del A

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def spectral_init(n, n_iters=200, rng=None):
    """Initialize using spectral method (alternating projections)."""
    if rng is None:
        rng = np.random.default_rng()
    
    a = np.ones(n) / n
    M_fft = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        # Forward: compute spectrum
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Modify amplitude to be flatter
        target = amp.mean()
        flatness = 0.5 + 0.3 * (it / n_iters)  # Gradually increase flatness
        amp_new = flatness * target + (1 - flatness) * amp
        # Avoid zeroing out DC component
        amp_new[0] = max(amp_new[0], target * 0.5)
        
        # Inverse
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Project: non-negative, normalize
        a_new = np.maximum(a_new, 0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
    
    return a
