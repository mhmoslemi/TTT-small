Now, the main algorithm:

**Iterative Reweighting (Multiplicative Weight Update):**

```
Initialize w (weights on lags 0 to 2n-2), e.g., uniform.
Initialize a (e.g., uniform).

For each iteration:
    1. Solve: minimize a^T W a subject to a ≥ 0, sum(a) = 1
       where W[i,j] = w[i+j] (Hankel matrix)
       Use projected gradient or L-BFGS on the simplex.
    
    2. Compute conv = convolve(a, a)
    
    3. Update weights: w[k] ∝ conv[k]^γ for some γ > 0
       (This is the multiplicative weight update)
    
    4. Optionally: track the current score and keep the best a.
```

The key insight is that in step 1, we're minimizing a weighted average of the convolution values, and in step 3, we update the weights to emphasize the lags that are currently large. This is a standard approach in game theory / online learning.

The convergence rate is O(1/sqrt(T)) for the multiplicative weight update, but in practice it can be much faster.

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using:
1. Phase retrieval initialization (flat autocorrelation target)
2. Multiplicative weight update (game-theoretic approach)
3. Frequency-domain error reduction
4. L-BFGS with LSE smoothing
5. Adaptive coordinate descent

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 1 << (2*n-2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:2*n-1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0*n*float(c.max())/(S*S)

def compute_Wa(a, w):
    """Compute W*a where W[i,j] = w[i+j] (Hankel matrix product)."""
    n = len(a)
    M = 1 << (3*n-3).bit_length()
    a_pad = np.zeros(M)
    a_pad[:n] = a
    w_pad = np.zeros(M)
    w_pad[:2*n-1] = w
    A = np.fft.rfft(a_pad, M)
    W = np.fft.rfft(w_pad, M)
    C = np.fft.irfft(A * np.conj(W), M)
    return C[:n]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_weighted_qp(a, w, n_iters=200, lr=None):
    """Solve min a^T W a s.t. a >= 0, sum(a) = 1 using projected gradient."""
    n = len(a)
    a = project_simplex(a.copy())
    
    # Estimate step size
    # Lipschitz constant ~ 2 * max_row_sum(W) ~ 2 * sum(w) = 2 (since w is a probability)
    # But more accurately, it's 2 * lambda_max(W)
    # For safety, use a smaller step
    if lr is None:
        lr = 0.5  # conservative
    
    for it in range(n_iters):
        g = compute_Wa(a, w)
        grad = 2.0 * g
        # Adaptive step size (Polyak)
        # t_k = f_k / ||g_k||^2
        f_val = float(a @ g)
        gn = float(grad @ grad)
        step = min(lr, f_val / (gn + 1e-30))
        step = max(step, 1e-6)
        
        a_new = a - step * grad
        a = project_simplex(a_new)
    
    return a

def multiplicative_weight_update(a_init, n, n_outer=200, n_inner=100, gamma=2.0):
    """
    Multiplicative weight update for the game:
    Player 1 (rowner): chooses a (sequence), payoff = conv(a,a)[k]
    Player 2 (column player): chooses lag k, payoff = conv(a,a)[k]
    
    The value of the game approximates the minimum max-convolution.
    """
    a = a_init.copy()
    a = project_simplex(a)
    w = np.ones(2*n-1) / (2*n-1)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for outer in range(n_outer):
        # Solve inner QP
        a = solve_weighted_qp(a, w, n_iters=n_inner)
        
        # Compute score
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Update weights (multiplicative weight)
        conv = fast_conv(a)
        # Shift to positive
        conv_shifted = conv + 1e-15
        w = conv_shifted ** gamma
        w = w / w.sum()
        
        # Cooling: reduce gamma over time for convergence
        gamma_current = max(0.5, gamma * (1.0 - outer/n_outer))
    
    return best_a, best_score

def phase_retrieval_init(n, n_iters=200):
    """
    Initialize using phase retrieval: find a non-negative sequence whose
    autocorrelation is as flat as possible.
    
    Target: |A(f)|^2 proportional to the DFT of the triangle function.
    The DFT of the triangle (length 2n-1, peak at center) is |sinc|^2.
    """
    M = 1 << (2*n-2).bit_length()
    
    # Target spectrum: flat autocorrelation means |A(f)|^2 = constant * triangle_DFT(f)
    # triangle_DFT(f) = |sinc(f*M/(2n))^2| approximately
    # For simplicity, target |A(f)| = 1/sqrt(triangle envelope)
    
    # The triangle of length 2n-1 has DFT that's a Dirichlet kernel squared
    # Let's just target |A(f)| = 1 (flat spectrum) as a starting point
    # and use the triangle constraint in the projection
    
    # Initialize with flat spectrum (random phase)
    rng = np.random.default_rng(42)
    A = np.ones(M//2+1)
    A[1:] *= rng.standard_normal(M//2) + 1j*rng.standard_normal(M//2)
    A[0] = A[0].real  # DC is real
    if M % 2 == 0:
        A[-1] = A[-1].real  # Nyquist is real
    
    a = np.fft.irfft(A * M, M)[:n]
    a = np.maximum(a, 0)
    
    for it in range(n_iters):
        if a.sum() < 1e-12:
            a = np.ones(n) / n
            continue
        a = a / a.sum()
        
        # Compute current spectrum
        A_cur = np.fft.rfft(a, M)
        
        # Target: |A_target|^2 should give flat autocorrelation
        # autocorrelation = ifft(|A|^2), we want this to be flat
        # So |A|^2 should be flat => |A| = constant
        # But we also need the support constraint: a is zero outside [0,n-1]
        
        # Iterate: 
        # 1. Set magnitude to target (flat)
        # 2. Compute time domain, project to [0,n-1] and non-negativity
        
        A_new = np.ones_like(A_cur) * np.abs(A_cur).mean()
        A_new *= A_cur / (np.abs(A_cur) + 1e-30)  # preserve phase
        
        a_new = np.fft.irfft(A_new * M, M)[:n]
        a_new = np.maximum(a_new, 0)
        
        if a_new.sum() < 1e-12:
            # Try different phase
            A = np.ones(M//2+1) * (rng.standard_normal(M//2) + 1j*rng.standard_normal(M//2))
            A[0] = abs(A[0])
            a = np.fft.irfft(A * M, M)[:n]
            a = np.maximum(a, 0)
            if a.sum() < 1e-12:
                a = np.ones(n) / n
            continue
        
        a = a_new
    
    return a / a.sum()

def lse_obj_grad(a, beta, normalize=True):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    # Gradient of L w.r.t. a[i]:
    # L = max_k c[k] + (1/beta) * log(sum_k exp(beta*(c[k]-c[k_max])))
    # dL/da[i] = (1/beta) * sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] + a[i-k] (if valid) ... actually dc[k]/da[i] = a[max(0,k-i)] + ... 
    # More precisely: c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k+... ]
    # Wait: c[k] = sum_j a[j]*a[k-j]. dc[k]/da[i] = a[k-i] (if 0<=k-i<=n-1) + a[i] (if 0<=k-i... no)
    # dc[k]/da[i] = sum_j [delta(j=i)*a[k-j] + delta(k-j=i)*a[j]]
    #             = a[k-i] (if 0<=k-i<n) + a[i-k+... ]
    # Actually: dc[k]/da[i] = a[k-i] (if 0 ≤ k-i ≤ n-1) + a[k-i] (if 0 ≤ k-i ≤ n-1)
    # No wait: c[k] = sum_j a[j] * a[k-j]
    # dc[k]/da[i] = sum_j [δ_{j,i} * a[k-j] + a[j] * δ_{k-j,i}]
    #             = a[k-i] * (1 if 0≤k-i<n) + a[k-i] * (1 if 0≤k-i<n)
    #             = 2*a[k-i] (if 0 ≤ k-i ≤ n-1)
    
    # So dL/da[i] = (1/beta) * sum_k p[k] * 2*a[k-i] (if 0≤k-i<n)
    #             = (2/beta) * sum_k p[k] * a[k-i] (if 0≤k-i<n)
    #             = (2/beta) * (cross-correlation of p and a)[i]
    
    # cross_corr_p_a[i] = sum_k p[k] * a[k-i] (for valid k-i)
    # = sum_m p[m+i] * a[m] (substituting m = k-i)
    
    # This is the "valid" cross-correlation of p (length 2n-1) and a (length n)
    # cross_corr[i] = sum_{m=0}^{n-1} a[m] * p[m+i] for i = 0, ..., n-1
    
    # Use FFT: same as compute_Wa but with p instead of w
    p_float = p.astype(np.float64)
    S2 = 3*n - 2
    M2 = 1 << (S2-1).bit_length()
    a_pad = np.zeros(M2)
    a_pad[:n] = a
    p_pad = np.zeros(M2)
    p_pad[:2*n-1] = p_float
    A2 = np.fft.rfft(a_pad, M2)
    P2 = np.fft.rfft(p_pad, M2)
    CC = np.fft.irfft(A2 * np.conj(P2), M2)[:n]
    
    dL_da = 2.0 * CC / beta
    
    # Full gradient of score = 2n * L / S^2
    grad = 2.0 * n * (dL_da / (S*S) - 2.0 * L / (S*S*S))
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0, 1000).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0, 1000)
    return compute_score(a_out), a_out

def error_reduction(a, n_iters=200, alpha=0.7):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n)/n
    a = a / S
    M = 1 << (2*n-2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A*A, M)[:2*n-1]
        
        conv_max = conv.max()
        conv_mean = conv.mean()
        if conv_max - conv_mean < 1e-15:
            break
        
        # Target: reduce peaks
        thresh = conv_mean + (1-alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0)
        
        err = -0.5 * excess
        err_pad = np.zeros(M)
        err_pad[:2*n-1] = err
        err_spec = np.fft.rfft(err_pad, M)
        
        # Update: A_new = A + err_spec / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain error reduction for fast initial flattening
2. Multiplicative weight update (game-theoretic/iterative reweighting)
3. L-BFGS with LSE smoothing for local refinement
4. Phase-retrieval-inspired initialization
5. Adaptive coordinate descent polish

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_Wa(a, w):
    """Compute W*a where W[i,j] = w[i+j] (Hankel matrix-vector product via FFT)."""
    n = len(a)
    M = 1 << (3 * n - 3).bit_length()
    a_pad = np.zeros(M)
    a_pad[:n] = a
    w_pad = np.zeros(M)
    w_pad[:2 * n - 1] = w
    A = np.fft.rfft(a_pad, M)
    W = np.fft.rfft(w_pad, M)
    C = np.fft.irfft(A * np.conj(W), M)
    return C[:n]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_weighted_qp(a, w, n_iters=100):
    """Solve min a^T W a s.t. a >= 0, sum(a)=1 via projected gradient with backtracking."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_f = float(a @ compute_Wa(a, w))

    for it in range(n_iters):
        g = compute_Wa(a, w)
        grad = 2.0 * g
        grad_norm_sq = float(grad @ grad)
        if grad_norm_sq < 1e-30:
            break

        # Step size: start with 1/lipschitz, use backtracking
        step = 1.0 / max(2.0, grad_norm_sq + 1e-10)
        step = min(step, 1.0)

        for _ in range(10):
            a_new = project_simplex(a - step * grad)
            f_new = float(a_new @ compute_Wa(a_new, w))
            if f_new < best_f + 1e-15:
                break
            step *= 0.5
        else:
            break

        a = a_new
        best_f = f_new

    return a


def multiplicative_weight_update(a_init, n, n_outer=200, n_inner=80, gamma=2.0):
    """
    Iterative reweighting: minimize max_k conv(a,a)[k] via game-theoretic MWU.
    """
    a = project_simplex(a_init.astype(np.float64).copy())
    w = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_score = compute_score(a)

    for outer in range(n_outer):
        # Solve inner QP with current weights
        a = solve_weighted_qp(a, w, n_iters=n_inner)

        # Evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

def error_reduction(a, n_iters=200, alpha=0.75):
    """Frequency-domain iterative error reduction to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-15:

def lse_obj_grad(a, beta):
    """LSE-smoothed objective and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    # dL/da[i] = (2/beta) * sum_k p[k] * a[k-i] (for valid 0<=k-i<n)
    # = (2/beta) * cross_corr[p, a](i) where cross_corr[i] = sum_m a[m]*p[m+i]
    M2 = 1 << (3 * n - 3).bit_length()
    a_pad = np.zeros(M2)
    a_pad[:n] = a
    p_pad = np.zeros(M2)
    p_pad[:2 * n - 1] = p
    A2 = np.fft.rfft(a_pad, M2)
    P2 = np.fft.rfft(p_pad, M2)
    CC = np.fft.irfft(A2 * np.conj(P2), M2)[:n]

    dL_da = 2.0 * CC / beta
    grad = 2.0 * n * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)

    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

    # to make the autocorrelation flat. The triangle of length 2n-1 has

    target_mag = np.ones(M // 2 + 1)

    A[0] = abs(A[0])

            A[0] = abs(A[0])
            a = np.fft.irfft(A * M, M)[:n]
            a = np.maximum(a, 0.0)
            if a.sum() < 1e-12:
                return np.ones(n) / n

        a = a_new

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        return np.ones(n) / n
    return a / a.sum()


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, np.maximum(parent, 0)).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))

    # Phase retrieval init
    try:
        inits.append(phase_retrieval_init(n, rng, n_iters=200))
    except Exception:
        pass

    return [np.maximum(a, 0.01) for a in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: error reduction, MWU, L-BFGS, coordinate descent.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [6962, 8192, 4096, 10240, 12288]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths), key=lambda x: x)[:5]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # ============================================================
    # Phase 1: Error Reduction from diverse inits (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (float(budget_s) - 30)
    print(f"\n=== Phase 1: Error Reduction (until t={phase1_end - t_start:.0f}s) ===")

    for n in lengths[:3]:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, int((phase1_end - time.monotonic()) / 15.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_er = error_reduction(a0, n_iters=200, alpha=0.75)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    # Also try with different alpha
    if time.monotonic() < phase1_end and len(best_a) > 0:
        a_er2 = error_reduction(best_a / best_a.sum(), n_iters=150, alpha=0.9)
        report(compute_score(a_er2), a_er2, "ER alpha=0.9")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Multiplicative Weight Update (30% time)
    # ============================================================
    phase2_end = t_start + 0.55 * (float(budget_s) - 30)
    print(f"\n=== Phase 2: Multiplicative Weight Update (until t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_norm = best_a / best_a.sum() if best_a.sum() > 1e-12 else np.ones(n_best) / n_best

    # Estimate how many outer iterations we can afford
    # Each outer iter: n_inner QP iters, each O(n log n) for FFT
    # Roughly 0.01-0.1s per outer iteration for n~7000
    time_per_outer = 0.15
    n_outer = max(30, int((phase2_end - time.monotonic()) / time_per_outer))
    n_outer = min(n_outer, 800)
    n_inner = 60

    print(f"  MWU: n_outer={n_outer}, n_inner={n_inner}")
    a_mwu, sc_mwu = multiplicative_weight_update(a_norm, n_best, n_outer=n_outer, n_inner=n_inner, gamma=2.0)
    report(sc_mwu, a_mwu, "MWU")

    # Also try MWU from a different starting point
    if time.monotonic() < phase2_end - 30:
        a_alt = error_reduction(np.ones(n_best) * 0.5, n_iters=50, alpha=0.5)
        a_mwu2, sc_mwu2 = multiplicative_weight_update(
            a_alt, n_best,
            n_outer=min(n_outer // 2, 200), n_inner=40, gamma=3.0
        )
        report(sc_mwu2, a_mwu2, "MWU alt")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS Refinement (30% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (float(budget_s) - 30)
    print(f"\n=== Phase 3: L-BFGS Refinement (until t={phase3_end - t_start:.0f}s) ===")

    beta_schedule = [100, 500, 2000, 5000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining_time = phase3_end - time.monotonic()
        maxiter = int(min(8000, remaining_time / 0.05))
        maxfun = int(min(200000, remaining_time / 0.005))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=max(500, maxiter), maxfun=max(2000, maxfun))
        report(sc, a_opt, f"LBFGS b={beta}")

    # Try from different starting points
    if time.monotonic() < phase3_end - 60:
        for n_try in lengths[:2]:
            if time.monotonic() > phase3_end:
                break
            if len(best_a) != n_try:
                a_interp = np.interp(
                    np.linspace(0, 1, n_try),
                    np.linspace(0, 1, len(best_a)),
                    best_a
                )
            else:
                a_interp = best_a.copy()
            for beta in [500, 3000]:
                if time.monotonic() > phase3_end:
                    break
                sc, a_opt = run_lbfgs(a_interp, beta, maxiter=4000, maxfun=100000)
                report(sc, a_opt, f"LBFGS alt n={n_try} b={beta}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final Polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    remaining = deadline - time.monotonic()

    if remaining > 30:
        a_cd = coord_descent_polish(best_a, n_sweeps=5, top_k=5)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_descent")

    if remaining > 25:
        a_er = error_reduction(best_a / max(best_a.sum(), 1e-12), n_iters=150, alpha=0.85)
        sc = compute_score(a_er)
        report(sc, a_er, "final_ER")

    if remaining > 20 and best_score > 1.503:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final_LBFGS b=20000")

    if remaining > 15:
        # One more ER + LBFGS combo
        a_combo = error_reduction(best_a / max(best_a.sum(), 1e-12), n_iters=80, alpha=0.7)
        sc, a_opt = run_lbfgs(a_combo, 5000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "ER+LBFGS")

    if remaining > 10:
        # Diversification: perturb and rerun
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_er_p = error_reduction(a_pert, n_iters=80, alpha=0.8)
            sc, a_opt = run_lbfgs(a_er_p, 2000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"diversify#{trial}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
