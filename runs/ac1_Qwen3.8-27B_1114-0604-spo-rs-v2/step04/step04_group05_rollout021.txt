def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)

    log_sum = np.log(exp_scaled.sum())

    # Actually, let me use the convolution approach:
    # c(k) = sum_j a[j] * a[k-j]
    # d c(k) / d a[i] = a[k-i] (if valid) + a[k-i] (if valid, from the other term)

    # = 2 * sum_{k: 0<=k-i<n} p(k) * a[k-i]
    # = 2 * sum_{j: 0<=j<n} p(i+j) * a[j]  (substituting j=k-i)
    # = 2 * (p_shifted * a)[i] where p_shifted[j] = p(i+j)

    # Define q[k] = p[k] (length 2n-1, the full convolution output)
    # Then h[i] = sum_k q[k] * a[k-i] * [0<=k-i<n]

    h = conv_result[n-1 : 2*n-1]  # h[0],...,h[n-1]
    
    g_L = 2.0 * h  # dL/da = 2*h
    
    # Full gradient of score = 2n*L/S^2:
    # d score / d a = 2n * (g_L * S^2 - L * 2*S) / S^4

def run_lbfgs(a0, beta, maxiter=3000, maxfun=20000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    if time_budget is not None and t_start is not None:
        deadline = t_start + time_budget
        def timed_obj(x):
            if time.monotonic() > deadline:
                raise TimeoutError
            return lse_score_grad(x, beta)
        try:
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                          options={'ftol': 1e-15, 'gtol': 1e-12,
                                   'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        except (TimeoutError, Exception):
            return compute_score(a0), a0
    else:
        res = minimize(lambda x: lse_score_grad(x, beta), a0, method='L-BFGS-B',
                      jac=True, bounds=bounds,
                      options={'ftol': 1e-15, 'gtol': 1e-12,
                               'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out
