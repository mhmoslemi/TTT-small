Hybrid optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_v2(a, n_iters=500, n_levels=5, tighten_rate=0.995):
    """
    Improved Fienup with multi-level soft thresholding.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adaptive parameters
    base_mult = 1.5  # Start with threshold at 1.5x mean
    mult = base_mult

    for it in range(n_iters):
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Multi-level soft thresholding
        scale = np.ones_like(P)
        for lv in range(n_levels):
            thresh = P_mean * mult * (1.0 + 0.1 * lv)
            above = P > thresh
            if above.any():
                # Soft reduction: blend between current and threshold
                alpha = 0.3 + 0.7 * progress  # Increase strength over time
                red = np.sqrt(np.maximum(thresh / P[above], 0.01))
                scale[above] *= (1.0 - alpha) + alpha * red

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Tighten the threshold over iterations
        mult *= tighten_rate

    return best_a

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_score_and_peaks(a, top_m=20):

    score = 2.0 * n * float(np.max(c)) / (S * S)
    idx = np.argsort(c)[::-1][:top_m]

12:

    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Compute dL/da = conv(rev_a, p)[n-1:2n-1] + conv(a, p)[0:n]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    A_fft = np.fft.rfft(a, M2_fft)
    RA_fft = np.fft.rfft(a[::-1], M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)

    conv_ap = np.fft.irfft(A_fft * P_fft, M2_fft)[:n]
    conv_rap = np.fft.irfft(RA_fft * P_fft, M2_fft)[n - 1:2 * n - 1]

    dL_da = conv_ap + conv_rap

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x, beta):
    """Top-level wrapper for scipy.minimize."""
    return lse_grad_correct(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective with correct gradient."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=5000, top_m=15, step_init=0.03, decay=0.9998):
    """
    Subgradient method using top-M peaks with Polyak step size.
    Directly targets reducing the maximum convolution value.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peak indices
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]

        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak-like step
        polyak_step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, polyak_step * 0.5)
        step = min(step, 0.06)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def mirror_descent(a, n_iters=3000, eta_init=0.1, top_m=10):
    """
    Mirror descent (multiplicative weights) on the probability simplex.
    a_new ∝ a_old * exp(-eta * grad)
    Particularly effective for simplex-constrained problems.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient for top-M peaks
        top_idx = np.argpartition(c, -top_m)[-top_m:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Mirror descent step: a_new ∝ a * exp(-eta * grad)
        # For numerical stability, subtract max
        log_update = -eta * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        # Decay eta
        eta *= 0.9997
        eta = max(eta, 1e-6)

    return best_a


def peak_shaven_coordinate_descent(a, n_sweeps=10, n_top=40):
    """
    Coordinate descent targeting peak-contributing elements.
    Tries multiplicative factors on the most important coordinates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peak
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg * seg[::-1]  # Product contribution to the peak
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        # Try perturbation on top contributors
        for j in sorted(top_global, key=lambda x: -contrib[x - j_start]):
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            for f in [0.4, 0.6, 0.8, 1.2, 1.4]:
                a[j] = a_old_j * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_sc = a / S_tmp
                sc_try = compute_score(a_sc)
                if sc_try < best_sc:
                    best_val = a_old_j * f
                    best_sc = sc_try
                    best_a = a_sc.copy()
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (2 * x - 1) ** 4, 0.01))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, None))
    inits.append(0.3 + 0.5 * np.cos(2 * np.pi * 2 * x) ** 2 + 0.2 * np.cos(4 * np.pi * x) ** 2)
    inits.append(np.maximum(0.1 + 0.9 * np.sin(np.pi * x) ** 2, 0.01))
    return inits


def cycle_optimize(a, n_fienup=80, n_cycles=2, lbfgs_betas=[500, 2000]):
    """Cycle between Fienup and L-BFGS."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for cyc in range(n_cycles):
        a = fienup_improved(a, n_iters=n_fienup, n_levels=4, tighten=0.996, start_mult=1.5)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        for beta in lbfgs_betas:
            sc2, a2 = run_lbfgs(a, beta, maxiter=3000, maxfun=40000)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
                a = a2

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    usable = total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Primary length
    n_primary = 6962
    if parent is not None:
        n_primary = len(parent)

    print(f"Primary length: {n_primary}, Parent: {'yes' if parent is not None else 'no'}")
    print(f"Time budget: {total_budget}s, usable: {usable}s")

    # Start from parent if available
    a_work = np.ones(n_primary) * 0.5
    if parent is not None and len(parent) == n_primary:
        a_work = parent.astype(np.float64).copy()
        a_work = np.maximum(a_work, 0.0)
        S = a_work.sum()
        if S > 1e-6:
            a_work = a_work / S
        sc_init = compute_score(a_work)
        update_best(sc_init, a_work, "INIT-parent")
    else:
        sc_init = compute_score(a_work)
        update_best(sc_init, a_work, "INIT-uniform")

    # ============================================================
    # Phase 1: Fienup warm-up (25% of time)
    # ============================================================
    phase1_end = t_start + 0.25 * usable
    print(f"\n=== Phase 1: Fienup (deadline {phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n_primary, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        print(f"  Fienup init {i} (n={n_primary})...")
        a_f = fienup_improved(a0, n_iters=300, n_levels=5, tighten=0.998, start_mult=1.8)
        sc = compute_score(a_f)
        update_best(sc, a_f, f"FIENUP-{i}")

        if time.monotonic() < phase1_end:
            a_ag = fienup_aggressive(a_f, n_iters=150)
            sc2 = compute_score(a_ag)
            update_best(sc2, a_ag, f"FIENUP-AGG-{i}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: L-BFGS with increasing beta (30% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * usable
    print(f"\n=== Phase 2: L-BFGS (deadline {phase2_end - t_start:.0f}s) ===")

    betas = [100, 300, 500, 1000, 2000, 5000]
    current_a = best_a.copy()
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining_frac = (phase2_end - time.monotonic()) / max(1, usable * 0.3)
        maxiter = int(4000 * min(1.0, remaining_frac * 2))
        maxfun = int(60000 * min(1.0, remaining_frac * 2))
        maxiter = max(maxiter, 500)
        maxfun = max(maxfun, 10000)

        print(f"  L-BFGS beta={beta}, maxiter={maxiter}...")
        sc, a_opt = run_lbfgs(current_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS-{beta}")
        if improved:
            current_a = a_opt

    # Also try from a perturbed version
    if time.monotonic() < phase2_end:
        a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_primary))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=3000, maxfun=40000)
        update_best(sc, a_opt, "LBFGS-pert-1000")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient + Mirror Descent (30% of time)
    # ============================================================
    phase3_end = t_start + 0.85 * usable
    print(f"\n=== Phase 3: Subgradient + Mirror Descent (deadline {phase3_end - t_start:.0f}s) ===")

    # Top-M subgradient with various parameters
    current_a = best_a.copy()
    for top_m in [5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        print(f"  Subgradient top_m={top_m}...")
        a_sg = top_m_subgradient(current_a, n_iters=4000, top_m=top_m, step_init=0.03)
        sc = compute_score(a_sg)
        improved = update_best(sc, a_sg, f"SG-{top_m}")
        if improved:
            current_a = a_sg

    # Mirror descent
    if time.monotonic() < phase3_end:
        print("  Mirror descent...")
        a_md = mirror_descent(current_a, n_iters=3000, eta_init=0.08, top_m=10)
        sc = compute_score(a_md)
        improved = update_best(sc, a_md, "MIRROR")
        if improved:
            current_a = a_md

    # Combined: mirror descent then subgradient
    if time.monotonic() < phase3_end:
        print("  Mirror + SG combo...")
        a_combo = mirror_descent(best_a, n_iters=2000, eta_init=0.05, top_m=15)
        a_combo = top_m_subgradient(a_combo, n_iters=3000, top_m=10, step_init=0.02)
        sc = compute_score(a_combo)
        update_best(sc, a_combo, "MD+SG")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Stochastic escape + re-optimization (15% of time)
    # ============================================================
    print(f"\n=== Phase 4: Stochastic Escape ===")

    n_perturbations = 5
    for pi in range(n_perturbations):
        if time.monotonic() > deadline:
            break
        # Different perturbation strategies
        strategy = pi % 3
        if strategy == 0:
            # Gaussian perturbation
            a_p = best_a * (1.0 + 0.05 * rng.standard_normal(n_primary))
        elif strategy == 1:
            # Random spike: zero out a random fraction
            mask = rng.random(n_primary) > 0.1
            a_p = best_a * mask
        else:
            # Mix with uniform
            a_p = 0.8 * best_a + 0.2 * np.ones(n_primary) / n_primary

        a_p = np.maximum(a_p, 0.0)
        S = a_p.sum()
        if S < 1e-6:
            continue
        a_p = a_p / S

        # Quick Fienup + L-BFGS
        a_p = fienup_improved(a_p, n_iters=50, n_levels=3, tighten=0.995, start_mult=1.4)
        sc = compute_score(a_p)
        if time.monotonic() < deadline:
            sc, a_opt = run_lbfgs(a_p, 500, maxiter=2000, maxfun=30000)
        improved = update_best(sc, a_opt, f"PERTURB-{pi}")

    # Coordinate descent polishing
    if time.monotonic() < deadline - 20:
        print("  Coordinate descent...")
        a_cd = peak_shaven_coordinate_descent(best_a, n_sweeps=5, n_top=50)
        sc = compute_score(a_cd)
        updated = update_best(sc, a_cd, "COORD")
        if updated and time.monotonic() < deadline - 10:
            sc2, a_opt = run_lbfgs(a_cd, 2000, maxiter=2000, maxfun=20000)
            update_best(sc2, a_opt, "COORD+LBFGS")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 5:
        print("  Final L-BFGS...")
        sc, a_opt = run_lbfgs(best_a, 8000, maxiter=3000, maxfun=30000)
        update_best(sc, a_opt, "FINAL-LBFGS-8000")

    # Try a different length if time permits
    if time.monotonic() < deadline - 60:
        n_alt = int(n_primary * 1.2)
        print(f"  Trying alternative length n={n_alt}...")
        x_old = np.linspace(0, 1, n_primary)
        x_new = np.linspace(0, 1, n_alt)
        a_alt = np.interp(x_new, x_old, best_a).copy()
        a_alt = np.maximum(a_alt, 0.01)
        a_alt = a_alt / a_alt.sum()
        a_alt = fienup_improved(a_alt, n_iters=100, n_levels=4, tighten=0.997, start_mult=1.6)
        sc = compute_score(a_alt)
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs(a_alt, 500, maxiter=2000, maxfun=20000)
        update_best(sc, a_opt, f"ALT-n={n_alt}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
