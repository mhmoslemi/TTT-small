```python
"""
LP-based linearized descent optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Novel approach: at each step, solve a linear program to find the perturbation
direction that best reduces the top peaks, then line-search for step size.
Combined with LSE-smoothed L-BFGS for polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Score = 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled - scaled.max())
    Z = exp_s.sum()
    log_sum = np.log(Z) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / Z
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    # This is: 2 * (reverse of a) convolved with p, evaluated at appropriate indices
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def lp_step(a, K=200, U=None):
    """
    Solve LP to find optimal perturbation direction.
    Returns (delta, t_opt) or (None, None) if infeasible.
    """
    n = len(a)
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:K]
    
    if U is None:
        U = 1.0 / n  # max change per element
    
    # Build constraint matrix
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i, k in enumerate(top_idx):
        k = int(k)
        # coeff of delta[m] = a[k-m] + a[m] (with out-of-range = 0)
        # Vectorized: for m in 0..n-1, coeff[m] = a[k-m] (if 0<=k-m<n) + a[m]
        coeffs = a.copy()  # a[m] part
        # Add a[k-m] where 0 <= k-m < n
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            # For m in [m_start, m_end]: a[k-m] means index k-m in [0, n-1]
            # a[k-m] for m = m_start..m_end = a[k-m_start]..a[k-m_end] reversed
            coeffs[m_start:m_end + 1] += a[k - m_end:k - m_start + 1][::-1]
        A_ub[i, :n] = coeffs
        A_ub[i, n] = -1.0
        b_ub[i] = -c[k]
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lower = np.maximum(-a, 0.0)  # delta[m] >= -a[m] (so a+delta >= 0)
    upper = np.full(n, U)
    bounds_list = list(zip(lower, upper))
    bounds_list.append((0.0, float(c.max())))
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_list, method='highs')
    
    if res.status != 0:
        return None, None
    
    delta = res.x[:n]
    t_opt = float(res.x[n])
    return delta, t_opt


def lp_descent(a, max_iters=500, K=200, time_limit=None, t_start=None):
    """Iterative LP-based linearized descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, compute_score(a)
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        delta, t_opt = lp_step(a, K=K)
        if delta is None:
            break
        
        # Check if there's a meaningful direction
        if np.abs(delta).max() < 1e-12:
            break
        
        # Line search
        improved = False
        for alpha in alphas:
            a_new = a + alpha * delta
            if a_new.min() < -1e-10:
                continue
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            sc = compute_score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
        
        if not improved:
            # Try even smaller steps
            for alpha in [0.01, 0.005, 0.001]:
                a_new = a + alpha * delta
                if a_new.min() < -1e-10:
                    continue
                a_new = np.maximum(a_new, 0.0)
                S_new = a_new.sum()
                if S_new < 1e-12:
                    continue
                a_new = a_new / S_new
                sc = compute_score(a_new)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            
            if not improved:
                break  # Stagnated
    
    return best_a, best_sc


def subgradient_polish(a, n_iters=2000, time_limit=None, t_start=None):
    """Direct subgradient method."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, compute_score(a)
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of M = conv[k*]
        sg = np.zeros(n)
        # sg[m] = a[k*-m] + a[m] (with out-of-range = 0)
        m_start = max(0, k_star - n + 1)
        m_end = min(k_star, n - 1)
        if m_end >= m_start:
            sg[m_start:m_end+1] += a[k_star - m_end:k_star - m_start + 1][::-1]
        sg += a  # the a[m] part
        
        # Full subgradient of the objective
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.02)
        step = max(step, 1e-7)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a, best_sc


def error_reduction(a, n_iters=100):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]
        
        conv_mean = conv.mean()
        conv_max = conv.max()
        if conv_max - conv_mean < 1e-15:
            break
        
        # Flatten: reduce excess above the mean
        excess = np.maximum(conv - conv_mean, 0.0)
        if excess.max() < 1e-15:
            break
        
        alpha = 0.3 + 0.5 * (it / n_iters)
        conv_mod = conv - alpha * excess
        err = conv_mod - conv
        
        err_spec = np.fft.rfft(np.concatenate([err, np.zeros(M - len(err))]))
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0
        
        A_new = A + d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a
```

```python
"""
LP-based linearized descent optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Primary innovation: at each step, solve a linear program to find the optimal
perturbation direction that reduces the top peaks, then line-search for step size.
Combined with LSE-smoothed L-BFGS, frequency-domain error reduction, and
subgradient polishing for comprehensive search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Score = 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled - scaled.max())
    Z = exp_s.sum()
    log_sum = np.log(Z) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / Z
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=10000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def lp_step(a, K=200, U=None):
    """
    Solve LP to find optimal perturbation that reduces top-K peaks.
    Returns (delta, t_opt) or (None, None).
    """
    n = len(a)
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:K]

    if U is None:
        U = 1.0 / n

    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)

    for i, k in enumerate(top_idx):
        k = int(k)
        coeffs = a.copy()
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            coeffs[m_start:m_end + 1] += a[k - m_start:k - m_end + 1][::-1]
        A_ub[i, :n] = coeffs
        A_ub[i, n] = -1.0
        b_ub[i] = -c[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    lower = -a.copy()
    upper = np.full(n, U)
    bounds_list = list(zip(lower, upper))
    bounds_list.append((0.0, float(c.max()) * 2))

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs')
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    delta = res.x[:n]
    t_opt = float(res.x[n])
    return delta, t_opt


def lp_descent(a, max_iters=300, K=200, U=None, time_limit=60.0, t_start=None):
    """Iterative LP-based linearized descent with line search."""
    if t_start is None:
        t_start = time.monotonic()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    max_no_improve = 50

    alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03, 0.01]

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        delta, t_opt = lp_step(a, K=K, U=U)
        if delta is None:
            break
        if np.abs(delta).max() < 1e-14:
            break

        improved = False
        for alpha in alphas:
            a_new = a + alpha * delta
            if a_new.min() < -1e-10:
                continue
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            sc = compute_score(a_new)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                break

    return best_a, best_sc


def subgradient_polish(a, n_iters=2000, time_limit=10.0, t_start=None):
    """Direct subgradient method with adaptive step."""
    if t_start is None:
        t_start = time.monotonic()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(k_star, n - 1)
        if m_end >= m_start:
            sg[m_start:m_end + 1] += a[k_star - m_start:k_star - m_end + 1][::-1]
        sg += a

        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.015)
        step = max(step, 1e-7)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a, best_sc


def error_reduction(a, n_iters=100, time_limit=10.0, t_start=None):
    """Frequency-domain error reduction for flattening convolution."""
    if t_start is None:
        t_start = time.monotonic()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = conv.mean()
        conv_max = conv.max()
        if conv_max - conv_mean < 1e-15:
            break

        alpha = 0.3 + 0.5 * (it / n_iters)
        excess = np.maximum(conv - conv_mean, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - alpha * excess
        err = conv_mod - conv
        err_pad = np.zeros(M)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0

        A_new = A + d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent))

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.3 + 0.7 * np.abs(2.0 * x - 1.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))

    # Two-scale structure
    period = max(1, n // 16)
    base = 0.3 + 0.5 * rng.random(16)
    inits.append(np.tile(base, n // len(base) + 1)[:n])

    # High-frequency modulation
    inits.append(np.clip(0.5 + 0.2 * np.sin(20 * np.pi * x) + 0.1 * np.sin(7 * np.pi * x), 0.01, 10.0))

    for a in inits:
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based linearized descent as primary method with LSE L-BFGS,
    subgradient, and frequency-domain error reduction as supporting methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] score={sc:.8f} n={len(a)} t={time.monotonic()-t_start:.0f}s")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 12288, 16384]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:5]

    print(f"Lengths: {lengths}, Parent: {'yes' if parent is not None else 'no'}")

    # ============ PHASE 0: Quick warm-up with subgradient (5%) ============
    p0_end = t_start + 0.05 * (total_budget - 20.0)
    print(f"\n=== Phase 0: Warm-up ===")

    for n in lengths[:3]:
        if time.monotonic() > p0_end:
            break
        a0 = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n)
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        a_out, sc = subgradient_polish(a0, n_iters=500, time_limit=3.0)
        report(sc, a_out, f"SGD warm n={n}")

    # ============ PHASE 1: LP Descent (35%) ============
    p1_end = t_start + 0.40 * (total_budget - 20.0)
    print(f"\n=== Phase 1: LP Descent ===")

    for n in lengths:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")

        # Get best starting point for this length
        if parent is not None:
            if len(parent) == n:
                a0 = parent.copy()
            else:
                a0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            a0 = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n)
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()

        # Also try from current best if different length
        if len(best_a) != n and np.isfinite(best_score):
            a_alt = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
            a_alt = np.maximum(a_alt, 0.01)
            a_alt = a_alt / a_alt.sum()
            # Quick LP from alt
            t_sub = time.monotonic()
            a_lp, sc = lp_descent(a_alt, max_iters=100, K=100,
                                  time_limit=5.0, t_start=t_sub)
            report(sc, a_lp, f"LP alt n={n}")
            if sc < compute_score(a0):
                a0 = a_lp.copy()

        # Progressive LP with increasing K
        for K in [50, 100, 200, 400]:
            if time.monotonic() > p1_end - 5:
                break
            t_sub = time.monotonic()
            remaining = p1_end - time.monotonic()
            max_iters = max(50, int(remaining / 0.3))
            a_out, sc = lp_descent(a0, max_iters=max_iters, K=K,
                                   time_limit=remaining - 1.0, t_start=t_sub)
            report(sc, a_out, f"LP n={n} K={K}")
            a0 = a_out.copy()

        # Polish with LSE after LP
        if time.monotonic() < p1_end - 10:
            sc, a_opt = run_lbfgs(a0, 200, maxiter=2000, maxfun=6000)
            report(sc, a_opt, f"LP+LSE n={n} b=200")

    print(f"After Phase 1: best={best_score:.8f}")

    # ============ PHASE 2: LSE L-BFGS with adaptive beta (25%) ============
    p2_end = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n=== Phase 2: LSE L-BFGS ===")

    for beta in [50, 100, 300, 1000, 3000, 8000]:
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        maxiter = max(2000, int(remaining / 0.04))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=min(maxiter * 3, 50000))
        report(sc, a_opt, f"LSE b={beta}")

    # Try from other good starts
    if time.monotonic() < p2_end - 30:
        n_best = len(best_a)
        for n in lengths[:2]:
            if time.monotonic() > p2_end - 15:
                break
            if n != n_best:
                a_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, n_best), best_a)
            else:
                a_interp = best_a.copy()
            a_interp = np.maximum(a_interp, 0.01)
            a_interp = a_interp / a_interp.sum()
            for beta in [100, 500]:
                if time.monotonic() > p2_end - 5:
                    break
                sc, a_opt = run_lbfgs(a_interp, beta, maxiter=3000, maxfun=9000)
                report(sc, a_opt, f"LSE alt n={n} b={beta}")

    print(f"After Phase 2: best={best_score:.8f}")

    # ============ PHASE 3: Diverse local search (25%) ============
    p3_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n=== Phase 3: Local Search ===")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < p3_end:
        trial += 1
        strategy = (trial - 1) % 5
        elapsed_phase = p3_end - time.monotonic()

        if strategy == 0:
            # LP from perturbed start
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.01, 1000.0)
            a_pert = a_pert / a_pert.sum()
            t_sub = time.monotonic()
            sc, a_out = lp_descent(a_pert, max_iters=150, K=200,
                                   time_limit=min(15.0, elapsed_phase * 0.1), t_start=t_sub)
            report(sc, a_out, f"LP pert #{trial}")

        elif strategy == 1:
            # Error reduction + LSE
            t_sub = time.monotonic()
            a_er, sc_er = error_reduction(best_a, n_iters=80,
                                          time_limit=min(8.0, elapsed_phase * 0.05))
            report(sc_er, a_er, f"ER #{trial}")
            sc, a_opt = run_lbfgs(a_er, 500, maxiter=2000, maxfun=6000)
            report(sc, a_opt, f"ER+LSE #{trial}")

        elif strategy == 2:
            # Subgradient polish
            t_sub = time.monotonic()
            sc, a_out = subgradient_polish(best_a, n_iters=3000,
                                            time_limit=min(10.0, elapsed_phase * 0.05),
                                            t_start=t_sub)
            report(sc, a_out, f"SGD #{trial}")

        elif strategy == 3:
            # LSE with random beta
            beta = int(10 ** (1.5 + 2.5 * rng.random()))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=2500, maxfun=7500)
            report(sc, a_opt, f"LSE b={beta} #{trial}")

        else:
            # Multi-peak targeted coordinate adjustment
            c = fast_conv(best_a)
            top_peaks = np.argsort(c)[::-1][:5]
            a_pert = best_a.copy()
            for k in top_peaks:
                k = int(k)
                m_start = max(0, k - n_best + 1)
                m_end = min(k, n_best - 1)
                if m_end < m_start:
                    continue
                seg = a_pert[m_start:m_end + 1]
                contrib = seg[::-1]
                top_count = max(3, n_best // 200)
                top_local = np.argsort(contrib)[::-1][:top_count]
                total_red = 0.0
                for li in top_local:
                    gi = m_start + li
                    red = a_pert[gi] * 0.02
                    a_pert[gi] = max(a_pert[gi] - red, 0.0)
                    total_red += red
                if total_red > 0:
                    bottom_count = len(contrib) - top_count
                    if bottom_count > 0:
                        bottom_local = np.argsort(contrib)[:bottom_count]
                        per_add = total_red / bottom_count
                        for li in bottom_local:
                            a_pert[m_start + li] += per_add
            a_pert = np.maximum(a_pert, 0.0)
            S_p = a_pert.sum()
            if S_p > 1e-12:
                a_pert = a_pert / S_p
            sc, a_opt = run_lbfgs(a_pert, 300, maxiter=2000, maxfun=5000)
            report(sc, a_opt, f"Coord #{trial}")

    print(f"After Phase 3: best={best_score:.8f}")

    # ============ PHASE 4: Final polish (10%) ============
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Very high beta LSE
        sc, a_opt = run_lbfgs(best_a, 15000,
                               maxiter=max(2000, int(remaining / 0.05)),
                               maxfun=30000)
        report(sc, a_opt, "Final LSE b=15000")

    if remaining > 30:
        # Final LP with high K
        t_sub = time.monotonic()
        sc, a_out = lp_descent(best_a, max_iters=500, K=500,
                               time_limit=min(remaining * 0.4, 60.0), t_start=t_sub)
        report(sc, a_out, "Final LP K=500")

    if remaining > 20:
        # One more LSE after final LP
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=10000)
        report(sc, a_opt, "Final LSE b=5000")

    if remaining > 15:
        # Subgradient
        t_sub = time.monotonic()
        sc, a_out = subgradient_polish(best_a, n_iters=5000,
                                        time_limit=remaining - 5, t_start=t_sub)
        report(sc, a_out, "Final SGD")

    if remaining > 15:
        # Error reduction
        t_sub = time.monotonic()
        a_er, sc_er = error_reduction(best_a, n_iters=100,
                                       time_limit=remaining - 5, t_start=t_sub)
        report(sc_er, a_er, "Final ER")
        if sc_er < best_score:
            sc, a_opt = run_lbfgs(a_er, 3000, maxiter=3000, maxfun=9000)
            report(sc, a_opt, "Final ER+LSE")

    # ============ Final verification ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={elapsed:.1f}s/{budget_s}s")
    print(f"{'=' * 60}")
    return out
```
