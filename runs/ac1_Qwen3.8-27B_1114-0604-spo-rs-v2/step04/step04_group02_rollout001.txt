import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_score_grad_multipeak(a, beta, n_peaks=3):
    """LSE with emphasis on top-n peaks to flatten the distribution."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    # Weight the top peaks more heavily
    top_idx = np.argsort(c)[::-1][:n_peaks]
    c_weighted = c.copy()
    c_weighted[top_idx] *= 1.0  # Keep as is, LSE naturally focuses on max
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    S = a0.sum()
    if S > 0:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def peak_coordinate_descent(a, n_sweeps=10, n_top=50):
    """
    Coordinate descent specifically targeting the convolution peak.
    For the current peak, identify top contributors and try to reduce them
    by redistributing mass to positions that don't contribute to the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            c = fast_conv(a)
            k_star = int(np.argmax(np.delete(c, k_star)))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                break

        # Get the segment that contributes to peak
        seg = a[j_start:j_end + 1].copy()
        # Contribution of a[j] to c[k_star] is a[k_star - j]
        contrib = np.array([a[k_star - j] for j in range(j_start, j_end + 1)])

        # Sort by contribution (descending)
        order = np.argsort(contrib)[::-1]
        top_count = min(n_top, len(order))

        # For top contributors, try reducing and adding to non-contributors
        non_contrib_mask = np.zeros(n, dtype=bool)
        non_contrib_mask[:j_start] = True
        non_contrib_mask[j_end + 1:] = True
        # Also positions not in the peak window
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[j_start:j_end + 1] = True

        for _ in range(3):  # Multiple sub-iterations per sweep
            improved = False
            for idx in order[:top_count]:
                j_global = j_start + idx
                a_old_val = a[j_global]
                if a_old_val < 1e-10:
                    continue

                # Try reducing this position and adding to random non-peak positions
                n_add = max(5, n // 20)
                add_positions = np.where(non_contrib_mask)[0]
                if len(add_positions) == 0:
                    continue
                # Pick positions far from the peak
                dist = np.abs(np.arange(len(add_positions)) - k_star)
                # Just pick randomly from non-peak
                chosen = add_positions[rng_cd.choice(len(add_positions), size=min(n_add, len(add_positions)), replace=False)]

                for frac in [0.5, 0.7, 0.9]:
                    reduction = a_old_val * (1.0 - frac)
                    if reduction < 1e-12:
                        continue
                    a[j_global] = a_old_val * frac
                    per_add = reduction / len(chosen)
                    for cp in chosen:
                        a[cp] += per_add
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        # Revert
                        a[j_global] = a_old_val
                        for cp in chosen:
                            a[cp] -= per_add
            if not improved:
                break

    return best_a


rng_cd = np.random.default_rng(12345)


def diversified_init(n, rng, parent=None, mode="mixed"):
    """Generate structurally diverse initializations."""
    x = np.arange(n, dtype=np.float64) / n
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent))

    # 1. Uniform
    inits.append(np.ones(n) * 0.5)

    # 2. Smooth random (low-pass filtered noise)
    r = rng.standard_normal(n)
    # Simple exponential smoothing
    alpha = 0.1 + 0.3 * rng.random()
    sm = np.zeros(n)
    sm[0] = r[0]
    for i in range(1, n):
        sm[i] = alpha * r[i] + (1 - alpha) * sm[i - 1]
    sm = np.abs(sm) + 0.01
    inits.append(sm / sm.sum())

    # 3. Multi-frequency modulation
    n_freqs = 2 + int(rng.integers(0, 4))
    sig = np.ones(n) * 0.5
    for _ in range(n_freqs):
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = 0.2 + 0.3 * rng.random()
        sig *= (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    inits.append(np.clip(sig, 0.01, None))

    # 4. Bimodal / multimodal
    n_modes = 2 + int(rng.integers(0, 3))
    sig = np.zeros(n)
    for _ in range(n_modes):
        center = rng.random()
        width = 0.02 + 0.1 * rng.random()
        sig += np.exp(-0.5 * ((x - center) / width) ** 2)
    inits.append(sig / (sig.sum() + 1e-12))

    # 5. Power law / heavy-tailed
    sig = (x + 0.01) ** -0.5 * rng.random()
    sig = np.clip(sig, 0.001, None)
    inits.append(sig)

    # 6. Cosine window (various widths)
    width = 0.3 + 0.7 * rng.random()
    center = 0.2 + 0.6 * rng.random()
    sig = np.maximum(0.1, np.cos(np.pi * (x - center) / width))
    inits.append(sig)

    # 7. Periodic + noise
    freq = 3.0 + 7.0 * rng.random()
    sig = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x) + 0.1 * rng.standard_normal(n)
    inits.append(np.clip(sig, 0.01, None))

    # 8. Random with a few bumps
    sig = 0.1 + 0.1 * rng.standard_normal(n)
    n_bumps = 3 + int(rng.integers(0, 5))
    for _ in range(n_bumps):
        c = rng.random()
        w = 0.01 + 0.05 * rng.random()
        sig += np.exp(-0.5 * ((x - c) / w) ** 2)
    inits.append(np.clip(sig, 0.001, None))

    # 9. Step-like (piecewise constant)
    n_steps = 4 + int(rng.integers(0, 8))
    sig = np.zeros(n)
    step_size = n // n_steps
    for i in range(n_steps):
        s = i * step_size
        e = min(s + step_size, n)
        sig[s:e] = 0.1 + 0.9 * rng.random()
    inits.append(sig)

    # 10. Interpolated parent with larger perturbation
    if parent is not None:
        n_p = len(parent)
        a_interp = np.interp(x, np.linspace(0, 1, n_p), parent)
        pert = 0.2 * rng.standard_normal(n)
        a_interp = a_interp * (1.0 + pert)
        inits.append(np.clip(a_interp, 0.01, None))

    # Normalize all
    normalized = []
    for s in inits:
        s = np.clip(np.asarray(s, dtype=np.float64), 0.0, 1000.0)
        S = s.sum()
        if S > 1e-6:
            normalized.append(s / S)
    return normalized


def subgradient_peak_reduction(a, n_iters=2000, f_target=1.0):
    """
    Fast subgradient descent focusing on the single peak.
    Very efficient: O(n) per iteration with FFT.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc0 = best_sc

    momentum = np.zeros(n)
    beta_mom = 0.95

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient: d max_conv / da[j] = a[k_star - j] for valid j
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = a[k_star - idx]

        # Normalize subgradient
        gnorm = np.linalg.norm(sg)
        if gnorm < 1e-15:
            break

        # Step size: proportional to (current - target) / ||g||^2
        step = 0.5 * (sc - f_target) / (gnorm * gnorm + 1e-10)
        step = min(step, 0.1)

        momentum = beta_mom * momentum - step * sg
        a_new = a + momentum
        a = np.clip(a_new, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        # Decay momentum
        if it % 500 == 0 and it > 0:
            beta_mom *= 0.9

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses multi-start L-BFGS with LSE smoothing, subgradient, and peak-aware refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 30.0

    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(np.asarray(a, dtype=np.float64).copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(best_a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Work with parent length or default
    if parent is not None:
        n_work = len(parent)
        print(f"Using parent length: {n_work}")
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")
    else:
        n_work = 8192
        print(f"Default length: {n_work}")

    # Also try a slightly different length
    n_alt = int(n_work * 0.9) if n_work > 1000 else n_work

    # ================================================================
    # Phase 1: Diverse multi-start with moderate-beta L-BFGS (35% time)
    # ================================================================
    phase1_end = t_start + 0.35 * (budget_s - 30)
    print(f"\n=== Phase 1: Multi-start L-BFGS (n={n_work}) ===")

    # Generate diverse initializations
    inits = diversified_init(n_work, rng, parent)
    n_starts = len(inits)
    print(f"  Generated {n_starts} diverse initializations")

    # For each, do a quick L-BFGS with beta=300
    time_per_start = max(10.0, (phase1_end - time.monotonic()) / n_starts)
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        maxiter = int(min(8000, max(2000, remaining * 10)))
        maxfun = int(min(40000, max(5000, remaining * 50)))
        sc, a_opt, _ = run_lbfgs(a_init, beta=300, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"start[{i}]")

    print(f"  After Phase 1: best = {best_score:.10f}")

    # ================================================================
    # Phase 2: Adaptive beta increase on best candidates (25% time)
    # ================================================================
    phase2_end = t_start + 0.60 * (budget_s - 30)
    print(f"\n=== Phase 2: Beta escalation ===")

    beta_schedule = [1000, 5000, 20000, 50000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 60:
            break
        maxiter = int(min(30000, max(5000, remaining * 30)))
        maxfun = int(min(100000, max(10000, remaining * 100)))
        sc, a_opt, _ = run_lbfgs(best_a, beta=beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"beta={beta}")

    print(f"  After Phase 2: best = {best_score:.10f}")

    # ================================================================
    # Phase 3: Subgradient peak reduction (10% time)
    # ================================================================
    phase3_end = t_start + 0.70 * (budget_s - 30)
    print(f"\n=== Phase 3: Subgradient ===")

    if time.monotonic() < phase3_end:
        remaining_iters = int(min(3000, max(500, (phase3_end - time.monotonic()) * 3)))
        a_sg = subgradient_peak_reduction(best_a, n_iters=remaining_iters)
        sc = compute_score(a_sg)
        report(sc, a_sg, "subgrad")

    # ================================================================
    # Phase 4: Iterated local search with structural perturbations (20% time)
    # ================================================================
    phase4_end = t_start + 0.90 * (budget_s - 30)
    print(f"\n=== Phase 4: ILS with structural perturbations ===")

    n_il_trials = 15
    for trial in range(n_il_trials):
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 30:
            break

        strategy = trial % 4
        if strategy == 0:
            # Large structural perturbation
            a_pert = perturbed_structural(best_a, rng, scale=0.15)
        elif strategy == 1:
            # Targeted peak reduction perturbation
            a_pert = perturbed_peak_targeted(best_a, rng, scale=0.1)
        elif strategy == 2:
            # Blend with a fresh diverse init
            fresh = diversified_init(len(best_a), rng, parent, mode="mixed")
            a_fresh = fresh[rng.integers(0, len(fresh))]
            mix = 0.3 + 0.4 * rng.random()
            a_pert = mix * best_a + (1 - mix) * a_fresh
        else:
            # Small refinement perturbation
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(len(best_a)))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = a_pert.sum()
        if S > 1e-12:
            a_pert = a_pert / S

        # Optimize with moderate beta
        beta = 2000 + int(5000 * rng.random())
        maxiter = int(min(15000, max(3000, remaining * 20)))
        maxfun = int(min(50000, max(5000, remaining * 60)))
        sc, a_opt, _ = run_lbfgs(a_pert, beta=beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS[{trial}] s={strategy}")

    print(f"  After Phase 4: best = {best_score:.10f}")

    # ================================================================
    # Phase 5: Peak coordinate descent + final high-beta polish (15% time)
    # ================================================================
    print(f"\n=== Phase 5: Final refinement ===")

    # Peak coordinate descent
    if time.monotonic() < deadline - 120:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=5, n_top=30)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_descent")

    # Final very high beta L-BFGS
    for beta in [100000, 200000, 500000]:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic() - 30
        if remaining < 30:
            break
        maxiter = int(min(20000, max(3000, remaining * 20)))
        maxfun = int(min(60000, max(5000, remaining * 60)))
        sc, a_opt, _ = run_lbfgs(best_a, beta=beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final_beta={beta}")

    # One more subgradient pass
    if time.monotonic() < deadline - 60:
        a_sg2 = subgradient_peak_reduction(best_a, n_iters=1000)
        sc = compute_score(a_sg2)
        report(sc, a_sg2, "final_subgrad")

    # Final coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd2 = peak_coordinate_descent(best_a, n_sweeps=3, n_top=20)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "final_cd")

    # ================================================================
    # Output preparation
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out


def perturbed_structural(a, rng, scale=0.1):
    """Large structural perturbation: mix different frequency components."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    multiplier = np.ones(n)

    # Add random low-frequency modulation
    n_harmonics = 1 + int(rng.integers(0, 3))
    for _ in range(n_harmonics):
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        multiplier *= (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))

    # Add some random smooth noise
    noise = rng.standard_normal(n) * scale * 0.5
    sm = np.convolve(noise, np.ones(max(1, n // 100)) / max(1, n // 100), mode='same')
    multiplier = multiplier * (1.0 + sm)

    a_pert = a * multiplier
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = a_pert.sum()
    if S > 1e-12:
        a_pert = a_pert / S
    return a_pert


def perturbed_peak_targeted(a, rng, scale=0.05):
    """Perturbation that specifically targets reducing the current peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)

    a_pert = a.copy()
    if j_end > j_start:
        # Reduce top contributors to the peak
        seg = a_pert[j_start:j_end + 1].copy()
        contrib = np.array([a[k_star - j] for j in range(j_start, j_end + 1)])
        n_reduce = max(5, n // 30)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        for idx in top_idx:
            a_pert[j_start + idx] *= (1.0 - scale * (0.5 + rng.random()))

        # Distribute mass to positions far from the peak
        reduction_total = a.sum() - a_pert.sum()
        if reduction_total > 1e-12:
            # Add to positions not in the peak window
            non_peak = np.concatenate([np.arange(0, j_start), np.arange(j_end + 1, n)])
            if len(non_peak) > 0:
                n_add = min(len(non_peak), max(10, n // 10))
                add_idx = non_peak[rng.choice(len(non_peak), size=n_add, replace=False)]
                per = reduction_total / len(add_idx)
                for idx in add_idx:
                    a_pert[idx] += per

    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = a_pert.sum()
    if S > 1e-12:
        a_pert = a_pert / S
    return a_pert
