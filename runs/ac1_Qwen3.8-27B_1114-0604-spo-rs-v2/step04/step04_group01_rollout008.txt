   - Subgradient makes a precise, targeted improvement
   - Alternating avoids the "spiky landscape" problem of high-beta LSE

3. **Two-phase beta schedule**: 
   - Phase A: beta = [10, 30, 80, 200, 500] for basin finding
   - Phase B: beta = [500, 1000, 2000, 5000, 10000, 20000, 50000] for refinement, with subgradient interleaved

4. **Adaptive active set in subgradient**: Try k = 1, 2, 3, 5, 8 and pick the best.

5. **Better perturbation**: Use the convolution structure to create informed perturbations (e.g., perturb elements that contribute most to the peaks).

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LSE-continuation L-BFGS-B
interleaved with exact subgradient multiplicative updates and adaptive
step-size bisection. Multi-start with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def fun_lse_tuple(a, beta):
    return lse_score_grad_tuple(a, beta)

def run_lbfgs(obj_func, a0, param, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
        # Estimate time per evaluation
        est_time = 0.02 * max(1.0, n / 4096.0)
        cap = int((0.85 * rem) / max(est_time, 0.001))
        cap = max(32, min(cap, 2000))
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun) if maxfun else 500

    if maxfun < 32:
        maxfun = 32
    maxiter = max(30, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            obj_func, a,
            method="L-BFGS-B", jac=True,
            args=(param,), bounds=bounds, options=opts,
        )
    except Exception:
        return exact_score_a(a), a

    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)

def subgradient_step(a, k_values=(1, 2, 3, 5, 8), etas=(0.01, 0.03, 0.08, 0.15, 0.30, 0.60), deadline=None):
    """
    One subgradient step: try different active set sizes and step sizes,
    pick the best improvement.
    """
    n = len(a)
    best_sc = exact_score_a(a)
    best_a = a.copy()
    
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return best_sc, best_a
    
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return best_sc, best_a
    
    for k in k_values:
        if deadline is not None and time.monotonic() > deadline:
            return best_sc, best_a
        w = active_weights(d, min(k, d.size))
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            continue
        
        for eta in etas:
            if deadline is not None and time.monotonic() > deadline:
                return best_sc, best_a
            q = a * np.exp(-eta * h / hmax)
            q = normalize_a(q)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
    
    return best_sc, best_a

def subgradient_polish(a, max_iters=30, deadline=None):
    """
    Multiple subgradient steps with adaptive step sizes.
    """
    best_a = normalize_a(a)
    best_sc = exact_score_a(best_a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Try a range of k values and step sizes
        sc, a2 = subgradient_step(
            best_a,
            k_values=(1, 2, 3, 5, 8),
            etas=(0.01, 0.03, 0.08, 0.15, 0.30, 0.60),
            deadline=deadline
        )
        
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        else:
            # No improvement, try smaller steps
            sc, a2 = subgradient_step(
                best_a,
                k_values=(1, 2),
                etas=(0.005, 0.01, 0.02, 0.05),
                deadline=deadline
            )
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            else:
                break  # Stuck
    
    return best_sc, best_a

def generate_inits(n, rng, parent, cap=30):
    inits = []
    n = int(n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.05, None)  # Keep positive to avoid log issues
        inits.append(normalize_a(arr))
    
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Spectral modification
        N = next_power_of_two(n)
        spec = np.fft.rfft(pr, N)
        # Attenuate high frequencies
        freqs = np.fft.rfftfreq(N) * N
        atten = 1.0 / (1.0 + (freqs / max(1, n // 4)) ** 2)
        spec2 = spec * atten
        add(np.fft.irfft(spec2, N)[:n])
    
    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n))
    
    # Linear ramps
    for alpha in (0.3, 0.5, 0.7, 0.9):
        add(1.0 + alpha * (2.0 * x - 1.0))
    
    # Cosine shapes
    for f in (1, 2, 3):
        add(1.0 + 0.3 * np.cos(2.0 * np.pi * f * x))
        add(1.0 - 0.3 * np.cos(2.0 * np.pi * f * x))
    
    # Trigonometric
    max_trig = 8
    for r in range(max_trig):
        vals = np.ones(n)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            freq = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.1, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * freq * x + ph)
        add(vals)
    
    # Random
    for _ in range(5):
        add(0.1 + 0.9 * rng.random(n))
    
    # Block structure
    for nb in (16, 32, 64):
        if nb > n:
            nb = max(2, n // 2)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals, n - a.size)])
        add(a[:n])
    
    return inits

def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Pressure-informed perturbation
        d, S = d_from_a(p)
        if S > 1e-12 and d.size > 0:
            dmax = float(np.max(d))
            if dmax > 0:
                w = active_weights(d, 5)
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * 0.5 * h / hmax)
                    return normalize_a(q)
        q = p * np.exp(0.1 * rng.standard_normal(n))
    else:
        # Smooth noise
        noise = rng.standard_normal(n)
        k = min(21, max(3, n // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    
    return normalize_a(q)

def optimize_with_continuation(a_start, deadline, rng, label, max_time=600.0):
    """
    Optimize using LSE continuation + interleaved subgradient polishing.
    """
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    n = len(a)
    
    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))
    
    # Phase 1: LSE continuation (basin finding)
    # Beta schedule: start low, end high
    lse_betas = [20, 50, 100, 200, 400, 800]
    # Time fractions for each beta
    lse_fracs = [0.10, 0.15, 0.20, 0.20, 0.15, 0.20]
    
    total_lse_time = min(max_time * 0.35, deadline - t0 - 60.0)
    if total_lse_time < 60.0:
        total_lse_time = max(30.0, min(60.0, deadline - t0 - 30.0))
    
    cum = 0.0
    for i, (beta, frac) in enumerate(zip(lse_betas, lse_fracs)):
        sub_dead = t0 + (cum + frac) * total_lse_time
        sub_dead = min(sub_dead, deadline - 30.0)
        cum += frac
        
        while time.monotonic() < sub_dead - 3.0:
            rem = sub_dead - time.monotonic()
            if rem < 5.0:
                break
            est_time = 0.02 * max(1.0, n / 4096.0)
            mf = int((0.8 * rem) / max(est_time, 0.001))
            mf = max(32, min(mf, 1000))
            
            sc, a2 = run_lbfgs(fun_lse_tuple, a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] LSE beta={:.0f} score={:.8f}".format(label, beta, best_sc))
            a = normalize_a(a2)
    
    # Phase 2: Interleaved LSE + subgradient (refinement)
    refine_dead = min(t0 + max_time * 0.85, deadline - 60.0)
    
    refine_betas = [400, 800, 1500, 3000, 6000, 12000, 25000, 50000]
    n_rounds = len(refine_betas)
    time_per_round = max(10.0, (refine_dead - time.monotonic()) / n_rounds)
    
    for i, beta in enumerate(refine_betas):
        if time.monotonic() > refine_dead - 5.0:
            break
        
        round_dead = time.monotonic() + time_per_round
        round_dead = min(round_dead, refine_dead)
        
        # LSE step
        if time.monotonic() < round_dead - 5.0:
            rem = round_dead - time.monotonic() - 5.0
            est_time = 0.02 * max(1.0, n / 4096.0)
            mf = int((0.7 * rem) / max(est_time, 0.001))
            mf = max(32, min(mf, 500))
            
            sc, a2 = run_lbfgs(fun_lse_tuple, a, float(beta), round_dead - 10.0, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] Refine LSE beta={:.0f} score={:.8f}".format(label, beta, best_sc))
            a = normalize_a(a2)
        
        # Subgradient step
        if time.monotonic() < round_dead - 3.0:
            sc, a2 = subgradient_polish(a, max_iters=15, deadline=round_dead - 3.0)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] Refine Subgrad beta={:.0f} score={:.8f}".format(label, beta, best_sc))
            a = normalize_a(a2)
    
    return best_sc, best_a
