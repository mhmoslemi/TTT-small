        step = min(step, 0.3)

        if Sa < 1e-12:

"""
Multi-strategy optimizer using mirror descent (multiplicative updates), L-BFGS 
with adaptive LSE smoothing, and exact subgradient polishing.
Key innovation: exponential-weights mirror descent for effective exploration 
in the non-negative simplex, combined with high-precision L-BFGS refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS callback
_current_beta = 1000.0


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # dL/da[i] = 2 * (p * rev_a)[n-1+i]
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_result = np.fft.irfft(P * RA, M2_fft)[:M2]
    dL_da = 2.0 * conv_result[n - 1:n - 1 + n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_callback(x):
    """Callback for L-BFGS using global beta."""
    return lse_score_and_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    global _current_beta
    _current_beta = beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_callback, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-12:
        a_out = a0 / a0.sum()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent(a0, beta, n_iters=10000, eta0=0.05, decay=0.9995):
    """Mirror descent (exponential weights) with LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        sc, grad = lse_score_and_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

        # Center gradient for normalization
        g_centered = grad - np.mean(grad)

        # Exponential weights update
        exponent = -eta * g_centered
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        eta *= decay

    return best_a


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters=3000, f_lb=1.0, top_k=5):
    """Subgradient using top-k peaks for a smoother descent direction."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.1

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        # Use top-k peaks weighted
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(c))
        for rank, idx in enumerate(top_idx):
            weights[idx] = (top_k - rank) / (top_k * (top_k + 1) / 2.0)

        Mak_weighted = float(np.sum(weights * c))
        sc = 2.0 * n * float(np.max(c)) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Weighted subgradient
        sg = np.zeros(n)
        S2 = S_a * S_a
        S3 = S2 * S_a
        for idx in top_idx:
            w = weights[idx]
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_end >= j_start:
                ii = np.arange(j_start, j_end + 1)
                sg[ii] += w * 2.0 * a[idx - ii]

        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak_weighted / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (0.999 ** it)
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a

def targeted_peak_flatting(a, n_sweeps=10, n_top=50):
    """Targeted coordinate descent: reduce peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()

        contrib = seg[::-1].copy()

        top_count = min(n_top, max(1, total_len // 3))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 0.9, 0.95]:

                a[j] = new_val

                    bottom_local = np.argsort(contrib)[:max(1, total_len - top_count)]
                    per = reduction / len(bottom_local)

                if S > 1e-12:

                    best_val = new_val

            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:

        report(sc_md, a_init, f"init_{i}")

        sc = compute_score(a_md)
        report(sc, a_md, f"mirror_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: L-BFGS with increasing beta (50% time)
    # ==================================================================
    phase2_deadline = t_start + 0.70 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: L-BFGS Refinement (n={n_work}) ---")

    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining_s = phase2_deadline - time.monotonic()
        if remaining_s < 30:
            break
        maxiter = int(min(50000, max(3000, remaining_s * 40)))
        maxfun = int(min(200000, max(10000, remaining_s * 150)))

        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")

        # If we have time, try a second pass from slightly different start
        if time.monotonic() < phase2_deadline - 60 and beta >= 1000:
            a_pert = perturbed_restart(best_a, rng, scale=0.01)
            sc2, a_opt2, _ = run_lbfgs(a_pert, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS beta={beta} pert")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Iterated local search with mirror + LBFGS (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: Iterated Local Search ---")

    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        remaining_s = phase3_deadline - time.monotonic()
        if remaining_s < 60:
            break

        scale = 0.02 + 0.04 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Mirror descent first for exploration
        a_md = mirror_descent(a_pert, beta=100, n_iters=2000, eta0=0.05)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, f"ILS#{trial} mirror")

        # Then L-BFGS for refinement
        beta = 500 + int(1000 * rng.random())
        maxiter = int(min(30000, max(2000, remaining_s * 15)))
        maxfun = int(min(100000, max(5000, remaining_s * 60)))
        sc, a_opt, _ = run_lbfgs(np.maximum(a_md, 1e-10), beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial} lbfgs b={beta}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Subgradient polishing + targeted peak flattening (10% time)
    # ==================================================================
    phase4_deadline = t_start + 0.95 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: Subgradient + Peak Flattening ---")

    # Nesterov subgradient
    a_ns = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
    sc = compute_score(a_ns)
    report(sc, a_ns, "nesterov")

    # Multi-peak subgradient
    a_mp = multi_peak_subgrad(best_a, n_iters=2000, f_lb=1.0, top_k=5)
    sc = compute_score(a_mp)
    report(sc, a_mp, "multi_peak_sg")

    # Targeted peak flattening
    a_tpf = targeted_peak_flatting(best_a, n_sweeps=10, n_top=50)
    sc = compute_score(a_tpf)
    report(sc, a_tpf, "peak_flat")

    # Another Nesterov pass on the best
    if time.monotonic() < phase4_deadline - 30:
        a_ns2 = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "nesterov2")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 30:
        for beta in [100000, 300000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000)
            report(sc, a_opt, f"final beta={beta}")

    # One more mirror descent pass at high precision
    if time.monotonic() < deadline - 20:
        a_md_final = mirror_descent(best_a, beta=5000, n_iters=3000, eta0=0.01)
        sc = compute_score(a_md_final)
        report(sc, a_md_final, "mirror_final")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
