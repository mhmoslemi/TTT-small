Convex optimization approach to flat-autocorrelation non-negative sequences.
Uses exact LP solving via cvxpy in an alternating minimization framework,
combined with LSE-smoothed L-BFGS polishing and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import coo_matrix
import time
import cvxpy as cp

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix C where C @ x = cross_conv(a, x)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_start < 2 * n - 1:
            k_range = range(k_start, k_end)
            rows.extend(k_range)
            cols.extend([j] * (k_end - k_start))
            vals.extend([a[j]] * (k_end - k_start))
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def lp_solve(a, M):
    """Solve: max sum(x) s.t. cross_conv(a,x)[k] <= M for all k, x >= 0"""
    n = len(a)
    x = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(a)
    obj = cp.Maximize(cp.sum(x))
    constraints = [C @ x <= M]
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order of preference
    for solver in [cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if x.value is not None:
                return np.maximum(x.value, 0)
        except Exception:
            continue
    return None

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a: 2 * cross-correlation(a, p)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:

    else:
        a = a / S

    best_a = a.copy()

    no_improve = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        x = lp_solve(a, M)
        if x is None:

        if Sx < 1e-12:
            break
        x = x / Sx

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:

        if sc < best_sc:

def peak_reduction(a, n_peaks=5, n_iters=50):
    """Targeted peak reduction: identify top peaks and reduce them."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        threshold = c[peak_indices[-1]]
        
        # For each peak at position k, the contributing pairs are (i, k-i)
        # Reduce the product a[i]*a[k-i] by reducing the larger of a[i] and a[k-i]
        reduction = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                other = k - j
                # We want to reduce a[j]*a[other]
                # Reduce the one that's larger (or both proportionally)
                if a[j] > a[other]:
                    reduction[j] += a[j] * 0.01
                else:
                    reduction[other] += a[other] * 0.01
        
        a_new = a - reduction
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

def _lp_fista(a, M, n_iters=3000, beta_start=5.0, beta_end=50.0):
    """FISTA-based LP solver (fallback when cvxpy fails)."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_iters):
        frac = min(1.0, it / (n_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x

def solve_dual_lp(b, n):
    """
    Solve: min sum(y) s.t. sum_{k=j}^{j+n-1} y[k] >= b[j] for j=0..n-1, y >= 0.
    Returns optimal y (length 2n-1) and optimal value.
    """
    y = np.zeros(2 * n - 1)
    # Process intervals left to right
    # Maintain a running sum of y in the current window
    # Window j = [j, j+n-1]
    
    # Use a "difference array" to track contributions
    # When we add to y[pos], it affects all windows that contain pos.
    # Window j contains pos if j <= pos <= j+n-1, i.e., pos-n+1 <= j <= pos.
    # So adding to y[pos] affects windows max(0, pos-n+1) through min(pos, n-1).
    
    # Greedy: for each window j (left to right), if the current sum in window j < b[j],
    # add the deficit to y[j+n-1] (the rightmost point of window j).
    
    # To efficiently compute the sum in window j, maintain a running total.
    # When we move from window j to j+1:
    #   - y[j] leaves the window
    #   - y[j+n] enters the window
    # But y[j] was never set (j < n-1 for the windows we process, and we only set y at positions >= n-1)
    
    # Actually, let's just track the running sum directly.
    window_sum = 0.0
    for j in range(n):
        # Enter y[j+n] (for j < n-1; for j = n-1, y[j+n] = y[2n-1] which doesn't exist)
        # Wait, window j covers [j, j+n-1]. When we go from j to j+1:
        #   - position j leaves
        #   - position j+n enters
        # But at j=0, window covers [0, n-1]. Sum = sum(y[0:n]).
        # At j=1, window covers [1, n]. Sum = sum(y[1:n+1]).
        
        # Let me just compute it directly for now (O(n) per step, O(n^2) total)
        # For n=6962, that's ~48.5M operations, which is fine.
        
        s = np.sum(y[j:j+n])
        if s < b[j]:
            deficit = b[j] - s
            y[j+n-1] += deficit
    
    return y, np.sum(y)

"""
LP-alternating optimizer using exact cvxpy LP solves for flat-autocorrelation
non-negative sequences. Combines exact LP solving (via SCS), LSE-smoothed
L-BFGS polishing, adaptive multi-length exploration, and targeted peak
reduction to minimize 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_structure(n):
    """Precompute row/col indices for convolution matrix C where C@x = cross_conv(a,x).
    C[k,j] = a[j] if j <= k <= j+n-1. Total nonzeros = n^2."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for j in range(n):
        count = n
        rows[idx:idx + count] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + count] = j
        idx += count
    return rows, cols


def make_conv_matrix(a, rows, cols):
    """Build sparse convolution matrix given precomputed structure and current a."""
    n = len(a)
    vals = np.empty(n * n, dtype=np.float64)
    for j in range(n):
        vals[j * n:(j + 1) * n] = a[j]
    from scipy.sparse import coo_array
    return coo_array((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_solve_cvxpy(a, M, rows, cols):
    """Solve: max sum(x) s.t. cross_conv(a,x)[k] <= M for all k, x >= 0."""
    n = len(a)
    C = make_conv_matrix(a, rows, cols)
    x = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(x))
    constraints = [C @ x <= M]
    prob = cp.Problem(obj, constraints)
    for solver in [cp.SCS, cp.ECOS, cp.MOSEK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if x.value is not None:
                return np.maximum(np.asarray(x.value, dtype=np.float64), 0.0)
        except Exception:
            continue
    return None


def lp_solve_fista(a, M, n_iters=5000, beta_start=10.0, beta_end=100.0):
    """FISTA-based LP solver (fast approximate)."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    for it in range(n_iters):
        frac = min(1.0, it / (n_iters * 0.25))
        beta = beta_start + (beta_end - beta_start) * frac
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    return x


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step_size = 0.02
    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_end - j_start:-1 if j_end == n - 1 else None][::-1] if j_end - j_start > 0 else 2.0 * a[j_start]
        # Simpler: gradient of max at k_star
        # d/d a[j] of conv[k_star] = a[k_star - j] for valid j
        for j in range(j_start, j_end + 1):
            other = k_star - j
            if 0 <= other < n:
                sg[j] = a[other]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step_size, (sc - 1.0) / (gn * gn + 1e-30))
        step = max(step, 1e-6)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step_size *= 0.999
    return best_a


def peak_shaving(a, n_peaks=5, n_iters=30, strength=0.05):
    """Targeted peak reduction by reducing contributors to top peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = self_conv(a)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        reduction = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                other = k - j
                prod = a[j] * a[other]
                if a[j] >= a[other]:
                    reduction[j] += strength * a[j]
                else:
                    reduction[other] += strength * a[other]
        a_new = a - reduction
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        strength *= 0.9
    return best_a
