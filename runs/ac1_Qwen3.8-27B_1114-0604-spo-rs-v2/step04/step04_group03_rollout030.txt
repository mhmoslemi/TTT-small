```python
    dL_da = 2.0 * r[2*n-2::-1][:n]  # r[2n-2], r[2n-3], ..., r[n-1]
    
    # Full objective: f = 2n*L / S^2
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L * np.ones(n) / (S**3))
    return score, grad
```

Wait, `r[2*n-2::-1][:n]` gives `r[2n-2], r[2n-3], ..., r[n-1]` which is exactly `r[2n-2-j]` for `j=0,...,n-1`. 

Now let me design the full algorithm. I'll focus on:

1. **Starting points**: Use the parent (height_sequence_1) and also try some hand-crafted asymmetric initializations.
2. **Main optimization**: 
   - L-BFGS with progressively increasing beta (LSE smoothing → true max)
   - Between L-BFGS runs, do a "peak flattening" step using mirror descent on the active peaks
3. **Perturbation and re-start**: Periodically perturb the best solution and re-optimize.

Let me also think about what makes this problem hard. The objective is non-smooth (due to the max), so L-BFGS on the LSE approximation works well only when beta is large enough that the LSE closely approximates the max. But for large beta, the problem becomes very "spiky" and hard for L-BFGS.

A good strategy is:
- Start with small beta (smooth landscape, easy for L-BFGS)
- Gradually increase beta
- At each beta level, run L-BFGS to convergence
- This is like a "continuation method"

The previous program does this, but maybe not aggressively enough. Let me try more beta levels and more iterations at high beta.

Also, I want to try a different approach: **projected gradient with the exact subgradient** (not LSE). At each step:
1. Compute `conv(a)`, find all peaks with value >= M - epsilon (active set)
2. Compute the averaged subgradient: `grad[j] = (2/|A|) * sum_{k in A} a[k-j]` (for valid k-j)
3. Take a step: `a_new = a - step * grad`
4. Project: `a_new = max(a_new, 0)`, then rescale to `sum = 1`

The key improvement is using the **active set** (multiple peaks) rather than just the argmax. This gives a smoother gradient signal.

Let me also implement a "smart coordinate descent":
- For the peak at `k*`, identify the top contributing pairs
- For each pair `(i, k*-i)`, try to reduce `max(a[i], a[k*-i])` while adding the mass to a "safe" location
- A "safe" location is one that minimally affects all peaks

```python
"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation L-BFGS with LSE smoothing (increasing beta)
2. Active-set subgradient descent with multiple peaks
3. Smart coordinate descent on top peak contributors
4. Diverse perturbation and re-optimization
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    p = exp_s / exp_s.sum()
    L = c_max + log_sum / beta
    
    # Gradient computation via FFT
    p_flipped = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PF_fft = np.fft.rfft(p_flipped, M2_fft)
    r = np.fft.irfft(A_fft * PF_fft, M2_fft)[:M2]
    dL_da = 2.0 * r[2*n-2::-1][:n]
    
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S**3))
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def active_set_subgradient(a, n_iters=5000, eps=1e-4):
    """
    Subgradient descent using active set of peaks.
    Work on simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        # Active set: peaks within eps of max
        active = np.where(c >= M - eps * M)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Averaged subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += 2.0 * seg[::-1]
        grad /= len(active)
        
        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Adaptive step size
        sc = 2.0 * n * M / (S * S)
        # Use Polyak step: step = (f(a) - f*) / ||g||^2, approximate f* 
        # Lower bound: 2n * mean(conv) / S^2 = 2n / ((2n-1) * S^2) since sum(conv) = S^2
        lb = 2.0 * n / (2 * n - 1)
        step = (sc - lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        step = max(step, 1e-7)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def coordinate_descent_peaks(a, n_sweeps=10, n_top=20):
    """
    Coordinate descent: for each top peak, reduce top contributors,
    add mass to bottom contributors (within the peak's support).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        M = float(c.max())
        # Top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]
        
        improved = False
        for k_star in top_peaks:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            # Contribution of each element to this peak
            contrib = seg[::-1].copy()  # contrib[i] = a[k_star - (j_start + i)]
            
            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            
            if len(bot_local) == 0:
                continue
            
            # Reduce top, add to bottom
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a[gi] * 0.05
                a[gi] = max(a[gi] - red, 0.0)
                total_red += red
            
            if total_red > 0:
                per = total_red / len(bot_local)
                for li in bot_local:
                    a[j_start + li] += per
            
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            improved = True
    
    return best_a


def mirror_descent_multi_peak(a, n_iters=5000, eta0=0.05, n_peaks=3):
    """Mirror descent with top-K peaks in the gradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        # Top K peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Average gradient from top peaks
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        grad /= n_peaks
        
        eta = eta0 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_asymmetric_inits(n, rng):
    """Build diverse asymmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Linear ramp (asymmetric)
    inits.append(0.5 + x)  # 0.5 to 1.5
    inits.append(1.5 - x)  # 1.5 to 0.5
    # Quadratic ramp
    inits.append(0.5 + x**2)
    inits.append(0.5 + (1-x)**2)
    # Exponential ramp
    inits.append(np.exp(2*x) / np.exp(2*x).sum())
    inits.append(np.exp(-2*x) / np.exp(-2*x).sum())
    # Cosine (asymmetric phase)
    inits.append(0.5 + 0.4 * np.cos(2*np.pi*x - np.pi/3))
    inits.append(0.5 + 0.4 * np.cos(2*np.pi*x + np.pi/3))
    # Triangular (asymmetric peak)
    inits.append(1.0 - np.abs(2*x - 0.3))
    inits.append(1.0 - np.abs(2*x - 0.7))
    # Sigmoid-like
    inits.append(1.0 / (1.0 + np.exp(-10*(x - 0.3))))
    inits.append(1.0 / (1.0 + np.exp(-10*(x - 0.7))))
    # Random smooth
    for _ in range(3):
        freq = 1 + 5*rng.random()
        phase = 2*np.pi*rng.random()
        amp = 0.3 + 0.3*rng.random()
        inits.append(0.5 + amp * np.cos(2*np.pi*freq*x + phase))
    # Random (positive)
    inits.append(np.exp(rng.standard_normal(n) * 0.5))
    
    # Ensure all non-negative
    inits = [np.maximum(np.asarray(i, dtype=np.float64), 1e-6) for i in inits]
    return inits
```

For the LSE objective:
`L(a) = c_max + (1/beta) * log(sum_k exp(beta * (c[k] - c_max))))`

where `c[k] = conv(k)` and `c_max = max(c)`.

Actually, let me use a simpler form. Let `c[k] = conv(a, a)[k]` for `k = 0, ..., 2n-2`.

The LSE (log-sum-exp) is: `LSE(c) = (1/beta) * log(sum_k exp(beta * c[k]))`.

`d(LSE)/d(a[j]) = sum_k p[k] * d(c[k])/d(a[j])` where `p[k] = exp(beta*c[k]) / sum_k exp(beta*c[k])`.

And `d(c[k])/d(a[j]) = 2*a[k-j]` if `0 <= k-j < n`, else 0.

So `d(LSE)/d(a[j]) = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]`.

Let `m = k-j`: `= 2 * sum_m p[m+j] * a[m] * 1[0<=m<n] = 2 * sum_{m=0}^{n-1} a[m] * p[m+j]`.

This is what I had before. Good.

`df/da[j] = 2n * [d(LSE)/d(a[j]) / (sum(a))^2 - 2 * LSE * 1 / (sum(a))^3]`
`= 2n * [d(LSE)/d(a[j]) / S^2 - 2*LSE / S^3]`

where `S = sum(a)`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using continuation L-BFGS with LSE
smoothing, active-set subgradient descent, coordinate descent on peak
contributors, and multi-peak mirror descent. Targets score < 1.5027.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective value and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.sum()) / beta
    # Gradient via FFT cross-correlation
    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    r_full = np.fft.irfft(A_fft * PR_fft, M2_fft)[:M2]
    dL_da = 2.0 * r_full[2 * n - 2::-1][:n]
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S ** 3))
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_func(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj_func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def active_set_subgrad(a, n_iters, eps_frac):
    """Subgradient descent using active set of near-max peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        threshold = M * (1.0 - eps_frac)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[int(np.argmax(c))] = True
            n_active = 1
        
        # Averaged subgradient from active peaks
        grad = np.zeros(n)
        for k in range(len(c)):
            if not active_mask[k]:
                continue
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        grad /= n_active
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        sc = 2.0 * n * M
        # Lower bound on objective (with sum=1): 2n * (1/(2n-1)) = 2n/(2n-1)
        lb = 2.0 * n / (2 * n - 1)
        step = (sc - lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        step = max(step, 1e-8)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        sc_now = compute_score(a)
        if sc_now < best_sc:
            best_sc = sc_now
            best_a = a.copy()
    
    return best_a

def mirror_descent_multi(a, n_iters, eta0, n_peaks):

    inits.append(0.3 + 0.5 * x ** 2)
    inits.append(0.3 + 0.5 * (1 - x) ** 2)
    inits.append(np.exp(1.5 * x))
    inits.append(np.exp(-1.5 * x))
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x - 0.5))
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x + 0.5))
    inits.append(1.0 - np.abs(2 * x - 0.4))
    inits.append(1.0 - np.abs(2 * x - 0.6))
    # Random positive
    inits.append(np.exp(0.5 * rng.standard_normal(n)))
    inits.append(np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None))
    
    inits = [np.maximum(np.asarray(i, dtype=np.float64), 1e-8) for i in inits]
    return inits
