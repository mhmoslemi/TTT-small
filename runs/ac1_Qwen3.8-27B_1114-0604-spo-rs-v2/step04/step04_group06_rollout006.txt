def fast_conv(a):

    n = len(a)

    if S < 1e-12:

    c = fast_conv(a)
    S = float(a.sum())

        return 1e10, np.zeros(n)

    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S

    # Also, when i = k-j, we need k-j in [0, n-1], so k in [j, j+n-1].
    # And when i = j, we need j in [0, n-1] (always true) and k-j in [0, n-1], so k in [j, j+n-1].
    
    # So k ranges from j to min(j+n-1, 2*n-2)
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    
    for k in range(k_start, k_end + 1):
        idx = k - j  # this is the index such that a_j * a[idx] or a[idx] * a_j
        if idx == j:
            # k = 2j, the term is a_j^2
            b[k] += new_val * new_val - old_val * old_val
        else:
            # Two terms: a_j * a_idx and a_idx * a_j
            b[k] += 2 * delta * a[idx]
    
    a[j] = new_val
    return b

Minimize 2*n*max(selfconv)/(sum)^2 for non-negative sequences.
Multi-phase optimization: adaptive-beta LSE L-BFGS, incremental
coordinate descent with 1D line search, and randomized local search.

def fast_conv(a):
    """Linear self-convolution via FFT. O(n log n)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed approximation of the score and its gradient.
    L = max(c) + log(sum(exp(beta*(c - max(c)))))/beta
    score = 2n * L / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())

    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
