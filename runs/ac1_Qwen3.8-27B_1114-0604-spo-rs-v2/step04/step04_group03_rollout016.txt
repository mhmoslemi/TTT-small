def peak_shave(a, n_steps=500, top_k=10):
    """
    Direct peak shaving: repeatedly identify the peak convolution lag,
    find top contributors, reduce them, and redistribute to low contributors.
    This is a fast local optimization that complements L-BFGS.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for step in range(n_steps):
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top K peak lags
        top_lags = np.argsort(c)[::-1][:top_k]

        for k_star in top_lags:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()

            # Find top contributors to this peak
            n_top = min(20, len(contrib) // 3)
            n_bot = min(20, len(contrib) // 3)
            top_local = np.argsort(contrib)[::-1][:n_top]
            bot_local = np.argsort(contrib)[:n_bot]

            # Reduce top contributors
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a[gi] * 0.01
                a[gi] = max(a[gi] - red, 0.0)
                total_red += red

            # Redistribute to bottom contributors (outside the peak segment if possible)
            if total_red > 1e-15:
                # Find positions that don't contribute to any top peak
                all_peak_positions = set()
                for ks in top_lags:
                    ks = int(ks)
                    js = max(0, ks - n + 1)
                    je = min(ks, n - 1)
                    all_peak_positions.update(range(js, je + 1))

                # Positions not in any peak segment
                free_positions = [i for i in range(n) if i not in all_peak_positions]
                if free_positions:
                    # Pick the smallest-valued free positions
                    free_sorted = sorted(free_positions, key=lambda i: a[i])
                    n_add = min(n_bot, len(free_sorted))
                    per = total_red / n_add
                    for fi in free_sorted[:n_add]:
                        a[fi] += per
                else:
                    # Redistribute within the segment to bottom contributors
                    per = total_red / max(1, len(bot_local))
                    for li in bot_local:
                        gi = j_start + li
                        a[gi] += per

            S = a.sum()
            if S > 1e-12:
                a = a / S

        # Check improvement
        sc_now = compute_score(a)
        if sc_now < best_sc:
            best_sc = sc_now
            best_a = a.copy()

    return best_a


def spectral_equality(a, n_iters=200):
    """
    In frequency domain, try to make the power spectrum more uniform.
    By Wiener-Khinchin, flat autocorrelation <-> flat power spectrum.
    We iteratively adjust the magnitude spectrum toward uniformity
    while preserving the phase (to keep the signal non-negative).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        amp = np.abs(A)
        phase = np.angle(A)

        # Target: make amp more uniform (geometric mean)
        log_amp = np.log(amp + 1e-15)
        target_log = np.mean(log_amp)
        # Partially move toward target
        alpha = 0.05 + 0.1 * (it / n_iters)
        new_log_amp = log_amp + alpha * (target_log - log_amp)
        new_amp = np.exp(new_log_amp)

        # Reconstruct
        A_new = new_amp * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def mirror_descent_polish(a, n_iters=3000):
    """
    Mirror descent (entropic) on the simplex for minimizing max(conv).
    Uses adaptive step size based on the gap to the theoretical lower bound.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv)
        grad_max = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad_max[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Also consider second-worst peak for smoother updates
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1:
            gap = c_sorted[0] - c_sorted[1]
            if gap < 0.1 * c_sorted[0]:
                k_second = int(np.argsort(c)[::-1][1])
                j_start2 = max(0, k_second - n + 1)
                j_end2 = min(k_second, n - 1)
                if j_end2 >= j_start2:
                    seg2 = a[j_start2:j_end2 + 1]
                    grad_max[j_start2:j_end2 + 1] += 0.5 * seg2[::-1]

        # Mirror descent step with adaptive eta
        eta = 0.5 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * grad_max
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def error_reduction_frequency(a, n_iters=300):
    """
    Frequency-domain error reduction: iteratively reshape the convolution
    to be flatter by modifying the spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-18:
            break

        # Soft threshold: reduce excess above a moving threshold
        alpha = 0.3 + 0.5 * min(1.0, it / n_iters)
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-18:
            break

        # Modify convolution to reduce excess
        conv_mod = conv.copy()
        conv_mod[:2 * n - 1] -= 0.5 * excess[:2 * n - 1]
        # Compensate below mean to preserve total
        below = np.maximum(thresh - conv[:2 * n - 1], 0.0)
        if below.sum() > 0:
            conv_mod[:2 * n - 1] += 0.5 * excess.sum() / below.sum() * below

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0

        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_descent_peak(a, n_sweeps=20):
    """
    Coordinate descent that targets the current peak convolution value.
    For the peak lag k*, adjusts the elements a[j] and a[k*-j] to reduce
    the product while maintaining sum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        for j in range(j_start, j_end + 1):
            partner = k_star - j
            if partner < 0 or partner >= n:
                continue
            if j == partner:
                # a[j]^2 term: reduce a[j]
                a_old = a[j]
                for factor in [0.8, 0.9, 0.95]:
                    a[j] = a_old * factor
                    # Redistribute to a random other position
                    other = np.argmin(a)
                    if other != j:
                        a[other] += a_old * (1 - factor)
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        break
                    a[j] = a_old
                    a[other] -= a_old * (1 - factor)
                    if S > 1e-12:
                        a = a / S
            else:
                # a[j]*a[partner] term: reduce one of them
                a_j = a[j]
                a_p = a[partner]
                # Try reducing the larger one
                if a_j >= a_p:
                    for factor in [0.85, 0.9, 0.95]:
                        a[j] = a_j * factor
                        a[partner] += a_j * (1 - factor)
                        S = a.sum()
                        if S > 1e-12:
                            a = a / S
                        sc = compute_score(a)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                            break
                        a[j] = a_j
                        a[partner] -= a_j * (1 - factor)
                        if S > 1e-12:
                            a = a / S
                else:
                    for factor in [0.85, 0.9, 0.95]:
                        a[partner] = a_p * factor
                        a[j] += a_p * (1 - factor)
                        S = a.sum()
                        if S > 1e-12:
                            a = a / S
                        sc = compute_score(a)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                            break
                        a[partner] = a_p
                        a[j] -= a_p * (1 - factor)
                        if S > 1e-12:
                            a = a / S

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive LSE-beta annealing, peak shaving, spectral methods,
    and mirror descent for comprehensive optimization.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Load parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: len={len(parent)}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # If we have a parent, start from it
    if parent is not None:
        report(compute_score(parent), parent, "parent")

    # Determine working lengths
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 8192

    print(f"Main length: {n_main}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")

    # === Phase 1: Quick exploration with multiple inits (10% time) ===
    phase1_end = t_start + 0.10 * (float(budget_s) - 15)
    print(f"\n--- Phase 1: Exploration (until t={phase1_end - t_start:.0f}s) ---")

    working = best_a.copy() if best_a is not None else None

    if working is None:
        x = np.arange(n_main, dtype=np.float64) / n_main
        inits = [
            0.5 + 0.4 * np.cos(2.0 * np.pi * x),
            0.3 + 0.7 * (2.0 * x - 1.0) ** 2,
            np.clip(0.5 + 0.3 * rng.standard_normal(n_main), 0.01, 1000.0),
            0.5 + 0.3 * np.cos(4.0 * np.pi * x),
        ]
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            sc, a_opt = run_lbfgs(a0, 100, maxiter=5000, maxfun=50000)
            report(sc, a_opt, f"init{i}")

    working = best_a.copy() if best_a is not None else np.ones(n_main) * 0.5

    # === Phase 2: Error reduction + spectral (15% time) ===
    phase2_end = t_start + 0.25 * (float(budget_s) - 15)
    print(f"\n--- Phase 2: Error Reduction + Spectral (until t={phase2_end - t_start:.0f}s) ---")

    a_er = error_reduction_frequency(working, n_iters=200)
    report(compute_score(a_er), a_er, "ER")
    working = best_a.copy()

    a_se = spectral_equality(working, n_iters=100)
    report(compute_score(a_se), a_se, "SPEC")
    working = best_a.copy()

    a_pe = peak_shave(working, n_steps=300, top_k=5)
    report(compute_score(a_pe), a_pe, "PEAK1")
    working = best_a.copy()

    # === Phase 3: Main L-BFGS beta annealing (50% time) ===
    phase3_end = t_start + 0.75 * (float(budget_s) - 15)
    print(f"\n--- Phase 3: L-BFGS Beta Annealing (until t={phase3_end - t_start:.0f}s) ---")

    betas = [50, 100, 200, 500, 1000, 2000, 5000]
    for beta in betas:
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        # Scale maxfun based on remaining time
        # Each fun eval ~1ms for n~8000, so maxfun ~ remaining*1000*0.8
        maxfun = int(min(500000, max(50000, remaining * 300)))
        maxiter = int(min(50000, max(5000, remaining * 30)))

        print(f"  L-BFGS beta={beta}, maxfun={maxfun}, time_left={remaining:.0f}s")
        sc, a_opt = run_lbfgs(working, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        working = best_a.copy()

        # Peak shave between L-BFGS calls
        if time.monotonic() < phase3_end - 60:
            a_ps = peak_shave(working, n_steps=200, top_k=5)
            sc_ps = compute_score(a_ps)
            if sc_ps < best_score:
                best_score = sc_ps
                best_a = a_ps.copy()
                working = best_a.copy()
                print(f"    Peak shave improved to {sc_ps:.8f}")

    # === Phase 4: Diversified local search (10% time) ===
    phase4_end = t_start + 0.88 * (float(budget_s) - 15)
    print(f"\n--- Phase 4: Local Search (until t={phase4_end - t_start:.0f}s) ---")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase4_end:
            break
        strategy = trial % 6
        n_work = len(working)

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.005 + 0.02 * rng.random()
            a_pert = working * (1.0 + scale * rng.standard_normal(n_work))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak reduction
            c = fast_conv(working)
            k_star = int(np.argmax(c))
            a_pert = working.copy()
            j_start = max(0, k_star - n_work + 1)
            j_end = min(k_star, n_work - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(10, n_work // 30)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.02
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                # Redistribute to positions far from peak
                all_peak = set()
                for ks in np.argsort(c)[::-1][:5]:
                    js = max(0, int(ks) - n_work + 1)
                    je = min(int(ks), n_work - 1)
                    all_peak.update(range(js, je + 1))
                free = [i for i in range(n_work) if i not in all_peak]
                if free:
                    free_sorted = sorted(free, key=lambda i: a_pert[i])
                    n_add = min(50, len(free_sorted))
                    per = total_red / n_add
                    for fi in free_sorted[:n_add]:
                        a_pert[fi] += per
                S = a_pert.sum()
                if S > 1e-12:
                    a_pert = a_pert / S
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 0.5 + 5.0 * rng.random()
            phase_r = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            x = np.arange(n_work, dtype=np.float64) / n_work
            a_pert = working * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_r))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Mirror descent from perturbed
            a_md = mirror_descent_polish(working * (1 + 0.01 * rng.standard_normal(n_work)),
                                         n_iters=500)
            a_pert = a_md
        elif strategy == 4:
            # Block rescale
            a_pert = working.copy()
            n_blocks = 5 + int(rng.integers(0, 10))
            block_size = max(1, n_work // n_blocks)
            for b in range(min(n_blocks, n_work // block_size)):
                s = b * block_size
                e = min(s + block_size, n_work)
                factor = 0.8 + 0.4 * rng.random()
                a_pert[s:e] *= factor
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
        else:
            # Coordinate descent from current
            a_pert = coordinate_descent_peak(working, n_sweeps=5)

        sc_pert = compute_score(a_pert)
        if sc_pert >= best_score * 0.99:
            # Close enough to be worth refining with L-BFGS
            remaining = phase4_end - time.monotonic()
            if remaining > 30:
                maxfun = int(min(100000, remaining * 500))
                maxiter = int(min(10000, remaining * 50))
                sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=maxiter, maxfun=maxfun)
                report(sc, a_opt, f"local#{trial} s={strategy}")
                working = best_a.copy()

    # === Phase 5: Final polish (7% time) ===
    print(f"\n--- Phase 5: Final Polish ---")
    remaining = deadline - time.monotonic()
    print(f"  Time remaining: {remaining:.0f}s")

    if remaining > 120:
        # Long L-BFGS with highest beta
        maxfun = int(min(300000, remaining * 500))
        maxiter = int(min(30000, remaining * 50))
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, "final LBFGS b=5000")

    if remaining > 60:
        # Final peak shave
        a_ps = peak_shave(best_a, n_steps=500, top_k=10)
        sc_ps = compute_score(a_ps)
        report(sc_ps, a_ps, "final peak shave")

    if remaining > 30:
        # Mirror descent polish
        a_md = mirror_descent_polish(best_a, n_iters=int(min(3000, remaining * 20)))
        sc_md = compute_score(a_md)
        report(sc_md, a_md, "final mirror")

    if remaining > 20:
        # One more L-BFGS from the best
        sc, a_opt = run_lbfgs(best_a, 3000, maxiter=int(min(10000, remaining * 200)),
                               maxfun=int(min(100000, remaining * 2000)))
        report(sc, a_opt, "very final LBFGS")

    # === Final: Try a couple more lengths if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Try a different length
        for n_try in [4096, 12288, 16384]:
            if time.monotonic() > deadline - 60:
                break
            if n_try == len(best_a):
                continue
            # Interpolate best solution to new length
            a_interp = np.interp(
                np.linspace(0, 1, n_try),
                np.linspace(0, 1, len(best_a)),
                best_a
            )
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            sc_i = compute_score(a_interp)
            if sc_i < best_score * 1.01:
                # Worth refining
                maxfun = int(min(100000, (deadline - time.monotonic()) * 300))
                if maxfun > 10000:
                    sc, a_opt = run_lbfgs(a_interp, 500, maxiter=5000, maxfun=maxfun)
                    report(sc, a_opt, f"alt_len n={n_try}")

    # === Final verification ===
    if best_a is not None:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    else:
        out = [0.5] * 64

    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
