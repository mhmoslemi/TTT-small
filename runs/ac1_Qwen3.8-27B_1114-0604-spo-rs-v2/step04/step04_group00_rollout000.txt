"""
Hybrid optimizer: Mirror Descent + Fienup + LSE-Anncaled L-BFGS + Top-M Subgradient.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: geometric beta annealing for LSE, multiplicative weights for
global progress, adaptive subgradient with curvature estimation.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def fienup_flat(a, n_iters=200, aggressive=False):

    Fienup-style iteration: flatten |DFT(a)|^2 to reduce auto-correlation peaks.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        if aggressive:
            # Target top 20-40% of bins
            top_frac = 0.2 + 0.2 * progress
            thresh = np.quantile(P, 1.0 - top_frac)
            above = P > thresh
            if above.any():
                scale = np.ones_like(P)
                scale[above] = np.sqrt(thresh / P[above]) * (0.9 + 0.05 * progress)
                A = A * scale
        else:
            # Gentle flattening
            thresh = P_mean * (1.2 + 0.3 * (1 - progress))
            above = P > thresh
            if above.any():
                scale = np.ones_like(P)
                clip = 0.3 + 0.4 * progress
                scale[above] = (thresh / P[above]) ** (clip * 0.5)
                A = A * scale
        
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def lse_objective_and_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = float(np.log(exp_scaled.sum())) + scaled.max()
    L = c_max + (log_sum - scaled.max()) / beta  # Wait, this is wrong
    # Let me redo: LSE = c_max + (1/beta) * log(sum(exp(beta*(c-c_max))))
    # = c_max + (1/beta) * log_sum_exp(beta*(c-c_max))
    # For numerical stability: log_sum_exp(x) = x_max + log(sum(exp(x - x_max)))
    # Here x = beta*(c-c_max), x_max = 0 (since c_max is the max)
    # So log_sum_exp = log(sum(exp(beta*(c-c_max))))
    # And LSE = c_max + (1/beta) * log(sum(exp(beta*(c-c_max))))
    
    # Let me just compute directly with the shift
    shifted = beta * (c - c_max)  # max is 0
    max_shifted = float(shifted.max())  # should be 0
    log_sum_exp = max_shifted + float(np.log(np.exp(shifted - max_shifted).sum()))
    L = c_max + log_sum_exp / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient of LSE w.r.t. a:
    # dLSE/da_j = (1/beta) * sum_k p_k * d(c_k)/da_j
    # where p_k = exp(beta*(c_k - c_max)) / sum(exp(beta*(c-c_max)))
    # d(c_k)/da_j = 2*a_{k-j} if 0 <= k-j < n
    p = np.exp(shifted - max_shifted)
    p = p / p.sum()
    
    # conv of p with reversed a, taking the relevant part
    # dLSE/da_j = (2/beta) * sum_{k: 0<=k-j<n} p_k * a_{k-j}
    # = (2/beta) * (p convolved with a) evaluated at appropriate indices
    # Actually: sum_k p_k * a_{k-j} for k such that 0<=k-j<n
    # Let m = k-j, then k = m+j, and we need 0<=m<n, 0<=m+j<2n-1
    # So sum_{m=0}^{n-1} p_{m+j} * a_m (where p_{m+j} is 0 if m+j >= 2n-1)
    # This is the cross-correlation of p and a, evaluated at lag j.
    
    # Use FFT: this is the convolution of p with reversed a, indexed appropriately
    # (p * a_rev)(j + n - 1) gives sum_m p_{m} * a_rev_{j+n-1-m} = sum_m p_m * a_{n-1-j-n+1+m} = sum_m p_m * a_{m-j}
    # Hmm, let me just do it directly with FFT.
    # We want: g_j = sum_{k} p_k * 2*a_{k-j} * [0<=k-j<n]
    # = 2 * sum_{m=0}^{n-1} p_{m+j} * a_m
    # This is the cross-correlation of p and a at lag j.
    # cross_corr(j) = sum_m a_m * p_{m+j}
    # = (a conv p)(j + n - 1) where a conv p has length 2n-1
    # Actually: (a * p)(k) = sum_m a_m * p_{k-m}
    # We want sum_m a_m * p_{m+j} = sum_m a_m * p_{(m+j)} 
    # Let l = m+j: sum_l a_{l-j} * p_l ... this is getting confusing.
    
    # Let me use the direct formula:
    # g_j = (2/beta) * sum_{k=max(j,0)}^{min(2n-2, j+n-1)} p_k * a_{k-j}
    #     = (2/beta) * sum_{m=0}^{n-1} p_{m+j} * a_m  [where p_{m+j}=0 if m+j>=2n-1]
    
    # This is the cross-correlation. Using FFT:
    # corr(j) = sum_m a_m * p_{m+j} = (FFT technique)
    # = IFFT(FFT(a) * conj(FFT(p))) ... no, that's not right either.
    
    # Actually, let's just use the convolution:
    # sum_m a_m * p_{m+j} = (a * p_flipped)(j + n - 1) where p_flipped[m] = p_{n-1-m}
    # No wait: (a * q)(k) = sum_m a_m * q_{k-m}
    # If q = p (length 2n-1), then (a * p)(j + n - 1) = sum_m a_m * p_{j+n-1-m}
    # That's not what we want.
    
    # We want: sum_m a_m * p_{m+j}
    # Let's define r = reversed p: r_k = p_{2n-2-k}
    # (a * r)(k) = sum_m a_m * r_{k-m} = sum_m a_m * p_{2n-2-k+m}
    # We want p_{m+j}, so 2n-2-k+m = m+j => k = 2n-2-j
    # So sum_m a_m * p_{m+j} = (a * r)(2n-2-j)
    
    # Or more simply, use the fact that:
    # sum_m a_m * p_{m+j} = sum_k p_k * a_{k-j} (reindexing)
    # This is the cross-correlation of p and a at lag j.
    # In FFT: cross_corr = IFFT(conj(FFT(a_padded)) * FFT(p_padded))
    # But the indexing is tricky. Let me just compute it directly for correctness.
    
    # For efficiency with large n, let's use FFT:
    # Define A = FFT(a, M2) and P = FFT(p, M2) where M2 >= 2n
    # Then the linear cross-correlation is:
    # corr[j] = sum_m a[m] * p[m+j] for m = max(0,-j) to min(n-1, 2n-2-j)
    # This equals IFFT(A * conj(P))[j] (with appropriate conventions)
    
    # Actually, the standard result is:
    # (a ⋆ p)[j] = sum_m a[m] * conj(p[m-j]) = IFFT(FFT(a) * conj(FFT(p)))[j]
    # For real sequences: (a ⋆ p)[j] = sum_m a[m] * p[m-j]
    # We want sum_m a[m] * p[m+j] = (a ⋆ p)[-j] = (p ⋆ a)[j]
    
    M2 = 3 * n  # enough for the cross-correlation
    M2_fft = 1 << (M2 - 1).bit_length()
    A_ft = np.fft.rfft(a, M2_fft)
    P_ft = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(A_ft * P_ft, M2_fft)  # This gives (a * p) linear conv
    # (a * p)[k] = sum_m a[m] * p[k-m]
    # We want sum_m a[m] * p[m+j] = sum_m a[m] * p[(m+j)]
    # = (a * p)[m+j] summed over m... no.
    # sum_m a[m] * p[m+j]: let k = m+j, then m = k-j
    # = sum_k a[k-j] * p[k] (with a[k-j] = 0 if k-j not in [0,n-1])
    # This is the cross-correlation (p ⋆ a)[j] = sum_k p[k] * a[k-j]
    # = (a * p_reversed)[(n-1) + j] where p_reversed[k] = p[2n-2-k]
    # Hmm, let me just verify with a small example mentally.
    
    # Actually, the simplest: sum_m a[m] * p[m+j] is the j-th value of the cross-correlation
    # In terms of convolution: it equals (a * p_flip)[n-1+j] where p_flip[k] = p[2n-2-k]
    # Because (a * p_flip)[n-1+j] = sum_m a[m] * p_flip[n-1+j-m] = sum_m a[m] * p[2n-2-(n-1+j-m)]
    # = sum_m a[m] * p[n-1-j+m]
    # That's sum_m a[m] * p[m + (n-1-j)], which is the (n-1-j)-th cross-correlation.
    
    # I'm going in circles. Let me just use a direct but vectorized approach:
    # g_j = (2/beta) * sum_{m=0}^{n-1} p[m+j] * a[m]
    # For each j, this is a dot product of a with a shifted version of p.
    # Using np.convolve or manual FFT.
    
    # Simplest correct approach: use the convolution
    # Let's define q of length 2n-1 where q[k] = p[k]
    # We want c[x] = sum_{m=0}^{n-1} a[m] * q[m+x] for x = 0, 1, ..., n-1
    # This is the cross-correlation. In numpy:
    # corr = np.correlate(q, a, mode='full') gives corr[k] = sum_n q[n+k] * a[n]... 
    # Actually np.correlate(v, w, 'full')[k] = sum_n v[n+k] * conj(w[n])
    # So np.correlate(p, a, 'full')[j] = sum_n p[n+j] * a[n] for valid n
    # That's exactly what we want!
    
    # But np.correlate is O(n^2). For n=7000, that's 49M operations per evaluation.
    # Too slow. Let's use FFT.
    
    # The cross-correlation via FFT:
    # (a ⋆ p)[j] = sum_m a[m] * p[m+j] = IFFT(conj(FFT(a)) * FFT(p)) at index j
    # Wait, the standard result is:
    # (f ⋆ g)[n] = sum_k f[k] * conj(g[k-n]) = IFFT(FFT(f) * conj(FFT(g)))[n]
    # For real: (f ⋆ g)[n] = sum_k f[k] * g[k-n]
    # We want sum_m a[m] * p[m+j] = sum_m a[m] * p[m-(-j)] = (a ⋆ p)[-j] = (p ⋆ a)[j]
    # So: what we want at index j = (p ⋆ a)[j] = IFFT(FFT(p) * conj(FFT(a)))[j]
    
    # Let me verify: IFFT(FFT(p) * conj(FFT(a)))[j] = sum_k p[k] * a[k-j] (circular)
    # For linear (with zero padding): sum_k p[k] * a[k-j] for valid k, k-j
    # = sum_k p[k] * a[k-j] where 0<=k<2n-1 and 0<=k-j<n
    # Let m = k-j: sum_m p[m+j] * a[m] where 0<=m<n and 0<=m+j<2n-1
    # = sum_{m=0}^{n-1} p[m+j] * a[m] (with p[m+j]=0 if m+j>=2n-1)
    # YES! That's exactly what we want.
    
    corr = np.fft.irfft(P_ft * np.conj(A_ft), M2_fft)
    g_lse = (2.0 / beta) * corr[:n]
    
    # Full gradient: d(score)/da_j = (2n/S^2) * (g_lse_j - 2*L/S)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * g_lse - (4.0 * n * L / S3) * np.ones(n)
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) * 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return a_out

def lse_annealing(a0, betas, maxiter_per_beta=3000):
    """
    Run L-BFGS with geometrically annealed beta values.
    Start smooth (low beta), end sharp (high beta).
    """
    a = normalize(a0)
    for beta in betas:
        a = run_lbfgs(a, beta, maxiter=maxiter_per_beta)
    return a

def mirror_descent(a, n_iters=5000, eta_init=0.1, eta_decay=0.9999):
    """
    Multiplicative weights / mirror descent with KL divergence.
    Naturally handles non-negativity.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) w.r.t. a:
        # g_j = 2 * a_{k_star - j} if 0 <= k_star - j < n
        g = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                g[j] = 2.0 * a[k_j]
        
        # Multiplicative update (mirror descent with KL):
        # a_j <- a_j * exp(-eta * (g_j - mean(g)))
        # Normalize after
        g_centered = g - g.mean()
        a = a * np.exp(-eta * g_centered)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        eta *= eta_decay
    
    return best_a

def top_m_subgradient(a, n_iters=5000, top_m=10, eta_init=0.05):
    """Top-M peaks subgradient with adaptive step."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                g[j_start:j_end+1] += 2.0 * a[j_start:j_end+1][::-1]
        
        # Normalize gradient
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        
        # Adaptive step: larger when far from bound
        gap = max(sc - sc_lb, 1e-6)
        eta = min(eta * 1.001, gap / (gn * gn + 1e-30), 0.1)
        
        a -= eta * g
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a
