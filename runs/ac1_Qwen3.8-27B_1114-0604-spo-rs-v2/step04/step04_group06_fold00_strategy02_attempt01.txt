def evaluate_sequence(seq):
    # Calculate self-convolution and score (as provided)
    n = len(seq)
    conv = np.convolve(seq, seq, mode='full')
    max_conv = float(conv.max())
    sum_seq = float(sum(seq))
    return 2.0 * n * max_conv / (sum_seq ** 2)

def uniform_sequence(length):
    return np.ones(length) / float(length)

    # Ensure non-negative and within bounds
    safe_seq = np.clip(seq, 0.0, 1000.0)
    # Normalize to sum >= 0.01 if using Adam/L-BFGS
    total = sum(safe_seq)
    if total < 0.01:
        safe_seq *= 1000 / 0.01
    return safe_seq

def pad_sequence(seq):
    # For convolution calculations (surrogate optimization convenience)
    pad_length = 2*len(seq) - 1
    extended_seq = np.pad(seq, (0, pad_length - len(seq)), mode='constant')
    return extended_seq
