def fast_conv(a):

    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    
    return a_new

def small_perturbation(a, rng, strength=0.05):
    """Small random perturbation."""
    a_new = a * (1.0 + strength * rng.standard_normal(len(a)))
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new

def coordinate_sparsification(a, n_tries=20):
    """Try zeroing out coordinates one at a time."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    
    # Sort by a[j] ascending (try zeroing small ones first)
    order = np.argsort(a)
    
    for j in order[:n_tries]:
        old_val = a[j]
        a[j] = 0.0
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
        else:
            a[j] = old_val
    
    return best_score, a
