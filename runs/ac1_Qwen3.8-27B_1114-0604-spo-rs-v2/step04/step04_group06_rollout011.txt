Minimize 2n*max(conv(a,a))/(sum a)^2 for non-negative sequences.
Uses LP-based iterative method (fix f, solve LP for g, update f)

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def conv_with_fixed_f(f, g):
    """Compute f * g via FFT (f is fixed, g varies)."""
    n = len(f)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    F = np.fft.rfft(f, M_fft)
    G = np.fft.rfft(g, M_fft)
    return np.fft.irfft(F * G, M_fft)[:M]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix_sparse(f):
    """Build sparse matrix T where (T@g)_k = (f*g)_k."""
    n = len(f)
    rows = 2 * n - 1
    cols = n
    A = lil_matrix((rows, cols))

        if d < 0 or d >= n:

        val = f[d]
        if val == 0:
            continue

        j_end = min(n, rows - d)
        for j in range(j_start, j_end):
            A[j + d, j] = val
    return A.tocsr()

def solve_lp_for_g(f, M, time_limit=30):
    """Solve: max sum(g) s.t. T_f @ g <= M, g >= 0."""
    import cvxpy as cp
    n = len(f)

    A = build_conv_matrix_sparse(f)
    
    g = cp.Variable(n)

            if g.value is not None:

        except Exception as e:

    """Run one LP iteration: fix f, solve for g, return g."""
    M = float(fast_conv(f).max())
    g = solve_lp_for_g(f, M, time_limit=time_per_lp)
    if g is None or g.sum() < 1e-12:
