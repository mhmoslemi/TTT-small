"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta, n_peaks=0):
    """
    LSE-smoothed score and gradient.
    If n_peaks > 0, only consider the top n_peaks values of c in the LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if n_peaks > 0:
        # Only consider top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        c_sub = c[top_idx]
        c_max = float(c_sub.max())
        scaled = beta * (c_sub - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()
        
        # Gradient: only contributions from top peaks
        dL_da = np.zeros(n)
        for j, k_j in enumerate(top_idx):
            j_start = max(0, int(k_j) - n + 1)
            j_end = min(int(k_j), n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end+1][::-1]
            dL_da[j_start:j_end+1] += 2.0 * p[j] * seg
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()
        
        # Gradient via FFT
        rev_a = a[::-1]
        M2 = 3 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        r = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
        dL_da = 2.0 * r
    
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000, n_peaks=0):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_obj_grad(x, beta, n_peaks)
    
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return a_out


def spectral_peak_reduction(a, n_iters=500, eta0=0.05):
    """
    Reduce peaks in the convolution by modifying the spectrum.
    Works in the frequency domain to distribute energy more evenly.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        power = np.abs(A) ** 2
        c = np.fft.irfft(power, L)[:2 * n - 1]
        
        k_star = int(np.argmax(c))
        sc = 2.0 * n * float(c.max())
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute the "peak correlation" in frequency domain
        # c[k*] = (1/L) * sum_m |A[m]|^2 * re(exp(2*pi*i*m*k*/L))
        # For rfft: m = 0, 1, ..., L//2
        m = np.arange(L // 2 + 1, dtype=np.float64)
        cos_vals = np.cos(2.0 * np.pi * m * k_star / L)
        
        # Reduce power where cos > 0 (constructive at peak)
        # Increase where cos < 0 (destructive at peak)
        eta = eta0 / (1.0 + 0.05 * it)
        log_factor = -eta * cos_vals
        log_factor[0] = 0.0  # Preserve DC (sum)
        
        power_new = power * np.exp(log_factor)
        power_new = np.maximum(power_new, 0.0)
        
        # Reconstruct A with new magnitude, same phase
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
    
    return best_a


def top_k_subgradient(a, n_iters=2000, K=5):
    """
    Subgradient descent using top-K peaks for smoother gradients.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-K peaks
        top_idx = np.argsort(c)[::-1][:K]
        
        # Gradient as average of subgradients at top peaks
        grad = np.zeros(n)
        for k_j in top_idx:
            k_j = int(k_j)
            j_start = max(0, k_j - n + 1)
            j_end = min(k_j, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end+1][::-1]
            grad[j_start:j_end+1] += 2.0 * seg
        grad /= K
        
        # Scale for the full objective subgradient
        S2 = S * S
        grad = (4.0 * n) * (grad / S2 - M / (S2 * S))
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        step = max(step, 1e-7)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def error_reduction(a, n_iters=100):
    """
    Frequency-domain error reduction: flatten the convolution spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    L = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, L)
        power = A * np.conj(A)
        conv = np.fft.irfft(power, L)[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        alpha = 0.5 + 0.4 * (it / max(1, n_iters // 2))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv
        
        err_full = np.zeros(L, dtype=np.float64)
        err_full[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_full, L)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0
        
        damp = 0.3 + 0.4 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * d
        
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE-smoothed L-BFGS with peak focusing, spectral-domain
peak reduction, top-K subgradient descent, error reduction, and
multi-length exploration with diverse restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta, n_peaks=0):
    """
    LSE-smoothed score and gradient.
    If n_peaks > 0, only consider top n_peaks values in the LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if n_peaks > 0:
        top_idx = np.argsort(c)[::-1][:n_peaks]
        c_sub = c[top_idx]
        c_max = float(c_sub.max())
        scaled = beta * (c_sub - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        L = c_max + float(np.log(exp_s.sum())) / beta
        p = exp_s / exp_s.sum()

        dL_da = np.zeros(n)
        for j in range(len(top_idx)):
            k_j = int(top_idx[j])
            j_start = max(0, k_j - n + 1)
            j_end = min(k_j, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1][::-1]
            dL_da[j_start:j_end + 1] += 2.0 * p[j] * seg
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        L = c_max + float(np.log(exp_s.sum())) / beta
        p = exp_s / exp_s.sum()

        rev_a = a[::-1]
        M2 = 3 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        dL_da = 2.0 * r

    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000, n_peaks=0):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta, n_peaks)

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)


def spectral_peak_reduction(a, n_iters=500, eta0=0.04):
    """
    Reduce peaks by modifying the power spectrum.
    Reduces |A[m]|^2 where it constructively interferes at the peak lag.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    m_arr = np.arange(L // 2 + 1, dtype=np.float64)

    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        power = np.abs(A) ** 2
        c = np.fft.irfft(power, L)[:2 * n - 1]

        k_star = int(np.argmax(c))
        sc = 2.0 * n * float(c.max())
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        cos_vals = np.cos(2.0 * np.pi * m_arr * k_star / L)

        # Also consider the second peak for more robust reduction
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.8 * c_sorted[0]:
            k_star2 = int(np.argmax(c - c.max() * np.ones(2 * n - 1) * (1 - 1e-9)))
            # Find second peak index
            idx_max = int(np.argmax(c))
            c_masked = c.copy()
            c_masked[idx_max] = -1
            k_star2 = int(np.argmax(c_masked))
            if k_star2 != k_star:
                cos_vals2 = np.cos(2.0 * np.pi * m_arr * k_star2 / L)
                cos_vals = 0.7 * cos_vals + 0.3 * cos_vals2

        eta = eta0 / (1.0 + 0.03 * it)
        log_factor = -eta * cos_vals
        log_factor[0] = 0.0

        power_new = power * np.exp(log_factor)
        power_new = np.maximum(power_new, 0.0)

        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

    return best_a


def top_k_subgradient(a, n_iters=2000, K=5):
    """Subgradient descent using top-K peaks for smoother gradients."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:K]

        grad = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            j_start = max(0, kk - n + 1)
            j_end = min(kk, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * seg
        grad /= float(K)

        S2 = S * S
        grad = (4.0 * n) * (grad / S2 - M / (S2 * S))

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.02)
        step = max(step, 1e-7)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def error_reduction(a, n_iters=100):
    """Frequency-domain error reduction: flatten the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    L = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, L)
        power = A * np.conj(A)
        conv = np.fft.irfft(power, L)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        alpha = 0.5 + 0.4 * min(1.0, it / max(1, n_iters // 2))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv

        err_full = np.zeros(L, dtype=np.float64)
        err_full[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_full, L)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0

        damp = 0.3 + 0.4 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent.astype(np.float64)).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 50.0))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: error reduction, L-BFGS (adaptive beta, peak-focused),
    spectral reduction, top-K subgradient, local search, final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [6962, 4096, 8192, 10240, 12288, 16384]
    if parent is not None:
        lengths.insert(0, len(parent))
    # Remove duplicates preserving order
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths

    print(f"Lengths: {lengths}")
    print(f"Parent: {'yes (len=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: Quick exploration - error reduction (12% time)
    # ============================================================
    p1_end = t_start + 0.12 * total_budget
    print(f"\n=== Phase 1: Error Reduction (t<{(p1_end-t_start):.0f}s) ===")

    for n in lengths[:3]:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > p1_end:
                break
            a_er = error_reduction(a0, n_iters=60)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    # ============================================================
    # Phase 2: L-BFGS with adaptive beta (38% time)
    # ============================================================
    p2_end = t_start + 0.50 * total_budget
    print(f"\n=== Phase 2: L-BFGS (t<{(p2_end-t_start):.0f}s) ===")

    # Main L-BFGS on best
    beta_schedule = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
    for beta in beta_schedule:
        if time.monotonic() > p2_end:
            break
        elapsed = time.monotonic() - t_start
        remaining = p2_end - time.monotonic()
        # Scale iterations by available time
        maxiter = min(10000, max(1000, int(remaining / 0.02)))
        maxfun = min(200000, max(5000, int(remaining / 0.001)))
        a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"LBFGS b={beta}")

    # Peak-focused LSE (concentrate on top peaks)
    if time.monotonic() < p2_end:
        for n_peaks in [3, 5, 8]:
            if time.monotonic() > p2_end:
                break
            a_opt = run_lbfgs(best_a, 500, maxiter=5000, maxfun=80000, n_peaks=n_peaks)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"LBFGS topK={n_peaks}")

    # Try other lengths
    if time.monotonic() < p2_end:
        for n in lengths[1:4]:
            if time.monotonic() > p2_end:
                break
            if n == len(best_a):
                continue
            a_interp = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, len(best_a)),
                best_a
            )
            a_opt = run_lbfgs(a_interp, 500, maxiter=4000, maxfun=60000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"LBFGS n={n}")

            a_opt2 = run_lbfgs(a_opt, 5000, maxiter=4000, maxfun=60000)
            sc2 = compute_score(a_opt2)
            report(sc2, a_opt2, f"LBFGS n={n} b=5000")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Spectral + Subgradient + Local search (25% time)
    # ============================================================
    p3_end = t_start + 0.75 * total_budget
    print(f"\n=== Phase 3: Spectral + Subgradient + Local (t<{(p3_end-t_start):.0f}s) ===")

    # Spectral peak reduction
    if time.monotonic() < p3_end:
        a_sp = spectral_peak_reduction(best_a, n_iters=200, eta0=0.03)
        sc = compute_score(a_sp)
        report(sc, a_sp, "spectral")

        # Follow up with L-BFGS
        if time.monotonic() < p3_end:
            a_opt = run_lbfgs(a_sp, 2000, maxiter=4000, maxfun=60000)
            sc = compute_score(a_opt)
            report(sc, a_opt, "spectral+LBFGS")

    # Top-K subgradient
    if time.monotonic() < p3_end:
        for K in [3, 5, 10]:
            if time.monotonic() > p3_end:
                break
            n_it = int(min(2000, (p3_end - time.monotonic()) / 0.03))
            a_sg = top_k_subgradient(best_a, n_iters=max(200, n_it), K=K)
            sc = compute_score(a_sg)
            report(sc, a_sg, f"topK-SG K={K}")

    # Local search with diverse perturbations
    if time.monotonic() < p3_end:
        n_best = len(best_a)
        n_trials = 15
        for trial in range(n_trials):
            if time.monotonic() > p3_end:
                break
            strategy = trial % 5
            if strategy == 0:
                # Gaussian perturbation
                scale = 0.01 + 0.03 * rng.random()
                a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            elif strategy == 1:
                # Targeted peak reduction
                c = fast_conv(best_a)
                k_star = int(np.argmax(c))
                a_pert = best_a.copy()
                j_start = max(0, k_star - n_best + 1)
                j_end = min(k_star, n_best - 1)
                seg = a_pert[j_start:j_end + 1]
                contrib = seg[::-1].copy()
                top_count = max(10, n_best // 40)
                top_local = np.argsort(contrib)[::-1][:top_count]
                total_red = 0.0
                for li in top_local:
                    gi = j_start + li
                    red = a_pert[gi] * 0.04
                    a_pert[gi] = max(a_pert[gi] - red, 0.0)
                    total_red += red
                if total_red > 0:
                    bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                    per = total_red / len(bottom_local)
                    for li in bottom_local:
                        a_pert[j_start + li] += per
                S = a_pert.sum()
                if S > 1e-12:
                    a_pert = a_pert / S
            elif strategy == 2:
                # Sinusoidal perturbation
                freq = 1.0 + 10.0 * rng.random()
                phase = 2.0 * np.pi * rng.random()
                amp = 0.015 + 0.02 * rng.random()
                x = np.arange(n_best, dtype=np.float64) / n_best
                a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            elif strategy == 3:
                # Error reduction restart
                a_pert = error_reduction(
                    best_a * (1 + 0.02 * rng.standard_normal(n_best)),
                    n_iters=50
                )
            else:
                # Block rescale
                a_pert = best_a.copy()
                n_blocks = 4 + int(rng.integers(0, 8))
                block_size = max(1, n_best // n_blocks)
                for b in range(min(n_blocks, n_best // block_size)):
                    s = b * block_size
                    e = min(s + block_size, n_best)
                    factor = 0.7 + 0.6 * rng.random()
                    a_pert[s:e] *= factor
                S = a_pert.sum()
                if S > 1e-12:
                    a_pert = a_pert / S

            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_opt = run_lbfgs(a_pert, 1000, maxiter=4000, maxfun=60000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"local#{trial} s={strategy}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final intensive polish (25% time)
    # ============================================================
    p4_end = deadline
    print(f"\n=== Phase 4: Final Polish (t<{(p4_end-t_start):.0f}s) ===")

    # Very high beta L-BFGS for precise optimization
    for beta in [5000, 20000, 50000]:
        if time.monotonic() > p4_end:
            break
        remaining = p4_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(15000, int(remaining / 0.015))
        maxfun = min(300000, int(remaining / 0.001))
        a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"final LBFGS b={beta}")

    # Error reduction + L-BFGS
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        if remaining > 60:
            a_er = error_reduction(best_a, n_iters=100)
            sc = compute_score(a_er)
            report(sc, a_er, "final ER")

            if time.monotonic() < p4_end:
                a_opt = run_lbfgs(a_er, 10000, maxiter=5000, maxfun=100000)
                sc = compute_score(a_opt)
                report(sc, a_opt, "ER+LBFGS b=10000")

    # Spectral reduction + L-BFGS
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        if remaining > 60:
            a_sp = spectral_peak_reduction(best_a, n_iters=150, eta0=0.02)
            sc = compute_score(a_sp)
            report(sc, a_sp, "final spectral")

            if time.monotonic() < p4_end:
                a_opt = run_lbfgs(a_sp, 10000, maxiter=5000, maxfun=100000)
                sc = compute_score(a_opt)
                report(sc, a_opt, "spectral+LBFGS b=10000")

    # Top-K SG + L-BFGS
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        if remaining > 40:
            n_it = int(min(1500, remaining / 0.04))
            a_sg = top_k_subgradient(best_a, n_iters=max(100, n_it), K=5)
            sc = compute_score(a_sg)
            report(sc, a_sg, "final topK-SG")

            if time.monotonic() < p4_end:
                a_opt = run_lbfgs(a_sg, 10000, maxiter=4000, maxfun=80000)
                sc = compute_score(a_opt)
                report(sc, a_opt, "SG+LBFGS b=10000")

    # One more round: error reduction on the very best
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        if remaining > 40:
            a_er2 = error_reduction(best_a, n_iters=80)
            sc = compute_score(a_er2)
            report(sc, a_er2, "ER2")
            if time.monotonic() < p4_end:
                a_opt = run_lbfgs(a_er2, 30000, maxiter=3000, maxfun=60000)
                sc = compute_score(a_opt)
                report(sc, a_opt, "ER2+LBFGS b=30000")

    # ============================================================
    # Final verification and return
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
