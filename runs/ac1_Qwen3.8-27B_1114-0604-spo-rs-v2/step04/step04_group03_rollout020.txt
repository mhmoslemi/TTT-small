"""
Optimized search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE smoothing, multi-peak coordinate descent,
and spectral peak shaving to approach the theoretical optimum.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S**3))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def spectral_peak_shave(a, n_iters=100, alpha=0.1):
    """
    Reduce peaks in the autocorrelation by modifying the spectrum.
    In frequency domain, autocorrelation ~ IDFT(|A|^2).
    We reduce |A|^2 at the frequencies contributing most to time-domain peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        power = A * A  # This is the spectrum of the autocorrelation (up to scaling)
        
        # Compute autocorrelation
        conv = np.fft.irfft(power, M)[:2*n-1]
        
        # Find peak
        k_star = int(np.argmax(conv))
        
        # The peak at k_star is contributed by all frequencies.
        # conv[k_star] = (1/M) * sum_omega |A(omega)|^2 * exp(i*omega*k_star)
        # To reduce conv[k_star], we want to reduce |A(omega)|^2 where exp(i*omega*k_star) is positive
        # relative to the mean.
        
        # Actually, let's use a simpler approach: reduce the magnitude of A
        # at frequencies where they contribute most to the peak.
        
        # The contribution of frequency omega to conv[k_star] is:
        # |A(omega)|^2 * cos(omega * k_star) (for real signals, the imaginary parts cancel)
        
        omega = np.fft.rfftfreq(M, d=1.0/M) * 2 * np.pi
        contrib_to_peak = (A * A).real * np.cos(omega * k_star)
        
        # Reduce frequencies with positive contribution
        reduce_mask = contrib_to_peak > 0
        if reduce_mask.sum() == 0:
            break
        
        # Scale down the magnitude at those frequencies
        reduction = alpha / (1.0 + it * 0.1)
        A[reduce_mask] *= (1.0 - reduction)
        
        # Also ensure we don't make A too small (maintain some baseline)
        # Redistribute: add a small uniform component
        A += 1e-10  # tiny regularization
        
        a_new = np.fft.irfft(A, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def iterative_peak_reduction(a, n_iters=500):
    """
    Iteratively reduce the maximum peak using steepest descent on the LSE objective
    with very high beta (essentially targeting the true max).
    Uses backtracking line search for stability.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    beta = 1000.0
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Gradient of the true max (subgradient)
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Proper gradient of 2n*M/S^2
        grad = (2.0 * n) * (grad / (S*S) - 2.0 * M_val / (S*S*S))
        
        # But we want to MINIMIZE, so move in -grad direction
        # with constraint sum = 1 (projected gradient)
        # a_new = a - step * grad, then project to {a >= 0, sum = 1}

        if gn < 1e-20:

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        
        # Backtracking line search

            trial_step = step * (0.5 ** ls)
            a_new = a - trial_step * grad
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()

                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                break
        
        # Gradually increase beta for LSE version (if we switch to LSE)
    
    return best_a


def run_lbfgs_normalized(a0, beta, maxiter=10000, maxfun=200000):
    """
    L-BFGS on LSE objective with explicit sum normalization.
    Works with a where sum(a) = 1.
    """
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None and len(parent) == n:
        p = parent.astype(np.float64).copy()
        p = p / p.sum()
        inits.append(p)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine shapes (symmetric)
    inits.append((0.5 + 0.4 * np.cos(2.0 * np.pi * x)) / n)
    inits.append((0.3 + 0.7 * np.cos(2.0 * np.pi * x)**2) / n)
    
    # Ramp (lopsided to reduce center peak)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # Exponential decay
    exp_seq = np.exp(-2.0 * x)
    inits.append(exp_seq / exp_seq.sum())
    
    # Double exponential
    dbl_exp = np.exp(-3.0 * np.abs(2.0 * x - 1.0))
    inits.append(dbl_exp / dbl_exp.sum())
    
    # Random
    r = np.abs(rng.standard_normal(n)) + 0.01
    inits.append(r / r.sum())
    
    return inits

"""
Optimized search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE smoothing with progressive beta, multi-peak coordinate descent,
spectral peak shaving, and subgradient refinement to approach the theoretical optimum.
Key innovations: active-set-aware coordinate descent, spectral domain peak reduction,
and aggressive high-beta LSE polishing with careful line search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] = 2 * (conv(rev_a, p))[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective with sum normalization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multi_peak_coord_descent(a, n_sweeps=10, n_top_peaks=5, n_top_contrib=100):
    """
    Coordinate descent targeting top-K peaks.
    For each peak, reduces top contributors and redistributes to low contributors.
    Uses line search to ensure improvement.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    line_factors = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        n_peaks = min(n_top_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        improved_this_sweep = False
        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg * seg[::-1]
            n_contrib = min(n_top_contrib, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_contrib]
            top_global = j_start + top_local

            for factor in line_factors:
                a_trial = a.copy()
                total_red = 0.0
                for gi in top_global:
                    red = a_trial[gi] * factor
                    a_trial[gi] = max(a_trial[gi] - red, 0.0)
                    total_red += red

                if total_red < 1e-15:
                    continue

                # Redistribute to lowest contributors
                bottom_count = max(1, len(contrib) - n_contrib)
                bottom_local = np.argsort(contrib)[:bottom_count]
                bottom_global = j_start + bottom_local
                per = total_red / len(bottom_global)
                for gi in bottom_global:
                    a_trial[gi] += per

                S_trial = a_trial.sum()
                if S_trial < 1e-12:
                    continue
                a_trial = a_trial / S_trial
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved_this_sweep = True
                    break

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if not improved_this_sweep and sweep > 3:
            break

    return best_a


def spectral_peak_shave(a, n_iters=200, alpha=0.05):
    """
    Reduce peaks in the autocorrelation by modifying the signal spectrum.
    Iteratively reduces |A(w)|^2 at frequencies contributing positively to the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()
    omega = np.fft.rfftfreq(M, d=1.0 / M) * 2.0 * np.pi

    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        power_real = (A * A).real
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        k_star = int(np.argmax(conv))
        if conv.max() < 1e-15:
            break

        # Contribution of each frequency to conv[k_star]
        contrib = power_real * np.cos(omega * k_star)
        reduce_mask = contrib > 0
        n_reduce = reduce_mask.sum()
        if n_reduce < 10:
            break

        # Scale down magnitude at contributing frequencies
        reduction = alpha / (1.0 + 0.05 * it)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_mag[reduce_mask] *= (1.0 - reduction)
        # Add small floor to prevent complete cancellation
        A_mag = np.maximum(A_mag, 1e-15)
        A_new = A_mag * np.exp(1j * A_phase)

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_refine(a, n_iters=5000, tol=1e-10):
    """
    Projected subgradient descent on the true max objective.
    Works with sum(a)=1 normalization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # theoretical lower bound ~ 1 (for large n)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - sc_lb < tol:
            break

        # Subgradient of max at the argmax
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Gradient of full objective 2n*M/S^2
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M_val / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-20:
            break

        # Adaptive step: Polyak step
        step = max(1e-10, (sc - sc_lb)) / (gn * gn + 1e-30)
        step = min(step, 0.02)

        # Backtracking
        for ls in range(15):
            a_new = a - step * (0.5 ** ls) * grad
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                break
        else:
            a = a_new if S_new > 1e-12 else a

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations for optimization."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            p = parent.astype(np.float64).copy()
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-12:
                inits.append(p / p.sum())
        else:
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, np.maximum(parent, 0.0))
            if p.sum() > 1e-12:
                inits.append(p / p.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine shapes
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01) / n)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2, 0.01) / n)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x), 0.01) / n)

    # Ramp (reduces center peak)
    ramp = np.linspace(0.05, 1.0, n)
    inits.append(ramp / ramp.sum())

    # Exponential decay
    exp_seq = np.exp(-2.0 * x)
    inits.append(exp_seq / exp_seq.sum())

    # Double exponential (U-shaped)
    dbl = np.exp(-4.0 * np.abs(2.0 * x - 1.0))
    inits.append(dbl / dbl.sum())

    # Sinc-like
    sinc_seq = np.sinc(3.0 * (x - 0.5))
    sinc_seq = np.maximum(sinc_seq, 0.01)
    inits.append(sinc_seq / sinc_seq.sum())

    # Random
    r = np.abs(rng.standard_normal(n)) + 0.01
    inits.append(r / r.sum())

    return inits
