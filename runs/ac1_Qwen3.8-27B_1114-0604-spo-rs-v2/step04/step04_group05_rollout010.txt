def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient via cross-correlation
    a_rev = a[::-1]
    M2 = (2 * n - 1) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    sg = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * L / S3)
    return score, grad


def peak_subgrad(a):
    """Exact subgradient of max(self-conv) at the peak lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    # Subgradient: 2 * a[k_star - i] for valid i

    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    return k_star, sg, float(c[k_star])


def subgradient_step(a, lr, beta_lse=None):
    """
    One subgradient step. If beta_lse is None, use exact peak subgradient.
    Otherwise use LSE-smoothed gradient.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    if beta_lse is None:
        k_star, sg, M = peak_subgrad(a)
        score = 2.0 * n * M / (S * S)
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S * S * S))
    else:
        score, grad = lse_score_grad(a, beta_lse)
    # Polyak step: (score - lower_bound) / ||grad||^2
    lower_bound = 2.0 * n / (2.0 * n - 1)
    gap = max(score - lower_bound, 0.01)
    g_norm_sq = float(np.dot(grad, grad))
    if g_norm_sq < 1e-20:
        return a
    polyak_lr = gap / g_norm_sq
    step = min(lr, polyak_lr)
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    # Renormalize to keep sum stable
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new * (S / S_new)
    return a_new

def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing and bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        obj_count[0] += 1
        if time_budget is not None and t_start is not None:

                raise TimeoutError
        return lse_score_grad(x, beta)

    try:

            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,

        return compute_score(a_out), a_out

    a_out = np.clip(res.x, 0.0, 1000.0)

    """
    Main search function: Adaptive LSE continuation + subgradient polishing
    + antisymmetric bias + diverse restarts + surgical coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    best_state = {'score': np.inf, 'a': None, 't_start': t_start}

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={n}, score={parent_score:.8f}")
        best_state['score'] = parent_score
        best_state['a'] = parent.copy()
    else:
        n = 4096
        parent = np.ones(n) / n
        parent_score = compute_score(parent)
        print(f"No parent, uniform n={n}, score={parent_score:.8f}")
        best_state['score'] = parent_score
        best_state['a'] = parent.copy()

    # ==================================================================
    # Phase 1: Intensive optimization from parent (45% of budget)
    # ==================================================================
    p1_budget = effective_budget * 0.45
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive optimization from parent (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    a_work = best_state['a'].copy()

    # 1a: Subgradient warm-up (10% of phase 1)
    sg_time = p1_budget * 0.10
    if time.monotonic() < t_start + effective_budget - 20:
        sc, a_sg = polyak_subgradient_descent(a_work, n_steps=5000, lr_init=1.0,
                                               time_budget=sg_time, t_start=t_start)
        try_update_global(best_state, sc, a_sg, "Subgradient warmup")
        a_work = best_state['a']

    # 1b: L-BFGS continuation with many stages (80% of phase 1)
    cont_time = p1_budget * 0.80
    if time.monotonic() < t_start + effective_budget - 20:
        # 2x increase per stage, 10 stages
        betas = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240]
        sc, a_opt = continuation_lbfgs(a_work, betas, cont_time, t_start,
                                        maxiter_per_stage=3000)
        try_update_global(best_state, sc, a_opt, "Continuation LBFGS")
        a_work = best_state['a']

    # 1c: Quick coordinate descent (10% of phase 1)
    cd_time = p1_budget * 0.10
    if time.monotonic() < t_start + effective_budget - 15:
        sc, a_opt = coord_descent_peak(a_work, n_sweeps=3, top_k=30,
                                       time_budget=cd_time, t_start=t_start)
        try_update_global(best_state, sc, a_opt, "Phase1 CoordDesc")

    # ==================================================================
    # Phase 2: Diverse restarts (35% of budget)
    # ==================================================================
    p2_budget = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget
    n_restarts = 10
    per_restart_time = p2_budget / (n_restarts + 1)

    strategies = [
        ("antisym", antisymmetric_bias),
        ("spectral", spectral_perturbation),
        ("block_swap", block_swap_perturbation),
        ("gaussian", gaussian_perturbation),
        ("chirp", chirp_perturbation),
        ("antisym_strong", None),  # handled inline
        ("spectral_weak", None),
        ("combined1", None),
        ("combined2", None),
        ("shuffled", None),
    ]

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break
        time_left = p2_deadline - time.monotonic()
        time_alloc = min(time_left * 0.85, per_restart_time * 1.5)
        if time_alloc < 20:
            break

        name, fn = strategies[ri % len(strategies)]
        base = best_state['a']

        if fn is not None:
            a_pert = fn(base, rng, **({"strength": 0.1} if "strength" in fn.__code__.co_varnames else {}))
        else:
            if name == "antisym_strong":
                a_pert = antisymmetric_bias(base, rng, strength=0.15)
            elif name == "spectral_weak":
                a_pert = spectral_perturbation(base, rng, strength=0.05)
            elif name == "combined1":
                a_pert = antisymmetric_bias(spectral_perturbation(base, rng, 0.08), rng, 0.08)
            elif name == "combined2":
                a_pert = gaussian_perturbation(block_swap_perturbation(base, rng, 2), rng, 3)
            else:  # shuffled
                a_pert = base.copy()
                # Random block shuffle
                n_blk = 5
                indices = list(range(n // n_blk))
                rng.shuffle(indices)
                a_new = np.empty_like(a_pert)
                for new_pos, old_blk in enumerate(indices):
                    s = old_blk * (n // n_blk)
                    e = (old_blk + 1) * (n // n_blk)
                    ns = new_pos * (n // n_blk)
                    ne = (new_pos + 1) * (n // n_blk)
                    a_new[ns:ne] = a_pert[s:e]
                a_pert = a_new

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            continue

        sc, a_opt = refine_candidate(a_pert, time_alloc, t_start)
        try_update_global(best_state, sc, a_opt, f"restart#{ri}:{name}")

    # ==================================================================
    # Phase 3: Intensive polish of best (15% of budget)
    # ==================================================================
    p3_budget = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive polish (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    a_work = best_state['a'].copy()
    p3_deadline = t_start + effective_budget * 0.80 - 10

    # High-beta L-BFGS
    for beta in [5000, 10000, 20000, 50000]:
        if time.monotonic() > p3_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=5000, maxfun=50000,
                              time_budget=60.0, t_start=t_start)
        try_update_global(best_state, sc, a_opt, f"Polish LBFGS beta={beta}")
        print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        a_work = best_state['a']

    # Extended coordinate descent
    if time.monotonic() < p3_deadline - 10:
        cd_time = min(p3_deadline - time.monotonic() - 5, 80.0)
        sc, a_opt = coord_descent_peak(a_work, n_sweeps=8, top_k=50,
                                       time_budget=cd_time, t_start=t_start)
        try_update_global(best_state, sc, a_opt, "Polish CoordDesc")

    # ==================================================================
    # Phase 4: Try slightly different lengths (5% of budget)
    # ==================================================================
    p4_budget = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 4: Length variations (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_state['a'])
    for factor in [0.75, 1.25, 1.5]:
        if time.monotonic() > t_start + effective_budget - 30:
            break
        tl = int(n_best * factor)
        if tl < 256 or tl > 30000:
            continue
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_state['a'])
        a_res = np.clip(a_res, 0.0, 1000.0)
        if a_res.sum() < 0.01:
            continue
        time_alloc = min(p4_budget * 0.5, 40.0)
        sc, a_opt = refine_candidate(a_res, time_alloc, t_start)
        try_update_global(best_state, sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 5: Final ultra-precise polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0
    a_work = best_state['a'].copy()

    # Very high beta L-BFGS
    if time.monotonic() < final_deadline - 15:
        sc, a_opt = run_lbfgs(a_work, 100000, maxiter=10000, maxfun=100000,
                              time_budget=60.0, t_start=t_start)
        try_update_global(best_state, sc, a_opt, "Final LBFGS beta=100000")

    # Final coordinate descent with more sweeps
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = coord_descent_peak(best_state['a'], n_sweeps=10, top_k=60,
                                       time_budget=min(remaining - 3, 60.0),
                                       t_start=t_start)
        try_update_global(best_state, sc, a_opt, "Final CoordDesc")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_state['a']]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_state['score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
