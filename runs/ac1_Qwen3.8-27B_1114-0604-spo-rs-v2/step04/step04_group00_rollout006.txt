import cvxpy as cp
import numpy as np

def lp_direction(a, M):
    """Solve LP to find direction of improvement.
    Maximize sum(b) subject to b >= 0, (a*b)_k <= M for all k.
    """
    n = len(a)
    b = cp.Variable(n, non_neg=True)
    
    # Build convolution constraint matrix
    # (a*b)_k = sum_j a_j * b_{k-j} for valid j, k-j
    # This is a matrix-vector product: C @ b <= M * ones
    
    # C is (2n-1) x n, C[k, j] = a[j] if 0 <= k-j <= n-1, else 0
    # But this is huge. Let's use the fact that it's a convolution.
    
    # Actually, for cvxpy, we can express this as:
    # For each k, sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * b[k-j] <= M
    # Let i = k-j: sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * b[i] <= M
    
    # The constraint matrix A has A[k, i] = a[k-i] if 0 <= k-i <= n-1
    # This is a "convolution matrix"
    
    # For efficiency, let's build it as a sparse matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            A[k, k-j] = a[j]
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
        return b.value
    except:
        return None

"""
LP-based block coordinate descent + Fienup + L-BFGS hybrid.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. LP-based direction finding (from state of the art) with reduced constraint set
2. Adaptive step size with backtracking
3. Cycling between LP, Fienup, and L-BFGS methods
4. High-beta LSE smoothing for final polish
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import time
import cvxpy as cp


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def solve_lp_direction(a, M, top_k=200, n_extra=100, rng=None):
    """
    Solve LP: max sum(b) s.t. (a*b)_k <= M for selected k, 0 <= b_j <= 1.
    Returns the optimal b or None.
    """
    n = len(a)
    c = fast_conv(a)
    
    # Select constraints
    n_c = len(c)
    if top_k >= n_c:
        sel_idx = np.arange(n_c)
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
        remaining = np.setdiff1d(np.arange(n_c), top_idx)
        if n_extra > 0 and len(remaining) > 0:
            extra_idx = rng.choice(remaining, size=min(n_extra, len(remaining)), replace=False)
            sel_idx = np.concatenate([top_idx, extra_idx])
        else:
            sel_idx = top_idx
    
    sel_idx = sel_idx.astype(int)
    n_sel = len(sel_idx)
    
    # Build sparse constraint matrix
    A = lil_matrix((n_sel, n))
    for i, k in enumerate(sel_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            cols = np.arange(j_start, j_end + 1)
            rows = np.full(len(cols), i)
            vals = a[k - cols]  # a[k-j] for j in [j_start, j_end]
            A[rows, cols] = vals
    
    b_var = cp.Variable(n, non_neg=True)
    constraints = [A.tocsr() @ b_var <= M, b_var <= 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        if b_var.value is not None:
            b_val = np.asarray(b_var.value).flatten()
            if np.all(np.isfinite(b_val)) and b_val.sum() > 0:
                return b_val
    except Exception:
        pass
    return None


def solve_lp_full(a, M, rng=None):
    """Solve LP with ALL constraints. Slower but more accurate."""
    n = len(a)
    n_c = 2 * n - 1
    
    # Build full sparse constraint matrix
    A = lil_matrix((n_c, n))
    for k in range(n_c):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        cols = np.arange(j_start, j_end + 1)
        rows = np.full(len(cols), k)
        vals = a[k - cols]
        A[rows, cols] = vals
    
    b_var = cp.Variable(n, non_neg=True)
    constraints = [A.tocsr() @ b_var <= M, b_var <= 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        if b_var.value is not None:
            b_val = np.asarray(b_var.value).flatten()
            if np.all(np.isfinite(b_val)) and b_val.sum() > 0:
                return b_val
    except Exception:
        pass
    return None


def lp_based_optimize(a, n_iters=200, top_k=200, n_extra=100, t_init=0.1, rng=None):
    """
    LP-based block coordinate descent.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    # Ensure positivity for LP boundedness
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t = t_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Solve LP
        b = solve_lp_direction(a, M, top_k=top_k, n_extra=n_extra, rng=rng)
        if b is None:
            break
        
        S_b = float(b.sum())
        if S_b < 1e-12:
            break
        
        # Normalize b
        b = b / S_b
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        
        sc_new = compute_score(a_new)
        
        if sc_new < sc:
            a = a_new
            # Adaptive step: increase if improvement is good
            t = min(0.3, t * 1.05)
        else:
            # Decrease step
            t = max(0.01, t * 0.7)
            a = a_new  # still use the new a (it's a convex combination, should be OK)
            # Actually, let's be more conservative
            if sc_new > sc * 1.01:
                t = max(0.01, t * 0.5)
    
    return best_a


def fienup_iteration(a, n_iters=200, mode='soft', clip_strength=0.5):
    """Fienup-style iteration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * 1.3 * (1.0 + 0.3 * (1.0 - progress))
        above = P > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (clip_strength * 0.5)
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def proximal_subgradient(a, n_iters=5000, top_m=20, rng=None):
    """Proximal subgradient with Barzilai-Borwein step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    
    a_prev = a.copy()
    sc_prev = best_sc
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-M subgradient
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Barzilai-Borwein step
        if it > 0:
            s = a - a_prev
            y = grad - 0  # approximate y as just grad (since we can't get previous grad easily)
            # Use simple adaptive step instead
            gs = float(np.dot(s, s))
            gg = float(np.dot(grad, grad))
            if gg > 1e-30:
                step = gs / (gg + 1e-30)
                step = np.clip(step, 1e-6, 0.05)
            else:
                step = 0.01
        else:
            step = 0.01
        
        a_prev = a.copy()
        sc_prev = sc
        
        # Grad
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

    return best_a
