**Approach: Iterative "peak shaving" with exact computation**

The key observation: the gradient of the objective at a peak lag k* is simply:
g[j] = a[k*-j] (for valid j)

This tells us: to reduce the peak at lag k*, we should decrease a[j] in proportion to a[k*-j].

So the update is: a[j] ← a[j] - η * a[k*-j] (for j where k*-j is valid)

This is a very simple update! And it has an interpretation: we're "shaving" the sequence by its mirror image (at the peak lag).

Let me formalize:
1. Find peak lag k*
2. Compute the "mirror" of a at lag k*: m[j] = a[k*-j] (zero-padded)
3. Update: a ← a - η * m
4. Project to non-negative: a ← max(a, 0)
5. Renormalize: a ← a / sum(a) * S_target

The step size η should be chosen carefully. If η is too large, we create zeros; if too small, convergence is slow.

This is essentially a form of "mirror descent" or "gradient flow" on the peak.

**Enhanced version: Use multiple peaks**

Instead of just the single peak, use the top-K peaks (or all peaks above a threshold):
g[j] = Σ_{k in peaks} w_k * a[k-j]

where w_k are weights (e.g., proportional to c[k]^p for some p).

This gives a smoother gradient that considers multiple "pressure points" simultaneously.

**Even better: Use the exact subgradient of the max**

If the max is achieved at multiple lags, the subgradient is the average of the individual gradients at those lags. This is exactly the multi-peak approach with equal weights.

```python
def subgradient_step(a, eta, top_k=10):
    """One step of subgradient descent on the peak."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    
    # Find top-k peaks
    if top_k >= len(c):
        peaks = np.arange(len(c))
        weights = np.ones(len(c))
    else:
        # Weight by c[k]^p to emphasize the peak
        weights_all = c ** 2  # or c, or c^p
        top_idx = np.argsort(weights_all)[::-1][:top_k]
        weights = np.zeros(len(c))
        weights[top_idx] = weights_all[top_idx]
        weights /= weights.sum()
        peaks = top_idx
    
    # Compute subgradient
    g = np.zeros(n)
    S2 = S * S
    S3 = S2 * S
    M = c.max()
    
    for k in peaks:
        w = weights[k]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            g[js:je+1] += w * a[idx]
    
    # Full gradient of 2n*M/S^2:
    # = 2n * (g/S^2 - 2*M/S^3)
    # But for the step, we can just use g (the subgradient of M) and normalize
    
    a_new = a - eta * g
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-12:
        return a
    a_new = a_new * (S / a_new.sum())
    return a_new
```

```python
Improved step function optimizer using hybrid LSE+flatness gradient,
multi-peak subgradient polish, and diverse structured initializations.

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_flat_score_grad(a, beta, lambda_flat=0.0):
    """
    Score and gradient for the combined LSE + flatness objective.
    F = (2n * LSE(c, beta) + lambda_flat * sum(c^2)) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    
    S2 = S * S
    
    # Gradient of L w.r.t. a:
    # dL/da[j] = sum_k p[k] * a[k-j] (for valid k-j)
    # = sum_{m=0}^{n-1} p[j+m] * a[m]  (letting m = k-j, so k = j+m)
    # This is a sliding dot product of p (extended to length 2n-1) with segments of a
    
    # For the flatness term:
    # d(sum c[k]^2)/da[j] = 2 * sum_{m=0}^{n-1} c[j+m] * a[m]
    
    # Combined: dF/da[j] = (2n*dL/da[j] + lambda_flat * 2*d(sum c^2)/da[j]) / S^2
    #                  - 2 * (2n*L + lambda_flat*sum(c^2)) / S^3
    
    # Compute dL/da[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is correlate(p, a, 'valid') where p has length 2n-1 and a has length n
    # Output has length (2n-1) - n + 1 = n
    
    # For FFT efficiency:
    # correlate(p, a, 'valid')[j] = convolve(p, a[::-1], 'full')[j + n - 1]
    
    a_rev = a[::-1]
    M_conv = len(p) + n - 1  # = 3n - 2
    M_conv_fft = 1 << (M_conv - 1).bit_length()
    P = np.fft.rfft(p, M_conv_fft)
    AR = np.fft.rfft(a_rev, M_conv_fft)
    conv_p = np.fft.irfft(P * AR, M_conv_fft)
    dL_da = conv_p[n-1:n-1+n]
    
    if lambda_flat > 0:
        C = np.fft.rfft(c, M_conv_fft)
        conv_c = np.fft.irfft(C * AR, M_conv_fft)
        dC2_da = 2.0 * conv_c[n-1:n-1+n]
    else:
        dC2_da = 0.0
    
    numerator = 2.0 * n * L + lambda_flat * float(np.sum(c * c))
    grad = (2.0 * n * dL_da + lambda_flat * dC2_da) / S2 - 2.0 * numerator / (S2 * S)
    
    score = numerator / S2
    
    return score, grad
```

```python
"""
Improved step function optimizer using hybrid LSE+flatness gradient,
multi-peak subgradient polish, diverse structured initializations,
and adaptive beta continuation for minimizing 2n*max(self-conv)/(sum)^2.
Key improvements over prior art:
- Combined LSE + quartic flatness gradient for denser optimization signal
- Multi-peak exact subgradient polish phase
- Structured initializations (sinusoidal, power-law, two-block)
- Aggressive high-beta continuation with more iterations
- Adaptive lambda_flat based on peak-to-mean ratio
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_flat_score_grad(a, beta, lam):
    """
    Score and gradient for combined LSE + flatness objective.
    F = (2n * LSE(c,beta) + lam * sum(c^2)) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta

    S2 = S * S

    # dL/da[j] = 2 * sum_m p[j+m] * a[m]
    # d(sum c^2)/da[j] = 4 * sum_m c[j+m] * a[m]
    # Combined: grad[j] = (4n/S2)*sum(p[j+m]*a[m]) + (4*lam/S2)*sum(c[j+m]*a[m]) - (4nL + 2*lam*sum(c^2))/S^3

    # Compute sliding dot products via FFT
    a_rev = a[::-1]
    M_conv = (2 * n - 1) + n - 1  # 3n-2
    M_conv_fft = 1 << (M_conv - 1).bit_length()
    AR = np.fft.rfft(a_rev, M_conv_fft)

    P = np.fft.rfft(p, M_conv_fft)
    conv_p = np.fft.irfft(P * AR, M_conv_fft)
    slide_p = conv_p[n - 1:n - 1 + n]

    if lam != 0.0:
        C = np.fft.rfft(c, M_conv_fft)
        conv_c = np.fft.irfft(C * AR, M_conv_fft)
        slide_c = conv_c[n - 1:n - 1 + n]
        sum_c2 = float(np.dot(c, c))
    else:
        slide_c = 0.0
        sum_c2 = 0.0

    S3 = S2 * S
    grad = (4.0 * n * slide_p + 4.0 * lam * slide_c) / S2 - (4.0 * n * L + 2.0 * lam * sum_c2) / S3

    score = (2.0 * n * L + lam * sum_c2) / S2
    return score, grad


def lse_score_grad_a(a, beta):
    """Standard LSE score and gradient (lam=0 special case, optimized)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta

    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def multi_peak_subgrad_step(a, eta, top_k=5):
    """
    One step of multi-peak subgradient descent.
    Uses top-k peaks of convolution for a denser gradient signal.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a

    # Weight peaks by their values (squared to emphasize top ones)
    c_clipped = np.maximum(c, 0)
    weights_all = c_clipped ** 2
    total_w = weights_all.sum()
    if total_w < 1e-30:
        return a
    weights_all /= total_w

    # Get top-k peak positions
    if top_k >= len(c):
        top_idx = np.arange(len(c))
        top_w = weights_all
    else:
        top_idx = np.argsort(c_clipped)[::-1][:top_k]
        top_w = weights_all[top_idx]
        top_w = top_w / top_w.sum()

    # Compute subgradient: g[j] = sum_k w_k * 2*a[k-j] (for valid k-j)
    g = np.zeros(n)
    for idx, w in zip(top_idx, top_w):
        k = int(idx)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            m = np.arange(js, je + 1)
            idx_a = k - m
            g[js:je + 1] += w * 2.0 * a[idx_a]

    # Update
    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    s_new = a_new.sum()
    if s_new < 1e-12:
        return a
    a_new = a_new * (S / s_new)
    return a_new


def golden_section_1d(a, j, n_evals=20):
    """Golden-section search for optimal value of a[j]."""
    n = len(a)
    a_old = a[j]
    lo = 0.0
    hi = max(a_old * 5.0, a.max() * 0.5, 1e-10)

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent_golden(a, n_sweeps=3, top_k=50, time_budget=60.0, t_start=None):
    """Coordinate descent with golden-section 1D search on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        k_star = int(c.argmax())
        M = float(c[k_star])
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Gradient at peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        grad = np.zeros(n)
        if je >= js:
            m = np.arange(js, je + 1)
            idx_a = k_star - m
            grad[js:je + 1] = 2.0 * a[idx_a]

        # Scale gradient
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)

        # Select top-k by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=15)
            if sc < best_score:
                best_score = sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def run_lbfgs_combined(a0, beta, lam, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B with combined LSE+flatness objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    if lam == 0.0:
        def objective_and_grad(x):
            return lse_score_grad_a(x, beta)
    else:
        def objective_and_grad(x):
            return lse_flat_score_grad(x, beta, lam)

    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    def timed_obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return objective_and_grad(x)

    try:
        res = minimize(
            timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0.copy()

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def adam_combined(a_init, lr=0.003, n_steps=50000, beta_lse=200, lam=0.0,
                  time_budget=None, t_start=None, normalize=True, lr_decay=5e-6):
    """Adam optimizer with combined LSE+flatness gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        lr_t = lr / (1.0 + lr_decay * t)
        if lam == 0.0:
            score, grad = lse_score_grad_a(a, beta_lse)
        else:
            score, grad = lse_flat_score_grad(a, beta_lse, lam)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a * (S0 / s)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def subgrad_polish(a, n_steps=2000, initial_eta=None, time_budget=30.0, t_start=None):
    """
    Multi-peak subgradient descent with geometrically decreasing step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    S = a.sum()

    if initial_eta is None:
        c = fast_conv(a)
        M = c.max()
        mean_c = 1.0 / (2 * n - 1)
        initial_eta = 0.1 * M / (mean_c * n)

    eta = initial_eta
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = multi_peak_subgrad_step(a, eta, top_k=8)

        if step % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Geometric decay
        eta *= 0.998

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def block_swap_perturbation(a, rng, n_blocks=3):
    """Swap random blocks."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        block_len = int(rng.integers(n // 20, n // 5))
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if i != j:
            tmp = a_new[i:i + block_len].copy()
            a_new[i:i + block_len] = a_new[j:j + block_len]
            a_new[j:j + block_len] = tmp
    return a_new


def antipalindromic_perturbation(a, rng, strength=0.1):
    """Add anti-palindromic component."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / max(center, 1.0)
    freq = float(rng.uniform(1, 5))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def gaussian_bump_perturbation(a, rng, n_bumps=5):
    """Add Gaussian bumps."""
    n = len(a)
    a_new = a.copy()
    S0 = a.sum()
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(n // 50, n // 10))
        amplitude = float(rng.uniform(-0.15, 0.15)) * S0 / n
        x = np.arange(n, dtype=np.float64)
        bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new += bump
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def sinusoidal_init(n, rng):
    """Create a sinusoidally-modulated initial sequence."""
    x = np.arange(n, dtype=np.float64)
    freq = float(rng.uniform(1, 4))
    phase = float(rng.uniform(0, 2 * np.pi))
    a = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def powerlaw_init(n, rng):
    """Create a power-law shaped initial sequence."""
    x = np.arange(n, dtype=np.float64)
    alpha = float(rng.uniform(-2, 2))
    center = (n - 1) / 2.0
    a = np.abs(x - center) ** alpha + 0.1
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def two_block_init(n, rng):
    """Two-block: high on one side, low on the other (anti-palindromic)."""
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1)
    switch = float(rng.uniform(0.3, 0.7))
    a = np.where(t < switch, 1.0 + 0.5, 0.5)
    # Add some noise
    a += 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def refine_candidate(a_init, time_budget, t_start):
    """Multi-stage refinement pipeline."""
    a = a_init.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    # Stage 1: Adam (25% of budget)
    if time.monotonic() < deadline - 5:
        adam_time = min(deadline - time.monotonic(), time_budget * 0.25)
        if adam_time > 3:
            lam_adam = 0.5
            sc, a_opt = adam_combined(a, lr=0.003, n_steps=50000, beta_lse=150,
                                      lam=lam_adam, time_budget=adam_time, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()

    # Stage 2: L-BFGS continuation (50% of budget)
    if time.monotonic() < deadline - 5:
        lbfgs_time = min(deadline - time.monotonic(), time_budget * 0.50)
        betas = [100, 400, 1500, 5000, 20000]
        for beta in betas:
            if time.monotonic() > deadline - 3:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            time_alloc = min(remaining * 0.6, lbfgs_time / len(betas) * 1.5)
            # Use flatness for lower betas, pure LSE for higher
            lam = 1.0 if beta < 2000 else 0.3
            sc, a_opt = run_lbfgs_combined(best_a, beta, lam,
                                           maxiter=3000, maxfun=20000,
                                           time_budget=time_alloc, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()

    # Stage 3: Subgradient polish (10% of budget)
    if time.monotonic() < deadline - 3:
        sg_time = min(deadline - time.monotonic() - 2, time_budget * 0.10)
        if sg_time > 5:
            sc, a_opt = subgrad_polish(best_a, n_steps=5000,
                                       time_budget=sg_time, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()

    # Stage 4: Coordinate descent (15% of budget)
    if time.monotonic() < deadline - 3:
        cd_time = min(deadline - time.monotonic() - 2, time_budget * 0.15)
        if cd_time > 5:
            sc, a_opt = coord_descent_golden(best_a, n_sweeps=5, top_k=40,
                                             time_budget=cd_time, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()

    return best_score, best_a


def continuation_lbfgs(a0, time_budget, t_start):
    """Continuation method with aggressive beta schedule and flatness."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    # Beta schedule: start moderate, go very high
    betas = [50, 150, 500, 2000, 8000, 30000, 100000]
    # Lambda schedule: high early (for smoothing), low later (for accuracy)
    lams = [2.0, 1.0, 0.5, 0.2, 0.1, 0.0, 0.0]

    for beta, lam in zip(betas, lams):
        if time.monotonic() > deadline - 3:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        # Allocate more time to higher beta stages
        if beta >= 8000:
            time_alloc = min(remaining * 0.5, 60.0)
            maxiter = 5000
            maxfun = 40000
        elif beta >= 2000:
            time_alloc = min(remaining * 0.35, 40.0)
            maxiter = 3000
            maxfun = 25000
        else:
            time_alloc = min(remaining * 0.25, 25.0)
            maxiter = 2000
            maxfun = 15000

        sc, a_opt = run_lbfgs_combined(a, beta, lam,
                                       maxiter=maxiter, maxfun=maxfun,
                                       time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            elapsed = time.monotonic() - t_start
            print(f"    Continuation beta={beta}, lam={lam}: score={sc:.8f} (t={elapsed:.0f}s)")
        a = a_opt

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function: Hybrid LSE+flatness with diverse initializations,
    aggressive continuation, multi-peak subgradient polish, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Intensive optimization from parent (30% of budget)
    # ==================================================================
    p1_budget = effective_budget * 0.30
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Intensive optimization from parent (budget={p1_budget:.0f}s)")
    print(f"{'=' * 60}")

    # Quick Adam to get to a better local minimum
    quick_time = min(15.0, p1_budget * 0.1)
    sc, a_quick = adam_combined(parent, lr=0.005, n_steps=30000, beta_lse=100,
                                lam=1.0, time_budget=quick_time, t_start=t_start)
    try_update(sc, a_quick, "Quick Adam")

    # Continuation L-BFGS
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 10:
        sc, a_cont = continuation_lbfgs(best_a, remaining, t_start)
        try_update(sc, a_cont, "Continuation LBFGS")

    # ==================================================================
    # Phase 2: Diverse restarts with structured initializations (35% of budget)
    # ==================================================================
    p2_budget = effective_budget * 0.35
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'=' * 60}")

    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget

    # Generate diverse initializations
    n_base = len(best_a)
    inits = []
    labels = []

    # Perturbations of current best
    inits.append(spectral_perturbation(best_a, rng, 0.2))
    labels.append("spectral(0.2)")
    inits.append(spectral_perturbation(best_a, rng, 0.4))
    labels.append("spectral(0.4)")
    inits.append(block_swap_perturbation(best_a, rng, 4))
    labels.append("block_swap(4)")
    inits.append(antipalindromic_perturbation(best_a, rng, 0.15))
    labels.append("anti-palin(0.15)")
    inits.append(gaussian_bump_perturbation(best_a, rng, 7))
    labels.append("gaussian(7)")
    inits.append(spectral_perturbation(best_a, rng, 0.3))
    labels.append("spectral(0.3)")

    # Structured initializations
    inits.append(sinusoidal_init(n_base, rng))
    labels.append("sinusoidal")
    inits.append(powerlaw_init(n_base, rng))
    labels.append("powerlaw")
    inits.append(two_block_init(n_base, rng))
    labels.append("two_block")
    inits.append(np.clip(rng.standard_normal(n_base) * 0.3 + 1.0, 0, None))
    labels.append("random_gauss")

    n_restarts = len(inits)
    per_restart_time = p2_budget / (n_restarts + 1)

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 8:
            break
        t_remain = p2_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.9, per_restart_time * 2.0)
        if time_alloc < 12:
            break

        a_pert = np.clip(inits[ri], 0.0, 1000.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert * (best_a.sum() / s)

        sc, a_opt = refine_candidate(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"restart#{ri}: {labels[ri]}")

    # ==================================================================
    # Phase 3: Intensive optimization of best (20% of budget)
    # ==================================================================
    p3_budget = effective_budget * 0.20
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'=' * 60}")

    if time.monotonic() < t_start + effective_budget - 30:
        # Very high beta L-BFGS
        for beta, lam in [(50000, 0.0), (100000, 0.0), (200000, 0.0)]:
            if time.monotonic() > t_start + effective_budget - 30:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs_combined(best_a, beta, lam,
                                           maxiter=5000, maxfun=50000,
                                           time_budget=30.0, t_start=t_start)
            try_update(sc, a_opt, f"Intensive LBFGS beta={beta}")
            print(f"    beta={beta}: {sc:.8f} ({time.monotonic() - t0:.1f}s)")

        # Subgradient polish
        remaining = t_start + effective_budget - 30 - time.monotonic()
        if remaining > 10:
            sc, a_opt = subgrad_polish(best_a, n_steps=10000,
                                       time_budget=min(remaining, 40.0),
                                       t_start=t_start)
            try_update(sc, a_opt, "Subgrad polish")

        # Coordinate descent
        remaining = t_start + effective_budget - 30 - time.monotonic()
        if remaining > 10:
            sc, a_opt = coord_descent_golden(best_a, n_sweeps=8, top_k=50,
                                             time_budget=min(remaining, 50.0),
                                             t_start=t_start)
            try_update(sc, a_opt, "Phase3 CoordDesc")

    # ==================================================================
    # Phase 4: Try different lengths (10% of budget)
    # ==================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Different lengths (budget={p4_budget:.0f}s)")
    print(f"{'=' * 60}")

    n_best = len(best_a)
    resample_targets = []
    for factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
        tl = int(n_best * factor)
        if 256 <= tl <= 20000:
            resample_targets.append(tl)

    # Also try some fixed lengths
    for fixed_n in [1024, 2048, 4096, 8192]:
        if fixed_n not in resample_targets and 256 <= fixed_n <= 20000:
            resample_targets.append(fixed_n)

    resample_targets = sorted(set(resample_targets))[:5]

    for tl in resample_targets:
        if time.monotonic() > t_start + effective_budget - 20:
            break
        time_left = t_start + effective_budget - 20 - time.monotonic()
        if time_left < 20:
            break

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)

        time_alloc = min(time_left * 0.8, 50.0)
        sc, a_opt = refine_candidate(a_res, time_alloc, t_start)
        try_update(sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 5: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Final polish")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget - 5.0

    # Final very-high-beta L-BFGS
    if time.monotonic() < final_deadline - 10:
        sc, a_opt = run_lbfgs_combined(best_a, 300000, 0.0,
                                       maxiter=10000, maxfun=80000,
                                       time_budget=30.0, t_start=t_start)
        try_update(sc, a_opt, "Final LBFGS 300k")

    # Final subgradient
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = subgrad_polish(best_a, n_steps=5000,
                                   time_budget=min(remaining * 0.4, 20.0),
                                   t_start=t_start)
        try_update(sc, a_opt, "Final subgrad")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = coord_descent_golden(best_a, n_sweeps=10, top_k=60,
                                         time_budget=min(remaining - 2, 30.0),
                                         t_start=t_start)
        try_update(sc, a_opt, "Final CoordDesc")

    # One more L-BFGS pass after coord descent
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = run_lbfgs_combined(best_a, 100000, 0.0,
                                       maxiter=5000, maxfun=40000,
                                       time_budget=min(remaining - 2, 20.0),
                                       t_start=t_start)
        try_update(sc, a_opt, "Post-CD LBFGS")

    # Final coord descent after last LBFGS
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = coord_descent_golden(best_a, n_sweeps=5, top_k=40,
                                         time_budget=min(remaining - 2, 15.0),
                                         t_start=t_start)
        try_update(sc, a_opt, "Very Final CoordDesc")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
