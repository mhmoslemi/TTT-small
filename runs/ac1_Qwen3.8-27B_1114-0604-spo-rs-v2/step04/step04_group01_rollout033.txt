Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses GS (Gerchberg-Saxton) pre-optimization, LSE continuation with L-BFGS-B,
and exact subgradient polishing.

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    a = a * (target / S)
    return a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64)
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Gerchberg-Saxton pre-optimization for flat-spectrum initialization,
LSE continuation with L-BFGS-B for smooth optimization, and exact
subgradient pressure for final polishing. Explores multiple sequence
lengths and diverse starting points.

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    return a * (target / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64)
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    if not np.isfinite(beta) or beta <= 1e-4:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(np.minimum(z, 500.0))
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(x, beta):
    return lse_score_grad(x, beta)


def gs_optimize(a, n_iter=80, relax=1.0):
    """Gerchberg-Saxton style iteration for flat spectrum."""
    a = normalize_a(a)
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    S_target = 1.0

    for it in range(n_iter):
        A = np.fft.rfft(np.zeros(N) * 0.0)
        a_pad = np.zeros(N, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)

        target_mag = np.full(len(mag), S_target / np.sqrt(N), dtype=np.float64)
        target_mag[0] = S_target

        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, N)[:n]

        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            a_new = np.full(n, S_target / n)
        else:
            a_new = a_new * (S_target / S)

        a = (1.0 - relax) * a + relax * a_new

    return normalize_a(a)


def gs_optimize_adaptive(a, n_iter=60, deadline=None):
    """GS with adaptive relaxation."""
    a = normalize_a(a)
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break
        a_pad = np.zeros(N, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        mag = np.abs(A)
        phase = np.ones_like(A, dtype=np.complex128)
        mask = mag > 1e-30
        phase[mask] = A[mask] / mag[mask]
        phase[~mask] = 1.0 + 0.0j

        target_mag = np.full(len(mag), 1.0 / np.sqrt(N), dtype=np.float64)
        target_mag[0] = 1.0

        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new * (1.0 / S)

        relax = min(1.0, 0.3 + 0.7 * it / max(1, n_iter))
        a = (1.0 - relax) * a + relax * a_new
        a = normalize_a(a)

        if it % 10 == 0 or it == n_iter - 1:
            sc = exact_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    rem = float(deadline - time.monotonic())
    if rem < 5.0:
        return exact_score(a), a

    est_eval_time = 0.02 * max(0.5, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    cap = int((0.85 * rem) / max(est_eval_time, 1e-4))
    maxfun = int(min(maxfun, cap))
    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-10,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(beta,), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score(a), a

    if hasattr(res, 'x') and res.x is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 3.0:
        return 64
    est = 0.02 * max(0.5, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    return int(np.clip(mf, 64, 2000))


def subgrad_polish(a, max_iters=30, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        cur_best = None
        cur_best_sc = best_sc

        for k in (1, 3, 5, 10):
            m = d.size
            if k >= m:
                w = np.ones(m) / m
            else:
                idx = np.argpartition(d, m - k)[m - k:]
                w = np.zeros(m)
                w[idx] = 1.0 / k

            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue

            for mult in (0.01, 0.03, 0.1, 0.3, 0.8):
                if deadline and time.monotonic() > deadline:
                    break
                eta = mult / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < cur_best_sc:
                    cur_best_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best
        if cur_best_sc < best_sc:
            best_sc = cur_best_sc
            best_a = cur_best.copy()

    return float(best_sc), best_a

def generate_inits(n, rng, parent):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None:
        pr = resample_sequence(parent, n)
        inits.append(normalize_a(pr))
        inits.append(normalize_a(pr[::-1]))

    inits.append(np.ones(n, dtype=np.float64) / n)

    for alpha in (0.2, 0.4, 0.6, 0.8):
        inits.append(normalize_a(1.0 + alpha * (2.0 * x - 1.0)))
        inits.append(normalize_a(1.0 - alpha * (2.0 * x - 1.0)))

    # Cosine ramps
    inits.append(normalize_a(0.5 + 0.5 * np.cos(np.pi * x)))
    inits.append(normalize_a(0.5 + 0.5 * np.sin(np.pi * x)))

    # Random trig
    for _ in range(min(8, n // 500 + 2)):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.1, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        inits.append(normalize_a(vals))

    # Random smooth (low-freq noise)
    for _ in range(min(4, n // 1000 + 1)):
        Nf = next_power_of_two(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(2.0, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        inits.append(normalize_a(1.0 + 0.5 * noise))

    # Plateau shapes
    for width in (0.1, 0.2, 0.3, 0.5):
        vals = np.ones(n) * 0.3
        center = int(n * float(rng.uniform(0.3, 0.7)))
        half_w = int(n * width / 2)
        lo = max(0, center - half_w)
        hi = min(n, center + half_w)
        vals[lo:hi] = 1.0
        inits.append(normalize_a(vals))

    # Random
    for _ in range(4):
        inits.append(normalize_a(rng.random(n)))

    return inits


def optimize_trajectory(a_start, deadline, rng, label=""):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    t0 = time.monotonic()

    print("  [{}] start={:.8f} n={}".format(label, best_sc, n))

    stages = [
        (50.0, 0.10),
        (100.0, 0.12),
        (200.0, 0.14),
        (400.0, 0.14),
        (800.0, 0.14),
        (2000.0, 0.14),
        (5000.0, 0.10),
        (12000.0, 0.08),
        (30000.0, 0.04),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + cum * max(60.0, deadline - t0 - 30.0)
        sub_dead = min(sub_dead, deadline - 10.0)

        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, n)
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                elapsed = time.monotonic() - t0
                print("  [{}] {:.8f} beta={:.0f} t={:.0f}s".format(label, best_sc, beta, elapsed))
            a = normalize_a(a2)

    return float(best_sc), best_a


def perturb(a, rng, strength=None):
    p = normalize_a(a)
    n = len(p)
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        if S > 1e-12 and d.size > 0:
            dmax = d.max()
            if dmax > 0:
                k = int(rng.integers(1, 6))
                idx = np.argpartition(d, d.size - k)[d.size - k:]
                w = np.zeros(d.size)
                w[idx] = 1.0 / k
                h = corr_pair(p, w)
                hmax = h.max()
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p * np.exp(0.05 * rng.standard_normal(n))
            else:
                q = p * np.exp(0.05 * rng.standard_normal(n))
        else:
            q = p * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 4:
        # Smooth Gaussian perturbation in log space
        logp = np.log(p + 1e-20)
        sig = max(3, n // 50)
        noise = rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-sig, sig + 1) / (sig / 3.0)) ** 2)
        kernel /= kernel.sum()
        noise = np.convolve(noise, kernel, mode='same')
        noise -= noise.mean()
        logp += 0.3 * strength * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Block perturbation
        nb = int(rng.integers(4, 20))
        per = n // nb
        if per < 1:
            per = 1
        multipliers = np.exp(rng.standard_normal(nb) * 0.3)
        q = np.repeat(multipliers, per)
        if len(q) < n:
            q = np.concatenate([q, np.repeat(multipliers, n - len(q))])
        q = p * q[:n]
    else:
        # Sparse perturbation
        q = p.copy()
        n_sparse = int(n * 0.1)
        idx = rng.choice(n, n_sparse, replace=False)
        q[idx] *= np.exp(rng.standard_normal(n_sparse) * strength)

    return normalize_a(q)


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for beta in (300, 1000, 3000, 10000, 30000):
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        mf = estimate_maxfun(rem * 0.85, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} len={}".format(parent is not None, parent.size if parent is not None else 0))

    # Choose lengths to try
    parent_len = parent.size if parent is not None else 0
    lengths = []
    if parent is not None:
        pl = min(parent_len, 16384)
        if pl < 2048:
            pl = 2048
        lengths.append(pl)
    lengths.extend([4096, 8192, 16384])

    # Deduplicate and filter
    seen = set()
    unique_lengths = []
    for n in lengths:
        n = int(n)
        if 1024 <= n <= 32768 and n not in seen:
            unique_lengths.append(n)
            seen.add(n)
    lengths = unique_lengths[:4]
    print("Lengths: {}".format(lengths))

    best_score = np.inf
    best_seq = None

    # Phase 1: Quick screening of multiple (length, init) pairs
    candidates = []  # (score, seq, length)

    for n in lengths:
        if time.monotonic() > deadline - 300.0:
            break

        inits = generate_inits(n, rng, parent)
        evaluated = 0
        for init in inits:
            if time.monotonic() > deadline - 200.0:
                break
            p = normalize_a(init)
            # GS pre-optimization
            p_gs = gs_optimize_adaptive(p, n_iter=30)
            sc = exact_score(p_gs)
            if np.isfinite(sc):
                candidates.append((sc, p_gs, n))
                evaluated += 1
                if evaluated >= 12:
                    break

    if not candidates:
        n0 = 8192
        p0 = np.ones(n0) / n0
        candidates.append((exact_score(p0), p0, n0))

    # Sort by score and pick top diverse candidates
    candidates.sort(key=lambda x: x[0])
    print("Screening done: {} candidates, best={:.8f}".format(len(candidates), candidates[0][0]))

    # Pick up to 3 diverse starting points (different lengths preferred)
    starts = []
    used_lengths = set()
    for sc, seq, n in candidates:
        if len(starts) >= 3:
            break
        if n not in used_lengths:
            starts.append((sc, seq, n))
            used_lengths.add(n)

    # If we don't have 3 different lengths, add perturbations
    while len(starts) < 3 and starts:
        base = starts[0][1]
        pert = perturb(base, rng)
        sc = exact_score(pert)
        starts.append((sc, pert, len(base)))

    print("Selected starts: {[{:.6f}, len={}] for sc, seq, n in starts}".format(
        [sc for sc, _, _ in starts]))

    # Phase 2: Full optimization on each start
    main_deadline = deadline - 180.0
    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.6, 0.4]
    else:
        fracs = [0.5, 0.3, 0.2]

    t_phase2 = time.monotonic()
    for i, (sc0, seq, n) in enumerate(starts):
        rem_total = main_deadline - t_phase2
        if rem_total < 60.0:
            break
        sub_dead = t_phase2 + fracs[i] * max(60.0, deadline - t_start - 200.0)
        sub_dead = min(sub_dead, main_deadline)

        if i > 0 and best_score < 1.6:
            base = best_seq
            if len(base) != n:
                base = resample_sequence(base, n)
            seq = normalize_a(base) if i == 1 else perturb(base, rng)

        sc, seq = optimize_trajectory(seq, sub_dead, rng, "traj{}".format(i + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(i + 1, best_score))

    best_seq = normalize_a(best_seq)
    print("After main optimization: {:.8f} len={}".format(best_score, len(best_seq)))

    # Phase 3: Subgradient polish
    if time.monotonic() < deadline - 120.0:
        sc, seq = subgrad_polish(best_seq, max_iters=25, deadline=deadline - 90.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

        sc, seq = smooth_weighted_subgrad(best_seq, beta_w=50.0, max_iters=20, deadline=deadline - 80.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Smooth subgrad: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts
    n_perturb = 0
    while time.monotonic() < deadline - 100.0 and n_perturb < 4:
        left = float(deadline - 100.0 - time.monotonic())
        if left < 80.0:
            break
        chunk = time.monotonic() + left / float(4 - n_perturb)
        p = perturb(best_seq, rng, strength=float(rng.uniform(0.05, 0.25)))
        sc, seq = quick_optimize(p, chunk)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(n_perturb, best_score))
        n_perturb += 1

    # Phase 5: Final high-beta LSE polish
    if time.monotonic() < deadline - 50.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 15.0:
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 20000.0, deadline, mf)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

            if time.monotonic() < deadline - 25.0:
                rem = float(deadline - time.monotonic() - 5.0)
                if rem > 10.0:
                    mf = estimate_maxfun(rem * 0.85, len(best_seq))
                    sc, seq = run_lbfgs(best_seq, 50000.0, deadline, mf)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print("VHigh beta final: {:.8f}".format(best_score))

    # Phase 6: Final subgradient cleanup
    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 10.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start
    ))
    return out
