"""
Multi-phase optimizer combining exact LP (HiGHS), spectral methods, and L-BFGS.
Improvements: exact LP solver, better adaptive mixing, multi-length search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def normalize(a):
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def build_conv_matrix(a):
    """Build sparse convolution matrix C where C[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C

    """Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0 using HiGHS."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_vec = M * np.ones(2 * n - 1)
    res = linprog(c_obj, A_ub=C, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return np.maximum(res.x, 0)
    return None

def solve_lp_fista(a, M, n_iters=5000):
    """FISTA-based LP solver (fallback for large n)."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    base_L = n * sum_a2
    
    for it in range(n_iters):
        # Adaptive step with backtracking
        step = 1.0 / (base_L * (1 + it / 1000.0))
        
        c = cross_conv(a, x)
        # Smooth max approximation
        beta = 20.0 + 30.0 * min(1.0, it / (n_iters * 0.3))
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        p = exp_z / exp_sum if exp_sum > 1e-300 else np.ones(len(c)) / len(c)
        
        # Gradient of smooth max of (c - M) w.r.t. x
        # c = conv(a, x), grad_x c[k] = a[k-x] (sliding)
        # grad = C^T @ p = a * sliding_sum(p)
        cum = np.concatenate([[0.0], np.cumsum(p)])
        len_p = len(p)
        sw = cum[n:] - cum[:len_p - n + 1]
        grad = a * sw
        
        # FISTA update
        v = u - step * (beta * grad - 1.0)
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x

def lp_alternating(a, max_outer=50, use_highs=True, time_limit=None):
    """LP-based alternating method."""
    n = len(a)
    a = normalize(np.maximum(a, 0.0))
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        
        if use_highs and n <= 4096:
            x = solve_lp_highs(a, M)
            if x is None:
                x = solve_lp_fista(a, M)
        else:
            x = solve_lp_fista(a, M)
        
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Line search for mixing parameter
        best_t = 0.5
        best_sc_t = best_sc
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize(np.maximum((1 - t) * a + t * x, 0.0))
            sc_try = compute_score(a_try)
            if sc_try < best_sc_t:
                best_sc_t = sc_try
                best_t = t
        
        if best_sc_t >= best_sc and best_t == 0.5:
            # No improvement, try smaller steps
            for t in [0.05, 0.1, 0.2]:
                a_try = normalize(np.maximum((1 - t) * a + t * x, 0.0))
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # Stuck
                break
        else:
            a = normalize(np.maximum((1 - best_t) * a + best_t * x, 0.0))
            best_sc = best_sc_t
            best_a = a.copy()
    
    return best_a, best_sc

def lse_score_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def error_reduction(a, n_iters=100, alpha=0.7):
    n = len(a)
    a = normalize(np.maximum(a, 0.0))
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def build_inits(n, rng, parent=None):

    x = np.arange(n, dtype=np.float64) / n

    # Multi-freq
    inits.append(0.4 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x))
    
    return inits

"""
Multi-phase optimizer for flat-autocorrelation non-negative sequences.
Key improvements over previous SOTA:
- Exact LP solver (HiGHS via scipy) for the LP subproblem when n is manageable
- Adaptive mixing with line search in LP alternating
- Multi-length exploration with diverse initializations
- Spectral error reduction for quick global improvement
- High-beta L-BFGS polish for final refinement
- Targeted local search around peak-contributing elements
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def build_conv_matrix(a):
    """Build sparse (2n-1) x n convolution matrix C where C[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def solve_lp_highs(a, M, time_limit_s=30.0):
    """Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0 using HiGHS."""
    n = len(a)
    try:
        C = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_vec = M * np.ones(2 * n - 1)
        res = linprog(c_obj, A_ub=C, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_s})
        if res.success or res.status == 1:
            return np.maximum(res.x, 0.0)
        return None
    except Exception as e:
        print(f"  LP HiGHS error: {e}")
        return None


def solve_lp_fista(a, M, n_iters=5000):
    """FISTA-based LP solver with adaptive smoothing."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    base_L = n * sum_a2

    for it in range(n_iters):
        frac = min(1.0, it / max(1, n_iters * 0.3))
        beta = 10.0 + 40.0 * frac

        step = 1.0 / (base_L * (1.0 + 0.5 * frac))

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        len_p = len(p)
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:max(1, len_p - n + 1)]
        grad = a * sw

        v = u - step * (beta * grad - 1.0)
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_mom  # keep momentum same
        t_mom = t_new

    return x


def lp_alternating(a, max_outer=50, use_highs=True, time_limit=None, fista_iters=3000):

            if sc_try < best_sc_t:

        sc = compute_score(a_new)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                # Try smaller steps
                found = False
                for t in [0.05, 0.1, 0.15]:
                    a_try = normalize((1 - t) * a + t * x)
                    sc_try = compute_score(a_try)
                    if sc_try < best_sc:

                        best_a = a_try.copy()
                        a = a_try

                        found = True
                        break
                if not found:
                    break

        a = a_new

    return best_a, best_sc


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def spectral_flatten(a, n_iters=50, strength=0.5):
    """Flatten the power spectrum iteratively."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2
        target = float(power.mean())
        # Interpolate between current and flat spectrum
        new_power = (1 - strength) * power + strength * target
        new_power = np.maximum(new_power, 1e-20)
        # Keep phase, adjust magnitude
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.02)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)), parent))

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(0.3 + 0.5 * np.sin(np.pi * x))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    inits.append(1.0 - 2.0 * np.abs(x - 0.5))
    inits.append(0.4 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10))

    # Smoothed noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 10))

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: exploration, LP alternating (exact), L-BFGS polish, local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [2048, 4096, 6962, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    print(f"Lengths: {lengths}")

    # ============================================================
    # Phase 1: Quick exploration with spectral methods (12% time)
    # ============================================================
    p1_deadline = t_start + 0.12 * (float(budget_s) - 30)
    print(f"\n{'='*50}")
    print(f"Phase 1: Exploration (deadline at t={p1_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    for n in lengths[:5]:
        if time.monotonic() > p1_deadline:
            break
        inits = build_inits(n, rng, parent)
        for i, a0 in enumerate(inits[:5]):
            if time.monotonic() > p1_deadline:
                break
            # Try both error reduction and spectral flatten
            a_er = error_reduction(a0, n_iters=60, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

            a_sf = spectral_flatten(a0, n_iters=40, strength=0.5)
            sc2 = compute_score(a_sf)
            report(sc2, a_sf, f"SF n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LP Alternating (55% time) - MAIN PHASE
    # ============================================================
    p2_deadline = t_start + 0.67 * (float(budget_s) - 30)
    print(f"\n{'='*50}")
    print(f"Phase 2: LP Alternating (deadline at t={p2_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    # Focus on best length found so far
    n_main = len(best_a)
    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    use_highs = n_main <= 4096
    print(f"  Main length: n={n_main}, HiGHS={use_highs}")

    # Run LP alternating on main length
    max_outer_highs = 40
    max_outer_fista = 60
    fista_it = 4000

    a_out, sc_out = lp_alternating(
        a_work,
        max_outer=max_outer_highs if use_highs else max_outer_fista,
        use_highs=use_highs,
        time_limit=p2_deadline - 120,
        fista_iters=fista_it
    )
    sc = compute_score(a_out)
    report(sc, a_out, f"LP main n={n_main}")

    # Try other lengths
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > p2_deadline - 60:
            break
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S_t = a_test.sum()
        if S_t > 1e-12:
            a_test = a_test / S_t
        use_h = n <= 4096
        print(f"  Trying n={n} (HiGHS={use_h})...")
        a_out2, sc_out2 = lp_alternating(
            a_test,
            max_outer=25 if use_h else 30,
            use_highs=use_h,
            time_limit=p2_deadline - 60,
            fista_iters=3000
        )
        sc2 = compute_score(a_out2)
        report(sc2, a_out2, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS Polish (20% time)
    # ============================================================
    p3_deadline = t_start + 0.87 * (float(budget_s) - 30)
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS Polish (deadline at t={p3_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    a_norm = best_a.copy()
    S = a_norm.sum()
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [30, 100, 300, 800, 1500]
    for beta in beta_schedule:
        if time.monotonic() > p3_deadline:
            break
        print(f"  L-BFGS beta={beta}...")
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=5000, maxfun=150000)
        report(sc, a_opt, f"LBFGS beta={beta}")
        a_norm = a_opt.copy()
        S = a_norm.sum()
        if S > 0:
            a_norm = a_norm / S

    # Subgradient polish
    if time.monotonic() < p3_deadline - 30:
        print("  Subgradient polish...")
        a_sp = subgradient_polish(a_norm, total_iters=2000)
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Targeted Local Search (7% time)
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Local Search (deadline at t={deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    n_best = len(best_a)
    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    n_trials = 10
    for trial in range(n_trials):
        if time.monotonic() > deadline - 20:
            break
        strategy = trial % 5

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.005 + 0.02 * rng.random()
            a_pert = a_work * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = self_conv(a_work)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = a_work.copy()
            contrib = a_work[j_start:j_end + 1][::-1].copy()
            top_count = max(10, n_best // 100)
            idx = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in idx:
                gi = j_start + li
                red = a_pert[gi] * (0.02 + 0.03 * rng.random())
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            bottom_idx = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = total_red / len(bottom_idx) if len(bottom_idx) > 0 else 0.0
            for li in bottom_idx:
                a_pert[j_start + li] += per

        elif strategy == 2:
            # Error reduction restart
            a_pert = error_reduction(
                a_work * (1 + 0.02 * rng.standard_normal(n_best)),
                n_iters=50, alpha=0.8
            )

        elif strategy == 3:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_work * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        else:
            # Spectral flatten restart
            a_pert = spectral_flatten(
                a_work * (1 + 0.01 * rng.standard_normal(n_best)),
                n_iters=30, strength=0.6
            )

        S_p = float(a_pert.sum())
        if S_p > 1e-12:
            a_pert = a_pert / S_p
            sc, a_opt, _ = run_lbfgs(a_pert, 200, maxiter=3000, maxfun=80000)
            report(sc, a_opt, f"local #{trial} s={strategy}")

    # Final LP polish if time permits
    if time.monotonic() < deadline - 30:
        print("  Final LP polish...")
        a_final = best_a.copy()
        S = a_final.sum()
        if S > 0:
            a_final = a_final / S
        use_h = len(a_final) <= 4096
        a_fp, sc_fp = lp_alternating(
            a_final, max_outer=15, use_highs=use_h,
            time_limit=deadline - 20, fista_iters=5000
        )
        sc = compute_score(a_fp)
        report(sc, a_fp, "final LP")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
