    if len(sequence) == 0:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = np.array(sequence, dtype=np.float64)
    sequence = np.clip(sequence, 0.0, 1000.0)
    n = len(sequence)
    A = np.fft.rfft(sequence, 1024)  # Extend to safe FFT size
    convolve_fft = np.fft.irfft(A * A, 1024)[n-1:2*n-1]  # Only compute full convolution
    convolve_fft[convolve_fft < 0] = 0  # Rectify due to symmetry
    S = np.sum(sequence)
    if S < 0.01:
        return np.inf
    M = np.max(convolve_fft)
    score = float(2.0 * n * M / (S * S))
    return score
