def run_lbfgs(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj, a0,

def peak_shaving(a, n_iters=500):
    """Directly shave the peak of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find elements contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1].copy()
        contrib = seg[::-1].copy()
        
        if contrib.max() < 1e-15:
            break
        
        # Reduce top contributors, add to bottom contributors
        top_frac = 0.1
        n_top = max(1, int(top_frac * len(contrib)))
        n_bot = max(1, int(0.5 * len(contrib)))
        
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:n_bot]

        seg[top_idx] -= transfer
        seg[top_idx] = np.maximum(seg[top_idx], 0.0)
        seg[bot_idx] += transfer * n_top / n_bot

        a = np.maximum(a, 0.0)

        a = a / S
        
        sc = compute_score(a)

def multigrid_refine(a_coarse, n_fine):
    """Upscale a coarse solution to fine resolution."""
    n_coarse = len(a_coarse)
    if n_coarse == n_fine:
        return a_coarse.copy()
    x_old = np.linspace(0, 1, n_coarse)
    x_new = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_new, x_old, a_coarse)
    return np.maximum(a_fine, 0.0)

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines adaptive LSE+L-BFGS, frequency-domain peak shaving,
coordinate descent, and diverse multi-start with secondary
length exploration. Key innovations: adaptive beta based on
residual, convolution-referencing perturbation that guarantees
a reduction, and cached convolution in coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fresh_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fresh_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fresh_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fresh_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def objfun(x):
        return lse_and_grad(x, beta)

    res = minimize(
        objfun, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return fresh_score(a_out), a_out


def adaptive_lbfgs(a0, time_limit=300.0):
    """Adaptive multi-beta L-BFGS with residual-based stopping."""
    t0 = time.monotonic()
    a = a0.copy()
    best_sc = fresh_score(a)
    best_a = a.copy()

    betas = [100, 500, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() - t0 > time_limit - 15:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=120000)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_opt.copy()
        else:
            # Diminishing returns, try next beta
            continue

    return best_sc, best_a


def peak_shave(a, n_iters=400):
    """Direct peak shaving of convolution with guaranteed reduction logic."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = fresh_score(a)
    c = fresh_conv(a)

    for it in range(n_iters):
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1].copy()
        contrib = seg[::-1].copy()
        if contrib.max() < 1e-15:
            break
        n_top = max(1, int(0.05 * len(contrib)))
        top_idx = np.argsort(contrib)[::-1][:n_top]
        n_bot = max(1, len(contrib) - n_top)
        bot_idx = np.argsort(contrib)[:n_bot]
        transfer = 0.008 * float(seg[top_idx].mean())
        if transfer < 1e-15:
            break
        seg[top_idx] -= transfer
        seg[top_idx] = np.maximum(seg[top_idx], 0.0)
        seg[bot_idx] += transfer * n_top / n_bot
        a[j_start:j_end + 1] = seg
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        c = fresh_conv(a)
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def freq_waterfill(a, n_iters=250):
    """Frequency-domain water-filling to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = fresh_score(a)
    M_fft = 1 << (2 * n - 2).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        power = np.abs(A) ** 2
        conv = np.fft.irfft(power, M_fft)[:2 * n - 1]
        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-14:
            break
        target = conv_mean + 0.4 * (conv_max - conv_mean)
        err = np.maximum(conv - target, 0.0)
        if err.max() < 1e-14:
            break
        err_pad = np.zeros(M_fft)
        err_pad[:2 * n - 1] = err
        err_freq = np.fft.rfft(err_pad, M_fft)
        power_sum = float(power.sum())
        if power_sum < 1e-15:
            break
        eta = 0.08 * (1.0 - 0.5 * it / n_iters)
        scale = 1.0 - eta * (np.abs(err_freq) / (power_sum + 1e-15))
        scale = np.clip(scale, 0.2, 1.3)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = fresh_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coord_desc_cached(a, n_sweeps=4, n_cand=8):
    """Coordinate descent with cached convolution updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = fresh_score(a)

    for sweep in range(n_sweeps):
        c = fresh_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        n_top = min(50, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        for j in top_local:
            gi = j_start + int(j)
            a_old = a[gi]
            if a_old < 1e-12:
                continue

def _damped_value(a_old, factor):
    """Compute damped candidate value."""
    if factor == 0.0:
        return 0.0
    return max(0.0, a_old * factor)


def conv_ref_perturb(a, rng, scale=0.03):
    """Perturbation that references the convolution structure."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    c = fresh_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end <= j_start:
        a += scale * rng.standard_normal(n)
        return np.clip(a, 0.0, 1000.0)
    # Perturb the contributing segment more
    noise = rng.standard_normal(n) * scale
    noise[j_start:j_end + 1] *= 2.0
    a += noise
    return np.clip(a, 0.0, 1000.0)

            parent = np.clip(p, 0.0, 1000.0)

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    lengths = [6962, 8192]
    if parent is not None:
        pl = len(parent)
        if pl not in lengths:
            lengths.insert(0, pl)
        lengths = [pl, 8192]
    print(f"Lengths: {lengths}, parent: {parent is not None}")

    # ============ PHASE 1: Quick explore (20%) ============
    p1_deadline = t_start + 0.20 * (budget_s - 15)
    print(f"\n=== Phase 1: Explore ===")

    candidates = []
    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        inits = build_inits(n, rng, parent)
        max_inits = max(4, int((p1_deadline - time.monotonic()) / 4.0))
        inits = inits[:min(max_inits, len(inits))]
        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline:
                break
            sc, a_opt = run_lbfgs(a0, 200, maxiter=3000, maxfun=60000)
            report(sc, a_opt, f"exp n={n} i={i}")
            candidates.append((sc, a_opt))

    candidates.sort(key=lambda x: x[0])
    top_cand = candidates[:4]
    print(f"Phase 1 done. Top: {top_cand[0][0]:.8f}")

    # ============ PHASE 2: Intensive LSE (35%) ============
    p2_deadline = t_start + 0.55 * (budget_s - 15)
    print(f"\n=== Phase 2: Intensive LSE ===")

    for ci, (sc, a_c) in enumerate(top_cand):
        if time.monotonic() > p2_deadline:
            break
        rem = p2_deadline - time.monotonic()
        if rem < 60:
            break
        sc2, a2 = adaptive_lbfgs(a_c, time_limit=min(rem * 0.7, 300.0))
        report(sc2, a2, f"LSE c{ci}")

    # Perturbation-based restarts from current best
    for trial in range(4):
        if time.monotonic() > p2_deadline:
            break
        n_b = len(best_a)
        a_pert = conv_ref_perturb(best_a, rng, scale=0.03 + 0.02 * trial)
        sc2, a2 = adaptive_lbfgs(a_pert, time_limit=100.0)
        report(sc2, a2, f"pert{trial}")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ============ PHASE 3: Peak shave + Freq + Coord (25%) ============
    p3_deadline = t_start + 0.80 * (budget_s - 15)
    print(f"\n=== Phase 3: Shave + Freq + Coord ===")

    if time.monotonic() < p3_deadline:
        a_ps = peak_shave(best_a, n_iters=400)
        sc = fresh_score(a_ps)
        report(sc, a_ps, "shave1")

    if time.monotonic() < p3_deadline:
        a_fw = freq_waterfill(best_a, n_iters=250)
        sc = fresh_score(a_fw)
        report(sc, a_fw, "freq_wf")

    if time.monotonic() < p3_deadline:
        a_cd = coord_desc_cached(best_a, n_sweeps=4, n_cand=8)
        sc = fresh_score(a_cd)
        report(sc, a_cd, "coord1")

    # LSE on the best after shaving
    if time.monotonic() < p3_deadline:
        sc2, a2 = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=100000)
        report(sc2, a2, "LSE_5k")

    # Chain: shave -> coord -> LSE
    if time.monotonic() < p3_deadline:
        a_chain = peak_shave(best_a, n_iters=200)
        a_chain = coord_desc_cached(a_chain, n_sweeps=2)
        sc2, a2 = run_lbfgs(a_chain, 3000, maxiter=3000, maxfun=80000)
        report(sc2, a2, "chain")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ============ PHASE 4: Secondary length (10%) ============
    p4_deadline = t_start + 0.90 * (budget_s - 15)
    print(f"\n=== Phase 4: Secondary Length ===")

    n_sec = 12288
    if time.monotonic() < p4_deadline:
        # Interpolate best to secondary length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_sec)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.0)
        sc, a_opt = run_lbfgs(a_interp, 500, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"sec n={n_sec}")

    # ============ PHASE 5: Final polish (remaining) ============
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 20:
        a_cd = coord_desc_cached(best_a, n_sweeps=5, n_cand=8)
        sc = fresh_score(a_cd)
        report(sc, a_cd, "coord_final")

    if time.monotonic() < deadline - 10:
        sc2, a2 = run_lbfgs(best_a, 8000, maxiter=4000, maxfun=100000)
        report(sc2, a2, "LSE_8k")

    if time.monotonic() < deadline - 5:
        a_ps = peak_shave(best_a, n_iters=200)
        sc = fresh_score(a_ps)
        report(sc, a_ps, "shave_final")

    if time.monotonic() < deadline - 3:
        # One final high-beta LSE
        sc2, a2 = run_lbfgs(best_a, 15000, maxiter=3000, maxfun=80000)
        report(sc2, a2, "LSE_15k")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
