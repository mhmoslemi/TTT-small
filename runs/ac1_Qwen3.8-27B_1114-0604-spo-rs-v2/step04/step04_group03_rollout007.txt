    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if it >= len(beta_schedule):
            beta = beta_schedule[-1]
        else:
            beta = beta_schedule[it]
        
        score, grad = lse_score_grad(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2

        S = a.sum()
        if S < 1e-12:
            break
        a = a / S  # normalize

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

    n = len(a)
    M = 2 * n - 1

        w = weights[j]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # For i in [i_start, i_end]: grad_conv[i] += w * a[k-i]

    # d(score)/d(a_i) = 2n * (grad_conv_i / S^2 - 2 * max_conv / S^3)
    grad = (2.0 * n) * (grad_conv / (S * S) - 2.0 * c_max / (S ** 3))
    
    return score, grad

Minimizes 2n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE-Adam with adaptive beta annealing, multi-start, and periodic restarts.
Works on the probability simplex (sum=1, a>=0) exploiting scale invariance.
"""

import numpy as np
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_simplex(a, beta):
    """
    LSE-smoothed score and gradient, assuming a is on simplex (sum=1, a>=0).
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    
    scaled = beta * (c - c_max)
    m = scaled.max()
    exp_s = np.exp(scaled - m)
    p = exp_s / exp_s.sum()
    log_sum = float(np.log(exp_s.sum())) + m
    L = c_max + log_sum / beta
    
    score = 2.0 * n * L  # since S=1, score = 2n*L/S^2 = 2n*L
    
    # Gradient of L w.r.t. a
    # dL/da_i = sum_k p_k * a_{k-i} (for valid k-i in [0, n-1])
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    
    # Gradient of score = 2n * L (since S=1)
    # But we need to account for the constraint sum(a)=1.
    # On the simplex, the effective gradient is:
    # grad_eff_i = d(score)/da_i - mean_j(d(score)/da_j)
    # This projects the gradient onto the tangent space of the simplex.
    # d(score)/da_i = 2n * dL_da_i (since S=1, dS/da_i = 1, and d(1/S^2)/da_i = -2/S^3 = -2)
    # Wait: score = 2n * L * S^{-2}. d(score)/da_i = 2n * (dL/da_i * S^{-2} + L * (-2) * S^{-3} * dS/da_i)
    # = 2n * (dL_da_i / S^2 - 2L / S^3)
    # With S = 1: = 2n * dL_da_i - 4nL
    
    raw_grad = 2.0 * n * dL_da - 4.0 * n * L
    
    # Project onto simplex tangent space (remove mean)
    grad = raw_grad - raw_grad.mean()
    
    return score, grad
