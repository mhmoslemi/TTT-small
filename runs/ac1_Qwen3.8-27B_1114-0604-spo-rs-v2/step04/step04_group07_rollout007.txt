```python
Uses the minimax theorem to alternate between:

- Primal update: QP solved via FISTA with Hankel matrix-vector products

import numpy as np
import time

def fast_conv(a, b=None):
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()

    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
```

```python
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)
```

"""
Alternating Min-Max (AMM) optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Uses the minimax/saddle-point structure: alternates between updating a dual
distribution (softmax on self-convolution) and solving a QP in the primal.
The QP is solved via FISTA with FFT-based Hankel matrix-vector products.
Complemented by peak-shaving and coordinate descent polishing.
"""

import numpy as np
import time

def fast_conv(a, b=None):
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def hankel_mv(w, a):
    """Compute H_w @ a where H_w[j,m] = w_{j+m}. O(n log n)."""
    n = len(a)
    a_rev = a[::-1]
    M = len(w) + n - 1
    M_fft = 1 << (M - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv = np.fft.irfft(W * AR, M_fft)
    return conv[n-1:2*n-1]

def project_simplex(a):
    """Project onto {x: sum(x)=1, x>=0} using sort-based method."""
    n = len(a)
    u = np.sort(a)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cv[rho] - 1) / (rho + 1.0)
    x = np.maximum(a - theta, 0)
    return x

def fista_qp(w, a0, n_iters=200, lr=0.45):
    """Solve min a^T H_w a s.t. sum(a)=1, a>=0 via FISTA."""
    n = len(a0)
    y = a0.copy().astype(np.float64)
    z = a0.copy().astype(np.float64)
    t = 1.0
    for k in range(n_iters):
        grad = 2.0 * hankel_mv(w, z)
        y_new = project_simplex(z - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_new = y_new + ((t - 1.0) / t_new) * (y_new - y)
        y, z, t = z_new, z_new, t_new
    return z

def softmax_weights(c, beta):
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_v = np.exp(scaled)
    s = exp_v.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_v / s

def amm_step(a, beta, fista_iters=200):
    """One AMM iteration: update w, solve QP."""
    c = fast_conv(a)
    w = softmax_weights(c, beta)
    a_new = fista_qp(w, a, n_iters=fista_iters)
    return a_new, compute_score(a_new)

def amm_optimize(a_init, beta_schedule, fista_iters, time_budget, t_start, damping=1.0):
    """Run AMM with a schedule of beta values."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
    a /= a.sum()
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for beta in beta_schedule:
        n_outer = 0
        while time.monotonic() < t_start + time_budget:
            a_new, sc = amm_step(a, beta, fista_iters)
            n_outer += 1
            # Damped update
            if damping < 1.0:
                a = (1.0 - damping) * a + damping * a_new
            else:
                a = a_new
            
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            # Check if we're making progress
            if n_outer % 50 == 0:
                c = fast_conv(a)
                new_max = c.max()
                # If the max hasn't changed much, break to next beta
                pass
    
    return best_score, best_a

def peak_shave(a, n_iters=100, lr=1.0):
    """Direct peak-shaving: reduce elements contributing to the peak."""
    a = a.copy().astype(np.float64)
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]
        
        # Contribution of each element to the peak
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            contrib[js:je + 1] = a[idx]
        
        # Scale down high-contribution elements
        a = a * np.exp(-lr * contrib / max(M, 1e-10))
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

def coord_polish(a, time_budget, t_start):
    """Coordinate descent polishing targeting peak reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_score = compute_score(a)
    best_a = a.copy()
    
    while time.monotonic() < t_start + time_budget:
        c = fast_conv(a)
        k_star = int(c.argmax())
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        improved = False
        # Try reducing peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > t_start + time_budget:
                break
            a_old = a[j]
            if a_old < 1e-12:
                continue
            for frac in [0.8, 0.6, 0.4]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                mask = np.ones(n, dtype=bool)
                mask[js:je+1] = False
                if mask.sum() > 0:
                    a[mask] += removed / mask.sum()
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[j] = a_old
        
        if not improved:
            # Try increasing non-peak elements
            mask = np.ones(n, dtype=bool)
            mask[js:je+1] = False
            nc_idx = np.where(mask)[0]
            if len(nc_idx) > 0:
                j = nc_idx[np.random.randint(len(nc_idx))]
                a_old = a[j]
                improved2 = False
                for mult in [1.3, 1.6, 2.0]:
                    a[j] = a_old * mult
                    sc = compute_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved2 = True
                        improved = True
                        break
                if not improved2:
                    a[j] = a_old
        
        if not improved:
            break
    
    return best_score, best_a
