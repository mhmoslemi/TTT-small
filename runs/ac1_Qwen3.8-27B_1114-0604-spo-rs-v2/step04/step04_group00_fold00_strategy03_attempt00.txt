def linear_score(x):
    """Compute score for a given sequence using direct linear convolution"""
    s = np.sum(x)
    if abs(s) < 1e-12 or s > 1000000:
        return np.inf
        
    # Linear convolution using fast method
    conv = np.convolve(x, x, mode='full')
    c_max = np.max(conv)
    n = len(x)
    return 2.0 * n * c_max / (s**2), True
