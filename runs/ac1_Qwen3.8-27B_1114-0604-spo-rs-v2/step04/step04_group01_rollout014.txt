"""
Search for minimum peak linear self-convolution of a non-negative sequence.
Strategy: LSE-smoothed gradient optimization with continuation in temperature,
interleaved with exact subgradient steps, multi-length exploration (4096/8192/16384),
spectral initialization, and population-based restarts.
Key improvements over previous: longer sequences, exact subgradient with line search,
better beta schedule, spectral flatness initialization.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def subgrad_step(a, k_active=1, eta_candidates=None):
    """Exact subgradient step with line search over top-k active lags."""
    a = normalize_a(a)
    n = len(a)
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return a
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return a
    m = d.size
    if k_active <= 0 or k_active >= m:
        idx = np.array([np.argmax(d)])
    else:
        idx = np.argpartition(d, m - k_active)[m - k_active:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / float(len(idx))
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a
    LDS = float(np.dot(w, d))
    score0 = exact_score_a(a)
    if eta_candidates is None:
        base_eta = 1.0 / (float(n) * hmax * 10.0)
        eta_candidates = [base_eta * f for f in (0.5, 1.0, 2.0, 4.0, 8.0)]
    best_a = a.copy()
    best_sc = score0
    for eta in eta_candidates:
        q = a - eta * (4.0 * float(n) * h / (S * S) - 4.0 * LDS / S)
        q = np.clip(q, 0.0, None)
        Sq = float(q.sum())
        if Sq < 1e-12:
            continue
        q = q / Sq
        sc = exact_score_a(q)
        if sc < best_sc:
            best_sc = sc
            best_a = q.copy()
    return best_a


def subgrad_polish(a, max_iters=30, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in (1, 3, 5, 10):
            q = subgrad_step(a, k_active=k)
            sc = exact_score_a(q)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = q.copy()
                a = q.copy()
                cur_sc = sc
                improved = True
        if not improved:
            break
    return float(best_sc), best_a


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def estimate_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):

    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    if mf < 64:

    if mf > 4000:

    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    rem = float(deadline - time.monotonic())
    if rem < 4.0:
        return exact_score_a(a), a
    est = estimate_eval_time(n)
    cap = int((0.85 * rem) / max(est, 1e-4))
    if cap < 64:
        cap = 64
    maxfun = int(min(int(maxfun), cap))
    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def spectral_init(n, rng):
    """Generate a non-negative sequence with somewhat flat magnitude spectrum."""
    N = next_power_of_two(n)
    # Random phases with slight low-frequency bias
    freqs = np.fft.rfftfreq(N) * N
    # Use 1/sqrt(f+1) magnitude for mild low-freq emphasis
    mag = 1.0 / np.sqrt(np.abs(freqs) + 1.0)
    mag[0] = float(n) ** 0.5  # DC component for sum
    phase = rng.uniform(-np.pi, np.pi, size=len(mag))
    spec = mag * np.exp(1j * phase)
    a = np.fft.irfft(spec, N)[:n]
    a = a - float(a.min()) + 0.01
    return a


def generate_inits(n, rng, parent, cap=20):
    inits = []

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-12:
            arr = np.ones(n) / n
        inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n, dtype=np.float64))

    # Spectral inits
    for _ in range(4):
        add(spectral_init(n, rng))

    # Tapered
    x = np.linspace(0.0, 1.0, n)
    for alpha in (0.3, 0.6, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))

    # Trigonometric
    for r in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.05, 0.30))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Random smooth
    for _ in range(4):
        noise = rng.standard_normal(n)
        k = min(21, max(3, int(n) // 300))
        if k % 2 == 0:
            k += 1
        if k >= 3:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        add(1.0 + 0.5 * noise)

    # Random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.60))
    mode = int(rng.integers(0, 5))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.40))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 0:
                w = np.zeros(d.size)
                idx = np.argmax(d)
                w[idx] = 1.0
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 3:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(21, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        logp += strength * 0.4 * (noise - float(noise.mean()))
        q = np.exp(logp - float(np.max(logp)))
    else:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    return normalize_a(q)


def optimize_trajectory(a_start, deadline, rng, label):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 20.0))
    n = len(a)
    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Beta schedule: more time at moderate beta for equalization
    stages = [
        (50.0, 0.06),
        (100.0, 0.08),
        (200.0, 0.10),
        (350.0, 0.12),
        (600.0, 0.14),
        (1000.0, 0.12),
        (1800.0, 0.12),
        (3500.0, 0.10),
        (7000.0, 0.08),
        (15000.0, 0.08),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 10.0))

        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 8.0:
                break
            mf = estimate_maxfun(rem * 0.8, n)
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, beta, sub_dead, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)
            # Interleave subgradient steps
            if time.monotonic() < sub_dead - 15.0:
                sc_sg, a_sg = subgrad_polish(a, max_iters=5, deadline=sub_dead - 10.0)
                if sc_sg < best_sc:
                    best_sc = sc_sg
                    best_a = a_sg.copy()
                    print("[{}] subgrad best {:.8f} beta={:.0f} t={:.0f}s".format(
                        label, best_sc, beta, time.monotonic() - t0))
                a = normalize_a(a_sg)

    return float(best_sc), best_a


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for beta in (200, 600, 1500, 4000, 10000):
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        mf = estimate_maxfun(rem * 0.85, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pl = int(min(parent.size, 16384))
        if pl < 2048:
            pl = 2048
        lengths.append(pl)
    # Add longer lengths for exploration
    extra_lens = [4096, 8192, 16384]
    for el in extra_lens:
        if el not in lengths:
            lengths.append(el)
    # Sort by length, keep at most 3
    lengths = sorted(set(lengths))
    # Prioritize longer lengths
    lengths = lengths[-3:] if len(lengths) > 3 else lengths
    if 16384 not in lengths:
        lengths.append(16384)
    lengths = lengths[-3:]

    print("Lengths: {}".format(lengths))
    print("Parent: {} (len={})".format(parent is not None, parent.size if parent is not None else 0))

    # Generate initials for all lengths, pick best starts
    all_cands = []
    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=12 if n > 8000 else 16)
        for arr in inits:
            sc = exact_score_a(arr)
            if np.isfinite(sc):
                all_cands.append((sc, arr, n))

    if not all_cands:
        n0 = lengths[0]
        arr0 = np.ones(n0) / n0
        all_cands.append((exact_score_a(arr0), arr0, n0))

    all_cands.sort(key=lambda x: x[0])
    print("Best initial: score={:.6f} n={}".format(all_cands[0][0], all_cands[0][2]))

    # Select diverse starts (different lengths if possible)
    starts = []
    used_lens = set()
    for sc, arr, n in all_cands:
        if n not in used_lens:
            starts.append(arr.copy())
            used_lens.add(n)
            if len(starts) >= 3:
                break
    while len(starts) < 3:
        starts.append(perturb_best(starts[0], rng))

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Selected best start: {:.8f} len={}".format(best_score, len(best_seq)))

    # Main optimization: allocate time to starts, longer sequences get more time
    # Weights based on length preference
    n_main = len(best_seq)
    weights = []
    for i, st in enumerate(starts):
        w = 1.0 + 0.3 * (len(st) / 4096.0)
        weights.append(w)
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    main_end = deadline - 300.0
    cum_alloc = 0.0
    for i, st in enumerate(starts):
        alloc = weights[i] * (main_end - t_start)
        sub_dead = t_start + cum_alloc + alloc
        sub_dead = min(sub_dead, main_end)
        cum_alloc += alloc

        if time.monotonic() > sub_dead - 30.0:
            continue

        # Start from best known if this is not the primary
        if i > 0:
            base = best_seq
            if len(base) != len(st):
                base = resample_sequence(base, len(st))
            base = normalize_a(base)
            st = base

        sc, seq = optimize_trajectory(st, sub_dead, rng, "traj{}".format(i + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f} len={}".format(i + 1, best_score, len(best_seq)))

    best_seq = normalize_a(best_seq)
    print("After main trajectories: {:.8f} len={}".format(best_score, len(best_seq)))

    # Subgradient polish
    if time.monotonic() < deadline - 200.0:
        sc, seq = subgrad_polish(best_seq, max_iters=40, deadline=deadline - 200.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # Perturbation restarts
    perturb_count = 0
    while time.monotonic() < deadline - 150.0 and perturb_count < 6:
        left = float(deadline - 150.0 - time.monotonic())
        if left < 80.0:
            break
        chunk_dead = time.monotonic() + left / float(6 - perturb_count)
        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk_dead)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb {} best: {:.8f}".format(perturb_count, best_score))
        perturb_count += 1

    # Final high-beta LSE
    if time.monotonic() < deadline - 50.0:
        rem = float(deadline - time.monotonic() - 10.0)
        if rem > 20.0:
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            if mf >= 64:
                sc, seq = run_lbfgs(best_seq, 8000.0, deadline, mf)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print("High beta final: {:.8f}".format(best_score))
                if time.monotonic() < deadline - 30.0:
                    rem2 = float(deadline - time.monotonic() - 10.0)
                    if rem2 > 15.0:
                        mf2 = estimate_maxfun(rem2 * 0.85, len(best_seq))
                        if mf2 >= 64:
                            sc2, seq2 = run_lbfgs(best_seq, 20000.0, deadline, mf2)
                            if sc2 < best_score:
                                best_score = sc2
                                best_seq = seq2.copy()
                                print("VHigh beta final: {:.8f}".format(best_score))

    # Final subgradient
    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 15.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: best={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
