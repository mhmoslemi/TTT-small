        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csc_matrix((vals, (rows, cols)), shape=(K, n))

"""
Optimizes non-negative sequence to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-guided direction finding, LSE-smoothed L-BFGS, and spectral methods.
Key innovations: reduced LP for efficient direction finding, adaptive step size
via line search, and simulated annealing for local minimum escape.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via real FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):

    # L = c_max + (1/beta)*log(sum_k exp(beta*(c_k - c_max)))
    # dL/da_i = sum_k p_k * dc_k/da_i
    # where p_k = exp(beta*(c_k-c_max)) / sum_j exp(beta*(c_j-c_max))
    # This is what I computed above as p. Good.

    # Actually: c_k = sum_j a[j]*a[k-j]
    # dc_k/da_i = a[k-i] (from the a[j] term with j=i) + a[i] (from the a[k-j] term with k-j=i, i.e., j=k-i)

    # In general: dc_k/da_i = a[k-i] * 1[0<=k-i<n] + a[i] * 1[0<=k-i<n]

    # dc_k/da_i = d/da_i [sum_j a[j] * a[k-j]]

    # This is the cross-correlation of p and a, evaluated at lag i.
    # cross_corr[i] = sum_m p[i+m] * a[m]
    
    # I can compute this via FFT:
    # cross_corr = ifft(FFT(p) * conj(FFT(a))) ... no, that's not quite right for linear cross-correlation.
    
    # Let me just use the direct formula with a convolution:
    # reverse a: ra[m] = a[n-1-m]
    # cross_corr[i] = sum_m p[i+m] * a[m] = (p * ra)[i + n - 1] ... hmm, let me think.
    
    # Actually, let me define: h[i] = sum_{m=0}^{n-1} p[i+m] * a[m] (for i = 0, ..., n-1, with p having length 2n-1)
    # This is a "sliding dot product" of a with windows of p.
    # I can compute it as: h = correlate(p, a, mode='valid') ... 
    # numpy correlate: correlate(p, a)[k] = sum_m p[m+k] * conj(a[m]) (for real: p[m+k]*a[m])
    # So h[i] = correlate(p, a)[i] for i=0,...,n-1 (if p has length >= 2n-1 and a has length n)
    # Actually correlate(p, a) has length len(p)+len(a)-1 = 3n-2
    # And correlate(p, a)[i] = sum_m p[m+i] * a[len(a)-1-m]
    # Hmm, that's not what I want.

    # Let j = m-i: = sum_{j=0}^{2n-2-i} q[i+j] * a[j] (with a[j]=0 for j>=n)

    # Or equivalently: h = ifft(FFT(q) * FFT(b))[n-1 : 2n-1]
    
    M2 = 3 * n - 1  # length of convolve(p, a_reversed)
    M2_fft = 1 << (M2 - 1).bit_length()
    Q = np.fft.rfft(p, M2_fft)

    h = np.fft.irfft(Q * B, M2_fft)[n-1:2*n-1]
    
    dL_da = 2.0 * h
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def solve_reduced_lp(a, M, K=200):
    """Solve reduced LP to find descent direction.
    max sum(g) s.t. conv(a,g)[k_i] <= M for top-K peaks, g >= 0
    """
    n = len(a)
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix (K x n), sparse
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            v = a[k - j]
            if v > 1e-15:
                rows.append(i)
                cols.append(j)
                vals.append(v)
    
    if len(rows) == 0:
        return np.ones(n) / n
    
    A_mat = csc_matrix((vals, (rows, cols)), shape=(K, n))
    
    # Solve: max 1^T g s.t. A_mat @ g <= M * 1, g >= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(K, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = res.x
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return np.ones(n) / n


def line_search(a, g, n_points=50):
    """Find best step size t for a_new = (1-t)*a + t*g."""
    S_a = a.sum()
    S_g = g.sum()
    if S_g < 1e-12:
        return 0.0
    g_norm = g / S_g  # normalize g to sum=1
    
    best_t = 0.0
    best_sc = compute_score(a)
    
    for t in np.linspace(0.01, 0.99, n_points):
        a_new = (1 - t) * a + t * g_norm
        if a_new.sum() < 1e-12:
            continue
        a_new = np.maximum(a_new, 0.0)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t


def lp_guided_iteration(a, n_iters=500, K=200, use_annealing=True, T0=0.01, T_end=0.001):
    """Main LP-guided optimization loop."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 0.5
        return a
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > 2000:
            break
        
        M = float(fast_conv(a).max())
        sc = 2.0 * n * M  # since sum(a)=1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get direction from LP
        g = solve_reduced_lp(a, M, K=K)
        
        # Line search
        t = line_search(a, g, n_points=30)
        
        if t < 0.005:
            # Try smaller K or different approach
            t = 0.05
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc_new = compute_score(a_new)
        
        # Simulated annealing: accept with probability exp(-Delta/T)
        if sc_new < best_sc:
            a = a_new
        elif use_annealing:
            T = T0 * (T_end / T0) ** (it / max(1, n_iters))
            delta = sc_new - sc
            if np.random.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_new
        
        if it % 50 == 0:
            print(f"    LP iter {it}: sc={sc:.8f}, best={best_sc:.8f}, t={t:.4f}")
    
    return best_a
