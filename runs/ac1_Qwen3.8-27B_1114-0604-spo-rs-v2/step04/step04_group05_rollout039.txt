```
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.where(cond)[0][-1]

        idx = k_star - np.arange(js, je + 1)
        grad[js:je + 1] = 2.0 * a[idx]
```

```python
def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)

        return 1e10, np.zeros(n)
    c_max = float(c.max())

    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]

    current_beta = beta_lse
    if beta_ramp:
        current_beta = max(10, beta_lse * 0.1)

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        if beta_ramp and t % 500 == 0:
            current_beta = min(beta_lse, current_beta * 1.05)

        lr_t = lr / (1.0 + lr_decay * t)
        score, grad = lse_score_grad_a(a, current_beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a * (S0 / s)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in DFT domain: modify amplitudes and phases."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))

    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)

    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def antipalindromic_perturbation(a, rng, strength=0.1):
    """Add anti-palindromic component to reduce center convolution peak."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / center
    freq = float(rng.uniform(1, 5))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    freq2 = float(rng.uniform(5, 20))
    tilt += 0.3 * np.sin(2.0 * np.pi * freq2 * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def chirp_perturbation(a, rng):
    """Add chirp (frequency-modulated sinusoid) component."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64) / n
    f0 = float(rng.uniform(1, 3))
    f1 = float(rng.uniform(5, 15))
    phase = float(rng.uniform(0, 2 * np.pi))
    chirp = np.sin(2.0 * np.pi * (f0 * x + (f1 - f0) * x * x / 2.0 + phase))
    amplitude = float(rng.uniform(0.02, 0.1)) * a.max()
    a_new += amplitude * chirp
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def gaussian_bump_perturbation(a, rng, n_bumps=3):
    """Add/subtract Gaussian bumps."""
    n = len(a)
    a_new = a.copy()
    S0 = a.sum()
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(n // 30, n // 8))
        amplitude = float(rng.uniform(-0.12, 0.12)) * S0 / n
        x = np.arange(n, dtype=np.float64)
        bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new += bump
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def sparse_mask_perturbation(a, rng, sparsity=0.1):
    """Zero out a fraction of elements (explore sparse structure)."""
    n = len(a)
    a_new = a.copy()
    n_zero = int(sparsity * n)
    mask = rng.random(n) > sparsity
    # Keep edges and center
    mask[:5] = True
    mask[-5:] = True
    mid = n // 2
    mask[max(0, mid - 5):mid + 5] = True
    a_new[~mask] = 0
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def golden_section_1d(a, j, n_evals=20):
    """Golden-section search for optimal value of a[j]."""
    a_old = a[j]
    if a_old < 1e-15:
        candidates = [0.0, 0.01 * a.max(), 0.05 * a.max(), 0.1 * a.max(), 0.3 * a.max()]
        best_f = np.inf
        best_v = 0.0
        for cv in candidates:
            a[j] = cv
            f = compute_score(a)
            if f < best_f:
                best_f = f
                best_v = cv
        a[j] = best_v
        return best_f

    lo = 0.0
    hi = a_old * 3.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent_golden(a, n_sweeps=5, top_k=40, time_budget=60.0, t_start=None):
    """Coordinate descent with golden-section 1D search on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            grad[js:je + 1] = a[idx]
        S2 = S * S
        S3 = S2 * S
        grad_score = (2.0 * n) * (grad / S2 - 2.0 * M / S3)

        order = np.argsort(np.abs(grad_score))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=12)
            if sc < best_score:
                best_score = sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, a


def continuation_lbfgs(a0, betas, time_budget, t_start, maxiter_per_stage=2000):
    """Continuation method: run L-BFGS with increasing beta."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        time_alloc = min(remaining * 0.45, 80.0)
        sc, a_out = run_lbfgs_lse(a, beta, maxiter=maxiter_per_stage,
                                   maxfun=15000, time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Continuation beta={beta}: score={sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a = a_out

    return best_score, best_a


def refine_candidate(a_init, time_budget, t_start):
    """
    Multi-stage refinement pipeline:
    Adam -> LSE continuation -> Projected gradient exact peak -> Coord descent
    """
    a = a_init.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False

    # Stage 1: Adam (30% of budget)
    if time.monotonic() < deadline - 5:
        adam_time = min(deadline - time.monotonic(), time_budget * 0.3)
        sc, a_opt = adam_lse(a, lr=0.003, n_steps=80000, beta_lse=300,
                             time_budget=adam_time, t_start=t_start, beta_ramp=True)
        try_update(sc, a_opt, "Adam")
        a = best_a

    # Stage 2: LSE continuation L-BFGS (30% of budget)
    if time.monotonic() < deadline - 5:
        lbfgs_time = min(deadline - time.monotonic(), time_budget * 0.3)
        sc, a_opt = continuation_lbfgs(a, [200, 500, 2000, 10000], lbfgs_time, t_start,
                                        maxiter_per_stage=2000)
        try_update(sc, a_opt, "L-BFGS cont")
        a = best_a

    # Stage 3: Projected gradient exact peak (25% of budget) [NEW]
    if time.monotonic() < deadline - 5:
        pg_time = min(deadline - time.monotonic(), time_budget * 0.25)
        a_norm = best_a.copy()
        S = a_norm.sum()
        if S > 1e-10:
            a_norm = a_norm / S
        peak, a_opt = projected_gradient_peak(a_norm, max_iter=300000,
                                               time_budget=pg_time, t_start=t_start)
        # Convert back to score
        sc = 2.0 * len(a_opt) * peak
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    [PG peak] {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Stage 4: Coordinate descent (15% of budget)
    if time.monotonic() < deadline - 3:
        coord_time = min(deadline - time.monotonic() - 2, time_budget * 0.15)
        sc, a_opt = coord_descent_golden(best_a, n_sweeps=5, top_k=35,
                                         time_budget=coord_time, t_start=t_start)
        try_update(sc, a_opt, "CoordDesc")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function: FISTA + LSE continuation + Exact peak PG + Diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 15.0

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Intensive optimization from parent (30% of budget)
    # ==================================================================
    p1_budget = effective_budget * 0.30
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive optimization from parent (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    # Quick Adam to improve
    quick_time = min(30.0, p1_budget * 0.2)
    sc, a_quick = adam_lse(parent, lr=0.005, n_steps=50000, beta_lse=100,
                           time_budget=quick_time, t_start=t_start, beta_ramp=True)
    try_update(sc, a_quick, "Quick Adam")

    # Continuation L-BFGS
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 15:
        sc, a_cont = continuation_lbfgs(best_a, [100, 300, 1000, 5000, 20000],
                                         remaining, t_start, maxiter_per_stage=2000)
        try_update(sc, a_cont, "Continuation LBFGS")

    # Projected gradient exact peak
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 10:
        a_norm = best_a / best_a.sum()
        peak, a_opt = projected_gradient_peak(a_norm, max_iter=500000,
                                               time_budget=remaining, t_start=t_start)
        sc = 2.0 * len(a_opt) * peak
        try_update(sc, a_opt, "PG Exact Peak")

    # ==================================================================
    # Phase 2: Diverse restarts (35% of budget)
    # ==================================================================
    p2_budget = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_deadline = t_start + (time.monotonic() - t_start) + p2_budget
    p2_deadline = min(p2_deadline, t_start + effective_budget - 30)

    n_restarts = 12
    per_restart_time = p2_budget / (n_restarts + 2)

    strategies = [
        ("spectral_0.1", 0.1),
        ("spectral_0.2", 0.2),
        ("spectral_0.3", 0.3),
        ("anti_pal_0.1", None),
        ("anti_pal_0.2", None),
        ("chirp", None),
        ("gauss", None),
        ("sparse_0.05", None),
        ("sparse_0.1", None),
        ("spectral+anti", None),
        ("spectral+sparse", None),
        ("gauss+chirp", None),
    ]

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break

        t_remain = p2_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.85, per_restart_time * 1.8)
        if time_alloc < 20:
            break

        strat_name = strategies[ri][0]
        strength = strategies[ri][1]

        # Generate perturbation
        if strength is not None:
            a_pert = spectral_perturbation(best_a, rng, strength=strength)
        elif strat_name.startswith("anti_pal"):
            s = float(strat_name.split("_")[-1])
            a_pert = antipalindromic_perturbation(best_a, rng, strength=s)
        elif strat_name == "chirp":
            a_pert = chirp_perturbation(best_a, rng)
        elif strat_name == "gauss":
            a_pert = gaussian_bump_perturbation(best_a, rng, n_bumps=5)
        elif strat_name.startswith("sparse"):
            s = float(strat_name.split("_")[-1])
            a_pert = sparse_mask_perturbation(best_a, rng, sparsity=s)
        elif strat_name == "spectral+anti":
            a_pert = antipalindromic_perturbation(
                spectral_perturbation(best_a, rng, 0.15), rng, 0.12)
        elif strat_name == "spectral+sparse":
            a_pert = sparse_mask_perturbation(
                spectral_perturbation(best_a, rng, 0.1), rng, 0.05)
        elif strat_name == "gauss+chirp":
            a_pert = chirp_perturbation(gaussian_bump_perturbation(best_a, rng, 3), rng)
        else:
            a_pert = spectral_perturbation(best_a, rng, 0.15)

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = refine_candidate(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"restart#{ri}: {strat_name}")

    # ==================================================================
    # Phase 3: Intensive optimization of best (20% of budget)
    # ==================================================================
    p3_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = t_start + effective_budget - 25
    if time.monotonic() < p3_deadline:
        # High-precision L-BFGS
        for beta in [10000, 30000, 100000]:
            if time.monotonic() > p3_deadline - 10:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=5000, maxfun=50000)
            try_update(sc, a_opt, f"Intensive LBFGS beta={beta}")
            print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

        # Projected gradient exact peak on best
        if time.monotonic() < p3_deadline - 5:
            remaining = p3_deadline - time.monotonic()
            a_norm = best_a / best_a.sum()
            peak, a_opt = projected_gradient_peak(a_norm, max_iter=500000,
                                                   time_budget=remaining, t_start=t_start)
            sc = 2.0 * len(a_opt) * peak
            try_update(sc, a_opt, "PG Peak Phase3")

    # ==================================================================
    # Phase 4: Try different lengths (10% of budget)
    # ==================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Different lengths (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    p4_deadline = t_start + effective_budget - 15

    resample_targets = []
    for factor in [0.75, 0.9, 1.1, 1.25, 1.4, 1.6]:
        tl = int(n_best * factor)
        if 1024 <= tl <= 50000:
            resample_targets.append(tl)

    for tl in resample_targets:
        if time.monotonic() > p4_deadline:
            break
        time_left = p4_deadline - time.monotonic()
        if time_left < 25:
            break

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)

        time_alloc = min(time_left * 0.7, 50.0)
        sc, a_opt = refine_candidate(a_res, time_alloc, t_start)
        try_update(sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 5: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # Final L-BFGS
    if time.monotonic() < final_deadline - 15:
        sc, a_opt = run_lbfgs_lse(best_a, 50000, maxiter=10000, maxfun=80000)
        try_update(sc, a_opt, "Final LBFGS")

    # Final projected gradient
    if time.monotonic() < final_deadline - 8:
        a_norm = best_a / best_a.sum()
        remaining = final_deadline - time.monotonic() - 5
        peak, a_opt = projected_gradient_peak(a_norm, max_iter=500000,
                                               time_budget=remaining, t_start=t_start)
        sc = 2.0 * len(a_opt) * peak
        try_update(sc, a_opt, "Final PG")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 8:
        sc, a_opt = coord_descent_golden(best_a, n_sweeps=8, top_k=50,
                                         time_budget=min(remaining - 3, 40.0),
                                         t_start=t_start)
        try_update(sc, a_opt, "Final CoordDesc")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
