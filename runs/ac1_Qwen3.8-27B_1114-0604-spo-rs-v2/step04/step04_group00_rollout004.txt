import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def next_pow2(m):
    """Next power of 2 >= m."""
    return 1 << (int(m) - 1).bit_length()


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = next_pow2(M)
    A = rfft(a, M_fft)
    return irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to unit sum, clip to non-negative."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

    """Standard Fienup with soft clipping."""

    a = normalize(a)

    best_sc = compute_score(a)

        A = rfft(a, M_fft)

        a_new = irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_peak_directed(a, n_iters=400, alpha=0.5, top_frac=0.3):
    """
    Peak-directed Fienup: preferentially reduces frequency bins that
    contribute most to the current time-domain peak location.
    """
    n = len(a)
    a = normalize(a)
    M_fft = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    freqs = np.arange(len(rfft(np.zeros(n), M_fft)), dtype=np.float64)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        A = rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Phase weight: how much each frequency bin contributes to c[k_star]
        phase_weight = np.cos(2.0 * np.pi * freqs * k_star / M_fft)
        # Also consider neighboring peaks
        top_peaks = np.argsort(c)[::-1][:min(5, len(c))]
        for kp in top_peaks:
            phase_weight += 0.3 * np.cos(2.0 * np.pi * freqs * int(kp) / M_fft)
        phase_weight = np.maximum(phase_weight, 0.0)
        phase_weight = phase_weight / (phase_weight.max() + 1e-15)

        # Combined target: high power AND high phase contribution
        target = P * (1.0 + alpha * phase_weight)
        target_mean = float(target.mean())

        thresh = target_mean * (1.2 + 0.4 * (1.0 - progress))
        above = target > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = (thresh / target[above]) ** 0.5
        A_mod = A * scale
        a_new = irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_multi_peak(a, n_iters=300, n_peaks=10, decay=0.98):
    """Fienup targeting multiple peaks simultaneously with decay."""
    n = len(a)
    a = normalize(a)
    M_fft = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    freqs = np.arange(len(rfft(np.zeros(n), M_fft)), dtype=np.float64)

    for it in range(n_iters):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        c_max = float(c[top_idx[0]])

        # Weight peaks by their excess over threshold
        thresh_c = c_max * 0.8
        weights = np.maximum(c[top_idx] - thresh_c, 0.0)
        if weights.max() < 1e-15:
            weights[:] = 1.0
        weights = weights / weights.sum()

        # Combined phase weight from top peaks
        phase_weight = np.zeros(len(freqs))
        for wp, kp in zip(weights, top_idx):
            phase_weight += wp * np.cos(2.0 * np.pi * freqs * int(kp) / M_fft)
        phase_weight = np.maximum(phase_weight, 0.0)
        if phase_weight.max() > 1e-15:
            phase_weight /= phase_weight.max()

        A = rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        target = P * (1.0 + 0.8 * phase_weight)
        thresh = float(np.quantile(target, 0.75))
        above = target > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        strength = 0.4 + 0.4 * (it / n_iters)
        scale[above] = (thresh / target[above]) ** strength
        A_mod = A * scale
        a_new = irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def solve_lp_power_spectrum(phases, n, M_fft):
    """
    Solve LP to find optimal power spectrum P(f) >= 0 given fixed phases,
    minimizing max_k |IDFT(sqrt(P) * phases)|^2 peak.
    
    Actually, we minimize max of the linear convolution given the spectral magnitudes.
    c_k = (1/M) sum_f P(f) * cos(2*pi*f*k/M)  [for real signal]
    
    We solve: minimize t
    s.t. sum_f P(f) * cos(2*pi*f*k/M) <= t * M  for all k
         sum(P) = E (fixed energy)
         P >= 0
    """
    n_freq = len(phases)
    # For real signal, P is symmetric: P[k] = P[M-k]
    # Free variables: P[0], P[1], ..., P[M/2]
    n_vars = n_freq  # rfft gives M/2+1 bins
    
    # Build the constraint matrix: C[k, f] = cos(2*pi*f*k/M)
    # But we only need k = 0, ..., 2n-2 (linear conv length)
    n_conv = 2 * n - 1
    freqs = np.arange(n_freq, dtype=np.float64)
    
    # For efficiency, only use a subset of constraints (the ones likely to be binding)
    # Use all constraints for accuracy
    C = np.cos(2.0 * np.pi * np.outer(np.arange(n_conv), freqs) / M_fft)
    
    # Energy constraint: sum of P = E
    # For rfft, the total energy is P[0] + 2*sum(P[1:-1]) + P[-1] (if M is even)
    # Simplify: just use sum(P) = E
    E = np.sum(phases.real ** 2 + phases.imag ** 2) / M_fft  # normalized
    
    P_var = cp.Variable(n_freq, nonneg=True)
    t = cp.Variable()
    
    constraints = [
        cp.sum(P_var) == E,
        C @ P_var <= t * M_fft
    ]
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=5000, abstol=1e-8)
        if P_var.value is not None:
            return np.maximum(P_var.value, 0.0), float(t.value) if t.value is not None else float('inf')
    except Exception:
        pass
    
    try:
        prob.solve(solver=cp.PDDLP, verbose=False)
        if P_var.value is not None:
            return np.maximum(P_var.value, 0.0), float(t.value) if t.value is not None else float('inf')
    except Exception:
        pass
    
    return None, float('inf')


def fienup_lp_guided(a, n_iters=20, n_fienup_per_lp=50):
    """
    LP-guided Fienup: periodically solve LP for optimal power spectrum shape,
    then use Fienup to approach that shape.
    """
    n = len(a)
    a = normalize(a)
    M_fft = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for lp_iter in range(n_iters):
        # Get current phases
        A = rfft(a, M_fft)
        phases = A / (np.abs(A) + 1e-15)
        
        # Solve LP (only for small enough problems or use approximation)
        if n_freq_samples(M_fft) <= 8192:
            P_opt, lp_val = solve_lp_power_spectrum(phases, n, M_fft)
            if P_opt is not None:
                # Use P_opt as target for Fienup
                a = fienup_toward_target(a, P_opt, M_fft, n_iters=n_fienup_per_lp)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                continue
        
        # Fallback: standard Fienup
        a = fienup_standard(a, n_iters=n_fienup_per_lp)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def n_freq_samples(M_fft):
    return M_fft // 2 + 1


def fienup_toward_target(a, target_P, M_fft, n_iters=100):
    """Fienup that drives |A(f)|^2 toward a specific target spectrum."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_safe = np.maximum(P, 1e-20)
        
        # Ratio of target to current
        ratio = target_P / P_safe
        # Smooth the update
        alpha = 0.5
        scale = np.ones_like(P)
        scale = 1.0 + alpha * (np.sqrt(ratio) - 1.0)
        scale = np.clip(scale, 0.1, 10.0)
        
        A_mod = A * scale
        a_new = irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective and gradient for L-BFGS."""
    n = len(x)
    a = np.maximum(x, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = a / S  # normalize
    
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50.0, 50.0)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient via LSE
    p = exp_scaled / exp_scaled.sum()
    # dL/da_j = (2/M_fft) sum_k p_k * a_{k-j} (circular, but we use linear)
    # More precisely: c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} + a_{k-j} (if j appears twice) ... actually:
    # dc_k/da_j = sum of a_i where i + j = k, i.e., a_{k-j} (if valid)
    # dL/da_j = sum_k p_k * dc_k/da_j = sum_k p_k * (a_{k-j} + a_{j-k} if both valid)
    # For linear conv: dc_k/da_j = a_{k-j} if 0<=k-j<n, plus a_{j-(k-j)}... 
    # Actually for self-conv: c_k = sum_i a_i a_{k-i}
    # dc_k/da_j = a_{k-j} (if 0 <= k-j < n) + a_{k-j} (same term, since it's self-conv)
    # Wait: c_k = sum_i a_i * a_{k-i}. d/dc_k/da_j: j appears as i=j (giving a_{k-j}) and as k-i=j (giving i=k-j, a_{k-j}).
    # So dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n, else a_{k-j} if j==k-j (i.e., 2j=k).
    # For simplicity, use: dc_k/da_j = a_{k-j} + a_{j} * delta_{2j,k} ... this is getting complex.
    # Let's use the FFT approach:
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = next_pow2(M2)
    RA = rfft(rev_a, M2_fft)
    PP = rfft(p, M2_fft)
    conv_res = irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res  # gradient of LSE w.r.t. a (before normalization)
    
    # Chain rule for normalization: a_norm = a / sum(a)
    # d(a_norm)/da_j = (a - a_j * 1) / S^2 ... 
    # Actually: a_norm_i = a_i / S, so da_norm_i/da_j = delta_{ij}/S - a_i/S^2
    # dL/da_j = sum_i (dL/da_norm_i) * (delta_{ij}/S - a_i/S^2)
    #         = g_j/S - (sum_i g_i * a_i)/S^2
    
    grad_norm = g / S - np.dot(g, a) / (S * S)
    
    # But we also need to account for the nonlinearity of max(0, x)
    # For L-BFGS with bounds, this is handled by the bound constraints
    grad = 2.0 * n * grad_norm
    
    return score, grad


def lse_objective_wrapper(x, beta):
    """Top-level wrapper for scipy.minimize."""
    return lse_objective_and_grad(x, beta)


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fista_lse(a0, beta, n_iters=5000, step_init=0.01):
    """FISTA (Nesterov-accelerated proximal gradient) on LSE objective."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t_k = 1.0
    a_prev = a.copy()
    step = step_init
    
    for it in range(n_iters):
        sc, grad = lse_objective_and_grad(a, beta)
        
        if np.isnan(sc) or np.isinf(sc):
            break
        
        # Gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        a_y = a_new + ((t_k - 1.0) / t_new) * (a_new - a_prev)
        a_y = np.maximum(a_y, 0.0)
        S_y = float(a_y.sum())
        if S_y < 1e-12:
            a_y = a_new
        else:
            a_y = a_y / S_y
        
        a_prev = a
        a = a_y
        t_k = t_new
        
        # Backtracking
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        elif it > 100:
            step *= 0.95
        
        # Adaptive step
        gnorm = float(np.linalg.norm(grad))
        if gnorm > 1e-15:
            step = min(step * 1.01, 0.05 / (gnorm + 1e-10))
    
    return best_a


def subgradient_topm(a, n_iters=5000, top_m=10):
    """Subgradient method with top-M peaks and BB step size."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.03
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    prev_grad_norm = 1.0
    prev_step = 0.0
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:top_m]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # BB step size
        if prev_step > 0 and prev_grad_norm > 1e-15:
            bb_step = 0.5 * prev_step * (prev_grad_norm / gn) ** 2
            step = np.clip(bb_step, 0.001, 0.08)
        
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30) * 0.5)
        step = min(step, 0.08)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        prev_grad_norm = gn
        prev_step = step
    
    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=40):
    """Coordinate descent on top-contributing elements."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j
            best_local_sc = cur_sc
            
            for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                cv = a_old_j * f
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
                a[j] = a_old_j
            
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    # Chirp-like
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * x ** 2), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    
    Strategy:
    1. Multi-length peak-directed Fienup exploration
    2. LP-guided spectral shaping
    3. High-beta LSE with L-BFGS (adaptive beta schedule)
    4. FISTA acceleration
    5. Subgradient + coordinate descent polishing
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = [4096, 6144, 8192]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: Peak-directed Fienup (35% of time)
    # ==================================================================
    phase1_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Peak-directed Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Processing n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits[:5]):  # Limit initializations
            if time.monotonic() > phase1_end:
                break
            
            # Peak-directed Fienup
            a_pd = fienup_peak_directed(a0, n_iters=300, alpha=0.6, top_frac=0.3)
            sc = compute_score(a_pd)
            update_best(sc, a_pd, f"PD-FIENUP n={n} init={i}")

            # Multi-peak Fienup
            if time.monotonic() < phase1_end:
                a_mp = fienup_multi_peak(a0, n_iters=250, n_peaks=10)
                sc2 = compute_score(a_mp)
                update_best(sc2, a_mp, f"MP-FIENUP n={n} init={i}")

            # Standard Fienup as baseline
            if time.monotonic() < phase1_end:
                a_std = fienup_standard(a0, n_iters=200)
                sc3 = compute_score(a_std)
                update_best(sc3, a_std, f"STD-FIENUP n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP-guided spectral shaping (15% of time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LP-guided (deadline t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    # Only try LP if sequence is small enough
    M_fft_best = next_pow2(2 * n_best - 1)
    if n_freq_samples(M_fft_best) <= 8192 and time.monotonic() < phase2_end:
        print(f"  Running LP-guided Fienup for n={n_best}")
        a_lp = fienup_lp_guided(best_a, n_iters=5, n_fienup_per_lp=50)
        sc = compute_score(a_lp)
        update_best(sc, a_lp, "LP-GUIDED")
    else:
        print(f"  Skipping LP (n_freq={n_freq_samples(M_fft_best)} too large)")
        # Instead, do refined peak-directed on best
        a_pd2 = fienup_peak_directed(best_a, n_iters=300, alpha=0.8)
        sc = compute_score(a_pd2)
        update_best(sc, a_pd2, "PD-FIENUP-2")

    # Try on a different length
    if time.monotonic() < phase2_end:
        n_alt = 8192 if len(best_a) != 8192 else 6144
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_alt)
        a_alt = np.interp(x_new, x_old, best_a)
        a_alt = np.maximum(a_alt, 0.01)
        a_mp_alt = fienup_multi_peak(a_alt, n_iters=200, n_peaks=8)
        sc = compute_score(a_mp_alt)
        update_best(sc, a_mp_alt, f"MP-FIENUP alt n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: High-beta LSE with L-BFGS (30% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: LSE L-BFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    
    # Progressive beta schedule
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    
    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        
        # Adjust iterations based on remaining time
        remaining = phase3_end - time.monotonic()
        maxfun = min(80000, int(remaining * 800))
        maxiter = min(5000, maxfun // 20)
        
        if maxfun < 1000:
            break
            
        print(f"  L-BFGS beta={beta}, maxfun={maxfun}, maxiter={maxiter}")
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        updated = update_best(sc, a_opt, f"LBFGS beta={beta}")
        
        # If we found improvement, also try the result at next beta
        if updated:
            best_a = a_opt.copy()

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: FISTA + Subgradient + Coord Descent (15% of time)
    # ==================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: FISTA + Subgradient + CoordDescent (deadline t={phase4_end - t_start:.0f}s) ===")

    # FISTA with moderate beta
    if time.monotonic() < phase4_end - 60:
        print("  Running FISTA...")
        a_fista = fista_lse(best_a, beta=500, n_iters=3000, step_init=0.02)
        sc = compute_score(a_fista)
        update_best(sc, a_fista, "FISTA")

    # Top-M subgradient
    for top_m in [5, 10, 20]:
        if time.monotonic() > phase4_end - 30:
            break
        a_sg = subgradient_topm(best_a, n_iters=3000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    # Coordinate descent
    if time.monotonic() < phase4_end - 30:
        a_cd = coordinate_descent(best_a, n_sweeps=3, n_top=60)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD-DESCENT")

    # Final L-BFGS polish with very high beta
    if time.monotonic() < phase4_end - 20:
        print("  Final L-BFGS polish...")
        sc, a_opt = run_lbfgs_lse(best_a, beta=10000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
