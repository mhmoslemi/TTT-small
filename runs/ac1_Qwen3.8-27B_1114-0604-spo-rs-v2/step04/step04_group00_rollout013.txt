```python
"""
Continuation L-BFGS + Backtracking Subgradient + Multi-start optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous:
1. Continuation method for LSE smoothing (gradually increase beta)
2. Backtracking line search for subgradient (more robust than fixed decay)
3. Active-set focusing (only optimize against top-K peaks)
4. Random perturbation + re-optimization for escaping local minima
5. Better time allocation: more effort on local refinement of best solution
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    score = 2n * LSE(beta*(c - c_max)) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient of L wrt a_i = sum_k p_k * dc_k/da_i = 2 * sum_k p_k * a_{k-i}
    # Compute cross-correlation R[i] = sum_m a[m] * p[m+i]
    N = 1
    while N < 3 * n - 2:
        N *= 2
    A_fft = np.fft.rfft(a, N)
    P_fft = np.fft.rfft(p, N)
    R = np.fft.irfft(np.conj(A_fft) * P_fft, N)
    dL_da = 2.0 * R[:n]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x, beta):
    """Top-level wrapper for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    # Use bounds but keep them loose to avoid slow convergence
    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def continuation_lbfgs(a0, beta_start=5.0, beta_end=10000.0, n_stages=None,
                       maxiter_per_stage=3000, maxfun_per_stage=60000,
                       time_budget=None, t_start=None):
    """
    Continuation method: gradually increase beta from beta_start to beta_end.
    At each stage, run L-BFGS to (near) convergence, then increase beta.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, None).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    if n_stages is None:
        n_stages = max(2, int(np.ceil(np.log2(beta_end / beta_start))))

    best_a = a.copy()
    best_sc = compute_score(a)

    for stage in range(n_stages):
        if time_budget is not None and t_start is not None:
            if time.monotonic() > t_start + time_budget:
                break

        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        # Adjust iterations: fewer for small beta (easier), more for large beta
        it_scale = 0.5 + 0.5 * frac
        mi = int(maxiter_per_stage * it_scale)
        mf = int(maxfun_per_stage * it_scale)

        sc, a_opt = run_lbfgs(a, beta, maxiter=mi, maxfun=mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    [cont-lbfgs] stage={stage}, beta={beta:.1f}, score={sc:.8f}")
        a = a_opt

        # Verify with exact score
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()

    return best_sc, best_a


def backtracking_subgradient(a, max_iter=5000, top_m=10, alpha_init=0.02,
                             alpha_max=0.05, contraction=0.5, expansion=1.05,
                             time_budget=None, t_start=None):
    """
    Subgradient method with backtracking line search.
    Uses top-M peaks for a richer subgradient signal.
    """
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, None).copy()
    S = a.sum()
    if S < 1e-12:
        return 1.0, np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iter):
        if time_budget is not None and t_start is not None:
            if time.monotonic() > t_start + time_budget:
                break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # Reset alpha on improvement
            alpha = min(alpha * expansion, alpha_max)

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # dc_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (approx)
                # More precisely: dc_k/da_j = a_{k-j} for j in valid range
                # (counting both the j and k-j terms)
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Backtracking line search
        a_current = a.copy()
        sc_current = sc
        found = False
        for _ in range(30):
            a_trial = a_current - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                alpha *= contraction
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            # Accept if improvement (simplified Armijo)
            if sc_trial < sc_current - 1e-14:
                a = a_trial
                found = True
                break
            alpha *= contraction

        if not found:
            # Too small step, break
            if alpha < 1e-10:
                break
            # Try a smaller top_m or just break
            break

        alpha = min(alpha * expansion, alpha_max)

    return best_sc, best_a

                break

        # Identify active peaks
        c = fast_conv(a)
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]

        # L-BFGS with very high beta (focuses on the max)
        beta = 5000 + 2000 * rnd
        sc, a_opt = run_lbfgs(a, beta, maxiter=2000, maxfun=40000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a = a_opt

        # Subgradient step on top peaks
        sc2, a_sg = backtracking_subgradient(a, max_iter=500, top_m=top_m,
                                             alpha_init=0.01, alpha_max=0.03,
                                             contraction=0.5, expansion=1.03,
                                             time_budget=30, t_start=time.monotonic())
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a_sg.copy()
            a = a_sg

        # Check stopping
        c_check = fast_conv(a)
        M_check = float(c_check.max())
        avg_check = float(c_check.sum()) / len(c_check)
        ratio = M_check / avg_check
        if ratio < 1.001:
            break

    return best_sc, best_a


def fienup_quick(a, n_iters=100, threshold_mult=1.5):
    """Quick Fienup for initialization/diversity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * threshold_mult
        above = P > thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = np.sqrt(np.maximum(thresh / P[above], 0.01))
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def random_perturb_reoptimize(a, n_tries=3, perturb_scale=0.05,
                              time_budget=None, t_start=None):
    """
    Perturb the solution randomly and re-optimize.
    Helps escape local minima.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    for trial in range(n_tries):
        if time_budget is not None and t_start is not None:
            if time.monotonic() > t_start + time_budget:
                break

        # Different perturbation strategies
        strategy = trial % 3
        if strategy == 0:
            # Gaussian perturbation
            a_pert = a * (1.0 + perturb_scale * rng.standard_normal(n))
        elif strategy == 1:
            # Spike at random positions
            a_pert = a.copy()
            n_spikes = max(1, n // 100)
            spike_idx = rng.choice(n, size=n_spikes, replace=False)
            for idx in spike_idx:
                a_pert[idx] *= rng.uniform(0.3, 3.0)
        else:
            # Smooth perturbation (low-frequency)
            freq = rng.integers(1, 5)
            x = np.arange(n) / n
            pert = 1.0 + perturb_scale * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
            a_pert = a * pert

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_pert, beta=500, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      [perturb-reopt] trial={trial}, score={sc:.8f}")

    return best_sc, best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # "Bump" initialization - wider than uniform
    inits.append(np.maximum(1.0 - 0.8 * np.abs(2.0 * x - 1.0) ** 3, 0.01))
    # Multi-frequency
    inits.append(np.maximum(0.5 + 0.2 * np.cos(2*np.pi*x) + 0.15 * np.cos(4*np.pi*x) + 0.1 * np.cos(6*np.pi*x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0  # 30s safety margin

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    n_parent = len(parent) if parent is not None else 4096
    print(f"Parent length: {n_parent}")
    print(f"Time budget: {total_budget}s, deadline at t={deadline - t_start:.0f}s")

    # ================================================================
    # Phase 1: Continuation L-BFGS from parent (40% of time)
    # ================================================================
    phase1_time = 0.40 * (total_budget - 30.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: Continuation L-BFGS from parent ({phase1_time:.0f}s)")
    print(f"{'='*60}")

    a_init = parent.copy() if parent is not None else np.ones(4096) * 0.5
    a_init = np.maximum(a_init, 0.0)
    a_init = a_init / a_init.sum()

    sc, a_opt = continuation_lbfgs(
        a_init,
        beta_start=3.0,
        beta_end=20000.0,
        n_stages=12,
        maxiter_per_stage=4000,
        maxfun_per_stage=80000,
        time_budget=phase1_time * 0.8,
        t_start=t_start
    )
    update_best(sc, a_opt, "CONT-LBFGS-parent")

    # Additional continuation with different parameters
    if time.monotonic() < t_start + phase1_time:
        sc2, a_opt2 = continuation_lbfgs(
            a_opt,
            beta_start=500.0,
            beta_end=100000.0,
            n_stages=6,
            maxiter_per_stage=3000,
            maxfun_per_stage=60000,
            time_budget=min(60.0, (t_start + phase1_time) - time.monotonic()),
            t_start=t_start
        )
        update_best(sc2, a_opt2, "CONT-LBFGS-high")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Active-set refinement (15% of time)
    # ================================================================
    phase2_time = 0.15 * (total_budget - 30.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Active-set refinement ({phase2_time:.0f}s)")
    print(f"{'='*60}")

    sc, a_ref = active_set_refinement(
        best_a, n_rounds=15, top_m=5, max_iter=2000,
        time_budget=phase2_time, t_start=time.monotonic()
    )
    update_best(sc, a_ref, "ACTIVE-SET")

    # Try with more peaks
    if time.monotonic() < t_start + phase1_time + phase2_time:
        sc2, a_ref2 = active_set_refinement(
            best_a, n_rounds=10, top_m=15, max_iter=1000,
            time_budget=min(60.0, (t_start + phase1_time + phase2_time) - time.monotonic()),
            t_start=time.monotonic()
        )
        update_best(sc2, a_ref2, "ACTIVE-SET-15")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Multi-start exploration (20% of time)
    # ================================================================
    phase3_time = 0.20 * (total_budget - 30.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-start ({phase3_time:.0f}s)")
    print(f"{'='*60}")

    n_try = n_parent if parent is not None else 4096
    # Also try a slightly different length
    lengths_to_try = [n_try]
    if n_try > 2048:
        lengths_to_try.append(n_try + 512)
    if n_try < 12000:
        lengths_to_try.append(min(n_try + 1024, 12000))

    p3_deadline = t_start + phase1_time + phase2_time + phase3_time
    for n in lengths_to_try:
        if time.monotonic() > p3_deadline:
            break
        inits = build_initializations(n, rng, parent)
        # Limit number of inits based on remaining time
        n_inits = min(len(inits), 4)
        for i in range(n_inits):
            if time.monotonic() > p3_deadline:
                break
            a0 = inits[i].copy()
            a0 = np.maximum(a0, 0.0)
            S0 = a0.sum()
            if S0 < 1e-12:
                continue
            a0 = a0 / S0

            # Quick Fienup + L-BFGS
            a_f = fienup_quick(a0, n_iters=80)
            sc_f = compute_score(a_f)
            update_best(sc_f, a_f, f"FIENUP n={n} init={i}")

            if time.monotonic() < p3_deadline:
                sc_c, a_c = continuation_lbfgs(
                    a_f, beta_start=10.0, beta_end=5000.0,
                    n_stages=5, maxiter_per_stage=2000, maxfun_per_stage=40000,
                    time_budget=45.0, t_start=time.monotonic()
                )
                update_best(sc_c, a_c, f"CONT n={n} init={i}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Random perturbation + re-optimization (10% of time)
    # ================================================================
    phase4_time = 0.10 * (total_budget - 30.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Perturb + Re-optimize ({phase4_time:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = t_start + phase1_time + phase2_time + phase3_time + phase4_time
    sc, a_pert = random_perturb_reoptimize(
        best_a, n_tries=5, perturb_scale=0.08,
        time_budget=phase4_time * 0.6, t_start=time.monotonic()
    )
    update_best(sc, a_pert, "PERTURB")

    # Stronger perturbation
    if time.monotonic() < p4_deadline:
        sc2, a_pert2 = random_perturb_reoptimize(
            best_a, n_tries=3, perturb_scale=0.15,
            time_budget=min(60.0, p4_deadline - time.monotonic()),
            t_start=time.monotonic()
        )
        update_best(sc2, a_pert2, "PERTURB-strong")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ================================================================
    # Phase 5: Final intensive refinement (10% of time)
    # ================================================================
    phase5_time = 0.10 * (total_budget - 30.0)
    print(f"\n{'='*60}")
    print(f"Phase 5: Final refinement ({phase5_time:.0f}s)")
    print(f"{'='*60}")

    # Very high beta L-BFGS
    sc, a_fin = run_lbfgs(best_a, beta=50000, maxiter=10000, maxfun=200000)
    update_best(sc, a_fin, "FINAL-LBFGS-high")

    # Backtracking subgradient with many iterations
    if time.monotonic() < deadline - 30:
        sc2, a_bg = backtracking_subgradient(
            best_a, max_iter=5000, top_m=5,
            alpha_init=0.005, alpha_max=0.02,
            contraction=0.5, expansion=1.02,
            time_budget=60.0, t_start=time.monotonic()
        )
        update_best(sc2, a_bg, "FINAL-SG")

    # Another round of continuation from the best
    if time.monotonic() < deadline - 20:
        sc3, a_cont = continuation_lbfgs(
            best_a, beta_start=2000.0, beta_end=200000.0,
            n_stages=5, maxiter_per_stage=3000, maxfun_per_stage=60000,
            time_budget=min(60.0, deadline - 20 - time.monotonic()),
            t_start=time.monotonic()
        )
        update_best(sc3, a_cont, "FINAL-CONT")

    # Final micro-optimization: L-BFGS with maximum beta
    if time.monotonic() < deadline - 10:
        sc4, a_final = run_lbfgs(best_a, beta=100000, maxiter=5000, maxfun=100000)
        update_best(sc4, a_final, "FINAL-MAXBETA")

    print(f"\n{'='*60}")
    print(f"FINAL: best_score = {best_score:.8f}")
    print(f"Length: {len(best_a)}, Time: {time.monotonic()-t_start:.1f}s / {total_budget}s")
    print(f"{'='*60}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"Verified score: {verified:.8f}")
    return out
```
