- For component j (0 <= j < n): `sum_k y_k * 2*a_{k-j}^* + z <= 0`
- For component n (t): `-sum_k y_k + 0 <= 1` → `sum_k y_k >= -1` (always true since y >= 0)

So the dual is:
```
max sum_k y_k * c_k(a^*) + z
s.t. sum_k y_k * 2*a_{k-j}^* + z <= 0 for j = 0,...,n-1
     y_k >= 0
     z free
```

This is a linear program in |P| + 1 variables with n constraints. For |P| = 10, that's 11 variables and 7000 constraints. This is much smaller than the primal!

We can solve this dual LP easily, and then the dual variables give us the "weights" that the KKT conditions require.

But more importantly, the **complementary slackness** tells us which constraints are active at the optimum, which helps us build the next iteration.

However, for our purpose, solving the primal LP directly is fine (it's not that large).

Let me now think about potential issues:
1. **LP infeasibility**: If the linearized constraints are too tight, the LP might be infeasible. This can happen if the active set is too large or the linearization is poor. Solution: reduce the active set or relax the constraints.

2. **LP unboundedness**: Since `t` can go to 0 (or negative if we allow it), and `a >= 0`, the objective `min t` could be unbounded below if the constraints don't prevent it. But since `sum(a) = 1` and `a >= 0`, and the convolution is non-negative (all terms are products of non-negative numbers), `t` should be bounded below by 0. Actually, the linearized constraint `sum_j 2*a_{k-j}^* * a_j - t <= c_k(a^*)` with `a_{k-j}^* >= 0` and `sum(a) = 1` gives a lower bound on t.

   Specifically, if we set `a = e_j` (unit vector), then `sum_j 2*a_{k-j}^* * a_j = 2*a_{k-j}^*`, so the constraint gives `2*a_{k-j}^* - t <= c_k(a^*)`, i.e., `t >= 2*a_{k-j}^* - c_k(a^*)`. ```python
"""
Cutting-plane LP optimizer for step function peak minimization.
Uses sequential linearization of the max-convolution objective,
solving an LP at each iteration to find the next iterate.
Combined with LSE-smoothed L-BFGS for initialization and final polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def func(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def find_active_peaks(c, frac=0.001, max_count=50):
    """Find the active peaks: those within frac of the max."""
    c_max = float(c.max())
    threshold = c_max * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) > max_count:
        # Take the top max_count
        top_idx = np.argsort(c[active])[::-1][:max_count]
        active = active[top_idx]
    return active, c_max


def solve_lp_cutting_plane(a, c, active_peaks, c_max):
    """
    Solve the LP:
    minimize t
    subject to: sum_j 2*a_{k-j} * a_j - t <= c_k for k in active_peaks
                sum_j a_j = 1
                a_j >= 0, t >= 0
    """
    n = len(a)
    K = len(active_peaks)
    
    # Build A_ub, b_ub
    # Row k: [2*a_{k-j} for j=0..n-1, -1] <= c_k
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i, k in enumerate(active_peaks):
        # a_{k-j} for j=0..n-1: non-zero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            A_ub[i, j_lo:j_hi+1] = 2.0 * a[k - np.arange(j_lo, j_hi + 1)]
        A_ub[i, n] = -1.0
        b_ub[i] = c[k]
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: a_j in [0, 1000], t in [0, 1000]
    bounds = [(0.0, 1000.0)] * n + [(0.0, 1000.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        a_new = res.x[:n]
        t_new = res.x[n]
        return a_new, t_new
    else:
        return None, None


def cutting_plane_loop(a, deadline, max_iters=5000, verbose_interval=100):
    """
    Iterative cutting-plane refinement.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        # Note: S hasn't changed if we normalize after each step
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 200:
            # Try to escape by perturbing the active set
            stagnant = 0
            # Slightly perturb a
            rng = np.random.default_rng(it)
            a_pert = a * (1.0 + 0.01 * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, None)
            S_pert = a_pert.sum()
            if S_pert > 1e-12:
                a = a_pert / S_pert
                c = fast_conv(a)
            continue
        
        # Find active peaks
        active_peaks, c_max_val = find_active_peaks(c, frac=0.001, max_count=30)
        
        if len(active_peaks) < 2:
            # Only one peak is active; use subgradient descent instead
            break
        
        # Solve LP
        t_deadline_lp = time.monotonic() + 2.0  # 2 second timeout for LP
        a_new, t_new = solve_lp_cutting_plane(a, c, active_peaks, c_max_val)
        
        if a_new is None:
            # LP failed; try with fewer active peaks
            active_peaks_small = active_peaks[:5]
            a_new, t_new = solve_lp_cutting_plane(a, c, active_peaks_small, c_max_val)
        
        if a_new is not None:
            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new / S_new
            
            # Check score
            c_new = fast_conv(a_new)
            sc_new = 2.0 * n * float(np.max(c_new)) / (S_new * S_new)
            
            if sc_new < sc - 1e-15:
                # Accept
                a = a_new
                c = c_new
                S = S_new
            else:
                # Try a fractional step
                accepted = False
                for alpha in [0.5, 0.25, 0.1, 0.05]:
                    a_try = (1.0 - alpha) * a + alpha * a_new
                    a_try = np.clip(a_try, 0.0, 1000.0)
                    S_try = a_try.sum()
                    if S_try < 1e-12:
                        continue
                    a_try = a_try / S_try
                    c_try = fast_conv(a_try)
                    sc_try = 2.0 * n * float(np.max(c_try)) / (S_try * S_try)
                    if sc_try < sc - 1e-15:
                        a = a_try
                        c = c_try
                        S = S_try
                        accepted = True
                        break
                if not accepted:
                    stagnant += 1
        else:
            stagnant += 1
        
        if it % verbose_interval == 0:
            print(f"  CP iter {it}: score={sc:.10f}, best={best_sc:.10f}, "
                  f"n_active={len(active_peaks)}, t={time.monotonic():.0f}s")
    
    return best_a
