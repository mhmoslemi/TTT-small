def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    a0 = a0 / a0.sum()

    res = minimize(
        score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                 'iprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_flatten(a, n_iters=200, top_frac=0.3):
    """
    Fienup-style iteration to flatten the power spectrum.
    Reduces high-frequency power to make the convolution flatter.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / float(n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        P_mean = float(P.mean())
        
        # Target: reduce power above threshold
        # Use quantile-based threshold
        thresh_quantile = 1.0 - top_frac * (1.0 - 0.5 * progress)
        thresh = np.quantile(P, thresh_quantile)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            # Soft reduction
            reduction = np.sqrt(thresh / P[above])
            # Blend: don't fully reduce, use a factor
            factor = 0.7 + 0.3 * progress  # Start at 0.7, go to 1.0 (full reduction)
            scale[above] = 1.0 - factor * (1.0 - reduction)
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_equalize(a, n_iters=500, n_peaks=5, step=0.01):
    """
    Try to equalize the top peaks of the convolution.
    For each of the top peaks, find which a_j contribute most and reduce them,
    while increasing the less-contributing elements to maintain sum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # dg_k/da_j = a_{k-j} + a_j for valid j
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                sg[j] += a[k - j] + a[j]
        
        # Gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n / S2) * sg - (2.0 * n * M_val / S3)
        
        # Take a step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        actual_step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        actual_step = min(actual_step, 0.05)
        
        a -= actual_step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        step *= 0.999
    
    return best_a

def perturbed_restart(a, rng, sigma=0.05):
    """Create a perturbed version of a for multi-start."""
    n = len(a)
    a_pert = a * (1.0 + sigma * rng.standard_normal(n))
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = a_pert.sum()
    if S < 1e-12:
        return a.copy()
    return a_pert / S

def local_search_decrease(a, rng, n_trials=200):
    """
    Try random local modifications that decrease the score.
    Focus on elements that contribute to the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for trial in range(n_trials):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find elements that contribute to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end+1]
        contrib = seg * a[k_star-j_start:k_star-j_end+1][::-1]
        
        # Pick a random element from the contributing set
        if len(contrib) == 0:
            break
        weights = contrib + 1e-10
        weights /= weights.sum()
        j_local = rng.choice(len(contrib), p=weights)
        j = j_start + j_local
        
        # Try to decrease a[j] and increase a random other element
        old_val = a[j]
        if old_val < 1e-10:
            continue
        
        # Decrease by a fraction
        frac = rng.uniform(0.05, 0.3)
        delta = old_val * frac
        a[j] = old_val - delta
        
        # Add delta to a random element (weighted towards edges)
        other_j = rng.integers(0, n)
        a[other_j] += delta
        
        S = a.sum()
        if S < 1e-12:
            a[j] = old_val
            a[other_j] -= delta
            continue
        a = a / S
        sc = compute_score(a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()  # Restore best
    
    return best_a
