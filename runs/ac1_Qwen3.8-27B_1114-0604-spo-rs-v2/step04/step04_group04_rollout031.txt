"""
Multi-phase optimizer combining frequency-domain spectral flattening,
LP-based alternating, and L-BFGS refinement for minimizing peak autocorrelation
of non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def spectral_flatten(a, n_iters=200, alpha=0.1, gamma=0.5):
    """
    Frequency-domain spectral flattening.
    Iteratively:
    1. Compute FFT of a (zero-padded)
    2. Modify amplitude to be more flat
    3. Inverse FFT
    4. Clip to non-negative
    5. Normalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Zero-pad and FFT
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        amp = np.abs(A)
        amp_mean = amp.mean()
        if amp_mean < 1e-15:
            break
        
        # Flatten: compress dynamic range
        # amp_new = amp^(1-alpha) * amp_mean^alpha
        # This reduces the contrast between high and low amplitude frequencies
        amp_new = np.power(amp + 1e-30, 1.0 - alpha) * np.power(amp_mean, alpha)
        
        # Also try: amp_new = amp / (1 + alpha * (amp - amp_mean) / amp_mean)
        # This is a softer flattening
        
        # Reconstruct with modified amplitude, keeping phase
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Blend: don't fully replace, to maintain stability
        a = gamma * a + (1.0 - gamma) * a_new
        
    return best_a

import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    c = self_conv(a)

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Combines POCS-style spectral flattening, exact LP alternating (via CVXPY for
moderate sizes and FISTA for large sizes), LSE-smoothed L-BFGS, and
diversified local search with adaptive perturbation.
Key differences from prior work: (1) spectral domain POCS iteration as a
distinct phase, (2) min-peak LP formulation instead of max-sum, (3) 
simulated-annealing-style diversification to escape local minima.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import diags
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    n = len(a)
    offsets = list(range(0, -n, -1))
    data = [np.full(n - k, a[k]) for k in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def spectral_flatten(a, n_iters=300, alpha_init=0.15, alpha_end=0.02, gamma=0.6, time_limit=None):
    n = len(a)
    a = normalize(a.copy())
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        frac = min(1.0, it / max(1, n_iters * 0.4))
        alpha = alpha_init + (alpha_end - alpha_init) * frac
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A_spec = np.fft.fft(a_pad)
        amp = np.abs(A_spec)
        amp[0] = np.abs(A_spec[0])
        amp_mean = amp[1:].mean() if n > 1 else 1.0
        if amp_mean < 1e-15:
            break
        amp_new = np.power(amp + 1e-30, 1.0 - alpha) * np.power(amp_mean + 1e-30, alpha)
        amp_new[0] = amp[0]
        phase = np.angle(A_spec)
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = gamma * a + (1.0 - gamma) * a_new
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
    return best_a


def lp_solve_fista(a, M, n_fista_iters=4000, beta_start=3.0, beta_end=80.0):
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    for it in range(n_fista_iters):
        frac = min(1.0, it / (n_fista_iters * 0.25))
        beta = beta_start + (beta_end - beta_start) * frac
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    return x


def solve_lp_cvxpy(a, timeout=120):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    n = len(a)
    A_mat = build_conv_matrix(a)
    x = cp.Variable(n)
    M_p = cp.Variable()
    constraints = [
        A_mat @ x <= M_p,
        cp.sum(x) == 1,
        x >= 0
    ]
    prob = cp.Problem(cp.Minimize(M_p), constraints)
    solvers = []
    for s in ['GUROBI', 'MOSEK', 'CLARABEL', 'PDLP', 'ECOS', 'SCS']:
        solvers.append(s)
    for solver in solvers:
        try:
            prob.solve(solver=cp.def_constr.get_attr_value
                       if False else None)
        except Exception:
            pass
        try:
            solver_obj = getattr(cp, solver, None)
            if solver_obj is not None:
                prob.solve(solver=solver_obj)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    x_val = np.array(x.value).flatten()
                    return np.clip(x_val, 0, None), float(M_p.value)
        except Exception:
            continue
    return None, None


def lp_alternating(a, max_outer_iters=50, fista_iters=4000, t_start=0.5, t_min=0.05,
                   time_limit=None, use_cvxpy=False, n_cvxpy_max=5000):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    t_current = t_start
    t_start_func = time.monotonic()
    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        M = float(self_conv(a).max())
        x = None
        if use_cvxpy and n <= n_cvxpy_max:
            x, M_p = solve_lp_cvxpy(a)
            if x is not None and x.sum() > 1e-12:
                x = x / x.sum()
        if x is None:
            x = lp_solve_fista(a, M, n_fista_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        a_new = (1.0 - t_current) * a + t_current * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                t_current *= 0.7
                if t_current < t_min * 0.5:
                    break
                no_improve = 0
        t_current = max(t_min, t_start * (1.0 - 0.4 * outer / max(1, max_outer_iters)))
        a = a_new
    return best_a, best_sc


def lse_score_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=150, alpha=0.75):
    n = len(a)
    a = normalize(a.copy())
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def diversify_search(a_best, n, rng, n_trials=10, time_limit=None, t_start_func=None):
    results = []
    t_ref = t_start_func if t_start_func is not None else time.monotonic()

            a_pert = a_pert * (1.0 + scale * rng.standard_normal(n))
        elif strategy == 1:

            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(3, n // 300)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_idx:
                gi = j_start + li
                red = a_pert[gi] * (0.05 + 0.05 * rng.random())
                a_pert[gi] -= red
                total_red += red
            if total_red > 0:
                bottom_idx = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_idx)
                for li in bottom_idx:
                    a_pert[j_start + li] += per
        elif strategy == 2:
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.06 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = a_pert * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 3:

        a_pert = np.maximum(a_pert, 0.0)
        S_p = float(a_pert.sum())

def subgradient_polish(a, total_iters=2000):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = self_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    return best_a

        sc = compute_score(a_out)

        a_work = a_out.copy()

    # Try other lengths
    for n_alt in lengths:
        if n_alt == n_main or time.monotonic() > phase2_deadline - 60:
            continue
        a_test = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_a)), best_a)
        a_test = normalize(a_test)
        print(f"  LP alt: n={n_alt}")
        a_out, sc_out = lp_alternating(
            a_test, max_outer_iters=25, n_fista_iters=2500,
            t_start=0.4, t_min=0.1, time_limit=phase2_deadline - 20
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    n_best = len(best_a)
    a_norm = normalize(best_a.copy())

    for beta in [50, 150, 400, 1000, 2000]:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=4000, maxfun=80000)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = normalize(a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Diversified local search + polish (15% time)
    # ==================================================================
    print(f"\n--- Phase 4: Diversified Search & Polish ---")

    n_best = len(best_a)
    a_norm = normalize(best_a.copy())

    # Generate diverse perturbations
    perturbations = diversify_search(a_norm, n_best, rng, n_trials=12,
                                     time_limit=deadline - 60)
    for pi, a_p in enumerate(perturbations):
        if time.monotonic() > deadline - 30:
            break
        sc, a_opt = run_lbfgs(a_p, 200, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"div #{pi}")

    # Subgradient polish on best
    if time.monotonic() < deadline - 120:
        print("  Subgradient polish...")
        a_sp = subgradient_polish(best_a, total_iters=2000)
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    # Final LP polish
    if time.monotonic() < deadline - 80:
        print("  Final LP polish...")
        a_final = normalize(best_a.copy())
        a_out, sc_out = lp_alternating(
            a_final, max_outer_iters=20, n_fista_iters=5000,
            t_start=0.15, t_min=0.02, time_limit=deadline - 30
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")
        # L-BFGS after LP
        if time.monotonic() < deadline - 15:
            sc2, a_opt2 = run_lbfgs(normalize(a_out), 3000, maxiter=3000, maxfun=50000)
            report(sc2, a_opt2, "final LBFGS")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
