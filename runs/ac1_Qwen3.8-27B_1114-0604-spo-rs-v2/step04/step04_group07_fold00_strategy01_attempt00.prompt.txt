def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)
    c = fast_conv(a)

def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

def adam_lse_direct(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start,
                    normalize=True, verbose_interval=5000):
    """Adam with LSE smoothing and scheduled beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a

            lr_t = lr / (1.0 + 1e-5 * t)
            score, grad = lse_score_grad_a(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if (bi * steps_per_beta + t) % verbose_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

def lp_dual_update(a, M, max_iter=200, lr=0.1):
    """
    LP-based dual update: find g maximizing sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Uses projected subgradient on the dual for scalability.
    Returns the optimal g (normalized to same sum as a).
    """
    n = len(a)
    # The primal: max 1^T g s.t. A g <= M*1, g >= 0
    # where Ag[k] = sum_j a[k-j]*g[j] (valid range)
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # Where A^T y [j] = sum_k a[k-j]*y[k]
    #
    # We solve the dual using projected subgradient, then recover primal from y.
    # Actually, let's use a simpler multiplicative weight approach.
    
    # Multiplicative weight: start with g = uniform, update based on constraints.
    g = np.ones(n) / n
    y = np.zeros(2 * n - 1)  # dual variables for each constraint
    
    for it in range(max_iter):
        # Compute constraint violations: (a*g)[k] - M for each k
        c = _cross_conv(a, g)
        violations = np.maximum(0, c - M)
        
        # Update dual variables
        y += lr * violations
        
        # Recover primal from dual: g[j] proportional to exp(-sum_k a[k-j]*y[k])
        # A^T y [j] = sum_k a[max(0, k-j)] * y[k] ... 
        # Let's compute: for each j, sum_k a[k-j]*y[k] where 0<=k-j<n
        # = sum_i a[i]*y[i+j] where i = k-j, so j = k-i
        # = (a * y)[j + n - 1] ... approximately
        # Let me just compute it directly via FFT
        M_fft = 1 << (3 * n - 2).bit_length()
        A_fft = np.fft.rfft(a, M_fft)
        Y_fft = np.fft.rfft(y, M_fft)
        Ay = np.fft.irfft(A_fft * Y_fft, M_fft)[:3 * n - 2]
        # A^T y [j] = sum_k y[k] * a[k-j] = (y * a_rev)[j + n - 1]
        a_rev = a[::-1]
        M_fft2 = 1 << (3 * n - 2).bit_length()
        Y_fft2 = np.fft.rfft(y, M_fft2)
        AR_fft2 = np.fft.rfft(a_rev, M_fft2)
        Aty = np.fft.irfft(Y_fft2 * AR_fft2, M_fft2)[:3 * n - 2]
        Aty = Aty[n - 1:2 * n - 1]
        
        # Multiplicative weight update on g
        g = g * np.exp(-lr * Aty)
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    
    return g


def _cross_conv(a, b):
    """Linear cross-convolution of a and b."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def multiplicative_weight_update(a, n_iters=100, lr=0.5):
    """
    Direct multiplicative weight update targeting the max convolution lag.
    At each step, find the peak lag and disproportionate mass away from
    elements contributing to that peak.
    """
    n = len(a)
    g = a.astype(np.float64).copy()
    g = g / g.sum()
    
    for it in range(n_iters):
        c = fast_conv(g)
        k_star = int(c.argmax())
        M = c[k_star]
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            contrib[js:je + 1] = g[idx]
        
        # Reduce elements with high contribution
        g = g * np.exp(-lr * contrib / max(M, 1e-10))
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    
    return g


def coord_descent_peak_target(a, n_sweeps=50, time_limit_s=None, t_start=None):
    """
    Coordinate descent that targets peak-reducing moves.
    At each step, identify elements contributing to the peak and try to reduce them,
    redistributing mass to non-contributing elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Reduce peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing
            for frac in [0.8, 0.6, 0.4, 0.2]:
                a[j] = a_old * frac
                sc = compute_score(a)
                # Redistribute to non-contributing elements
                removed = a_old - a[j]
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    a[non_contrib] += removed / non_contrib.sum()
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            a[j] = a[j]  # keep best

        # Also try increasing non-contributing elements
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) > 0:
            # Pick a random non-contributing element and increase it
            j = nc_idx[np.random.randint(len(nc_idx))]
            a_old = a[j]
            S = a.sum()
            for mult in [1.2, 1.5, 2.0]:
                a[j] = a_old * mult
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            # Revert if no improvement
            if sc >= score:
                a[j] = a_old

    return best_score, a


def generate_initializations(rng, n):
    """Generate diverse initializations for a sequence of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (centered)
    x = np.arange(n) / n
    sigma = 0.15 + 0.2 * rng.random()
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Sinusoidal modulation
    freq = 2 + 8 * rng.random()
    phase = 2 * np.pi * rng.random()
    s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Multi-frequency
    f1 = 1 + 3 * rng.random()
    f2 = 4 + 8 * rng.random()
    p1 = 2 * np.pi * rng.random()
    p2 = 2 * np.pi * rng.random()
    m = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * x + p1) + 0.1 * np.cos(2 * np.pi * f2 * x + p2)
    m = np.maximum(m, 0.01)
    inits.append(m / m.sum())

    # Two-bump
    c1 = 0.25 + 0.1 * rng.random()
    c2 = 0.65 + 0.1 * rng.random()
    s1 = 0.05 + 0.1 * rng.random()
    s2 = 0.05 + 0.1 * rng.random()
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    # Smooth with a running average
    k = max(3, n // 50)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.5
    h_smooth = np.convolve(h, np.ones(max(3, half_n // 50)) / max(3, half_n // 50), mode='same')
    pal = np.concatenate([h_smooth, h_smooth[::-1]])
    inits.append(pal / pal.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def resample_sequence(a, new_n):
    """Resample to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
