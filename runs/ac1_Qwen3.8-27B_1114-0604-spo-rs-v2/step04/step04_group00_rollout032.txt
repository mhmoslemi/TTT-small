"""
Hybrid optimizer: LSE-LBFGS + Exact Subgradient with Backtracking + Targeted Fienup + Support Pruning.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of sum_k p_k * c_k w.r.t. a_j
    # c_k = sum_{j+k-j <= n-1} a_j * a_{k-j}
    # d(sum_k p_k c_k)/da_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
    # = 2 * (p * rev_a)_j where the convolution is truncated
    rev_a = a[::-1]
    M2 = 3 * n - 2  # max index in p is 2n-2, max index in rev_a is n-1, so max in conv is 3n-3
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    # (p * rev_a)_j = sum_k p_k * rev_a_{j-k} = sum_k p_k * a_{n-1-(j-k)} = sum_k p_k * a_{n-1-j+k}
    # We need indices where n-1-j+k is in [0, n-1], i.e., j-(n-1) <= k <= j
    # conv_res[j] = sum_k p_k * a_{n-1-(j-k)} for valid indices
    # Actually let me think more carefully.
    # p has length 2n-1 (indices 0 to 2n-2)
    # rev_a has length n (indices 0 to n-1)
    # Full convolution has length 2n-1 + n - 1 = 3n-2 (indices 0 to 3n-3)
    # conv_res[m] = sum_k p[k] * rev_a[m-k] for valid k, m-k
    # We want g_j = 2 * sum_k p[k] * a[k-j] for valid k-j (i.e., 0 <= k-j <= n-1)
    # rev_a[i] = a[n-1-i]
    # sum_k p[k] * rev_a[m-k] where m-k = n-1-(k-j) = n-1-k+j, so m = n-1+j
    # So g_j = 2 * conv_res[n-1+j]
    g = 2.0 * conv_res[n-1:n-1+n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3 * a)
    return score, grad

def lse_objective_and_jac(x, beta):
    """Top-level objective for scipy."""
    return lse_score_and_grad(x, beta)

def run_lbfgs_ratcheted(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS with ratchet: only keep improvements."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_jac, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    return a_out

def exact_subgradient_backtrack(a, n_iters=5000, top_m=5, alpha_init=0.01):
    """
    Exact subgradient with backtracking line search.
    At each step, compute the exact subgradient of the max term,
    then backtrack to find a good step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = a.sum()
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-m peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient of max (or sum of top-m)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += 2.0 * a[j_start:j_end+1][::-1]
        
        # Full gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3 * a)
        
        # Backtracking line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Normalize gradient direction
        direction = -grad / (grad_norm + 1e-30)
        
        found = False
        for ls_iter in range(30):
            a_trial = a + alpha * direction
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                alpha *= 0.5
                continue
            a_trial_norm = a_trial / S_trial
            sc_trial = compute_score(a_trial_norm)
            if sc_trial < sc - 1e-10:
                a = a_trial_norm
                found = True
                break
            alpha *= 0.5
        
        if not found:
            # Try a very small step
            alpha = max(alpha * 0.5, 1e-8)
            if alpha < 1e-12:
                break
            a_trial = a + alpha * direction
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = a_trial.sum()
            if S_trial > 1e-12:
                a = a_trial / S_trial
            else:
                break
        
        # Slowly increase alpha
        alpha = min(alpha * 1.05, 0.1)
    
    return best_a

def fienup_targeted(a, n_iters=200, target_flatness=0.9):
    """
    Fienup with a target spectrum: try to make |FFT(a)|^2 close to a target profile.
    The target is a flattened version of the current spectrum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Target: geometric mean of spectrum (flattened)
        # Blend between current and flat
        t = 0.3 + 0.5 * progress
        P_target = P_mean * (1 - t) + t * np.ones_like(P) * P_mean
        
        # Actually, let's use a more principled target:
        # We want P to be as flat as possible, but the DC component is fixed
        # (it's S^2 = 1 for normalized a)
        # So we flatten all non-DC bins
        
        # Clip top bins toward the target
        scale = np.ones_like(P)
        thresh = P_mean * (1.0 + 0.5 * (1.0 - progress))
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])
        
        # Also try to boost very low bins (optional)
        low_thresh = P_mean * 0.3
        below = P < low_thresh
        if below.any() and progress > 0.5:
            scale[below] = np.sqrt(1.0 + 0.1 * (low_thresh / P[below]))
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def support_prune(a, threshold=1e-6):
    """
    Try zeroing out small elements and see if the score improves.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Find small elements
    small = np.where(a < threshold)[0]
    if len(small) == 0:
        return best_a
    
    # Try zeroing them all
    a_pruned = a.copy()
    a_pruned[small] = 0.0
    S_pruned = a_pruned.sum()
    if S_pruned > 1e-12:
        a_pruned = a_pruned / S_pruned
        sc_pruned = compute_score(a_pruned)
        if sc_pruned < best_sc:
            best_sc = sc_pruned
            best_a = a_pruned.copy()
    
    return best_a

"""
Hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Combines LSE-LBFGS with geometric continuation, exact subgradient with backtracking,
targeted spectral shaping (Fienup), and peak-kicking. Exploits the structure of
the autocorrelation flattening problem (MPSL) with non-negativity constraint.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: dLSE/da_j = sum_k p_k * 2*a_{k-j} (for valid k-j)
    # = 2 * (p convolved with rev_a)[n-1+j]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3 * a)
    return score, grad


def lse_objective_and_jac(x, beta):
    """Top-level for scipy.minimize."""
    x_clipped = np.clip(x, 0.0, 1000.0)
    sc, g = lse_score_and_grad(x_clipped, beta)
    return sc, g


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_jac, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    return a_out


def subgradient_backtrack(a, n_iters=5000, top_m=5, alpha_init=0.01, momentum=0.0):
    """
    Exact subgradient with backtracking line search and optional momentum.
    Targets top-M peaks for a smoother effective gradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init
    v = np.zeros(n)  # momentum

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-m peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3 * a)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Momentum update
        v = momentum * v - alpha * grad
        v_norm = float(np.linalg.norm(v))
        if v_norm < 1e-15:
            break

        # Backtracking line search
        direction = v / v_norm
        found = False
        for _ in range(40):
            a_trial = a + direction
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                alpha *= 0.5
                v *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial < sc - 1e-12:
                a = a_trial
                found = True
                break
            alpha *= 0.4
            v = momentum * v - alpha * grad
            v_norm = float(np.linalg.norm(v))
            if v_norm < 1e-15:
                break
            direction = v / v_norm

        if not found:
            # Increase alpha slightly for next iteration
            alpha = min(alpha * 1.1, 0.05)
            if alpha < 1e-10:
                break
            # Very small step
            a_trial = a - alpha * grad / (grad_norm + 1e-30)
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = a_trial.sum()
            if S_trial > 1e-12:
                a = a_trial / S_trial
            else:
                break
        else:
            alpha = min(alpha * 1.02, 0.05)

    return best_a
