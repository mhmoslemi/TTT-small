"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE continuation, projected subgradient, coordinate descent, and
Fienup-style spectral iterations. Explores multiple sequence lengths
and diverse initializations to escape local optima.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy.minimize."""
    n = len(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c = fast_conv(x)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, None)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    S_out = float(a_out.sum())
    if S_out < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def subgradient_refine(a, n_iters=5000, top_m=20, step_init=0.02):
    """Subgradient method on the exact max-convection objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_m)[-top_m:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max(0.01, (sc - sc_lb)) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9997

    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=40):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if sweep > 0 and sweep % 2 == 0:
            c = fast_conv(a)
            k_star = int(np.argmax(c))
        else:
            c = fast_conv(a)
            k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        local_top = np.argsort(contrib)[::-1][:top_count]
        global_top = j_start + local_top

        cur_sc = best_sc
        for j in global_top:
            a_old = a[j]
            if a_old < 1e-10:
                continue
            factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6]
            best_val = a_old
            best_local = cur_sc
            for f in factors:
                cv = a_old * f
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a


def fienup_flatten(a, n_iters=200, top_frac=0.25):
    """Fienup-style: flatten power spectrum to reduce convolution peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        prog = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac * (1.0 + 0.5 * prog))
        above = P > thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        strength = 0.4 + 0.4 * prog
        scale[above] = (thresh / P[above]) ** strength
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, None).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, np.clip(parent, 0, None)).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)) ** 2)
    sigma = 0.3
    inits.append(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)))
    sigma = 0.4
    inits.append(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)))
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.4 + 0.6 * np.sin(np.pi * x) ** 2)
    return inits


def continuation_lbfgs(a, betas=None, maxiter_per_beta=3000, maxfun_per_beta=50000):
    """LSE continuation: optimize with increasing beta."""
    if betas is None:
        betas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(len(a)) * 0.5
        S = len(a) * 0.5
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        sc, a_opt = run_lbfgs(a, beta, maxiter=maxiter_per_beta, maxfun=maxfun_per_beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a = a_opt
    return best_sc, best_a
