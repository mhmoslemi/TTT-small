def _npt(a):
    """Normalize to non-negative, sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / S


def _self_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.inf
    c = _self_conv(a)
    cmax = float(c.max())
    return float(2.0 * n * cmax / (S * S))


def _lse_obj_grad(x, beta):
    """LSE-smoothed score and its gradient."""
    n = len(x)
    S = float(x.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n, dtype=np.float64)
    c = _self_conv(x)
    d = float(n) * c / (S * S)
    dm = float(d.max())
    if dm <= 0 or not np.isfinite(dm):
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = max(beta, 10.0)
    z = beta * (d - dm)
    z_shift = z - z.max()
    em = np.exp(z_shift)
    se = float(em.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = em / se
    L = dm + (z.max() + np.log(se)) / beta

    # h[j] = sum_{m=0}^{n-1} x[m] * w[m+j]
    x_rev = x[::-1]
    Nc = _next_pow2(3 * n)
    Xr = np.fft.rfft(x_rev, Nc)
    W = np.fft.rfft(w, Nc)
    Hfull = np.fft.irfft(Xr * W, Nc)
    h = Hfull[n - 1:2 * n - 1]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)

    wac = float(np.dot(w, c))
    grad = (2.0 * n / (S * S)) * h - (2.0 * n / (S ** 3)) * wac * x
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * L), grad


def _run_lbfgs(a, beta, deadline, maxfun):
    """Run L-BFGS-B on LSE objective."""
    a = _npt(a)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return _score(a), a
        est = 0.02 * max(1.0, float(n) / 4096.0)
        cap = int((0.85 * rem) / max(est, 1e-4))
        maxfun = int(min(maxfun, max(64, cap)))

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(_lse_obj_grad, a, method="L-BFGS-B", jac=True,
                       args=(beta,), bounds=bounds, options=opts)
        a_out = _npt(res.x)
    except Exception:
        a_out = a
    sc = _score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _score(a)
    return float(sc), a_out


def _cross_corr(a, w):
    """Compute h[j] = sum_m a[m]*w[m+j] for j=0..n-1 via FFT."""
    n = len(a)
    x_rev = a[::-1]
    Nc = _next_pow2(3 * n)
    Xr = np.fft.rfft(x_rev, Nc)
    W = np.fft.rfft(w, Nc)
    Hfull = np.fft.irfft(Xr * W, Nc)
    return np.nan_to_num(Hfull[n - 1:2 * n - 1], nan=0.0, posinf=0.0, neginf=0.0)


def _coord_descent(a, deadline, max_sweeps=8):
    """Coordinate descent: for each coordinate, find optimal value by 1D search.
    Focus on coordinates that contribute to the current peaks."""
    a = _npt(a.copy())
    n = len(a)
    best_sc = _score(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for sweep in range(int(max_sweeps)):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S = float(a.sum())
        cmax = float(c.max())
        # Find top peaks
        npeaks = min(8, 2 * n - 1)
        peak_idx = np.argpartition(-c, npeaks)[:npeaks]
        peak_set = set(peak_idx.tolist())

        # Coordinates to try: those contributing to peaks
        # For peak at position k, contributing coords are j where 0<=k-j<n → j in [k-n+1, k]
        coord_set = set()
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                coord_set.add(j)
                if len(coord_set) > 400:
                    break
            if len(coord_set) > 400:
                break

        if not coord_set:
            coord_list = list(range(n))
        else:
            coord_list = list(coord_set)

        # Shuffle for diversity
        rng_local = np.random.default_rng(sweep * 1000 + 42)
        rng_local.shuffle(coord_list)

        improved = 0
        for j in coord_list:
            if time.monotonic() > deadline:
                break

            x = a[j]
            # Compute C = conv with a[j] set to 0
            b = a.copy()
            b[j] = 0.0
            S_b = S - x

            # C[k] = c[k] - x*b[k-j] for k in [j, j+n-1], k != 2j
            # C[2j] = c[2j] - x^2
            # C[k] = c[k] for k outside [j, j+n-1]

            # For candidate x': new_c[k] = C[k] + x'*b[k-j] (for k in range) or C[2j]+x'^2
            # score(x') = 2n * max(new_c) / (S_b + x')^2

            # Key: find x' that minimizes the score
            # The max of new_c is the max of:
            #   (a) max of C[k] for k not in [j, j+n-1] (constant, call it M_const)
            #   (b) max of C[k] + x'*b[k-j] for k in [j,j+n-1], k!=2j (linear in x')
            #   (c) C[2j] + x'^2 (quadratic)

            lo_k = j
            hi_k = j + n - 1
            hi_k = min(hi_k, 2 * n - 2)

            # M_const: max of c[k] for k outside [lo_k, hi_k]
            if lo_k > 0:
                M_const_left = float(c[:lo_k].max())
            else:
                M_const_left = 0.0
            if hi_k < 2 * n - 2:
                M_const_right = float(c[hi_k + 1:].max())
            else:
                M_const_right = 0.0
            M_const = max(M_const_left, M_const_right) if n > 1 else 0.0

            # For the range [lo_k, hi_k]:
            # new_c[k] = c[k] - x*b[k-j] + x'*b[k-j] = c[k] + (x'-x)*b[k-j] (for k!=2j)
            # new_c[2j] = c[2j] - x^2 + x'^2
            delta = 0  # placeholder

            # Vectorized: new_c in range = c[lo_k:hi_k+1] + (x'-x)*b[lo_k-j:hi_k-j+1]
            # But for k=2j (if in range), it's c[2j] + (x'^2 - x^2) instead of (x'-x)*b[2j-j] = (x'-x)*0 = 0
            # Since b[j] = 0, the formula c[k] + (x'-x)*b[k-j] gives c[2j] + 0 = c[2j] for k=2j.
            # But the true value is c[2j] - x^2 + x'^2. So we need to handle k=2j separately.

            b_range = b[lo_k - j:hi_k - j + 1]  # length = hi_k - lo_k + 1 = n
            c_range = c[lo_k:hi_k + 1]

            # k=2j position in the range: index = 2j - lo_k = 2j - j = j
            idx_2j = j  # position in the range array

            # For a given x', max_in_range = max(c_range + (x'-x)*b_range, but at idx_2j: c[2j] - x^2 + x'^2)
            # = max( max(c_range + (x'-x)*b_range, with idx_2j replaced by c[2j]-x^2+x'^2) )

            # Try several candidate x' values:
            # x'=0, x'=x, and values that balance the peaks
            candidates = [0.0, x]
            # Find x' that makes the max in range = M_const (if M_const > current max in range at x'=0)
            # At x'=0: max_in_range_0 = max(c_range - x*b_range, with idx_2j: c[2j]-x^2)
            # We want max_in_range(x') to be minimized.

            # Simple approach: try geometric interpolation
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                candidates.append(x * frac)
            # Also try increasing
            for mult in [1.5, 2.0, 3.0]:
                candidates.append(x * mult)

            best_x_prime = x
            best_local_sc = best_sc + 1.0

            for xp in candidates:
                if xp < 0:
                    continue
                # Compute new max
                new_range = c_range + (xp - x) * b_range
                if 0 <= idx_2j < n:
                    new_range = new_range.copy()
                    new_range[idx_2j] = c[2 * j] - x * x + xp * xp
                max_range = float(new_range.max())
                max_total = max(M_const, max_range)
                S_new = S_b + xp
                if S_new < 1e-15:
                    continue
                sc_trial = 2.0 * n * max_total / (S_new * S_new)
                if sc_trial < best_local_sc:
                    best_local_sc = sc_trial
                    best_x_prime = xp

            # Golden section search around best candidate for refinement
            if best_x_prime > 0 or x > 0:
                lo = max(0.0, best_x_prime * 0.5 - x * 0.2)
                hi = best_x_prime * 1.5 + x * 0.5
                # Evaluate at golden ratio points
                gr = (np.sqrt(5) - 1) / 2
                for _ in range(15):
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)

                    def eval_at(xp):
                        new_range = c_range + (xp - x) * b_range
                        if 0 <= idx_2j < n:
                            new_range = new_range.copy()
                            new_range[idx_2j] = c[2 * j] - x * x + xp * xp
                        max_total = max(M_const, float(new_range.max()))
                        S_new = S_b + xp
                        if S_new < 1e-15:
                            return 1e10
                        return 2.0 * n * max_total / (S_new * S_new)

                    f1 = eval_at(c1)
                    f2 = eval_at(c2)
                    if f1 < f2:
                        hi = c2
                    else:
                        lo = c1

                    if eval_at(best_x_prime) > f1:
                        best_x_prime = c1
                    if eval_at(best_x_prime) > f2:
                        best_x_prime = c2

                mid = (lo + hi) / 2
                fm = eval_at(mid)
                if fm < eval_at(best_x_prime):
                    best_x_prime = mid

            if best_x_prime + 1e-15 != x:
                a[j] = best_x_prime
                S += best_x_prime - x
                # Update c approx: skip full recompute, just track improvement
                sc_now = _score(a)
                if sc_now < best_sc - 1e-10:
                    best_sc = sc_now
                    best_a = a.copy()
                    improved += 1
                elif sc_now > best_sc + 0.001:
                    # Revert
                    a[j] = x
                    S -= best_x_prime - x

        if improved == 0 and sweep > 2:
            break

        # Full recompute after sweep
        best_sc = _score(a)
        if a is not best_a:
            # a was modified in place, need to check
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

        print("  Coord descent sweep {}: score={:.8f}, improved_coords={}".format(
            sweep + 1, best_sc, improved))

    return float(best_sc), _npt(best_a)


def _subgrad_polish(a, deadline, max_iters=50):
    """Momentum-based subgradient on the exact max with adaptive step."""
    a = _npt(a.copy())
    n = len(a)
    best_sc = _score(a)
    best_a = a.copy()
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.5  # momentum coefficient
    t0 = time.monotonic()

    for it in range(int(max_iters)):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S = float(a.sum())
        cmax = float(c.max())
        if cmax < 1e-15:
            break

        # Find top-K peaks
        K = min(5, 2 * n - 1)
        peak_idx = np.argpartition(-c, K)[:K]

        # Compute pressure: for each peak, which coordinates contribute
        # h[j] = sum over peaks k of (c[k]/cmax) * a[k-j] (normalized)
        h = np.zeros(n, dtype=np.float64)
        for k in peak_idx:
            weight = c[k] / cmax
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            h[lo:hi + 1] += weight * a[lo:hi + 1]  # a[k-j] for j in [lo,hi] → a[lo:hi+1] reversed?
            # Actually: for peak at k, dc_k/da[j] = 2*a[k-j] for 0<=k-j<n
            # So contribution to pressure on j is proportional to a[k-j]
            # h[j] += weight * a[k-j] for j in [k-n+1, k]
            # Let m = k-j, j = k-m, m in [0, n-1], j in [k-n+1, k]
            # h[j] += weight * a[k-j] = weight * a[m]
            # So for m=0..n-1: h[k-m] += weight * a[m]
            # i.e., h[k-0:k-n+1-(-1)] += weight * a reversed... 
            # Let me just do: for m in range(min(n, k+1)): h[k-m] += weight * a[m]
            # But that's O(n) per peak, O(Kn) total. Fine for K=5.

        # Recompute h properly
        h = np.zeros(n, dtype=np.float64)
        for k in peak_idx:
            weight = c[k] / cmax
            m_max = min(n, k + 1)
            # m from 0 to m_max-1: j = k-m, h[j] += weight * a[m]
            ms = np.arange(m_max)
            js = k - ms
            h[js] += weight * a[ms]

        hmax = float(h.max())
        if hmax < 1e-15:
            break

        # Adaptive step: try a few step sizes and pick the best
        base_eta = 0.05 / (1.0 + 0.1 * it)
        best_q = a.copy()
        best_trial_sc = best_sc

        for mult in [0.3, 0.7, 1.0, 1.5, 2.0]:
            eta = base_eta * mult
            # Momentum update
            momentum = mu * momentum + (1 - mu) * h / hmax
            q = a * np.exp(-eta * momentum)
            q = _npt(q)
            sc_q = _score(q)
            if np.isfinite(sc_q) and sc_q < best_trial_sc:
                best_trial_sc = sc_q
                best_q = q.copy()

        if best_trial_sc < best_sc:
            a = best_q
            best_sc = best_trial_sc
            best_a = a.copy()
        else:
            # Reduce step
            mu = max(0.1, mu * 0.9)

        if it % 10 == 0:
            print("  Subgrad iter {}: score={:.8f}".format(it + 1, best_sc))

    return float(best_sc), _npt(best_a)


def _perturb(a, rng, strength=0.3):
    """Perturb a sequence for diversity."""
    n = len(a)
    p = _npt(a.copy())
    mode = int(rng.integers(0, 4))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * strength * rng.standard_normal(n))
    else:
        # Smooth perturbation
        noise = rng.standard_normal(n)
        k = min(21, max(5, n // 100))
        if k % 2 == 0:
            k += 1
        sig = max(1.0, k / 4.0)
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        noise -= noise.mean()
        q = p * np.exp(strength * 0.5 * noise)
    return _npt(q)


def _resample(a, n_new):
    """Resample sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400):
    """Search for a low-score nonnegative sequence under the 2400s budget."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        pass

    print("Parent available: {}, len={}".format(
        parent is not None, len(parent) if parent is not None else 0))

    # Choose lengths to try
    lengths = []
    if parent is not None:
        plen = min(len(parent), 12288)
        lengths.append(plen)
        if plen != 8192:
            lengths.append(8192)
    else:
        lengths = [8192, 6144]

    # Deduplicate and cap
    seen = set()
    unique_lengths = []
    for n in lengths:
        if n not in seen and 1024 <= n <= 16384:
            unique_lengths.append(n)
            seen.add(n)
    if not unique_lengths:
        unique_lengths = [8192]

    print("Lengths to try: {}".format(unique_lengths))

    best_seq = None
    best_score = np.inf

    for n in unique_lengths:
        if time.monotonic() > deadline - 300:
            break

        # Generate starting point
        if parent is not None and len(parent) > 0:
            start = _resample(parent, n)
        else:
            start = np.ones(n) / n
        start = _npt(start)

        sc0 = _score(start)
        print("n={}, initial score={:.8f}".format(n, sc0))

        # Phase 1: LSE continuation
        sub_deadline = min(deadline - 120, t_start + 0.45 * float(budget_s))
        a_cur = start.copy()
        sc_cur = sc0

        beta_schedule = [30, 60, 100, 200, 400, 800, 1500, 3000, 6000, 12000]
        for beta in beta_schedule:
            if time.monotonic() > sub_deadline - 10:
                break
            rem = float(sub_deadline - time.monotonic())
            if rem < 10:
                break
            # Estimate max function evaluations
            est_time = 0.02 * max(1.0, float(n) / 4096.0)
            maxfun = int((0.7 * rem) / max(est_time, 1e-4))
            maxfun = max(64, min(maxfun, 2000))

            sc, a2 = _run_lbfgs(a_cur, beta, sub_deadline, maxfun)
            if sc < sc_cur:
                sc_cur = sc
                a_cur = a2.copy()
                if sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("  LSE beta={}: score={:.8f}".format(beta, sc))
            a_cur = _npt(a2)

        # Phase 2: Coordinate descent
        if time.monotonic() < deadline - 150:
            coord_dead = min(deadline - 120, time.monotonic() + 0.25 * float(budget_s))
            sc_cd, a_cd = _coord_descent(a_cur, coord_dead, max_sweeps=6)
            if sc_cd < sc_cur:
                sc_cur = sc_cd
                a_cur = a_cd
                if sc_cur < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                    print("  Coord descent: score={:.8f}".format(sc_cd))

        # Phase 3: Subgradient polishing
        if time.monotonic() < deadline - 60:
            subg_dead = min(deadline - 30, time.monotonic() + 0.15 * float(budget_s))
            sc_sg, a_sg = _subgrad_polish(a_cur, subg_dead, max_iters=40)
            if sc_sg < sc_cur:
                sc_cur = sc_sg
                a_cur = a_sg
                if sc_cur < best_score:
                    best_score = sc_sg
                    best_seq = a_sg.copy()
                    print("  Subgrad: score={:.8f}".format(sc_sg))

        # Phase 4: Quick LSE with very high beta
        if time.monotonic() < deadline -
