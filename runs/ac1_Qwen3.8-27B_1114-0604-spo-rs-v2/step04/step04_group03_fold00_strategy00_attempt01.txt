```python
    max_conv = np.max(conv_result)
    score_after = 2.0*n_opt*max_conv/seq.sum()**2
    
    if score_after < best_score:
        best_score = score_after
        best_sequence = seq
        
    # Check if we can do better with different initialization
    block_size = n_opt//8
    seq_block = np.ones(block_size)*0.5
    best_candidate = run_lbfgs(seq_block, 300, maxiter=10000)
    if best_candidate[0] < best_score:
        best_score = best_candidate[0]
        best_sequence = best_candidate[1].copy()
        
    # Final evaluation
    out_seq = np.clip(best_sequence, 0, 1000)
    verified_score = evaluate_sequence(out_seq)
    
    # Return the best candidate found
    return out_seq, verified_score
```
