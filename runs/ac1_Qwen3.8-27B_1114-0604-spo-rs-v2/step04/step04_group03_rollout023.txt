import numpy as np
from scipy.optimize import minimize
import time

# Helper functions for FFT-based convolution and scoring

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

# Parametric approach

def make_basis(n, K):
    """Create cosine basis matrix."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    B = np.zeros((n, K + 1))
    for k in range(K + 1):
        B[:, k] = np.cos(k * np.pi * x)
    return B

def parametric_score_grad(w, B, beta):
    """Score and gradient for parametric model."""
    a = B @ w
    a = np.maximum(a, 0.0)  # Project to non-negative
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(len(w))
    a_norm = a / S  # Normalize (score is scale-invariant)
    
    c = fast_conv(a_norm)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L  # Since sum(a_norm) = 1
    
    # Gradient w.r.t. w
    p = exp_s / exp_s.sum()
    # dL/da_norm[i] = 2 * conv(rev(a_norm), p)[n-1+i]
    rev_a = a_norm[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    # Chain rule: da/dw = B (but with the max(.,0) projection)
    # For simplicity, ignore the projection in the gradient (it's a subgradient)
    grad = 2.0 * n * B.T @ dL_da
    return score, grad

def run_parametric(n, K, w0, betas, max_time=300):
    """Run parametric optimization with increasing beta."""
    B = make_basis(n, K)
    best_w = w0.copy()
    best_sc = compute_score(np.maximum(B @ w0, 0))
    
    t_start = time.monotonic()
    for beta in betas:
        if time.monotonic() - t_start > max_time:
            break
        
        def obj(x):
            s, g = parametric_score_grad(x, B, beta)
            return s, g
        
        res = minimize(
            obj, best_w,
            method='L-BFGS-B', jac=True,
            options={'ftol': 1e-15, 'gtol': 1e-10,
                     'maxiter': 5000, 'maxfun': 50000}
        )
        w_new = res.x
        a_new = np.maximum(B @ w_new, 0.0)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_w = w_new.copy()
    
    return best_sc, np.maximum(B @ best_w, 0.0)

Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Frequency-domain peak targeting, LSE continuation (L-BFGS),
and subgradient polish. Novel approach: targeted spectral redistribution
to reduce autocorrelation peaks, combined with adaptive LSE continuation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (a is assumed normalized, sum=1)."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    Z = exp_s.sum()
    L = c_max + np.log(Z) / beta
    score = 2.0 * n * L
    p = exp_s / Z
    # dL/da[i] = 2 * (rev(a) * p)[n-1+i]
    rev_a = a[::-1]
    conv_len = 2 * n - 1
    M_tree = 3 * n - 1
    M_fft = 1 << (M_tree - 1).bit_length()
    RA = np.fft.rfft(rev_a, M_fft)
    P = np.fft.rfft(p, M_fft)
    r = np.fft.irfft(RA * P, M_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = 2.0 * n * dL_da
    return score, grad


def full_lse_score_grad(a, beta):
    """LSE score and grad w.r.t. unnormalized a."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:

    a_norm = a / S
    score_norm, grad_norm = lse_score_grad(a_norm, beta)
    # Chain rule: d/d a[i] f(a/S) = (1/S) * df/da_norm[i] - (a[i]/S^2) * sum_j j * df/da_norm[j]
    # Actually, since f is scale invariant, d/da[i] f(a) = (1/S) * grad_norm[i] - (a[i]/S) * (1/S) * sum(grad_norm)
    # Wait, let me be more careful.

    # df/da[i] = sum_j dg/da_norm[j] * da_norm[j]/da[i]
    # a_norm[j] = a[j] / S
    # da_norm[j]/da[i] = delta_{ij}/S - a[j]/S^2
    # So df/da[i] = sum_j grad_norm[j] * (delta_{ij}/S - a[j]/S^2)
    # = grad_norm[i]/S - (sum_j grad_norm[j] * a[j])/S^2
    # But sum_j grad_norm[j] * a[j] should be 0 for scale-invariant g (Euler's theorem)

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Targeted spectral redistribution to reduce autocorrelation peaks,
fine LSE continuation via L-BFGS, Nesterov-accelerated subgradient polish,
and multi-length exploration. Novel: frequency-domain peak targeting
that shifts spectral energy away from constructive interference at the
dominant lag, combined with adaptive continuation schedule.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalized_lse(a, beta):
    """LSE-smoothed 2n*L where a is normalized (sum=1), plus gradient."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    Z = exp_s.sum()
    L = c_max + np.log(Z) / beta
    score = 2.0 * n * L
    p = exp_s / Z
    rev_a = a[::-1]
    M_tree = 3 * n - 1
    M_fft = 1 << (M_tree - 1).bit_length()
    RA = np.fft.rfft(rev_a, M_fft)
    P = np.fft.rfft(p, M_fft)
    r = np.fft.irfft(RA * P, M_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = 2.0 * n * dL_da
    return score, grad


def run_lbfgs_normalized(a0, beta, maxiter=8000, maxfun=200000):
    """L-BFGS-B on LSE-smoothed objective (a normalized to sum=1)."""
    n = len(a0)
    a_init = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a_init.sum()
    if S < 1e-10:
        a_init[:] = 1.0 / n
    else:
        a_init /= S
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        x2 = np.maximum(x, 0.0)
        S2 = x2.sum()
        if S2 < 1e-10:
            return 1e10, np.zeros(n)
        x2 = x2 / S2
        s, g = normalized_lse(x2, beta)
        # Project gradient back (account for normalization)
        g = g / S2 - (g.dot(x2) / (S2 * S2))
        g = np.maximum(g, -1e6)
        return s, g

    res = minimize(
        obj, a_init,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    S_out = a_out.sum()
    if S_out > 1e-10:
        a_out /= S_out
    sc = compute_score(a_out)
    return sc, a_out


def spectral_flattening(a, n_iters=300, alpha_init=0.3):
    """
    Targeted spectral redistribution: shift energy away from frequencies
    that constructively interfere at the dominant autocorrelation lag.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a /= S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(np.concatenate([a, np.zeros(M - n)]), M)
        c = np.fft.irfft(A * A, M)[:2 * n - 1]
        k_star = int(np.argmax(c))
        c_max = float(c.max())
        c_mean = float(c.mean())
        if c_max - c_mean < 1e-15:
            break

        # Compute interference pattern at k_star
        freqs = np.arange(len(A), dtype=np.float64)
        cos_pattern = np.cos(2.0 * np.pi * freqs * k_star / M)
        cos_pattern[0] = 0.0  # Don't modify DC

        # Adaptive alpha: start strong, decay
        alpha = alpha_init * (1.0 - 0.005 * it)
        alpha = max(alpha, 0.01)

        # Shift energy: reduce amplitude where constructive, increase where destructive
        with np.errstate(divide='ignore', invalid='ignore'):
            mod = np.exp(-alpha * cos_pattern)
        mod[0] = 1.0  # Keep DC
        A_new = A * mod

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new /= S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def amplitude_compression(a, n_iters=200, gamma_start=0.6, gamma_end=0.95):
    """Compress spectral amplitude to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a /= S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        gamma = gamma_start + (gamma_end - gamma_start) * (it / max(1, n_iters - 1))
        A = np.fft.rfft(np.concatenate([a, np.zeros(M - n)]), M)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        mag_new = np.ones_like(mag)
        mask = mag > 1e-10
        mag_new[mask] = mag[mask] ** gamma
        mag_new[0] = mag[0]  # Keep DC
        A_new = phase * mag_new
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new /= S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def nesterov_subgradient(a, n_iters=3000, mu=0.5):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        S = float(y.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Subgradient of max(c) w.r.t. y
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = y[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Scale for the full objective
        gn = np.linalg.norm(sg)
        if gn < 1e-18:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.02)

        y_new = y - step * sg
        y_new = np.maximum(y_new, 1e-15)
        y_new /= y_new.sum()

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu * (it + 1) ** 2)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu * it ** 2)) / 2.0 if it > 0 else 1.0
        beta_k = (t_k - 1.0) / (t_prev + 1e-10)
        a = y_new + beta_k * (y_new - y)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        y = a

    return best_a


def peak_coord_descent(a, n_sweeps=10):
    """Coordinate descent targeting the convolution peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a /= S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        seg = a[j_start:j_end + 1].copy()
        contrib = seg[::-1].copy()
        n_top = max(3, n // 100)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bottom_idx = np.argsort(contrib)[:n_top]

        for li in top_idx:
            gi = j_start + li
            old_val = a[gi]
            best_val = old_val
            best_local = best_sc
            for factor in [0.5, 0.7, 0.85, 0.95]:
                a[gi] = old_val * factor
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = old_val * factor
            a[gi] = best_val

        # Redistribute to bottom contributors
        S = a.sum()
        if S > 1e-12:
            a /= S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
