def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)

    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )

    except Exception:
        a_out = a0

    return exact_sc, a_out


def lp_solve_fista(a, M, n_fista_iters=5000, beta=30.0, nesterov=True):
    """
    Solve approximate LP: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on the LSE-smoothed surrogate.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    L = beta * n * sum_a2 * 1.1 + 1e-10
    step = 1.0 / L

    for it in range(n_fista_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        if nesterov:
            v = u - step * grad
            x_new = np.maximum(v, 0.0)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
            u = np.maximum(u, 0.0)
            x = x_new
            t_mom = t_new
        else:
            x_new = np.maximum(x - step * grad, 0.0)
            x = x_new

    return x

def peak_targeted_reduction(a, n_iters=50, K=10):

        a = a_new

def perturb_and_refine(a, strategy='gaussian', scale=0.02):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 'gaussian':
        a_pert = a * (1.0 + scale * np.random.standard_normal(n))
    elif strategy == 'sinusoidal':
        freq = 1.0 + 10.0 * np.random.random()
        phase = 2.0 * np.pi * np.random.random()
        a_pert = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 'anti_peak':
        # Reduce elements that contribute to the current peak
        c = self_conv(a)
        k_star = int(np.argmax(c))
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        a_pert = a.copy()
        contrib = a[k_star - i_start:k_star - i_start + (i_end - i_start + 1)][::-1]
        top_count = max(3, n // 200)
        top_local = np.argsort(contrib)[::-1][:top_count]
        total_red = 0.0
        for li in top_local:
            gi = i_start + li
            red = a[gi] * 0.02
            a_pert[gi] = max(a[gi] - red, 0.0)
            total_red += red
        if total_red > 0:
            bottom_local = np.argsort(contrib)[:max(1, n - top_count)]
            per = total_red / len(bottom_local)
            for li in bottom_local:
                a_pert[i_start + li] += per
    elif strategy == 'shift':
        # Circular shift by a small amount
        shift = int(np.random.randint(1, max(2, n // 100)))
        a_pert = np.roll(a, shift)
        # Blend to avoid discontinuity
        fade = min(10, n // 100)
        a_pert[:fade] = (1 - np.linspace(1, 0, fade)) * a[:fade] + \
                        (np.linspace(1, 0, fade)) * a_pert[:fade]
        a_pert[-fade:] = (1 - np.linspace(0, 1, fade)) * a[-fade:] + \
                         (np.linspace(0, 1, fade)) * a_pert[-fade:]
    else:
        a_pert = a.copy()

    a_pert = np.maximum(a_pert, 0.0)
    S = float(a_pert.sum())
    if S > 1e-12:
        a_pert = a_pert / S

                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is not None else len(parent)}")

    # Determine working length
    if parent is not None:
        n_main = len(parent)
        a_work = parent.copy()
    else:
        n_main = 6962
        a_work = np.ones(n_main) * 0.5

    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    sc_initial = compute_score(a_work)
    print(f"Initial score: {sc_initial:.8f}, n={n_main}")
    report(sc_initial, a_work, "initial")

    # ==================================================================
    # Phase 1: LP Alternating - Round 1 (moderate) [15% time]
    # ==================================================================
    phase1_end = t_start + 0.15 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: LP Alternating (moderate) ---")

    a_out, sc_out = lp_alternating_improved(
        a_work, max_outer_iters=50, fista_iters=3000,
        t_start=0.4, t_min=0.05, beta_lp=25.0,
        time_limit=phase1_end - 5, patience=15
    )
    sc = compute_score(a_out)
    report(sc, a_out, "LP r1")
    a_work = a_out.copy()

    # ==================================================================
    # Phase 2: LP Alternating - Round 2 (intensive) [35% time]
    # ==================================================================
    phase2_end = t_start + 0.50 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LP Alternating (intensive) ---")

    # Use higher beta for tighter LP approximation
    a_out, sc_out = lp_alternating_improved(
        a_work, max_outer_iters=100, fista_iters=6000,
        t_start=0.25, t_min=0.02, beta_lp=40.0,
        time_limit=phase2_end - 5, patience=25
    )
    sc = compute_score(a_out)
    report(sc, a_out, "LP r2")
    a_work = a_out.copy()

    # ==================================================================
    # Phase 3: LSE L-BFGS with increasing beta [25% time]
    # ==================================================================
    phase3_end = t_start + 0.75 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: LSE L-BFGS (high beta) ---")

    beta_schedule = [50, 100, 200, 500, 1000, 3000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end - 30:
            break
        remaining_frac = (phase3_end - time.monotonic()) / max(1, phase3_end - t_start)
        maxiter = int(2000 + 3000 * remaining_frac)
        maxfun = int(50000 + 100000 * remaining_frac)
        
        a_norm = a_work.copy()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        if sc < compute_score(a_work):
            a_work = a_opt.copy()

    # ==================================================================
    # Phase 4: Iterative perturb + LP + L-BFGS [20% time]
    # ==================================================================
    phase4_end = deadline
    print(f"\n--- Phase 4: Perturb & Refine ---")

    strategies = ['anti_peak', 'gaussian', 'sinusoidal', 'anti_peak', 'gaussian', 'sinusoidal']
    n_perturb_trials = 0
    no_improve_perturb = 0

    while time.monotonic() < phase4_end - 60:
        strategy = strategies[n_perturb_trials % len(strategies)]
        scale = 0.01 + 0.02 * (n_perturb_trials % 3)
        
        a_pert = perturb_and_refine(a_work, strategy=strategy, scale=scale)
        
        # Quick LP alternating on perturbed
        remaining = phase4_end - 60 - time.monotonic()
        if remaining < 10:
            break
        fista_it = max(2000, int(remaining / 15))
        a_out, sc_out = lp_alternating_improved(
            a_pert, max_outer_iters=20, fista_iters=fista_it,
            t_start=0.3, t_min=0.05, beta_lp=30.0,
            time_limit=phase4_end - 30 - time.monotonic() + t_start, patience=10
        )
        
        # Follow with L-BFGS
        if time.monotonic() < phase4_end - 15:
            sc_lb, a_lb = run_lbfgs(a_out, 500, maxiter=3000, maxfun=80000)
            if sc_lb < sc_out:
                a_out = a_lb
                sc_out = sc_lb
        
        sc = compute_score(a_out)
        improved = report(sc, a_out, f"pert #{n_perturb_trials} {strategy}")
        
        if improved:
            a_work = a_out.copy()
            no_improve_perturb = 0
        else:
            no_improve_perturb += 1
            if no_improve_perturb >= 4:
                # Try a larger perturbation
                scale = 0.05
                a_pert = perturb_and_refine(a_work, strategy='gaussian', scale=scale)
                a_out, sc_out = lp_alternating_improved(
                    a_pert, max_outer_iters=15, fista_iters=3000,
                    t_start=0.4, t_min=0.1, beta_lp=25.0,
                    time_limit=phase4_end - 30, patience=10
                )
                sc = compute_score(a_out)
                improved = report(sc, a_out, f"pert #{n_perturb_trials} big_gauss")
                if improved:
                    a_work = a_out.copy()
                    no_improve_perturb = 0

        n_perturb_trials += 1

    # ==================================================================
    # Phase 5: Final intensive polish [remaining time]
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # One more round of LP alternating on the best
    if time.monotonic() < deadline - 30:
        a_out, sc_out = lp_alternating_improved(
            a_work, max_outer_iters=30, fista_iters=8000,
            t_start=0.15, t_min=0.01, beta_lp=50.0,
            time_limit=deadline - 30, patience=20
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")
        if sc < best_score:
            a_work = a_out

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(a_work, 15000, maxiter=5000, maxfun=150000)
        report(sc, a_opt, "final LBFGS b=15000")

    # Peak targeted reduction as last resort
    if time.monotonic() < deadline - 5:
        a_pt = peak_targeted_reduction(a_work, n_iters=30, K=20)
        sc = compute_score(a_pt)
        report(sc, a_pt, "peak_target")

        a_sf = spectral_flattening(a_work, n_iters=50, alpha=0.85)
        sc = compute_score(a_sf)
        report(sc, a_sf, "spectral")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
