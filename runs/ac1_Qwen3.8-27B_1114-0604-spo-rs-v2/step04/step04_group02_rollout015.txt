    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    c_max = c.max()

    g = 2.0 * conv_res

    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

Hybrid LP + LSE-L-BFGS optimizer for step function peak minimization.

Multi-phase with adaptive time allocation and multi-start exploration.

from scipy import sparse
import cvxpy as cp
import time


def fast_conv(a):
    """FFT-based linear self-convolution. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_val(x, beta):
    """LSE-smoothed score and gradient (for scipy minimize)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500))
    s = float(exp_scaled.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        L = c_max
    else:

        L = c_max + float(np.log(s)) / beta
    score = 2.0 * n * L / (S * S)

    M2 = 3 * n - 2

def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    y = a.copy()
    t_param = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak * 2.0 / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        theta = (t_param - 1.0) / t_new
        t_param = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = float(a.sum())
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def subgrad_momentum(a, n_iters=2000, f_lb=1.0, decay=0.9995):
    """Subgradient with heavy-ball momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.05

    for it in range(n_iters):
        c = fast_conv(a)
        Sa = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sa * Sa)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * a[k_star - idx]

        S2 = Sa * Sa
        S3 = S2 * Sa
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak * 2.0 / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (decay ** it)
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        v = 0.85 * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa2 = float(a.sum())
        if Sa2 < 1e-12:
            break
        a = a / Sa2

    return best_a


def generate_perturbation(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = int(rng.integers(0, 6))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 8.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(3, n // 100)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S_p = float(a_p.sum())
        if S_p > 1e-12:
            deficit = float(a.sum()) - S_p
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        # Sinusoidal mod at random frequency
        f1 = rng.uniform(0.5, 5.0)
        f2 = rng.uniform(5.0, 20.0)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        a_p = a * (1.0 + scale * np.sin(2 * np.pi * f1 * x + ph1) +
                   scale * 0.5 * np.sin(2 * np.pi * f2 * x + ph2))
    else:
        # Random subset rescale
        n_sel = n // 3
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = float(a_p.sum())
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses hybrid LP + LSE-L-BFGS + subgradient approach.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 25.0

    # Get parent sequence if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length from parent
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}, parent score: {compute_score(parent/parent.sum()):.10f}")
        report(compute_score(parent / parent.sum()), parent / parent.sum(), "parent")
    else:
        n_work = 8192
        print(f"No parent, using n={n_work}")

    # Working array
    if parent is not None and len(parent) == n_work:
        a_cur = parent / parent.sum()
    elif parent is not None:
        a_cur = np.interp(np.linspace(0, 1, n_work),
                          np.linspace(0, 1, len(parent)), parent)
        a_cur = np.clip(a_cur, 0.0, 1000.0)
        a_cur = a_cur / a_cur.sum()
    else:
        a_cur = np.ones(n_work) * 0.5

    # ================================================================
    # Phase 1: LP-based optimization (25% time)
    # ================================================================
    phase1_deadline = t_start + 0.25 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Optimization (n={n_work}, budget={phase1_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < phase1_deadline:
        a_lp = lp_optimize(a_cur, max_iters=25, time_limit_lp=40,
                           deadline=phase1_deadline)
        report(compute_score(a_lp), a_lp, "LP_main")

    # Try LP from an alternative initialization
    if time.monotonic() < phase1_deadline - 60:
        # Edge-concentrated initialization
        x = np.arange(n_work) / n_work
        a_edge = (0.1 + 0.9 * (2.0 * x - 1.0) ** 3)
        a_edge = np.clip(a_edge, 0.01, 10.0)
        a_edge = a_edge / a_edge.sum()
        a_lp2 = lp_optimize(a_edge, max_iters=10, time_limit_lp=30,
                            deadline=phase1_deadline)
        report(compute_score(a_lp2), a_lp2, "LP_edge")

    print(f"  After Phase 1: best = {best_score:.10f}")

    # ================================================================
    # Phase 2: Progressive L-BFGS refinement (45% time)
    # ================================================================
    phase2_deadline = t_start + 0.70 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Refinement (budget={phase2_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    beta_schedule = [20, 50, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline - 10:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(30000, max(2000, remaining * 30)))
        maxfun = int(min(100000, max(5000, remaining * 100)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = report(sc, a_opt, f"LBFGS_b{beta}")

        # If we got stuck (no improvement), try a slightly perturbed start
        if not improved and remaining > 60:
            a_pert = generate_perturbation(best_a, rng, scale=0.01)
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=maxiter // 2,
                                     maxfun=maxfun // 2)
            report(sc2, a_opt2, f"LBFGS_b{beta}_pert")

    print(f"  After Phase 2: best = {best_score:.10f}")

    # ================================================================
    # Phase 3: Multi-start iterated local search (20% time)
    # ================================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-start ILS (budget={phase3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline - 20:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.01 + 0.04 * rng.random()
        a_pert = generate_perturbation(best_a, rng, scale=scale)
        beta = int(500 * (1.0 + rng.random() * 5.0))
        maxiter = int(min(20000, max(2000, remaining * 25)))
        maxfun = int(min(60000, max(5000, remaining * 80)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS_{trial}_b{beta}")

        # If good result, try to refine further
        if sc < best_score + 0.001 and remaining > 60:
            sc2, a_opt2 = run_lbfgs(a_opt, int(beta * 3),
                                     maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"ILS_{trial}_refine")

    print(f"  After Phase 3: best = {best_score:.10f}")

    # ================================================================
    # Phase 4: Final polish (10% time)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    # Very high beta L-BFGS
    for beta in [200000, 500000]:
        if time.monotonic() > deadline - 20:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=50000)
        report(sc, a_opt, f"polish_b{beta}")

    # Subgradient polishing
    if time.monotonic() < deadline - 30:
        a_sg = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        report(compute_score(a_sg), a_sg, "subgrad_final")

    # One more LP pass if time allows
    if time.monotonic() < deadline - 60:
        a_lp_final = lp_optimize(best_a, max_iters=5, time_limit_lp=30,
                                  deadline=deadline - 20)
        if a_lp_final is not None:
            report(compute_score(a_lp_final), a_lp_final, "LP_final")

    # ================================================================
    # Try alternative lengths if time permits
    # ================================================================
    if time.monotonic() < deadline - 90 and parent is not None:
        print(f"\n--- Length Exploration ---")
        for n_try in [n_work // 2, int(n_work * 1.3)]:
            if time.monotonic() > deadline - 40:
                break
            if n_try < 256 or n_try > 30000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = float(a_interp.sum())
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt = run_lbfgs(a_interp, 1000, maxiter=10000, maxfun=40000)
            report(sc, a_opt, f"len_{n_try}")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
