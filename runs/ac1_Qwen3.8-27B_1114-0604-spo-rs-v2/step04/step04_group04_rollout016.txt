replaces FISTA approximation of the LP with exact HiGHS solutions,

and multi-scale refinement.

from scipy.sparse import lil_matrix, csr_matrix

    M_fft = 1 << (M - 1).bit_length()

    if S < 1e-12:

    return 2.0 * n * float(c.max()) / (S * S)


def build_lp_matrix(a):
    """Build sparse constraint matrix for LP: conv(a,x)[k] <= M for all k.
    A[k,j] = a[j] if max(0,k-n+1) <= j <= min(k,n-1), else 0.
    """

    m = 2 * n - 1
    # Column j has non-zeros at rows j through j+n-1, all with value a[j]
    total_nz = n * n  # each of n columns has n non-zeros (approximately)
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive
        length = k_end - k_start
        rows[idx:idx + length] = np.arange(k_start, k_end, dtype=np.int32)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[j]
        idx += length
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def solve_lp_for_x(a, M, n):

    Returns optimal x (unnormalized).

            return res.x
        else:

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a: 2 * (a flipped) * p convolved, extracted
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

    )

def spectral_flatten(a, n_iters=100, alpha=0.5):
    """
    Frequency-domain iterative flattening.
    Works by modifying the spectrum to reduce peak in time-domain convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2
        # Target: flatten the power spectrum (reduce peaks in spectrum)
        # This makes the time-domain convolution more concentrated (worse)
        # Instead, we want to make the time-domain convolution flatter
        # which means making the spectrum more peaked at DC

        # Alternative: directly reduce the peak of the convolution
        conv = self_conv(a)
        conv_mean = float((conv[:n]).mean())  # approximate mean of middle part
        conv_max = float(conv.max())

        if conv_max - conv_mean < 1e-14:
            break

        # Compute error signal
        k_star = int(np.argmax(conv))
        # The gradient of max(conv) w.r.t. a points in direction that increases the peak
        # We want to go opposite: reduce the peak
        # Subgradient: d(max_k conv[k])/da = 2 * (flipped a) restricted to valid indices
        # For peak at k_star: gradient[j] = 2*a[k_star - j] if valid

        # Use a soft threshold approach
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)

        # Compute correction in frequency domain
        err = excess.copy()
        err_pad = np.zeros(M_fft)
        err_pad[:len(err)] = err
        err_fft = np.fft.rfft(err_pad)

        # Newton step: dA/d|A|^2 relationship
        # |A|^2 -> |A|^2 - gamma * err_fft
        gamma = 0.5 + 0.3 * (it / max(1, n_iters))
        new_power = np.maximum(power.real - gamma * err_fft, 0.0)

        # Take square root to get new amplitude
        new_amp = np.sqrt(new_power + 1e-30)
        phase = np.exp(1j * np.angle(A))
        A_new = new_amp * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shave(a, n_rounds=50, frac=0.02):
    """Directly shave the peak by reducing elements that contribute most."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        # Elements contributing to c[k_star]: a[j] * a[k_star - j]
        # Gradient of c[k_star] w.r.t. a[j] = a[k_star - j] (for valid j)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[k_star - j]

        # Reduce top contributors
        top_k = max(1, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        total_red = 0.0
        for idx in top_idx:
            red = a[idx] * frac
            a[idx] = max(a[idx] - red, 0.0)
            total_red += red

        # Redistribute to bottom contributors
        if total_red > 0:
            bottom_idx = np.argsort(contrib)[:top_k]
            per = total_red / len(bottom_idx)
            for idx in bottom_idx:
                a[idx] += per

        S = float(a.sum())
        if S > 0:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=5000):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[k_star - np.arange(j_start, j_end + 1)]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Primary: LP-based alternating with exact HiGHS.
    Secondary: LSE-LBFGS, spectral flatten, peak shave, subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    if parent is not None:
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")
        n_main = len(parent)
    else:
        n_main = 6962

    # ============================================================
    # Phase 1: LP-based alternating (65% of time)
    # ============================================================
    phase1_time = 0.65 * (float(budget_s) - 25.0)
    phase1_deadline = t_start + phase1_time
    print(f"\n--- Phase 1: LP Alternating (time budget {phase1_time:.0f}s) ---")

    a_work = best_a if len(best_a) == n_main else np.interp(
        np.linspace(0, 1, n_main), np.linspace(0, 1, len(best_a)), best_a
    )
    a_work = np.maximum(a_work, 0.0)
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Run LP alternating in stages with decreasing aggressiveness
    stages = [
        (80, 0.6, 0.1),    # (max_iters, t_init, t_min)
        (80, 0.4, 0.05),
        (60, 0.25, 0.03),
        (40, 0.15, 0.02),
        (30, 0.1, 0.01),
    ]

    for stage_idx, (max_it, t_s, t_m) in enumerate(stages):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"  Stage {stage_idx}: max_iters={max_it}, t=[{t_s},{t_m}], remaining={remaining:.0f}s")
        a_out, sc_out = lp_alternating(
            a_work, max_iters=max_it, t_init=t_s, t_min=t_m,
            time_limit=phase1_deadline - 5, verbose=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP stage{stage_idx}")
        a_work = a_out.copy()

    # Spectral flatten on the LP result
    if time.monotonic() < phase1_deadline - 60:
        print("  Spectral flatten...")
        a_sf = spectral_flatten(a_work, n_iters=80, alpha=0.4)
        sc = compute_score(a_sf)
        report(sc, a_sf, "spectral")

    # Peak shave
    if time.monotonic() < phase1_deadline - 30:
        print("  Peak shave...")
        a_ps = peak_shave(best_a, n_rounds=50, frac=0.02)
        sc = compute_score(a_ps)
        report(sc, a_ps, "peak_shave")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS refinement (20% of time)
    # ============================================================
    phase2_time = 0.20 * (float(budget_s) - 25.0)
    phase2_deadline = t_start + phase1_time + phase2_time
    print(f"\n--- Phase 2: LSE-LBFGS (time budget {phase2_time:.0f}s) ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 400, 800]
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        maxiter = min(5000, int(remaining * 30))
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=maxiter, maxfun=maxiter * 40)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    # Multiple perturbation restarts with LBFGS
    n_restarts = 6
    for i in range(n_restarts):
        if time.monotonic() > phase2_deadline - 30:
            break
        # Different perturbation strategies
        if i % 3 == 0:
            scale = 0.02 + 0.02 * rng.random()
            a_pert = a_norm * (1.0 + scale * rng.standard_normal(len(a_norm)))
        elif i % 3 == 1:
            # Targeted: reduce peak contributors
            c = self_conv(a_norm)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - len(a_norm) + 1)
            j_end = min(k_star, len(a_norm) - 1)
            a_pert = a_norm.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, len(a_norm) // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.05
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            bottom_local = np.argsort(contrib)[:top_count]
            per = total_red / max(1, len(bottom_local))
            for li in bottom_local:
                a_pert[j_start + li] += per
        else:
            # Smooth perturbation
            freq = 1.0 + 5.0 * rng.random()
            x = np.arange(len(a_norm), dtype=np.float64) / len(a_norm)
            amp = 0.01 + 0.03 * rng.random()
            a_pert = a_norm * (1.0 + amp * np.cos(2 * np.pi * freq * x + rng.random() * 6.28))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        Sp = float(a_pert.sum())
        if Sp > 1e-12:
            a_pert = a_pert / Sp
            sc, a_opt = run_lbfgs(a_pert, 150, maxiter=3000, maxfun=100000)
            report(sc, a_opt, f"restart{i}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient + final LP polish (10% of time)
    # ============================================================
    phase3_time = 0.10 * (float(budget_s) - 25.0)
    phase3_deadline = t_start + phase1_time + phase2_time + phase3_time
    print(f"\n--- Phase 3: Polish (time budget {phase3_time:.0f}s) ---")

    # Subgradient
    if time.monotonic() < phase3_deadline - 60:
        print("  Subgradient polish...")
        a_sg = subgradient_polish(best_a, total_iters=5000)
        sc = compute_score(a_sg)
        report(sc, a_sg, "subgrad")

    # Final LP
    if time.monotonic() < phase3_deadline - 30:
        print("  Final LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 0:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating(
            a_final_in, max_iters=30, t_init=0.1, t_min=0.01,
            time_limit=phase3_deadline - 5, verbose=False
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final_LP")

    # Final high-beta LBFGS
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 1500, maxiter=4000, maxfun=150000)
        report(sc, a_opt, "final_LBFGS_hi_beta")

    # ============================================================
    # Phase 4: Try a different length if time permits (5%)
    # ============================================================
    remaining_time = deadline - time.monotonic()
    if remaining_time > 120 and len(best_a) != 8192:
        print(f"\n--- Phase 4: Try n=8192 ({remaining_time:.0f}s remaining) ---")
        n_try = 8192
        a_interp = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, len(best_a)), best_a)
        a_interp = np.maximum(a_interp, 0.0)
        S = float(a_interp.sum())
        if S > 0:
            a_interp = a_interp / S
        sc_interp = compute_score(a_interp)
        report(sc_interp, a_interp, f"interpolated n={n_try}")

        a_out, sc_out = lp_alternating(
            a_interp, max_iters=40, t_init=0.3, t_min=0.05,
            time_limit=deadline - 60, verbose=False
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n_try}")

        if sc < compute_score(best_a):
            a_norm = a_out.copy()
            S = float(a_norm.sum())
            if S > 0:
                a_norm = a_norm / S
            sc2, a_opt = run_lbfgs(a_norm, 200, maxiter=3000, maxfun=80000)
            report(sc2, a_opt, f"LBFGS n={n_try}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
