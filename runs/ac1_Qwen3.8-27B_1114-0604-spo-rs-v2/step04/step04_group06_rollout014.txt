Starts from the best known parent sequence.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def adam_lse_schedule(a_init, betas, lr, n_steps_per_beta, time_limit_s, t_start):
    """Adam with LSE, with increasing beta schedule."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S  # Normalize to sum = 1
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t_global = 0
    for beta in betas:
        if time.monotonic() - t_start > time_limit_s:
            break
        t0 = time.monotonic()
        for t in range(1, n_steps_per_beta + 1):
            t_global += 1
            if time.monotonic() - t_start > time_limit_s:
                break
            if time.monotonic() - t0 > 30:  # Check every 30s
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"  [Adam] beta={beta}, t={t}: score={sc:.8f}")
            
            lr_t = lr / (1.0 + 1e-4 * t)
            score, grad = lse_smootheval(a, beta)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [Adam] beta={beta} done: score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_smootheval, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def coord_descent_golden(a, top_k=50, n_sweeps=20, time_limit_s=None, t_start=None):

                break
            a_old = a[j]
            if a_old < 1e-10:

                    a[j] = m1
                    sc1 = compute_score(a)
                    a[j] = m2
                    sc2 = compute_score(a)
                    if sc1 < sc2:
                        hi = m2
                    else:
                        lo = m1
                a[j] = (lo + hi) / 2
                sc = compute_score(a)
            a[j] = a_old  # Reset

            # Actually let me redo this more carefully

                for x in [a.mean() * 0.1, a.mean() * 0.5, a.mean()]:
                    a[j] = x
                    sc = compute_score(a)
                    if sc < best_score:
                        best_score = sc
                    else:
                        a[j] = a_old

```python
def lse_smootheval(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    # Compute cross-correlation of p and a via FFT
    L_fft = 1 << (2 * n - 1 - 1).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    cross_corr = np.fft.irfft(P * np.conj(A), L_fft)[:n]
    
    # Gradient of L w.r.t. a
    grad_L = 2.0 * cross_corr
    
    # Full gradient of score = 2n * L / S^2
    score = 2.0 * n * L / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad
```
