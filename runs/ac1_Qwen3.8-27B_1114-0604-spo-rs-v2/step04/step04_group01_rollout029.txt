def normalize(a, target=1.0):
    """Clip to non-negative, normalize sum to target."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size))
    return a * (target / S)


def resample(a, n):
    """Linearly resample sequence to length n."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def corr_pair(a, w):
    """h_j = sum_k w_k * a_{k-j} for j=0..n-1 (middle slice of conv)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def lse_obj_grad(a, beta):
    """
    LSE-smooth objective: 2 * LSE_beta(d) where d = n*conv(a,a)/sum(a)^2.
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    value = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(value), grad


def l2_obj_grad(a):
    """
    L2 objective: sum(conv(a,a)^2). Returns (value, gradient).
    Gradient: grad[m] = 2 * sum_j c[j] * (a[j-m] + a[m-j])
    Computed via FFT cross-correlation.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    c = fast_self_conv(a)
    val = float(np.sum(c * c))

    N = next_power_of_two(3 * n)
    A = np.fft.rfft(a, N)
    C = np.fft.rfft(c, N)
    part1 = np.fft.irfft(C * np.conj(A), N)[:n]
    part2 = np.fft.irfft(A * C, N)[:n]
    grad = 2.0 * (part1 + part2)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return val, grad


def l2_normed_obj_grad(a):
    """
    L2 of normalized d = n*c/S^2. Returns (value, gradient).
    This is a smoother proxy for the max that's scale-invariant.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    val = float(np.sum(d * d))

    # grad of sum(d^2):
    # d[j] = n*c[j]/S^2
    # d(d[j]^2)/da[m] = 2*d[j]*n*(a[j-m]+a[m-j])/S^2 - 4*d[j]*n*c[j]/S^3
    N = next_power_of_two(3 * n)
    A = np.fft.rfft(a, N)
    D = np.fft.rfft(d, N)
    part1 = np.fft.irfft(D * np.conj(A), N)[:n]
    part2 = np.fft.irfft(A * D, N)[:n]
    second = float(np.dot(d, c))
    grad = (2.0 * float(n) / (S * S)) * (part1 + part2) - (4.0 * float(n) / (S ** 3)) * second
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return val, grad


def run_lbfgs(obj_func, a0, param, deadline, maxfun, maxiter=None):
    """Run L-BFGS-B with normalization after."""
    a = normalize(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score(a), a
    if maxfun is None:
        maxfun = 500
    if maxiter is None:
        maxiter = max(30, maxfun // 2)

    def wrapper(x):
        x = np.clip(x, 0.0, None)
        v, g = obj_func(x, param)
        return v, g

    try:
        res = minimize(
            wrapper, a, method="L-BFGS-B", jac=True,
            args=(param,), bounds=[(0.0, 10.0)] * n,
            options={"ftol": 1e-15, "gtol": 1e-12, "maxiter": maxiter, "maxfun": maxfun, "maxls": 50}
        )
    except Exception:
        return exact_score(a), a

    a_out = normalize(res.x) if res.x is not None else a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def active_weights(d, k):
    """Uniform weights on the top-k entries of d."""
    m = int(d.size)
    if m <= 0:
        return np.array([])
    if k <= 0 or k >= m:
        return np.ones(m) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m)
    w[idx] = 1.0
    return w / k


def subgrad_step(a, k, mult):
    """One multiplicative subgradient step on top-k peaks."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    d = float(n) * c / (S * S)
    dmax = float(np.max(d))
    if dmax <= 0 or not np.isfinite(dmax):
        return a
    w = active_weights(d, k)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    eta = mult / hmax
    q = a * np.exp(-eta * h)
    return normalize(q)


def subgrad_polish(a, max_iters=20, deadline=None):
    """Iterative subgradient polishing."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > deadline:
            break
        cur_a = None
        cur_sc = best_sc
        for k in (1, 2, 3, 5, 8):
            for mult in (0.05, 0.15, 0.30, 0.50, 0.80):
                if deadline is not None and time.monotonic() > deadline:
                    break
                q = subgrad_step(a, k, mult)
                sc = exact_score(q)
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_a = q.copy()
        if cur_a is None:
            break
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cur_a.copy()
        a = cur_a

    return float(best_sc), best_a


def freq_domain_init(n, rng):
    """
    Generate a sequence with approximately flat auto-correlation using
    frequency-domain construction.
    """
    N = next_power_of_two(2 * n)
    # Random phase, flat magnitude (except DC)
    phase = rng.uniform(-np.pi, np.pi, size=N // 2 + 1)
    mag = np.ones(N // 2 + 1)
    mag[0] = 0.0  # Remove DC
    # Add some smooth variation to magnitude
    freqs = np.arange(N // 2 + 1, dtype=np.float64)
    mag *= 1.0 + 0.3 * np.cos(2 * np.pi * freqs / (N / 4.0))
    spec = mag * np.exp(1j * phase)
    x = np.fft.irfft(spec, N)[:n]
    x = np.abs(x)  # Take absolute value to ensure non-negativity
    # Also try a different construction: random real signal, take positive part
    x2 = np.fft.irfft(spec, N)[:n]
    x2 = np.maximum(x2, 0.0)
    # Pick whichever has better score
    if x2.sum() > 1e-6:
        x2 = normalize(x2)
        sc2 = exact_score(x2)
    else:
        sc2 = np.inf
    x = normalize(x)
    sc1 = exact_score(x)
    if sc1 <= sc2:
        return x
    return x2


def gen_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # From parent
    if parent is not None:
        inits.append(normalize(resample(parent, n)))
        inits.append(normalize(resample(parent[::-1], n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramps
    inits.append(normalize(1.0 + 0.5 * (2 * x - 1)))
    inits.append(normalize(1.0 - 0.5 * (2 * x - 1)))

    # Cosine shapes
    for alpha in (0.3, 0.6, 0.9):
        inits.append(normalize(1.0 + alpha * np.cos(2 * np.pi * x)))
        inits.append(normalize(1.0 + alpha * np.cos(4 * np.pi * x)))

    # Frequency domain
    for _ in range(4):
        inits.append(freq_domain_init(n, rng))

    # Low-frequency smooth noise
    for _ in range(4):
        Nf = next_power_of_two(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** rng.uniform(1.0, 2.5))
        power[0] = 0.0
        ph = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * ph)
        spec[0] = 0.0
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - sig.mean()
        inits.append(normalize(1.0 + 0.5 * sig))

    # Random
    for _ in range(3):
        inits.append(normalize(rng.random(n) + 0.1))

    # Block structure
    for nb in (16, 32, 64, 128):
        if nb >= n:
            continue
        per = n // nb
        bv = 0.2 + 0.8 * rng.random(nb)
        blk = np.repeat(bv, per)
        if blk.size < n:
            blk = np.concatenate([blk, np.repeat(bv, n - blk.size)])
        inits.append(normalize(blk[:n]))

    return inits


def optimize_l2_phase(a, deadline):
    """Phase 1: minimize L2 norm of normalized convolution."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    # Estimate time per eval
    est_time = 0.025 * max(1.0, n / 4096.0)
    maxfun = max(50, int((deadline - time.monotonic() - 2.0) / max(est_time, 0.01)))
    maxfun = min(maxfun, 800)

    sc, a2 = run_lbfgs(l2_normed_obj_grad, a, 0.0, deadline, maxfun)
    if sc < best_sc:
        best_sc = sc
        best_a = a2.copy()
    a = normalize(a2)

    # Second L2 pass with more iterations
    if time.monotonic() < deadline - 5.0:
        maxfun2 = max(100, int((deadline - time.monotonic() - 3.0) / max(est_time, 0.01)))
        maxfun2 = min(maxfun2, 1000)
        sc, a2 = run_lbfgs(l2_normed_obj_grad, a, 0.0, deadline, maxfun2)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    return float(best_sc), best_a


def optimize_lse_phase(a, deadline):
    """Phase 2: LSE continuation with increasing beta."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    t0 = time.monotonic()
    total = max(10.0, deadline - t0 - 5.0)

    # Beta schedule: start smooth, get sharp
    stages = [
        (20, 0.06),
        (50, 0.08),
        (100, 0.10),
        (200, 0.10),
        (400, 0.12),
        (800, 0.12),
        (1500, 0.12),
        (3000, 0.12),
        (6000, 0.10),
        (12000, 0.08),
    ]

    est_time = 0.03 * max(1.0, n / 4096.0)
    cum = 0.0
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + cum * total
        sub_dead = min(sub_dead, deadline - 3.0)

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = max(50, int(rem * 0.8 / max(est_time, 0.01)))
            mf = min(mf, 2000)
            sc, a2 = run_lbfgs(lse_obj_grad, a, float(beta), deadline, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  LSE beta={:.0f} score={:.8f}".format(beta, best_sc))
            a = normalize(a2)

    return float(best_sc), best_a


def perturb(a, rng, strength=None):
    """Random perturbation of a normalized sequence."""
    p = normalize(a)
    n = len(p)
    if strength is None:
        strength = rng.uniform(0.15, 0.60)
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.1, 0.4)
        q = (1 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        q = p[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        # Pressure-based: reduce peaks
        S = float(p.sum())
        c = fast_self_conv(p)
        d = float(n) * c / (S * S)
        k = int(rng.integers(2, 8))
        w = active_weights(d, k)
        h = corr_pair(p, w)
        hmax = float(np.max(h))
        if hmax > 1e-12:
            q = p * np.exp(-strength * h / hmax)
        else:
            q = p * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 4:
        # Smooth noise in log domain
        noise = rng.standard_normal(n)
        k = min(31, max(5, n // 100))
        if k > 2:
            c_ = k // 2
            sig = max(1.0, c_ / 2.0)
            kern = np.exp(-((np.arange(k) - c_) ** 2) / (2 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp = np.log(p + 1e-30)
        logp += strength * 0.4 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Blend with frequency-domain init
        q2 = freq_domain_init(n, rng)
        t = rng.uniform(0.3, 0.7)
        q = (1 - t) * p + t * q2
    else:
        # Gaussian smoothing
        k = int(rng.integers(5, 25))
        if k % 2 == 0:
            k += 1
        c_ = k // 2
        sig = max(1.0, c_ / 2.5)
        kern = np.exp(-((np.arange(k) - c_) ** 2) / (2 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")

    return normalize(q)


def quick_optimize(a, deadline):
    """Quick optimization: LSE at moderate beta + subgrad."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    est_time = 0.03 * max(1.0, n / 4096.0)

    for beta in (100, 500, 2000):
        rem = float(deadline - time.monotonic())
        if rem < 8.0:
            break
        mf = max(40, int(rem * 0.6 / max(est_time, 0.01)))
        mf = min(mf, 600)
        sc, a2 = run_lbfgs(lse_obj_grad, a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize(a2)

    if time.monotonic() < deadline - 3.0:
        sc, a2 = subgrad_polish(a, max_iters=8, deadline=deadline - 2.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    return float(best_sc), best_a


def full_optimize(a, deadline, label):
    """Full optimization pipeline for one starting sequence."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    t0 = time.monotonic()
    total = max(20.0, deadline - t0 - 10.0)

    print("[{}] n={} initial={:.8f}".format(label, n, best_sc))

    # Phase 1: L2 warm-up (20% of time)
    l2_dead = t0 + 0.20 * total
    l2_dead = min(l2_dead, deadline - 10.0)
    if time.monotonic() < l2_dead - 3.0:
        sc, a2 = optimize_l2_phase(a, l2_dead)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] L2 phase: {:.8f}".format(label, best_sc))
        a = normalize(a2)

    # Phase 2: LSE continuation (55% of time)
    lse_dead = t0 + 0.75 * total
    lse_dead = min(lse_dead, deadline - 10.0)
    if time.monotonic() < lse_dead - 3.0:
        sc, a2 = optimize_lse_phase(a, lse_dead)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] LSE phase: {:.8f}".format(label, best_sc))
        a = normalize(a2)

    # Phase 3: Subgradient polish (10%)
    sg_dead = t0 + 0.85 * total
    sg_dead = min(sg_dead, deadline - 5.0)
    if time.monotonic() < sg_dead:
        sc, a2 = subgrad_polish(a, max_iters=25, deadline=sg_dead)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] subgrad: {:.8f}".format(label, best_sc))

    # Phase 4: Perturbation restarts (15%)
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - 30.0 - time.monotonic())
        n_restarts = min(5, max(2, int(rem / 60.0)))
        for ri in range(n_restarts):
            rl = float(deadline - 30.0 - time.monotonic())
            if rl < 30.0:
                break
            chunk = t0 + (0.85 + 0.15 * (ri + 1) / n_restarts) * total
            chunk = min(chunk, deadline - 25.0)
            p = perturb(best_a, np.random.default_rng(42 + ri * 1000), strength=0.3)
            sc, a2 = quick_optimize(p, chunk)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] restart{}: {:.8f}".format(label, ri, best_sc))

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L2 warm-up, LSE continuation, subgradient polishing, and stochastic restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                        parent.size if parent is not None else 0))

    # Choose lengths to try
    lengths = []
    if parent is not None:
        pl = int(parent.size)
        lengths.append(min(pl, 10000))
    lengths.extend([7000, 9000, 11000])

    # Deduplicate and filter
    seen = set()
    final_lengths = []
    for n in lengths:
        n = int(n)
        if 2048 <= n <= 12000 and n not in seen:
            final_lengths.append(n)
            seen.add(n)
    if not final_lengths:
        final_lengths = [7000]
    final_lengths = final_lengths[:4]

    print("Lengths: {}".format(final_lengths))

    best_seq = None
    best_score = np.inf

    # For each length, generate inits and pick best
    all_starts = []
    init_dead = t_start + 60.0

    for n in final_lengths:
        if time.monotonic() > init_dead:
            break
        inits = gen_inits(n, rng, parent)
        best_init_sc = np.inf
        best_init = None
        for arr in inits:
            p = normalize(arr)
            sc = exact_score(p)
            if np.isfinite(sc) and sc < best_init_sc:
                best_init_sc = sc
                best_init = p
        if best_init is not None:
            all_starts.append((best_init, float(best_init_sc)))
            print("  n={} best_init={:.6f}".format(n, best_init_sc))

    if not all_starts:
        n0 = final_lengths[0]
        all_starts.append((np.ones(n0) / n0, 2.0))

    # Sort by initial score
    all_starts.sort(key=lambda x: x[1])

    # Allocate time: give more time to first two, less to others
    n_starts = len(all_starts)
    if n_starts == 1:
        fracs = [0.90]
    elif n_starts == 2:
        fracs = [0.60, 0.30]
    elif n_starts == 3:
        fracs = [0.45, 0.30, 0.15]
    else:
        fracs = [0.35, 0.28, 0.20, 0.12]

    fracs = fracs[:n_starts]
    # Normalize fracs to sum to available time
    available = float(deadline - t_start - 60.0)
    total_frac = sum(fracs)
    fracs = [f / total_frac for f in fracs]

    cum = 0.0
    for i, (start, init_sc) in enumerate(all_starts):
        cum += fracs[i]
        sub_dead = t_start + cum * available
        sub_dead = min(sub_dead, deadline - 60.0)
        if time.monotonic() > sub_dead - 10.0:
            continue

        n = len(start)
        # For starts after the first, blend with current best
        if i > 0 and best_seq is not None:
            base = normalize(resample(best_seq, n))
            start = 0.5 * base + 0.5 * normalize(start)
            start = normalize(start)

        sc, seq = full_optimize(start, sub_dead, "L{}_{}".format(n, i))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("GLOBAL BEST: {:.8f} (n={})".format(best_score, n))

    if best_seq is None:
        best_seq = all_starts[0][0]
        best_score = all_starts[0][1]

    # Final polish phase
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - 30.0 - time.monotonic())
        # Subgradient
        if rem > 20.0:
            sc, seq = subgrad_polish(best_seq, max_iters=30, deadline=deadline - 25.0)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final subgrad: {:.8f}".format(best_score))

        # High-beta LSE
        if time.monotonic() < deadline - 15.0:
            rem2 = float(deadline - 15.0 - time.monotonic())
            if rem2 > 10.0:
                n = len(best_seq)
                est = 0.03 * max(1.0, n / 4096.0)
                mf = max(50, int(rem2 * 0.7 / max(est, 0.01)))
                sc, seq = run_lbfgs(lse_obj_grad, best_seq, 20000.0, deadline, mf)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print("Final LSE: {:.8f}".format(best_score))

        # One more subgrad
        if time.monotonic() < deadline - 5.0:
            sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 3.0)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Very final subgrad: {:.8f}".format(best_score))

    # Prepare output
    out = np.asarray(best_seq, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S
        out = np.clip(out, 0.0, 1000.0)

    out_list = [float(x) for x in out]
    if sum(out_list) < 0.01:
        out_list = [0.5] * 64

    # Verify
    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out_list), time.monotonic() - t_start))
    return out_list
