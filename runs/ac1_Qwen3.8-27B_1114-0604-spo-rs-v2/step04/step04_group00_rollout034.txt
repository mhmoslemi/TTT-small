import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

# LSE-based objective for L-BFGS
def lse_obj_and_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    # For numerical stability
    exp_shifted = np.exp(shifted - shifted.max())
    log_sum_exp = shifted.max() + np.log(exp_shifted.sum())
    L = c_max + log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient of L with respect to x
    # dL/dx_j = sum_k p_k * dc_k/dx_j = sum_k p_k * 2*x[k-j] * 1[0<=k-j<n]
    # = 2 * sum_{m=0}^{n-1} x[m] * p[j+m]  (where m = k-j, so k = j+m)
    # This is 2 * (cross-correlation of x and p at lag j) for j=0..n-1
    
    # Compute via FFT:
    # r[j] = sum_m x[m] * p[j+m] for j = 0..n-1
    # = conv(x[::-1], p)[n-1+j]  (verified above)
    N_fft = 1 << (3 * n - 2).bit_length()
    x_rev = x[::-1]
    Xr = np.fft.rfft(x_rev, N_fft)
    P = np.fft.rfft(p, N_fft)
    conv_val = np.fft.irfft(Xr * P, N_fft)
    dL_dx = 2.0 * conv_val[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_dx / S2 - 2.0 * L * x / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        j_end = min(k, n - 1)
        if j_end >= j_start:
            # For j in [j_start, j_end]: k-j is in [k-j_end, k-j_start] = [0 or k-n+1, n-1 or k]
            # We need x[k-j] for j in range, which is x reversed segment
            indices = k - np.arange(j_start, j_end + 1)
            sg[j_start:j_end+1] += 2.0 * a[indices]
    
    # Normalize subgradient
    sg_norm = np.linalg.norm(sg)
    if sg_norm < 1e-15:
        return a
    
    # Step
    a_new = a - (lr / sg_norm) * sg
    
    # Also add gradient of normalization penalty
    # We want to maintain sum(a) = 1, so add -2*(sum(a)-1)*ones
    a_new = a_new - 2.0 * (a_new.sum() - 1.0) / n * np.ones(n)
    
    a_new = np.clip(a_new, 0.0, 1000.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a
    a_new = a_new / S_new
    return a_new

def subgradient_descent(a, n_iters=5000, n_top=5, lr_init=0.05, decay=0.9999):
    """Subgradient descent with averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        a = subgradient_step(a, n_top=n_top, lr=lr)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        lr *= decay
    
    return best_a

def fienup_iter(a, n_iters=200, strength=0.5):
    """Fienup-style: flatten power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        thresh = P_mean * 1.5
        scale = np.ones_like(P)
        above = P > thresh
        scale[above] = (thresh / P[above]) ** 0.5
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def symmetry_break(a, rng, strength=0.05):

"""
Progressive L-BFGS + Peak-targeted Subgradient + Symmetry-breaking optimizer.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: very low starting beta, symmetry-breaking perturbations,
peak-equalization subgradient, and multi-length exploration.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(x, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    log_sum_exp = shifted_max + np.log(exp_shifted.sum())
    L = c_max + log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    p = exp_shifted / exp_shifted.sum()

    # dL/dx_j = 2 * sum_{m=0}^{n-1} x[m] * p[j+m]
    # = 2 * conv(x[::-1], p)[n-1 + j]
    N_fft = 1 << (3 * n - 2).bit_length()
    x_rev = x[::-1]
    Xr = np.fft.rfft(x_rev, N_fft)
    P = np.fft.rfft(p, N_fft)
    conv_val = np.fft.irfft(Xr * P, N_fft)
    dL_dx = 2.0 * conv_val[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_dx / S2 - 2.0 * L * x / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / a0.sum()

    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0

    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_descent(a, n_iters=3000, n_top=5, lr_init=0.04, decay=0.9998):
    """Subgradient descent targeting top-M peaks with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if n_top >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:n_top]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                indices = k - np.arange(j_start, j_end + 1)
                sg[j_start:j_end + 1] += 2.0 * a[indices]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Adaptive step: cap by distance to lower bound
        step = min(lr, max(0.0, (sc - sc_lb)) / (sg_norm * sg_norm + 1e-30))
        step = min(step, 0.06)

        a_new = a - step * sg
        # Renormalize
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new
        lr *= decay

    return best_a


def fienup_iter(a, n_iters=200, strength=0.5):
    """Fienup-style: flatten power spectrum to reduce peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

        P_mean = float(P.mean())
        thresh = P_mean * (1.3 + 0.7 * (1.0 - progress))

        S_new = float(a_new.sum())

        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)

            best_a = a.copy()

def peak_shave(a, n_peaks=3, n_shaves=200, strength=0.02):
    """Directly reduce elements contributing most to top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_shaves):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # For each top peak, find the element with max contribution and reduce it
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            # Contribution of element j to peak k is a[j] * a[k-j]
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            # Reduce the top contributor
            top_local = js[np.argmax(contribs)]
            a[top_local] *= (1.0 - strength)
            # Also reduce the mirrored element if different
            mirror = k - top_local
            if 0 <= mirror < n and mirror != top_local:
                a[mirror] *= (1.0 - strength * 0.5)

        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        print(f"  beta={beta:>5d}: score={sc:.8f}, time={elapsed:.1f}s")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    print(f"After Phase 1: best={best_score:.8f}")

    # ================================================================
    # Phase 2: Peak shaving + Subgradient (20%)
    # ================================================================
    phase2_end = t_start + 0.65 * (total - 30.0)
    print(f"\n=== Phase 2: Peak Shaving + Subgradient (t={phase2_end - t_start:.0f}s) ===")

    # Peak shaving
    if time.monotonic() < phase2_end:
        a_ps = peak_shave(best_a, n_peaks=3, n_shaves=300, strength=0.015)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PeakShave")
        if sc < best_score:
            # Polish with L-BFGS
            sc2, a2 = run_lbfgs(a_ps, 100, maxiter=3000, maxfun=40000)
            update(sc2, a2, "PeakShave+LBFGS")

    # Subgradient with various top_m
    for top_m in [3, 5, 8, 15]:
        if time.monotonic() > phase2_end:
            break
        a_sg = subgradient_descent(best_a, n_iters=3000, n_top=top_m, lr_init=0.03)
        sc = compute_score(a_sg)
        update(sc, a_sg, f"SG top={top_m}")
        if sc < best_score:
            sc2, a2 = run_lbfgs(a_sg, 200, maxiter=3000, maxfun=40000)
            update(sc2, a2, f"SG{top_m}+LBFGS")

    # Fienup
    if time.monotonic() < phase2_end:
        a_f = fienup_iter(best_a, n_iters=300, strength=0.6)
        sc = compute_score(a_f)
        update(sc, a_f, "Fienup")
        if time.monotonic() < phase2_end:
            sc2, a2 = run_lbfgs(a_f, 100, maxiter=3000, maxfun=40000)
            update(sc2, a2, "Fienup+LBFGS")

    print(f"After Phase 2: best={best_score:.8f}")

    # ================================================================
    # Phase 3: Symmetry breaking + re-optimize (20%)
    # ================================================================
    phase3_end = t_start + 0.85 * (total - 30.0)
    print(f"\n=== Phase 3: Symmetry Breaking (t={phase3_end - t_start:.0f}s) ===")

    for attempt in range(15):
        if time.monotonic() > phase3_end:
            break
        strength = 0.02 + 0.02 * attempt
        a_pert = symmetry_break(best_a, rng, strength=strength)

        # L-BFGS from perturbed
        sc, a_opt = run_lbfgs(a_pert, 30 + attempt * 20, maxiter=3000, maxfun=40000)
        updated = update(sc, a_opt, f"SymbRk {attempt} lbfgs")

        # Subgradient from perturbed
        if time.monotonic() < phase3_end:
            a_sg = subgradient_descent(a_pert, n_iters=1500, n_top=5, lr_init=0.03)
            sc_sg = compute_score(a_sg)
            update(sc_sg, a_sg, f"SymbRk {attempt} sg")
            if sc_sg < best_score:
                sc_pol, a_pol = run_lbfgs(a_sg, 100, maxiter=2000, maxfun=30000)
                update(sc_pol, a_pol, f"SymbRk {attempt} sg+lbfgs")

        # Fienup from perturbed
        if time.monotonic() < phase3_end:
            a_f = fienup_iter(a_pert, n_iters=150, strength=0.5)
            sc_f = compute_score(a_f)
            update(sc_f, a_f, f"SymbRk {attempt} fienup")

    print(f"After Phase 3: best={best_score:.8f}")

    # ================================================================
    # Phase 4: Multi-length + diverse inits (10%)
    # ================================================================
    phase4_end = t_start + 0.95 * (total - 30.0)
    print(f"\n=== Phase 4: Multi-length (t={phase4_end - t_start:.0f}s) ===")

    lengths_to_try = sorted(set([n0, n0 - 500, n0 + 500, n0 + 1000,
                                  n0 // 2 + 500, n0 * 3 // 2]))
    lengths_to_try = [l for l in lengths_to_try if 1000 <= l <= 20000 and l != n0]

    for n_try in lengths_to_try:
        if time.monotonic() > phase4_end:
            break
        # Interpolate
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.01)
        a_interp = a_interp / a_interp.sum()

        sc, a_opt = run_lbfgs(a_interp, 50, maxiter=3000, maxfun=40000)
        update(sc, a_opt, f"Len {n_try}")

    # Try a ramp initialization at original length
    if time.monotonic() < phase4_end:
        x = np.arange(n0, dtype=np.float64)
        a_ramp = 0.3 + 0.6 * x / n0 + 0.1 * np.cos(2 * np.pi * x / n0)
        a_ramp = np.maximum(a_ramp, 0.01)
        a_ramp = a_ramp / a_ramp.sum()
        sc, a_opt = run_lbfgs(a_ramp, 30, maxiter=5000, maxfun=60000)
        update(sc, a_opt, "Ramp init")

    # ================================================================
    # Phase 5: Final polish (5%)
    # ================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 10:
        for beta in [500, 2000, 8000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=60000)
            update(sc, a_opt, f"Final LBFGS b

    # Final subgradient pass
    if time.monotonic() < deadline - 5:
        a_sg = subgradient_descent(best_a, n_iters=2000, n_top=5, lr_init=0.02)
        sc = compute_score(a_sg)
        update(sc, a_sg, "Final SG")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
