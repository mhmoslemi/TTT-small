**Let me try a different angle: the problem is equivalent to a max-cut-like problem.**

Define the score as:
score = 2n * max_k sum_i a[i]*a[k-i] / (sum a)²

Let's define p_i = a_i / sum(a) (probability distribution). Then:
score = 2n * max_k sum_i p_i * p_{k-i}

The quantity sum_i p_i * p_{k-i} is the probability that two independent draws from p differ by k (if we think of p as a distribution on {0,...,n-1}).

So we want to minimize the maximum "difference probability" over all lags k. This is related to the "sphere packing" problem or "code design" in information theory.

For a uniform distribution, the max difference probability is 1/n (at lag 0: sum p_i² = 1/n). Wait no, at lag n-1 (the middle of the convolution), it's sum p_i * p_{n-1-i}. For uniform p, this is also 1/n.

So the uniform distribution gives score = 2n * 1/n = 2.

To do better, we need a distribution where the "difference distribution" is flatter. This is possible if the distribution has some structure.

**Key insight**: If we think of p as a distribution on a cyclic group Z_n, then the "circular autocorrelation" at lag k is sum p_i * p_{i+k} (mod n). The sum of all circular autocorrelations equals sum p_i² (the energy). To minimize the max circular autocorrelation, we'd want them all equal, giving max = (sum p_i²)/n >= 1/n².

But we're dealing with LINEAR convolution, not circular. The linear convolution has 2n-1 lags, and the relationship to circular is more complex.

For the LINEAR convolution, the peak is at the middle (lag n-1) for "symmetric" sequences. For asymmetric sequences, the peak could be elsewhere.

```python
"""
Multi-stage optimizer for step function peak minimization.
Implements: (1) LSE-smoothed L-BFGS, (2) adaptive multi-peak hinge loss
with projected subgradient, (3) coordinate descent polish, (4) high-beta
final refinement. Focuses on multi-peak simultaneous reduction via
hinge-loss subgradients computed via FFT cross-correlation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1].astype(np.float64)
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def hinge_grad(a, epsilon):
    """
    Compute gradient of sum_{k: c[k] > M} (c[k] - M) where M = max(c)*(1-epsilon).
    Returns (gradient, M, active_set_size).
    Gradient formula: df/da[j] = 2 * sum_{k in S} a[k-j]
    Computed via cross-correlation using FFT.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(np.max(c))
    M = c_max * (1.0 - epsilon)
    S_active = c > M
    n_active = int(S_active.sum())
    if n_active == 0:
        return np.zeros(n), M, 0

    # h[j] = sum_k s[k] * a[k-j] = (s * a_rev)[n-1+j]
    s = S_active.astype(np.float64)
    a_rev = a[::-1].astype(np.float64)
    M_conv = len(s) + n - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    S_fft = np.fft.rfft(s, M_fft)
    AR_fft = np.fft.rfft(a_rev, M_fft)
    h_full = np.fft.irfft(S_fft * AR_fft, M_fft)
    # h[j] = h_full[n-1+j] for j = 0,...,n-1
    h = h_full[n - 1:n - 1 + n]
    grad = 2.0 * h
    return grad, M, n_active


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def run_lbfgs_lse(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def hinge_subgradient_descent(a, n_iters=5000, epsilon=0.005, eta0=0.05, decay=0.9995):
    """
    Projected subgradient descent on the hinge loss.
    Maintains sum(a)=1, a>=0 via simplex projection.
    """
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    v = np.zeros(n)  # momentum

    for it in range(n_iters):
        grad, M, n_active = hinge_grad(a, epsilon)
        if n_active == 0:
            break

        # Also add a small regularization toward the peak position
        # to ensure we're actually reducing the max
        S_a = a.sum()
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        sc = 2.0 * n * c_max / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Add subgradient of the max term (for the actual peak)
        sg_max = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_max[idx] = a[k_star - idx]
        # Combine: hinge gradient + small weight on max subgradient
        combined_grad = grad + 0.5 * sg_max

        # Heavy ball momentum
        v = 0.85 * v - eta * combined_grad
        a_new = a + v
        a = project_simplex(a_new)

        # Decay step size
        eta *= decay

        if it % 500 == 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def coordinate_descent_polish(a, n_rounds=20, n_evals_per_coord=3):
    """
    Coordinate descent: for each coordinate, try a few perturbations
    and keep the best one.
    """
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_sc = compute_score(a)
    best_a = a.copy()

    for rnd in range(n_rounds):
        # Focus on top-50% of coordinates by value
        order = np.argsort(a)[::-1]
        n_focus = max(10, n // 2)
        focus = order[:n_focus]

        improved = False
        for j in focus:
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_local = a_old
            best_local_sc = best_sc
            for frac in [0.7, 0.85, 1.15, 1.3]:
                a[j] = a_old * frac
                S_new = a.sum()
                if S_new < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_new
                sc = compute_score(a_norm)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = a_old * frac
            a[j] = best_local
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            improved = True
        elif not improved and rnd > 3:
            break

    return best_a


def multi_peak_lbfgs(a, maxiter, maxfun, beta_start=200, beta_end=50000, n_phases=5):
    """
    Multi-phase LSE L-BFGS with increasing beta for progressively
    sharper peak approximation.
    """
    n = len(a)
    a_cur = np.clip(a.astype(np.float64).copy(), 0.0, None)
    S = a_cur.sum()
    if S < 1e-12:
        return a_cur
    a_cur = a_cur / S
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()

    for phase in range(n_phases):
        frac = (phase + 1) / n_phases
        beta = int(beta_start * (beta_end / beta_start) ** frac)
        sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter, maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a_cur = best_a

    return best_a


def nesterov_subgrad_peak(a, n_iters=3000, f_lb=1.4):
    """Nesterov-accelerated subgradient on the exact peak objective."""
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Subgradient
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, None)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def get_parent_sequence():
    """Retrieve the parent sequence from global scope."""
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                return np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-stage approach: LSE-LBFGS -> hinge subgradient -> CD polish -> final LSE.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 25.0
    deadline = t_start + effective_budget

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Get parent
    parent = get_parent_sequence()
    if parent is not None:
        print(f"Parent length: {len(parent)}, sum: {parent.sum():.2f}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        n_work = 4096
        parent_norm = np.ones(n_work) * 0.5
        report(compute_score(parent_norm), parent_norm, "uniform")

    n_work = len(parent_norm)
    a_cur = parent_norm.copy()
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: LSE L-BFGS with moderate beta (15% time)
    # ============================================================
    phase1_deadline = t_start + 0.15 * effective_budget
    print(f"\n--- Phase 1: LSE L-BFGS (moderate beta) ---")

    for beta in [50, 200, 500]:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter=2000, maxfun=10000)
        report(compute_score(a_opt), a_opt, f"LSE b={beta}")
        a_cur = best_a

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Hinge loss subgradient descent (25% time)
    # ============================================================
    phase2_deadline = t_start + 0.40 * effective_budget
    print(f"\n--- Phase 2: Hinge Subgradient ---")

    for epsilon in [0.002, 0.005, 0.01, 0.02]:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 20:
            break
        n_iters = int(min(5000, max(500, remaining * 10)))
        a_hinge = hinge_subgradient_descent(a_cur, n_iters=n_iters, epsilon=epsilon)
        sc_h = compute_score(a_hinge)
        report(sc_h, a_hinge, f"hinge eps={epsilon}")

        # Also try a smaller eta
        if time.monotonic() < phase2_deadline - 10:
            a_hinge2 = hinge_subgradient_descent(a_cur, n_iters=n_iters // 2,
                                                  epsilon=epsilon, eta0=0.02)
            sc_h2 = compute_score(a_hinge2)
            report(sc_h2, a_hinge2, f"hinge small eta eps={epsilon}")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Multi-phase LSE L-BFGS (30% time)
    # ============================================================
    phase3_deadline = t_start + 0.70 * effective_budget
    print(f"\n--- Phase 3: Multi-phase LSE L-BFGS ---")

    remaining3 = phase3_deadline - time.monotonic()
    if remaining3 > 30:
        maxiter3 = int(min(30000, max(3000, remaining3 * 30)))
        maxfun3 = int(min(100000, max(5000, remaining3 * 100)))
        a_multi = multi_peak_lbfgs(best_a, maxiter=maxiter3, maxfun=maxfun3,
                                    beta_start=100, beta_end=100000, n_phases=6)
        sc_multi = compute_score(a_multi)
        report(sc_multi, a_multi, "multi-phase LSE")
        a_cur = best_a

    # Additional LSE passes with high beta
    for beta in [10000, 50000]:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 30:
            break
        sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter=15000, maxfun=50000)
        report(sc, a_opt, f"LSE high b={beta}")
        a_cur = best_a

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Nesterov subgradient refinement (15% time)
    # ============================================================
    phase4_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 4: Nesterov Subgradient ---")

    if time.monotonic() < phase4_deadline:
        remaining4 = phase4_deadline - time.monotonic()
        n_iters4 = int(min(5000, max(1000, remaining4 * 8)))
        a_nest = nesterov_subgrad_peak(best_a, n_iters=n_iters4, f_lb=1.4)
        sc_nest = compute_score(a_nest)
        report(sc_nest, a_nest, "nesterov")

    # ============================================================
    # Phase 5: Coordinate descent + final polish (15% time)
    # ============================================================
    phase5_deadline = deadline
    print(f"\n--- Phase 5: CD Polish + Final ---")

    if time.monotonic() < phase5_deadline:
        a_cd = coordinate_descent_polish(best_a, n_rounds=10)
        sc_cd = compute_score(a_cd)
        report(sc_cd, a_cd, "CD polish")

    # Final high-precision LSE
    if time.monotonic() < deadline - 20:
        for beta in [100000, 200000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=20000, maxfun=60000)
            report(sc, a_opt, f"final LSE b={beta}")

    # One more hinge pass
    if time.monotonic() < deadline - 20:
        a_h = hinge_subgradient_descent(best_a, n_iters=2000, epsilon=0.003, eta0=0.01)
        sc_h = compute_score(a_h)
        report(sc_h, a_h, "final hinge")

    # Final small perturbations
    if time.monotonic() < deadline - 15:
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_work))
            a_pert = np.clip(a_pert, 0.0, None)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
            sc_p, a_opt_p = run_lbfgs_lse(a_pert, 5000, maxiter=5000, maxfun=20000)
            report(sc_p, a_opt_p, f"perturb #{trial}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
