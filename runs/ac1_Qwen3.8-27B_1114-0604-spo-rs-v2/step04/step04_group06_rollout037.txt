```python
    n = len(a)
    c = 2*n - 1  # number of convolution constraints

    # A is c x n, A[k,j] = a[k-j] if 0 <= k-j <= n-1 else 0
    
    # For efficiency, use scipy.sparse
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(c):

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

        g_sum = g.sum()
        if g_sum > 0:
            g = g / g_sum  # normalize to sum 1
            return g
    return None
```

```python
"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.

Algorithm:
1. Try multiple analytical initializations (sinusoidal, raised cosine, etc.)
2. Main optimization: LP-based iterative method (find direction via LP, line search)
3. Polish: LSE-smoothed gradient descent + coordinate descent
4. Return best sequence found
"""

import numpy as np
from scipy.optimize import linprog, minimize
import scipy.sparse as sp
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix C where (C @ g)[k] = (a*g)[k]."""
    n = len(a)
    c_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(c_len):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        ks = k - js
        rows.extend(k for _ in js)
        cols.extend(js.tolist())
        vals.extend(a[ks].tolist())
    C = sp.csr_matrix((vals, (rows, cols)), shape=(c_len, n))
    return C

def lp_direction(a, M, time_limit=5.0):
    """Solve LP to find optimal direction g."""
    n = len(a)
    C = build_conv_matrix(a)
    
    # max 1^T g s.t. C g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(C.shape[0], M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.success and res.x is not None:
        g = np.clip(res.x, 0, None)
        g_sum = g.sum()
        if g_sum > 1e-12:
            g = g / g_sum
            return g
    return None

def lse_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # gradient of L w.r.t. a: convolve reversed a with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    # Find all argmax points (within tolerance)
    tol = 1e-10
    max_idx = np.where(c >= M - tol)[0]
    score = 2.0 * n * M / (S * S)
    # Subgradient: average over argmax points
    sg = np.zeros(n)
    for k in max_idx:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        idx = k - np.arange(js, je + 1)
        sg[js:je + 1] += 2.0 * a[idx]
    sg /= len(max_idx)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad

def adam_optimize(a_init, lr=0.01, n_steps=20000, beta=200, time_limit=None, t_start=None):
    """Adam with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0/n
        s = 1.0
    a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        score, grad = lse_grad(a, beta)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        lr_t = lr / (1 + 1e-5 * t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a

def coord_descent(a, top_k=50, n_sweeps=5, time_limit=None, t_start=None):
    """Coordinate descent on top-K coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit) if (time_limit is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
```

```python
"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.

Algorithm:
1. Try diverse analytical initializations (sinusoidal, super-Gaussian, etc.)
2. LP-based iterative direction finding (solve LP for optimal direction each step)
3. LSE-smoothed Adam gradient optimization with multiple betas
4. Spectral relaxation steps (Fourier-domain phase/magnitude updates)
5. Coordinate descent polish
6. Second LP round for refinement
"""

import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
from scipy.interpolate import interp1d
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    """Build sparse matrix C where (C @ g)[k] = (a*g)[k] (linear conv)."""
    n = len(a)
    c_len = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(c_len):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        cnt = j_end - j_start + 1
        if cnt > 0:
            js = np.arange(j_start, j_end + 1)
            ks = k - js
            rows[idx:idx + cnt] = k
            cols[idx:idx + cnt] = js
            vals[idx:idx + cnt] = a[ks]
            idx += cnt
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    C = sp.csr_matrix((vals, (rows, cols)), shape=(c_len, n))
    return C


def solve_lp_direction(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g normalized to sum 1."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(C.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            g_sum = g.sum()
            if g_sum > 1e-12:
                g = g / g_sum
                return g
    except Exception:
        pass
    return None


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and (sub)gradient using all argmax points."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    tol = M * 1e-8 + 1e-15
    max_idx = np.where(c >= M - tol)[0]
    sg = np.zeros(n)
    for k in max_idx:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        idx = k - np.arange(js, je + 1)
        sg[js:je + 1] += 2.0 * a[idx]
    sg /= len(max_idx)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, lr, n_steps, beta, time_limit, t_start):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        score, grad = lse_score_grad(a, beta)
        if not np.isfinite(score):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_t = lr / (1.0 + 1e-5 * t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def spectral_relax(a, target_flat, n_iters=50, strength=1.0):
    """Fourier-based relaxation: adjust magnitude to push convolution towards target."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, None).copy()
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    target = np.zeros(N)
    target[:M] = target_flat
    target_hat = np.fft.rfft(target)
    for _ in range(n_iters):
        a_hat = np.fft.rfft(a, N)
        a_mag = np.abs(a_hat)
        a_phase = np.angle(a_hat)
        # We want |a_hat|^2 ~ target_hat, so |a_hat| ~ sqrt(|target_hat|)
        target_mag = np.sqrt(np.abs(target_hat) + 1e-30)
        current_mag_sq = a_mag ** 2
        # Blend: new_mag = current_mag * (target_mag / current_mag)^strength
        ratio = target_mag / (a_mag + 1e-15)
        ratio = np.clip(ratio, 0.01, 100.0)
        new_mag = a_mag * (ratio ** strength)
        # Keep DC component fixed (to preserve sum)
        new_mag[0] = a_mag[0]
        new_hat = new_mag * np.exp(1j * a_phase)
        # Enforce conjugate symmetry for real output
        if N % 2 == 0:
            new_hat[-1] = np.conj(new_hat[-1])
        a_new = np.fft.irfft(new_hat, N)[:n]
        a_new = np.clip(a_new, 0.0, None)
        s2 = a_new.sum()
        if s2 < 1e-12:
            break
        a = a_new / s2
    return a


def subgrad_method(a_init, n_steps, time_limit, t_start, lr_init=0.001, lr_decay=0.999):
    """Subgradient method with averaging."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s
    a_avg = a.copy()
    lr = lr_init
    best_score = compute_score(a)
    best_a = a.copy()
    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        score, grad = subgrad_score(a)
        if not np.isfinite(score):
            break
        a = a - lr * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        a_avg = a_avg + (a - a_avg) / t
        lr *= lr_decay
        if t % 5000 == 0:
            sc = compute_score(a_avg)
            if sc < best_score:
                best_score = sc
                best_a = a_avg.copy()
    sc = compute_score(a_avg)
    if sc < best_score:
        best_score = sc
        best_a = a_avg.copy()
    return best_score, best_a


def coord_descent(a_init, top_k, n_sweeps, time_limit, t_start):
    """Coordinate descent on top-K coordinates by gradient magnitude."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    s = a.sum()
    if s < 1e-12:
        return np.inf, a
    a = a / s
    best_score = compute_score(a)
    deadline = t_start + time_limit
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score, grad = subgrad_score(a)
        if not np.isfinite(score):
            break
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-15:
                a[j] = 0.0
                continue
            if grad[j] > 1e-10:
                mults = [0.3, 0.5, 0.7, 0.85, 0.95]
            elif grad[j] < -1e-10:
                mults = [0.0, 1.3, 1.7, 2.5]
            else:
                continue
            best_x = a_old
            best_sc = score
            for mult in mults:
                x = a_old * mult
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def lp_optimize(a_init, max_iters, time_per_lp, t_start, budget_remaining_func):
    """LP-based iterative optimization. Returns (best_score, best_a)."""
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    s = a.sum()
    if s < 1e-12:
        return np.inf, a
    a = a / s
    best_score = compute_score(a)
    best_a = a.copy()
    prev_score = best_score
    stuck_count = 0
    for it in range(max_iters):
        rem = budget_remaining_func()
        if rem < 15:
            break
        a_norm = a / a.sum()
        conv = fast_conv(a_norm)
        M = float(conv.max())
        g = solve_lp_direction(a_norm, M, time_limit=time_per_lp)
        if g is None:
            break
        similarity = float(np.sum(g * a_norm))
        if similarity > 0.9995:
            stuck_count += 1
            if stuck_count > 5:
                break
            # Add small perturbation
            rng_local = np.random.default_rng(it)
            perturb = rng_local.dirichlet(np.ones(len(a)) * 0.1)
            a = 0.95 * a_norm + 0.05 * perturb
            a = a / a.sum()
            continue
        stuck_count = 0
        # Line search
        best_t = 0.01
        best_sc = compute_score(a_norm)
        for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_trial = (1.0 - t) * a_norm + t * g
            a_trial = np.clip(a_trial, 0.0, None)
            s2 = a_trial.sum()
            if s2 < 1e-12:
                continue
            a_trial = a_trial / s2
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        a = (1.0 - best_t) * a_norm + best_t * g
        a = np.clip(a, 0.0, None)
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {it}: score={sc:.10f}, t={best_t:.3f}, sim={similarity:.6f}, elapsed={elapsed:.0f}s")
        # Convergence check
        if abs(prev_score - sc) < 1e-12 and it > 50:
            break
        prev_score = sc
    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    Uses LP-based direction finding, LSE-smoothed gradients, spectral relaxation,
    and coordinate descent in a multi-phase strategy.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = float(budget_s) - 25.0

    def time_remaining():
        return effective_budget - (time.monotonic() - t_start)

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                p = np.clip(p, 0.0, None)
                parent = p / p.sum()
    except Exception:
        pass

    best_score = [np.inf]
    best_a = [None]

    def update_best(sc, a, label):
        if sc < best_score[0] and np.isfinite(sc):
            best_score[0] = sc
            best_a[0] = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Phase 1: Analytical initializations
    print("=" * 60)
    print("Phase 1: Analytical initializations")
    print("=" * 60)

    n_main = 4096
    j = np.arange(n_main)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n_main) / n_main

    # Sinusoidal
    s = np.sin(np.pi * (j + 1) / (n_main + 1))
    inits['sin'] = s / s.sum()

    # Sin squared
    s = s ** 2
    inits['sin2'] = s / s.sum()

    # Sin cubed
    s = np.sin(np.pi * (j + 1) / (n_main + 1)) ** 3
    inits['sin3'] = s / s.sum()

    # Raised cosine (Bartlett-like)
    s = (1.0 + np.cos(2.0 * np.pi * j / (n_main - 1))) / 2.0
    s = np.clip(s, 1e-8, None)
    inits['raised_cos'] = s / s.sum()

    # Super-Gaussian p=4
    x = (j - n_main / 2.0) / (n_main / 4.0)
    s = np.exp(-x ** 4)
    inits['super_gauss4'] = s / s.sum()

    # Super-Gaussian p=6
    s = np.exp(-x ** 6)
    inits['super_gauss6'] = s / s.sum()

    # Laplace
    s = np.exp(-np.abs(j - n_main / 2.0) / (n_main / 5.0))
    inits['laplace'] = s / s.sum()

    # Quadratic parabola
    s = (j + 1.0) * (n_main - j)
    inits['para'] = s / s.sum()

    # Power 1.5
    s = ((j + 1.0) * (n_main - j)) ** 1.5
    inits['para15'] = s / s.sum()

    # Power 2.5
    s = ((j + 1.0) * (n_main - j)) ** 2.5
    inits['para25'] = s / s.sum()

    # Borel-type
    s = 1.0 / (1.0 + ((j - n_main / 2.0) / (n_main / 6.0)) ** 2)
    inits['borel'] = s / s.sum()

    # Borel p=4
    s = 1.0 / (1.0 + ((j - n_main / 2.0) / (n_main / 6.0)) ** 4)
    inits['borel4'] = s / s.sum()

    # Mixtures
    for alpha in [0.2, 0.4, 0.6]:
        uni = np.ones(n_main) / n_main
        gauss = np.exp(-((j - n_main / 2.0) / (n_main / 6.0)) ** 2)
        gauss = gauss / gauss.sum()
        s = alpha * uni + (1.0 - alpha) * gauss
        inits[f'mix_{alpha}'] = s / s.sum()

    # Two-lobe
    lobe1 = np.exp(-((j - n_main * 0.25) / (n_main / 8.0)) ** 2)
    lobe2 = np.exp(-((j - n_main * 0.75) / (n_main / 8.0)) ** 2)
    s = lobe1 + lobe2
    inits['two_lobe'] = s / s.sum()

    # Parent (resampled and original)
    if parent is not None:
        inits['parent_orig'] = parent.copy()
        if len(parent) != n_main:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_main)
            try:
                f_interp = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
                a_p = np.clip(f_interp(x_new), 0, None)
                if a_p.sum() > 0.01:
                    inits['parent_resamp'] = a_p / a_p.sum()
            except Exception:
                pass

    for name, a_init in inits.items():
        sc = compute_score(a_init)
        print(f"  {name:20s}: score={sc:.10f}")
        update_best(sc, a_init, name)

    print(f"  Phase 1 best: {best_score[0]:.10f}")

    # Phase 2: LP-based optimization
    print("\n" + "=" * 60)
    print("Phase 2: LP-based optimization")
    print("=" * 60)

    # Choose starting point for LP: best at n=4096 or parent
    a_lp = None
    if best_a[0] is not None and len(best_a[0]) == n_main:
        a_lp = best_a[0] / best_a[0].sum()
    elif parent is not None:
        # Resample parent to n_main for LP
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n_main)
        try:
            f_interp = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
            a_lp = np.clip(f_interp(x_new), 0, None)
            a_lp = a_lp / a_lp.sum()
        except Exception:
            a_lp = np.ones(n_main) / n_main
    else:
        a_lp = best_a[0].copy() if best_a[0] is not None else np.ones(n_main) / n_main
        a_lp = a_lp / a_lp.sum()

    lp_budget = min(1200.0, effective_budget * 0.5)
    sc, a_opt = lp_optimize(a_lp, max_iters=300, time_per_lp=4.0,
                            t_start=t_start, budget_remaining_func=time_remaining)
    update_best(sc, a_opt, "LP round 1")

    # Phase 3: LSE Adam optimization
    print("\n" + "=" * 60)
    print("Phase 3: LSE Adam optimization")
    print("=" * 60)

    if best_a[0] is not None:
        a_work = best_a[0] / best_a[0].sum()
        n_work = len(a_work)
        # For very large n, downsample for gradient methods
        if n_work > 8192:
            x_old = np.linspace(0, 1, n_work)
            x_new = np.linspace(0, 1, 4096)
            f_interp = interp1d(x_old, a_work, kind='cubic', fill_value='extrapolate')
            a_work = np.clip(f_interp(x_new), 0, None)
            a_work = a_work / a_work.sum()
            n_work = len(a_work)

        for beta, lr in [(50, 0.02), (200, 0.01), (1000, 0.005), (5000, 0.002)]:
            if time_remaining() < 30:
                break
            alloc = min(time_remaining() * 0.25, 250.0)
            sc, a_opt = adam_lse(a_work, lr=lr, n_steps=100000, beta=beta,
                                 time_limit=effective_budget - (time.monotonic() - t_start),
                                 t_start=t_start)
            update_best(sc, a_opt, f"Adam beta={beta}")
            a_work = a_opt  # chain the improvements

    # Phase 4: Spectral relaxation
    print("\n" + "=" * 60)
    print("Phase 4: Spectral relaxation")
    print("=" * 60)

    if best_a[0] is not None and time_remaining() > 30:
        a_spec = best_a[0] / best_a[0].sum()
        n_spec = len(a_spec)
        if n_spec <= 8192:
            # Target: flat convolution
            target_flat = np.full(2 * n_spec - 1, 1.0 / (2 * n_spec - 1))
            for strength in [0.3, 0.5, 0.7, 1.0]:
                if time_remaining() < 10:
                    break
                a_new = spectral_relax(a_spec, target_flat, n_iters=30, strength=strength)
                sc = compute_score(a_new)
                update_best(sc, a_new, f"spectral s={strength}")
                if sc < best_score[0] + 1e-10:
                    a_spec = a_new

    # Phase 5: Subgradient method
    print("\n" + "=" * 60)
    print("Phase 5: Subgradient method")
    print("=" * 60)

    if best_a[0] is not None and time_remaining() > 30:
        a_sub = best_a[0] / best_a[0].sum()
        if len(a_sub) <= 8192:
            alloc = min(time_remaining() * 0.3, 200.0)
            sc, a_opt = subgrad_method(a_sub, n_steps=100000,
                                       time_limit=effective_budget - (time.monotonic() - t_start),
                                       t_start=t_start, lr_init=0.005, lr_decay=0.9995)
            update_best(sc, a_opt, "Subgrad")

    # Phase 6: Coordinate descent
    print("\n" + "=" * 60)
    print("Phase 6: Coordinate descent polish")
    print("=" * 60)

    if best_a[0] is not None and time_remaining() > 20:
        alloc = min(time_remaining() * 0.4, 300.0)
        sc, a_opt = coord_descent(best_a[0], top_k=200, n_sweeps=20,
                                  time_limit=alloc, t_start=t_start)
        update_best(sc, a_opt, "Coord descent")

    # Phase 7: Second LP round (if time allows)
    print("\n" + "=" * 60)
    print("Phase 7: Second LP round")
    print("=" * 60)

    if best_a[0] is not None and time_remaining() > 60:
        a_lp2 = best_a[0] / best_a[0].sum()
        if len(a_lp2) <= 4096:
            sc, a_opt = lp_optimize(a_lp2, max_iters=200, time_per_lp=3.0,
                                    t_start=t_start, budget_remaining_func=time_remaining)
            update_best(sc, a_opt, "LP round 2")
        elif len(a_lp2) <= 8192:
            # Downsample for LP
            x_old = np.linspace(0, 1, len(a_lp2))
            x_new = np.linspace(0, 1, 4096)
            f_interp = interp1d(x_old, a_lp2, kind='cubic', fill_value='extrapolate')
            a_down = np.clip(f_interp(x_new), 0, None)
            a_down = a_down / a_down.sum()
            sc, a_opt = lp_optimize(a_down, max_iters=200, time_per_lp=3.0,
                                    t_start=t_start, budget_remaining_func=time_remaining)
            update_best(sc, a_opt, "LP round 2 (down)")

    # Phase 8: Final coordinate descent on the best
    print("\n" + "=" * 60)
    print("Phase 8: Final coordinate descent")
    print("=" * 60)

    if best_a[0] is not None and time_remaining() > 15:
        alloc = min(time_remaining() * 0.5, 150.0)
        sc, a_opt = coord_descent(best_a[0], top_k=300, n_sweeps=10,
                                  time_limit=alloc, t_start=t_start)
        update_best(sc, a_opt, "Final coord")

    # Final output
    if best_a[0] is None:
        out = [1.0] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a[0]]
        if sum(out) < 0.01:
            out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score[0]:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
