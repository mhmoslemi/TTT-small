"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- FFT-based exact scoring
- LSE-smoothed objective with adaptive temperature continuation
- L-BFGS-B local optimization
- Peak deflation via active-set subgradient

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Compute (a * w)_k for k = 0..n-1 where a has length n, w has length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.empty(0, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / a.size, dtype=np.float64)
    a = a * (target / S)
    return a


def exact_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = max(0.0, float(np.max(c)))
    return float(2.0 * n * cmax / (S * S))


def d_vals(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(2 * n - 1), S
    c = fast_self_conv(a)
    d = n * c / (S * S)
    return np.maximum(d, 0.0), S


def lse_objective_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    d, S = d_vals(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se

    L = dmax + np.log(se) / beta
    LDS = float(np.dot(w, d))

    h = cross_corr(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * n * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return float(score), grad


def make_obj(a0_len):
    def wrapper(x, beta):
        return lse_objective_grad(x, beta)
    return wrapper


def run_lbfgs(a0, beta, deadline, maxfun=500):
    a = normalize(a0)
    n = len(a)
    if n < 2:
        return exact_score(a), a

    rem = deadline - time.monotonic()
    if rem < 5:
        return exact_score(a), a

    est = 0.02 * max(0.5, n / 4096.0)
    cap = int((0.85 * rem) / max(est, 1e-4))
    maxfun = int(min(max(64, maxfun), max(64, cap)))
    maxiter = max(30, maxfun // 2)

    def fun_and_jac(x, b):
        v, g = lse_objective_grad(x, b)
        return v, g

    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-11, "maxiter": maxiter,
            "maxfun": maxfun, "maxls": 40}

    try:
        res = minimize(fun_and_jac, a, method="L-BFGS-B", jac=True,
                       args=(beta,), bounds=bounds, options=opts)
        a_out = normalize(res.x) if res.x is not None else a
    except Exception:
        a_out = a

    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, a)


def make_inits(n, rng, parent=None):
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None:
        p = normalize(resample(parent, n))
        inits.append(p)
        inits.append(p[::-1].copy())

    inits.append(np.ones(n) / n)

    for alpha in (0.2, 0.5, 0.8):
        inits.append(normalize(1.0 + alpha * (2 * x - 1)))
        inits.append(normalize(1.0 - alpha * (2 * x - 1)))

    # Cosine-based inits
    for r in range(8):
        v = np.ones(n)
        K = int(rng.integers(1, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 16)))
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            v += amp * np.cos(2 * np.pi * f * x + ph)
        inits.append(normalize(v))

    # Random smooth
    for _ in range(4):
        v = 0.3 + 0.7 * rng.random(n)
        # Smooth
        k = max(3, n // 200)
        if k % 2 == 0:
            k += 1
        if k < n:
            kern = np.ones(k) / k
            v = np.convolve(v, kern, mode="same")
        inits.append(normalize(v))

    # Block inits
    for nb in (4, 8, 16, 32):
        if nb > n:
            nb = max(2, n // 2)
        bv = 0.2 + 0.8 * rng.random(nb)
        per = max(1, n // nb)
        v = np.repeat(bv, per)[:n]
        if v.size < n:
            v = np.concatenate([v, np.repeat(bv, n - v.size)])
        inits.append(normalize(v))

    # Parent-perturbed
    if parent is not None:
        p = normalize(resample(parent, n))
        for strength in (0.1, 0.3, 0.5):
            q = p * np.exp(strength * 0.3 * rng.standard_normal(n))
            inits.append(normalize(q))

    return [normalize(v) for v in inits]


def deflation_step(a, k_active=3, n_trials=8, deadline=None):
    """Try multiplicative deflation on active lags."""
    a = normalize(a)
    n = len(a)
    best_sc = exact_score(a)
    best_a = a.copy()

    d, S = d_vals(a)
    if d.size == 0 or S < 1e-12:
        return best_sc, best_a

    m = d.size
    kk = min(k_active, m)
    idx = np.argpartition(d, m - kk)[m - kk:]
    w = np.zeros(m)
    w[idx] = 1.0 / kk

    h = cross_corr(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return best_sc, best_a

    base = 1.0 / hmax
    for mult in (0.01, 0.03, 0.08, 0.15, 0.3, 0.5, 0.8, 1.2):
        if deadline and time.monotonic() > deadline:
            break
        eta = mult * base
        q = a * np.exp(-eta * h)
        q = normalize(q)
        sc = exact_score(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()

    return best_sc, best_a


def coordinate_descent(a, n_sweeps=3, deadline=None):
    """Simple coordinate descent: for each j, try scaling a[j] up or down."""
    a = normalize(a)
    n = len(a)
    best_sc = exact_score(a)

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        for j in range(0, n, max(1, n // 200)):
            if deadline and time.monotonic() > deadline:
                break
            orig = a[j]
            if orig < 1e-15:
                continue
            for factor in (0.7, 0.85, 1.15, 1.3):
                a[j] = orig * factor
                sc = exact_score(a)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    improved = True
                    break
            else:
                a[j] = orig

        if not improved:
            break

    return best_sc, normalize(a)

def perturb(a, rng, strength=0.3):
    a = normalize(a)
    n = len(a)
    mode = int(rng.integers(0, 4))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * a + t * np.ones(n) / n
    elif mode == 2:
        h, S = d_vals(a)
        idx = np.argmax(h)
        w = np.zeros(len(h))
        w[max(0, idx-2):idx+3] = 1.0
        w /= w.sum()
        hcorr = cross_corr(a, w)
        hmax = max(1e-12, float(np.max(hcorr)))
        q = a * np.exp(-strength * 2 * hcorr / hmax)
    else:
        shift = int(rng.integers(1, max(2, n // 10)))
        q = np.roll(a, shift)
        q = q * np.exp(0.05 * rng.standard_normal(n))
    return normalize(q)
