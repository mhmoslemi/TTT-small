        P_new = np.maximum(P_new, 1e-20)

        A_phase = A / np.maximum(np.sqrt(P), 1e-30)
        A_new = A_amp * A_phase
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Non-negativity constraint
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a = a_new
        
        # Check improvement

def fienup_peak_shave(a, n_iters=300):
    """Fienup that specifically targets reducing the peak of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        # Identify high-power bins and reduce them
        # Use quantile-based threshold
        q = 0.7 + 0.2 * (it / n_iters)  # tighten over time
        thresh = np.quantile(P, q)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():

            scale[above] = np.sqrt(thresh / P[above])
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new

        if sc < best_sc:

            best_a = a.copy()

def peak_repair(a, n_iters=2000, top_k=5):
    """
    Directly reduce the top peaks of the convolution.
    For each peak at lag k*, the contributing elements are a[j] and a[k*-j].
    We reduce the elements with largest contribution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top peaks
        k_peaks = np.argsort(c)[::-1][:top_k]
        
        # For each peak, find the elements with largest product contribution
        reductions = np.zeros(n)
        for k in k_peaks:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max < j_min:
                continue
            # Contribution of a[j] to c[k] is a[j] * a[k-j]
            # Reducing a[j] by delta reduces c[k] by delta * a[k-j]
            # We want to reduce the largest contributors
            js = np.arange(j_min, j_max + 1)
            contribs = a[js] * a[k - js]
            # Reduce proportionally to contribution
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue
            # Reduce each a[j] proportional to its contribution
            for j in js:
                reductions[j] += a[j] * contribs[j - j_min] / total_contrib
        
        # Apply reduction
        S = float(a.sum())
        reduction_scale = 0.01 * (1.0 + 0.5 * (1 - it / n_iters))
        a -= reduction_scale * reductions
        
        # Compensate: add back to elements not contributing much to peaks
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 over non-negative sequences.
Uses: Fienup frequency-domain iteration, FISTA on LSE-smoothed objective,
L-BFGS with adaptive beta scheduling, peak-targeted coordinate descent,
and multi-restart with structured perturbations. Focuses on squeezing out
marginal improvements from the parent sequence.

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L = sum_k p_k * c_k with respect to x
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(PA * RA, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    """Top-level wrapper for scipy minimize."""
    return lse_obj_grad(x, beta)

                best_sc = sc
                best_x = x.copy()

        f_y, g_y = lse_obj_grad(z, beta)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - x)

        # Gradient step with projection
        x_new = y - g_y / L
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            x_new = x.copy()
        else:
            x_new = x_new / S_new

        # Backtracking line search
        f_new, _ = lse_obj_grad(x_new, beta)
        quadratic = 0.5 * L * np.sum((x_new - y) ** 2)
        if f_new > f_y + g_y @ (x_new - y) + quadratic:
            L *= 1.5
            if L > 1e10:
                break
            continue
        else:
            L = max(0.8 * L, 1.0)

        x = x_new
        z = x + ((t_k - 1.0) / t_new) * (x - z)
        z = np.maximum(z, 0.0)
        S_z = z.sum()
        if S_z > 1e-12:
            z = z / S_z
        t_k = t_new

    sc = compute_score(x)
    if sc < best_sc:
        best_x = x.copy()

    return best_x


def fienup_flatten(a, n_iters=300):
    """Fienup: flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Target: blend towards flat spectrum
        alpha = 0.2 + 0.6 * progress
        P_mean = float(P.mean())
        P_target = np.full_like(P, P_mean)
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        # Update amplitude, keep phase
        A_new = A * (np.sqrt(P_blend / P) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=300):
    """Fienup with quantile-based thresholding."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Tightening quantile threshold
        q = 0.75 + 0.2 * (it / n_iters)
        thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - it / n_iters))

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave(a, n_iters=3000, top_k=5, step_init=0.02, decay=0.999):
    """
    Subgradient method targeting top-K peaks with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-K peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_k]

        # Subgradient from top peaks
        sg = np.zeros(n)
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            k = int(k)
            # Weight peaks by how close they are to the max
            weights[i] = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * (weights.sum() / len(top_idx)) / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coord_descent_peaks(a, n_sweeps=10, n_top=100):
    """
    Coordinate descent: for each top-contributing element at the peak lag,
    try multiplicative perturbations.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution of each element to the peak
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        total = contribs.sum()
        if total < 1e-15:
            break

        # Focus on top contributors
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        cur_sc = compute_score(a)
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                continue
            candidates = [a_old_j * f for f in [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3]]
            best_val = a_old_j
            best_local = cur_sc
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def structured_perturb(a, rng, n_modes=8, scale=0.03):
    """Add low-frequency structured perturbations to escape local minima."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = rng.integers(1, 15)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / np.sqrt(m)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def noise_perturb(a, rng, scale=0.02):
    """Add Gaussian noise perturbation."""
    a_new = a * (1.0 + scale * rng.standard_normal(len(a)))
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def interpolated_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: Fienup -> L-BFGS (multi-beta) -> Peak shave ->
    Coordinate descent -> Multi-restart with perturbations.
    Returns the best non-negative sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        n0 = len(parent)
        sc_p = compute_score(parent)
        print(f"Parent: n={n0}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        n0 = 4096
        print("No parent available")

    print(f"Starting best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Fienup (10% of time) - quick global mixing
    # ==================================================================
    phase1_end = t_start + 0.10 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (until {phase1_end - t_start:.0f}s) ===")

    a_f1 = fienup_flatten(best_a, n_iters=200)
    sc = compute_score(a_f1)
    update(sc, a_f1, "FIENUP-flatten")

    a_f2 = fienup_quantile(best_a, n_iters=200)
    sc = compute_score(a_f2)
    update(sc, a_f2, "FIENUP-quantile")

    # ==================================================================
    # Phase 2: L-BFGS multi-beta sweep (40% of time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS sweep (until {phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000, 12000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 60:
            break
        # Scale maxiter based on remaining time
        maxiter = min(15000, int(remaining * 30))
        maxfun = min(300000, int(remaining * 600))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter})...")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

    # Second pass with different betas on the best found
    if time.monotonic() < phase2_end - 120:
        print("  L-BFGS second pass...")
        for beta in [500, 2000, 8000]:
            if time.monotonic() > phase2_end - 30:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
            update(sc, a_opt, f"LBFGS-2 b={beta}")

    # ==================================================================
    # Phase 3: Peak shaving + FISTA (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak shave + FISTA (until {phase3_end - t_start:.0f}s) ===")

    # Peak shaving
    if time.monotonic() < phase3_end:
        a_ps = peak_shave(best_a, n_iters=2000, top_k=5, step_init=0.015)
        sc = compute_score(a_ps)
        update(sc, a_ps, "PEAK-SHAVE")

        # Fisten after peak shave
        if time.monotonic() < phase3_end - 60:
            a_fi = run_fista(a_ps, 500, n_iters=2000)
            sc = compute_score(a_fi)
            update(sc, a_fi, "FISTA-after-shave")

    # FISTA directly
    if time.monotonic() < phase3_end - 60:
        for beta in [200, 1000, 3000]:
            if time.monotonic() > phase3_end - 30:
                break
            a_fi = run_fista(best_a, beta, n_iters=2000)
            sc = compute_score(a_fi)
            update(sc, a_fi, f"FISTA b={beta}")

    # ==================================================================
    # Phase 4: Coordinate descent + multi-restart (20% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Coord descent + restarts (until {phase4_end - t_start:.0f}s) ===")

    # Coordinate descent
    if time.monotonic() < phase4_end - 120:
        a_cd = coord_descent_peaks(best_a, n_sweeps=15, n_top=80)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCENT")

        # L-BFGS polish after coord descent
        if sc < best_score and time.monotonic() < phase4_end - 60:
            sc2, a2 = run_lbfgs(a_cd, 3000, maxiter=5000, maxfun=100000)
            update(sc2, a2, "CD+LBFGS")

    # Multi-restart with perturbations
    n_restarts = 8
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 30:
            break
        # Alternate between structured and noise perturbations
        if r % 2 == 0:
            a_pert = structured_perturb(best_a, rng, n_modes=8, scale=0.02 + 0.01 * r)
        else:
            a_pert = noise_perturb(best_a, rng, scale=0.01 + 0.005 * r)

        # Quick L-BFGS from perturbed start
        maxiter = min(5000, int((phase4_end - time.monotonic() - 30) * 20))
        if maxiter < 500:
            break
        sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 5: Final polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish ===")

    # Best beta L-BFGS run on the best found
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = int(remaining * 25)
        sc, a_final = run_lbfgs(best_a, 5000, maxiter=maxiter, maxfun=maxiter * 25)
        update(sc, a_final, "FINAL-LBFGS-5000")

    if time.monotonic() < deadline - 20:
        sc, a_final = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=100000)
        update(sc, a_final, "FINAL-LBFGS-20000")

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_cd = coord_descent_peaks(best_a, n_sweeps=5, n_top=120)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
