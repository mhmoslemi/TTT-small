"""
Convolution Flattening + LSE L-BFGS + Multi-Peak Subgradient Hybrid Optimizer.

Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. Direct convolution flattening via projected gradient (minimizes sum c(k)^2)
2. Adaptive LSE-smoothed L-BFGS with multiple beta values
3. Multi-peak subgradient with adaptive step size
4. LP-based direction search for escaping local minima
5. Multi-length exploration with interpolation
6. Bathtub-shape initialization motivated by theoretically optimal edge boosting

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def conv_flatten_grad(a, c):
    """
    Gradient of sum_k c(k)^2 with respect to a, where c = conv(a,a).
    grad(m) = 4 * sum_j a[j] * c[m+j]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, L)
    C = np.fft.rfft(c, L)
    result = np.fft.irfft(A * np.conj(C), L)
    return 4.0 * result[:n]


def peak_penalty_grad(a, c, threshold):
    """
    Gradient of sum_k max(0, c(k) - threshold)^2 with respect to a.
    Only penalizes peaks above threshold.
    """
    n = len(a)
    excess = np.maximum(c - threshold, 0.0)
    # grad(m) = 4 * sum_k excess(k) * a_{k-m}
    L = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, L)
    E = np.fft.rfft(excess, L)
    result = np.fft.irfft(A * np.conj(E), L)
    return 4.0 * result[:n]


def convolution_flattening(a0, n_iters=500, alpha=1.0, beta_peak=0.5, use_penalty=True):
    """
    Projected gradient method to flatten the self-convolution.
    Minimizes a combination of sum(c^2) and peak penalty.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    c = fast_conv(a)
    c_mean = 1.0 / (2 * n - 1)

    step = 0.1
    for it in range(n_iters):
        c = fast_conv(a)
        S = a.sum()
        if S < 1e-12:
            break
        a_norm = a / S
        sc = compute_score(a_norm)
        if sc < best_sc:
            best_sc = sc
            best_a = a_norm.copy()

        # Compute gradient
        if use_penalty:
            threshold = c.mean() + 0.5 * (c.max() - c.mean())
            grad = alpha * conv_flatten_grad(a_norm, c) + beta_peak * peak_penalty_grad(a_norm, c, threshold)
        else:
            grad = conv_flatten_grad(a_norm, c)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        c_max = c.max()
        c_min = c.min()
        spread = c_max - c_min + 1e-30
        step = min(step, 0.5 * spread / (gnorm * np.max(np.abs(grad)) + 1e-30))
        step = max(step, 1e-6)
        step = min(step, 0.15)

        # Gradient step
        a = a_norm - step * grad / gnorm
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        # Decay step
        step *= 0.998

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: 2n * (2 * (p * a_rev) / S^2 - 2*L / S^3)
    # where (p * a_rev)(m) = sum_k p(k) * a_{k-m}
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:n]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize."""
    x_clipped = np.maximum(x, 0.0)
    return lse_score_grad(x_clipped, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multi_peak_subgradient(a0, n_iters=5000, top_m=20, step_init=0.03, decay=0.9998):
    """
    Subgradient method using top-M peaks with normalized score gradient.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Weight peaks by how close they are to max
        peak_vals = c[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * 5.0)
        weights /= weights.sum()

        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weights[i] * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        step = max(step, 1e-7)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def lp_direction(a, max_time=60):
    """
    Solve the LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0.
    Returns the optimal b (normalized) or None if failed.
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    S = float(a.sum())

    # For efficiency, use a smaller version if n is too large
    if n > 4096:
        # Interpolate to smaller size
        n_small = 4096
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_small)
        a_small = np.interp(x_new, x_old, a)
        a_small = np.maximum(a_small, 0.01)
        a_small = a_small / a_small.sum()
    else:
        a_small = a
        n_small = n

    n_s = len(a_small)
    c_s = fast_conv(a_small)
    M_s = float(c_s.max())

    # Build constraint matrix: (a_small * b)(k) <= M_s for k=0,...,2n_s-2
    # (a_small * b)(k) = sum_j a_small[k-j] * b[j]
    # For each k, the coefficients on b[j] are a_small[k-j] for valid j
    n_constraints = 2 * n_s - 1
    # Build sparse-like matrix (dense for small n_s)
    Amat = np.zeros((n_constraints, n_s))
    for k in range(n_constraints):
        j_start = max(0, k - n_s + 1)
        j_end = min(k, n_s - 1)
        for j in range(j_start, j_end + 1):
            Amat[k, j] = a_small[k - j]

    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n_s)

    t_lp_start = time.monotonic()
    try:
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=Amat, b_ub=np.full(n_constraints, M_s),
                      bounds=[(0, None)] * n_s, method='highs',
                      options={'time_limit': max_time})
        if time.monotonic() - t_lp_start > max_time:
            return None
        if res.status != 0:
            return None
        b_opt = res.x
        if b_opt.sum() < 1e-12:
            return None
    except Exception:
        return None

    # Interpolate back to original size
    if n_s != n:
        x_old = np.linspace(0, 1, n_s)
        x_new = np.linspace(0, 1, n)
        b_full = np.interp(x_new, x_old, b_opt)
    else:
        b_full = b_opt

    b_full = np.maximum(b_full, 0.0)
    if b_full.sum() < 1e-12:
        return None
    return b_full / b_full.sum()


def bathtub_init(n, sharpness=2.0, rng=None):
    """
    Generate a bathtub-shaped initialization.
    Theoretically, the optimal sequence has larger values at edges.
    """
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    # Bathtub: high at edges, low in middle
    shape = 0.3 + 0.7 * (2.0 * x - 1.0) ** sharpness
    if rng is not None:
        shape = shape * (1.0 + 0.05 * rng.standard_normal(n))
    shape = np.maximum(shape, 0.01)
    return shape / shape.sum()


def cosine_init(n, rng=None):
    """Cosine-shaped initialization (high in middle)."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    shape = 0.5 + 0.5 * np.cos(np.pi * (2.0 * x - 1.0))
    shape = np.maximum(shape, 0.01)
    if rng is not None:
        shape = shape * (1.0 + 0.05 * rng.standard_normal(n))
        shape = np.maximum(shape, 0.01)
    return shape / shape.sum()


def power_init(n, p=1.5, rng=None):
    """
    Power-law initialization: a_j proportional to (j/n)^p or (1 - j/n)^p.
    Tries to make convolution flatter by boosting edges.
    """
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    shape = np.maximum(1.0 - x, 0.01) ** p
    if rng is not None:
        shape = shape * (1.0 + 0.05 * rng.standard_normal(n))
        shape = np.maximum(shape, 0.01)
    return shape / shape.sum()


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, parent)
            inits.append(normalize(p))

    inits.append(bathtub_init(n, 2.0, rng))
    inits.append(bathtub_init(n, 3.0, rng))
    inits.append(cosine_init(n, rng))
    inits.append(power_init(n, 1.0, rng))
    inits.append(power_init(n, 2.0, rng))

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = np.maximum(rng.standard_normal(n), 0.1)
    inits.append(r / r.sum())

    return inits


def cycle_flatten_lbfgs(a, n_flatten=200, n_cycles=3, betas=[100, 300, 800]):
    """
    Cycling: Convolution Flattening -> L-BFGS -> repeat.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for cyc in range(n_cycles):
        # Flattening step
        a = convolution_flattening(a, n_iters=n_flatten, alpha=1.0, beta_peak=0.5)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # L-BFGS step
        beta = betas[cyc % len(betas)]
        sc2, a2 = run_lbfgs(a, beta, maxiter=3000, maxfun=50000)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            a = a2

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    usable_time = total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = [4096, 6962, 8192, 10240]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: Convolution Flattening from diverse inits (35% of time)
    # ==================================================================
    phase1_end = t_start + 0.35 * usable_time
    print(f"\n=== Phase 1: Convolution Flattening (deadline {phase1_end-t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Processing n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            # Convolution flattening
            a_f = convolution_flattening(a0, n_iters=300, alpha=1.0, beta_peak=0.5, use_penalty=True)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"FLATTEN n={n} init={i}")

            # Second round with different parameters
            if time.monotonic() < phase1_end:
                a_f2 = convolution_flattening(a_f, n_iters=200, alpha=0.5, beta_peak=1.0, use_penalty=True)
                sc2 = compute_score(a_f2)
                update_best(sc2, a_f2, f"FLATTEN2 n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Cycling Flatten+L-BFGS (20% of time)
    # ==================================================================
    phase2_end = t_start + 0.55 * usable_time
    print(f"\n=== Phase 2: Cycling (deadline {phase2_end-t_start:.0f}s) ===")

    # Cycle from current best
    a_cyc = cycle_flatten_lbfgs(best_a, n_flatten=100, n_cycles=3, betas=[100, 300, 800])
    sc = compute_score(a_cyc)
    update_best(sc, a_cyc, "CYCLE-best")

    # Cycle from perturbed best
    if time.monotonic() < phase2_end:
        n_best = len(best_a)
        a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
        a_pert = normalize(a_pert)
        a_cyc2 = cycle_flatten_lbfgs(a_pert, n_flatten=80, n_cycles=2, betas=[200, 500])
        sc2 = compute_score(a_cyc2)
        update_best(sc2, a_cyc2, "CYCLE-pert")

        n_best = len(best_a)
        a_bat = bathtub_init(n_best, 2.5, rng)
        a_cyc3 = cycle_flatten_lbfgs(a_bat, n_flatten=80, n_cycles=2, betas=[100, 400])
        sc3 = compute_score(a_cyc3)
        update_best(sc3, a_cyc3, "CYCLE-bathtub")

    # Try different length

            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.01)
            a_alt = a_alt / a_alt.sum()
            a_cyc4 = cycle_flatten_lbfgs(a_alt, n_flatten=80, n_cycles=2, betas=[200, 600])
