"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Aggressive LSE-smoothed optimization with very high beta,
combined with targeted peak reduction and multi-length exploration.
Key improvements: much higher final beta, adaptive step sizing,
symmetry exploitation, and structured perturbations.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_s / exp_sum
    log_sum = np.log(exp_sum) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = (2.0 / beta) * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S**3))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_reduction(a, n_peaks=5, n_sweeps=3):
    """Targeted reduction of top convolution peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            n_top = max(3, len(contrib) // 10)
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            # Reduce top contributors, add to bottom
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a[gi] * 0.05
                a[gi] = max(a[gi] - red, 0.0)
                total_red += red
            
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - n_top)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a[j_start + li] += per
            
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def error_reduction(a, n_iters=100):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + 0.3 * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        A_new = A + 0.5 * d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=2000):
    """Subgradient method with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        grad = (4.0 * n) * (sg / (S * S) - M / (S**3))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        step = min((best_sc - sc_lb) / (gn * gn + 1e-30), 0.03)
        step = max(step, 1e-8)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

**Alternative: "Peak elimination" via linear programming**

For the final phase, instead of using LSE, we can directly solve:
`minimize M`
`subject to: (a*a)[k] <= M for all k`
`sum(a) = 1`
`a >= 0`

This is a QCQP. We can solve it using a successive linearization approach:
1. Fix `a` at current point
2. Linearize the quadratic constraints around `a`
3. Solve the resulting linear program
4. Update `a`
5. Repeat

The linearization of `(a*a)[k] <= M` around `a0` is:
`(a*a)[k] ≈ c[k] + 2 * sum_i (a[i] - a0[i]) * a0[k-i]` (for the relevant range)

So the linear constraint becomes:
`c[k] + 2 * sum_i (a[i] - a0[i]) * a0[k-i] <= M`

This is linear in `a` and `M`. We can solve this LP efficiently.

1. Compute the linear approximation of the objective at the current point

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Progressive LSE-smoothed L-BFGS with very high beta (up to 100000),
targeted peak reduction, structured diversification, and subgradient polish.
Key improvements: much higher final beta for accurate max approximation,
adaptive peak targeting, symmetry-aware perturbations, and extended runtime
utilization for deep local refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_s / exp_sum
    log_sum = np.log(exp_sum) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = (2.0 / beta) * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S**3))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_reduction(a, n_peaks=5, n_sweeps=3, strength=0.04):
    """Targeted reduction of top convolution peaks with redistribution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            n_top = max(3, len(contrib) // 10)
            top_local = np.argsort(contrib)[::-1][:n_top]

            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a[gi] * strength
                a[gi] = max(a[gi] - red, 0.0)
                total_red += red

            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - n_top)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a[j_start + li] += per

            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def error_reduction(a, n_iters=100):
    """Frequency-domain error reduction to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + 0.3 * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        A_new = A + 0.6 * d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Subgradient method with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        grad = (4.0 * n) * (sg / (S * S) - M / (S**3))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = min((best_sc - sc_lb) / (gn * gn + 1e-30), 0.02)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def tempered_lse_obj_grad(a, beta, temperature=0.5):
    """Tempered LSE: mixture of peaked and uniform distributions for better exploration."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-300:
        p_peaked = np.zeros(len(c))
        p_peaked[np.argmax(c)] = 1.0
    else:
        p_peaked = exp_s / exp_sum

    p_flat = np.ones(len(c)) / len(c)
    p = temperature * p_peaked + (1.0 - temperature) * p_flat

    log_sum = np.log(exp_sum) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = (2.0 / beta) * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S**3))
    return score, grad


def run_tempered_lbfgs(a0, beta, temperature, maxiter=5000, maxfun=200000):
    """L-BFGS with tempered LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = tempered_lse_obj_grad(x, beta, temperature)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent_peaks(a, n_sweeps=5, top_k=3, candidates_list=None):
    """Coordinate descent targeting top peaks."""
    if candidates_list is None:
        candidates_list = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15]
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:top_k]

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            n_top = max(5, len(contrib) // 20)
            top_local = np.argsort(contrib)[::-1][:n_top]
            top_global = j_start + top_local

            for j in top_global:
                a_old_j = a[j]
                best_val = a_old_j
                best_local_sc = best_sc
                for cv in candidates_list:
                    cv = max(0.0, min(1000.0, a_old_j * cv))
                    if abs(cv - a_old_j) < 1e-10:
                        continue
                    a[j] = cv
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = cv
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
