"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses multi-phase optimization: LSE-Adam with adaptive beta, mirror descent,
coordinate descent, and LP-based Frank-Wolfe steps.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    return irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient computation
    p_rev = p[::-1]
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = rfft(a_rev, M2_fft)
    P = rfft(p, M2_fft)
    conv_res = irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad

def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad

def adam_lse_adaptive(a_init, time_limit_s, t_start, beta_start=50, beta_end=10000, 
                      lr_start=0.01, lr_end=0.0001):
    """Adam with adaptive LSE beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    total_steps = 100000
    
    for t in range(1, total_steps + 1):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        # Adaptive beta and learning rate
        frac = (t - 1) / total_steps
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def mirror_descent(a_init, time_limit_s, t_start, eta=0.01, n_max_steps=200000):
    """Mirror descent (multiplicative updates) with exact subgradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 1e-15, 1e10)
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_max_steps + 1):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        score, grad = subgrad_score(a)
        
        # Adaptive learning rate
        eta_t = eta / (1.0 + 0.001 * t)
        
        # Multiplicative update (mirror step)
        a = a * np.exp(-eta_t * grad)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def coord_descent(a_init, time_limit_s, t_start, top_k=50, n_sweeps=100):
    """Coordinate descent with exact subgradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        score, grad = subgrad_score(a)
        
        # Find top-k coordinates by gradient magnitude
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() - t_start > time_limit_s:
                break
            
            a_old = a[j]
            
            if grad[j] > 1e-12:
                # Try to decrease
                candidates = a_old * np.array([0.1, 0.3, 0.5, 0.7, 0.85, 0.95])
            elif grad[j] < -1e-12:
                # Try to increase
                candidates = np.concatenate([[0.0], a_old * np.array([1.3, 1.5, 2.0, 3.0])])
            else:
                continue
            
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a

def lp_frank_wolfe(a_init, time_limit_s, t_start, max_iter=50, step_size=0.1):
    """LP-based Frank-Wolfe step."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Build the constraint matrix A where (A @ b)[k] = (a*b)[k]
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # A has shape (2n-1) x n
    
    # For efficiency, build as sparse
    # Each column j has non-zeros at rows k where 0 <= k-j < n, i.e., j <= k <= j+n-1
    # So column j has non-zeros at rows j, j+1, ..., j+n-1 with values a[j], a[j], ..., a[j]
    # Wait, A[k,j] = a[j] if 0 <= k-j < n
    # So for column j: A[k,j] = a[j] for k = j, j+1, ..., j+n-1
    
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T b s.t. A_mat @ b <= M, b >= 0
    S = a.sum()
    c = fast_conv(a)
    M = float(c.max())
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = A_mat
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    a_new = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        t0 = time.monotonic()
        
        # Rebuild A_mat with current a
        # (Actually, A changes when a changes, so we need to rebuild each time)
        rows = []
        cols = []
        vals = []
        for j in range(n):
            if a[j] > 1e-15:
                k_start = j
                k_end = min(j + n, 2*n - 1)
                for k in range(k_start, k_end):
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[j])
        
        A_mat = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            print(f"  LP iteration {it}: failed - {res.message}")
            break
        
        b_opt = res.x
        b_sum = b_opt.sum()
        if b_sum < 1e-12:
            break
        
        # Rescale to match current sum
        b_rescaled = b_opt * (S / b_sum)
        
        # Update: convex combination
        a_mix = (1 - step_size) * a + step_size * b_rescaled
        a_mix = np.clip(a_mix, 0.0, 1e10)
        a_mix = a_mix / a_mix.sum()
        
        sc = compute_score(a_mix)
        if sc < best_score:
            best_score = sc
            best_a = a_mix.copy()
            print(f"  LP iter {it}: score={sc:.10f} ({time.monotonic()-t0:.1f}s)")
        
        a = a_mix
        # Update M
        c = fast_conv(a)
        M = float(c.max())
    
    return best_score, best_a
