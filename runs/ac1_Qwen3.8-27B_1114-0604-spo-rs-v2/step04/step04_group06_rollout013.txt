"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses a multi-phase approach: LSE-smoothed gradient descent, LP-based direction
finding (Frank-Wolfe type), and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: L depends on c = conv(a,a)
    # dL/da = sum_k p[k] * dc_k/da
    # dc_k/da[i] = a[k-i] + a[i-k] (for valid indices) = 2*a[k-i] if k != 0, else a[0]
    # More carefully: c_k = sum_j a[j]*a[k-j], dc_k/da[i] = a[k-i] + a[i-k] (with bounds)
    # In vector form: dc_k/da = shift(a, k) + shift(a, -k) (circular, but we need linear)
    # For FFT: dc/da = 2 * (a * p) in the appropriate sense
    # Using correlation: dc_k/da[i] = sum over valid j of [a[j]*a[k-j], d/da[i]]
    # = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # This is the correlation of a with the "reversed" p... 
    # Actually: c = conv(a,a), dc/da[i] = conv(a, p)[i] + conv(a, p)[2n-2-i]... no
    # Let me just use: if c = conv(a, a), then dc_k/da_i = a_{k-i} + a_{i-k} (with zero padding)
    # So dL/da_i = sum_k p_k * (a_{k-i} + a_{i-k})
    # = sum_k p_k * a_{k-i} + sum_k p_k * a_{i-k}
    # = (p * a)[i] + (p_rev * a)[i] where p is reversed... 
    # Actually: sum_k p_k * a_{k-i} = (p_convolve_a)[i] where we reverse p first...
    # Let me think again. sum_k p_k * a_{k-i}: let j = k-i, so k = j+i
    # = sum_j p_{j+i} * a_j = (shifted_p * a) at position... 
    # This is the cross-correlation of p and a!
    # corr(p, a)[i] = sum_j p[j] * a[j+i] = sum_j p[j-i] * a[j]
    # Hmm, let me just use the FFT approach:
    # sum_k p_k * a_{k-i} for each i is the convolution of p_reversed with a, evaluated at specific positions.
    # Actually, the simplest way: 
    # d_L/da = 2 * ifft(conj(fft(a)) * fft(p_reversed))... this is getting complicated.
    # Let me use the direct formula with FFT:
    # c_k = sum_j a_j a_{k-j}, so dc_k/da_i = a_{k-i} + a_{i-k} (with zero outside [0,n-1])
    # dL/da_i = sum_k p_k (a_{k-i} + a_{i-k})
    # Let r = p reversed: r_k = p_{2n-2-k}
    # sum_k p_k a_{k-i} = (p * a_flip) where... 
    # OK let me just compute it directly using the correlation theorem:
    # sum_k p_k a_{k-i} is the cross-correlation of p and a at lag i (with appropriate definition)
    # In FFT: cross_correlation(p, a) = ifft(conj(fft(p)) * fft(a))
    # But we need linear correlation, not circular.
    
    # Simpler approach: use the fact that dL/da = 2 * (p correlated with a) in linear sense
    # corr_linear(p, a)[i] = sum_k p[k] * a[k-i] (with zero padding)
    # This equals the middle part of the convolution of p_reversed with a.
    
    p_rev = p[::-1]  # p reversed
    # conv(p_rev, a) has length (n + 2n-2 - 1) = 3n - 2
    # corr[i] = conv(p_rev, a)[n-1+i] for i = -(n-1), ..., n-1
    # Wait, let me be more careful.
    # conv(p_rev, a)[m] = sum_j p_rev[j] * a[m-j] = sum_j p[2n-2-j] * a[m-j]
    # Let k = 2n-2-j, so j = 2n-2-k:
    # = sum_k p[k] * a[m-2n+2+k]
    # We want sum_k p[k] * a[k-i] = sum_k p[k] * a[(k-i)]
    # So m - 2n + 2 + k = k - i => m = i + 2n - 2
    # So corr[i] = conv(p_rev, a)[i + 2n - 2]
    # And the valid range: 0 <= i + 2n-2 < 3n-2 => 2-2n <= i < n
    # For i in [0, n-1]: corr[i] = conv(p_rev, a)[i + 2n - 2]
    
    # Similarly, sum_k p_k a_{i-k} = corr of a and p at lag i = same thing by symmetry
    # Actually sum_k p_k a_{i-k} = sum_j p_{i-j} a_j (let j = i-k)
    # = conv(p, a)[i] (standard convolution)
    
    # So dL/da_i = conv(p, a)[i] + corr[i]
    # Where corr[i] = conv(p_rev, a)[i + 2n - 2]
    # But conv(p, a) = conv(a, p) by commutativity
    # And corr[i] = sum_k p[k] a[k-i] = (a * p_rev)[i + 2n - 2 - (n-1)] ... I'm confusing myself.
    
    # Let me just do it the simplest way:
    # dL/da = 2 * np.convolve(a, p, mode='same')  ... no that's not right either.
    
    # Actually the cleanest way: 
    # c = conv(a, a), so c is a vector of length 2n-1
    # dc_k/da_i = a[k-i] + a[i-k] (with out-of-bounds = 0)
    # This means: dc/da is a matrix where row k, col i has value a[k-i] + a[i-k]
    # dL/da_i = sum_k p[k] * (a[k-i] + a[i-k])
    # = sum_k p[k] a[k-i] + sum_k p[k] a[i-k]
    
    # First term: sum_k p[k] a[k-i]. Let's call this f[i].
    # f[i] = sum_k p[k] a[k-i] = (p flipped, convolved with a) evaluated at position...
    # Let pf = p[::-1] (length 2n-1)
    # conv(pf, a)[m] = sum_j pf[j] a[m-j] = sum_j p[2n-2-j] a[m-j]
    # Let k = 2n-2-j: = sum_k p[k] a[m-2n+2+k]
    # We want a[k-i], so m-2n+2+k = k-i => m = i+2n-2
    # So f[i] = conv(pf, a)[i + 2n - 2]
    
    # Second term: sum_k p[k] a[i-k] = (p convolved with a)[i] = conv(p, a)[i]
    # But conv(p, a) where p has length 2n-1 and a has length n: result has length 3n-2
    # conv(p, a)[i] for i = 0, ..., n-1
    
    # Hmm, this is getting complicated. Let me just use a simpler formula.
    # Note that c = conv(a,a) and p is a probability distribution on [0, 2n-2].
    # dL/da = sum_k p[k] * dc_k/da
    # dc_k/da is the gradient of the k-th convolution coefficient.
    # 
    # For k < n: c_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] a[k-i]
    # dc_k/da[i] = a[k-i] if 0 <= k-i < n and i is in range, plus a[i-k] if 0 <= i-k < n
    # 
    # The simplest efficient computation: use FFT to compute the "score function"
    # dL/da[i] = sum_k p[k] (a[k-i] + a[i-k]) for valid indices
    # = 2 * sum_k p[k] a[k-i] (by symmetry of the sum, since p is not necessarily symmetric)
    # 
    # Actually, let me just compute it as:
    # dL/da = 2 * ifft(fft(a) * fft(p_padded)) ... no, that's circular convolution.
    
    # You know what, let me just use a straightforward O(n * 2n) computation for the gradient,
    # or better, use the FFT trick:
    # 
    # The gradient of sum_k p[k] * conv(a,a)[k] w.r.t. a is:
    # = 2 * (a * p) in the "cross-correlation" sense
    # Specifically, g[i] = 2 * sum_k p[k] * a[k-i] (with a[j] = 0 for j outside [0,n-1])
    # Wait, that's only if p is symmetric. In general:
    # g[i] = sum_k p[k] * (a[k-i] + a[i-k])
    # 
    # Let's define: g1[i] = sum_k p[k] a[k-i] (zero-padded a)
    #            g2[i] = sum_k p[k] a[i-k] (zero-padded a)
    # g = g1 + g2
    # 
    # g1 is the cross-correlation of p and a (with p as the "template"):
    # g1[i] = <p, shift_i(a)> where shift_i(a)[k] = a[k-i]
    # In FFT: g1 = ifft(conj(fft(p_pad)) * fft(a_pad)) ... circular
    # For linear: use zero-padding to length >= (2n-1) + n - 1 = 3n-2
    # 
    # g2[i] = sum_k p[k] a[i-k] = (p * a)[i] = convolution of p and a at position i
    # In FFT: g2 = ifft(fft(p_pad) * fft(a_pad)) (linear conv with enough padding)
    
    # OK I'll just compute both using FFT with padding
    L_fft = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    # g2[i] = conv(p, a)[i] for i = 0..n-1
    conv_pa = np.fft.irfft(P * A, L_fft)
    g2 = conv_pa[:n]
    # g1[i] = sum_k p[k] a[k-i] = cross-correlation
    # = conj part: ifft(conj(P) * A) gives circular cross-correlation
    # For linear: reverse p, convolve with a
    P_rev = np.fft.rfft(p[::-1], L_fft)
    corr = np.fft.irfft(P_rev * A, L_fft)
    # corr[m] = sum_j p[2n-2-j] a[m-j] = sum_k p[k] a[m-2n+2+k]
    # We want a[k-i], so m-2n+2+k = k-i => m = i+2n-2
    g1 = corr[2*n-2 : 2*n-2+n]
    
    grad_L = g1 + g2
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def adam_optimize(a_init, lr=0.01, n_steps=50000, beta_lse=200,
                  time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        score, grad = lse_score_grad(a, beta_lse)
        if not np.isfinite(score):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_t = lr / (1.0 + 1e-5 * t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lp_direction(f, M, top_k=200, max_iter=50):
    """
    LP-based direction finding (from the state of the art).
    Find g that maximizes sum(g) subject to (f*g)[k] <= M for top-k lags, g >= 0.
    Returns g normalized to sum=1, or None if LP fails.
    """
    n = len(f)
    conv_f = fast_conv(f)
    # Get top_k lag indices
    top_indices = np.argsort(conv_f)[::-1][:top_k]
    
    # Build constraint matrix: A[i, j] = coefficient of g[j] in (f*g)[top_indices[i]]
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
    A = np.zeros((top_k, n))
    for ii, k in enumerate(top_indices):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        A[ii, js:je+1] = f[k - np.arange(js, je + 1)]
    
    # Solve: maximize sum(g) subject to A@g <= M, g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(top_k, M), 
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def frank_wolfe_lp(a, max_iters=200, time_limit_s=None, t_start=None, 
                   top_k=200, print_interval=10):
    """
    Frank-Wolfe type iteration using LP direction finding.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if t_start is not None and time_limit_s is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break
        
        S = a.sum()
        conv_a = fast_conv(a)
        M = conv_a.max()
        
        g = lp_direction(a, M, top_k=top_k)
        if g is None:
            print(f"  LP failed at iter {it}")
            break
        
        # Line search: find best t in [0, 1] for (1-t)*a + t*g
        best_t = 0.0
        best_sc = compute_score(a)
        
        # Try several t values
        ts = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]
        for t in ts:
            a_trial = (1 - t) * a + t * g
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.001, 0.005]:
                a_trial = (1 - t) * a + t * g
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.clip(a, 0, None)
        a = a / a.sum()
        
        if compute_score(a) < best_score:
            best_score = compute_score(a)
            best_a = a.copy()
        
        if it % print_interval == 0:
            print(f"    FW-LP iter {it}: score={compute_score(a):.8f}, t={best_t:.4f}")
    
    return best_score, best_a


def coord_descent(a, n_sweeps=5, top_k=50, time_limit_s=None, t_start=None):
    """Selective coordinate descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s and t_start) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score = compute_score(a)
        conv_a = fast_conv(a)
        S = a.sum()
        
        # Find coordinates to optimize: those contributing most to the peak
        k_star = conv_a.argmax()
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        # Also consider coordinates with large gradient
        grad_approx = np.zeros(n)
        for k in range(max(0, k_star-5), min(2*n-1, k_star+6)):
            js2 = max(0, k - n + 1)
            je2 = min(k, n - 1)
            for j in range(js2, je2 + 1):
                grad_approx[j] += conv_a[k] * a[k - j]
        
        # Select top-k coordinates by |contribution|
        contrib = np.abs(grad_approx)
        order = np.argsort(contrib)[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            # Try several values
            candidates = np.unique(np.concatenate([
                [0.0],
                [a_old * f for f in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]],
                [a.mean() * f for f in [0.5, 1.0, 1.5, 2.0]]
            ]))
            candidates = np.clip(candidates, 0, 1000)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
            score = best_sc
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a

Multi-phase approach: LSE-smoothed Adam, LP-based Frank-Wolfe direction finding,

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a. Assumes sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[i] = sum_k p[k] * (a[k-i] + a[i-k]) with zero padding
    L_fft = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, L_fft)
    A_fft = np.fft.rfft(a, L_fft)
    # g2[i] = conv(p, a)[i] = sum_k p[k] * a[i-k]
    conv_pa = np.fft.irfft(P * A_fft, L_fft)
    g2 = conv_pa[:n]
    # g1[i] = sum_k p[k] * a[k-i]: use reversed p convolution
    P_rev = np.fft.rfft(p[::-1], L_fft)
    corr = np.fft.irfft(P_rev * A_fft, L_fft)
    g1 = corr[2 * n - 2: 2 * n - 2 + n]

    grad_L = g1 + g2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def adam_optimize(a_init, lr=0.01, n_steps=50000, beta_lse=200,
                  time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient, normalizing to sum=1."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        score, grad = lse_score_grad(a, beta_lse)
        if not np.isfinite(score):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_t = lr / (1.0 + 1e-5 * t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lp_direction(f, M, top_k=200):
    """
    LP-based direction finding.
    Find g maximizing sum(g) subject to (f*g)[k] <= M for top_k lags, g >= 0.
    Returns g normalized to sum=1, or None.
    """
    n = len(f)
    conv_f = fast_conv(f)
    top_indices = np.argsort(conv_f)[::-1][:top_k]

    A = np.zeros((top_k, n))
    for ii, k in enumerate(top_indices):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            A[ii, js:je + 1] = f[k - np.arange(js, je + 1)]

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(top_k, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def frank_wolfe_lp(a, max_iters=100, time_limit_s=None, t_start=None,
                   top_k=150, print_interval=5):
    """Frank-Wolfe type iteration using LP direction finding."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if t_start is not None and time_limit_s is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break

        conv_a = fast_conv(a)
        M = float(conv_a.max())

        g = lp_direction(a, M, top_k=top_k)
        if g is None:
            break

        best_t = 0.0
        best_sc = compute_score(a)
        ts = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            for t in [0.005, 0.002]:
                a_trial = (1.0 - t) * a + t * g
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t

        if best_t == 0.0:
            break

        a = (1.0 - best_t) * a + best_t * g
        a = np.clip(a, 0, None)
        a = a / a.sum()

        sc_now = compute_score(a)
        if sc_now < best_score:
            best_score = sc_now
            best_a = a.copy()

        if it % print_interval == 0:
            elapsed = time.monotonic() - t_start if t_start else 0
            print(f"    FW-LP iter {it}: score={sc_now:.8f}, t={best_t:.4f}, t_elapsed={elapsed:.0f}s")

    return best_score, best_a


def coord_descent(a, n_sweeps=5, top_k=80, time_limit_s=None, t_start=None):
    """Selective coordinate descent focusing on peak-contributing coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s and t_start) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score = compute_score(a)
        conv_a = fast_conv(a)
        k_star = int(conv_a.argmax())

        # Compute approximate gradient at peak
        grad_approx = np.zeros(n)
        for k in range(max(0, k_star - 3), min(2 * n - 1, k_star + 4)):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                grad_approx[j] += conv_a[k] * a[k - j]

        order = np.argsort(np.abs(grad_approx))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            mean_val = a.mean()
            candidates = np.unique(np.concatenate([
                [0.0],
                [a_old * f for f in [0.2, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]],
                [mean_val * f for f in [0.5, 1.0, 1.5, 2.0]]
            ]))
            candidates = np.clip(candidates, 0, 1000)

            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
            score = best_sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: Adam LSE, LP Frank-Wolfe, coordinate descent, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = budget_s - 20.0

    # Get parent
    parent = None
    try:
        p = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
        if p.size >= 64 and np.all(np.isfinite(p)) and p.sum() > 0.01:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Evaluate parent
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={ps:.8f}")
        try_update(ps, parent, "parent")

    # Phase 1: Adam LSE
    print(f"\n{'='*50}")
    print("Phase 1: Adam LSE optimization")
    print(f"{'='*50}")

    if parent is not None and len(parent) >= 1024:
        n_opt = len(parent)
        a_init = parent / parent.sum()
    else:
        n_opt = 4096
        a_init = np.ones(n_opt) / n_opt

    time_alloc = min(effective_budget * 0.25, 400.0)
    sc, a_opt = adam_optimize(a_init, lr=0.01, n_steps=100000, beta_lse=200,
                              time_limit_s=time_alloc, t_start=t_start)
    try_update(sc, a_opt, f"Adam n={n_opt}")

    # Phase 1b: Higher beta Adam
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        time_alloc2 = min(remaining * 0.15, 250.0)
        sc2, a_opt2 = adam_optimize(a_opt.copy(), lr=0.005, n_steps=100000,
                                    beta_lse=500, time_limit_s=time_alloc2, t_start=t_start)
        try_update(sc2, a_opt2, "Adam beta=500")
        a_opt = a_opt2 if sc2 < sc else a_opt

    # Phase 2: LP-based Frank-Wolfe
    print(f"\n{'='*50}")
    print("Phase 2: LP-based Frank-Wolfe")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60 and best_a is not None:
        time_alloc3 = min(remaining * 0.35, 700.0)
        a_fw = best_a.copy()
        sc3, a_fw_out = frank_wolfe_lp(a_fw, max_iters=80,
                                       time_limit_s=time_alloc3, t_start=t_start,
                                       top_k=150, print_interval=5)
        try_update(sc3, a_fw_out, "FW-LP")

    # Phase 3: Coordinate descent
    print(f"\n{'='*50}")
    print("Phase 3: Coordinate descent")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30 and best_a is not None:
        time_alloc4 = min(remaining * 0.25, 400.0)
        a_cd = best_a.copy()
        sc4, a_cd_out = coord_descent(a_cd, n_sweeps=5, top_k=100,
                                      time_limit_s=time_alloc4, t_start=t_start)
        try_update(sc4, a_cd_out, "Coord descent")

    # Phase 4: Try different starting points/lengths
    print(f"\n{'='*50}")
    print("Phase 4: Multi-length exploration")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    lengths_to_try = [2048, 8192, 3000, 5000]
    for n_try in lengths_to_try:
        if time.monotonic() - t_start > effective_budget - 30:
            break
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 30:
            break
        # Different initializations
        inits = [
            np.ones(n_try) / n_try,
            np.cos(np.pi * np.arange(n_try) / (n_try - 1)) ** 2 + 0.01,
        ]
        for ii, a_init2 in enumerate(inits):
            a_init2 = np.clip(a_init2, 0, None)
            a_init2 = a_init2 / a_init2.sum()
            time_alloc5 = min(remaining * 0.2, 100.0)
            sc5, a_test = adam_optimize(a_init2, lr=0.01, n_steps=50000,
                                        beta_lse=200, time_limit_s=time_alloc5,
                                        t_start=t_start)
            label = f"Adam n={n_try} init={ii}"
            try_update(sc5, a_test, label)
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 30:
                break

    # Phase 5: Final LP refinement on best
    print(f"\n{'='*50}")
    print("Phase 5: Final LP refinement")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30 and best_a is not None:
        time_alloc6 = min(remaining * 0.5, 400.0)
        sc6, a_final = frank_wolfe_lp(best_a.copy(), max_iters=50,
                                       time_limit_s=time_alloc6, t_start=t_start,
                                       top_k=200, print_interval=5)
        try_update(sc6, a_final, "Final FW-LP")

    # Phase 6: Final coordinate descent
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 20 and best_a is not None:
        time_alloc7 = min(remaining * 0.5, 200.0)
        sc7, a_final2 = coord_descent(best_a.copy(), n_sweeps=3, top_k=150,
                                      time_limit_s=time_alloc7, t_start=t_start)
        try_update(sc7, a_final2, "Final coord descent")

    # Finalize
    if best_a is None:
        out = [1.0] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
