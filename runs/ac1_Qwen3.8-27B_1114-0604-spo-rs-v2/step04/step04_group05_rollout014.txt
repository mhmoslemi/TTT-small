```python
"""
LP-guided optimization for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses iterated LP direction finding with adaptive constraint relaxation, golden-section
line search, continuation L-BFGS polish, and coordinate descent refinement.
Key innovation: The LP finds the optimal "shape direction" to move towards while
preserving the autocorrelation bound, giving much better directions than subgradient.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    return irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_peak_info(a):
    """Return convolution, max value, argmax, and sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M_val = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M_val, S


def build_conv_matrix_a(a):
    """Build sparse matrix A where (a*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(a)
    N = 2 * n - 1
    k_vals = np.arange(N)
    j_starts = np.maximum(0, k_vals - n + 1)
    j_ends = np.minimum(k_vals, n - 1)
    lengths = j_ends - j_starts + 1
    total_nnz = int(lengths.sum())

    row_idx = np.repeat(np.arange(N), lengths)
    cum_before = np.zeros(N, dtype=np.int64)
    cum_before[1:] = np.cumsum(lengths)[:-1]
    cum_before_rep = cum_before[row_idx]
    positions = np.arange(total_nnz) - cum_before_rep
    j_starts_rep = j_starts(row_idx) if callable(j_starts) else j_starts[row_idx]
    cols = j_starts_rep + positions
    k_rep = row_idx
    indices = k_rep - cols
    vals = a[indices]
    return csr_matrix((vals, (row_idx, cols)), shape=(N, n))


def build_conv_matrix_restricted(a, active_indices):
    """Build sparse matrix for only a subset of convolution lags."""
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return csr_matrix((K, n))

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            L = je - js + 1
            rows.extend([idx] * L)
            cols.extend(range(js, je + 1))
            vals.extend(a[k - je:k - js + 1])

    if not rows:
        return csr_matrix((K, n))
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_scaled_sum = exp_scaled.sum()
    log_sum = np.log(exp_scaled_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = rfft(rev_a, M2_fft)
    P = rfft(p, M2_fft)
    conv_res = irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lp_direction_finding(a, c, M_val, threshold=0.3, max_constraints=500):
    """
    Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M_val for active k.
    Returns the optimal g or None if LP fails.
    """
    n = len(a)
    # Find active constraints (lags where conv is near peak)
    active_thresh = threshold * M_val
    active = np.where(c >= active_thresh)[0]
    if len(active) > max_constraints:
        # Keep only the top max_constraints
        active_vals = c[active]
        top_idx = np.argsort(active_vals)[::-1][:max_constraints]
        active = active[top_idx]

    if len(active) == 0:
        active = np.array([int(c.argmax())])

    # Build restricted constraint matrix
    A = build_conv_matrix_restricted(a, active)
    b = np.full(len(active), M_val)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            return res.x
        elif res.status == 9:  # timeout
            return res.x if res.x is not None and res.x.sum() > 0 else None
    except Exception:
        pass
    return None


def golden_section_search(a, g, t_lo=0.0, t_hi=1.5, n_evals=30):
    """
    Golden-section search for optimal t in h(t) = (1-t)*a + t*g.
    Both a and g should have the same sum.
    """
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)

    h1 = (1.0 - c1) * a + c1 * g
    f1 = compute_score(h1)
    h2 = (1.0 - c2) * a + c2 * g
    f2 = compute_score(h2)

    for _ in range(n_evals - 2):
        if f1 < f2:
            t_hi = c2
            c2 = c1
            f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            h1 = (1.0 - c1) * a + c1 * g
            f1 = compute_score(h1)
        else:
            t_lo = c1
            c1 = c2
            f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            h2 = (1.0 - c2) * a + c2 * g
            f2 = compute_score(h2)

    if f1 < f2:
        best_t = c1
        best_f = f1
    else:
        best_t = c2
        best_f = f2

    return best_t, best_f


def run_lp_iteration(a, c, M_val, k_star, threshold=0.3, max_constraints=500):
    """
    One LP-based iteration: find direction, line search, return new a and score.
    """
    n = len(a)
    S = float(a.sum())

    g = lp_direction_finding(a, c, M_val, threshold, max_constraints)
    if g is None or g.sum() <= S * (1 + 1e-6):
        return a, compute_score(a), 0.0

    # Rescale g to have same sum as a
    g = g * (S / g.sum())
    g = np.clip(g, 0.0, 1000.0)

    # Line search
    best_t, best_f = golden_section_search(a, g, t_lo=0.0, t_hi=1.5, n_evals=25)

    if best_f < compute_score(a) - 1e-10:
        new_a = (1.0 - best_t) * a + best_t * g
        new_a = np.clip(new_a, 0.0, 1000.0)
        return new_a, best_f, best_t
    else:
        return a, compute_score(a), 0.0


def lbfgs_refine(a0, beta, maxiter=3000, maxfun=50000, time_limit=60.0):
    """L-BFGS-B with LSE smoothing and time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    t_start = time.monotonic()

    def obj_and_grad(x):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError
        return lse_score_grad(x, beta)

    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-14, 'gtol': 1e-10,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    return a_out


def coord_descent(a, n_sweeps=3, top_k=40, time_budget=30.0):
    """Coordinate descent with golden-section 1D search on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + time_budget
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Subgradient at peak
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 3.0, a.max() * 0.5)
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)

            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, a


def perturbation_spectral(a, rng, strength=0.15):
    """DFT-domain perturbation."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new * (a.sum() / a_new.sum())


def perturbation_antipalin(a, rng, strength=0.12):
    """Anti-palindromic perturbation to reduce center peak."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / center
    freq = float(rng.uniform(1, 8))
    tilt += 0.3 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_block(a, rng, n_blocks=3):
    """Block swap perturbation."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        block_len = int(rng.integers(max(1, n // 30), max(2, n // 8)))
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if abs(i - j) > block_len:
            tmp = a_new[i:i + block_len].copy()
            a_new[i:i + block_len] = a_new[j:j + block_len]
            a_new[j:j + block_len] = tmp
    return a_new


def lp_optimize_loop(a, max_iters, time_budget, t_start, threshold=0.3,
                     max_constraints=400, verbose=True):
    """
    Main LP-based optimization loop with adaptive threshold.
    """
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break

        c, k_star, M_val, S = conv_peak_info(a)
        current_score = 2.0 * n * M_val / (S * S)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: score={current_score:.8f}, "
                  f"best={best_score:.8f}, stag={stagnation}, "
                  f"t={time.monotonic()-t_start:.0f}s")

        if stagnation > 20:
            # Tighten threshold to focus on the worst lags
            threshold = max(0.1, threshold * 0.7)
            stagnation = 0

        new_a, new_score, step = run_lp_iteration(
            a, c, M_val, k_star, threshold, max_constraints)

        if step == 0.0:
            # Try with tighter threshold
            new_a, new_score, step = run_lp_iteration(
                a, c, M_val, k_star, 0.7, 200)
            if step == 0.0:
                # LP is stuck, break
                break

        a = new_a
        a = np.clip(a, 0.0, 1000.0)

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_score, best_a


def continuation_lbfgs(a, betas, time_budget, t_start):
    """Continuation L-BFGS with increasing beta."""
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.4, 80.0)
        if alloc < 5:
            break
        a = lbfgs_refine(a, beta, maxiter=3000, maxfun=30000, time_limit=alloc)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose_lbfgs:
                print(f"    L-BFGS beta={beta}: {sc:.8f}")
        a = best_a if sc < best_score else a

    return best_score, best_a


def refine_full(a_init, time_budget, t_start, use_lp=True):
    """Multi-stage refinement: LP -> L-BFGS -> Coord descent."""
    a = a_init.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False

    # Stage 1: LP-based optimization (50% of budget if available)
    if use_lp and time.monotonic() < deadline - 10:
        lp_time = min((deadline - time.monotonic()) * 0.5, time_budget * 0.5)
        sc, a_opt = lp_optimize_loop(a, max_iters=100, time_budget=lp_time,
                                     t_start=t_start, threshold=0.3,
                                     max_constraints=400, verbose=False)
        try_update(sc, a_opt)
        a = best_a

    # Stage 2: L-BFGS continuation (30%)
    if time.monotonic() < deadline - 10:
        lbfgs_time = min((deadline - time.monotonic()) * 0.4, time_budget * 0.3)
        betas = [100, 500, 2000, 8000]
        sc, a_opt = continuation_lbfgs(a, betas, lbfgs_time, t_start)
        try_update(sc, a_opt)
        a = best_a

    # Stage 3: Coordinate descent (20%)
    if time.monotonic() < deadline - 5:
        coord_time = min(deadline - time.monotonic() - 3, time_budget * 0.2)
        if coord_time > 5:
            sc, a_opt = coord_descent(a, n_sweeps=5, top_k=40,
                                      time_budget=coord_time)
            try_update(sc, a_opt)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP-guided optimization + L-BFGS continuation + Coord descent.
    Multiple restarts with diverse perturbations.
    """
    global verbose_lbfgs
    verbose_lbfgs = False

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Intensive LP optimization from parent (40% budget)
    # ==================================================================
    p1_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization from parent (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    # Quick L-BFGS warmup to get to a good starting point
    quick_time = min(15.0, p1_budget * 0.1)
    a_warm = lbfgs_refine(parent, 200, maxiter=2000, maxfun=20000,
                          time_limit=quick_time)
    sc_warm = compute_score(a_warm)
    try_update(sc_warm, a_warm, "Quick LBFGS warmup")

    # Main LP loop
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 10:
        sc, a_opt = lp_optimize_loop(best_a, max_iters=200, time_budget=remaining,
                                     t_start=t_start, threshold=0.3,
                                     max_constraints=500, verbose=True)
        try_update(sc, a_opt, "LP main loop")

    # ==================================================================
    # Phase 2: L-BFGS continuation polish (20% budget)
    # ==================================================================
    p2_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS continuation (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < t_start + effective_budget - 30:
        lbfgs_time = min(p2_budget, t_start + effective_budget - 30 - time.monotonic())
        betas = [200, 800, 3000, 10000, 30000]
        sc, a_opt = continuation_lbfgs(best_a, betas, lbfgs_time, t_start)
        try_update(sc, a_opt, "L-BFGS continuation")

    # ==================================================================
    # Phase 3: Diverse restarts (25% budget)
    # ==================================================================
    p3_budget = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse restarts (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_start = time.monotonic()
    p3_deadline = p3_start + p3_budget
    strategies = [
        "spectral_01",
        "spectral_03",
        "anti_palin",
        "anti_palin_strong",
        "block_swap",
        "spectral_anti",
        "spectral_weak",
        "block_1",
    ]

    for ri in range(len(strategies)):
        if time.monotonic() > p3_deadline - 15:
            break
        t_remain = p3_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.85, p3_budget / len(strategies) * 1.3)
        if time_alloc < 20:
            break

        name = strategies[ri]
        if name == "spectral_01":
            a_pert = perturbation_spectral(best_a, rng, strength=0.1)
        elif name == "spectral_03":
            a_pert = perturbation_spectral(best_a, rng, strength=0.3)
        elif name == "spectral_anti":
            a_pert = perturbation_antipalin(
                perturbation_spectral(best_a, rng, 0.1), rng, 0.1)
        elif name == "anti_palin":
            a_pert = perturbation_antipalin(best_a, rng, strength=0.1)
        elif name == "anti_palin_strong":
            a_pert = perturbation_antipalin(best_a, rng, strength=0.2)
        elif name == "block_swap":
            a_pert = perturbation_block(best_a, rng, n_blocks=3)
        elif name == "spectral_weak":
            a_pert = perturbation_spectral(best_a, rng, strength=0.05)
        elif name == "block_1":
            a_pert = perturbation_block(best_a, rng, n_blocks=1)
        else:
            a_pert = perturbation_spectral(best_a, rng, strength=0.15)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        sc, a_opt = refine_full(a_pert, time_alloc, t_start, use_lp=True)
        try_update(sc, a_opt, f"restart#{ri}: {name}")

    # ==================================================================
    # Phase 4: Try different lengths via resampling (5% budget)
    # ==================================================================
    p4_budget = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 4: Length variation (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    for factor in [0.8, 1.2]:
        if time.monotonic() > t_start + effective_budget - 20:
            break
        tl = int(n_best * factor)
        if tl < 512 or tl > 20000:
            continue
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)
        time_left = t_start + effective_budget - 20 - time.monotonic()
        if time_left < 15:
            continue
        sc, a_opt = refine_full(a_res, min(time_left, 30.0), t_start, use_lp=True)
        try_update(sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 5: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    if time.monotonic() < final_deadline - 5:
        # High-beta L-BFGS
        alloc = min(final_deadline - time.monotonic() - 3, 40.0)
        if alloc > 5:
            a_polish = lbfgs_refine(best_a, 20000, maxiter=5000,
                                    maxfun=50000, time_limit=alloc)
            sc = compute_score(a_polish)
            try_update(sc, a_polish, "Final LBFGS high-beta")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = coord_descent(best_a, n_sweeps=8, top_k=50,
                                  time_budget=min(remaining - 2, 30.0))
        try_update(sc, a_opt, "Final CoordDesc")

    # Final LP polish if time allows
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = lp_optimize_loop(best_a, max_iters=50,
                                     time_budget=min(remaining - 2, 30.0),
                                     t_start=t_start, threshold=0.5,
                                     max_constraints=300, verbose=False)
        try_update(sc, a_opt, "Final LP polish")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
