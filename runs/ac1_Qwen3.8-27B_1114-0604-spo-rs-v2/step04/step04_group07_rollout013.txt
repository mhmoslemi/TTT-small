import numpy as np
from scipy.optimize import minimize
import time

def fast_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def lse_grad_f(a, beta):
    """Compute LSE-smoothed score and its gradient w.r.t. a."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    LSE = c_max + np.log(exp_scaled.sum()) / beta
    
    score = 2.0 * n * LSE / (S * S)
    
    # Compute gradient
    # grad[j] = (2n/S^2) * dLSE/da[j] - (4n*LSE/S^3)
    # dLSE/da[j] = (f_rev * p)[n-1+j] + (p * f)[j]
    
    M_fft = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    
    # (p * f)[j] for j = 0..n-1
    pf = np.fft.irfft(P * F, M_fft)[:n]
    
    # (f_rev * p)[n-1+j] for j = 0..n-1
    f_rev = a[::-1].copy()
    FR = np.fft.rfft(f_rev, M_fft)
    frp = np.fft.irfft(FR * P, M_fft)[n-1:2*n-1]
    
    dLSE = frp + pf
    grad = (2.0 * n / (S * S)) * dLSE - (4.0 * n * LSE / (S ** 3))
    
    return score, grad

def compute_score(a):
    """Exact score computation."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lbfgs_optimize(a0, beta, maxiter=10000, maxfun=100000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    bounds = [(0.0, None)] * n
    
    def obj(x):
        x = np.maximum(x, 0)
        sc, gr = lse_grad_f(x, beta)
        return sc, gr
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-10, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.maximum(res.x, 0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:

    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)

def peak_local_search(a, n_steps=5000, time_limit_s=60.0, t_start=None, rng=None):

    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a)
    a = a.copy()
    a = a / a.sum()  # Normalize

    best_a = a.copy()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))

        nc_idx = np.where(nc_mask)[0]
        if len(nc_idx) == 0:
            break
        target = nc_idx[rng.integers(0, len(nc_idx))]
        
        # Try several step sizes

            a[j] = a_save - moved
            a[target] = t_save + moved

                best_a = a.copy()
                improved = True
                break
            # Also check current score for the "annealing" accept
            # (skip for simplicity)
        
        if not improved:
            a[j] = a_save
            a[target] = t_save
    
    return best_score, best_a


def adam_lse(a0, beta_schedule, lr=0.01, time_limit_s=120.0, t_start=None):
    """Adam optimizer with LSE smoothing and beta schedule."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    deadline = t_start + time_limit_s
    t_global = 0
    
    for beta in beta_schedule:
        while time.monotonic() < deadline:
            t_global += 1
            sc, gr = obj_with_norm(a, beta)
            
            m = b1 * m + (1 - b1) * gr
            v = b2 * v + (1 - b2) * gr * gr
            m_hat = m / (1 - b1 ** t_global)
            v_hat = v / (1 - b2 ** t_global)
            
            lr_t = lr / (1 + 1e-5 * t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            
            S = a.sum()
            if S < 1e-12:
                a = a0.copy()
            else:
                a = a / S
            
            # Periodic check
            if t_global % 1000 == 0:
                sc_exact = compute_score(a)
                if sc_exact < best_score:
                    best_score = sc_exact
                    best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

"""
Optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses LSE-smoothed L-BFGS-B with geometric beta continuation, peak-targeting local search,
multi-length exploration, and simulated annealing for escaping local minima.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def lse_score_grad_normed(a, beta):
    """
    LSE-smoothed score and gradient for a normalized sequence (sum(a)=1).
    score = 2n * LSE_beta(a*a)
    Returns (score, grad) where grad is d(score)/d(a) with sum(a)=1 implicit.
    """
    n = len(a)
    c = fast_self_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    Z = exp_s.sum()
    p = exp_s / Z
    LSE = c_max + np.log(Z) / beta
    score = 2.0 * n * LSE

    # dLSE/da[j] = (f_rev * p)[n-1+j] + (p * f)[j]
    M_fft = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    pf = np.fft.irfft(P * F, M_fft)[:n]
    FR = np.fft.rfft(a[::-1], M_fft)
    frp = np.fft.irfft(FR * P, M_fft)[n - 1:2 * n - 1]
    dLSE = frp + pf
    # score = 2n * LSE / S^2, with S=1: grad = 2n * (dLSE - 2*LSE*1)
    # Actually: d(score)/da[j] = 2n*(dLSE[j]/S^2 - 2*LSE/S^3) = 2n*(dLSE[j] - 2*LSE) when S=1
    grad = 2.0 * n * (dLSE - 2.0 * LSE)
    return score, grad


def compute_exact_score(a):
    """Exact objective: 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def objective_fn(x, beta):
    """Objective with internal normalization for L-BFGS-B."""
    n = len(x)
    x = np.maximum(x, 1e-15)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = x / S
    sc, gr = lse_score_grad_normed(a, beta)
    # Chain rule: a = x/S, da[i]/dx[j] = (delta_ij - a[i])/S
    # d(score)/dx[j] = sum_i gr[i] * (delta_ij - a[i]) / S = (gr[j] - dot(gr,a)) / S
    dot_val = float(np.dot(gr, a))
    grad = (gr - dot_val * a) / S
    return sc, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000, deadline=None):
    """L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    bounds = [(0.0, None)] * n

    def obj(x):
        if deadline is not None and time.monotonic() > deadline:
            a = np.maximum(x, 0)
            return compute_exact_score(a), np.zeros(n)
        return objective_fn(x, beta)

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50, 'disp': False})
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-12:
        a_out = np.ones(n) / n
    sc = compute_exact_score(a_out)
    return sc, a_out


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1.0, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def peak_local_search(a, n_steps=10000, deadline=None, rng=None):
    """
    Peak-targeting local search with simulated annealing.
    Reduces elements contributing to the convolution peak and redistributes mass.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_exact_score(a)
    best_a = a.copy()
    T = 0.001  # Initial temperature for annealing

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Decay temperature
        T *= 0.9995

        c = fast_self_conv(a)
        k_star = int(c.argmax())

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            break

        # Pick contributor to reduce
        j = int(rng.integers(js, je + 1))
        if a[j] < 1e-12:
            continue

        # Pick non-contributor(s) to receive mass
        nc_mask = np.ones(n, dtype=bool)
        nc_mask[js:je + 1] = False
        nc_idx = np.where(nc_mask)[0]
        if len(nc_idx) == 0:
            break

        if rng.random() < 0.3:
            # Spread to all non-contributors
            targets = nc_idx
            weights = np.ones(len(targets))
        else:
            # Single target
            targets = np.array([int(nc_idx[rng.integers(0, len(nc_idx))])])
            weights = np.array([1.0])

        a_save = a.copy()
        cur_score = compute_exact_score(a)
        improved = False

        for frac in [0.15, 0.3, 0.5]:
            moved = a[j] * frac
            a[j] = a_save[j] - moved
            for t_idx, t_w in zip(targets, weights):
                a[t_idx] += moved * t_w / weights.sum()

            sc = compute_exact_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                improved = True
                break
            elif sc < cur_score + T:
                # Accept with probability (annealing)
                if rng.random() < np.exp(-(sc - cur_score) / max(T, 1e-10)):
                    a = a.copy()
                    break
            else:
                a = a_save.copy()

        if not improved:
            a = a_save.copy()

    return best_score, best_a


def adam_lse_refine(a0, beta_schedule, lr, deadline, verbose=False):
    """Adam with LSE smoothing and scheduled beta, with periodic exact checks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_global = 0

    best_a = a.copy()
    best_score = compute_exact_score(a)

    for beta in beta_schedule:
        while time.monotonic() < deadline:
            t_global += 1
            sc, gr = objective_fn(a, beta)

            m = b1 * m + (1 - b1) * gr
            v = b2 * v + (1 - b2) * gr * gr
            m_hat = m / (1 - b1 ** t_global)
            v_hat = v / (1 - b2 ** t_global)

            lr_t = lr / (1.0 + 1e-5 * t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                a = best_a.copy()
            else:
                a = a / S

            if t_global % 2000 == 0:
                sc_ex = compute_exact_score(a)
                if sc_ex < best_score:
                    best_score = sc_ex
                    best_a = a.copy()
                    if verbose:
                        print(f"    Adam beta={beta} t={t_global}: {sc_ex:.8f}")

    sc_final = compute_exact_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def generate_diverse_inits(rng, n, parent=None):
    """Generate multiple diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Gaussian centered
    for sigma in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Sinusoidal modulation
    for freq in [3, 5, 8, 12]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Random smooth
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.3
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        r_s = np.convolve(r, kernel, mode='same')
        inits.append(r_s / r_s.sum())

    # Two-bump
    c1, c2 = 0.3, 0.7
    b = np.exp(-0.5 * ((x - c1) / 0.08) ** 2) + np.exp(-0.5 * ((x - c2) / 0.08) ** 2)
    inits.append(b / b.sum())

    # Parent-based perturbation
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum()
        inits.append(p)
        # Log-space perturbation
        log_p = np.log(np.maximum(p, 1e-10))
        for scale in [0.05, 0.1, 0.2]:
            pert = log_p + scale * rng.standard_normal(n)
            a_p = np.exp(pert - pert.max())
            inits.append(a_p / a_p.sum())

    return inits

            elapsed = time.monotonic() - t_start

    parent = None

                    parent = parent / parent.sum()
                    print(f"Parent loaded: n={len(parent)}, score={compute_exact_score(parent):.8f}")
                    try_update(compute_exact_score(parent), parent, "parent")
                else:
                    parent = None
    except Exception:
        parent = None

    if parent is None:
        print("No valid parent, starting fresh at n=4096.")
        n_main = 4096
        best_a = np.ones(n_main) / n_main

    a_work = a_work / a_work.sum()

    beta_schedule = [20, 80, 300, 1000, 3000, 8000]
    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > p1_deadline:
            break
        # Allocate time proportionally (more time for higher beta)
        remaining = p1_deadline - time.monotonic()
        n_remaining = len(beta_schedule) - i
        time_alloc = min(remaining / n_remaining * 1.3, remaining)

    # Phase 2: Adam refinement (20% time)
    # ==========================================

                                                     rng=rng)
                    try_update(sc2, a_opt2, f"restart {i} + search")

        try_update(sc, a_opt, "final LBFGS")

    # Final local search
    if time.monotonic() < deadline_final - 5:
        sc, a_opt = peak_local_search(best_a, n_steps=10000,
                                       deadline=min(time.monotonic() + 30, deadline_final - 2),
                                       rng=rng)
        try_update(sc, a_opt, "final search")

    # ==========================================
