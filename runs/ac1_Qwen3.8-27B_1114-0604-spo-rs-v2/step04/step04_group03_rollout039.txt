def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros_like(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # For numerical stability, shift so max is 0
    shift = scaled.max()
    scaled -= shift
    exp_s = np.exp(scaled)
    log_sum_exp = np.log(exp_s.sum()) + shift
    L = c_max + log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient: dL/da[i] = 2 * sum_j p[i+j] * a[j]
    # Computed as cross-correlation via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L * 1.0 / (S * S * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

        return lse_score_and_grad(x, beta)

        obj, a0,

        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    """
    Coordinate descent that targets the maximum convolution peak.
    Reduces contributions to the peak and redistributes mass.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        
        # Find indices that contribute to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        # Contribution of each pair (i, k_star-i) to c[k_star]
        # For each j in [j_start, j_end], the contribution is a[j] * a[k_star - j]
        seg = a[j_start:j_end + 1]
        partner = a[k_star - j_end:k_star - j_start + 1][::-1]
        contrib = seg * partner
        
        # Reduce the top contributors
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        
        total_reduction = 0.0
        for li in top_local:
            gi = j_start + li
            reduction = a[gi] * 0.02  # reduce by 2%

            total_reduction += reduction

            bot_count = min(n_bottom, len(contrib) - top_count)

"""
Optimized step function minimizer using progressive LSE-smoothed L-BFGS,
peak-directed coordinate descent, frequency-domain error flattening,
and multi-length exploration. Exploits scale invariance (optimizes on simplex),
uses high-beta LSE for near-exact max approximation, and applies targeted
peak-reduction moves inspired by the active-set LP approach from the
state-of-the-art description.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    shift = scaled.max()
    scaled_shifted = scaled - shift
    exp_s = np.exp(scaled_shifted)
    log_sum = np.log(exp_s.sum()) + shift
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] via cross-correlation FFT
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B on LSE-smoothed objective with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_reduction_lp(a, M_target, max_iter=100):
    """
    Active-set inspired peak reduction.
    Solves: maximize sum(a) subject to conv(a,a)[k] <= M_target for all k, a >= 0.
    Uses a multiplicative update (like matrix scaling / Sinkhorn) to flatten peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iter):
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max <= M_target * (1 + 1e-12):
            break
        # Soft penalty: reduce weights where convolution exceeds target
        excess = np.maximum(c - M_target, 0.0)
        # Compute gradient pressure: dp[i] = sum_k excess[k] * dc[k]/da[i]
        # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
        rev_a = a[::-1]
        M2 = 3 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        E = np.fft.rfft(excess, M2_fft)
        r = np.fft.irfft(RA * E, M2_fft)[n - 1:2 * n - 1]
        pressure = 2.0 * r
        # Multiplicative update: reduce high-pressure elements
        eta = 0.1 / (1.0 + 0.01 * it)
        a_new = a * np.exp(-eta * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def error_flatten(a, n_iters=200):
    """Frequency-domain flattening of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        alpha = 0.6 + 0.3 * min(1.0, it / (n_iters * 0.7))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-14)
        d[np.abs(A) < 1e-12] = 0.0

        damp = 0.3 + 0.5 * min(1.0, it / n_iters)
        A_new = A + damp * d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            break
        a_new = a_new / a_sum

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_peak_descent(a, n_sweeps=5, top_k=80):
    """Coordinate descent reducing top contributors to the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue
        seg = a[j_start:j_end + 1]
        partner_idx = k_star - np.arange(j_end + 1, j_start - 1, -1)
        partner = a[partner_idx]
        contrib = seg * partner

        top_count = min(top_k, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        total_red = 0.0
        for li in top_local:
            gi = j_start + li
            red = a[gi] * 0.03
            a[gi] = max(a[gi] - red, 0.0)
            total_red += red

        if total_red > 1e-15:
            bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = total_red / len(bot_local)
            for li in bot_local:
                a[j_start + li] += per

        S = a.sum()
        if S > 1e-12:
            a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
