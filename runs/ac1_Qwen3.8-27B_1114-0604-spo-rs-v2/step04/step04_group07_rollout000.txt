The center convolution c[n-1] = sum_j a[j]*a[n-1-j]. For a palindromic sequence, this equals sum_j a[j]^2 which is minimized when all a[j] are equal (giving 1/n). For a non-palindromic sequence, this is sum_j a[j]*a[n-1-j] which can be smaller!

In fact, if a[j] = a[n-1-j] for all j (palindromic), then c[n-1] = sum a[j]^2 >= 1/n.
If a is "anti-correlated" with its reverse (a[j] is large where a[n-1-j] is small), then c[n-1] can be much less than 1/n!

But we also need to check the other lags. If we make a[j] and a[n-1-j] anti-correlated, the center lag drops, but other lags might increase.

This is the key insight the previous approach missed by forcing palindromicity!

**Revised plan:**
1. Do NOT force palindromicity
2. Allow the sequence to "anti-correlate" with its reverse to reduce the center peak
3. Use LSE-smoothed optimization in the full (non-palindromic) space
4. Use a larger n for more "degrees of freedom"

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv_fft(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum_exp = np.log(exp_scaled.sum())
    L = c_max + log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a[i]:
    # L = (1/beta) log(sum_k exp(beta * C[k]))
    # dL/da[i] = sum_k p[k] * dC[k]/da[i] = sum_k p[k] * 2*a[k-i] (for valid k-i)
    # = 2 * (p * a_rev_conv)[i + n - 1] ... 
    # Actually: sum_k p[k] * a[k-i] where 0 <= k-i < n
    # Let m = k - i, so k = m + i, m ranges over [0, n-1]
    # sum_{m=0}^{n-1} p[m+i] * a[m]
    # This is the cross-correlation of p and a at shift i
    # = (p * a_rev)[n-1+i] where a_rev = a[::-1]
    # Wait: (p * a_rev)[j] = sum_k p[k] * a_rev[j-k] = sum_k p[k] * a[n-1-(j-k)] = sum_k p[k] * a[n-1-j+k]
    # We want sum_k p[k] * a[k-i]
    # Let's set n-1-j+k = k-i => n-1-j = -i => j = n-1+i
    # So (p * a_rev)[n-1+i] = sum_k p[k] * a[k-i] ✓
    
    a_rev = a[::-1]
    M2 = len(p) + n - 1  # len(p) = 2n-1, so M2 = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)[:M2]
    # We need conv_pa[n-1 : 2n-1] which gives sum_k p[k]*a[k-i] for i=0..n-1
    profile = conv_pa[n-1:2*n-1]
    dL_da = 2.0 * profile
    
    # d(score)/da[i] = 2n * (dL_da[i] / S^2 - 2*L*dS/da[i] / S^2)
    # where dS/da[i] = 1
    # = 2n * dL_da[i] / S^2 - 4n*L / S^3
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * dL_da / S2 - (2.0 * n) * 2.0 * L / S3
    
    return score, grad

def adam_optimize(a_init, beta_schedule, lr_schedule, n_steps, time_limit_s, t_start):
    """Adam with LSE smoothing and scheduled beta/lr."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    t_global = 0
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            
            t_global += 1
            lr_t = lr_schedule[min(bi, len(lr_schedule)-1)] / (1.0 + 1e-5 * t)
            
            score, grad = lse_score_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e6)
            
            # Renormalize to keep sum constant (scale invariance)
            s = a.sum()
            if s > 1e-12:
                a = a / s * a_init.sum()  # keep same sum
        
        # Check improvement at end of each beta phase
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    
    # Fix sum to 1 for scale invariance
    a0 = a0 / a0.sum()
    n_free = n - 1  # fix last element to maintain sum=1
    
    def obj_and_grad(x):
        a = np.empty(n)
        a[:n_free] = x
        a[n-1] = max(0.0, 1.0 - x.sum())
        if a[n-1] < 0:
            a = np.clip(a, 0, None)
            a = a / a.sum()
        sc, gr = lse_score_grad(a, beta)
        # Project gradient: only first n_free components matter (last is determined)
        gr_free = gr[:n_free] - gr[n-1]  # since a[n-1] = 1 - sum(x)
        return sc, gr_free
    
    x0 = a0[:n_free].copy()
    bounds = [(0.0, 1.0)] * n_free
    
    res = minimize(obj_and_grad, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.empty(n)
    a_out[:n_free] = res.x
    a_out[n-1] = max(0.0, 1.0 - res.x.sum())
    a_out = np.clip(a_out, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
