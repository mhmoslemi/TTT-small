    n = len(a)

    # Actually, we need at least the sum constraint to be bounded
    # Let's start with a few initial constraints
    
    active_constraints = [n - 1]  # Start with the center lag
    
    for it in range(max_iters):

        m_active = len(active_constraints)
        A_ub = np.zeros((m_active, n))
        for i, k in enumerate(active_constraints):
            # (a * x)[k] = sum_j a[k-j] * x[j] (valid j)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                A_ub[i, j] = a[k - j]
        
        c = -np.ones(n)  # Minimize -sum(x)
        b_ub = np.full(m_active, M)

                      bounds=[(0, None)] * n)

            break
        
        x = res.x
        
        # Check all constraints
        full_conv = cross_conv(a, x)
        violated = [k for k in range(len(full_conv)) if full_conv[k] > M + 1e-8]
        
        if not violated:
            return x
        
        # Add the most violated constraint
        worst = max(violated, key=lambda k: full_conv[k])
        if worst not in active_constraints:
            active_constraints.append(worst)
        
        # Also add nearby constraints for efficiency
        for dk in range(-2, 3):
            k2 = worst + dk
            if 0 <= k2 < len(full_conv) and k2 not in active_constraints and full_conv[k2] > M * 0.9:
                active_constraints.append(k2)
    
    return res.x

A better bound: We know that conv(a,a)[n-1] = sum_j a[j]*a[n-1-j] >= (sum(a))^2 / n (by Cauchy-Schwarz, since the vectors (a[j]) and (a[n-1-j]) have the same entries).

Wait, that's not Cauchy-Schwarz. Let me think again.

By the rearrangement inequality or just by noting that for non-negative a:
sum_j a[j]*a[n-1-j] >= min(a) * sum_j a[n-1-j] = min(a) * sum(a)

That's not helpful.

Actually, by Cauchy-Schwarz:
(sum_j a[j]*a[n-1-j])^2 <= (sum_j a[j]^2) * (sum_j a[n-1-j]^2) = (sum_j a[j]^2)^2

So conv[n-1] <= sum(a^2). That's an upper bound.

For a lower bound on max(conv): We know conv[n-1] = sum_j a[j]*a[n-1-j]. If a is "symmetric" (a[j] = a[n-1-j]), then conv[n-1] = sum_j a[j]^2 >= (sum(a))^2/n.

But a doesn't have to be symmetric. If a is concentrated at one end, conv[n-1] can be small, but other lags will be large.

I think the practical lower bound is around 1.4-1.5, and we're at 1.507. Let me focus on practical improvements.

LP-Alternating optimizer using HiGHS LP solver for exact subproblem solving,

import numpy as np
from scipy.optimize import minimize, linprog

import time

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp_subproblem(a, M, max_cut_iters=50, verbose=False):
    """
    Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using cutting-plane with HiGHS.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Initialize with the peak lag and a few neighbors
    c_aa = self_conv(a)
    k_peak = int(np.argmax(c_aa))
    active_lags = sorted(set([k_peak] + [k_peak + d for d in range(-3, 4) if 0 <= k_peak + d < conv_len]))
    
    x_best = None
    obj_best = -np.inf
    
    for cp_iter in range(max_cut_iters):
        m_act = len(active_lags)
        A_ub = np.zeros((m_act, n))
        for i, k in enumerate(active_lags):
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                A_ub[i, j] = a[k - j]
        
        c_obj = -np.ones(n)
        b_ub = np.full(m_act, M)
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            break
        
        x = res.x
        obj = float(res.fun)
        
        if obj < obj_best:
            obj_best = obj
        
        # Check all constraints
        full_conv = cross_conv(a, x)
        violations = full_conv - M
        max_viol = float(violations.max())
        
        if max_viol < 1e-7:
            if verbose:
                print(f"    LP converged in {cp_iter+1} cut iterations, obj={-obj:.6f}")
            return x
        
        # Add most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active_lags:
            active_lags.append(worst_k)
            active_lags.sort()
        
        # Add neighbors of violated constraints
        viol_ks = np.where(violations > M * 0.1)[0]
        for vk in viol_ks:
            for d in range(-1, 2):
                k2 = int(vk) + d
                if 0 <= k2 < conv_len and k2 not in active_lags:
                    active_lags.append(k2)
        active_lags.sort()
        
        # Cap the number of active constraints
        if len(active_lags) > 200:
            # Keep only the most violated
            viol_scores = [(full_conv[k] - M, k) for k in active_lags]
            viol_scores.sort(reverse=True)
            active_lags = sorted([k for _, k in viol_scores[:200]])
    
    # Return best x found
    if x_best is not None:
        return x_best
    return x  # Return last x

def lp_alternating_higs(a, max_outer=50, t_start=0.4, t_min=0.05,
                        time_limit=None, verbose=True):
    """LP alternating with HiGHS solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    t = t_start
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        t0 = time.monotonic()
        M = float(self_conv(a).max())
        x = solve_lp_subproblem(a, M)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Line search on mixing parameter
        best_t = t
        best_sc_ls = compute_score(a)
        for t_try in [t, t*0.5, t*0.7, t*1.2, t*1.5]:
            t_try = min(t_try, 1.0)
            a_try = (1.0 - t_try) * a + t_try * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc_ls:
                best_sc_ls = sc_try
                best_t = t_try
        
        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        elapsed_outer = time.monotonic() - t0
        
        if sc < best_sc:
            if verbose:
                print(f"  LP iter {outer}: score={sc:.8f}, t={best_t:.3f}, time={elapsed_outer:.1f}s")
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if verbose:
                print(f"  LP iter {outer}: no improve (sc={sc:.8f} vs best={best_sc:.8f}), no_improve={no_improve}")
            t *= 0.7
            if no_improve >= 20:
                break
        
        a = a_new
        t = max(t_min, t * 0.95)
    
    return best_a, best_sc

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    # Phase 3: Local search with perturbations (10% time)
    phase3_deadline = t_start + 0.92 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: Local Search ---")
    
    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S
    
    for trial in range(10):
        if time.monotonic() > phase3_deadline:
            break
        scale = 0.005 + 0.02 * rng.random()
        a_pert = a_norm * (1.0 + scale * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_p = float(a_pert.sum())
        if S_p > 1e-12:
            a_pert = a_pert / S_p
            sc, a_opt = run_lbfgs(a_pert, 300, maxiter=2000, maxfun=50000)
            report(sc, a_opt, f"local #{trial}")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # Phase 4: Final LP polish (8% time)
    if time.monotonic() < deadline - 30:
        print(f"\n--- Phase 4: Final LP Polish ---")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 0:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating_higs(
            a_final_in, max_outer=30, t_start=0.1, t_min=0.01,
            time_limit=deadline - 10, verbose=True
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")
    
    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    
    return out
