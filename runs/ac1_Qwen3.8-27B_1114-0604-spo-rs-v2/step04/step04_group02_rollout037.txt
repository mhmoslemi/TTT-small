```python
"""
Peak-shaping optimizer for step function convolution minimization.
Combines: (1) adaptive LSE-LBFGS, (2) peak-targeted coordinate descent,
(3) structural diversification, and (4) multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    rev_a = a[::-1]
    M2 = 2 * n - 1  # convolution length
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    # conv_res[j] = sum_i rev_a[i] * p[j-i] = sum_i a[n-1-i] * p[j-i]
    # Let k = n-1-i: = sum_k a[k] * p[j-n+1+k]
    # We want dL/da_j = 2 * sum_l a[l] * p[l+j]
    # conv_res[j] = sum_k a[k] * p[j-n+1+k], so with j' = j-n+1: conv_res[n-1+j'] = sum_k a[k] * p[j'+k]
    # So dL/da_j = 2 * conv_res[n-1+j] for j = 0,...,n-1, i.e., positions n-1,...,2n-2 in conv_res
    g = 2.0 * conv_res[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L * np.ones(n) / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def peak_coordinate_descent(a, n_iters=5000, verbose=False):
    """
    Coordinate descent that specifically targets reducing the convolution peak.
    At each step, find the peak, identify the top contributor, and move mass
    from it to a position that minimally increases other convolution values.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_current = float(c[k_star])
        sc = 2.0 * n * M_current
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 1000 == 0:
                print(f"    PCD iter {it}: score={sc:.8f}")
        
        # Get the "partner" array for the peak
        # c_{k*} = sum_j a_j * a_{k*-j} for valid j
        # The "partner" of position j is a_{k*-j}
        j_valid = np.arange(max(0, k_star - n + 1), min(k_star + 1, n))
        partners = a[k_star - j_valid]
        
        # The contribution of position j to the peak is a_j * a_{k*-j}
        contribs = a[j_valid] * partners
        
        # Find the top contributor
        top_idx_local = int(np.argmax(contribs))
        top_j = j_valid[top_idx_local]
        
        if a[top_j] < 1e-10:
            # Try next top
            top_indices = np.argsort(contribs)[::-1]
            found = False
            for ti in top_indices:
                if a[j_valid[ti]] > 1e-10:
                    top_j = j_valid[ti]
                    found = True
                    break
            if not found:
                break
        
        # Find a good target: a position k such that moving mass from top_j to k
        # reduces c_{k*} and doesn't increase other c_m too much
        # Change in c_m: 2*delta*(a_{m-top_j} - a_{m-k})
        # We want: a_{k*-top_j} > a_{k*-k} (to reduce the peak)
        # And: a_{m-top_j} <= a_{m-k} for m where c_m is high (to not increase peaks)
        
        # Simplified: find k that maximizes (a_{k*-top_j} - a_{k*-k})
        # while minimizing the max increase elsewhere.
        # For speed, just try a few candidate targets.
        
        target_candidates = []
        # Positions where a_{k*-position} is small
        all_partners = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                all_partners[j] = a[k2]
        
        # Sort by partner value (ascending) - these are good targets
        sorted_by_partner = np.argsort(all_partners)[:20]  # top 20 candidates
        
        best_move = None
        best_improvement = 0
        delta = a[top_j] * 0.1  # move 10% of the mass
        
        for k_target in sorted_by_partner:
            if k_target == top_j:
                continue
            # Estimate improvement: change in c_{k*}
            improvement = delta * (a[k_star - top_j] - a[k_star - k_target]) if (0 <= k_star - top_j < n) else 0
            if k_star - k_target < 0 or k_star - k_target >= n:
                improvement = delta * a[k_star - top_j]
            
            # Estimate worst-case increase: max over m of delta*(a_{m-k_target} - a_{m-top_j})
            # This is expensive to compute exactly, so use a proxy:
            # The max of a shifted by k_target minus the max of a shifted by top_j
            # Approximate by looking at the top few m values
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = k_target
        
        if best_move is None:
            break
        
        a[top_j] -= delta
        a[best_move] += delta
        S = a.sum()
        if S < 1e-12:
            break
        # Don't renormalize here to maintain exact sum=1
        
    return best_a
```

```python
"""
Multi-phase optimizer for minimizing step function self-convolution peak.
Strategy: adaptive LSE-LBFGS + peak-targeted coordinate descent + structural
diversification + multi-length exploration. Key differences from previous:
- New peak coordinate descent with line search
- Structural perturbations (frequency-domain, chirp, mirror mixing)
- Adaptive beta schedule based on convergence
- Better time allocation with early stopping
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def peak_coordinate_descent(a, n_iters=3000, verbose=False):

    Coordinate descent targeting the convolution peak.
    At each step: find peak, move mass from top contributor to best target,
    with line search to ensure improvement.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        if len(js) < 2:
            break
        contribs = a[js] * a[k_star - js]

        # Sort contributors in descending order, try top few
        sorted_idx = np.argsort(contribs)[::-1]
        moved = False

        for si in sorted_idx[:5]:
            top_j = js[si]
            a_old = a[top_j]
            if a_old < 1e-12:
                continue

            # Compute partner array: partner[j] = a[k_star - j] if valid
            ar = np.arange(n)

        target_vals = np.where(mask, partner, np.inf)
        bottom_n = max(3, len(js) // 10)
        bottom_local = np.argsort(target_vals)[:bottom_n]
        bottom_js = np.arange(n)[bottom_local]

        # Try moving from each top to each bottom
        best_move = None
        best_sc_move = best_sc

        for tj in top_js[:3]:
            a_tj = a[tj]
            if a_tj < 1e-12:
                continue
            for bk in bottom_js[:3]:
                for frac in [0.2, 0.5]:
                    delta = a_tj * frac
                    a[tj] = a_tj - delta
                    a[bk] += delta
                    sc = compute_score(a)
                    a[tj] = a_tj
                    a[bk] -= delta
                    if sc < best_sc_move:
                        best_sc_move = sc
                        best_move = (tj, bk, delta)

        if best_move is not None:
            tj, bk, delta = best_move
            a[tj] -= delta
            a[bk] += delta
            if best_sc_move < best_sc:
                best_sc = best_sc_move
                best_a = a.copy()

        if it % 200 == 0 and verbose:
            print(f"    MSD iter {it}: best={best_sc:.10f}")

    return best_a


def structural_perturbation(a, rng, strength=0.08):
    """Generate a structurally perturbed version of a."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 7)

    if strategy == 0:
        # Frequency-domain sinusoidal modulation
        freq = rng.integers(1, max(2, n // 8))
        phase = 2 * np.pi * rng.random()
        amp = strength * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.sin(2 * np.pi * freq * x + phase))
    elif strategy == 1:
        # Chirp modulation
        chirp_freq = rng.integers(2, max(3, n // 10))
        a_p = a * (1.0 + strength * 0.5 * np.cos(np.pi * chirp_freq * x * (x + 0.5)))
    elif strategy == 2:
        # Mirror blending
        a_ref = a[::-1]
        t = rng.random() * 0.25
        a_p = (1.0 - t) * a + t * a_ref
    elif strategy == 3:
        # Peak-aware: reduce top contributors, boost bottom
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        a_p = a.copy()
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        contribs = a_p[js] * a_p[k_star - js]
        top_n = max(3, n // 15)
        top_local = np.argsort(contribs)[::-1][:top_n]
        for li in top_local:
            a_p[js[li]] *= (1.0 - strength * rng.random())
        deficit = a.sum() - a_p.sum()
        if deficit > 1e-10:
            bot_local = np.argsort(contribs)[:len(contribs) - top_n]
            weights = a_p[js[bot_local]] + 1e-10
            weights = weights / weights.sum()
            a_p[js[bot_local]] += deficit * weights
    elif strategy == 4:
        # Block-level perturbation
        n_blocks = int(rng.integers(4, max(5, n // 500)))
        block_size = n // n_blocks
        a_p = a.copy()
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + strength * (rng.random() - 0.5))
    elif strategy == 5:
        # Random walk perturbation
        a_p = a.copy()
        walk = np.cumsum(rng.standard_normal(n))
        walk -= walk.mean()
        walk /= (np.max(np.abs(walk)) + 1e-10)
        a_p = a * (1.0 + strength * walk)
    else:
        # Gaussian noise
        a_p = a * (1.0 + strength * 0.5 * rng.standard_normal(n))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # U-shape (high at edges)
    u_shape = 0.3 + 0.7 * (2.0 * x - 1.0) ** 2
    inits.append(u_shape / u_shape.sum())

    # Inverted U (high at center)
    iu_shape = 0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2)
    inits.append(iu_shape / iu_shape.sum())

    # Alternating (period 2)
    alt = np.ones(n)
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())

    # Period 3 oscillation
    p3 = np.ones(n)
    p3[1::3] = 0.6
    p3[2::3] = 0.3
    inits.append(p3 / p3.sum())

    # Cosine window
    cw = np.maximum(0.05, np.cos(np.pi * x))
    inits.append(cw / cw.sum())

    # Random
    r = np.abs(rng.standard_normal(n)) + 0.1
    inits.append(r / r.sum())

    # Multi-frequency
    mf = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    mf = np.clip(mf, 0.01, None)
    inits.append(mf / mf.sum())

    # Chirp-inspired
    chirp = 1.0 + 0.3 * np.cos(np.pi * 5 * x * (x + 1))
    chirp = np.clip(chirp, 0.01, None)
    inits.append(chirp / chirp.sum())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: LSE-LBFGS + peak coordinate descent + diversification.
    Returns the best non-negative sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    best_score = np.inf
    best_a = None
    parent = None

    # Load parent
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Parent loaded, length={len(parent)}")
    except Exception:
        pass

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working setup
    if parent is not None:
        n_work = len(parent)
        a_start = parent / parent.sum()
        sc_parent = compute_score(a_start)
        report(sc_parent, a_start, "parent")
    else:
        n_work = 4096
        a_start = np.ones(n_work) / n_work

    print(f"Working length: {n_work}, initial score: {best_score:.10f}")

    # ============================================================
    # Phase 1: Initial LSE-LBFGS from parent (10% time)
    # ============================================================
    t1_end = t_start + 0.10 * total_budget
    print(f"\n--- Phase 1: Initial LSE-LBFGS ---")

    for beta in [50, 200, 500, 1000]:
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=int(min(15000, remaining * 20)),
                              maxfun=int(min(60000, remaining * 80)))
        report(sc, a_opt, f"P1 b={beta}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Peak coordinate descent (15% time)
    # ============================================================
    t2_end = t_start + 0.25 * total_budget
    print(f"\n--- Phase 2: Peak Coordinate Descent ---")

    while time.monotonic() < t2_end:
        remaining = t2_end - time.monotonic()
        if remaining < 30:
            break
        n_iters = int(min(3000, max(500, remaining * 5)))
        a_pcd = peak_coordinate_descent(best_a, n_iters=n_iters)
        sc_pcd = compute_score(a_pcd)
        report(sc_pcd, a_pcd, "PCD")

        # Occasionally try multi-source
        if rng.random() < 0.4 and time.monotonic() < t2_end - 30:
            a_msd = multi_source_peak_descent(best_a, n_iters=n_iters // 2)
            sc_msd = compute_score(a_msd)
            report(sc_msd, a_msd, "MSD")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Diversified optimization (45% time)
    # ============================================================
    t3_end = t_start + 0.70 * total_budget
    print(f"\n--- Phase 3: Diversified Optimization ---")

    trial = 0
    beta_schedule = [100, 300, 500, 1000, 2000, 5000, 10000, 20000]
    bi = 0

    while time.monotonic() < t3_end:
        remaining = t3_end - time.monotonic()
        if remaining < 60:
            break

        # Generate perturbation with varying strength
        strength = 0.03 + 0.06 * rng.random()
        a_pert = structural_perturbation(best_a, rng, strength=strength)

        # Choose beta from schedule
        beta = beta_schedule[bi % len(beta_schedule)]
        bi += 1

        maxiter = int(min(20000, max(2000, remaining * 25)))
        maxfun = int(min(80000, max(8000, remaining * 80)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        improved = report(sc, a_opt, f"P3 #{trial} b={beta}")
        trial += 1

        # If improved, follow up with PCD
        if improved and time.monotonic() < t3_end - 60:
            a_pcd = peak_coordinate_descent(best_a, n_iters=1500)
            sc_pcd = compute_score(a_pcd)
            report(sc_pcd, a_pcd, f"P3+PCD #{trial}")

        # Every 5 trials, try a fresh initialization
        if trial % 5 == 0 and time.monotonic() < t3_end - 120:
            inits = generate_initializations(n_work, rng)
            for a_init in inits[:3]:
                if time.monotonic() > t3_end - 60:
                    break
                sc_i, a_i = run_lbfgs(a_init, 300, maxiter=5000, maxfun=20000)
                report(sc_i, a_i, f"P3 fresh #{trial}")

    print(f"After Phase 3: best = {best_score:.10f} (trials={trial})")

    # ============================================================
    # Phase 4: High-precision refinement (15% time)
    # ============================================================
    t4_end = t_start + 0.85 * total_budget
    print(f"\n--- Phase 4: High-Precision Refinement ---")

    for beta in [5000, 20000, 50000, 100000, 200000]:
        if time.monotonic() > t4_end:
            break
        remaining = t4_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(50000, max(5000, remaining * 60)))
        maxfun = int(min(200000, max(20000, remaining * 200)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"P4 b={beta}")

        if time.monotonic() < t4_end - 30:
            a_pcd = peak_coordinate_descent(best_a, n_iters=2000)
            sc_pcd = compute_score(a_pcd)
            report(sc_pcd, a_pcd, "P4+PCD")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Length exploration + final polish (10% time)
    # ============================================================
    print(f"\n--- Phase 5: Length Exploration & Final Polish ---")

    # Try different lengths
    for n_try in [n_work // 2, int(n_work * 1.25), int(n_work * 0.75)]:
        if time.monotonic() > deadline - 30:
            break
        if n_try < 256 or n_try > 40000:
            continue
        a_interp = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, len(best_a)), best_a)
        a_interp = np.clip(a_interp, 0.0, 1000.0)
        S = a_interp.sum()
        if S > 0:
            a_interp = a_interp / S
        sc, a_opt = run_lbfgs(a_interp, 500, maxiter=10000, maxfun=40000)
        report(sc, a_opt, f"P5 len={n_try}")
        if sc < best_score and time.monotonic() < deadline - 30:
            a_pcd = peak_coordinate_descent(a_opt, n_iters=1000)
            report(compute_score(a_pcd), a_pcd, f"P5+PCD len={n_try}")

    # Final PCD pass on the best
    if time.monotonic() < deadline - 20:
        a_final = peak_coordinate_descent(best_a, n_iters=3000)
        sc_final = compute_score(a_final)
        report(sc_final, a_final, "final PCD")

    # Very last L-BFGS with highest beta
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 200000, maxiter=20000, maxfun=80000)
        report(sc, a_opt, "final LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
