"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses multi-scale LSE-optimized gradient descent with peak-directed
coordinate descent and random restart perturbations, starting from
the parent construction.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smoothed_max(c, beta):
    """LSE approximation of max(c). Returns value and softmax weights."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    L = c_max + np.log(sum_exp) / beta
    p = exp_scaled / sum_exp
    return L, p


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Score = 2n * LSE(c) / (sum a)^2
    where LSE(c) ≈ max(c).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, p = lse_smoothed_max(c, beta)
    score = 2.0 * n * L / (S * S)
    
    # dLSE/d c_k = p_k
    # d c_k / d a_j = a_{k-j} (for valid indices), which is a[flip] convolved with p
    # Specifically: d/d a_j [sum_k p_k * c_k] = sum_k p_k * a_{k-j}
    # = (p * flip(a))[n-1-j] ... let me think more carefully.
    
    # c_k = sum_i a_i * a_{k-i}
    # dc_k / da_j = a_{k-j} (if 0 <= k-j < n) + a_{k-j} (if 0 <= j-k < n)... 
    # Wait, c_k = sum_i a_i * a_{k-i}. 
    # dc_k/da_j: the term a_j appears when i=j (giving a_{k-j}) and when k-i=j i.e. i=k-j (giving a_j * a_{k-(k-j)} = a_j * a_j)... 
    # No wait. c_k = sum_i a_i * a_{k-i}. 
    # dc_k/da_j = a_{k-j} (from the term i=j) + a_{j} (from the term k-i=j, i.e., i=k-j, giving a_{k-j} * a_j, derivative w.r.t. a_j is a_{k-j})
    # Wait, that's only if j != k-j, i.e., j != k/2.
    # If j = k/2 (k even), then dc_k/da_j = 2*a_j.
    # In general: dc_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} if k-j != j, else a_j (since it's a_j^2, derivative is 2*a_j).
    # Actually, let me just use the product rule carefully.
    # c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j = sum_i [delta_{i,j} * a_{k-i} + a_i * delta_{k-i,j}]
    # = a_{k-j} (if 0 <= k-j < n) + a_{k-j} (if 0 <= k-j < n)
    # = 2 * a_{k-j} (if 0 <= k-j < n)
    # Wait, that's the same term twice? Let me be more careful.
    # dc_k/da_j = d/da_j [sum_i a_i * a_{k-i}]
    # For a fixed i, d/da_j [a_i * a_{k-i}]:
    #   If j = i: a_{k-i} = a_{k-j}
    #   If j = k-i (i.e., i = k-j): a_i = a_{k-j}
    #   If j = i and j = k-i simultaneously (i.e., j = k/2): both terms contribute, giving 2*a_{k-j}
    # So: dc_k/da_j = a_{k-j} * [1_{0<=k-j<n} + 1_{0<=k-j<n}] = 2*a_{k-j} * 1_{0<=k-j<n}
    # Hmm, that's not right either. Let me do it step by step.
    # c_k = sum_i a_i * a_{k-i}
    # Let's write this as c_k = a_j * a_{k-j} + sum_{i != j} a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} + a_j * 0 + ... 
    # Wait, I need to be careful about which terms involve a_j.
    # The term a_i * a_{k-i} involves a_j if either i = j or k-i = j.
    # If i = j: the term is a_j * a_{k-j}, derivative w.r.t. a_j is a_{k-j}.
    # If k-i = j, i.e., i = k-j: the term is a_{k-j} * a_j, derivative w.r.t. a_j is a_{k-j}.
    # If j = k-j (i.e., k = 2j): both conditions give the same term a_j * a_j = a_j^2, derivative is 2*a_j.
    # So in general (for k != 2j): dc_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (if 0 <= k-j < n)
    # For k = 2j: dc_k/da_j = 2*a_j.
    # But wait, 2*a_{k-j} for k=2j gives 2*a_j, which is the same! So the formula dc_k/da_j = 2*a_{k-j} works in all cases (when 0 <= k-j < n).
    
    # So: d/d a_j [sum_k p_k * c_k] = sum_k p_k * 2 * a_{k-j} * 1_{0 <= k-j < n}
    # = 2 * sum_k p_k * a_{k-j} * 1_{0 <= k-j < n}
    # Let m = k-j, then k = m+j, and we need 0 <= m < n, so 0 <= m+j <= 2n-2 (always true for m < n, j < n).
    # = 2 * sum_{m=0}^{n-1} p_{m+j} * a_m (where p is defined for indices 0 to 2n-2)
    # This is the cross-correlation of p (padded to length 2n-1) with a, evaluated at lag j.
    # = 2 * (p_full * flip(a))[n-1+j] ... let me use FFT.
    
    # Actually, let me define: g_j = sum_{m=0}^{n-1} a_m * p_{m+j} for j = 0, ..., n-1
    # This is the cross-correlation. We can compute it via FFT:
    # g = IFFT(conj(FFT(a, M)) * FFT(p_padded, M)) ... no, that gives circular cross-correlation.
    # Actually, g_j = sum_m a_m * p_{m+j} is the cross-correlation of a and p at lag j.
    # In FFT terms: g = IFFT(FFT(a, M) * conj(FFT(p, M))) ... hmm, I need to be careful.
    
    # Let's use: g_j = sum_m a_m * p_{j+m}
    # This is the "asymmetric" cross-correlation. We can compute it as:
    # g = convolve(flip(a), p) evaluated at appropriate indices.
    # flip(a) = a[n-1], a[n-2], ..., a[0]
    # convolve(flip(a), p)[k] = sum_i flip(a)[i] * p[k-i] = sum_i a[n-1-i] * p[k-i]
    # Let j = n-1-i, so i = n-1-j:
    # = sum_j a[j] * p[k-(n-1-j)] = sum_j a[j] * p[k-n+1+j]
    # We want g_j = sum_m a_m * p[j+m], so we need k-n+1 = j, i.e., k = j+n-1.
    # So g_j = convolve(flip(a), p)[j+n-1] for j = 0, ..., n-1.
    
    # Computationally:
    M2 = 3 * n - 1  # length of convolve(flip(a), p) = n + (2n-1) - 1 = 3n-1
    M2_fft = 1 << (M2 - 1).bit_length()
    fa = a[::-1]
    FA = np.fft.rfft(fa, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_result = np.fft.irfft(FA * P, M2_fft)
    
    # g_j = conv_result[j + n - 1] for j = 0, ..., n-1
    g = conv_result[n-1:2*n-1]  # indices n-1 to 2n-2, which is n elements
    
    # grad_j = 2 * g_j / S^2 - 4 * n * L / S^3 * ... 
    # Wait, let me redo the gradient.
    # score = 2n * L / S^2
    # d(score)/da_j = 2n * (dL/da_j) / S^2 + 2n * L * d(1/S^2)/da_j
    # dL/da_j = 2 * g_j (from above, where g_j = sum_m a_m * p_{m+j})
    # d(1/S^2)/da_j = -2/S^3
    # So d(score)/da_j = 2n * 2*g_j / S^2 - 2n * L * 2/S^3
    # = 4n*g_j/S^2 - 4nL/S^3
    
    S2 = S * S
    S3 = S2 * S
    grad = 4.0 * n * (g / S2 - L / S3)
    
    return score, grad


def adam_optimize(a_init, beta_schedule, lr_schedule, n_steps, time_limit_s, t_start):
    """
    Adam optimization with scheduled beta and learning rate.
    a_init: initial sequence (will be normalized to sum=1)
    beta_schedule: list of (beta, n_steps_at_this_beta)
    lr_schedule: list of initial learning rates
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t_global = 0
    for beta, n_beta_steps, lr_init in zip(beta_schedule[0], beta_schedule[1], beta_schedule[2]):
        m = np.zeros(n)
        v = np.zeros(n)
        b1, b2 = 0.9, 0.999
        eps = 1e-8
        
        for t in range(n_beta_steps):
            t_global += 1
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            
            lr_t = lr_init / (1.0 + 1e-4 * t)
            score, grad = lse_score_and_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** (t + 1))
            v_hat = v / (1.0 - b2 ** (t + 1))
            
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a[:] = 1.0 / n
                s = 1.0
            a = a / s
            
            if (t + 1) % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    [Adam] beta={beta:.0f}, step={t+1}, score={sc:.8f}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def peak_coord_descent(a, n_rounds=20, time_limit_s=None, t_start=None):
    """
    Coordinate descent targeting the peaks of the convolution.
    For each peak, identify contributing elements and adjust them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()  # Normalize
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        
        # Find the top peaks
        peak_idx = np.argsort(c)[::-1][:5]
        
        improved = False
        for k_star in peak_idx:
            if time.monotonic() > deadline:
                break
            
            # Elements contributing to c[k_star] are a[j] for j in [max(0, k_star-n+1), min(k_star, n-1)]
            js = max(0, int(k_star) - n + 1)
            je = min(int(k_star), n - 1)
            
            # The contribution of a[j] to c[k_star] is 2*a[j]*a[k_star-j] (or a[j]^2 if j=k_star-j)
            # We want to reduce a[j] where a[k_star-j] is large
            contrib = np.abs(a[js:je+1] * a[k_star - np.arange(js, je+1)])
            top_contrib_idx = np.argsort(contrib)[::-1][:10]
            
            for local_idx in top_contrib_idx:
                j = js + local_idx
                old_val = a[j]
                
                # Try reducing this element
                for factor in [0.5, 0.7, 0.85, 0.95]:
                    a[j] = old_val * factor
                    sc = compute_score(a)
                    if sc < best_score:
                        best_score = sc
                        improved = True
                        break
                    elif factor == 0.95:
                        a[j] = old_val  # Restore if none worked
                        break
                else:
                    a[j] = old_val
        
        if not improved:
            break
    
    return best_score, a


def lbfgs_polish(a, beta, maxiter=5000, time_limit_s=None, t_start=None):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    elif a.sum() > 0:
        a = a / a.sum()
    
    bounds = [(0.0, None)] * n
    t0 = time.monotonic()
    
    def obj(x):
        if time_limit_s and t_start and time.monotonic() - t_start > time_limit_s + 60:
            # Return a large value to stop
            x[:] = 0  # Signal to stop
            return 1e10
        val, grad = lse_score_and_grad(x, beta)
        return val, grad
    
    res = minimize(
        obj, a, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': maxiter,
                 'maxfun': maxiter * 20, 'maxls': 50}
    )
    
    a_out = np.clip(res.x, 0.0, None)
    if a_out.sum() > 1e-12:
        a_out = a_out / a_out.sum()
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def random_perturb_and_optimize(a, n_iter=5, time_limit_s=None, t_start=None):
    """Add random noise and re-optimize."""
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf
    
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Add relative noise
        noise_scale = 0.05 + 0.1 * i  # Increasing noise
        a_pert = a + noise_scale * a.mean() * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() < 1e-12:
            continue
        a_pert = a_pert / a_pert.sum()
        
        # Short Adam optimization
        sc, a_opt = adam_optimize(
            a_pert,
            beta_schedule=([200, 500], [2000, 2000], [0.01, 0.005]),
            lr_schedule=[0.01, 0.005],
            n_steps=4000,
            time_limit_s=60,
            t_start=time.monotonic()
        )
        
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    [Perturb {i+1}] score={sc:.8f}")
    
    return best_score, best_a

    p = exp_scaled / sum_exp
    score = 2.0 * n * L / (S * S)

    # dL/da_j = 2 * sum_m a[m] * p[m+j]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    fa = a[::-1]
    FA = np.fft.rfft(fa, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(FA * P, M2_fft)
    g = conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 4.0 * n * (g / S2 - L / S3)

    return score, grad


def adam_optimize(a_init, betas, steps_per_beta, lrs, time_limit_s, t_start):
    """Adam with beta annealing schedule."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for idx in range(len(betas)):
        beta = betas[idx]
        n_steps = steps_per_beta[idx]
        lr_init = lrs[idx]
        m = np.zeros(n)
        v = np.zeros(n)
        b1, b2 = 0.9, 0.999
        eps = 1e-8

        for t in range(n_steps):
            if time.monotonic() - t_start > time_limit_s:
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                return best_score, best_a

            lr_t = lr_init / (1.0 + 5e-5 * t)
            score, grad = lse_score_and_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** (t + 1))
            v_hat = v / (1.0 - b2 ** (t + 1))

            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a[:] = 1.0 / n
                s = 1.0
            a = a / s

            if (t + 1) % 5000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    el = time.monotonic() - t_start
                    print(f"      Adam b={beta:.0f} s={t+1}/{n_steps} sc={sc:.8f} t={el:.0f}s")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_score, best_a


def lbfgs_polish(a, beta, maxiter=3000):
    """L-BFGS-B polish."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()

    def obj(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-12:
            return 1e10, np.zeros(len(x))
        x_n = x / s
        val, grad = lse_score_and_grad(x_n, beta)
        dot = np.dot(grad, x_n)
        grad_p = grad - dot * x_n
        return val, grad_p

    bounds = [(0.0, None)] * n
    res = minimize(obj, a, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': maxiter,
                            'maxfun': maxiter * 30, 'maxls': 50})
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() > 1e-12:
        a_out = a_out / a_out.sum()
    return compute_score(a_out), a_out
