import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 5e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def es_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 1024.0)
    return base * scale


def est_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = es_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    return max(64, min(5000, mf))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        cap = int((0.9 * rem) / max(es_eval_time(n), 1e-4))
        maxfun = int(min(int(maxfun), max(64, cap)))
    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-11, "maxiter": int(maxiter),
            "maxfun": int(maxfun), "maxls": 50}

    try:
        res = minimize(lambda x: lse_score_grad(x, beta), a, method="L-BFGS-B",
                       jac=True, bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a

    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def spectral_gs_init(n, rng, n_iter=150, relax=0.5):
    """Gerchberg-Saxton with relaxation for flat-spectrum non-negative init."""
    N = next_power_of_two(2 * n - 1)
    target_mag = np.ones(N // 2 + 1, dtype=np.float64)

    phi = rng.uniform(-np.pi, np.pi, size=N // 2 + 1)
    A = target_mag * np.exp(1j * phi)
    A[0] = abs(A[0])
    if N % 2 == 0:
        A[N // 2] = abs(A[N // 2])

    a = np.zeros(n, dtype=np.float64)
    for it in range(n_iter):
        a_full = np.fft.irfft(A, N)
        a = np.maximum(a_full[:n], 0.0)
        S = float(a.sum())
        if S < 1e-10:
            a = rng.random(n)
            continue
        A_new = np.fft.rfft(a, N)
        mag = np.abs(A_new)
        mag[mag < 1e-10] = 1e-10
        phase = np.angle(A_new)
        # Relaxed magnitude update
        new_mag = relax * target_mag + (1.0 - relax) * mag
        A = new_mag * np.exp(1j * phase)
        A[0] = abs(A[0])
        if N % 2 == 0:
            A[N // 2] = abs(A[N // 2])

    return normalize_a(a)


def spectral_random_init(n, rng, n_iter=80):
    """Random spectral init with varying relax."""
    relax = float(rng.uniform(0.3, 0.7))
    return spectral_gs_init(n, rng, n_iter=n_iter, relax=relax)

Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using:
- Spectral (Gerchberg-Saxton/Fienup) initialization for flat power spectrum,
- LSE-smoothed objective with L-BFGS-B optimization and continuation,

- Multiple sequence lengths and diverse restarts with spectral perturbations.

PEN_SUM = 5e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta

    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

def spectral_gs_init(n, rng, n_iter=150, relax=0.5):
    """Gerchberg-Saxton with relaxation for flat-spectrum non-negative init."""
    N = next_power_of_two(2 * n - 1)
    target_mag = np.ones(N // 2 + 1, dtype=np.float64)
    phi = rng.uniform(-np.pi, np.pi, size=N // 2 + 1)
    A = target_mag * np.exp(1j * phi)
    A[0] = abs(A[0])
    if N % 2 == 0:
        A[N // 2] = abs(A[N // 2])
    a = np.zeros(n, dtype=np.float64)
    for it in range(n_iter):
        a_full = np.fft.irfft(A, N)
        a = np.maximum(a_full[:n], 0.0)
        S = float(a.sum())
        if S < 1e-10:
            a = rng.random(n)
            continue
        A_new = np.fft.rfft(a, N)
        mag = np.abs(A_new)
        mag[mag < 1e-10] = 1e-10
        phase = np.angle(A_new)
        new_mag = relax * target_mag + (1.0 - relax) * mag
        A = new_mag * np.exp(1j * phase)
        A[0] = abs(A[0])
        if N % 2 == 0:
            A[N // 2] = abs(A[N // 2])
    return normalize_a(a)
