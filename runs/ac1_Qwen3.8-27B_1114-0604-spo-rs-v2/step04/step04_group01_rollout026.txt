What if I use a **semidefinite programming (SDP) relaxation**? Let `X = a*a^T`. Then `conv(a,a)[k] = sum_i a[i]*a[k-i] = sum_{i: k-i >= 0} X[i, k-i]`. The constraint `X = a*a^T` (rank 1) is the non-convex part. Relaxing to `X >= a*a^T` (i.e., X - a*a^T is PSD) gives an SDP. But this is expensive for large n.

Actually, a better relaxation: note that `a*i a[i-1]` terms appear. If I define `b_k = a[k]*a[k+1]` (adjacent products), then the problem becomes non-linear but perhaps more tractable.

Let me not go down the SDP path and instead focus on practical improvements to the existing approach.

     - Stage C: LSE with high beta (1000-10000) - target peaks

   - Periodically try perturbations of the best solution

3. **Final polish** (~400s):

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed gradient optimization, multiplicative pressure updates
with adaptive line search, diverse structured initializations, and
multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4  # Penalty for sum deviation from 1

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    
    def obj(x):
        val, g = lse_score_grad(x, beta)
        return val, g
    
    maxiter = max(50, min(maxfun // 2, 500))
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": int(maxiter),
        "maxfun": int(min(maxfun, 2000)),
        "maxls": 30,
    }
    
    try:
        res = minimize(obj, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def adaptive_pressure_update(a, deadline, max_steps=20):
    """Adaptive multiplicative pressure update with line search on eta."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc
    n = len(a)
    
    for step in range(max_steps):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0.0:
            break
        
        cur_a = None
        cur_new_sc = cur_sc
        
        # Try different active set sizes
        for k in (1, 3, 5, 10):
            if k >= d.size:
                continue
            idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(d.size, dtype=np.float64)
            w[idx] = 1.0 / k
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            
            # Binary search on eta for best improvement
            best_eta = None
            best_eta_sc = cur_new_sc
            
            for log_eta in np.linspace(-4, 1, 12):  # eta from ~0.018 to ~2.7
                eta = float(np.exp(log_eta)) / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_eta_sc:
                    best_eta_sc = sc
                    best_eta = eta
                    cur_a = q.copy()
            
            if best_eta is not None and best_eta_sc < best_sc:
                best_sc = best_eta_sc
                best_a = cur_a.copy()
            if best_eta is not None and best_eta_sc < cur_new_sc:
                cur_new_sc = best_eta_sc
                cur_a = cur_a.copy()
        
        if cur_a is None:
            break
        a = cur_a
        cur_sc = cur_new_sc
    
    return float(best_sc), best_a

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def generate_inits(n, rng, parent):
    """Generate diverse initializations for a sequence of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(arr.min())
        if mn < 0.0:
            arr = arr - mn + 0.01
        inits.append(normalize_a(arr))
    
    # Parent-based
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Local modifications of parent
        for _ in range(3):
            mod = pr * np.exp(0.3 * rng.standard_normal(n))
            add(mod)
    
    # Uniform
    add(np.ones(n))
    
    # Linear ramps
    for alpha in (0.3, 0.5, 0.7, 0.9):
        add(1.0 + alpha * (2.0 * x - 1.0))
    
    # Quadratic
    for c in (0.5, 1.0, 2.0, 5.0):
        add(1.0 + c * (1.0 - 4.0 * x * (1.0 - x)))
    
    # Cosine windows
    for bw in (0.3, 0.5, 0.7, 0.9):
        add(0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5) / bw)))
    
    # Chirp-based (flat spectrum)
    for fmax in (3, 5, 8, 12):
        chirp = np.cos(np.pi * fmax * x * x)
        add(abs(chirp) + 0.1)
    
    # Sine-modulated
    for f in (2, 3, 5, 7, 10):
        for amp in (0.3, 0.6, 0.9):
            add(1.0 + amp * np.sin(2.0 * np.pi * f * x))
    
    # Product of sines (creates more complex patterns)
    for f1 in (2, 3, 5):
        for f2 in (4, 7, 11):
            add(1.0 + 0.4 * np.sin(2.0 * np.pi * f1 * x) * np.sin(2.0 * np.pi * f2 * x))
    
    # Frequency-domain random (flat-ish spectrum)
    Nf = next_power_of_two(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(0.5, 2.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + 0.5 * noise)
    
    # Random binary/ternary patterns
    for _ in range(3):
        vals = np.where(rng.random(n) < 0.5, 0.0, 1.0)
        add(0.2 + 0.8 * vals)
    
    # Block patterns
    for nb in (4, 8, 16, 32, 64):
        if nb >= n:
            continue
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = n // nb
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals[-1], n - a.size)])
        add(a[:n])
    
    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed gradient optimization, adaptive multiplicative pressure
updates with log-eta line search, diverse structured initializations
(chirps, frequency-domain random, modulated patterns), hierarchical
multi-scale exploration, and careful time management.

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    d, _ = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a

    def obj(x):
        val, g = lse_score_grad(x, beta)
        return val, g

    maxiter = max(40, min(maxfun // 2, 400))
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": int(maxiter),
        "maxfun": int(min(maxfun, 2000)),
        "maxls": 20,
    }

    try:
        res = minimize(obj, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def adaptive_pressure(a, deadline, max_steps=25):
    """Adaptive multiplicative pressure with log-eta line search."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc
    n = len(a)

    for _ in range(max_steps):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0.0:
            break

        improved = False

        for k in (1, 3, 5, 12):
            if k >= d.size:
                continue
            idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(d.size, dtype=np.float64)
            w[idx] = 1.0 / float(k)
            h = corr_pair(a, w)
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-12:
                continue

            cur_a = None
            cur_sc_local = cur_sc

            for log_eta in np.linspace(-4.5, 0.5, 14):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = float(np.exp(log_eta)) / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < cur_sc_local:
                    cur_sc_local = sc
                    cur_a = q.copy()

            if cur_a is not None:
                if cur_sc_local < best_sc:
                    best_sc = cur_sc_local
                    best_a = cur_a.copy()
                cur_a2 = cur_a.copy()
                a = cur_a2
                cur_sc = cur_sc_local
                improved = True
            else:
                # Try positive pressure (increase contributing elements)
                cur_a = None
                cur_sc_local = cur_sc
                for log_eta in np.linspace(-4.0, -1.0, 8):
                    if deadline is not None and time.monotonic() > float(deadline):
                        break
                    eta = float(np.exp(log_eta)) / hmax
                    q = a * np.exp(eta * (h - float(np.mean(h))))
                    q = normalize_a(q)
                    sc = exact_score_a(q)
                    if np.isfinite(sc) and sc < cur_sc_local:
                        cur_sc_local = sc
                        cur_a = q.copy()
                if cur_a is not None:
                    if cur_sc_local < best_sc:
                        best_sc = cur_sc_local
                        best_a = cur_a.copy()
                    a = cur_a.copy()
                    cur_sc = cur_sc_local
                    improved = True

        if not improved:
            break

    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(arr.min())
        if mn < 0.0:
            arr = arr - mn + 0.01
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        for _ in range(4):
            add(pr * np.exp(0.25 * rng.standard_normal(n)))
        # Smooth parent
        k = min(7, n if n % 2 == 1 else n - 1)
        kern = np.ones(k) / k
        add(np.convolve(pr, kern, mode='same'))

    add(np.ones(n))

    for alpha in (0.2, 0.4, 0.6, 0.8):
        add(1.0 + alpha * (2.0 * x - 1.0))

    for c in (0.5, 1.0, 2.0, 4.0):
        add(1.0 + c * (1.0 - 4.0 * x * (1.0 - x)))

    for f in (1, 2, 3, 4, 6, 8):
        add(0.5 + 0.5 * np.cos(2.0 * np.pi * f * x))

    for f1 in (2, 3, 5):
        for f2 in (5, 8, 13):
            add(1.0 + 0.3 * np.sin(2.0 * np.pi * f1 * x) * np.sin(2.0 * np.pi * f2 * x))

    # Chirp-based
    for fmax in (2, 4, 8, 16, 32):
        chirp = np.abs(np.cos(np.pi * fmax * x * x))
        add(0.2 + 0.8 * chirp)

    # Frequency-domain random with different spectral shapes
    Nf = next_power_of_two(n)
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        decay = float(rng.uniform(0.3, 2.5))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** decay)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + 0.4 * noise)

    # Flat-spectrum random (target white noise)
    for _ in range(4):
        phase = rng.uniform(-np.pi, np.pi, size=(Nf + 1) // 2 + 1)
        spec = np.exp(1j * phase)
        spec[0] = 0.0
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - float(sig.mean())
        add(1.0 + 0.3 * sig)

    # Block patterns
    for nb in (4, 8, 16, 32, 64, 128):
        if nb >= n:
            continue
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = n // nb
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals[-1], n - a.size)])
        add(a[:n])

    # Alternating/block
    for per in (2, 4, 8):
        if per >= n:
            continue
        vals = np.ones(n)
        for i in range(0, n, 2 * per):
            vals[i:i+per] = 0.3
        add(vals)

    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits

        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
