**Information-theoretic / entropy approach:**
The problem is equivalent to finding a probability distribution a on {0,...,n-1} that minimizes the maximum self-convolution value. This is related to the concept of "saturation" in additive combinatorics.

Max = 1/2. Mean = 1/(2n-1). Ratio = (1/2)/(1/(2n-1)) = (2n-1)/2 ≈ n.

That's much worse! So concentrating mass is bad.

For a distribution with mass spread uniformly over m consecutive positions (out of n):

a[j] = 1/m for j in [p, p+m-1], 0 otherwise.
conv[k] = (number of valid j)/m^2.

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)

    for bi, beta in enumerate(betas):
        lr = lrs[bi] if bi < len(lrs) else lrs[-1]
        ns = steps_per[bi] if bi < len(steps_per) else steps_per[-1]
        for t in range(ns):
            if time.monotonic() - t0 > tlimit:
                return best_sc, best_a
            tg += 1
            sc, g = _lse_grad(a, beta)
            if not np.isfinite(sc):
                break
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            mh = m / (1 - b1 ** tg)
            vh = v / (1 - b2 ** tg)
            a = a - (lr / (1 + 1e-4 * t)) * mh / (np.sqrt(vh) + eps)
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-12:
                a /= ss
            if t % 1500 == 0:
                cs = _score(a)
                if cs < best_sc:
                    best_sc, best_a = cs, a.copy()
    fs = _score(a)
    if fs < best_sc:
        best_sc, best_a = fs, a.copy()
    return best_sc, best_a


def _peak_rebalance(a, n_iters=200, tlimit=30.0, t0=0.0):
    """Iteratively reduce the peak by shifting mass from peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return _score(a), a
    a /= s
    best_sc, best_a = _score(a), a.copy()
    deadline = t0 + tlimit
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _fast_conv(a)
        ks = int(c.argmax())
        M = c[ks]
        sc = 2.0 * n * M / (a.sum() ** 2)
        js, je = max(0, ks - n + 1), min(ks, n - 1)
        if je < js:
            continue
        idx = ks - np.arange(js, je + 1)
        contrib = a[idx]
        tot = contrib.sum()
        if tot < 1e-15:
            continue
        # Try several redistribution strengths
        best_trial_sc = sc
        best_trial_a = a.copy()
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.25]:
            atr = a.copy()
            reduction = alpha * (contrib / tot) * a[js:je + 1].sum()
            atr[js:je + 1] -= reduction
            atr = np.maximum(atr, 0.0)
            removed = a[js:je + 1].sum() - atr[js:je + 1].sum()
            nc = np.ones(n, dtype=bool)
            nc[js:je + 1] = False
            if nc.sum() > 0:
                w = a[nc].copy()
                ws = w.sum()
                if ws > 1e-15:
                    atr[nc] += removed * w / ws
                else:
                    atr[nc] += removed / nc.sum()
            atr /= atr.sum()
            tsc = _score(atr)
            if tsc < best_trial_sc:
                best_trial_sc = tsc
                best_trial_a = atr.copy()
        if best_trial_sc < sc - 1e-15:
            a = best_trial_a
            if a.sum() > 1e-12:
                a /= a.sum()
            if best_trial_sc < best_sc:
                best_sc, best_a = best_trial_sc, a.copy()
        else:
            # Stuck - try a smaller perturbation or break
            if alpha < 0.03:
                break
    return best_sc, best_a


def _lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a0.sum()
    if s < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 /= s
    bounds = [(0.0, None)] * n

    def obj(x):
        xc = np.maximum(x, 0.0)
        sc, gr = _lse_grad(xc, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.maximum(res.x, 0.0)
    s = a_out.sum()
    if s > 1e-12:
        a_out /= s
    return _score(a_out), a_out


def _make_init(rng, n, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    # Uniform
    inits.append(np.ones(n) / n)
    # Gaussian
    for sig in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        inits.append(g / g.sum())
    # Edge-concentrated
    e = np.exp(-x / 0.15) + np.exp(-(1 - x) / 0.15)
    inits.append(e / e.sum())
    # Sinusoidal
    for f in [1, 2, 3, 5, 8]:
        sgn = 0.5 + 0.5 * np.cos(2 * np.pi * f * x)
        inits.append(sgn / sgn.sum())
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 100)
    rs = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(rs / rs.sum())
    # Two bump
    c1, c2 = rng.uniform(0.1, 0.4), rng.uniform(0.6, 0.9)
    b = np.exp(-((x - c1) / 0.08) ** 2) + np.exp(-((x - c2) / 0.08) ** 2)
    inits.append(b / b.sum())
    return [np.maximum(i, 1e-15) for i in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Main optimization: non-palindrome LSE continuation + peak rebalancing + L-BFGS.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s) - 30.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 128:
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 1e-6:
                parent /= ps
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_up(sc, a, lbl):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            old = best_score
            best_score = sc
            best_a = np.maximum(a, 0.0).copy()
            el = time.monotonic() - t0
            print(f"  NEW BEST: {sc:.8f} (was {old:.8f}) | {lbl} | n={len(a)} | t={el:.0f}s")
            return True
        return False

    if parent is not None:
        n = len(parent)
        print(f"Parent loaded: n={n}, score={_score(parent):.8f}")
        try_up(_score(parent), parent, "parent")
    else:
        n = 4096
        print(f"No parent, starting fresh with n={n}")
        best_a = np.ones(n) / n
        best_score = _score(best_a)
        print(f"  Uniform: {best_score:.8f}")

    # === Phase 1: Quick LSE Adam screening (10%) ===
    print(f"\n--- Phase 1: Quick screening ({budget*0.10:.0f}s) ---")
    p1_end = t0 + budget * 0.10
    # Refine parent with quick Adam
    if parent is not None:
        sc, ao = _adam_lse(parent, [5, 20, 50], [0.02, 0.01, 0.005],
                           [3000, 3000, 3000], tlimit=10.0, t0=time.monotonic())
        try_up(sc, ao, "quick Adam parent")

    # Try a few inits if time permits
    if time.monotonic() < p1_end:
        inits = _make_init(rng, n, parent)
        for i, ini in enumerate(inits[:4]):
            if time.monotonic() > p1_end:
                break
            sc, ao = _adam_lse(ini, [5, 20], [0.02, 0.01], [2000, 2000],
                               tlimit=3.0, t0=time.monotonic())
            try_up(sc, ao, f"init{i}")

    # === Phase 2: Main LSE continuation Adam (40%) ===
    print(f"\n--- Phase 2: LSE continuation Adam ({budget*0.40:.0f}s) ---")
    p2_end = t0 + budget * 0.50
    betas = [5, 10, 20, 50, 100, 200, 500, 1000, 3000]
    lrs = [0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
    # Allocate more steps to higher betas
    steps = [2000, 2000, 3000, 3000, 4000, 4000, 5000, 5000, 5000]
    sc, ao = _adam_lse(best_a, betas, lrs, steps,
                       tlimit=p2_end - time.monotonic(), t0=time.monotonic())
    try_up(sc, ao, "main Adam")

    # === Phase 3: Peak rebalancing (15%) ===
    print(f"\n--- Phase 3: Peak rebalancing ({budget*0.15:.0f}s) ---")
    p3_end = t0 + budget * 0.65
    remaining = max(1.0, p3_end - time.monotonic())
    sc, ao = _peak_rebalance(best_a, n_iters=500, tlimit=remaining, t0=time.monotonic())
    try_up(sc, ao, "peak rebalance")

    # === Phase 4: L-BFGS polishing (20%) ===
    print(f"\n--- Phase 4: L-BFGS polish ({budget*0.20:.0f}s) ---")
    p4_end = t0 + budget * 0.85
    for beta in [200, 500, 1000, 3000]:
        rem = p4_end - time.monotonic()
        if rem < 3:
            break
        t1 = time.monotonic()
        sc, ao = _lbfgs_lse(best_a, beta, maxiter=2000, maxfun=30000)
        try_up(sc, ao, f"LBFGS beta={beta}")
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t1:.1f}s)")
        # Intersperse peak rebalancing
        if time.monotonic() < p4_end - 5:
            sc2, ao2 = _peak_rebalance(best_a, n_iters=50,
                                       tlimit=min(5.0, p4_end - time.monotonic()),
                                       t0=time.monotonic())
            try_up(sc2, ao2, f"rebalance after LBFGS beta={beta}")

    # === Phase 5: Structured restarts (10%) ===
    print(f"\n--- Phase 5: Restarts ({budget*0.10:.0f}s) ---")
    p5_end = t0 + budget * 0.95
    for r in range(6):
        rem = p5_end - time.monotonic()
        if rem < 5:
            break
        # Different perturbation strategies
        if r % 3 == 0:
            # Log-space Gaussian
            la = np.log(np.maximum(best_a, 1e-15))
            ap = np.exp(la + 0.1 * rng.standard_normal(len(best_a)))
        elif r % 3 == 1:
            # Peak avoidance
            c = _fast_conv(best_a)
            ks = int(c.argmax())
            js, je = max(0, ks - len(best_a) + 1), min(ks, len(best_a) - 1)
            ap = best_a.copy()
            ap[js:je + 1] *= 0.6
            ap = np.maximum(ap, 1e-15)
        else:
            # Block modulation
            nb = 8
            bs = len(best_a) // nb
            ap = best_a.copy()
            for b in range(nb):
                ap[b * bs:(b + 1) * bs] *= rng.uniform(0.5, 1.5)
            ap = np.maximum(ap, 1e-15)
        ap /= ap.sum()
        sc, ao = _adam_lse(ap, [20, 100, 500], [0.01, 0.005, 0.002],
                           [2000, 2000, 2000], tlimit=min(8.0, rem - 2), t0=time.monotonic())
        try_up(sc, ao, f"restart{r}")

    # === Phase 6: Final intensive polish (5%) ===
    print(f"\n--- Phase 6: Final polish ---")
    final_end = t0 + budget - 5.0
    if time.monotonic() < final_end:
        rem = final_end - time.monotonic()
        sc, ao = _lbfgs_lse(best_a, 5000, maxiter=5000, maxfun=100000)
        try_up(sc, ao, "final LBFGS")
    if time.monotonic() < final_end - 1:
        sc, ao = _peak_rebalance(best_a, n_iters=100,
                                 tlimit=min(5.0, final_end - time.monotonic() - 1),
                                 t0=time.monotonic())
        try_up(sc, ao, "final rebalance")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={time.monotonic()-t0:.1f}s/{budget_s}s")
    return out
