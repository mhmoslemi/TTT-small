"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed L-BFGS-B optimization with multi-length exploration,
direct subgradient methods, active-set-aware perturbations, and
structured initializations (V-shapes, cosines, exponential tapers).

import numpy as np
import time
from scipy.optimize import minimize


PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))

    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    mf = max(64, min(3000, mf))
    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    maxfun = max(64, maxfun)
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def subgrad_step(a, eta, k_active=3):
    d, S = d_from_a(a)
    if S < 1e-12 or d.size == 0:
        return a
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return a
    w = active_weights(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    return a * np.exp(-eta * h / hmax)


def subgrad_polish(a, max_iters=30, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        cur_best = None
        cur_sc = best_sc

        for k in (1, 2, 3, 5, 8):
            for mult in (0.01, 0.03, 0.08, 0.15, 0.30, 0.60):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step(a, mult, k)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    cur_best = q.copy()

        if cur_best is None:
            break
        a = cur_best

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=30):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shape (high at ends, low in middle) - key for reducing peak convolution
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Cosine shapes
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.cos(np.pi * x))

    # Linear ramps
    for alpha in (0.3, 0.6, 1.0, 1.5):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))

    # Exponential taper (one-sided)
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Bimodal (two peaks)
    for width in (0.1, 0.15, 0.25):
        for center in (0.25, 0.5):
            add(np.exp(-(((x - 0.5) / width) ** 2)) + np.exp(-(((x + 0.5) / width) ** 2)))
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1-center)) / width) ** 2)))

    # Sine-based
    for f in (1, 2, 3, 4):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth (low-pass)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    # Random trig
    for _ in range(5):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 8)))
            amp = float(rng.uniform(0.05, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Block patterns
    for nb in (4, 8, 16, 32):
        if nb > n:
            nb = max(2, n // 2)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals[-1], n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.60))
    mode = int(rng.integers(0, 8))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.40))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights(d, int(rng.integers(1, 6)))
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.4 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # Shift mass from high-conv to low-conv positions
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights(d, 3)
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * 0.5 * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 6:
        # V-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = p * vshape
    else:
        # Gaussian kernel smoothing
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")

    return normalize_a(np.clip(q, 0.0, None))


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for beta in (200, 600, 2000, 6000, 15000):
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        mf = estimate_maxfun(rem * 0.85, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    rem = float(deadline - time.monotonic())
    if rem > 15.0:
        mf = estimate_maxfun(rem * 0.85, len(a))
        if mf >= 64:
            sc, a2 = run_lbfgs(a, 30000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()

    return float(best_sc), best_a


def optimize_trajectory(a_start, deadline, rng, label):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 10.0))
    n = len(a)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # More aggressive continuation schedule
    stages = [
        (30.0, 0.06),
        (80.0, 0.08),
        (150.0, 0.08),
        (300.0, 0.10),
        (600.0, 0.10),
        (1200.0, 0.12),
        (3000.0, 0.12),
        (8000.0, 0.14),
        (20000.0, 0.12),
        (50000.0, 0.08),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, n)
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} elapsed={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0
                ))
            a = normalize_a(a2)

        # Add subgradient polish between stages
        if time.monotonic() < sub_dead - 5.0:
            sc_sg, a_sg = subgrad_polish(a, max_iters=10, deadline=sub_dead)
            if np.isfinite(sc_sg) and sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                print("[{}] subgrad best {:.8f} after beta={:.0f}".format(label, best_sc, beta))
            a = normalize_a(a_sg)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine which lengths to try
    lengths = []
    parent_len = 0
    if parent is not None:
        parent_len = int(min(parent.size, 16384))
        if parent_len < 2048:
            parent_len = 2048
        lengths.append(parent_len)
        # Try nearby and different lengths
        for L in [4096, 6144, 8192, 10240, 12288, 16384]:
            if L != parent_len:
                lengths.append(L)
    else:
        lengths = [8192, 6144, 10240, 4096, 12288]

    # Deduplicate and sort by priority
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 2048 <= L <= 16384:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:5]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Quick evaluation of diverse inits across lengths
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=20 if n > 8000 else 28)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))

    if not all_scores:
        n0 = lengths[0] if lengths else 8192
        all_seqs.append(np.ones(n0, dtype=np.float64) / float(n0))
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))
    print("Best init score: {:.8f} (from length {})".format(
        all_scores[int(order[0])], all_lens[int(order[0])]))

    # Select top starts (at most 3, from different lengths if possible)
    starts = []
    used_lens = set()

    # Always include parent if available
    if parent is not None and parent_len > 0:
        ps = normalize_a(resample_sequence(parent, parent_len))
        starts.append(ps)
        used_lens.add(parent_len)

    for idx in order:
        if len(starts) >= 3:
            break
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best: {:.8f} (length {})".format(best_score, len(best_seq)))

    # Phase 2: Main optimization with time allocation
    # Prioritize the best start, then others
    starts_sorted = sorted(range(len(starts)), key=lambda i: exact_score_a(starts[i]))

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.55, 0.25, 0.20]

    main_end = deadline - 300.0
    main_budget = max(60.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(main_end))

        if time.monotonic() > sub_dead - 20.0:
            continue

        sq = starts[si].copy()
        # For secondary starts, blend with best
        if rank > 0 and np.isfinite(best_score):
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            t_blend = 0.5 if rank == 1 else 0.7
            sq = normalize_a((1.0 - t_blend) * sq + t_blend * base)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = normalize_a(best_seq)

    # Phase 3: Subgradient polish on the best
    if time.monotonic() < deadline - 200.0:
        sc, seq = subgrad_polish(best_seq, max_iters=25, deadline=deadline - 200.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgradient polish: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 150.0 and perturb_done < 6:
        left = float(deadline - 150.0 - time.monotonic())
        if left < 60.0:
            break
        n_remaining = 6 - perturb_done
        chunk = time.monotonic() + left / float(n_remaining)
        p = perturb_best(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(perturb_done, best_score))
        perturb_done += 1

    # Phase 5: Final high-beta LSE polish
    if time.monotonic() < deadline - 50.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 10.0:
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 50000.0, deadline, mf)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

            if time.monotonic() < deadline - 30.0:
                rem = float(deadline - time.monotonic() - 5.0)
                if rem > 10.0:
                    mf = estimate_maxfun(rem * 0.85, len(best_seq))
                    sc, seq = run_lbfgs(best_seq, 100000.0, deadline, mf)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print("Very high beta final: {:.8f}".format(best_score))

    # Phase 6: Final subgradient cleanup
    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 15.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
