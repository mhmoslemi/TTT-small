```python
import time

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S*S)

def lse_grad_fft(a, beta):
    """Compute LSE score and gradient using FFT for the gradient."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S*S)
    
    # Gradient: dL/da[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * (a_rev * p)[n-1+i]
    a_rev = a[::-1]
    M2 = 3*n - 2  # length of a_rev * p
    M2_fft = 1 << (M2-1).bit_length()
    A_rev = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(A_rev * P, M2_fft)[:M2]
    dL_da = 2.0 * conv_res[n-1:2*n-1]
    
    S2 = S*S
    S3 = S2*S
    grad = 2.0*n * (dL_da/S2 - 2.0*L/S3)
    return score, grad

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {a >= 0, sum a = target_sum}."""
    n = len(v)
    v = v.copy()
    # Scale to target_sum
    v = v / target_sum  # work with sum=1, then scale back
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) * target_sum / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v - theta, 0)
    return result * target_sum

    a = a0.copy()
    # Normalize to sum=1
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    prev_grad = None
    prev_a = None
    alpha = 0.01  # initial step size
    
    for bi, beta in enumerate(beta_schedule):
        n_steps_beta = 0
        max_steps_beta = 50000
        
        while time.monotonic() - t_start < time_limit:
            score, grad = lse_grad_fft(a, beta)
            
            # BB step size
            if prev_grad is not None and prev_a is not None:
                s = a - prev_a
                y = grad - prev_grad
                sy = s @ y
                yy = y @ y
                if yy > 1e-20 and sy > 1e-20:
                    alpha_new = sy / yy
                    alpha = 0.5 * alpha + 0.5 * np.clip(alpha_new, 0.1*alpha, 2.0*alpha)
            
            # Gradient step
            a_new = a - alpha * grad

            a_new = project_simplex(a_new, 1.0)

                alpha *= 0.5
                a_new = a - alpha * grad
                a_new = project_simplex(a_new, 1.0)

                if sc_new > score * (1 + 1e-10):
                    # Accept anyway (non-monotone)
                    pass

            a = a_new
            
            n_steps_beta += 1
            if n_steps_beta % 1000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            
            if n_steps_beta >= max_steps_beta:
                break
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
```

One more thought: the **active-set peeling** idea. At the optimum, only a few lags are "active" (at the max). We can:

- Taking larger steps (since the subgradient is more accurate with fewer active lags).

```
  3. Take a step: a <- project_simplex(a - alpha * grad_k*).
  4. Choose alpha by line search (or BB).
  5. Check if conv[k*] has decreased sufficiently.
  6. If not, reduce alpha and retry.
```

```python
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient using FFT-based gradient computation."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / np.sum(exp_scaled)
    L = c_max + np.log(np.sum(exp_scaled)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient via FFT: dL/da[i] = 2 * (a_rev * p)[n-1+i]
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(A_rev * P, M2_fft)[:M2]
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient at the argmax lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(np.max(c))
    k_star = int(np.argmax(c))
    score = 2.0 * n * M / (S * S)
    # grad_k[i] = 2*a[k-i] for valid i
    grad_k = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    if hi >= lo:
        grad_k[lo:hi + 1] = 2.0 * a[k_star - np.arange(lo, hi + 1)]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_k / S2 - 2.0 * M / S3)
    return score, grad


def subgrad_score_active(a, active_lags, weights):
    """Score and weighted average subgradient over active lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(np.max(c))
    score = 2.0 * n * M / (S * S)
    grad_avg = np.zeros(n)
    for k, w in zip(active_lags, weights):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            grad_avg[lo:hi + 1] += w * 2.0 * a[k - np.arange(lo, hi + 1)]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_avg / S2 - 2.0 * M / S3)
    return score, grad


def project_simplex(v):
    """Euclidean projection onto {a >= 0, sum a = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def run_lse_continuation(a0, betas, alphas, time_limit_s, t_start, verbose=True):
    """LSE continuation with BB-adapted step size and simplex projection."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_score = compute_score(a)

    prev_a = a.copy()
    prev_grad = None
    step = alphas[0] if alphas else 0.01

    for bi, beta in enumerate(betas):
        initial_step = alphas[bi] if bi < len(alphas) else alphas[-1]
        step = initial_step
        local_best = best_score
        local_best_a = best_a.copy()
        stagnation_count = 0

        while time.monotonic() - t_start < time_limit_s:
            score, grad = lse_score_grad(a, beta)

            # BB step size adaptation
            if prev_grad is not None:
                s_vec = a - prev_a
                y_vec = grad - prev_grad
                sy = float(np.dot(s_vec, y_vec))
                yy = float(np.dot(y_vec, y_vec))
                if yy > 1e-20 and sy > 1e-20:
                    bb_step = sy / yy
                    step = 0.5 * step + 0.5 * np.clip(bb_step, 0.5 * step, 2.0 * step)
                else:
                    step *= 0.9

            # Gradient step
            a_new = a - step * grad
            a_new = project_simplex(a_new)

            # Occasional line search for safety
            sc_new = compute_score(a_new)
            if sc_new > score * (1 + 1e-6) + 1e-12:
                # Halve step and retry once
                step *= 0.5
                a_new = a - step * grad
                a_new = project_simplex(a_new)
                sc_new = compute_score(a_new)

            prev_a = a.copy()
            prev_grad = grad.copy()
            a = a_new

            if sc_new < local_best - 1e-10:
                local_best = sc_new
                local_best_a = a.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1

            if stagnation_count > 5000:
                break

            if verbose and (bi * 100000 + int((time.monotonic() - t_start) * 10)) % 200000 == 0:
                print(f"    LSE beta={beta}: score={local_best:.8f}, step={step:.6f}")

        # Update global best
        final_sc = compute_score(a)
        if final_sc < best_score:
            best_score = final_sc
            best_a = a.copy()
        if local_best < best_score:
            best_score = local_best
            best_a = local_best_a.copy()
        a = best_a.copy()

        if verbose:
            print(f"  LSE beta={beta}: best={best_score:.8f}")

    return best_score, best_a


def run_adam_lse(a0, betas, lr_init, time_limit_s, t_start, verbose=True):
    """Adam optimizer with LSE smoothing (for comparison/backup)."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / np.sum(a)

    m = np.zeros(n)
    v = np.zeros(n)

    eps = 1e-8
    t_count = 0

    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        while time.monotonic() - t_start < time_limit_s:
            t_count += 1
            lr = lr_init / (1.0 + 1e-5 * t_count)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_count)
            v_hat = v / (1.0 - b2 ** t_count)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, None)

                    best_a = a.copy()

    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B polishing with LSE smoothing."""
    n = len(a0)
    a_init = a0.astype(np.float64).copy()
    a_init = np.clip(a_init, 0.0, 1000.0)
    if a_init.sum() < 1e-6:
        a_init = np.ones(n) / n
    bounds = [(0.0, 1000.0)] * n

    def obj_fun(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj_fun, a_init, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def run_frank_wolfe(a0, beta, time_limit_s, t_start, verbose=True):
    """Frank-Wolfe (conditional gradient) with LSE smoothing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s
    t_iter = 0

    while time.monotonic() < deadline:
        t_iter += 1
        score, grad = lse_score_grad(a, beta)

        # FW step: v = e_{argmin grad}
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = 1.0

        # Step size: gamma = 2/(t+2) or exact line search
        gamma = 2.0 / (t_iter + 2)

        # Exact line search along [a, v]
        best_t = gamma
        best_sc = compute_score((1 - gamma) * a + gamma * v)
        for t_try in [0.5 * gamma, 1.5 * gamma, 2.0 * gamma, 0.1 * gamma]:
            t_try = np.clip(t_try, 0.01, 2.0)
            a_try = (1 - t_try) * a + t_try * v
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try

        a = (1 - best_t) * a + best_t * v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a = a / s

        if best_sc < best_score - 1e-12:
            best_score = best_sc
            best_a = a.copy()

        if verbose and t_iter % 5000 == 0:
            print(f"    FW iter {t_iter}: best={best_score:.8f}")

    return best_score, best_a


def generate_initializations(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    sigma = 0.15 + 0.1 * rng.random()
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Sinusoidal
    freq = 2 + 6 * rng.random()
    phase = 2 * np.pi * rng.random()
    s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Multi-frequency
    f1 = 1 + 4 * rng.random()
    f2 = 3 + 7 * rng.random()
    m = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * x) + 0.1 * np.cos(2 * np.pi * f2 * x)
    m = np.maximum(m, 0.01)
    inits.append(m / m.sum())

    # Smooth random
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 100)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    return inits


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

        sc, a_opt = run_lse_continuation(a_cur, [beta], [alpha], t_actual, t_start, verbose=False)
        if sc < best_score:
            try_update(sc, a_opt, f"LSE-BB beta={beta}")
            a_cur = a_opt.copy()
        else:
            # Check if a_cur improved internally
            sc_cur = compute_score(a_cur)
            if sc_cur < best_score:
                try_update(sc_cur, a_cur, f"LSE-BB beta={beta} (cur)")
        print(f"  Beta={beta}: score={min(sc, compute_score(a_cur)):.10f}, t={time.monotonic()-t_start:.0f}s")

    # ============================================
    # Phase 2: L-BFGS Polishing (25% time)
    # ============================================
    p2_end = t_start + budget * 0.60
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Polishing (budget={(p2_end-t_start):.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        a_cur = best_a.copy()
        for beta in [1000, 3000, 8000, 20000]:
            t_remaining = p2_end - time.monotonic()
            if t_remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=40000)
            elapsed_lb = time.monotonic() - t0
            if sc < best_score:
                try_update(sc, a_opt, f"LBFGS beta={beta}")
                a_cur = a_opt.copy()
            print(f"  LBFGS beta={beta}: {sc:.10f} ({elapsed_lb:.1f}s)")

    # ============================================
    # Phase 3: Peeling Coordinate Descent (15% time)
    # ============================================
    p3_end = t_start + budget * 0.75
    print(f"\n{'='*60}")
    print(f"Phase 3: Peeling CD (budget={(p3_end-time.monotonic()):.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        p3_time = p3_end - time.monotonic()
        if p3_time > 5:
            sc, a_opt = run_peeling_cd(best_a, p3_time, t_start, verbose=True)
            if sc < best_score:
                try_update(sc, a_opt, "Peeling CD")
            print(f"  Peeling CD final: {sc:.10f}")

    # ============================================
    # Phase 4: Frank-Wolfe + Restarts (12% time)
    # ============================================
    p4_end = t_start + budget * 0.87
    print(f"\n{'='*60}")
    print(f"Phase 4: FW + Restarts (budget={(p4_end-time.monotonic()):.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        p4_time = p4_end - time.monotonic()
        if p4_time > 10:
            # Frank-Wolfe on current best
            fw_time = min(p4_time * 0.4, 120.0)
            sc, a_opt = run_frank_wolfe(best_a, 500, fw_time, t_start, verbose=False)
            if sc < best_score:
                try_update(sc, a_opt, "Frank-Wolfe")

            # Restarts with perturbation
            remaining = p4_end - time.monotonic()
            n_restarts = min(4, int(remaining / 30))
            for r in range(n_restarts):
                if time.monotonic() > p4_end - 5:
                    break
                # Perturb in log space
                log_a = np.log(np.clip(best_a, 1e-10, 1e10))
                noise = 0.03 + 0.03 * rng.random()
                log_pert = log_a + noise * rng.standard_normal(len(best_a))
                a_pert = np.exp(log_pert - np.max(log_pert))
                a_pert = a_pert / a_pert.sum()

                # Quick LSE refinement
                t_left = p4_end - time.monotonic() - 3
                if t_left < 3:
                    break
                sc, a_r = run_lse_continuation(a_pert, [200, 1000], [0.01, 0.005],
                                                t_left, t_start, verbose=False)
                if sc < best_score:
                    try_update(sc, a_r, f"Restart {r}")

    # ============================================
    # Phase 5: Final Intensive Polish (8% time)
    # ============================================
    p5_end = deadline
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish (budget={(p5_end-time.monotonic()):.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        a_cur = best_a.copy()

        # Very high beta L-BFGS
        for beta in [10000, 30000]:
            t_remaining = p5_end - time.monotonic()
            if t_remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=5000, maxfun=60000)
            if sc < best_score:
                try_update(sc, a_opt, f"Final LBFGS beta={beta}")
                a_cur = a_opt.copy()
            print(f"  Final LBFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

        # Final peeling CD
        t_remaining = p5_end - time.monotonic()
        if t_remaining > 5:
            sc, a_opt = run_peeling_cd(best_a, min(t_remaining, 60.0), t_start, verbose=False)
            if sc < best_score:
                try_update(sc, a_opt, "Final Peeling CD")

        # One more LSE sweep
        t_remaining = p5_end - time.monotonic()
        if t_remaining > 5:
            sc, a_opt = run_lse_continuation(best_a, [5000], [0.001],
                                              min(t_remaining, 30.0), t_start, verbose=False)
            if sc < best_score:
                try_update(sc, a_opt, "Final LSE")

    # ============================================
    # Final verification and output
    # ============================================
    if best_a is None:
        out = [1.0 / 64] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [1.0 / 64] * 64

    # Verify with the actual evaluator
    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget}s")
    print(f"{'='*60}")

    return out
```
