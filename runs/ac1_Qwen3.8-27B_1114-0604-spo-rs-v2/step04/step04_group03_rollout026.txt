"""
Spectral flattening optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: iterative spectral domain flattening with non-negativity projection,
combined with LSE-smoothed L-BFGS for fine-tuning.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_flattening(a, n_iters=500, beta=0.3, relax=1.0, seed=42):
    """
    Iteratively flatten the power spectrum of a while maintaining a >= 0.
    
    Algorithm:
    1. Compute FFT of a (zero-padded to M >= 2n-1)
    2. Raise the power spectrum to beta (< 0.5 flattens)
    3. Reconstruct with same phase
    4. Project to a >= 0 and normalize
    """
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        amp = np.abs(A)
        S_spec = amp * amp
        
        # Flatten: new_amp = S_spec^beta = amp^(2*beta)
        # For beta < 0.5, this compresses the dynamic range
        new_amp = S_spec ** beta
        # Fix DC: A[0] = sum(a) = 1
        new_amp[0] = 1.0
        
        # Reconstruct: same phase, new amplitude
        phase = np.ones_like(A, dtype=np.complex128)
        mask = amp > 1e-15
        phase[mask] = A[mask] / amp[mask]
        A_new = phase * new_amp
        
        # Relaxation
        if relax < 1.0:
            A_final = (1.0 - relax) * A + relax * A_new
        else:
            A_final = A_new
        
        a_new = np.fft.irfft(A_final, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * (rev(a) * p)[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_targeted_reduction(a, n_iters=200):
    """
    Target the maximum peak in convolution and reduce it.
    Uses the subgradient direction with adaptive step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) at a
        # d(conv(k*))/da[i] = a[k*-i] if 0 <= k*-i < n
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Full gradient of score = 2n*M/S^2
        # d(score)/da[i] = 2n/S^2 * grad[i] - 4n*M/S^3 (since d(S)/da[i] = 1)
        gn = float(np.sum(grad**2))
        if gn < 1e-30:
            break
        
        # Step size: proportional to (score - lower_bound) / ||grad||^2
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        step = (sc - lower_bound) / (gn / (S * S) + 1e-30)
        step = min(step, 0.03)
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

```python
"""
Spectral flattening optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Primary method: iterative spectral domain flattening with non-negativity projection.
Secondary: LSE-smoothed L-BFGS for fine-tuning.
The spectral approach directly targets making the power spectrum (and hence
self-convolution) as flat as possible, which is the fundamental goal.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_flatten_iter(a, M, beta, alpha=1.0):
    """Single iteration of spectral flattening."""
    A = np.fft.rfft(a, M)
    amp = np.abs(A)
    S_spec = amp * amp
    
    new_amp = S_spec ** beta
    new_amp[0] = 1.0  # Fix DC
    
    # Phase
    phase = np.ones_like(A, dtype=np.complex128)
    mask = amp > 1e-15
    phase[mask] = A[mask] / amp[mask]
    A_new = phase * new_amp
    
    if alpha < 1.0:
        A_final = (1.0 - alpha) * A + alpha * A_new
    else:
        A_final = A_new
    
    a_new = np.fft.irfft(A_final, M)[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return None
    return a_new / S_new


def spectral_flattening(a, n_iters=500, beta=0.3, alpha=1.0):
    """Iterative spectral flattening."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        a_new = spectral_flatten_iter(a, M, beta, alpha)
        if a_new is None:
            break
        a = a_new
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a


def log_spectral_flattening(a, n_iters=500, alpha=0.3):
    """
    Log-domain spectral flattening.
    Works on log(power spectrum) and shifts towards the mean.
    """
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        amp = np.abs(A)
        # Log spectrum (add small constant for stability)
        log_S = np.log(amp * amp + 1e-30)
        
        # Shift towards mean (flattening in log domain)
        log_S_mean = log_S.mean()
        log_S_new = (1.0 - alpha) * log_S + alpha * log_S_mean
        
        # Convert back to amplitude
        new_amp = np.exp(0.5 * log_S_new)
        # Normalize so that DC = 1
        new_amp = new_amp / new_amp[0]
        
        # Reconstruct
        phase = np.ones_like(A, dtype=np.complex128)
        mask = amp > 1e-15
        phase[mask] = A[mask] / amp[mask]
        A_new = phase * new_amp
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_reduction(a, n_iters=200):
    """Subgradient-based peak reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Subgradient
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Weighted step: reduce the peak contributors
        # Normalize grad to have unit norm
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Step size based on gap to lower bound
        lb = 2.0 * n / (2.0 * n - 1.0)
        step = max(1e-4, min(0.02, (sc - lb) / (gn * gn / (S * S) + 1e-30)))
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:

            inits.append(parent.astype(np.float64).copy())

            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)

    x = np.arange(n, dtype=np.float64) / n

    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))

    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))

def conv_error_reduction(a, n_iters=500, beta=0.5):
    """
    Error reduction in the frequency domain.
    Targets the peaks of the linear convolution.
    """
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]
        
        conv_mean = conv.mean()
        conv_max = conv.max()
        if conv_max - conv_mean < 1e-16:
            break
        
        # Excess above mean (what we want to reduce)
        excess = np.maximum(conv - conv_mean, 0.0)
        if excess.max() < 1e-16:
            break
        
        # Compute correction in frequency domain
        # We want: IFFT(2*Re(A*conj(ΔA))) = -excess (approximately)
        # So: 2*Re(A*conj(ΔA)) ≈ -FFT(excess)
        # ΔA ≈ -FFT(excess) / (2*conj(A))
        
        excess_pad = np.zeros(M, dtype=np.float64)
        excess_pad[:2 * n - 1] = excess
        E = np.fft.rfft(excess_pad, M)
        
        # Correction
        denom = 2.0 * np.conj(A)
        # Regularize: where |A| is small, don't correct (to avoid division by zero)
        mask = np.abs(A) > 1e-10
        Delta_A = np.zeros_like(A)
        Delta_A[mask] = -E[mask] / denom[mask]
        
        # Apply correction with step size
        damp = beta  # step size
        A_new = A + damp * Delta_A
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
```

- We can modify a, which changes conv quadratically.
- A small change Δa changes conv by approximately 2 * (a * Δa) (linearization).

We want to find Δa that reduces the peaks of conv:

This is a least-squares problem: find Δa to minimize ||2*(a * Δa) + excess||^2.

The normal equations: (J^T J) Δa = -J^T excess.
J^T J is a 2-n^2 matrix (for n = 6962, that's 48.5 million entries - too large for direct solve).

But we can use the conjugate gradient method to solve this without forming J^T J explicitly. Each J and J^T application is a convolution (O(n log n)).

    Gauss-Newton method for minimizing max(conv) via linearized subproblem.
    Uses CG to solve the normal equations without forming the full matrix.
    """

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)

        target = 1.0 / (2 * n - 1)
        excess = np.maximum(conv - target, 0.0)

            return jtjx

            # -J^T excess = -2 * (rev(a) * excess)[n-1 : 2n-2]
            return -2.0 * inner_conv(a, excess)

        b = rhs()
        x_cg = cg_solve(matvec, b, n_iters_cg=50, reg=1e-6 * np.mean(np.abs(b)))
