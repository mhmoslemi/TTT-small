Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE-smoothed L-BFGS, LP-inspired peak projection,
sonic peak-shaving, multi-length exploration, and coordinate descent.

import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_shift = scaled.max()
    scaled -= scaled_shift
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + (log_sum + scaled_shift) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient: dL/da[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    rev_a = a[::-1]
    M2 = 3 * n
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

def sonic_peak_shave(a, n_iters=200, n_peaks=8, alpha=0.5):
    """
    Sonic-style peak shaving: iteratively reduce the dynamic range
    of the autocorrelation by adjusting the sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_mean = float(c.sum()) / len(c)
        c_max = float(c.max())
        if c_max - c_mean < 1e-14:
            break

        # Target: reduce peaks toward the mean
        target = c_mean + alpha * (c_max - c_mean)
        # Find lags exceeding target
        excess = np.maximum(c - target, 0.0)
        if excess.max() < 1e-14:
            break

        # For each excess lag, compute gradient of excess wrt a
        # and take a step to reduce it
        grad = np.zeros(n)
        excess_lags = np.where(excess > 1e-14)[0]
        # Limit to top lags for efficiency
        if len(excess_lags) > n_peaks:
            top_lags = excess_lags[np.argsort(excess[excess_lags])[::-1][:n_peaks]]
        else:
            top_lags = excess_lags

        for k in top_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            # d(c[k])/da[i] for i in [j_start, j_end] = a[k-i]
            # which is seg reversed
            grad[j_start:j_end + 1] += excess[k] * seg[::-1]

        # Step: reduce a in direction of grad
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        step = min(0.02, (c_max - target) / (gn * gn + 1e-30))
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lp_projection_step(a, n_iters=30):
    """
    LP-inspired projection: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k, where M = max(a*a).
    Approximated by iterative constraint reduction.
    """
    n = len(a)
    a_norm = a.astype(np.float64).copy()
    S = float(a_norm.sum())
    if S < 1e-12:
        return a.copy()
    a_norm = a_norm / S
    M = float(fast_conv(a_norm).max())

    g = np.ones(n)
    for it in range(n_iters):
        # Compute cross-convolution a_norm * g
        c = fast_conv_cross(a_norm, g)
        # Find most violated constraints
        excess = c - M
        viol = np.where(excess > 1e-12)[0]
        if len(viol) == 0:
            break

        # Process top violated lags
        viol_sorted = viol[np.argsort(excess[viol])[::-1][:20]]
        for k in viol_sorted:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg_a = a_norm[j_start:j_end + 1][::-1]
            # Reduce g proportionally to a's contribution
            total = float(np.sum(seg_a * g[j_start:j_end + 1]))
            if total < 1e-15:
                continue
            reduction = min((c[k] - M) / total, 0.3)
            g[j_start:j_end + 1] -= reduction * seg_a
            g = np.maximum(g, 0.0)

    S_g = float(g.sum())
    if S_g < 1e-12:
        return a.copy()
    return g / S_g


def mirror_descent_v2(a, n_iters=2000, eta0=0.05, n_soft_peaks=5):

def frequency_flatten(a, n_iters=300, alpha=0.3):
    """
    Frequency-domain flattening: modify the spectrum to reduce
    peak-to-avg ratio of the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        # Compute error: where conv exceeds mean by too much
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        # Modify spectrum to reduce excess
        err_pad = np.zeros(M_fft)
        err_pad[:2 * n - 1] = -excess * 0.5
        err_spec = np.fft.rfft(err_pad)

        # Update: A_new = A + err_spec / (2*A) (Newton step on |A|^2)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-20)
        d[np.abs(A) < 1e-12] = 0.0
        damp = 0.3 + 0.3 * (it / max(1, n_iters // 3))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def coordinate_descent_peaks(a, n_sweeps=5, n_top=40):

    print(f"Parent: {'yes n=' + str(len(parent)) if parent is not None else 'no'}")
    print(f"Lengths: {lengths}")

    # ============================================================
    # Phase 1: Quick LSE optimization from parent and inits (25% time)
    # ============================================================
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: LSE optimization ---")

    candidates = []  # list of (score, array, n)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        # Limit number of inits based on time
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 12.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-6:
                a0 = np.ones(n) * 0.5
            # Quick LSE with moderate beta
            sc, a_opt = run_lbfgs(a0, beta=200, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"p1 n={n} i={i}")
            candidates.append((sc, a_opt.copy(), n))

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Refined LSE with increasing beta (30% time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: Refined LSE (increasing beta) ---")

    beta_schedule = [500, 1000, 2000, 5000, 10000, 20000]
    n_best = len(best_a)

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining_for_beta = phase2_deadline - time.monotonic()
        maxiter = min(8000, int(remaining_for_beta / 0.05))
        maxfun = min(150000, int(remaining_for_beta / 0.003))
        if maxiter < 500:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"p2 b={beta}")

    # Also try from second-best candidate
    if len(candidates) > 1 and time.monotonic() < phase2_deadline:
        sc2, a2, n2 = candidates[1]
        if n2 != n_best:
            a2_interp = np.interp(
                np.linspace(0, 1, n_best),
                np.linspace(0, 1, n2), a2
            )
        else:
            a2_interp = a2
        sc, a_opt = run_lbfgs(a2_interp, 3000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "p2 alt b=3000")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: LP projection + sonic + frequency flatten (25% time)
    # ============================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Specialized methods ---")

    # LP projection step
    if time.monotonic() < phase3_deadline:
        a_lp = lp_projection_step(best_a, n_iters=50)
        # Follow up with LSE
        sc, a_opt = run_lbfgs(a_lp, 2000, maxiter=4000, maxfun=60000)
        report(sc, a_opt, "p3 LP+LSE")

    # Sonic peak shaving
    if time.monotonic() < phase3_deadline:
        a_sonic = sonic_peak_shave(best_a, n_iters=300, n_peaks=10, alpha=0.4)
        sc = compute_score(a_sonic)
        report(sc, a_sonic, "p3 sonic")

    # Frequency flattening
    if time.monotonic() < phase3_deadline:
        a_freq = frequency_flatten(best_a, n_iters=300, alpha=0.4)
        sc = compute_score(a_freq)
        report(sc, a_freq, "p3 freq")

    # Combine: run all three and take best, then LSE polish
    if time.monotonic() < phase3_deadline:
        for label, a_try in [("lp2", a_lp), ("sonic2", a_sonic), ("freq2", a_freq)]:
            if time.monotonic() > phase3_deadline:
                break
            sc, a_opt = run_lbfgs(a_try, 5000, maxiter=5000, maxfun=80000)
            report(sc, a_opt, f"p3 {label}+LSE")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Coordinate descent + mirror descent polish (12% time)
    # ============================================================
    phase4_deadline = t_start + 0.92 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Polish ---")

    if time.monotonic() < phase4_deadline:
        a_cd = coordinate_descent_peaks(best_a, n_sweeps=5, n_top=50)
        sc = compute_score(a_cd)
        report(sc, a_cd, "p4 coord")

    if time.monotonic() < phase4_deadline:
        a_md = mirror_descent_v2(best_a, n_iters=1500, eta0=0.03)
        sc = compute_score(a_md)
        report(sc, a_md, "p4 mirror")

    # ============================================================
    # Phase 5: Final high-beta LSE + multi-restart (3% time)
    # ============================================================
    print(f"\n--- Phase 5: Final ---")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        sc, a_opt = run_lbfgs(best_a, 30000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "p5 hi-beta")

    if remaining > 20:
        # Try a few random perturbations + LSE
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"p5 pert{trial}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
