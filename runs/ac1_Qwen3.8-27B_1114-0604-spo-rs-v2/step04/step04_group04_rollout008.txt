"""
Peak-minimization optimizer for non-negative sequences.
Uses CVXPY-exact LP alternating method with LSE-LBFGS polish.

import numpy as np
from scipy.optimize import minimize
import time

# Import cvxpy if available
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def solve_lp_cvxpy(a, M):
    """Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using CVXPY."""
    n = len(a)
    x = cp.Variable(n)
    
    # Build constraint matrix implicitly
    # conv(a, x)[k] = sum_i a[i] * x[k-i] for valid i
    # We need: for all k in 0..2n-2: sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*x[k-i] <= M
    
    # Efficient way: use the structure
    # Actually, let's just build the matrix for moderate n
    if n > 8000:
        return None  # Too large for dense CVXPY
    
    # Build constraint matrix
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a (2n-1) x n matrix
    # Use sparse representation
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    constraints = [A @ x <= M, x >= 0]
    objective = cp.Maximize(cp.sum(x))
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference
    for solver in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.GLPK, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if x.value is not None:
                return np.maximum(x.value, 0)
        except Exception:
            continue
    
    return None


def solve_lp_fista(a, M, n_iters=5000):
    """FISTA-based LP solver (fallback)."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    beta = 50.0
    L = beta * n * sum_a2 + 1e-12
    step = 1.0 / L
    
    for it in range(n_iters):
        # conv(a, x)
        M_conv = 2 * n - 1
        M_fft = 1 << (M_conv - 1).bit_length()
        A_f = np.fft.rfft(a, M_fft)
        X_f = np.fft.rfft(x, M_fft)
        c = np.fft.irfft(A_f * X_f, M_fft)[:M_conv]
        
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        p = exp_z / max(exp_sum, 1e-300)
        
        # C^T p: gradient of sum_k p_k * c_k w.r.t. x
        # = a * sliding_window_sum(p)
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    
    # Gradient of L w.r.t. a
    # L = (1/beta) * log(sum exp(beta * c_k))
    # dL/da_j = sum_k p_k * dc_k/da_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * (p * reverse(a)) at appropriate positions

    M_fft = 1 << (M_conv - 1).bit_length()
    P_f = np.fft.rfft(p, M_fft)
    RA_f = np.fft.rfft(a[::-1], M_fft)
    conv_res = np.fft.irfft(P_f * RA_f, M_fft)[:M_conv]
    
    # We need: for each j, 2 * sum_k p_k * a_{k-j} (if 0 <= k-j < n)
    # This is 2 * (conv of p and reverse(a)) evaluated at positions n-1-j
    # Actually, let me think again...
    # sum_k p_k * a_{k-j}: let m = k-j, so k = m+j
    # = sum_m p_{m+j} * a_m (for valid m and m+j)
    # This is the cross-correlation of p and a, evaluated at lag j
    # = sum_m a_m * p_{m+j}
    
    # Using FFT: this is ifft(conj(FFT(a)) * FFT(p)) at lag j
    # Or equivalently, conv(a, reverse(p)) at position (len(p)-1 - j + ... )
    
    # Let me use a different approach:
    # grad_L[j] = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j]
    # = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    
    # This is the cross-correlation of p and a at lag j
    # = (a ⊛ p)[j] where ⊛ is cross-correlation
    
    # Using the fact that cross-correlation = conv(a, reverse(p)):
    # conv(a, reverse(p))[j + (n-1)] = sum_m a[m] * reverse(p)[j+(n-1)-m] = sum_m a[m] * p[n-1-j+m]
    # Hmm, this is getting confusing. Let me just compute it directly using the sliding sum approach.
    
    # grad_L[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # = 2 * a^T * (sliding window of p starting at position j)
    
    # Precompute cumulative sum of p
    cum_p = np.concatenate([[0.0], np.cumsum(p)])
    # For each j, sum_{m=0}^{n-1} p[j+m] = cum_p[j+n] - cum_p[j]
    window_sums = cum_p[n:len(p)+1] - cum_p[:len(p)-n+1]
    # window_sums[j] = sum_{m=0}^{n-1} p[j+m]
    
    # But we need sum_{m=0}^{n-1} a[m] * p[j+m], not just the sum of p.
    # This is (a * p_shifted_j) = dot(a, p[j:j+n])
    
    # This is the convolution (a * p) evaluated at position j + (n-1)
    # Let's verify: (a * p)[k] = sum_i a[i] * p[k-i]
    # At k = j + n - 1: sum_i a[i] * p[j+n-1-i]
    # Let m = n-1-i: i = n-1-m
    # = sum_m a[n-1-m] * p[j+m]
    
    # Hmm, that's a reversed version. Let me try:
    # (reverse(a) * p)[k] = sum_i reverse(a)[i] * p[k-i] = sum_i a[n-1-i] * p[k-i]
    # At k = j: sum_i a[n-1-i] * p[j-i]
    # Let m = n-1-i: i = n-1-m
    # = sum_m a[m] * p[j-n+1+m]
    
    # I want sum_m a[m] * p[j+m]. So I need j-n+1+m = j+m, which means n-1 = 0, i.e., n=1. Doesn't work.
    
    # Let me try: (a * reverse(p))[k] = sum_i a[i] * reverse(p)[k-i] = sum_i a[i] * p[len(p)-1-k+i]
    # At k = len(p)-1-j: sum_i a[i] * p[len(p)-1-(len(p)-1-j)+i] = sum_i a[i] * p[j+i]
    # Yes! So grad_L[j] = 2 * (a * reverse(p))[len(p)-1-j]

    rp = p[::-1]
    M2 = n + len(p) - 1  # n + 2n-1 - 1 = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_f = np.fft.rfft(a, M2_fft)
    RP_f = np.fft.rfft(rp, M2_fft)
    conv_res = np.fft.irfft(A_f * RP_f, M2_fft)
    
    positions = (2 * n - 2) - np.arange(n)
    grad_L = 2.0 * conv_res[positions]
    
    S2 = S * S
    S3 = S2 * S

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_alternating(a, max_outer=50, time_limit=None, use_cvxpy=True, fista_iters=5000):
    """LP-based alternating method."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        
        # Solve LP
        x = None
        if use_cvxpy and HAS_CVXPY and n <= 8000:
            x = solve_lp_cvxpy(a, M)
        if x is None:
            x = solve_lp_fista(a, M, n_iters=fista_iters)
        
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Line search for mixing parameter
        best_t = 0.1
        best_sc_t = best_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = (1 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_t = compute_score(a_try)
            if sc_t < best_sc_t:
                best_sc_t = sc_t
                best_t = t
        
        a = (1 - best_t) * a + best_t * x
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
    
    return best_a, best_sc


def error_reduction(a, n_iters=100, alpha=0.7):
    """Frequency domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = min(1.0, 0.3 + 0.7 * (it / max(1, n_iters // 4)))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer: CVXPY-exact LP alternating + LSE-LBFGS polish.
