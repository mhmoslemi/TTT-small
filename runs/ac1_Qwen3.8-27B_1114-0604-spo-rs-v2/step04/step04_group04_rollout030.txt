    A = np.fft.rfft(a, M_fft)

"""
Optimized multi-phase search for flat-autocorrelation non-negative sequences.
Uses improved FISTA-based LP alternating with adaptive beta, exact Lipschitz
constants, periodic restarts, spectral refinement, and L-BFGS polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _fft_size(m):
    return 1 << (m - 1).bit_length()


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    fs = _fft_size(M)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    fs = _fft_size(M)
    A = np.fft.rfft(a, fs)
    B = np.fft.rfft(b, fs)
    return np.fft.irfft(A * B, fs)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def conv_spectral_norm(a, n_fft):
    """Spectral norm of the convolution operator x -> conv(a, x)."""
    A = np.fft.rfft(a, n_fft)
    return float(np.max(np.abs(A)))


def fista_lp(a, M_val, n_iters=5000, beta_start=10.0, beta_end=500.0,
             time_limit=None, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M_val, b >= 0 via FISTA."""
    n = len(a)
    M_total = 2 * n - 1
    n_fft = _fft_size(4 * M_total)
    A_norm = conv_spectral_norm(a, n_fft)
    
    b = np.ones(n) / n
    u = b.copy()
    t_mom = 1.0
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        frac = min(1.0, it / max(1, int(n_iters * 0.4)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        L = beta * A_norm * A_norm + 1e-10
        step = 1.0 / L
        
        c = cross_conv(a, b)
        z = beta * (c - M_val)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        # (C^T p)[j] = a[j] * sum_{k=j}^{j+n-1} p[k]
        cum = np.empty(len(c) + 1)
        cum[0] = 0.0
        np.cumsum(p, out=cum[1:])
        sw = cum[n:] - cum[:len(c) - n + 1]
        grad = a * sw - 1.0
        
        v = u - step * grad
        b_new = np.maximum(v, 0.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        u = b_new + (t_mom - 1.0) / t_new * (b_new - b)
        u = np.maximum(u, 0.0)
        b = b_new
        t_mom = t_new
        
        if verbose and it % 1000 == 0:
            print(f"    FISTA iter {it}: beta={beta:.1f}, sum(b)={b.sum():.6f}")
    
    return b


def lp_alternating(a, max_outer=100, fista_iters=5000,
                   beta_start=10.0, beta_end=500.0,
                   t_start=0.3, t_min=0.03, time_limit=None, verbose=False):
    """LP-based alternating method with FISTA sub-solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        x = fista_lp(a, M, n_iters=fista_iters,
                     beta_start=beta_start, beta_end=beta_end,
                     time_limit=time_limit)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        t = max(t_min, t_start * (1.0 - 0.5 * outer / max(1, max_outer)))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
            if verbose:
                print(f"  LP outer {outer}: score={sc:.8f} (improved)")
        else:
            no_improve += 1
            if no_improve >= 20:
                t = max(0.01, t * 0.5)
                if t < 0.005:
                    break
                no_improve = 0
        
        a = a_new
    
    return best_a, best_sc


def spectral_refine(a, n_iters=200, gamma=0.5, time_limit=None):
    """Spectral domain refinement to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M = 4 * n
    M_fft = _fft_size(M)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        c = self_conv(a)
        
        c_mean = float(c.mean())
        c_max = float(c.max())
        if c_max - c_mean < 1e-14:
            break
        
        # Modify spectrum to reduce peaks
        # Target: reduce values above the mean
        excess = np.maximum(c - c_mean, 0.0)
        c_mod = c - gamma * excess
        
        # Ensure c_mod is PSD (non-negative FFT)
        c_mod_pad = np.zeros(M_fft)
        c_mod_pad[:len(c)] = c_mod
        C_mod = np.fft.rfft(c_mod_pad)
        
        # Phase-preserving update: A_new = A * sqrt(C_mod / |A|^2)
        A_abs2 = A.real * A.real + A.imag * A.imag
        ratio = np.sqrt(np.maximum(C_mod.real, 0.0) / np.maximum(A_abs2, 1e-30))
        A_new = A * ratio
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a
