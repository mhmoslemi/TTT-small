```python
L-BFGS continuation + peak-aware optimization for minimum-crest-factor convolution.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences a.

Key innovations:
1. L-BFGS continuation with beta from 10 to 100000 (very tight max approximation)
2. Peak-aware coordinate descent with golden section search

4. "Anti-peak" perturbation for local optimum escaping

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * np.max(c)) / (S * S)


def lse_smooth(a, beta):
    """
    Compute LSE-smoothed objective and gradient.
    Objective: 2n * LSE(conv(a,a), beta) / sum(a)^2
    where LSE(x, beta) = (1/beta) * log(sum(exp(beta * x)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # For numerical stability, subtract max before exp
    beta_c = beta * (c - c_max)
    exp_vals = np.exp(np.clip(beta_c, -500, 500))
    log_sum_exp = c_max + float(np.log(exp_vals.sum())) / beta
    S2 = S * S
    S3 = S2 * S
```

Score = `2n * L / S^2` where `L = LSE(c, beta)` and `S = sum(a)`.

`dScore/da[j] = 2n * (dL/da[j] * S^2 - L * 2*S * dS/da[j]) / S^4`
`= 2n * (dL/da[j] / S^2 - 2*L / S^3)`

And `dL/da[j] = dLSE/da[j]`.

Now, `LSE(c, beta) = (1/beta) * log(sum_k exp(beta * c[k]))`.

`dLSE/da[j] = (1/beta) * sum_k exp(beta*c[k]) / sum_k exp(beta*c[k]) * d(c[k])/da[j]`
`= sum_k p[k] * d(c[k])/da[j]`

where `p[k] = exp(beta*c[k]) / sum_k exp(beta*c[k])`.

And `c[k] = conv(a,a)[k] = sum_j a[j] * a[k-j]`.

`d(c[k])/da[j] = a[k-j]` if `0 <= k-j < n`, else 0.

So `dLSE/da[j] = sum_k p[k] * a[k-j]` (where the sum is over k such that `0 <= k-j < n`, i.e., `j <= k < j+n`).

This is what I computed as `g[j]` above. Good.

Now let me also think about an alternative: instead of the log-sum-exp, I could use the **power mean**:
`PM_p(c) = (1/m * sum_k c[k]^p)^(1/p)` where `m = 2n-1`.

As `p → ∞`, this approaches `max(c)`. For finite p, it's a smooth approximation.

The gradient:
`dPM_p/da[j] = (1/m * sum_k c[k]^p)^(1/p - 1) * (1/m) * p * sum_k c[k]^(p-1) * d(c[k])/da[j]`

```python
"""
L-BFGS continuation + peak-aware optimization for minimum-crest-factor self-convolution.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Load parent sequence as starting point
2. L-BFGS continuation: beta from 20 to 50000, doubling each round
3. Peak-aware coordinate descent for fine-tuning
4. Fienup with cosine-annealed threshold for global exploration
5. Final L-BFGS polish with very high beta
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    log_sum = c_max + float(np.log(exp_scaled.sum())) / beta
    S2 = S * S
    S3 = S2 * S
    p = exp_scaled / exp_scaled.sum()
    # Compute g[j] = sum_m x[m] * p[j+m] for j=0..n-1
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    p_rev = p[::-1]
    X = np.fft.rfft(x, M2_fft)
    PR = np.fft.rfft(p_rev, M2_fft)
    conv_xp = np.fft.irfft(X * PR, M2_fft)
    g = conv_xp[n-1:2*n-1][::-1]
    obj = 2.0 * n * log_sum / S2
    grad = 2.0 * n * (g / S2 - 2.0 * log_sum / S3)
    return float(obj), grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n)
    bounds = [(0.0, None)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def fienup_cosine(a, n_iters=200, strength=0.5):
    """Fienup with cosine-annealed threshold."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Cosine annealing: start aggressive, end gentle
        t = it / n_iters
        anim = 0.5 * (1 + np.cos(np.pi * t))  # 1 -> 0
        thresh_mult = 1.0 + 0.5 * anim  # 1.5 -> 1.0
        
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        thresh = P_mean * thresh_mult
        above = P > thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (strength * anim * 0.5 + 0.05)
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def fienup_quantile(a, n_iters=200, top_frac_start=0.5, top_frac_end=0.2, strength=0.7):
    """Fienup targeting top fraction of power spectrum bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / n_iters
        top_frac = top_frac_start + (top_frac_end - top_frac_start) * t
        
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        thresh = np.quantile(P, 1.0 - top_frac)
        
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            s = (thresh / P[above]) ** strength
            scale[above] = s * 0.99
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def peak_shave(a, n_rounds=50, n_peaks=5, reduce_frac=0.1):
    """
    Direct peak shaving: identify peaks of conv(a,a), reduce contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rd in range(n_rounds):
        c = fast_conv(a)
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        M_val = float(c.max())
        
        # Build subgradient from top peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end+1]
                sg[j_start:j_end+1] += seg[::-1]
        
        # Normalize subgradient
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break
        
        # Step size: proportional to gap between current and theoretical minimum
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        sc_cur = compute_score(a)
        gap = max(sc_cur - sc_lb, 1e-6)
        step = reduce_frac * gap / (sg_norm / S + 1e-15)
        step = min(step, 0.1)
        
        a_new = a - step * sg
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def anti_peak_perturb(a, rng, n_peaks=3, magnitude=0.05):
    """
    Perturb elements that contribute LESS to the peak, to escape local optima.
    """
    n = len(a)
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:n_peaks]
    
    # Find elements that contribute most to peaks
    peak_contrib = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end+1]
            peak_contrib[j_start:j_end+1] += seg[::-1]
    
    # Perturb elements with LOW peak contribution
    contrib_thresh = float(np.median(peak_contrib))
    low_mask = peak_contrib < contrib_thresh
    
    a_new = a.copy()
    if low_mask.any():
        n_low = low_mask.sum()
        # Add random perturbation to low-contribution elements
        pert = rng.standard_normal(n) * magnitude
        pert[~low_mask] = 0  # only perturb low-contrib elements
        a_new = a + pert
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S > 1e-12:
            a_new = a_new / S
    return a_new


def coordinate_descent_conv(a, n_sweeps=3, n_top=40):
    """
    Coordinate descent: try to reduce each top-contributing element.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sw in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end+1]
        contrib = seg[::-1].copy()
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local
        
        cur_sc = best_sc
        for j in top_global:
            j = int(j)
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing
            best_val = a_old
            best_local = cur_sc
            for frac in [0.5, 0.7, 0.8, 0.9]:
                candidate = a_old * frac
                a[j] = candidate
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_tmp = a / S_tmp
                sc = compute_score(a_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = candidate
                a[j] = a_old
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
```

```python
"""
L-BFGS continuation with beta schedule + peak shaving + Fienup.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and its gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -700, 700)
    exp_vals = np.exp(scaled)
    L = c_max + float(np.log(exp_vals.sum())) / beta
    S2 = S * S
    S3 = S2 * S
    p = exp_vals / exp_vals.sum()
    
    # g[j] = sum_m x[m] * p[j+m]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    p_rev = p[::-1]
    X = np.fft.rfft(x, M2_fft)
    PR = np.fft.rfft(p_rev, M2_fft)
    conv_xp = np.fft.irfft(X * PR, M2_fft)
    g = conv_xp[n-1:2*n-1][::-1]
    
    obj = 2.0 * n * L / S2
    grad = 2.0 * n * (2.0 * g / S2 - 2.0 * L / S3)
    return float(obj), grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, None)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum()

 < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def fienup_v2(a, n_iters, strength):
    """Fienup with linearly decreasing threshold."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh_mult = 1.3 - 0.3 * t  # 1.3 -> 1.0
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        thresh = P_mean * thresh_mult
        above = P > thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (strength * 0.5)
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def peak_shave(a, n_rounds, n_peaks, step_frac):
    """Subgradient peak shaving."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for rd in range(n_rounds):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += a[js:je+1][::-1]
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break
        sc_cur = 2.0 * n * float(c.max()) / (S * S)
        gap = max(sc_cur - sc_lb, 1e-8)
        step = step_frac * gap * S / sg_norm
        step = min(step, 0.05 * S)
        a_new = a - step * sg / S
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def coord_desc(a, n_sweeps, n_top):
    """Coordinate descent on peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sw in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je <= js:
            break
        seg = a[js:je+1]
        contrib = seg[::-1].copy()
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = js + top_local
        for j in top_global:
            j = int(j)
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_val = a_old
            best_local = best_sc
            for frac in [0.5, 0.7, 0.85]:
                a[j] = a_old * frac
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * frac
                a[j] = a_old
            a[j] = best_val
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
```

```python
"""
L-BFGS continuation with beta schedule + peak shaving + Fienup + coordinate descent.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences a.

Key approach:
1. Load parent sequence as warm start
2. L-BFGS continuation: beta from 10 to 80000 (progressively tighter max approximation)
3. Peak shaving via subgradient on top-M convolution peaks
4. Fienup-style spectral shaping with annealed threshold
5. Coordinate descent on peak-contributing elements
6. Final high-beta L-BFGS polish
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and gradient.
    Objective: 2n * LSE(conv(x,x), beta) / sum(x)^2
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -700, 700)
    exp_vals = np.exp(scaled)
    L = c_max + float(np.log(exp_vals.sum())) / beta
    S2 = S * S
    S3 = S2 * S
    p = exp_vals / exp_vals.sum()

    # g[j] = sum_m x[m] * p[j+m] for j=0..n-1
    # Computed as conv(x, p_reversed)[n-1:2n-1][::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    p_rev = p[::-1]
    X = np.fft.rfft(x, M2_fft)
    PR = np.fft.rfft(p_rev, M2_fft)
    conv_xp = np.fft.irfft(X * PR, M2_fft)
    g = conv_xp[n - 1:2 * n - 1][::-1]

    obj = 2.0 * n * L / S2
    # dc[k]/dx[j] = 2*x[k-j] if 0<=k-j<n else 0
    # dLSE/dx[j] = 2 * sum_m x[m] * p[j+m] = 2*g[j]
    # dScore/dx[j] = 2n * (2*g[j]/S^2 - 2*L/S^3)
    grad = 2.0 * n * (2.0 * g / S2 - 2.0 * L / S3)
    return float(obj), grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, None)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def fienup_v2(a, n_iters, strength, thresh_start=1.4, thresh_end=1.0):
    """Fienup with linearly annealed threshold multiplier."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        thresh_mult = thresh_start + (thresh_end - thresh_start) * t
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * thresh_mult
        above = P > thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (strength * 0.5)
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_quantile(a, n_iters, top_start, top_end, strength):
    """Fienup targeting top fraction of power spectrum bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        top_frac = top_start + (top_end - top_start) * t
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        thresh = np.quantile(P, 1.0 - top_frac)
        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            scale[above] = (thresh / P[above]) ** strength * 0.99
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def peak_shave(a, n_rounds, n_peaks, step_init):
    """Subgradient peak shaving with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    step = step_init
    for rd in range(n_rounds):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += a[js:je + 1][::-1]
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break
        sc_cur = 2.0 * n * float(c.max()) / (S * S)
        gap = max(sc_cur - sc_lb, 1e-8)
        step_use = min(step * gap, 0.08)
        update = step_use * sg / (sg_norm / S)
        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            step *= 0.95
    return best_a


def coord_desc(a, n_sweeps, n_top):
    """Coordinate descent: reduce top-contributing elements to peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sw in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je <= js:
            break
        seg = a[js:je + 1]
        contrib = seg[::-1].copy()
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = js + top_local
        for j in top_global:
            j = int(j)
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_val = a_old
            best_local = best_sc
            for frac in [0.5, 0.7, 0.85]:
                a[j] = a_old * frac
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * frac
                a[j] = a_old
            a[j] = best_val
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def perturb_and_reopt(a, rng, magnitude, beta):
    """Perturb and re-optimize with L-BFGS."""
    n = len(a)
    a_pert = a * (1.0 + magnitude * rng.standard_normal(n))
    a_pert = np.maximum(a_pert, 0.0)
    S = float(a_pert.sum())
    if S < 1e-12:
        return a
    a_pert = a_pert / S
    sc, a_opt = run_lbfgs(a_pert, beta, 2000, 30000)
    return a_opt

            print(f"  final_lbfgs(beta={beta}): {sc:.8f} *** NEW BEST *** ({elapsed:.1f}s)")
        else:
            print(f"  final_lbfgs(beta={beta}): {sc:.8f} ({elapsed:.1f}s)")

    # Very final peak shave
    if time.monotonic() < deadline - 15:
        a_ps = peak_shave(best_a, n_rounds=20, n_peaks=5, step_init=0.02)
        sc = compute_score(a_ps)
        if sc < best_score:
            best_score = sc
            best_a = a_ps.copy()
            print(f"  final_peak_shave: {sc:.8f} *** NEW BEST ***")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, x)) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
