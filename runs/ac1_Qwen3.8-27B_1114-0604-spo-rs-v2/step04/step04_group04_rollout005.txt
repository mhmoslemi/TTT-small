def lp_alternating(a, max_outer_iters, fista_iters, beta, t_start, t_min, time_limit):
    """LP-based alternating method with FISTA sub-solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters, beta=beta)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Adaptive t: geometric decay
        frac = outer / max(1, max_outer_iters - 1)
        t = t_start * (t_min / t_start) ** frac if t_start > 0 else t_min
        
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # Try smaller t
            t_half = t * 0.5
            a_try = (1.0 - t_half) * a + t_half * x
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try > 1e-12:
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    continue
            # No improvement, reduce t for next iteration
            a = a  # keep old a
        
        a = a_new
    
    return best_a, best_sc


def frequency_domain_equalize(a, n_iters=100, alpha=0.5):
    """Newton-like frequency domain equalization to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        V = A * A
        conv = np.fft.ifft(V).real[:2*n-1]
        
        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-15:
            break
        
        # Target: flatten toward mean
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2*n-1] = err
        err_spec = np.fft.fft(err_pad)
        
        # Newton step: dA = err_spec / (2A)
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = err_spec / (2.0 * A + 1e-15)
        dA[np.abs(A) < 1e-8] = 0.0
        
        # Damping: start aggressive, become conservative
        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * dA
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - 1.4) / (gn * gn + 1e-30)
        step = min(step, step_size)
        step = max(step, 0.001)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay step size
        if it % 500 == 499:
            step_size *= 0.7
    
    return best_a
