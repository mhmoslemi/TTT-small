```python
"""
High-precision LSE-smoothed L-BFGS optimizer with adaptive peak targeting.
Uses very high beta values for near-exact max approximation, combined with
iterated local search and targeted peak reduction for final improvement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """
    LSE-smoothed score with gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: clip to avoid underflow issues
    scaled = np.clip(scaled, -700, 0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(c.max() > c)] = 1.0  # fallback
    else:
        p = exp_scaled / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    
    return score, grad


def lse_topk_objective_and_grad(a, beta, k_frac):
    """
    LSE-smoothed score restricted to top fraction of peaks.
    k_frac: fraction of peaks to include (e.g., 0.01 for top 1%)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # Include peaks within a threshold of the max
    threshold = c_max * (1.0 - k_frac)
    mask = c >= threshold
    
    if mask.sum() == 0:
        mask = np.zeros(len(c), dtype=bool)
        mask[int(np.argmax(c))] = True
    
    c_sel = c[mask]
    c_sel_max = float(c_sel.max())
    scaled = beta * (c_sel - c_sel_max)
    scaled = np.clip(scaled, -700, 0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p_sel = exp_scaled / exp_sum
    
    # Build full p vector
    p = np.zeros(len(c))
    p[mask] = p_sel
    
    L = c_sel_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
```

```python
"""
High-precision LSE-smoothed L-BFGS optimizer with continuation method.
Uses very high beta values for near-exact max approximation, adaptive top-k
for plateau detection, iterated local search, and length exploration.
Key improvements: much higher beta schedule, tighter convergence, adaptive
peak targeting, and multi-scale refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    out = np.fft.irfft(A * A, M_fft)[:M]
    return out


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient (full convolution)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.where(scaled > -700, scaled, -700)
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    p = exp_s / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2f = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2f)
    P = np.fft.rfft(p, M2f)
    conv_res = np.fft.irfft(RA * P, M2f)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_topk_obj_grad(a, beta, k):
    """LSE-smoothed score restricted to top-k peaks for plateau handling."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    k = min(k, len(c))
    topk_vals = np.sort(c)[::-1][:k]
    threshold = topk_vals[-1]
    mask = c >= threshold - 1e-15
    if mask.sum() < 1:
        mask = np.zeros(len(c), dtype=bool)
        mask[int(np.argmax(c))] = True
    c_sel = c[mask]
    c_sel_max = float(c_sel.max())
    scaled = beta * (c_sel - c_sel_max)
    scaled = np.where(scaled > -700, scaled, -700)
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    p_sel = exp_s / exp_sum
    p = np.zeros(len(c))
    p[mask] = p_sel
    L = c_sel_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2f = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2f)
    P = np.fft.rfft(p, M2f)
    conv_res = np.fft.irfft(RA * P, M2f)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, topk=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def func(x):
        if topk is not None:
            return lse_topk_obj_grad(x, beta, topk)
        else:
            return lse_obj_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-20,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 500}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def detect_plateau(c, rel_tol=0.001):
    """Detect how many peaks are within rel_tol of the maximum."""
    c_max = float(c.max())
    if c_max <= 0:
        return 1
    threshold = c_max * (1.0 - rel_tol)
    count = int(np.sum(c >= threshold))
    return max(1, count)


def targeted_peak_fix(a, n_iters=2000):
    """
    Targeted coordinate descent on the peak contributors.
    For the current peak position, reduce the largest contributing pairs.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        Sc = compute_score(a)
        if Sc < best_sc:
            best_sc = Sc
            best_a = a.copy()

        # Get contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        # Contribution of a[i] to peak is a[i] * a[k_star - i]
        paired = a[k_star - j_end:k_star - j_start + 1][::-1]
        contrib = seg * paired

        # Find top contributor
        top_local = int(np.argmax(contrib))
        top_global = j_start + top_local
        a_top = a[top_global]
        a_partner = a[k_star - top_global]

        if a_top < 1e-12 and a_partner < 1e-12:
            break

        # Try reducing the larger of the pair
        if a_top >= a_partner:
            j_reduce = top_global
        else:
            j_reduce = k_star - top_global

        a_red = a[j_reduce]
        if a_red < 1e-12:
            break

        # Try various reduction fractions
        improved = False
        for frac in [0.5, 0.3, 0.1, 0.05, 0.01]:
            new_val = a_red * (1.0 - frac)
            a[j_reduce] = new_val
            reduction = a_red - new_val
            # Redistribute to the median contributor
            med_local = int(np.argsort(contrib)[len(contrib) // 2])
            a[j_start + med_local] += reduction
            a[j_reduce] = new_val - reduction  # compensate
            a[j_reduce] = new_val
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new
            sc_new = compute_score(a)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
                improved = True
                break
            else:
                # Revert
                a = best_a.copy()
        
        if not improved:
            # Try a different approach: equalize top 3 contributors
            top3_local = np.argsort(contrib)[::-1][:3]
            a_before = a.copy()
            total_top3 = sum(a[j_start + li] for li in top3_local)
            avg = total_top3 / 3.0
            for li in top3_local:
                a[j_start + li] = avg
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new
            sc_new = compute_score(a)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
            else:
                a = a_before.copy()
                break

    return best_a


def perturb_and_refine(a, rng, beta, maxiter, maxfun, scale=0.01):
    """Generate perturbation, refine with L-BFGS, return best."""
    n = len(a)
    strategy = rng.integers(0, 4)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 2.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Targeted: nudge peak reducers
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        paired = a_p[k_star - j_end:k_star - j_start + 1][::-1]
        contrib = seg * paired
        top_count = max(10, n // 100)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            if len(bottom_local) > 0:
                per = deficit / len(bottom_local)
                for li in bottom_local:
                    a_p[j_start + li] += per
    else:
        # Random block rescale
        n_blocks = 5 + int(rng.integers(0, 10))
        bs = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // bs)):
            s = b * bs
            e = min(s + bs, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S

    sc, a_opt, _ = run_lbfgs(a_p, beta, maxiter=maxiter, maxfun=maxfun)
    return sc, a_opt
```
