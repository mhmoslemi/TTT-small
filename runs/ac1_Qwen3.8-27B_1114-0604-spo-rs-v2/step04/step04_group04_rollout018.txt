def cross_conv(a, b):

    n = len(a)

    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

def conv_T(a, y):
    """Compute C^T * y where C is convolution matrix for a, y has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    M = n + len(y) - 1

    A = np.fft.rfft(a_rev, M_fft)
    Y = np.fft.rfft(y, M_fft)
    result = np.fft.irfft(A * Y, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_z / exp_sum
    log_sum = z.max() - z_shift.max() + np.log(exp_sum)
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def lse_objective_full(a, beta):
    """Wrapper returning (value, grad) for scipy."""
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=300000):
    """Run L-BFGS-B on LSE-smoothed objective with projection."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    res = minimize(

        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)

def entropy_lp_solve(a, M, eps, n_dual_iters=2000, warm_start_y=None):
    """
    Solve: max 1^T x s.t. Cx <= M, x >= 0
    using entropy-regularized dual ascent.
    x*(y) = exp(eps*(1 - C^T y) - 1)
    Dual update: y += eta * (C x* - M), projected to y >= 0
    Returns x* at the final dual iterate.
    """
    n = len(a)
    if warm_start_y is not None:
        y = warm_start_y.copy()
    else:
        y = np.zeros(2 * n - 1)

    # Compute ||C||^2 for step size: max eigenvalue of C^T C
    # C^T C is Toeplitz with first row = autocorrelation of a
    # lambda_max(C^T C) = max_theta |FFT(a)(theta)|^2
    M_fft = 1 << (2 * n - 1)
    A_spec = np.fft.rfft(a, M_fft)
    norm_C_sq = float(np.max(np.abs(A_spec) ** 2))
    if norm_C_sq < 1e-15:
        norm_C_sq = 1.0

    eta = eps / (norm_C_sq * 1.5 + 1e-12)

    x = np.ones(n) / n  # initial feasible guess
    best_x = x.copy()
    best_obj = -float(x.sum())

    for it in range(n_dual_iters):
        # Compute C^T y
        Cty = conv_T(a, y)
        # Compute x* = exp(eps*(1 - C^T y) - 1)
        log_x = eps * (1.0 - Cty) - 1.0
        log_x = np.clip(log_x, -50.0, 50.0)  # prevent overflow
        x = np.exp(log_x)
        x = np.maximum(x, 0.0)

        obj = -float(x.sum())
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()

        # Dual gradient: C x* - M
        Cx = cross_conv(a, x)
        dual_grad = Cx - M

        # Backtracking to ensure dual improves
        step = eta
        for _bt in range(20):
            y_new = np.maximum(y + step * dual_grad, 0.0)
            Cty_new = conv_T(a, y_new)
            x_new = np.exp(np.clip(eps * (1.0 - Cty_new) - 1.0, -50.0, 50.0))
            obj_new = -float(x_new.sum())
            if obj_new < obj or step < 1e-10:
                break
            step *= 0.5
        else:
            y_new = np.maximum(y + step * dual_grad, 0.0)

        y = y_new

        # Check convergence
        if it > 50 and it % 50 == 0:
            viol = float(np.max(np.maximum(Cx - M, 0.0)))
            if viol < 1e-10 * M:
                break

    return x, y


def entropy_lp_alternating(a, max_outer=50, eps_schedule=None, t_start=0.4, t_min=0.05,
                           dual_iters=2000, time_limit=None):
    """LP alternating with entropy-regularized sub-solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    if eps_schedule is None:
        eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    best_a = a.copy()
    best_sc = compute_score(a)
    warm_y = None
    no_improve = 0

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        eps = eps_schedule[min(outer, len(eps_schedule) - 1)]

        x, warm_y = entropy_lp_solve(a, M, eps, n_dual_iters=dual_iters, warm_start_y=warm_y)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        t = max(t_min, t_start * (1.0 - 0.5 * outer / max(1, max_outer)))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                t *= 0.5
                if t < 0.01:
                    break
                no_improve = 0
            y = self_conv(a)
        a = a_new

    return best_a, best_sc

    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    L = beta * n * sum_a2 + 1e-12
    step = 1.0 / L

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_mom  # keep for simplicity (use standard FISTA)
        t_mom = t_new
    return x


def subgradient_polish(a, total_iters=5000, step_scale=0.5):
    """Subgradient descent on the exact max-conv objective with normalization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max c_k: gradient of c_{k*}
        # (grad c_{k*})[m] = 2 * a[k*-m] if 0 <= k*-m < n
        sg = np.zeros(n)
        for m in range(max(0, k_star - n + 1), min(k_star, n - 1) + 1):
            idx = k_star - m
            if 0 <= idx < n:
                sg[m] = 2.0 * a[idx]
        # Account for the 1/S^2 normalization
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = step_scale * (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a, best_sc


def error_reduction(a, n_iters=100, alpha=0.7):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Smooth noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    # Piecewise constant (step-like)
    n_blocks = max(4, n // 200)
    block_vals = np.clip(0.5 + 0.3 * rng.standard_normal(n_blocks), 0.1, 1.0)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses entropy-regularized LP alternating + aggressive LSE L-BFGS + subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s))
    deadline = t_start + effective_budget - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Primary length: parent length or 6962
    n_primary = len(parent) if parent is not None and 1000 <= len(parent) <= 12000 else 6962
    lengths = [n_primary]
    for n_try in [4096, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))

    print(f"Lengths: {lengths}, primary: {n_primary}")

    # ============================================================
    # Phase 0: Quick LSE L-BFGS on parent (if available)
    # ============================================================
    phase0_deadline = t_start + min(120.0, 0.05 * effective_budget)
    print(f"\n--- Phase 0: Quick LSE on parent ---")

    if parent is not None:
        a_p = parent.copy()
        S = float(a_p.sum())
        if S > 1e-12:
            a_p = a_p / S
        for beta in [50, 200]:
            if time.monotonic() > phase0_deadline:
                break
            sc, a_opt, _ = run_lbfgs(a_p, beta, maxiter=3000, maxfun=100000)
            report(sc, a_opt, f"Q-LBFGS b={beta}")
            a_p = a_opt.copy()
    print(f"Phase 0 done: best={best_score:.8f}")

    # ============================================================
    # Phase 1: Error reduction initialization (5% time)
    # ============================================================
    phase1_deadline = t_start + 0.10 * effective_budget
    print(f"\n--- Phase 1: Error Reduction Init ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        inits = build_initializations(n, rng, parent)
        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=80, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: LP Alternating with entropy-regularized solver (45% time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * effective_budget
    print(f"\n--- Phase 2: LP Alternating (Entropy-Reg) ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Multi-round with decreasing epsilon
    for round_idx in range(5):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        eps_sched = [0.3, 0.1, 0.05, 0.02, 0.01][:max(1, 5 - round_idx)]
        max_outer = max(3, int(remaining / max(2, n_main / 5000.0)))
        dual_it = max(500, min(3000, int(remaining * 50)))

        print(f"  LP round {round_idx}: n={n_main}, outer~{max_outer}, dual_it~{dual_it}, eps={eps_sched}")
        a_out, sc_out = entropy_lp_alternating(
            a_work, max_outer=max_outer, eps_schedule=eps_sched,
            t_start=0.3, t_min=0.03, dual_iters=dual_it,
            time_limit=phase2_deadline - 5
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP-ent r{round_idx}")
        a_work = a_out.copy()

    # Try other lengths briefly
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_deadline - 30:
            break
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = entropy_lp_alternating(
            a_test, max_outer=15, eps_schedule=[0.2, 0.1, 0.05],
            t_start=0.3, t_min=0.05, dual_iters=1000,
            time_limit=phase2_deadline - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Aggressive LSE L-BFGS continuation (25% time)
    # ============================================================
    phase3_deadline = t_start + 0.82 * effective_budget
    print(f"\n--- Phase 3: LSE L-BFGS Continuation ---")

    n_best = len(best_a)
    a_curr = best_a.copy()
    S = float(a_curr.sum())
    if S > 1e-12:
        a_curr = a_curr / S

    beta_schedule = [50, 150, 400, 1000, 2500, 5000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 15:
            break
        maxit = max(2000, int(remaining * 20))
        maxfu = max(50000, int(remaining * 1000))
        print(f"  LSE beta={beta}, maxiter={maxit}")
        sc, a_opt, _ = run_lbfgs(a_curr, beta, maxiter=maxit, maxfun=maxfu)
        report(sc, a_opt, f"LSE b={beta}")
        a_curr = a_opt.copy()

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (remaining time)
    # ============================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # Subgradient on exact max
    a_sp, sc_sp = subgradient_polish(best_a, total_iters=5000, step_scale=0.3)
    report(sc_sp, a_sp, "subgrad")

    # One more LSE with very high beta
    if time.monotonic() < deadline - 30:
        a_curr = best_a.copy()
        S = float(a_curr.sum())
        if S > 1e-12:
            a_curr = a_curr / S
        sc, a_opt, _ = run_lbfgs(a_curr, 8000, maxiter=4000, maxfun=100000)
        report(sc, a_opt, "final-LSE")

    # Targeted peak reduction
    if time.monotonic() < deadline - 20:
        n_b = len(best_a)
        a_t = best_a.copy()
        S = float(a_t.sum())
        if S > 1e-12:
            a_t = a_t / S
        c = self_conv(a_t)
        k_star = int(np.argmax(c))
        # Reduce elements that contribute most to the peak
        j_start = max(0, k_star - n_b + 1)
        j_end = min(k_star, n_b - 1)
        contrib = np.zeros(j_end - j_start + 1)
        for j in range(j_start, j_end + 1):
            idx = k_star - j
            if 0 <= idx < n_b:
                contrib[j - j_start] = a_t[idx]
        # Reduce top contributors, add to bottom
        n_reduce = max(10, n_b // 200)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        bot_idx = np.argsort(contrib)[:n_reduce]
        total_red = 0.0
        for li in top_idx:
            gi = j_start + li
            red = a_t[gi] * 0.02
            a_t[gi] = max(0.0, a_t[gi] - red)
            total_red += red
        per_add = total_red / len(bot_idx)
        for li in bot_idx:
            a_t[j_start + li] += per_add
        a_t = np.clip(a_t, 0.0, 1000.0)
        S = float(a_t.sum())
        if S > 1e-12:
            a_t = a_t / S
        sc, a_opt, _ = run_lbfgs(a_t, 2000, maxiter=3000, maxfun=80000)
        report(sc, a_opt, "peak-peak")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
