"""
Adam-based optimizer with LSE curriculum for flat-autocorrelation non-negative sequences.
Key innovations: per-coordinate adaptive learning rates (Adam), smooth beta curriculum
for progressive sharpening, simplex exploitation via scale invariance, multi-scale
initialization, and periodic restarts to escape local minima. Combines error reduction
pre-processing with Adam fine-tuning and L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    exp_z = np.exp(z)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
        L_val = c_max
    else:
        p = exp_z / exp_sum
        L_val = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L_val / (S * S)

    rev_a = a[::-1]
    len_c = len(c)
    M2 = n + len_c - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def error_reduction(a, n_iters=200, alpha=0.7, time_limit=None, t_start=None):
    """Newton-like error reduction in frequency domain."""
    t_start = t_start or time.monotonic()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def adam_optimize(a0, beta_schedule, total_lr, beta1=0.9, beta2=0.999,
                  eps=1e-8, time_limit=None, t_start=None, clip_norm=5.0,
                  restart_every=20000, restart_scale=0.02):
    """
    Adam optimization on LSE-smoothed objective with simplex projection.
    beta_schedule: list of (beta, n_iters) tuples.
    """
    t_start = t_start or time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S > 1e-12:
        a = a / S
    else:
        a = np.ones(n) / n

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    check_interval = 100

    for beta, n_iters in beta_schedule:
        for it in range(n_iters):
            t_adam += 1
            if time_limit is not None and time.monotonic() > time_limit:
                break

            # Learning rate: inverse sqrt decay
            lr = total_lr / np.sqrt(1.0 + 0.001 * t_adam)

            sc, grad = lse_score_grad(a, beta)

            # Project onto simplex tangent
            grad_tan = grad - grad.mean()

            # Gradient clipping
            gnorm = float(np.linalg.norm(grad_tan))
            if gnorm > clip_norm:
                grad_tan = grad_tan * (clip_norm / gnorm)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_tan
            v = beta2 * v + (1.0 - beta2) * grad_tan ** 2
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project to simplex
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                a = np.ones(n) / n
                m = np.zeros(n)
                v = np.zeros(n)
            else:
                a = a / S

            # Periodic evaluation
            if (it + 1) % check_interval == 0 or it == n_iters - 1:
                sc_exact = compute_score(a)
                if sc_exact < best_sc - 1e-10:
                    best_sc = sc_exact
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += check_interval

            # Restart if stuck
            if no_improve >= restart_every:
                no_improve = 0
                a = a * (1.0 + restart_scale * np.random.standard_normal(n))
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S > 1e-12:
                    a = a / S
                m = np.zeros(n)
                v = np.zeros(n)

        if time_limit is not None and time.monotonic() > time_limit:
            break

    return best_a, best_sc


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    def obj(x):
        s, g = lse_score_grad(x, beta)
        return s, g

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def subgradient_polish(a, total_iters=5000, time_limit=None, t_start=None):
    """Projected subgradient descent on the exact max-conv objective."""
    t_start = t_start or time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        grad = grad - grad.mean()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min((sc - 1.0) / (gn * gn + 1e-30), 0.03)
        step = max(step, 0.001)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) >= 8:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p = np.interp(x_new, x_old, parent)
        p = np.maximum(p, 0.0)
        S = float(p.sum())
        if S > 1e-12:
            inits.append(p / S)

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    inits.append((0.5 + 0.4 * np.cos(2.0 * np.pi * x)))
    inits.append((0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))

    # Normalize all to simplex
    normalized = []
    for a in inits:
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            normalized.append(a / S)
    return normalized
