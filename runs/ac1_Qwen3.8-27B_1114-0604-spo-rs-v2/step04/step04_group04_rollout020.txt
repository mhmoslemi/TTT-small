from scipy.optimize import minimize

def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    A = np.fft.rfft(a, M_fft)

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def fista_lp_solve(a, M, n_iters=5000, warm_start=None):
    """
    Solve LP: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA with backtracking and momentum restart on the LSE-smoothed surrogate.
    """
    n = len(a)
    if warm_start is not None and len(warm_start) == n:
        x = np.maximum(warm_start.copy(), 0.0)
    else:
        x = np.ones(n) / n
    
    u = x.copy()
    t_mom = 1.0
    
    # Estimate Lipschitz constant
    sum_a2 = float(np.sum(a * a))
    L = 2.0 * n * sum_a2  # upper bound on Lipschitz of the smooth part
    # Use a more careful estimate: the gradient is a * (sliding_window_sum(p))
    # The Lipschitz constant is bounded by ||a||_2^2 * (max possible sum of p in window) * 2
    # But for safety, use the standard bound.
    
    best_x = x.copy()
    best_obj = -float(x.sum())
    
    for it in range(n_iters):
        # Adaptive beta
        frac = min(1.0, it / max(1, n_iters * 0.2))
        beta = 10.0 + (100.0 - 10.0) * frac
        
        # Compute gradient of LSE surrogate
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        
        # Gradient: a * (sliding_window_sum of p) - 1
        cum = np.concatenate([[0.0], np.cumsum(p)])
        p_len = len(p)
        sw = cum[n:] - cum[:p_len - n + 1]
        grad = a * sw - 1.0
        
        # Backtracking line search
        step = 1.0 / L
        c_new = cross_conv(a, x - step * grad)
        z_new = beta * (c_new - M)
        z_new_max = float(z_new.max())
        obj_new = float(z_new_max) + np.log(np.exp(z_new - z_new_max).sum()) / beta
        
        for bt in range(20):
            if obj_new < float(z_max) + np.log(exp_sum) / beta + 1e-10:
                break
            step *= 0.5
            L *= 2.0
            c_new = cross_conv(a, x - step * grad)
            z_new = beta * (c_new - M)
            z_new_max = float(z_new.max())
            obj_new = float(z_new_max) + np.log(np.exp(z_new - z_new_max).sum()) / beta
        
        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        
        # Momentum with restart
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        # Check for restart: if inner product of (x_new - x) and (x - u_prev) is positive, restart
        if it > 0 and np.dot(x_new - x, x - u) > 0:
            t_new = 1.0
        
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
        
        # Track best
        current_obj = -float(x.sum())
        if current_obj < best_obj:
            best_obj = current_obj
            best_x = x.copy()
    
    return best_x


def lp_alternating_v2(a, max_outer=100, fista_iters=5000, t_start=0.5, t_min=0.02,
                      time_limit=None, time_check=10):
    """
    Improved LP-based alternating method.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    warm_x = None
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        
        # Solve LP
        x = fista_lp_solve(a, M, n_iters=fista_iters, warm_start=warm_x)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        warm_x = x.copy()
        
        # Adaptive mixing
        progress = outer / max(1, max_outer)
        t = t_start * (1.0 - progress) + t_min * progress
        # Cosine annealing
        t = t_min + 0.5 * (t_start - t_min) * (1.0 + np.cos(np.pi * outer / max(1, max_outer)))
        
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                t = max(0.01, t * 0.5)
                if t < 0.01:
                    break
                no_improve = 0
        
        a = a_new
    
    return best_a, best_sc


def lbfgs_refine(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def freq_domain_shave(a, n_iters=200, alpha=0.5):
    """
    Frequency-domain peak shaving: iteratively reduce peaks in the autocorrelation
    by modifying the phase of the FFT.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Current autocorrelation
        power = A * np.conj(A)
        conv = np.fft.ifft(power).real[:2*n-1]
        
        conv_max = float(conv.max())
        conv_mean = float(conv[:n].mean())  # focus on the "main" part
        
        if conv_max - conv_mean < 1e-14:
            break
        
        # Reduce peaks: where conv > threshold, reduce power spectrum
        threshold = conv_mean + alpha * (conv_max - conv_mean)
        excess = np.maximum(conv - threshold, 0.0)
        
        # Modify power spectrum to reduce excess
        # The excess in time domain corresponds to adding a negative component in freq domain
        excess_pad = np.zeros(M_fft, dtype=np.complex128)
        excess_pad[:2*n-1] = excess
        excess_spec = np.fft.fft(excess_pad)
        
        # Reduce: new_power = power - lambda * excess_spec
        # But we need to keep power >= 0 (it's a power spectrum)
        # Use: new_A = A * exp(-0.5 * excess_spec / power) approximately
        # Or simply: reduce the magnitude of A where it contributes to peaks
        
        # Simpler approach: directly modify A to reduce |A|^2 at problematic frequencies
        # The contribution of frequency f to conv[k] is |A[f]|^2 * exp(2j*pi*f*k/M)
        # To reduce conv at specific k, we want to reduce |A|^2 at frequencies where
        # the phase alignment causes constructive interference at that k.
        
        # Practical approach: use gradient of max(conv) w.r.t. |A|^2
        # This is complex, so use a simpler heuristic:
        # Reduce the overall magnitude of A proportionally to the excess
        
        # Even simpler: iterate on a in time domain using subgradient
        k_star = int(np.argmax(conv))
        # Subgradient direction
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end+1] = a[j_start:j_end+1][::-1]
        
        # Normalize direction
        d_norm = float(np.linalg.norm(direction))
        if d_norm < 1e-15:
            break
        direction = direction / d_norm
        
        # Step size: Barzilai-Borwein style
        step = 0.1 * (conv_max - conv_mean) / (np.max(a) + 1e-15)
        step = min(step, 0.5 * float(np.min(a[np.arange(n) >= 0])) / (np.max(a) + 1e-15))
        step = min(step, 0.05)
        
        a_new = a - step * direction
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def bb_gradient_descent(a0, total_iters=5000, beta=50):
    """
    Barzilai-Borwein gradient descent on the LSE-smoothed objective.
    More robust than L-BFGS for this non-smooth problem.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S > 1e-12:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    prev_grad = None
    prev_a = None
    
    for it in range(total_iters):
        sc, grad = lse_score_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-12:
            break
        
        # BB step size
        if prev_grad is not None:
            s = a - prev_a
            y = grad - prev_grad
            sy = float(np.dot(s, y))
            yy = float(np.dot(y, y))
            ss = float(np.dot(s, s))
            if yy > 1e-15:
                if ss > 1e-15:
                    step_bb1 = ss / sy if sy > 1e-15 else 1.0 / g_norm
                    step_bb2 = sy / yy
                    # Alternate between the two
                    if it % 2 == 0:
                        step = max(1e-8, min(step_bb1, 10.0 / g_norm))
                    else:
                        step = max(1e-8, min(step_bb2, 10.0 / g_norm))
                else:
                    step = 1.0 / g_norm
            else:
                step = 1.0 / g_norm
        else:
            step = 1.0 / g_norm
        
        step = min(step, 0.1 / g_norm)  # cap the step
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Check for improvement; if not, halve the step
        sc_new = compute_score(a_new)
        while sc_new > sc * (1 + 1e-6) and step > 1e-10:
            step *= 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            sc_new = compute_score(a_new)
        
        prev_grad = grad
        prev_a = a
        a = a_new
    
    return best_a


def build_init_raised_cosine(n):
    """Raised cosine (Hann) window - has good sidelobe properties."""
    x = np.arange(n) / n
    return 0.5 + 0.5 * np.cos(2 * np.pi * x)


def build_init_tukey(n, r=0.5):
    """Tukey window."""
    x = np.arange(n) / n
    a = np.ones(n)
    # Left taper
    left = int(r * n / 2)
    for i in range(left):
        a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / r / n - 1)))
    # Right taper
    for i in range(left):
        a[n - 1 - i] = a[i]
    return a


def build_init_kaiser(n, beta_k=8.0):
    """Kaiser window."""
    x = np.arange(n) / n - 0.5
    x = np.abs(x) * 2
    # Bessel function approximation (use scipy)
    from scipy.special import i0
    a = i0(beta_k * np.sqrt(np.maximum(1 - x*x, 0))) / i0(beta_k)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search for optimal non-negative sequence.
    Strategies: LP-alternating (improved FISTA), BB gradient descent,
    LSE L-BFGS, frequency-domain shaving, multi-resolution.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 30.0
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t0
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    print(f"Parent: {'yes n='+str(len(parent)) if parent is not None else 'no'}")
    
    # Choose lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 12288, 16384]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:5]
    
    # Phase 1: Quick exploration with diverse initializations (15% time)
    p1_end = t0 + 0.15 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Quick Exploration (t < {p1_end-t0:.0f}s) ---")
    
    all_candidates = []
    
    for n in lengths[:3]:
        if time.monotonic() > p1_end:
            break
        inits = [
            build_init_raised_cosine(n),
            build_init_tukey(n, 0.3),
            build_init_tukey(n, 0.7),
            build_init_kaiser(n, 5.0),
            build_init_kaiser(n, 10.0),
            np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1.0),
        ]
        if parent is not None and len(parent) == n:
            inits.insert(0, parent.copy())
        
        for i, a0 in enumerate(inits[:4]):
            if time.monotonic() > p1_end:
                break
            a0 = np.maximum(a0, 0.0)
            S = float(a0.sum())
            if S < 1e-12:
                continue
            a0 = a0 / S
            sc = compute_score(a0)
            all_candidates.append((sc, a0.copy(), n))
            report(sc, a0, f"P1 n={n} i={i}")
    
    # Phase 2: BB gradient descent on best candidates (25% time)
    p2_end = t0 + 0.40 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: BB Gradient Descent (t < {p2_end-t0:.0f}s) ---")
    
    # Sort candidates by score
    all_candidates.sort(key=lambda x: x[0])
    
    seen_lengths = set()
    for sc, a0, n in all_candidates[:5]:
        if time.monotonic() > p2_end:
            break
        if n in seen_lengths and len(seen_lengths) >= 2:
            continue
        seen_lengths.add(n)
        
        print(f"  BB on n={n}, init_score={sc:.6f}")
        for beta in [20, 50, 100]:
            if time.monotonic() > p2_end:
                break
            total_iters = max(500, int((p2_end - time.monotonic()) / max(0.01, n / 1e6)))
            total_iters = min(total_iters, 5000)
            a_out = bb_gradient_descent(a0, total_iters=total_iters, beta=beta)
            sc_out = compute_score(a_out)
            report(sc_out, a_out, f"BB n={n} beta={beta}")
            if sc_out < sc:
                a0 = a_out
    
    # Phase 3: LP alternating on best length (35% time)
    p3_end = t0 + 0.75 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: LP Alternating (t < {p3_end-t0:.0f}s) ---")
    
    n_main = len(best_a) if best_a is not None else 8192
    a_main = best_a.copy() if best_a is not None else np.ones(n_main) * 0.5
    S = float(a_main.sum())
    if S > 1e-12:
        a_main = a_main / S
    
    # Progressive LP alternating with increasing FISTA iterations
    fista_schedules = [
        (3000, 0.4, 0.05, 30),
        (5000, 0.3, 0.03, 40),
        (8000, 0.2, 0.02, 50),
        (10000, 0.15, 0.01, 60),
    ]
    
    for fi, (f_iters, t_s, t_m, max_out) in enumerate(fista_schedules):
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP round {fi}: fista={f_iters}, t=[{t_s},{t_m}], remaining={remaining:.0f}s")
        
        a_out, sc_out = lp_alternating_v2(
            a_main, max_outer=max_out, fista_iters=f_iters,
            t_start=t_s, t_min=t_m, time_limit=p3_end - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_main = a_out.copy()
    
    # Phase 4: L-BFGS refinement (15% time)
    p4_end = t0 + 0.90 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 4: L-BFGS Refinement (t < {p4_end-t0:.0f}s) ---")
    
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S
    
    for beta in [50, 100, 300, 1000]:
        if time.monotonic() > p4_end:
            break
        maxiter = max(2000, int((p4_end - time.monotonic()) / 0.05))
        maxiter = min(maxiter, 8000)
        sc, a_opt = lbfgs_refine(a_work, beta, maxiter=maxiter, maxfun=maxiter*2)
        report(sc, a_opt, f"LBFGS beta={beta}")
        a_work = a_opt.copy()
    
    # Phase 5: Final polish (10% time)
    print(f"\n--- Phase 5: Final Polish ---")
    
    # Multiple short LP runs with very small step
    if time.monotonic() < deadline - 30:
        a_final = best_a.copy()
        S = float(a_final.sum())
        if S > 1e-12
