def fast_conv(a):

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_v2(a, n_iters=400, gamma=1.0, top_frac_init=0.15, top_frac_end=0.40,
              soft_gate_alpha=50.0, sum_preserve=True):
    """
    Improved Fienup with:
    - Adaptive top-fraction that grows over iterations
    - Power-law compression with tunable gamma
    - Sigmoid soft-gate for non-negativity
    - Optional sum preservation (add back clipped mass uniformly)
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Cosine-annealed top fraction
        top_frac = top_frac_init + (top_frac_end - top_frac_init) * 0.5 * (1 - np.cos(np.pi * progress))

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        n_bins = len(P)

        # Determine threshold: top `top_frac` of bins
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            # Power-law compression: bring above-thresh down to thresh
            ratio = thresh / P[above]
            scale[above] = ratio ** 0.5 * gamma
            # If gamma > 1, we actually expand slightly (exploration)
            # If gamma < 1, we compress more aggressively

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]

        # Soft non-negativity: sigmoid gate
        if soft_gate_alpha > 0:
            gate = 1.0 / (1.0 + np.exp(-soft_gate_alpha * a_new))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break

        if sum_preserve:
            # Renormalize to preserve sum
            a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_objective_grad(x, beta):
    """LSE-smoothed objective and gradient for scipy.minimize."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient
    p = exp_scaled / exp_sum  # weights on peaks
    # dL/dx_i = sum_k p_k * (a[k-i]) where a is reversed
    # conv(p, rev_a)[n-1 : 2n-1] gives sum_k p[k] * a[k-i] for i=0..n-1
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res  # dL/dx = 2 * sum_k p_k * x[k-i]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=2000, top_m=15, step_init=0.03, decay=0.9998):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def build_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Half-sine
    inits.append(np.maximum(0.5 * np.sin(np.pi * x) + 0.1, 0.01))
    # Raised cosine
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.5)), 0.01))
    # Gaussian bump (centered)
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.1)
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    # Asymmetric ramp
    inits.append(0.3 + 0.7 * x)
    # Inverse ramp
    inits.append(0.3 + 0.7 * (1.0 - x))
    # Two-peak
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    # Oscillatory
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = [6962, 8192, 12288]
    if parent is not None:
        lengths = [len(parent)] + [l for l in lengths if l != len(parent)]
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ', score=' + str(compute_score(parent)) + ')' if parent is not None else 'no'}")

    # If we have a parent, quickly evaluate it
    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ==================================================================
    # Phase 0: Quick L-BFGS from parent (5% time) - get fast improvement
    # ==================================================================
    phase0_end = t_start + 0.05 * (total_budget - 30.0)
    print(f"\n=== Phase 0: Quick LBFGS from parent ===")
    if parent is not None and time.monotonic() < phase0_end:
        for beta in [50, 200]:
            if time.monotonic() > phase0_end:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"P0-LBFGS b={beta}")

    # ==================================================================
    # Phase 1: Fienup from diverse inits (30% time)
    # ==================================================================
    phase1_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            # Variant 1: Standard Fienup v2
            a_f = fienup_v2(a0, n_iters=200, gamma=0.8, top_frac_init=0.1,
                            top_frac_end=0.5, soft_gate_alpha=30.0)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"FIENUP2 n={n} i={i}")

            # Variant 2: Spectral equalization
            if time.monotonic() < phase1_end:
                a_eq = fienup_spectral_eq(a0, n_iters=150, alpha_init=0.2, alpha_end=0.7)
                sc2 = compute_score(a_eq)
                update_best(sc2, a_eq, f"SPEQ n={n} i={i}")

            # Variant 3: Water-filling
            if time.monotonic() < phase1_end:
                a_wf = fienup_waterfill(a0, n_iters=150, n_levels=6)
                sc3 = compute_score(a_wf)
                update_best(sc3, a_wf, f"WATER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Extended Fienup from best (25% time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Extended Fienup (deadline t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    # Run multiple Fienup variants from best, chaining them
    a_cur = best_a.copy() / best_a.sum()

    chains = [
        ("fienup2_g0.6", lambda x: fienup_v2(x, 300, gamma=0.6, top_frac_init=0.08, top_frac_end=0.6, soft_gate_alpha=20.0)),
        ("fienup2_g1.0", lambda x: fienup_v2(x, 300, gamma=1.0, top_frac_init=0.12, top_frac_end=0.45, soft_gate_alpha=40.0)),
        ("speq", lambda x: fienup_spectral_eq(x, 250, alpha_init=0.3, alpha_end=0.9)),
        ("water", lambda x: fienup_waterfill(x, 250, n_levels=8)),
        ("fienup2_g0.4", lambda x: fienup_v2(x, 300, gamma=0.4, top_frac_init=0.15, top_frac_end=0.7, soft_gate_alpha=10.0)),
    ]

    for name, func in chains:
        if time.monotonic() > phase2_end:
            break
        a_out = func(a_cur)
        sc = compute_score(a_out)
        improved = update_best(sc, a_out, f"P2-{name}")
        if improved:
            a_cur = a_out.copy() / a_out.sum()

    # Also try from a perturbed version
    a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    a_pert = a_pert / a_pert.sum()
    if time.monotonic() < phase2_end:
        a_out = fienup_v2(a_pert, 200, gamma=0.8, top_frac_init=0.1, top_frac_end=0.5)
        sc = compute_score(a_out)
        update_best(sc, a_out, "P2-PERT")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS with increasing beta (25% time)
    # ==================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: LBFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    betas = [100, 300, 500, 1000, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        # Allocate more iterations for lower beta (easier to converge)
        maxiter = min(15000, int(remaining // (len(best_a) / 1000 + 1) * 20))
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            # Re-run with higher beta from new best
            pass

    # L-BFGS from parent at native length if different
    if parent is not None and len(parent) != len(best_a):
        if time.monotonic() < phase3_end:
            for beta in [200, 1000, 3000]:
                if time.monotonic() > phase3_end:
                    break
                sc, a_opt = run_lbfgs(parent, beta, maxiter=8000, maxfun=100000)
                update_best(sc, a_opt, f"LBFGS-PARENT b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient + coordinate refinement (10% time)
    # ==================================================================
    print(f"\n=== Phase 4: Subgradient + Refinement ===")

    for top_m in [5, 10, 20, 40]:
        if time.monotonic() > deadline:
            break
        a_sg = top_m_subgradient(best_a, n_iters=2000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG m={top_m}")

    # Final Fienup + LBFGS combo
    if time.monotonic() < deadline - 60:
        a_final = fienup_v2(best_a, 100, gamma=0.7, top_frac_init=0.1, top_frac_end=0.4)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-FIENUP")

        if time.monotonic() < deadline - 30:
            sc2, a_opt = run_lbfgs(a_final, 5000, maxiter=3000, maxfun=50000)
            update_best(sc2, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final: one more round of perturbation + Fienup if time allows
    # ==================================================================
    if time.monotonic() < deadline - 20:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.02 * (1 + trial * 0.5)
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            a_r = fienup_v2(a_p, 150, gamma=0.8, top_frac_init=0.1, top_frac_end=0.5)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART-{trial}")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
