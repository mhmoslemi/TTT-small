    """Exact score."""
    n = len(a)

        return np.inf

def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * LSE_beta(conv) / (sum)^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

"""
Multi-phase optimizer for step function peak minimization.
Strategy: Adaptive LSE with cross-correlation gradient, combined with
Nesterov subgradient, peak-targeted perturbation, and multi-beta L-BFGS.
Key improvement: variance-aware LSE that penalizes both peak height and
convolution irregularity, plus adaptive beta scheduling.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * LSE_beta(conv) / (sum)^2
    Gradient computed via cross-correlation FFT trick.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: clip to avoid underflow
    scaled = np.maximum(scaled, -80.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
        L = c_max
    else:
        p = exp_scaled / exp_sum
        L = c_max + float(np.log(exp_sum)) / beta

    score = 2.0 * n * L / (S * S)

    # Gradient: dg/da[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is the cross-correlation (p (circ) a)[j] = ifft(FFT(p) * conj(FFT(a)))[j]
    M_cpx = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.fft(p, M_cpx)
    A = np.fft.fft(a, M_cpx)
    h = np.fft.ifft(P * np.conj(A)).real[:n]

    # Full gradient of 2n*L/S^2:
    # d(2nL/S^2)/da[j] = 2n/S^2 * (2*h[j] - 2L/S)
    grad = (4.0 * n / (S * S)) * (h - L / S)

    return score, grad


def lse_var_score_grad(a, beta, var_weight=0.1):
    """
    Combined LSE + variance objective.
    LSE penalizes the peak; variance term penalizes irregularity.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.maximum(scaled, -80.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
        L = c_max
    else:
        p = exp_scaled / exp_sum
        L = c_max + float(np.log(exp_sum)) / beta

    # Variance of convolution values
    c_mean = float(c.mean())
    var = float(np.mean((c - c_mean) ** 2))

    score = 2.0 * n * L / (S * S) + var_weight * var / (S * S)

    # Gradient of LSE part (same as before)
    M_cpx = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.fft(p, M_cpx)
    A = np.fft.fft(a, M_cpx)
    h = np.fft.ifft(P * np.conj(A)).real[:n]

    # Gradient of variance part:
    # var = mean((c - c_mean)^2)
    # d(var)/da[j] = (2/(2n-1)) * sum_k (c[k]-c_mean) * dc[k]/da[j] / ... 
    # Actually: var = (1/M_c) * sum_k (c[k]-c_mean)^2 where M_c = 2n-1
    # d(var)/da[j] = (2/M_c) * sum_k (c[k]-c_mean) * dc[k]/da[j] - (2/M_c)*(1/M_c)*sum_k dc[k]/da[j] * sum_l (c[l]-c_mean)
    # The second term is zero since sum_k (c[k]-c_mean) = 0.
    # So d(var)/da[j] = (2/M_c) * sum_k (c[k]-c_mean) * 2*a[k-j] (when valid)
    # = (4/M_c) * sum_{m=0}^{n-1} (c[j+m]-c_mean) * a[m]
    # This is (4/M_c) * [(c-c_mean) (circ) a][j]

    M_c = 2 * n - 1
    Pc = np.fft.fft(c - c_mean, M_cpx)
    h_var = np.fft.ifft(Pc * np.conj(A)).real[:n]
    grad_var = (4.0 / M_c) * h_var / (S * S)

    # Full gradient
    grad_lse = (4.0 * n / (S * S)) * (h - L / S)
    grad = grad_lse + var_weight * grad_var

    return score, grad


def optimize_lbfgs(a0, beta, maxiter, maxfun, var_weight=0.0):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = a0.sum()
    bounds = [(0.0, 1000.0)] * n

    if var_weight > 0:
        func = lambda x: lse_var_score_grad(x, beta, var_weight)
    else:
        func = lambda x: lse_score_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a, n_iters, f_lb=1.0):
    """Nesterov-accelerated subgradient on exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Subgradient: dc[k_star]/da[j] = 2*y[k_star-j] when 0<=k_star-j<=n-1
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def peak_shaving(a, n_iters=500):
    """
    Peak-targeted coordinate descent: identify the peak in the convolution,
    find its top contributors, and try to reduce them by redistributing mass.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        Sc = float(c[k_star])
        sc = 2.0 * n * Sc / (S * S)

        # Identify contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_end < j_start:
            break

        seg = a[j_start:j_end + 1]
        # Contribution of a[j] to c[k_star] is 2*a[k_star-j] for the pair
        # So a[j] contributes 2*a[k_star-j] to the peak
        contrib = 2.0 * seg[::-1].copy()

        # Find top and bottom contributors
        n_top = min(20, len(contrib) // 4)
        n_bot = min(20, len(contrib) // 4)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:n_bot]

        improved = False
        for ti in top_idx:
            j = j_start + ti
            a_orig = a[j]
            if a_orig < 1e-12:
                continue

            # Try reducing a[j] by a small amount and adding to bottom
            for frac in [0.7, 0.5, 0.3]:
                new_val = a_orig * frac
                reduction = a_orig - new_val
                if reduction < 1e-12:
                    continue

                a[j] = new_val
                # Redistribute to bottom contributors
                per = reduction / max(1, len(bot_idx))
                for bi in bot_idx:
                    a[j_start + bi] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc_new = compute_score(a)
                if sc_new < sc:
                    sc = sc_new
                    improved = True
                    break
                else:
                    a[j] = a_orig / S  # revert (approximately)
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S

        if not improved:
            # Try a different strategy: reduce top-2 simultaneously
            if n_top >= 2:
                j1 = j_start + top_idx[0]
                j2 = j_start + top_idx[1]
                a1, a2 = a[j1], a[j2]
                a[j1] = a1 * 0.6
                a[j2] = a2 * 0.6
                reduction = 0.4 * (a1 + a2)
                per = reduction / max(1, len(bot_idx))
                for bi in bot_idx:
                    a[j_start + bi] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc_new = compute_score(a)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                break
        else:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a for iterated local search."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bot_local)
            for li in bot_local:
                a_p[j_start + li] += per
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    # Raised cosine
    inits.append(np.maximum(0.01, 0.5 * (1 + np.cos(2 * np.pi * x))))
    # Gaussian-like
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    # U-shape
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    # Random
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + total_budget

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Setup working length
    if parent is not None:
        n_work = len(parent)
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
    else:
        n_work = 8192

    if best_a is None:
        best_a = np.ones(n_work) * 0.5

    # Ensure best_a has correct length
    if len(best_a) != n_work:
        best_a = np.interp(np.linspace(0, 1, n_work),
                           np.linspace(0, 1, len(best_a)), best_a)
        best_a = np.clip(best_a, 0.0, 1000.0)
        S = best_a.sum()
        if S > 0:
            best_a = best_a / S

    print(f"Working length: {n_work}, initial score: {best_score:.10f}")

    # ==================================================================
    # Phase 1: Adaptive LSE optimization (40% of time)
    # Start with low beta for exploration, increase for exploitation
    # ==================================================================
    print("\n=== Phase 1: Adaptive LSE ===")
    phase1_deadline = t_start + 0.40 * total_budget

    beta_schedule = [30, 50, 100, 200, 500, 1000, 2000, 5000]
    prev_score = best_score

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline:
            break
        remaining = max(5.0, phase1_deadline - time.monotonic())
        maxiter = int(min(30000, max(2000, remaining * 40)))
        maxfun = int(min(80000, max(8000, remaining * 100)))

        # Use variance-aware objective for medium betas
        vw = 0.05 if 100 <= beta <= 2000 else 0.0
        sc, a_opt = optimize_lbfgs(best_a, beta, maxiter, maxfun, var_weight=vw)
        improved = report(sc, a_opt, f"LSE b={beta}")

        # If no improvement, try a perturbed version
        if not improved and remaining > 30:
            a_pert = perturbed_restart(best_a, rng, scale=0.01)
            sc2, a_opt2 = optimize_lbfgs(a_pert, beta, maxiter // 2, maxfun // 2)
            report(sc2, a_opt2, f"LSE b={beta} pert")

        # Check for stall
        if prev_score - best_score < 1e-8:
            # Try a larger perturbation
            a_pert = perturbed_restart(best_a, rng, scale=0.05)
            sc3, a_opt3 = optimize_lbfgs(a_pert, max(50, beta // 2),
                                          max(2000, maxiter // 3),
                                          max(5000, maxfun // 3))
            report(sc3, a_opt3, f"LSE b={beta//2} big_pert")

        prev_score = best_score

    print(f"  After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: Nesterov subgradient (15% of time)
    # ==================================================================
    print("\n=== Phase 2: Nesterov Subgradient ===")
    phase2_deadline = t_start + 0.55 * total_budget
    if time.monotonic() < phase2_deadline:
        n_iter_ns = max(1000, int((phase2_deadline - time.monotonic()) * 3))
        a_ns = nesterov_subgrad(best_a, n_iters=min(5000, n_iter_ns), f_lb=1.0)
        sc = compute_score(a_ns)
        report(sc, a_ns, "nesterov")

        # Try again from perturbed start
        if time.monotonic() < phase2_deadline - 30:
            a_p = perturbed_restart(best_a, rng, scale=0.02)
            a_ns2 = nesterov_subgrad(a_p, n_iters=2000, f_lb=1.0)
            sc = compute_score(a_ns2)
            report(sc, a_ns2, "nesterov_pert")

    print(f"  After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: High-beta LSE refinement (20% of time)
    # ==================================================================
    print("\n=== Phase 3: High-Beta Refinement ===")
    phase3_deadline = t_start + 0.75 * total_budget

    for beta in [10000, 30000, 100000, 300000]:
        if time.monotonic() > phase3_deadline:
            break
        remaining = max(5.0, phase3_deadline - time.monotonic())
        maxiter = int(min(30000, max(3000, remaining * 50)))
        maxfun = int(min(100000, max(10000, remaining * 150)))

        sc, a_opt = optimize_lbfgs(best_a, beta, maxiter, maxfun)
        report(sc, a_opt, f"high b={beta}")

    print(f"  After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Peak shaving (10% of time)
    # ==================================================================
    print("\n=== Phase 4: Peak Shaving ===")
    phase4_deadline = t_start + 0.85 * total_budget
    if time.monotonic() < phase4_deadline:
        n_ps_iters = min(500, int((phase4_deadline - time.monotonic()) * 5))
        a_ps = peak_shaving(best_a, n_iters=n_ps_iters)
        sc = compute_score(a_ps)
        report(sc, a_ps, "peak_shave")

        # Follow up with LSE
        if time.monotonic() < phase4_deadline - 30:
            sc, a_opt = optimize_lbfgs(a_ps, 5000, 10000, 50000)
            report(sc, a_opt, "peak_shave+LSE")

    print(f"  After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Iterated local search (remaining time)
    # ==================================================================
    print("\n=== Phase 5: Iterated Local Search ===")
    trial = 0
    while time.monotonic() < deadline - 30:
        trial += 1
        # Adaptive perturbation scale
        scale = 0.01 + 0.03 * (1.0 / (1.0 + trial * 0.1))
        a_p = perturbed_restart(best_a, rng, scale=scale)

        # Adaptive beta
        beta = int(1000 * (1.0 + rng.random() * 5.0))
        remaining = deadline - 30 - time.monotonic()
        if remaining < 20:
            break
        maxiter = int(min(15000, max(2000, remaining * 25)))
        maxfun = int(min(50000, max(5000, remaining * 75)))

        sc, a_opt = optimize_lbfgs(a_p, beta, maxiter, maxfun)
        improved = report(sc, a_opt, f"ILS #{trial} b={beta}")

        # If improved, do a quick polish
        if improved and time.monotonic() < deadline - 30:
            sc2, a_opt2 = optimize_lbfgs(a_opt, min(200000, beta * 10),
                                          max(3000, maxiter // 3),
                                          max(5000, maxfun // 3))
            report(sc2, a_opt2, f"ILS #{trial} polish")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
