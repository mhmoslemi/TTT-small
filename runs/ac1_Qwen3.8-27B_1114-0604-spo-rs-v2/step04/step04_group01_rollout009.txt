"""
Adaptive Multiplicative Pressure with Soft Exponential Weighting, Temperature Annealing,
and LSE+L-BFGS-B Polish, for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative
sequences. Uses longer sequences (up to 16384) and a multi-phase schedule.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_middle(a, w):
    """
    Compute h[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1.
    This is the middle slice of conv(a_reversed, w).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / S


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_smooth_score(a, beta):
    """
    LSE-smoothed version of 2*n*max(conv)/sum^2.
    Returns (score, gradient).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    beta = float(max(beta, 10.0))
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    
    dmax = float(np.max(d))
    if dmax <= 0.0:
        return 0.0, np.zeros(n)
    
    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    
    # Gradient of L w.r.t. a:
    # dL/da[j] = sum_k w[k] * d(d[k])/da[j]
    # d(d[k])/da[j] = n * (a[k-j] + (k==0 ? 0 : a[j-k])) / S^2 ... 
    # Actually d[k] = n * c[k] / S^2, and c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (if both indices are valid)
    # But for the middle slice approach, we use corr:
    # h[j] = sum_k w[k] * (c[k])/S^2 ... 
    
    # Let me use the corr approach:
    # dL/da[j] = (n/S^2) * sum_k w[k] * (dc[k]/da[j])
    # dc[k]/da[j] = a[k-j] (when 0<=k-j<n) + a[j-k] (when 0<=j-k<n, k≠j case)
    # Actually for self-convolution, dc[k]/da[j] = a[k-j] + a[j-k] if we count both terms
    # But since c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = a[k-j] + (if j≠k-j then a[j-k] else a[j])
    # For j ≠ k-j, dc[k]/da[j] = a[k-j] + a[j-k]
    # For j = k-j (i.e., k=2j), dc[k]/da[j] = 2*a[j]
    
    # The corr_middle function computes: h[j] = sum_k w[k] * a[k-j]
    # This gives us sum_k w[k] * a[k-j], which is half of what we need (the other half is sum_k w[k] * a[j-k])
    # By symmetry of the problem (c is symmetric around n-1), we can use:
    # h[j] ≈ sum_k w[k] * (a[k-j] + a[j-k]) / 2 ... 
    
    # Actually, let me just compute it directly using the correction:
    # The full gradient is:
    # grad[j] = (2n/S^2) * [sum_k w[k]*a[k-j] * 1(0<=k-j<n) + sum_k w[k]*a[j-k] * 1(0<=j-k<n, j≠k-j)]
    
    # For simplicity, and since the convolution is symmetric (c[k] = c[2n-2-k]),
    # and w is symmetric (since it depends on d which is symmetric),
    # we have: sum_k w[k]*a[k-j] = sum_k w[k]*a[j-k] (by reindexing)
    # So the gradient is approximately:
    # grad[j] = (4n/S^2) * sum_k w[k]*a[k-j] = (4n/S^2) * h[j]
    # (with a correction for the diagonal terms, which is negligible for large n)
    
    # But wait, we also need to account for the derivative of S^2 in the denominator:
    # d = n*c/S^2, so dd/da[j] = n*(dc/da[j])/S^2 - 2*n*c/S^3 * S (for the S dependence)
    # = (n/S^2) * dc[k]/da[j] - 2*d[k] * (1/S)
    
    # So: dL/da[j] = sum_k w[k] * [(n/S^2) * dc[k]/da[j] - 2*d[k]/S]
    # = (n/S^2) * sum_k w[k] * dc[k]/da[j] - (2/S) * sum_k w[k]*d[k]
    
    # The second term is: (2/S) * weighted_mean(d) ≈ (2/S) * L (approximately)
    
    # For the first term, using h[j] = sum_k w[k] * a[k-j]:
    # sum_k w[k] * dc[k]/da[j] ≈ 2 * h[j] (by the symmetry argument above)
    
    # So: grad[j] ≈ (2n/S^2) * 2*h[j] - (2/S) * LDS
    # = (4n/S^2) * h[j] - (2/S) * LDS
    
    # where LDS = sum_k w[k]*d[k]
    
    h = corr_middle(a, w)
    LDS = float(np.dot(w, d))
    
    grad = (4.0 * float(n) * h) / (S * S) - (2.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    
    return float(score), grad


def lse_optimize(a, beta, deadline, maxfun):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = normalize_a(a)
    n = len(a)
    
    if deadline - time.monotonic() < 5.0:
        return exact_score(a), a
    
    def func_and_grad(x):
        x_clipped = np.clip(np.asarray(x, dtype=np.float64), 0.0, None)
        sc, g = lse_smooth_score(x_clipped, beta)
        return sc, g
    
    maxiter = max(30, maxfun // 3)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    
    try:
        res = minimize(
            func_and_grad,
            a,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=opts,
        )
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def multiplicative_pressure(a, T, eta):
    """
    One step of multiplicative pressure with soft exponential weighting.
    T: temperature (controls softness of weighting)
    eta: step size
    Returns new sequence (not normalized).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    
    c = fast_self_conv(a)
    c = np.maximum(c, 1e-30)  # avoid log(0)
    
    # Soft weights: w[k] ∝ exp(c[k]/T)
    c_scaled = c / max(T, 1e-10)
    c_max = float(np.max(c_scaled))
    e = np.exp(c_scaled - c_max)  # shift for numerical stability
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return a.copy()
    w = e / sum_e
    
    # Pressure: p[j] = sum_k w[k] * a[k-j]
    p = corr_middle(a, w)
    pmax = float(np.max(p))
    if pmax < 1e-12:
        return a.copy()
    
    # Multiplicative update
    a_new = a * np.exp(-eta * p / pmax)
    return a_new


def adaptive_pressure_optimize(a, deadline, T_init=1.0, T_min=0.01, verbose=True, label="pressure"):
    """
    Main optimization loop: adaptive multiplicative pressure with temperature annealing
    and backtracking line search.
    """
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    cur_sc = best_sc
    
    n = len(a)
    c = fast_self_conv(a)
    c_range = float(np.max(c) - np.min(c))
    if c_range < 1e-10:
        c_range = 1.0
    
    # Initial temperature: spread over a reasonable range
    T = T_init * c_range
    eta = 0.5
    
    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)
    n_iter = 0
    stuck_count = 0
    
    while time.monotonic() < deadline - 1.0:
        progress = (time.monotonic() - t_start) / total_time
        
        # Anneal temperature
        T = max(T_min * c_range, T_init * c_range * (1.0 - progress) ** 1.5)
        
        # Try the step
        a_new = multiplicative_pressure(a, T, eta)
        a_new = normalize_a(a_new)
        
        sc_new = exact_score(a_new)
        
        if sc_new < cur_sc - 1e-12:
            # Accept
            a = a_new
            cur_sc = sc_new
            stuck_count = 0
            # Slightly increase eta (be more aggressive)
            eta = min(eta * 1.05, 3.0)
            
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
                if verbose and n_iter % 200 == 0:
                    elapsed = time.monotonic() - t_start
                    print("[{}] iter={} score={:.8f} T={:.4f} eta={:.3f} elapsed={:.0f}s".format(
                        label, n_iter, best_sc, T, eta, elapsed))
        else:
            # Reject: decrease eta
            stuck_count += 1
            eta = max(eta * 0.7, 0.01)
            
            # If stuck for too long, try a larger temperature (more exploration)
            if stuck_count > 50:
                T = T * 1.5
                stuck_count = 0
                eta = 0.3
        
        n_iter += 1
    
    # Final: try a few LSE polish steps if time allows
    rem = deadline - time.monotonic()
    if rem > 30.0:
        mf = int(rem * 20)  # rough estimate
        mf = max(64, min(mf, 2000))
        sc, a2 = lse_optimize(best_a, 5000.0, deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            if verbose:
                print("[{}] LSE polish: {:.8f}".format(label, best_sc))
    
    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1.0, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.01
        if arr.sum() < 1e-10:
            arr = np.ones(n) / n
        inits.append(arr / float(arr.sum()))
    
    # Parent
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    # Uniform
    add(np.ones(n, dtype=np.float64))
    
    # Linear ramps
    for alpha in (0.3, 0.5, 0.7, 0.9):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))
    
    # Cosine windows
    for bw in (0.3, 0.5, 0.7):
        add(1.0 + bw * np.cos(2 * np.pi * x))
        add(1.0 + bw * np.cos(4 * np.pi * x))
    
    # Multi-frequency
    for _ in range(8):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.1, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
            if rng.random() < 0.5:
                vals += 0.5 * amp * np.sin(2.0 * np.pi * f * x + ph)
        add(vals)
    
    # Random smooth (low-pass noise)
    for _ in range(6):
        Nf = next_power_of_two(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        sigma = float(rng.uniform(1.0, 3.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** sigma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 0.8)) * noise)
    
    # Block structure
    for nb in (8, 16, 32, 64, 128):
        if nb > n:
            nb = n
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        arr = np.repeat(block_vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block_vals, n - arr.size)])
        add(arr[:n])
    
    # Two-scale: smooth envelope × periodic
    for _ in range(4):
        m = int(rng.integers(4, 32))
        env = 0.5 + 0.5 * np.cos(2 * np.pi * x)  # smooth envelope
        flen = max(2, n // m)
        fine = 0.7 + 0.3 * rng.random(flen)
        fine_full = np.interp(x * m, np.linspace(0, 1, flen), fine)
        add(env * fine_full)
    
    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))
    
    # Palindromic
    for _ in range(3):
        half = rng.uniform(0.1, 1.0, size=n // 2 + 1)
        arr = np.concatenate([half, half[-2::-1]])
        if arr.size < n:
            arr = np.concatenate([arr, np.ones(n - arr.size) * 0.5])
        add(arr[:n])
    
    return inits


def subgrad_polish(a, max_iters=30, deadline=None):
    """Exact subgradient polish with adaptive step sizes."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    cur_sc = best_sc
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        if cmax <= 0:
            break
        
        # Use soft weights
        T = (cmax * 0.1)  # weight on values within 10% of max
        c_s = (c - cmax) / max(T, 1e-10)
        e = np.exp(np.clip(c_s, -50, 50))
        w = e / float(e.sum())
        
        p = corr_middle(a, w)
        pmax = float(np.max(p))
        if pmax < 1e-12:
            break
        
        best_step_sc = cur_sc
        best_step_a = a
        
        for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
            q = normalize_a(a * np.exp(-eta * p / pmax))
            sc = exact_score(q)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step_a = q
        
        if best_step_a is not a and best_step_sc < cur_sc - 1e-10:
            a = best_step_a
            cur_sc = best_step_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        else:
            break
    
    return float(best_sc), best_a
