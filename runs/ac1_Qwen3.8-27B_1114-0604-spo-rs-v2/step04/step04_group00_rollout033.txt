"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: multiplicative subgradient, Adam in log-space with LSE smoothing,
iterative spectral shaping, and perturbation-based exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_max_and_prob(c, beta):
    """Compute LSE-approximation of max and the softmax probabilities."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = max_scaled + float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    return L, p


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, p = lse_max_and_prob(c, beta)
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a: 2 * conv(rev_a, p)[n-1:2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

    a = a / a.sum()
    z = np.log(a) - np.log(a).mean()  # log-parameterization
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    for it in range(n_iters):
        # Get current beta
        frac = it / max(1, n_iters - 1)
        if isinstance(beta_schedule, (list, tuple)):
            # Linear interpolation through the schedule
            idx = frac * (len(beta_schedule) - 1)
            i0 = int(idx)
            i1 = min(i0 + 1, len(beta_schedule) - 1)
            w = idx - i0
            beta = beta_schedule[i0] * (1 - w) + beta_schedule[i1] * w
        else:
            beta = beta_schedule
        
        # Compute score and gradient w.r.t. a
        sc_a, grad_a = lse_score_grad(a, beta)
        
        # Convert to gradient w.r.t. z (natural gradient for softmax)
        # dL/dz_j = a_j * (dL/da_j - sum_i a_i * dL/da_i)
        mean_grad = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - mean_grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Update a from z
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a_sum = a.sum()
        if a_sum < 1e-12:
            break
        a = a / a_sum
        
        # Check true score
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def multiplicative_subgradient(a0, n_iters=5000, top_m=20, eta0=0.01, decay=0.999):
    """
    Multiplicative (log-space) subgradient method using top-M peaks.
    More stable than additive because it maintains positivity.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    sc_lb = 2.0 * n / (2.0 * n - 1)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient: pressure on each position
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # c_k = sum_j a_j * a_{k-j}
            # dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # sg[j] += 2 * a_{k-j} for j in [j_start, j_end]
                # a_{k-j} for j in [j_start, j_end] is a[k-j_end : k-j_start+1] reversed
                idx = k - np.arange(j_start, j_end + 1)
                sg[j_start:j_end+1] += 2.0 * a[idx]
        
        # Multiplicative update in log-space
        # z = log(a), z -= eta * sg (then normalize)
        # Equivalent: a *= exp(-eta * sg), then normalize
        # For stability, use: a *= (1 - eta * sg / a_mean) or similar
        # Simpler: log-space update
        log_a = np.log(a + 1e-30)
        log_a -= eta * sg
        # Re-normalize
        log_a_shifted = log_a - log_a.max()
        a = np.exp(log_a_shifted)
        a_sum = a.sum()
        a = a / a_sum
        
        eta *= decay
    
    return best_sc, best_a


def fienup_project(a, n_iters=200, clip_strength=0.5):
    """
    Fienup-style spectral shaping with projection back to non-negative.
    Flattens the power spectrum iteratively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        # Adaptive threshold
        thresh = P_mean * (1.2 + 0.3 * (1 - progress))
        
        above = P > thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (clip_strength * 0.5)
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def perturbed_restart(a, rng, sigma=0.05):
    """Apply Gaussian perturbation in log-space."""
    n = len(a)
    log_a = np.log(np.maximum(a, 1e-10))
    log_a += sigma * rng.standard_normal(n)
    a_new = np.exp(log_a - log_a.max())
    a_new = a_new / a_new.sum()
    return a_new

    """Top-level function for scipy minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    a0 = a0 / a0.sum()
    bounds = [(1e-10, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),

    )
    a_out = np.clip(res.x, 1e-10, 1000.0)
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Start from parent, apply multiplicative subgradient for quick progress,
then Adam with LSE smoothing (increasing beta schedule), Fienup spectral shaping,
high-beta L-BFGS fine-tuning, and perturbation-based exploration.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_max_and_prob(c, beta):
    """Compute LSE-approximation of max and softmax probabilities over c."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = max_scaled + float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    return L, p


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (unconstrained)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, p = lse_max_and_prob(c, beta)
    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a: 2 * conv(rev_a, p)[n-1:2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Subgradient: pressure on each position (sum over top-M peaks)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = k - np.arange(j_start, j_end + 1)
                sg[j_start:j_end + 1] += 2.0 * a[idx]

        # Log-space multiplicative update
        log_a = np.log(a + 1e-30)
        log_a -= eta * sg
        log_a_shifted = log_a - log_a.max()
        a = np.exp(log_a_shifted)
        a_sum = a.sum()
        if a_sum < 1e-12:
            break
        a = a / a_sum
        eta *= decay

    return best_sc, best_a


def method_adam_lse(a0, beta_start, beta_end, n_iters, lr=0.01, b1=0.9, b2=0.999):
    """
    Adam optimizer with LSE smoothing, working directly on a with projection.
    Beta increases linearly from beta_start to beta_end over iterations.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        sc_a, grad_a = lse_score_grad(a, beta)

        # Project gradient to the tangent space of the simplex (sum-preserving)
        mean_g = float(np.dot(a, grad_a)) / S2 if (S2 := float(a.sum())**2) > 1e-20 else 0.0
        # Actually just use the raw gradient; the normalization step handles sum
        t += 1
        m = b1 * m + (1 - b1) * grad_a
        v = b2 * v + (1 - b2) * (grad_a * grad_a)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            break
        a_new = a_new / a_sum
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a
