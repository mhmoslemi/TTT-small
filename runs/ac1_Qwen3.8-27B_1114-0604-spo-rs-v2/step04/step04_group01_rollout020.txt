def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)

    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return np.zeros(n, dtype=np.float64)
    w = active_weights_topk(d, k)
    return corr_pair(a, w)


def estimate_eval_time(n):
    base = 0.016
    scale = max(0.4, float(n) / 8192.0)
    log_scale = max(1.0, np.log2(max(16, n)) / np.log2(8192.0))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.75 * t_left) / max(est, 1e-4))
    mf = max(64, min(mf, 4000))
    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        cap = int((0.9 * rem) / max(estimate_eval_time(n), 1e-4))
        maxfun = int(min(maxfun, max(64, cap)))
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-15, "gtol": 1e-12, "maxiter": int(maxiter),
            "maxfun": int(maxfun), "maxls": 50}
    try:
        res = minimize(lse_objective, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def subgrad_step(a, k, eta):
    h = pressure_dir(a, k)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return a
    q = a * np.exp(-eta * h / hmax)
    return normalize_a(q)


def subgrad_polish(a, max_iters=30, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in (1, 2, 3, 5, 8):
            for mult in (0.01, 0.03, 0.08, 0.15, 0.30):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = subgrad_step(a, k, mult)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < cur_sc:
                    cur_sc = sc
                    a = q.copy()
        if not improved:
            # Try a gentler step
            q = subgrad_step(a, 3, 0.01)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
            else:
                break
    return float(best_sc), best_a


def perturb_sequence(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 7))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        q = p[::-1].copy()
        q *= np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        h = pressure_dir(p, 4)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = p.copy()
        else:
            q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp += strength * 0.5 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Local random rebalancing
        nblocks = int(rng.integers(4, 16))
        block_size = max(1, n // nblocks)
        q = p.copy()
        for _ in range(3):
            i = int(rng.integers(0, n - block_size))
            j = int(rng.integers(0, n - block_size))
            if i != j:
                swap_amt = 0.1 * min(p[i:i+block_size].mean(), p[j:j+block_size].mean())
                q[i:i+block_size] -= swap_amt
                q[j:j+block_size] += swap_amt
        q = np.clip(q, 0, None)
    else:
        # Gaussian smoothing
        k = int(rng.integers(3, 11))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(3, n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.5)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    return normalize_a(q)


def lse_trajectory(a0, deadline, rng, betas=None, label=""):
    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total_time = max(10.0, float(deadline - t0 - 10.0))

    if betas is None:
        betas = [30, 60, 100, 200, 400, 800, 1500, 3000, 6000, 12000, 25000]

    n = len(a)
    n_betas = len(betas)
    # Allocate time proportionally (more time at middle-to-high betas)
    weights = [1.0] * n_betas
    # Give more weight to middle and high betas
    for i in range(n_betas):
        if i >= n_betas // 3:
            weights[i] = 1.5
        if i >= 2 * n_betas // 3:
            weights[i] = 2.0
    total_w = sum(weights)
    time_fracs = [w / total_w for w in weights]

    print("[{}] start: score={:.8f} n={}".format(label, best_sc, n))

    cum_time = 0.0
    for bi, beta in enumerate(betas):
        cum_time += time_fracs[bi] * total_time
        sub_dead = t0 + cum_time
        sub_dead = min(sub_dead, deadline - 5.0)

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.8, n)
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] improved: {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = normalize_a(a2)
            # Occasional subgradient step between LSE rounds
            if time.monotonic() < sub_dead - 10.0:
                for k in (3, 5):
                    for mult in (0.03, 0.1):
                        q = subgrad_step(a, k, mult)
                        sc2 = exact_score_a(q)
                        if np.isfinite(sc2) and sc2 < best_sc:
                            best_sc = sc2
                            best_a = q.copy()
                            a = q.copy()

    # Final subgradient polish within this trajectory
    if time.monotonic() < deadline - 10.0:
        rem = deadline - time.monotonic()
        if rem > 15.0:
            sc, seq = subgrad_polish(a, max_iters=min(20, int(rem // 3)), deadline=deadline - 5.0)
            if sc < best_sc:
                best_sc = sc
                best_a = seq.copy()

    return float(best_sc), best_a


def quick_lse_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for beta in (200, 1000, 5000, 15000):
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        mf = estimate_maxfun(rem * 0.8, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)
    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None and parent.size > 0:
        pr = normalize_a(resample_sequence(parent, n))
        inits.append(pr)
        inits.append(normalize_a(pr[::-1]))

    inits.append(np.ones(n) / n)

    for alpha in (0.3, 0.5, 0.7):
        inits.append(normalize_a(1.0 + alpha * (2 * x - 1)))
        inits.append(normalize_a(1.0 - alpha * (2 * x - 1)))

    # Cosine bump (good for this problem - concentrated in middle)
    inits.append(normalize_a(0.3 + 0.7 * np.cos(np.pi * (x - 0.5)) ** 2))
    inits.append(normalize_a(0.5 + 0.5 * np.sin(np.pi * x) ** 4))

    # Trigonometric mixtures
    for r in range(6):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.05, 0.3))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2 * np.pi * f * x + ph)
        inits.append(normalize_a(vals))

    # Low-frequency noise
    Nf = next_power_of_two(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        inits.append(normalize_a(1.0 + 0.5 * noise))

    # Block structures
    for nb in (8, 16, 32):
        if nb > n:
            continue
        block_vals = 0.2 + 0.8 * rng.random(nb)
        per = max(1, n // nb)
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals[-1], n - a.size)])
        inits.append(normalize_a(a[:n]))

    # Random
    for _ in range(3):
        inits.append(normalize_a(rng.random(n)))

    return inits[:cap]


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    # Determine target length
    if parent is not None:
        target_n = int(min(parent.size, 10240))
        if target_n < 4096:
            target_n = 4096
        if target_n % 2 != 0:
            target_n -= 1
    else:
        target_n = 8192

    print("Target length: {}, parent: {}".format(target_n, parent is not None))

    # Phase 1: Multilevel coarse-to-fine (if no parent or parent is short)
    best_seq = None
    best_score = np.inf

    if parent is not None:
        # Start directly from parent at target length
        best_seq = normalize_a(resample_sequence(parent, target_n))
        best_score = exact_score_a(best_seq)
        print("Parent resampled, initial score: {:.8f}".format(best_score))

        # Also try a few diverse inits
        inits = generate_inits(target_n, rng, parent, cap=8)
        for arr in inits:
            sc = exact_score_a(arr)
            if sc < best_score:
                best_score = sc
                best_seq = arr.copy()
    else:
        # Multilevel: 64 -> 128 -> 256 -> 512 -> 1024 -> 2048 -> 4096 -> 8192
        levels = [64, 128, 256, 512, 1024, 2048, 4096, target_n]
        coarse_deadline = t_start + 120.0
        cur = None
        for i, lvl in enumerate(levels):
            rem = coarse_deadline - time.monotonic()
            if rem < 5.0:
                break
            if cur is None:
                inits = generate_inits(lvl, rng, None, cap=6)
                # Pick best init
                best_init = inits[0]
                best_init_sc = exact_score_a(best_init)
                for arr in inits[1:]:
                    sc = exact_score_a(arr)
                    if sc < best_init_sc:
                        best_init_sc = sc
                        best_init = arr
                cur = best_init
            else:
                cur = normalize_a(resample_sequence(cur, lvl))

            # Quick optimize at this level
            time_per_level = max(8.0, rem * 0.7 / (len(levels) - i))
            lvl_dead = time.monotonic() + time_per_level
            sc, seq = lse_trajectory(cur, lvl_dead, rng,
                                     betas=[50, 200, 800, 3000],
                                     label="ml{}".format(lvl))
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

        # If we didn't reach target_n, resample
        if best_seq is None:
            best_seq = normalize_a(resample_sequence(cur, target_n))
            best_score = exact_score_a(best_seq)
        elif len(best_seq) != target_n:
            best_seq = normalize_a(resample_sequence(best_seq, target_n))
            best_score = exact_score_a(best_seq)

    print("After phase 1: best={:.8f} n={}".format(best_score, len(best_seq)))

    # Phase 2: Main optimization at target length
    # Allocate most of the remaining time here
    main_deadline = deadline - 300.0  # reserve 300s for final polish
    main_start = time.monotonic()
    main_time = max(60.0, main_deadline - main_start)

    # Run multiple trajectories with different initializations
    n_trajs = 3
    traj_time = main_time / n_trajs

    for ti in range(n_trajs):
        traj_dead = main_start + (ti + 1) * traj_time
        traj_dead = min(traj_dead, main_deadline)
        if time.monotonic() > traj_dead - 30.0:
            break

        if ti == 0:
            start_seq = best_seq.copy()
            label = "main0"
        elif ti == 1:
            # Perturbed version of best
            start_seq = perturb_sequence(best_seq, rng, strength=0.3)
            label = "main1"
        else:
            # Fresh diverse init
            inits = generate_inits(target_n, rng, best_seq, cap=6)
            start_seq = inits[0]
            for arr in inits[1:]:
                if exact_score_a(arr) < exact_score_a(start_seq):
                    start_seq = arr
            label = "main2"

        # Use an aggressive temperature schedule
        if ti == 0:
            betas = [40, 80, 150, 300, 600, 1200, 2500, 5000, 10000, 20000, 40000]
        else:
            betas = [100, 300, 800, 2000, 5000, 12000, 30000]

        sc, seq = lse_trajectory(start_seq, traj_dead, rng, betas=betas, label=label)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after {}: {:.8f}".format(label, best_score))

    best_seq = normalize_a(best_seq)
    print("After phase 2: best={:.8f}".format(best_score))

    # Phase 3: Subgradient polish
    if time.monotonic() < deadline - 150.0:
        sd = deadline - 150.0
        sc, seq = subgrad_polish(best_seq, max_iters=50, deadline=sd)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # Phase 4: Random restarts with quick optimization
    n_restarts = 4
    restart_start = time.monotonic()
    restart_dead = deadline - 80.0
    restart_time = max(10.0, restart_dead - restart_start)

    for ri in range(n_restarts):
        rem = restart_dead - time.monotonic()
        if rem < 30.0:
            break
        chunk = min(rem * 0.9, restart_time / (n_restarts - ri))
        p = perturb_sequence(best_seq, rng, strength=float(rng.uniform(0.05, 0.3)))
        sc, seq = quick_lse_optimize(p, time.monotonic() + chunk)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {}: {:.8f}".format(ri, best_score))

    # Phase 5: Final high-temperature LSE + subgradient
    if time.monotonic() < deadline - 30.0:
        rem = deadline - time.monotonic() - 10.0
        if rem > 20.0:
            mf = estimate_maxfun(rem * 0.8, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 50000.0, deadline, mf)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final LSE: {:.8f}".format(best_score))

    if time.monotonic() < deadline - 15.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 10.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
