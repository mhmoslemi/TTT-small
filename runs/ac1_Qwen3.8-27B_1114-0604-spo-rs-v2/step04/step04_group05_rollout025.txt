"""
LP-based iterative refinement + LSE gradient optimization for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.
New: Uses LP to find optimal mixing direction (from state-of-the-art theory),
combined with LSE-smoothed gradient polish and adaptive step sizes.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time
import math
import cvxpy as cp


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of LSE w.r.t. a[j] = sum_k p[k] * 2*a[k-j] (valid indices)
    # This is 2 * conv(p, a_reversed) viewed appropriately
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_conv = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_conv / S2 - 2.0 * L / S3)
    return score, grad


def lp_direction(f, M, method='highs'):
    """
    Find g >= 0 maximizing sum(g) subject to ||f*g||_inf <= M.
    Uses LP. f is the current sequence, M is the current peak.
    Returns the optimal g, or None if infeasible.
    """
    n = len(f)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Build constraint matrix: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j: 0<=k-j<n} f[k-j] * g[j]
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build the banded matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_direction_sparse(f, M):
    """
    More efficient LP: exploit the banded structure.
    Only include non-zero constraints.
    """
    n = len(f)
    c_obj = -np.ones(n)
    
    # For each k, the constraint is sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j] <= M
    # Build sparse representation
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        A_ub[k, js] = f[k - js]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def adam_optimize(a_init, lr=0.005, n_steps=50000, beta=200,
                  time_budget=None, t_start=None, normalize=True):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a * (S0 / s)
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    try:
        if deadline != np.inf:
            def timed_obj(x):
                if time.monotonic() > deadline:
                    raise TimeoutError
                return lse_score_grad(x, beta)
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-14, 'gtol': 1e-11, 'maxiter': maxiter, 
                                  'maxfun': maxfun, 'maxls': 50})
        else:
            res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-14, 'gtol': 1e-11, 'maxiter': maxiter,
                                  'maxfun': maxfun, 'maxls': 50})
    except (TimeoutError, Exception):
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def lp_iteration(f, M, t_values=None):
    """
    One LP-based iteration: find direction g, try mixing with different t.
    Returns (best_score, best_a).
    """
    n = len(f)
    S = f.sum()
    
    # Solve LP
    g = lp_direction(f, M)
    if g is None:
        return compute_score(f), f
    
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-10:
        return compute_score(f), f
    
    # Rescale g to match sum of f
    g = g * (S / g.sum())
    
    # Try different mixing parameters
    if t_values is None:
        t_values = [0.1, 0.2, 0.3, 0.5, 0.7]
    
    best_score = compute_score(f)
    best_a = f.copy()
    
    for t in t_values:
        a_new = (1 - t) * f + t * g
        a_new = np.clip(a_new, 0.0, None)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

"""
LP-guided iterative refinement + LSE gradient optimization for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.

Key innovations over previous approach:
1. LP-based direction finding (from state-of-the-art theoretical framework)
2. Adaptive step size via line search in LP iterations
3. Multi-scale LSE continuation with very high final betas
4. Structured initializations (raised cosine, Gaussian, chirp-based)
5. Aggressive coordinate descent polish with wider search ranges
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_peak_info(a):
    """Return convolution, peak value, argmax, and sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M, S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum_val = np.log(exp_scaled.sum())
    L = c_max + log_sum_val / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # d(LSE)/d(a[j]) = sum_k p[k] * 2*a[k-j] (for valid k-j)
    # = 2 * (p_rev * a)[j] where * denotes appropriate convolution
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_conv = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_conv / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_direction(f, M):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            A_ub[k, js] = f[k - js]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'time_limit': 30})
        if res.success:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def adam_optimize(a_init, lr=0.005, n_steps=100000, beta=200,
                  time_budget=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient and normalization."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)
        
        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=50000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    try:
        if deadline != np.inf:
            def timed_obj(x):
                if time.monotonic() > deadline:
                    raise TimeoutError
                return lse_score_grad(x, beta)
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                                  'maxfun': maxfun, 'maxls': 50})
        else:
            res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                                  'maxfun': maxfun, 'maxls': 50})
    except (TimeoutError, Exception):
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def golden_section_1d(a, j, n_evals=20):
    """Golden-section search for optimal value of a[j]."""
    a_old = a[j]
    a_max = a.max()
    
    if a_old < 1e-15:
        lo = 0.0
        hi = 0.5 * a_max
    else:
        lo = 0.0
        hi = a_old * 5.0
        if hi < 1e-10:
            hi = a_max
    
    if hi <= lo:
        hi = lo + 1e-6
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)
    
    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)
    
    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent(a, n_sweeps=5, top_k=40, time_budget=30.0, t_start=None):
    """Coordinate descent with golden-section 1D search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if t_start is not None else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at peak
        n = len(a)
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=15)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_score, best_a


def raised_cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a


def bisection_init(n, depth=4):
    """.Iterative bisection: split into halves and alternate weights."""
    a = np.ones(n)
    for d in range(depth):
        block = n // (2 ** (d + 1))
        for i in range(0, n, 2 * block):
            a[i:i+block] *= 1.3
            a[i+block:i+2*block] *= 1.0/1.3
    a = np.clip(a, 0.01, None)
    return a


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    
    return a_new


def antipalindromic_perturbation(a, rng, strength=0.1):
    """Add anti-palindromic component."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / center
    freq = float(rng.uniform(1, 5))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def lp_refine(f, n_iters=50, t_start=None, time_budget=120.0):
    """
    LP-based refinement: iterate the LP direction finding.
    f should be normalized (sum = sqrt(2n)).
    """
    n = len(f)
    target_sum = math.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())
    
    best_score = compute_score(f)
    best_a = f.copy()
    deadline = t_start + time_budget
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c, k_star, M, S = conv_peak_info(f)
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        
        if g.sum() < 1e-10:
            break
        
        # Rescale
        g = g * (target_sum / g.sum())
        
        # Try different step sizes via line search
        best_local_score = best_score
        best_local_a = best_a
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_new = (1 - t) * f + t * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            sc = compute_score(a_new)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = a_new.copy()
        
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()
            f = best_a.copy()
            print(f"    LP iter {it}: score={best_score:.8f} (lp_time={lp_time:.2f}s)")
        else:
            # No improvement from LP step, break
            break
    
    return best_score, best_a


def continuation_lbfgs(a0, betas, time_budget, t_start, maxiter_per_stage=3000):
    """Continuation method with increasing beta."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.35, 80.0)
        sc, a_out = lbfgs_optimize(a, beta, maxiter=maxiter_per_stage,
                                    maxfun=20000, time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a = a_out
    
    return best_score, best_a

```python
"""
LP-guided iterative refinement + LSE gradient optimization for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.

Key innovations:
1. LP-based direction finding exploiting the linear structure of f*g in g
2. Continuation L-BFGS with very high final betas for exact peak minimization
3. Multi-scale structured initializations (raised cosine, bisection, spectral)
4. Aggressive coordinate descent with wider search ranges
5. Adaptive step-size line search in LP iterations
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_peak_info(a):
    """Return convolution, peak value, argmax, and sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M, S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum_val = np.log(exp_scaled.sum())
    L = c_max + log_sum_val / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_conv = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_conv / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_direction(f, M, time_limit=20):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses sparse constraint matrix.
    """
    n = len(f)
    if n > 8000:
        return None

    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    A = A.tocsr()

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def adam_optimize(a_init, lr=0.005, n_steps=100000, beta=200,
                  time_budget=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient and normalization."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=50000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    try:
        if deadline != np.inf:
            def timed_obj(x):
                if time.monotonic() > deadline:
                    raise TimeoutError
                return lse_score_grad(x, beta)
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                                  'maxfun': maxfun, 'maxls': 50})
        else:
            res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                         options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                                  'maxfun': maxfun, 'maxls': 50})
    except (TimeoutError, Exception):
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def golden_section_1d(a, j, n_evals=18):
    """Golden-section search for optimal value of a[j]."""
    a_old = a[j]
    a_max = a.max()

    if a_old < 1e-15:
        lo = 0.0
        hi = 0.5 * a_max
    else:
        lo = 0.0
        hi = a_old * 5.0
        if hi < 1e-10:
            hi = a_max

    if hi <= lo:
        hi = lo + 1e-6

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent(a, n_sweeps=5, top_k=40, time_budget=30.0, t_start=None):
    """Coordinate descent with golden-section 1D search on top-gradient coords."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if t_start is not None else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        n = len(a)
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())

        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
