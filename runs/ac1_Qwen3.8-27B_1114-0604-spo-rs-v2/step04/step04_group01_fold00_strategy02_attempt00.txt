def calculate_score(seq):
    n = len(seq)
    arr = np.array(seq, dtype=np.float64)
    if arr.size < 10:
        return np.inf

    # Calculate the self-convolution
    conv_result = np.convolve(arr, arr, mode='full')
    max_conv_val = np.max(conv_result)
    sum_val = np.sum(arr)

    if np.min(arr) < 0.0:
        return np.inf

    if sum_val < 0.01:
        return np.inf

    score = float(2.0 * n * max_conv_val / (sum_val ** 2))
    if np.isnan(score):
        return np.inf
    return score
