    # Then g[j] = sum_m a[m] * q[2n-2-j-m]
    # = (a * q)[2n-2-j] (linear convolution of a with q)
    # 
    # Verify: (a*q)[k] = sum_m a[m] * q[k-m] = sum_m a[m] * q[k-m]
    # (a*q)[2n-2-j] = sum_m a[m] * q[2n-2-j-m] = sum_m a[m] * p[2n-2-(2n-2-j-m)] = sum_m a[m] * p[j+m] ✓
    
    # So: compute conv(a, q) where q = p[::-1], then g[j] = conv_result[2n-2-j]
    # g has length n: g[0] = conv[2n-2], g[1] = conv[2n-3], ..., g[n-1] = conv[n-1]
    # So g = conv[n-1 : 2n-1][::-1]
    
    # Verify with n=2: a=[a0,a1], p=[p0,p1,p2], q=[p2,p1,p0]
    # conv = [a0*p2, a0*p1+a1*p2, a0*p0+a1*p1, a1*p0]
    # g[0] = conv[2] = a0*p0+a1*p1 ✓
    # g[1] = conv[1] = a0*p1+a1*p2 ✓
    # g = conv[n-1:2n-1][::-1] = conv[1:3][::-1] = [conv[2], conv[1]] ✓
    
    # Great! So the formula is:
    # g = convolve(a, p[::-1])[n-1 : 2*n-1][::-1]
    
    grad_LSE = conv_ap[n-1:2*n-1][::-1]
    
    # Full gradient of score = 2n * L / S²
    # d(score)/da = 2n * (grad_LSE / S² - 2L / S³)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_LSE / S2 - 2.0 * L / S3)
    
    return score, grad

def subgradient_peak_reduction(a, step_size=None, n_steps=50):
    """
    Reduce the peak of the convolution using subgradient-like steps.
    At each step, find the peak position and move a in the direction that reduces it.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        c = fast_conv(a)
        # Find the top few peaks
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Subgradient of c[k_star] w.r.t. a
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # d(c[k_star])/da[j] = a[k_star - j] (for valid j)
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        # Also consider nearby peaks (within a window)
        # Use a weighted combination of top few gradients
        c_sorted = np.sort(c)[::-1]
        # How many peaks are close to the max?
        threshold = M * 0.95
        near_peaks = np.where(c >= threshold)[0]
        
        if len(near_peaks) > 1:
            # Weight by proximity to max
            weights = c[near_peaks] ** 2  # emphasize the highest
            weights = weights / weights.sum()
            for w, kp in zip(weights, near_peaks):
                js2 = max(0, int(kp) - n + 1)
                je2 = min(int(kp), n - 1)
                if je2 >= js2:
                    idx2 = int(kp) - np.arange(js2, je2 + 1)
                    sg[js2:je2 + 1] += w * a[idx2]
            # Renormalize
            sg = sg / len(near_peaks)
        
        # Normalize the step to maintain sum(a) = S0
        # Project sg onto the subspace sum(x) = 0
        sg_centered = sg - sg.mean()
        
        # Step size: use a simple heuristic
        if step_size is None:
            step_size = 0.1 * S0 / (np.linalg.norm(sg_centered) + 1e-10)
        
        a = a - step_size * sg_centered
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
        
        # Decrease step size
        step_size *= 0.98
    
    final_sc = compute_score(best_a)
    return final_sc, best_a

def freq_domain_projection(a, n_iters=100, alpha=0.5):
    """
    Gerchberg-Saxton style frequency domain projection.
    Alternates between:
    1. Time domain: enforce non-negativity and fixed sum
    2. Frequency domain: flatten the convolution (i.e., make |A|² more uniform)
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = float(a.sum())
    N = 1 << (2 * n - 2 - 1).bit_length()
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        # Compute current convolution
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        c = np.fft.irfft(power, N)[:2*n-1]
        
        # Target: flatten c
        c_mean = S0 * S0 / (2 * n - 1)
        c_target = (1 - alpha) * c + alpha * c_mean
        
        # Ensure c_target is non-negative (it should be since c >= 0 and c_mean > 0)
        c_target = np.maximum(c_target, 0)
        
        # New power spectrum
        P_target = np.fft.rfft(c_target, N)
        # We need |A_new|² = P_target, but P_target might not be non-negative everywhere
        # (since FFT of a real sequence can have negative values)
        # Take the positive part and renormalize
        P_target = np.maximum(P_target, 0)
        
        # Preserve the DC component (which determines the sum)
        P_target[0] = power[0]
        
        # New amplitude
        A_new = A * (np.sqrt(P_target / (power + 1e-30)) )
        # But this might not preserve sum(a) = S0
        
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Renormalize
        S_new = a_new.sum()
        if S_new > 1e-12:
            a_new = a_new * (S0 / S_new)
        else:
            a_new = a.copy()
        
        # Check if it improved
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def objective_and_grad(x):
        return lse_score_and_grad(x, beta)
    
    deadline = t_start + time_budget if (time_budget is not None and t_start is not None) else np.inf
    
    def timed_obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)
    
    try:
        res = minimize(
            timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
    except (TimeoutError, Exception) as e:
        a_out = a0
        return compute_score(a_out), a_out
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def refine_with_continuation(a0, betas, time_budget, t_start):
