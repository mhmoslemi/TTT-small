```python
"""
Coarse-to-fine LSE-LBFGS with adaptive beta schedule, multi-peak subgradient,
and structured exploration for minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    ess = float(exp_scaled.sum())
    if ess < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(ess)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / ess
    # Gradient: d(score)/d(a_i)
    # score = 2n * L / S^2
    # dL/da_i = sum_k p_k * dc_k/da_i = sum_k p_k * 2 * a_{k-i} (reversed)
    # dS/da_i = 1
    # d(score)/da_i = 2n * (dL_da_i / S^2 - 2*L / S^3)
    
    # dL/da_i = 2 * sum_k p_k * a_{k-i} (where k-i is valid)
    # This is 2 * (p * a_reversed)[i] for the linear convolution
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000, time_limit=60.0):
    """Run L-BFGS-B on LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    t0 = time.monotonic()
    
    def obj_and_grad(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return lse_obj_and_grad(x, beta)
    
    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except TimeoutError:
        a_out = a0
    except Exception as e:
        a_out = a0
    
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
```

```python
"""
Coarse-to-fine LSE-LBFGS optimizer with Nesterov subgradient refinement.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Start from parent or structured initializations
2. Coarse-to-fine L-BFGS with increasing LSE sharpness (beta)
3. Nesterov-accelerated top-M subgradient for final refinement
4. Perturbation-based restarts to escape local optima
5. Multi-length exploration
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad_computation(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Use log-sum-exp trick for numerical stability
    max_scaled = float(scaled.max())
    exp_scaled = np.exp(scaled - max_scaled)
    ess = float(exp_scaled.sum())
    L = c_max + (max_scaled + float(np.log(ess))) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / ess  # Softmax probabilities
    # dL/da = 2 * (p * a_reversed) truncated to length n
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def top_m_subgradient_nesterov(a, n_iters=20000, top_m=10, 
                                step_init=0.02, momentum=0.9):
    """
    Nesterov-accelerated subgradient method using top-M peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n, np.inf
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    v = np.zeros(n)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0.0)
        S_look = float(a_look.sum())
        if S_look < 1e-12:
            a_look = a
            S_look = S
        
        c = fast_conv(a_look)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute subgradient at top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a_look[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = min(step, 2.0 * (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        
        # Nesterov update
        v = momentum * v + step * grad
        a = a - v
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        
        # Slowly decay step
        step = max(step * 0.9999, 1e-6)
    
    return best_a, best_sc


def fienup_quick(a, n_iters=100):
    """Quick Fienup iteration to flatten power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * 1.2
        above = P > thresh
        
        scale = np.ones_like(P)
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def run_lbfgs_timed(a0, beta, time_limit=120.0, maxiter=5000):
    """L-BFGS with wall-clock time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    t0 = time.monotonic()
    best_x = a0.copy()
    best_f = compute_score(a0)
    
    # Use callback to check time and track best
    state = {'count': 0}
    
    def obj_and_grad(x):
        state['count'] += 1
        if time.monotonic() - t0 > time_limit:
            raise ValueError("timeout")
        f, g = lse_grad_computation(x, beta)
        return f, g
    
    def callback(xk):
        state['count'] += 1
        if time.monotonic() - t0 > time_limit:
            raise ValueError("timeout")
        sc = compute_score(xk)
        if sc < best_f:
            best_f = sc
            best_x = xk.copy()
        return True
    
    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=callback,
            options={'ftol': 1e-15, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': 100000, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        sc = compute_score(a_out)
        if sc < best_f:
            best_f = sc
            best_x = a_out.copy()
    except (ValueError, Exception):
        pass
    
    if best_x.sum() < 1e-6:
        best_x = a0
    return best_f, best_x
```

```python
"""
Coarse-to-fine LSE-LBFGS with Nesterov-accelerated subgradient refinement.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Quick exploration with structured initializations and Fienup
2. Coarse-to-fine L-BFGS with increasing LSE sharpness (beta 20->20000)
3. Nesterov-accelerated top-M subgradient for fine refinement
4. High-beta L-BFGS polish
5. Perturbation-based restarts to escape local optima
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """Compute LSE-smoothed score and gradient. Handles numerical stability."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_scaled = np.exp(scaled - max_scaled)
    ess = float(exp_scaled.sum())
    if ess < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        L = c_max
    else:
        L = c_max + (max_scaled + float(np.log(ess))) / beta
        p = exp_scaled / ess
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_timed(a0, beta, time_limit=60.0, maxiter=10000):
    """L-BFGS-B on LSE objective with wall-clock time limit. Returns (score, a)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    best_x = a0.copy()
    best_f = compute_score(a0)
    timeout_flag = [False]

    def obj_and_grad(x):
        if time.monotonic() - t0 > time_limit or timeout_flag[0]:
            timeout_flag[0] = True
            return float('inf'), np.zeros(n)
        f, g = lse_obj_grad(x, beta)
        if not np.isfinite(f):
            return 1e10, np.zeros(n)
        return f, g

    def callback(xk):
        if time.monotonic() - t0 > time_limit:
            timeout_flag[0] = True
            return False
        sc = compute_score(xk)
        if sc < best_f and np.isfinite(sc):
            best_f = sc
            best_x = xk.copy()
        return True

    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=callback,
            options={'ftol': 1e-15, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': 500000, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() > 1e-6:
            sc = compute_score(a_out)
            if sc < best_f and np.isfinite(sc):
                best_f = sc
                best_x = a_out.copy()
    except Exception:
        pass

    if best_x.sum() < 1e-6:
        best_x = a0
        best_f = compute_score(a0)
    return best_f, best_x


def nesterov_subgradient(a, n_iters=20000, top_m=10, step_init=0.015,
                         momentum=0.9, step_decay=0.9999):
    """Nesterov-accelerated subgradient using top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n, np.inf
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    v = np.zeros(n)
    step = step_init
    stagnation = 0
    best_sc_stag = best_sc

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 2.0 * max(sc - sc_lb, 1e-10) / (gn * gn + 1e-30))
        step = min(step, 0.04)
        step = max(step, 1e-8)

        v = momentum * v + step * grad
        a_new = a - v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            step *= 0.5
            v = v * 0.5
            continue
        a = a_new / S_new
        step *= step_decay

        if stagnation > 5000 and step > 1e-7:
            step *= 0.9
            v = v * 0.5
            stagnation = 0

    return best_a, best_sc


def fienup_flat(a, n_iters=100, thresh_mult=1.2):
    """Fienup iteration to flatten the power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n, np.inf
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * thresh_mult
        above = P > thresh

        scale = np.ones_like(P)
        if above.any():
            scale[above] = np.sqrt(thresh / P[above])

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def peak_shave(a, n_iters=5000, n_peaks=5):
    """
    Iterative peak shaving: at each step, identify the top peaks of the
    convolution and apply a targeted reduction to the contributing elements.
    Uses a smoothed max (LSE over top peaks) for a better gradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n, np.inf
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    step = 0.02
    momentum_v = np.zeros(n)
    momentum = 0.85

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peaks with some separation
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:n_peaks]

        # Weight peaks by how much they exceed the mean
        c_mean = float(c.mean())
        weights = np.maximum(c[top_indices] - c_mean, 0.0)
        wsum = float(weights.sum())
        if wsum < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / wsum

        # Compute weighted subgradient
        sg = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 1.5 * max(sc - sc_lb, 1e-10) / (gn * gn + 1e-30))
        step = min(step, 0.03)
        step = max(step, 1e-8)

        momentum_v = momentum * momentum_v + step * grad
        a_new = a - momentum_v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            step *= 0.5
            momentum_v *= 0.5
            continue
        a = a_new / S_new
        step *= 0.99995

    return best_a, best_sc


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            p = np.clip(parent.astype(np.float64), 0.0, 1000.0).copy()
            if p.sum() > 1e-6:
                inits.append(p / p.sum())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, parent)
            interp = np.maximum(interp, 0.0)
            if interp.sum() > 1e-6:
                inits.append(interp / interp.sum())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01) / n)
    inits.append(np.maximum(0.3 + 0.3 * np.cos(2.0 * np.pi * 2 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01) / n)
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x), 0.01) / n)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0) / n)
    return inits

    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS coarse-to-fine (deadline t={phase2_end - t_start:.0f}s) ===")

    n_work = len(work_a)
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        n_remaining = len(beta_schedule) - bi
        time_alloc = min(150.0, 0.75 * remaining / max(n_remaining, 1))
        time_alloc = max(time_alloc, 10.0)
        maxiter = int(time_alloc * 80)
        maxiter = max(maxiter, 500)

        sc, a_opt = run_lbfgs_timed(work_a, beta, time_limit=time_alloc, maxiter=maxiter)
        if sc < best_score:
            update_best(sc, a_opt, f"LBFGS b={beta}")
        work_a = a_opt.copy()
        work_score = compute_score(work_a)
        print(f"    beta={beta}: score={work_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Subgradient refinement (22% of time)
    # ==================================================================
    phase3_end = t_start + 0.72 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Subgradient (deadline t={phase3_end - t_start:.0f}s) ===")

    for top_m in [3, 5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        n_iters = int(remaining * 200)
        n_iters = max(n_iters, 3000)

        a_sg, sc_sg = nesterov_subgradient(
            work_a, n_iters=n_iters, top_m=top_m,
            step_init=0.01, momentum=0.9, step_decay=0.99995
        )
        if sc_sg < best_score:
            update_best(sc_sg, a_sg, f"SG top_m={top_m}")
            work_a = a_sg.copy()

    # Peak shaving
    if time.monotonic() < phase3_end - 30:
        remaining = phase3_end - time.monotonic()
        n_iters = int(remaining * 80)
        a_ps, sc_ps = peak_shave(work_a, n_iters=n_iters, n_peaks=5)
        if sc_ps < best_score:
            update_best(sc_ps, a_ps, "peak-shave")
            work_a = a_ps.copy()

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: High-beta L-BFGS polish (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.87 * (total_budget - 30.0)
    print(f"\n=== Phase 4: High-beta polish (deadline t={phase4_end - t_start:.0f}s) ===")

    for beta in [5000, 10000, 20000]:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break
        time_alloc = min(90.0, remaining * 0.7)
        sc, a_opt = run_lbfgs_timed(work_a, beta, time_limit=time_alloc, maxiter=5000)
        if sc < best_score:
            update_best(sc, a_opt, f"polish b={beta}")
            work_a = a_opt.copy()

    # More subgradient after polish
    if time.monotonic() < phase4_end - 5:
        remaining = phase4_end - time.monotonic()
        n_iters = int(remaining * 300)
        a_sg, sc_sg = nesterov_subgradient(
            work_a, n_iters=n_iters, top_m=5,
            step_init=0.005, momentum=0.92, step_decay=0.99998
        )
        if sc_sg < best_score:
            update_best(sc_sg, a_sg, "SG-final")
            work_a = a_sg.copy()

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Perturbation restarts (remaining time)
    # ==================================================================
    print(f"\n=== Phase 5: Perturbation restarts ===")

    n_best = len(best_a) if best_a is not None else n_work
    perturb_scales = [0.01, 0.02, 0.03, 0.015, 0.025]

    for trial in range(15):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        time_per = remaining / (15 - trial) * 0.7

        scale = perturb_scales[trial % len(perturb_scales)]
        a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        a_pert = np.maximum(a_pert, 0.0)
        S_pert = float(a_pert.sum())
        if S_pert < 1e-6:
            continue
        a_pert = a_pert / S_pert

        # Quick L-BFGS
        sc, a_opt = run_lbfgs_timed(a_pert, 2000, time_limit=time_per * 0.5, maxiter=3000)
        if sc < best_score:
            update_best(sc, a_opt, f"pert-{trial}")
            best_a = a_opt.copy()
            n_best = len(best_a)

        # Quick subgradient
        if time.monotonic() < deadline - 5:
            n_it = int(time_per * 100)
            a_sg, sc_sg = nesterov_subgradient(
                a_opt, n_iters=n_it, top_m=5,
                step_init=0.005, momentum=0.9, step_decay=0.99998
            )
            if sc_sg < best_score:
                update_best(sc_sg, a_sg, f"pert-{trial}-sg")
                best_a = a_sg.copy()
                n_best = len(best_a)

    print(f"After Phase 5: best = {best_score:.8f}")

    # ==================================================================
    # Final: one more L-BFGS + subgradient on absolute best
    # ==================================================================
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        sc, a_opt = run_lbfgs_timed(best_a, 10000, time_limit=remaining * 0.4, maxiter=3000)
        if sc < best_score:
            update_best(sc, a_opt, "final-polish")
            best_a = a_opt.copy()

        n_it = int(remaining * 150)
        a_sg, sc_sg = nesterov_subgradient(
            best_a, n_iters=n_it, top_m=3,
            step_init=0.003, momentum=0.93, step_decay=0.99999
        )
        if sc_sg < best_score:
            update_best(sc_sg, a_sg, "final-sg")
            best_a = a_sg.copy()

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
