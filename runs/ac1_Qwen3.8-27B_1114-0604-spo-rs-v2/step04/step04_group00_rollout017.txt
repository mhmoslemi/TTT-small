Hybrid optimizer combining Direct Fienup (targeting conv directly via complex sqrt),
Standard Fienup (targeting |FFT|^2), L-BFGS with adaptive LSE smoothing,
and Top-M subgradient. Key novelty: direct convolution-domain Fienup that

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_direct(a, n_iters=400, damping=0.5):
    """
    Direct Fienup: modifies conv(a,a) directly to reduce its peak,
    then recovers a via complex square root of FFT.
    a_new ~ IFFT(sqrt(FFT(c_modified)))
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        c = fast_conv(a)
        c_max = float(c.max())
        c_min = float(c.min())

        # Adaptive threshold: start gentle, get more aggressive
        thresh_frac = 0.85 + 0.15 * (1.0 - progress)
        thresh = c_max * thresh_frac

        # Soft modification of c
        c_mod = c.copy()
        above = c_mod > thresh
        if above.any():
            # Power-law soft clip
            alpha = 0.4 + 0.4 * progress
            ratio = c_mod[above] / thresh
            c_mod[above] = thresh * (ratio * (thresh / c_max) ** (1.0 - alpha)) ** 1.0
            # Ensure non-negative
            c_mod = np.maximum(c_mod, 0.0)

        # FFT of modified convolution
        C = np.fft.rfft(c_mod, M_fft)

        # Complex square root to recover A
        A = np.sqrt(C.astype(np.complex128))

        # Ensure DC is positive
        if A[0].real < 0:
            A = -A

        # Inverse FFT
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a = (1.0 - damping) * a + damping * a_new
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient for L-BFGS.
    Score: 2n * LSE(c; beta) / S^2
    Where LSE is the log-sum-exp of the convolution values.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Gradient: g_j = sum_k p_k * a_{k-j}
    # = cross_correlation(p, a) computed via FFT
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(P_fft * np.conj(A_fft), M2_fft)
    g = cross[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_topm(a, n_iters=5000, top_m=10, step_init=0.04, decay=0.9997):
    """
    Subgradient method using top-M peaks of the convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        subgrad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                subgrad[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (subgrad / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.06)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def spectral_phase_optimize(a, n_iters=200):
    """
    Phase-only optimization: modify the phase of FFT(a) to reduce
    the peak of IFFT(A^2), while keeping |A| fixed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current convolution peak
        c = fast_conv(a)
        m_peak = int(np.argmax(c))

        # Subgradient in frequency domain:
        # dc[m_peak]/dA_k = conj(A_k) * exp(2*pi*i*k*m_peak/M)
        # We want to decrease c[m_peak], so move A in direction -grad
        # This changes both magnitude and phase

        # Simpler: just nudge the phase
        k_vals = np.arange(len(A))
        phase_grad = -np.real(A * np.exp(2j * np.pi * k_vals * m_peak / M_fft))
        # Normalize
        pg_norm = np.linalg.norm(phase_grad)
        if pg_norm > 1e-15:
            phase_grad /= pg_norm

        # Small phase update
        step = 0.01 * (1.0 + it / n_iters)
        new_phase = phase + step * phase_grad

        # Reconstruct A with modified phase, same magnitude
        A_new = mag * np.exp(1j * new_phase)

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Revert if worse (simple rejection)
            pass

    return best_a


def two_phase_refine(a, n_direct=200, n_std=200, n_iters_inner=50):
    """
    Two-phase refinement: alternating direct Fienup and standard Fienup
    with L-BFGS in between.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for cycle in range(n_iters_inner):
        # Direct Fienup
        a_d = fienup_direct(a, n_iters=n_direct // n_iters_inner + 5, damping=0.6)
        sc_d = compute_score(a_d)
        if sc_d < best_sc:
            best_sc = sc_d
            best_a = a_d.copy()
            a = a_d

        # Standard Fienup
        a_s = fienup_standard(a, n_iters=n_std // n_iters_inner + 5, mode='soft', clip_strength=0.5)
        sc_s = compute_score(a_s)
        if sc_s < best_sc:
            best_sc = sc_s
            best_a = a_s.copy()
            a = a_s

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    # Log-normal shaped
    inits.append(np.maximum(0.2 + 0.6 * np.exp(-3.0 * (2.0 * x - 1.0) ** 2), 0.01))
    # Multi-peak
    inits.append(np.maximum(0.2 + 0.4 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Direct Fienup (novel) + Standard Fienup + L-BFGS + Subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine target length
    if parent is not None:
        n_target = len(parent)
    else:
        n_target = 6962

    print(f"Target length: {n_target}")
    print(f"Parent available: {parent is not None}")

    # Starting sequences
    start_seqs = []
    if parent is not None:
        start_seqs.append(('parent', parent.copy()))
    start_seqs.append(('uniform', np.ones(n_target) * 0.5))
    start_seqs.append(('cos', np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * np.arange(n_target) / n_target), 0.01)))
    start_seqs.append(('noise', np.clip(0.5 + 0.3 * rng.standard_normal(n_target), 0.01, 100.0)))

    # ==================================================================
    # Phase 1: Direct Fienup from multiple starts (35% of time)
    # ==================================================================
    phase1_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Direct Fienup (deadline +{phase1_end-t_start:.0f}s) ===")

    for name, a0 in start_seqs:
        if time.monotonic() > phase1_end:
            break
        print(f"  Direct Fienup from '{name}'...")
        a_d = fienup_direct(a0, n_iters=300, damping=0.5)
        sc = compute_score(a_d)
        update_best(sc, a_d, f"DIRECT-FIENUP {name}")

        # Hard version
        if time.monotonic() < phase1_end:
            a_dh = fienup_direct_hard(a0, n_iters=200)
            sc2 = compute_score(a_dh)
            update_best(sc2, a_dh, f"DIRECT-HARD {name}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Standard Fienup + Multi-scale (15% of time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Standard Fienup (deadline +{phase2_end-t_start:.0f}s) ===")

    a_work = best_a.copy()
    a_s = fienup_standard(a_work, n_iters=300, mode='soft', clip_strength=0.5)
    sc = compute_score(a_s)
    update_best(sc, a_s, "STD-FIENUP")

    if time.monotonic() < phase2_end:
        a_ms = fienup_multi_scale(a_work, n_iters=250)
        sc2 = compute_score(a_ms)
        update_best(sc2, a_ms, "MULTI-SCALE")

    # Combine: run standard Fienup on best, then direct Fienup
    if time.monotonic() < phase2_end:
        a_comb = fienup_direct(best_a, n_iters=200, damping=0.4)
        sc3 = compute_score(a_comb)
        update_best(sc3, a_comb, "COMBINED")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS with multiple betas (30% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (deadline +{phase3_end-t_start:.0f}s) ===")

    n_best = len(best_a)

    # Use adaptive beta schedule: start low (wide smoothing), end high (sharp)
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        maxfun = min(200000, int(remaining * 5000))
        maxiter = min(10000, int(remaining * 500))
        if maxfun < 1000:
            break
        print(f"  L-BFGS beta={beta} (maxfun={maxfun})...")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS beta={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient refinement (10% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Subgradient (deadline +{phase4_end-t_start:.0f}s) ===")

    for top_m in [5, 10, 20]:
        if time.monotonic() > phase4_end:
            break
        remaining = max(1000, int((phase4_end - time.monotonic()) * 500))
        a_sg = subgradient_topm(best_a, n_iters=min(remaining, 5000), top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final refinement cycle (remaining time)
    # ==================================================================
    print(f"\n=== Phase 5: Final refinement ===")

    if time.monotonic() < deadline - 20:
        # One more direct Fienup + L-BFGS cycle
        a_f = fienup_direct(best_a, n_iters=150, damping=0.3)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FINAL-DIRECT")

    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=50000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    if time.monotonic() < deadline - 10:
        # Short subgradient polish
        a_sg = subgradient_topm(best_a, n_iters=2000, top_m=5, step_init=0.02)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "FINAL-SG")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
