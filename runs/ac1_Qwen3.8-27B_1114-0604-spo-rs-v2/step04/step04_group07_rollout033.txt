    """Project v onto {a >= 0, sum a = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - tau, 0.0)

def subgrad_line_search(a0, n_iters=2000, time_limit_s=None, t_start=None):

    best_x = x.copy()

        else:
            x = project_simplex(x - best_alpha * sg)
            sc = compute_score(x)

        if sc < best_score:
            best_score = sc
            best_x = x.copy()
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha *= 0.7

    return best_score, best_x


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1e6)] * n

    def obj(x):
        x_c = np.maximum(x, 0.0)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() > 1e-12:
            a_out = a_out / a_out.sum()
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception:
        return compute_score(a0), a0


def generate_inits(rng, n):
    """Generate diverse initializations emphasizing edges."""
    inits = []
    x = np.arange(n) / max(n - 1, 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine edge-emphasis (high at edges, low at center)
    for amp in [0.3, 0.5, 0.7]:
        c = 1.0 + amp * np.cos(2.0 * np.pi * x)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Quadratic edge-emphasis
    q = 1.0 + 0.5 * (2.0 * x - 1.0) ** 2
    inits.append(q / q.sum())

    # Strong edge emphasis
    e = np.abs(2.0 * x - 1.0) ** 0.5 + 0.1
    inits.append(e / e.sum())

    # Gaussian (centered)
    for sigma in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Two-bump (off-center)
    for c1, c2 in [(0.2, 0.8), (0.15, 0.75), (0.3, 0.7)]:
        b = np.exp(-0.5 * ((x - c1) / 0.08) ** 2) + np.exp(-0.5 * ((x - c2) / 0.08) ** 2)
        inits.append(b / b.sum())

    # Periodic modulation
    for freq, amp in [(4, 0.3), (8, 0.2), (16, 0.15)]:
        p = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
        p = np.maximum(p, 0.01)
        inits.append(p / p.sum())

    # Linear ramp (high on one side)
    r = np.linspace(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Symmetric with dip
    d = 1.0 - 0.4 * np.exp(-((x - 0.5) / 0.15) ** 2)
    d = np.maximum(d, 0.01)
    inits.append(d / d.sum())

    return inits


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def perturb_sequence(a, rng, strategy=0):
    """Perturb a sequence using different strategies."""
    n = len(a)
    x = np.arange(n) / max(n - 1, 1)

    if strategy == 0:
        # Gaussian perturbation in log-space
        log_a = np.log(np.clip(a, 1e-15, 1e10))
        noise = 0.1 * rng.standard_normal(n)
        a_pert = np.exp(log_a + noise)
    elif strategy == 1:
        # Peak-avoidance: reduce elements contributing to peak
        c = fast_conv(a)
        k_star = int(c.argmax())
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        a_pert = a.copy()
        a_pert[js:je + 1] *= (0.6 + 0.2 * rng.random())
        a_pert = project_simplex(a_pert)
    elif strategy == 2:
        # Fourier perturbation
        freqs = rng.uniform(1, 12, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.05, 0.15, size=3)
        mod = 1.0
        for f, p, am in zip(freqs, phases, amps):
            mod += am * np.cos(2.0 * np.pi * f * x + p)
        a_pert = a * np.maximum(mod, 0.1)
        a_pert = project_simplex(a_pert)
    elif strategy == 3:
        # Block modification
        n_blocks = 4 + int(6 * rng.random())
        a_pert = a.copy()
        block_size = max(1, n // n_blocks)
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            a_pert[s:e] *= rng.uniform(0.5, 1.5)
        a_pert = project_simplex(a_pert)
    elif strategy == 4:
        # Edge boost
        edge_width = n // 10
        a_pert = a.copy()
        a_pert[:edge_width] *= 1.3
        a_pert[-edge_width:] *= 1.3
        a_pert = project_simplex(a_pert)
    elif strategy == 5:
        # Center dip
        center = n // 2
        width = n // 8
        a_pert = a.copy()
        dip_idx = np.arange(max(0, center - width), min(n, center + width))
        a_pert[dip_idx] *= 0.7
        a_pert = project_simplex(a_pert)
    else:
        a_pert = a + 0.05 * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = project_simplex(a_pert)

    return a_pert
