"""
Multi-phase optimizer for step function peak minimization.
Uses adaptive multi-peak gradient descent, L-BFGS with progressive smoothing,
peak-targeted coordinate descent, and simulated annealing for exploration.
Key innovations: adaptive peak weighting, frequency-domain spectral balancing,
and length-aware search strategy.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50.0))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_smooth_grad(a, beta, top_k):
    """LSE over top-k peaks only. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    sorted_idx = np.argsort(c)[::-1]
    top_indices = sorted_idx[:top_k]
    c_top = c[top_indices]
    c_max = float(c_top.max())
    scaled = beta * (c_top - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50.0))
    p_top = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    p_full = np.zeros(len(c))
    p_full[top_indices] = p_top
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p_full, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    if top_k is not None:
        def func(x):
            return topk_smooth_grad(x, beta, top_k)
    else:
        def func(x):
            return lse_grad(x, beta)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def adaptive_peak_descent(a, n_iters=3000, f_lb=1.0):
    """
    Adaptive multi-peak subgradient descent.
    Uses softmax weighting over top-K peaks with adaptive K.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.05
    decay = 0.9995
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive top-K selection
        sorted_c = np.sort(c)[::-1]
        # K = number of peaks within 10% of max
        threshold = c_max * 0.95
        K = max(3, int(np.sum(c >= threshold)))
        K = min(K, 50)
        
        # Softmax weights over top-K
        top_k_idx = np.argsort(c)[::-1][:K]
        c_top = c[top_k_idx]
        beta_w = 50.0
        exp_w = np.exp(beta_w * (c_top - c_max))
        w = exp_w / exp_w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(K):
            k = top_k_idx[ki]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += w[ki] * 2.0 * a[k - idx]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (grad / S2 - 2.0 * c_max / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-18:
            break
        
        step = step_scale * decay ** it
        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    
    return best_a


def peak_targeted_cd(a, n_sweeps=10):
    """
    Coordinate descent targeting the peak positions.
    For the current peak, identify top contributors and try to
    reduce them while redistributing to non-contributing positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Get all peaks near the max
        c_max = float(c[k_star])
        near_peaks = np.where(c >= c_max * 0.99)[0]
        
        c_improved = False
        for k in near_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            
            # Top and bottom contributors
            top_count = max(3, n // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[:top_count]
            
            # Try reducing top contributors
            for li in top_local:
                j_global = j_start + li
                a_orig = a[j_global]
                if a_orig < 1e-12:
                    continue
                
                for frac in [0.0, 0.2, 0.5, 0.7, 0.9]:
                    new_val = a_orig * frac
                    reduction = a_orig - new_val
                    if reduction < 1e-15:
                        continue
                    a[j_global] = new_val
                    
                    # Redistribute to bottom contributors
                    if len(bottom_local) > 0:
                        per = reduction / len(bottom_local)
                        for bi in bottom_local:
                            a[j_start + bi] += per
                    
                    S_tmp = a.sum()
                    if S_tmp > 1e-12:
                        a = a / S_tmp
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        c_improved = True
                    else:
                        a[j_global] = a_orig
                        if len(bottom_local) > 0:
                            for bi in bottom_local:
                                a[j_start + bi] -= reduction / len(bottom_local)
                        S_tmp = a.sum()
                        if S_tmp > 1e-12:
                            a = a / S_tmp
        
        if not c_improved and sweep > 3:
            break
    
    return best_a


def spectral_balance(a, n_iters=100):
    """
    Frequency-domain spectral balancing.
    Try to flatten |A(f)|^2 to reduce concentration in the time domain.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        power = np.abs(A) ** 2
        total_power = power.sum()
        
        # Identify high-power frequency bins
        p_max = power.max()
        p_min = power[1:].min() if len(power) > 1 else 0.0
        if p_max <= 1e-15:
            break
        
        # Create a target: compressed dynamic range
        alpha = 0.3 + 0.4 * (it / n_iters)  # increasing compression
        target = np.sign(A) * (np.abs(A) ** alpha)
        
        # Transform back
        a_new = np.fft.irfft(target, M_fft)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a


def simulated_annealing(a, n_iter=5000, T_start=0.05, T_end=0.001):
    """Simulated annealing on the sequence."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    rng = np.random.default_rng(123)
    
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        # Random perturbation
        strategy = rng.integers(0, 4)
        a_new = a.copy()
        
        if strategy == 0:
            # Gaussian noise
            noise = rng.standard_normal(n) * T * 0.1
            a_new += noise
        elif strategy == 1:
            # Random subset rescale
            n_sel = max(10, n // 20)
            sel = rng.choice(n, size=n_sel, replace=False)
            a_new[sel] *= (1.0 + T * rng.standard_normal(n_sel) * 0.5)
        elif strategy == 2:
            # Peak-targeted: reduce top contributors
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            seg = a_new[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            n_top = max(5, n // 50)
            top_local = np.argsort(contrib)[::-1][:n_top]
            a_new[j_start + top_local] *= (1.0 - T * rng.random(n_top) * 0.3)
        else:
            # Swap: move mass from one position to another
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            delta = a[i1] * T * 0.5
            a_new[i1] -= delta
            a_new[i2] += delta
        
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        
        sc_new = compute_score(a_new)
        delta = sc_new - cur_sc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
    
    return best_a

def diversify_init(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(parent)), parent))
    
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Gaussian noise
    a_g = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 5.0)
    inits.append(a_g)
    
    # U-shaped
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 4)
    
    # Cosine window
    inits.append(np.maximum(0.05, np.cos(np.pi * x) ** 2))
    
    # Random smooth (low-pass filtered noise)
    a_rand = rng.standard_normal(n)
    kernel_size = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3.0)) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    a_smooth = np.clip(a_smooth, 0.05, 5.0)
    inits.append(a_smooth)
    
    # Bump functions
    center = 0.2 + 0.6 * rng.random()
    width = 0.05 + 0.3 * rng.random()
    inits.append(np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.05)
    
    # Two bumps
    c1, c2 = sorted(rng.random(2) * 0.6 + 0.2)
    inits.append(np.exp(-((x - c1) ** 2) / 0.01) + np.exp(-((x - c2) ** 2) / 0.01) + 0.05)
    
    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.0, 1000.0)
        S = inits[i].sum()
        if S > 1e-12:
            inits[i] = inits[i] / S
    
    return inits

        remaining = phase4_deadline - time.monotonic()
        n_sa = int(min(3000, max(500, remaining * 5)))
        a_sa = simulated_annealing(best_a, n_iter=n_sa)
        sc = compute_score(a_sa)
        report(sc, a_sa, "annealing")
        
        # Polish after annealing
        if sc < best_score * 1.01 and time.monotonic() < phase4_deadline - 20:
            sc2, a2 = run_lbfgs(a_sa, 10000, maxiter=10000, maxfun=40000)
            report(sc2, a2, "anneal+LBFGS")
    
    print(f"After Phase 4: best = {best_score:.10f}")
    
    # ==================================================================
    # Phase 5: Final high-precision refinement (10% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Refinement ---")
    
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > deadline - 20:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000)
        report(sc, a_opt, f"final beta={beta}")
    
    # Final adaptive gradient pass
    if time.monotonic() < deadline - 20:
        a_final = adaptive_peak_descent(best_a, n_iters=5000, f_lb=1.0)
        sc = compute_score(a_final)
        report(sc, a_final, "final PGD")
    
    # One more CD pass
    if time.monotonic() < deadline - 15:
        a_cd_final = peak_targeted_cd(best_a, n_sweeps=15)
        sc = compute_score(a_cd_final)
        report(sc, a_cd_final, "final CD")
    
    # Try a different length if time permits
    if time.monotonic() < deadline - 60 and len(best_a) > 1024:
        print(f"\n--- Phase 6: Length Exploration ---")
        for n_try in [len(best_a) // 2, int(len(best_a) * 1.3)]:
            if time.monotonic() > deadline - 30:
                break
            if 256 <= n_try <= 50000:
                sc_l, a_l = multi_length_optimize(best_a, n_try)
                report(sc_l, a_l, f"len={n_try}")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
