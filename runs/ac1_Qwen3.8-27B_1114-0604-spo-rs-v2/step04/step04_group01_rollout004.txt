import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    a = a * (target / S)
    return a


def resample_seq(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a)


def exact_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(c.max())
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def soft_active_weights(d, k, temperature=0.05):
    m = d.size
    if m == 0:
        return np.array([], dtype=np.float64)
    dmax = float(d.max())
    if dmax <= 0:
        return np.ones(m) / m
    z = d / (dmax * temperature)
    e = np.exp(z - z.max())
    w = e / e.sum()
    # Boost top-k
    if k < m:
        idx = np.argpartition(d, m - k)[m - k:]
        w_boost = w.copy()
        w_boost[idx] *= 3.0
        w_boost /= w_boost.sum()
        w = 0.5 * w + 0.5 * w_boost
    return w


def lse_objective_grad(a, beta, pen=1e-4):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    d = fast_self_conv(a)
    d = (float(n) * d) / (S * S)
    dmax = float(d.max())
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + pen * (S - 1.0) ** 2
    grad = (4.0 * n * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * pen * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    rem = deadline - time.monotonic()
    if rem < 5.0:
        return exact_score(a), a
    est = 0.01 * max(0.5, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    cap = int((0.85 * rem) / max(est, 1e-4))
    maxfun = min(maxfun, max(64, min(cap, 2000)))
    maxiter = max(30, maxfun // 2)
    bounds = [(0.0, 10.0)] * n

    def obj(x):
        return lse_objective_grad(x, beta)

    try:
        res = minimize(obj, a, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-14, 'gtol': 1e-11,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40})
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def mwu_step(a, eta, k, temperature):
    d = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    d_norm = d / S
    w = soft_active_weights(d_norm, k, temperature)
    h = corr_pair(a, w)
    hmax = float(h.max())
    if hmax < 1e-15:
        return a, np.inf
    q = a * np.exp(-eta * h / hmax)
    q = normalize_a(q)
    return q, exact_score(q)


def adaptive_mwu(a, deadline, max_iters=300, k=5, temp_start=0.1, temp_end=0.03):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, max_iters)
        temp = temp_start + (temp_end - temp_start) * progress
        eta_base = 0.3 * (1.0 - 0.5 * progress)

        best_local_a = a.copy()
        best_local_sc = best_sc

        for mult in (0.1, 0.3, 0.6, 1.0, 2.0):
            eta = eta_base * mult
            q, sc = mwu_step(a, eta, k, temp)
            if np.isfinite(sc) and sc < best_local_sc:
                best_local_sc = sc
                best_local_a = q.copy()

        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best_a = best_local_a.copy()
        a = best_local_a

        if it % 50 == 0:
            print("  MWU iter {} score={:.8f}".format(it, best_sc))

    return best_sc, best_a


def generate_inits(n, rng, parent):
    inits = []

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(normalize_a(p))
        inits.append(normalize_a(p[::-1]))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    # Linear ramps
    x = np.linspace(0, 1, n)
    for alpha in (0.3, 0.5, 0.7):
        v = 1.0 + alpha * (2.0 * x - 1.0)
        inits.append(normalize_a(v))
        v = 1.0 - alpha * (2.0 * x - 1.0)
        inits.append(normalize_a(v))

    # Cosine
    inits.append(normalize_a(1.0 + 0.4 * np.cos(2 * np.pi * x)))
    inits.append(normalize_a(1.0 + 0.4 * np.cos(4 * np.pi * x)))
    inits.append(normalize_a(1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)))

    # Multi-frequency (incommensurate)
    for _ in range(8):
        v = np.ones(n, dtype=np.float64)
        K = rng.integers(3, 7)
        for _ in range(K):
            f = rng.uniform(0.5, n / 4.0)
            amp = rng.uniform(0.05, 0.30)
            ph = rng.uniform(0, 2 * np.pi)
            v += amp * np.cos(2 * np.pi * f * x + ph)
        inits.append(normalize_a(v))

    # Power-law spectrum
    for _ in range(6):
        Nf = next_power_of_two(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        exp_p = rng.uniform(1.0, 3.0)
        power = 1.0 / ((np.abs(freqs) + 1.0) ** exp_p)
        power[0] = 0.0
        phase = rng.uniform(0, 2 * np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - sig.mean()
        v = 1.0 + rng.uniform(0.3, 0.8) * sig
        inits.append(normalize_a(v))

    # Gapped (sparse blocks)
    for frac in (0.4, 0.5, 0.6, 0.7):
        block_size = max(2, n // 20)
        pattern = np.zeros(n)
        for start in range(0, n, block_size * 2):
            end = min(start + block_size, n)
            pattern[start:end] = 1.0
        mask = rng.random(n) < frac
        v = np.where(mask, rng.uniform(0.5, 1.5, n), 0.0)
        inits.append(normalize_a(v))

    # Random
    for _ in range(4):
        inits.append(normalize_a(rng.random(n)))

    # Smoothed random (low-pass)
    for _ in range(4):
        v = rng.random(n)
        k = max(3, n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (k / 4.0))
        kern /= kern.sum()
        v = np.convolve(v, kern, mode='same')
        inits.append(normalize_a(v))

    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(a)
    n = len(p)
    if strength is None:
        strength = rng.uniform(0.1, 0.5)
    mode = rng.integers(0, 5)
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.1, 0.4)
        q = (1 - t) * p + t / n
    elif mode == 2:
        q = p[::-1].copy() * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        d = fast_self_conv(p)
        dmax = d.max()
        if dmax > 0:
            w = soft_active_weights(d / dmax, 4, 0.05)
            h = corr_pair(p, w)
            hmax = h.max()
            if hmax > 1e-15:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    else:
        logp = np.log(p + 1e-30)
        noise = rng.standard_normal(n)
        k = max(3, min(31, n // 50))
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (k / 4.0))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        noise -= noise.mean()
        logp += strength * 0.5 * noise
        q = np.exp(logp - logp.max())
    return normalize_a(q)


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # MWU phase
    rem = deadline - time.monotonic()
    if rem > 60:
        sc, a2 = adaptive_mwu(a, time.monotonic() + rem * 0.5, max_iters=150)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = a2

    # LSE polish
    for beta in (200, 800, 3000, 10000):
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        sc, a2 = run_lbfgs(a, beta, deadline, 500)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    return best_sc, best_a


def optimize_trajectory(a_start, deadline, label):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    t0 = time.monotonic()
    total = max(20.0, deadline - t0 - 10.0)

    print("[{}] start score={:.8f} n={}".format(label, best_sc, n))

    # Phase 1: MWU with cooling
    mwu_dead = t0 + 0.4 * total
    sc, a2 = adaptive_mwu(a, mwu_dead, max_iters=200, k=5, temp_start=0.12, temp_end=0.04)
    if sc < best_sc:
        best_sc = sc
        best_a = a2.copy()
    a = a2

    # Phase 2: LSE continuation
    for beta, frac in [(100, 0.1), (300, 0.1), (1000, 0.15), (3000, 0.15), (8000, 0.1), (20000, 0.1)]:
        sub_dead = t0 + frac * total
        sub_dead = min(sub_dead, deadline - 10)
        while time.monotonic() < sub_dead - 3:
            rem = sub_dead - time.monotonic()
            if rem < 8:
                break
            sc, a2 = run_lbfgs(a, beta, sub_dead, 300)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={}".format(label, best_sc, beta))
            a = normalize_a(a2)

    # Phase 3: More MWU with smaller temp
    rem = deadline - time.monotonic()
    if rem > 30:
        sc, a2 = adaptive_mwu(a, deadline - 15, max_iters=100, k=3, temp_start=0.05, temp_end=0.02)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    # Phase 4: High-beta LSE
    rem = deadline - time.monotonic()
    if rem > 20:
        sc, a2 = run_lbfgs(a, 30000, deadline - 5, 400)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] final high-beta {:.8f}".format(label, best_sc))

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} len={}".format(parent is not None, parent.size if parent is not None else 0))

    # Lengths to try
    lengths = []
    if parent is not None:
        pl = min(parent.size, 12000)
        if pl >= 1024:
            lengths.append(pl)
        if pl < 8000:
            lengths.append(8000)
        if pl > 5000:
            lengths.append(5000)
    else:
        lengths = [8000, 6000, 10000]

    # Deduplicate
    seen = set()
    unique_lengths = []
    for n in lengths:
        n = int(n)
        if 1024 <= n <= 12000 and n not in seen:
            unique_lengths.append(n)
            seen.add(n)
    lengths = unique_lengths[:3]
    print("Lengths: {}".format(lengths))

    # Generate and score initial sequences
    all_seqs = []
    all_scores = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score(p)
            if np.isfinite(sc) and sc < 2.0:
                all_seqs.append(p)
                all_scores.append(sc)
                all_lens.append(n)

    if not all_seqs:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(exact_score(all_seqs[0]))
        all_lens.append(n0)

    order = np.argsort(np.array(all_scores))
    print("Best initial: {:.8f} ({} candidates)".format(all_scores[order[0]], len(all_scores)))

    # Select top starts (max 3, different lengths if possible)
    starts = []
    used_lens = set()
    for idx in order:
        nl = all_lens[idx]
        if nl not in used_lens and len(starts) < 3:
            starts.append(all_seqs[idx].copy())
            used_lens.add(nl)
    while len(starts) < min(3, len(lengths)):
        starts.append(perturb(starts[0], rng))

    best_score = min(exact_score(s) for s in starts)
    best_seq = [s for s in starts if exact_score(s) == best_score][0].copy()
    print("Selected starts: scores =", [round(exact_score(s), 6) for s in starts])

    # Allocate time to trajectories
    main_budget = deadline - t_start - 180.0
    if len(starts) == 1:
        fr = [1.0]
    elif len(starts) == 2:
        fr = [0.60, 0.40]
    else:
        fr = [0.50, 0.30, 0.20]

    cum = 0.0
    for i, st in enumerate(starts):
        cum += fr[i]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, deadline - 180.0)
        if time.monotonic() > sub_dead - 30:
            continue

        sc, seq = optimize_trajectory(st, sub_dead, r"traj{}".format(i + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(i + 1, best_score))

    best_seq = normalize_a(best_seq)
    print("After trajectories: {:.8f} n={}".format(best_score, len(best_seq)))

    # Final intensive polishing of global best
    if time.monotonic() < deadline - 80:
        rem = deadline - time.monotonic() - 80
        # More MWU
        sc, a2 = adaptive_mwu(best_seq, t_start + rem * 0.6 + time.monotonic(),
                              max_iters=150, k=3, temp_start=0.04, temp_end=0.015)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Polish MWU: {:.8f}".format(best_score))

    # Perturbation restarts
    perturb_count = 0
    while time.monotonic() < deadline - 100 and perturb_count < 4:
        left = deadline - 100 - time.monotonic()
        if left < 60:
            break
        chunk = time.monotonic() + left / (4 - perturb_count)
        p = perturb(best_seq, rng, strength=rng.uniform(0.15, 0.4))
        sc, seq = quick_optimize(p, min(chunk, deadline - 60))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {} best: {:.8f}".format(perturb_count, best_score))
        perturb_count += 1

    # Final high-beta LSE
    if time.monotonic() < deadline - 30:
        sc, a2 = run_lbfgs(best_seq, 25000, deadline - 10, 500)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Final LSE: {:.8f}".format(best_score))

    # Final micro-MWU
    if time.monotonic() < deadline - 15:
        sc, a2 = adaptive_mwu(best_seq, deadline - 12, max_iters=40, k=2,
                              temp_start=0.02, temp_end=0.01)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()

    best_seq = normalize_a(best_seq)
    out_arr = best_seq.astype(np.float64)
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=0.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
