def solve_lp_for_g(a, n_constraints_margin=0):
    """
    Given a fixed a (sum=1, non-negative), solve:
    min t
    s.t. (a*g)[k] <= t for k=0,...,2n-2
         sum(g) = 1
         g >= 0
    
    Returns the optimal g and t.
    """
    n = len(a)
    # Build the constraint matrix
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    
    # Using scipy linprog
    # c = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
    # A_ub * [g, t] <= b_ub
    # (a*g)[k] - t <= 0 for each k
    # A_eq * [g, t] = b_eq
    # sum(g) = 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: (a*g)[k] - t <= 0
    # A_ub is (2n-1) x (n+1)
    A_ub = np.zeros((2*n-1, n+1))

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
from scipy import linalg
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def solve_best_response(a):
    """
    Given fixed a (non-negative), solve:
    min t
    s.t. (a*g)[k] <= t for k=0,...,2n-2
         sum(g) = 1
         g >= 0
    Returns (g_opt, t_opt).
    """
    n = len(a)
    
    # Build sparse A_ub matrix: (2n-1) x (n+1)
    # Row k: sum_j a[k-j]*g[j] - t <= 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        rows.append(k)
        cols.append(n)  # t variable
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n - 1)
    
    # Equality: sum(g) = 1
    A_eq = csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n+1))
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

"""
LP-based block coordinate descent for minimizing 2n*max(self-conv)/(sum)^2.
Key innovation: exact LP solution for the best response at each BCD iteration,
combined with line search and L-BFGS polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    na, nb = len(a), len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(self-conv) / (sum)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def build_sparse_conv_matrix(a):
    """
    Build sparse matrix A where (A @ g)[:2n-1] = (a * g) (linear cross-conv).
    A is (2n-1) x n.
    """
    n = len(a)
    n_rows = 2 * n - 1
    
    ks = np.arange(n_rows, dtype=np.int64)
    js_start = np.maximum(0, ks - n + 1)
    js_end = np.minimum(ks, n - 1)
    lengths = js_end - js_start + 1
    total = int(lengths.sum())
    
    row_g = np.repeat(ks, lengths)
    cum = np.cumsum(lengths)
    offsets = np.arange(total) - np.repeat(cum - lengths, lengths)
    col_g = np.repeat(js_start, lengths) + offsets
    idx = row_g - col_g
