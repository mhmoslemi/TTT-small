"""
Multi-phase optimizer combining alternating exponential (MWUA) global search
with L-BFGS LSE local refinement for minimizing step function peak autocorrelation.
Key innovations: alternating multiplicative weight updates for global exploration,
spectral domain perturbations for diversity, and multi-length exploration.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_pair(a, b):
    """Linear convolution of a and b via real FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def func(x):
        return lse_score_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def compute_v_alpha_b(alpha, b):

    n = len(b)
    alpha_pad = np.concatenate([alpha, np.zeros(1)])

def spectral_perturb(a, rng, method=0):
    """Perturb a in the spectral domain for diversity."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)

    if method == 0:
        # Random phase perturbation
        phases = 2 * np.pi * rng.random(A.shape)
        A_pert = A * np.exp(1j * phases)
    elif method == 1:
        # Magnitude perturbation
        mag_scale = 1.0 + 0.3 * np.tanh(0.5 * rng.standard_normal(A.shape))
        A_pert = A * mag_scale
    elif method == 2:
        # Low-pass emphasis (smooth the spectrum)
        freqs = np.linspace(0, 1, A.shape[0])
        cutoff = 0.5 + 0.5 * rng.random()
        mask = np.exp(-((freqs - cutoff) ** 2) / 0.1)
        A_pert = A * (1.0 + 0.5 * mask)
    elif method == 3:
        # Band-stop perturbation
        center = rng.random()
        width = 0.05 + 0.1 * rng.random()
        freqs = np.linspace(0, 1, A.shape[0])
        rejection = np.exp(-((freqs - center) ** 2) / (2 * width ** 2))
        A_pert = A * (1.0 - 0.5 * rejection)
    else:
        # Mixed
        phases = 2 * np.pi * rng.random(A.shape)
        mag_scale = 1.0 + 0.2 * rng.standard_normal(A.shape)
        A_pert = A * mag_scale * np.exp(1j * phases * 0.3)

    a_pert = np.fft.irfft(A_pert, M_fft)[:n]
    a_pert = np.clip(a_pert, 0.0, None)
    a_sum = a_pert.sum()
    if a_sum < 1e-10:
        # Fallback: time-domain perturbation
        a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        a_pert = np.clip(a_pert, 0.0, None)
        a_sum = a_pert.sum()
    if a_sum > 1e-10:
        a_pert /= a_sum
    else:
        a_pert = a.copy()
    return a_pert


def nesterov_subgrad(a, n_iters=3000, f_lb=1.45):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()

        return a
    a = a / S

    t = 1.0

    best_sc = compute_score(a)

        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_end = min(k_star, n - 1)

        if j_end >= j_start:

            sg_M[idx] = 2.0 * y[k_star - idx]

                mid = k_star // 2
                if 0 <= mid < n and j_start <= mid <= j_end:
                    sg_M[mid] = y[mid]  # Only count once for the square term

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def peak_surgery(a, n_rounds=30, n_top=5):
    """Targeted reduction of top peaks via local adjustments."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        # Get top n_top peaks
        peak_indices = np.argsort(c)[::-1][:n_top]

        for k_star in peak_indices:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            # Find top contributors
            top_count = max(3, len(contrib) // 20)
            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[:top_count]

            for li in top_local:
                j = j_start + li
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                # Try reducing
                reduction = old_val * 0.3
                a[j] = old_val - reduction
                # Redistribute to bottom
                per = reduction / len(bottom_local)
                for bi in bottom_local:
                    a[j_start + bi] += per
                a = np.clip(a, 0.0, None)
                S_tmp = a.sum()
                if S_tmp > 1e-12:
                    a = a / S_tmp
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    # Revert
                    a[j] = old_val
                    for bi in bottom_local:
                        a[j_start + bi] -= per
                    a = np.clip(a, 0.0, None)
                    S_tmp = a.sum()
                    if S_tmp > 1e-12:
                        a = a / S_tmp

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: alternating exponential (global) + L-BFGS LSE (local)
    + spectral perturbations + multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 25.0

    # Load parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1e10)
                if parent.sum() > 1e-12:
                    parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Evaluate parent
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        report(sc_p, parent, "parent")

    # Working lengths to try
    if parent is not None:
        n_parent = len(parent)
        lengths = sorted(set([n_parent, 8192, 12288, 16384]))
    else:
        lengths = [8192, 12288, 16384]
    # Filter reasonable lengths
    lengths = [n for n in lengths if 1024 <= n <= 32768]

    print(f"Lengths to explore: {lengths}")
    print(f"Best so far: {best_score:.10f}")

    # ==================================================================
    # Phase 1: Alternating exponential for global exploration (~35% time)
    # ==================================================================
    phase1_frac = 0.35
    phase1_deadline = t_start + phase1_frac * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: Alternating Exponential (deadline {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    candidates = []  # list of (score, array)

    for n_try in lengths:
        if time.monotonic() > phase1_deadline:
            break
        # Initialize from parent or uniform
        if parent is not None:
            if len(parent) == n_try:
                a_init = parent.copy()
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n_try)
                a_init = np.interp(x_new, x_old, parent)
                a_init = np.clip(a_init, 1e-10, None)
                a_init /= a_init.sum()
        else:
            x = np.linspace(0, 1, n_try)
            a_init = 0.3 + 0.7 * np.cos(np.pi * x)
            a_init = np.clip(a_init, 0.01, None)
            a_init /= a_init.sum()

        # Run alternating exponential with different parameters
        for (eta_s, eta_e, beta_s, beta_e, gamma, n_iter) in [
            (5.0, 0.5, 20.0, 200.0, 0.5, 200),
            (10.0, 1.0, 50.0, 500.0, 0.3, 150),
            (3.0, 0.2, 10.0, 100.0, 0.7, 300),
        ]:
            if time.monotonic() > phase1_deadline:
                break
            a_alt = alternating_exponential(
                a_init, n_iters=n_iter,
                eta_start=eta_s, eta_end=eta_e,
                beta_start=beta_s, beta_end=beta_e,
                gamma=gamma
            )
            sc = compute_score(a_alt)
            label = f"alt n={n_try} e={eta_s} b={beta_s}"
            report(sc, a_alt, label)
            candidates.append((sc, a_alt))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best = {best_score:.10f}, {len(candidates)} candidates")

    # ==================================================================
    # Phase 2: L-BFGS refinement on best candidates (~35% time)
    # ==================================================================
    phase2_deadline = t_start + 0.70 * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Refinement (deadline {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Take top candidates and refine
    n_refine = min(5, len(candidates))
    for ci in range(n_refine):
        if time.monotonic() > phase2_deadline:
            break
        sc_init, a_init = candidates[ci]
        n_curr = len(a_init)

        # Progressive beta refinement
        for beta in [100, 500, 2000, 10000, 50000]:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 60:
                break
            maxiter = int(min(30000, max(3000, remaining * 30)))
            maxfun = int(min(100000, max(10000, remaining * 150)))
            sc, a_opt = run_lbfgs(a_init, beta, maxiter=maxiter, maxfun=maxfun)
            report(sc, a_opt, f"LBFGS b={beta} n={n_curr}")
            a_init = a_opt

    print(f"Phase 2 done. Best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Spectral perturbations + refinement (~20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.90 * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral Perturbations (deadline {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 60:
            break

        method = trial % 5
        a_pert = spectral_perturb(best_a, rng, method=method)
        sc_pert = compute_score(a_pert)

        # Quick L-BFGS on perturbation
        beta = 200 + 300 * trial
        maxiter = int(min(20000, max(2000, remaining * 20)))
        maxfun = int(min(60000, max(5000, remaining * 100)))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"spect #{trial} m={method}")

        # If perturbation itself was good, try another beta
        if time.monotonic() < phase3_deadline - 30 and sc_pert < best_score + 0.01:
            sc2, a_opt2 = run_lbfgs(a_pert, 5000, maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"spect #{trial} refine")

    print(f"Phase 3 done. Best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Final polish (~10% time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    # Peak surgery
    if time.monotonic() < deadline - 60:
        a_surg = peak_surgery(best_a, n_rounds=50, n_top=5)
        sc = compute_score(a_surg)
        report(sc, a_surg, "peak_surgery")
        if sc < best_score:
            # Refine the surgery result
            sc_r, a_r = run_lbfgs(a_surg, 10000, maxiter=20000, maxfun=60000)
            report(sc_r, a_r, "surgery+refine")

    # High-beta L-BFGS
    if time.monotonic() < deadline - 40:
        for beta in [100000, 300000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=30000, maxfun=100000)
            report(sc, a_opt, f"final beta={beta}")

    # Nesterov subgradient
    if time.monotonic() < deadline - 30:
        a_ns = nesterov_subgrad(best_a, n_iters=5000, f_lb=1.45)
        sc = compute_score(a_ns)
        report(sc, a_ns, "nesterov_final")

    # One more alternating pass on the best
    if time.monotonic() < deadline - 20:
        n_best_len = len(best_a)
        a_alt2 = alternating_exponential(
            best_a, n_iters=100,
            eta_start=2.0, eta_end=0.3,
            beta_start=100.0, beta_end=1000.0,
            gamma=0.3
        )
        sc = compute_score(a_alt2)
        report(sc, a_alt2, "alt_final")
        if sc < best_score:
            sc_r, a_r = run_lbfgs(a_alt2, 50000, maxiter=10000, maxfun=30000)
            report(sc_r, a_r, "alt_final+refine")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if not out or sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
