Let me also think about the step size. If we use $x_j \leftarrow x_j - \eta \cdot sg_j / S$, then:
$a_j \leftarrow a_j \cdot e^{-\eta \cdot sg_j / S}$

For the score to decrease, we need the peak to decrease faster than the sum squared. Since we renormalize after each step, the sum is always 1, so we just need the peak to decrease.

The reduction in the peak: $\Delta b_{k^*} \approx -\eta \cdot (sg \cdot a)_{k^*} / S = -\eta \cdot \sum_j sg_j a_j / S$

So the peak decreases by a factor of $(1 - 2\eta)$ per step (approximately). We need $\eta < 1/2$ for convergence.

But we also need to make sure other peaks don't increase too much. This is where the LSE-smoothed approach is better: it considers all peaks, not just the maximum.

        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': 50000, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    a_out = np.maximum(a_out, 0.0)
    s = a_out.sum()
    if s > 1e-12:
        a_out = a_out / s
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multi_peak_subgrad_descent(a_init, n_iters, n_peaks, lr, time_limit_s, t_start):
    """Subgradient descent using top-K peaks of the convolution."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break

        c = fast_conv(a)
        S = float(a.sum())
        S2 = S * S

        # Find top peaks
        top_k = np.argsort(c)[::-1][:n_peaks]
        # Weight by how much above average
        c_avg = float(c.mean())
        w = np.array([max(0, c[k] - c_avg) for k in top_k])
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(top_k)) / len(top_k)
        else:
            w = w / w_sum

        # Accumulate subgradients
        sg = np.zeros(n)
        for wi, k in zip(w, top_k):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                idx = np.clip(idx, 0, n - 1)
                sg[js:je + 1] += wi * 2.0 * a[idx]

        # Update
        a = a - lr * (sg / S2 - 2.0 * float(c[top_k].max()) / (S2 * S))
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    Uses LSE-smoothed Adam in log-space with adaptive temperature,
    multiplicative weight updates, random perturbations, and L-BFGS polish.
    """
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def do_update(sc, arr, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(arr, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(arr)} | t={elapsed:.0f}s")
            return True
        return False

    # Phase 0: Initial candidates
    print("=" * 60)
    print("Phase 0: Initial evaluation")
    print("=" * 60)

    if parent is not None and parent.sum() > 0.01:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
        do_update(parent_score, parent, "parent")

    for n_try in [2048, 4096, 8192]:
        a_u = np.ones(n_try)
        sc = compute_score(a_u)
        do_update(sc, a_u, f"uniform n={n_try}")

    if best_a is None:
        best_a = np.ones(4096)
        best_score = 2.0

    n_best = len(best_a)
    print(f"Starting best: {best_score:.10f}, n={n_best}")

    # Phase 1: Main LSE-smoothed Adam in log-space
    print("\n" + "=" * 60)
    print("Phase 1: Adam log-space (adaptive beta)")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.35, 800.0)
    if t1 > 60:
        sc, a_opt = adam_log_a(best_a, 50, 2000, 300000, 0.02, t1, t_start)
        do_update(sc, a_opt, "Adam log main")
        best_a = a_opt

        # Second run with different params
        remaining = effective_budget - (time.monotonic() - t_start)
        t2 = min(remaining * 0.15, 300.0)
        if t2 > 30:
            sc, a_opt = adam_log_a(best_a, 200, 5000, 100000, 0.005, t2, t_start)
            do_update(sc, a_opt, "Adam log hi-beta")
            if sc < best_score:
                best_a = a_opt

    # Phase 2: Multiplicative weight updates
    print("\n" + "=" * 60)
    print("Phase 2: Multiplicative weight updates")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 300.0)
    if t3 > 20:
        for eta in [0.01, 0.005, 0.002]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 15:
                break
            sc, a_opt = multiplicative_updates_fixed(best_a, 500000, eta,
                                                     min(remaining * 0.3, 80.0), t_start)
            do_update(sc, a_opt, f"MultUpdate eta={eta}")
            if sc < best_score:
                best_a = a_opt

    # Phase 3: Multi-peak subgradient descent
    print("\n" + "=" * 60)
    print("Phase 3: Multi-peak subgradient")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        for n_peaks, lr_val in [(3, 0.001), (10, 0.0005), (50, 0.0002)]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 15:
                break
            sc, a_opt = multi_peak_subgrad_descent(best_a, 200000, n_peaks, lr_val,
                                                   min(remaining * 0.2, 60.0), t_start)
            do_update(sc, a_opt, f"MultiPeak K={n_peaks}")
            if sc < best_score:
                best_a = a_opt

    # Phase 4: Random perturbations + re-optimization
    print("\n" + "=" * 60)
    print("Phase 4: Random perturbations")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        n_pert = max(2, int(remaining / 60))
        n_pert = min(n_pert, 15)
        time_per = remaining * 0.6 / max(1, n_pert)

        for scale in [0.3, 0.1, 0.05]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 30:
                break
            sc, a_opt = random_perturb_reoptimize(
                best_a, n_pert, scale,
                min(remaining * 0.25, 120.0) / max(1, n_pert),
                t_start, seed=seed + int(scale * 100))
            do_update(sc, a_opt, f"Perturb scale={scale}")
            if sc < best_score:
                best_a = a_opt

    # Phase 5: L-BFGS polish
    print("\n" + "=" * 60)
    print("Phase 5: L-BFGS polish")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        for beta_pol in [500, 2000, 5000]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 20:
                break
            t0 = time.monotonic()
            try:
                sc, a_opt = lbfgs_polish(best_a, beta_pol, maxiter=2000)
                do_update(sc, a_opt, f"L-BFGS beta={beta_pol}")
                if sc < best_score:
                    best_a = a_opt
                print(f"    L-BFGS beta={beta_pol}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
            except Exception as e:
                print(f"    L-BFGS beta={beta_pol} failed: {e}")

    # Phase 6: Final Adam refinement
    print("\n" + "=" * 60)
    print("Phase 6: Final Adam refinement")
    print("=" * 60)

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        sc, a_opt = adam_log_a(best_a, 1000, 5000, 100000, 0.003,
                               min(remaining * 0.5, 200.0), t_start)
        do_update(sc, a_opt, "Final Adam")

        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining > 30:
            try:
                sc2, a_opt2 = lbfgs_polish(best_a if sc < best_score else a_opt,
                                           3000, maxiter=3000)
                do_update(sc2, a_opt2, "Final L-BFGS")
            except Exception:
                pass

    # Final output
    out_a = best_a
    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
