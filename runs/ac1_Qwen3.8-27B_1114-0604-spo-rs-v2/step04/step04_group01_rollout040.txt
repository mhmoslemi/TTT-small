"""
Search for non-negative sequences minimizing

    2 * n * max(conv(a, a)) / sum(a)^2

using FFT-based LSE-smoothed objectives, multi-length refinement,
active-set multiplicative pressure steps, and perturbation restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Constants and caches.
PEN_SUM = 1e-4
MAX_LEN = 12288
TIMING_CACHE = {}
BOUNDS_CACHE = {}


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def get_parent_sequence():
    try:
        h = height_sequence_1
    except Exception:
        return None
    if h is None:
        return None
    try:
        arr = np.asarray(h, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0 or float(arr.sum()) <= 0.0:
            return None
        return normalize_a(arr)
    except Exception:
        return None


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-300:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    S2 = float(a.sum())
    if S2 < 1e-300:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    out = out[:M]
    return np.maximum(out, 0.0)


def exact_score_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    S = float(a.sum())
    if S <= 0.0:
        return float("inf")
    z = a / S
    c = fast_self_conv(z)
    if c.size == 0:
        return float("inf")
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return float("inf")
    return float(2.0 * n * cmax)


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    m = int(a.size)
    if m == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if m == 1:
        return np.full(n, float(a[0]) / float(n), dtype=np.float64)
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def corr_direct(z, w):
    z = np.asarray(z, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(z.size)
    h = np.zeros(n, dtype=np.float64)
    if n == 0 or w.size == 0:
        return h
    msz = min(int(w.size), 2 * n - 1)
    if msz <= 0:
        return h
    nz = np.nonzero(w)[0]
    if nz.size == 0:
        return h
    rev = z[::-1]
    for idx in nz:
        l = int(idx)
        if l < 0 or l >= msz:
            continue
        wl = float(w[l])
        if wl == 0.0:
            continue
        if l < n:
            h[:l + 1] += wl * rev[n - 1 - l:n]
        else:
            start = l - (n - 1)
            if start >= n:
                continue
            h[start:] += wl * rev[0:n - start]
    return np.maximum(h, 0.0)


def fun_lse(x, beta, extra=0.0):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(x.size)
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)

    S = float(x.sum())
    if S <= 1e-12:
        return 1e100, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    z = x / S
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()

    A = np.fft.rfft(z, N)
    c = np.fft.irfft(A * A, N)[:M]
    c = np.maximum(c, 0.0)

    d = float(n) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    u = beta * (d - dmax)
    e = np.exp(u)
    sum_e = float(np.sum(e))
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e100, np.zeros(n, dtype=np.float64)

    w = e / sum_e
    L = dmax + float(np.log(sum_e)) / beta
    LDS = float(np.dot(w, d))

    # If weights are already sparse, avoid an FFT for the correlated gradient.
    nz_count = int(np.count_nonzero(w))
    if nz_count <= 16:
        h = corr_direct(z, w)
    else:
        W = np.fft.rfft(w, N)
        h = np.fft.irfft(A * np.conj(W), N)[:n]
        h = np.maximum(h, 0.0)

    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)
    const_term = (4.0 * LDS) / S
    pen_term = 2.0 * PEN_SUM * (S - 1.0)
    grad = (4.0 * float(n) * h) / S - const_term + pen_term

    grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    grad = np.clip(grad, -1e6, 1e6)

    if not np.isfinite(score):
        score = 1e100
    return float(score), grad


def run_lbfgs_lse(x0, beta, maxfun, deadline=None):
    x = normalize_a(x0)
    n = int(x.size)
    maxfun = int(max(0, maxfun))
    if maxfun < 24:
        return exact_score_a(x), x

    if n in BOUNDS_CACHE:
        bounds = BOUNDS_CACHE[n]
    else:
        bounds = [(0.0, 100.0)] * n
        BOUNDS_CACHE[n] = bounds

    opts = {
        "maxfun": maxfun,
        "maxiter": max(24, maxfun),
        "ftol": 1e-12,
        "gtol": 1e-6,
        "maxls": 50,
        "disp": False,
    }

    try:
        res = minimize(
            fun_lse,
            x,
            method="L-BFGS-B",
            jac=True,
            args=(float(beta), 0.0),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return exact_score_a(x), x

    if getattr(res, "x", None) is not None:
        xr = normalize_a(res.x)
        sc = exact_score_a(xr)
        if np.isfinite(sc):
            return float(sc), xr

    return exact_score_a(x), x


def get_eval_time(n, sample=None):
    n = int(n)
    key = n
    if key in TIMING_CACHE:
        return TIMING_CACHE[key]

    if sample is None:
        sample = np.ones(n, dtype=np.float64) / float(n)
    sample = np.asarray(sample, dtype=np.float64).ravel()
    if sample.size != n:
        sample = np.ones(n, dtype=np.float64) / float(n)

    reps = 3
    t0 = time.monotonic()
    for _ in range(reps):
        exact_score_a(sample)
    dt = (time.monotonic() - t0) / float(reps)

    val = max(0.005, dt * 3.2)
    TIMING_CACHE[key] = val
    return val


def maxfun_for_time(t_left, eval_time, minfun=30, capfun=2500):
    t_left = float(max(0.0, t_left))
    if t_left < 8.0:
        return 0
    if eval_time <= 1e-6:
        eval_time = 0.01
    mf = int((0.55 * t_left) / float(eval_time))
    if mf < minfun:
        mf = minfun
    if mf > capfun:
        mf = capfun
    return mf


def pressure_polish(x, max_steps=8, deadline=None):
    x = normalize_a(x)
    best = x.copy()
    best_sc = exact_score_a(x)
    cur = x.copy()
    cur_sc = best_sc
    step = 0
    no_improve = 0

    etas = (0.05, 0.15, 0.40)
    small_ks = (1, 3, 8)
    large_ks = (8, 16, 32, 55)

    while step < int(max_steps):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        c = fast_self_conv(cur)
        if c.size == 0:
            break
        d = float(cur.size) * c
        if not np.isfinite(np.max(d)):
            break

        if no_improve < 2:
            ks = [k for k in small_ks if 1 <= k < c.size]
        else:
            ks = [k for k in large_ks if 1 <= k < c.size]
        if not ks:
            ks = [1]
        ks = ks[:4]

        improved = False

        for K in ks:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            idx = np.argpartition(d, c.size - K)[c.size - K:]
            w = np.zeros(c.size, dtype=np.float64)
            w[idx] = 1.0 / float(K)

            h = corr_direct(cur, w)
            hmax = float(np.max(h))
            if not np.isfinite(hmax) or hmax <= 1e-15:
                continue

            direction = h / hmax
            for eta in etas:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                logv = np.log(cur + 1e-300) - float(eta) * direction
                maxv = float(np.max(logv))
                q = np.exp(logv - maxv)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = float(sc)
                    best = q.copy()
                    improved = True
                if sc < cur_sc:
                    cur_sc = float(sc)
                    cur = q.copy()
                    improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
        step += 1

    return float(best_sc), best


def smooth_noise(n, rng, strength):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    noise = rng.standard_normal(n)
    N = next_power_of_two(n)
    F = np.fft.rfft(noise, N)
    fr = np.fft.rfftfreq(N) * N
    cutoff = max(2.0, N * float(rng.uniform(0.03, 0.18)))
    F[fr > cutoff] = 0.0
    noise = np.fft.irfft(F, N)[:n]
    noise = noise - float(noise.mean())
    return noise * float(strength)


def perturb_best(a, rng, strength=None):
    p = normalize_a(a)
    n = int(p.size)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.12, 0.38))

    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.30))
        q = (1.0 - t) * p + t * (np.ones(n, dtype=np.float64) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        noise = smooth_noise(n, rng, strength * 0.6)
        logp = np.log(p + 1e-300) + noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 4:
        nb = int(min(max(4, n // 192), 64))
        block = 0.5 + 0.5 * rng.random(nb)
        q = np.repeat(block, max(1, n // nb))
        if q.size < n:
            q = np.concatenate([q, q[:n - q.size]])
        q = q[:n]
        t = float(rng.uniform(0.35, 0.70))
        q = (1.0 - t) * p + t * q
    else:
        q = p.copy()
        lo = max(1, n // 20)
        hi = max(lo + 1, n // 2)
        seg_len = int(rng.integers(lo, hi))
        s = int(rng.integers(0, max(1, n - seg_len)))
        e = min(n, s + seg_len)
        q[s:e] = p[s:e][::-1]
        q = q * np.exp(0.03 * rng.standard_normal(n))

    return normalize_a(q)


def add_candidate(lst, arr, cap):
    if len(lst) >= int(cap):
        return
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return
    arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
    mn = float(np.min(arr))
    if mn < 0.0:
        arr = arr - mn + 0.05
    p = normalize_a(arr)
    if p.size > 0 and np.isfinite(p.sum()):
        lst.append(p)


def make_inits(n, rng, parent, cap=None):
    n = int(n)
    inits = []
    if cap is None:
        cap = 18 if n > 8192 else 24

    if parent is not None and parent.size > 0:
        if n == parent.size:
            add_candidate(inits, parent, cap)
        else:
            add_candidate(inits, resample_sequence(parent, n), cap)
        add_candidate(inits, resample_sequence(parent[::-1], n), cap)

    x = np.linspace(0.0, 1.0, n)

    add_candidate(inits, np.ones(n, dtype=np.float64), cap)

    for alpha in (0.25, 0.45, 0.65, 0.85):
        add_candidate(inits, 1.0 + alpha * (2.0 * x - 1.0), cap)
        add_candidate(inits, 1.0 - alpha * (2.0 * x - 1.0), cap)

    max_trig = 8 if n > 8192 else 12
    max_f = max(2, n // 8)
    for _ in range(max_trig):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max_f + 1))
            amp = float(rng.uniform(0.08, 0.35))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
            if rng.random() < 0.5:
                vals += 0.5 * amp * np.sin(2.0 * np.pi * f * x + ph)
        add_candidate(inits, vals, cap)

    Nf = next_power_of_two(n)
    max_low = 4 if n > 8192 else 7
    for _ in range(max_low):
        freqs = np.fft.rfftfreq(Nf) * Nf
        p = float(rng.uniform(1.3, 3.2))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** p)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=power.size)
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        vals = 1.0 + float(rng.uniform(0.3, 0.95)) * noise
        add_candidate(inits, vals, cap)

    for frac in (0.35, 0.55, 0.75):
        mask = rng.random(n) < frac
        vals = np.where(mask, 0.25 + 0.75 * rng.random(n), 0.0)
        add_candidate(inits, vals, cap)

    for width in (0.04, 0.08, 0.15):
        vals = 0.2 + 0.8 * (
            np.exp(-((x / width) ** 2)) +
            np.exp(-(((1.0 - x) / width) ** 2))
        )
        add_candidate(inits, vals, cap)

    for nb in (8, 16, 32, 64):
        if nb > n:
            nb = max(2, n)
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals, n - a.size)])
        add_candidate(inits, a[:n], cap)

    for _ in range(3):
        add_candidate(inits, 0.1 + 0.9 * rng.random(n), cap)

    return inits


def choose_lengths(parent):
    raw = []
    if parent is not None:
        pl = int(parent.size)
        if 1024 <= pl <= MAX_LEN:
            raw.append(pl)
        elif pl > MAX_LEN:
            raw.append(MAX_LEN)

    for n in (8192, MAX_LEN, 4096):
        if n not in raw:
            raw.append(n)

    out = []
    for n in raw:
        n = int(n)
        if 1024 <= n <= MAX_LEN and n not in out:
            out.append(n)
    if not out:
        out = [8192]
    return out[:3]


def select_candidates(cands, max_count=8):
    if not cands:
        return []
    scores = np.array([float(c[0]) for c in cands], dtype=np.float64)
    order = np.argsort(scores)
    chosen = []
    counts = {}
    for idx in order:
        if len(chosen) >= max_count:
            break
        sc = float(cands[idx][0])
        seq = np.asarray(cands[idx][1], dtype=np.float64).ravel()
        L = int(seq.size)
        cnt = counts.get(L, 0)
        if cnt >= 3 and len(chosen) >= 3:
            continue
        chosen.append((sc, seq.copy()))
        counts[L] = cnt + 1
    return chosen


def quick_optimize(x, deadline):
    x = normalize_a(x)
    best = x.copy()
    best_sc = exact_score_a(x)
    n = int(x.size)
    eval_t = get_eval_time(n, x)
    capfun = 1500 if n <= 8192 else 800

    betas = (60.0, 200.0, 600.0, 2000.0, 6000.0)
    for beta in betas:
        if time.monotonic() >= float(deadline) - 8.0:
            break
        rem = float(deadline) - time.monotonic() - 5.0
        if rem < 12.0:
            break
        mf = maxfun_for_time(rem * 0.85, eval_t, capfun=capfun)
        if mf < 24:
            break
        sc, x2 = run_lbfgs_lse(x, beta, mf)
        if sc < best_sc:
            best_sc = float(sc)
            best = np.asarray(x2, dtype=np.float64).copy()
        x = normalize_a(x2)

        if time.monotonic() < float(deadline) - 15.0:
            ps, px = pressure_polish(x, max_steps=3, deadline=float(deadline) - 8.0)
            if ps < best_sc:
                best_sc = float(ps)
                best = np.asarray(px, dtype=np.float64).copy()
            x = normalize_a(px)

    return float(best_sc), best


def optimize_sequence(x0, deadline, label=""):
    t0 = time.monotonic()
    x = normalize_a(x0)
    if x.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    best = x.copy()
    best_sc = exact_score_a(x)
    n = int(x.size)
    eval_t = get_eval_time(n, x)

    if float(deadline) - t0 < 30.0:
        return float(best_sc), best

    total = max(15.0, float(deadline) - t0 - 8.0)
    stages = [
        (20.0, 0.04),
        (50.0, 0.08),
        (120.0, 0.16),
        (300.0, 0.22),
        (700.0, 0.22),
        (1500.0, 0.14),
        (3000.0, 0.14),
        (6000.0, 0.10),
    ]
    total_frac = sum(fr for _, fr in stages)
    cum = 0.0
    capfun = 2500 if n <= 8192 else 1200

    for beta, frac in stages:
        cum += float(frac) / float(total_frac)
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline) - 8.0)

        while time.monotonic() < sub_dead - 5.0:
            rem = float(sub_dead) - time.monotonic() - 3.0
            if rem < 8.0:
                break
            mf = maxfun_for_time(rem * 0.9, eval_t, capfun=capfun)
            if mf < 24:
                break

            sc, x2 = run_lbfgs_lse(x, beta, mf)
            if sc < best_sc:
                best_sc = float(sc)
                best = np.asarray(x2, dtype=np.float64).copy()
                print("[{}] new best {:.8f} beta={:.0f}".format(label, best_sc, beta))
            x = normalize_a(x2)

            if time.monotonic() < sub_dead - 12.0:
                ps, px = pressure_polish(x, max_steps=3, deadline=float(sub_dead) - 6.0)
                if ps < best_sc:
                    best_sc = float(ps)
                    best = np.asarray(px, dtype=np.float64).copy()
                    print("[{}] pressure best {:.8f}".format(label, best_sc))
                x = normalize_a(px)

            eval_t = get_eval_time(n, x)

    if time.monotonic() < float(deadline) - 25.0:
        rem = float(deadline) - time.monotonic() - 8.0
        if rem > 20.0:
            mf = maxfun_for_time(rem * 0.8, eval_t, capfun=1200)
            if mf >= 24:
                sc, x2 = run_lbfgs_lse(x, 10000.0, mf)
                if sc < best_sc:
                    best_sc = float(sc)
                    best = np.asarray(x2, dtype=np.float64).copy()
                    print("[{}] high beta best {:.8f}".format(label, best_sc))
                x = normalize_a(x2)

                if time.monotonic() < float(deadline) - 12.0:
                    ps, px = pressure_polish(x, max_steps=4, deadline=float(deadline) - 6.0)
                    if ps < best_sc:
                        best_sc = float(ps)
                        best = np.asarray(px, dtype=np.float64).copy()
                        print("[{}] final pressure best {:.8f}".format(label, best_sc))

    return float(best_sc), best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    actual_deadline = t0 + float(max(120.0, float(budget_s)))
    work_deadline = actual_deadline - 35.0
    rng = np.random.default_rng(seed)

    parent = get_parent_sequence()
    lengths = choose_lengths(parent)

    print("Start search: parent={}".format(parent is not None))
    print("Lengths: {}".format(lengths))

    all_cands = []
    for n in lengths:
        inits = make_inits(n, rng, parent, cap=(14 if n > 8192 else 22))
        for arr in inits:
            arr = normalize_a(arr)
            sc = exact_score_a(arr)
            if np.isfinite(sc):
                all_cands.append((float(sc), arr))

    if not all_cands:
        n0 = 8192
        u = np.ones(n0, dtype=np.float64) / float(n0)
        all_cands.append((exact_score_a(u), u))

    top_cands = select_candidates(all_cands, max_count=10)
    best_seq = np.asarray(top_cands[0][1], dtype=np.float64).copy()
    best_sc = float(top_cands[0][0])
    print("Initial best: {:.8f} length={}".format(best_sc, len(best_seq)))

    # Quick multi-start refinement.
    time_total = max(60.0, work_deadline - t0)
    long_reserve = max(700.0, 0.70 * time_total)
    quick_budget = min(360.0, max(60.0, 0.15 * time_total))
    quick_end = min(work_deadline - long_reserve, t0 + quick_budget)
    if quick_end < t0 + 30.0:
        quick_end = min(work_deadline - 60.0, t0 + 30.0)

    quick_list = []
    used_lengths = set()
    for sc, seq in top_cands:
        if len(quick_list) >= 4:
            break
        L = int(seq.size)
        if L not in used_lengths:
            quick_list.append((float(sc), np.asarray(seq, dtype=np.float64).copy()))
            used_lengths.add(L)

    for sc, seq in top_cands:
        if len(quick_list) >= 4:
            break
        if all(int(seq.size) != int(q.size) for _, q in quick_list):
            quick_list.append((float(sc), np.asarray(seq, dtype=np.float64).copy()))

    if quick_list:
        per = max(18.0, (quick_end - t0) / float(len(quick_list)))
        for i, item in enumerate(quick_list):
            if time.monotonic() >= quick_end - 10.0:
                break
            sub_dead = min(quick_end - 5.0, time.monotonic() + per)
            if sub_dead - time.monotonic() < 15.0:
                break
            sc2, seq2 = quick_optimize(item[1], sub_dead)
            if sc2 < best_sc:
                best_sc = float(sc2)
                best_seq = np.asarray(seq2, dtype=np.float64).copy()
                print("After quick{}: {:.8f} length={}".format(i + 1, best_sc, len(best_seq)))

    # Long refinement of the best sequence.
    final_reserve = max(160.0, 0.08 * time_total)
    long_end = work_deadline - final_reserve

    first_dead = min(long_end, max(time.monotonic() + 60.0,
                                   quick_end + 0.65 * max(60.0, long_end - quick_end)))
    if time.monotonic() < first_dead - 30.0:
        sc2, seq2 = optimize_sequence(best_seq, first_dead, label="long1")
        if sc2 < best_sc:
            best_sc = float(sc2)
            best_seq = np.asarray(seq2, dtype=np.float64).copy()
            print("After long1: {:.8f} length={}".format(best_sc, len(best_seq)))

    # Second trajectory: different length or perturbed global best.
    start2 = None
    for sc, seq in top_cands:
        if int(seq.size) != int(best_seq.size) and float(sc) < best_sc + 0.04:
            start2 = np.asarray(seq, dtype=np.float64).copy()
            break
    if start2 is None:
        target_len = 8192 if int(best_seq.size) != 8192 else MAX_LEN
        start2 = normalize_a(resample_sequence(best_seq, target_len))

    if time.monotonic() < long_end - 90.0:
        sub_dead = min(long_end - 25.0, time.monotonic() + 220.0)
        sc2, seq2 = optimize_sequence(start2, sub_dead, label="long2")
        if sc2 < best_sc:
            best_sc = float(sc2)
            best_seq = np.asarray(seq2, dtype=np.float64).copy()
            print("After long2: {:.8f} length={}".format(best_sc, len(best_seq)))

    # Final perturbation restarts.
    perturb_end = work_deadline - 25.0
    perturb_done = 0
    while time.monotonic() < perturb_end - 90.0 and perturb_done < 4:
        rem = float(perturb_end) - time.monotonic()
        if rem < 120.0:
            break
        chunk = min(240.0, rem / float(4 - perturb_done))
        if chunk < 80.0:
            break
        p = perturb_best(best_seq, rng)
        sc2, seq2 = quick_optimize(p, time.monotonic() + chunk)
        if sc2 < best_sc:
            best_sc = float(sc2)
            best_seq = np.asarray(seq2, dtype=np.float64).copy()
            print("Final restart {}: {:.8f} length={}".format(perturb_done + 1, best_sc, len(best_seq)))
        perturb_done += 1

    # Final high-temperature polish.
    if time.monotonic() < work_deadline - 18.0:
        rem = float(work_deadline) - time.monotonic() - 8.0
        if rem > 20.0:
            n = int(best_seq.size)
            eval_t = get_eval_time(n, best_seq)
            mf = maxfun_for_time(rem * 0.8, eval_t, capfun=800)
            if mf >= 24:
                sc2, seq2 = run_lbfgs_lse(best_seq, 12000.0, mf)
                if sc2 < best_sc:
                    best_sc = float(sc2)
                    best_seq = np.asarray(seq2, dtype=np.float64).copy()
                    print("Final high beta: {:.8f}".format(best_sc))

    # Final exact pressure cleanup.
    if time.monotonic() < work_deadline - 10.0:
        sc2, seq2 = pressure_polish(best_seq, max_steps=8, deadline=work_deadline - 5.0)
        if sc2 < best_sc:
            best_sc = float(sc2)
            best_seq = np.asarray(seq2, dtype=np.float64).copy()
            print("Final pressure: {:.8f}".format(best_sc))

    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(v) for v in out_arr]
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t0
    ))
    return out
