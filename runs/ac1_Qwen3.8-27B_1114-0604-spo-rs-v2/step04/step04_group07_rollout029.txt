"""
Optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key insights:
- Palindrome constraint is HARMFUL (forces peak to center = ||a||_2^2, giving min score 2)
- Focus on non-palindromic sequences with flat autocorrelation
- Use LSE-smoothed Adam as primary optimizer with careful beta scheduling
- Periodic "shock" moves to escape local minima
- Multiple sequence lengths
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Assumes sum(a) is fixed (we track it)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    L = c_max + np.log(sum_exp) / beta
    p = exp_scaled / sum_exp  # probability distribution over lags
    
    # Gradient of L w.r.t. a:
    # dL/da_j = (1/beta) * sum_k p_k * dc_k/da_j = (1/beta) * sum_k p_k * 2*a_{k-j}
    # = (2/beta) * (p cross-correlate a)[j]
    # (p * a_rev)[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want sum_k p[k] * a[k-j], which is (p * a_rev)[j + n - 1]
    a_rev = a[::-1]
    M_cross = n + (2*n-1) - 1  # length of p (2n-1) convolved with a_rev (n) = 3n-2
    M_cross = 3 * n - 2
    M_fft = 1 << (M_cross - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_pa = np.fft.irfft(P * AR, M_fft)[:M_cross]
    b = conv_pa[n-1:2*n-1]  # b[j] = sum_k p[k] * a[k-j]
    
    dL_da = (2.0 / beta) * b
    
    # Score = 2n * L / S^2
    # d(score)/da_j = 2n * dL_da_j / S^2 - 4n * L / S^3
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def adam_lse(a_init, beta_schedule, lr, n_steps_per_beta, time_limit_s, t_start,
             verbose_every=2000):
    """Adam optimizer with LSE smoothing and beta schedule."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    t_global = 0
    
    for beta in beta_schedule:
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            
            lr_t = lr / (1.0 + 0.001 * t)
            score, grad = lse_score_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global += 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e6)
            
            # Normalize (score is scale-invariant, but normalization helps stability)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            
            if (t_global) % verbose_every == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    [Adam] beta={beta}, step={t_global}, score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def subgrad_step(a, step_size):
    """One subgradient step on the exact objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())
    k_star = int(c.argmax())
    
    # Subgradient of max_k c_k at a:
    # For the active lag k*, gradient is 2 * a shifted (as derived)
    sg = np.zeros(n)
    # sg[j] = 2 * a[k_star - j] (with boundary: a[m] = 0 if m < 0 or m >= n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            sg[j] = 2.0 * a[idx]
    
    # Gradient of score = 2n*M/S^2:
    # = 2n * (sg / S^2 - 2*M/S^3)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    
    # Step
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new

def peak_target_perturb(a, strength, rng):
    """Perturb elements contributing to the peak, boost others."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = c[k_star]
    
    # Identify contributing range
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    
    a_new = a.copy()
    # Reduce peak contributors
    if je >= js:
        reduction = strength * (0.5 + 0.5 * rng.random())
        a_new[js:je+1] *= (1.0 - reduction)
        removed = a[js:je+1].sum() - a_new[js:je+1].sum()
        
        # Redistribute to non-contributing elements
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) > 0:
            a_new[nc_idx] += removed / len(nc_idx)
    
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new

def random_perturb(a, scale, rng):
    """Random multiplicative perturbation."""
    n = len(a)
    log_noise = scale * rng.standard_normal(n)
    a_new = a * np.exp(log_noise)
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new

def coordinate_descent(a, n_sweeps, time_limit_s, t_start, step_fracs=None):
    """Coordinate descent trying to reduce the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    
    if step_fracs is None:
        step_fracs = [0.7, 0.5, 0.3, 0.1]
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = 2.0 * n * c[k_star] / (a.sum() ** 2)
        
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        # Try reducing each peak contributor
        for j in range(js, je + 1):
            if time.monotonic() - t_start > time_limit_s:
                break
            a_old = a[j]
            if a_old < 1e-14:
                continue
            for frac in step_fracs:
                a[j] = a_old * frac
                # Redistribute
                removed = a_old - a[j]
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je+1] = False
                nc_idx = np.where(non_contrib)[0]
                if len(nc_idx) > 0:
                    a[nc_idx] += removed / len(nc_idx)
                c_new = fast_conv(a)
                sc_new = 2.0 * n * c_new.max() / (a.sum() ** 2)
                if sc_new < score:
                    score = sc_new
                    if sc_new < best_score:
                        best_score = sc_new
                    break
                else:
                    a[j] = a_old  # revert
                    a[nc_idx] -= removed / len(nc_idx) if len(nc_idx) > 0 else 0
            
            a = np.clip(a, 0.0, 1e6)
            s = a.sum()
            if s > 1e-12:
                a = a / s
    
    return best_score, a

def run_lsq_polish(a0, beta, maxiter=3000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1e6).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1e6)] * n
    
    def obj(x):
        x_c = np.clip(x, 0.0, 1e6)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': 20000, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1e6)
        s = a_out.sum()
        if s > 1e-12:
            a_out = a_out / s
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception as e:
        print(f"    LBFGS error: {e}")
        return compute_score(a0), a0

def generate_init(n, rng, mode='gaussian'):
    """Generate initial sequence of length n."""
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'gaussian':
        x = np.arange(n) / n
        sigma = 0.1 + 0.2 * rng.random()
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif mode == 'raised_cosine':
        x = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 'random_smooth':
        a = np.abs(rng.standard_normal(n)) + 0.3
        k = max(3, n // 20)
        a = np.convolve(a, np.ones(k)/k, mode='same')
    elif mode == 'two_bump':
        x = np.arange(n) / n
        c1 = 0.2 + 0.2 * rng.random()
        c2 = 0.6 + 0.2 * rng.random()
        s1 = 0.05 + 0.1 * rng.random()
        s2 = 0.05 + 0.1 * rng.random()
        a = np.exp(-0.5*((x-c1)/s1)**2) + np.exp(-0.5*((x-c2)/s2)**2)
    elif mode == 'fourier':
        x = np.arange(n) / n
        a = np.ones(n)
        for _ in range(3):
            f = 1.0 + 5.0 * rng.random()
            p = 2 * np.pi * rng.random()
            amp = 0.1 + 0.3 * rng.random()
            a += amp * np.cos(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.0, 1e6)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1e6)

```python
"""
Optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key insight: Palindrome constraint is PROBABLY suboptimal (for palindromic a, peak = ||a||_2^2, min score = 2).
Focus on non-palindromic sequences with flat autocorrelation using LSE-smoothed Adam.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    L = c_max + np.log(sum_exp) / beta
    p = exp_scaled / sum_exp

    a_rev = a[::-1]
    M_cross = 3 * n - 2
    M_fft = 1 << (M_cross - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_pa = np.fft.irfft(P * AR, M_fft)[:M_cross]
    b = conv_pa[n - 1:2 * n - 1]

    dL_da = (2.0 / beta) * b
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a_init, beta_schedule, lr, n_steps_per_beta, time_limit_s, t_start,
             verbose_every=5000):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    t_global = 0

    for beta in beta_schedule:
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a

            lr_t = lr / (1.0 + 0.0005 * t)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global += 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)

            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e6)
            s = a.sum()
            if s > 1e-12:
                a = a / s

            if t_global % verbose_every == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    [Adam] beta={beta}, step={t_global}, score={sc:.8f}")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def subgrad_step(a, step_size):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())
    k_star = int(c.argmax())

    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)

    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new


def peak_target_perturb(a, strength, rng):
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)

    a_new = a.copy()
    if je >= js:
        reduction = strength * (0.3 + 0.5 * rng.random())
        a_new[js:je + 1] *= (1.0 - reduction)
        removed = a[js:je + 1].sum() - a_new[js:je + 1].sum()
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) > 0:
            a_new[nc_idx] += removed / len(nc_idx)

    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new


def random_perturb(a, scale, rng):
    n = len(a)
    log_noise = scale * rng.standard_normal(n)
    a_new = a * np.exp(log_noise)
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new


def run_lsq_polish(a0, beta, maxiter=3000):
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1e6).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1e6)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e6)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': 30000, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1e6)
        s = a_out.sum()
        if s > 1e-12:
            a_out = a_out / s
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception as e:
        print(f"    LBFGS error: {e}")
        return compute_score(a0), a0


def generate_init(n, rng, mode='gaussian'):
    x = np.arange(n, dtype=np.float64) / n
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'gaussian':
        sigma = 0.1 + 0.2 * rng.random()
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif mode == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif mode == 'random_smooth':
        a = np.abs(rng.standard_normal(n)) + 0.3
        k = max(3, n // 20)
        a = np.convolve(a, np.ones(k) / k, mode='same')
    elif mode == 'two_bump':
        c1 = 0.2 + 0.2 * rng.random()
        c2 = 0.6 + 0.2 * rng.random()
        s1 = 0.05 + 0.1 * rng.random()
        s2 = 0.05 + 0.1 * rng.random()
        a = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    elif mode == 'fourier':
        a = np.ones(n)
        for _ in range(4):
            f = 1.0 + 6.0 * rng.random()
            p = 2.0 * np.pi * rng.random()
            amp = 0.05 + 0.25 * rng.random()
            a += amp * np.cos(2.0 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 'bimodal_flat':
        a = np.ones(n)
        mask1 = (x > 0.1) & (x < 0.4)
        mask2 = (x > 0.6) & (x < 0.9)
        a[mask1] *= 0.3
        a[mask2] *= 0.3
    else:
        a = np.ones(n)
    a = np.clip(a, 0.0, 1e6)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1e6)


def coordinate_descent(a, n_sweeps, time_limit_s, t_start, step_fracs=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    if step_fracs is None:
        step_fracs = [0.6, 0.4, 0.2]

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit_s:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = 2.0 * n * c[k_star] / (a.sum() ** 2)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        step_idx = sweep % 3
        j = js + (sweep * 7 + step_idx) % max(1, je - js + 1)
        j = int(min(max(j, js), je))

        a_old = a[j]
        if a_old < 1e-14:
            continue
        improved = False
        for frac in step_fracs:
            a[j] = a_old * frac
            removed = a_old - a[j]
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je + 1] = False
            nc_idx = np.where(non_contrib)[0]
            if len(nc_idx) == 0:
                break
            a[nc_idx] += removed / len(nc_idx)
            c_new = fast_conv(a)
            sc_new = 2.0 * n * c_new.max() / (a.sum() ** 2)
            if sc_new < score:
                score = sc_new
                improved = True
                if sc_new < best_score:
                    best_score = sc_new
                break
            else:
                a[j] = a_old
                a[nc_idx] -= removed / len(nc_idx)
        if not improved:
            a[j] = a_old
        a = np.clip(a, 0.0, 1e6)
        s = a.sum()
        if s > 1e-12:
            a = a / s
    return best_score, a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer focusing on NON-palindromic sequences.
    Uses LSE-smoothed Adam with geometric beta schedule, periodic shocks,
    L-BFGS polish, subgradient steps, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 30.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and a is not None:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                s = parent.sum()
                if s > 1e-6:
                    parent = parent / s
                    print(f"Parent loaded: n={len(parent)}, score={compute_score(parent):.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # =====================
    # Phase 1: Exploration (15% time)
    # =====================
    p1_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Exploration (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    n_options = [8192]
    if parent is not None:
        np_ = len(parent)
        n_options.insert(0, np_)
        if np_ // 2 >= 1024:
            n_options.append(np_ // 2)
        if np_ * 2 <= 32768:
            n_options.append(min(np_ * 2, 16384))
    n_options.append(4096)
    n_options.append(16384)
    # Deduplicate and filter
    n_options = sorted(set(n for n in n_options if 1024 <= n <= 32768))

    candidates = []
    modes = ['gaussian', 'two_bump', 'fourier', 'random_smooth', 'raised_cosine', 'bimodal_flat']

    for n in n_options:
        if time.monotonic() - t_start > p1_time - 10:
            break
        for mode in modes:
            if time.monotonic() - t_start > p1_time - 5:
                break
            a_init = generate_init(n, rng, mode)
            if parent is not None:
                a_parent_res = resample(parent, n)
                blend = 0.5 + 0.3 * rng.random()
                a_init = blend * a_parent_res + (1 - blend) * a_init
                a_init = np.clip(a_init, 0, 1e6)
                a_init = a_init / max(a_init.sum(), 1e-12)
            sc, a_opt = adam_lse(a_init, [30, 80], lr=0.04, n_steps_per_beta=3000,
                                  time_limit_s=2.5, t_start=time.monotonic() - 0.1,
                                  verbose_every=999999)
            candidates.append((sc, a_opt))
            print(f"  n={n}, {mode}: {sc:.8f}")

    if parent is not None:
        candidates.append((compute_score(parent), parent.copy()))
        print(f"  Parent: {candidates[-1][0]:.8f}")

    candidates.sort(key=lambda x: x[0])
    for sc, a_opt in candidates[:5]:
        try_update(sc, a_opt, "exploration")

    print(f"  Best after exploration: {best_score:.8f}")

    # =====================
    # Phase 2: Main LSE-Adam (50% time)
    # =====================
    p2_time = effective_budget * 0.50
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Main LSE-Adam (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    a_work = best_a.astype(np.float64).copy()
    n_work = len(a_work)

    # Geometric beta schedule
    beta_schedule = [30, 80, 200, 500, 1200, 3000, 8000, 20000, 50000]
    lr_schedule = [0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0003]
    time_weights = [1, 1, 2, 2, 3, 3, 4, 5, 6]
    total_w = sum(time_weights)
    beta_times = [p2_time * w / total_w for w in time_weights]

    p2_deadline = t_start + p2_time
    shock_count = 0
    last_shock_wall = time.monotonic()

    for i, (beta, lr, t_alloc) in enumerate(zip(beta_schedule, lr_schedule, beta_times)):
        if time.monotonic() > p2_deadline - 5:
            break
        remaining = p2_deadline - time.monotonic()
        t_actual = min(t_alloc * 1.3, remaining)
        if t_actual < 2:
            break

        print(f"  Beta={beta}, lr={lr}, time={t_actual:.0f}s")
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_work, [beta], lr=lr, n_steps_per_beta=10**9,
                              time_limit_s=t_actual, t_start=t_start,
                              verbose_every=10000)
        try_update(sc, a_opt, f"Adam b={beta}")
        a_work = a_opt
        print(f"  Beta={beta} done: {sc:.8f} ({time.monotonic() - t0:.1f}s)")

        # Periodic shock
        wall_elapsed = time.monotonic() - last_shock_wall
        if wall_elapsed > 50 and time.monotonic() < p2_deadline - 40:
            shock_count += 1
            last_shock_wall = time.monotonic()
            print(f"  SHOCK #{shock_count}: trying escape moves...")

            best_shock = a_work
            best_shock_sc = compute_score(a_work)

            for trial in range(6):
                if time.monotonic() > p2_deadline - 30:
                    break
                if trial == 0:
                    a_sh = peak_target_perturb(a_work, 0.3 + 0.3 * rng.random(), rng)
                elif trial == 1:
                    a_sh = random_perturb(a_work, 0.15, rng)
                elif trial == 2:
                    a_sh = random_perturb(a_work, 0.3, rng)
                elif trial == 3:
                    a_rand = generate_init(n_work, rng, modes[trial % len(modes)])
                    a_sh = 0.5 * a_work + 0.5 * a_rand
                    a_sh = a_sh / max(a_sh.sum(), 1e-12)
                elif trial == 4:
                    # High-beta gradient step (greedy)
                    _, grad_hb = lse_score_grad(a_work, 10000)
                    step = 0.1
                    a_sh = a_work - step * grad_hb
                    a_sh = np.clip(a_sh, 0, 1e6)
                    a_sh = a_sh / max(a_sh.sum(), 1e-12)
                else:
                    # Blend with uniform
                    a_sh = 0.8 * a_work + 0.2 * np.ones(n_work) / n_work
                    a_sh = a_sh / max(a_sh.sum(), 1e-12)

                sc_s, a_s_ref = adam_lse(a_sh, [beta * 2], lr=lr * 1.5,
                                          n_steps_per_beta=3000,
                                          time_limit_s=3.0, t_start=time.monotonic() - 0.1,
                                          verbose_every=999999)
                if sc_s < best_shock_sc:
                    best_shock_sc = sc_s
                    best_shock = a_s_ref

            try_update(best_shock_sc, best_shock, f"shock#{shock_count}")
            if best_shock_sc < compute_score(a_work):
                a_work = best_shock
                print(f"    Shock improved to {best_shock_sc:.8f}")

    # =====================
    # Phase 3: Local refinement (20% time)
    # =====================
    p3_time = effective_budget * 0.20
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Local Refinement (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    p3_deadline = t_start + p3_time

    # L-BFGS polish
    for beta in [1000, 5000, 20000]:
        if time.monotonic() > p3_deadline - 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lsq_polish(best_a, beta, maxiter=5000)
        try_update(sc, a_opt, f"LBFGS b={beta}")
        print(f"  LBFGS b={beta}: {sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # Coordinate descent
    if time.monotonic() < p3_deadline - 5:
        p3_cd = min(60.0, p3_deadline - time.monotonic() - 3)
        t0 = time.monotonic()
        sc, a_opt = coordinate_descent(best_a, n_sweeps=500,
                                        time_limit_s=p3_cd, t_start=t_start)
        try_update(sc, a_opt, "coord_descent")
        print(f"  Coord descent: {sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # Subgradient
    if time.monotonic() < p3_deadline - 5:
        sg_deadline = min(time.monotonic() + 30.0, p3_deadline - 2)
        a_sg = best_a.copy()
        sg_score = compute_score(a_sg)
        step = 0.01
        iters = 0
        while time.monotonic() < sg_deadline:
            a_sg = subgrad_step(a_sg, step)
            sc = compute_score(a_sg)
            if sc < sg_score:
                sg_score = sc
                step = min(step * 1.05, 0.05)
            else:
                step *= 0.9
            iters += 1
        try_update(sg_score, a_sg, f"subgrad({iters})")
        print(f"  Subgrad: {sg_score:.8f} ({iters} iters)")

    # =====================
    # Phase 4: Final polish (15% time)
    # =====================
    p4_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Final Polish (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    p4_deadline = t_start + effective_budget - 10.0

    # High-beta Adam
    if time.monotonic() <
