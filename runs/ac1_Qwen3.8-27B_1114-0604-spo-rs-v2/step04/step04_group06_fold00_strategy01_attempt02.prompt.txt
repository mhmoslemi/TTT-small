    # Verify that the input is a list

    if not sequence:

    # Check each element in the list for validity

        if np.isnan(x) or np.isinf(x):

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

        idx = k_star - np.arange(js, je + 1)

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, lr=0.005, n_steps=50000, beta_lse=100,
             time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient in a-space."""

    a = a_init.astype(np.float64).copy()

def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,

    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def selective_coord_descent(a, top_k=30, n_sweeps=10, time_limit_s=None, t_start=None):
    """Coordinate descent optimizing top-K coordinates per sweep."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score, grad = subgrad_score_a(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if grad[j] > 1e-10:
                candidates = a_old * np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0])
            elif grad[j] < -1e-10:
                candidates = np.concatenate([[0.0], a_old * np.array([1.5, 2.0, 3.0])])
            else:
                candidates = np.array([a_old])
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a
