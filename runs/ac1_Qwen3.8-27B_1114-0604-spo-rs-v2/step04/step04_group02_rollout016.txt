Using FFT: cross_correlation can be computed as:

Or equivalently: cross_correlation(a, p) = conv(a, p_flipped) shifted appropriately.

But we only need i = 0, ..., n-1, so m = 2n-2, 2n-3, ..., n-1.

So the gradient (up to the factor of 2n) is:
g[i] = 2 * conv(a, p_flipped)[2n-2-i] for i = 0, ..., n-1

Let me code this up properly:

```python
def lse_grad(a, beta, top_k=None):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        threshold = np.partition(c, len(c) - top_k)[len(c) - top_k]
        c_eff = np.where(c >= threshold, c, threshold)
    else:
        c_eff = c
    
    c_max = float(c_eff.max())
    scaled = beta * (c_eff - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -200))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Compute gradient: g[i] = 2 * sum_k p[k] * a[k-i] (approx)
    # = 2 * cross_correlation(a, p, lag=i)
    # Using: cross_correlation(a, p, lag=i) = conv(a, p[::-1])[2n-2-i]
    p_flipped = p[::-1]  # length 2n-1
    # conv of a (length n) and p_flipped (length 2n-1) has length n + 2n-2 = 3n-2
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    FA = np.fft.rfft(a, M_fft)
    FP = np.fft.rfft(p_flipped, M_fft)
    conv_result = np.fft.irfft(FA * FP, M_fft)[:M]
    
    # g[i] = 2 * conv_result[2n-2-i] for i = 0, ..., n-1
    # So g = 2 * conv_result[n-1 : 2n-1][::-1]
    g = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
```
