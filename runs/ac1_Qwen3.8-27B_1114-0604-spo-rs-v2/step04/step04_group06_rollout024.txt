```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses multi-strategy optimization: LSE-smoothed Adam/L-BFGS, Nesterov subgradient,
frequency-domain alternating projections, and random restarts.
Starts from parent construction and polishes with high-precision methods.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    N2 = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, N2)
    P = np.fft.rfft(p, N2)
    conv_res = np.fft.irfft(RA * P, N2)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_exact(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, lr, n_steps, beta, time_limit, t_start):
    """Adam with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s > 1e-12:
        a = a / s
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def nesterov_subgrad(a_init, n_steps, c_step, momentum, time_limit, t_start):
    """Nesterov-accelerated subgradient method for exact objective."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s > 1e-12:
        a = a / s
    y = a.copy()
    prev_a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    best_beta = 0.0

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        beta_t = (1.0 + best_beta) / (1.0 + best_beta + t)
        y = a + (beta_t - 1.0) * (a - prev_a)
        y = np.clip(y, 0.0, None)
        sy = y.sum()
        if sy > 1e-12:
            y = y / sy
        score, grad = subgrad_exact(y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = c_step * score / (grad_norm * grad_norm + 1e-10)
        step = min(step, 1.0 / (t + 1))
        a_new = y - step * grad
        a_new = np.clip(a_new, 0.0, None)
        sn = a_new.sum()
        if sn > 1e-12:
            a_new = a_new / sn
        prev_a = a
        a = a_new
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def heavy_ball_subgrad(a_init, n_steps, lr0, decay, momentum, time_limit, t_start):
    """Heavy ball subgradient method."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s > 1e-12:
        a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        lr_t = lr0 / (1.0 + decay * t)
        score, grad = subgrad_exact(a)
        v = momentum * v - lr_t * grad
        a = a + v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_lse(a0, beta, maxiter, maxfun, time_limit, t_start):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def freq_domain_polish(a, n_iters=50, time_limit=60, t_start=None):
    """Frequency-domain alternating projection for flat spectrum."""
    if t_start is None:
        t_start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s > 1e-12:
        a = a / s
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        # Step 1: Compute spectrum
        A = np.fft.rfft(a, N)
        b = A.real ** 2 + A.imag ** 2  # |A|^2, shape (N//2+1,)
        c_full = np.fft.irfft(b, N)[:M]  # = conv(a, a)

        # Step 2: Find the target - make c as flat as possible
        c_mean = c_full.mean()
        c_max = c_full.max()
        if c_max < c_mean * 1.001:
            break  # Already very flat

        # Compute correction in frequency domain
        # We want to reduce the peak at k* and redistribute
        k_star = int(c_full.argmax())
        # The contribution of frequency bin j to c[k_star] is:
        # (1/N) * b[j] * cos(2*pi*k_star*j/N)  (for the real part)
        # Reduce b[j] where cos(2*pi*k_star*j/N) > 0
        # Increase b[j] where cos(2*pi*k_star*j/N) < 0

        j_vals = np.arange(N // 2 + 1)
        cos_vals = np.cos(2.0 * np.pi * k_star * j_vals / N)
        # Apply multiplicative correction
        alpha = 0.1  # correction strength
        correction = np.exp(-alpha * cos_vals * (c_max - c_mean) / c_mean)
        b_new = b * correction
        # Enforce symmetry (already satisfied since b is from |A|^2)
        # Enforce non-negativity
        b_new = np.maximum(b_new, 0.0)
        # Normalize to maintain DC
        b_new = b_new * (b[0] / b_new[0]) if b_new[0] > 0 else b_new

        # Step 3: New A with same phase, new magnitude
        magnitude = np.sqrt(b_new)
        phase = np.angle(A)
        A_new = magnitude * np.exp(1j * phase)
        # Ensure Hermitian symmetry in the full spectrum
        a_new = np.fft.irfft(A_new, N)[:n]

        # Step 4: Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new = a_new / sn

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new + 0.1 * (a_new - a)  # blend for stability
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa > 1e-12:
            a = a / sa

    return best_score, best_a


def random_restart_optimize(a_best, n_restarts, time_per_restart, rng, t_start, deadline):
    """Try random perturbations of the best solution."""
    n = len(a_best)
    best_score = compute_score(a_best)
    best_a = a_best.copy()

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Perturb
        noise_scale = 0.01 * (1.0 + r * 0.5)
        a_perturbed = best_a + noise_scale * rng.standard_normal(n)
        a_perturbed = np.maximum(a_perturbed, 0.0)
        sp = a_perturbed.sum()
        if sp < 1e-12:
            continue
        a_perturbed = a_perturbed / sp

        # Quick LSE optimization
        sc, a_opt = adam_lse(a_perturbed, lr=0.001, n_steps=20000,
                             beta=2000, time_limit=time_per_restart, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"    Restart {r}: {best_score:.10f}")

    return best_score, best_a


def try_trim_and_extend(a, rng, t_start, deadline):
    """Try trimming zeros from ends and extending with small values."""
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()

    # Trim trailing/leading zeros
    a_trimmed = a.copy()
    # Find first and last non-negligible element
    threshold = 1e-10 * a.max()
    mask = a_trimmed > threshold
    if mask.any():
        first = np.argmax(mask)
        last = n - 1 - np.argmax(mask[::-1])
        a_t = a_trimmed[first:last + 1]
        if len(a_t) >= 100:
            sc = compute_score(a_t)
            if sc < best_score:
                best_score = sc
                best_a = a_t
                print(f"    Trimmed to {len(a_t)}: {best_score:.10f}")

    return best_score, best_a


def block_coord_descent(a, n_blocks=20, n_sweeps=5, time_limit=120, t_start=None):
    """Block coordinate descent - optimize blocks one at a time."""
    if t_start is None:
        t_start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    block_size = n // n_blocks + 1
    n_actual_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit:
            break
        for b in range(n_actual_blocks):
            if time.monotonic() - t_start > time_limit:
                break
            start = b * block_size
            end = min(start + block_size, n)
            if end - start < 2:
                continue
            # Optimize this block using L-BFGS with high beta
            a0 = a[start:end].copy()
            if a0.sum() < 1e-15:
                a0[:] = a.sum() / n / (end - start)
                if a0.sum() < 1e-15:
                    a0[:] = 1.0 / (end - start)

            def block_obj(x, _a=a, _start=start, _end=end, _beta=5000):
                a_full = _a.copy()
                a_full[_start:_end] = np.maximum(x, 0.0)
                sc, g_full = lse_score_grad(a_full, _beta)
                g = g_full[_start:_end]
                return sc, g

            try:
                res = minimize(block_obj, a0, method='L-BFGS-B', jac=True,
                               bounds=[(0.0, None)] * (end - start),
                               options={'maxiter': 200, 'maxfun': 5000,
                                        'ftol': 1e-15, 'gtol': 1e-12})
                a[start:end] = np.clip(res.x, 0.0, None)
            except Exception:
                pass

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-strategy optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LSE-smoothed gradient methods, subgradient methods,
    frequency-domain projections, block coordinate descent, and random restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0  # safety margin

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            raise ValueError("Parent too small")
    except Exception:
        a = np.ones(4096) / 4096
        print("No valid parent, using uniform initialization")

    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()
    print(f"Initial: n={n}, score={best_score:.10f}")

    def try_update(sc, a_new, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = np.clip(a_new, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {best_score:.10f} | {label} | t={elapsed:.0f}s")
            return True
        return False

    # === Phase 1: LSE Adam with increasing beta ===
    print(f"\n{'='*50}")
    print("Phase 1: LSE Adam optimization")
    print(f"{'='*50}")

    for beta, lr in [(200, 0.003), (500, 0.002), (1000, 0.001), (2000, 0.0005), (5000, 0.0002)]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 180.0)
        sc, a_opt = adam_lse(best_a, lr=lr, n_steps=500000, beta=beta,
                             time_limit=alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam beta={beta},lr={lr}")

    # === Phase 2: L-BFGS with high beta ===
    print(f"\n{'='*50}")
    print("Phase 2: L-BFGS polish")
    print(f"{'='*50}")

    for beta in [1000, 3000, 8000, 20000]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        t0 = time.monotonic()
        try:
            sc, a_opt = lbfgs_lse(best_a, beta, maxiter=5000, maxfun=50000,
                                   time_limit=min(remaining * 0.1, 120), t_start=t_start)
            try_update(sc, a_opt, f"L-BFGS beta={beta}")
            print(f"    L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
        except Exception as e:
            print(f"    L-BFGS beta={beta} failed: {e}")

    # === Phase 3: Heavy ball subgradient ===
    print(f"\n{'='*50}")
    print("Phase 3: Heavy ball subgradient")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        for lr0, decay, mom in [(0.01, 0.001, 0.9), (0.005, 0.0005, 0.95), (0.02, 0.002, 0.8)]:
            alloc = min(remaining * 0.1, 120.0)
            sc, a_opt = heavy_ball_subgrad(best_a, n_steps=200000, lr0=lr0,
                                           decay=decay, momentum=mom,
                                           time_limit=alloc, t_start=t_start)
            try_update(sc, a_opt, f"HeavyBall lr={lr0},m={mom}")

    # === Phase 4: Frequency domain polish ===
    print(f"\n{'='*50}")
    print("Phase 4: Frequency domain polish")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 120.0)
        sc, a_opt = freq_domain_polish(best_a, n_iters=100,
                                       time_limit=alloc, t_start=t_start)
        try_update(sc, a_opt, "Freq domain")

    # === Phase 5: Block coordinate descent ===
    print(f"\n{'='*50}")
    print("Phase 5: Block coordinate descent")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 180.0)
        sc, a_opt = block_coord_descent(best_a, n_blocks=30, n_sweeps=3,
                                        time_limit=alloc, t_start=t_start)
        try_update(sc, a_opt, "Block CD")

    # === Phase 6: Nesterov subgradient ===
    print(f"\n{'='*50}")
    print("Phase 6: Nesterov subgradient")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        for c_step in [0.01, 0.005, 0.02]:
            alloc = min(remaining * 0.1, 120.0)
            sc, a_opt = nesterov_subgrad(best_a, n_steps=100000, c_step=c_step,
                                         momentum=0.9, time_limit=alloc, t_start=t_start)
            try_update(sc, a_opt, f"Nesterov c={c_step}")

    # === Phase 7: More LSE with very high beta (final polish) ===
    print(f"\n{'='*50}")
    print("Phase 7: Final high-beta LSE polish")
    print(f"{'='*50}")

    for beta, lr in [(10000, 0.0001), (20000, 0.00005), (50000, 0.00002)]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.1, 120.0)
        sc, a_opt = adam_lse(best_a, lr=lr, n_steps=500000, beta=beta,
                             time_limit=alloc, t_start=t_start)
        try_update(sc, a_opt, f"Final Adam beta={beta}")

    # === Phase 8: Random restarts ===
    print(f"\n{'='*50}")
    print("Phase 8: Random restarts")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_restarts = min(20, int(remaining / 20))
        time_per = max(5.0, remaining / (n_restarts * 2))
        sc, a_opt = random_restart_optimize(best_a, n_restarts=n_restarts,
                                            time_per_restart=time_per,
                                            rng=rng, t_start=t_start,
                                            deadline=deadline)
        try_update(sc, a_opt, "Random restarts")

    # === Phase 9: Final L-BFGS on best ===
    print(f"\n{'='*50}")
    print("Phase 9: Final L-BFGS")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        for beta in [5000, 10000, 50000]:
            if time.monotonic() > deadline - 10:
                break
            try:
                sc, a_opt = lbfgs_lse(best_a, beta, maxiter=10000, maxfun=100000,
                                       time_limit=min(remaining * 0.1, 90), t_start=t_start)
                try_update(sc, a_opt, f"Final L-BFGS beta={beta}")
            except Exception:
                pass

    # === Phase 10: Try to trim or extend ===
    print(f"\n{'='*50}")
    print("Phase 10: Trim/Extend")
    print(f"{'='*50}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = try_trim_and_extend(best_a, rng, t_start, deadline)
        try_update(sc, a_opt, "Trim/Extend")

    # Final verification and output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
