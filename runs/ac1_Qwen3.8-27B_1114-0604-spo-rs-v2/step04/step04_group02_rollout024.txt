def fast_conv(a):

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

"""
LP-guided optimizer for step function peak minimization.
Innovation: Uses linear programming to find optimal mixing directions that
simultaneously respect all active convolution constraints, combined with
adaptive L-BFGS refinement and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Fast self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=None):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_work = c.copy()
    if top_k is not None and top_k < len(c_work):
        kth_val = np.partition(c_work, len(c_work) - top_k)[len(c_work) - top_k]
        c_work = np.minimum(c_work, kth_val)

    c_max = float(c_work.max())
    scaled = beta * (c_work - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def make_lse_func(beta, top_k):
    """Create a function for L-BFGS (no closures allowed, so we use a class)."""
    class LSEFunc:
        def __call__(self, a):
            return lse_score_grad(a, beta, top_k)
    return LSEFunc()


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000, top_k=None, time_limit=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    bounds = [(0.0, 1000.0)] * n
    t_start = time.monotonic()

    def objective(x):
        return lse_score_grad(x, beta, top_k)

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_lp_matrix(constraint_indices, a):
    """Build sparse constraint matrix for LP. A[i,j] = a[k_i - j] if valid."""
    n = len(a)
    T = len(constraint_indices)
    A = lil_matrix((T, n))
    for i, k in enumerate(constraint_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            vals = a[k - j_end:k - j_start + 1]
            A[i, j_start:j_start + len(vals)] = vals
    return A.tocsr()


def solve_lp(a, M, active_constraints, max_cutting_iters=5):
    """
    Solve LP: max sum(b) s.t. (a*b)(k) <= M for k in active_constraints, b >= 0.
    Uses cutting-plane if new constraints are violated.
    Returns b or None if infeasible.
    """
    n = len(a)
    active = list(active_constraints)

    for _ in range(max_cutting_iters):
        if len(active) == 0:
            # Return uniform b
            return np.ones(n) * (M / max(float(a.max()), 1e-10))

        A = build_lp_matrix(active, a)
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            return None

        if not res.success:
            return None

        b = np.maximum(res.x, 0.0)

        # Check for violated constraints
        if len(active) < 2 * n - 1:
            conv_ab = cross_conv(a, b)
            all_idx = np.arange(len(conv_ab))
            active_set = set(active)
            non_active = np.array([i for i in all_idx if i not in active_set])
            if len(non_active) > 0:
                violations = conv_ab[non_active]
                violated_mask = violations > M * (1 + 1e-7)
                if np.any(violated_mask):
                    violated_idx = non_active[violated_mask]
                    # Sort by violation magnitude, add worst ones
                    viol_vals = conv_ab[violated_idx]
                    order = np.argsort(viol_vals)[::-1]
                    new_constrs = list(violated_idx[order[:10]])
                    active = list(set(active) | set(new_constrs))
                    # Keep only top constraints by value
                    all_vals = conv_ab[active]
                    sort_order = np.argsort(all_vals)[::-1]
                    active = [active[i] for i in sort_order[:80]]
                    continue

        return b

    return b


def lp_guided_descent(a, n_iters=200, alpha_init=0.3, n_top=40, f_lb=1.0):
    """
    LP-guided descent: iteratively find b via LP, mix with a.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init
    stall_count = 0
    prev_sc = best_sc

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (float(a.sum()) ** 2)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
            alpha = min(0.5, alpha * 1.05)
        else:
            stall_count += 1
            if stall_count > 20:
                alpha *= 0.7
                if alpha < 0.01:
                    break
            elif stall_count > 40:
                break

        # Get top constraint indices
        top_idx = np.argsort(c)[::-1][:n_top]

        # Solve LP
        b = solve_lp(a, M, list(top_idx), max_cutting_iters=3)
        if b is None:
            break

        Sb = float(b.sum())
        if Sb < 1e-12:
            break

        # Check b*b peak to ensure safety
        c_bb = fast_conv(b)
        M_bb = float(c_bb.max())
        Sa = float(a.sum())
        Sa2 = Sa * Sa

        # Adaptive alpha: reduce if b*b peak is large
        if M_bb > 2.0 * M:
            alpha_safe = min(alpha, 0.1)
        elif M_bb > 1.5 * M:
            alpha_safe = min(alpha, 0.15)
        else:
            alpha_safe = alpha

        # Mix
        b_rescaled = b * (Sa / Sb)
        a_new = (1.0 - alpha_safe) * a + alpha_safe * b_rescaled
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        if it % 50 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_sc:.10f}, alpha={alpha_safe:.3f}")

    return best_a


def nesterov_subgrad(a, n_iters=3000, f_lb=1.4):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            a_p = a_p / S
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def coordinate_descent_peak(a, n_sweeps=3):
    """Coordinate descent focused on reducing the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        n_top = max(10, n // 20)
        top_local = np.argsort(contrib)[::-1][:n_top]
        bottom_local = np.argsort(contrib)[:n_top]

        for _ in range(5):
            for li in top_local:
                j = j_start + li
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                a[j] = old_val * 0.5
                reduction = old_val * 0.5
                n_bottom = len(bottom_local)
                if n_bottom > 0:
                    per = reduction / n_bottom
                    for bi in bottom_local:
                        a[j_start + bi] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = old_val
                    for bi in bottom_local:
                        a[j_start + bi] -= per
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy optimizer using LP-guided descent, L-BFGS, and
    Nesterov subgradient. Returns the best sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.sum() > 0.01:
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
        n_work = len(parent)
    else:
        n_work = 6962
        print(f"No parent found, using n={n_work}")

    print(f"Working length: {n_work}, budget: {total_budget}s")

    # Start with best available
    a_cur = best_a.copy()
    if len(a_cur) != n_work:
        a_cur = np.interp(np.linspace(0, 1, n_work),
                          np.linspace(0, 1, len(a_cur)), a_cur)
        a_cur = np.clip(a_cur, 0.0, 1000.0)
        S = a_cur.sum()
        if S > 0:
            a_cur = a_cur / S

    # ============================================================
    # Phase 1: Quick L-BFGS refinement (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 25.0)
    print(f"\n=== Phase 1: Quick L-BFGS (deadline t={phase1_end-t_start:.0f}s) ===")

    for beta in [500, 2000]:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=20000, maxfun=80000)
        report(sc, a_opt, f"P1 LBFGS beta={beta}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: LP-guided descent (35% time)
    # ============================================================
    phase2_end = t_start + 0.45 * (total_budget - 25.0)
    print(f"\n=== Phase 2: LP-Guided Descent (deadline t={phase2_end-t_start:.0f}s) ===")

    n_lp_iters = 150
    a_lp = lp_guided_descent(best_a, n_iters=n_lp_iters, alpha_init=0.25, n_top=30)
    sc = compute_score(a_lp)
    report(sc, a_lp, "LP descent")

    # Second LP pass with different parameters
    if time.monotonic() < phase2_end - 30:
        a_lp2 = lp_guided_descent(best_a, n_iters=100, alpha_init=0.15, n_top=50)
        sc = compute_score(a_lp2)
        report(sc, a_lp2, "LP descent v2")

    # Coordinate descent on the peak
    if time.monotonic() < phase2_end - 20:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=3)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord descent")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: High-precision L-BFGS (30% time)
    # ============================================================
    phase3_end = t_start + 0.75 * (total_budget - 25.0)
    print(f"\n=== Phase 3: High-Precision L-BFGS (deadline t={phase3_end-t_start:.0f}s) ===")

    beta_schedule = [5000, 15000, 50000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(50000, max(5000, remaining * 40)))
        maxfun = int(min(200000, max(20000, remaining * 150)))

        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"P3 LBFGS beta={beta}")

        # Top-k variant
        if time.monotonic() < phase3_end - 30:
            for top_k in [3, 5]:
                sc_k, a_k, _ = run_lbfgs(best_a, beta, maxiter=maxiter // 2,
                                          maxfun=maxfun // 2, top_k=top_k)
                report(sc_k, a_k, f"P3 LBFGS beta={beta} tk={top_k}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Iterated local search (15% time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 25.0)
    print(f"\n=== Phase 4: Iterated Local Search (deadline t={phase4_end-t_start:.0f}s) ===")

    for trial in range(15):
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 20:
            break

        scale = 0.01 + 0.02 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        beta = 3000 + int(20000 * rng.random())
        maxiter = int(min(30000, max(3000, remaining * 15)))
        maxfun = int(min(100000, max(10000, remaining * 60)))

        sc, a_opt, _ = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS #{trial}")

        # Small perturbation follow-up
        if time.monotonic() < phase4_end - 15:
            a_pert2 = perturbed_restart(best_a, rng, scale=0.005)
            sc2, a_opt2, _ = run_lbfgs(a_pert2, 10000,
                                        maxiter=maxiter // 2, maxfun=maxfun // 2)
            report(sc2, a_opt2, f"ILS #{trial} small")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Final polish and exploration (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Nesterov pass
    if time.monotonic() < deadline - 30:
        a_ns = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.4)
        sc = compute_score(a_ns)
        report(sc, a_ns, "final nesterov")

    # One more LP pass with small alpha
    if time.monotonic() < deadline - 30:
        a_lp_final = lp_guided_descent(best_a, n_iters=50, alpha_init=0.05, n_top=20)
        sc = compute_score(a_lp_final)
        report(sc, a_lp_final, "final LP")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 200000, maxiter=30000, maxfun=100000)
        report(sc, a_opt, "final LBFGS beta=200k")

    # Try a shorter length via interpolation
    if time.monotonic() < deadline - 40 and parent is not None:
        for n_try in [len(parent) // 2, int(len(parent) * 0.75)]:
            if time.monotonic() > deadline - 20:
                break
            if n_try < 512:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt, _ = run_lbfgs(a_interp, 5000, maxiter=20000, maxfun=60000)
            report(sc, a_opt, f"len={n_try}")

    # Try a different initialization strategy
    if time.monotonic() < deadline - 40:
        n_try = n_work
        x = np.arange(n_try, dtype=np.float64) / n_try
        # Raised cosine
        a_rc = 0.3 + 0.7 * np.cos(np.pi * (x - 0.5)) ** 2
        a_rc = np.clip(a_rc, 0.01, 10.0)
        S = a_rc.sum()
        a_rc = a_rc / S
        sc, a_opt, _ = run_lbfgs(a_rc, 3000, maxiter=30000, maxfun=100000)
        report(sc, a_opt, "raised cosine init")

        # Tukey window
        r = 0.3
        a_tk = np.ones(n_try)
        mask_lo = x < r
        mask_hi = x > 1 - r
        a_tk[mask_lo] = 0.5 + 0.5 * np.cos(np.pi * (x[mask_lo] - r) / r)
        a_tk[mask_hi] = 0.5 + 0.5 * np.cos(np.pi * (x[mask_hi] - (1 - r)) / r)
        a_tk = np.clip(a_tk, 0.01, 10.0)
        S = a_tk.sum()
        a_tk = a_tk / S
        sc, a_opt, _ = run_lbfgs(a_tk, 3000, maxiter=30000, maxfun=100000)
        report(sc, a_opt, "tukey init")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
