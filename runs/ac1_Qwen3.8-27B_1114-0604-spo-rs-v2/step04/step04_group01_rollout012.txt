"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS-B with geometric beta continuation, multi-length
exploration, exact subgradient polish, and adaptive time management.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Full linear self-convolution via FFT."""
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """h_j = sum_k w_k * a_{k-j}, middle slice of conv(a[::-1], w)."""
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    """Clip to non-negative, normalize to sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a * (1.0 / S)


def resample_sequence(a, n):
    """Linearly resample a non-negative sequence to length n."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(max(1, int(n)), dtype=np.float64)
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score(a):
    """Exact score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_obj_grad(a, beta):
    """
    LSE-smooth objective and gradient.
    F = 2 * LSE_beta(d) + pen * (sum(a)-1)^2
    where d = n*conv(a,a)/sum(a)^2.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    pen = 5e-4
    score = 2.0 * L + pen * (S - 1.0) * (S - 1.0)

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * pen * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun):
    """Run L-BFGS-B on the LSE objective."""
    a = normalize_a(a0)
    n = len(a)

    def wrapper(x):
        return lse_obj_grad(x, beta)

    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score(a), a

    bounds = [(0.0, 50.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(wrapper, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def estimate_bufuncs(t_left, n):
    """Estimate max function calls for remaining time."""
    t_left = float(max(1.0, t_left))
    base = 0.015 * max(0.5, float(n) / 4096.0)
    mf = int((0.75 * t_left) / max(base, 0.005))
    return max(50, min(mf, 2000))


def subgrad_polish(a, max_iters=30, deadline=None):
    """Exact subgradient descent with multiplicative updates."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        n = len(a)
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0:
            break
        S = float(a.sum())

        # Find top-k active lags
        for k_active in (1, 3, 5):
            if k_active >= c.size:
                idx = np.arange(c.size)
            else:
                idx = np.argpartition(c, c.size - k_active)[c.size - k_active:]
            w = np.zeros(c.size, dtype=np.float64)
            w[idx] = 1.0 / float(k_active)

            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue

            for mult in (0.01, 0.05, 0.15, 0.40, 0.80):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < exact_score(a):
                    a = q

    return float(best_sc), best_a


def smooth_noise(n, rng, strength):
    """Generate smoothed Gaussian noise."""
    noise = rng.standard_normal(int(n))
    k = min(31, max(3, int(n) // 150))
    if k % 2 == 0:
        k += 1
    if k > int(n):
        k = int(n) if int(n) % 2 == 1 else max(3, int(n) - 1)
    if k >= 3:
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
    noise = noise - float(noise.mean())
    return noise * float(strength)

    mode = int(rng.integers(0, 5))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.30))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        c = fast_self_conv(p)
        cmax = float(np.max(c))
        if cmax > 0:
            idx = np.argpartition(c, c.size - 3)[c.size - 3:]
            w = np.zeros(c.size, dtype=np.float64)
            w[idx] = 1.0 / 3.0
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * 2.0 * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    else:
        logp = np.log(p + 1e-300)
        logp = logp + smooth_noise(n, rng, strength * 0.4)
        q = np.exp(logp - float(np.max(logp)))

    return normalize_a(q)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None and parent.size > 0:
        pr = normalize_a(resample_sequence(parent, n))
        inits.append(pr)
        inits.append(pr[::-1].copy())

    inits.append(np.ones(n, dtype=np.float64) / float(n))

    for alpha in (0.3, 0.6, 0.85):
        inits.append(normalize_a(1.0 + alpha * (2.0 * x - 1.0)))
        inits.append(normalize_a(1.0 - alpha * (2.0 * x - 1.0)))

    for r in range(8):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 8)))
            amp = float(rng.uniform(0.05, 0.30))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        inits.append(normalize_a(vals))

    for _ in range(4):
        Nf = next_power_of_two(2 * n - 1)
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        vals = 1.0 + float(rng.uniform(0.3, 0.8)) * noise
        inits.append(normalize_a(vals))

    for _ in range(3):
        inits.append(normalize_a(0.1 + 0.9 * rng.random(n)))

    return inits


def optimize_trajectory(a_start, deadline, rng, label, time_frac=1.0):
    """
    Main optimization loop: geometric beta continuation.
    """
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    print("[{}] start score={:.8f} n={}".format(label, best_sc, len(a)))

    # Geometric beta schedule
    beta_start = 30.0
    beta_end = 15000.0
    n_betas = 12
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_betas) / (n_betas - 1))

    # Allocate time proportionally to beta (higher beta gets more time)
    weights = 0.5 + 0.5 * (np.arange(n_betas) / (n_betas - 1))
    weights = weights / weights.sum()

    cum = 0.0
    for i, beta in enumerate(betas):
        cum += weights[i]
        sub_dead = t0 + cum * (deadline - t0 - 10.0) * time_frac
        sub_dead = min(float(sub_dead), float(deadline - 5.0))

        iters = 0
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_bufuncs(rem, len(a))
            if mf < 50:
                break
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            iters += 1
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] new best {:.8f} beta={:.0f} it={} t={:.0f}s".format(
                    label, best_sc, beta, iters, time.monotonic() - t0))
            a = normalize_a(a2)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a low-score nonnegative sequence under the 2400s budget."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Get parent sequence if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Determine lengths to try
    lengths = []
    if parent is not None:
        plen = int(min(parent.size, 16384))
        if plen < 2048:
            plen = 2048
        lengths.append(plen)
        # Try a nearby power of 2
        p2 = 1 << (plen - 1).bit_length()
        if p2 != plen and p2 >= 2048 and p2 <= 16384:
            lengths.append(p2)
        # Try a larger length
        larger = min(16384, max(4096, plen + 2048))
        if larger != plen and larger != p2:
            lengths.append(larger)
    else:
        lengths = [8192, 16384]

    # Deduplicate and limit
    unique_lengths = []
    for n in lengths:
        n = int(n)
        if 1024 <= n <= 16384 and n not in unique_lengths:
            unique_lengths.append(n)
    if not unique_lengths:
        unique_lengths = [8192]
    lengths = unique_lengths[:3]

    print("Lengths to try: {}".format(lengths))

    best_score = np.inf
    best_seq = None

    # Phase 1: Main optimization with LSE continuation (70% of time)
    phase1_dead = t_start + 0.70 * (deadline - t_start)

    if len(lengths) == 1:
        fracs = [1.0]
    elif len(lengths) == 2:
        fracs = [0.55, 0.45]
    else:
        fracs = [0.45, 0.30, 0.25]

    # Sort lengths by preference (start with parent length)
    n_trajectories = len(lengths)
    cum = 0.0

    for i, n in enumerate(lengths):
        sub_dead = t_start + (cum + fracs[i]) * (phase1_dead - t_start)
        sub_dead = min(float(sub_dead), float(phase1_dead - 5.0))

        if time.monotonic() >= sub_dead - 30.0:
            continue

        # Generate initializations
        inits = generate_inits(n, rng, parent)

        # Score all initializations
        init_scores = []
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score(p)
            if np.isfinite(sc):
                init_scores.append((sc, p))

        if not init_scores:
            init_scores = [(2.0, np.ones(n, dtype=np.float64) / float(n))]

        init_scores.sort(key=lambda x: x[0])
        print("[n={}] best init score={:.8f} ({} inits)".format(n, init_scores[0][0], len(init_scores)))

        # Pick top 2 starts
        start_a = init_scores[0][1].copy()
        start_sc = init_scores[0][0]

        # Optimize
        sc, seq = optimize_trajectory(start_a, sub_dead, rng, "L{}".format(n),
                                      time_frac=1.0)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after L{}: {:.8f}".format(n, best_score))

        # Try second start if time remains
        if len(init_scores) > 1 and time.monotonic() < sub_dead - 30.0:
            start2 = init_scores[1][1].copy()
            # Blend with best
            if best_seq is not None:
                b2 = normalize_a(resample_sequence(best_seq, n))
                start2 = normalize_a(0.5 * start2 + 0.5 * b2)
            sc2, seq2 = optimize_trajectory(start2, sub_dead, rng, "L{}b".format(n),
                                            time_frac=0.7)
            if sc2 < best_score:
                best_score = sc2
                best_seq = seq2.copy()
                print("Global best after L{}b: {:.8f}".format(n, best_score))

        cum += fracs[i]

    if best_seq is None:
        best_seq = np.ones(4096, dtype=np.float64) / 4096.0
        best_score = exact_score(best_seq)

    best_seq = normalize_a(best_seq)

    # Phase 2: Subgradient polish (15% of time)
    phase2_dead = t_start + 0.85 * (deadline - t_start)
    if time.monotonic() < phase2_dead - 10.0:
        sc, seq = subgrad_polish(best_seq, max_iters=50, deadline=phase2_dead)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish best: {:.8f}".format(best_score))

    # Phase 3: Random restarts + high-beta final (remaining time)
    phase3_dead = deadline - 10.0
    n_restarts = 0
    while time.monotonic() < phase3_dead - 60.0 and n_restarts < 8:
        left = float(phase3_dead - 60.0 - time.monotonic())
        if left < 60.0:
            break
        chunk = time.monotonic() + min(left / float(8 - n_restarts), 120.0)

        p = perturb(best_seq, rng, strength=float(rng.uniform(0.1, 0.35)))
        # Quick LSE optimization
        sc, seq = optimize_trajectory(p, chunk, rng, "R{}".format(n_restarts),
                                      time_frac=0.6)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {} best: {:.8f}".format(n_restarts, best_score))
        n_restarts += 1

    # Final high-beta polish
    if time.monotonic() < phase3_dead - 20.0:
        rem = float(phase3_dead - 20.0 - time.monotonic())
        if rem > 15.0:
            mf = estimate_bufuncs(rem, len(best_seq))
            if mf >= 50:
                sc, seq = run_lbfgs(best_seq, 30000.0, phase3_dead - 10.0, mf)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print("High-beta final: {:.8f}".format(best_score))

    # Final tiny subgradient cleanup
    if time.monotonic() < phase3_dead - 5.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=phase3_dead - 3.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL best={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
