    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    score = 2n * LSE(c; beta) / S^2
    LSE(c; beta) = (1/beta) * log(sum_k exp(beta * c_k))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a_j = 2 * sum_l a[l] * p[l+j] (p zero outside [0,2n-2])
    # Computed as: 2 * convolve(a, p[::-1])[2n-2-j] for j=0,...,n-1
    p_rev = p[::-1]
    M2 = n + (2 * n - 1) - 1  # 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_rev_fft = np.fft.rfft(p_rev, M2_fft)
    conv_full = np.fft.irfft(A_fft * P_rev_fft, M2_fft)[:M2]
    # g_raw[j] = conv_full[2n-2-j] for j=0,...,n-1
    g_raw = conv_full[n - 1:2 * n - 1][::-1]
    dL_da = 2.0 * g_raw

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * dL_da - (4.0 * n * L / S3) * a
    return score, grad


def lse_score_grad_pal(half_a, beta):
    """LSE-smoothed score and gradient w.r.t. half of palindromic sequence."""
    n_half = len(half_a)
    n = 2 * n_half
    a = np.concatenate([half_a, half_a[::-1]])
    sc, grad_full = lse_score_grad(a, beta)
    # grad_half[i] = grad_full[i] + grad_full[n-1-i]
    grad_half = grad_full[:n_half] + grad_full[n - 1:n_half - 1:-1]
    return sc, grad_half


def nesterov_lse(a_init, beta_schedule, lr_base, n_steps_per_beta, time_limit_s, t_start,
                 verbose_interval=5000):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed score.
    Works in the space with normalization (sum=1) enforced after each step.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S0

    a_prev = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    total_steps = 0
    for beta in beta_schedule:
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            total_steps += 1

            # Nesterov extrapolation
            momentum = (total_steps - 1) / (total_steps + 1)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum > 1e-12:
                y = y / y_sum

            # Gradient at y
            sc, grad = lse_score_grad(y, beta)

            # Adaptive step size
            lr_t = lr_base / (1.0 + 0.01 * total_steps)
            a_new = y - lr_t * grad
            a_new = np.clip(a_new, 0.0, 1e10)
            a_new_sum = a_new.sum()
            if a_new_sum > 1e-12:
                a_new = a_new / a_new_sum
            else:
                a_new = a.copy()

            a_prev = a
            a = a_new

            if total_steps % verbose_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def nesterov_lse_pal(half_init, beta_schedule, lr_base, n_steps_per_beta, time_limit_s, t_start,
                     verbose_interval=5000):
    """Nesterov-accelerated gradient descent in palindrome half-space."""
    n_half = len(half_init)
    h = half_init.astype(np.float64).copy()
    S0 = h.sum()
    if S0 < 1e-12:
        h[:] = 1.0 / n_half
    else:
        h = h / S0

    h_prev = h.copy()
    full = np.concatenate([h, h[::-1]])
    best_h = h.copy()
    best_score = compute_score(full)

    total_steps = 0
    for beta in beta_schedule:
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_h
            total_steps += 1

            momentum = (total_steps - 1) / (total_steps + 1)
            y = h + momentum * (h - h_prev)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum > 1e-12:
                y = y / y_sum

            sc, grad = lse_score_grad_pal(y, beta)

            lr_t = lr_base / (1.0 + 0.01 * total_steps)
            h_new = y - lr_t * grad
            h_new = np.clip(h_new, 0.0, 1e10)
            h_new_sum = h_new.sum()
            if h_new_sum > 1e-12:
                h_new = h_new / h_new_sum
            else:
                h_new = h.copy()

            h_prev = h
            h = h_new

            if total_steps % verbose_interval == 0:
                full = np.concatenate([h, h[::-1]])
                sc = compute_score(full)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()

    full = np.concatenate([h, h[::-1]])
    sc_final = compute_score(full)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h.copy()
    return best_score, best_h


def lbfgs_free(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B optimization in free space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1e10).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lbfgs_pal(half0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B in palindrome half-space."""
    n_half = len(half0)
    h0 = np.clip(half0.astype(np.float64), 0.0, 1e10).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half

    def obj(h):
        h_c = np.clip(h, 0.0, 1e10)
        sc, gr = lse_score_grad_pal(h_c, beta)
        return sc, gr

    res = minimize(obj, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def peak_shaving_cd(a, n_sweeps=200, time_limit_s=None, t_start=None, rng=None):
    """
    Intelligent coordinate descent for peak shaving.
    At each step, identifies the peak lag and tries many possible redistributions.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        current_score = compute_score(a)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        peak_indices = list(range(js, je + 1))
        non_peak_indices = list(set(range(n)) - set(peak_indices))

        # Strategy 1: Reduce a random peak element, distribute to random non-peak elements
        for _ in range(5):
            if time.monotonic() > deadline:
                break
            j = peak_indices[rng.integers(len(peak_indices))]
            a_old = a[j].copy()
            if a_old < 1e-10:
                continue
            for frac in [0.7, 0.5, 0.3, 0.1]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                # Distribute to random non-peak positions
                n_dest = min(3, len(non_peak_indices))
                dests = rng.choice(non_peak_indices, size=n_dest, replace=False)
                for d in dests:
                    a[d] += removed / n_dest
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    break
                elif sc >= current_score:
                    a[j] = a_old
                    for d in dests:
                        a[d] -= removed / n_dest
                    continue
        # Restore if we didn't find improvement
        if compute_score(a) > best_score + 1e-15:
            # Keep the best known state - we need to track this more carefully
            pass

        # Strategy 2: Increase a random non-peak element, reduce peak elements proportionally
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            d = non_peak_indices[rng.integers(len(non_peak_indices))]
            a_old_d = a[d]
            for mult in [1.3, 1.5, 2.0]:
                added = a_old_d * (mult - 1)
                a[d] = a_old_d * mult
                # Reduce peak elements proportionally
                peak_total = a[peak_indices].sum()
                if peak_total < 1e-12:
                    a[d] = a_old_d
                    continue
                for j in peak_indices:
                    a[j] *= (peak_total - added) / peak_total
                a = np.maximum(a, 0.0)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    break
                else:
                    a[d] = a_old_d
                    for j in peak_indices:
                        a[j] /= (peak_total - added) / peak_total

        # Strategy 3: Average peak and non-peak (smoothing)
        if sweep % 5 == 0:
            peak_mean = a[peak_indices].mean()
            non_peak_mean = a[non_peak_indices].mean() if non_peak_indices else 0
            a_new = a.copy()
            for j in peak_indices:
                a_new[j] = 0.7 * a[j] + 0.3 * non_peak_mean
            for d in non_peak_indices[:10]:
                a_new[d] = 0.7 * a[d] + 0.3 * peak_mean
            a_new = np.maximum(a_new, 0)
            if a_new.sum() > 1e-12:
                sc = compute_score(a_new)
                if sc < best_score:
                    a = a_new
                    best_score = sc

    return best_score, a


def upsample(a, new_n):
    """Upsample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def init_arcsine(n, rng):
    """Arcsine-like initialization: concentrated at edges."""
    x = np.arange(n) / (n - 1)
    # Arcsine density: 1/(pi*sqrt(x*(1-x)))
    eps = 1e-6
    density = 1.0 / np.sqrt(np.maximum(x * (1 - x), eps))
    # Blend with uniform to avoid too much concentration
    alpha = 0.3 + 0.7 * rng.random()
    init = alpha * density + (1 - alpha)
    return init / init.sum()


def init_edge_spread(n, rng):
    """Edge-spread initialization with tunable concentration."""
    x = np.arange(n) / (n - 1)
    # Beta distribution-like: (x)^p * (1-x)^p
    p = 0.5 + rng.random() * 2.0
    eps = 1e-8
    init = np.maximum(x, eps) ** p * np.maximum(1 - x, eps) ** p
    # Add small uniform component
    init += 0.05
    return init / init.sum()


def init_gaussian(n, rng):
    """Gaussian initialization."""
    x = np.arange(n) / (n - 1)
    sigma = 0.15 + 0.25 * rng.random()
    init = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return init / init.sum()


def init_two_bump(n, rng):
    """Two-bump initialization."""
    x = np.arange(n) / (n - 1)
    c1 = 0.2 + 0.1 * rng.random()
    c2 = 0.7 - 0.1 * rng.random()
    s = 0.05 + 0.1 * rng.random()
    init = np.exp(-0.5 * ((x - c1) / s) ** 2) + np.exp(-0.5 * ((x - c2) / s) ** 2)
    return init / init.sum()


def init_fourier_flat(n, rng):
    """
    Initialize to approximately flat |A(f)|^2 using inverse of a flat spectrum.
    Start with random phases, compute a = real(fft^{-1}(flat_magnitude * phases)),
    then project to non-negative.
    """
    n_fft = 2 * n
    # Random phases
    phases = np.exp(1j * 2 * np.pi * rng.random(n_fft // 2 + 1))
    # Flat magnitude (with DC correction)
    mag = np.ones(n_fft // 2 + 1)
    # Inverse FFT
    spectrum = mag * phases
    full_spec = np.zeros(n_fft)
    full_spec[:n_fft // 2 + 1] = spectrum
    full_spec[n_fft // 2 + 1:] = np.conj(spectrum[1:n_fft // 2][::-1])
    a = np.real(np.fft.ifft(full_spec))[:n]
    # Project to non-negative
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    Phase 1: Diverse initializations + quick Nesterov screening
    Phase 2: Nesterov LSE continuation (free space)
    Phase 3: L-BFGS polishing
    Phase 4: Multi-scale upsampling
    Phase 5: Peak-shaving coordinate descent
    Phase 6: Palindrome-specific optimization
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 30.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
                print(f"Loaded parent: n={len(parent)}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    # ============================
    # Phase 1: Diverse initializations + screening (15% time)
    # ============================
    p1_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Diverse init + screening (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    n_work = 4096

    # Generate diverse initializations
    inits = []
    inits.append(('uniform', np.ones(n_work) / n_work))
    inits.append(('arcsine1', init_arcsine(n_work, rng)))
    inits.append(('arcsine2', init_arcsine(n_work, rng)))
    inits.append(('edge_spread1', init_edge_spread(n_work, rng)))
    inits.append(('edge_spread2', init_edge_spread(n_work, rng)))
    inits.append(('gaussian', init_gaussian(n_work, rng)))
    inits.append(('two_bump', init_two_bump(n_work, rng)))
    inits.append(('fourier_flat', init_fourier_flat(n_work, rng)))
    inits.append(('fourier_flat2', init_fourier_flat(n_work, rng)))

    # Add parent
    if parent is not None:
        if len(parent) == n_work:
            inits.append(('parent', parent.copy()))
        else:
            inits.append(('parent_up', upsample(parent, n_work)))

    # Screen each initialization with short Nesterov run
    candidates = []
    p1_deadline = t_start + p1_time
    per_init_time = p1_time / len(inits)

    for name, init in inits:
        if time.monotonic() > p1_deadline - 3:
            break
        time_left = p1_deadline - time.monotonic()
        t_alloc = min(per_init_time, time_left)

        sc, a_opt = nesterov_lse(init, [20, 50, 100, 300], lr_base=0.05,
                                  n_steps_per_beta=max(200, int(t_alloc * 800)),
                                  time_limit_s=t_alloc, t_start=time.monotonic() - 0.1)
        candidates.append((sc, a_opt, name))
        print(f"  {name}: score={sc:.8f}")
        try_update(sc, a_opt, f"screen-{name}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Best after screening: {best_score:.8f}")

    # Use best candidate as starting point
    a_current = best_a.copy()
    if a_current is None:
        a_current = np.ones(n_work) / n_work

    # ============================
    # Phase 2: Nesterov LSE continuation - free space (25% time)
    # ============================
    p2_time = effective_budget * 0.25
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Nesterov LSE continuation (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    betas = [100, 300, 1000, 3000, 10000, 30000, 80000]
    lrs = [0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]

    p2_deadline = t_start + p1_time + p2_time
    for i, (beta, lr) in enumerate(zip(betas, lrs)):
        time_left = p2_deadline - time.monotonic()
        if time_left < 3:
            break
        t_alloc = time_left / (len(betas) - i)
        sc, a_opt = nesterov_lse(a_current, [beta], lr_base=lr,
                                  n_steps_per_beta=1000000,
                                  time_limit_s=t_alloc, t_start=time.monotonic() - 0.1)
        if sc < best_score:
            try_update(sc, a_opt, f"Nesterov beta={beta}")
        a_current = a_opt
        elapsed_beta = time.monotonic()
        print(f"  beta={beta}: score={sc:.8f} (total t={elapsed_beta - t_start:.0f}s)")

    # ============================
    # Phase 3: L-BFGS polishing (15% time)
    # ============================
    p3_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 3: L-BFGS polishing (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    p3_deadline = t_start + p1_time + p2_time + p3_time
    for beta in [3000, 10000, 30000, 80000, 200000]:
        time_left = p3_deadline - time.monotonic()
        if time_left < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_free(a_current, beta, maxiter=5000, maxfun=80000)
        if sc < best_score:
            try_update(sc, a_opt, f"LBFGS beta={beta}")
        a_current = a_opt
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # ============================
    # Phase 4: Multi-scale upsampling (15% time)
    # ============================
    p4_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Multi-scale upsampling (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    p4_deadline = t_start + p1_time + p2_time + p3_time + p4_time

    # Try upsampling to 8192
    n_up = 2 * len(a_current)
    if n_up <= 16384:
        a_up = upsample(a_current, n_up)
        time_left = p4_deadline - time.monotonic()
        if time_left > 5:
            print(f"  Upsampling to n={n_up}")
            sc, a_up_opt = nesterov_lse(a_up, [100, 500, 2000, 8000], lr_base=0.02,
                                         n_steps_per_beta=500000,
                                         time_limit_s=time_left * 0.6,
                                         t_start=time.monotonic() - 0.1)
            if sc < best_score:
                try_update(sc, a_up_opt, f"upsampled n={n_up}")
                a_current = a_up_opt

            # Quick L-BFGS on upsampled
            time_left = p4_deadline - time.monotonic()
            if time_left > 10:
                sc, a_up_opt2 = lbfgs_free(a_up_opt, 5000, maxiter=3000, maxfun=50000)
                if sc < best_score:
                    try_update(sc, a_up_opt2, f"upsampled+LBFGS n={n_up}")
                    a_current = a_up_opt2
                print(f"  Upsampled+LBFGS: {sc:.8f}")

    # Also try a fresh run at a different size
    n_alt = 8192
    if n_alt != len(a_current):
        time_left = p4_deadline - time.monotonic()
        if time_left > 10:
            a_alt = upsample(a_current, n_alt)
            sc, a_alt_opt = nesterov_lse(a_alt, [200, 1000, 5000], lr_base=0.015,
                                          n_steps_per_beta=300000,
                                          time_limit_s=time_left * 0.7,
                                          t_start=time.monotonic() - 0.1)
            if sc < best_score:
                try_update(sc, a_alt_opt, f"alt n={n_alt}")
                a_current = a_alt_opt
            print(f"  Alt n={n_alt}: {sc:.8f}")

    # ============================
    # Phase 5: Peak-shaving coordinate descent (10% time)
    # ============================
    p5_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Peak-shaving CD (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    p5_deadline = t_start + p1_time + p2_time + p3_time + p4_time + p5_time
    time_left = p5_deadline - time.monotonic()
    if time_left > 3:
        sc, a_cd = peak_shaving_cd(a_current, n_sweeps=500,
                                    time_limit_s=time_left, t_start=time.monotonic(),
                                    rng=rng)
        if sc < best_score:
            try_update(sc, a_cd, "peak-shaving CD")
        a_current = a_cd if sc < compute_score(a_current) else a_current
        print(f"  Peak-shaving: {sc:.8f}")

    # ============================
    # Phase 6: Palindrome-specific optimization (10% time)
    # ============================
    p6_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Palindrome optimization (budget={p6_time:.0f}s)")
    print(f"{'=' * 60}")

    p6_deadline = t_start + p1_time + p2_time + p3_time + p4_time + p5_time + p6_time

    # Symmetrize current best
    n_cur = len(a_current)
    if n_cur % 2 != 0:
        n_cur += 1
        a_current = upsample(a_current, n_cur)
    n_half = n_cur // 2
    half_best = (a_current[:n_half] + a_current[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    # Check palindromic score
    pal_check = np.concatenate([half_best, half_best[::-1]])
    pal_sc = compute_score(pal_check)
    if pal_sc < best_score:
        try_update(pal_sc, pal_check, "palindromized")

    # Nesterov in palindrome space
    time_left = p6_deadline - time.monotonic()
    if time_left > 5:
        sc, h_opt = nesterov_lse_pal(half_best, [500, 2000, 8000], lr_base=0.01,
                                      n_steps_per_beta=300000,
                                      time_limit_s=time_left * 0.6,
                                      t_start=time.monotonic() - 0.1)
        a_pal = np.concatenate([h_opt, h_opt[::-1]])
        if sc < best_score:
            try_update(sc, a_pal, "Nesterov-pal")
            a_current = a_pal

        # L-BFGS in palindrome space
        time_left = p6_deadline - time.monotonic()
        if time_left > 5:
            sc, h_opt2 = lbfgs_pal(h_opt, 10000, maxiter=5000, maxfun=80000)
            a_pal2 = np.concatenate([h_opt2, h_opt2[::-1]])
            if sc < best_score:
                try_update(sc, a_pal2, "LBFGS-pal")
                a_current = a_pal2
            print(f"  LBFGS-pal: {sc:.8f}")

    # ============================
    # Phase 7: Final intensive polish (10% time)
    # ============================
    p7_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 7: Final polish (budget={p7_time:.0f}s)")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget

    # Very high beta L-BFGS
    time_left = final_deadline - time.monotonic()
    if time_left > 5:
        sc, a_opt = lbfgs_free(a_current, 50000, maxiter=8000, maxfun=150000)
        if sc < best_score:
            try_update(sc, a_opt, "final LBFGS high-beta")
        a_current = a_opt if sc < compute_score(a_current) else a_current
        print(f"  Final LBFGS: {sc:.8f}")

    # Final Nesterov with very high beta
    time_left = final_deadline - time.monotonic()
    if time_left > 5:
        sc, a_opt = nesterov_lse(a_current, [50000, 150000], lr_base=0.0005,
                                  n_steps_per_beta=500000,
                                  time_limit_s=time_left * 0.5,
                                  t_start=time.monotonic() - 0.1)
        if sc < best_score:
            try_update(sc, a_opt, "final Nesterov high-beta")
        a_current = a_opt if sc < compute_score(a_current) else a_current
        print(f"  Final Nesterov: {sc:.8f}")

    # Final peak shaving
    time_left = final_deadline - time.monotonic()
    if time_left > 3:
        sc, a_cd = peak_shaving_cd(a_current, n_sweeps=200,
                                    time_limit_s=min(time_left, 30.0),
                                    t_start=time.monotonic(), rng=rng)
        if sc < best_score:
            try_update(sc, a_cd, "final peak-shaving")
        print(f"  Final peak-shaving: {sc:.8f}")

    # One more L-BFGS
    time_left = final_deadline - time.monotonic()
    if time_left > 5:
        sc, a_opt = lbfgs_free(best_a, 100000, maxiter=5000, maxfun=100000)
        if sc < best_score:
            try_update(sc, a_opt, "final LBFGS 2")
        print(f"  Final LBFGS 2: {sc:.8f}")

    # ============================
    # Final verification
    # ============================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
