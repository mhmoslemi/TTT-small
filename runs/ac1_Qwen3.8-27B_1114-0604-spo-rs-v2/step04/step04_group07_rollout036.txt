def spectral_peak_reduction(a, n_peaks=5, strength=0.1):
    """
    Reduce peaks in the self-convolution by modifying the spectral content.
    """
    n = len(a)
    N = 2 * n - 1
    M_fft = 1 << (N - 1).bit_length()
    
    A = np.fft.rfft(a, M_fft)
    # A[m] for m = 0, 1, ..., M_fft//2
    # The relevant frequencies are m = 0, ..., N-1 (but due to real input, 
    # only m = 0, ..., N//2 are independent, and the rest are conjugate)
    
    # Compute c = IDFT(A^2)
    A_sq = A * A
    c = np.fft.irfft(A_sq, M_fft)[:N]
    
    # Find top peaks
    peak_indices = np.argsort(c)[-n_peaks:]
    
    # For each peak at k, the contribution from frequency m is A[m]^2 * exp(2πimk/N) / N
    # We want to reduce |A[m]|^2 for m that contribute to peaks
    
    # Compute "damage" of each frequency: how much does it contribute to peaks?
    damage = np.zeros(len(A))
    for k in peak_indices:
        # Contribution of freq m to peak k: |A[m]|^2 * |exp(2πimk/N)| = |A[m]|^2
        # But the PHASE matters: it's A[m]^2 * exp(2πimk/N), not |A[m]|^2
        # The real part contributes to c[k]
        A_sq_m = A * A  # complex
        phases = np.exp(2j * np.pi * np.arange(len(A)) * k / M_fft)
        damage += np.real(A_sq_m * phases)
    
    # Reduce A where damage is positive (contributing to peaks)
    # A_new = A * (1 - strength * damage / |A|^2)
    A_abs_sq = np.abs(A) ** 2
    reduction = np.zeros(len(A))
    mask = A_abs_sq > 1e-20
    reduction[mask] = strength * damage[mask] / A_abs_sq[mask]
    reduction = np.clip(reduction, -1.0, 0.5)  # Don't increase, small decreases
    
    A_new = A * (1.0 + reduction)
    
    # Invert
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.maximum(a_new, 0)
    
    if a_new.sum() > 1e-12:
        a_new = a_new / a_new.sum()
    
    return a_new

def spectral_equalize(a, n_iters=100, strength=0.1):
    """
    Try to make the power spectrum |A[m]|^2 as flat as possible.
    This helps flatten the self-convolution.
    """
    n = len(a)
    N = 2 * n - 1
    M_fft = 1 << (N - 1).bit_length()
    
    a_work = a.astype(np.float64).copy()
    
    for it in range(n_iters):
        A = np.fft.rfft(a_work, M_fft)
        power = np.abs(A) ** 2
        # Target: flat power spectrum (exclude DC)
        # The DC component A[0] = sum(a) is fixed
        # For m >= 1, target power = total_power / (N//2)
        target = np.mean(power[1:])  # approximate flat level
        # Adjust: reduce high-power bins, increase low-power bins
        # A_new[m] = A[m] * sqrt(target / power[m])
        ratio = np.ones(len(A))
        mask = power[1:] > 1e-20
        ratio[1:][mask] = np.sqrt(target / power[1:][mask])
        # Smooth the ratio to avoid wild changes
        ratio = 1.0 + strength * (ratio - 1.0)
        ratio = np.clip(ratio, 0.5, 2.0)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-12:
            a_new = a_new / a_new.sum()
        else:
            break
        a_work = a_new
    
    return a_work

For real a, A[N-m] = conj(A[m]), so A^2 has the property A^2[N-m] = conj(A[m])^2 = conj(A^2[m]). So c is real.

But the KEY point is: c[k] depends on the PHASE of A[m], not just its magnitude. Two sequences with the same power spectrum can have very different self-convolutions.

So spectral equalization of |A|^2 alone is not sufficient. We also need to control the phases.

"""
Optimized step function sequence optimizer combining:
- LSE-smoothed Adam with geometric beta continuation (adapted parameters)
- Top-K peak multiplicative weight updates (novel simultaneous peak reduction)
- Spectral flattening via power spectrum equalization
- L-BFGS-B polishing at multiple beta levels
- Palindrome constraint for reduced dimensionality
- Multi-start with diverse perturbations and re-optimization
- Adaptive trust-region stepping at high beta to handle non-smoothness

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    L = c_max + log(sum(exp(beta*(c - c_max))))/beta
    score = 2n*L/S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 0.0)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        L = c_max
        p = np.zeros(len(c))
        p[int(c.argmax())] = 1.0
    else:
        L = c_max + np.log(exp_sum) / beta
        p = exp_scaled / exp_sum
    score = 2.0 * n * L / (S * S)
    a_flipped = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P_fft = np.fft.rfft(p, M2_fft)
    AF_fft = np.fft.rfft(a_flipped, M2_fft)
    conv_pa = np.fft.irfft(P_fft * AF_fft, M2_fft)[:M2]
    grad_L = 2.0 * conv_pa[n - 1:n - 1 + n]
    S2 = S * S
    S3 = S2 * S
    grad_score = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad_score


def lse_score_grad_pal(half, beta):
    """LSE-smoothed score and gradient w.r.t. half of palindromic sequence."""
    n_half = len(half)
    n = 2 * n_half
    a = np.concatenate([half, half[::-1]])
    sc, grad = lse_score_grad(a, beta)
    grad_half = grad[:n_half] + grad[n_half:n][::-1]
    return sc, grad_half


def adam_lse_optimize(half_init, beta_schedule, lr_schedule, time_limit, t_start,
                      normalize=True, verbose=False):
    """Adam optimizer with LSE smoothing in palindrome space."""
    n_half = len(half_init)
    h = half_init.astype(np.float64).copy()
    if h.sum() < 1e-12:
        h[:] = 1.0 / n_half

    m = np.zeros(n_half)
    v = np.zeros(n_half)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_h = h.copy()
    best_score = compute_score(np.concatenate([h, h[::-1]]))

    total_steps = sum(s for s in [max(1, len(beta_schedule)) ] )
    step_count = 0

    for bi, beta in enumerate(beta_schedule):
        lr = lr_schedule[min(bi, len(lr_schedule) - 1)]
        steps_in_phase = 1
        while time.monotonic() - t_start < time_limit and steps_in_phase < 500000:
            steps_in_phase += 1
            step_count += 1
            score, grad = lse_score_grad_pal(h, beta)

            # Adaptive learning rate based on gradient norm
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                lr_t = lr * min(1.0, 0.1 / gnorm)
            else:
                lr_t = lr

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** step_count)
            v_hat = v / (1.0 - b2 ** step_count)
            h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0.0, 1e10)

            if normalize:
                s = h.sum()
                if s > 1e-12:
                    h = h / s

            if steps_in_phase % 2000 == 0:
                a_full = np.concatenate([h, h[::-1]])
                sc = compute_score(a_full)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()

    a_full = np.concatenate([h, h[::-1]])
    sc_final = compute_score(a_full)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h.copy()

    return best_score, best_h


def top_k_mw_update(half, n_peaks=10, n_iters=200, lr=0.3):
    """
    Top-K multiplicative weight update: simultaneously reduce top-K peaks.
    Works in palindrome space.
    """
    n_half = len(half)
    n = 2 * n_half
    h = half.astype(np.float64).copy()
    h = h / h.sum()

    for it in range(n_iters):
        a = np.concatenate([h, h[::-1]])
        c = fast_conv(a)
        S = a.sum()

        # Find top-K peaks
        peak_indices = np.argsort(c)[-n_peaks:]

        # Compute total contribution vector
        contrib = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                contrib[js:je + 1] += a[idx]

        # Normalized damage
        max_contrib = contrib.max() if contrib.max() > 1e-20 else 1.0
        damage = contrib / max_contrib

        # Update: reduce high-damage elements, boost low-damage ones
        h_full = np.concatenate([h, h[::-1]])
        h_full = h_full * np.exp(-lr * damage)
        # Slight boost to zero-damage elements
        zero_mask = damage < 1e-10
        if zero_mask.sum() > 0:
            h_full[zero_mask] *= np.exp(lr * 0.3)

        h_full = np.maximum(h_full, 1e-15)
        h_full = h_full / h_full.sum()

        # Enforce palindrome
        h = (h_full[:n_half] + h_full[n_half:][::-1]) / 2.0
        h = h / h.sum()

    return h


def spectral_flatten(half, n_iters=50, strength=0.05):
    """
    Spectral flattening: try to equalize the power spectrum to flatten convolution.
    Works in palindrome space.
    """
    n_half = len(half)
    n = 2 * n_half
    h = half.astype(np.float64).copy()
    h = h / h.sum()

    for it in range(n_iters):
        a = np.concatenate([h, h[::-1]])
        N = 2 * n - 1
        M_fft = 1 << (N - 1).bit_length()

        A = np.fft.rfft(a, M_fft)
        power = np.abs(A) ** 2

        # Target: equal power across non-DC bins
        non_dc_power = power[1:]
        if non_dc_power.sum() < 1e-30:
            break
        target_level = non_dc_power.mean()

        # Adjust amplitude: A_new = A * sqrt(target/power)
        ratio = np.ones(len(A))
        mask = power[1:] > 1e-20
        ratio[1:][mask] = np.sqrt(target_level / power[1:][mask])
        ratio = 1.0 + strength * (ratio - 1.0)
        ratio = np.clip(ratio, 0.5, 2.0)

        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        # Enforce palindrome
        h = (a_new[:n_half] + a_new[n_half:][::-1]) / 2.0
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-12:
            break
        h = h / h.sum()

    return h


def run_lbfgs_pal(half0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in palindrome half-space with LSE smoothing."""
    n_half = len(half0)
    h0 = np.clip(half0.astype(np.float64), 0.0, 1000.0).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half

    def obj_fn(h):
        h_c = np.clip(h, 0.0, 1e10)
        sc, gr = lse_score_grad_pal(h_c, beta)
        return sc, gr

    res = minimize(obj_fn, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in full a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_fn(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad_a(x_c, beta)
        return sc, gr

    res = minimize(obj_fn, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def generate_initializations(rng, n):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (various widths)
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Sinusoidal modulation
    for freq in [2, 4, 6, 8, 12]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Two-bump
    for (c1, c2, s1, s2) in [(0.25, 0.75, 0.08, 0.08), (0.2, 0.8, 0.05, 0.05),
                               (0.3, 0.7, 0.15, 0.15), (0.15, 0.6, 0.1, 0.2)]:
        b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
        inits.append(b / b.sum())

    # Random smooth
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.5
        k = max(3, n // 50)
        r_smooth = np.convolve(r, np.ones(k) / k, mode='same')
        inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    for _ in range(3):
        half_n = n // 2
        h = np.abs(rng.standard_normal(half_n)) + 0.5
        h_smooth = np.convolve(h, np.ones(max(3, half_n // 50)) / max(3, half_n // 50), mode='same')
        pal = np.concatenate([h_smooth, h_smooth[::-1]])
        inits.append(pal / pal.sum())

    # Power-law (decreasing)
    inits.append(x ** 2 / (x ** 2).sum())
    inits.append((1 - x) ** 2 / ((1 - x) ** 2).sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def coord_descent_peak(half, n_sweeps=50, time_limit=None, t_start=None):
    """Coordinate descent targeting peak reduction in palindrome space."""
    n_half = len(half)
    n = 2 * n_half
    h = half.astype(np.float64).copy()
    best_score = compute_score(np.concatenate([h, h[::-1]]))
    deadline = t_start + time_limit if (time_limit is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        a = np.concatenate([h, h[::-1]])
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing peak contributors (in palindrome space)
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            # Map to palindrome half index
            hj = min(j, n - 1 - j)
            if hj >= n_half:
                continue
            h_old = h[hj]
            if h_old < 1e-10:
                continue
            for frac in [0.85, 0.7, 0.5]:
                h[hj] = h_old * frac
                a_test = np.concatenate([h, h[::-1]])
                sc = compute_score(a_test)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            if h[hj] >= h_old * 0.99:
                pass  # No improvement, keep original

        # Try boosting non-peak elements
        non_peak_mask = np.ones(n, dtype=bool)
        non_peak_mask[js:je + 1] = False
        non_peak_half = np.zeros(n_half, dtype=bool)
        for j in range(n_half):
            a_idx_1 = j
            a_idx_2 = n - 1 - j
            if not non_peak_mask[a_idx_1] and not non_peak_mask[a_idx_2]:
                non_peak_half[j] = True

        nc_idx = np.where(non_peak_half)[0]
        if len(nc_idx) > 0:
            for _ in range(min(5, len(nc_idx))):
                if time.monotonic() > deadline:
                    break
                j = nc_idx[np.random.randint(len(nc_idx))]
                h_old = h[j]
                for mult in [1.1, 1.3, 1.5]:
                    h[j] = h_old * mult
                    a_test = np.concatenate([h, h[::-1]])
                    sc = compute_score(a_test)
                    if sc < score:
                        score = sc
                        if sc < best_score:
                            best_score = sc
                        break
                h[j] = h[j]

    return best_score, h
