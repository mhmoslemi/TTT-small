"""
Hybrid LP-direction + LSE-LBFGS continuation + Coordinate Descent
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key innovation: Linear Programming direction finding (maximize sum(b)
s.t. (a*b) <= M) provides globally-informed search directions that
complement local LSE-based gradient methods. Combined with Nesterov-style
averaging, active-set LP solving, and diverse spectral restarts.

import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_objective_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    AR = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_ap = np.fft.irfft(AR * P, M2_fft)
    dL_da = conv_ap[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def build_conv_matrix_sparse(a, n):
    rows = np.zeros(2 * n - 1, dtype=np.int32)
    cols = np.zeros(2 * n - 1, dtype=np.int32)
    vals = np.zeros(2 * n - 1, dtype=np.float64)
    # Use vectorized construction
    all_rows = []
    all_cols = []
    all_vals = []
    for j in range(2 * n - 1):
        l_min = max(0, j - n + 1)
        l_max = min(j, n - 1)
        l_range = np.arange(l_min, l_max + 1)
        all_rows.extend([j] * len(l_range))
        all_cols.extend(l_range.tolist())
        all_vals.extend(a[j - l_range].tolist())
    A = sp.csr_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n))
    return A


def solve_lp_direction(a, M, active_indices=None):
    n = len(a)
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        rows_sub = active_indices
    else:
        rows_sub = range(2 * n - 1)
    
    all_rows = []
    all_cols = []
    all_vals = []
    row_idx = 0
    n_rows = len(rows_sub) if isinstance(rows_sub, (list, np.ndarray)) else 2*n-1
    
    if isinstance(rows_sub, (list, np.ndarray)):
        row_list = list(rows_sub)
    else:
        row_list = list(range(2*n-1))
    
    for j in row_list:
        l_min = max(0, j - n + 1)
        l_max = min(j, n - 1)
        for l in range(l_min, l_max + 1):
            all_rows.append(row_idx)
            all_cols.append(l)
            all_vals.append(a[j - l])
        row_idx += 1
    
    A = sp.csr_matrix((all_vals, (all_rows, all_cols)), shape=(row_idx, n))
    c_obj = -np.ones(n)

def subgrad_peak_info(a):
    n = len(a)
    c = fast_conv(a)

    je = min(k_star, n - 1)
    if je >= js:

        sg[js:je + 1] = a[idx]

def nesterov_subgrad(a, n_steps=5000, time_budget=None, t_start=None):
    """
    Nesterov-accelerated subgradient method on max(self-conv) with sum=1 constraint.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    a = a / S0  # work with sum=1
    best_score = compute_score(a)
    best_a = a.copy()
    
    a_avg = a.copy()
    y = a.copy()  # Nesterov iterate
    alpha = 0.5  # initial step
    beta_nest = 1.0  # momentum parameter
    
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        c = fast_conv(y)
        k_star = int(c.argmax())
        M = float(c[k_star])
        
        # Subgradient of max(a*a) w.r.t. a at index k_star
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = y[idx]
        
        # Projected gradient step: minimize max(a*a) s.t. a>=0, sum(a)=1
        # Descent direction: -sg, but we also need to maintain sum=1
        # Use mirror descent / exponential weights
        lr_t = 2.0 / (t + 2)  # standard subgradient step
        
        # Update: y_new = project(y - lr * (sg - mean(sg)))
        sg_centered = sg - sg.sum() / n
        y_new = y - lr_t * sg_centered
        y_new = np.clip(y_new, 0, None)
        s = y_new.sum()
        if s < 1e-12:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s
        
        # Nesterov averaging
        a_avg = a_avg + y_new / (t + 1)
        
        # Update y with momentum
        beta_new = 2.0 / (t + 2)
        y = y_new
        
        if t % 200 == 0:
            sc = compute_score(a_avg)
            if sc < best_score:
                best_score = sc
                best_a = a_avg.copy()

        best_score = sc_final
        best_a = a_avg.copy()
    
    best_a = best_a * S0
    return compute_score(best_a), best_a
