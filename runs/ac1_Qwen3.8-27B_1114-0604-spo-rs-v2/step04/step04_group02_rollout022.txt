"""
Dual-ascent multiplicative weight optimizer with palindromic constraint and LSE-polished L-BFGS.
Uses KKT structure: at optimum, pressure p[i] = sum_k lambda_k * a[k-i] is constant for a[i]>0.
Multiplicative update a[i] *= (target/p[i])^eta drives pressure toward uniformity.
Combined with annealed beta schedule and high-precision L-BFGS polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    la = len(a)
    lb = len(b)
    M = la + lb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def make_palindromic(a):
    """Make a sequence palindromic by symmetric averaging."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def compute_pressure(a, lambda_w):
    """
    Compute pressure vector p[i] = sum_j a[j] * lambda_w[i+j].
    This is the gradient of sum_k lambda_k * conv_k w.r.t. a[i].
    """
    n = len(a)
    lambda_rev = lambda_w[::-1]
    cross = fast_cross_conv(a, lambda_rev)
    # p[i] = cross[2n-2-i] for i = 0,...,n-1
    p = cross[n - 2:2 * n - 1][::-1]
    return p


def dual_ascent_step(a, beta, eta):
    """One step of dual-ascent multiplicative weight update."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a_norm = a / S

    conv = fast_self_conv(a_norm)
    conv_max = float(conv.max())
    scaled = beta * (conv - conv_max)
    exp_vals = np.exp(np.maximum(scaled, -100))
    lam_sum = exp_vals.sum()
    if lam_sum < 1e-30:
        lambda_w = np.ones(len(conv)) / len(conv)
    else:
        lambda_w = exp_vals / lam_sum

    p = compute_pressure(a_norm, lambda_w)
    p_safe = np.maximum(p, 1e-30)

    log_target = float(np.mean(np.log(p_safe[np.where(a_norm > 1e-15)])))
    target_p = np.exp(log_target)

    update = np.power(target_p / p_safe, eta)
    a_new = a_norm * update
    a_new = np.clip(a_new, 0.0, 10000.0)
    a_new = make_palindromic(a_new)
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new / S_new
    return a_new


def dual_ascent_optimize(a0, n_phases=8, iters_per_phase=200, eta_start=0.3, eta_decay=0.7,
                         beta_start=20.0, beta_end=20000.0):
    """Annealed dual-ascent optimization."""
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    a = make_palindromic(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for phase in range(n_phases):
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        eta = eta_start * (eta_decay ** phase)

        for it in range(iters_per_phase):
            a = dual_ascent_step(a, beta, eta)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -200))
    p_w = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: d(score)/da[i]
    # score = 2n*L/S^2
    # dL/da[i] = sum_k p_w[k] * d(conv_k)/da[i] = sum_k p_w[k] * a[k-i] (valid)
    # = sum_j p_w[i+j] * a[j] = pressure of a with p_w
    p_w_rev = p_w[::-1]
    cross = fast_cross_conv(a, p_w_rev)
    dL_da = cross[n - 2:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def active_set_ld(a0, n_iters=200):
    """
    Active-set style: identify top-k active convolution peaks,
    then solve a constrained problem to equalize them.
    Uses iterative proportional fitting on the active set.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    a = make_palindromic(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        conv = fast_self_conv(a)
        M = float(conv.max())
        # Active set: entries within 1% of max
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]

        # Build soft weights concentrating on active set
        w = np.zeros(len(conv))
        w[active] = 1.0
        # Smooth slightly
        w = w * 0.7 + np.ones(len(conv)) * 0.3 / len(conv)
        w = w / w.sum()

        p = compute_pressure(a, w)
        p_safe = np.maximum(p, 1e-30)
        pos_mask = a > 1e-15
        if pos_mask.sum() > 0:
            log_target = float(np.mean(np.log(p_safe[pos_mask])))
        else:
            log_target = 0.0
        target_p = np.exp(log_target)

        eta = 0.15 * (1.0 - 0.5 * it / n_iters)
        update = np.power(target_p / p_safe, eta)
        a_new = a * update
        a_new = np.clip(a_new, 0.0, 10000.0)
        a_new = make_palindromic(a_new)
        S_new = a_new.sum()
        if S_new > 1e-12:
            a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def perturb_sequence(a, rng, scale=0.02):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 4)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 8.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 3 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // max(1, block_size))):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        # Random subset rescale
        n_sel = max(10, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return make_palindromic(a_p)


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(make_palindromic(parent.astype(np.float64).copy()))
    elif parent is not None:
        inits.append(make_palindromic(np.interp(
            np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    # U-shaped
    inits.append(0.1 + 0.9 * (2 * x - 1) ** 2)
    # Cosine
    inits.append(np.maximum(0.05, np.cos(np.pi * (x - 0.5))))
    # Random
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    # Bimodal
    inits.append(0.1 + 0.9 * np.exp(-((x - 0.3) ** 2) / 0.01) + 0.9 * np.exp(-((x - 0.7) ** 2) / 0.01))

    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.0, 1000.0)
        s = inits[i].sum()
        if s > 1e-12:
            inits[i] = inits[i] / s
        inits[i] = make_palindromic(inits[i])
    return inits
