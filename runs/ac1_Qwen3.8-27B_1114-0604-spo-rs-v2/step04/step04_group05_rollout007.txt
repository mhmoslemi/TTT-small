    """
    Loss = (1/beta) * log(sum_k exp(beta * max(0, c[k] - T)))
    This penalizes convolution values exceeding threshold T.
    """

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a
