"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using:
1. LP-guided alternating optimization (fix a, solve LP for b)
2. Subgradient method with Nesterov averaging
3. L-BFGS on log-sum-exp smoothed objective
4. Multi-length exploration

import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_lp_matrix(a):
    """Build the constraint matrix for the LP subproblem.
    Row i, col j = a[i-j] if 0 <= i-j <= n-1, else 0.
    """
    n = len(a)
    # Column j has a[0] at row j, a[1] at row j+1, ..., a[n-1] at row j+n-1
    row_ind = np.repeat(np.arange(2*n-1), 0)  # placeholder
    # Build column by column
    all_rows = []
    all_cols = []
    all_vals = []
    for j in range(n):
        r = j + np.arange(n)
        all_rows.append(r)
        all_cols.append(np.full(n, j))
        all_vals.append(a)
    all_rows = np.concatenate(all_rows)
    all_cols = np.concatenate(all_cols)
    all_vals = np.concatenate(all_vals)
    A = sp.csc_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n))
    return A

def solve_lp_subproblem(a, M):
    """Given fixed a and target M, find b >= 0 maximizing sum(b) s.t. conv(a,b) <= M."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    # Gradient: dL/da[i] = 2 * sum_j p[i+j] * a[j]
    # This is a correlation
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_obj_grad(x, beta)
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def mirror_descent_smoothed(a, n_iters=3000, beta=100):
    """Mirror descent on the LSE-smoothed objective (entropic regularization)."""
    n = len(a)

    best_a = a.copy()

        sc, grad = lse_obj_grad(a, beta)

            best_sc = sc
            best_a = a.copy()
        
        # Mirror descent: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        eta = 1.0 / (np.linalg.norm(grad) + 1e-10)
        eta *= 0.5  # damping
        log_a = np.log(a) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        if sc_new < best_sc:

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using:
1. LP-guided alternating optimization (fix a, solve LP for b via HiGHS)
2. Accelerated subgradient with Nesterov averaging and Polyak step
3. L-BFGS on log-sum-exp smoothed objective with adaptive beta
4. Multi-length exploration and diverse initializations
Key improvement: LP subproblem gives exact optimal direction at each step,
combined with Nesterov-accelerated subgradient for fast convergence.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Bisection for the threshold
    lo = -v.max() - 1.0
    hi = 0.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.maximum(v - mid, 0.0)
        s = proj.sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    result = np.maximum(v - (lo + hi) / 2.0, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def subgradient_descent(a, n_iters=10000, time_limit=60.0):
    """
    Projected subgradient with Nesterov averaging.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    # Nesterov averaging
    a_avg = np.zeros(n)
    a_y = a.copy()  # momentum variable

    t0 = time.monotonic()
    mu = 1.0  # interpolation parameter (Polyak)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a_y)
        k_star = int(np.argmax(c))
        M_cur = float(c[k_star])
        sc_cur = 2.0 * n * M_cur

        # Subgradient of max(conv) at a_y
        s = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a_y[j_start:j_end + 1]
            s[j_start:j_end + 1] = 2.0 * seg[::-1]
            # Correct diagonal: if 2i = k_star
            if k_star % 2 == 0:
                i_diag = k_star // 2
                if j_start <= i_diag <= j_end:
                    s[i_diag] = a_y[i_diag]

        # Polyak step size
        sc_lb = 2.0 * n / (2.0 * n)  # lower bound ~ 1 (use conservative)
        step = (sc_cur - sc_lb) / (float(np.dot(s, s)) + 1e-30)
        step = min(step, 0.2)
        step = max(step, 1e-8)

        # Update y
        a_y = a_y - step * s
        a_y_new = project_simplex(a_y)

        # Nesterov momentum
        t_it = (it + 1) / (n_iters + 1.0)
        a = (1.0 - mu) * a + mu * a_y_new + (1.0 - mu) * (a_y_new - a)
        a = project_simplex(a)

        # Running average
        a_avg = a_avg + (a - a_avg) / (it + 1.0)

        # Check both current and averaged
        sc_check = 2.0 * n * float(fast_conv(a).max())
        if sc_check < best_sc:
            best_sc = sc_check
            best_a = a.copy()

        if it % 100 == 0:
            sc_avg = 2.0 * n * float(fast_conv(a_avg).max())
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = a_avg.copy()

    # Final check of averaged
    sc_avg = 2.0 * n * float(fast_conv(a_avg).max())
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = a_avg.copy()

    return best_a


def build_lp_constraint_matrix(a):
    """
    Build sparse constraint matrix for LP subproblem.
    Row i, col j = a[i-j] if 0 <= i-j <= n-1, else 0.
    Each column j has entries a[0],...,a[n-1] at rows j,...,j+n-1.
    """
    n = len(a)
    # Vectorized construction
    all_rows = np.concatenate([j + np.arange(n) for j in range(n)])
    all_cols = np.concatenate([np.full(n, j) for j in range(n)])
    all_vals = np.tile(a, n)
    A = sp.csc_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_subproblem(a, M):
    """
    Given fixed a (sum=1) and target M, solve:
    max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns b or None if infeasible/convergence failure.
    """
    n = len(a)
    A = build_lp_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def run_lp_alternating(a, n_iters=50, time_limit=120.0):
    """
    Alternating LP optimization.
    At each step: fix a, solve LP for b, interpolate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n)
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        M = float(c.max())

        b = solve_lp_subproblem(a, M)
        if b is None:
            break

        S_b = float(b.sum())
        if S_b < 1e-12:
            break
        b = b / S_b

        # Adaptive interpolation
        t_lr = 0.4 / (1.0 + 0.05 * it)
        a_new = (1.0 - t_lr) * a + t_lr * b
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc_new = 2.0 * n * float(fast_conv(a_new).max())
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    LP iter {it}: score={sc_new:.8f}")
        a = a_new

    return best_a


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    # dL/da[i] = 2 * (correlation of p with reversed a)
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def error_reduction(a, n_iters=200):
    """
    Frequency-domain error reduction: iteratively flatten the convolution.
    Works on normalized (sum=1) sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M_fft)
        conv = np.fft.irfft(A * A, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        # Threshold: reduce values above threshold
        thresh = conv_mean + 0.3 * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M_fft)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.5 * (it / max(1, n_iters // 2))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = 2.0 * n * float(fast_conv(a_new).max())
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            s[j_start:j_end + 1] = 2.0 * seg[::-1]
            if k_star % 2 == 0:
                i_diag = k_star // 2
                if j_start <= i_diag <= j_end:
                    s[i_diag] = a[i_diag]

        # FW step: move towards e_{argmin s}
        i_min = int(np.argmin(s))
        t_lr = 2.0 / (it + 2.0)  # standard FW rate
        a_new = (1.0 - t_lr) * a + t_lr * np.zeros(n)
        a_new[i_min] = t_lr
        # Ensure non-negative
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-guided optimization, accelerated subgradient, error reduction,
    and L-BFGS polishing in a multi-phase pipeline.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine target lengths
    target_n = [2048, 4096, 6962, 8192]
    if parent is not None:
        target_n.insert(0, len(parent))
    target_n = sorted(set(min(max(x, 512), 16384) for x in target_n))[:5]

    print(f"Target lengths: {target_n}")
    print(f"Parent available: {parent is not None}")

    # ============================================================
    # Phase 1: Quick initialization search (15% time)
    # ============================================================
    p1_deadline = t_start + 0.15 * (total_budget - 20)
    print("\n=== Phase 1: Initialization ===")

    for n in target_n:
        if time.monotonic() > p1_deadline:
            break
        print(f"  n={n}")
        inits = []
        if parent is not None:
            if len(parent) == n:
                inits.append(parent.copy())
            else:
                inits.append(np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(parent)), parent))
        x = np.arange(n, dtype=np.float64) / n
        inits.append(np.ones(n) * 0.5)
        inits.append(0.3 + 0.7 * x)
        inits.append(0.3 + 0.7 * (1.0 - x))
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
        inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))

        max_inits = max(2, int((p1_deadline - time.monotonic()) / 5.0))
        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > p1_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-6:
                continue
            a_res = subgradient_descent(a0, n_iters=3000, time_limit=4.0)
            sc = compute_score(a_res)
            report(sc, a_res, f"init n={n} i={i}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Error reduction + subgradient (20% time)
    # ============================================================
    p2_deadline = t_wrap_deadline = t_start + 0.35 * (total_budget - 20)
    print("\n=== Phase 2: Error Reduction + Subgradient ===")

    n_work = len(best_a)
    a_work = best_a.copy()
    S = a_work.sum()
    if S > 1e-6:
        a_work = a_work / S

    # Error reduction
    a_er = error_reduction(a_work, n_iters=300)
    sc = compute_score(a_er)
    report(sc, a_er, f"ER n={n_work}")

    # Subgradient polish on ER result
    remaining = min(60.0, p2_deadline - time.monotonic())
    if remaining > 5:
        a_sg = subgradient_descent(a_er, n_iters=5000, time_limit=remaining)
        sc = compute_score(a_sg)
        report(sc, a_sg, f"SG-ER n={n_work}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: LP Alternating (25% time)
    # ============================================================
    p3_deadline = t_start + 0.60 * (total_budget - 20)
    print("\n=== Phase 3: LP Alternating ===")

    # Use moderate n for LP
    n_lp = min(len(best_a), 4096)
    if n_lp < 1024:
        n_lp = 2048
    a_lp = best_a.copy()
    if n_lp != len(best_a):
        a_lp = np.interp(np.linspace(0, 1, n_lp),
                         np.linspace(0, 1, len(best_a)), best_a)
    a_lp = np.maximum(a_lp, 0.0)
    S = a_lp.sum()
    if S > 1e-6:
        a_lp = a_lp / S

    print(f"  LP size: n={n_lp}")
    t0 = time.monotonic()
    a_lp_res = run_lp_alternating(a_lp, n_iters=80,
                                  time_limit=max(10.0, p3_deadline - time.monotonic() - 30))
    sc = compute_score(a_lp_res)
    print(f"  LP done in {time.monotonic()-t0:.1f}s, score={sc:.8f}")
    report(sc, a_lp_res, f"LP n={n_lp}")

    # If LP was at smaller n, try to upscale and polish
    if n_lp < len(best_a) and best_score < 1.52:
        n_full = max(len(best_a), 6962)
        a_up = np.interp(np.linspace(0, 1, n_full),
                         np.linspace(0, 1, n_lp), a_lp_res)
        a_up = np.maximum(a_up, 0.0)
        if a_up.sum() > 1e-6:
            a_up = a_up / a_up.sum()
        a_pol = subgradient_descent(a_up, n_iters=5000, time_limit=30.0)
        sc = compute_score(a_pol)
        report(sc, a_pol, f"LP+up n={n_full}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Extended subgradient + Frank-Wolfe (15% time)
    # ============================================================
    p4_deadline = t_start + 0.75 * (total_budget - 20)
    print("\n=== Phase 4: Extended Optimization ===")

    n_ext = len(best_a)
    a_ext = best_a.copy()
    S = a_ext.sum()
    if S > 1e-6:
        a_ext = a_ext / S

    remaining = min(90.0, p4_deadline - time.monotonic())
    if remaining > 10:
        a_sg2 = subgradient_descent(a_ext, n_iters=20000, time_limit=remaining * 0.5)
        sc = compute_score(a_sg2)
        report(sc, a_sg2, f"SG2 n={n_ext}")

    if time.monotonic() < p4_deadline:
        remaining_fw = min(30.0, p4_deadline - time.monotonic())
        a_fw = frank_wolfe(a_ext, n_iters=5000, time_limit=remaining_fw)
        sc = compute_score(a_fw)
        report(sc, a_fw, f"FW n={n_ext}")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: L-BFGS with increasing beta (15% time)
    # ============================================================
    p5_deadline = deadline
    print("\n=== Phase 5: L-BFGS Polish ===")

    for beta in [50, 200, 1000, 3000]:
        if time.monotonic() > p5_deadline - 10:
            break
        maxfun = 50000
        maxiter = 3000
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")

    print(f"  After Phase 5: best = {best_score:.8f}")

    # ============================================================
    # Phase 6: Final combined attempts
    # ============================================================
    print("\n=== Phase 6: Final Polish ===")

    if time.monotonic() < deadline - 30:
        # One more ER + SG round
        a_fin = error_reduction(best_a, n_iters=150)
        a_fin = subgradient_descent(a_fin, n_iters=5000, time_limit=20.0)
        sc = compute_score(a_fin)
        report(sc, a_fin, "final ER+SG")

    if time.monotonic() < deadline - 20:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=100000)
        report(sc, a_opt, "final LBFGS b=5000")

    if time.monotonic() < deadline - 15:
        # Try a few random perturbations + quick polish
        for trial in range(3):
            if time.monotonic() > deadline - 15:
                break
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() < 1e-6:
                continue
            a_pert = a_pert / a_pert.sum()
            a_pert = subgradient_descent(a_pert, n_iters=3000, time_limit=5.0)
            sc = compute_score(a_pert)
            report(sc, a_pert, f"pert#{trial}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
