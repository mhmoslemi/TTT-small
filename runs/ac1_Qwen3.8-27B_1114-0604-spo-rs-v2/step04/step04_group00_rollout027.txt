def lse_objective_and_grad(a, beta):
    """LSE-smoothed score and gradient for the full objective 2n*LSE/S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    LSE_val = c_max + np.log(exp_scaled.sum()) / beta
    
    score = 2.0 * n * LSE_val / (S * S)
    
    # Gradient of score = 2n/S^2 * dLSE/da - 4n*LSE/S^3 * S (sum of all partial derivatives of S)
    # dLSE/da_j = beta * sum_k p_k * dc_k/da_j = beta * sum_k p_k * a_{k-j}
    # = beta * g_j (from our computation above)
    # dS/da_j = 1 (for all j)
    # So grad_j = 2n/S^2 * beta * g_j - 4n*LSE/S^3 * 1
    
    n = len(a)
    a_rev = a[::-1].copy()
    M = 3 * n
    M_fft = 1 << (M - 1).bit_length()
    conv_ap = np.fft.irfft(np.fft.rfft(a_rev, M_fft) * np.fft.rfft(p, M_fft), M_fft)
    g = conv_ap[n-1 : 2*n-1].copy()
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n * beta * g) / S2 - (4.0 * n * LSE_val) / S3
    
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_objective_and_grad(x, beta)
    
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_top_m(a, n_iters=5000, top_m=20, step_scale=0.1):
    """Subgradient with top-M peaks and Polyak step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1)  # theoretical lower bound
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if sc - lb < 1e-8:
            break
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
            c_top = c[top_idx]
            weights = (c_top - c_top.min() + 1e-10)
            weights /= weights.sum()
        
        # Subgradient: for each peak k, the subgradient wrt a is 2*a_rev_window
        sg = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = int(idx)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]
        
        # Gradient of score = 2n/S^2 * (sg - 2*M_val/S * S_vec)
        # Actually: score = 2n * M / S^2
        # d(score)/da_j = 2n * (dM/da_j * S^2 - M * 2S * 1) / S^4
        # = 2n * (dM/da_j / S^2 - 2M / S^3)
        # dM/da_j = sg_j (the subgradient of max at the active peaks)
        # But more precisely, the subgradient of max(c) is in the convex hull of
        # the gradients of the active c(k).
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Polyak step size
        gn2 = float(np.dot(grad, grad))
        step = min(step_scale, (sc - lb) / (gn2 + 1e-30))
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def fienup_flat(a, n_iters=200, clip=0.5):
    """Simplified Fienup: flatten the power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        thresh = P_mean * (1.2 + 0.3 * it / n_iters)
        
        above = P > thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = (thresh / P[above]) ** (clip * 0.5)
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def coordinate_descent_peaks(a, n_sweeps=3, n_top=40):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        cur_sc = compute_score(a)
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_val = a_old
            best_local = cur_sc
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                cv = a_old * factor
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_tmp = a / S_tmp
                sc = compute_score(a_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
