```python
"""
LSE Continuation + Adaptive Subgradient + Multi-start optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
1. Fine-grained LSE beta continuation (geometric schedule)
2. Adaptive top-M subgradient with Polyak step size
3. Fienup spectral shaping interleave for escaping local optima
4. Multi-length exploration around parent length
5. Coordinate descent on peak contributors for final polish
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    # Gradient: dL/dx[i] = 2 * sum_j x[j] * p[i+j]
    p_rev = p[::-1]
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    PR = np.fft.rfft(p_rev, M_fft)
    conv_result = np.fft.irfft(A * PR, M_fft)[:M]
    g = conv_result[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=None, t_start=None):
    """Run L-BFGS-B on LSE-smoothed objective with optional time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    # Time-aware maxiter
    if time_limit is not None and t_start is not None:
        elapsed = time.monotonic() - t_start
        remaining = time_limit - elapsed
        if remaining < 2:
            return compute_score(a0), a0
        # Estimate: each fun evaluation ~ n*log(n) / 1e7 seconds (rough)
        est_per_eval = n * np.log2(n) / 5e7
        est_per_iter = 15 * est_per_eval
        maxiter = min(maxiter, max(50, int(remaining * 0.8 / max(est_per_iter, 1e-6))))
        maxfun = min(maxfun, maxiter * 20)

    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=5000, top_m=20, step_init=0.03, decay=0.9998,
                      time_limit=None, t_start=None):
    """Subgradient with top-M peaks and Polyak step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound

    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step_p = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, step_p)
        step = min(step, 0.05)
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_shapes(a, n_iters=200, target='uniform', strength=0.4):
    """Fienup-style spectral shaping to flatten the power spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if target == 'uniform':
            # Reduce top power bins toward mean
            thresh = P_mean * (1.2 + 0.8 * (1.0 - progress))
        else:
            thresh = P_mean * 1.5

        above = P > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        red = np.sqrt(np.maximum(thresh / P[above], 0.01))
        scale[above] = (1.0 - strength) + strength * red

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_descent_peak(a, n_sweeps=5, n_top=100, time_limit=None, t_start=None):
    """Coordinate descent optimizing top contributors to the max peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            if time_limit is not None and t_start is not None:
                if time.monotonic() - t_start > time_limit:
                    break
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j
            best_local_sc = cur_sc
            for f in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]:
                cv = max(0.0, min(1000.0, a_old_j * f))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_continuation(a0, betas, maxiter_per_beta=3000, time_limit=None, t_start=None):
    """
    Run LSE continuation: optimize with increasing beta values.
    Returns best (score, sequence) found.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        sc, a_opt = run_lbfgs(a, beta, maxiter=maxiter_per_beta,
                              time_limit=time_limit, t_start=t_start)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a = a_opt
        else:
            a = a_opt  # still use as warm start

        print(f"    beta={beta:>6.0f}: score={sc:.8f}")

    return best_sc, best_a


def make_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    # Chirp-like
    inits.append(np.abs(np.exp(2j * np.pi * x ** 2)) * 0.5 + 0.3)
    # Two-tone
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.3 * np.cos(6.0 * np.pi * x), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    # Choose primary length
    n_primary = len(parent) if parent is not None else 7000
    print(f"Parent length: {n_primary if parent is not None else 'none'}")
    print(f"Primary length: {n_primary}")

    # ==================================================================
    # Phase 1: LSE Continuation (50% of time)
    # ==================================================================
    phase1_end = t_start + 0.50 * (deadline - t_start)
    print(f"\n=== Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s) ===")

    # Fine geometric beta schedule
    betas = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240, 20000, 40000]

    if parent is not None and len(parent) == n_primary:
        a_start = parent.copy()
    else:
        inits = make_initializations(n_primary, rng, parent)
        a_start = inits[0]  # parent interpolation or first init

    a_start = np.maximum(a_start, 0.0)
    if a_start.sum() < 1e-12:
        a_start = np.ones(n_primary)
    a_start = a_start / a_start.sum()

    sc, a_opt = lse_continuation(a_start, betas, maxiter_per_beta=4000,
                                  time_limit=phase1_end, t_start=t_start)
    update_best(sc, a_opt, "LSE-CONT")

    # ==================================================================
    # Phase 2: Subgradient refinement (15% of time)
    # ==================================================================
    phase2_end = t_start + 0.65 * (deadline - t_start)
    print(f"\n=== Phase 2: Subgradient (until t={phase2_end - t_start:.0f}s) ===")

    for top_m in [10, 20, 50]:
        if time.monotonic() > phase2_end:
            break
        a_sg = top_m_subgradient(best_a, n_iters=5000, top_m=top_m,
                                 step_init=0.03, decay=0.9999,
                                 time_limit=phase2_end, t_start=t_start)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")
        # Follow up with L-BFGS
        if time.monotonic() < phase2_end - 10:
            sc2, a2 = run_lbfgs(a_sg, 5000, maxiter=3000, maxfun=50000)
            update_best(sc2, a2, f"SG+LBFGS top_m={top_m}")

    # ==================================================================
    # Phase 3: Fienup + cycling (15% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (deadline - t_start)
    print(f"\n=== Phase 3: Fienup Cycling (until t={phase3_end - t_start:.0f}s) ===")

    for strength in [0.3, 0.5, 0.7]:
        if time.monotonic() > phase3_end:
            break
        a_f = fienup_shapes(best_a, n_iters=200, strength=strength)
        sc = compute_score(a_f)
        update_best(sc, a_f, f"FIENUP s={strength}")
        if time.monotonic() < phase3_end - 10:
            sc2, a2 = run_lbfgs(a_f, 2000, maxiter=2000, maxfun=30000)
            update_best(sc2, a2, f"FIENUP+LBFGS s={strength}")

    # Perturbed restart
    if time.monotonic() < phase3_end:
        a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_primary))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        a_pert = a_pert / a_pert.sum()
        a_f2 = fienup_shapes(a_pert, n_iters=150, strength=0.5)
        sc = compute_score(a_f2)
        update_best(sc, a_f2, "FIENUP-pert")
        if time.monotonic() < phase3_end - 10:
            sc2, a2 = run_lbfgs(a_f2, 3000, maxiter=2000, maxfun=30000)
            update_best(sc2, a2, "FIENUP-pert+LBFGS")

    # ==================================================================
    # Phase 4: Multi-length exploration (10% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (deadline - t_start)
    print(f"\n=== Phase 4: Multi-length (until t={phase4_end - t_start:.0f}s) ===")

    alt_lengths = []
    if n_primary > 1000:
        alt_lengths.append(n_primary + 500)
        alt_lengths.append(n_primary + 1000)
        alt_lengths.append(n_primary - 500)

    for n_alt in alt_lengths:
        if time.monotonic() > phase4_end:
            break
        if n_alt < 256:
            continue
        # Interpolate best solution to new length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_alt)
        a_alt = np.interp(x_new, x_old, best_a)
        a_alt = np.maximum(a_alt, 0.01)
        a_alt = a_alt / a_alt.sum()

        # Quick LSE at a few betas
        sc, a_opt = lse_continuation(a_alt, [50, 200, 1000, 5000],
                                      maxiter_per_beta=2000,
                                      time_limit=phase4_end, t_start=t_start)
        update_best(sc, a_opt, f"MULTI n={n_alt}")

    # ==================================================================
    # Phase 5: Final polish (remaining 5%)
    # ==================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Very high beta LSE
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "FINAL-LBFGS-hi")

    # Coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=4, n_top=100,
                                       time_limit=deadline, t_start=t_start)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD-DESCENT")
        if time.monotonic() < deadline - 10:
            sc2, a2 = run_lbfgs(a_cd, 10000, maxiter=2000, maxfun=30000)
            update_best(sc2, a2, "CD+LBFGS")

    # One more subgradient pass with many peaks
    if time.monotonic() < deadline - 10:
        a_sg = top_m_subgradient(best_a, n_iters=3000, top_m=100,
                                 step_init=0.02, decay=0.9999,
                                 time_limit=deadline, t_start=t_start)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "FINAL-SG")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
