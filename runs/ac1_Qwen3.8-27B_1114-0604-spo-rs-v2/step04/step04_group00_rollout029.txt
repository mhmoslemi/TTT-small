import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def conv_fienup(a, n_iters=300, n_peaks=5):
    """
    Convolution-domain Fienup: directly reduce peaks in conv(a,a)
    by modifying A^2 in the frequency domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        B = A * A  # This gives DFT of circular conv, but with zero-padding it's linear
        b = np.fft.irfft(B, M_fft)[:M]  # This is the linear convolution
        
        # Find peaks to reduce
        b_max = float(b.max())
        # Target: reduce top peaks toward the mean
        b_mean = float(b.mean())
        
        # Create modified b
        b_mod = b.copy()
        # Reduce values above threshold
        threshold = b_mean + 0.5 * (b_max - b_mean)  # Reduce top 50% of the excess
        above = b_mod > threshold
        if above.any():
            # Soft reduction
            b_mod[above] = threshold + (b_mod[above] - threshold) * 0.7
        
        B_mod = np.fft.rfft(b_mod, M_fft)
        
        # Take square root with phase alignment
        # Phase of A_prev tells us which branch of sqrt to use
        phase_ref = np.angle(A)
        magnitude_new = np.sqrt(np.abs(B_mod))
        # Phase: we want A_new such that A_new^2 = B_mod
        # angle(A_new) = angle(B_mod) / 2, but aligned with previous A
        angle_B = np.angle(B_mod)
        angle_A = angle_B / 2.0
        # Align with previous phase (choose the sign to be closest)
        # If angle difference > pi/2, flip by pi
        diff = np.angle(np.exp(1j * (angle_A - phase_ref)))
        flip = diff > np.pi / 2
        angle_A[flip] += np.pi
        
        A_new = magnitude_new * np.exp(1j * angle_A)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

    # c[k] = sum_j a[j] * a[k-j], so dc[k]/da[j] = a[k-j] + a[j] * (if j appears twice, i.e., k=2j)
    # More carefully: c[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * a[k-j]
    # dc[k]/da[j] = a[k-j] if j is in the valid range for c[k]
    # (each a[j] appears exactly once in the sum for c[k] unless k=2j where it appears as a[j]*a[j])
    # Wait no: in c[k] = sum_j a[j]*a[k-j], the term a[j] appears as a[j]*a[k-j] when the first index is j,
    # and also when a[k-j] is the j-th element, i.e., k-j=n-1-j... no.
    # Let me be very careful:
    # c[k] = sum_{j=0}^{n-1} a[j] * a[k-j] where a[m] = 0 for m outside [0,n-1]
    # dc[k]/da[j] = a[k-j] + a[j] (the second term appears because a[j] also appears as a[k-(k-j)] = a[j] when we differentiate the a[k-j] term)

    # c[k] = sum_i a[i] * a[k-i]
    # dc[k]/da[j] = (d/dj) sum_i a[i]*a[k-i] = a[k-j] + a[j] if the index j appears in both positions

    # The second term a[k-i] with k-i=j gives i=k-j, so a[k-j]*... wait
    # dc[k]/da[j] = sum_i [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
    # = a[k-j] + a[k-j] ... no
    # = a[k-j] (from first term, i=j) + a[k-j] (from second term, k-i=j means i=k-j, so a[k-j])
    # Wait: second term: a[i]*delta(k-i-j), this is non-zero when i = k-j, giving a[k-j]
    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (when both are in valid range)
    # EXCEPT when j = k-j (i.e., k = 2j), in which case it's just a[j] (the same term)
    # Hmm, actually no. Let me redo:
    # c[k] = sum_i a[i] * a[k-i]
    # dc[k]/da[j] = sum_i [d(a[i])/da[j] * a[k-i] + a[i] * d(a[k-i])/da[j]]
    # = sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
    # = a[k-j] + a[k-j]  (first sum: i=j, second sum: k-i=j => i=k-j)
    # = 2*a[k-j] when j != k-j (i.e., k != 2j)
    # = a[j] when k = 2j (since both terms give the same a[j])
    # Wait no, when k=2j: first term gives a[2j-j] = a[j], second term gives a[2j-j] = a[j], so total = 2*a[j]
    # Actually it's always 2*a[k-j]! The two terms are always the same value.
    # So dc[k]/da[j] = 2*a[k-j] for all valid j, k.
    
    # Great! So the gradient is:
    # dL/da[j] = (1/beta) * sum_k p[k] * 2*a[k-j]
    # = (2/beta) * sum_k p[k] * a[k-j]
    # = (2/beta) * (p * a_rev)[j] where a_rev[m] = a[n-1-m] and * is appropriate convolution

    # This is the cross-correlation (p ⋆ a)[j] = sum_k p[k] * a[k-j]

    # Or equivalently: ifft(fft(p) * conj(fft(a))) ... let me use the standard formula
    # corr(p, a)[j] = ifft(conj(FFT(p)) * FFT(a))[j] ... no
    # Standard: (f * g)[n] = ifft(FFT(f) * FFT(g))
    # Correlation: (f ⋆ g)[n] = sum_k f[k+n] * conj(g[k]) = ifft(FFT(f) * conj(FFT(g)))
    # We want sum_k p[k] * a[k-j] = sum_k p[k] * a[k-j]
    # Let m = k-j: sum_m p[m+j] * a[m] = (p_shifted_j ⋆ a)[0] ... this is getting circular
    
    # Simplest: just compute it using FFT with zero-padding
    # We want g[j] = sum_{k=0}^{2n-2} p[k] * a[k-j] for j=0..n-1
    # Note a[k-j] is non-zero only when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    # So g[j] = sum_{k=j}^{j+n-1} p[k] * a[k-j]
    # Let's define: g[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is a "sliding dot product" of a with shifted versions of p
    # We can compute this as: g = ifft(FFT(p_padded) * FFT(a_padded)) [appropriate slice]
    # where p_padded has p at positions 0..2n-2 and a_padded has a at positions 0..n-1
    # The convolution (p * a)[j] = sum_k p[k] * a[j-k] ... not quite what we want
    # We want sum_m p[j+m] * a[m] = sum_m a[m] * p[j+m]
    # = (a * p_rev)[j] where p_rev is p reversed... 
    
    # OK let me just use the approach from the previous code which uses:
    # conv of rev_a and p
    rev_a = a[::-1]
    # (rev_a * p)[j + n - 1] = sum_k rev_a[k] * p[j+n-1-k]
    # rev_a[k] = a[n-1-k]
    # = sum_k a[n-1-k] * p[j+n-1-k]
    # Let m = n-1-k: = sum_m a[m] * p[j+m]
    # = g[j] ✓
    
    M2 = 3 * n - 2  # length of rev_a * p
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    
    g = (2.0 / beta) * conv_res
    # This is dL/da
    
    # Now the full score gradient:
    # score = 2n * L / S^2
    # d(score)/da[j] = 2n * (dL/da[j] * S^2 - L * 2*S) / S^4
    # = 2n * (dL/da[j] / S^2 - 2*L / S^3)
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def objective_and_grad(x, beta):
    """Top-level function for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_gradient(a, n_iters=5000, beta=200, lr_init=0.01, momentum=0.9):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed objective.
    With adaptive learning rate and projection onto a >= 0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # velocity
    lr = lr_init
    
    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        
        # Nesterov step
        v = momentum * v + lr * grad
        a_new = a - v
        
        # Project onto a >= 0
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a
        else:
            a_new = a_new / S_new
        
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Backtracking if no improvement
        if sc_new > sc * 1.001:
            lr *= 0.9
            a = a  # don't update
        else:
            a = a_new
            # Slowly increase lr back
            lr = min(lr * 1.01, lr_init)
    
    return best_a


def lse_anneal_lbfgs(a0, betas=None, maxiter_each=5000):
    """
    LSE annealing: start with low beta (smooth), gradually increase.
    Each step runs L-BFGS with the current beta.
    """
    if betas is None:
        betas = [10, 50, 100, 200, 500, 1000, 2000, 5000]
    
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for beta in betas:
        sc, a_new = run_lbfgs(a, beta, maxiter=maxiter_each, maxfun=maxiter_each*20)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new  # continue from this even if not best (for annealing)
    
    return best_a


def proposed_phase_retrieval(a, n_iters=500, alpha=0.5):
    """
    Phase retrieval approach: alternate between
    (1) modifying the convolution to be flatter
    (2) recovering a from the modified convolution via sqrt in frequency domain
    
    This is the key novel approach vs the previous code.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        # Get current convolution
        A = np.fft.rfft(a, M_fft)
        B = A * A
        c = np.fft.irfft(B, M_fft)[:M]
        
        # Modify c to be flatter
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        # Adaptive reduction strength
        strength = alpha * (0.5 + 0.5 * progress)
        
        # Reduce peaks: soft threshold
        c_mod = c.copy()
        # Target: bring everything closer to the mean
        excess = np.maximum(c_mod - c_mean, 0)
        c_mod = c_mod - strength * excess
        
        # Ensure the sum is preserved (important for DC component)
        c_mod = np.maximum(c_mod, 0.0)
        
        # Frequency domain: take square root
        C_mod = np.fft.rfft(c_mod, M_fft)
        mag_C = np.abs(C_mod)
        phase_C = np.angle(C_mod)
        
        # A_new = sqrt(C_mod), with phase alignment to previous A
        mag_A = np.sqrt(mag_C)
        phase_A = phase_C / 2.0
        
        # Phase alignment: make A_new close to A_old in phase
        phase_diff = np.angle(np.exp(1j * (phase_A - np.angle(A))))
        flip_mask = phase_diff > np.pi / 2
        phase_A[flip_mask] += np.pi
        
        A_new = mag_A * np.exp(1j * phase_A)
        
        # Recover a
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def multi_peak_subgradient(a, n_iters=5000, top_k=20, beta_smooth=50):
    """
    Subgradient method using top-k peaks with smoothing.
    Uses LSE gradient for better direction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.02
    
    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta_smooth)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size approximation
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999

    return best_a

Wait, I need to check a few things:
1. The `objective_and_grad` function needs to be top-level (no closures)
2. No lambda functions
3. All helper functions at top level

Let me also reconsider the phase retrieval approach. The key issue is: when we compute A^2 and take the square root, we need to be careful about:
- The DC bin: A[0] should be real and positive (since a >= 0, sum(a) >= 0)

"""
Convolution-domain phase retrieval + LSE annealing + Nesterov gradient optimizer.
Directly targets flatness of self-convolution via A^2 (not |A|^2) phase manipulation,
combined with careful LSE-smoothed L-BFGS annealing and momentum-based methods.

Key novelty: operates on A^2 in frequency domain (the DFT of the convolution)
rather than |A|^2 (the DFT of the autocorrelation), directly targeting the objective.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed objective and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da[j] = (2/beta) * sum_k p[k] * a[k-j]
    # Compute via FFT: (rev_a * p)[j+n-1] = sum_m a[m] * p[j+m]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = (2.0 / beta) * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def objective_and_grad(x, beta):
    """Top-level function for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def conv_phase_retrieval(a, n_iters=400, alpha=0.5, mode='soft'):
    """
    Convolution-domain phase retrieval: directly modify conv(a,a) to be flatter,
    then recover a via sqrt of DFT in frequency domain.
    
    This targets A^2 (DFT of convolution) rather than |A|^2 (DFT of autocorrelation).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        B = A * A
        c = np.fft.irfft(B, M_fft)[:M]

        c_max = float(c.max())
        c_median = float(np.median(c))

        # Adaptive peak reduction
        if mode == 'soft':
            # Soft: reduce excess above median proportionally
            excess = np.maximum(c - c_median, 0.0)
            strength = alpha * (0.3 + 0.7 * progress)
            c_mod = c - strength * excess
        elif mode == 'hard':
            # Hard: clip top peaks to a threshold
            thresh = c_median + 0.5 * (c_max - c_median) * (1.0 - 0.5 * progress)
            c_mod = np.minimum(c, thresh)
        else:
            # Quantile-based: reduce top fraction
            top_frac = 0.3 + 0.2 * progress
            thresh = np.quantile(c, 1.0 - top_frac)
            c_mod = c.copy()
            above = c_mod > thresh
            c_mod[above] = thresh + (c_mod[above] - thresh) * 0.5

        c_mod = np.maximum(c_mod, 0.0)

        # Frequency domain: recover A from modified convolution
        C_mod = np.fft.rfft(c_mod, M_fft)
        mag_C = np.abs(C_mod)
        phase_C = np.angle(C_mod)

        mag_A = np.sqrt(mag_C)
        phase_A = phase_C / 2.0

        # Phase alignment with previous A
        phase_ref = np.angle(A)
        phase_diff = np.angle(np.exp(1j * (phase_A - phase_ref)))
        flip_mask = phase_diff > np.pi / 2
        phase_A = phase_A.copy()
        phase_A[flip_mask] += np.pi

        A_new = mag_A * np.exp(1j * phase_A)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def conv_phase_retrieval_multi(a, n_rounds=3, iters_per_round=200):
    """
    Multi-round phase retrieval with varying alpha and mode.
    Each round uses a different reduction strategy.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    configs = [
        (0.4, 'soft'),
        (0.6, 'soft'),
        (0.3, 'hard'),
        (0.5, 'quantile'),
        (0.3, 'soft'),
        (0.2, 'soft'),
    ]

    for rnd in range(n_rounds):
        cfg_idx = rnd % len(configs)
        alpha, mode = configs[cfg_idx]
        iters = max(50, iters_per_round - rnd * 30)
        a = conv_phase_retrieval(a, n_iters=iters, alpha=alpha, mode=mode)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_proximal(a, n_iters=4000, beta=200, lr_init=0.003, momentum=0.85):
    """
    Nesterov-accelerated proximal gradient with adaptive LR.
    Works on the LSE-smoothed objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    lr = lr_init
    stalled = 0

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)

        # Nesterov: evaluate gradient at lookahead point
        v_new = momentum * v
        a_lookahead = a + v_new
        a_lookahead = np.maximum(a_lookahead, 0.0)
        S_la = float(a_lookahead.sum())
        if S_la < 1e-12:
            a_lookahead = a
        else:
            a_lookahead = a_lookahead / S_la

        sc_la, grad_la = lse_score_grad(a_lookahead, beta)
        v_new = momentum * v + lr * grad_la
        a_candidate = a - v_new
        a_candidate = np.maximum(a_candidate, 0.0)
        S_c = float(a_candidate.sum())
        if S_c < 1e-12:
            a_candidate = a
        else:
            a_candidate = a_candidate / S_c

        sc_c = compute_score(a_candidate)
        if sc_c < best_sc:
            best_sc = sc_c
            best_a = a_candidate.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled > 200:
            lr *= 0.95
            stalled = 0

        a = a_candidate
        v = v_new
        # Slightly increase LR over time if making progress
        if it % 500 == 0 and it > 0:
            lr = min(lr * 1.02, lr_init * 2.0)

    return best_a


def hybrid_cycle(a, n_cycles=4):
    """
    Alternate between phase retrieval and L-BFGS to escape local optima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for cyc in range(n_cycles):
        # Phase retrieval
        a = conv_phase_retrieval(a, n_iters=150, alpha=0.4, mode='soft')
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # L-BFGS polish
        beta = 100 + 300 * cyc
        sc2, a2 = run_lbfgs(a, beta, maxiter=2000, maxfun=40000)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            a = a2

    return best_a


def diverse_initializations(n, rng, parent):
    """Generate diverse starting points."""
    x = np.arange(n, dtype=np.float64) / n
    inits = []

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        # Perturbed parent
        p = parent / parent.sum()
        for scale in [0.02, 0.05]:
            pert = p * (1.0 + scale * rng.standard_normal(n))
            pert = np.clip(pert, 0.0, 1000.0)
            inits.append(pert / pert.sum())

    # Structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.4 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * x)) ** 2)
    inits.append(np.clip(0.5 + 0.15 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.2 + 0.8 * (1.0 - np.abs(2.0 * x - 1.0)) ** 2)

    return inits

        report(sc, a_opt, "POLISH-8k")

    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 15000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "POLISH-15k")

    # ========== Final Output ==========
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
