"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) BB subgradient with KL mirror descent and adaptive active sets,
(2) LSE-smoothed L-BFGS-B with continuation, (3) spectral flattening moves,
(4) coordinate-wise line search, and (5) diversification restarts.

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / S


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, np.maximum(a, 0)).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    return np.maximum(d, 0.0), S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def active_weights(d, k, threshold=0.9):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.ones(m) / m
    idx_active = np.where(d >= threshold * dmax)[0]
    if len(idx_active) == 0:
        idx_active = np.argmax(d)
        idx_active = np.array([idx_active])
    if len(idx_active) > k:
        idx_top = np.argpartition(d, m - k)[m - k:]
        idx_active = idx_top
    w = np.zeros(m)
    w[idx_active] = 1.0
    return w / float(len(idx_active))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    d, _ = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    if beta < 1:
        beta = 1.0
    z = beta * (d - dmax)
    z = np.clip(z, -700, 700)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (4.0 * n * h) / (S * S) - (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    score = 2.0 * L
    return float(score), grad


def bb_subgradient_optimize(a, deadline, verbose=False):
    """BB subgradient with KL mirror descent and adaptive active set."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    s_prev = None
    y_prev = None
    alpha = 1.0 / n
    momentum = np.zeros(n)
    mu = 0.7
    k_active = max(3, n // 256)
    
    it = 0
    stall_count = 0
    stall_patience = 50
    
    while time.monotonic() < deadline:
        it += 1
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax):
            break
        
        cur_sc = 2.0 * dmax
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            stall_count = 0
            if verbose and it % 200 == 0:
                print("  BB-subgrad it={}: score={:.8f}".format(it, best_sc))
        
        if stall_count > stall_patience:
            # Diversification: add noise
            noise = np.exp(0.1 * np.random.standard_normal(n))
            a = normalize_a(a * noise)
            stall_count = 0
            s_prev = None
            y_prev = None
            continue
        
        w = active_weights(d, k_active, threshold=0.85)
        g = corr_pair(a, w)
        g = g / max(float(np.max(g)), 1e-12)
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * g
        g_use = momentum
        
        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_prev, y_prev))
            yy = float(np.dot(y_prev, y_prev))
            ss = float(np.dot(s_prev, s_prev))
            if yy > 1e-12 and sy > 1e-12:
                alpha_bb1 = sy / yy
                alpha_bb2 = ss / sy
                alpha_new = 0.5 * (alpha_bb1 + alpha_bb2)
                alpha = max(1e-8, min(alpha_new, 1.0))
            else:
                alpha *= 0.9
        
        # KL mirror descent update: a <- a * exp(-alpha * g_projected)
        g_mean = float(np.mean(g_use))
        g_centered = g_use - g_mean
        step = alpha * g_centered
        step = np.clip(step, -2.0, 2.0)
        a_new = a * np.exp(-step)
        a_new = normalize_a(a_new)
        
        s_curr = a_new - a
        y_curr = corr_pair(a_new, active_weights(d_from_a(a_new)[0], k_active, 0.85)) - g
        s_prev = s_curr
        y_prev = y_curr
        a = a_new
        
        # Adapt k_active
        if it % 100 == 0:
            d_check, _ = d_from_a(a)
            dmax_c = float(np.max(d_check))
            frac_active = float(np.sum(d_check > 0.8 * dmax_c)) / d_check.size
            if frac_active > 0.1:
                k_active = max(1, int(frac_active * d_check.size * 0.5))
            elif frac_active < 0.01:
                k_active = min(d_check.size, k_active * 2)
    
    return float(best_sc), best_a


def lse_lbfgs_optimize(a, beta, deadline, maxfun=None):
    """LSE-smoothed L-BFGS-B optimization."""
    a = normalize_a(a)
    n = len(a)
    if maxfun is None:
        maxfun = max(100, min(2000, int((deadline - time.monotonic()) * 20)))
    
    def obj(x):
        val, grad = lse_score_grad(x, beta)
        return val, grad
    
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": max(30, maxfun // 3),
            "maxfun": maxfun, "maxls": 40}
    
    try:
        res = minimize(obj, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = normalize_a(res.x)
    except Exception:
        a_out = a
    
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def spectral_flat_move(a, n_iters=20, deadline=None):
    """Iteratively flatten the power spectrum to reduce conv peak."""
    a = normalize_a(a)
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for _ in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_mean = float(np.mean(P))
        
        # Soft clipping of peaks, boosting of valleys
        clip_hi = 0.7
        clip_lo = 0.3
        P_mod = np.where(P > P_mean, P_mean + (P - P_mean) * clip_hi,
                        P + (P_mean - P) * clip_lo)
        P_mod = np.maximum(P_mod, 1e-10)
        
        phase = np.angle(A)
        A_new = np.sqrt(P_mod) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = normalize_a(a_new)
        
        sc = exact_score_a(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try gentler version
            P_mod2 = np.where(P > P_mean, P_mean + (P - P_mean) * 0.9,
                            P + (P_mean - P) * 0.1)
            A_new2 = np.sqrt(np.maximum(P_mod2, 1e-10)) * np.exp(1j * phase)
            a_new2 = np.fft.irfft(A_new2, N)[:n]
            a_new2 = np.maximum(a_new2, 0)
            a_new2 = normalize_a(a_new2)
            sc2 = exact_score_a(a_new2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break
    
    return float(best_sc), best_a


def coordinate_line_search(a, deadline, n_passes=5):
    """Coordinate-wise line search to reduce the peak."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        d, S = d_from_a(a)
        dmax = float(np.max(d))
        
        # Find which coordinates most affect the peak
        active_idx = np.where(d >= 0.9 * dmax)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(d)])
        
        # For each active lag k, the gradient component is a[k-i]
        # Reducing a[i] where a[k-i] is large helps most
        for k in active_idx[:5]:
            if time.monotonic() > deadline:
                break
            # Gradient w.r.t. a from this lag: g[i] = a[k-i] (circular-like)
            g = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] = a[j]
            
            # Try reducing elements with high gradient
            top_idx = np.argsort(-g)[:max(3, n // 100)]
            for i in top_idx:
                if a[i] < 1e-10:
                    continue
                for frac in (0.5, 0.7, 0.85, 0.95):
                    a_try = a.copy()
                    a_try[i] *= frac
                    a_try = normalize_a(a_try)
                    sc = exact_score_a(a_try)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_try.copy()
                        a = a_try
                        break
    
    return float(best_sc), best_a


def perturb_and_optimize(a, rng, deadline, strength=0.3):
    """Perturb the sequence and do a quick optimization."""
    n = len(a)
    mode = int(rng.integers(0, 4))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * a + t * np.ones(n) / n
    elif mode == 2:
        # Block shuffle
        nb = int(rng.integers(4, 20))
        perm = rng.permutation(nb)
        per = n // nb
        q = np.zeros(n)
        for i in range(nb):
            start = i * per
            end = start + per if i < nb - 1 else n
            pstart = perm[i] * per
            pend = pstart + per if perm[i] < nb - 1 else n
            q[start:end] = a[pstart:pend]
        if nb > 1:
            perm2 = rng.permutation(nb)
            for i in range(nb):
                start = i * per
                end = start + per if i < nb - 1 else n
                pstart = perm2[i] * per
                pend = pstart + per if perm2[i] < nb - 1 else n
                q[start:end] = q[pstart:pend]
    else:
        # Smoothed random
        k = int(rng.integers(3, 31))
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * (k // 4 + 1) ** 2))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode='same')
        q = a * np.exp(0.2 * noise)
    
    q = normalize_a(q)
    
    # Quick BB subgradient
    sub_dead = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.4))
    sc, q = bb_subgradient_optimize(q, sub_dead)
    
    # Quick LSE
    rem = deadline - time.monotonic()
    if rem > 5:
        sc2, q2 = lse_lbfgs_optimize(q, 5000.0, time.monotonic() + min(rem * 0.6, 30.0), maxfun=200)
        if sc2 < sc:
            sc, q = sc2, q2
    
    return sc, q


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines BB subgradient, LSE+L-BFGS-B, spectral moves, and coordinate search.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        pass
    
    if parent is not None:
        n = min(int(parent.size), 12288)
        a = normalize_a(resample_sequence(parent, n))
    else:
        n = 8192
        a = np.ones(n) / n
    
    print("Starting: n={}, initial_score={:.8f}".format(n, exact_score_a(a)))
    
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    # Phase 1: BB Subgradient (40% of time)
    p1_dead = t_start + 0.40 * (deadline - t_start)
    print("Phase 1: BB Subgradient (until t={:.0f}s)".format(p1_dead - t_start))
    sc, a_new = bb_subgradient_optimize(a, p1_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a_new.copy()
        print("Phase 1 best: {:.8f}".format(best_sc))
    a = normalize_a(a_new)
    
    # Phase 2: LSE + L-BFGS-B with continuation (30% of time)
    p2_dead = t_start + 0.70 * (deadline - t_start)
    print("Phase 2: LSE+L-BFGS-B (until t={:.0f}s)".format(p2_dead - t_start))
    for beta in (100, 300, 1000, 3000, 5000):
        if time.monotonic() > p2_dead - 10:
            break
        sub_dead = min(p2_dead, time.monotonic() + 60.0)
        mf = max(50, int((sub_dead - time.monotonic()) * 15))
        sc, a_new = lse_lbfgs_optimize(a, float(beta), sub_dead, maxfun=mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print("  LSE beta={}: {:.8f}".format(beta, best_sc))
        a = normalize_a(a_new)
        # Interleave with BB subgradient
        if time.monotonic() < p2_dead - 10:
            sub_dead2 = min(p2_dead, time.monotonic() + 30.0)
            sc2, a2 = bb_subgradient_optimize(a, sub_dead2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
                print("  BB subgrad: {:.8f}".format(best_sc))
            a = normalize_a(a2)
    
    # Phase 3: Spectral moves + coordinate search (15% of time)
    p3_dead = t_start + 0.85 * (deadline - t_start)
    print("Phase 3: Spectral + Coordinate (until t={:.0f}s)".format(p3_dead - t_start))
    
    sc, a_spec = spectral_flat_move(a, n_iters=50, deadline=p3_dead)
    if sc < best_sc:
        best_sc = sc
        best_a = a_spec.copy()
        print("  Spectral: {:.8f}".format(best_sc))
    a = normalize_a(a_spec)
    
    if time.monotonic() < p3_dead - 5:
        sc, a_coord = coordinate_line_search(a, p3_dead, n_passes=3)
        if sc < best_sc:
            best_sc = sc
            best_a = a_coord.copy()
            print("  Coordinate: {:.8f}".format(best_sc))
        a = normalize_a(a_coord)
    
    # Phase 4: Diversification restarts (15% of time)
    p4_dead = deadline
    print("Phase 4: Diversification (until t={:.0f}s)".format(p4_dead - t_start))
    
    n_restarts = 0
    while time.monotonic() < p4_dead - 30 and n_restarts < 12:
        left = p4_dead - 30.0 - time.monotonic()
        if left < 30:
            break
        chunk = left / max(1, (12 - n_restarts))
        chunk_dead = time.monotonic() + chunk
        
        p = normalize_a(best_a)
        sc_r, a_r = perturb_and_optimize(p, rng, chunk_dead, strength=0.25)
        if sc_r < best_sc:
            best_sc = sc_r
            best_a = a_r.copy()
            print("  Restart {}: {:.8f}".format(n_restarts + 1, best_sc))
        n_restarts += 1
    
    # Phase 5: Final polish (last 30 seconds)
    if time.monotonic() < deadline - 5:
        print("Phase 5: Final polish")
        sc, a_f = bb_subgradient_optimize(best_a, deadline - 10)
        if sc < best_sc:
            best_sc = sc
            best_a = a_f.copy()
        
        if time.monotonic() < deadline - 10:
            sc, a_f = lse_lbfgs_optimize(best_a, 10000.0, deadline - 5, maxfun=300)
            if sc < best_sc:
                best_sc = sc
                best_a = a_f.copy()
                print("  Final LSE: {:.8f}".format(best_sc))
    
    # Final output preparation
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc
    
    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
