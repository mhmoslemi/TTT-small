def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * float(c.max()) / (S * S))


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / max(1, a.size))
    return a * (target / S)


def _resample(a, n_new):

    m = a.size
    if m == 0 or n_new <= 0:
        return np.ones(n_new)
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)

        return 1e10, np.zeros(n)
    c = _self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    if not np.isfinite(dmax) or dmax <= 0:
        return 2.0 * dmax + pen * (S - 1.0) ** 2, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -400.0, 0.0)

    w = e / se

    # Subgradient: 2n*(h_k/S - M*a_k/S^2) ... move opposite

    q = a * np.exp(-eta * d / dmax)

        kernel /= kernel.sum()
        noise = np.convolve(noise, kernel, mode='same')
        noise -= noise.mean()
        log_a += strength * 0.5 * noise
        q = np.exp(log_a - log_a.max())

    out = _normalize(best_seq, 1.0)
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    result = [float(x) for x in out]

    # Verify with evaluate_sequence
    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(result), time.monotonic() - t0))
    return result
