from scipy.optimize import linprog
import time

def solve_lp(a, M):
    """Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k"""
    n = len(a)
    # c = -ones (we want to minimize -sum(g))
    c = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_j a[j] * g[k-j] where 0 <= j < n, 0 <= k-j < n
    # So for a given k, j ranges from max(0, k-n+1) to min(k, n-1)
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x

def alternating_minimize(a_init, max_iters=1000, t=0.1):
    a = a_init.copy().astype(np.float64)

        if Sg < 1e-10:

```python
"""
Alternating LP + LSE L-BFGS + Coordinate descent for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.
Primary method: Block coordinate descent via LP (fix one factor, optimize the other).
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: d/d a[j] of sum_k p_k * (a*a)[k] = sum_k p_k * a[k-j]
    # This is the cross-correlation of a with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_for_g(a, M, n=None):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k.
    """
    if n is None:
        n = len(a)
    c_vec = -np.ones(n)
    
    # Build constraint matrix
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.full(n_constraints, M)
    
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x, 0, None)
    return None


def alternating_lp(a_init, max_iters=200, t_init=0.3, t_min=1e-4, 
                   time_budget=None, t_start=None, verbose=True):
    """
    Alternating LP method: fix a, solve LP for g, update a.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    t = t_init
    
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        conv = fast_conv(a)
        M = float(conv.max())
        score = 2.0 * n * M / (S * S)
        
        if verbose and it % 10 == 0:
            print(f"    AltLP iter {it}: score={score:.8f}, M={M:.6f}, t={t:.4f}")
        
        # Solve LP
        g = solve_lp_for_g(a, M, n)
        if g is None:
            if verbose:
                print("    LP failed, stopping")
            break
        
        Sg = g.sum()
        if Sg < 1e-12:
            break
        
        # Rescale
        g = g * (S / Sg)
        
        # Try update
        a_new = (1 - t) * a + t * g
        sc_new = compute_score(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            t = min(t * 1.1, 0.5)  # Slightly increase step
        else:
            # Try binary search on t
            lo, hi = 0.0, t
            for _ in range(5):
                mid = (lo + hi) / 2
                a_trial = (1 - mid) * a + mid * g
                if compute_score(a_trial) < best_score:
                    lo = mid
                else:
                    hi = mid
            t = max(lo * 0.9, t_min)
            if lo > 0:
                a = (1 - lo) * a + lo * g
                sc_check = compute_score(a)
                if sc_check < best_score:
                    best_score = sc_check
                    best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        if verbose and it % 10 == 0:
            print(f"      LP time: {elapsed:.1f}s")

def run_lbfgs(a0, beta, maxiter=3000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing."""

    if a0.sum() < 1e-6:

    except (TimeoutError, Exception):
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def coord_descent(a_init, n_sweeps=5, top_k=30, time_budget=60.0, t_start=None):
    """Coordinate descent with golden section on top-k coordinates by gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        
        # Subgradient
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)

"""
Alternating LP + LSE L-BFGS hybrid for minimizing 2n*max(self-conv)/(sum)^2.
Key innovation: Direct LP-based block coordinate descent with adaptive constraint
set reduction, combined with LSE-smoothed L-BFGS for local refinement.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time
import math


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def build_conv_matrix(a, k_indices):
    """Build submatrix of the convolution matrix for given lag indices."""
    n = len(a)
    K = len(k_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[idx, j] = a[k - j]
    return A


def solve_lp_reduced(a, M, top_k=None, time_limit=30.0):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for top_k lags.
    """
    n = len(a)
    conv = fast_conv(a)
    
    if top_k is None or top_k >= 2*n - 1:
        k_indices = np.arange(2*n - 1)
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    
    c_vec = -np.ones(n)
    A_ub = build_conv_matrix(a, k_indices)
    b_ub = M * np.ones(len(k_indices))
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def alternating_lp(a_init, max_iters=500, t_init=0.2, time_budget=None, t_start=None,
                   top_k_frac=0.5, verbose=True):
    """
    Alternating LP: fix a, solve LP for g, update a.
    Uses reduced constraint set for speed.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    t = t_init
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    top_k = max(n, int(top_k_frac * (2*n - 1)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        conv = fast_conv(a)
        M = float(conv.max())
        score = 2.0 * n * M / (S * S)
        
        if verbose and it % 50 == 0:
            print(f"    AltLP[{it}]: score={score:.8f}, M={M:.6g}, t={t:.4f}")
        
        # Solve reduced LP
        g = solve_lp_reduced(a, M, top_k=top_k, time_limit=min(30.0, max(5.0, (deadline - time.monotonic()) * 0.1)))
        if g is None:
            if verbose:
                print("    LP failed")
            break
        
        Sg = g.sum()
        if Sg < 1e-12:
            break
        
        g = g * (S / Sg)
        
        # Line search on t
        a_new = (1 - t) * a + t * g
        sc_new = compute_score(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            # Increase t slightly if improvement is good
            t = min(t * 1.05, 0.5)
        else:
            # Binary search for valid t
            lo, hi = 0.0, t
            for _ in range(8):
                mid = (lo + hi) / 2.0
                a_trial = (1 - mid) * a + mid * g
                if compute_score(a_trial) < best_score:
                    lo = mid
                else:
                    hi = mid
            if lo > 1e-8:
                a = (1 - lo) * a + lo * g
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            t = max(lo * 0.5, 1e-6)
        
        if verbose and it % 50 == 0:
            print(f"      time={time.monotonic()-t0:.1f}s")
        
        # Convergence check
        if t < 1e-6:
            if verbose:
                print("    Converged (t too small)")
            break
    
    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=5000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    def obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-14, 'gtol': 1e-11,
                                'maxiter': maxiter, 'maxfun': 100000, 'maxls': 50})
    except (TimeoutError, Exception):
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def coord_descent_polish(a_init, n_sweeps=3, top_k=30, time_budget=30.0, t_start=None):
    """Coordinate descent with golden section on top-gradient coordinates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            
            lo = 0.0
            hi = max(a_old * 3.0, 0.1 * a.max()) if a_old > 0 else 0.5 * a.max()
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(12):
                if time.monotonic() > deadline:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            
            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def perturbation_spectral(a, rng, strength=0.15):
    """Perturb in frequency domain."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = 1.0 + strength * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.3, 2.0)
    phase = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_taper(a, rng):
    """Apply a random taper modification."""
    n = len(a)
    a_new = a.copy()
    # Random cosine taper
    alpha = float(rng.uniform(0.5, 2.0))
    x = np.arange(n) / (n - 1)
    taper = 1.0 + float(rng.uniform(-0.3, 0.3)) * np.cos(np.pi * alpha * (2*x - 1))
    a_new = a_new * taper
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_noise(a, rng, strength=0.05):
    """Add multiplicative noise."""
    noise = 1.0 + strength * rng.standard_normal(len(a))
    a_new = a * noise
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def structured_init(n, rng, style="bessel_like"):
    """Create structured initial sequences."""
    if style == "bessel_like":
        # Based on Fejér kernel shape (sinc^2)
        x = np.arange(n) - n / 2
        a = np sinc(x / n) if hasattr(np, 'sinc') else np.where(x == 0, 1.0, np.sin(np.pi * x / n) / (np.pi * x / n)))
        a = np.abs(a)
    elif style == "gaussian":
        x = np.arange(n) - n / 2
        sigma = n / 6
        a = np.exp(-0.5 * (x / sigma) ** 2)
    elif style == "u_shape":
        x = np.arange(n) / (n - 1)
        a = (2 * x - 1) ** 2 + 0.1
    elif style == "anti_pal":
        x = np.arange(n) - n / 2
        a = np.abs(x) / (n / 2) + 0.3
    else:
        a = np.ones(n)
    a = a - a.min() + 1e-6
    return a

```python
"""
Alternating LP with sparse constraints + LSE L-BFGS + coordinate descent polish
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Primary innovation: direct LP-based block coordinate descent with adaptive
sparse constraint sets, avoiding LSE approximation bias.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def build_conv_matrix_sparse(a, k_indices):
    """Build sparse convolution constraint matrix for given lag indices."""
    n = len(a)
    K = len(k_indices)
    rows = []
    cols = []
    vals = []
    for idx in range(K):
        k = int(k_indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not vals:
        return csr_matrix((K, n))
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def solve_lp_reduced(a, M, top_k=None, time_limit=60.0):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for top_k tightest lags.
    Returns g or None on failure.
    """
    n = len(a)
    conv = fast_conv(a)
    n_lags = 2 * n - 1

    if top_k is None or top_k >= n_lags:
        k_indices = np.arange(n_lags)
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]

    c_vec = -np.ones(n)
    A_ub = build_conv_matrix_sparse(a, k_indices)
    b_ub = M * np.ones(len(k_indices))
    bounds = [(0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return np.clip(res.x, 0, None)
        elif res.status == 9:
            # numerical difficulties - try with slightly relaxed M
            return None
    except Exception:
        pass
    return None


def alternating_lp(a_init, max_iters=200, t_init=0.3, time_budget=None,
                   t_start=None, top_k_frac=0.2, full_constraints_last=20,
                   verbose=True):
    """
    Alternating LP method with adaptive constraint set.
    Fix a, solve LP for g (max sum(g) s.t. a*g <= M, g>=0), update a.
    Uses reduced constraints for speed, full constraints near convergence.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    t = t_init
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    n_lags = 2 * n - 1
    top_k = max(n // 2, int(top_k_frac * n_lags))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        time_remaining = deadline - time.monotonic()
        if time_remaining < 3:
            break

        conv = fast_conv(a)
        M = float(conv.max())
        score = 2.0 * n * M / (S * S)

        if verbose and it % 20 == 0:
            print(f"    AltLP[{it}]: score={score:.8f}, M={M:.6g}, t={t:.4f}, topk={top_k}")

        # Use full constraints in last few iterations
        use_full = (it >= max_iters - full_constraints_last) and (n_lags <= 15000)
        cur_top_k = n_lags if use_full else top_k

        # Time allocation for LP
        lp_time = min(60.0, time_remaining * 0.3)
        g = solve_lp_reduced(a, M, top_k=cur_top_k, time_limit=lp_time)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        Sg = g.sum()
        if Sg < 1e-12:
            break

        g = g * (S / Sg)

        # Line search on mixing parameter t
        a_new = (1.0 - t) * a + t * g
        sc_new = compute_score(a_new)

        improved = False
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            t = min(t * 1.08, 0.5)
            improved = True
        else:
            # Binary search for valid step
            lo, hi = 0.0, t
            for _ in range(10):
                if time.monotonic() > deadline:
                    break
                mid = (lo + hi) * 0.5
                a_trial = (1.0 - mid) * a + mid * g
                if compute_score(a_trial) < best_score:
                    lo = mid
                else:
                    hi = mid
            if lo > 1e-8:
                a = (1.0 - lo) * a + lo * g
                sc_check = compute_score(a)
                if sc_check < best_score:
                    best_score = sc_check
                    best_a = a.copy()
                    improved = True
            t = max(lo * 0.5, 1e-7)

        if verbose and it % 20 == 0:
            print(f"      lp_time={time.monotonic()-t0:.1f}s, improved={improved}")

        if t < 1e-7:
            if verbose:
                print(f"    Converged at iter {it}")
            break

    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=5000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    def obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-14, 'gtol': 1e-11,
                                'maxiter': maxiter, 'maxfun': 150000, 'maxls': 50})
    except (TimeoutError, Exception):
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def run_adam(a_init, lr=0.003, n_steps=50000, beta_lse=200,
             time_budget=None, t_start=None):
    """Adam optimizer with LSE gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        _, grad = lse_score_and_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a
