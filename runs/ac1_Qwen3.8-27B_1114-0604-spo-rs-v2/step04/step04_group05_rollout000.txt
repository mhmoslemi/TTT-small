    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, time_budget, t_start, maxiter=3000, maxfun=30000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = t_start + time_budget
    
    def obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_grad(x, beta)
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter, 
                                'maxfun': maxfun, 'maxls': 40})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out

def peak_flattening(a, n_iters=100, t_start=None, time_budget=30.0):
    """
    Iteratively reduce the top peaks by scaling down coordinates
    that contribute most to those peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        M = c.max()
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top peaks
        top_k = min(5, len(c))
        peak_values = np.sort(c)[::-1][:top_k]
        peak_positions = [int(np.where(c == pv)[0][0]) for pv in peak_values]
        # Actually let's use argsort properly
        sorted_idx = np.argsort(c)[::-1][:top_k]
        
        # For each top peak, compute the "pressure" on each coordinate
        # pressure[j] = sum over peaks k of a[j] * a[k-j] / c[k] (weighted)
        pressure = np.zeros(n)
        for k in sorted_idx:
            weight = c[k] / peak_values[0]  # normalize by max
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                pressure[js:je+1] += weight * a[idx]
        
        # Scale down: a *= (1 - eta * pressure / M)
        # Choose eta adaptively
        max_pressure = pressure.max()
        if max_pressure > 1e-15:
            eta = 0.5 * M / max_pressure  # This gives max reduction of 0.5
            # Actually let's be more conservative
            eta = min(0.3, 0.1 * M / max_pressure)
            a *= (1.0 - eta * pressure / M)
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() > 1e-10:
                a *= S0 / a.sum()
    
    return best_score, best_a

def null_space_perturb(a, rng, n_directions=5):
    """
    Generate perturbations in the null space of the sum constraint
    that are orthogonal to the current peak's subgradient.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = c[k_star]
    S = a.sum()
    
    # Subgradient at the peak
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je+1] = a[idx]
    
    # Remove the mean (project onto null space of sum constraint)
    sg_centered = sg - sg.mean()
    
    results = []
    for d in range(n_directions):
        # Random direction in null space
        r = rng.standard_normal(n)
        r -= r.mean()  # null space of sum
        # Make it have a component against the subgradient
        r -= (r @ sg_centered) / (sg_centered @ sg_centered + 1e-15) * sg_centered
        r /= (np.linalg.norm(r) + 1e-15)
        
        # Try both directions
        for sign in [1, -1]:
            for scale in [0.05, 0.1, 0.2]:
                a_new = a + sign * scale * r * a.max()
                a_new = np.clip(a_new, 0.0, 1000.0)
                if a_new.sum() > 1e-10:
                    a_new *= S / a_new.sum()
                results.append(a_new)
    
    return results

def coord_descent(a, n_sweeps=5, top_k=50, n_evals=20, time_budget=30.0, t_start=None):
    """Coordinate descent with golden-section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]
        S = a.sum()
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Gradient
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je+1] = a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 3, a.max() * 0.5)
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(n_evals - 2):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            
            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a

def spectral_perturb(a, rng, strength=0.15):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.3 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new *= a.sum() / a_new.sum()
    return a_new

def resample_sequence(a, new_len):
    n = len(a)
    if new_len == n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_len)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

def refine(a_init, time_budget, t_start):
    """Multi-stage refinement pipeline."""
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    def try_update(sc, a_cand):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False
    
    # Stage 1: L-BFGS continuation
    if time.monotonic() < deadline - 10:
        t_remaining = deadline - time.monotonic()
        n = len(a)
        # Scale-aware beta
        c = fast_conv(a)
        M = c.max()
        S = a.sum()
        base_beta = max(50, 5000.0 / (M / (S * S) + 1e-10))
        
        betas = [base_beta * 0.1, base_beta * 0.5, base_beta, base_beta * 5, base_beta * 20]
        for beta in betas:
            if time.monotonic() > deadline - 5:
                break
            t_alloc = min((deadline - time.monotonic()) * 0.3, 30.0)
            if t_alloc < 3:
                break
            sc, a_out = run_lbfgs(a, beta, t_alloc, t_start, maxiter=2000, maxfun=20000)
            try_update(sc, a_out)
            a = best_a
    
    # Stage 2: Peak flattening
    if time.monotonic() < deadline - 5:
        t_alloc = min(deadline - time.monotonic() - 3, time_budget * 0.15)
        if t_alloc > 5:
            sc, a_out = peak_flattening(a, n_iters=200, t_start=t_start, time_budget=t_alloc)
            try_update(sc, a_out)
            a = best_a
    
    # Stage 3: Coordinate descent
    if time.monotonic() < deadline - 3:
        t_alloc = min(deadline - time.monotonic() - 2, time_budget * 0.25)
        if t_alloc > 5:
            sc, a_out = coord_descent(a, n_sweeps=8, top_k=40, n_evals=20,
                                     time_budget=t_alloc, t_start=t_start)
            try_update(sc, a_out)
            a = best_a
    
    # Stage 4: Final L-BFGS with very high beta
    if time.monotonic() < deadline - 3:
        t_alloc = min(deadline - time.monotonic() - 2, 20.0)
        if t_alloc > 3:
            sc, a_out = run_lbfgs(a, 50000, t_alloc, t_start, maxiter=5000, maxfun=50000)
            try_update(sc, a_out)
    
    return best_score, best_a
