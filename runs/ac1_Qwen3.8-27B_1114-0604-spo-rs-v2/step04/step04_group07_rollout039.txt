from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    """Adam with LSE smoothing and scheduled beta."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 1e-30, 1e10).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    if normalize:
        a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    t_global = 0
    for bi in range(len(betas)):
        beta = betas[bi]
        lr = lrs[bi] if bi < len(lrs) else lrs[-1]
        steps_this = n_steps // len(betas)
        for t in range(steps_this):
            t_global += 1
            if time.monotonic() - t_start > time_limit_s:
                break

            score, grad = lse_score_grad_a(a, beta, lamb)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)

            lr_t = lr / (1.0 + 1e-5 * t)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if t % verbose_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if time.monotonic() - t_start > time_limit_s:
            break

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=80000, lamb=0.0):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad_a(x_c, beta, lamb)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    a_out = np.maximum(a_out, 0.0)
    if a_out.sum() < 1e-12:
        a_out[:] = 1.0 / n
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coord_descent(a, n_sweeps=200, time_limit_s=None, t_start=None, verbose=False):
    """
    Peak-targeting coordinate descent.
    At each step, identify the peak lag and try to reduce it by
    moving mass from peak-contributing elements to non-contributing ones.
    """
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1e10).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        current_score = 2.0 * n * float(c[k_star]) / (float(a.sum()) ** 2)

        if verbose:
            print(f"    sweep {sweep}: k*={k_star}, c[k*]={c[k_star]:.8f}, score={current_score:.8f}")

        # Identify contributing range
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing each contributing element and redistributing
        improved = False
        # Sort contributors by impact (a[j] * a[k_star - j])
        if je >= js:
            contributions = a[js:je + 1] * a[k_star - np.arange(js, je + 1)][::-1]
            # Actually: contribution of a[j] to c[k_star] is a[j]*a[k_star-j] (counted once)
            # But in the sum, each pair (j, k_star-j) appears, so contribution = a[j]*a[k_star-j]
            contrib_vals = np.zeros(n)
            for j in range(js, je + 1):
                other = k_star - j
                if 0 <= other < n:
                    contrib_vals[j] = a[j] * a[other]
            # Sort by contribution descending
            order = np.argsort(contrib_vals[je - js + 1:if je > js else 1])[::-1] if je > js else np.array([0])
            # Simpler: just try all in the range
            indices = list(range(js, je + 1))
            np.random.shuffle(indices)

        else:
            indices = []

        for j in indices:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-12:
                continue

            # Try reducing a[j] by various fractions
            best_frac = 1.0
            best_local_score = current_score

            for frac in [0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                removed = a_old * (1.0 - frac)

                # Redistribute to non-contributing elements
                non_contrib = np.ones(n, dtype=bool)
                # Mark all elements that contribute to c[k_star]
                for jj in range(max(0, k_star - n + 1), min(k_star, n - 1) + 1):
                    non_contrib[jj] = False
                # Also mark elements that contribute to other top peaks
                # (keep it simple for now)

                nc_count = int(non_contrib.sum())
                if nc_count > 0:
                    a[non_contrib] += removed / nc_count

                sc = compute_score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_frac = frac
                else:
                    a[j] = a_old
                    # Undo redistribution
                    a[non_contrib] -= removed / nc_count if nc_count > 0 else 0

            if best_frac < 1.0 and best_local_score < current_score:
                # Re-apply the best move
                a[j] = a_old * best_frac
                removed = a_old * (1.0 - best_frac)
                non_contrib = np.ones(n, dtype=bool)
                for jj in range(max(0, k_star - n + 1), min(k_star, n - 1) + 1):
                    non_contrib[jj] = False
                nc_count = int(non_contrib.sum())
                if nc_count > 0:
                    a[non_contrib] += removed / nc_count
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                    if verbose:
                        print(f"      improved: a[{j}]*{best_frac}, score={best_score:.8f}")

        if not improved:
            # Try increasing non-contributing elements
            non_contrib = np.ones(n, dtype=bool)
            for jj in range(max(0, k_star - n + 1), min(k_star, n - 1) + 1):
                non_contrib[jj] = False
            nc_idx = np.where(non_contrib)[0]
            if len(nc_idx) > 0:
                np.random.shuffle(nc_idx)
                for j in nc_idx[:10]:
                    if time.monotonic() > deadline:
                        break
                    a_old = a[j]
                    for mult in [1.3, 1.6, 2.0]:
                        a[j] = a_old * mult
                        sc = compute_score(a)
                        if sc < current_score:
                            current_score = sc
                            if sc < best_score:
                                best_score = sc
                                best_a = a.copy()
                            break
                        else:
                            a[j] = a_old

    return best_score, best_a


def multiplicative_weight_update(a, n_iters=200, lr=0.3, verbose=False):
    """
    Multiplicative weight update targeting peak reduction.
    Reduces elements that contribute to the peak, boosts others.
    """
    n = len(a)
    g = np.clip(a.astype(np.float64), 1e-15, 1e10).copy()
    g = g / g.sum()

    for it in range(n_iters):
        c = fast_conv(g)
        k_star = int(c.argmax())
        M = c[k_star]

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        for j in range(js, je + 1):
            other = k_star - j
            if 0 <= other < n:
                contrib[j] += g[other]

        # Also look at the 2nd and 3rd largest peaks
        c_sorted_idx = np.argsort(c)[::-1]
        for pk in c_sorted_idx[1:min(4, len(c_sorted_idx))]:
            js2 = max(0, pk - n + 1)
            je2 = min(pk, n - 1)
            for j in range(js2, je2 + 1):
                other = pk - j
                if 0 <= other < n:
                    contrib[j] += 0.5 * g[other]

        # Multiplicative update: reduce high contributors, boost low
        g = g * np.exp(-lr * contrib / max(M, 1e-15))
        g = np.maximum(g, 1e-15)
        g = g / g.sum()

    return g


def generate_initializations(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (centered)
    sigma = 0.15
    g = np.exp(-0.5 * ((x - 0.4) / sigma) ** 2)
    inits.append(g / g.sum())

    # Skewed Gaussian (off-center)
    g2 = np.exp(-0.5 * ((x - 0.3) / 0.12) ** 2)
    inits.append(g2 / g2.sum())

    # Strongly skewed
    g3 = np.exp(-0.5 * ((x - 0.2) / 0.08) ** 2)
    inits.append(g3 / g3.sum())

    # Triangular (centered)
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Skewed triangle
    tri2 = np.maximum(0, 1 - np.abs((x - 0.3) / 0.4))
    if tri2.sum() > 0:
        inits.append(tri2 / tri2.sum())

    # Sinusoidal modulation
    freq = 5.0
    s = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Two bumps (asymmetric)
    c1, c2 = 0.25, 0.7
    s1, s2 = 0.1, 0.15
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + 0.6 * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 30)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Exponential decay (geometric)
    r_val = 0.999
    exp_seq = r_val ** np.arange(n)
    inits.append(exp_seq / exp_seq.sum())

    # Power law
    pw = 0.5
    pl = (x + 0.01) ** (-pw)
    inits.append(pl / pl.sum())

    return [np.clip(xi, 1e-10, 1000.0) for xi in inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase hybrid optimizer for minimizing 2n*max(self-conv)/(sum)^2.
    Phase 1: Diverse initializations + quick Adam screening
    Phase 2: LSE continuation Adam (non-palindromic) on best candidates
    Phase 3: LBFGS polishing with high beta
    Phase 4: Multiplicative weight + peak-targeting coordinate descent
    Phase 5: Structured restarts with diverse perturbations
    Phase 6: Final intensive polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 30.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                if parent.sum() > 0.01:
                    parent = parent / parent.sum()
    except Exception:
        pass

    if parent is not None:
        n_work = len(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={n_work}, score={parent_score:.8f}")
        try_update(parent_score, parent, "parent")
    else:
        n_work = 4096
        print("No parent available, starting fresh with n=4096.")

    # ==========================================
    # Phase 1: Diverse initializations + screening (12% time)
    # ==========================================
    p1_time = effective_budget * 0.12
    deadline1 = t_start + p1_time
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Diverse init + screening (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    inits = generate_initializations(rng, n_work)
    if parent is not None and len(parent) == n_work:
        inits.insert(0, parent.copy())

    candidates = []
    for i, init in enumerate(inits):
        if time.monotonic() > deadline1 - 3:
            break
        sc_init = compute_score(init)
        # Quick Adam
        beta_sched = [30, 100, 300]
        lr_sched = [0.02, 0.01, 0.005]
        t_remain = deadline1 - time.monotonic() - 2
        if t_remain < 2:
            candidates.append((sc_init, init))
            continue
        sc, a_opt = adam_lse(init, beta_sched, lr_sched, n_steps=5000,
                             time_limit_s=min(t_remain, 8.0),
                             t_start=time.monotonic(), normalize=True)
        candidates.append((sc, a_opt))
        print(f"  Init {i}: score={sc:.8f}")
        try_update(sc, a_opt, f"screening#{i}")

    candidates.sort(key=lambda x: x[0])
    top_k = min(4, len(candidates))
    print(f"  Top {top_k} candidates, best={candidates[0][0]:.8f}")

    # ==========================================
    # Phase 2: LSE continuation Adam - Main (35% time)
    # ==========================================
    p2_time = effective_budget * 0.35
    deadline2 = t_start + p1_time + p2_time
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LSE continuation Adam (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    # Start from the best candidate
    a_cur = candidates[0][1].copy()

    # Beta schedule: geometric increase
    betas_p2 = [50, 150, 500, 1500, 5000, 15000]
    lrs_p2 = [0.015, 0.01, 0.006, 0.003, 0.001, 0.0004]
    n_steps_p2 = max(1, int(p2_time / len(betas_p2) / 0.001))  # rough estimate

    for bi, (beta, lr) in enumerate(zip(betas_p2, lrs_p2)):
        remaining = deadline2 - time.monotonic()
        if remaining < 3:
            break
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_cur, [beta], [lr], n_steps=1000000,
                             time_limit_s=min(remaining * 0.45, p2_time / len(betas_p2)),
                             t_start=time.monotonic(), normalize=True)
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_cur = a_opt
        print(f"  Adam beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Also try with small regularization
    remaining = deadline2 - time.monotonic()
    if remaining > 10:
        for lamb in [0.01, 0.1]:
            remaining = deadline2 - time.monotonic()
            if remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = adam_lse(a_cur, [500], [0.003], n_steps=100000,
                                 time_limit_s=min(remaining * 0.4, 20.0),
                                 t_start=time.monotonic(), normalize=True, lamb=lamb)
            try_update(sc, a_opt, f"Adam reg lamb={lamb}")
            a_cur = a_opt
            print(f"  Adam reg lamb={lamb}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 3: LBFGS Polishing (25% time)
    # ==========================================
    p3_time = effective_budget * 0.25
    deadline3 = t_start + p1_time + p2_time + p3_time
    print(f"\n{'=' * 60}")
    print(f"Phase 3: LBFGS Polish (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    for beta in [2000, 5000, 10000, 20000]:
        remaining = deadline3 - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_a(a_cur, beta, maxiter=8000, maxfun=120000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        a_cur = a_opt
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 4: MW + Peak Coord Descent (15% time)
    # ==========================================
    p4_time = effective_budget * 0.15
    deadline4 = t_start + p1_time + p2_time + p3_time + p4_time
    print(f"\n{'=' * 60}")
    print(f"Phase 4: MW + Coord Descent (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    # Multiplicative weight update
    t0 = time.monotonic()
    g = multiplicative_weight_update(a_cur, n_iters=200, lr=0.3)
    # Blend
    for t_blend in [0.2, 0.4, 0.6, 0.8]:
        a_blend = (1 - t_blend) * a_cur + t_blend * g
        a_blend = np.clip(a_blend, 0, 1000)
        if a_blend.sum() > 1e-12:
            a_blend = a_blend / a_blend.sum()
            sc = compute_score(a_blend)
            try_update(sc, a_blend, f"MW blend={t_blend}")
    print(f"  MW: best={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # Peak-targeting coordinate descent
    remaining = deadline4 - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = peak_coord_descent(a_cur, n_sweeps=500,
                                       time_limit_s=min(remaining, p4_time * 0.6),
                                       t_start=time.monotonic())
        try_update(sc, a_opt, "Peak coord descent")
        a_cur = a_opt
        print(f"  Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 5: Structured restarts (10% time)
    # ==========================================
    p5_time = effective_budget * 0.10
    deadline5 = t_start + p1_time + p2_time + p3_time + p4_time + p5_time
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Structured Restarts (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    n_restarts = 10
    restart_budget = p5_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > deadline5 - 5:
            break

        noise_scale = 0.03 + 0.04 * rng.random()
        strategy = r % 5

        if strategy == 0:
            # Gaussian perturbation in log-space
            log_a = np.log(np.clip(a_cur, 1e-15, 1e10))
            log_pert = log_a + noise_scale * rng.standard_normal(n_work)
            a_pert = np.exp(log_pert - np.max(log_pert))
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Peak-avoidance perturbation
            c = fast_conv(a_cur)
            k_star = int(c.argmax())
            js = max(0, k_star - n_work + 1)
            je = min(k_star, n_work - 1)
            a_pert = a_cur.copy()
            a_pert[js:je + 1] *= (0.4 + 0.3 * rng.random())
            non_peak = np.ones(n_work, dtype=bool)
            non_peak[js:je + 1] = False
            if non_peak.sum() > 0:
                a_pert[non_peak] *= (1.0 + 0.2 * rng.random())
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Fourier perturbation
            x = np.arange(n_work) / n_work
            freqs = rng.uniform(1, 12, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amps = rng.uniform(0.02, 0.1, size=3)
            mod = 1.0
            for f, p, a_m in zip(freqs, phases, amps):
                mod += a_m * np.cos(2 * np.pi * f * x + p)
            a_pert = a_cur * np.maximum(mod, 0.1)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 3:
            # Block modification
            n_blocks = 6 + int(6 * rng.random())
            a_pert = a_cur.copy()
            block_size = max(1, n_work // n_blocks)
            for b in range(n_blocks):
                s = b * block_size
                e = min(s + block_size, n_work)
                a_pert[s:e] *= rng.uniform(0.3, 2.0)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            # Shift-and-modify: try shifting mass between halves
            half = n_work // 2
            a_pert = a_cur.copy()
            swap_frac = 0.1 + 0.2 * rng.random()
            a_pert[:half] += swap_frac * a_pert[half:]
            a_pert[half:] += swap_frac * a_pert[:half]
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

        # Quick refinement with Adam
        time_left = deadline5 - 5 - time.monotonic()
        if time_left < 3:
            break
        adam_t = min(restart_budget * 0.6, time_left)
        sc, a_r = adam_lse(a_pert, [200, 1000], [0.005, 0.002], n_steps=50000,
                           time_limit_s=adam_t, t_start=time.monotonic(), normalize=True)
        # Quick LBFGS
        if time.monotonic() < deadline5 - 3:
            sc2, a_r2 = run_lbfgs_a(a_r, 3000, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")

    # ==========================================
    # Phase 6: Final intensive polish (8% time)
    # ==========================================
    p6_time = effective_budget * 0.08
    deadline6 = t_start + effective_budget - 10.0
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final Polish (budget={max(0, deadline6 - time.monotonic()):.0f}s)")
    print(f"{'=' * 60}")

    # Very high beta LBFGS
    for beta in [15000, 30000, 50000]:
        remaining = deadline6 - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_a(a_cur, beta, maxiter=10000, maxfun=200000)
        try_update(sc, a_opt, f"final LBFGS beta={beta}")
        a_cur = a_opt
        print(f"  Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Very high beta Adam
    remaining = deadline6 - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_cur, [30000], [0.0002], n_steps=200000,
                             time_limit_s=min(remaining * 0.5, 30.0),
                             t_start=time.monotonic(), normalize=True)
        try_update(sc, a_opt, "final Adam high-beta")
        a_cur = a_opt
        print(f"  Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final coordinate descent
    remaining = deadline6 - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = peak_coord_descent(a_cur, n_sweeps=500,
                                       time_limit_s=min(remaining, 30.0),
                                       t_start=time.monotonic())
        try_update(sc, a_opt, "final coord descent")
        print(f"  Final coord: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # One more restart cycle if time allows
    remaining = deadline6 - time.monotonic()
    if remaining > 20:
        for r in range(3):
            if time.monotonic() > deadline6 - 3:
                break
            log_a = np.log(np.clip(a_cur, 1e-15, 1e10))
            log_pert = log_a + 0.02 * rng.standard_normal(n_work)
            a_pert = np.exp(log_pert - np.max(log_pert))
            a_pert = a_pert / a_pert.sum()
            t_rem = deadline6 - 2 - time.monotonic()
            if t_rem < 5:
                break
            sc, a_r = adam_lse(a_pert, [5000], [0.001], n_steps=100000,
                               time_limit_s=t_rem * 0.5, t_start=time.monotonic(), normalize=True)
            sc2, a_r2 = run_lbfgs_a(a_r, 20000, maxiter=5000, maxfun=80000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
            try_update(sc, a_r, f"final restart#{r}")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
