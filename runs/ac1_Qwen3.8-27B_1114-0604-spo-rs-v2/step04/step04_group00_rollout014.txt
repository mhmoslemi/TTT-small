def run_lbfgs(a0, beta, maxiter=8000, maxfun=100000):

def anti_palindromic_step(a, strength=0.1):
    """
    Adjust the sequence to be more anti-palindromic.
    For each pair (a[j], a[n-1-j]), increase the difference while keeping the sum.
    This reduces the center convolution value.
    """
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    a = a / S  # normalize
    
    for j in range(n // 2):
        k = n - 1 - j
        s = a[j] + a[k]  # sum to preserve
        if s < 1e-15:
            continue
        # Current product: a[j] * a[k]
        # Target: increase |a[j] - a[k]| by factor (1+strength)
        diff = a[j] - a[k]
        new_diff = diff * (1.0 + strength)
        # Keep sum the same: a[j] + a[k] = s, a[j] - a[k] = new_diff
        a[j] = (s + new_diff) / 2.0
        a[k] = (s - new_diff) / 2.0
        # Enforce non-negativity
        a[j] = max(0.0, a[j])
        a[k] = max(0.0, a[k])
    
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def spectral_flatten(a, n_iters=200, aggressive=False):
    """
    Flatten the power spectrum to reduce convolution peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if aggressive:
            # Hard clip top 50%
            thresh = np.quantile(P, 0.5)
            scale = np.ones_like(P)
            above = P > thresh
            scale[above] = np.sqrt(thresh / P[above]) * 0.9
        else:
            # Soft flatten: scale by P_mean / P to the power of a strength
            strength = 0.3 + 0.4 * progress
            scale = (np.maximum(P_mean / P, 0.01)) ** strength
            # But don't scale the DC bin too much
            scale[0] = 1.0

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_directed_refine(a, n_iters=2000):
    """
    Directly reduce the peak of the convolution by adjusting contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        max_val = float(c[k_star])
        sc = 2.0 * n * max_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find elements contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        # Contribution of a[j] to c[k_star] is a[k_star - j]
        contribs = np.array([a[k_star - j] for j in range(j_start, j_end + 1)])
        
        # Try reducing the element with highest contribution
        # and redistributing to a non-contributing element
        max_contrib_idx = np.argmax(contribs)
        j_max = j_start + max_contrib_idx
        
        # Find a "safe" element to redistribute to (one that doesn't contribute to the peak)
        # An element at position i contributes to c[k] if 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # So a[i] does NOT contribute to c[k_star] if i < j_start or i > j_end
        safe_indices = np.where((np.arange(n) < j_start) | (np.arange(n) > j_end))[0]
        if len(safe_indices) == 0:
            break
        
        # Choose the safest index (farthest from the peak range)
        if len(safe_indices) > 0:
            j_safe = safe_indices[np.argmin(np.abs(safe_indices - k_star))]
            # Actually, we want the one that contributes least to ALL peaks
            # For now, just pick one
            j_safe = safe_indices[0]

        # Reduce a[j_max] and add to a[j_safe]
        epsilon = a[j_max] * 0.01  # reduce by 1%
        for frac in [0.5, 1.0, 2.0, 5.0, 10.0]:
            test_a = a.copy()
            test_a[j_max] -= epsilon * frac
            if test_a[j_max] < 0:
                test_a[j_max] = 0
                test_a[j_safe] += test_a[j_max] + epsilon * frac  # hmm this is wrong
                continue
            test_a[j_safe] += epsilon * frac
            S_test = test_a.sum()
            if S_test < 1e-12:
                continue
            test_a = test_a / S_test
            test_sc = compute_score(test_a)
            if test_sc < best_sc:
                best_sc = test_sc
                best_a = test_a.copy()
                a = test_a
                break

    return best_a


def multi_beta_lbfgs(a0, betas, maxiter_each=5000):
    """Run L-BFGS with a schedule of increasing betas."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

Multi-strategy optimizer combining anti-palindromic shaping, adaptive LSE L-BFGS,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    return lse_score_grad(x, beta)

                continue
            diff = a[j] - a[k]
            new_diff = diff * (1.0 + strength)
            a[j] = max(0.0, (s + new_diff) / 2.0)
            a[k] = max(0.0, (s - new_diff) / 2.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return a


def spectral_flatten_equality(a, n_iters=300, target_flatness=0.8):
    """
    Flatten power spectrum by scaling bins toward the mean.
    More aggressive than standard Fienup.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Scale toward flat: P_new = P_mean + (P - P_mean) * (1-target)
        # Equivalent to: scale_k = (P_mean + (P_k - P_mean)*(1-target)) / P_k
        target = target_flatness * (0.5 + 0.5 * progress)
        P_target = np.maximum(P_mean + (P - P_mean) * (1.0 - target), 1e-15 * P_mean)
        scale = np.sqrt(P_target / P)
        # Don't touch DC (bin 0)
        scale[0] = 1.0

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def bb_subgradient(a, n_iters=5000, top_m=5):
    """
    Subgradient method with Barzilai-Borwein step size and top-M peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1.0)

    prev_a = a.copy()
    prev_grad = None
    step = 0.01

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks for subgradient
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        # Compute subgradient (direction to decrease the max)
        sg = np.zeros(n)
        weights = c[top_idx] / c[top_idx].sum()
        for w, k in zip(weights, top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Barzilai-Borwein step
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = float(s_vec @ y_vec)
            yy = float(y_vec @ y_vec)
            ss = float(s_vec @ s_vec)
            if yy > 1e-30 and abs(sy) > 1e-30:
                step_bb = sy / yy
                step_bb = max(1e-6, min(step_bb, 0.1))
                step = 0.5 * step + 0.5 * step_bb

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        prev_a = a.copy()
        prev_grad = grad.copy()

        # Decay step slightly
        step *= 0.9999

    return best_a


def focused_peak_reduction(a, n_iters=2000, n_peaks=3):
    """
    For each of the top peaks, try to reduce it by adjusting contributing elements.
    Uses a "push out" strategy: reduce contributing elements, add to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        sc = compute_score(a)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peaks
        top_k = np.argsort(c)[::-1][:n_peaks]

        for k_peak in top_k:
            k_peak = int(k_peak)
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            if j_end < j_start:
                continue

            # Contributions to this peak
            contribs = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contribs[j] = a[k_peak - j]

            # Find the element with max contribution
            j_max = j_start + int(np.argmax(contribs[j_start:j_end + 1]))

            # Find a safe element (doesn't contribute to any top peak)
            dangerous = set()
            for kp in top_k:
                kp = int(kp)
                js = max(0, kp - n + 1)
                je = min(kp, n - 1)
                dangerous.update(range(js, je + 1))

            safe = [i for i in range(n) if i not in dangerous]
            if not safe:
                safe = list(set(range(n)) - set(dangerous) - {j_max})
            if not safe:
                continue

            # Try moving mass from j_max to a safe location
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[j_max] * frac
                for j_safe in safe[:3]:
                    test_a = a.copy()
                    test_a[j_max] -= amt
                    test_a[j_safe] += amt
                    if test_a[j_max] < 0 or test_a[j_safe] > 1000:
                        continue
                    S_t = test_a.sum()
                    if S_t < 1e-12:
                        continue
                    test_a = test_a / S_t
                    test_sc = compute_score(test_a)
                    if test_sc < best_sc:
                        best_sc = test_sc
                        best_a = test_a.copy()
                        a = test_a
                        break

    return best_a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    return np.maximum(new_a, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Start from parent, apply anti-palindromic adjustment + LSE
    2. Ramping LSE (low to high beta) for global shape optimization
    3. Spectral equality flattening
    4. BB subgradient for fine-tuning
    5. Focused peak reduction
    6. Final high-beta LSE polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 35.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) / 4096.0

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            old = best_score
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (was {old:.8f}, n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        n = len(parent)
        a_work = parent.astype(np.float64).copy()
        init_sc = compute_score(a_work)
        print(f"Parent: n={n}, score={init_sc:.8f}")
        update_best(init_sc, a_work, "parent")
    else:
        n = 4096
        a_work = np.ones(n) / n
        print(f"No parent, using constant n={n}, score={compute_score(a_work):.8f}")
        update_best(compute_score(a_work), a_work, "constant")

    n = len(a_work)

    # ==================================================================
    # Phase 1: Anti-palindromic + short LSE (20%)
    # ==================================================================
    phase1_end = t_start + 0.20 * (total_budget - 35.0)
    print(f"\n=== Phase 1: Anti-palindromic (deadline t={phase1_end-t_start:.0f}s) ===")

    for strength in [0.05, 0.1, 0.15, 0.2, 0.3]:
        if time.monotonic() > phase1_end:
            break
        a_ap = anti_palindromic_adjust(a_work, strength=strength, n_reps=2)
        sc = compute_score(a_ap)
        update_best(sc, a_ap, f"anti s={strength}")

        if time.monotonic() < phase1_end:
            sc2, a2 = run_lbfgs(a_ap, 200, maxiter=2000, maxfun=30000)
            update_best(sc2, a2, f"anti+LSE s={strength}")

    # ==================================================================
    # Phase 2: Ramping LSE from best (30%)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 35.0)
    print(f"\n=== Phase 2: Ramping LSE (deadline t={phase2_end-t_start:.0f}s) ===")

    a_cur = best_a.copy()
    betas = [30, 60, 100, 200, 400, 800, 1500, 3000, 6000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        # More iterations for lower beta (easier problem)
        iters = max(1500, min(6000, 200000 // (beta + 10)))
        sc, a_new = run_lbfgs(a_cur, beta, maxiter=iters, maxfun=iters * 15)
        update_best(sc, a_new, f"LSE b={beta}")
        a_cur = a_new

    # ==================================================================
    # Phase 3: Spectral equality (15%)
    # ==================================================================
    phase3_end = t_start + 0.65 * (total_budget - 35.0)
    print(f"\n=== Phase 3: Spectral equality (deadline t={phase3_end-t_start:.0f}s) ===")

    for flatness in [0.6, 0.8, 0.9]:
        if time.monotonic() > phase3_end:
            break
        a_sf = spectral_flatten_equality(best_a, n_iters=250, target_flatness=flatness)
        sc = compute_score(a_sf)
        update_best(sc, a_sf, f"spec flat={flatness}")

        if time.monotonic() < phase3_end:
            sc2, a2 = run_lbfgs(a_sf, 1000, maxiter=3000, maxfun=50000)
            update_best(sc2, a2, f"spec+LSE flat={flatness}")

    # ==================================================================
    # Phase 4: BB Subgradient (15%)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 35.0)
    print(f"\n=== Phase 4: BB Subgradient (deadline t={phase4_end-t_start:.0f}s) ===")

    for top_m in [3, 5, 10]:
        if time.monotonic() > phase4_end:
            break
        a_bb = bb_subgradient(best_a, n_iters=4000, top_m=top_m)
        sc = compute_score(a_bb)
        update_best(sc, a_bb, f"BB top={top_m}")

    # ==================================================================
    # Phase 5: Focused peak reduction (10%)
    # ==================================================================
    phase5_end = t_start + 0.90 * (total_budget - 35.0)
    print(f"\n=== Phase 5: Peak reduction (deadline t={phase5_end-t_start:.0f}s) ===")

    a_fp = focused_peak_reduction(best_a, n_iters=2000, n_peaks=3)
    sc = compute_score(a_fp)
    update_best(sc, a_fp, "focused peak")

    if time.monotonic() < phase5_end:
        sc2, a2 = run_lbfgs(a_fp, 5000, maxiter=2000, maxfun=40000)
        update_best(sc2, a2, "peak+LSE5000")

    # ==================================================================
    # Phase 6: Try different lengths (5%)
    # ==================================================================
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 6: Length exploration ===")
        for delta in [-256, 256]:
            if time.monotonic() > deadline - 20:
                break
            new_n = n + delta
            if new_n < 1024 or new_n > 20000:
                continue
            a_res = resample_sequence(best_a, new_n)
            a_res = a_res / a_res.sum()
            sc = compute_score(a_res)
            if sc < best_score:
                # Optimize at new length
                sc2, a2 = run_lbfgs(a_res, 1000, maxiter=3000, maxfun=50000)
                update_best(sc2, a2, f"resample n={new_n}")

    # ==================================================================
    # Phase 7: Final polish
    # ==================================================================
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final polish ===")
        for beta in [10000, 20000, 50000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a2 = run_lbfgs(best_a, beta, maxiter=2000, maxfun=30000)
            update_best(sc, a2, f"final b={beta}")

    # Final anti-palindromic nudge + LSE
    if time.monotonic() < deadline - 5:
        a_final = anti_palindromic_adjust(best_a, strength=0.05, n_reps=1)
        sc = compute_score(a_final)
        update_best(sc, a_final, "final anti-0.05")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
