    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=None):
    """LSE-smoothed score with optional top-k restriction. Returns (score, gradient)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_work = c.copy()
    if top_k is not None and top_k < len(c):
        # Keep only top_k values, set rest to the k-th largest value
        sorted_vals = np.sort(c_work)[::-1]
        threshold = sorted_vals[min(top_k, len(c)) - 1]
        c_work = np.maximum(c_work, threshold)
        # Actually we want to soft-clip: values above threshold contribute, below don't
        # Better approach: mask
        top_indices = np.argsort(c)[::-1][:top_k]
        c_masked = np.full_like(c, c[top_indices[0]])  # fill with max
        c_masked[top_indices] = c[top_indices]
        # Use a soft version: replace non-top with the minimum of top
        min_top = c[top_indices].min()
        c_masked = np.where(np.isin(np.arange(len(c)), top_indices), c, min_top)

    c_max = float(c_masked.max())
    scaled = beta * (c_masked - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da_j = sum_k p_k * dc_k/da_j
    # dc_k/da_j = 2*a_{k-j} (with boundary handling)
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_func_wrapper(x, beta, top_k):
    """Wrapper for scipy minimize (no closures)."""
    return lse_score_grad(x, beta, top_k)


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_func_wrapper, a0, args=(beta, top_k),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def projected_peak_step(a, n_peaks=5, n_substeps=10, step_frac=0.1):
    """
    LP-inspired projected step: for each top peak, compute the gradient
    direction and take a projected (non-negative) step that reduces the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for substep in range(n_substeps):
        c = fast_conv(a)
        top_k_indices = np.argsort(c)[::-1][:n_peaks]

        for k_star in top_k_indices:
            Mak = float(c[k_star])
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            idx = np.arange(j_start, j_end + 1)
            weights = a[k_star - idx]  # gradient direction for this peak

            # We want to reduce sum(weights * a[idx])
            # Move mass from high-weight positions to low-weight positions
            w_sorted = np.sort(weights)
            n_move = max(1, len(weights) // 4)
            top_positions = np.argsort(weights)[::-1][:n_move]
            bot_positions = np.argsort(weights)[:n_move]

            total_move = 0.0
            for tp in top_positions:
                total_move += a[j_start + tp] * step_frac
            if total_move < 1e-15:
                continue

            # Redistribute mass from top to bottom
            per_bot = total_move / len(bot_positions)
            for tp in top_positions:
                a[j_start + tp] *= (1.0 - step_frac)
            for bp in bot_positions:
                a[j_start + bp] += per_bot

        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            break

    return best_a


def structured_perturbation(a, rng, mode=0):
    """Generate structured perturbation based on peak structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    seg = a[j_start:j_end + 1]
    contrib = seg[::-1].copy()

    if mode == 0:
        # Anti-peak: reduce top contributors, boost bottom
        top_count = max(3, n // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:top_count]
        a_p = a.copy()
        reduction = 0.0
        for li in top_local:
            a_p[j_start + li] *= 0.85
            reduction += a[ j_start + li] - a_p[j_start + li]
        per = reduction / len(bot_local)
        for li in bot_local:
            a_p[j_start + li] += per
    elif mode == 1:
        # Frequency-domain perturbation: add a flat-spectrum component
        a_p = a.copy()
        freq = int(rng.integers(1, n // 4))
        phase = 2.0 * np.pi * rng.random()
        amp = 0.05 * a.std()
        a_p += amp * np.cos(2.0 * np.pi * freq * x + phase)
    elif mode == 2:
        # Random subset rescale
        n_sel = n // 5
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + 0.1 * (rng.random(n_sel) - 0.5))
    elif mode == 3:
        # Block perturbation
        n_blocks = 4 + int(rng.integers(0, 5))
        block_size = n // n_blocks
        a_p = a.copy()
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + 0.08 * (rng.random() - 0.5))
    else:
        # Gaussian noise
        a_p = a * (1.0 + 0.05 * rng.standard_normal(n))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def flat_spectrum_perturbation(n, rng, amplitude=0.1):
    """Generate a perturbation with flat power spectrum (white in frequency)."""
    # Generate random phases in frequency domain
    freqs = np.fft.rfftfreq(n)
    mask = freqs > 1.0 / n  # exclude DC
    phases = 2.0 * np.pi * rng.random(len(freqs))
    spectrum = np.zeros(len(freqs), dtype=np.complex128)
    spectrum[mask] = amplitude / np.sqrt(n) * np.exp(1j * phases[mask])
    pert = np.fft.irfft(spectrum * np.sqrt(n), n)
    return pert


def two_scale_optimize(a_coarse, ratio=4, beta=500, maxiter=10000):
    """Two-scale: optimize coarse version, then upsample and refine."""
    n_coarse = len(a_coarse)
    n_fine = n_coarse * ratio

    # Upsample coarse to fine
    a_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        a_fine[i * ratio:(i + 1) * ratio] = a_coarse[i] / ratio

    # Optimize fine
    sc, a_opt = run_lbfgs(a_fine, beta, maxiter=maxiter, maxfun=maxiter * 5)
    return sc, a_opt


def coarse_from_fine(a_fine, ratio=4):
    """Downsample by block averaging."""
    n_fine = len(a_fine)
    n_coarse = n_fine // ratio
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a_fine[i * ratio:(i + 1) * ratio].sum()
    return a_coarse


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-phase strategy: initialization, L-BFGS refinement, ILS, final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(64) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize from parent or generate
    if parent is not None:
        n_work = len(parent)
        a_cur = parent / parent.sum()
        sc = compute_score(a_cur)
        report(sc, a_cur, "parent")
        print(f"Parent: n={n_work}, score={sc:.10f}")
    else:
        n_work = 6962
        a_cur = np.ones(n_work) * 0.5
        sc = compute_score(a_cur)
        report(sc, a_cur, "uniform")

    # ==================================================================
    # Phase 1: Coarse-to-fine initialization (10% time)
    # ==================================================================
    phase1_end = t_start + 0.10 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Initialization & Coarse Optimization ---")

    # Generate diverse initializations
    n = n_work
    x = np.arange(n, dtype=np.float64) / n
    inits = [a_cur.copy()]

    # U-shaped
    u_shape = 0.1 + 0.9 * (2.0 * x - 1.0) ** 2
    inits.append(u_shape / u_shape.sum())

    # Cosine bump at 1/4 and 3/4
    two_peak = np.exp(-((x - 0.25) / 0.15) ** 2) + np.exp(-((x - 0.75) / 0.15) ** 2)
    inits.append(two_peak / two_peak.sum())

    # Flat spectrum perturbation of uniform
    fs_pert = flat_spectrum_perturbation(n, rng, amplitude=0.3)
    fs_init = np.abs(np.ones(n) / n + fs_pert)
    inits.append(fs_init / fs_init.sum())

    # Random with structure
    rand_struct = np.abs(1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x))
    inits.append(rand_struct / rand_struct.sum())

    # Quick L-BFGS on each initialization
    for i, init in enumerate(inits[:4]):
        if time.monotonic() > phase1_end:
            break
        sc, a_opt = run_lbfgs(init, beta=200, maxiter=5000, maxfun=20000, top_k=20)
        report(sc, a_opt, f"init_{i}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: Progressive L-BFGS with top-k narrowing (40% time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: Progressive L-BFGS Refinement ---")

    # Beta schedule with progressive top_k narrowing
    # (beta, top_k, fraction_of_phase2_time)
    schedule = [
        (100, 50, 0.05),
        (300, 30, 0.05),
        (1000, 20, 0.05),
        (3000, 10, 0.05),
        (10000, 7, 0.05),
        (30000, 5, 0.05),
        (100000, 4, 0.05),
        (300000, 3, 0.05),
        (1000000, 3, 0.05),
        (3000000, 2, 0.05),
        (10000000, 2, 0.05),
    ]

    for beta, top_k, time_frac in schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        alloc = min(remaining, time_frac * (float(budget_s) - 15.0))
        if alloc < 10:
            break
        maxiter = int(min(30000, max(2000, alloc * 40)))
        maxfun = int(min(100000, max(5000, alloc * 150)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun, top_k=top_k)
        report(sc, a_opt, f"LBFGS b={beta} k={top_k}")

        # Also try without top-k restriction at high beta
        if beta >= 10000 and time.monotonic() < phase2_end - 30:
            sc2, a_opt2 = run_lbfgs(best_a, beta // 2, maxiter=maxiter // 2,
                                     maxfun=maxfun // 2, top_k=None)
            report(sc2, a_opt2, f"LBFGS b={beta//2} full")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Projected peak reduction (10% time)
    # ==================================================================
    phase3_end = t_start + 0.60 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Projected Peak Reduction ---")

    for iteration in range(5):
        if time.monotonic() > phase3_end:
            break
        a_pr = projected_peak_step(best_a, n_peaks=5, n_substeps=15, step_frac=0.08)
        sc = compute_score(a_pr)
        report(sc, a_pr, f"proj_{iteration}")

        # Follow up with L-BFGS
        if time.monotonic() < phase3_end - 20:
            sc2, a_opt2 = run_lbfgs(a_pr, 50000, maxiter=10000, maxfun=50000, top_k=3)
            report(sc2, a_opt2, f"proj_{iteration}_lbfgs")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated Local Search (20% time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 30:
            break

        mode = trial % 5
        a_pert = structured_perturbation(best_a, rng, mode=mode)

        # Quick L-BFGS on perturbation
        beta = 1000 + int(5000 * rng.random())
        top_k = 3 + int(5 * rng.random())
        maxiter = int(min(15000, max(2000, remaining * 15)))
        maxfun = int(min(50000, max(5000, remaining * 50)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun, top_k=top_k)
        report(sc, a_opt, f"ILS {trial} m={mode}")

        # If improved, do a second round with smaller perturbation
        if sc < best_score and time.monotonic() < phase4_end - 30:
            a_pert2 = structured_perturbation(a_opt, rng, mode=(mode + 1) % 5)
            sc2, a_opt2 = run_lbfgs(a_pert2, beta * 3, maxiter=maxiter // 2,
                                    maxfun=maxfun // 2, top_k=2)
            report(sc2, a_opt2, f"ILS {trial} refine")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Nesterov + Final Polish (10% time)
    # ==================================================================
    phase5_end = t_start + 0.90 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 5: Nesterov & Final Polish ---")

    # Nesterov subgradient
    if time.monotonic() < phase5_end:
        a_ns = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
        sc = compute_score(a_ns)
        report(sc, a_ns, "nesterov_p5")

    # Final high-precision L-BFGS
    for beta in [500000, 2000000, 5000000]:
        if time.monotonic() > deadline - 30:
            break
        maxiter = int(min(50000, max(5000, (deadline - time.monotonic()) * 30)))
        maxfun = int(min(200000, max(10000, (deadline - time.monotonic()) * 100)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun, top_k=2)
        report(sc, a_opt, f"final_b={beta}")

    # One more projected step + L-BFGS
    if time.monotonic() < deadline - 30:
        a_pr = projected_peak_step(best_a, n_peaks=3, n_substeps=20, step_frac=0.05)
        sc = compute_score(a_pr)
        report(sc, a_pr, "final_proj")
        if time.monotonic() < deadline - 15:
            sc2, a_opt2 = run_lbfgs(a_pr, 2000000, maxiter=20000, maxfun=80000, top_k=2)
            report(sc2, a_opt2, "final_proj_lbfgs")

    # ==================================================================
    # Phase 6: Length exploration if time permits (remaining time)
    # ==================================================================
    print(f"\n--- Phase 6: Length Exploration ---")

    n_best = len(best_a)
    length_tries = [n_best - 512, n_best + 512, n_best // 2, min(20000, n_best * 2)]
    length_tries = [lt for lt in length_tries if 512 <= lt <= 30000]
    # Remove duplicates and sort
    length_tries = sorted(set(length_tries))

    for n_try in length_tries:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Interpolate best_a to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.clip(a_interp, 0.0, 1000.0)
        S = a_interp.sum()
        if S > 1e-12:
            a_interp = a_interp / S

        maxiter = int(min(20000, max(3000, remaining * 20)))
        maxfun = int(min(60000, max(5000, remaining * 60)))
        sc, a_opt = run_lbfgs(a_interp, 2000, maxiter=maxiter, maxfun=maxfun, top_k=10)
        report(sc, a_opt, f"len={n_try}")

        # Quick refine
        if time.monotonic() < deadline - 15:
            sc2, a_opt2 = run_lbfgs(a_opt, 50000, maxiter=10000, maxfun=40000, top_k=3)
            report(sc2, a_opt2, f"len={n_try}_refine")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
