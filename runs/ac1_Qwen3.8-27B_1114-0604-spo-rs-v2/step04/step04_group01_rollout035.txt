"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using Exponentiated Gradient (multiplicative weights) with LSE annealing,
coordinate descent refinement, and multi-start diversification.
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def cross_corr(a, w):
    """
    h[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1
    where a[k] = 0 for k >= n.
    Computed as middle slice of conv(a[::-1], w).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def multiplicative_eg_step(a, beta, eta):
    """
    One Exponentiated Gradient (multiplicative weights) step.
    Uses LSE-smoothed gradient for the max of the self-convolution.
    Returns updated a (normalized to sum=1).
    """
    n = len(a)
    c = fast_self_conv(a)
    dmax = float(np.max(c))
    if dmax < 1e-12:
        return a.copy()
    
    z = beta * (c - dmax)
    # Clip z to prevent overflow
    z = np.clip(z, -500.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e < 1e-12:
        return a.copy()
    w = e / sum_e
    
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] (approximate, ignoring self-term)
    h = cross_corr(a, w)
    g = 2.0 * h
    
    # For the sum-normalized problem, the effective gradient is g - mean(g)
    # (the mean is the Lagrange multiplier for the sum constraint)
    g_centered = g - float(np.mean(g))
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g_centered)
    a_new = np.clip(a_new, 0.0, None)
    S = float(a_new.sum())
    if S < 1e-12:
        return a.copy()
    a_new = a_new / S
    return a_new

    a = normalize_a(a)

            best_a = a.copy()
            print("[{}] beta={:.1f} score={:.8f}".format(label, beta, best_sc))

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using Exponentiated Gradient (multiplicative weights) with LSE annealing,
adaptive step sizes, multi-start diversification, and final polish.

import numpy as np
import time


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1, via middle slice of conv(a[::-1], w)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def eg_step(a, beta, eta, momentum_buf=None, mu=0.0):
    """
    One Exponentiated Gradient step for normalized a (sum=1).
    Uses LSE-smoothed gradient of n*max(conv(a,a)).
    Returns (a_new, grad_magnitude) for adaptive step sizing.
    """
    n = len(a)
    c = fast_self_conv(a)
    d = float(n) * c
    dmax = float(np.max(d))
    if dmax < 1e-12:
        return a.copy(), 0.0

    z = beta * (d - dmax)
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e < 1e-10:
        return a.copy(), 0.0
    w = e / sum_e

    h = cross_corr(a, w)
    g = float(2.0 * n) * h
    g = g - float(np.mean(g))

    if momentum_buf is not None and mu > 0.0:
        momentum_buf[:] = mu * momentum_buf + (1.0 - mu) * g
        g = momentum_buf

    gmag = float(np.max(np.abs(g)))
    if gmag < 1e-12:
        return a.copy(), 0.0

    log_update = -eta * g
    log_update = np.clip(log_update, -80.0, 80.0)
    a_new = a * np.exp(log_update)
    a_new = np.clip(a_new, 0.0, None)
    S = float(a_new.sum())
    if S < 1e-12:
        return a.copy(), gmag
    a_new = a_new / S
    return a_new, gmag


def eg_anneal(a, deadline, label="EG", n_betas=12, steps_per_beta=300,
              beta_start=2.0, beta_end=50000.0, mu=0.7):
    """
    Run EG with LSE temperature annealing.
    """
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    momentum_buf = np.zeros(n, dtype=np.float64)

    log_betas = np.geomspace(beta_start, beta_end, n_betas)
    total_steps = 0

    for bi, beta in enumerate(log_betas):
        beta = float(beta)
        eta = min(1.0, 0.5 / max(1.0, beta ** 0.3))
        if beta > 1000:
            eta = min(0.1, 50.0 / beta)
        if beta > 10000:
            eta = min(0.05, 500.0 / beta)

        for step in range(steps_per_beta):
            if time.monotonic() > deadline:
                return best_sc, best_a
            a, gmag = eg_step(a, beta, eta, momentum_buf, mu)
            total_steps += 1
            if step % 100 == 0 or step == steps_per_beta - 1:
                sc = exact_score_a(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

        sc = exact_score_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("[{}] beta={:.1f} score={:.8f} steps={}".format(label, beta, best_sc, total_steps))

    return best_sc, best_a


def eg_fine_tune(a, deadline, label="fine", n_iter=5000, beta_init=5000.0):
    """
    Fine-tuning with high beta and small steps, gradually increasing beta.
    """
    a = normalize_a(a)

def perturb(a, rng, strength=0.3):
    """Perturb a normalized sequence."""
    n = len(a)
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.30))
        q = (1.0 - t) * a + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = a[::-1].copy()
    elif mode == 3:
        c = fast_self_conv(a)
        idx = int(np.argmax(c))
        a_rev = a[::-1].copy()
        shift = int(rng.integers(-3, 4))
        q = np.roll(a_rev, shift)
    elif mode == 4:
        loga = np.log(a + 1e-30)
        k = min(31, max(3, n // 200))
        if k % 2 == 0:
            k += 1
        sig = max(1.0, k / 4.0)
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise_c = np.convolve(noise, kern, mode="same")
        noise_c -= noise_c.mean()
        loga += strength * 0.5 * noise_c
        q = np.exp(loga - loga.max())
    elif mode == 5:
        k = int(rng.integers(max(3, n // 100), max(5, n // 50)))
        k = k if k % 2 == 1 else k + 1
        k = min(k, n if n % 2 == 1 else n - 1)
        sig = max(1.0, k / 4.0)
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode="same")
    elif mode == 6:
        perm = rng.permutation(n)
        q = a[perm].copy()
        q = (1.0 - 0.5) * a + 0.5 * q
    else:
        half = n // 2
        q = np.concatenate([a[half:], a[:half]])

    return normalize_a(q)


def zero_polish(a, deadline, max_rounds=3):
    """
    Try zeroing out the smallest elements to see if score improves.
    Shorter effective sequences can have better scores.
    """
    a = normalize_a(a).copy()
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break
        sorted_idx = np.argsort(a)
        improved = False
        for frac in (0.01, 0.03, 0.05, 0.10):
            k = int(n * frac)
            if k < 1 or k >= n - 10:
                continue
            a2 = a.copy()
            a2[sorted_idx[:k]] = 0.0
            S = a2.sum()
            if S < 1e-12:
                continue
            a2 = a2 / S
            sc = exact_score_a(a2)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                improved = True
                a = a2.copy()
        if not improved:
            break

    return best_sc, best_a

def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(normalize_a(resample_sequence(parent, n)))
        inits.append(normalize_a(resample_sequence(parent, n)[::-1]))

    x = np.linspace(0.0, 1.0, n)
    inits.append(np.ones(n, dtype=np.float64) / float(n))

    # Linear ramps
    for alpha in (0.3, 0.6, 0.9):
        inits.append(normalize_a(1.0 + alpha * (2.0 * x - 1.0)))
        inits.append(normalize_a(1.0 - alpha * (2.0 * x - 1.0)))

    # Sinusoidal
    for f in (2, 3, 5, 8, 13, 21):
        if f * 4 < n:
            vals = 1.0 + 0.4 * np.cos(2.0 * np.pi * f * x)
            inits.append(normalize_a(vals))

    # Two-peak
    for w in (0.1, 0.2, 0.3):
        vals = np.exp(-((x - 0.25) / w) ** 2) + np.exp(-((x - 0.75) / w) ** 2) + 0.01
        inits.append(normalize_a(vals))

    # Log-spectral
    Nf = 1 << (n - 1).bit_length()
    freqs = np.fft.rfftfreq(Nf) * Nf
    for power_idx in (2.0, 3.0, 4.0):
        power = 1.0 / ((np.abs(freqs) + 1.0) ** power_idx)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        vals = 1.0 + 0.5 * noise / (np.std(noise) + 1e-10)
        inits.append(normalize_a(vals))

    # Random
    for _ in range(4):
        inits.append(normalize_a(rng.random(n)))

    # Block structure
    for nb in (16, 32, 64, 128):
        if nb > n // 4:
            continue
        per = n // nb
        block_vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals[-1], n - a.size)])
        inits.append(normalize_a(a[:n]))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low-score nonnegative sequence.
    Uses EG (multiplicative weights) with LSE annealing, multi-start,
    perturbation restarts, truncation optimization, and zero-polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 128:
                parent = p
    except Exception:
        pass

    # Choose lengths to try
    lengths = []
    if parent is not None:
        plen = int(min(parent.size, 12288))
        if plen < 1024:
            plen = 1024
        lengths.append(plen)
        if plen != 8192:
            lengths.append(8192)
        if plen != 4096:
            lengths.append(4096)
    else:
        lengths = [8192, 4096]

    lengths = list(dict.fromkeys(lengths))[:3]
    print("Parent: {}, lengths: {}".format(
        "yes({})".format(parent.size) if parent is not None else "no", lengths))

    best_seq = None
    best_score = np.inf

    # Phase 1: Multi-start EG optimization (70% of budget)
    phase1_deadline = t_start + 0.70 * (budget_s - 20.0)

    # Generate and score initial sequences
    all_cands = []
    for n in lengths:
        inits = generate_diverse_inits(n, rng, parent)
        for arr in inits:
            sc = exact_score_a(arr)
            if np.isfinite(sc):
                all_cands.append((sc, arr, n))

    all_cands.sort(key=lambda x: x[0])
    print("Best initial: {:.8f} (length={})".format(all_cands[0][0], all_cands[0][2]))

    # Select top starts with length diversity
    starts = []
    used_lens = set()
    for sc, arr, n in all_cands:
        if n not in used_lens and len(starts) < 3:
            starts.append(normalize_a(arr))
            used_lens.add(n)
    while len(starts) < min(3, len(lengths)):
        starts.append(perturb(starts[0], rng, 0.5))

    # Update global best from initial candidates
    for sc, arr, n in all_cands[:10]:
        if sc < best_score:
            best_score = sc
            best_seq = normalize_a(arr)

    # Run EG on each start
    n_starts = len(starts)
    per_start_budget = (phase1_deadline - time.monotonic()) / n_starts

    for i, st in enumerate(starts):
        sub_dead = t_start + (i + 1) * per_start_budget
        sub_dead = min(sub_dead, phase1_deadline - 5.0)
        if time.monotonic() > sub_dead - 10.0:
            continue

        print("Starting EG traj {} with score {:.8f} n={}".format(i, exact_score_a(st), len(st)))
        sc, seq = eg_anneal(st, sub_dead, "T{}".format(i),
                            n_betas=10, steps_per_beta=400,
                            beta_start=2.0, beta_end=30000.0, mu=0.7)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("New global best: {:.8f} (traj {})".format(sc, i))

    # Phase 2: Fine-tuning (20% of budget)
    phase2_deadline = t_start + 0.90 * (budget_s - 20.0)
    if time.monotonic() < phase2_deadline - 30.0 and best_seq is not None:
        print("Phase 2: Fine-tuning, current best={:.8f}".format(best_score))
        sc, seq = eg_fine_tune(best_seq, phase2_deadline, "fine",
                               n_iter=8000, beta_init=5000.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After fine-tune: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts (10% of budget)
    if time.monotonic() < deadline - 30.0 and best_seq is not None:
        print("Phase 3: Perturbation restarts")
        n_restarts = 5
        rem = deadline - 30.0 - time.monotonic()
        per_restart = rem / n_restarts

        for r in range(n_restarts):
            sub_dead = time.monotonic() + per_restart
            if time.monotonic() > deadline - 30.0:
                break
            p = perturb(best_seq, rng, strength=float(rng.uniform(0.1, 0.5)))
            sc, seq = eg_anneal(p, sub_dead, "R{}".format(r),
                                n_betas=6, steps_per_beta=300,
                                beta_start=5.0, beta_end=20000.0, mu=0.6)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Restart {} improved: {:.8f}".format(r, sc))

    # Phase 4: Zero polish and truncation
    if time.monotonic() < deadline - 20.0 and best_seq is not None:
        sc, seq = zero_polish(best_seq, deadline - 10.0, max_rounds=3)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Zero polish: {:.8f}".format(best_score))

    if time.monotonic() < deadline - 20.0 and best_seq is not None:
        sc, seq = trunc_optimize(best_seq, deadline - 10.0, "trunc")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Final: one more EG pass on the best
    if time.monotonic() < deadline - 10.0 and best_seq is not None:
        rem_dead = deadline - 5.0
        sc, seq = eg_fine_tune(best_seq, rem_dead, "final", n_iter=3000, beta_init=10000.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = float(out_arr.sum())
    out_arr = out_arr / S
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
