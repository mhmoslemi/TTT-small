Key innovations:

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def convolve_xy(x, y):
    """Linear convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    M_fft = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, M_fft)
    Y = np.fft.rfft(y, M_fft)
    return np.fft.irfft(X * Y, M_fft)[:n]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lp_direction(f, M, n_constr=100, method='highs'):
    """
    Find direction b maximizing sum(b) subject to (f*b)_k <= M, b >= 0.
    Uses working set of top n_constr constraints.
    """
    n = len(f)
    # The most binding constraints are where (f*f) is largest (proxy for (f*b) when b~f)
    # But actually we need to check (f*b) at the solution
    # For the working set, use the top peaks of f*f as initial guess
    
    ff = fast_conv(f)
    top_idx = np.argsort(ff)[::-1][:n_constr]
    
    # Build constraint matrix: A[i, j] = f[top_idx[i] - j] if 0 <= top_idx[i]-j < n
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A[i, j_start:j_end+1] = f[k - j_end:k - j_start + 1][::-1]
    
    # Solve LP: maximize 1^T b subject to A b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method=method)
    if res.success and res.x is not None:
        b = np.clip(res.x, 0, None)
        if b.sum() > 1e-12:
            return b
    return None

    for _ in range(20):
        mid = (lo + hi) / 2
        f_test = (1 - mid) * f + mid * b
        sc_test = float(np.max(fast_conv(f_test)))
        if sc_test <= M * (1 + 1e-10):
            lo = mid
        else:
            hi = mid
    
    f_new = (1 - lo) * f + lo * b
    f_new = np.clip(f_new, 0, None)
    S = f_new.sum()
    if S > 1e-12:
        f_new = f_new / S
    return f_new


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=50):
    """
    Targeted coordinate descent: for each top peak contributor,
    try adjusting it while redistributing to reduce the overall peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    # Redistribute to non-peak positions
                    mask = np.ones(n, dtype=bool)
                    mask[j_start:j_end+1] = False
                    if mask.sum() > 0:
                        a[mask] += reduction / mask.sum()
                S_new = a.sum()
                if S_new > 1e-12:
                    a_norm = a / S_new
                    sc = compute_score(a_norm)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = new_val
                else:
                    a = a / S  # restore
            a[j] = best_val
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def diverse_init(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Gaussian
    inits.append(np.exp(-20 * (x - 0.5) ** 2))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # U-shaped
    inits.append(0.1 + 0.9 * (2 * x - 1) ** 2)
    # Random
    inits.append(np.clip(rng.standard_normal(n), 0.01, None))
    # Triangular
    inits.append(1 - np.abs(2 * x - 1))
    # Bell-shaped (narrower)
    inits.append(np.exp(-50 * (x - 0.5) ** 2))
    
    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.0, 1000.0)
        s = inits[i].sum()
        if s > 1e-12:
            inits[i] = inits[i] / s
    return inits

    return np.fft.irfft(X * Y, M_fft)[:n]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_topk_obj_grad(a, beta, top_k):
    """LSE-smoothed objective restricted to top_k peaks."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k < len(c):
        threshold = np.sort(c)[::-1][top_k - 1]
        c_clipped = np.minimum(c, threshold)
    else:
        c_clipped = c

    c_max = float(c_clipped.max())
    scaled = beta * (c_clipped - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    if top_k is not None:
        func = partial(lse_topk_obj_grad, beta=beta, top_k=top_k)
    else:
        func = partial(lse_obj_grad, beta=beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_lp_matrix(f, constraint_indices, n):
    """Build the LP constraint matrix for convolution constraints."""
    A = np.zeros((len(constraint_indices), n))
    for i, k in enumerate(constraint_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A[i, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_direction(f, M, constraint_indices, n):
    """Solve the LP: max 1^T b s.t. (f*b)_k <= M for k in constraint_indices, b >= 0."""
    A = build_lp_matrix(f, constraint_indices, n)
    c_obj = -np.ones(n)
    b_ub = np.full(len(constraint_indices), M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        b = np.clip(res.x, 0, None)
        if b.sum() > 1e-12:
            return b
    return None


def lp_iteration(f, n_constr=100, max_inner=5):
    """
    One LP-based refinement iteration using working set.
    Returns updated f with non-increasing peak.
    """
    n = len(f)
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S

    ff = fast_conv(f)
    M = float(np.max(ff))

    # Initial working set: top peaks
    top_idx = np.argsort(ff)[::-1][:n_constr]

    # Solve LP
    b = solve_lp_direction(f, M, top_idx, n)
    if b is None:
        return f

    # Verify all constraints, add violations if needed
    for _ in range(max_inner):
        fb = convolve_xy(f, b)
        violated = np.where(fb > M * (1.0 + 1e-9))[0]
        if len(violated) == 0:
            break
        # Add violated constraints
        all_idx = np.concatenate([top_idx, violated])
        unique_idx = np.unique(all_idx)
        # Keep most important (highest f*f values)
        if len(unique_idx) > 2 * n_constr:
            scores = ff[unique_idx]
            keep = np.argsort(scores)[::-1][:2 * n_constr]
            unique_idx = unique_idx[keep]
        b = solve_lp_direction(f, M, unique_idx, n)
        if b is None:
            return f

    if b is None or b.sum() < 1e-12:
        return f

    b = b / b.sum()

    # Bisection to find largest step that doesn't increase peak
    lo, hi = 0.0, 0.8
    for _ in range(25):
        mid = (lo + hi) / 2.0
        f_test = (1.0 - mid) * f + mid * b
        peak_test = float(np.max(fast_conv(f_test)))
        if peak_test <= M * (1.0 + 1e-10):
            lo = mid
        else:
            hi = mid

    f_new = (1.0 - lo) * f + lo * b
    f_new = np.clip(f_new, 0.0, None)
    S = f_new.sum()
    if S > 1e-12:
        f_new = f_new / S
    return f_new


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def targeted_coord_descent(a, n_sweeps=3, n_top=50):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j_start:j_end + 1] = False
                    if mask.sum() > 0:
                        a[mask] += reduction / mask.sum()
                S_new = a.sum()
                if S_new > 1e-12:
                    a_norm = a / S_new
                    sc = compute_score(a_norm)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = new_val
                a[j] = a_old_j  # restore for next trial
            # Apply best
            a[j] = best_val
            reduction = a_old_j - best_val
            if reduction > 1e-15:
                mask = np.ones(n, dtype=bool)
                mask[j_start:j_end + 1] = False
                if mask.sum() > 0:
                    a[mask] += reduction / mask.sum()
            S_new = a.sum()
            if S_new > 1e-12:
                a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multiscale_coord_descent(a, n_levels=4):
    """
    Multi-scale coordinate descent: work on coarsened version, then refine.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_sc = compute_score(a)

    # Coarsen
    a_coarse = a.copy()
    for level in range(n_levels):
        if len(a_coarse) <= 2:
            break
        # Reduce by 2
        n_c = len(a_coarse)
        n_next = n_c // 2
        a_next = np.zeros(n_next)
        for i in range(n_next):
            a_next[i] = a_coarse[2 * i] + a_coarse[2 * i + 1]
        a_coarse = a_next

    # Optimize coarse version with LSE
    sc, a_opt_coarse = run_lbfgs(a_coarse, beta=50, maxiter=5000, maxfun=20000)

    # Upsample back
    a_up = a_opt_coarse.copy()
    for level in range(n_levels):
        n_cur = len(a_up)
        n_target = n_cur * 2
        if n_target > n:
            break
        a_new = np.zeros(n_target)
        a_new[0::2] = a_up
        a_new[1::2] = a_up
        a_up = a_new
        if len(a_up) > n:
            a_up = a_up[:n]
            break
    if len(a_up) < n:
        # Pad
        a_pad = np.zeros(n)
        a_pad[:len(a_up)] = a_up
        a_up = a_pad

    a_up = np.clip(a_up, 0, None)
    S = a_up.sum()
    if S > 1e-12:
        a_up = a_up / S

    sc_up = compute_score(a_up)
    if sc_up < best_sc:
        a = a_up
    return a


def diverse_init(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Gaussian (narrow)
    inits.append(np.exp(-30 * (x - 0.5) ** 2))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # U-shaped
    inits.append(0.1 + 0.9 * (2 * x - 1) ** 2)
    # Bell (medium)
    inits.append(np.exp(-10 * (x - 0.5) ** 2))
    # Triangular
    inits.append(np.maximum(0, 1 - np.abs(2 * x - 1)))
    # Random
    inits.append(np.clip(rng.standard_normal(n), 0.01, None))
    # Two-hump
    inits.append(np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2))

    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.0, 1000.0)
        s = inits[i].sum()
        if s > 1e-12:
            inits[i] = inits[i] / s
    return inits


def perturbed(a, rng, scale=0.03):
    """Generate a perturbation of a."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    else:
        # Random subset rescale
        n_sel = max(10, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def spectral_flat_init(n, rng, n_trials=20):
    """
    Try to find a non-negative sequence with flat |DFT|^2 (white spectrum).
    This should give approximately flat autocorrelation.
    """
    best_a = None
    best_score = np.inf

    for _ in range(n_trials):
        # Random phases
        phases = rng.uniform(0, 2 * np.pi, n)
        # Random magnitude profile (try different shapes)
        shape = rng.integers(0, 3)
        x = np.linspace(0, 1, n)
        if shape == 0:
            magnitudes = np.ones(n)
        elif shape == 1:
            magnitudes = 1.0 + 0.5 * rng.random(n)
        else:
            magnitudes = np.exp(-5 * (x - 0.5) ** 2)

        # Construct signal
        spectrum = magnitudes * np.exp(1j * phases)
        # Make it conjugate-symmetric for real signal
        a_real = np.fft.irfft(spectrum, n)
        # Project to non-negative
        a_nn = np.clip(a_real, 0, None)
        S = a_nn.sum()
        if S > 1e-6:
            a_nn = a_nn / S
            sc = compute_score(a_nn)
            if sc < best_score:
                best_score = sc
                best_a = a_nn.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/(sum)^2.

    Multi-phase approach:
    1. Diverse initialization + quick LSE optimization
    2. LP-based direction finding with working set
    3. High-precision LSE L-BFGS with increasing beta
    4. Iterated local search with diverse perturbations
    5. Targeted coordinate descent + final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        try_update(sc_parent, parent_norm, "parent")
    else:
        n_work = 6962

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 1: Diverse initialization + quick LSE (10% time)
    # ==================================================================
    phase1_deadline = t_start + 0.10 * total_budget
    print(f"\n--- Phase 1: Diverse Initialization (n={n_work}) ---")

    inits = diverse_init(n_work, rng, parent)
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        sc, a_opt = run_lbfgs(a_init, beta=100, maxiter=5000, maxfun=20000)
        try_update(sc, a_opt, f"init_{i}")

    # Try spectral flat init
    if time.monotonic() < phase1_deadline - 30:
        a_spec, sc_spec = spectral_flat_init(n_work, rng, n_trials=10)
        if a_spec is not None:
            sc_opt, a_opt = run_lbfgs(a_spec, beta=200, maxiter=10000, maxfun=30000)
            try_update(sc_opt, a_opt, "spectral_flat")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: LP-based refinement (30% time)
    # ==================================================================
    phase2_deadline = t_start + 0.40 * total_budget
    print(f"\n--- Phase 2: LP-based Direction Finding ---")

    a_lp = best_a.copy()
    n_lp_iters = 40
    for it in range(n_lp_iters):
        if time.monotonic() > phase2_deadline - 15:
            break
        t_before = time.monotonic()

        a_lp = lp_iteration(a_lp, n_constr=80, max_inner=3)
        sc = compute_score(a_lp)
        improved = try_update(sc, a_lp, f"LP iter {it}")

        t_elapsed = time.monotonic() - t_before
        speedup = ""
        if improved:
            speedup = " *"

        # Periodic LSE polishing every 5 iterations
        if it % 5 == 4 and time.monotonic() < phase2_deadline - 30:
            sc, a_opt = run_lbfgs(a_lp, beta=500, maxiter=15000, maxfun=40000)
            try_update(sc, a_opt, f"LP+LSE@{it}")
            a_lp = best_a.copy()

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: High-precision LSE L-BFGS (25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.65 * total_budget
    print(f"\n--- Phase 3: High-Precision LSE L-BFGS ---")

    beta_schedule = [200, 500, 1000, 2000, 5000, 10000, 30000, 80000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 10:
            break
        remaining = phase3_deadline - time.monotonic()
        maxiter = int(min(40000, max(5000, remaining * 20)))
        maxfun = int(min(150000, max(10000, remaining * 80)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(sc, a_opt, f"LSE beta={beta}")

        # Also try top-k variant for some betas
        if beta in [1000, 10000] and time.monotonic() < phase3_deadline - 30:
            for tk in [3, 5]:
                sc_k, a_k = run_lbfgs(best_a, beta, maxiter=maxiter // 2,
                                       maxfun=maxfun // 2, top_k=tk)
                try_update(sc_k, a_k, f"LSE beta={beta} topk={tk}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated local search (20% time)
    # ==================================================================
    phase4_deadline = t_start + 0.85 * total_budget
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 0
    stagnation_count = 0
    while time.monotonic() < phase4_deadline - 30:
        remaining = phase4_deadline - time.monotonic()
        if remaining < 60:
            break

        # Adaptive perturbation scale
        base_scale = 0.02 + 0.03 * rng.random()
        if stagnation_count > 3:
            base_scale *= 2.0  # Break out of local min

        a_pert = perturbed(best_a, rng, scale=base_scale)

        # Choose beta based on remaining time
        beta = 500 + int(5000 * rng.random())
        maxiter = int(min(25000, max(3000, remaining * 10)))
        maxfun = int(min(80000, max(5000, remaining * 40)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        improved = try_update(sc, a_opt, f"ILS #{n_trials}")

        if not improved:
            stagnation_count += 1
        else:
            stagnation_count = 0
            # If improved, try smaller perturbation for exploitation
            if time.monotonic() < phase4_deadline - 30:
                a_pert2 = perturbed(best_a, rng, scale=base_scale * 0.2)
                sc2, a_opt2 = run_lbfgs(a_pert2, 3000,
                                         maxiter=maxiter // 2, maxfun=maxfun // 2)
                try_update(sc2, a_opt2, f"ILS #{n_trials} exploit")

        #偶尔 try LP after ILS
        if n_trials % 5 == 4 and time.monotonic() < phase4_deadline - 60:
            a_lp2 = lp_iteration(best_a, n_constr=60, max_inner=2)
            sc_lp = compute_score(a_lp2)
            try_update(sc_lp, a_lp2, f"ILS+LP #{n_trials}")

        n_trials += 1

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final targeted methods (15% remaining)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Targeted coordinate descent
    if time.monotonic() < deadline - 60:
        a_cd = targeted_coord_descent(best_a, n_sweeps=3, n_top=40)
        sc = compute_score(a_cd)
        try_update(sc, a_cd, "coord_descent")

    # Multi-scale approach
    if time.monotonic() < deadline - 60:
        a_ms = multiscale_coord_descent(best_a, n_levels=3)
        sc = compute_score(a_ms)
        try_update(sc, a_ms, "multiscale")
        if sc < best_score:
            # Refine the multi-scale result
            sc, a_opt = run_lbfgs(a_ms, beta=2000, maxiter=20000, maxfun=60000)
            try_update(sc, a_opt, "multiscale refined")

    # Final high-beta LSE
    if time.monotonic() < deadline - 30:
        for beta in [50000, 200000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=20000, maxfun=60000)
            try_update(sc, a_opt, f"final beta={beta}")

    # Final Nesterov
    if time.monotonic() < deadline - 20:
        a_ns = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns)
        try_update(sc, a_ns, "final nesterov")

    # Final LP pass
    if time.monotonic() < deadline - 30:
        a_lp_final = lp_iteration(best_a, n_constr=120, max_inner=3)
        sc = compute_score(a_lp_final)
        try_update(sc, a_lp_final, "final LP")

    # One more round of targeted descent on the best
    if time.monotonic() < deadline - 20:
        a_final_cd = targeted_coord_descent(best_a, n_sweeps=2, n_top=30)
        sc = compute_score(a_final_cd)
        try_update(sc, a_final_cd, "final coord")

    # ==================================================================
    # Try different lengths if time permits
    # ==================================================================
    if time.monotonic() < deadline - 90 and parent is not None:
        print(f"\n--- Phase 6: Length Exploration ---")
        for n_try in [n_work // 2, int(n_work * 1.3)]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 256 or n_try > 20000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt = run_lbfgs(a_interp, 500, maxiter=15000, maxfun=50000)
            try_update(sc, a_opt, f"len={n_try}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
