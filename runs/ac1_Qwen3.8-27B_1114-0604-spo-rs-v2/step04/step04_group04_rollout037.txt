"""
Improved flat-autocorrelation optimizer with fixed FISTA gradient,
adaptive beta scheduling, warm-start LP alternating, and LSE L-BFGS polish.
Key fix: correct cross-correlation gradient in the LP sub-solver.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _self_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = _next_pow2(M)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(_self_conv(a).max()) / (S * S)


def _lp_fista(a, M_val, x_init, n_iters, beta):
    n = len(a)
    M_len = 2 * n - 1
    Mf = _next_pow2(3 * n - 2)

    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    A = np.fft.rfft(a, Mf)
    sum_a = float(a.sum())
    L = beta * sum_a * sum_a + 1e-12
    step = 1.0 / L

    for it in range(n_iters):
        X = np.fft.rfft(x, Mf)
        c = np.fft.irfft(A * X, Mf)[:M_len]

        z = beta * (c - M_val)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        z_sum = float(exp_z.sum())
        if z_sum < 1e-300:
            p = np.ones(M_len) / M_len
        else:
            p = exp_z / z_sum

        P = np.fft.rfft(p, Mf)
        cc = np.fft.irfft(np.conj(A) * P, Mf)[:n]
        grad = cc - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def _lp_alternating(a, max_outer, fista_iters, t_start, t_min, deadline, beta_start, beta_end):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = _score(a)
    x_warm = np.ones(n) / n
    no_improve = 0

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        M_val = float(c.max())

        frac = outer / max(1, max_outer - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        x = _lp_fista(a, M_val, x_warm, fista_iters, beta)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        t = max(t_min, t_start * (1.0 - 0.5 * frac))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                t = max(0.01, t * 0.5)
                a_new = (1.0 - t) * a + t * x
                a_new = np.maximum(a_new, 0.0)
                S_new = float(a_new.sum())
                if S_new > 1e-12:
                    a_new = a_new / S_new
                    sc = _score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                no_improve = 0
                if t < 0.01:
                    break

        a = a_new

    return best_a, best_sc


def _lse_score_grad(a, beta):
    n = len(a)
    M_len = 2 * n - 1
    Mf = _next_pow2(M_len)
    Mf_cc = _next_pow2(3 * n - 2)

    A = np.fft.rfft(a, Mf)
    c = np.fft.irfft(A * A, Mf)[:M_len]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    z = beta * (c - c_max)
    z_max = float(z.max())
    exp_z = np.exp(z - z_max)
    z_sum = float(exp_z.sum())
    if z_sum < 1e-300:
        L_val = c_max
        p = np.ones(M_len) / M_len
    else:
        L_val = c_max + np.log(z_sum) / beta + z_max / beta
        p = exp_z / z_sum

    score = 2.0 * n * L_val / (S * S)

    A_cc = np.fft.rfft(a, Mf_cc)
    P_cc = np.fft.rfft(p, Mf_cc)
    cc = np.fft.irfft(np.conj(A_cc) * P_cc, Mf_cc)[:n]

    grad_LSE = (2.0 / beta) * cc
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_LSE / S2 - 2.0 * L_val / S3)

    return score, grad


def _lbfgs_refine(a0, beta, maxiter, maxfun, deadline):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        if time.monotonic() > deadline:
            return 1e10, np.zeros(len(x))
        return _lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 20}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = _score(a_out)
    return sc, a_out


def _error_reduction(a, n_iters, alpha, deadline):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.4 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def _subgradient_polish(a, total_iters, deadline):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        c = _self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            a_seg = a[j_start:j_end + 1]
            kc = k_star
            for jj in range(j_start, j_end + 1):
                m = kc - jj
                if 0 <= m < n:
                    sg_M[jj] += 2.0 * a[m]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min((sc - 1.0) / (gn * gn + 1e-30), 0.03)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a
