def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def q_lambda_matvec(lam, v):
    """
    Compute Q_lambda @ v where (Q_lambda v)[i] = sum_k lam[k] * v[k-i].
    This is the cross-correlation of lam (len 2n-1) and v (len n), lags 0..n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(v)
    total_len = len(lam) + n - 1
    M_fft = 1 << (total_len - 1).bit_length()
    L = np.fft.rfft(lam, M_fft)
    V = np.fft.rfft(v, M_fft)
    # cross-correlation: ifft(conj(fft(lam)) * fft(v))
    CC = np.fft.irfft(np.conj(L) * V, M_fft)
    return CC[:n]


def proj_simplex(v):
    """Project v onto {a >= 0, sum a = 1} via water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return out / s


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # dL/da_i = sum_k p_k * (dc_k/da_i) = cross_corr(p, a)[i] + p[2i]*a[i]
    total_len = len(p) + n - 1
    M_fft = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    CC = np.fft.irfft(np.conj(P) * A, M_fft)[:n]
    idx2 = 2 * np.arange(n)
    valid = idx2 < len(p)
    diag_term = np.zeros(n)
    diag_term[valid] = p[idx2[valid]] * a[valid]
    dL_da = CC + diag_term
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def dual_mw_polish(a, time_limit=300.0, n_inner_max=30, gamma_init=2.0, gamma_growth=1.05):
    """
    Multiplicative weights (dual) polish.
    Outer: update lambda_k proportional to exp(gamma * c_k / max_c)
    Inner: FISTA on min a^T Q_lambda a s.t. a>=0, sum a=1
    Returns (best_score, best_a).
    """
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    M_conv = 2 * n - 1
    lam = np.ones(M_conv) / M_conv
    gamma = gamma_init
    step = 0.4  # 1/L where L ~ 2.5 (slightly conservative)

    outer_it = 0
    while time.monotonic() - t_start < time_limit:
        outer_it += 1

        # Inner FISTA loop
        a_curr = a.copy()
        a_prev = a.copy()
        t_k = 1.0
        n_inner = min(n_inner_max, max(5, int(3.0 / (outer_it * 0.01 + 1.0))))

        for inner_it in range(n_inner):
            if time.monotonic() - t_start > time_limit:
                break
            grad = 2.0 * q_lambda_matvec(lam, a_curr)
            a_candidate = a_curr - step * grad
            a_candidate = proj_simplex(a_candidate)
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            a_next = a_candidate + (t_k - 1.0) / (t_new + 1.0) * (a_candidate - a_prev)
            a_next = proj_simplex(a_next)
            a_prev = a_curr
            a_curr = a_next
            t_k = t_new

        a = a_curr
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        # Compute convolution and score
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            c_max = 1e-15
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Update lambda via multiplicative weights
        # lambda_k proportional to exp(gamma * c_k / c_max)
        scaled_c = gamma * c / c_max
        # Clip for numerical stability

        lam_sum = lam_new.sum()
        if lam_sum < 1e-15:
            lam = np.ones(M_conv) / M_conv
        else:

        # Increase gamma slowly
        gamma *= gamma_growth

        if outer_it % 50 == 0:
            lam = 0.8 * lam + 0.2 * (1.0 / M_conv)
            lam = lam / lam.sum()

    return best_sc, best_a

        # For each active k: grad_i = a_{k-i} + a_i * 1[2i=k]
        # Sum over active k: sum_k active[k] * (a_{k-i} + a_i*1[2i=k])
        # = cross_corr(active, a)[i] + a[i] * sum_k active[k]*1[2i=k]

        total_len = (2 * n - 1) + n - 1
        M_fft = 1 << (total_len - 1).bit_length()
        Act = np.fft.rfft(active.astype(np.float64), M_fft)
        A = np.fft.rfft(a, M_fft)
        CC = np.fft.irfft(np.conj(Act) * A, M_fft)[:n]

        # Diagonal term: for each i, check if 2i is active
        idx2 = 2 * np.arange(n)
        valid = idx2 < (2 * n - 1)
        diag = np.zeros(n)
        diag[valid] = active[idx2[valid]]

        sg = (CC + diag * a) / n_active

        # Nesterov momentum update
        v = mu * v + sg
        step = step_scale / (np.sqrt(it + 1.0) + 1.0)
        a = a - step * v
        a = proj_simplex(a)

        # Adaptive step: if score is improving, keep step; if not, reduce
        c_new = fast_conv(a)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new > sc * 1.05:
            step_scale *= 0.8
            # Revert
            a = a + step * v
            a = proj_simplex(a)

    return best_sc, best_a

# Initialization builders
# ---------------------------------------------------------------------------

def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Parent (resampled if needed)
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            # Resample parent to length n
            old_x = np.linspace(0, 1, len(parent))
            new_x = np.linspace(0, 1, n)
            resampled = np.interp(new_x, old_x, parent)
            inits.append(np.clip(resampled, 0.0, 1000.0))

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Tapered (Hann window)
    inits.append(np.maximum(0.01, 0.5 * (1 + np.cos(2 * np.pi * x))))

    # Asymmetric taper
    inits.append(np.clip(0.3 + 0.5 * np.sin(np.pi * x) ** 0.5, 0.01, 1000.0))

    # Cosine modulations
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    inits.append(0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x))

    # Quadratic / parabolic
    inits.append(0.1 + 0.9 * (2 * x - 1) ** 2)
    inits.append(0.5 + 0.5 * (1 - (2 * x - 1) ** 2))

    # Ramp
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))

    # Gaussian bump
    sigma = 0.2 + 0.3 * rng.random()
    center = 0.3 + 0.4 * rng.random()
    inits.append(np.exp(-0.5 * ((x - center) / sigma) ** 2))

    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))

    # Block structure
    for nb in (4, 8):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a_blk = np.repeat(vals, per)
        if len(a_blk) < n:
            a_blk = np.concatenate([a_blk, vals[:n - len(a_blk)]])
        inits.append(a_blk[:n].copy())

    # "Anti-symmetric" structure: high on left, low on right (reduces c_{n-1})
    inits.append(np.clip(np.exp(-3.0 * x) + 0.1, 0.01, 1000.0))
    inits.append(np.clip(np.exp(3.0 * (x - 1.0)) + 0.1, 0.01, 1000.0))

    # Bimodal
    inits.append(np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2))

    return inits

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = int(min(max(parent.size, 1024), 16384))
        lengths.append(pa)
        # Also try nearby lengths
        lengths.append(max(512, pa - pa // 10))
        lengths.append(min(16384, pa + pa // 10))
    for n in [1024, 2048, 4096, 8192]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit to 3-4 lengths
    lengths = lengths[:4]

    print(f"Lengths: {lengths}, Parent: {parent is not None}")

    # ==================================================================
    # Phase 1: L-BFGS continuation (55% of time)
    # ==================================================================
    phase1_end = t_start + 0.55 * (total_budget - 15.0)
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 600, 1200, 2500, 5000]

    print("\n=== Phase 1: L-BFGS Continuation ===")
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  -- n = {n} --")
        inits = build_inits(n, rng, parent)

        # Limit inits based on time
        elapsed = time.monotonic() - t_start
        remaining = phase1_end - time.monotonic()
        max_inits = max(3, min(len(inits), int(remaining / 8.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_cur = a0.copy()
            t0 = time.monotonic()

            for beta in beta_schedule:
                if time.monotonic() > phase1_end:
                    break
                # Adaptive maxiter
                rem = phase1_end - time.monotonic()
                mi = max(200, min(5000, int(rem / 3.0)))
                mf = min(100000, mi * 30)

                sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=mi, maxfun=mf)
                a_cur = a_opt
                if i == 0 or beta in [40, 320, 2500]:
                    report(sc, a_opt, f"n={n},init={i},b={beta}")

            elapsed_i = time.monotonic() - t0
            if i < 3:
                print(f"    init {i}: {elapsed_i:.1f}s, score={compute_score(a_cur):.6f}")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Dual MW polish (25% of time)
    # ==================================================================
    phase2_end = t_start + 0.80 * (total_budget - 15.0)
    print(f"\n=== Phase 2: Dual MW Polish ===")

    remaining2 = phase2_end - time.monotonic()
    if remaining2 > 30:
        # Run dual polish on best solution
        sc, a_polished = dual_mw_polish(best_a, time_limit=remaining2 * 0.7,
                                         n_inner_max=25, gamma_init=3.0, gamma_growth=1.03)
        report(sc, a_polished, "dual MW")

        # Also try a second dual run with different params
        if time.monotonic() < phase2_end - 10:
            sc2, a2 = dual_mw_polish(best_a, time_limit=remaining2 * 0.2,
                                      n_inner_max=15, gamma_init=5.0, gamma_growth=1.08)
            report(sc2, a2, "dual MW v2")

    # ==================================================================
    # Phase 3: Subgradient momentum (10% of time)
    # ==================================================================
    phase3_end = t_start + 0.90 * (total_budget - 15.0)
    print(f"\n=== Phase 3: Subgradient Momentum ===")

    remaining3 = phase3_end - time.monotonic()
    if remaining3 > 15:
        sc, a_sg = subgrad_momentum(best_a, time_limit=remaining3, n_iters=20000)
        report(sc, a_sg, "subgrad")

    # ==================================================================
    # Phase 4: Final L-BFGS polish (10% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    remaining4 = deadline - time.monotonic()
    betas_final = [2000, 5000, 10000]
    for beta in betas_final:
        if time.monotonic() > deadline - 5:
            break
        rem = deadline - time.monotonic()
        mi = max(1000, min(10000, int(rem / 2.0)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=mi, maxfun=min(200000, mi * 20))
        report(sc, a_opt, f"final b={beta}")

    # One more subgradient round on the final best
    if time.monotonic() < deadline - 5:
        sc, a_final = subgrad_momentum(best_a, time_limit=min(20.0, deadline - time.monotonic()), n_iters=5000)
        report(sc, a_final, "final subgrad")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
