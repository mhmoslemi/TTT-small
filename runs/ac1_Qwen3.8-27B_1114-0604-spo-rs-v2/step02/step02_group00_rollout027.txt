def fast_conv(a):

            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Pairs swap: reduce high elements, increase low ones
            a_pert = best_a.copy()
            thresh = np.median(a_pert)
            high_mask = a_pert > thresh
            low_mask = ~high_mask
            if high_mask.sum() > 0 and low_mask.sum() > 0:
                transfer = 0.03 * a_pert[high_mask].sum() / high_mask.sum()
                a_pert[high_mask] = a_pert[high_mask] - transfer
                a_pert[low_mask] = a_pert[low_mask] + transfer * high_mask.sum() / low_mask.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Frequency domain perturbation
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.fft(best_a, L)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            # Perturb magnitudes slightly
            A_mag_new = A_mag * (1.0 + 0.02 * rng.standard_normal(L))
            A_mag_new = np.clip(A_mag_new, 0, None)
            A_new = A_mag_new * np.exp(1j * A_phase)
            a_pert = np.fft.ifft(A_new).real[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LSE optimization from perturbed start
        a_opt = run_lbfgs(a_pert, lse_obj, (2000,), maxiter=500, maxfun=10000)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"restart #{restart_count} (strat={strategy})")
        restart_count += 1

    # ===================================================================
    # Phase 7: Final polish
    # ===================================================================
    print("\n=== Phase 7: Final Polish ===")
    if time.monotonic() < deadline:
        # One more LSE at very high beta
        a_final = run_lbfgs(best_a, lse_obj, (10000,), maxiter=1000, maxfun=20000)
        sc = compute_score(a_final)
        report(sc, a_final, "final LSE")

        # Subgradient polish
        sc, a_polished = peak_shave(best_a, iters=1000)
        report(sc, a_polished, "final subgrad")

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Verify with the actual evaluation function
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
