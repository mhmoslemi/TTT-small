LP-guided optimization for non-negative sequences minimizing

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    la_l = len(a)
    lb_l = len(b)
    if la_l * lb_l <= 2000000:
        return np.convolve(a, b).astype(float)
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _conv_self(a):
    n = len(a)
    if n * n <= 2000000:
        return np.convolve(a, a).astype(float)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A * A, L)
    return out[:2*n-1].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_self(p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _lp_direction(a, M, active_set=None, max_iters=50000):
    """
    Solve LP: max sum(d) s.t. (a*d)_k <= M for k in active_set, d >= 0.
    Returns the optimal d (unnormalized) or None if infeasible.
    """
    n = len(a)
    n_c = 2 * n - 1
    
    if active_set is None:
        active_set = np.arange(n_c)
    
    m = len(active_set)
    
    # Build constraint matrix: (a*d)_k for k in active_set
    # (a*d)_k = sum_i a[i] * d[k-i] (with 0 <= i < n, 0 <= k-i < n)
    # For a given k, d[j] appears with coefficient a[k-j] where j = k-i, i.e., i = k-j
    # So coefficient of d[j] in row k is a[k-j] if 0 <= k-j < n, else 0.
    
    # This is a Toeplitz-like matrix. Let me build it.
    # Row k (0-indexed in active_set), column j (0-indexed in d):
    # coeff = a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build the matrix explicitly (it's sparse but scipy linprog 
    # doesn't handle sparse well with all methods, so let's just do it for moderate n)
    
    if n * m > 50000000:  # Too large, use approximation
        return None
    
    A_ub = np.zeros((m, n), dtype=float)
    for idx, k in enumerate(active_set):
        # d[j] coefficient: a[k-j] for j where 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = a[k - j]
    
    # LP: max 1^T d s.t. A_ub @ d <= M, d >= 0
    # scipy linprog minimizes, so negate: min -1^T d
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x
            s = d.sum()
            if s > 1e-10:
                return d
            else:
                return None
        else:
            return None
    except Exception:
        return None

def _gradient_descent(a0, time_budget, beta=100.0, verbose=False):
    """
    Multiplicative gradient descent on the non-negative simplex.
    """
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    alpha = 0.1
    
    while time.monotonic() < deadline:
        c = _conv_self(a)
        d = n * c  # "scaled" convolution
        cmax = c.max()
        
        if cmax < 1e-15:
            break
        
        # Soft-max weights
        shift = d.max()
        w = np.exp(beta * (d - shift))
        ws = w.sum()
        if ws < 1e-300 or not np.isfinite(ws):
            w = np.ones(len(d)) / len(d)
        else:
            w = w / ws
        
        # Gradient: h[j] = 2 * sum_m w[m] * a[m-j] (cross-correlation)
        # Use FFT
        L = _next_pow2(3 * n)
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a, L)
        R = np.fft.irfft(W * np.conj(A), L)
        h = 2.0 * R[:n]
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 1e-30)
        
        h_max = h.max()
        if h_max < 1e-15:
            stall += 1
            continue
        h = h / h_max
        
        # Try multiple step sizes
        improved = False
        for a_try in (alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0):
            a_try = float(np.clip(a_try, 0.002, 3.0))
            a_new = a * np.exp(-a_try * h)
            a_new = np.maximum(a_new, 1e-20)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = _score(a_new)
            if sc < cur_score - 1e-14:
                a = a_new
                cur_score = sc
                improved = True
                break
        
        if improved:
            stall = 0
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose:
                    print(f"      GD: score={best_score:.6f}")
        else:
            stall += 1
            if stall > 5:
                alpha = max(0.005, alpha * 0.7)
            if stall > 30:
                break
        
        # Adapt beta
        if stall == 0 and np.random.random() < 0.1:
            beta = min(5000.0, beta * 1.2)

def _block_refine(a0, time_budget, verbose=False):

    a = a0.copy()
    n = len(a)
    best = a.copy()

        r = rng.random()
        if r < 0.3:
            bs = int(rng.integers(1, 6))
        elif r < 0.7:
            bs = int(rng.integers(max(2, n // 128), max(4, n // 16)))
        else:
            bs = int(rng.integers(max(8, n // 16), max(16, n // 2)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        for f in factors:
            a_try = a.copy()
            a_try[start:start+bs] *= f
            s = a_try.sum()
            if s < 1e-10:

            sc = _score(a_try)

def _resample(p_in, n_new):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    if n_old == 1:
        return np.ones(n_new, dtype=float) / n_new
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s
