"""
Step-function optimizer using FISTA (accelerated projected subgradient)
with active-set refinement and coordinate polish.
Minimizes score(a) = 2n * max_k conv(a,a)_k / (sum a)^2.
"""

import numpy as np
import time

def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()

def _autocorr(a):
    n = a.size
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(a, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    return out

def _grad_conv_k(a, k):
    """Gradient of conv[k] w.r.t. a. conv_k = sum_i a[i]*a[k-i] (valid)."""
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        # For j in [lo, hi]: grad[j] = 2 * a[k-j]
        # k-j ranges from k-hi to k-lo
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad

def _logsumexp_grad(a, T, topK=20):
    """
    Compute gradient of (1/T) * log(sum_k exp(T * conv_k)) at a.
    Uses only top-K lags for efficiency.
    """
    n = a.size
    conv = _autocorr(a)
    L = conv.size
    # Find top-K lags
    if topK >= L:
        active = np.arange(L)
    else:
        active = np.argpartition(conv, L - topK)[L - topK:]
    # Compute weights
    conv_active = conv[active]
    max_c = conv_active.max()
    exps = np.exp(T * (conv_active - max_c))
    p = exps / exps.sum()
    # Weighted sum of gradients
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(len(active)):
        if p[idx] > 1e-10:
            grad += p[idx] * _grad_conv_k(a, int(active[idx]))
    return grad, conv

def _fista(a0, budget_s, T_init=100.0, T_final=1.0, topK=20):
    """
    FISTA optimization: minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a0 = _normalize(a0)
    n = a0.size
    deadline = time.monotonic() + budget_s
    
    x = a0.copy()
    y = a0.copy()
    t = 1.0
    L = 2.0  # Lipschitz constant estimate
    alpha = 1.0 / L
    T = T_init
    
    best_x = x.copy()
    best_score = 2.0 * n * float(_autocorr(x).max())
    
    it = 0
    n_iter = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        it += 1
        n_iter += 1
        
        # Compute gradient at y
        grad, conv_y = _logsumexp_grad(y, T, topK)
        
        # Step
        z = y - alpha * grad
        x_new = _project_simplex(z)
        x_new = _normalize(x_new)  # Ensure exact normalization
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _normalize(y)  # Keep y in the feasible set
        t = t_new
        
        x = x_new
        
        # Update score
        conv_x = _autocorr(x)
        score = 2.0 * n * float(conv_x.max())
        if score < best_score:
            best_score = score
            best_x = x.copy()
        
        # Anneal T
        frac = (time.monotonic() - start_time) / budget_s
        T = T_init * (T_final / T_init) ** min(1.0, frac * 1.5)
        
        # Adaptive step size
        # If the step doesn't improve, reduce alpha
        if it % 100 == 0:
            # Check if we're making progress
            pass
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  FISTA iter={it} elapsed={elapsed:.1f}s best_score={best_score:.8f}")
    
    return best_x, best_score

                    # Quick update

"""
Step-function optimizer using FISTA (accelerated projected subgradient with log-sum-exp
smoothing), efficient 2-coordinate polish with incremental convolution updates,
Fourier subspace optimization, and multiple diverse starting points.
Minimizes score(a) = 2n * max_k conv(a,a)_k / (sum a)^2 over non-negative sequences.
"""

import numpy as np
import time
import math


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(a, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    return out


def _score_normalized(a):
    c = _autocorr(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * c.max())


def _grad_conv_k(a, k):
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad


def _soft_gradient(a, T, topK):
    conv = _autocorr(a)
    L = conv.size
    n = a.size
    if topK >= L:
        active = np.arange(L)
    else:
        active = np.argpartition(conv, L - topK)[L - topK:]
    conv_a = conv[active]
    max_c = conv_a.max()
    exps = np.exp(T * (conv_a - max_c))
    p = exps / float(exps.sum())
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(len(active)):
        if p[idx] > 1e-12:
            grad += p[idx] * _grad_conv_k(a, int(active[idx]))
    return grad, conv


def _fista_optimize(a0, budget_s, T_init=50.0, T_min=0.5, topK=30):
    a0 = _normalize(a0)
    n = a0.size

    best_sc = _score_normalized(x)

    stall = 0
    stall_thresh = 1e-12

    while time.monotonic() < deadline:
        it += 1
        grad, _ = _soft_gradient(y, T, topK)
        z = y - alpha * grad
        x_new = _project_simplex(z)
        x_new = _normalize(x_new)

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _normalize(y)
        t = t_new

        x = x_new
        sc = _score_normalized(x)

            stall = 0

            stall = 0

def _transfer_delta(a, i, j, dv):
    n = a.size
    L = 2 * n - 1
    delta = np.zeros(L, dtype=np.float64)
    for k in range(L):
        v1 = 0.0
        v2 = 0.0
        ki = k - i
        if 0 <= ki < n:
            v1 += a[ki]
        ki2 = i - k
        if 0 <= ki2 < n:
            v1 += a[ki2]
        kj = k - j
        if 0 <= kj < n:
            v2 += a[kj]
        kj2 = j - k
        if 0 <= kj2 < n:
            v2 += a[kj2]
        delta[k] = dv * (v2 - v1)
    delta[2 * i] += dv * dv
    delta[2 * j] += dv * dv
    delta[i + j] -= 2.0 * dv * dv
    return delta


def _transfer_delta_fast(a, i, j, dv, buf):
    n = a.size
    L = 2 * n - 1
    buf.fill(0.0)
    # -dv * (a[k-i] + a[i-k])
    lo_i = max(0, i - (n - 1))
    hi_i = min(2 * n - 2, i + (n - 1))
    for k in range(lo_i, hi_i + 1):
        ki = k - i
        if 0 <= ki < n:
            buf[k] -= dv * a[ki]
        ki2 = i - k
        if 0 <= ki2 < n:
            buf[k] -= dv * a[ki2]
    # +dv * (a[k-j] + a[j-k])
    lo_j = max(0, j - (n - 1))
    hi_j = min(2 * n - 2, j + (n - 1))
    for k in range(lo_j, hi_j + 1):
        kj = k - j
        if 0 <= kj < n:
            buf[k] += dv * a[kj]
        kj2 = j - k
        if 0 <= kj2 < n:
            buf[k] += dv * a[kj2]
    # Quadratic terms
    buf[2 * i] += dv * dv
    buf[2 * j] += dv * dv
    buf[i + j] -= 2.0 * dv * dv
    return buf

def _fourier_optimize(a0, n_fourier, budget_s, rng):
    """Project onto first K Fourier modes, optimize coefficients, project back."""
    a0 = _normalize(a0)
    n = a0.size
    deadline = time.monotonic() + budget_s

    # Project to Fourier subspace
    freqs = np.fft.rfftfreq(n)
    # Keep DC + first K positive freqs
    K = min(n_fourier, n // 2)
    mask = np.zeros(n // 2 + 1, dtype=bool)
    mask[0] = True
    mask[1:K + 1] = True
    A = np.fft.rfft(a0)
    A[~mask] = 0.0
    a_fourier = np.fft.irfft(A, n)
    a_fourier = _normalize(a_fourier)

    best_a = a_fourier.copy()
    best_sc = _score_normalized(best_a)

    # Optimize in the Fourier subspace using gradient descent
    # Parameters: magnitudes and phases of the kept Fourier coefficients
    kept_idx = np.where(mask)[0]
    mags = np.abs(A[kept_idx]).copy()
    phases = np.angle(A[kept_idx]).copy()
    # Fix DC
    mags[0] = float(a0.sum())
    phases[0] = 0.0

    alpha = 0.02
    for it in range(500):
        if time.monotonic() > deadline:
            break
        # Reconstruct
        A_cur = np.zeros(n // 2 + 1, dtype=np.complex128)
        for ii, idx in enumerate(kept_idx):
            A_cur[idx] = mags[ii] * np.exp(1j * phases[ii])
        a_cur = np.fft.irfft(A_cur, n)
        a_cur = np.maximum(a_cur, 0.0)
        s = a_cur.sum()
        if s < 1e-10:
            break
        a_cur /= s

        sc = _score_normalized(a_cur)
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()

        # Gradient step on magnitudes (keep phases fixed for simplicity)
        conv = _autocorr(a_cur)
        topK = min(len(conv), 15)
        active = np.argpartition(conv, len(conv) - topK)[len(conv) - topK:]
        grad_a = np.zeros(n, dtype=np.float64)
        for k in active:
            grad_a += _grad_conv_k(a_cur, int(k))
        grad_a /= len(active)

        # Project gradient to Fourier subspace
        G = np.fft.rfft(grad_a)
        G[~mask] = 0.0
        G[0] = 0.0  # Don't change DC (preserves normalization)
        a_grad = np.fft.irfft(G, n)

        mags[1:] = np.maximum(0.1, mags[1:] - alpha * np.abs(a_grad[kept_idx[1:]]))
        alpha *= 0.995

    return _normalize(best_a), best_sc


def _beta_seq(n, alpha_p, beta_p):
    x = np.linspace(0.0, 1.0, n)
    f = (x + 1e-6) ** alpha_p * (1.0 - x + 1e-6) ** beta_p
    return _normalize(f)

            fa, sc_a = _fista_optimize(s, alloc, T_init=50.0, T_min=0.5, topK=30)
            if sc_a < best_score:
                best_score = sc_a
                best_f = fa.copy()
                print(f"    New best: {best_score:.8f}")

    # Phase 1b: Fourier subspace optimization
    t1 = time.monotonic()
    fourier_time = min(200.0, total_deadline - t1 - 400)
    if fourier_time > 50:
        print(f"\n=== Phase 1b: Fourier subspace (budget={fourier_time:.0f}s) ===")
        for K in (50, 100, 200, 500):
            if time.monotonic() > total_deadline - 300:
                break
            alloc = min(fourier_time / 4.0, total_deadline - time.monotonic() - 250)
            if alloc < 10:
                break
            print(f"  Fourier K={K} (alloc={alloc:.0f}s)")
            ff, sc_f = _fourier_optimize(best_f, K, alloc, rng)
            if sc_f < best_score:
                best_score = sc_f
                best_f = ff.copy()
                print(f"    New best: {best_score:.8f}")

    # Phase 2: Coordinate polish
    t2 = time.monotonic()
    coord_time = min(800.0, total_deadline - t2 - 200)
    if coord_time > 30:
        print(f"\n=== Phase 2: Coordinate polish (budget={coord_time:.0f}s) ===")
        pc, sc_c, _ = _coordinate_polish(best_f, coord_time, rng)
        if sc_c < best_score:
            best_score = sc_c
            best_f = pc.copy()
            print(f"  After polish: {best_score:.8f}")

    # Phase 3: Second round FISTA from polished solution
    t3 = time.monotonic()
    fista2_time = min(300.0, total_deadline - t3 - 50)
    if fista2_time > 30:
        print(f"\n=== Phase 3: FISTA round 2 (budget={fista2_time:.0f}s) ===")
        fa2, sc_a2 = _fista_optimize(best_f, fista2_time, T_init=20.0, T_min=0.3, topK=40)
        if sc_a2 < best_score:
            best_score = sc_a2
            best_f = fa2.copy()
            print(f"  New best: {best_score:.8f}")

    # Phase 4: Final coordinate polish
    t4 = time.monotonic()
    coord2_time = max(0.0, total_deadline - t4 - 20)
    if coord2_time > 20:
        print(f"\n=== Phase 4: Final polish (budget={coord2_time:.0f}s) ===")
        pf, sc_f, _ = _coordinate_polish(best_f, coord2_time, rng)
        if sc_f < best_score:
            best_score = sc_f
            best_f = pf.copy()
            print(f"  Final best: {best_score:.8f}")

    # Final: try perturbing and polishing a few times if time remains
    t5 = time.monotonic()
    extra = total_deadline - t5 - 15
    if extra > 60:
        print(f"\n=== Extra: perturb + polish (budget={extra:.0f}s) ===")
        n_tries = min(5, int(extra // 80))
        per_try = extra / (n_tries + 1)
        for trial in range(n_tries):
            if time.monotonic() > total_deadline - 20:
                break
            sigma = 0.05 + 0.05 * trial
            pert = _normalize(best_f * np.exp(sigma * rng.standard_normal(best_f.size)))
            alloc = min(per_try, total_deadline - time.monotonic() - 15)
            if alloc < 30:
                break
            pp, sc_p, _ = _coordinate_polish(pert, alloc, rng)
            if sc_p < best_score:
                best_score = sc_p
                best_f = pp.copy()
                print(f"  Trial {trial}: {best_score:.8f}")

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_score = _score_normalized(best_f)

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    out = [float(x) for x in best_f]
    print(f"\nFINAL score = {best_score:.8f} (target: 1.5027)")
    return out
