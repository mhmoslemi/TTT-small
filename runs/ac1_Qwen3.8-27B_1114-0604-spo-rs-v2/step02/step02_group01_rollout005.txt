import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _gradient_topk(f, c, K=10):
    """Compute gradient targeting top-K peaks."""
    n = len(f)
    # Get top K lags
    top_k = np.argpartition(c, -K)[-K:]
    top_k = top_k[np.argsort(c[top_k])[::-1]]
    top_vals = c[top_k]
    
    # Weights: log-sum-exp style
    T = 10.0
    w = np.exp(T * (top_vals - top_vals[0]))
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k, wi in zip(top_k, w):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * f[idx]
    return grad, top_k

def _mirror_step(f, grad, step):
    """Exponentiated gradient step."""
    span = grad.max() - grad.min()
    if span < 1e-30:
        return f
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _lp_direction(f, c, M, K=50):
    """Solve LP to find optimal direction g."""
    n = len(f)
    # Get top K constraint lags
    top_k = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix row by row
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    for k in top_k:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row = np.zeros(n)
            idx = k - np.arange(j_start, j_end + 1)
            row[j_start:j_end + 1] = f[idx]
            rows.append(row)
    
    if not rows:
        return np.ones(n) / n
    
    A = np.array(rows)  # shape (K, n)
    b = np.full(len(rows), M)
    
    # Maximize sum(g) = minimize -sum(g)

    """Compute gradient = 2 * (w cross-corr f) for full log-sum-exp weights."""
    n = len(f)
    if N is None:
        N = 1 << (3 * n - 2).bit_length()
    W = np.fft.fft(w, N)
    F = np.fft.fft(f, N)
    corr = np.fft.ifft(W * np.conj(F)).real[:n]
