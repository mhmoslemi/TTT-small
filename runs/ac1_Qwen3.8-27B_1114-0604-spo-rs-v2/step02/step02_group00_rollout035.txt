Hybrid Adam (softmax parametrization) + L-BFGS + coordinate descent
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (direct, not through softmax)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())

    rev_a = a[::-1]

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_score_grad_x(x, beta):
    """LSE-smoothed score and gradient w.r.t. x (softmax parametrization)."""
    a = softmax(x)
    score, grad_a = lse_score_grad_a(a, beta)
    # Chain rule: grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    d = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - d)
    return score, grad_x


def run_lbfgs_x(x0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in x-space (softmax parametrization)."""
    res = minimize(
        lse_score_grad_x, x0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = softmax(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.x


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def adam_softmax(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                 time_limit_s=None, t_start=None, verbose_interval=5000):
    """
    Adam optimizer in softmax x-space with LSE-smoothed gradient.
    """
    n = len(a_init)
    # Initialize x from a_init
    a_init = a_init.astype(np.float64).copy()
    a_init = np.clip(a_init, 1e-10, 1e10)
    x = np.log(a_init) - np.log(a_init).mean()  # center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Adaptive learning rate (decay)
        lr_t = lr / (1.0 + 0.0001 * t)
        
        score, grad_x = lse_score_grad_x(x, beta_lse)
        
        m = b1 * m + (1 - b1) * grad_x
        v = b2 * v + (1 - b2) * grad_x * grad_x
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if t % verbose_interval == 0:
            a_cur = softmax(x)
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"    Adam step {t}: score = {sc:.8f}")
    
    a_final = softmax(x)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_score, best_a

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using FFT-based interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    # Use linear interpolation
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

"""
Hybrid Adam (direct a-space) + L-BFGS + selective coordinate descent
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses LSE-smoothed gradients for Adam, exact subgradient for coord descent,
and L-BFGS-B for final polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def multiple_subgrad_score_a(a, top_k=5):
    """Score and averaged subgradient over top-k convolution lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    # Use top-k lags for a "smoothed" subgradient
    sorted_idx = np.argsort(c)[::-1][:top_k]
    weights = c[sorted_idx] - c.max() + 1.0  # Ensure positive
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for k in sorted_idx:
        js = max(0, int(k) - n + 1)
        je = min(int(k), n - 1)
        if je >= js:
            idx = int(k) - np.arange(js, je + 1)
            wg = weights[sorted_idx == int(k)][0] if len(np.where(sorted_idx == int(k))[0]) > 0 else 0.0
            # Find the weight for this k
            pos = np.where(sorted_idx == int(k))[0]
            if len(pos) > 0:
                w = weights[pos[0]]
                sg[js:je + 1] += 2.0 * w * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with bounds and LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_score_grad_x(x, beta):
    """LSE-smoothed score and gradient w.r.t. x (softmax parametrization)."""
    a = softmax_fn(x)
    score, grad_a = lse_score_grad_a(a, beta)
    d = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - d)
    return score, grad_x


def run_lbfgs_x(x0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in x-space (softmax parametrization)."""
    res = minimize(
        lse_score_grad_x, x0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = softmax_fn(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.x


def adam_direct(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                time_limit_s=None, t_start=None, normalize=True,
                verbose_interval=20000):
    """
    Adam optimizer in a-space with LSE-smoothed gradient.
    Optionally normalizes to sum=1 after each step.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        
        score, grad = lse_score_grad_a(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        if t % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    Adam t={t}: score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def adam_softmax_space(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                       time_limit_s=None, t_start=None, verbose_interval=20000):
    """Adam in x-space (softmax) with LSE-smoothed gradient."""
    n = len(a_init)
    a_init_safe = np.clip(a_init, 1e-10, 1e10)
    x = np.log(a_init_safe) - np.log(a_init_safe).mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a_init_safe.copy()
    best_score = compute_score(a_init_safe)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        
        score, grad_x = lse_score_grad_x(x, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad_x
        v = b2 * v + (1.0 - b2) * grad_x * grad_x
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if t % verbose_interval == 0:
            a_cur = softmax_fn(x)
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"    AdamX t={t}: score={sc:.8f}")
    
    a_final = softmax_fn(x)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_score, best_a


def selective_coord_descent(a, top_k=30, n_sweeps=10, time_limit_s=None, t_start=None):
    """
    Coordinate descent optimizing top-K coordinates per sweep.
    Uses exact score evaluation for candidate selection.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        score, grad = subgrad_score_a(a)
        if score < best_score:
            best_score = score
        
        # Select top-k coordinates by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            
            # Generate candidates based on gradient direction
            if grad[j] > 1e-10:
                # Positive gradient: reducing a[j] helps
                ratios = np.array([0.2, 0.4, 0.6, 0.75, 0.85, 0.95, 1.0, 1.05])
                candidates = a_old * ratios
            elif grad[j] < -1e-10:
                # Negative gradient: increasing a[j] helps
                ratios = np.array([0.5, 0.8, 1.0, 1.1, 1.3, 1.5, 2.0, 3.0])
                candidates = a_old * ratios
                candidates = np.concatenate([[0.0], candidates])
            else:
                candidates = np.array([a_old])
            
            candidates = np.clip(candidates, 0.0, 1000.0)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def nesterov_subgrad(a_init, lr_init=0.01, n_steps=20000, time_limit_s=None, t_start=None):
    """Nesterov-accelerated subgradient descent (exact, not smoothed)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        momentum = 2.0 / (t + 2.0)
        lr_t = lr_init * momentum
        
        score, grad = subgrad_score_a(y)
        
        a_new = y - lr_t * grad
        a_new = np.clip(a_new, 0.0, 1e10)
        
        y = a_new + (momentum - 1.0) * (a_new - a)
        y = np.clip(y, 0.0, 1e10)
        a = a_new
        
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid optimizer: Adam (LSE-smoothed) + Nesterov + L-BFGS + Coord Descent.
    Multi-phase approach with time management.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 15.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) * 0.5
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")
    
    # Track best
    best_score = parent_score
    best_a = parent.copy()
    
    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    # ==================================================================
    # Phase 1: Adam direct (normalized) - 35% of time
    # ==================================================================
    p1 = effective_budget * 0.35
    print(f"\n{'='*50}")
    print(f"Phase 1: Adam direct (budget={p1:.0f}s)")
    print(f"{'='*50}")
    
    # Normalize parent to sum=1 for better conditioning
    a_norm = parent / parent.sum()
    
    # Multi-beta continuation
    for beta, lr in [(30, 0.01), (100, 0.005), (300, 0.002)]:
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 5:
            break
        time_alloc = remaining * 0.4
        sc, a_opt = adam_direct(a_norm, lr=lr, n_steps=200000, beta_lse=beta,
                               time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam direct beta={beta}")
        a_norm = a_opt
    
    # ==================================================================
    # Phase 2: Adam in softmax space - 15% of time
    # ==================================================================
    p2 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 2: Adam softmax-space (budget={p2:.0f}s)")
    print(f"{'='*50}")
    
    for beta, lr in [(100, 0.01), (300, 0.005)]:
        remaining = p2 - (time.monotonic() - t_start)
        if remaining < 5:
            break
        time_alloc = remaining * 0.45
        sc, a_opt = adam_softmax_space(best_a, lr=lr, n_steps=200000, beta_lse=beta,
                                       time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam softmax beta={beta}")
    
    # ==================================================================
    # Phase 3: L-BFGS polish - 20% of time
    # ==================================================================
    p3 = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS Polish (budget={p3:.0f}s)")
    print(f"{'='*50}")
    
    for beta in [300, 800, 1500, 3000]:
        remaining = p3 - (time.monotonic() - t_start)
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=100000)
        try_update(sc, a_opt, f"L-BFGS-a beta={beta}")
        print(f"    L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # L-BFGS in x-space
    if time.monotonic() - t_start < t_start + p1 + p2 + p3:
        x0 = np.log(np.clip(best_a, 1e-10, 1e10))
        x0 = x0 - x0.mean()
        for beta in [500, 1500]:
            if time.monotonic() - t_start > t_start + p1 + p2 + p3:
                break
            t0 = time.monotonic()
            sc, a_opt, x_out = run_lbfgs_x(x0, beta, maxiter=3000, maxfun=50000)
            try_update(sc, a_opt, f"L-BFGS-x beta={beta}")
            x0 = x_out
            print(f"    L-BFGS-x beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 4: Selective coordinate descent - 10% of time
    # ==================================================================
    p4 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 4: Coordinate Descent (budget={p4:.0f}s)")
    print(f"{'='*50}")
    
    t0 = time.monotonic()
    sc, a_opt = selective_coord_descent(best_a, top_k=40, n_sweeps=30,
                                       time_limit_s=p4, t_start=t_start)
    try_update(sc, a_opt, "Coord descent")
    print(f"    Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 5: Nesterov + restarts - 15% of time
    # ==================================================================
    p5 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 5: Nesterov + Restarts (budget={p5:.0f}s)")
    print(f"{'='*50}")
    
    # Nesterov subgradient
    remaining = p5 - (time.monotonic() - t_start)
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_opt = nesterov_subgrad(best_a, lr_init=0.005, n_steps=50000,
                                     time_limit_s=remaining * 0.3, t_start=t_start)
        try_update(sc, a_opt, "Nesterov")
        print(f"    Nesterov: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Restarts with perturbations
    n_best = len(best_a)
    restart_deadline = t_start + effective_budget - 30.0  # Leave time for final polish
    
    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > restart_deadline:
            break
        
        strategy = r % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Targeted reduction of peak contributors
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            # Reduce elements that contribute to peak
            for j in range(js, je + 1):
                a_pert[j] *= 0.6
            # Redistribute to non-contributing elements
            removed = float(best_a[js:je+1].sum() - a_pert[js:je+1].sum())
            mask = np.arange(n_best) < js
            mask |= np.arange(n_best) > je
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            # Multi-frequency sinusoidal
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            freq2 = 3.0 + 7.0 * rng.random()
            phase2 = 2.0 * np.pi * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + 0.03 * np.cos(2*np.pi*freq*x + phase)
                               + 0.02 * np.cos(2*np.pi*freq2*x + phase2))
        else:
            # Local block modification
            center = int(rng.integers(n_best // 4, 3 * n_best // 4))
            width = max(10, n_best // 50)
            a_pert = best_a.copy()
            a_pert[center-width:center+width] *= rng.uniform(0.3, 1.3)
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick Adam refinement
        time_left = restart_deadline - time.monotonic()
        if time_left < 5:
            break
        adam_time = min(20.0, time_left * 0.5)
        sc, a_r = adam_direct(a_pert, lr=0.003, n_steps=30000, beta_lse=200,
                              time_limit_s=adam_time, t_start=t_start)
        # Quick L-BFGS
        if time.monotonic() < restart_deadline - 3:
            sc2, a_r2, _ = run_lbfgs_a(a_r, 800, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")
    
    # ==================================================================
    # Phase 6: Try different lengths (resample parent)
    # ==================================================================
    print(f"\n{'='*50}")
    print("Phase 6: Different Lengths")
    print(f"{'='*50}")
    
    n_parent = len(best_a)
    time_left = t_start + effective_budget - time.monotonic()
    target_lengths = [int(n_parent * 0.8), int(n_parent * 1.2), int(n_parent * 1.5)]
    for tl in target_lengths:
        if time_left < 30:
            break
        if tl < 512 or tl > 30000:
            continue
        a_res = resample_sequence(best_a, tl)
        # Quick optimization
        sc, a_opt = adam_direct(a_res, lr=0.003, n_steps=50000, beta_lse=200,
                               time_limit_s=min(25.0, time_left * 0.3), t_start=t_start)
        if time.monotonic() < t_start + effective_budget - 5:
            sc2, a_opt2, _ = run_lbfgs_a(a_opt, 500, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"n={tl}")
        time_left = t_start + effective_budget - time.monotonic()
    
    # ==================================================================
    # Phase 7: Final intensive polish
    # ==================================================================
    print(f"\n{'='*50}")
    print("Phase 7: Final Polish")
    print(f"{'='*50}")
    
    final_deadline = t_start + effective_budget - 3.0
    
    # High-beta L-BFGS
    for beta in [2000, 5000]:
        if time.monotonic() > final_deadline:
            break
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"final LBFGS beta={beta}")
    
    # Final Adam refinement
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        sc, a_opt = adam_direct(best_a, lr=0.001, n_steps=100000, beta_lse=2000,
                               time_limit_s=min(30.0, final_deadline - time.monotonic()),
                               t_start=t_start)
        try_update(sc, a_opt, "final Adam")
    
    # One more coord descent
    if time.monotonic() < final_deadline:
        sc, a_opt = selective_coord_descent(best_a, top_k=50, n_sweeps=5,
                                           time_limit_s=min(20.0, final_deadline - time.monotonic()),
                                           t_start=t_start)
        try_update(sc, a_opt, "final coord")
    
    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
