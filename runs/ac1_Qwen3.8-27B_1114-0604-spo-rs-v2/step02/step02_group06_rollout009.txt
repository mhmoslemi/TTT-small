    global _RNG, _START_TIME
    _RNG = np.random.default_rng(seed)
    _START_TIME = time.monotonic()

def _remaining_time(budget):
    return max(0.0, budget - (time.monotonic() - _START_TIME))

def _conv_len(n):
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    return L

def _convolve_full(a, L=None):
    """Compute full linear convolution using FFT."""
    n = a.size
    if L is None:
        L = _conv_len(n)
    Fa = np.fft.rfft(a, L)
    c = np.fft.irfft(Fa * Fa, L)
    return c[:2*n-1]

def _score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = a.size
    c = _convolve_full(a)
    max_c = c.max()
    return 2.0 * n * max_c, c

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

def _init_ramp(n, rng):
    """Initialize with a ramp-like anti-symmetric structure."""
    t = np.arange(n, dtype=float) / n
    # Random asymmetric smooth function
    mode = rng.integers(0, 5)
    if mode == 0:
        # Sigmoid ramp
        c = rng.uniform(0.3, 0.7)
        k = rng.uniform(5, 30)
        a = 1.0 / (1.0 + np.exp(-k * (t - c)))
    elif mode == 1:
        # Power law
        p = rng.uniform(0.3, 3.0)
        a = t**p
    elif mode == 2:
        # Two bumps (asymmetric)
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        s1 = rng.uniform(0.05, 0.2)
        s2 = rng.uniform(0.05, 0.2)
        w1 = rng.uniform(0.5, 2.0)
        w2 = rng.uniform(0.5, 2.0)
        a = w1 * np.exp(-((t-c1)/s1)**2) + w2 * np.exp(-((t-c2)/s2)**2)
    elif mode == 3:
        # Cosine modulation
        f = rng.uniform(1, 5)
        a = 0.5 + 0.4 * np.cos(2*np.pi*f*t) + 0.1*rng.normal(0,1,n)
    else:
        # Linear with noise
        a = t + 0.1*rng.normal(0,1,n)
    a = np.clip(a, 0.0, None) + 1e-10
    return _normalize(a)

def _init_from_parent(parent, n, rng):
    """Resample parent to length n with small perturbation."""
    p = np.asarray(parent, dtype=float)
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return _init_ramp(n, rng)
    p = p / s
    # Resample
    old_x = (np.arange(p.size) + 0.5) / p.size
    new_x = (np.arange(n) + 0.5) / n
    a = np.interp(new_x, old_x, p)
    # Add small noise
    a = a * np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 0) + 1e-10
    return _normalize(a)

def _optimize_mma(a, time_budget, target_obj=1.50, beta_init=50, beta_max=5000):
    """
    Main optimization loop using multiplicative updates with adaptive softmax.
    """
    n = a.size
    L = _conv_len(n)
    best = a.copy()
    best_sc, _ = _score_normalized(a)
    cur_sc = best_sc
    
    beta = float(beta_init)
    alpha = 0.1
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and time_budget > 0:
        # Compute convolution
        c = _convolve_full(a, L)
        max_c = c.max()
        cur_sc = 2.0 * n * max_c
        
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Target for the convolution values
        target_m = target_obj / (2.0 * n)
        
        # Compute weights based on how much each convolution value exceeds target
        if max_c > target_m:
            # Focus on values above target
            excess = np.maximum(0, c - target_m)
            if excess.max() > 1e-15:
                w = excess / excess.max()
                w = w ** 4  # Sharpen
            else:
                # Use softmax
                d = beta * (c - c.max())
                d = d - d.max()
                w = np.exp(d)
                w = w / w.sum()
        else:
            # All below target, use softmax to find the worst
            d = beta * (c - c.max())
            d = d - d.max()
            w = np.exp(d)
            w = w / w.sum()
        
        # Compute gradient direction: h[j] = sum_m w[m] * a[m-j]
        # This is conv(w, a) truncated to length n
        # Actually: grad_j of sum_m w[m] * conv[m] = sum_m w[m] * a[m-j]
        # which is (w * a_rev)[...] hmm, let me think.
        # conv[m] = sum_k a[k]*a[m-k]
        # d(conv[m])/da[j] = a[m-j]
        # So grad_j = sum_m w[m] * a[m-j]
        # This is the cross-correlation of w and a, which is conv(w, a[::-1]) reversed
        # Or equivalently, it's the convolution of w with the reversed a
        # Let me just compute it directly using FFT
        Fw = np.fft.rfft(w, 2*L)  # Need larger FFT to avoid wrap-around
        Fa = np.fft.rfft(a, 2*L)
        # conv(w, a) has length len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2
        # But we only need the first n elements (j=0..n-1)
        # Actually I need sum_m w[m]*a[m-j] for j=0..n-1
        # This is the cross-correlation: (w ⋆ a)[j] = sum_m w[m]*a[m-j]
        # = conv(w, a_flip)[j + len(a) - 1 - 1]... this is getting confusing
        # Let me just use the fact that sum_m w[m] * a[m-j] = 
        #   the (m-j) indexing means we're computing a "sliding dot product"
        #   which is the same as conv(w, a) at specific indices
        # Actually: sum_m w[m] * a[m-j] for fixed j
        # Let k = m-j, then m = k+j
        # = sum_k w[k+j] * a[k]
        # This is the cross-correlation of w and a at lag j
        # = sum_k w[k+j]*a[k]
        # This equals conv(w_rev, a)[j] where w_rev = w[::-1]
        # Hmm, let me just compute it with a direct approach for small n,
        # or use the FFT trick.
        # 
        # Using FFT: cross-correlation of w and a = ifft(fft(w) * conj(fft(a)))
        # But w has length 2n-1 and a has length n, so I need to be careful.
        #
        # Actually, the simplest approach: h[j] = sum_m w[m]*a[m-j]
        # For j >= 0: a[m-j] is non-zero only when 0 <= m-j < n, i.e., j <= m < j+n
        # So h[j] = sum_{m=j}^{min(j+n-1, 2n-2)} w[m]*a[m-j]
        #
        # This is the same as: take the array w (length 2n-1), multiply by a padded to
        # the right, and sum... 
        #
        # Actually the cleanest way: h = ifft(fft(w, 2L) * conj(fft(a, 2L)))[:n]
        # Let me verify: ifft(fft(w)*conj(fft(a)))[j] = sum_m w[m] * a[(m-j) mod L]
        # For L >= 2n-1 + n - 1 = 3n-2, there's no wrap-around for the valid range.
        # But our w has length 2n-1 and a has length n, so the cross-correlation
        # has valid range from -(n-1) to (2n-2), which fits in L >= 3n-2.
        #
        # Actually, numpy's cross-correlation: np.correlate(w, a, mode='full')
        # gives result[i] = sum_k w[k]*a[k+i-(n-1)] ... this is getting complicated.
        #
        # Let me just use the direct formula with FFT:
        # h[j] = sum_m w[m] * a[m-j] (with a[k]=0 for k<0 or k>=n)
        # This is the "a-correlation" of w at lag j.
        # = (w * a[::-1])[j + n - 1] (in the convolution sense)
        # Let me verify: (w * a_flip)[k] = sum_p w[p] * a_flip[k-p] = sum_p w[p] * a[n-1-(k-p)]
        # = sum_p w[p] * a[n-1-k+p]
        # For k = j + n - 1: sum_p w[p] * a[n-1-j-1+n-1-... 
        # This is getting too complicated. Let me just use a simple approach.
        
        # Direct computation using the property that this is a cross-correlation:
        # h[j] = sum_m w[m]*a[m-j] = <w, shift_j(a)>
        # where shift_j(a)[m] = a[m-j] (zero-padded)
        #
        # Using FFT with zero-padding to avoid circular effects:
        N = 2 * L  # L >= 2n-1, so N >= 4n-2, which is enough
        Fw_N = np.fft.rfft(w, N)
        Fa_N = np.fft.rfft(a, N)
        # Cross-correlation: ifft(Fw * conj(Fa))
        h_full = np.fft.irfft(Fw_N * np.conj(Fa_N), N)
        # h_full[j] = sum_m w[m] * a[(m-j) mod N]
        # For j = 0, 1, ..., n-1: (m-j) mod N = m-j (since m < 2n-1 <= L < N and j < n)
        # So h_full[j] = sum_m w[m] * a[m-j] (with a[k]=0 for k<0 or k>=n)
        # This is exactly what we want!
        h = h_full[:n]
        h = np.maximum(h, 0.0)
        
        if h.max() < 1e-30:
            # Degenerate case, perturb
            a = a * np.exp(_RNG.normal(0, 0.1, n))
            a = _normalize(a)
            continue
        
        h = h / h.max()
        
        # Try multiple step sizes
        best_p = None
        best_p_sc = cur_sc
        for mult in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
            a_try = a * np.exp(-alpha * mult * h)
            a_try = np.maximum(a_try, 1e-15)
            a_try = _normalize(a_try)
            if a_try is None:
                continue
            sc_try, _ = _score_normalized(a_try)
            if sc_try < best_p_sc:
                best_p_sc = sc_try
                best_p = a_try
        
        if best_p is not None:
            a = best_p
            cur_sc = best_p_sc
            # Adapt alpha
            if cur_sc < best_sc:
                alpha = min(1.0, alpha * 1.1)
            else:
                alpha = max(0.01, alpha * 0.85)
        else:
            alpha = max(0.01, alpha * 0.7)
            if stall > 20:
                # Perturb to escape
                a = a * np.exp(_RNG.normal(0, 0.1, n))
                a = _normalize(a)
            if stall > 100:
                # Reset to best
                a = best.copy()
                cur_sc = best_sc
                alpha = 0.1
            stall = 0  # (already incremented above)
        
        # Increase beta periodically
        if it % 50 == 0:
            beta = min(beta_max, beta * 1.5)
        
        it += 1
        
        if it % 200 == 0 and time_budget > 60:
            print(f"  MMA iter {it}: best={best_sc:.6f}, cur={cur_sc:.6f}, beta={beta:.0f}, alpha={alpha:.4f}, stall={stall}")
    
    return best, best_sc
