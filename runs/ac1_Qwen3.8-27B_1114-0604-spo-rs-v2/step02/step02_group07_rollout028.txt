def fast_self_conv(a):

"""
Optimizes non-negative step function to minimize 2n*max(conv(a,a))/(sum a)^2.
Uses multi-start gradient optimization with softmax subgradients,
cosine-annealed Adam, and periodic diversification.
Explores multiple sequence lengths and initialization strategies.

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_grad(a, c, topk, temp_frac=0.05):
    """
    Compute gradient of 2n*max(c)/S^2 using softmax over top-k peaks.
    Returns (gradient, soft_max_value).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    k_act = min(topk, c.size)
    thresh = np.partition(c, c.size - k_act)[c.size - k_act]
    part = np.where(c >= thresh - 1e-15)[0]
    # If there are ties at the boundary, we might have more than k_act; that's fine
    m = len(part)
    if m == 0:
        part = np.array([int(np.argmax(c))])
        m = 1
    
    vals = c[part]
    vmax = vals.max()
    temp = max(temp_frac * vmax, 1e-15)
    wts = np.exp((vals - vmax) / temp)
    wts /= wts.sum()
    M_soft = float(np.dot(vals, wts))
    
    # Gradient of M_soft w.r.t. a
    # M_soft = sum_{k in part} w_k * c[k]
    # d(c[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0
    grad_M = np.zeros(n)
    for idx in range(m):
        wi = wts[idx]
        if wi < 1e-12:
            continue
        k = int(part[idx])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js > je:
            continue
        # grad_M[js:je+1] += 2*wi*a[k-je : k-js+1][::-1]
        # k-je to k-js is the range of (k-j) for j in [js, je]
        # a[k-je], a[k-je+1], ..., a[k-js]
        # We want this reversed: a[k-js], a[k-js-1], ..., a[k-je]
        # which is a[k-je:k-js+1][::-1]
        lo = k - je  # = max(0, k-(n-1)) = max(0, k-n+1) = js  ... wait
        # k - je: if je = k (when k < n), then k-je = 0
        #          if je = n-1 (when k >= n), then k-je = k-n+1
        # k - js: if js = 0 (when k < n), then k-js = k
        #          if js = k-n+1 (when k >= n), then k-js = n-1
        # So the range is [max(0, k-n+1), min(k, n-1)] which is [js, je] for the paired indices
        # a[paired_lo : paired_hi+1] where paired_lo = k-je, paired_hi = k-js
        # We need this reversed to align with j going from js to je
        paired_lo = k - je
        paired_hi = k - js
        grad_M[js:je+1] += 2.0 * wi * a[paired_lo:paired_hi+1][::-1]
    
    # Objective: f = 2n * M_soft / S^2
    # df/da[i] = 2n * (grad_M[i]/S^2 - 2*M_soft/S^3)
    grad = (2.0 * n) * (grad_M / S2 - 2.0 * M_soft / S3)
    
    return grad, M_soft


def adam_step(a, g, m, v, t, lr):
    """In-place Adam step."""
    bc1 = 1.0 - 0.9**t
    bc2 = 1.0 - 0.999**t
    m[:] = 0.9*m + 0.1*g
    v[:] = 0.999*v + 0.001*g*g
    a -= lr * (m/bc1) / (np.sqrt(v/bc2) + 1e-8)


def make_init(n, rng, mode):
    """Create initial sequence based on mode."""
    if mode == 0:
        # Uniform
        return np.ones(n) * 1.0
    elif mode == 1:
        # Cosine bump (reduces center peak)
        x = np.linspace(0, 1, n)
        return 1.0 + 0.3*np.cos(2*np.pi*x)
    elif mode == 2:
        # Random Gaussian
        return np.clip(1.0 + 0.5*rng.standard_normal(n), 0.01, 1000.0)
    elif mode == 3:
        # Two-frequency
        x = np.linspace(0, 1, n)
        return np.clip(1.0 + 0.2*np.cos(2*np.pi*3*x) + 0.2*np.cos(2*np.pi*5*x), 0.01, 1000.0)
    elif mode == 4:

    elif mode == 5:

        a = np.fft.ifft(A * np.exp(1j*phase)).real
        return np.clip(a - a.min() + 0.1, 0.01, 1000.0)
    else:

"""
Multi-start gradient optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses softmax subgradient with adaptive temperature, cosine-annealed Adam,
periodic structured restarts, and explores multiple sequence lengths.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_grad(a, c, topk, temp_frac):
    """
    Compute gradient of 2n*M/S^2 using softmax over top-k peaks.
    Returns (gradient_array, soft_max_value).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S

    k_act = min(topk, c.size)
    thresh_idx = c.size - k_act
    thresh = np.partition(c, thresh_idx)[thresh_idx]
    part = np.where(c >= thresh - 1e-12)[0]
    m = len(part)
    if m == 0:
        part = np.array([int(np.argmax(c))])
        m = 1

    vals = c[part]
    vmax = float(vals.max())
    temp = max(temp_frac * vmax, 1e-15)
    wts = np.exp((vals - vmax) / temp)
    w_sum = wts.sum()
    if w_sum < 1e-30:
        wts = np.zeros(m)
        wts[0] = 1.0
    else:
        wts /= w_sum
    M_soft = float(np.dot(vals, wts))

    # grad_M[i] = 2 * sum_k w_k * a[k-i] (where 0<=k-i<n)
    grad_M = np.zeros(n)
    for idx in range(m):
        wi = wts[idx]
        if wi < 1e-12:
            continue
        k = int(part[idx])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js > je:
            continue
        paired_lo = k - je
        paired_hi = k - js
        grad_M[js:je + 1] += 2.0 * wi * a[paired_lo:paired_hi + 1][::-1]

    # grad = 2n * (S * grad_M - 2*M_soft) / S^3
    #      = 2n * grad_M / S^2 - 4n * M_soft / S^3
    grad = (2.0 * n * S * grad_M - 4.0 * n * M_soft) / S3
    return grad, M_soft


def adam_step_inplace(a, g, m, v, t, lr):
    """In-place Adam optimizer step."""
    bc1 = 1.0 - 0.9 ** t
    bc2 = 1.0 - 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)


def make_init_a(n, rng, mode):
