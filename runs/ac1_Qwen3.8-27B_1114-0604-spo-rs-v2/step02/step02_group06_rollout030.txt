    p_weights = exp_bc / sum_exp
    P = _compute_pressure(a, p_weights)
    grad = (2.0 * n / beta) * P / (S * S) - 4.0 * n * M_beta * a / (S ** 3)
    return float(f_val), grad


def _lbfgs_optimize(a0, betas_list, time_limit, verbose=False):
    current = a0.copy()
    n = len(current)
    best = current.copy()
    best_f = _exact_score(best)
    deadline = time.time() + time_limit

    for beta in betas_list:
        if time.time() > deadline:
            break
        t0 = time.time()

        def obj_func(x):
            a = np.maximum(x, 0.0)
            return _smooth_obj_and_grad(a, beta)[0]

        def grad_func(x):
            a = np.maximum(x, 0.0)
            return _smooth_obj_and_grad(a, beta)[1]

        maxiter = 300 if n <= 4096 else 150
        res = minimize(
            obj_func, current, method='L-BFGS-B', jac=grad_func,
            bounds=[(0.0, 100.0)] * n,
            options={'maxiter': maxiter, 'maxls': 30, 'ftol': 1e-14,
                     'gtol': 1e-9, 'disp': False}
        )
        elapsed = time.time() - t0
        new_a = np.maximum(res.x, 0.0)
        S_new = new_a.sum()
        if S_new > 1e-10:
            new_a = new_a / S_new
            sc = _exact_score(new_a)
            current = new_a.copy()
            if sc < best_f:
                best_f = sc
                best = new_a.copy()
                if verbose:
                    print(f"  L-BFGS beta={beta}: {best_f:.7f} ({elapsed:.1f}s)")
            elif verbose:
                print(f"  L-BFGS beta={beta}: no improve, smooth={res.fun:.7f}, elapsed={elapsed:.1f}s")
        else:
            if verbose:
                print(f"  L-BFGS beta={beta}: degenerate")
    return best, best_f


def _multiplicative_updates(a0, time_limit, target_score=1.500, verbose=False):
    p = a0.copy().astype(float)
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_sc = _exact_score(best)
    deadline = time.time() + time_limit
    stall = 0
    alpha = 0.1
    beta = 50.0
    eps = 0.1

    while time.time() < deadline:
        Fp = np.fft.rfft(p, L)
        c = np.fft.irfft(Fp * Fp, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        maxc = float(np.max(c))
        cur_sc = 2.0 * n * maxc
        p_sum = np.sum(p)
        if p_sum > 0:
            cur_sc_norm = cur_sc / (p_sum * p_sum)
        else:
            cur_sc_norm = float('inf')

        if cur_sc_norm < best_sc:
            best_sc = cur_sc_norm
            best = p.copy()
            if verbose and stall == 0:
                print(f"  MU update: {best_sc:.7f}")
            stall = 0
        else:
            stall += 1

        d = n * c
        t_thresh = (1.0 - eps) * maxc + eps * (target_score * p_sum * p_sum / (2.0 * n))
        if t_thresh >= maxc:
            t_thresh = maxc * (1.0 - 1e-6)
        viol = np.maximum(0.0, c - t_thresh)
        maxviol = float(np.max(viol))

        if maxviol > 1e-14 * maxc:
            w = viol / (maxviol + 1e-300)
            w = w * (0.3 + 0.7 * w)
            w = w * w
        else:
            shift = np.max(d)
            e = np.exp(beta * (d - shift))
            w = e / np.sum(e)

        Fw = np.fft.rfft(w, L)
        h = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
        h = np.maximum(h, 0.0)
        hm = float(np.max(h))

        if hm < 1e-300:
            p = p * np.exp(np.random.default_rng().normal(0, 0.05, n))
            p = np.maximum(p, 1e-16)
            p = p / p.sum()
            continue

        h = h / hm
        improved = False
        for mult in (0.5, 1.0, 2.0, 4.0):
            a_step = alpha * mult
            p_try = p * np.exp(-a_step * h)
            p_try = np.maximum(p_try, 1e-16)
            s = float(np.sum(p_try))
            if s < 1e-10:
                continue
            p_try = p_try / s
            sc_try = _exact_score(p_try)
            if sc_try < cur_sc_norm - 1e-12:
                p = p_try
                cur_sc_norm = sc_try
                improved = True
                break

        if not improved:
            alpha = max(0.01, alpha * 0.7)
            eps = min(1.0, eps * 1.01)
            if stall >= 15:
                rng = np.random.default_rng()
                std = 0.05 + 0.002 * min(stall, 50)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = np.maximum(p_try, 1e-16)
                s = float(np.sum(p_try))
                if s > 0:
                    p_try = p_try / s
                    sc2 = _exact_score(p_try)
                    if sc2 < best_sc * 1.03:
                        p = p_try
                        cur_sc_norm = sc2
                    else:
                        p = best.copy()
                        cur_sc_norm = best_sc
                stall = 0
            if stall > 100:
                p = best.copy()
                cur_sc_norm = best_sc
                alpha = 0.1
                eps = 0.1
                beta = max(50.0, beta * 0.5)
                stall = 0

        if (stall % 50) == 0 and stall > 0:
            beta = min(3000.0, beta * 1.1)

    return best, best_sc


def _local_block_search(a0, time_limit, verbose=False):
    p = a0.copy().astype(float)
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_sc = _exact_score(best)
    deadline = time.time() + time_limit
    stagnant = 0
    rng = np.random.default_rng(42)

    small = max(1, n // 4096)
    mid = max(small + 1, n // 512)
    large = max(mid + 1, min(n, n // 32))

    while time.time() < deadline:
        r = rng.random()
        if r < 0.25:
            w = 1
        elif r < 0.5:
            w = int(rng.integers(small, max(small + 1, small * 5)))
        elif r < 0.8:
            w = int(rng.integers(mid, max(mid + 1, min(n, mid * 5))))
        else:
            w = int(rng.integers(large, max(large + 1, min(n, large * 3))))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        cur_sc = _exact_score(p)
        improved = False

        for f in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.0, 3.0):
            cand = p.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            if s < 1e-10:
                continue
            cand = cand / s
            sc = _exact_score(cand)
            if sc < cur_sc - 1e-13:
                p = cand
                cur_sc = sc
                improved = True
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best = p.copy()

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                std = 0.03 + 0.002 * min(stagnant, 60)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = np.maximum(p_try, 1e-16)
                s = float(np.sum(p_try))
                if s > 0:
                    p_try = p_try / s
                    sc2 = _exact_score(p_try)
                    if sc2 < best_sc * 1.02:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0
            if stagnant > 200:
                p = best.copy()
                stagnant = 0

    return best, best_sc


def _coordinate_descent(a0, time_limit, verbose=False):
    p = a0.copy().astype(float)
    n = len(p)
    best = p.copy()
    best_sc = _exact_score(best)
    deadline = time.time() + time_limit
    rng = np.random.default_rng(7)

    while time.time() < deadline:
        idx = int(rng.integers(0, n))
        cur_sc = best_sc
        orig_val = best[idx]
        improved = False

        for f in (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 2.5, 4.0):
            if f == 1.0:
                continue
            cand = best.copy()
            cand[idx] = orig_val * f
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            if s < 1e-10:
                continue
            cand = cand / s
            sc = _exact_score(cand)
            if sc < cur_sc - 1e-13:
                cur_sc = sc
                improved = True
        if improved:
            best_sc = cur_sc
            best = cand

    return best, best_sc


def _resample(p, m):
    m = int(max(1, m))
    n = len(p)
    if n == m:
        return p.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0) + 1e-15
    s = float(np.sum(out))
    if s > 0:
        out = out / s
    return out


def propose_candidate(budget_s=2400, **kwargs):
    start_time = time.time()
    deadline = start_time + float(budget_s) - 90

    try:
        parent = np.array(height_sequence_1, dtype=float)
    except Exception:
        parent = np.ones(3072) / 3072.0

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    S = float(np.sum(parent))
    if S < 1e-10:
        parent = np.ones(3072) / 3072.0
        S = 1.0
    parent = parent / S

    print(f"Parent: n={len(parent)}, score={_exact_score(parent):.7f}")

    all_candidates = [(parent.copy(), _exact_score(parent))]
    verbose = True

    # Phase 1: L-BFGS with continuation at parent length
    n_parent = len(parent)
    rem = deadline - time.time()
    if rem > 120:
        t1 = min(rem - 60, 300)
        print(f"\nPhase 1: L-BFGS at n={n_parent}, time={t1:.0f}s")
        betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
        res_a, sc_a = _lbfgs_optimize(parent, betas, t1, verbose)
        all_candidates.append((res_a, sc_a))
        print(f"  Phase 1 best: {sc_a:.7f}")

    # Phase 2: Multiplicative updates
    rem = deadline - time.time()
    if rem > 100:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t2 = min(rem - 60, 250)
        print(f"\nPhase 2: Multiplicative updates at n={len(best_so_far[0])}, time={t2:.0f}s")
        res_b, sc_b = _multiplicative_updates(best_so_far[0], t2, verbose=verbose)
        all_candidates.append((res_b, sc_b))
        print(f"  Phase 2 best: {sc_b:.7f}")

    # Phase 3: L-BFGS from multiplicative result
    rem = deadline - time.time()
    if rem > 80:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t3 = min(rem - 40, 200)
        print(f"\nPhase 3: L-BFGS from best, time={t3:.0f}s")
        betas2 = [50, 200, 1000, 5000, 20000, 100000]
        res_c, sc_c = _lbfgs_optimize(best_so_far[0], betas2, t3, verbose)
        all_candidates.append((res_c, sc_c))
        print(f"  Phase 3 best: {sc_c:.7f}")

    # Phase 4: Try different lengths
    rem = deadline - time.time()
    if rem > 150:
        for try_n in [2048, 4096]:
            rem = deadline - time.time()
            if rem < 80:
                break
            best_so_far = min(all_candidates, key=lambda x: x[1])
            resampled = _resample(best_so_far[0], try_n)
            t4 = min(rem - 40, 150)
            print(f"\nPhase 4a: L-BFGS at n={try_n}, time={t4:.0f}s")
            betas3 = [20, 100, 500, 2000, 10000, 50000]
            res_d, sc_d = _lbfgs_optimize(resampled, betas3, t4, verbose)
            all_candidates.append((res_d, sc_d))
            print(f"  n={try_n} L-BFGS: {sc_d:.7f}")

    # Phase 5: Block local search on best
    rem = deadline - time.time()
    if rem > 60:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t5 = min(rem - 30, 200)
        print(f"\nPhase 5: Block local search, time={t5:.0f}s")
        res_e, sc_e = _local_block_search(best_so_far[0], t5, verbose)
        all_candidates.append((res_e, sc_e))
        print(f"  Phase 5 best: {sc_e:.7f}")

    # Phase 6: L-BFGS polish from block search result
    rem = deadline - time.time()
    if rem > 60:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t6 = min(rem - 30, 150)
        print(f"\nPhase 6: L-BFGS polish, time={t6:.0f}s")
        betas4 = [200, 1000, 5000, 50000, 200000]
        res_f, sc_f = _lbfgs_optimize(best_so_far[0], betas4, t6, verbose)
        all_candidates.append((res_f, sc_f))
        print(f"  Phase 6 best: {sc_f:.7f}")

    # Phase 7: Final local search
    rem = deadline - time.time()
    if rem > 40:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t7 = min(rem - 20, 120)
        print(f"\nPhase 7: Final block search, time={t7:.0f}s")
        res_g, sc_g = _local_block_search(best_so_far[0], t7, verbose)
        all_candidates.append((res_g, sc_g))
        print(f"  Phase 7 best: {sc_g:.7f}")

    # Phase 8: Coordinate descent polish
    rem = deadline - time.time()
    if rem > 30:
        best_so_far = min(all_candidates, key=lambda x: x[1])
        t8 = min(rem - 15, 80)
        print(f"\nPhase 8: Coordinate descent, time={t8:.0f}s")
        res_h, sc_h = _coordinate_descent(best_so_far[0], t8, verbose)
        all_candidates.append((res_h, sc_h))
        print(f"  Phase 8 best: {sc_h:.7f}")

    # Select best
    all_candidates.sort(key=lambda x: x[1])
    best_arr = all_candidates[0][0].copy()
    best_score = all_candidates[0][1]

    print(f"\nInternal best: {best_score:.7f} (n={len(best_arr)})")

    # Verify with official evaluator
    arr_list = [float(x) for x in best_arr]
    try:
        official = evaluate_sequence(arr_list)
        if np.isfinite(official) and official < best_score:
            best_score = official
            print(f"Official eval confirms improvement: {official:.7f}")
        elif np.isfinite(official):
            print(f"Official eval: {official:.7f} (internal: {best_score:.7f})")
    except Exception as e:
        print(f"Official eval failed: {e}")

    # Also check a few other candidates
    for (arr, sc) in all_candidates[1:4]:
        try:
            lst = [float(x) for x in arr]
            off = evaluate_sequence(lst)
            if np.isfinite(off) and off < best_score:
                best_score = off
                best_arr = arr.copy()
                print(f"Found better via official eval: {off:.7f}")
        except Exception:
            pass

    # Final cleanup
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    S_final = float(np.sum(best_arr))
    if S_final > 0:
        best_arr = best_arr / S_final

    return [float(x) for x in best_arr]
