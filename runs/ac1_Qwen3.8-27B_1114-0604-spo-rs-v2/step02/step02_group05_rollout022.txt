                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-13:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and iter_count % 100 == 0:
                        print(f"      iter={iter_count}, score={best_score:.6f}, k={k_current}, alpha={alpha:.4f}")
            else:
                stall += 1
                if stall > 10:
                    alpha = max(0.005, alpha * 0.7)
                if stall > 50:
                    p = best.copy()
                    cur_score = best_score
                    alpha = 0.15
                    stall = 10

        elif r < 0.75:
            # Frequency domain: convolution flattening
            alpha_f = float(rng.uniform(1.5, 8.0))
            p_new = _freq_flatten_conv(p, alpha_f, L_conv)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-14:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                        if verbose:
                            print(f"      CONV_FREQ iter={iter_count}, alpha_f={alpha_f:.2f}, score={best_score:.6f}")
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.85:
            # Frequency domain: power spectrum flattening
            alpha_f = float(rng.uniform(1.3, 6.0))
            p_new = _freq_flatten_power(p, alpha_f, L_conv)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-14:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.92:
            # Two-point exchange
            j1 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            if j1 == j2:
                j2 = (j1 + n // 2) % n
            p_new, sc_new = _two_point_exchange(p, j1, j2)
            if sc_new < cur_score - 1e-14:
                p = p_new
                cur_score = sc_new
                stall = 0
                if sc_new < best_score - 1e-13:
                    best = p.copy()
                    best_score = sc_new
            else:
                stall += 1

        else:
            # Random perturbation (escape local minima)
            if stall > 10:
                std = 0.01 + 0.003 * min(stall, 50)
                p_try = p * np.exp(rng.normal(0.0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.10:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
            else:
                stall += 1

        # Periodic pullback
        if cur_score > best_score * 1.12 and best_score < float('inf'):
            if rng.random() < 0.4:
                p = best.copy()
                cur_score = best_score
                stall = 0
                alpha = 0.15

        iter_count += 1

    return best, best_score


def _block_refine(p0, rng, time_budget):
    """Block-wise multiplicative refinement."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 3))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 256), max(3, n // 32)))
        else:
            bs = int(rng.integers(max(8, n // 64), max(16, n // 8)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 60:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 4.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.05 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(3, 10))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.03, 0.25)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_fourier(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 16):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Focused exploitation with top-k subgradient, frequency-domain methods,
    two-point exchange, and adaptive step sizes.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 15

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={time.monotonic()-start_time:.0f}s ***")

    # ========== PHASE 1: Quick exploration (5%) ==========
    print("=" * 60)
    print(f"Phase 1: Quick exploration")
    t1_end = start_time + 0.05 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    # Try a few initializations at key lengths
    base_lengths = [2048, 3072, 4096]
    if parent is not None:
        pn = len(parent)
        if 512 <= pn <= 8192:
            if pn not in base_lengths:
                base_lengths.append(pn)
            base_lengths.sort()

    for n in base_lengths:
        if time.monotonic() > t1_end:
            break
        inits = [
            _init_uniform(n),
            _init_two_level(n, rng),
            _init_gaussian_mixture(n, rng),
            _init_fourier(n, rng),
        ]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f} (n={len(best_overall) if best_overall is not None else 0})")

    # ========== PHASE 2: Main focused optimization (65%) ==========
    print("Phase 2: Main focused optimization")
    t2_end = start_time + 0.72 * budget

    if best_overall is None:
        best_overall = _init_uniform(3072)
        best_score_overall = _score(best_overall)

    n_main = len(best_overall)

    # Run 1: Standard focused optimization on the best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.40 * t_rem, 400.0)
        t_alloc = max(30.0, t_alloc)
        print(f"  Run 1: focused opt, n={n_main}, alloc={t_alloc:.0f}s")
        p, sc = _optimize_focused(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    # Run 2: Restart with perturbed initialization, higher diversity
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.25 * t_rem, 250.0)
        t_alloc = max(20.0, t_alloc)
        p_start = _resample(best_overall, n_main, rng)
        if p_start is None:
            p_start = best_overall.copy()
        print(f"  Run 2: perturbed restart, alloc={t_alloc:.0f}s")
        p, sc = _optimize_focused(p_start, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    # Run 3: Try larger length
    t_rem = t2_end - time.monotonic()
    if t_rem > 40:
        n_up = n_main * 2
        n_up = min(n_up, 8192)
        if n_up != n_main:
            t_alloc = min(0.20 * t_rem, 200.0)
            t_alloc = max(20.0, t_alloc)
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                print(f"  Run 3: upsized n={n_up}, alloc={t_alloc:.0f}s")
                p, sc = _optimize_focused(p_up, rng, t_alloc, verbose=True)
                _try_update(p, sc)

    # Run 4: Block refinement on current best
    t_rem = t2_end - time.monotonic()
    if t_rem > 20:
        t_alloc = min(0.15 * t_rem, 120.0)
        t_alloc = max(15.0, t_alloc)

        p, sc = _block_refine(best_overall, rng, t_alloc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts (15%) ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.87 * budget

                if sp is None:
                    sp = _init_uniform(n_best)
            elif mode == 4:

                sp = best_overall * np.exp(rng.normal(0, 0.1, n_best))
                sp = _sanitize(sp)
                if sp is None:
                    sp = _init_uniform(n_best)
            else:
                # Inverted structure
                sp = 1.0 / (best_overall + 1e-6)
                sp = _sanitize(sp)
                if sp is None:
                    sp = _init_uniform(n_best)

            t_alloc = min(50.0, restart_budget)
            p, sc = _optimize_focused(sp, rng, t_alloc, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (13%) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # Polish 1: focused optimization with aggressive settings
        t_rem = deadline - time.monotonic()
        if t_rem > 25:
            t_alloc = min(0.35 * t_rem, 180.0)
            p, sc = _optimize_focused(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # Polish 2: Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.25 * t_rem, 100.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # Polish 3: Another focused pass
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.30 * t_rem, 150.0)
            p_start = _resample(best_overall, len(best_overall), rng)
            if p_start is None:
                p_start = best_overall.copy()
            p, sc = _optimize_focused(p_start, rng, t_alloc, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = eval_score
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
