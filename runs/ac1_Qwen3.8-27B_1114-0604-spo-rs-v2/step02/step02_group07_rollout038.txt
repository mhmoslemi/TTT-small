import numpy as np
from scipy.optimize import minimize
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def smooth_max(c, beta=10.0):
    """Smooth approximation of max using log-sum-exp."""
    c_max = np.max(c)
    if c_max < 1e-30:
        return 0.0, 0.0
    shifted = c - c_max
    log_sum = np.log(np.sum(np.exp(beta * shifted)))
    return c_max + log_sum / beta

def objective_and_grad(a, beta=10.0):
    """Objective and gradient for L-BFGS-B."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    c_max = np.max(c)
    
    if c_max < 1e-30:
        return 2.0 * n * 1e-30 / (S * S), np.zeros(n)
    
    # Soft max weights
    shifted = c - c_max
    exp_shifted = np.exp(beta * shifted)
    w = exp_shifted / np.sum(exp_shifted)
    M_smooth = c_max + np.log(np.sum(exp_shifted)) / beta
    
    # Gradient of M_smooth w.r.t. a
    grad_M = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        # d(c[k])/d(a[i]) = a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            grad_M[i] += w[k] * a[k - i]
    
    # f = 2n * M_smooth / S^2
    f = 2.0 * n * M_smooth / (S * S)
    
    # df/da[i] = 2n * (grad_M[i] / S^2 - 2 * M_smooth / S^3)
    grad_f = 2.0 * n * (grad_M / (S * S) - 2.0 * M_smooth / (S ** 3))
    
    return f, grad_f

    """Custom gradient descent with momentum and BB step size."""
    n = len(a0)

    best_a = a.copy()

            yy = y @ y

        a = a + v
        a = np.clip(a, 0, 1000)

        prev_a = a.copy()
        t += 1

`= 2 * sum_{j=0}^{n-1} w[i+j] * a[j]` (where `i+j < 2n-1` is automatically satisfied since `i < n` and `j < n`, so `i+j < 2n-1`)

So I need: `T[i] = sum_{j=0}^{n-1} a[j] * w[i+j]` for `i = 0, ..., n-1`.

This is the cross-correlation of `a` and `w` at lag `i`, where the cross-correlation is defined as:
`(a ⋆ w)[i] = sum_j a[j] * w[j+i]`

Using the FFT: `(a ⋆ w) = ifft(conj(fft(a, N)) * fft(w, N))`

Let me verify: `ifft(conj(FFT(a)) * FFT(w))[m]`

`FFT(a)[k] = sum_j a[j] e^{-2πijk/N}`
`conj(FFT(a)[k]) = sum_j a[j] e^{2πijk/N}`
`FFT(w)[k] = sum_l w[l] e^{-2πilk/N}`

Product: `sum_j sum_l a[j] w[l] e^{2πijk/N} e^{-2πilk/N} = sum_j sum_l a[j] w[l] e^{2πik(j-l)/N}`

IFFT at position m: `(1/N) sum_k [product] e^{2πikm/N}`
`= (1/N) sum_j sum_l a[j] w[l] sum_k e^{2πik(j-l+m)/N}`
`= (1/N) sum_j sum_l a[j] w[l] * N * I((j-l+m) mod N == 0)`
`= sum_j sum_l a[j] w[l] * I(l = j+m) (for N large enough)`
`= sum_j a[j] w[j+m]`

So `ifft(conj(FFT(a, N)) * FFT(w, N))[m] = sum_j a[j] w[j+m] = (a ⋆ w)[m]`

This is exactly what I want! `T[i] = (a ⋆ w)[i] = ifft(conj(FFT(a, N)) * FFT(w, N))[i]`

And I need `N >= n + len(w) - 1 = n + (2n-1) - 1 = 3n-2` to avoid wraparound.

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with FFT-based gradients, adaptive soft-max temperature,
random restarts with structured perturbations, and multiple initializations
including flattop windows and cosine-based patterns.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def get_score(a):
    """Score from array."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    return compute_score(n, c, S)


def obj_func(a, beta):
    """Smooth objective using log-sum-exp."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10
    c = fast_self_conv(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 0.0
    shifted = c - c_max
    M_smooth = c_max + np.log(np.sum(np.exp(beta * shifted))) / beta
    return 2.0 * n * M_smooth / (S * S)


def obj_grad(a, beta):
    """Gradient of smooth objective using FFT-based cross-correlation."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.zeros(n)
    c = fast_self_conv(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    shifted = c - c_max
    exp_shifted = np.exp(beta * shifted)
    w = exp_shifted / np.sum(exp_shifted)
    M_smooth = c_max + np.log(np.sum(exp_shifted)) / beta

    # T[i] = sum_j a[j] * w[i+j] via FFT cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(np.conj(A) * W, N)[:n]

    # df/da[i] = 4n * (T[i]/S^2 - M/S^3)
    grad = 4.0 * n * (cross / (S * S) - M_smooth / (S ** 3))
    return grad


def lbgfss_run(a0, beta, maxiter=300, maxfun=1000, time_limit=None):
    """Run L-BFGS-B with smooth objective."""
    n = len(a0)
    stop_condition = {'flag': False}

    def fun(x):
        if time_limit is not None and time.monotonic() > time_limit:
            stop_condition['flag'] = True
        return obj_func(x, beta)

    def jac(x):
        if time_limit is not None and time.monotonic() > time_limit:
            stop_condition['flag'] = True
        return obj_grad(x, beta)

    try:
        result = minimize(
            fun=fun, x0=a0.copy(), jac=jac,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxfun, 'disp': False}
        )
        if np.isfinite(result.x).all() and np.all(result.x >= -1e-10):
            return np.clip(result.x, 0.0, 1000.0)
    except Exception:
        pass
    return a0.copy()


def custom_gd(a0, end_time, beta, lr_init=0.003):
    """Gradient descent with BB step size and clipping."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    best_a = a.copy()
    c = fast_self_conv(a)
    best_score = compute_score(n, c, a.sum())
    prev_a = a.copy()
    prev_g = None
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)

    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = a.sum()
        if S < 1e-10:
            break
        score = compute_score(n, c, S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        g = obj_grad(a, beta)

        # BB step size
        if prev_g is not None:
            s = a - prev_a
            y = g - prev_g
            sy = float(s @ y)
            yy = float(y @ y)
            ss = float(s @ s)
            if yy > 1e-30 and sy > 1e-12:
                lr_bb = ss / sy
                lr = 0.5 * lr + 0.5 * lr_bb
            lr = float(np.clip(lr, 1e-8, 0.5))

        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Backtracking if score worsens too much
        c_new = fast_self_conv(a_new)
        score_new = compute_score(n, c_new, a_new.sum())
        if score_new > best_score * 5:
            v = v * 0.1
            a_new = np.clip(a + v, 0.0, 1000.0)

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new

    return best_score, best_a


def flattop_init(n):
    """Flattop window."""
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    k = np.arange(n)
    a = (a0 - a1 * np.cos(2.0 * np.pi * k / n)
         + a2 * np.cos(4.0 * np.pi * k / n)
         - a3 * np.cos(6.0 * np.pi * k / n)
         + a4 * np.cos(8.0 * np.pi * k / n))
    return np.clip(a, 0.0, 1000.0)


def cosine_init(n, freq, amp, phase=0.0):
    k = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * k / n + phase)
    return np.clip(a, 0.0, 1000.0)
