Gradient-based multi-scale optimization for minimizing the self-convolution peak
of non-negative sequences. Uses L-BFGS-B with smooth-max surrogate in log-space,
coarse-to-fine multi-scale search, and local block refinement.
"""

import numpy as np

_CONV_LEN_CACHE = {}

def _conv_len(n):
    n = int(max(1, n))
    if n in _CONV_LEN_CACHE:
        return _CONV_LEN_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_LEN_CACHE[n] = L
    return L

def _sanitize(p):
    """Sanitize and normalize a sequence to have sum=1, non-negative."""
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

def _score_normalized(a):
    """Score for a normalized sequence (sum=1): 2*n*max(conv(a,a))"""
    n = a.size
    L = _conv_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    return 2.0 * n * c.max(), c

def _smooth_obj_grad(x, beta):
    """
    Smooth objective and gradient in log-space.
    x: array of log-values (length n)
    Returns: (value, gradient)
    """
    # Numerical stability: shift x
    x_shift = x - x.max()
    a_raw = np.exp(x_shift)
    s = a_raw.sum()
    a = a_raw / s
    n = a.size
    L = _conv_len(n)
    
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    # Smooth max via log-sum-exp
    c_max = c.max()
    beta_c = beta * (c - c_max)
    # Clip for numerical stability
    beta_c = np.clip(beta_c, -500, 500)
    exp_b = np.exp(beta_c)
    w_sum = exp_b.sum()
    smooth_max = c_max + np.log(w_sum) / beta
    value = 2.0 * n * smooth_max
    
    # Gradient
    w_norm = exp_b / w_sum
    W = np.fft.rfft(w_norm, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    
    grad_a = 4.0 * n * beta * cross
    grad_x = a * grad_a
    
    return value, grad_x

def _lbfgs_optimize(a_init, beta_start, beta_end, n_beta_steps, time_limit, verbose=False):
    """
    Run L-BFGS-B optimization with increasing beta.
    a_init: initial normalized sequence (sum=1)
    Returns: (best_sequence, best_score)
    """
    n = a_init.size
    x0 = np.log(np.maximum(a_init, 1e-30))
    
    best_a = a_init.copy()
    best_sc, _ = _score_normalized(a_init)
    
    deadline = time.monotonic() + time_limit
    current_beta = beta_start
    
    for step in range(n_beta_steps):
        # Interpolate beta (exponential schedule)
        frac = step / max(1, n_beta_steps - 1)
        current_beta = beta_start * (beta_end / beta_start) ** frac
        
        if time.monotonic() > deadline - 5:
            break
        
        def obj(x):
            val, grad = _smooth_obj_grad(x, current_beta)
            return val, grad
        
        try:
            result = minimize(
                obj, x0, method='L-BFGS-B',
                options={
                    'maxiter': 500,
                    'maxfun': 5000,
                    'gtol': 1e-8,
                    'ftol': 1e-12,
                    'maxls': 50,
                }
            )
            x_new = result.x
            a_new = np.exp(x_new - x_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            sc, _ = _score_normalized(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if verbose:
                    print(f"  L-BFGS step {step}: beta={current_beta:.1f}, score={sc:.8f}, converged={result.success}")
            
            x0 = x_new
        except Exception as e:
            if verbose:
                print(f"  L-BFGS step {step} failed: {e}")
            break
    
    return best_a, best_sc

def _resample(a, m):
    """Resample sequence a to length m using linear interpolation."""
    n = a.size
    if n == m:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()

"""
Gradient-based multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B in log-space with smooth-max surrogate, threshold-aware gradient enhancement,
coarse-to-fine multi-scale search, and local block refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_RNG = np.random.default_rng(2024)
_CONV_LEN_CACHE = {}

def _conv_len(n):
    n = int(max(1, n))
    if n in _CONV_LEN_CACHE:
        return _CONV_LEN_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_LEN_CACHE[n] = L
    return L

def _sanitize(p):
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s

def _score_normalized(a):
    n = a.size
    L = _conv_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    return 2.0 * n * c.max(), c

def _smooth_obj_grad(x, beta, threshold_frac=0.0):
    x_shift = x - x.max()
    a_raw = np.exp(x_shift)
    s = a_raw.sum()
    a = a_raw / s
    n = a.size
    L = _conv_len(n)
    
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    c_max = c.max()
    beta_c = beta * (c - c_max)
    beta_c = np.clip(beta_c, -600, 600)
    exp_b = np.exp(beta_c)
    
    if threshold_frac > 0:
        thresh = c_max - threshold_frac * c_max
        mask = (c > thresh).astype(float)
        if mask.sum() > 0:
            exp_b = exp_b * (1.0 + 3.0 * mask)
    
    w_sum = exp_b.sum()
    if w_sum <= 0 or not np.isfinite(w_sum):
        w_sum = 1.0
        exp_b = np.ones_like(c) / len(c)
    
    smooth_max = c_max + np.log(w_sum) / beta
    value = 2.0 * n * smooth_max
    
    w_norm = exp_b / w_sum
    W = np.fft.rfft(w_norm, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    
    grad_a = 4.0 * n * beta * cross
    grad_x = a * grad_a
    
    return value, grad_x

def _lbfgs_optimize(a_init, beta_start, beta_end, n_steps, time_limit, threshold_frac=0.0, verbose=False):
    n = a_init.size
    x0 = np.log(np.maximum(a_init, 1e-30))
    best_a = a_init.copy()
    best_sc, _ = _score_normalized(a_init)
    
    deadline = time.monotonic() + time_limit
    total_iters = 0
    
    for step in range(n_steps):
        if time.monotonic() > deadline - 3:
            break
        
        frac = step / max(1, n_steps - 1)
        current_beta = beta_start * (beta_end / beta_start) ** frac
        
        def obj(x):
            val, grad = _smooth_obj_grad(x, current_beta, threshold_frac)
            return val, grad
        
        maxiter = max(50, min(800, int(30.0 / max(0.01, (deadline - time.monotonic()) / time_limit))))
        
        try:
            result = minimize(
                obj, x0, method='L-BFGS-B',
                options={
                    'maxiter': maxiter,
                    'maxfun': maxiter * 10,
                    'gtol': 1e-8,
                    'ftol': 1e-14,
                    'maxls': 40,
                }
            )
            x_new = result.x
            a_new = np.exp(x_new - x_new.max())
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            sc, _ = _score_normalized(a_new)
            total_iters += result.nit if hasattr(result, 'nit') else 100
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                if verbose:
                    print(f"    beta={current_beta:.0f}: score={sc:.10f}, iters={result.nit}, ok={result.success}")
            
            x0 = x_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS step {step} error: {e}")
            break
    
    return best_a, best_sc

def _threshold_iteration(a_init, time_limit, beta_base=200, n_peaks=5, verbose=False):

    a = a_init.copy()

    it = 0

    while time.monotonic() < deadline:

            stall += 1

            continue
        
        cross = cross / cross.max()
        cross = np.power(cross, 1.5)

            step_size = eta * mult
            a_try = a * np.exp(-step_size * cross)
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            
            sc_try, _ = _score_normalized(a_try)

                a_pert = a * noise

                a_pert /= a_pert.sum()
                sc_p, _ = _score_normalized(a_pert)
                if sc_p < best_sc * 1.02:
                    a = a_pert
                else:
                    a = best_a.copy()
                stall = 0
                eta = 0.3
            
            if stall > 200:
                break
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    thresh it={it}: best={best_sc:.10f}, curr={2.0*n*c.max():.10f}")
    
    return best_a, best_sc

def _local_block_search(a_init, time_limit, verbose=False):
    a = a_init.copy()
    n = a.size
    L = _conv_len(n)
    best_a = a.copy()
    best_sc, _ = _score_normalized(a)
    
    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0

            a = current_best_a

            stagnant = 0

def _resample(a, m):
    n = a.size
    if n == m:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()

def _make_initial(n, rng, mode):
    t = np.linspace(0, 1, n)
    if mode == 0:
        p = np.ones(n)
    elif mode == 1:
        alpha = rng.uniform(0.3, 8.0)
        beta_v = rng.uniform(0.3, 8.0)
        p = np.power(t + 1e-6, alpha) * np.power(1.0 - t + 1e-6, beta_v)
    elif mode == 2:
        K = int(rng.integers(2, 6))
        p = np.ones(n) * 0.3
        for _ in range(K):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.4)
            amp = rng.uniform(0.5, 3.0)
            p += amp * np.exp(-((t - c) / s) ** 2)
    elif mode == 3:
        cut = int(rng.integers(n // 4, 3 * n // 4))
        ratio = rng.uniform(0.2, 5.0)
        p = np.ones(n)
        p[cut:] = ratio
    elif mode == 4:
        p = np.ones(n)
        for k in range(1, 10):
            amp = rng.normal(0, 0.4 / np.sqrt(k))
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.maximum(p, 0.05)
    elif mode == 5:
        K = int(rng.integers(4, 12))
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0.0], edges, [1.0]])
        heights = rng.lognormal(0, 0.8, K)
        p = np.zeros(n)
        for i in range(K):
            s = int(round(blocks[i] * n))
            e = max(s + 1, min(n, int(round(blocks[i+1] * n))))
            p[s:e] = heights[i]
        p = np.maximum(p, 0.01)
    else:
        p = rng.lognormal(0, 0.5, n)
    
    p = np.maximum(p, 1e-10)
    return p / p.sum()
