    """Build the matrix A such that A @ q = conv(p, q)."""
    n = len(p)
    # A has shape (2n-1, n)
    # A[k, j] = p[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))

        if j_start <= j_end:

    n = len(p)

    b = np.full(2*n-1, M)

    if res.success:

        q = np.maximum(q, 0)
        s = q.sum()

```python
"""
LP-guided and gradient-based optimization for non-negative sequences
minimizing 2*n*max(conv(a,a))/sum(a)^2.

Novel approach: Uses a linear program to find the optimal "spreading direction"
that reduces the convolution peak while maintaining non-negativity. Combined
with multiplicative gradient descent and structural moves (power transforms,
block swaps) for escape from local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 200000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _convolve_score_and_max(p):
    """Return (score, max_b, b_array) for normalized non-negative p."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf'), 0.0, None
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    max_c = float(c.max())
    sc = 2.0 * n * max_c
    return sc, max_c, c


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _compute_gradient(p, w, L=None):
    """
    Compute gradient g[j] = 2 * sum_k w[k] * p[k-j] for the given peak weights w.
    w has length 2n-1 (the convolution length).
    """
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    g = 2.0 * np.fft.irfft(np.conj(W) * P, L)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return g


def _gradient_descent_step(p, c, top_k=3, alphas=None):
    """
    One step of multiplicative gradient descent.
    c is the convolution array (length 2n-1).
    Returns (new_p, new_score).
    """
    n = len(p)
    conv_len = len(c)
    
    # Find top-k peak positions
    k = min(top_k, conv_len)
    top_idx = np.argsort(c)[-k:][::-1]  # sorted descending
    
    # Build weights
    w = np.zeros(conv_len)
    w[top_idx] = 1.0 / k
    
    # Compute gradient
    L = _next_pow2(3 * n)
    g = _compute_gradient(p, w, L)
    
    if g.max() < 1e-15:
        sc = 2.0 * n * c.max()
        return p.copy(), sc
    
    g_norm = g / g.max()
    
    if alphas is None:
        alphas = [0.03, 0.06, 0.1, 0.2, 0.4, 0.8, 1.5]
    
    best_p = p.copy()
    best_sc = 2.0 * n * c.max()
    
    for alpha in alphas:
        p_try = p * np.exp(-alpha * g_norm)
        p_try = np.maximum(p_try, 1e-15 / n)
        s = p_try.sum()
        if s < 1e-10:
            continue
        p_try = p_try / s
        sc = _score(p_try)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_p = p_try.copy()
    
    return best_p, best_sc


def _build_conv_matrix(p):
    """Build matrix A (2n-1 x n) such that A @ q = conv(p, q)."""
    n = len(p)
    # A[k, j] = p[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    # Row k has p[k], p[k-1], ..., p[k-j], ..., p[k-(n-1)] in positions 0..n-1
    # but only where the index is in [0, n-1]
    
    # More efficiently: row k of A is the cross-correlation of delta_k with p
    # A[k, :] = [p[k-0], p[k-1], ..., p[k-(n-1)]] with out-of-range = 0
    
    # We can build this using the convolution structure
    # A is the "convolution matrix" of p (transposed)
    
    # For efficiency with large n, we'll build it row by row
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: A[k, j] = p[k-j] for max(0, j) <= k <= min(2n-2, j+n-1)
        k_start = j  # k - j >= 0 => k >= j
        k_end = min(2 * n - 2, j + n - 1)  # k - j <= n-1 => k <= j+n-1
        if k_start <= k_end:
            A[k_start:k_end + 1, j] = p[0:k_end - k_start + 1]
    
    return A


def _lp_spread(p, M, max_iter_lp=50000):
    """
    Solve LP: maximize sum(q) subject to conv(p,q) <= M (element-wise), q >= 0.
    Returns normalized q or None if infeasible/failed.
    """
    n = len(p)
    A = _build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': max_iter_lp, 'time_limit': 30})
        if res.success or (hasattr(res, 'status') and res.status in [0, 3]):
            q = res.x
            q = np.maximum(q, 0.0)
            s = q.sum()
            if s > 1e-10:
                return q / s
    except Exception:
        pass
    return None


def _power_transform(p, gamma):
    """Apply power transform: p <- normalized(p^gamma)."""
    p_pos = np.maximum(p, 1e-15)
    p_new = np.power(p_pos, gamma)
    s = p_new.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p_new / s


def _block_swap(p, rng):
    """Swap two random blocks of the sequence."""
    n = len(p)
    # Choose block sizes
    bs1 = int(rng.integers(2, max(3, n // 8)))
    bs2 = int(rng.integers(2, max(3, n // 8)))
    # Choose positions
    s1 = int(rng.integers(0, n - bs1))
    s2 = int(rng.integers(0, n - bs2))
    if abs(s1 - s2) < max(bs1, bs2):
        return None  # Overlapping blocks, skip
    
    p_try = p.copy()
    block1 = p_try[s1:s1 + bs1].copy()
    block2 = p_try[s2:s2 + bs2].copy()
    p_try[s1:s1 + bs1] = block2[:bs1] if len(block2) >= bs1 else np.pad(block2, (0, bs1 - len(block2)))
    p_try[s2:s2 + bs2] = block1[:bs2] if len(block1) >= bs2 else np.pad(block1, (0, bs2 - len(block1)))
    
    # This doesn't work well for different-sized blocks. Let's use same size.
    bs = min(bs1, bs2)
    p_try = p.copy()
    block1 = p_try[s1:s1 + bs].copy()
    block2 = p_try[s2:s2 + bs].copy()
    p_try[s1:s1 + bs] = block2
    p_try[s2:s2 + bs] = block1
    
    return _sanitize(p_try)


def _symmetrize(p):
    """Make sequence symmetric: p_new[i] = (p[i] + p[n-1-i]) / 2."""
    n = len(p)
    p_new = (p + p[::-1]) / 2.0
    return _sanitize
