def _sanitize(arr, min_value=0.0, normalize=True, max_value=1000.0):
    if arr is None:
        return None
    try:
        out = np.asarray(arr, dtype=float).ravel()
    except Exception:
        return None
    if out.size == 0:
        return None
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, max_value)
    if min_value > 0.0:
        out = np.maximum(out, float(min_value))
    if normalize:
        s = float(out.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        out = out / s
    return out

def _grad_lag_weight(p, w_lag, L):
    n = int(p.size)
    W = np.zeros(L, dtype=float)
    if n == 0:
        return np.array([])
    W[0] = float(w_lag[0])
    if n > 1:
        half = 0.5 * w_lag[1:]
        W[1:n] = half

    FW = np.fft.rfft(W, L)

```
def _gaussian_smooth(v, sigma):
    v=np.asarray(v,dtype=float)
    n=v.size
    if n==0: return v
    sigma=float(sigma)
    if sigma<=0 or n<=1: return v
    rad=int(max(1, math.ceil(3.0*sigma)))
    k=np.arange(-rad, rad+1, dtype=float)
    kern=np.exp(-0.5*(k/sigma)**2)
    s=kern.sum()
    if s<=0: return v

       out=np.fft.irfft(V*K, L)[:n]
       return out
    out=np.convolve(v, kern, mode='same')
```

```
def _gaussian_smooth(v, sigma):
    v=np.asarray(v,dtype=float).ravel()
    n=int(v.size)
    if n==0: return v
    sigma=float(sigma)
    if sigma <= 0.0 or n <= 2: return v
    rad=int(max(1, math.ceil(3.0*sigma)))
    k=np.arange(-rad, rad+1, dtype=float)
    kern=np.exp(-0.5*(k/sigma)**2)
    s=float(kern.sum())
    if s <= 0.0: return v
```

```python
"""
Search for low-scoring non-negative step-height sequences.

Strategy:
- Use the provided parent sequence when available, plus diverse starts.
- Diagnose the official evaluator at runtime to adapt priorities.
- Use fast FFT mirror-descent-style optimization for several smooth/threshold
  autocorrelation proxies (full, off-zero, and mixed zero-weight variants).
- Refine top candidates with black-box official-evaluator local searches:
  low-rank multiplicative coordinate moves and block scaling.
- Finally, rank candidates by the official evaluator where possible and return
  the best normalized non-negative sequence.
"""

import numpy as np
import time
import math

_GLOBAL_RNG = np.random.default_rng(12345)
_EVAL_CALLS = 0
_EVAL_MEAN = 0.0
_MIN_P = 1e-15


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _sanitize(arr, min_value=0.0, normalize=True, max_value=1000.0):
    """Finite, clipped, non-negative numpy array, optionally normalized to sum 1."""
    if arr is None:
        return None
    try:
        out = np.asarray(arr, dtype=float).ravel()
    except Exception:
        return None
    if out.size == 0:
        return None
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, float(max_value))
    if min_value > 0.0:
        out = np.maximum(out, float(min_value))
    if normalize:
        s = float(out.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        out = out / s
    return out


def _uniform(n):
    n = int(max(1, n))
    return np.ones(n, dtype=float) / float(n)


def _autocorr_pos(p, L=None):
    """
    Positive-lag linear autocorrelation c[0:n], where c[d] = sum_i p_i p_{i+d}.
    c[0] is sum(p^2).
    """
    n = int(p.size)
    if n == 0:
        return np.array([], dtype=float)
    if L is None:
        L = _next_pow2(2 * n - 1)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * np.conj(Fp), L)
    c = c[:n].copy()
    c = np.maximum(c, 0.0)
    return c


def _grad_lag_weight(p, w_lag, L):
    """
    Gradient wrt p of sum_{d=0}^{n-1} w_lag[d] * c[d],
    where c[d] is positive-lag linear autocorrelation.
    """
    n = int(p.size)
    if n == 0:
        return np.array([], dtype=float)
    W = np.zeros(L, dtype=float)
    W[0] = float(w_lag[0])
    if n > 1:
        half = 0.5 * w_lag[1:]
        W[1:n] = half
        W[L - n + 1:L] = half[::-1]
    Fp = np.fft.rfft(p, L)
    FW = np.fft.rfft(W, L)
    g = np.fft.irfft(np.conj(Fp) * FW, L)
    g = g[:n]
    return 2.0 * g


def _cost_array(p, c, mask, reg_type, reg_weight):
    """Internal proxy cost: weighted peak autocorrelation plus optional regularization."""
    n = int(p.size)
    if n <= 0 or c.size == 0:
        return float("inf"), 0.0
    v = c * mask
    maxv = float(np.max(v)) if v.size else 0.0
    if maxv < 0.0:
        maxv = 0.0
    score = 2.0 * float(n) * maxv

    reg = 0.0
    if reg_weight > 0.0:
        c0 = float(c[0])
        if reg_type == "c0":
            reg = float(reg_weight) * c0
        elif reg_type == "center":
            reg = float(reg_weight) * max(0.0, c0 - 1.0 / float(n))
        elif reg_type == "entropy":
            pp = np.maximum(p, 1e-300)
            H = -float(np.sum(p * np.log(pp)))
            Hmax = math.log(max(2, n))
            reg = float(reg_weight) * (1.0 - H / Hmax)
    return score + reg, maxv


def _smooth_weights_v(v, beta, mask):
    """Softmax weights over active lags for smooth maximization."""
    n = int(v.size)
    w = np.zeros(n, dtype=float)
    active = mask > 0.0
    if not np.any(active):
        return w
    pos = np.nonzero(active)[0]
    va = v[pos]
    if va.size == 0:
        return w
    maxva = float(np.max(va))
    if beta <= 0.0:
        w[pos] = 1.0 / float(pos.size)
        return w
    z = float(beta) * (va - maxva)
    z = np.where(z > -745.0, z, -745.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w[pos[int(np.argmax(va))]] = 1.0
        return w
    w[pos] = e / s
    return w


def _threshold_weights_v(v, mask, t):
    """Hard threshold weights over active lags above t."""
    n = int(v.size)
    w = np.zeros(n, dtype=float)
    active = (mask > 0.0) & (v > float(t))
    if not np.any(active):
        return w
    pos = np.nonzero(active)[0]
    val = v[pos] - float(t)
    mx = float(np.max(val))
    if mx <= 0.0:
        return w
    val = val / mx
    val = val * val
    s = float(val.sum())
    if s <= 0.0 or not np.isfinite(s):
        m = int(val.size)
        val = np.ones(m, dtype=float) / float(m)
    w[pos] = val / s
    return w


def _fft_mirror_optimize(p0, time_limit, mask, reg_type="none", reg_weight=0.0,

    Fast internal optimizer for proxy objective min 2*n*max(mask*c(pc)).
    Uses multiplicative mirror gradients derived from smoothed/threshold max.
    """

    if p is None:
        return None, float("inf")
    n = int(p.size)
    if n <= 1:
        return p, 0.0
    time_limit = float(max(0.0, time_limit))
    deadline = time.monotonic() + time_limit

    if mask.size != n:
        mask = np.ones(n, dtype=float)
    active = mask > 0.0
    if not np.any(active):
        c = _autocorr_pos(p, L)
        cost, _ = _cost_array(p, c, mask, reg_type, reg_weight)
        return p, cost

    cur_cost, _ = _cost_array(p, c, mask, reg_type, reg_weight)

    eta = float(eta_start)
    min_eta = 0.02
    max_eta = 3.0
    stall = 0
    it = 0

            p_try = _sanitize(p_try, min_p, True)

                c_try = _autocorr_pos(p_try, L)
                cost_try, _ = _cost_array(p_try, c_try, mask, reg_type, reg_weight)
                if cost_try < cur_cost:
                    p = p_try
                    cur_cost = cost_try
            stall += 1
            it += 1
            if stall > 100:
                p = best_p.copy()
                cur_cost = best_cost
                stall = 0
                eta = eta_start
            continue

        gn = g / gmax
        best_cost_try = cur_cost
        best_p_try = None

        for a in (eta, 0.5 * eta, 2.0 * eta):
            if a <= 0.0:
                continue
            ex = -a * gn
            ex = np.clip(ex, -50.0, 50.0)
            p_try = p * np.exp(ex)
            p_try = _sanitize(p_try, min_p, True)
            if p_try is None:

            cost_try, _ = _cost_array(p_try, c_try, mask, reg_type, reg_weight)
            if np.isfinite(cost_try) and cost_try < best_cost_try - 1e-14:
                best_cost_try = float(cost_try)
                best_p_try = p_try

        if best_p_try is not None:
            p = best_p_try
            cur_cost = best_cost_try
            stall = 0

        else:
            eta = max(min_eta, eta * 0.72)
            stall += 1

            if stall > 15:
                sigma = 0.02 + 0.001 * min(100.0, float(stall))
                p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, sigma, n))
                p_try = _sanitize(p_try, min_p, True)
                if p_try is not None:
                    c_try = _autocorr_pos(p_try, L)
                    cost_try, _ = _cost_array(p_try, c_try, mask, reg_type, reg_weight)

                        cur_cost = cost_try
                        stall = 0

                cur_cost = best_cost

        if it % 100 == 0 and time.monotonic() < deadline:
            beta_cur = min(1000.0, beta_cur * 1.15)

    return best_p, best_cost


def _full_internal_score(p):
    p = _sanitize(p, 0.0, True)
    if p is None:
        return float("inf")
    c = _autocorr_pos(p)
    if c.size == 0:
        return float("inf")
    return 2.0 * float(p.size) * float(c.max())


def _official_score(arr):
    """Call the official evaluator, with fallback to the internal full objective."""
    global _EVAL_CALLS, _EVAL_MEAN
    a = _sanitize(arr, 0.0, True)
    if a is None:
        return float("inf")
    lst = a.tolist()
    t0 = time.monotonic()
    try:
        val = float(evaluate_sequence(lst))
    except NameError:
        val = _full_internal_score(a)
    except Exception:
        val = _full_internal_score(a)
    dt = time.monotonic() - t0
    _EVAL_CALLS += 1
    if 0.0 < dt < 5.0:
        if _EVAL_MEAN <= 0.0:
            _EVAL_MEAN = dt
        else:
            _EVAL_MEAN = 0.8 * _EVAL_MEAN + 0.2 * dt
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def _official_budget(seconds):
    per = _EVAL_MEAN if _EVAL_MEAN > 0.0 else 0.005
    per = max(0.0005, 1.6 * per)
    calls = int(float(seconds) / per)
    if calls < 30:
        calls = 30
    if calls > 8000:
        calls = 8000
    return calls


def _candidate_sort_key(item):
    off = item[0]
    if off >= 1e99:
        return (1, 1e99, item[1])
    return (0, off, item[1])


def _add_candidate(pool, arr, official=None, key=None, max_pool=512):
    a = _sanitize(arr, 0.0, True)
    if a is None:
        return
    if key is None:
        c = _autocorr_pos(a)
        if c.size:
            full = 2.0 * float(a.size) * float(c.max())
        else:
            full = float("inf")
        off = float("inf")
        if a.size > 1 and c.size > 1:
            off = 2.0 * float(a.size) * float(c[1:].max())
        key = min(full, off)
    if official is None:
        official = 1e100
    pool.append((float(official), float(key), int(a.size), a.copy()))
    if len(pool) > max_pool:
        pool.sort(key=_candidate_sort_key)
        del pool[max_pool:]


def _resample(p, m):
    m = int(max(1, m))
    a = _sanitize(p, 0.0, True)
    if a is None:
        return None
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=float)
    x_old = (np.arange(n, dtype=float) + 0.5) / float(n)
    x_new = (np.arange(m, dtype=float) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out, _MIN_P, True)


def _gaussian_smooth(v, sigma):
    v = np.asarray(v, dtype=float).ravel()
    n = int(v.size)
    if n == 0:
        return v
    sigma = float(sigma)
    if sigma <= 0.0 or n <= 2:
        return v
    rad = int(max(1, math.ceil(3.0 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    s = float(kern.sum())
    if s <= 0.0:
        return v
    kern = kern / s
    if (2 * rad + 1) > max(16, n // 8):
        L = _next_pow2(n + 2 * rad + 1)
        Fv = np.fft.rfft(v, L)
        Fk = np.fft.rfft(kern, L)
        out = np.fft.irfft(Fv * Fk, L)[:n]
        return out
    return np.convolve(v, kern, mode="same")


def _smooth_array(p, sigma):
    a = _sanitize(p, 0.0, True)
    if a is None or sigma <= 0.0 or a.size < 5:
        return a
    b = _gaussian_smooth(a, float(sigma))
    return _sanitize(b, _MIN_P, True)


def _noise_array(p, sigma):
    a = _sanitize(p, 0.0, True)
    if a is None or sigma <= 0.0:
        return a
    b = a * np.exp(float(sigma) * _GLOBAL_RNG.normal(0.0, 1.0, a.size))
    return _sanitize(b, _MIN_P, True)


def _random_block_vector(n, block):
    n = int(max(1, n))
    block = int(max(1, block))
    nb = max(1, int(math.ceil(n / float(block))))
    vals = _GLOBAL_RNG.normal(0.0, 1.0, nb)
    v = np.repeat(vals, block)
    v = v[:n]
    v = v.astype(float, copy=False) - float(np.mean(v))
    m = float(np.max(np.abs(v)))
    if m <= 1e-15:
        return None
    return v / m


def _make_local_directions(n, count=18):
    dirs = []
    n = int(max(1, n))
    if n <= 1:
        return dirs

    t = np.arange(n, dtype=float) / float(n)

    wave_specs = ((1, 0.0), (1, 0.7), (2, 0.0), (3, 0.5), (5, 0.2), (8, 0.0), (13, 0.1))
    for k, ph in wave_specs:
        if len(dirs) >= count:
            break
        v = np.cos(2.0 * np.pi * float(k) * t + float(ph))
        v = v - float(np.mean(v))
        m = float(np.max(np.abs(v)))
        if m > 1e-15:
            dirs.append(v / m)

    for sig in (4.0, 16.0, 64.0, 256.0):
        if len(dirs) >= count:
            break
        v = _GLOBAL_RNG.normal(size=n)
        v = _gaussian_smooth(v, sig)
        v = v - float(np.mean(v))
        m = float(np.max(np.abs(v)))
        if m > 1e-15:
            dirs.append(v / m)

    for block in (16, 64, 256, 1024):
        if len(dirs) >= count:
            break
        v = _random_block_vector(n, block)
        if v is not None:
            dirs.append(v)

    i = 0
    while len(dirs) < count and i < 100:
        sig = 1.0 + float(_GLOBAL_RNG.integers(1, 200))
        v = _GLOBAL_RNG.normal(size=n)
        v = _gaussian_smooth(v, sig)
        v = v - float(np.mean(v))
        m = float(np.max(np.abs(v)))
        if m > 1e-15:
            dirs.append(v / m)
        i += 1

    if not dirs:
        v = np.zeros(n, dtype=float)
        v[0] = 1.0
        dirs.append(v)

    return dirs[:count]


def _official_local_lowrank(base, time_limit, max_evals):
    """Black-box local search over low-rank multiplicative directions."""
    p = _sanitize(base, 1e-20, True)
    if p is None:
        return None, float("inf")
    cur = _official_score(p)
    if not np.isfinite(cur):
        return p, cur

    n = int(p.size)
    dirs = _make_local_directions(n, 18)
    deadline = time.monotonic() + float(max(0.0, time_limit))
    base_s = p.copy()
    stall = 0

    while time.monotonic() < deadline and _EVAL_CALLS < max_evals:
        order = _GLOBAL_RNG.permutation(len(dirs))
        improved = False

        for idx in order:
            if _EVAL_CALLS >= max_evals or time.monotonic() >= deadline:
                break
            v = dirs[int(idx)]
            p_safe = np.maximum(p, 1e-20)
            s = float(p_safe.sum())
            if s <= 0.0:
                continue
            p_safe = p_safe / s

            best_p = None
            best_s = cur
            for step in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5):
                for sign in (1.0, -1.0):
                    if _EVAL_CALLS >= max_evals:
                        break
                    p_try = p_safe * np.exp(np.clip(sign * step * v, -30.0, 30.0))
                    p_try = _sanitize(p_try, 0.0, True)
                    if p_try is None:
                        continue
                    sc = _official_score(p_try)
                    if np.isfinite(sc) and sc < best_s - 1e-12:
                        best_s = sc
                        best_p = p_try

            if best_p is not None and best_s < cur - 1e-12:
                p = best_p
                cur = best_s
                base_s = p.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if len(dirs) < 28:
                sig = 8.0 + 16.0 * float(_GLOBAL_RNG.integers(0, 8))
                v = _GLOBAL_RNG.normal(size=n)
                v = _gaussian_smooth(v, sig)
                v -= float(np.mean(v))
                m = float(np.max(np.abs(v)))
                if m > 1e-15:
                    dirs.append(v / m)

            if stall > 3:
                p_try = _noise_array(base_s, 0.05 + 0.01 * min(10.0, float(stall)))
                if p_try is not None:
                    sc = _official_score(p_try)
                    if np.isfinite(sc) and sc < cur:
                        p = p_try
                        cur = sc
                        base_s = p.copy()
                        stall = 0
            if stall > 10:
                break

    return p, cur


def _official_local_blocks(base, time_limit, max_evals):
    """Black-box local search by scaling random contiguous blocks."""
    p = _sanitize(base, 1e-20, True)
    if p is None:
        return None, float("inf")
    cur = _official_score(p)
    if not np.isfinite(cur):
        return p, cur

    n = int(p.size)
    deadline = time.monotonic() + float(max(0.0, time_limit))
    raw_blocks = [max(1, n // 4096), 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    block_list = [int(b) for b in raw_blocks if 1 <= int(b) <= n]
    if not block_list:
        block_list = [1]

    stall = 0
    while time.monotonic() < deadline and _EVAL_CALLS < max_evals:
        w = int(_GLOBAL_RNG.choice(block_list))
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))
        best_p = None
        best_s = cur

        for f in (0.5, 0.75, 0.9, 1.1, 1.3, 1.6):
            if _EVAL_CALLS >= max_evals:
                break
            cand = p.copy()
            cand[start:start + w] *= float(f)
            cand = _sanitize(cand, 0.0, True)
            if cand is None:
                continue
            sc = _official_score(cand)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best_s = sc
                best_p = cand

        if best_p is not None:
            p = best_p
            cur = best_s
            stall = 0
        else:
            stall += 1
            if stall > 4:
                p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, 0.05, n))
                p_try = _sanitize(p_try, 0.0, True)
                if p_try is not None:
                    sc = _official_score(p_try)
                    if np.isfinite(sc) and sc < cur:
                        p = p_try
                        cur = sc
                        stall = 0
            if stall > 25:
                break

    return p, cur


def _initial_sequence(n, mode=None):
    n = int(max(2, n))
    if mode is None:
        mode = int(_GLOBAL_RNG.integers(0, 6))
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    p = np.ones(n, dtype=float)

    if mode == 0:
        z = 1.0
        for kk in range(1, 13):
            amp = float(_GLOBAL_RNG.normal(0.0, 1.0 / (kk + 1)))
            ph = float(_GLOBAL_RNG.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * float(kk) * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        a = float(_GLOBAL_RNG.uniform(0.05, 3.0))
        b = float(_GLOBAL_RNG.uniform(0.05, 3.0))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)

    elif mode == 2:
        K = int(_GLOBAL_RNG.integers(1, 5))
        p = np.zeros(n, dtype=float)
        centers = np.sort(_GLOBAL_RNG.uniform(0.0, 1.0, K))
        sigmas = _GLOBAL_RNG.uniform(0.02, 0.35, K)
        weights = _GLOBAL_RNG.uniform(0.5, 2.0, K)
        for ii in range(K):
            s = max(1e-6, float(sigmas[ii]))
            p = p + float(weights[ii]) * np.exp(-((t - float(centers[ii])) / s) ** 2)

    elif mode == 3:
        K = int(_GLOBAL_RNG.integers(3, 10))
        knots = np.sort(_GLOBAL_RNG.uniform(0.0, 1.0, K))
        heights = _GLOBAL_RNG.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        K = int(_GLOBAL_RNG.integers(3, 17))
        edges = np.sort(_GLOBAL_RNG.uniform(0.0, 1.0, max(1, K - 1)))
        blocks = [0.0]
        blocks.extend([float(x) for x in edges])
        blocks.append(1.0)
        heights = _GLOBAL_RNG.lognormal(0.0, 1.0, K)
        p = np.zeros(n, dtype=float)
        for ii in range(K):
            s = max(0, int(round(blocks[ii] * n)))
            e = max(s + 1, int(round(blocks[ii + 1] * n)))
            e = min(n, e)
            p[s:e] = float(heights[ii])

    else:
        c = float(_GLOBAL_RNG.uniform(0.2, 0.8))
        k = float(_GLOBAL_RNG.uniform(5.0, 30.0))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50.0, 50.0)))
        p = p * np.exp(-((t - c) / 0.25) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0) + 1e-12
    return _sanitize(p, _MIN_P, True)


def _make_starts(n, parent, limit=4):
    n = int(max(2, n))
    starts = []
    if parent is not None:
        sp = _resample(parent, n)
        if sp is not None:
            starts.append(sp)
            sp2 = _smooth_array(sp, 1.0)
            if sp2 is not None:
                starts.append(sp2)
            sp3 = _noise_array(sp, 0.03)
            if sp3 is not None:
                starts.append(sp3)

    if len(starts) < limit:
        starts.append(_uniform(n))

    for mode in (1, 2, 4, 0, 3, 5):
        if len(starts) >= limit:
            break
        rp = _initial_sequence(n, mode)
        if rp is not None:
            starts.append(rp)

    out = []
    for sp in starts:
        a = _sanitize(sp, 0.0, True)
        if a is not None:
            out.append(a)
    return out[:limit]


def _make_mask(n, kind, gamma):
    n = int(max(1, n))
    m = np.ones(n, dtype=float)
    if n <= 1:
        m[0] = 1.0
        return m
    if kind == "full":
        m[:] = 1.0
    elif kind == "off":
        m[0] = 0.0
    elif kind == "mix":
        m[0] = float(gamma)
    else:
        m[:] = 1.0
    return m


def _run_fft_candidate(pool, p_start, n, kind, gamma, reg_type, reg_weight,
                       target, beta, time_limit, min_p, threshold_prob):
    if p_start is None or time_limit <= 0.0:
        return
    p0 = _resample(p_start, n)
    if p0 is None:
        p0 = _uniform(n)
    mask = _make_mask(n, kind, gamma)
    p, _ = _fft_mirror_optimize(
        p0, time_limit, mask, reg_type, reg_weight, beta, target,
        threshold_prob, min_p
    )
    if p is not None:
        _add_candidate(pool, p)
        for sig in (1.0, 4.0):
            p2 = _smooth_array(p, sig)
            if p2 is not None:
                _add_candidate(pool, p2)

        if alloc > 700.0:
            alloc = 700.0
        cap = _EVAL_CALLS + _official_budget(alloc * 0.75)
        p, sc = _official_local_lowrank(top_arrs[i], alloc, cap)
        if p is not None:
            _add_candidate(pool, p, sc)

    rem = float(local_end - time.monotonic())
    if rem > 90.0 and pool:
        pool.sort(key=_candidate_sort_key)
        best_arr = pool[0][3]
        alloc = min(rem * 0.75, 450.0)
        cap = _EVAL_CALLS + _official_budget(alloc * 0.75)
        p, sc = _official_local_blocks(best_arr, alloc, cap)
        if p is not None:
            _add_candidate(pool, p, sc)

    print(f"after local: pool={len(pool)}, eval_calls={_EVAL_CALLS}", flush=True)

    # Final official ranking of candidates.
    final_end = deadline - 10.0
    pool.sort(key=_candidate_sort_key)

    cap = _EVAL_CALLS + _official_budget(max(0.0, final_end - time.monotonic()) * 0.65)
    for item in pool:
        if _EVAL_CALLS >= cap or time.monotonic() > final_end:
            break
        if item[0] < 1e99:
            continue
        sc = _official_score(item[3])
        _add_candidate(pool, item[3], sc)

    pool.sort(key=_candidate_sort_key)
    cap2 = _EVAL_CALLS + _official_budget(max(0.0, final_end - time.monotonic()) * 0.35)
    if cap2 > _EVAL_CALLS and len(pool) > 1:
        order = _GLOBAL_RNG.permutation(len(pool))
        for idx in order:
            if _EVAL_CALLS >= cap2 or time.monotonic() > final_end:
                break
            item = pool[int(idx)]
            if item[0] < 1e99:
                continue
            sc = _official_score(item[3])
            _add_candidate(pool, item[3], sc)

    best_off = float("inf")
    best_arr = None
    for item in pool:
        if item[0] < best_off:
            best_off = item[0]
            best_arr = item[3]

    if best_arr is None:
        if pool:
            best_arr = pool[0][3]
        else:
            best_arr = _uniform(3072)

    out = _sanitize(best_arr, 0.0, True)
    if out is None:
        out = _uniform(3072)

    out = np.asarray(out, dtype=float).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = _uniform(3072)
    else:
        out = out / s

    # Remove obviously numerical dust; keep sum positive.
    out[out < 1e-18] = 0.0
    s = float(out.sum())
    if s <= 0.0:
        out = _uniform(3072)
        s = 1.0
    else:
        out = out / s

    try:
        final_score = _official_score(out)
    except Exception:
        final_score = _full_internal_score(out)

    print(f"final length={int(out.size)}, official_score={final_score:.7f}, "
          f"eval_calls={_EVAL_CALLS}", flush=True)

    return [float(x) for x in out]
```
