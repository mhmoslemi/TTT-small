import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_lp(a, M, top_constraints=None):
    """
    Solve the LP: maximize sum(g) subject to g >= 0 and (a*g)[k] <= M for all k.
    If top_constraints is specified, only use those constraints.
    """
    n = len(a)
    
    # Build constraint matrix A (sparse)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We want: A @ g <= M (vector of M's)
    
    if top_constraints is not None:
        rows = top_constraints
    else:
        rows = range(2*n - 1)
    
    n_rows = len(rows) if isinstance(rows, (list, np.ndarray)) else 2*n-1
    
    # Build sparse matrix
    A = lil_matrix((n_rows, n))
    b = np.full(n_rows, M)
    
    for idx, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            A[idx, j] = a[k - j]
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A.tocsr(), b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def gradient_topk(a, c, topk, temp):
    """Compute gradient using top-k soft-max."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    
    # Find top-k peaks
    k = min(topk, c.size)
    part = np.argpartition(c, -k)[-k:]
    vals = c[part]
    
    # Soft-max weights
    temp_eff = max(temp, 1e-15)
    wts = np.exp((vals - vals.max()) / temp_eff)
    wts /= wts.sum()
    
    # Weighted max
    M_weighted = float(np.dot(vals, wts))
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(k):
        wi = float(wts[idx])
        if wi < 1e-10:
            continue
        ki = int(part[idx])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses adaptive soft-max gradient with curriculum-style hyperparameter scheduling,
multiple initialization strategies, and LP-assisted direction finding for
escaping local minima. Combines Adam optimization with periodic Frank-Wolfe
style LP steps for robust global exploration.

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via real FFT. Both a and b have same length n."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_gradient(a, c, topk, temp):
    """
    Compute gradient using top-k soft-max approximation.
    Returns (gradient, weighted_max, sum).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S

    k = min(topk, c.size)
    part = np.argpartition(c, -k)[-k:]
    vals = c[part].copy()

    v_max = float(vals.max())
    if v_max < 1e-15:
        return np.zeros(n), 0.0, S

    exps = np.exp((vals - v_max) / max(temp, 1e-15))
    wts = exps / exps.sum()
    M_w = float(np.dot(vals, wts))

    grad = np.zeros(n)
    for idx in range(k):
        wi = float(wts[idx])
        if wi < 1e-12:
            continue
        ki = int(part[idx])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            grad[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    return 2.0 * n * (grad / S2 - 2.0 * M_w / (S2 * S)), M_w, S


def adam_update(a, g, m, v, t, lr):
    """In-place Adam optimizer step."""
    b1, b2 = 0.9, 0.999
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    a[:] -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)


def solve_lp_direction(a, M, top_k_constraints=200):
    """
    Solve reduced LP to find Frank-Wolfe direction.
    maximize sum(g) s.t. g >= 0, (a*g)[k] <= M for top-k constraints.
    Returns the optimal g or None on failure.
    """
    from scipy.optimize import linprog

    n = len(a)
    c = fast_self_conv(a)

    # Select top constraints (peaks closest to M)
    k = min(top_k_constraints, c.size)
    peak_indices = np.argpartition(c, -k)[-k:]
    vals = c[peak_indices]

    # Build sparse constraint matrix
    rows = peak_indices
    n_rows = len(rows)

    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)

    # Build constraint matrix A (n_rows x n)
    # A[i, j] = a[rows[i] - j] if 0 <= rows[i] - j < n
    A_list = []
    b_ub = np.full(n_rows, M)

    for idx in range(n_rows):
        ki = int(rows[idx])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        row = np.zeros(n)
        if js <= je:
            row[js:je + 1] = a[ki - js:ki - je - 1:-1] if je > js else a[ki - js]
            # Actually: A[idx, j] = a[ki - j] for j in [js, je]
            row[js:je + 1] = a[ki - np.arange(js, je + 1)]
        A_list.append(row)

    A_mat = np.array(A_list)

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    per_init = (p1_end - start) / n_inits

    for i, a0 in enumerate(inits):
        t_end = min(p1_end, start + per_init * (i + 1) + per_init * 0.3)
        if time.monotonic() >= p1_end:
            break
        sc, aa = optimize_phase(a0, t_end, rng,
                                lr_start=0.08, topk_start=8, temp_start=0.1,
                                patience=80, use_lp=False)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1 init {i}/{n_inits}: {sc:.6f}")

    if best_a is None:
        best_a = np.ones(n)
        best_score = compute_score(n, fast_self_conv(best_a), float(best_a.sum()))

    print(f"Phase 1 best: {best_score:.6f}")

    # === Phase 2: Focused optimization (33% of time) ===
    p2_end = start + 0.45 * (total - 5.0)
    use_lp = n <= 4096
    sc, aa = optimize_phase(best_a, p2_end, rng,
                            lr_start=0.02, topk_start=12, temp_start=0.05,
                            patience=250, use_lp=use_lp, lp_every=50)
    if sc < best_score:
        best_score = sc
        best_a = aa.copy()
    print(f"Phase 2: {best_score:.6f}")

    # === Phase 3: Multi-restart exploration (25% of time) ===
    p3_end = start + 0.70 * (total - 5.0)
    if time.monotonic() < p3_end:
        # Try several perturbed versions of the best solution
        n_restarts = 3
        per_restart = (p3_end - time.monotonic()) / n_restarts
        for r in range(n_restarts):
            if time.monotonic() >= p3_end - 10:
                break
            t_rend = min(p3_end, time.monotonic() + per_restart)
            # Different perturbation strategies
            if r == 0:
                a_pert = best_a * (1.0 + 0.15 * rng.standard_normal(n))
            elif r == 1:
                # Block perturbation
                a_pert = best_a.copy()
                block_size = n // 4
                for b in range(4):
                    s = b * block_size
                    e = s + block_size
                    a_pert[s:e] *= (1.0 + 0.2 * rng.standard_normal(e - s))
            else:
                # Spectral perturbation
                A_fft = np.fft.rfft(best_a)
                A_fft *= (1.0 + 0.1 * rng.standard_normal(A_fft.size))
                a_pert = np.fft.irfft(A_fft, n)

            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, aa = optimize_phase(a_pert, t_rend, rng,
                                    lr_start=0.01, topk_start=16, temp_start=0.03,
                                    patience=200, use_lp=False)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"Phase 3 restart {r}: {best_score:.6f}")

    # === Phase 4: Fine polishing (20% of time) ===
    p4_end = start + 0.90 * (total - 5.0)
    if time.monotonic() < p4_end:
        sc, aa = optimize_phase(best_a, p4_end, rng,
                                lr_start=0.003, topk_start=32, temp_start=0.012,
                                patience=500, use_lp=False)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4: {best_score:.6f}")

    # === Phase 5: Ultra-fine final polish ===
    if time.monotonic() < deadline - 2.0:
        sc, aa = optimize_phase(best_a, deadline - 2.0, rng,
                                lr_start=0.0008, topk_start=48, temp_start=0.005,
                                patience=800, use_lp=False, lr_decay=0.99999)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 5: {best_score:.6f}")

    # Final output
    if best_a is None:
        best_a = np.ones(n)

    # Ensure non-negative and within bounds
    best_a = np.clip(best_a, 0.0, 1000.0)

    out = [float(x) for x in best_a]

    # Verify
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass

    elapsed = time.monotonic() - start
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, "
          f"n={len(out)}, time={elapsed:.1f}s")

    return out
