def score_fast(f):
    """Compute 2*n*max(conv(f,f))/(sum(f))^2"""

    c = fftconvolve(f, f)
    return 2.0 * n * c.max() / (s * s)

def smooth_score(f, top_k=10, temperature=1.0):
    """Smoothed score using log-sum-exp of top peaks."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = fftconvolve(f, f)
    # Get top-k values
    if top_k < len(c):
        top = np.partition(c, -top_k)[-top_k:]
    else:
        top = c
    # Log-sum-exp smoothing
    max_c = top.max()
    lse = max_c + np.log(np.sum(np.exp((top - max_c) / temperature)) / len(top)) * temperature
    return 2.0 * n * lse / (s * s)

from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def _autocorr(f):
    """Linear autocorrelation via FFT."""
    return fftconvolve(f, f)

def _exact_score(f):
    """Compute the exact score."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr(f / s)
    return float(2.0 * n * c.max())

    Smoothed objective and gradient.
    f: raw variable (will be normalized internally)
    Returns (objective_value, gradient)
    """
    n = f.shape[0]
    f_pos = np.maximum(f, 0.0)
    s = f_pos.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    fn = f_pos / s  # normalized to sum=1
    
    c = _autocorr(fn)  # length 2n-1
    
    # Select top-k
    if top_k < c.shape[0]:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(c.shape[0])
    top_vals = c[top_idx]

    max_tv = top_vals.max()
    exp_vals = np.exp((top_vals - max_tv) / temp)
    w_sum = exp_vals.sum()
    w = exp_vals / w_sum

"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with adaptive softmax-smoothed objective, FFT-based gradient,
multiple n values, diverse restarts, annealing, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time


def _autocorr(f):
    """Linear autocorrelation via FFT."""
    return fftconvolve(f, f)


def _exact_score(f):
    """Compute exact score 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    fn = f / s
    c = _autocorr(fn)
    return float(2.0 * n * c.max())


def _smooth_obj_grad(f, temp, top_k):
    """
    Smoothed objective and gradient for L-BFGS-B.
    Uses top-k softmax smoothing of the convolution peaks.
    f is the raw variable; internally normalized to sum=1.
    Returns (objective, gradient).
    """
    n = f.shape[0]
    f_pos = np.maximum(f, 0.0)
    s = f_pos.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    fn = f_pos / s

    c = _autocorr(fn)
    c_len = c.shape[0]

    # Select top-k lags
    if top_k < c_len:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(c_len)
    top_vals = c[top_idx]

    # Softmax weights
    max_tv = top_vals.max()
    exp_vals = np.exp((top_vals - max_tv) / temp)
    w_sum = exp_vals.sum()
    w = exp_vals / w_sum

    # Smoothed objective
    obj_base = max_tv + temp * (np.log(w_sum) - np.log(len(top_idx)))
    obj = 2.0 * n * obj_base

    # Gradient w.r.t. fn (the normalized variable)
    grad_fn = np.zeros(n)
    for kk, wi in zip(top_idx, w):
        k = int(kk)
        # Term 1: fn[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        js1 = max(0, k - n + 1)
        je1 = min(n - 1, k)
        if js1 <= je1:
            idx1 = k - np.arange(js1, je1 + 1)
            grad_fn[js1:je1 + 1] += wi * fn[idx1]
        # Term 2: fn[j-k] for j in [max(0,k), min(n-1,k+n-1)]
        js2 = max(0, k)
        je2 = min(n - 1, k + n - 1)
        if js2 <= je2:
            idx2 = np.arange(js2, je2 + 1) - k
            grad_fn[js2:je2 + 1] += wi * fn[idx2]

    # Chain rule for normalization: fn = f/s
    # d obj / df[i] = (2n/s) * (grad_fn[i] - fn[i] * grad_fn.sum())
    sum_gf = grad_fn.sum()
    grad = (2.0 * n / s) * (grad_fn - fn * sum_gf)

    return obj, grad


def _run_lbfgsb(f0, n, deadline, temp_init, temp_final, top_k, max_iter, verbose=False, label=""):
    """Run L-BFGS-B optimization from starting point f0."""
    f0 = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f0.sum()
    if s < 1e-15:
        f0 = np.ones(n)
    else:
        f0 = f0 / s

    start_time = time.monotonic()
    local_best = _exact_score(f0)
    local_best_f = f0.copy()

    # Annealing schedule
    n_outer = 5  # number of annealing stages
    best_score = local_best
    best_f = f0.copy()
    current_f = f0.copy()

    for stage in range(n_outer):
        if time.monotonic() > deadline:
            break

        # Linear annealing of temperature
        frac = stage / max(1, n_outer - 1)
        temp = temp_init * (temp_final / temp_init) ** frac

        def objective_and_gradient(x):
            if time.monotonic() > deadline:
                return 1e10, np.zeros(n)
            o, g = _smooth_obj_grad(x, temp, top_k)
            return o, g

        # Run L-BFGS-B
        res = minimize(
            objective_and_gradient,
            current_f,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0, None)] * n,
            options={
                'maxiter': max_iter,
                'maxls': 20,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxfun': max_iter * 50,
            }
        )

        if res.x is not None:
            current_f = np.maximum(res.x, 0.0)
            s_cur = current_f.sum()
            if s_cur > 1e-15:
                current_f = current_f / s_cur

        # Evaluate exact score
        score = _exact_score(current_f)
        if score < best_score:
            best_score = score
            best_f = current_f.copy()
            if verbose:
                print(f"      {label} stage {stage} (temp={temp:.4f}): {score:.7f}")
        else:
            if verbose:
                print(f"      {label} stage {stage} (temp={temp:.4f}): {score:.7f} (no improvement)")

        # Also try the L-BFGS result directly
        if res.success:
            score_res = _exact_score(current_f)
            if score_res < best_score:
                best_score = score_res
                best_f = current_f.copy()

    final_score = _exact_score(current_f)
    if final_score < best_score:
        best_score = final_score
        best_f = current_f.copy()

    elapsed = time.monotonic() - start_time
    return best_f, best_score


def _polish_coordinate(f, n, deadline, rng, max_passes=200):
    """
    Coordinate descent polish: find peak lag, identify contributing positions,
    transfer mass from high-contributing to low-contributing positions.
    """
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s

    local_best = _exact_score(f)
    local_best_f = f.copy()

    c = _autocorr(f)
    peak = c.max()
    n_stuck = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Find top 3 peak lags
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]

        improved = False
        for k_star in top3:
            k = int(k_star)
            # Contribution of each position to c[k]
            contrib = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = k - np.arange(js, je + 1)
                contrib[js:je + 1] = f[js:je + 1] * f[idx]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if js2 <= je2:
                idx2 = np.arange(js2, je2 + 1) - k
                contrib[js2:je2 + 1] += f[js2:je2 + 1] * f[idx2]

            if contrib.max() < 1e-20:
                continue

            # Try moving mass from high-contrib to low-contrib
            j_don = int(np.argmax(contrib))
            # Find best receiver: low contribution, not the donor
            contrib_copy = contrib.copy()
            contrib_copy[j_don] = np.inf
            j_rec = int(np.argmin(contrib_copy))

            # Try several transfer fractions
            best_peak = peak
            best_amt = 0.0
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
                amt = f[j_don] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = np.maximum(f_try, 0)
                st = f_try.sum()
                if st < 1e-15:
                    continue
                f_try = f_try / st
                c_try = _autocorr(f_try)
                pk = c_try.max()
                if pk < best_peak - 1e-14:
                    best_peak = pk
                    best_amt = amt

            if best_amt > 0:
                f[j_don] -= best_amt
                f[j_rec] += best_amt
                f = np.maximum(f, 0)
                f = f / f.sum()
                c = _autocorr(f)
                peak = c.max()
                local_best = _exact_score(f)
                local_best_f = f.copy()
                improved = True
                n_stuck = 0
            else:
                n_stuck += 1

        if not improved and n_stuck > 20 and p > 30:
            # Random perturbation
            n_pert = int(n * 0.01)
            idx = rng.integers(0, n, size=n_pert)
            f[idx] *= np.exp(rng.normal(0, 0.3, size=n_pert))
            f = np.maximum(f, 0)
            f = f / f.sum()
            c = _autocorr(f)
            peak = c.max()
            n_stuck = 0

    return local_best_f, local_best


def _polish_gradient(f, n, deadline, rng):
    """
    Short gradient-based polish using tiny steps on the exact (non-smooth)
    objective, targeting only the argmax lag each time.
    """
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s

    local_best = _exact_score(f)
    local_best_f = f.copy()

    c = _autocorr(f)
    peak = c.max()

    n_iters = 50000
    step = 0.05
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        if it % 100 == 0:
            # Recompute convolution
            c = _autocorr(f)
            peak = c.max()

        k_star = int(np.argmax(c))

        # Gradient of c[k_star] w.r.t. f
        grad = np.zeros(n)
        js1 = max(0, k_star - n + 1)
        je1 = min(n - 1, k_star)
        if js1 <= je1:
            idx1 = k_star - np.arange(js1, je1 + 1)
            grad[js1:je1 + 1] += f[idx1]
        js2 = max(0, k_star)
        je2 = min(n - 1, k_star + n - 1)
        if js2 <= je2:
            idx2 = np.arange(js2, je2 + 1) - k_star
            grad[js2:je2 + 1] += f[idx2]

        # Normalize gradient (project out the 1-direction for scale invariance)
        grad = grad - f * grad.dot(f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break

        step = 0.05 * max(0.1, 1.0 - it / n_iters)
        f_new = f - step * grad / grad_norm
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new = f_new / sn

        # Check if peak decreased
        c_new = _autocorr(f_new)
        if c_new.max() < peak:
            f = f_new
            peak = c_new.max()
            if it % 500 == 0:
                sc = _exact_score(f)
                if sc < local_best:
                    local_best = sc
                    local_best_f = f.copy()
        else:
            step *= 0.7

    sc_final = _exact_score(f)
    if sc_final < local_best:
        local_best = sc_final
        local_best_f = f.copy()

    return local_best_f, local_best

            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                best_n = n_try
                print(f"    {name}: {sc0:.5f} -> {sc_opt:.7f} (NEW BEST)")
            else:
                print(f"    {name}: {sc0:.5f} -> {sc_opt:.7f}")

    print(f"\n  After Phase 1: best_score={best_score:.7f}, best_n={best_n}")

    # Phase 2: Focused optimization on the best n
    print(f"\n=== Phase 2: Focused optimization (n={best_n}) ===")
    phase2_budget = min(1400.0, (deadline - time.monotonic()) * 0.55)
    phase2_deadline = time.monotonic() + phase2_budget

    n = best_n
    f_best = best_f.copy()

    # Multiple runs with different parameters
    n_runs = 6
    t_per_run = phase2_budget / n_runs

    for run_i in range(n_runs):
        if time.monotonic() > phase2_deadline:
            break

        d = time.monotonic() + t_per_run * 0.9

        # Generate a perturbation of the best
        if run_i == 0:
            f_start = f_best.copy()
        elif run_i == 1:
            # Sinusoidal perturbation
            j = np.arange(n, dtype=np.float64)
            freq = rng.integers(1, max(2, n // 5) + 1)
            phase = rng.uniform(0, 2 * np.pi)
            f_start = f_best * (1.0 + 0.2 * np.sin(2 * np.pi * freq * j / n + phase))
        elif run_i == 2:
            # Exponential tilt
            alpha = rng.uniform(-0.5, 0.5)
            j = np.arange(n, dtype=np.float64)
            f_start = f_best * np.exp(alpha * (j / n - 0.5))
        elif run_i == 3:
            # Random perturbation
            f_start = f_best * np.exp(rng.normal(0, 0.2, size=n))
        elif run_i == 4:
            # Windowing
            j = np.arange(n, dtype=np.float64)
            width = rng.uniform(0.4, 0.9)
            f_start = f_best * (1.0 - width * ((j / n - 0.5) / 0.5) ** 2)
        else:
            # Fresh random start
            f_start = rng.dirichlet(np.ones(n) * 0.5)

        f_start = np.maximum(f_start, 0.0)
        ss = f_start.sum()
        if ss < 1e-10:
            f_start = np.ones(n)
        else:
            f_start = f_start / ss

        # varying top_k and temperature
        top_k = 10 + 5 * run_i
        temp_init = 0.02 * (0.7 ** run_i)
        temp_final = 0.002 * (0.7 ** run_i)

        f_opt, sc_opt = _run_lbfgsb(
            f_start, n, d,
            temp_init=temp_init, temp_final=temp_final,
            top_k=top_k, max_iter=800,
            verbose=True, label=f"run{run_i}"
        )

        sc_opt = _exact_score(f_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    run{run_i}: NEW BEST {sc_opt:.7f}")
        else:
            print(f"    run{run_i}: {sc_opt:.7f}")

    print(f"\n  After Phase 2: best_score={best_score:.7f}")

    # Phase 3: Polish
    print(f"\n=== Phase 3: Coordinate polish ===")
    polish_budget = min(200.0, max(60.0, (deadline - time.monotonic()) * 0.3))
    polish_deadline = time.monotonic() + polish_budget
    f_pol, sc_pol = _polish_coordinate(best_f, n, polish_deadline, rng, max_passes=500)
    sc_pol = _exact_score(f_pol)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"    Polish: {sc_pol:.7f} (improved)")
    else:
        print(f"    Polish: {sc_pol:.7f}")

    # Phase 4: Gradient polish
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Gradient polish ===")
        grad_budget = min(150.0, max(30.0, (deadline - time.monotonic()) * 0.4))
        grad_deadline = time.monotonic() + grad_budget
        f_grad, sc_grad = _polish_gradient(best_f, n, grad_deadline, rng)
        sc_grad = _exact_score(f_grad)
        if sc_grad < best_score:
            best_score = sc_grad
            best_f = f_grad.copy()
            print(f"    Gradient polish: {sc_grad:.7f} (improved)")
        else:
            print(f"    Gradient polish: {sc_grad:.7f}")

    # Phase 5: One more L-BFGS-B pass with very tight parameters
    if time.monotonic() < deadline - 40:
        print(f"\n=== Phase 5: Final L-BFGS-B pass ===")
        final_deadline = deadline - 10
        f_final, sc_final = _run_lbfgsb(
            best_f, n, final_deadline,
            temp_init=0.005, temp_final=0.0005,
            top_k=30, max_iter=2000,
            verbose=True, label="final"
        )
        sc_final = _exact_score(f_final)
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()
            print(f"    Final: {sc_final:.7f} (improved)")
        else:
            print(f"    Final: {sc_final:.7f}")

    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial score: {final_score:.7f}")

    if not np.isfinite(final_score):
        # Fallback to the given sequence
        best_f = f_given
        final_score = score_given

    total_time = time.monotonic() - t_start
    print(f"\n=== FINAL SCORE: {final_score:.7f} (time: {total_time:.1f}s) ===")

    return [float(x) for x in best_f]
