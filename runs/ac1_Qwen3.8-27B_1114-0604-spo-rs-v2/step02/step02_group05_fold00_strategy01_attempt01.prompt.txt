import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()

def _soft_weights(d, beta):
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-8:
        return np.ones(n, dtype=float) / n
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=float) / n
    return e / s


def _gradient_info(p, beta, L=None):
    """Return (score, gradient h of length n, max_h)."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    c = _conv_full(p, p)
    d = n * c
    exact_score = float(2.0 * d.max())
    w = _soft_weights(d, beta)
    # h[j] = 2 * sum_m w[j+m] * p[m] for j = 0..n-1
    # = 2 * cross_correlation(w, p)[j]
    # Using FFT: R = IFFT(rfft(w,L) * conj(rfft(p,L))) gives circular xcorr
    # For L >= (2n-1)+n-1 = 3n-2, first n elements = valid xcorr
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 2.0 * R[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    max_h = float(h.max())
    return exact_score, h, max_h


def _freq_flatten(p, alpha, L=None):
    """
    Frequency-domain convolution flattening (convolution square root).
    alpha > 1: more flattening of the convolution.
    Returns new normalized non-negative sequence or None.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.3:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

def _block_refine(p0, rng, time_budget):
    """Block-wise multiplicative refinement with exact scoring."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.2:
            bs = int(rng.integers(1, 5))
        elif r < 0.6:
            bs = int(rng.integers(max(2, n // 256), max(3, n // 32)))
        else:
            bs = int(rng.integers(max(8, n // 32), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 50:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 10):
        amp = rng.uniform(-0.6, 0.6) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_block_random(n, rng):
    K = int(rng.integers(4, 20))
    p = np.zeros(n, dtype=float)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1, K)
    for i in range(K):
        s = int(round(blocks[i] * n))
        e = max(s + 1, int(round(blocks[i + 1] * n)))
        e = min(n, e)
        p[s:e] = heights[i]
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.03, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s
