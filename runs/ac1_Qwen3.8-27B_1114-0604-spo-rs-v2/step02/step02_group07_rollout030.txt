    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None and res.x.sum() > 0:
            return res.x
    except Exception as e:
        print(f"LP failed: {e}")
    return None

def lp_phase(a_best, end_time, rng):
    """LP-based refinement phase."""
    n = a_best.size
    a = a_best.copy()
    best_score = np.inf
    best_a = a.copy()
    lr = 0.1
    iter_count = 0
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        M = float(np.max(c))
        score = compute_score(n, c, S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Solve LP
        g = solve_lp_direction(a, M, n)
        if g is None:
            # LP failed, try a smaller M
            M_render = M * 0.9
            g = solve_lp_direction(a, M_render, n)
            if g is None:
                break
        
        # Rescale g to match sum of a
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)
        
        # Update
        t = lr
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000)
        
        # Check if improvement
        c_new = fast_self_conv(a_new)
        score_new = compute_score(n, c_new, float(a_new.sum()))
        
        if score_new < score:
            a = a_new
            # Increase lr slightly
            lr = min(lr * 1.1, 0.5)
        else:
            # Decrease lr
            lr = max(lr * 0.5, 0.01)
            a = (1 - lr) * a + lr * g
            a = np.clip(a, 0, 1000)
        
        iter_count += 1
        if iter_count % 10 == 0:
            print(f"  LP iter {iter_count}: score={compute_score(n, fast_self_conv(a), float(a.sum())):.6f}, lr={lr:.4f}")
    
    return best_score, best_a

def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 10
    
    # Determine n
    n = 4096
    
    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0:
                parent = p
    except:
        pass
    
    best_score = np.inf
    best_a = None
    
    # Generate initializations
    inits = []
    
    if parent is not None and parent.size == n:
        inits.append(('parent', parent.copy()))
    elif parent is not None:
        # Resample parent to size n
        p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        inits.append(('parent_resampled', np.clip(p_resampled, 0, 1000)))
    
    # Ramp
    inits.append(('ramp', np.linspace(1, n, n)))
    # Reversed ramp
    inits.append(('rev_ramp', np.linspace(n, 1, n)))
    # Gaussian (center)
    x = np.arange(n)
    inits.append(('gauss_center', np.exp(-2.0 * ((x - n//2) / (n//4))**2)))
    # Bimodal
    inits.append(('bimodal', np.exp(-2.0 * ((x - n//4) / (n//8))**2) + np.exp(-2.0 * ((x - 3*n//4) / (n//8))**2)))
    # Cosine
    inits.append(('cosine', 1.0 + 0.5 * np.cos(2*np.pi*x/n)))
    # Random
    inits.append(('random', rng.random(n) + 0.1))
    # Edge-heavy
    inits.append(('edge', np.abs(x - n//2) / (n//2) + 0.1))
    
    # Phase 1: Explore with gradient descent (40% time)
    p1_end = start + 0.40 * (budget_s - 10)
    per_init_time = (p1_end - start) / len(inits)
    
    for name, a0 in inits:
        if time.monotonic() >= p1_end:
            break
        a0 = np.clip(a0, 0.01, 1000.0)
        x0 = np.log(a0)
        t_end = min(p1_end, time.monotonic() + per_init_time)
        
        sc, x_best = optimize_logspace(x0, t_end, lr=0.1, temp=1.0, rng=rng, patience=100)
        a_best = np.exp(np.clip(x_best, -20, 7))
        
        if sc < best_score:
            best_score = sc
            best_a = a_best.copy()
        print(f"Phase 1 [{name}]: {sc:.6f}")
    
    if best_a is None:
        best_a = np.ones(n)
        best_score = compute_score(n, fast_self_conv(best_a), float(best_a.sum()))
    
    print(f"After Phase 1: {best_score:.6f}")
    
    # Phase 2: LP refinement (30% time)
    p2_end = start + 0.70 * (budget_s - 10)
    if time.monotonic() < p2_end:
        sc, aa = lp_phase(best_a, p2_end, rng)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"After Phase 2: {best_score:.6f}")
    
    # Phase 3: Fine gradient polish (25% time)
    p3_end = start + 0.95 * (budget_s - 10)
    if time.monotonic() < p3_end:
        x0 = np.log(np.clip(best_a, 1e-10, 1000))
        sc, x_best = optimize_logspace(x0, p3_end, lr=0.01, temp=0.1, rng=rng, patience=500)
        a_best = np.exp(np.clip(x_best, -20, 7))
        if sc < best_score:
            best_score = sc
            best_a = a_best.copy()
        print(f"After Phase 3: {best_score:.6f}")
    
    # Phase 4: Final micro-polish
    if time.monotonic() < deadline:
        x0 = np.log(np.clip(best_a, 1e-10, 1000))
        sc, x_best = optimize_logspace(x0, deadline, lr=0.003, temp=0.01, rng=rng, patience=1000)
        a_best = np.exp(np.clip(x_best, -20, 7))
        if sc < best_score:
            best_score = sc
            best_a = a_best.copy()
        print(f"After Phase 4: {best_score:.6f}")
    
    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = evaluate_sequence(out)
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}")
    return out
