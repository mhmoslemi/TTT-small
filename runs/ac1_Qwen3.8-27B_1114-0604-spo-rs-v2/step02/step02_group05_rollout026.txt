"""
Multi-strategy optimization for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: exact top-K subgradient with adaptive step size, structured initializations
(tapered windows, spectral seeding), and multi-scale refinement.

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_convolve(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 500000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:nout].astype(float)

def _score_normalized(p):
    """Score assuming p is already normalized (sum=1)."""
    n = len(p)
    c = _fft_convolve(p, p)
    return float(2.0 * n * c.max())

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    return _score_normalized(p / s)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _subgradient_topK(a, K=10):
    """Compute weighted subgradient using top-K convolution peaks.
    Returns (grad, max_conv, peak_positions)."""
    n = len(a)
    c = _fft_convolve(a, a)
    max_c = c.max()
    # Get top-K positions
    # For efficiency, use argpartition for large arrays
    if len(c) <= 2 * K:
        idx = np.argsort(c)[::-1][:K]
    else:
        part = np.argpartition(c, -K)[-K:]
        idx = part[np.argsort(c[part])[::-1]]
    # Weights: proportional to peak values (normalized)
    w = c[idx] / c[idx].sum()
    # grad[i] = sum_m w_m * 2 * a[k_m - i]
    grad = np.zeros(n)
    for m in range(K):
        km = int(idx[m])
        # a[km - i] for i=0..n-1, valid when 0 <= km-i < n
        # i ranges from max(0, km-n+1) to min(km, n-1)
        i_lo = max(0, km - (n - 1))
        i_hi = min(km, n - 1)
        if i_lo <= i_hi:
            # a[km - i] for i in [i_lo, i_hi] = a[km - i_hi : km - i_lo + 1] reversed
            # km - i ranges from km - i_hi to km - i_lo
            # a[km-i_hi : km-i_lo+1]
            vals = a[km - i_hi : km - i_lo + 1][::-1]
            grad[i_lo:i_hi+1] += 2.0 * w[m] * vals
    return grad, float(max_c), idx

def _freq_flatten(a, alpha, L=None):
    """Frequency-domain flattening: raise periodogram to 1/alpha power and invert."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    P = A * A.conjugate()  # |A|^2, real and non-negative
    P = np.maximum(P, 0)
    Pmax = P.max()
    if Pmax < 1e-15:
        return a.copy()
    P_norm = P / Pmax
    P_new = Pmax * np.power(P_norm, 1.0 / alpha)
    # Invert: take sqrt of P_new in frequency domain
    # P_new = |A_new|^2, so A_new = sqrt(P_new) * exp(i*angle(A))
    # Preserve the phase of A
    angle = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * angle)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.5:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

    if s < 1e-10:

                cur_score = sc

The gradient of the score with respect to a[i] (for the active peak k*):
d(score)/d(a[i]) ∝ 2*a[k*-i]

But we could also use a "flattening" gradient:
grad_flat[i] = sum_k excess[k] * d(conv[k])/d(a[i]) = sum_k excess[k] * 2*a[k-i]

This is a weighted sum of shifts of a, where the weights are the "excess" values. If the excess is positive (peak above target), we push that peak down; if negative (dip below target), we don't care (since we're minimizing the max).

So: grad_flat[i] = 2 * sum_{k: excess[k] > 0} excess[k] * a[k-i]

This is a "soft" version of the max that considers all peaks above the target, not just the single maximum. It might give better convergence.

def _flat_gradient(a, threshold_frac=0.9):
    """Compute gradient based on flattening the convolution above a threshold.
    threshold_frac: only consider peaks above this fraction of the max."""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = c.max()
    threshold = c_max * threshold_frac
    excess = np.maximum(c - threshold, 0)
    # grad[i] = 2 * sum_k excess[k] * a[k-i]
    # This is a cross-correlation of excess with a
    # Use FFT: grad = 2 * ifft(fft(excess, L) * conj(fft(a, L)))[:n]
    L = _next_pow2(2 * n + n)  # Need L >= (2n-1) + n - 1 = 3n - 2
    E = np.fft.rfft(excess, L)
    A = np.fft.rfft(a, L)
    R = np.fft.irfft(E * np.conj(A), L)
    grad = 2.0 * R[:n]
    grad = np.maximum(grad, 0)
    return grad, float(c_max), excess

"""
Multi-strategy optimization for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses excess-based flattening gradient (novel), top-K exact subgradient, adaptive multiplicative
descent, frequency-domain spectral flattening, diverse structured initializations, and
block coordinate refinement. Exploits the convolution gradient structure d(conv_k)/d(a[i])=2*a[k-i].
"""

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_convolve(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 400000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:nout].astype(float)


def _score_norm(p):
    n = len(p)
    c = _fft_convolve(p, p)
    return float(2.0 * n * c.max())


def _score(p):
    p = np.maximum(np.asarray(p, dtype=float), 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    return _score_norm(p / s)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _flatten_grad(a, thresh_frac=0.85):
    """Excess-based flattening gradient: push down all conv peaks above threshold."""
    n = len(a)
    c = _fft_convolve(a, a)
    cmax = c.max()
    if cmax < 1e-15:
        return np.zeros(n), 0.0
    threshold = cmax * thresh_frac
    excess = np.maximum(c - threshold, 0.0)
    L = _next_pow2(3 * n)
    E = np.fft.rfft(excess, L)
    A = np.fft.rfft(a, L)
    R = np.fft.irfft(E * np.conj(A), L)
    grad = 2.0 * R[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad = np.maximum(grad, 0.0)
    return grad, float(cmax)


def _topk_grad(a, K=10):
    """Exact top-K subgradient for max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    cmax = c.max()
    if cmax < 1e-15:
        return np.zeros(n), 0.0
    if len(c) <= 2 * K:
        idx = np.argsort(c)[::-1][:K]
    else:
        part = np.argpartition(c, -K)[-K:]
        idx = part[np.argsort(c[part])[::-1]]
    w = c[idx] / max(c[idx].sum(), 1e-30)
    grad = np.zeros(n)
    for m in range(len(idx)):
        km = int(idx[m])
        i_lo = max(0, km - (n - 1))
        i_hi = min(km, n - 1)
        if i_lo <= i_hi:
            vals = a[km - i_hi:km - i_lo + 1][::-1]
            grad[i_lo:i_hi + 1] += 2.0 * w[m] * vals
    return grad, float(cmax)


def _freq_flatten(a, alpha):
    """Spectral flattening: raise periodogram to fractional power and invert."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    P = (A.real ** 2 + A.imag ** 2).astype(float)
    P = np.maximum(P, 0.0)
    Pmax = P.max()
    if Pmax < 1e-15:
        return a.copy()
    Pnorm = P / Pmax
    Pnew = Pmax * np.power(Pnorm, 1.0 / alpha)
    angle = np.arctan2(A.imag, A.real)
    Anew = np.sqrt(Pnew) * (np.cos(angle) + 1j * np.sin(angle))
    a_new = np.fft.irfft(Anew, L)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

    return np.ones(n, dtype=float) / n


def _init_tapered(n, kind='cosine'):

    elif kind == 'tri':

    else:
        p = np.ones(n)
    p = np.maximum(p, 0.02)
    return p / p.sum()

    p = np.ones(n)
    p[:k] = ratio
    p = p[rng.permutation(n)]
    t = np.linspace(0, 1, n)
    taper = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
    p = p * taper
    p = np.maximum(p, 0.01)
    return p / p.sum()


def _init_cosine_sum(n, rng, nterms=10):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n) * 4.0
    for _ in range(nterms):
        k = int(rng.integers(1, 20))
        amp = rng.uniform(-1.5, 1.5)
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0.01)
    return p / p.sum()


def _init_spectral(n, rng):
    L = _next_pow2(2 * n - 1)
    nfreq = L // 2 + 1
    P = np.ones(nfreq)
    freqs = np.linspace(0.0, np.pi, nfreq)
    P *= (1.0 + 0.8 * np.exp(-3.0 * freqs))
    phase = rng.uniform(0, 2 * np.pi, nfreq)
    phase[0] = 0
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.fft.irfft(A * L, L)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return a / s


def _init_resample(parent, n, rng):
    if parent is None:
        return _init_uniform(n)
    parent = _sanitize(parent)
    if parent is None:
        return _init_uniform(n)
    nold = len(parent)
    if nold == n:
        result = parent.copy()
    else:
        xold = (np.arange(nold) + 0.5) / nold
        xnew = (np.arange(n) + 0.5) / n
        result = np.interp(xnew, xold, parent)
    result = np.maximum(result, 0.0)
    result = result * np.exp(rng.normal(0, 0.04, n))
    result = np.maximum(result, 1e-10)
    s = result.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return result / s


def _grad_optimize(a0, rng, time_budget, K=10, thresh=0.85, verbose=False):
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _score_norm(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    eta = 0.25
    stall = 0
    it = 0
    mode_timer = 0

    if verbose:
        print(f"    GradOpt: n={n}, init={best_score:.6f}, T={time_budget:.0f}s")

    while time.monotonic() < deadline:
        mode_timer += 1
        # Choose gradient type
        if mode_timer % 8 == 0:
            use_flat = rng.random() < 0.6
        else:
            use_flat = (it % 3 != 0)

        if use_flat:
            grad, cmax = _flatten_grad(a, thresh)
        else:
            grad, cmax = _topk_grad(a, K)

        gmax = grad.max()
        if gmax < 1e-15:
            stall += 5
            continue
        grad_n = grad / gmax

        improved = False
        for et in (eta, eta * 0.6, eta * 0.4, eta * 0.25):
            a_try = a * np.exp(-et * grad_n)
            a_try = np.maximum(a_try, 1e-18 / n)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc = _score_norm(a_try)
            if sc < cur_score - 1e-14:
                a = a_try
                cur_score = sc
                eta = et
                improved = True
                break

        if improved:
            stall = 0
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose and it % 200 == 0:
                    print(f"      it={it}, score={best_score:.6f}, eta={eta:.3f}")
        else:
            stall += 1
            eta = max(0.01, eta * 0.75)
            if stall > 15:
                a = best.copy()
                cur_score = best_score
                eta = 0.25
                stall = 0
                # Try freq flatten
                al = float(rng.uniform(1.4, 3.5))
                an = _freq_flatten(a, al)
                if an is not None:
                    sc_n = _score_norm(an)
                    if sc_n < cur_score - 1e-12:
                        a = an
                        cur_score = sc_n
                        best = a.copy()
                        best_score = sc_n
                        if verbose:
                            print(f"      FREQ: {best_score:.6f}")
                elif rng.random() < 0.5:
                    # Perturb
                    a_try = a * np.exp(rng.normal(0, 0.03, n))
                    a_try = np.maximum(a_try, 1e-18 / n)
                    st = a_try.sum()
                    if st > 0:
                        a_try /= st
                        sc_n = _score_norm(a_try)
                        if sc_n < best_score * 1.05:
                            a = a_try
                            cur_score = sc_n
                            stall = 0

        # Periodic freq flatten
        if it % 60 == 0 and it > 0:
            al = float(rng.uniform(1.2, 2.5))
            an = _freq_flatten(a, al)
            if an is not None:
                sc_n = _score_norm(an)
                if sc_n < cur_score * 1.01:
                    a = an
                    cur_score = sc_n
                    if sc_n < best_score - 1e-12:
                        best = a.copy()
                        best_score = sc_n

        it += 1

    return best, best_score


def _block_optimize(a0, rng, time_budget, verbose=False):
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _score_norm(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.4, 0.6, 0.75, 0.88, 1.12, 1.25, 1.5, 2.0, 3.0)

    if verbose:
        print(f"    BlockOpt: n={n}, init={best_score:.6f}, T={time_budget:.0f}s")

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.1:
            bs = int(rng.integers(1, 4))
        elif r < 0.45:
            bs = int(rng.integers(4, max(8, n // 64)))
        elif r < 0.8:
            bs = int(rng.integers(max(8, n // 32), max(16, n // 10)))
        else:
            bs = int(rng.integers(max(16, n // 16), max(32, n // 5)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            a_try = a.copy()
            a_try[start:start + bs] *= f
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            sc = _score_norm(a_try)
            if sc < cur_score - 1e-14:
                a = a_try
                cur_score = sc
                improved = True
                break

        if not improved and bs >= 4 and rng.random() < 0.4:
            h = bs // 2
            for fu in (1.3, 1.7, 2.2):
                for fd in (0.5, 0.7, 0.85):
                    a_try = a.copy()
                    a_try[start:start + h] *= fu
                    a_try[start + h:start + bs] *= fd
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-10:
                        continue
                    a_try /= st
                    sc = _score_norm(a_try)
                    if sc < cur_score - 1e-14:
                        a = a_try
                        cur_score = sc
                        improved = True
                        break
                if improved:
                    break

        if improved:
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose:
                    print(f"      Block: {best_score:.6f}")
            stall = 0
        else:
            stall += 1
            if stall > 200:
                a = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score
