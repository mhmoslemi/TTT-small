Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam-based multiplicative gradient descent with top-K peak weighting,

"""

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _topk_weights(d, K, beta):
    """Weight function that combines top-K selection with softmax smoothing."""
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if K >= n:
        # Full softmax
        shift = d.max()
        e = np.exp(beta * (d - shift))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return np.ones(n, dtype=float) / n
        return e / s
    # Top-K: assign equal weight to top K, zero to rest
    # With smooth transition
    idx = np.argsort(d)[::-1][:K]
    w = np.zeros(n, dtype=float)
    threshold = d[idx[-1]]
    # Smooth: elements above threshold get weight proportional to exp(beta*(d-threshold))
    above = d >= threshold
    if above.sum() == 0:
        return np.ones(n, dtype=float) / n
    e = np.where(above, np.exp(beta * (d - threshold)), 0.0)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w[idx] = 1.0 / K
        return w
    return e / s

def _gradient_descent_adam(p, rng, time_budget, beta_start=50.0, beta_end=5000.0, 
                           K_frac=0.05, lr=0.1, beta_adam=0.9, eps=1e-8, verbose=False):
    """Adam-based multiplicative gradient descent with top-K peak targeting."""
    p = _sanitize(p)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(3 * n + 10)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    v = np.zeros(n, dtype=float)  # second moment
    t = 0
    
    deadline = time.monotonic() + time_budget
    stall = 0
    K = max(1, int(n * K_frac))
    
    if verbose:
        print(f"    Adam GD: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s, K={K}")
    
    while time.monotonic() < deadline:
        t += 1
        # Interpolate beta
        frac = min(1.0, t / max(1, int(time_budget / 0.05)))  # ramp over budget
        beta = beta_start * (beta_end / beta_start) ** frac
        K_cur = max(1, int(n * K_frac * (1 + frac)))  # increase K over time
        
        # Compute gradient
        c = _conv_full(p, p)
        d = n * c  # d[k] = n * (a*a)[k]
        exact_score = float(2.0 * d.max())
        
        w = _topk_weights(d, K_cur, beta)

        # = 2 * cross_correlation(w, p)[j]
        W = np.fft.rfft(w, L)
        P = np.fft.rfft(p, L)
        R = np.fft.irfft(W * np.conj(P), L)
        h = 2.0 * R[:n]
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        
        if h.max() < 1e-15:
            stall += 1
            continue
        
        h_norm = h / h.max()
        
        # Adam update
        m = beta_adam * m + (1 - beta_adam) * h_norm
        v = beta_adam * v + (1 - beta_adam) * h_norm ** 2
        m_hat = m / (1 - beta_adam ** t)
        v_hat = v / (1 - beta_adam ** t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        update = update / (np.max(np.abs(update)) + eps)  # normalize
        
        # Try multiple step sizes
        best_trial = None
        best_trial_sc = cur_score
        for a_try in (lr * 0.5, lr, lr * 1.5, lr * 2.0):

            if s_try < 1e-10:

            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-13:
                best_trial = p_try
                best_trial_sc = sc_try
        
        if best_trial is not None:

            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-12:

                best_score = cur_score

        else:

                cur_score = best_score
                lr = 0.1
                m = np.zeros(n, dtype=float)
                v = np.zeros(n, dtype=float)
                stall = 10
        
        # Periodic pullback

            if rng.random() < 0.2:
                p = best.copy()
                cur_score = best_score
                stall = 0
    
    return best, best_score

def _topk_weights(d, K, beta):

    if n == 0:
        return np.array([], dtype=float)

    if K >= n:
        shift = d.max()
        e = np.exp(beta * (d - shift))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return np.ones(n, dtype=float) / n
        return e / s
    idx = np.argsort(d)[::-1][:K]
    threshold = d[idx[-1]]
    above = d >= threshold
    if above.sum() == 0:
        return np.ones(n, dtype=float) / n
    e = np.where(above, np.exp(beta * (d - threshold)), 0.0)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(n, dtype=float)
        w[idx] = 1.0 / K
        return w
    return e / s

"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam-based multiplicative gradient descent with top-K peak weighting,
iterative spectral shaping (POCS-style), multi-length exploration, and
cosine-annealed beta scheduling.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b).astype(float)
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0.0)
    if rng is not None:
        result = result * np.exp(rng.normal(0.0, 0.03, n_new))
        result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _topk_weights(d, K, beta):
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if K >= n:
        shift = d.max()
        e = np.exp(beta * (d - shift))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return np.ones(n, dtype=float) / n
        return e / s
    idx = np.argsort(d)[::-1][:K]
    threshold = d[idx[-1]]
    above = d >= threshold
    if above.sum() == 0:
        return np.ones(n, dtype=float) / n
    e = np.where(above, np.exp(beta * (d - threshold)), 0.0)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(n, dtype=float)
        w[idx] = 1.0 / K
        return w
    return e / s

                             K_frac=0.04, lr=0.12, verbose=True)
            _try_update(p, sc)

    t_rem = t4_end - time.monotonic()
    if t_rem > 30 and best_overall is not None:
        t_alloc = min(0.5 * t_rem, 250.0)
        p, sc = _adam_gd(best_overall, rng, t_alloc,
                         beta_start=500.0, beta_end=20000.0,
                         K_frac=0.02, lr=0.06, verbose=True)
        _try_update(p, sc)
    print(f"  Phase 4 best: {best_score_overall:.6f}")

    # Phase 5: Final polishing
    print("=" * 60)
    print("Phase 5: Final polishing")
    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 180.0)
            p, sc = _adam_gd(best_overall, rng, t_alloc,
                             beta_start=1000.0, beta_end=40000.0,
                             K_frac=0.015, lr=0.05, verbose=True)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.35 * t_rem, 150.0)
            p, sc = _block_refine(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.35 * t_rem, 120.0)
            p, sc = _adam_gd(best_overall, rng, t_alloc,
                             beta_start=2000.0, beta_end=50000.0,
                             K_frac=0.01, lr=0.04, verbose=True)
            _try_update(p, sc)

    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    final_score = _score(out)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in out]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(out)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    return [float(x) for x in out]
