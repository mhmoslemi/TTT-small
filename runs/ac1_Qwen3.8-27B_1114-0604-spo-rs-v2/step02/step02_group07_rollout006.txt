"""

import time
from scipy.optimize import minimize as scipy_minimize


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def exact_score(n, a):
    """Compute exact objective value."""
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    M = float(np.max(c))
    return 2.0 * n * M / (S * S)


def smooth_obj_and_grad(a, tau):
    """
    Compute smooth objective and gradient using log-sum-exp max approximation.
    f_smooth = 2*n * m(c, tau) / S^2
    where m(c,tau) = tau * log(sum(exp(c/tau)))
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    # Soft weights
    c_shifted = (c - np.max(c)) / tau
    wts = np.exp(c_shifted)
    wts = wts / wts.sum()
    
    # Smooth max value
    smooth_max = np.max(c) + tau * np.log(np.sum(wts))
    
    # Gradient of smooth_max w.r.t. a_i
    grad_max = np.zeros(n)
    for k in range(len(c)):
        wk = wts[k]
        if wk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js > je:
            continue
        grad_max[js:je+1] += wk * a[js:je+1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    obj = 2.0 * n * smooth_max / S2
    grad = 2.0 * n * (grad_max / S2 - 2.0 * smooth_max / S3)
    
    return obj, grad


def subgradient_step(a, c, topk, lr):
    """
    Take a subgradient step using the top-k peaks.
    Returns updated a.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    # Get top-k peaks
    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part]
    # Weight by proximity to max
    max_v = vals.max()
    spread = max_v - vals.min()
    if spread < 1e-15:
        wts = np.ones(k_act) / k_act
    else:
        wts = np.exp(-(max_v - vals) / (0.1 * max_v))
        wts = wts / wts.sum()
    
    M = float(np.dot(vals, wts))
    
    # Gradient
    grad = np.zeros(n)
    for i in range(k_act):
        wi = wts[i]
        if wi < 1e-12:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
    
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new
