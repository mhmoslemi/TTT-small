```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def autocorr(f):
    """Compute conv(f, f) using FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = autocorr(f)
    return 2.0 * n * c.max() / (s * s)

def solve_best_response(f, M, n):
    """
    Solve: min M' s.t. conv(f, g)[k] <= M' for all k, g >= 0, sum(g) = 1.
    Returns the optimal g and M'.
    """
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n
    row_idx = []
    col_idx = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(f[k - j])
    
    # Add the -1 column for M'
    # A_full is (2n-1) x (n+1): last column is -1
    row_idx_full = row_idx + list(range(2*n - 1))
    col_idx_full = col_idx + [n] * (2*n - 1)
    vals_full = vals + [-1.0] * (2*n - 1)
    
    A = csr_matrix((vals_full, (row_idx_full, col_idx_full)), shape=(2*n-1, n+1))
    
    # Objective: minimize M' (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A @ x <= 0
    b_ub = np.zeros(2*n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    else:
        # Fallback: return uniform
        return np.ones(n) / n, 1.0 / n
```

```python
def solve_best_response_active_set(f, n, max_active=100, max_iters=50):
    """
    Solve best-response LP using active set method.
    """
    # Start with the constraint at k = n-1 (usually the most binding)
    # and a few neighbors
    active_k = list(range(max(0, n-10), min(2*n-1, n+10)))
    
    best_g = np.ones(n) / n
    best_M = autocorr(f).max()  # initial upper bound
    
    for outer in range(max_iters):
        # Build LP with active constraints
        # Variables: g[0], ..., g[n-1], M'
        # Constraints: conv(f, g)[k] <= M' for k in active_k
        #              sum(g) = 1
        #              g >= 0, M' >= 0
        
        m = len(active_k)
        
        # Build A matrix (m x n) for the conv constraints
        row_idx = []
        col_idx = []
        vals = []
        for i, k in enumerate(active_k):
            j_start = max(0, k - n + 1)
            j_end = min(n-1, k)
            js = np.arange(j_start, j_end+1)
            row_idx.extend([i]*len(js))
            col_idx.extend(js.tolist())
            vals.extend(f[js].tolist())
        
        # Full constraint matrix (m x (n+1)): [A_conv, -1]
        from scipy.sparse import csr_matrix, vstack, hstack, csr_array
        A_conv = csr_matrix((vals, (row_idx, col_idx)), shape=(m, n))
        neg_one = csr_matrix((-np.ones(m), (range(m), [n]*m)), shape=(m, 1))
        A_full = hstack([A_conv, neg_one], format='csr')
        
        c = np.zeros(n+1)
        c[n] = 1.0  # minimize M'
        b_ub = np.zeros(m)
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        
        if not res.success:
            break
        
        g = res.x[:n]
        M_prime = res.x[n]
        
        # Check all constraints
        c_full = np.convolve(f, g)  # full convolution
        violations = c_full - M_prime
        max_violation = violations.max()
        
        if max_violation <= 1e-10:
            return g, M_prime  # feasible!
        
        # Add worst constraints
        k_new = np.argsort(violations)[::-1][:20]
        for k in k_new:
            if k not in active_k:
                active_k.append(k)
        
        # Limit active set size
        if len(active_k) > max_active:
            # Keep only the most violated
            violations = c_full - M_prime
            active_k = list(np.argsort(violations)[::-1][:max_active])
        
        # Update best
        actual_M = c_full.max()
        if actual_M < best_M:
            best_M = actual_M
            best_g = g.copy()
    
    return best_g, best_M
```

Uses active-set LP to find best responses, with line search for step size.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix, diags, hstack

def _autocorr(f):
    """Compute full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _cross_corr(f, g):
    """Compute full linear cross-correlation conv(f, g) via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(f, size) * np.fft.rfft(g, size))[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr(f)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_constraint_matrix(f, n, active_k=None):
    """
    Build sparse constraint matrix for conv(f, b)[k] <= M.
    If active_k is None, use all k. Otherwise use only those k values.
    Returns (A, b_ub) where A is (m, n+1) and b_ub is (m,).
    """
    if active_k is None:
        active_k = range(2 * n - 1)
    active_k = list(active_k)
    m = len(active_k)
    
    # Build A_conv (m x n): A_conv[i, j] = f[k_i - j] if valid
    row_idx = []
    col_idx = []
    vals = []
    for i, k in enumerate(active_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        row_idx.extend([i] * len(js))
        col_idx.extend(js.tolist())
        vals.extend(f[js].tolist())
    
    A_conv = csr_matrix((vals, (row_idx, col_idx)), shape=(m, n))
    
    # Add -1 column for M'
    neg_col = csr_matrix((-np.ones(m), (list(range(m)), [n] * m)), shape=(m, 1))
    A_full = hstack([A_conv, neg_col], format='csr')
    
    b_ub = np.zeros(m)
    return A_full, b_ub


def _solve_best_response(f, n, max_lp_time=15):

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max_lp_time})
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s < 1e-15:

        g /= s

def _structured_perturb(f, rng, n):

    j = np.arange(n, dtype=np.float64)
    mode = int(rng.integers(0, 4))
    if mode == 0:
        alpha = rng.uniform(-0.5, 0.5)
        f = f * np.exp(alpha * (j / n - 0.5))
    elif mode == 1:
        freq = int(rng.integers(1, max(2, n // 5) + 1))
        phase = rng.uniform(0, 2 * np.pi)
        f = f * (1.0 + 0.2 * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        f = f + rng.uniform(-0.01,
