"""
Hybrid optimizer combining log-parameterized gradient descent with soft-max
gradient blending, periodic peak-shaving, and LP-inspired refinement for
minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
import time


def fast_self_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_objective(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    M = float(c.max())
    return 2.0 * n * M / (S * S)


def grad_softmax(a, c, topk=16, temp_scale=0.05):
    """
    Compute gradient of 2n*M/S^2 using soft-max over top-k peaks.
    Returns (gradient, soft_max_value, sum).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)

    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part]
    top_val = float(vals.max())
    temp = max(temp_scale * top_val, 1e-15)
    wts = np.exp((vals - top_val) / temp)
    wts = wts / wts.sum()
    M = float(np.dot(vals, wts))

    # Gradient of M w.r.t. a
    grad_M = np.zeros(n)
    for idx in range(k_act):
        wi = float(wts[idx])
        if wi < 1e-15:
            continue
        ki = int(part[idx])
        k_start = max(0, ki - n + 1)
        k_end = min(ki, n - 1)
        if k_start > k_end:
            continue
        j_start = ki - k_end
        j_end = ki - k_start
        grad_M[k_start:k_end + 1] += 2.0 * wi * a[j_start:j_end + 1][::-1]

    grad = (2.0 * n) * (grad_M / S2 - 2.0 * M * a / S3)
    return grad, M, S


def grad_exact(a, c, topk=1):
    """Exact subgradient using only the argmax peak."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    M = float(c[part].max())

    grad_M = np.zeros(n)
    for ki in part:
        ki = int(ki)
        k_start = max(0, ki - n + 1)
        k_end = min(ki, n - 1)
        if k_start > k_end:
            continue
        j_start = ki - k_end
        j_end = ki - k_start
        grad_M[k_start:k_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

    grad = (2.0 * n) * (grad_M / S2 - 2.0 * M * a / S3)
    return grad, M, S


def peak_shave(a, c, n_shave=5, strength=0.01):
    """Directly reduce elements contributing to the highest peaks."""
    n = len(a)
    peaks = np.argsort(c)[-n_shave:]
    for ki in peaks:
        ki = int(ki)
        k_start = max(0, ki - n + 1)
        k_end = min(ki, n - 1)
        if k_start > k_end:
            continue
        j_start = ki - k_end
        j_end = ki - k_start
        # Reduce a[j] for j in [j_start, j_end]
        a[j_start:j_end + 1] *= (1.0 - strength)
    return a


def adam_update(params, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer update in-place."""
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)


def optimize_phase(a0, end_time, rng, init_lr, topk, temp_scale, patience,
                   use_log=True, shave_every=200, shave_strength=0.005):
    """
    Run optimization phase until end_time.
    Returns (best_score, best_a, best_c).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    # Normalize to sum=1
    a = a / a.sum()

    c = fast_self_conv(a)
    score = 2.0 * n * float(c.max()) / 1.0  # S=1
    best_score = score
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    cur_pat = patience
    step_count = 0

    # Cosine schedule parameters
    total_estimated_steps = max(1, int((end_time - time.monotonic()) * 100))  # rough estimate
    final_lr = init_lr * 0.01

    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        score = 2.0 * n * float(c.max())

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Restart if stuck
        if no_imp >= cur_pat:
            # Perturb from best
            pert_scale = 0.01 + 0.05 * rng.random()
            a = best_a * (1.0 + pert_scale * rng.standard_normal(n))
            a = np.clip(a, 1e-10, None)
            a = a / a.sum()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(final_lr, init_lr * 0.3)
            no_imp = 0
            cur_pat = max(20, cur_pat * 2 // 3)
            step_count = 0
            continue

        # Compute gradient
        g, M, S = grad_softmax(a, c, topk, temp_scale)

        # Adam step
        t += 1
        # Cosine LR schedule
        progress = min(1.0, t / max(1, total_estimated_steps))
        lr_curr = final_lr + 0.5 * (lr - final_lr) * (1 + np.cos(np.pi * progress))
        adam_update(a, g, m, v, t, lr_curr)

        # Enforce positivity and normalize
        a = np.clip(a, 1e-10, None)
        a = a / a.sum()

        # Periodic peak shaving
        step_count += 1
        if shave_every > 0 and step_count % shave_every == 0:
            a = peak_shave(a, c, n_shave=3, strength=shave_strength)
            a = np.clip(a, 1e-10, None)
            a = a / a.sum()

    return best_score, best_a, c

            a = np.clip(a, 0.0, None)

        a = np.clip(a, 0.0, None)

def try_lp_refine(a, M, n_iterations=50, end_time=None):
    """
    LP-inspired refinement: find direction g that maximizes sum(g)
    subject to (a*g)[k] <= M. Use a simplified inner-approximation.
    """
    n = len(a)
    a = a / a.sum()
    M = fast_self_conv(a).max()

    # Use a random subset of constraints to make LP tractable
    n_constraints = min(2000, 2 * n - 1)
    all_indices = np.arange(2 * n - 1)
    # Always include the top peaks
    c = fast_self_conv(a)
    top_peaks = np.argsort(c)[-50:]
    remaining = np.setdiff1d(all_indices, top_peaks)
    sampled = np.concatenate([top_peaks, rng_sample(remaining, n_constraints - len(top_peaks))])

    # Build constraint matrix for sampled constraints
    n_vars = n
    A_ub = np.zeros((len(sampled), n_vars))
    for i, k in enumerate(sampled):
        k = int(k)
        k_start = max(0, k - n + 1)
        k_end = min(k, n - 1)
        if k_start > k_end:
            continue
        j_start = k - k_end
        j_end = k - k_start
        A_ub[i, j_start:j_end+1] = a[j_start:j_end+1][::-1]

    try:
        from scipy.optimize import linprog
        c_obj = -np.ones(n_vars)
        bounds = [(0, None)] * n_vars
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.full(len(sampled), M),
                         bounds=bounds, method='highs', options={'time_limit': 5})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                return g
    except Exception:
        pass
    return None


def rng_sample(arr, k):
    """Random sample without replacement."""
    rng = np.random.default_rng()
    if k >= len(arr):
        return arr
    return rng.choice(arr, size=k, replace=False)


def init_strategies(n, rng):
    """Generate diverse initial guesses."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n))

    # Linear ramp
    inits.append(1.0 + 0.0 * x)  # uniform
    inits.append(0.5 + x)  # ramp up
    inits.append(1.5 - x)  # ramp down

    # Quadratic
    inits.append(0.5 + x * x)
    inits.append(0.5 + (x - 0.5) ** 2)

    # Cosine variations
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(1.0 + 0.5 * np.cos(4 * np.pi * x))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * x))

    # Exponential
    inits.append(np.exp(0.5 * x))
    inits.append(np.exp(-0.5 * x))
    inits.append(np.exp(1.0 * (x - 0.5)))

    # Gapped
    a_g = np.ones(n)
    a_g[1::2] = 0.5
    inits.append(a_g)
    a_g2 = np.ones(n)
    a_g2[1::3] = 0.3
    inits.append(a_g2)

    # Bimodal
    a_b = np.zeros(n)
    a_b[n//4] = 1.0
    a_b[3*n//4] = 1.0
    # Smooth it
    from scipy.ndimage import gaussian_filter1d
    a_b = gaussian_filter1d(a_b, sigma=n//20)
    a_b = np.clip(a_b, 0.01, None)
    inits.append(a_b)

    # Random smooth
    a_r = np.clip(0.5 + 0.5 * rng.random(n), 0.01, None)
    a_r = gaussian_filter1d(a_r, sigma=n//10)
    inits.append(a_r)

    # Triangular
    inits.append(1.0 - 2.0 * np.abs(x - 0.5))

    # Power law
    inits.append((x + 0.1) ** 0.5)
    inits.append((x + 0.1) ** 2.0)

    return inits

        sc = optimize_phase(best_a, deadline - 3.0, rng, init_lr=0.003, topk=64,
                            temp_scale=0.005, patience=800, shave_every=500,
                            shave_strength=0.001)
        if sc[0] < best_score:
            best_score, best_a = sc[0], sc[1]
        print(f"Phase 5: {best_score:.6f}")

    # Final: verify and prepare output
    if best_a is None:
        best_a = np.ones(best_n)

    # Ensure non-negative and within bounds
    best_a = np.clip(best_a, 0.0, 1000.0)
    # Scale so sum > 0.01 (to pass the evaluate_sequence check)
    if best_a.sum() < 0.01:
        best_a = best_a * 10.0

    out = [float(x) for x in best_a]

    # Verify
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception as e:
        print(f"Verification error: {e}")

    elapsed = time.monotonic() - start
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, "
          f"n={len(out)}, time={elapsed:.1f}s")

    return out
