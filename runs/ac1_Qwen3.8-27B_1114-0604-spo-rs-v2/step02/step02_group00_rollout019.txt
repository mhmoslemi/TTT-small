```python
"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with LSE-smoothed objective, active-set local search,
and spectral insights for the final push.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        val, grad = lse_score_grad(x, beta)
        return val, grad
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def active_set_search(a, total_budget_s=300, rng=None):
    """
    Active-set local search:
    - Identify lags within epsilon of the max
    - Compute combined subgradient
    - Take adaptive steps with line search
    - Also try redistribution moves
    """

        rng = np.random.default_rng(42)
    
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    # Initial step size
    step = 0.01
    S = float(a.sum())
    
    for it in range(50000):
        if time.monotonic() - t_start > total_budget_s:
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find active lags (within 0.1% of max)
        threshold = c_max * 0.999
        active_lags = np.where(c >= threshold)[0]
        
        # Compute combined subgradient
        sg = np.zeros(n)
        for k in active_lags:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)

        effective_step = step * a_scale / sg_norm

def spectral_refine(a, total_budget_s=100):
    """
    Try to flatten the convolution by working in the spectral domain.
    The idea: if conv(a,a) = ifft(|FFT(a)|^2), we can try to
    modify |FFT(a)|^2 to be "flatter" in a way that reduces the peak,
    then find a non-negative time-domain signal matching the modified spectrum.
    
    Simplified approach: iterate between:
    1. Modify spectrum to reduce peak (projection in frequency)
    2. Modify time domain to be non-negative (projection in time)
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    for it in range(2000):
        if time.monotonic() - t_start > total_budget_s:
            break
        
        # Current spectrum
        A = np.fft.rfft(a, M_fft)
        P = A * A  # |A|^2 (since a is real, this is the power spectrum)
        
        # Inverse to get convolution
        c = np.fft.irfft(P, M_fft)[:M]
        
        # Soft-threshold the convolution to reduce peak
        c_max = float(c.max())
        c_target = np.minimum(c, c_max * 0.95)  # try to cap at 95% of peak
        
        # Get modified spectrum
        P_new = np.fft.rfft(c_target, M_fft)
        # We need |A_new|^2 = P_new, so |A_new| = sqrt(P_new)
        # Keep the phase of A
        P_new = np.abs(P_new)  # ensure non-negative
        A_new = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0) * np.sqrt(P_new)
        
        # Inverse FFT to get time domain
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Project to non-negative
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Normalize to same sum
        S_old = float(a.sum())
        S_new = float(a_new.sum())
        if S_new > 1e-10:
            a_new = a_new * S_old / S_new
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_sc, best_a
```

```python
def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    """Top-level objective for scipy: returns (value, gradient)."""
    return lse_score_grad(x, beta)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S3 = S2 * S
```

Let me also think about what's special about the active-set search. At the optimum:
- The max of conv is achieved at 1 or a few lags
- The subgradient at those lags gives the direction of steepest increase
- We want to move in the negative subgradient direction
- But we need to be careful: moving too far might create a new, higher peak elsewhere

The key challenge is the "sawtooth" nature of the landscape: reducing one peak might create a higher one elsewhere. The LSE smoothing helps with this, but at high beta, it becomes essentially the same as the non-smooth problem.

A better strategy for the local search:
1. Compute the full convolution c
2. Find the top K lags (e.g., K = 5 or 10)
3. We want a direction d such that c[k] + 2*(d*a)[k] decreases for all top K lags
4. This is equivalent to finding d that minimizes max_k (c[k] + 2*(d*a)[k]) over the top K
5. By LP duality, this is equivalent to finding weights w_k >= 0, sum w_k = 1, such that d = -sum_k w_k * v_k where v_k is the subgradient at lag k, and this minimizes the max

Actually, the problem "find d to minimize max_k (c[k] + 2*(d*a)[k])" is a linear program in variables (d_0, ..., d_{n-1}, t):
- Minimize t
- Subject to: c[k_i] + 2*sum_j d[j]*a[k_i - j] <= t for i = 1,...,K
- And: ||d||_2 <= 1 (or some normalization)
- And: a + t_step * d >= 0

The constraint ||d||_2 <= 1 makes it non-linear. If we replace with ||d||_1 <= 1, it becomes an LP. But the number of variables is n + K + 1, and for n = 8192, this is a large LP.

Alternatively, we can use a simpler heuristic:
- d = -average of subgradients at active lags (weighted by how much each contributes to the max)
- This is what I already have

Let me just go with a practical implementation and focus on getting the details right.

```python
"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Phase 1: L-BFGS with LSE continuation (beta 200->50000) from parent/structured inits.
Phase 2: Active-set local search with redistribution and periodic L-BFGS restarts.
Phase 3: Spectral-domain projection + final high-beta polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT, length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (top-level for scipy)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -60.0, 60.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: 2n * (g/S^2 - 2L/S^3) where g[j] = 2*sum_m a[m]*p[m+j]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed score. Returns (exact_score, array, result)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def get_subgrad_at_lag(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    j_s = max(0, k - n + 1)
    j_e = min(k, n - 1)
    if j_e >= j_s:
        sg[j_s:j_e + 1] = 2.0 * a[j_s:j_e + 1][::-1]
    return sg


def active_set_search(a, budget_s, rng):
    """
    Active-set local search:
    - Identifies lags within 0.05% of max
    - Computes combined subgradient
    - Adaptive step with backtracking line search
    - Periodic redistribution moves
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    step = 0.005
    no_improve = 0

    for it in range(200000):
        if time.monotonic() - t0 > budget_s:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        # Active lags within 0.05% of max
        threshold = c_max * (1.0 - 5e-4)
        active = np.where(c >= threshold)[0]

        # Combined subgradient
        sg = np.zeros(n)
        for k in active:
            sg += get_subgrad_at_lag(a, k)

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Relative step size
        a_scale = float(np.mean(a[a > 0])) if np.any(a > 0) else 1.0
        base_step = step * a_scale / sg_norm

        # Backtracking line search
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_try = a - base_step * shrink * sg
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc_try = compute_score(a_try)
            if sc_try < sc * (1.0 - 1e-12):
                a = a_try
                step = min(step * 1.1, 0.03)
                improved = True
                break

        if not improved:
            step *= 0.8
            if step < 1e-12:
                break

        # Redistribution move every 200 iterations
        if (it + 1) % 200 == 0 and len(active) >= 1:
            k_star = int(np.argmax(c))
            k_low = int(np.argmin(c))
            if c[k_low] < c_max * 0.4:
                v_peak = get_subgrad_at_lag(a, k_star)
                v_low = get_subgrad_at_lag(a, k_low)
                for w in [0.3, 0.2, 0.1]:
                    d = -v_peak + w * v_low
                    dn = np.linalg.norm(d)
                    if dn < 1e-10:
                        continue
                    for scale in [0.005, 0.002, 0.001]:
                        a_try = np.clip(a + scale * (a_scale / dn) * d, 0.0, 1000.0)
                        sc_try = compute_score(a_try)
                        if sc_try < best_sc:
                            a = a_try
                            best_sc = sc_try
                            best_a = a.copy()
                            no_improve = 0
                            break
                        break

    return best_sc, best_a


def spectral_refine(a, budget_s):
    """
    Spectral projection: flatten conv(a,a) by modifying the power spectrum,
    then project back to non-negative time domain.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    M_fft = 1 << (2 * n - 2).bit_length()

    for it in range(10000):
        if time.monotonic() - t0 > budget_s:
            break

        A = np.fft.rfft(a, M_fft)
        c = fast_conv(a)
        c_max = float(c.max())

        # Soft-threshold the convolution
        cap = c_max * 0.96
        c_soft = np.where(c > cap, cap + (c - cap) * 0.2, c.copy())

        # Get spectrum of softened convolution
        P_target = np.abs(np.fft.rfft(c_soft, M_fft))**2

        # Match amplitude, preserve phase
        amp_current = np.abs(A)
        amp_target = np.sqrt(P_target[:len(A)])
        # Damped amplitude update
        amp_new = 0.7 * amp_current + 0.3 * amp_target
        A_new = amp_new * np.exp(1j * np.angle(A))

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Preserve sum
        S_old = float(a.sum())
        S_new = float(a_new.sum())
        if S_new > 1e-10:
            a_new *= S_old / S_new

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        # Damped time-domain update
        a = 0.6 * a + 0.4 * a_new

    return best_sc, best_a
```
