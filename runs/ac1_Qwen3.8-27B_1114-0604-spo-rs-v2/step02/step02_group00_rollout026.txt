```python
def lp_direction(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, conv(a,b)_k <= M for all k.
    Returns the optimal b.
    """
    n = len(a)
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    
    # Build the convolution matrix A (2n-1) x n
    # Row k, col j: A[k,j] = a[j] if 0 <= k-j < n, else 0
    # i.e., A[k,j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: minimize -sum(b) s.t. A @ b <= M (vector of M's), b >= 0
    c_vec = -np.ones(n)
    res = linprog(c_vec, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

def build_conv_mat(a):
    n = len(a)
    # For each column j, it appears in rows j, j+1, ..., j+n-1 with value a[j]
    # So column j has non-zeros at rows j to j+n-1
    col_indices = []
    row_indices = []
    values = []
    for j in range(n):
        r = np.arange(j, j+n)
        col_indices.extend(np.full(n, j))
        row_indices.extend(r)
        values.extend(np.full(n, a[j]))
    A = csr_matrix((values, (row_indices, col_indices)), shape=(2*n-1, n))
    return A

"""
LP-guided refinement for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Algorithm:
1. LP-based direction finding: at each step, solve an LP to find the best
   direction b that can be mixed with current a without increasing the peak.
2. Continuation: gradually tighten the peak constraint.
3. L-BFGS polish with LSE-smoothed objective.
4. Local search with diverse perturbations.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(n):
    """Build the (2n-1) x n convolution matrix structure (row/col indices)."""
    # Column j has non-zeros at rows j, j+1, ..., j+n-1
    # Total non-zeros: sum_{j=0}^{n-1} n = n^2
    # But we only need rows 0 to 2n-2, so column j has rows j to min(j+n-1, 2n-2)
    # For j <= n-1: rows j to j+n-1 (all valid since j+n-1 <= 2n-2)
    # So each column has exactly n non-zeros.
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    return rows, cols


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Handle overflow
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)_k <= M for all k.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    rows, cols = build_conv_matrix(n)
    vals = np.repeat(a, n)  # a[j] is the value for all n entries in column j
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_vec = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None


def lp_refine(a, n_iters=50, t_start=0.5, t_min=0.05):
    """
    Iterate LP-based refinement.
    Returns (best_score, best_a, final_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return np.inf, a, a
    
    best_sc = compute_score(a)
    best_a = a.copy()
    t = t_start
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        b = lp_direction(a, M)
        if b is None:
            t *= 0.5
            continue
        
        # Scale b to have same sum as a
        b_sum = b.sum()
        if b_sum < 1e-12:
            t *= 0.5
            continue
        b = b * (S / b_sum)
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            # Try to increase t
            t = min(t * 1.1, 0.9)
            a = a_new
        else:
            # Peak increased, reduce t
            t *= 0.5
            if t < t_min:
                break
    
    return best_sc, best_a, a

    max sum(b) s.t. b >= 0, conv(a,b)_{k_i} <= M for top K lags.
    """
    n = len(a)
    c = fast_conv(a)
    # Select top K lags
    top_k = np.argsort(c)[::-1][:K]
    
    # Build K x n constraint matrix
    rows = np.zeros(K * n, dtype=np.int32)  # Overestimate
    cols = np.zeros(K * n, dtype=np.int32)
    vals = np.zeros(K * n, dtype=np.float64)

"""
LP-guided refinement + L-BFGS polish for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
1. LP-based direction finding with active-set constraint selection
2. Momentum-based convex combination updates
3. Adaptive step-size (line search) for LP updates
4. Multi-phase: LP refinement -> L-BFGS continuation -> local search
5. Better time management and fallback strategies
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


# ---------------------------------------------------------------------------
# Core numerical helpers
# ---------------------------------------------------------------------------

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -200, 200)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


# ---------------------------------------------------------------------------
# LP-based direction finding
# ---------------------------------------------------------------------------

def lp_direction_active(a_normalized, M_p, K=300):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)_{k_i} <= M_p for top K lags.
    a_normalized has sum = 1. M_p = max(conv(a,a)).
    Returns optimal b (unnormalized) or None.
    """
    n = len(a_normalized)
    c = fast_conv(a_normalized)
    # Select K most active lags
    if K >= 2 * n - 1:
        top_lags = np.arange(2 * n - 1)
    else:
        top_lags = np.argsort(c)[::-1][:K]

    # Build K x n constraint matrix
    row_idx = []
    col_idx = []
    vals = []
    for ii, k in enumerate(top_lags):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(k_int, n - 1)
        if j_end >= j_start:
            row_idx.extend([ii] * (j_end - j_start + 1))
            col_idx.extend(range(j_start, j_end + 1))
            vals.extend(a_normalized[j_start:j_end + 1])

    if len(vals) == 0:
        return None

    A = csr_matrix((vals, (row_idx, col_idx)), shape=(len(top_lags), n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(top_lags), M_p),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if res.success:
        return res.x
    return None


def lp_refine_phase(a_init, t_start, deadline, max_lps=200):
    """
    LP-based refinement phase.
    a_init: initial sequence (will be normalized internally).
    Returns (best_score, best_a).
    """
    n = len(a_init)
    # Normalize to sum = 1
    S0 = float(a_init.sum())
    if S0 < 1e-12:
        p = np.ones(n) / n
    else:
        p = a_init / S0

    best_sc = compute_score(a_init)
    best_p = p.copy()
    t = 0.4
    b_momentum = None
    alpha_mom = 0.3  # momentum coefficient for direction averaging
    K = 200  # number of active constraints

    for it in range(max_lps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(p)
        M_p = float(c.max())

        # Solve LP
        b = lp_direction_active(p, M_p, K=K)
        if b is None:
            # Fallback: use subgradient direction
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            b = np.zeros(n)
            b[j_start:j_end + 1] = p[j_start:j_end + 1][::-1]
            b = b / (b.sum() + 1e-30)

        # Normalize b to sum = 1
        b_sum = float(b.sum())
        if b_sum < 1e-12:
            break
        b = b / b_sum

        # Momentum: average with previous direction
        if b_momentum is not None:
            b = alpha_mom * b + (1 - alpha_mom) * b_momentum
            b = b / (b.sum() + 1e-30)
        b_momentum = b.copy()

        # Line search: try to find good t
        best_t = t
        best_sc_trial = best_sc
        for t_try in [t * 0.5, t, t * 1.5, t * 2.0]:
            t_try = min(max(t_try, 0.01), 0.9)
            p_trial = (1 - t_try) * p + t_try * b
            p_trial = np.clip(p_trial, 0.0, None)
            sc_trial = compute_score_from_normalized(p_trial)
            if sc_trial < best_sc_trial:
                best_sc_trial = sc_trial
                best_t = t_try

        p_new = (1 - best_t) * p + best_t * b
        p_new = np.clip(p_new, 0.0, None)
        sc_new = compute_score_from_normalized(p_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new
            t = min(best_t * 1.1, 0.8)
        else:
            t *= 0.7
            p = (1 - t * 0.5) * p + t * 0.5 * b
            p = np.clip(p, 0.0, None)
            sc_check = compute_score_from_normalized(p)
            if sc_check < best_sc:
                best_sc = sc_check
                best_p = p.copy()

        # Gradually increase K for later iterations
        if it % 20 == 0 and K < 500:
            K = min(K + 50, 500)

        if it % 10 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {it}: score={sc_new:.8f}, best={best_sc:.8f}, t={t:.3f}, K={K}, elapsed={elapsed:.0f}s")

    return best_sc, best_p * S0


def compute_score_from_normalized(p):
    """Score from a normalized (sum=1) sequence."""
    n = len(p)
    c = fast_conv(p)
    return 2.0 * n * float(np.max(c))

def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(5, n // 2)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def sinusoidal_perturb(a, rng):
    """Apply sinusoidal perturbation."""
    n = len(a)
    freq = 1.0 + 8.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    amp = 0.02 + 0.08 * rng.random()
    x = np.arange(n, dtype=np.float64) / n
    a_pert = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    return np.clip(a_pert, 0.0, 1000.0)

                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"final polish beta={beta}")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
