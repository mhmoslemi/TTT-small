Key differences from the previous L-BFGS approach:

  - Adam optimizer for per-coordinate adaptive learning rates
  - Adaptive beta (temperature) annealing from smooth to sharp
  - Very fast per-iteration cost (O(n log n)) enabling 100K+ iterations
  - Cross-correlation gradient computed via FFT
  - Multiple length exploration with structured initializations
  - Final L-BFGS-B polish on the best log-domain solution

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_s_vector(a, p, M2):
    """
    Compute s_j = sum_m a[m] * p[m+j] for j=0,...,n-1.
    Using: s = conv(a[::-1], p)[n-1 : 2n-1]
    """
    n = len(a)
    q = a[::-1]
    Q = np.fft.rfft(q, M2)
    P = np.fft.rfft(p, M2)
    conv = np.fft.irfft(Q * P, M2)
    return conv[n - 1: 2 * n - 1]

        n_iters: total number of iterations

    # Adam state
    m = np.zeros(n)

    beta2 = 0.999
    eps = 1e-8

    best_a = a0.copy()
    
    # Precompute schedule: beta at each iteration
    n_phases = len(beta_schedule)
    iters_per_phase = n_iters // n_phases

        # Determine current beta
        phase = min(it // max(1, iters_per_phase), n_phases - 1)
        beta = beta_schedule[phase]
        
        # Current a
        a = np.exp(x)
        S = float(a.sum())
        if S < 1e-10:
            x += np.log(1e6)
            continue

        exp_scaled = np.exp(scaled)
        log_sum = np.log(exp_scaled.sum())
        F = c_max + log_sum / beta
        p = exp_scaled / exp_scaled.sum()
        
        # Compute score
        sc = 2.0 * n * F / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute s vector: s_j = sum_k p_k * a[k-j]
        s = compute_s_vector(a, p, M2)
        
        # Gradient in log-domain:
        # g_j = 4n * a_j * (s_j * S - F) / S^2  (simplified, absorbed 1/S)
        # Actually: d(score)/dx_j = 2n * (dF/dx_j * S - 2F * a_j) / S^2
        # where dF/dx_j = 2 * a_j * s_j
        # So: g_j = 2n * (2 * a_j * s_j * S - 2F * a_j) / S^2
        #      = 4n * a_j * (s_j * S - F) / S^2
        S2 = S * S
        grad = 4.0 * n * a * (s * S - F) / S2
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        
        # Adaptive learning rate: decay over time
        lr_t = lr * 0.999 ** (it // 100)
        x -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Center x periodically to prevent drift
        if it % 1000 == 0:
            x = x - np.mean(x)
        # Clip for numerical safety
        x = np.clip(x, -30, 30)
    
    a_final = np.exp(x)
    sc_final = compute_score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_sc, best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    M1 = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M1)
    c = np.fft.irfft(A * A, M1)[:2 * n - 1]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    M2 = 1 << (3 * n - 3).bit_length()
    q = a[::-1]
    Q = np.fft.rfft(q, M2)
    P = np.fft.rfft(p, M2)
    s = np.fft.irfft(Q * P, M2)[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * s / S - 2.0 * L / S3)
    return score, grad


def run_lbfgs_polish(a0, beta=500, maxiter=5000):
    """L-BFGS-B polish on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    bounds = [(1e-10, 1000.0)] * n
    
    def obj_and_grad(a):
        sc, g = lse_score_grad(a, beta)
        return sc, g
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': 100000,
                            'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    # Parent if available
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy() + 1e-4)
        else:
            # Resample parent to length n
            from scipy.ndimage import zoom
            z = zoom(parent, n / len(parent), order=3)
            inits.append(np.maximum(z, 1e-4).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine modulations
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.cos(4.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    # Quadratic bell
    inits.append(0.1 + 0.8 * np.exp(-4.0 * (x - 0.5) ** 2))
    # Two peaks
    inits.append(0.1 + 0.7 * (np.exp(-30*(x-0.3)**2) + np.exp(-30*(x-0.7)**2)))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.3 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    # Triangle
    inits.append(np.abs(1.0 - 2.0 * x) + 0.1)
    # Flat with noise
    inits.append(np.clip(0.5 + 0.1 * rng.standard_normal(n), 0.01, 10.0))
    
    return [np.maximum(a, 0.01).copy() for a in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses log-domain Adam with adaptive LSE temperature as the primary method.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget_s = max(30.0, float(budget_s))
    deadline = t_start + budget_s - 10.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(2048, dtype=np.float64) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    print(f"Budget: {budget_s}s, Parent available: {parent is not None}")
    if parent is not None:
        print(f"  Parent length: {len(parent)}, parent score: {compute_score(parent):.8f}")
        report(compute_score(parent), parent, "parent")
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        pn = len(parent)
        # Round to reasonable size
        if pn >= 1024:
            lengths.append(pn)
        else:
            lengths.append(2048)
    for n in [4096, 8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    
    # Time allocation per length (more time for longer sequences)
    n_lengths = len(lengths)
    # Phase 1: Log-domain Adam (70% of time)
    # Phase 2: L-BFGS polish (20% of time)
    # Phase 3: Final refinement (10% of time)
    phase1_end = t_start + 0.70 * (budget_s - 10)
    phase2_end = t_start + 0.90 * (budget_s - 10)
    
    # ==================================================================
    # Phase 1: Log-domain Adam optimization
    # ==================================================================
    print(f"\n{'='*60}")
    print("Phase 1: Log-domain Adam")
    print(f"{'='*60}")
    
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n--- n = {n} ---")
        t_phase = time.monotonic()
        
        # FFT sizes
        M1 = 1 << (2 * n - 2).bit_length()
        M2 = 1 << (3 * n - 3).bit_length()
        
        inits = build_initializations(n, rng, parent)
        n_inits = len(inits)
        
        # Time per init
        time_per_init = max(30.0, (phase1_end - time.monotonic()) / max(1, n_inits) * 0.8)
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            
            # Budget for this init
            init_deadline = min(time.monotonic() + time_per_init, phase1_end)
            elapsed_budget = init_deadline - time.monotonic()
            
            if elapsed_budget < 5:
                break
            
            # Estimate iterations: each iteration ~ (2 FFTs of size ~n) 
            # For n=8192: ~0.5ms per iter. For n=16384: ~1ms per iter.
            est_iter_time = max(0.001, 0.0005 * (n / 8192.0) ** 1.5)
            n_iters = int(min(500000, elapsed_budget / est_iter_time * 0.7))
            n_iters = max(5000, n_iters)
            
            # Beta schedule: start smooth, end sharp
            # More phases for more iterations
            n_phases = min(8, max(3, n_iters // 10000))
            beta_schedule = np.geomspace(5, 500, n_phases)
            
            # Learning rate: scale with problem size
            lr = 0.05 * (2048.0 / n) ** 0.5
            lr = min(lr, 0.1)
            
            t0 = time.monotonic()
            sc, a_opt = log_domain_adam(a0, beta_schedule, n_iters, lr, M2, M1)
            elapsed = time.monotonic() - t0
            report(sc, a_opt, f"n={n}, init={i}, adam({n_iters}it)")
            
            # Quick L-BFGS polish if still have time
            if time.monotonic() < phase1_end - 10:
                sc2, a_opt2 = run_lbfgs_polish(a_opt, beta=300, maxiter=2000)
                report(sc2, a_opt2, f"n={n}, init={i}, adam+lbfgs")
    
    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")
    
    # ==================================================================
    # Phase 2: Focused L-BFGS polish on best candidate
    # ==================================================================
    print(f"\n{'='*60}")
    print("Phase 2: L-BFGS Polish")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    M1_best = 1 << (2 * n_best - 2).bit_length()
    M2_best = 1 << (3 * n_best - 3).bit_length()
    
    # Increase beta for sharper optimization
    for beta in [500, 1000, 2000]:
        if time.monotonic() > phase2_end:
            break
        sc, a_opt = run_lbfgs_polish(best_a, beta=beta, maxiter=5000)
        report(sc, a_opt, f"polish beta={beta}")
    
    # Additional log-domain Adam with high beta
    if time.monotonic() < phase2_end:
        a_cur = best_a.copy()
        beta_high = np.geomspace(200, 2000, 5)
        est_iter_time = max(0.001, 0.0005 * (n_best / 8192.0) ** 1.5)
        n_iters = int(min(200000, (phase2_end - time.monotonic()) / est_iter_time * 0.8))
        n_iters = max(5000, n_iters)
        lr = 0.02 * (2048.0 / n_best) ** 0.5
        sc, a_opt = log_domain_adam(a_cur, beta_high, n_iters, lr, M2_best, M1_best)
        report(sc, a_opt, f"high-beta adam({n_iters}it)")
    
    # ==================================================================
    # Phase 3: Local search + final polish
    # ==================================================================
    print(f"\n{'='*60}")
    print("Phase 3: Local Search & Final Polish")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    
    # Try a few perturbation strategies
    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > deadline - 30:
            break
        
        strategy = trial % 5
        if strategy == 0:
            # Small Gaussian perturbation
            a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            contrib = a_pert[j_start:j_end+1][::-1] * a_pert[j_start:j_end+1]
            top_k = max(5, n_best // 100)
            local_top = np.argsort(contrib)[::-1][:top_k]
            for j in local_top:
                a_pert[j_start + j] *= 0.9
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        elif strategy == 2:
            # Frequency-domain perturbation
            A = np.fft.rfft(best_a, M1_best)
            amp_pert = 1.0 + 0.03 * rng.standard_normal(len(A))
            A = A * np.abs(amp_pert) / (np.abs(amp_pert) + 1e-10)
            a_pert = np.fft.irfft(A, M1_best)[:n_best]
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        elif strategy == 3:
            # Log-space perturbation
            x = np.log(best_a)
            x_pert = x + 0.1 * rng.standard_normal(n_best)
            a_pert = np.exp(x_pert)
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        else:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x_norm = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + 0.04 * np.cos(2.0 * np.pi * freq * x_norm + phase))
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        
        # Quick polish
        sc, a_opt = run_lbfgs_polish(a_pert, beta=500, maxiter=3000)
        report(sc, a_opt, f"local#{trial} (strat={strategy})")
    
    # Final: high-beta log-domain Adam on the absolute best
    if time.monotonic() < deadline - 20:
        n_final = len(best_a)
        M1_f = 1 << (2 * n_final - 2).bit_length()
        M2_f = 1 << (3 * n_final - 3).bit_length()
        beta_final = np.geomspace(500, 5000, 4)
        est_t = max(0.001, 0.0005 * (n_final / 8192.0) ** 1.5)
        n_iters_f = int(min(100000, (deadline - 20 - time.monotonic()) / est_t * 0.9))
        n_iters_f = max(5000, n_iters_f)
        lr_f = 0.01 * (2048.0 / n_final) ** 0.5
        sc, a_opt = log_domain_adam(best_a, beta_final, n_iters_f, lr_f, M2_f, M1_f)
        report(sc, a_opt, f"final adam({n_iters_f}it)")
        
        # One more L-BFGS
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs_polish(a_opt, beta=3000, maxiter=3000)
            report(sc, a_opt, "final lbfgs")
    
    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL RESULT")
    print(f"  Internal score: {best_score:.8f}")
    print(f"  Verified score: {verified:.8f}")
    print(f"  Sequence length: {len(out)}")
    print(f"  Time used: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
