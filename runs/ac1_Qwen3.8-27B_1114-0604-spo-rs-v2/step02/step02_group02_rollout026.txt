def _autocorr_fft(f):
    """Linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

    return 2.0 * n * peak

```python
"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2 using mirror descent
with adaptive step size on the probability simplex, diverse initialization,
and coordinate-descent polish. Exploits convexity of peak convolution.
"""

import numpy as np
import time


def _convolve_fft(f):
    """Linear convolution of f with itself via zero-padded FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _grad_conv(f, k, n):
    """Gradient of (f*f)[k] with respect to f (length n)."""
    grad = np.zeros(n)
    # Term 1: f[k-i] for i in [max(0,k-n+1), min(k, n-1)]
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if hi1 > lo1:
        grad[lo1:hi1] += f[k - np.arange(lo1, hi1)]
    # Term 2: f[i-k] for i in [k, min(n-1, k+n-1)]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if hi2 > lo2:
        grad[lo2:hi2] += f[np.arange(lo2, hi2) - k]
    return grad


def _subgrad(f, conv, k):
    """Subgradient: average gradient over top-k lags."""
    n = len(f)
    if k >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        g += _grad_conv(f, int(ki), n)
    return g / len(idx)


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _generate_starts(n, rng, base):
    """Generate diverse starting points (all normalized to sum 1, non-negative)."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = (n - 1) / 2.0

    def _add(w):
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 1e-10:
            starts.append(w / s)

    # Base sequence (from previous best)
    if base is not None:
        b = np.maximum(np.asarray(base, dtype=np.float64), 0.0)
        if b.sum() > 1e-10:
            starts.append(b / b.sum())

    # Uniform
    _add(np.ones(n))

    # Hann window
    _add(0.5 * (1.0 + np.cos(2 * np.pi * j / n)))

    # Polynomial tapers (raised to various powers)
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0]:
        x = np.abs(j - center) / half
        _add((1.0 - x) ** p + 1e-10)

    # Gaussian windows
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0]:
        _add(np.exp(-0.5 * ((j - center) / (sigma * half)) ** 2))

    # Raised cosine powers (Dolph-Chebyshev-like)
    for p in [0.5, 1.0, 2.0, 4.0, 8.0]:
        x = np.abs(j - center) / half
        w = 0.5 * (1.0 + np.cos(np.pi * x))
        _add(w ** p + 1e-10)

    # Bimodal
    for pos in [0.15, 0.2, 0.25, 0.3, 0.4]:
        w = (np.exp(-((j - pos * (n - 1)) / (0.04 * n)) ** 2) +
             np.exp(-((j - (1 - pos) * (n - 1)) / (0.04 * n)) ** 2) + 0.001)
        _add(w)

    # U-shaped
    for a in [0.5, 1.0, 2.0, 5.0, 10.0]:
        x = (j - center) / half
        _add(1.0 + a * x ** 2)

    # Sine windows
    for freq in [1, 2, 3, 5]:
        _add(np.abs(np.sin(np.pi * j / n)) ** 0.5 + 0.01)
        _add(np.sin(np.pi * j / n) ** 2 + 0.01)

    # Smooth random
    for _ in range(20):
        cutoff = rng.choice([0.005, 0.01, 0.02, 0.05, 0.1])
        freqs = np.fft.rfftfreq(n)
        spec = rng.standard_normal(len(freqs)) * np.exp(-freqs / cutoff)
        spec[0] = 0  # Remove DC component for variety
        w = np.fft.irfft(spec * n, n)
        _add(np.abs(w) + 0.01)

    return starts


def _optimize(f0, deadline, rng, n):
    """Mirror descent optimization with adaptive step size."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = _convolve_fft(f)
    peak = conv.max()
    best_peak = peak
    best_f = f.copy()

    eta = 1.0
    iter_count = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Every 5000 iterations, print progress
        if iter_count % 5000 == 0 and iter_count > 0:
            print(f"    iter {iter_count}: peak={peak:.12f}, best={best_peak:.12f}, eta={eta:.4f}")

        # Adaptive top-k: start small, grow when stuck
        if no_improve < 1000:
            k = 2
        elif no_improve < 5000:
            k = 4
        elif no_improve < 15000:
            k = 8
        else:
            k = 16

        grad = _subgrad(f, conv, k)

        # Mirror descent step
        f_new = _mirror_step(f, grad, eta)
        conv_new = _convolve_fft(f_new)
        peak_new = conv_new.max()

        if peak_new < peak - 1e-15:
            f = f_new
            conv = conv_new
            peak = peak_new
            if peak < best_peak:
                best_peak = peak
                best_f = f.copy()
                no_improve = 0
            eta *= 1.02  # Slightly increase step
            eta = min(eta, 10.0)
        else:
            no_improve += 1
            eta *= 0.7  # Decrease step
            eta = max(eta, 0.01)

            # If stuck a lot, try perturbation
            if no_improve % 2000 == 0 and no_improve > 0:
                mag = min(0.05, 0.005 * np.sqrt(no_improve / 2000.0))
                f = f + mag / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                conv = _convolve_fft(f)
                peak = conv.max()
                eta = max(eta, 0.5)

        iter_count += 1

    # Final check
    conv = _convolve_fft(f)
    if conv.max() < best_peak:
        best_peak = conv.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak


def _polish(f0, deadline, rng, n):
    """Coordinate descent polish."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    f /= f.sum()

    conv = _convolve_fft(f)
    best_peak = conv.max()
    best_f = f.copy()
    
    # Precompute: for coordinate i, the "neighbors" that interact with it
    # (at the peak lag, these are f[k*-i] and f[i-k*])
    # We'll just do sweeps with trial values.
    
    trial_fracs = [0.0, 0.25, 0.5, 0.75, 1.0, 1.33, 1.5, 2.0, 3.0]
    sweep_count = 0
    
    while time.monotonic() < deadline:
        if sweep_count % 5 == 0:
            print(f"    polish sweep {sweep_count}: peak={best_peak:.12f}")
        
        improved = False
        # Forward sweep
        for i in range(n):
            x = f[i]
            if x < 1e-15:
                continue
            best_trial = x
            best_trial_peak = best_peak  # Use current best as threshold
            
            for frac in trial_fracs:
                y = x * frac
                f_try = f.copy()
                f_try[i] = y
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try = f_try / s
                # Quick check: if the change is small, skip expensive computation
                # Full computation
                conv_try = _convolve_fft(f_try)
                p_try = conv_try.max()
                if p_try < best_trial_peak - 1e-15:
                    best_trial_peak = p_try
                    best_trial = y
            
            if best_trial != x:
                f[i] = best_trial
                f /= f.sum()
                conv = _convolve_fft(f)
                best_peak = conv.max()
                best_f = f.copy()
                improved = True
        
        sweep_count += 1
        
        # If no improvement in a full sweep, try random perturbations
        if not improved:
            # Try a few random coordinate changes
            for _ in range(50):
                i = int(rng.integers(0, n))
                frac = rng.choice([0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 5.0])
                f_try = f.copy()
                f_try[i] *= frac
                if f_try[i] < 0:
                    continue
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                conv_try = _convolve_fft(f_try)
                if conv_try.max() < best_peak - 1e-15:
                    f = f_try
                    best_peak = conv_try.max()
                    best_f = f.copy()
                    break
            
            # Try moving mass from one coordinate to another
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
