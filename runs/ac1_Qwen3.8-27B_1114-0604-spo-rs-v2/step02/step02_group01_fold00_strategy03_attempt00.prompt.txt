import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation without calling evaluate_sequence."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _gradient_at_lags(f, lags, weights):
    """
    Compute weighted gradient of sum_w w_t * conv(f,f)[k_t] w.r.t. f.
    grad[j] = 2 * sum_t w_t * f[k_t - j] (where valid)
    """
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        # f[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # indices into f: k - j for j in [j_start, j_end]
            # = k - j_start down to k - j_end
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    return grad


def _mirror_step(f, grad, step, momentum_vec=None, beta=0.9):
    """Exponentiated gradient (mirror descent) step with momentum."""
    if momentum_vec is not None:
        grad_eff = grad + beta * momentum_vec
        new_momentum = grad + beta * momentum_vec
    else:
        grad_eff = grad
        new_momentum = np.zeros_like(grad)

    # Scale by range for numerical stability
    span = grad_eff.max() - grad_eff.min()
    if span < 1e-30:
        return f, new_momentum
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad_eff
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f, new_momentum
    f_new /= s
    return f_new, new_momentum


def _project_nonneg_sum1(f):
    """Project onto {x >= 0, sum x = 1}."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _spectral_regularizer(f, n_harmonics=5):
    """Penalty for non-flat power spectrum (encourages low autocorrelation)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    power = np.abs(F) ** 2
    # Focus on low-mid frequencies where convolution peaks tend to come from
    mid = size // 2
    power_mid = power[1:min(n, mid)]  # exclude DC
    if len(power_mid) == 0:
        return 0.0
    mean_p = power_mid.mean()
    if mean_p < 1e-30:
        return 0.0
    var_p = np.var(power_mid) / (mean_p * mean_p)
    return var_p


def _structured_perturb(f, rng, mode=0):

    elif mode == 3:
        # Bimodal injection

        bw = rng.uniform(0.05, 0.2)
        boost = np.exp(-((j / n - c1) / bw) ** 2) + np.exp(-((j / n - c2) / bw) ** 2)
        f_try = f * (1.0 + 0.3 * boost)
    elif mode == 4:
        # Random local swap
        i1 = int(rng.integers(0, n // 2))
        i2 = int(rng.integers(n // 2, n))
        amt = f[i1] * rng.uniform(0.05, 0.3)

    elif mode == 5:

def _optimize_focused(f0, n, deadline, rng, max_iters=100000, verbose=False, label=""):

    Focused gradient descent with:
    - Top-K peak targeting
    - Momentum
    - Spectral regularizer
    - Adaptive step size
    - Structured perturbations on stagnation
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.85
    K = 4  # number of top peaks to target
    reg_weight = 0.01  # spectral regularizer weight

    step_base = 0.8
    step_min = 0.002
    n_total = max_iters

    stuck_count = 0
    patience = 8
    improvement_thresh = 1e-7
    last_eval_score = local_best

    for it in range(n_total):
        if time.monotonic() > deadline:
            break

        progress = it / n_total

        # Adaptive step size: large early, small late
        step = step_base * (step_min / step_base) ** (progress ** 0.6)

        # Compute convolution
        c = _autocorr_fft(f)
        peak = c.max()

        # Find top K lags
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]  # sort descending
        top_vals = c[top_idx]

        # Weights: emphasize the top peaks more strongly
        # Use inverse-gap weighting
        gaps = top_vals[0] - top_vals
        gaps = np.maximum(gaps, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Compute gradient at top lags
        grad = _gradient_at_lags(f, top_idx, weights)

        # Add spectral regularizer gradient (approximate)
        if reg_weight > 0 and it % 10 == 0:
            # Gradient of spectral variance - approximate by modulating f
            # Simple approach: penalize positions where f is correlated with its own shifts
            # Use a cheap proxy: penalty on f * reversed_f (the middle convolution)
            n_mid = n - 1
            if 0 <= n_mid < 2 * n - 1:
                mid_grad = np.zeros(n)
                for j in range(n):
                    idx = n_mid - j
                    if 0 <= idx < n:
                        mid_grad[j] = f[idx]
                grad += reg_weight * mid_grad

        # Mirror step with momentum
        f, momentum = _mirror_step(f, grad, step, momentum, beta)
        f = _project_nonneg_sum1(f)

        # Periodic evaluation
        eval_interval = max(100, 5000 // max(1, stuck_count))
        if it % eval_interval == 0 or it == n_total - 1:
            score = _score_fast(f)
            if score < local_best - improvement_thresh:
                local_best = score
                local_best_f = f.copy()
                stuck_count = 0
                last_eval_score = score
                if verbose and it % 5000 == 0:
                    print(f"    {label} iter {it}: {score:.7f}")
            else:
                stuck_count += 1

            # If stuck, apply structured perturbation
            if stuck_count >= patience:
                mode = int(rng.integers(0, 6))
                f = _structured_perturb(f, rng, mode)
                momentum *= 0.5  # damp momentum after perturbation
                stuck_count = 0

                # Also try: reduce K temporarily for exploration
                # and increase step
                step *= 2.0

        # Periodically reduce momentum for stability
        if it % 10000 == 0 and it > 0:
            momentum *= 0.5

    # Final evaluation
    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best


def _polish_coordinate(f, n, deadline, rng):
    """
    Coordinate descent polish: iteratively find the peak lag and
    transfer mass from high-contribution positions to low ones.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()

    c = _autocorr_fft(f)
    peak = c.max()

    n_passes = 500
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Find top 3 peak lags
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]

        improved = False
        for k_star in top3:
            # Contribution of each position to c[k_star]
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            if j_start <= j_end:
                idx = k_star - np.arange(j_start, j_end + 1)
                contrib[j_start:j_end + 1] = f[np.arange(j_start, j_end + 1)] * f[idx]

            # Find best donor (high contrib) and receiver (low total contribution)
            total_contrib = np.zeros(n)
            for k in top3:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    idx_k = k - np.arange(js, je + 1)
                    total_contrib[js:je + 1] += f[np.arange(js, je + 1)] * f[idx_k]

            # Donor: position with highest total contribution
            j_don = int(np.argmax(total_contrib))
            # Receiver: position with lowest total contribution
            j_rec = int(np.argmin(total_contrib))

            if j_don == j_rec or total_contrib[j_don] < 1e-15:
                continue

            # Try several transfer fractions
            best_local = None
            best_amt = 0
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                amt = f[j_don] * frac
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                if f_try[j_don] < -1e-15:
                    continue
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    if best_local is None or c_try.max() < best_local:
                        best_local = c_try.max()
                        best_amt = amt

            if best_local is not None:
                f[j_don] -= best_amt
                f[j_rec] += best_amt
                f = np.maximum(f, 0)
                f /= f.sum()
                c = _autocorr_fft(f)
                peak = c.max()
                improved = True
                local_best = _score_fast(f)
                local_best_f = f.copy()

        if not improved and p > 50:
            # Try random coordinate moves
            for _ in range(5):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                f_try = f.copy()
                amt = f_try[i] * 0.1
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    local_best = _score_fast(f)
                    local_best_f = f.copy()
                    break

    return local_best_f, local_best


def _finetune(f, n, deadline, rng):
    """
    Final fine-tuning: very small mirror steps focused on the single peak lag.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    K = 2
    step = 0.01

    n_iters = 50000
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        peak = c.max()

        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        grad = _gradient_at_lags(f, top_idx, weights)

        # Decrease step over time
        step = 0.01 * (0.001 / 0.01) ** (it / n_iters)

        f, momentum = _mirror_step(f, grad, step, momentum, beta=0.9)
        f = _project_nonneg_sum1(f)

        if it % 2000 == 0:
            score = _score_fast(f)
            if score < local_best:
                local_best = score
                local_best_f = f.copy()

    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best
