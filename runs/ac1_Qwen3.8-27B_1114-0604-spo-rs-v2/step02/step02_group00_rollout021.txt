def lp_iteration(a, M=None):
    """
    Given current a, solve LP to find g that maximizes sum(g)
    subject to (a*g)[k] <= M for all k, g >= 0.
    """
    from scipy.optimize import linprog
    n = len(a)
    if M is None:
        M = float(np.max(fast_conv(a)))
    
    # Constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # for k = 0, ..., 2n-2, j = 0, ..., n-1
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    # Build sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) subject to A @ g <= M, g >= 0
    # linprog minimizes, so minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return a.copy(), M
    
    g = res.x
    return g, M
```

def build_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid."""
    from scipy.sparse import csr_matrix
    # For each k, the valid j range is [max(0,k-n+1), min(k,n-1)]
    # A[k,j] = a[k-j]
    # Let's think of it differently:
    # For each pair (j, i) where i = k-j, we have A[j+i, j] = a[i]
    # So A[k, j] = a[k-j] means: for each j and each valid i, A[j+i, j] = a[i]
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # i ranges from 0 to min(n-1, 2n-2-j)
        i_max = min(n-1, 2*n-2-j)
        i_range = range(0, i_max + 1)
        k_range = range(j, j + i_max + 1)
        rows.extend(k_range)
        cols.extend([j] * (i_max + 1))
        vals.extend(a[:i_max+1])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

"""
Hybrid optimizer combining LP-based iteration (from state-of-the-art) with LSE-continuation
L-BFGS and exact subgradient polishing. Minimizes 2n * max(conv(a,a)) / (sum(a))^2
over non-negative sequences. Key improvements: LP iteration exploiting the linear
subproblem structure, adaptive LSE temperature, and multi-scale length exploration.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_constraint_matrix(n, a):
    rows_parts = []
    cols_parts = []
    vals_parts = []
    for j in range(n):
        i_max = min(n - 1, 2 * n - 2 - j)
        cnt = i_max + 1
        rows_parts.append(np.arange(j, j + cnt))
        cols_parts.append(np.full(cnt, j))
        vals_parts.append(a[:cnt])
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_iteration(a, M=None):
    n = len(a)
    c = fast_conv(a)
    if M is None:
        M = float(c.max())
    A = build_conv_constraint_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': 60})
    if res.status == 0:
        return np.asarray(res.x, dtype=np.float64), M
    return a.copy(), M


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M_conv = n + (2 * n - 1) - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    RA = np.fft.rfft(rev_a, M_fft)
    P = np.fft.rfft(p, M_fft)
    conv_full = np.fft.irfft(RA * P, M_fft)[:M_conv]
    h = conv_full[n - 1: 2 * n - 1]
    grad_L = 2.0 * h
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def subgrad_polish(a, total_iters=2000, lr_init=0.01):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    sc_lb = 1.0
    mu = 0.9
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * a[k_star - idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        lr = lr_init / (1.0 + 0.05 * it)
