"""
Hybrid projected subgradient + L-BFGS optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses simplex projection, multi-lag subgradients, adaptive step sizes,
and periodic random restarts to escape local minima.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def score_normalized(x):
    """Score assuming sum(x)=1: 2n*max(conv(x,x))."""
    n = len(x)
    c = fast_conv(x)
    return 2.0 * n * float(c.max())


def simplex_project(v):
    """Project v onto the probability simplex {y >= 0, sum(y) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    rho_star = int(np.max(rho[cond])) if np.any(cond) else 1
    tau = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - tau, 0.0)


def subgradient_max_conv(x, top_k=1):
    """
    Compute (approximate) subgradient of max_k (x*x)[k] at x.
    Uses average of gradients over top_k peak lags.
    Returns gradient vector of length n.
    """
    n = len(x)
    c = fast_conv(x)
    # Get top_k indices
    if top_k >= c.size:
        idxs = np.arange(c.size)
    else:
        idxs = np.argpartition(c, -top_k)[-top_k:]
    # For each lag k, the gradient of (x*x)[k] w.r.t. x is:
    # g[j] = 2*x[k-j] if 0 <= k-j <= n-1, else 0
    # This is 2 * (x reversed, shifted)
    rev_x = x[::-1]
    g = np.zeros(n)
    for k in idxs:
        # g[j] = 2*x[k-j] for valid j
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # j ranges from j_start to j_end, k-j ranges from k-j_start to k-j_end
            m_start = k - j_end
            m_end = k - j_start
            g[j_start:j_end + 1] += 2.0 * x[m_start:m_end + 1]
    return g / top_k


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for scipy optimize."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score_val = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    # Extract the right portion: conv[a_rev, p] has length (2n-1)+(n-1) = 3n-2
    # We need the part corresponding to correlation
    # conv_res[i] = sum_j rev_a[j] * p[i-j]
    # We want sum_m a[m] * p[m+j] for each j
    # = sum_m a[m] * p[j+m]
    # rev_a[j] = a[n-1-j]
    # conv_res[i] = sum_j a[n-1-j] * p[i-j]

    # We want: for each j (0..n-1), sum_m a[m]*p[j+m]
    # = conv_res[j + n - 1] ... let's verify

    g_conv = conv_res[n - 1:2 * n - 1]
    g = 2.0 * g_conv
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

        return val, grad

def projected_subgradient(x, n_iters=5000, top_k=3, step_init=0.01, rng=None):

    Returns the best x found.

    x = x.copy()
    x = x / x.sum()
    best_x = x.copy()
    best_sc = score_normalized(x)
    step = step_init
    no_improve = 0
    patience = 200

    for it in range(n_iters):
        c = fast_conv(x)
        # Multi-lag subgradient
        g = subgradient_max_conv(x, top_k=top_k)
        # Normalize gradient

        if gn < 1e-15:
            break
        g = g / gn
        # Tentative step
        x_new = x - step * g
        x_new = simplex_project(x_new)
        sc_new = score_normalized(x_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_x = x_new.copy()
            no_improve = 0
            # Occasionally increase step

                step = min(step * 1.1, 0.1)
        else:
            no_improve += 1
            if no_improve >= patience:
                step *= 0.7
                no_improve = 0
        x = x_new

    return best_x, best_sc


def run_projected_subgradient_phase(x0, n_iters, top_k_list, step_list, rng):

    n_reduce = max(1, n // 20)
    top_idx = np.argsort(contrib)[::-1][:n_reduce]
    bottom_idx = np.argsort(contrib)[:n_reduce]
    x_pert = x.copy()
    total_red = 0.0
    for j in top_idx:
        amt = x_pert[j] * strength
        x_pert[j] -= amt
        x_pert[j] = max(x_pert[j], 0.0)
        total_red += amt
    if total_red > 0 and len(bottom_idx) > 0:
        per = total_red / len(bottom_idx)
        for j in bottom_idx:
            x_pert[j] += per
    x_pert = simplex_project(x_pert)
    return x_pert


def try_multiple_lengths(x_parent, lengths, rng, t_start, deadline):
    """Try to improve the solution at various lengths."""
    best_x = x_parent.copy()
    best_sc = score_normalized(x_parent)
    best_n = len(x_parent)

    for n_target in lengths:
        if time.monotonic() > deadline:
            break
        # Resize: if parent is shorter, pad with small values; if longer, truncate
        if n_target == len(best_x):
            x_init = best_x.copy()
        elif n_target > len(best_x):
            # Pad with interpolated values
            old_n = len(best_x)
            x_interp = np.interp(np.linspace(0, old_n - 1, n_target),
                                  np.arange(old_n), best_x)
            x_init = simplex_project(x_interp)
        else:
            # Downsample
            old_n = len(best_x)
            x_interp = np.interp(np.arange(n_target),
                                  np.linspace(0, old_n - 1, n_target), best_x)
            x_init = simplex_project(x_interp)
        # Optimize at this length
        x_opt, sc = run_projected_subgradient_phase(
            x_init, n_iters=3000, top_k_list=[3, 5],
            step_list=[0.005, 0.01], rng=rng)
        # Convert to actual score
        actual_sc = 2.0 * n_target * sc / 2.0 / n_target  # sc = 2n*maxconv, score = 2n*maxconv (since sum=1)
        actual_sc = sc  # score_normalized already includes the 2n factor
        if actual_sc < best_sc:
            best_sc = actual_sc
            best_x = x_opt.copy()
            best_n = n_target
            print(f"  n={n_target}: improved to {actual_sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    return best_x, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses projected subgradient on the simplex, L-BFGS refinement,
    and random restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        print(f"Parent loaded: n={n}")
        x = parent / parent.sum()  # normalize to simplex
        best_sc = score_normalized(x)
        print(f"Parent normalized score: {best_sc:.8f}")
    else:
        n = 4096
        x = np.ones(n) / n
        best_sc = score_normalized(x)
        print(f"Using uniform init: n={n}, score={best_sc:.8f}")

    x_best = x.copy()
    best_score = best_sc

    def report(sc, x_opt, label):
        nonlocal best_score, x_best
        if sc < best_score:
            best_score = sc
            x_best = x_opt.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(x_opt)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ================================================================
    # Phase 1: Projected subgradient optimization (main workhorse)
    # ================================================================
    print("\n=== Phase 1: Projected Subgradient ===")
    phase1_deadline = t_start + 0.45 * (float(budget_s) - 15.0)

    # Multiple rounds with different top_k and step sizes
    current_x = x_best.copy()
    for round_num in range(8):
        if time.monotonic() > phase1_deadline:
            break
        top_k_list = [2, 3, 5, 8]
        step_list = [0.003, 0.008, 0.015]
        x_opt, sc = run_projected_subgradient_phase(
            current_x, n_iters=4000, top_k_list=top_k_list[:2],
            step_list=step_list, rng=rng)
        report(sc, x_opt, f"PS round={round_num}")
        current_x = x_best.copy()

    # ================================================================
    # Phase 2: L-BFGS refinement at high beta
    # ================================================================
    print("\n=== Phase 2: L-BFGS Refinement ===")
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 15.0)

    # Scale x to a reasonable range for L-BFGS (avoid very small values)
    x_scaled = x_best * 1.0  # already sum=1
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs(x_best, beta, maxiter=8000, maxfun=150000)
        report(sc, a_opt, f"LBFGS beta={beta}")
        # Convert back to simplex for next iteration
        if a_opt.sum() > 1e-10:
            x_best = a_opt / a_opt.sum()

    # ================================================================
    # Phase 3: Random restarts with targeted perturbations
    # ================================================================
    print("\n=== Phase 3: Random Restarts ===")
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 15.0)

    max_restarts = 30
    restart_count = 0
    while restart_count < max_restarts and time.monotonic() < phase3_deadline:
        # Choose perturbation strategy
        strat = restart_count % 4
        if strat == 0:
            # Gaussian perturbation
            strength = 0.02 + 0.06 * rng.random()
            x_pert = random_perturb_simplex(x_best, rng, strength)
        elif strat == 1:
            # Targeted perturbation
            strength = 0.03 + 0.07 * rng.random()
            x_pert = targeted_perturb_simplex(x_best, rng, strength,
                                                top_k=3 + int(4 * rng.random()))
        elif strat == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            t_arr = np.arange(len(x_best)) / len(x_best)
            pert = 1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr + phase)
            x_pert = simplex_project(x_best * pert)
        else:
            # Block perturbation
            blk = max(2, len(x_best) // 100)
            n_blk = len(x_best) // blk
            block_noises = rng.standard_normal(n_blk) * 0.03
            pert = np.repeat(block_noises, blk)[:len(x_best)]
            x_pert = simplex_project(x_best * (1.0 + pert))

        # Optimize the perturbation
        x_opt, sc = run_projected_subgradient_phase(
            x_pert, n_iters=2000, top_k_list=[3],
            step_list=[0.008], rng=rng)
        report(sc, x_opt, f"restart #{restart_count} (strat={strat})")
        restart_count += 1

    # ================================================================
    # Phase 4: Final polish
    # ================================================================
    print("\n=== Phase 4: Final Polish ===")

    # More aggressive subgradient with very small steps
    x_polished, sc = projected_subgradient(x_best, n_iters=10000,
                                            top_k=2, step_init=0.002, rng=rng)
    report(sc, x_polished, "final subgrad")

    # One more L-BFGS pass
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(x_best, 10000, maxiter=10000, maxfun=200000)
        report(sc, a_opt, "final LBFGS")

    # Try one more subgradient round
    if time.monotonic() < deadline - 10:
        x_final, sc = projected_subgradient(x_best, n_iters=5000,
                                             top_k=1, step_init=0.001, rng=rng)
        report(sc, x_final, "final subgrad 2")

    # ================================================================
    # Prepare output
    # ================================================================
    # Scale x_best to a reasonable range (avoid 0 and 1000 issues)
    # The score is scale-invariant, so any scale works
    out_arr = x_best * 1.0  # keep as is (sum=1)
    # Make sure all values are in [0, 1000]
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(v) for v in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
