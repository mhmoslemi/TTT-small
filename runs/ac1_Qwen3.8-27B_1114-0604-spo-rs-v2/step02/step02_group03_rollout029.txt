    m = 2 * n - 1
    size = 1
    while size < m:

def _score(c, n):
    """Compute score from convolution and length."""
    return float(2.0 * n * c.max())


def _transfer_delta(f, i, j, dv, L):
    """
    Compute the change in convolution for transferring dv from i to j.
    f has length n, L = 2n-1.
    Returns a new array of the delta.
    """
    n = f.size
    delta = np.zeros(L, dtype=np.float64)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    q = dv * dv
    if 2 * j < L:
        delta[2 * j] += q
    if 2 * i < L:
        delta[2 * i] += q
    if i + j < L:
        delta[i + j] -= 2.0 * q
    return delta


def _max_with_delta(c, delta):
    """Compute max of c + delta without allocating a new array."""
    return float(np.max(c + delta))


def _multi_hump_seed(n, nh, positions, widths, heights, rng):
    """Generate a multi-hump seed sequence."""
    f = np.zeros(n, dtype=np.float64)
    for h, (pos, wid, amp) in enumerate(zip(positions, widths, heights)):
        # Gaussian hump centered at pos with width wid
        idx = np.arange(n, dtype=np.float64)
        f += amp * np.exp(-((idx - pos) ** 2) / (2.0 * wid * wid))
    return _normalize(np.maximum(f, 0.0))


def _trapezoid_seed(n, n_humps, rng):
    """Generate a trapezoid/hat-function based seed."""
    f = np.zeros(n, dtype=np.float64)
    if n_humps == 1:
        w = int(n * 0.6)
        start = (n - w) // 2
        f[start:start + w] = 1.0
    elif n_humps == 2:
        w = int(n * 0.35)
        # Two humps
        s1 = 0
        s2 = n - w
        f[s1:s1 + w] = 1.0
        f[s2:s2 + w] = 1.0
    elif n_humps == 3:
        w = int(n * 0.22)
        positions = [0, n // 2 - w // 2, n - w]
        for p in positions:
            f[max(0, p):min(n, p + w)] = 1.0
    else:
        w = max(2, n // (n_humps * 2))
        for h in range(n_humps):
            center = int((h + 0.5) * n / n_humps)
            s = center - w // 2
            f[max(0, s):min(n, s + w)] = 1.0
    return _normalize(f)


def _fourier_seed(n, rng):
    """Smooth low-frequency perturbation of uniform."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for m in (1, 2, 3, 5, 8, 13):
        amp = float(rng.uniform(0.2, 0.8)) / math.sqrt(m)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * np.cos(2.0 * np.pi * m * x + phase)
        phase2 = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * 0.7 * np.sin(2.0 * np.pi * m * x + phase2)
    f = np.exp(0.8 * u)
    return _normalize(f)


def _power_law_seed(n, gamma, rng):
    """Power-law shaped seed."""
    x = np.arange(n, dtype=np.float64)
    if gamma > 0:
        f = (x + 1.0) ** (-gamma)
    else:
        f = (x + 1.0) ** (abs(gamma) * 0.5)
    # Add mirror for symmetry
    f_mirror = f[::-1]
    blend = float(rng.uniform(0.3, 0.7))
    f = blend * f + (1.0 - blend) * f_mirror
    return _normalize(f)


def _two_component_seed(n, split, gamma_l, gamma_r, rng):
    """Two-component seed: different power laws on left and right halves."""
    f = np.zeros(n, dtype=np.float64)
    x_left = np.arange(split, dtype=np.float64)
    x_right = np.arange(n - split, dtype=np.float64)
    f[:split] = (x_left + 1.0) ** (-gamma_l)
    f[split:] = (n - split - x_right) ** (-gamma_r)
    return _normalize(np.maximum(f, 1e-30))


def _smooth(f, window):
    """Uniform moving-average smoothing."""
    n = f.size
    if n <= 2 or window < 3:
        return f
    window = min(window, n)
    kernel = np.ones(window, dtype=np.float64) / float(window)
    out = np.convolve(f, kernel, mode="same")
    return _normalize(out)


def _interp_to_length(f, n):
    """Resample f to length n."""
    f = _normalize(f)
    m = f.size
    if m == n:
        return f
    if n == 1:
        return np.array([1.0])
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m)
    x_new = np.linspace(0.0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f))


def _coordinate_descent(f, c, budget_s, rng, verbose=False):
    """
    Local search by exact single mass transfers with adaptive step sizes.
    Returns (f, score, c).
    """
    f = _normalize(f)
    n = f.size
    L = 2 * n - 1
    if n < 2 or budget_s <= 0.0:
        c = _autocorr(f)
        return f, _score(c, n), c

    c = np.asarray(c, dtype=np.float64).copy()
    maxc = float(c.max())
    best_f = f.copy()
    best_mc = maxc
    best_score = _score(c, n)

    deadline = time.monotonic() + float(budget_s)
    no_improve = 0
    move_count = 0

    # Adaptive parameters based on n
    if n > 10000:
        D, R = 8, 8
        activeK = min(L, max(50, n // 20))
    elif n > 3000:
        D, R = 12, 12
        activeK = min(L, max(40, n // 10))
    else:
        D, R = 16, 16
        activeK = min(L, max(30, n // 8))

    delta_buf = np.zeros(L, dtype=np.float64)
    total_buf = np.zeros(L, dtype=np.float64)

    while time.monotonic() < deadline and no_improve < 30:
        maxc = float(c.max())
        if maxc <= 0:
            break

        # Identify active lags (near-maximum convolution positions)
        if activeK >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, L - activeK)[L - activeK:]

        # Compute gradient-like vector: g[i] = sum over active k of c[k] * contribution
        # Position i contributes to C[k] for k in [i, i+n-1]
        # Use FFT-based approach for efficiency
        p = np.zeros(L, dtype=np.float64)
        p[active] = 1.0
        # g[i] = 2 * sum_k p[k] * f[k-i] (for valid indices)
        # This is a cross-correlation
        m_size = L + n - 1
        fft_sz = 1
        while fft_sz < m_size:
            fft_sz <<= 1
        P = np.fft.rfft(p, fft_sz)
        Fr = np.fft.rfft(f[::-1], fft_sz)
        conv_g = np.fft.irfft(P * Fr, fft_sz)
        g = 2.0 * conv_g[n - 1:n - 1 + n]

        # Select donors (high g) and receivers (low g)
        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)

        donors = []
        donor_set = set()
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                donor_set.add(idx)
                if len(donors) >= D:
                    break

        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            continue

        # Precompute unit vectors for donors and receivers
        donor_units = []
        for i in donors:
            u = np.zeros(L, dtype=np.float64)
            u[i:i + n] = -f
            donor_units.append((i, u, float(f[i])))

        recv_units = []
        for j in receivers:
            u = np.zeros(L, dtype=np.float64)
            u[j:j + n] = f
            recv_units.append((j, u))

        best_mc = maxc
        best_move = None

        # Try candidate transfers with adaptive step sizes
        for i, ui, fi in donor_units:
            for j, uj in recv_units:
                # Compute the linear part L(k)
                np.add(ui, uj, out=delta_buf)
                # For the linear part: delta_lin(k) = 2*dv * (f[k-j] - f[k-i])
                # We stored (ui+uj) which is (f[k-j] - f[k-i]) for valid k
                # Try multiple step sizes using ternary search on dv
                # f(dv) = max_k [c(k) + 2*dv*(ui+uj)(k) + dv^2*Q(k)]
                # Q is nonzero only at 2i, 2j, i+j

                # Quick evaluation at a few points
                fracs = (0.05, 0.15, 0.3, 0.5, 0.8)
                local_best = maxc
                local_best_dv = 0.0

                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    np.multiply(delta_buf, 2.0 * dv, out=total_buf)
                    q = dv * dv
                    total_buf[2 * j] += q
                    total_buf[2 * i] += q
                    total_buf[i + j] -= 2.0 * q
                    mc = _max_with_delta(c, total_buf)
                    if mc < local_best:
                        local_best = mc
                        local_best_dv = dv

                if local_best < best_mc - 1e-18:
                    # Affine search around local_best_dv
                    step = local_best_dv * 0.2
                    for sgn in (-1.0, 1.0):
                        dv_try = local_best_dv + sgn * step
                        if dv_try <= 0 or dv_try > fi:
                            continue
                        np.multiply(delta_buf, 2.0 * dv_try, out=total_buf)
                        q = dv_try * dv_try
                        total_buf[2 * j] += q
                        total_buf[2 * i] += q
                        total_buf[i + j] -= 2.0 * q
                        mc = _max_with_delta(c, total_buf)
                        if mc < local_best:
                            local_best = mc
                            local_best_dv = dv_try

                    if local_best < best_mc - 1e-18:
                        best_mc = local_best
                        best_move = (i, j, local_best_dv)

        # Random fallback if no improvement found
        if best_move is None:
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                dv = float(f[i]) * float(rng.uniform(0.05, 0.5))
                d = _transfer_delta(f, i, j, dv, L)
                mc = _max_with_delta(c, d)
                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)

        if best_move is None or best_mc >= maxc - 1e-15:
            no_improve += 1
            # Try larger random moves when stuck
            if no_improve % 5 == 3:
                i = int(rng.integers(0, n))
                if f[i] > 1e-6:
                    j = int(rng.integers(0, n))
                    if j != i:
                        dv = float(f[i]) * float(rng.uniform(0.3, 0.9))
                        d = _transfer_delta(f, i, j, dv, L)
                        mc = _max_with_delta(c, d)
                        if mc < maxc:
                            best_move = (i, j, dv)
                            best_mc = mc
            continue

        i, j, dv = best_move
        d = _transfer_delta(f, i, j, dv, L)
        f[i] -= dv
        f[j] += dv
        c = c + d
        move_count += 1
        no_improve = 0

        cur_score = _score(c, n)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Periodic full re-evaluation to avoid drift
        if move_count % 80 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            mc = float(c.max())
            cur_score = _score(c, n)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    mc = float(c.max())
    sc = _score(c, n)
    if sc <= best_score:
        return f, sc, c
    return best_f, best_score, _autocorr(best_f)


def _multiplicative_reweight(f, c, budget_s, rng, verbose=False):
    """
    Multiplicative reweighting: reduce mass at positions contributing
    to high-convolution lags.
    """
    f = _normalize(f)
    n = f.size
    L = 2 * n - 1
    c = np.asarray(c, dtype=np.float64).copy()
    maxc = float(c.max())
    best_f = f.copy()
    best_score = _score(c, n)

    deadline = time.monotonic() + float(budget_s)
    alpha = 0.3

    while time.monotonic() < deadline:
        # Compute position-weighted load
        # W[i] = sum of c[k] for k where position i contributes
        # Position i contributes to C[k] for k in [i, i+n-1]
        # Use cumulative sum for efficiency
        c_clipped = np.maximum(c, 0.0)
        # For position i, the lags it contributes to are max(0,i) to min(L-1, i+n-1)
        # W[i] = cumsum_c[min(L-1, i+n-1)] - (cumsum_c[i-1] if i > 0 else 0)
        cumc = np.concatenate(([0.0], np.cumsum(c_clipped)))
        W = np.zeros(n, dtype=np.float64)
        starts = np.arange(n)
        ends = np.minimum(L, np.arange(n) + n)
        W = cumc[ends] - cumc[starts]

        # Softmax weights over lags (sharper = more targeted)
        T = maxc * 0.1 + 1e-300
        z = (c - maxc) / T
        np.clip(z, -50.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / L)
        else:
            p /= ps

        # Weighted load
        cumw = np.concatenate(([0.0], np.cumsum(p * c_clipped)))
        Ww = cumw[ends] - cumw[starts]

        # Multiplicative update
        Ww_safe = np.maximum(Ww, 1e-15)
        avg_load = float(np.mean(Ww_safe))
        update = (avg_load / Ww_safe) ** alpha
        f = f * update
        f = _normalize(f)
        c = _autocorr(f)
        maxc = float(c.max())
        cur_score = _score(c, n)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            if verbose:
                print(f"    reweight: score={cur_score:.6f}")

        # Decay alpha
        alpha *= 0.98
        if alpha < 0.02:
            break

    return best_f, best_score, _autocorr(best_f)


def _gradient_step(f, c, budget_s, rng, verbose=False):
    """
    Annealed gradient descent in log-domain with periodic coordinate polish.
    """
    f = _normalize(f)
    n = f.size
    L = 2 * n - 1
    c = np.asarray(c, dtype=np.float64).copy()
    maxc = float(c.max())
    best_f = f.copy()
    best_score = _score(c, n)

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.06
    min_sigma = 1e-10
    no_accept = 0
    it = 0
    last_coord = time.monotonic()

    if n < 3000:
        coord_budget = 0.3
        coord_interval = 0.5
    else:
        coord_budget = 0.5
        coord_interval = 0.7

    while time.monotonic() < deadline:
        it += 1
        maxc = float(c.max())
        if maxc <= 0:
            break

        # Softmax weights
        T = maxc * sigma_rel + 1e-300
        z = (c - maxc) / T
        np.clip(z, -50.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / L)
        else:
            p /= ps

        # Weighted gradient via FFT
        m_size = L + n - 1
        fft_sz = 1
        while fft_sz < m_size:
            fft_sz <<= 1
        P = np.fft.rfft(p, fft_sz)
        Fr = np.fft.rfft(f[::-1], fft_sz)
        conv_g = np.fft.irfft(P * Fr, fft_sz)
        g = 2.0 * conv_g[n - 1:n - 1 + n]
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 20:
                exps = (0.005, 0.02, 0.1, 0.5, 2.0)
            else:
                exps = (0.02, 0.08, 0.3, 1.0, 4.0)

            for e in exps:
                arg = -e * (gm / scale)
                np.clip(arg, -40.0, 40.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = _score(c, n)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
            sigma_rel = max(min_sigma, sigma_rel * 0.995)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_coord > coord_interval or no_accept > 30:
                fp, sc, cp = _coordinate_descent(f, c, coord_budget, rng)
                if sc < best_score - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    best_score = sc
                    best_f = f.copy()
                    accepted = True
                    no_accept = 0
                elif no_accept > 60:
                    # Perturb and restart
                    r = float(rng.random())
                    if r < 0.5:
                        sig = float(rng.uniform(0.05, 0.8))
                        f = f * np.exp(sig * rng.standard_normal(n))
                    else:
                        alpha = float(rng.uniform(0.1, 0.5))
                        noise = rng.random(n)
                        noise /= noise.sum()
                        f = (1.0 - alpha) * best_f + alpha * noise
                    f = _normalize(f)
                    c = _autocorr(f)
                    maxc = float(c.max())
                    sigma_rel = min(0.3, sigma_rel * 3.0 + 1e-4)
                    no_accept = 0
                last_coord = time.monotonic()
            else:
                sigma_rel = max(min_sigma, sigma_rel * 0.998)

        if it % 5000 == 0 and verbose:
            cur_score = _score(c, n)
            print(f"    grad iter={it} best={best_score:.6f} cur={cur_score:.6f} sigma={sigma_rel:.3e}")

    c = _autocorr(f)
    sc = _score(c, n)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    return best_f, best_score, _autocorr(best_f)


def _combined_optimize(f0, budget_s, rng, tag="", verbose=False):
    """
    Combined optimization: reweighting -> gradient -> coordinate descent.
    """
    f0 = _normalize(f0)
    n = f0.size
    if n == 0 or budget_s <= 0:
        c = _autocorr(f0)
        return f0, _score(c, n)

    c0 = _autocorr(f0)
    score0 = _score(c0, n)

    if verbose:
        print(f"  [{tag}] start score={score0:.6f}")

    remaining = budget_s
    f = f0

    # Phase 1: Multiplicative reweighting (fast global improvement)
    rw_time = min(30.0, 0.12 * budget_s)
    if remaining > rw_time + 60:
        f, sc_rw, _ = _multiplicative_reweight(f, c0, rw_time, rng)
        remaining -= (budget_s - (remaining))  # just track elapsed
        elapsed = budget_s - remaining
        remaining = budget_s - (time.monotonic() - (time.monotonic() - elapsed))
        # Simpler: just compute remaining time
        total_elapsed = time.monotonic() - (time.monotonic() - (time.monotonic() - time.monotonic()))
        # Just recompute
        pass

    # Let me restructure with proper time tracking
    start_time = time.monotonic()
    f = _normalize(f0)
    c = _autocorr(f)

    # Phase 1: Reweighting
    rw_budget = min(25.0, 0.10 * budget_s)
    if budget_s - (time.monotonic() - start_time) > rw_budget + 80:
        f, sc1, c = _multiplicative_reweight(f, c, rw_budget, rng)
        if verbose:
            print(f"  [{tag}] after reweight: {sc1:.6f}")

    # Phase 2: Gradient optimization
    remaining = budget_s - (time.monotonic() - start_time)
    grad_budget = min(remaining * 0.5, max(30.0, remaining * 0.4))
    if remaining > grad_budget + 40:
        f, sc2, c = _gradient_step(f, c, grad_budget, rng, verbose=verbose)
        if verbose:
            print(f"  [{tag}] after gradient: {sc2:.6f}")

    # Phase 3: Coordinate descent polish
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 20:
        f, sc3, c = _coordinate_descent(f, c, remaining - 5, rng)
        if verbose:
            print(f"  [{tag}] after coord: {sc3:.6f}")

    # Final evaluation
    f = _normalize(f)
    c = _autocorr(f)
    sc = _score(c, n)
    if verbose:
        print(f"  [{tag}] final: {sc:.6f}")
    return f, sc


def _build_seed_pool(n, rng, f_given=None):
    """Build a diverse pool of initial seeds."""
    seeds = []

    def add_seed(name, f):
        f = _normalize(f)
        try:
            sc = _score(_autocorr(f), n)
        except Exception:
            return
        if np.isfinite(sc):
            seeds.append((sc, len(seeds), name, f))

    # Given sequence
    if f_given is not None and f_given.size > 0:
        fg = _interp_to_length(f_given, n)
        add_seed("given", fg)
        add_seed("given_rev", fg[::-1])
        for w in (5, 15, 51):
            add_seed(f"given_sm{w}", _smooth(fg, w))

    # Uniform
    add_seed("uniform", np.ones(n, dtype=np.float64))

    # Multi-hump seeds
    hump_configs = [
        (2, [0.25 * n, 0.75 * n], [0.15 * n, 0.15 * n], [1.0, 1.0]),
        (2, [0.2 * n, 0.8 * n], [0.2 * n, 0.1 * n], [1.0, 0.6]),
        (2, [0.3 * n, 0.7 * n], [0.12 * n, 0.12 * n], [1.0, 1.0]),
        (3, [0.2 * n, 0.5 * n, 0.8 * n], [0.1 * n, 0.1 * n, 0.1 * n], [1.0, 0.8, 1.0]),
        (3, [0.15 * n, 0.5 * n, 0.85 * n], [0.15 * n, 0.1 * n, 0.15 * n], [1.0, 0.5, 1.0]),
        (4, [0.15 * n, 0.35 * n, 0.65 * n, 0.85 * n], [0.1 * n] * 4, [1.0, 0.7, 0.7, 1.0]),
    ]
    for hi, (nh, poss, wids, amps) in enumerate(hump_configs):
        add_seed(f"hump{hi}", _multi_hump_seed(n, nh, poss, wids, amps, rng))

    # Trapezoid seeds
    for nh in (2, 3, 4, 5):
        add_seed(f"trap{nh}", _trapezoid_seed(n, nh, rng))

    # Fourier seeds
    for i in range(8):
        add_seed(f"fourier{i}", _fourier_seed(n, rng))

    # Power law seeds
    for gam in (0.3, 0.5, 0.8, 1.2, 2.0):
        add_seed(f"pow{gam}", _power_law_seed(n, gam, rng))

    # Two-component seeds
    for split_frac in (0.3, 0.4, 0.5):
        split = int(n * split_frac)
        for gl in (0.5, 1.0, 2.0):
            for gr in (0.5, 1.0, 2.0):
                add_seed(f"2comp_{split_frac}_{gl}_{gr}",
                         _two_component_seed(n, split, gl, gr, rng))

    # Random smooth seeds
    for i in range(6):
        f_rand = np.exp(0.5 * rng.standard_normal(n))
        add_seed(f"rand_sm{i}", _smooth(f_rand, max(3, n // 100)))

    seeds.sort()
    return seeds


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    total_deadline = total_start + max(60.0, float(budget_s)) - 30.0

    # Load the provided construction if available
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(1024, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    print(f"Sequence length: {n}")
    c0 = _autocorr(f0)
    best_score = _score(c0, n)
    best_f = f0.copy()
    print(f"Initial score: {best_score:.6f}")

    # Build seed pool
    print("Building seed pool...")
    pool = _build_seed_pool(n, rng, f0)
    print(f"Seed pool size: {len(pool)}, best seed score: {pool[0][0]:.6f}")

    # Sort seeds and select top candidates for optimization
    # Allocate time strategically
    time_since_start = time.monotonic() - total_start
    remaining = total_deadline - time.monotonic()

    # Select top seeds, ensuring diversity
    top_seeds = []
    seen_scores = set()
    for sc, idx, name, f in pool:
        if len(top_seeds) >= 8:
            break
        # Avoid near-duplicates (same score within 1e-6)
        score_key = round(sc, 5)
        if score_key not in seen_scores:
            top_seeds.append((sc, name, f))
            seen_scores.add(score_key)

    # If the given sequence is the best, include it first
    # Already handled by sorting

    print(f"Selected {len(top_seeds)} seeds for optimization")
    for sc, name, f in top_seeds[:5]:
        print(f"  {name}: {sc:.6f}")

    # Optimize each seed with time allocation
    # First seed gets the most time, decreasing for subsequent
    weights = [0.35, 0.20, 0.12, 0.08, 0.06, 0.05, 0.05, 0.05, 0.04][:len(top_seeds)]
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    for idx, (sc, name, f) in enumerate(top_seeds):
        time_left = total_deadline - time.monotonic()
        if time_left < 60:
            break
        alloc = min(time_left - 40, 0.85 * remaining * weights[idx])
        alloc = max(30.0, alloc)

        if time_left - alloc < 60 and idx < len(top_seeds) - 1:
            alloc = time_left - 60
            if alloc < 30:
                break

        print(f"\n=== Optimizing seed {idx+1}/{len(top_seeds)}: {name} "
              f"(init={sc:.6f}, alloc={alloc:.0f}s, current_best={best_score:.6f}) ===")

        f_opt, sc_opt = _combined_optimize(f, alloc, rng, tag=name, verbose=(idx < 2))

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f} (target: {TARGET_SCORE})")
            if best_score < TARGET_SCORE:
                print("  *** BELOW TARGET! ***")

    # Final extended polish on the best found
    time_left = total_deadline - time.monotonic()
    print(f"\n=== Final polish (budget={time_left:.0f}s) ===")
    if time_left > 30:
        f_final, sc_final = _combined_optimize(best_f, time_left - 10, rng, tag="final", verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))

    # Final score computation
    c_final = _autocorr(best_f)
    best_score = _score(c_final, best_f.size)
    print(f"\nFINAL internal score = {best_score:.6f}")
    print(f"FINAL target = {TARGET_SCORE}")
    print(f"Total time = {time.monotonic() - total_start:.1f}s")

    # Try official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
            print(f"Official score = {best_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_f]
