import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score_from_conv(a, c):
    """Compute score given the sequence and its precomputed convolution."""
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_from_weights(w, a):
    """Compute gradient: 2 * cross-correlation of w with a."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:

    else:
        w_trimmed = w
    fw = np.fft.rfft(w_trimmed, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

"""
Low-peak step-function optimization via adaptive mirror descent with momentum,

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_from_conv(a, c):
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_from_weights(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size >= N:
        w_use = w[:N]
    else:
        w_use = w
    fw = np.fft.rfft(w_use, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_multi_sine(n, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s > 0.01:
        a /= s

def _make_raised_cosine(n, width=0.8):

    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)

def _make_chirp(n, alpha=0.4):
    i = np.arange(n, dtype=np.float64)

    else:
        a3, s3 = a2, s2
        print("[Phase 3] Skipped", flush=True)

    # Phase 4: Multi-restart refinement
    d4 = hard_deadline - 25.0
    remaining4 = d4 - time.monotonic()
    if remaining4 > 60:
        print(f"[Phase 4] Multi-restart refinement ({remaining4:.0f}s)...", flush=True)
        a4, s4 = _multi_restart_refine(a3, s3, d4, n_restarts=10, seed=seed + 100)
        print(f"  Multi-restart best: {s4:.8f}", flush=True)
    else:
        a4, s4 = a3, s3
        print("[Phase 4] Skipped", flush=True)

    # Also try hard-peaks from the scan init (different trajectory)
    d4b = hard_deadline - 20.0
    remaining4b = d4b - time.monotonic()
    a5, s5 = init_a, init_score
    if remaining4b > 60 and init_a.size == a4.size:
        print(f"[Phase 4b] Hard-peak from init ({remaining4b:.0f}s)...", flush=True)
        a5, s5 = _optimize_hard_peaks(init_a, d4b, K=8,
                                       lr0=0.4 * init_a.size)
        print(f"  Hard-peak from init: {s5:.8f}", flush=True)

    # Phase 5: Validate
    print("[Phase 5] Validating...", flush=True)
    candidates = [
        ("mirror", a2, s2),
        ("hard", a3, s3),
        ("multi", a4, s4),
        ("hard_init", a5, s5),
        ("init", init_a, init_score),
    ]
    if prev_arr is not None:
        candidates.append(("prev", prev_arr, prev_score))

    candidates.sort(key=lambda x: x[2])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in candidates:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)

    if best_out is None:
        best_out = [float(x) for x in start_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = start_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time={total_time:.1f}s, Best={best_actual:.8f}", flush=True)
    return best_out
