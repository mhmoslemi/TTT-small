```python
"""
High-resolution mirror-descent search for low-peak nonnegative step functions.

The score is scale invariant, so all sequences are kept normalized to sum 1.
The main subroutine repeatedly:
  1. computes the self-convolution c = a * a by FFT,
  2. builds a temperature-annealed subgradient p on the large convolution lags,
  3. computes L_i = sum_j p_j a_{j-i} by FFT cross-correlation,
  4. performs a multiplicative update a_i <- a_i * (M / L_i)^beta.

This is a smoothed multiplicative-weights descent for minimizing the maximum
convolution coefficient.  The search is further strengthened by:
  - multi-lengthbruck (1024 -> 2048 -> optionally 4096),
  - warm starts from the provided height_sequence_1 when available,
  - analytic and random smooth initializations,
  - occasional power-transform and random multiplicative restarts,
  - a final hard top peak polish,
  - an optional scipy HiGHS LP direction (used only when time permits).
"""

import numpy as np
import time
import math
from operator import itemgetter

try:
    import scipy.optimize as _sco
    import scipy.sparse as _sps
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
    else:
        a = a / s
    return a


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = _fft_convolve(a)
    if c.size == 0:
        return float("inf")
    M = float(np.max(c))
    return float(2.0 * n * M / (s * s))


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == 0:
        return _normalize(np.ones(n, dtype=np.float64))
    if m == 1:
        return _normalize(np.full(n, float(a[0]), dtype=np.float64))
    if m == n:
        return _normalize(a)
    x_old = np.arange(m, dtype=np.float64)
    x_new = np.linspace(0.0, float(m - 1), int(n), dtype=np.float64)
    y = np.interp(x_new, x_old, a)
    y = np.where(np.isfinite(y) & (y >= 0.0), y, 0.0)
    s = float(y.sum())
    if s <= 1e-12:
        return _normalize(np.ones(n, dtype=np.float64))
    return y / s


def _power_transform(a, gamma):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    if gamma == 1.0:
        return _normalize(a)
    b = np.power(a, float(gamma))
    b = np.where(np.isfinite(b) & (b >= 0.0), b, 0.0)
    s = float(b.sum())
    if s <= 1e-12:
        return _normalize(np.ones(a.size, dtype=np.float64))
    return b / s


def _make_sine(n, amp, phase, mode=1):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(mode) * i / float(n) + float(phase))
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
    return a / s


def _make_power(n, p):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(x, -float(p))
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    return _normalize(a)


def _make_geometric(n, r):
    i = np.arange(n, dtype=np.float64)
    a = np.power(float(r), i)
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    return _normalize(a)


def _make_random_fourier(n, rng, max_amp=0.9):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for mode in (1, 2, 3, 5, 8, 13):
        amp = float(rng.uniform(0.0, max_amp / float(mode)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.sin(2.0 * np.pi * float(mode) * i / float(n) + phase)
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
    return _normalize(a)


def _weights_global(c, M, temp):
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)
    if M <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64)
    tau = max(1e-15, float(M) * float(temp))
    x = (c - float(M)) / tau
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return ex / s


def _weights_top(c, M, K, temp):
    L = c.size
    K = int(max(1, min(K, L)))
    if M <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64)
    if K >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(c, -K)[-K:]
    vals = c[idx]
    vmax = float(np.max(vals))
    tau = max(1e-15, float(M) * float(temp))
    x = (vals - vmax) / tau
    w = np.exp(x)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        w = w / s
    p = np.zeros(L, dtype=np.float64)
    p[idx] = w
    return p


def _optimize_ratio(a0, deadline, mode="global", k_factor=1.0,
                    temp0=1.5, temp1=0.08, beta0=0.5, beta1=0.05,
                    seed=123):
    """
    Multiplicative ratio-descent.  Returns (best_array, best_internal_score).
    mode: 'global', 'top', or 'mix'.
    """
    t_end = float(deadline)
    a = _normalize(a0)
    n = a.size
    if n == 0:
        a = np.array([1.0], dtype=np.float64)
        n = 1
    N = _next_pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_fft(best)

    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    local_it = 0
    no_imp = 0
    restarts = 0
    patience = 30000
    beta_scale = 1.0
    max_log = math.log(3.5)
    K = int(max(8, k_factor * math.sqrt(float(n))))

    while time.monotonic() < t_end:
        frac = min(1.0, float(local_it) / 100000.0)
        beta = (float(beta0) + (float(beta1) - float(beta0)) * frac) * beta_scale
        temp = float(temp0) + (float(temp1) - float(temp0)) * frac

        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.where(np.isfinite(c) & (c >= 0.0), c, 0.0)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            a = best.copy()
            local_it += 1
            continue

        score = float(2.0 * n * M)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-10):
            best_score = score
            best = a.copy()
            no_imp = 0
            if beta_scale < 1.0:
                beta_scale = min(1.0, beta_scale * 1.04)
        else:
            no_imp += 1
            if score > best_score * 1.01 + 1e-9:
                a = best.copy()
                beta_scale = max(0.25, beta_scale * 0.7)
                no_imp = 0
                continue
            if no_imp >= patience:
                # Escape: power transform, multiplicative noise, or both alternating.
                r = restarts % 4
                if r == 1:
                    a = _power_transform(best, 0.85)
                elif r == 2:
                    a = _power_transform(best, 1.15)
                else:
                    sigma = 0.06 + 0.045 * min(restarts, 6)
                    a = best * np.exp(rng.standard_normal(n) * sigma)
                    a = _normalize(a)
                local_it = 0
                no_imp = 0
                restarts += 1
                beta_scale = 1.0
                continue

        if mode == "top":
            p = _weights_top(c, M, K, temp)
        elif mode == "mix":
            p_top = _weights_top(c, M, K, temp)
            p_glob = _weights_global(c, M, max(temp * 1.5, 0.05))
            mix = 1.0 - 0.5 * frac
            p = mix * p_top + (1.0 - mix) * p_glob
            s = float(p.sum())
            if s > 1e-300:
                p = p / s
            else:
                p = p_top
        else:
            p = _weights_global(c, M, temp)

        fw = np.fft.rfft(p, N)
        L = np.fft.irfft(np.conj(fa) * fw, N)[:n]
        L = np.where(np.isfinite(L) & (L >= 0.0), L, 0.0)
        eps = max(1e-15, 1e-8 * M)
        ratio = M / (L + eps)
        logf = beta * np.log(ratio)
        logf = np.where(np.isfinite(logf), logf, 0.0)
        logf = np.clip(logf, -max_log, max_log)

        cap_frac = float(np.sum(np.abs(logf) >= max_log * 0.95)) / float(n)
        if cap_frac > 0.12:
            beta_scale = max(0.30, beta_scale * 0.87)
        elif cap_frac < 0.005 and beta_scale < 1.0:
            beta_scale = min(1.0, beta_scale * 1.015)

        a = a * np.exp(logf)
        a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
        s = float(a.sum())
        if s <= 1e-10:
            a = best.copy()
            local_it += 1
            continue
        a = a / s
        local_it += 1

        if local_it % 20000 == 0:
            a = (1.0 - 1e-5) * a + 1e-5 * (1.0 / float(n))
            a = _normalize(a)

    return best, best_score


def _build_hankel_sparse(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    base_rows = np.tile(np.arange(n, dtype=np.int32), n)
    rows = cols + base_rows
    data = np.tile(a, n)
    B = _sps.coo_matrix((data, (rows, cols)), shape=(L, n)).tocsr()
    return B


def _lp_refine(a0, deadline, max_calls=3, call_budget=12.0):
    """
    Optional iterative LP-direction polish, based on max sum g s.t. a*g <= M.
    Only used for moderate n to avoid excessive LP solve time.
    """
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_score = _score_fft(best)
    if not _HAVE_SCIPY or n > 1024:
        return best, best_score

    calls = 0
    while calls < int(max_calls) and time.monotonic() + 3.0 < float(deadline):
        rem = float(deadline) - time.monotonic()
        cb = min(float(call_budget), max(2.0, rem - 5.0))
        if cb < 2.0:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break
        cur_score = float(2.0 * n * M)
        try:
            B = _build_hankel_sparse(a)
            b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
            bounds = [(0.0, None)] * n
            res = _sco.linprog(
                c=-np.ones(n, dtype=np.float64),
                A_ub=B,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": cb}
            )
        except Exception as exc:
            print("    LP exception:", exc, flush=True)
            break

        accepted = False
        if getattr(res, "success", False) and getattr(res, "x", None) is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.where(np.isfinite(g) & (g >= 0.0), g, 0.0)
            gs = float(g.sum())
            if gs > 1e-12:
                g = g / gs
                best_mix = a.copy()
                best_mix_score = cur_score
                for alpha in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12):
                    mix = (1.0 - alpha) * a + alpha * g
                    mix = _normalize(mix)
                    sc = _score_fft(mix)
                    if np.isfinite(sc) and sc < best_mix_score:
                        best_mix_score = sc
                        best_mix = mix
                if best_mix_score < cur_score or best_mix_score < best_score:
                    a = best_mix
                    if best_mix_score < best_score:
                        best_score = best_mix_score
                        best = a.copy()
                    accepted = True
        calls += 1
        if not accepted:
            # If LP gives no local valley move, stop to avoid wasting time.
            if calls >= 2:
                break
    return best, best_score


def _choose_resampled(n, sources, transforms=(0.85, 1.0, 1.15, 1.35)):
    best_a = None
    best_sc = float("inf")
    for src in sources:
        if src is None:
            continue
        ar = _resample(src, n)
        for g in transforms:
            ag = _power_transform(ar, g)
            sc = _score_fft(ag)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = ag
    if best_a is None:
        best_a = _make_sine(n, 0.7, 0.0)
        best_sc = _score_fft(best_a)
    return best_a, best_sc


def _add_candidate(bests, a, name):
    a = np.asarray(a, dtype=np.float64)
    sc = _score_fft(a)
    if np.isfinite(sc):
        bests.append((a.copy(), str(name), float(sc), int(a.size)))

def _sanitize_sequence(a):
    out = []
    for x in np.asarray(a, dtype=np.float64):
        if not np.isfinite(x):
            v = 0.0
        else:
            v = float(x)
        if v < 0.0:
            v = 0.0
        if v > 1000.0:
            v = 1000.0
        out.append(v)
    s = sum(out)
    if s < 0.01 or len(out) == 0:
        return [1.0]
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):

        return _sanitize_sequence(_make_sine(256, 0.6, 0.0))

                    print(f"[parent] exact score={pe:.8f}", flush=True)
    except Exception:
        parent = None

    # Candidate storage: (array, name, internal_score, length)

            for ph in (0.0, 0.17, 0.35, 0.52):
                a = _make_sine(n, amp, ph)
                _add_candidate(bests, a, f"sine_n{n}_a{amp}_p{ph}")

        # Power / geometric biases.
        for p in (0.35, 0.5, 0.65):
            _add_candidate(bests, _make_power(n, p), f"power_n{n}_p{p}")
        for r in (0.93, 0.97):
            _add_candidate(bests, _make_geometric(n, r), f"geom_n{n}_r{r}")

        # A few random smooth starts.
        for it in range(4):
            a = _make_random_fourier(n, rng, max_amp=0.85)
            _add_candidate(bests, a, f"rand_n{n}_{it}")

    if not bests:
        a = _make_sine(1024, 0.6, 0.0)
        _add_candidate(bests, a, "fallback_sine")

    # Keep only strongest per length for use as warm starts.
    best_by_len = {}
    for arr, name, sc, nn in bests:
        if nn not in best_by_len or sc < best_by_len[nn][1]:
            best_by_len[nn] = (arr, sc, name)

    def _get_len_best(nn):
        if nn in best_by_len:
            return best_by_len[nn][0]
        if parent is not None:
            return _resample(parent, nn)
        return _make_sine(nn, 0.65, 0.0)

    # ---- Phase 1: solid 1024 polish ------------------------------------------------
    low_n = 1024
    low_init = _get_len_best(low_n)
    low_budget = min(360.0, max(90.0, 0.12 * total_budget))
    low_deadline = min(hard_deadline, t0 + low_budget)
    if low_deadline - time.monotonic() > 60.0:
        print(f"[phase1] 1024 global optimization for {low_deadline - time.monotonic():.0f}s", flush=True)
        a_low, s_low = _optimize_ratio(
            low_init, low_deadline, mode="global",
            temp0=1.5, temp1=0.08, beta0=0.5, beta1=0.05,
            seed=seed
        )
        _add_candidate(bests, a_low, "low_global")
        print(f"[phase1] 1024 internal score={s_low:.8f}", flush=True)

    # ---- Phase 2: 2048 main optimization ------------------------------------------
    mid_n = 2048
    mid_sources = [parent, _get_len_best(1024), _get_len_best(mid_n)]
    mid_init, mid_init_sc = _choose_resampled(mid_n, mid_sources)
    _add_candidate(bests, mid_init, "mid_init_best")

    mid_budget = min(1500.0, max(500.0, 0.62 * total_budget))
    mid_deadline = min(hard_deadline, time.monotonic() + mid_budget)
    if mid_deadline - time.monotonic() > 240.0:
        first_frac = 0.68
        first_deadline = time.monotonic() + first_frac * (mid_deadline - time.monotonic())
        print(f"[phase2] 2048 wide global for {first_deadline - time.monotonic():.0f}s", flush=True)
        a_mid, s_mid = _optimize_ratio(
            mid_init, first_deadline, mode="global",
            temp0=1.6, temp1=0.085, beta0=0.5, beta1=0.05,
            seed=seed + 17
        )
        _add_candidate(bests, a_mid, "mid_global1")
        print(f"[phase2] 2048 wide global score={s_mid:.8f}", flush=True)

        if mid_deadline - time.monotonic() > 60.0:
            print(f"[phase2b] 2048 hard-top polish for {mid_deadline - time.monotonic():.0f}s", flush=True)
            a_mid2, s_mid2 = _optimize_ratio(
                a_mid, mid_deadline, mode="top",
                k_factor=0.45, temp0=0.32, temp1=0.012, beta0=0.36, beta1=0.04,
                seed=seed + 31
            )
            _add_candidate(bests, a_mid2, "mid_top1")
            print(f"[phase2b] 2048 hard-top score={s_mid2:.8f}", flush=True)

    # ---- Phase 3: optional 4096 high-resolution ------------------------------------
    high_n = 4096
    if high_n in lengths:
        high_sources = [parent, _get_len_best(mid_n), _get_len_best(low_n)]
        high_init, high_init_sc = _choose_resampled(high_n, high_sources)
        _add_candidate(bests, high_init, "high_init_best")
        high_budget = min(700.0, max(120.0, 0.22 * total_budget))
        high_deadline = min(hard_deadline, time.monotonic() + high_budget)
        if high_deadline - time.monotonic() > 180.0:
            print(f"[phase3] 4096 global for {high_deadline - time.monotonic():.0f}s", flush=True)
            a_high, s_high = _optimize_ratio(
                high_init, high_deadline, mode="global",
                temp0=1.8, temp1=0.10, beta0=0.45, beta1=0.05,
                seed=seed + 53
            )
            _add_candidate(bests, a_high, "high_global")
            print(f"[phase3] 4096 score={s_high:.8f}", flush=True)

    # Sort current internal scores.
    if bests:
        bests.sort(key=itemgetter(2))
        cur_best_arr = bests[0][0].copy()
        cur_best_name = bests[0][1]
        cur_best_sc = bests[0][2]
        cur_best_n = bests[0][3]
        print(f"[before-polish] best internal={cur_best_sc:.8f} len={cur_best_n}", flush=True)

        # Optional LP polish for 1024 only, if there is room.
        rem = hard_deadline - time.monotonic()
        if rem > 260.0 and cur_best_n == 1024 and _HAVE_SCIPY:
            lp_deadline = time.monotonic() + min(150.0, rem - 170.0)
            print(f"[polish] LP refine 1024 for {lp_deadline - time.monotonic():.0f}s", flush=True)
            a_lp, s_lp = _lp_refine(cur_best_arr, lp_deadline, max_calls=3, call_budget=18.0)
            _add_candidate(bests, a_lp, "lp_1024")
            print(f"[polish] LP internal score={s_lp:.8f}", flush=True)

        # Final hard top peak polish on current best.
        rem = hard_deadline - time.monotonic()
        if rem > 110.0:
            polish_budget = min(160.0, rem - 55.0)
            polish_deadline = time.monotonic() + polish_budget
            print(f"[polish] final top polish for {polish_budget:.0f}s", flush=True)
            a_pol, s_pol = _optimize_ratio(
                cur_best_arr, polish_deadline, mode="top",
                k_factor=0.38, temp0=0.28, temp1=0.011, beta0=0.32, beta1=0.038,
                seed=seed + 101
            )
            _add_candidate(bests, a_pol, "final_top")
            print(f"[polish] final top internal score={s_pol:.8f}", flush=True)

        # Re-sort and exact-evaluate the strongest candidates.
        bests.sort(key=itemgetter(2))
        best_exact_score = float("inf")
        best_exact_arr = None

        evaluated = 0
        for arr, name, sc, nn in bests:
            if evaluated >= 8:
                break
            if time.monotonic() > hard_deadline - 4.0:
                break
            actual = _exact_score_array(arr)
            evaluated += 1
            print(f"[exact] {name} len={nn} exact={actual:.8f}", flush=True)
            if np.isfinite(actual) and actual < best_exact_score:
                best_exact_score = actual
                best_exact_arr = arr

        if best_exact_arr is not None:
            return _sanitize_sequence(best_exact_arr)

    # Fallback: best internal or uniform.
    if bests:
        bests.sort(key=itemgetter(2))
        return _sanitize_sequence(bests[0][0])
    return _sanitize_sequence(_make_sine(1024, 0.6, 0.0))
```
