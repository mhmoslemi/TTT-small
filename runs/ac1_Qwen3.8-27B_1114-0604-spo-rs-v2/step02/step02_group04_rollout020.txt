Improved low-peak step-function optimization using variance minimization warm-start
followed by soft/hard max optimization with structural restarts.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant, matches evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """
    Compute 2 * cross-correlation of a and w (first n elements).
    This gives the gradient of c_k = conv(a,a)[k] with respect to a,
    when w is an indicator vector for position k.
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _variance_score(a):
    """Compute ||conv(a,a)||^2 (the variance-related objective)."""
    c = _fft_convolve(a)
    return float(np.dot(c, c))


def _variance_grad(a):
    """Gradient of ||conv(a,a)||^2 with respect to a."""
    c = _fft_convolve(a)
    return 2.0 * _grad_via_fft(c, a)


def _make_sine(n, amp, phase=0.5):
    """Anti-palindromic sinusoidal sequence normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_sine(n, mods):
    """Multi-mode sinusoidal. mods is list of (amp, phase, freq)."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, ph, freq in mods:
        theta = 2.0 * np.pi * freq * (i + ph) / n
        a += amp * np.sin(theta)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size) / a.size
    return a / s


def _optimize_variance(a, deadline, lr0=None):
    """
    Minimize ||conv(a,a)||^2 using mirror descent.
    This makes the convolution as flat as possible.
    """
    a = _normalize(a.copy())
    n = a.size
    if lr0 is None:
        lr0 = 0.1 * n
    
    best_a = a.copy()
    best_vs = _variance_score(best_a)
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.002 * it)
        
        grad = _variance_grad(a)
        grad_proj = grad - np.mean(grad)
        
        update = -lr * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)
        
        if it % 10 == 0:
            vs = _variance_score(a)
            sc = _score(a)
            if vs < best_vs:
                best_vs = vs
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = _normalize(a)
                patience = max(1000, int(patience * 1.5))
                lr0 = max(0.02 * n, lr0 * 0.7)
        
    return best_a, best_score


def _optimize_soft(a, deadline, T_start=5.0, T_end=50.0, lr0=None):
    """
    Soft-max mirror descent for minimizing max(conv(a,a)).
    """
    a = _normalize(a.copy())
    n = a.size
    if lr0 is None:
        lr0 = 0.2 * n
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 8000
    rng = np.random.RandomState(12345)
    
    while time.monotonic() < deadline:
        it += 1
        
        t_frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.005 * it)
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Softmax weights
        c_max = float(np.max(c))
        c_shift = c - c_max
        exp_c = np.exp(T * c_shift)
        s = exp_c.sum()
        if s < 1e-300:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
        else:
            w = exp_c / s
        
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        
        update = -lr * grad_proj
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)
        
        if it % 5 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Structured restart
                a = best_a.copy()
                strategy = rng.randint(3)
                if strategy == 0:
                    # Noise
                    noise = rng.standard_normal(n) * 0.02
                    a = a + noise * a.mean()
                elif strategy == 1:
                    # Reverse
                    a = a[::-1].copy()
                else:
                    # Random shift
                    shift = rng.randint(1, max(2, n // 4))
                    a = np.roll(a, shift)
                a = _normalize(a)
                patience = max(3000, int(patience * 1.3))
                lr0 = max(0.05 * n, lr0 * 0.8)
        
        if time.monotonic() < deadline and it % 100 == 0:
            if best_score < 1.50:
                print(f"    [soft] it={it}, score={best_score:.8f}", flush=True)
    
    return best_a, best_score


def _optimize_hard(a, deadline, K=8, lr0=None):
    """
    Hard-max top-K active set refinement.
    """
    a = _normalize(a.copy())
    n = a.size
    if lr0 is None:
        lr0 = 0.3 * n
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 5000
    rng = np.random.RandomState(99999)
    
    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.003 * it)
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Top-K with soft weights
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        top_vals = c[top_idx]
        T_local = 2.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()
        
        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]
        
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        
        update = -lr * grad_proj
        update = np.clip(update, -6.0, 6.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)
        
        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = _normalize(a)
                patience = max(2000, int(patience * 1.4))
                lr0 = max(0.1 * n, lr0 * 0.8)
        
        if it % 200 == 0:
            if best_score < 1.51:

import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant, matches evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """Compute 2 * cross-correlation of a and w (first n elements)."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size) / a.size
    return a / s

    """Gradient of ||conv(a,a)||^2 with respect to a."""

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()

        hard_time = time_left * (1 - soft_frac) * 0.8

        # Soft phase
        soft_deadline = time.monotonic() + soft_time
        if soft_time > 30:
            print(f"  Round {round_num}: Soft T=[{T_starts[ti]},{T_ends[ti]}] "
                  f"({soft_time:.0f}s), from score={current_score:.8f}", flush=True)
            a_soft, sc_soft = _optimize_soft(
                current_a, soft_deadline,
                T_start=T_starts[ti], T_end=T_ends[ti]
            )
            if sc_soft < current_score:
                current_score = sc_soft
                current_a = a_soft.copy()
                if sc_soft < global_best_score:
                    global_best_score = sc_soft
                    global_best_a = current_a.copy()
                    print(f"    -> new global best: {sc_soft:.8f}", flush=True)

        # Hard phase
        if time.monotonic() < d3_end - 20:
            hard_deadline = min(time.monotonic() + hard_time, d3_end - 20)
            if hard_deadline - time.monotonic() > 20:
                print(f"  Round {round_num}: Hard K={K_vals[Ki]} "
                      f"({hard_deadline - time.monotonic():.0f}s)", flush=True)
                a_hard, sc_hard = _optimize_hard(
                    current_a, hard_deadline, K=K_vals[Ki]
                )
                if sc_hard < current_score:
                    current_score = sc_hard
                    current_a = a_hard.copy()
                if sc_hard < global_best_score:
                    global_best_score = sc_hard
                    global_best_a = a_hard.copy()
                    print(f"    -> new global best: {sc_hard:.8f}", flush=True)

        # Also try optimizing the raw init in parallel (if different length)
        if raw_a.size != current_a.size and time.monotonic() < d3_end - 200:
            raw_deadline = time.monotonic() + 150
            a_raw_opt, sc_raw = _optimize_soft(raw_a, raw_deadline, T_start=5.0, T_end=50.0)
            if sc_raw < global_best_score:
                global_best_score = sc_raw
                global_best_a = a_raw_opt.copy()
                print(f"  Raw init improved to: {sc_raw:.8f}", flush=True)
            raw_a = a_raw_opt
            raw_score = sc_raw

    # Phase 4: Final hard polish on global best
    if time.monotonic() < hard_deadline - 5:
        final_budget = min(200, hard_deadline - 5 - time.monotonic())
        if final_budget > 30:
            print(f"[Phase 4] Final polish ({final_budget:.0f}s)...", flush=True)
            a_final, sc_final = _optimize_hard(global_best_a,
                                               time.monotonic() + final_budget, K=4)
            if sc_final < global_best_score:
                global_best_score = sc_final
                global_best_a = a_final.copy()
                print(f"  Final: {sc_final:.8f}", flush=True)

    # Phase 5: Validation
    print("[Phase 5] Validating...", flush=True)

    all_candidates = [
        ("global", global_best_a, global_best_score),
        ("current", current_a, current_score),
        ("var", best_after_var, best_after_var_score),
        ("init", candidates[0][1], candidates[0][0]),
    ]

    # Deduplicate by identity
    unique_candidates = []
    for name, arr, sc in all_candidates:
        if not any(np.array_equal(arr, ua) for _, ua, _ in unique_candidates):
            unique_candidates.append((name, arr, sc))

    unique_candidates.sort(key=lambda x: x[2])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in unique_candidates[:5]:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)

    if best_out is None:
        arr = candidates[0][1]
        best_out = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = candidates[0][0]
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
