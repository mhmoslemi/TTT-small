"""
Search for non-negative sequences minimizing
    score(a) = 2 * n * max(convolve(a, a)) / sum(a)^2.

The main novelty here is an exact coordinate-scaling local search:
for coordinate i, rescale one coordinate relative to all others while
keeping sum(a)=1. The full convolution after such a move can be updated
/ evaluated in O(n), and several candidate values are tested per
coordinate.  This is combined with multiplicative gradient descent using
FFT-accelerated convolution/gradient, frequency-domain convolution
flattening, block scaling, diverse restarts/resamplings, and final
polishing.

Only numpy/time are used.
"""

import numpy as np
import time


_FACTORS = (0.25, 0.5, 1.25, 2.0, 4.0)
_EXP_SIZE = 3
_RAND_SIZE = 1


def _next_pow2(x):
    x = max(1, int(x))
    y = 1
    while y < x:
        y <<= 1
    return y


def _sanitize_array(x):
    p = np.asarray(x, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p[p < 0.0] = 0.0
    s = float(p.sum())
    if s <= 1e-14 or not np.isfinite(s):
        return None
    return p / s


def _init_uniform(n):
    n = max(1, int(n))
    return np.ones(n, dtype=float) / float(n)


def _init_two_level(n, rng):
    n = max(1, int(n))
    if n == 1:
        return _init_uniform(n)
    k = int(n * float(rng.uniform(0.3, 0.7)))
    ratio = float(rng.uniform(1.5, 4.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    p = p[int(rng.integers(0, n - 1)):] + p[:int(rng.integers(0, n - 1))]
    p = p * (1.0 + 0.03 * float(rng.normal(0.0, 1.0)))
    p = np.maximum(p, 0.0)
    q = _sanitize_array(p)
    if q is None:
        return _init_uniform(n)
    return q


def _init_gaussian_mixture(n, rng):
    n = max(1, int(n))
    p = np.zeros(n, dtype=float)
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    K = int(rng.integers(2, 7))
    for _ in range(K):
        c = float(rng.uniform(0.0, 1.0))
        s = float(rng.uniform(0.02, 0.28))
        w = float(rng.uniform(0.5, 2.0))
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-12)
    q = _sanitize_array(p)
    if q is None:
        return _init_uniform(n)
    return q


def _init_cosine(n, rng):
    n = max(1, int(n))
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    p = np.ones(n, dtype=float)
    for k in range(1, 9):
        amp = float(rng.uniform(-0.5, 0.5)) / k
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        p += amp * np.cos(2.0 * np.pi * k * t + ph)
    p = np.maximum(p, 0.0)
    q = _sanitize_array(p)
    if q is None:
        return _init_uniform(n)
    return q


def _resample_sequence(p, n_new):
    p = _sanitize_array(p)
    if p is None:
        return None
    n_old = len(p)
    n_new = int(n_new)
    if n_old == n_new:
        return p.copy()
    if n_old == 1:
        return np.ones(n_new, dtype=float) / float(n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) / float(n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) / float(n_new)
    q = np.interp(x_new, x_old, p)
    q = np.maximum(q, 0.0)
    q = _sanitize_array(q)
    return q


def _convolution(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    N = 2 * n - 1
    if n <= 512:
        c = np.convolve(a, a).astype(float)
    else:
        L = _next_pow2(N)
        A = np.fft.rfft(a, L)
        c = np.fft.irfft(A * A, L)[:N]
        c = c.astype(float)
    c = np.where(np.isfinite(c), c, 0.0)
    if c.size > 0 and c.min() < 0.0:
        c = np.maximum(c, 0.0)
    return c


def _score_c(a):
    a = _sanitize_array(a)
    if a is None:
        return float("inf"), None
    c = _convolution(a)
    return float(2.0 * len(a) * c.max()), c


def _pref_suff(c):
    N = len(c)
    if N == 0:
        z = np.zeros(1, dtype=float)
        return 0.0, z, z
    acc = np.maximum.accumulate(c)
    pref = np.empty(N + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = acc

    rev_acc = np.maximum.accumulate(c[::-1])
    suff = np.empty(N + 1, dtype=float)
    suff[N] = 0.0
    suff[N - 1::-1] = rev_acc

    m = float(suff[0])
    return m, pref, suff


def _make_coord_candidates(x0, uniform, expdeltas, rands):
    m = 4 + len(_FACTORS) + len(expdeltas) + len(rands)
    cand = np.empty(m, dtype=float)
    cand[0] = x0
    cand[1] = 0.0
    cand[2] = uniform
    cand[3] = 1.0
    idx = 4
    for f in _FACTORS:
        cand[idx] = x0 * f
        idx += 1
    for e in expdeltas:
        cand[idx] = x0 * e
        idx += 1
    for r in rands:
        cand[idx] = r
        idx += 1
    np.clip(cand, 0.0, 1.0, out=cand)
    return cand


def _eval_coord_trial(a, c, pref, suff, i, x):
    n = len(a)
    x0 = float(a[i])
    b = 1.0 - x0
    if b <= 1e-14:
        return float(suff[0])
    beta = (1.0 - float(x)) / b
    alpha = beta * beta
    coeff = 2.0 * (float(x) * beta - x0 * alpha)
    point = float(x) * float(x) - x0 * x0 * alpha
    outside = max(pref[i], suff[i + n])
    cs = c[i:i + n]
    local = alpha * cs + coeff * a
    local[i] = alpha * cs[i] + point
    lm = float(local.max())
    am = float(alpha * outside)
    return lm if lm > am else am


def _optimize_coordinates(a0, deadline, rng, verbose=False):
    a = _sanitize_array(a0)
    if a is None:
        return None, float("inf")
    n = len(a)
    if n == 0:
        return None, float("inf")

    c = _convolution(a)
    m, pref, suff = _pref_suff(c)
    score = float(2.0 * n * m)
    best = a.copy()
    bestscore = score

    start_t = time.monotonic()
    acc = 0
    no_acc = 0
    sweep = 0
    last_print = 0.0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1.0:
            break

        if bestscore < float("inf") and (score > bestscore * 1.18 or no_acc > 3 * n):
            a = best.copy()
            c = _convolution(a)
            m, pref, suff = _pref_suff(c)
            score = float(2.0 * n * m)
            no_acc = 0
            if bestscore < float("inf") and rng.random() < 0.35:
                a = a * np.exp(rng.normal(0.0, 0.015, n))
                a = _sanitize_array(a)
                if a is None:
                    a = best.copy()
                c = _convolution(a)
                m, pref, suff = _pref_suff(c)
                score = float(2.0 * n * m)

        idx = rng.permutation(n)
        expdeltas = np.exp(rng.normal(0.0, 0.35, _EXP_SIZE))
        rands = rng.random(_RAND_SIZE)

        for i in idx:
            if time.monotonic() >= deadline:
                break

            x0 = float(a[i])
            b = 1.0 - x0
            if b < 1e-14:
                no_acc += 1
                continue

            xs = _make_coord_candidates(x0, 1.0 / float(n), expdeltas, rands)
            if xs.size == 0:
                no_acc += 1
                continue

            beta = (1.0 - xs) / b
            alpha = beta * beta
            coeff = 2.0 * (xs * beta - x0 * alpha)
            point = xs * xs - x0 * x0 * alpha

            cs = c[i:i + n]
            outside = float(max(pref[i], suff[i + n]))

            local = alpha[:, None] * cs[None, :] + coeff[:, None] * a[None, :]
            local[:, i] = alpha * cs[i] + point
            tmax = np.maximum(local.max(axis=1), alpha * outside)
            tmax = np.where(np.isfinite(tmax), tmax, np.inf)

            if tmax.size == 0:
                no_acc += 1
                continue

            bi = int(np.argmin(tmax))
            mbest = float(tmax[bi])

            if mbest < m - 1e-13:
                xb = float(xs[bi])
                if 1e-9 < xb < 1.0 - 1e-9:
                    step = max(1e-5, 0.15 * abs(xb), 1e-4 / float(n))
                    cand2 = np.array([xb - step, xb - 0.5 * step,
                                      xb + 0.5 * step, xb + step], dtype=float)
                    cand2 = cand2[(cand2 > 0.0) & (cand2 < 1.0)]
                    for xv in cand2:
                        mv = _eval_coord_trial(a, c, pref, suff, i, float(xv))
                        if mv < mbest - 1e-14:
                            mbest = float(mv)
                            xb = float(xv)

                if mbest < m - 1e-13:
                    xb = float(min(1.0, max(0.0, xb)))
                    beta_b = (1.0 - xb) / b
                    alpha_b = beta_b * beta_b
                    coeff_b = 2.0 * (xb * beta_b - x0 * alpha_b)
                    point_b = xb * xb - x0 * x0 * alpha_b

                    local_u = alpha_b * cs + coeff_b * a
                    local_u[i] = alpha_b * cs[i] + point_b

                    c *= alpha_b
                    c[i:i + n] = local_u
                    a *= beta_b
                    a[i] = xb

                    s = float(a.sum())
                    if s > 0.0 and np.isfinite(s):
                        if abs(s - 1.0) > 1e-12:
                            a /= s
                            c /= (s * s)
                    else:
                        a = best.copy()
                        c = _convolution(a)
                        m, pref, suff = _pref_suff(c)
                        score = float(2.0 * n * m)
                        continue

                    if not np.all(np.isfinite(c)) or np.min(c) < -1e-12:
                        c = _convolution(a)
                    else:
                        c = np.maximum(c, 0.0)

                    m, pref, suff = _pref_suff(c)
                    score = float(2.0 * n * m)
                    acc += 1
                    no_acc = 0

                    if score < bestscore:
                        best = a.copy()
                        bestscore = score
                        nowp = time.monotonic()
                        if verbose and (nowp - last_print > 30.0 or score < 1.505):
                            print(f"coord n={n} score={score:.6f} t={nowp - start_t:.0f}s")
                            last_print = nowp

                    if acc % 128 == 0:
                        c = _convolution(a)
                        m, pref, suff = _pref_suff(c)
                        score = float(2.0 * n * m)
                        if score < bestscore:
                            best = a.copy()
                            bestscore = score

                    if time.monotonic() > deadline - 1.0:
                        break
                else:
                    no_acc += 1
            else:
                no_acc += 1

        sweep += 1
        if verbose and sweep % 20 == 0:
            print(f"coord sweep={sweep} current={score:.6f} best={bestscore:.6f} "
                  f"t={time.monotonic() - start_t:.0f}s")

    c = _convolution(best)
    bestscore = float(2.0 * n * c.max())
    best = _sanitize_array(best)
    if best is None:
        best = _init_uniform(n)
    return best, bestscore


def _gradient_h(a, w):
    n = len(a)
    N = 2 * n - 1
    L = _next_pow2(N + n - 1)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a[::-1], L)
    conv = np.fft.irfft(W * A, L)[:N + n - 1]
    conv = np.where(np.isfinite(conv), conv, 0.0)
    h = 2.0 * conv[n - 1::-1][:n]
    h = np.where(np.isfinite(h), h, 0.0)
    return np.maximum(h, 0.0)


def _gradient_refine(a0, deadline, rng, beta_start=30.0, beta_max=4000.0, verbose=False):
    a = _sanitize_array(a0)
    if a is None:
        return None, float("inf")
    n = len(a)
    if n == 0:
        return None, float("inf")

    c = _convolution(a)
    m = float(c.max())
    score = float(2.0 * n * m)
    best = a.copy()
    bestscore = score

    start_t = time.monotonic()
    eta = 0.12
    beta = float(beta_start)
    stall = 0
    last_print = 0.0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1.0:
            break

        c = _convolution(a)
        m = float(c.max())
        score = float(2.0 * n * m)
        if score < bestscore:
            best = a.copy()
            bestscore = score

        if bestscore < float("inf") and score > bestscore * 1.25 and stall > 40:
            a = best.copy()
            c = _convolution(a)
            m = float(c.max())
            score = float(2.0 * n * m)
            stall = 0

        z = c - m
        w = np.exp(beta * z)
        w = np.where(np.isfinite(w), w, 0.0)
        sw = float(w.sum())
        if sw <= 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= sw

        h = _gradient_h(a, w)
        hp = h - float(h.mean())
        scale = float(np.max(np.abs(hp)))
        if scale < 1e-14:
            scale = 1e-14
            hp = np.zeros_like(h)

        improved = False
        for f in (1.0, 0.5, 2.0):
            arg = -eta * f * hp / scale
            arg = np.clip(arg, -4.0, 4.0)
            arg = np.where(np.isfinite(arg), arg, 0.0)
            at = a * np.exp(arg)
            at = np.where(np.isfinite(at), at, 1e-18 / float(n))
            at = np.maximum(at, 1e-18 / float(n))
            s = float(at.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            at /= s
            cn = _convolution(at)
            scn = float(2.0 * n * cn.max())
            if scn < score - 1e-12:
                a = at
                c = cn
                m = float(c.max())
                score = scn
                improved = True
                eta = min(1.5, eta * 1.06)
                break

        if improved:
            stall = 0
            if score < bestscore:
                best = a.copy()
                bestscore = score
                nowp = time.monotonic()
                if verbose and (nowp - last_print > 30.0 or score < 1.505):
                    print(f"grad n={n} score={score:.6f} beta={beta:.0f} t={nowp - start_t:.0f}s")
                    last_print = nowp
        else:
            eta = max(1e-4, eta * 0.75)
            stall += 1
            if stall % 25 == 0:
                if bestscore < float("inf") and score > bestscore * 1.05:
                    a = best.copy()
                    c = _convolution(a)
                    m = float(c.max())
                    score = float(2.0 * n * m)
                else:
                    sig = 0.02 + 0.002 * min(stall, 25)
                    at = a * np.exp(rng.normal(0.0, sig, n))
                    at = _sanitize_array(at)
                    if at is not None:
                        a = at
                        c = _convolution(a)
                        m = float(c.max())
                        score = float(2.0 * n * m)
                stall = 0
                eta = 0.12
            if stall > 35:
                beta = max(5.0, beta * 0.6)

        if stall == 0:
            beta = min(float(beta_max), beta * 1.04)
        else:
            beta = min(float(beta_max), beta * 1.01)

    c = _convolution(best)
    bestscore = float(2.0 * n * c.max())
    best = _sanitize_array(best)
    if best is None:
        best = _init_uniform(n)
    return best, bestscore


def _freq_flatten(a, alpha):
    a = _sanitize_array(a)
    if a is None:
        return None
    n = len(a)
    N = 2 * n - 1
    L = _next_pow2(N)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)[:N]
    C = np.where(np.isfinite(C), C, 0.0)
    C = np.maximum(C, 0.0)
    cmax = float(C.max())
    if cmax <= 1e-16:
        return a.copy()
    Cn = C / cmax
    Cnew = cmax * np.power(Cn, 1.0 / alpha)
    Cnew = np.maximum(Cnew, 0.0)
    Anew = np.sqrt(np.fft.rfft(Cnew, L))
    an = np.fft.irfft(Anew, L)[:n]
    an = np.where(np.isfinite(an), an, 0.0)
    neg_frac = float((an < -1e-12).sum()) / float(n)
    if neg_frac > 0.3:
        return None
    an = np.maximum(an, 0.0)
    an = _sanitize_array(an)
    return an


def _freq_phase(a0, deadline, rng, verbose=False):
    a = _sanitize_array(a0)
    if a is None:
        return None, float("inf")
    n = len(a)
    c = _convolution(a)
    score = float(2.0 * n * c.max())
    best = a.copy()
    bestscore = score
    alphas = np.array([1.2, 1.4, 1.7, 2.2, 3.0, 4.5, 7.0, 12.0], dtype=float)
    start_t = time.monotonic()
    no_imp = 0
    last_print = 0.0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1.0:
            break
        order = rng.permutation(len(alphas))
        improved = False
        for idx in order:
            alpha = float(alphas[idx])
            p = _freq_flatten(a, alpha)
            if p is None:
                continue
            c2 = _convolution(p)
            s2 = float(2.0 * n * c2.max())
            if s2 < score - 1e-12:
                a = p
                score = s2
                improved = True
                if score < bestscore:
                    best = a.copy()
                    bestscore = score
                    nowp = time.monotonic()
                    if verbose and (nowp - last_print > 30.0 or score < 1.505):
                        print(f"freq n={n} alpha={alpha:.2f} score={score:.6f} t={nowp - start_t:.0f}s")
                        last_print = nowp
                break

        if not improved:
            no_imp += 1
            if no_imp % 8 == 0:
                if bestscore < float("inf") and score > bestscore * 1.10:
                    a = best.copy()
                    c = _convolution(a)
                    score = float(2.0 * n * c.max())
                else:
                    at = a * np.exp(rng.normal(0.0, 0.03, n))
                    at = _sanitize_array(at)
                    if at is not None:
                        a = at
                        c = _convolution(a)
                        score = float(2.0 * n * c.max())
                no_imp = 0

    c = _convolution(best)
    bestscore = float(2.0 * n * c.max())
    best = _sanitize_array(best)
    if best is None:
        best = _init_uniform(n)
    return best, bestscore


def _block_phase(a0, deadline, rng, verbose=False):
    a = _sanitize_array(a0)
    if a is None:
        return None, float("inf")
    n = len(a)
    c = _convolution(a)
    score = float(2.0 * n * c.max())
    best = a.copy()
    bestscore = score
    factors = (0.35, 0.6, 0.85, 1.15, 1.5, 2.2)
    no_imp = 0
    start_t = time.monotonic()
    last_print = 0.0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1.0:
            break

        if n == 1:
            break

        lo = max(1, n // 512)
        hi = max(lo + 1, n // 8)
        if rng.random() < 0.20:
            bs = int(rng.integers(1, min(5, n) + 1))
        else:
            bs = int(rng.integers(lo, min(hi, n) + 1))
        bs = max(1, min(n, bs))
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            at = a.copy()
            at[start:start + bs] *= float(f)
            at = _sanitize_array(at)
            if at is None:
                continue
            cn = _convolution(at)
            sn = float(2.0 * n * cn.max())
            if sn < score - 1e-12:
                a = at
                c = cn
                score = sn
                improved = True
                if score < bestscore:
                    best = a.copy()
                    bestscore = score
                    nowp = time.monotonic()
                    if verbose and (nowp - last_print > 45.0 or score < 1.505):
                        print(f"block n={n} bs={bs} f={f:.2f} score={score:.6f} "
                              f"t={nowp - start_t:.0f}s")
                        last_print = nowp
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 120 == 0:
                if bestscore < float("inf") and score > bestscore * 1.10:
                    a = best.copy()
                    c = _convolution(a)
                    score = float(2.0 * n * c.max())
                else:
                    at = a * np.exp(rng.normal(0.0, 0.03, n))
                    at = _sanitize_array(at)
                    if at is not None:
                        a = at
                        c = _convolution(a)
                        score = float(2.0 * n * c.max())
                no_imp = 0

    c = _convolution(best)
    bestscore = float(2.0 * n * c.max())
    best = _sanitize_array(best)
    if best is None:
        best = _init_uniform(n)
    return best, bestscore


def _symmetrize(a0):
    a = _sanitize_array(a0)
    if a is None:
        return None, float("inf")
    s = (a + a[::-1]) * 0.5
    s = _sanitize_array(s)
    if s is None:
        return a.copy(), float(2.0 * len(a) * _convolution(a).max())
    return s.copy(), float(2.0 * len(s) * _convolution(s).max())


def propose_candidate(seed=12345, budget_s=2400):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 120.0:
        budget = 2400.0
    deadline = time.monotonic() + budget - 15.0

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    parent = None
    try:
        parent = _sanitize_array(height_sequence_1)
    except Exception:
        parent = None

    if parent is None:
        parent = _init_uniform(2048)

    best = parent.copy()
    bestscore, _ = _score_c(best)
    print(f"initial: n={len(best)} score={bestscore:.6f}")

    if bestscore >= float("inf"):
        best = _init_uniform(2048)
        bestscore, _ = _score_c(best)
        print(f"fallback initial: n={len(best)} score={bestscore:.6f}")

    # ---------------- Primary coordinate search ----------------
    main_len = len(best)
    if main_len > 4500:
        main = _resample_sequence(best, min(4096, max(2048, main_len // 2)))
        if main is None:
            main = best.copy()
    else:
        main = best.copy()

    mscore, _ = _score_c(main)
    if mscore < bestscore:
        best = main.copy()
        bestscore = mscore
        print(f"main candidate improves: n={len(best)} score={bestscore:.6f}")

    if main is not None and len(main) <= 4500:
        coord_alloc = min(1150.0, 0.48 * budget)
        now = time.monotonic()
        end = min(deadline, now + coord_alloc)
        if end - now > 45.0:
            print(f"coordinate search: n={len(main)} for {end - now:.0f}s")
            main, mscore = _optimize_coordinates(main, end, rng, verbose=True)
            if main is not None and np.isfinite(mscore) and mscore < bestscore:
                best = main.copy()
                bestscore = float(mscore)
                print(f"coordinate best: n={len(best)} score={bestscore:.6f}")

    # ---------------- Gradient polish ----------------
    now = time.monotonic()
    grad_alloc = min(420.0, 0.18 * budget)
    end = min(deadline, now + grad_alloc)
    if end - now > 45.0 and len(best) <= 9000:
        gb, gsc = _gradient_refine(best, end, rng, verbose=True)
        if gb is not None and np.isfinite(gsc) and gsc < bestscore:
            best = gb.copy()
            bestscore = float(gsc)
            print(f"gradient best: n={len(best)} score={bestscore:.6f}")

    # ---------------- Frequency flattening ----------------
    now = time.monotonic()
    freq_alloc = min(260.0, 0.12 * budget)
    end = min(deadline, now + freq_alloc)
    if end - now > 30.0:
        fb, fsc = _freq_phase(best, end, rng, verbose=False)
        if fb is not None and np.isfinite(fsc) and fsc < bestscore:
            best = fb.copy()
            bestscore = float(fsc)
            print(f"freq best: n={len(best)} score={bestscore:.6f}")

    # ---------------- Block scaling ----------------
    now = time.monotonic()
    block_alloc = min(220.0, 0.10 * budget)
    end = min(deadline, now + block_alloc)
    if end - now > 30.0:
        bb, bsc = _block_phase(best, end, rng, verbose=False)
        if bb is not None and np.isfinite(bsc) and bsc < bestscore:
            best = bb.copy()
            bestscore = float(bsc)
            print(f"block best: n={len(best)} score={bestscore:.6f}")

    # ---------------- Alternative lengths / restarts ----------------
    alt_lengths = [2048, 4096, 6144, 8192]
    for na in alt_lengths:
        now = time.monotonic()
        if deadline - now < 120.0:
            break
        if na == len(best):
            continue
        if na > 9000:
            continue

        aalt = _resample_sequence(best, na)
        if aalt is None:
            continue
        s0, _ = _score_c(aalt)
        if bestscore < 2.0 and s0 > bestscore + 0.35:
            continue

        t = min(160.0, max(35.0, (deadline - now - 120.0) / 2.0))
        end = min(deadline, now + t)
        if end - now < 20.0:
            end = now + 20.0
        if end <= now:
            break

        aalt, s1 = _gradient_refine(aalt, end, rng, verbose=False)
        if aalt is not None and np.isfinite(s1) and s1 < bestscore:
            best = aalt.copy()
            bestscore = float(s1)
            print(f"alt gradient n={na} score={bestscore:.6f}")

            now = time.monotonic()
            if len(best) <= 4096 and deadline - now > 220.0:
                t2 = min(200.0, deadline - now - 60.0)
                end2 = now + t2
                aalt2, s2 = _optimize_coordinates(aalt, end2, rng, verbose=True)
                if aalt2 is not None and np.isfinite(s2) and s2 < bestscore:
                    best = aalt2.copy()
                    bestscore = float(s2)
                    print(f"alt coordinate n={na} score={bestscore:.6f}")

    # ---------------- Symmetrization may help ----------------
    if len(best) > 1:
        sym, sym_score = _symmetrize(best)
        if sym is not None and np.isfinite(sym_score) and sym_score < bestscore:
            best = sym.copy()
            bestscore = float(sym_score)
            print(f"sym best: n={len(best)} score={bestscore:.6f}")

    # ---------------- Final short polish ----------------
    now = time.monotonic()
    rem = deadline - now
    if rem > 90.0:
        if len(best) <= 4500:
            fb, fsc = _optimize_coordinates(best, deadline - 5.0, rng, verbose=True)
        else:
            fb, fsc = _gradient_refine(best, deadline - 5.0, rng, verbose=True)
        if fb is not None and np.isfinite(fsc) and fsc < bestscore:
            best = fb.copy()
            bestscore = float(fsc)
            print(f"final polish: n={len(best)} score={bestscore:.6f}")

    final = _sanitize_array(best)
    if final is None:
        final = _init_uniform(2048)

    # One last exact local check and optional official evaluator consistency check.
    final_score, _ = _score_c(final)
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            print(f"official evaluator check: {official:.6f}")
    except Exception:
        pass

    out = np.asarray(final, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = _init_uniform(2048)
        s = float(out.sum())
        out = out / s
    else:
        out = out / s

    print(f"FINAL returned: n={len(out)} local_score={float(2.0 * len(out) * _convolution(out).max()):.6f}")
    return [float(x) for x in out]
