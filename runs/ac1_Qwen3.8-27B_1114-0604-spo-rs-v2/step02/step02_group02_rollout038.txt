2. Compute `c = full_convolution(a)`.
3. Compute `p = softmax(beta * c)` (Boltzmann weights).
4. Compute `g[j] = sum_k p[k] * a[k-j]` via FFT (cross-correlation).
5. Gradient: `grad = 4n * g`.
6. Take a step: `a_new = a - eta * grad`.
7. Project onto simplex: `a_new = max(a_new, 0)`, then normalize `a_new /= sum(a_new)`.
8. Repeat.

This is essentially a projected gradient method (or mirror descent if we use a different step).

For the step size, we can use backtracking line search or a fixed schedule.

This is similar to the previous approach but using a smooth approximation (log-sum-exp) instead of the max function. The key advantage is that we get a proper gradient (not just a subgradient) when `beta` is finite.

- `a = softmax(x)` (i.e., `a[i] = exp(x[i] - max(x)) / sum(exp(x[j] - max(x)))`).
- Objective: `F(a) = (2n/beta) * log(sum_k exp(beta * c_k(a)))`.
- Gradient w.r.t. x: `dF/dx_i = sum_j (dF/da_j) * (da_j/dx_i) = sum_j grad_a[j] * a[j] * (delta_{ji} - a[i])`
  `= a[i] * (grad_a[i] - sum_j a[j] * grad_a[j])`
  `= a[i] * (grad_a[i] - mean_weighted_by_a)`

where `grad_a = (4n/beta) * g`.

So: `dF/dx_i = a[i] * ((4n/beta) * g[i] - (4n/beta) * sum_j a[j]*g[j])`
`= (4n/beta) * a[i] * (g[i] - sum_j a[j]*g[j])`

Note that `sum_j a[j]*g[j] = sum_j a[j] * sum_k p[k]*a[k-j] = sum_k p[k] * sum_j a[j]*a[k-j] = sum_k p[k]*c[k]`.

So: `dF/dx_i = (4n/beta) * a[i] * (g[i] - E[p])` where `E[p] = sum_k p[k]*c[k]`.

This is a nice closed form! Let me implement this.

For L-BFGS with n=6121 parameters, each gradient evaluation requires:
1. Softmax: O(n)
2. FFT for convolution: O(n log n)
3. Cross-correlation via FFT: O(n log n)
4. Gradient computation: O(n)

Total: O(n log n) ≈ 6121 * 13 ≈ 80K operations per gradient evaluation. Very fast!

L-BFGS typically converges in O(sqrt(n)) to O(n) iterations for smooth problems. So we might need 100-1000 gradient evaluations, each taking <1ms. Total: <1s per optimization run. That's great!

"""
Step function optimizer using log-sum-exp smoothing with L-BFGS local optimization,
multiple diverse restarts, and coordinate descent polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def _compute_g(p, a):
    """Compute g[m] = sum_k p[k]*a[k-m] for m=0,...,n-1 using FFT.
    Verified: ifft(FFT(p) * conj(FFT(a))) gives g[m].
    """
    n = len(a)
    size = 1 << (3 * n + 2).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = np.fft.irfft(P * np.conj(A), size)
    return g[:n]

def _smooth_obj_and_grad(x, n, beta):
    """Objective and gradient for L-BFGS using softmax parameterization.
    x: unconstrained parameter vector (length n)
    Returns (objective_value, gradient_vector)
    """
    # Softmax
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum()
    
    # Convolution
    c = _full_conv(a)
    
    # Boltzmann weights
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    p = exp_bc / Z
    
    # Objective
    F = 2.0 * n * (bc_max + np.log(Z)) / beta
    
    # Cross-correlation
    g = _compute_g(p, a)
    
    # Gradient: dF/dx_i = (4n/beta) * a[i] * (g[i] - E)
    E = np.dot(p, c)  # Expected value of c under p
    grad = (4.0 * n / beta) * a * (g - E)
    
    return F, grad

def _softmax_to_sequence(x):
    """Convert unconstrained x to a non-negative unit-sum sequence."""
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()

def _optimize_lbfgs(a_init, beta, maxiter=500, tol=1e-10):
    """Run L-BFGS-B to optimize the smooth objective."""
    n = len(a_init)
    # Convert a_init to x (inverse softmax)
    x0 = np.log(a_init + 1e-300)
    
    result = minimize(
        fun=lambda x: _smooth_obj_and_grad(x, n, beta)[0],
        x0=x0,
        jac=lambda x: _smooth_obj_and_grad(x, n, beta)[1],
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8, 'maxfun': maxiter * 10}
    )
    
    a_opt = _softmax_to_sequence(result.x)
    c_opt = _full_conv(a_opt)
    score = 2.0 * n * c_opt.max()
    return a_opt, score, result

def _optimize_adaptive_beta(a_init, betas, budget_per_beta=300, maxiter=300):
    """Run L-BFGS with increasing beta values."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _full_conv(a).max()
    
    for beta in betas:
        t_start = time.monotonic()
        a, score, result = _optimize_lbfgs(a, beta, maxiter=maxiter)
        elapsed = time.monotonic() - t_start
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        print(f"    beta={beta:.0f}, score={score:.8f}, best={best_score:.8f}, time={elapsed:.1f}s")
    
    return best_a, best_score

"""
Step function optimizer using log-sum-exp smoothing with L-BFGS local optimization,
mirror descent with simulated annealing for exploration, diverse restarts,
and coordinate descent polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np
import time
from scipy.optimize import minimize


def _full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _compute_g(p, a):
    """Compute g[m] = sum_k p[k]*a[k-m] for m=0,...,n-1 using FFT."""
    n = len(a)
    size = 1 << (3 * n + 2).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = np.fft.irfft(P * np.conj(A), size)
    return g[:n]


def _softmax(x):
    """Stable softmax."""
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def _smooth_obj_and_grad(x, n, beta):
    """Objective and gradient for L-BFGS using softmax parameterization."""
    a = _softmax(x)
    c = _full_conv(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    p = exp_bc / Z
    F = 2.0 * n * (bc_max + np.log(Z)) / beta
    g = _compute_g(p, a)
    E = np.dot(p, c)
    grad = (4.0 * n / beta) * a * (g - E)
    return F, grad


def _optimize_lbfgs(a_init, beta, maxiter=2000, tol=1e-12):
    """Run L-BFGS-B to optimize the smooth objective from a given starting point."""
    n = len(a_init)
    x0 = np.log(a_init + 1e-300)

    def fun_wrapper(x):
        return _smooth_obj_and_grad(x, n, beta)[0]

    def jac_wrapper(x):
        return _smooth_obj_and_grad(x, n, beta)[1]

    result = minimize(
        fun=fun_wrapper,
        x0=x0,
        jac=jac_wrapper,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-9, 'maxfun': maxiter * 20}
    )

    a_opt = _softmax(result.x)
    c_opt = _full_conv(a_opt)
    score = 2.0 * n * c_opt.max()
    return a_opt, score


def _mirror_descent_step(a, grad, step_size):
    """Mirror descent (exponentiated gradient) step."""
    log_a = np.log(a + 1e-300)
    grad_range = grad.max() - grad.min()
    if grad_range < 1e-15:
        scale = step_size
    else:
        scale = step_size / grad_range
    log_a -= scale * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s


def _gradient_from_smooth(a, n, beta):
    """Compute the smooth gradient dF/da for a non-negative unit-sum sequence."""
    c = _full_conv(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    p = exp_bc / Z
    g = _compute_g(p, a)
    E = np.dot(p, c)
    grad = (4.0 * n / beta) * (g - E)
    return grad


def _optimize_mirror(a_init, deadline, rng, beta_start=1.0, beta_end=300.0):
    """Mirror descent with SA acceptance and adaptive beta/step."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    c = _full_conv(a)
    cur_score = 2.0 * n * c.max()
    best_score = cur_score
    best_a = a.copy()

    beta = beta_start
    step_size = 1.0
    T = cur_score * 0.03
    T_min = cur_score * 1e-14
    last_improve = time.monotonic()
    total_iters = 0

    while time.monotonic() < deadline:
        c = _full_conv(a)
        cur_score = 2.0 * n * c.max()

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()

        stuck = time.monotonic() - last_improve

        if stuck > 10.0:
            T = cur_score * 0.1
            step_size = max(step_size, 0.5)
            mag = min(0.03, 0.003 + 0.002 * (stuck / 30.0))
            a = a + mag / np.sqrt(n) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = time.monotonic()
            T = cur_score * 0.05
        else:
            T = max(T_min, T * 0.999)
            step_size *= 0.9999

        # Gradually increase beta
        beta = min(beta_end, beta * 1.002)

        grad = _gradient_from_smooth(a, n, beta)
        a_new = _mirror_descent_step(a, grad, step_size)

        c_new = _full_conv(a_new)
        new_score = 2.0 * n * c_new.max()

        delta = new_score - cur_score
        if delta <= 0:
            a = a_new
            step_size = min(step_size * 1.01, 3.0)
        elif T > 1e-30:
            if np.log(max(rng.uniform(), 1e-300)) < -delta / (T * n):
                a = a_new
            else:
                step_size *= 0.9
        else:
            step_size *= 0.85

        total_iters += 1

        if total_iters % 5000 == 0:
            print(f"    MD iter {total_iters}: score={cur_score:.8f}, best={best_score:.8f}, "
                  f"beta={beta:.1f}, step={step_size:.4f}")

    c = _full_conv(a)
    if 2.0 * n * c.max() < best_score:
        best_score = 2.0 * n * c.max()
        best_a = a.copy()

    return best_a, best_score


def _coordinate_polish(a, deadline, rng):
    """Coordinate descent polish: try redistributing mass between pairs."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    c = _full_conv(a)
    cur_score = 2.0 * n * c.max()
    best_score = cur_score
    best_a = a.copy()
    passes = 0

    while time.monotonic() < deadline:
        c = _full_conv(a)
        cur_score = 2.0 * n * c.max()

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        idx_range = np.arange(lo, hi)
        contrib[idx_range] = 2.0 * a[k_star - idx_range]

        improved = False

        # Try moving mass from high-contribution to low-contribution positions
        donor = int(np.argmax(contrib))
        low_candidates = np.argsort(contrib)[:max(5, n // 100)]

        for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
            for recv in low_candidates[:10]:
                if recv == donor:
                    continue
                amt = a[donor] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[donor] -= amt
                a_try[recv] += amt
                a_try /= a_try.sum()
                c_try = _full_conv(a_try)
                if 2.0 * n * c_try.max() < cur_score - 1e-14:
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        # Random pair moves
        if not improved:
            for _ in range(50):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.005, 0.01, 0.03, 0.08])
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                if a_try[i] < 0:
                    continue
                a_try /= a_try.sum()
                c_try = _full_conv(a_try)
                if 2.0 * n * c_try.max() < cur_score - 1e-14:
                    a = a_try
                    break

        # Small global perturbation
        if not improved and passes % 100 == 50:
            a_try = a + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            a_try = np.maximum(a_try, 0.0)
            a_try /= a_try.sum()
            c_try = _full_conv(a_try)
            if 2.0 * n * c_try.max() < cur_score - 1e-14:
                a = a_try

        passes += 1

    c = _full_conv(a)
    if 2.0 * n * c.max() < best_score:
        best_score = 2.0 * n * c.max()
        best_a = a.copy()

    return best_a, best_score


def _generate_starts(a_init, n, rng, count=60):
    """Generate diverse starting points."""
    starts = [a_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    norm = np.sqrt(n)

    # Linear tilts
    for t in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, -0.3, -0.5, -0.8, -1.0]:
        s = 1.0 + t * (2.0 * j / (n - 1) - 1.0)
        s = np.maximum(s, 0.01)
        starts.append(s)

    # Quadratic shapes
    for t in [0.3, 0.5, 1.0, 2.0, -0.3, -0.5, -1.0]:
        x = 2.0 * j / (n - 1) - 1.0
        s = 1.0 + t * (x * x - 1.0 / 3.0)
        s = np.maximum(s, 0.01)
        starts.append(s)

    # Exponential ramps
    for r in [0.5, 1.0, 2.0, 3.0, 5.0, -0.5, -1.0, -2.0]:
        s = np.exp(r * j / (n - 1))
        starts.append(s)

    # Sine modulations
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.3, 0.6, 0.9]:
            for phase in [0, np.pi / 2, np.pi]:
                s = 1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase)
                s = np.maximum(s, 0.01)
                starts.append(s)
                if len(starts) >= count:
                    break
            if len(starts) >= count:
                break
        if len(starts) >= count:
            break

    # Bimodal
    for pos in [0.2, 0.25, 0.3, 0.35, 0.4]:
        for width in [0.05, 0.08, 0.12, 0.2]:
            s = (np.exp(-((j - pos * n) / (width * n)) ** 2) +
                 np.exp(-((j - (1 - pos) * n) / (width * n)) ** 2) + 0.01)
            starts.append(s)
            if len(starts) >= count:
                break
        if len(starts) >= count:
            break

    # U-shaped
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        x = (j - center) / center
        starts.append(1.0 + alpha * x * x)

    # Random starts
    while len(starts) < count:
        r = rng.random()
        if r < 0.25:
            starts.append(rng.exponential(1.0, n) + 0.01)
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.7:
            starts.append(np.ones(n) + 0.5 * rng.standard_normal(n))
            starts[-1] = np.maximum(starts[-1], 0.01)
        elif r < 0.85:
            # Block structure
            s = np.ones(n)
            block = n // 4
            s[:block] *= 0.3
            s[-block:] *= 0.3
            starts.append(s + 0.01)
        else:
            # Power law
            p = rng.uniform(-1, 1)
            s = (j + 1) ** p
            starts.append(np.maximum(s, 0.01))

    # Normalize all
    normed = []
    for s in starts:
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss < 1e-10:
            s = np.ones(n) / n
        else:
            s = s / ss
        normed.append(s)

    return normed

    for idx in range(num_short):
        remaining = opt_deadline - time.monotonic()
        if remaining < 10:
            break

        peak0, a0 = scored[idx]
        t_start = time.monotonic()
        a_opt, score = _optimize_lbfgs(a0, beta=30.0, maxiter=500)

        elapsed = time.monotonic() - t_start
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Start {idx+1}: improved to {best_score:.8f} ({elapsed:.1f}s)")
        else:
            print(f"  Start {idx+1}: {score:.8f} (no improvement) ({elapsed:.1f}s)")

    # Phase 2: L-BFGS with increasing beta on top candidates
    print(f"\n--- Phase 2: Adaptive beta L-BFGS ---")
    opt2_deadline = deadline - budget_s * 0.20

    # Collect candidates: best so far + top starts that improved
    candidates = [best_a.copy()]
    for idx in range(min(5, len(scored))):
        if scored[idx][1] is not best_a:
            candidates.append(scored[idx][1].copy())

    num_phase2 = min(5, len(candidates))
    time_per_p2 = (opt2_deadline - time.monotonic()) / num_phase2 - 5.0
    betas = [30, 60, 120, 250, 500]

    for ci in range(num_phase2):
        remaining = opt2_deadline - time.monotonic()
        if remaining < 15:
            break

        a_cur = candidates[ci].copy()
        cur_score = 2.0 * n * _full_conv(a_cur).max()
        t_start = time.monotonic()
        print(f"\n  Phase 2 candidate {ci+1} (score={cur_score:.8f}):")

        for beta in betas:
            t_rem = opt2_deadline - time.monotonic()
            if t_rem < 5:
                break
            maxiter = min(2000, int(t_rem * 50))
            if maxiter < 10:
                break
            a_opt, score = _optimize_lbfgs(a_cur, beta=beta, maxiter=maxiter)
            if score < cur_score - 1e-12:
                a_cur = a_opt
                cur_score = score
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"    beta={beta}: NEW BEST {best_score:.8f}")
                else:
                    print(f"    beta={beta}: {score:.8f}")
            else:
                print(f"    beta={beta}: no improvement ({score:.8f})")
                if beta >= 120:
                    # Diminishing returns, move on
                    break

    # Phase 3: Mirror descent with SA on the best solution
    print(f"\n--- Phase 3: Mirror descent + SA ---")
    md_deadline = deadline - budget_s * 0.15
    if time.monotonic() < md_deadline:
        t_start = time.monotonic()
        a_md, score_md = _optimize_mirror(best_a, md_deadline, rng,
                                           beta_start=10.0, beta_end=200.0)
        elapsed = time.monotonic() - t_start
        if score_md < best_score:
            best_score = score_md
            best_a = a_md.copy()
            print(f"  MD improved to {best_score:.8f} ({elapsed:.1f}s)")
        else:
            print(f"  MD best: {score_md:.8f} (no improvement, {elapsed:.1f}s)")

    # Phase 4: Coordinate descent polish
    print(f"\n--- Phase 4: Coordinate descent polish ---")
    polish_deadline = deadline - 10.0
    t_start = time.monotonic()
    a_pol, score_pol = _coordinate_polish(best_a, polish_deadline, rng)
    elapsed = time.monotonic() - t_start
    if score_pol < best_score:
        best_score = score_pol
        best_a = a_pol.copy()
        print(f"  Polish improved to {best_score:.8f} ({elapsed:.1f}s)")
    else:
        print(f"  Polish best: {score_pol:.8f} (no improvement, {elapsed:.1f}s)")

    # Final L-BFGS polish if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Final L-BFGS polish ({remaining:.0f}s left) ---")
        a_fin, score_fin = _optimize_lbfgs(best_a, beta=500, maxiter=int(remaining * 100))
        if score_fin < best_score:
            best_score = score_fin
            best_a = a_fin.copy()
            print(f"  Final L-BFGS: {best_score:.8f}")

    # Verify with official evaluator
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nOfficial evaluation: {final_score:.8f}")

    if not np.isfinite(final_score):
        best_a = np.ones(n)
        final_score = evaluate_sequence(best_a.tolist())
        print(f"Fallback uniform: {final_score:.8f}")

    best_score = final_score
    print(f"=== FINAL SCORE: {best_score:.8f} ===")
    return [float(x) for x in best_a]
