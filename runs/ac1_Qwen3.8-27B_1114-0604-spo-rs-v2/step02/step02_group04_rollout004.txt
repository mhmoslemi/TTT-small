```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def _score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    n = a.size
    # FFT convolution
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)

def _solve_lp(fixed_a, n):
    """
    Given fixed_a (length n), find g (length n) minimizing M
    subject to (fixed_a * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M).
    """
    # Build constraint matrix
    # For each k = 0, ..., 2n-2:
    #   sum_j A[k,j] * g[j] <= M
    # where A[k,j] = fixed_a[k-j] if 0 <= k-j < n else 0
    
    # A is (2n-1) x n
    # Instead of building the full matrix, use the structure
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Constraints: A @ g - M <= 0 for each k
    #              sum(g) = 1
    #              g >= 0, M free (or M >= 0)
    
    # In linprog form: minimize c @ x
    # c = [0, 0, ..., 0, 1] (minimize M)
    # A_ub @ x <= b_ub: [A, -I_n x 1] @ [g; M] <= 0
    # A_eq @ x = b_eq: [ones, 0] @ [g; M] = 1
    # bounds: g >= 0, M free
    
    # But building the full (2n-1) x (n+1) matrix is O(n^2) memory
    # For n=1024, that's about 2047*1025*8 = 16.8 MB, which is fine.
    
    A_mat = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A_mat[k, j] = fixed_a[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A_mat[k, j] = fixed_a[k - j]
    
    # Full constraint matrix: [A_mat, -e] where e is the last basis vector
    # Actually: A_mat @ g - M <= 0
    # So: [A_mat | -1] @ [g; M] <= 0
    
    A_ub = np.hstack([A_mat, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n - 1)
    
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.ones(1)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, M free
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        # Fallback: return uniform
        return np.ones(n) / n, _score(np.ones(n) / n) / (2.0 * n)
    
    g = result.x[:n]
```

Looking at this problem, I need to minimize `2*n*max(convolution)/(sum)^2` for a non-negative sequence. The key insight is that this is a scale-invariant optimization where we need to minimize the peak of the auto-convolution relative to its total sum.

My approach uses **alternating linear programming** - a fundamentally different method from the gradient-based approach used before. The key observation is that for fixed `a`, finding the best `g` to minimize `max(a*g)` subject to `g>=0, sum(g)=1` is a **linear program** (the convolution is linear in `g`). This allows exact solving of each subproblem rather than approximate gradient steps.

```python
"""
Alternating LP optimization for low-peak non-negative step functions.

Core idea: The objective 2n*max(a*a)/(sum a)^2 is scale-invariant. Fixing sum(a)=1,
we minimize max(a*a). The alternating method solves:
  - min_g max(a*g) s.t. g>=0, sum(g)=1  [LP in g for fixed a]
  - min_f max(f*g) s.t. f>=0, sum(f)=1  [LP in f for fixed g]
Each subproblem is a linear program (convolution is linear in the free variable).
Combined with gradient polishing and multiple restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    n = a.size
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _build_conv_matrix(a, n):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    a = np.asarray(a, dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = np.arange(lo, hi)
            A[k, lo:hi] = a[k - idx]
    return A


_cache = {}

def _solve_lp(fixed, n):
    """Solve: min M s.t. fixed*g[k] <= M for all k, sum(g)=1, g>=0. Returns optimal g."""
    key = (n, np.round(fixed, 10).tobytes())
    if key in _cache:
        return _cache[key].copy()
    
    A_mat = _build_conv_matrix(fixed, n)
    
    # LP variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    # Inequality: A_mat @ g - M <= 0 for each of (2n-1) rows
    # Equality: sum(g) = 1
    # Bounds: g >= 0, M free
    
    n_vars = n + 1
    n_ineq = 2 * n - 1
    
    A_ub = np.zeros((n_ineq, n_vars))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_ineq)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
    
    if not res.success:
        g = np.ones(n) / n
    else:
        g = res.x[:n]
        g = np.clip(g, 0.0, None)
        s = g.sum()
        if s < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / s
    
    _cache.clear()
    _cache[key] = g.copy()
    return g.copy()

def _gradient_polish(a, deadline, verbose=True):
    """Gradient-based polishing using log-sum-exp surrogate."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = _score(a)
    
    T = 20.0
    lr = 0.3
    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(777)
    
    while time.monotonic() < deadline:
        it += 1
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Log-sum-exp weights
        c_max = c.max()
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / exp_c.sum()
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (where valid)
        # This is a convolution-like operation
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[:N], N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        grad = 2.0 * xcorr[:n]
        
        # Project to simplex tangent
        grad_proj = grad - np.mean(grad)
        
        # Update (multiplicative for positivity)
        update = -lr * grad_proj * a.mean()
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Anneal temperature and lr
        T = T * 1.002
        lr = lr * 0.9999
        
        if it % 50 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                if verbose and sc < 1.52:
                    print(f"    Grad iter {it}: score={sc:.8f}, T={T:.1f}", flush=True)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = np.clip(a, 0, None)
                a = a / a.sum()
                patience = max(1000, int(patience * 1.2))
                lr = max(0.05, lr * 0.7)
                T = max(2.0, T * 0.8)
    
    return best_a, best_score


def _make_sine(n, amp, phase=0.3):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_random(n, seed):
    rng = np.random.RandomState(seed)
    a = np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def _make_ramp(n, asym=1.0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + asym * i / n
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining alternating LP with gradient polishing.
    
    Phase 1: Generate diverse initializations and evaluate.
    Phase 2: Run alternating LP from best initializations.
    Phase 3: Gradient-based polishing.
    Phase 4: Final validation.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + budget_s - 20.0
    
    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s", flush=True)
    
    # Phase 1: Initializations
    print("[Phase 1] Generating initializations...", flush=True)
    
    n_default = 1024
    candidates = []
    
    # Try to use provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        if hs.size > 0 and hs.sum() > 0.01:
            hs = hs / hs.sum()
            sc = _score(hs)
            candidates.append(("provided", hs, sc))
            print(f"  provided: n={hs.size}, score={sc:.8f}", flush=True)
    except NameError:
        pass
    except Exception as e:
        print(f"  Warning: could not load height_sequence_1: {e}", flush=True)
    
    # Sinusoidal inits
    for n in [1024, 2048]:
        for amp in [0.6, 0.8, 1.0, 1.2]:
            for phase in [0.1, 0.3, 0.5]:
                a = _make_sine(n, amp, phase)
                sc = _score(a)
                candidates.append((f"sine_n{n}_a{amp}_p{phase}", a, sc))
    
    # Random inits
    for n in [1024, 2048]:
        for s in [1, 2, 3]:
            a = _make_random(n, seed + s)
            sc = _score(a)
            candidates.append((f"rand_n{n}_s{s}", a, sc))
    
    # Ramp inits
    for n in [1024]:
        for asym in [0.5, 1.0, 2.0]:
            a = _make_ramp(n, asym)
            sc = _score(a)
            candidates.append((f"ramp_n{n}_a{asym}", a, sc))
    
    candidates.sort(key=lambda x: x[2])
    print(f"  Best init: {candidates[0][0]}, score={candidates[0][2]:.8f}", flush=True)
    print(f"  Second: {candidates[1][0]}, score={candidates[1][2]:.8f}", flush=True)
    
    # Phase 2: Alternating LP
    # Run LP from top-3 candidates, split time
    print("[Phase 2] Alternating LP optimization...", flush=True)
    
    best_lp_a = candidates[0][1].copy()
    best_lp_score = candidates[0][2]
    
    # Allocate time: 60% to best init, 20% to second, 20% to third
    d_pct = hard_deadline - t0
    time_per_init = d_pct * 0.6
    d2_best = t0 + time_per_init
    
    print(f"  Running LP on best init (budget {time_per_init:.0f}s)...", flush=True)
    a_lp, s_lp = _alternating_lp(candidates[0][1], d2_best, verbose=True)
    if s_lp < best_lp_score:
        best_lp_score = s_lp
        best_lp_a = a_lp
    
    # Run on second-best if time allows
    remaining = hard_deadline - time.monotonic()
    if remaining > 60 and len(candidates) > 1:
        d2_second = time.monotonic() + min(remaining * 0.4, 300)
        print(f"  Running LP on 2nd init (budget {d2_second - time.monotonic():.0f}s)...", flush=True)
        a_lp2, s_lp2 = _alternating_lp(candidates[1][1], d2_second, verbose=False)
        if s_lp2 < best_lp_score:
            best_lp_score = s_lp2
            best_lp_a = a_lp2
            print(f"    New best from LP: {s_lp2:.8f}", flush=True)
    
    # Run on third-best if time allows
    remaining = hard_deadline - time.monotonic()
    if remaining > 60 and len(candidates) > 2:
        d2_third = time.monotonic() + min(remaining * 0.3, 200)
        print(f"  Running LP on 3rd init (budget {d2_third - time.monotonic():.0f}s)...", flush=True)
        a_lp3, s_lp3 = _alternating_lp(candidates[2][1], d2_third, verbose=False)
        if s_lp3 < best_lp_score:
            best_lp_score = s_lp3
            best_lp_a = a_lp3
            print(f"    New best from LP: {s_lp3:.8f}", flush=True)
    
    print(f"  Best after LP: {best_lp_score:.8f}", flush=True)
    
    # Phase 3: Gradient polishing
    remaining = hard_deadline - time.monotonic()
    if remaining > 30:
        print(f"[Phase 3] Gradient polishing (budget {remaining:.0f}s)...", flush=True)
        d3 = hard_deadline - 30
        a_pol, s_pol = _gradient_polish(best_lp_a, d3, verbose=True)
        if s_pol < best_lp_score:
            best_lp_score = s_pol
            best_lp_a = a_pol
            print(f"  Polished score: {s_pol:.8f}", flush=True)
    else:
        print("[Phase 3] Skipped (insufficient time)", flush=True)
    
    # Phase 4: Additional refinements with different blending
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        print(f"[Phase 4] Fine-tuning LP (budget {remaining:.0f}s)...", flush=True)
        d4 = hard_deadline - 15
        # Run LP with more aggressive blending from current best
        a_fin, s_fin = _alternating_lp(best_lp_a, d4, blend_start=0.7, blend_end=0.95, verbose=False)
        if s_fin < best_lp_score:
            best_lp_score = s_fin
            best_lp_a = a_fin
            print(f"  Fine-tuned score: {s_fin:.8f}", flush=True)
    
    # Phase 5: Validate and return
    print("[Phase 5] Validating...", flush=True)
    
    final_arr = np.clip(best_lp_a, 0.0, 1000.0)
    final_list = [float(x) for x in final_arr]
    
    try:
        actual = float(evaluate_sequence(final_list))
        print(f"  Final actual score: {actual:.8f}", flush=True)
    except Exception as e:
        print(f"  Validation error: {e}, using internal score", flush=True)
        actual = best_lp_score
    
    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {actual:.8f}", flush=True)
    
    return final_list
```
