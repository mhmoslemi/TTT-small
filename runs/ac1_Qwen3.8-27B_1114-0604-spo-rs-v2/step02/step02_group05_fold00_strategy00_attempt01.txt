def _next_pow2(x):
    """Smallest power of two >= x."""
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(p, q=None):
    """Fast convolution using FFT."""
    if q is None:
        return np.convolve(p, p, mode='full')
    n_out = len(p) + len(q) - 1
    if n_out == 0:
        return np.array([0.0])
    # Use FFT for efficiency
    p_fft = np.fft.rfft(p.astype(complex))
    q_fft = np.fft.rfft(q.astype(complex))
    conv = np.fft.irfft(p_fft * q_fft, n_out).astype(float)
    # Return only the non-circular part
    return conv
