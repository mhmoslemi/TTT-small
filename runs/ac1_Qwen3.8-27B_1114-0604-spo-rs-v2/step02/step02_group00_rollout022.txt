```python
"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:
  1. High-beta L-BFGS from parent (quick baseline improvement)
  2. Multi-length exploration (8192, 16384) with L-BFGS continuation
  3. Mirror descent / multiplicative weights for non-smooth refinement
  4. Multi-start perturbation with L-BFGS
  5. Final subgradient polish
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, ftol=1e-15, gtol=1e-13):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mirror_descent_step(a, eta):
    """
    One mirror descent step on the simplex.
    Minimizes max_k (a*a)[k] using multiplicative weights.
    Returns (new_a, score_of_new_a).
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Subgradient: sg[j] = 2*a[k_star - j] if 0 <= k_star - j < n
    sg = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    
    # Mirror descent update: a_new[j] = a[j] * exp(-eta * sg[j]) / Z
    # Since sg >= 0 and a >= 0, no overflow.
    log_factor = -eta * sg
    # For numerical stability, shift by max
    max_lf = float(np.max(log_factor[a > 0])) if np.any(a > 0) else 0.0
    weights = np.zeros(n)
    mask = a > 0
    weights[mask] = a[mask] * np.exp(log_factor[mask] - max_lf)
    Z = float(weights.sum())
    if Z < 1e-300:
        return a.copy(), compute_score(a)
    a_new = weights / Z
    return a_new, compute_score(a_new)


def run_mirror_descent(a0, max_iters=100000, eta0=0.1, reheat_every=5000, reheat_factor=1.3):
    """
    Run mirror descent with adaptive step size and periodic reheat.
    Returns best (score, a) found.
    """
    n = len(a0)
    a = a0.copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()  # Normalize to simplex
    
    best_sc = compute_score(a)
    best_a = a.copy()
    eta = eta0
    
    for it in range(max_iters):
        a_new, sc = mirror_descent_step(a, eta)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            # Slowly increase eta if making progress
            a = a_new
            eta = min(eta * 1.001, 1.0)
        else:
            # No improvement: try smaller step
            a = a_new  # Still move (for exploration)
            eta *= 0.95  # Reduce step
        
        # Reheat periodically
        if (it + 1) % reheat_every == 0:
            eta = min(eta * reheat_factor, 1.0)
        
        # Check convergence
        if it % 1000 == 0 and it > 0:
            pass  # Could add early stopping here
    
    return best_sc, best_a
```

```python
"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:
  1. High-beta L-BFGS from parent for quick baseline improvement
  2. Multi-length exploration (8192, 16384) with LSE continuation
  3. Mirror descent (multiplicative weights) for non-smooth refinement
  4. LSE-smoothed mirror descent for intermediate smoothness
  5. Multi-start perturbation with L-BFGS
  6. Final subgradient polish and verification
"""

import numpy as np
from scipy.optimize import minimize
import time


# ---------------------------------------------------------------------------
# Core numerical helpers
# ---------------------------------------------------------------------------

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: cross-correlation of a and p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def get_exact_subgradient(a, k_star):
    """Exact subgradient of (a*a)[k_star] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    return sg


def lse_subgradient(a, beta):
    """LSE-smoothed subgradient (same as LSE gradient without normalization)."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    return 2.0 * conv_res


def mirror_step(a, sg, eta):
    """One mirror descent (multiplicative weights) step on the simplex."""
    n = len(a)
    # a_new[j] = a[j] * exp(-eta * sg[j]) / Z
    # sg >= 0, a >= 0, so no overflow
    log_factor = -eta * sg
    mask = a > 0
    if not np.any(mask):
        return a.copy()
    max_lf = float(np.max(log_factor[mask]))
    weights = np.zeros(n)
    weights[mask] = a[mask] * np.exp(log_factor[mask] - max_lf)
    Z = float(weights.sum())
    if Z < 1e-300:
        return a.copy()
    return weights / Z


def run_mirror_descent(a0, max_iters=50000, eta0=0.5, use_lse=False, lse_beta=200):
    """
    Mirror descent with adaptive step size and reheat.
    If use_lse, uses LSE-smoothed subgradient instead of exact.
    """
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    best_sc = compute_score(a)
    best_a = a.copy()
    eta = eta0
    no_improve_count = 0

    for it in range(max_iters):
        if use_lse:
            sg = lse_subgradient(a, lse_beta)
        else:
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            sg = get_exact_subgradient(a, k_star)

        # Normalize subgradient for consistent step sizing
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break
        sg_unit = sg / sg_norm

        a_new = mirror_step(a, sg_unit, eta)

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        a = a_new
        # Adaptive step: reduce if not improving, slowly grow if improving
        if no_improve_count > 200:
            eta *= 0.5
            no_improve_count = 0
        elif no_improve_count == 0:
            eta = min(eta * 1.005, 2.0)

    return best_sc, best_a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    # Resampled parent
    if parent is not None and len(parent) > 1:
        inits.append(resample_sequence(parent, n).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    nb = 8
    per = max(1, n // nb)
    vals = 0.2 + 0.6 * rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    inits.append(a[:n].copy())
    return inits


def exact_subgrad_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg_M = get_exact_subgradient(a, k_star)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Small additional step for stability
        a -= 0.0005 * grad / gn
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a
```
