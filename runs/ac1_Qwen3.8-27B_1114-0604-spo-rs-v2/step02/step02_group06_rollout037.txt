import time

def _fft_conv(a, L=None):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if L is None:
        need = 2 * n - 1
        L = 1
        while L < need:
            L *= 2
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2*n-1]

def _objective(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_conv(a)
    return 2.0 * n * c.max()

"""
Optimization of non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient-based methods (smoothed max), multiplicative updates, threshold
penalties, multi-resolution search, and block-level local refinement.
"""

def _init_rng(seed=42):

    _RNG = np.random.default_rng(seed)

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_convolve(a, L=None):
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2 * n - 1], L

def _score(a):
    n = len(a)
    c, _ = _fft_convolve(a)
    return 2.0 * n * float(c.max())

def _gradient_smoothed(a, beta, top_k=5, L=None):
    """Compute approximate gradient of 2n * max(c) using top-k peaks."""
    n = len(a)
    c, L = _fft_convolve(a, L)
    mc = float(c.max())
    
    # Use top_k peaks for a smoother gradient
    if top_k > 1:
        # Softmax weights over all peaks
        shift = mc
        exp_bc = np.exp(beta * (c - shift))
        w = exp_bc / exp_bc.sum()
    else:
        # Subgradient at argmax
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    
    # grad[i] = 4*n * sum_k w[k] * a[k-i]
    # Compute using cross-correlation via FFT
    # r[m] = sum_k w[k]*a[k+m], computed as ifft(fft(a)*conj(fft(w)))
    L2 = _next_pow2(3 * n - 2)
    Wf = np.fft.rfft(w, L2)
    Af = np.fft.rfft(a, L2)
    r = np.fft.irfft(Af * np.conjugate(Wf), L2)
    # g[i] = r[-i mod L2] = r[L2-i]
    grad = 4.0 * n * r[L2 - np.arange(n)]
    # Wrap: r[L2-i mod L2]
    # Actually r is length L2, indices 0..L2-1
    # g[i] = r[(L2-i) % L2]
    idx = (L2 - np.arange(n)) % L2
    grad = 4.0 * n * r[idx]
    return grad, mc, c

def _project_simplex(a, min_val=0.0):
    """Project to {x >= min_val, sum x = 1}."""
    a = np.maximum(a, min_val)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s

def _grad_descent_step(a, grad, lr, min_val=1e-16):
    """Take a gradient step and project."""
    a_new = a - lr * grad
    return _project_simplex(a_new, min_val)

def _multiplicative_update(a, c, threshold, L):
    """Multiply a by exp(-alpha * violation) where violation = max(0, c - threshold)."""
    n = len(a)
    viol = np.maximum(0.0, c - threshold)
    max_viol = viol.max()
    if max_viol < 1e-30:
        return a
    # Compute the "gradient direction" from violations
    L2 = _next_pow2(3 * n - 2)
    Vf = np.fft.rfft(viol, L2)
    Af = np.fft.rfft(a, L2)
    r = np.fft.irfft(Af * np.conjugate(Vf), L2)
    idx = (L2 - np.arange(n)) % L2
    direction = r[idx]
    direction = np.maximum(direction, 0.0)
    return direction

def _resample(a, m):
    """Resample sequence to length m."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()

def _initializations(n, k=4):

    a = np.ones(n) / n

    inits.append(a / a.sum())
    
    # Bell-shaped
    sigma = 0.3
    a = np.exp(-((t - 0.5) / sigma) ** 2)
    inits.append(a / a.sum())
    
    # Sine-modulated
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t)
    a = np.maximum(a, 0.1)
    inits.append(a / a.sum())
    
    # Random smooth
    from numpy.polynomial import polynomial as P
    coeffs = _RNG.normal(0, 1, min(8, n)) / 3
    a = np.abs(P.polyval(t, coeffs)) + 0.1
    inits.append(a / a.sum())
    
    # Triangular (peaked in middle)
    a = 1.0 - np.abs(2 * t - 1)
    inits.append(a / a.sum())
    
    # Random step function
    n_steps = int(_RNG.integers(3, max(4, n // 100)))
    edges = np.sort(_RNG.uniform(0, 1, n_steps - 1))
    blocks = np.concatenate([[0], edges, [1]])
    a = np.zeros(n)
    for i in range(n_steps):
        s = max(0, int(blocks[i] * n))
        e = min(n, max(s+1, int(blocks[i+1] * n)))
        a[s:e] = _RNG.lognormal(0, 0.5)
    a = np.maximum(a, 1e-6)
    inits.append(a / a.sum())
    
    return inits
