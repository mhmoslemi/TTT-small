"""
Soft-max annealing optimizer with frequency-domain initialization for
minimizing 2n*max(conv(a,a))/(sum a)^2 over non-negative sequences.

Key improvements over previous approach:
1. Smooth soft-max gradient with temperature annealing (handles multiple peaks)
2. Frequency-domain IFT initialization for diverse starting points
3. Adaptive learning rate with cosine scheduling
4. Multi-phase optimization: explore -> refine -> polish
5. Constraint-aware restart strategy
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score_n(n, C, S):
    """Objective value given n, convolution array C, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(C)) / (S * S)


def soft_max_grad(a, T):
    """
    Compute gradient of soft_max(conv(a,a)) and the C array.
    Returns (gradient, C, m_value, p_weights).
    """
    n = len(a)
    M_len = 2 * n - 1
    C = fast_self_conv(a)

    # Stable softmax
    C_max = float(C.max())
    exp_C = np.exp(T * (C - C_max))
    exp_sum = float(exp_C.sum())
    if exp_sum < 1e-300:
        p = np.zeros(M_len)
        p[int(np.argmax(C))] = 1.0
        m_val = C_max
    else:
        p = exp_C / exp_sum
        m_val = C_max + np.log(exp_sum) / T

    # Compute dm/da via FFT
    # dm/da[i] = sum_k p[k]*(a[k-i] if valid + a[i-k] if valid)
    # = R1[i] + R2[i]
    # R1[i] = sum_j a[j]*p[j+i] = cross-correlation(p, a)[i]
    # R2[i] = sum_j a[j]*p[i-j] = convolution(a, p)[i]
    L = 1 << (3 * n - 2).bit_length()
    A_ft = np.fft.rfft(a, L)
    P_ft = np.fft.rfft(p, L)

    # R2 = ifft(fft(a)*fft(p)) = conv(a,p)
    R2 = np.fft.irfft(A_ft * P_ft, L)[:n]
    # R1 = ifft(fft(p)*conj(fft(a))) = xcorr(p,a)
    R1 = np.fft.irfft(P_ft * np.conj(A_ft), L)[:n]

    dm_da = R1 + R2
    return dm_da, C, m_val, p


def full_gradient(a, T):
    """
    Full gradient of f(a) = 2n * soft_max(conv(a,a)) / (sum a)^2
    """
    n = len(a)
    S = float(a.sum())
    dm_da, C, m_val, p = soft_max_grad(a, T)
    # f = 2n * m / S^2
    # df/da[i] = 2n * (dm_da[i] * S^2 - m * 2*S) / S^4
    #          = 2n * (dm_da[i]/S - 2*m/S^2)
    grad = (2.0 * n / S) * (dm_da - 2.0 * m_val / S)
    return grad, C, m_val


def adam_step(a, g, m_buf, v_buf, step, lr):
    """In-place Adam step."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf[:] = b1 * m_buf + (1.0 - b1) * g
    v_buf[:] = b2 * v_buf + (1.0 - b2) * g * g
    t = step
    m_hat = m_buf / (1.0 - b1 ** t)
    v_hat = v_buf / (1.0 - b2 ** t)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)


def init_ift_flat(n, rng, n_iter=200, L_target=None):
    """
    Initialize via IFT targeting flat power spectrum.
    Iteratively: FFT -> flatten magnitude -> IFFT -> clip non-negative.
    """
    L = 1 << (3 * n - 2).bit_length()
    a = np.ones(n) / np.sqrt(n)

    for it in range(n_iter):
        A_ft = np.fft.rfft(a, L)
        mag = np.abs(A_ft)
        phase = A_ft / np.maximum(mag, 1e-30)

        # Target magnitude: flat (all equal to the mean)
        # But keep DC component somewhat larger
        target = np.full(mag.shape, np.mean(mag))
        # Blend: keep some of the current magnitude for stability
        blend = 0.3 + 0.7 * (it / n_iter)  # increase flatness over iterations
        target = blend * target + (1.0 - blend) * mag

        A_new = target * phase
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.clip(a_new, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / np.sqrt(n)
        else:
            a = a / s * np.sqrt(n)

    return a


def init_ift_band(n, rng, center_freq, bandwidth, n_iter=150):
    """
    IFT initialization targeting a specific frequency band.
    """
    L = 1 << (3 * n - 2).bit_length()
    a = np.ones(n) / np.sqrt(n)

    for it in range(n_iter):
        A_ft = np.fft.rfft(a, L)
        mag = np.abs(A_ft)
        phase = A_ft / np.maximum(mag, 1e-30)

        # Target: emphasize frequencies near center_freq
        freqs = np.arange(len(mag)) / L
        target = np.exp(-((freqs - center_freq) ** 2) / (2 * bandwidth ** 2))
        target = target / target.max() * np.mean(mag)

        blend = 0.3 + 0.7 * (it / n_iter)
        target = blend * target + (1.0 - blend) * mag

        A_new = target * phase
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.clip(a_new, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / np.sqrt(n)
        else:
            a = a / s * np.sqrt(n)

    return a

def optimize_anneal(a0, end_time, rng, T_start, T_end, init_lr, patience,

    Main optimization loop with soft-max gradient and temperature annealing.
    Returns (best_score, best_array).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    v_buf = np.zeros(n)
    step = 0
    no_imp = 0
    lr = init_lr
    cur_patience = patience
    T = T_start

    while time.monotonic() < end_time:
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(a0, 0.01, 1000.0)
            S = float(a.sum())

        C = fast_self_conv(a)
        score = compute_score_n(n, C, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Restart if stuck

            scale = 0.01 + 0.05 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)

            m_buf[:] = 0.0

            T = T_start  # Reset temperature on restart
            continue

        # Compute gradient

        step += 1
        adam_step(a, grad, m_buf, v_buf, step, lr)
        a = np.clip(a, 0.0, 1000.0)

        progress = min(1.0, step / 5000.0)
        T = T_start * (T_end / T_start) ** progress

        sc, aa = optimize_anneal(a0, t_end, rng, T_start=T_s, T_end=T_s * 0.1,
                                 init_lr=0.5, patience=100, lr_decay=0.998)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"  Init {i}: score={sc:.6f}, best={best_score:.6f}")

    # === Phase 1: Main soft-max annealing (40% time) ===
    p1_end = start + 0.50 * (total - 10.0)
    if best_a is not None:
        C_b = fast_self_conv(best_a)
        current_max = float(C_b.max())
        S_b = float(best_a.sum())
        T_s = max(1.0, 10.0 / current_max * S_b * S_b / (2.0 * n))
        T_e = T_s * 0.01
        sc, aa = optimize_anneal(best_a, p1_end, rng, T_start=T_s, T_end=T_e,
                                 init_lr=0.15, patience=300, lr_decay=0.9995)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1 (soft-max anneal): {best_score:.6f}")

    # === Phase 2: Multi-peak targeted optimization (30% time) ===
    p2_end = start + 0.80 * (total - 10.0)
    if best_a is not None:
        sc, aa = optimize_multi_peak(best_a, p2_end, rng, init_lr=0.05)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2 (multi-peak): {best_score:.6f}")

    # === Phase 3: Fine soft-max polish (15% time) ===
    p3_end = start + 0.95 * (total - 10.0)
    if best_a is not None:
        C_b = fast_self_conv(best_a)
        S_b = float(best_a.sum())
        T_s = 5.0 / max(float(C_b.max()), 1e-10)
        sc, aa = optimize_anneal(best_a, p3_end, rng, T_start=T_s, T_end=T_s * 0.05,
                                 init_lr=0.02, patience=500, lr_decay=0.9999)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3 (fine polish): {best_score:.6f}")

    # === Phase 4: Final ultra-fine (5% time) ===
    if best_a is not None and time.monotonic() < deadline:
        C_b = fast_self_conv(best_a)
        S_b = float(best_a.sum())
        T_s = 20.0 / max(float(C_b.max()), 1e-10)
        sc, aa = optimize_anneal(best_a, deadline, rng, T_start=T_s, T_end=T_s * 0.1,
                                 init_lr=0.005, patience=800, lr_decay=0.99995)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4 (ultra-fine): {best_score:.6f}")

    # Final output
    if best_a is None:
        best_a = np.ones(n) / n

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
