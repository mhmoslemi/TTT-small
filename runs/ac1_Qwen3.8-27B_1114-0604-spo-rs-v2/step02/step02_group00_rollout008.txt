"""
Power-sum continuation optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses power-sum smoothing (sum c_k^p)^(1/p) with continuation in p,
combined with L-BFGS-B, multi-length exploration, structured initializations,
and iterative reweighting for peak reduction.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def power_sum_obj_grad(a, p):
    """
    Power-sum smoothed objective: 2n * (sum c_k^p)^(1/p) / S^2.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 0.0, np.zeros(n)
    c_scaled = c / c_max
    cs = c_scaled ** p
    Q = float(cs.sum())
    f_p = c_max * Q ** (1.0 / p)
    score = 2.0 * n * f_p / (S * S)
    # Gradient of f_p w.r.t. a_j
    # df_p/da_j = 2 * c_max^(p-1) * Q^(1/p-1) * sum_k c_scaled[k]^(p-1) * a[k-j]
    c_scaled_p_minus_1 = c_scaled ** (p - 1)
    w = 2.0 * c_max ** (p - 1) * Q ** (1.0 / p - 1.0)
    # Cross-correlation: sum_k c_scaled_p_minus_1[k] * a[k-j]
    a_rev = a[::-1]
    M_conv = 2 * n - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    CS = np.fft.rfft(c_scaled_p_minus_1, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    cross = np.fft.irfft(CS * AR, M_fft)[:M_conv]
    grad_fp = w * cross[n - 1:2 * n - 1]
    # Full gradient: 2n * (grad_fp / S^2 - 2*f_p / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_fp / S2 - 2.0 * f_p / S3)
    return score, grad


def run_opt(a0, p, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B optimization on power-sum objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(a):
        val, g = power_sum_obj_grad(a, p)
        return val, g

    res = minimize(
        obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_inits_v2(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Raised cosine
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    inits.append(np.clip(0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5))), 0.0, 1000.0) + 0.05)
    # Parabolic (inverted)
    t = 2.0 * x - 1.0
    inits.append(np.clip(1.0 - 0.8 * t * t, 0.01, 1000.0))
    # Parabolic (normal)
    inits.append(np.clip(0.2 + 0.8 * t * t, 0.0, 1000.0))
    # Cosine modulations
    for freq, amp in [(1, 0.3), (2, 0.25), (3, 0.2), (5, 0.15)]:
        inits.append(np.clip(0.5 + amp * np.cos(2.0 * np.pi * freq * x), 0.01, 1000.0))
    # Gaussian bump
    center = n / 2.0
    sigma = n / 4.0
    inits.append(np.clip(0.1 + 0.9 * np.exp(-0.5 * ((x * n - center) / sigma) ** 2), 0.0, 1000.0))
    # Double bump
    inits.append(np.clip(0.1 + 0.8 * (np.exp(-0.5 * ((x * n - n * 0.3) / (n * 0.15)) ** 2) +
                                       np.exp(-0.5 * ((x * n - n * 0.7) / (n * 0.15)) ** 2)), 0.0, 1000.0))
    # Random smooth (low-pass filtered)
    raw = rng.standard_normal(n)
    # Simple moving average smoothing
    k = min(5, n // 4)
    if k > 1:
        kernel = np.ones(k) / k
        smoothed = np.convolve(raw, kernel, mode='same')
        inits.append(np.clip(0.5 + 0.3 * smoothed / (np.std(smoothed) + 1e-10), 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # Fewer inits for large n
    if n > 8192:
        inits = inits[:8]
    return inits


def peak_reduction_polish(a, n_iters=2000):
    """
    Iterative peak reduction: at each step, reduce the contribution to the
    top convolution peaks by shifting mass from large to small elements
    that contribute to the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:top_k]
        # For each peak, compute gradient contribution
        grad_total = np.zeros(n)
        for k in peak_idx:
            kt = int(k)
            j_start = max(0, kt - n + 1)
            j_end = min(kt, n - 1)
            if j_end >= j_start:
                # d c[kt] / d a[j] = 2 * a[kt-j]
                seg = a[j_start:j_end + 1]
                grad_total[j_start:j_end + 1] += 2.0 * seg[::-1] * c[kt] ** 2
        # Normalize gradient
        gn = np.linalg.norm(grad_total)
        if gn < 1e-15:
            break
        # Step size: adaptive
        step = 0.01 * S / gn
        a -= step * grad_total / S
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize to keep sum roughly constant
        S_new = a.sum()
        if S_new > 1e-10:
            a = a * (S0 / S_new)
            a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def cos_basis_optimize(n, n_basis, maxiter=500):
    """
    Optimize a sequence represented as a sum of cosines:
    a_j = bias + sum_m c_m * cos(2*pi*m*j/n + phase_m)
    This is a much lower-dimensional optimization.
    """
    # Parameters: bias, and for each basis function: amplitude, phase, frequency
    # a_j = bias + sum_m amp_m * cos(2*pi*freq_m*j/n + phase_m)
    # To ensure non-negativity, we'll project after each step.
    x0 = np.concatenate([[0.5], np.zeros(2 * n_basis)])
    for i in range(n_basis):
        x0[1 + 2 * i] = 0.1 + 0.3 * np.random.random()  # amplitude
        x0[2 + 2 * i] = np.random.random() * 2 * np.pi  # phase

    def objective_flat(params):
        bias = params[0]
        a = np.full(n, bias)
        for m in range(n_basis):
            amp = params[1 + 2 * m]
            phase = params[2 + 2 * m]
            freq = m + 1
            j = np.arange(n, dtype=np.float64)
            a += amp * np.cos(2.0 * np.pi * freq * j / n + phase)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-10:
            return 1e10
        return compute_score(a)

    # Use Nelder-Mead for this lower-dimensional problem
    res = minimize(objective_flat, x0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
    # Extract final sequence
    bias = res.x[0]
    a = np.full(n, bias)
    for m in range(n_basis):
        amp = res.x[1 + 2 * m]
        phase = res.x[2 + 2 * m]
        freq = m + 1
        j = np.arange(n, dtype=np.float64)
        a += amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    a = np.clip(a, 0.0, 1000.0)
    return compute_score(a), a


def multi_scale_refine(a_coarse, n_target, rng):
    """Upsample coarse solution to target length using sinc interpolation."""
    n_coarse = len(a_coarse)
    # Use FFT-based upsampling
    ratio = n_target / n_coarse
    # Zero-pad in frequency domain
    M_fft = 1 << (n_target - 1).bit_length()
    A_coarse = np.fft.rfft(a_coarse, M_fft)
    # Apply a low-pass filter to avoid aliasing
    freqs = np.fft.rfftfreq(M_fft)
    cutoff = n_coarse / M_fft
    A_filtered = A_coarse * (freqs < cutoff)
    a_fine = np.fft.irfft(A_filtered, M_fft)[:n_target]
    a_fine = np.clip(a_fine, 0.0, 1000.0)
    # Slight random perturbation
    a_fine = a_fine * (1.0 + 0.01 * rng.standard_normal(n_target))
    a_fine = np.clip(a_fine, 0.0, 1000.0)
    return a_fine

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase2_deadline:
                break
            if max_inits_tried >= 6:
                break
            max_inits_tried += 1
            a_cur = a0.copy()
            for pi, p_val in enumerate(p_schedule):
                if time.monotonic() > phase2_deadline:
                    break
                remaining = phase2_deadline - time.monotonic()
                if remaining < 5:
                    break
                # Scale iterations based on remaining time
                maxiter = min(3000, max(200, int(remaining / 3.0)))
                sc, a_opt = run_opt(a_cur, p_val, maxiter=maxiter, maxfun=maxiter * 20)
                a_cur = a_opt
                report(sc, a_opt, f"n={n},i={i},p={p_val}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: High-power refinement on best solution
    # ==================================================================
    print(f"\n--- Phase 3: High-Power Refinement ---")
    phase3_deadline = t_start + 0.75 * (float(budget_s) - 15.0)

    for p_val in [128, 256, 512]:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt = run_opt(best_a, p_val, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"refine p={p_val}")

    # ==================================================================
    # Phase 4: Peak reduction polish
    # ==================================================================
    print(f"\n--- Phase 4: Peak Reduction Polish ---")
    phase4_deadline = t_start + 0.87 * (float(budget_s) - 15.0)

    sc, a_polished = peak_reduction_polish(best_a, n_iters=3000)
    report(sc, a_polished, "peak_polish")

    # Another round of L-BFGS from polished
    if time.monotonic() < phase4_deadline:
        sc, a_opt = run_opt(a_polished, 256, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "post-polish lbfgs")

    # ==================================================================
    # Phase 5: Local search with diverse perturbations
    # ==================================================================
    print(f"\n--- Phase 5: Local Search ---")
    n_best = len(best_a)
    local_deadline = deadline

    n_trials = 50
    for trial in range(n_trials):
        if time.monotonic() > local_deadline:
            break
        remaining = local_deadline - time.monotonic()
        if remaining < 10:
            break
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            contrib = np.zeros(n_best)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            top_n = max(5, n_best // 40)
            top_idx = np.argsort(contrib)[::-1][:top_n]
            bot_idx = np.argsort(contrib)[:n_best - top_n]
            a_pert = best_a.copy()
            total_red = 0.0
            for j in top_idx:
                red = a_pert[j] * 0.05
                a_pert[j] -= red
                total_red += red
            if len(bot_idx) > 0:
                a_pert[bot_idx] += total_red / len(bot_idx)
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Sparse random perturbation
            n_pert = max(10, n_best // 20)
            idx = rng.choice(n_best, size=n_pert, replace=False)
            a_pert = best_a.copy()
            a_pert[idx] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
        else:
            # Swap two segments
            if n_best > 100:
                seg_len = min(50, n_best // 10)
                i = int(rng.integers(0, n_best - 2 * seg_len))
                j = int(rng.integers(0, n_best - 2 * seg_len))
                a_pert = best_a.copy()
                a_pert[i:i+seg_len], a_pert[j:j+seg_len] = a_pert[j:j+seg_len].copy(), a_pert[i:i+seg_len].copy()
            else:
                a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)

        p_use = 64 if remaining > 30 else 32
        maxiter_use = min(2000, max(200, int(remaining / 2.0)))
        sc, a_opt = run_opt(a_pert, p_use, maxiter=maxiter_use, maxfun=maxiter_use * 15)
        report(sc, a_opt, f"local #{trial} s={strategy}")

    # ==================================================================
    # Phase 6: Final polish
    # ==================================================================
    print(f"\n--- Phase 6: Final Polish ---")
    if time.monotonic() < deadline - 5:
        sc, a_polished = peak_reduction_polish(best_a, n_iters=2000)
        report(sc, a_polished, "final polish")

    if time.monotonic() < deadline - 3:
        sc, a_opt = run_opt(best_a, 512, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "final lbfgs")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
