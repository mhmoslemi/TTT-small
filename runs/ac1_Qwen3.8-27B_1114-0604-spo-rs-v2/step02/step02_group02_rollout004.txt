def _lp_update(f, solver='ECOS'):
    """Solve the LP to find g that minimizes max(f*g) subject to g>=0, sum(g)=1."""
    n = len(f)
    # Variables: g[0..n-1], t
    # Minimize: t (coefficient is 0 for g, 1 for t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag k: sum_i f[i]*g[k-i] - t <= 0
    # The coefficient vector for constraint k has f[i] in positions corresponding to g[k-i] and -1 in position n (for t)
    
    # Build A_ub as a sparse matrix
    from scipy.sparse import lil_matrix
    A_ub = lil_matrix((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # (f*g)[k] = sum_{i} f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i  # index into g
            A_ub[k, j] = f[i]
        A_ub[k, n] = -1.0  # -t term
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0 (already handled by bounds), t is free
    bounds = [(0, None)] * n + [(None, None)]
    
    # Solve
    result = linprog(c_obj, A_ub=A_ub.tocsr(), A_eq=A_eq, b_ub=b_ub, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    return result.x[:n]

"""
Step function optimizer using LP-based alternating minimization.
Minimizes 2*n*max(f*f)/(sum(f))^2 by alternating between:
  (1) Fixing f and solving an LP to find optimal g
  (2) Updating f from g
With diverse initialization and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, coo_matrix, csr_matrix
from scipy.signal import convolve

def _autocorr(f):
    """Full autocorrelation via np.convolve."""
    return np.convolve(f, f)

def _build_conv_matrix(f):
    """Build matrix T such that T @ g = convolve(f, g, 'full')."""
    n = len(f)
    T = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        T[k, lo:hi] = f[k-hi+1:k-lo+1][::-1]
    return T.tocsr()

def _lp_solve(f):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(f)
    T = _build_conv_matrix(f)
    
    # Full constraint matrix: [T | -ones] @ [g; t] <= 0
    A_ub = T.copy()
    A_ub[:, n] = -1.0  # Add -t column (but T is sparse, need to be careful)
    
    # Actually, let's build the full matrix
    A_full = lil_matrix((2*n-1, n+1))
    A_full[:, :n] = T
    A_full[:, n] = -1.0
    A_full = A_full.tocsr()
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # t >= 0 (since f,g >= 0, t must be >= 0)
    
    result = linprog(c_obj, A_ub=A_full, b_ub=np.zeros(2*n-1), 
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    return result.x[:n]

def _objective(f):
    """Compute 2*n*max(f*f)/(sum(f))^2."""
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * len(f) * c.max() / (s**2)

def _generate_starts(f_ref, n, rng, count=30):
    """Generate diverse starting points."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n-1) / 2.0
    
    # Reference
    starts.append(f_ref.copy())
    
    # Uniform
    starts.append(np.ones(n))
    
    # Gaussian bells
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        starts.append(np.exp(-0.5 * ((j - center) / (sigma * n))**2))
    
    # U-shaped
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        starts.append(1.0 + alpha * ((j - center) / center)**2)
    
    # Bimodal
    for sep in [0.15, 0.2, 0.25, 0.3]:
        starts.append(
            np.exp(-((j - sep*n)/(0.05*n))**2) + 
            np.exp(-((j - (1-sep)*n)/(0.05*n))**2) + 0.01
        )
    
    # Sine-based
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            starts.append(1.0 + amp * np.sin(freq * 2 * np.pi * j / n))
    
    # Polynomial tapers
    for degree in [1, 2, 3, 4]:
        x = (j - center) / center
        starts.append(np.abs(x)**degree + 0.01)
        starts.append(1.0 + 0.5 * np.abs(x)**degree)
    
    # Random
    while len(starts) < count:
        r = rng.random()
        if r < 0.3:
            starts.append(rng.exponential(1.0, n))
        elif r < 0.5:
            starts.append(rng.dirichlet(np.ones(n)*0.1) * n)
        elif r < 0.7:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        else:
            base = np.zeros(n)
            nnz = n // 2
            idx = rng.choice(n, nnz, replace=False)
            base[idx] = rng.exponential(1.0, nnz)
            starts.append(base + 0.001)
    
    return starts[:count]

def _polish(f, deadline, rng, n):
    """Coordinate descent polishing."""
    f = f.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    
    c = _autocorr(f)
    best_peak = c.max()
    best_f = f.copy()
    
    while time.monotonic() < deadline:
        c = _autocorr(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-15:
            best_peak = cur_peak
            best_f = f.copy()
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Compute contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        # d(c[k_star])/df[j] = 2*f[k_star-j] for j in [lo, hi)
        for jj in range(lo, hi):
            contrib[jj] = 2.0 * f[k_star - jj]
        
        # Move mass from high-contribution to low-contribution coordinates
        # Find top contributors
        top_idx = np.argsort(contrib)[::-1][:20]
        bottom_idx = np.argsort(contrib)[:20]
        
        improved = False
        for ti in top_idx[:10]:
            if contrib[ti] < 1e-10:
                continue
            for bj in bottom_idx[:10]:
                if bj == ti:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = f[ti] * frac
                    if amt < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[ti] -= amt
                    f_try[bj] += amt
                    f_try = np.maximum(f_try, 0.0)
                    f_try /= f_try.sum()
                    c_try = _autocorr(f_try)
                    if c_try.max() < cur_peak - 1e-15:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        # Random moves if no structured improvement
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i == j2:
                    continue
                frac = rng.choice([0.005, 0.01, 0.02, 0.05])
                amt = f[i] * frac
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j2] += amt
                if f_try[i] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _autocorr(f_try)
                if c_try.max() < cur_peak - 1e-15:
                    f = f_try
                    break
        
        # Periodic perturbation
        if not improved and (int((time.monotonic() - (deadline - 2400)) * 10) % 100 == 0):
            f_try = f + 0.001/np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _autocorr(f_try)
            if c_try.max() < cur_peak - 1e-15:
                f = f_try
    
    return best_f, _objective(best_f)

        scored.append((obj, f))
        if (i+1) % 10 == 0:

    best_score = scored[0][0]
    best_f = scored[0][1].copy()
    
    # LP-based alternating optimization
    # Time allocation: 75% for LP optimization, 20% for polish, 5% buffer
    lp_budget = budget_s * 0.70

                f_new /= f_new.sum()

                    if obj_mixed < best_score:
                        best_score = obj_mixed

    final_score = evaluate_sequence([float(x) for x in np.clip(best_f, 0.0, 1000.0)])

        print(f"Fallback uniform: {final_score:.6f}")
    
    best_score = min(best_score, final_score)
    print(f"=== FINAL: {best_score:.6f} ===")
    
    return [float(x) for x in np.clip(best_f, 0.0, 1000.0)]
