"""
Spectral shaping + LP + gradient optimization for minimizing
2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.
Uses frequency-domain convolution flattening (error reduction),
LP-based direction finding, and multiplicative gradient descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

        a_pad[:n] = a
        B = np.fft.rfft(a_pad)
        C = B * B  # this is the spectrum of the linear convolution
        c = np.fft.irfft(C, L)
        c = c[:2*n-1]  # valid part
        
        c_mean = c.sum() / (2*n - 1)  # should be ~1/(2n-1) if sum(a)=1
        
        # Geometric mean shaping towards flat
        if c_mean > 1e-15:
            c_shaped = np.power(np.maximum(c, 1e-30), 1-t) * np.power(c_mean, t)
            # Ensure non-negative
            c_shaped = np.maximum(c_shaped, 0)
        else:
            c_shaped = c.copy()
        
        C_shaped_pad = np.zeros(L, dtype=float)
        C_shaped_pad[:2*n-1] = c_shaped
        C_shaped = np.fft.rfft(C_shaped_pad)
        
        # Take square root (complex)
        B_new = np.sqrt(C_shaped.astype(complex))
        
        a_new_full = np.fft.irfft(B_new, L)
        a_new = a_new_full[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    return a

def _gradient_step(a, beta, L):
    """
    Multiplicative gradient step using smoothed max.
    Returns (new_a, score, gradient_norm)
    """
    n = len(a)
    c = _conv_full(a, a)
    d = n * c  # so that score = 2 * max(d)
    
    # Softmax weights
    d_shifted = d - d.max()
    e = np.exp(beta * d_shifted)
    s = e.sum()
    if s < 1e-300:
        return a, float(2.0 * d.max()), 0.0
    w = e / s
    
    # Gradient: h[i] = 2 * sum_k w[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] + a[i-k] (with boundary handling)
    # h[i] = 2 * (cross_correlation(w, a))[i]
    # Using FFT
    L_fft = _next_pow2(3 * n)
    W = np.fft.rfft(w, L_fft)
    A = np.fft.rfft(a, L_fft)
    R = np.fft.irfft(W * np.conj(A), L_fft)
    h = 2.0 * R[:n]
    h = np.maximum(h, 0.0)
    
    score = float(2.0 * d.max())
    max_h = h.max()
    if max_h < 1e-15:
        return a, score, 0.0
    
    return a, score, max_h

def _do_gradient_update(a, learning_rate, beta, L):
    """Perform one gradient update."""
    n = len(a)
    c = _conv_full(a, a)
    d = n * c
    d_shifted = d - d.max()
    e = np.exp(beta * d_shifted)
    s = e.sum()
    if s < 1e-300:
        return a
    w = e / s
    
    L_fft = _next_pow2(3 * n)
    W = np.fft.rfft(w, L_fft)
    A = np.fft.rfft(a, L_fft)
    R = np.fft.irfft(W * np.conj(A), L_fft)
    h = 2.0 * R[:n]
    h = np.maximum(h, 0.0)
    max_h = h.max()
    if max_h < 1e-15:
        return a
    
    h_norm = h / max_h
    a_new = a * np.exp(-learning_rate * h_norm)
    a_new = np.maximum(a_new, 1e-18 / n)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    return a_new / s_new

def _lp_direction(f, M):
    """
    Solve LP: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns g (normalized to sum=1) or None.
    """
    n = len(f)
    # Constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # Number of constraints: 2n-1
    # Number of variables: n
    
    # For efficiency, use sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n-1)
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def _lp_refine(a, M_scale=1.0, max_iters=5, blend=0.3):
    """
    Use LP to find a good direction and blend with current.
    """
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        c = _conv_full(a, a)
        M = c.max() * M_scale
        
        g = _lp_direction(a, M)
        if g is None:
            break
        
        # Blend
        a_try = (1 - blend) * a + blend * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a_try = a_try / s
        
        sc = _score(a_try)
        if sc < best_score:
            best = a_try.copy()
            best_score = sc
            a = a_try
        else:
            # Try smaller blend
            a_try2 = (1 - blend*0.5) * a + blend*0.5 * g
            a_try2 = np.maximum(a_try2, 0)
            s2 = a_try2.sum()
            if s2 > 1e-10:

                sc2 = _score(a_try2)
                if sc2 < best_score:
                    best = a_try2.copy()
                    best_score = sc2
                    a = a_try2
                else:
                    break
            else:
                break
    
    return best, best_score

def _init_barker_like(n):

"""
Spectral shaping + LP + gradient optimization for minimizing
2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.
Uses log-magnitude spectral reshaping (error reduction in frequency domain),
LP-based direction finding from the convex optimization approach, and
multiplicative gradient descent with smoothed-max.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _spectral_shape(a, t, L_cache=None):
    """
    Log-magnitude spectral reshaping step.
    Compresses the log-magnitude of the convolution spectrum towards its mean,
    effectively reducing peaks in the convolution.
    a: non-negative sequence, sum=1
    t: shaping strength in (0, 1); 0=no change, 1=perfectly flat magnitude
    """
    n = len(a)
    if L_cache is None or L_cache < 2 * n - 1:
        L = _next_pow2(2 * n - 1)
    else:
        L = L_cache

    a_pad = np.zeros(L)
    a_pad[:n] = a
    B = np.fft.rfft(a_pad)
    C = B * B  # spectrum of linear convolution (length L/2+1)

    mag_C = np.abs(C) + 1e-30
    phase_C = np.angle(C)
    log_mag_C = np.log(mag_C)
    mean_log_mag = np.mean(log_mag_C)

    # Compress log-magnitude towards mean
    log_mag_shaped = (1.0 - t) * log_mag_C + t * mean_log_mag
    mag_shaped = np.exp(log_mag_shaped)

    # Reconstruct B_new with shaped magnitude, same phase
    B_new = np.sqrt(mag_shaped) * np.exp(0.5j * phase_C)

    # Enforce conjugate symmetry (should already hold, but numerical safety)
    # rfft output is [DC, ..., Nyquist], irfft handles the rest
    a_new_full = np.fft.irfft(B_new, L)
    a_new = a_new_full[:n]

    # Project to non-negative and normalize
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a  # failed, return original
    return a_new / s


def _spectral_shape_multi(a, t, n_substeps, L_cache=None):
    """Multiple sub-steps of spectral shaping for stability."""
    n = len(a)
    if L_cache is None or L_cache < 2 * n - 1:
        L = _next_pow2(2 * n - 1)
    else:
        L = L_cache

    for _ in range(n_substeps):
        t_sub = t / n_substeps
        a = _spectral_shape(a, t_sub, L)
    return a


def _gradient_update(a, lr, beta):
    """
    Multiplicative gradient step using smoothed max of n*conv(a,a).
    Returns new a (normalized).
    """
    n = len(a)
    c = _conv_full(a, a)
    d = n * c

    # Softmax weights (focus on peaks)
    d_max = d.max()
    if d_max < 1e-15:
        return a
    d_shifted = d - d_max
    e = np.exp(beta * d_shifted)
    s = e.sum()
    if s < 1e-300:
        return a
    w = e / s

    # Gradient h[i] = 2 * sum_k w[k] * dc[k]/da[i]
    # = 2 * cross_correlation(w, a)[i] (valid part)
    L_fft = _next_pow2(3 * n)
    W = np.fft.rfft(w, L_fft)
    A = np.fft.rfft(a, L_fft)
    R = np.fft.irfft(W * np.conj(A), L_fft)
    h = 2.0 * R[:n]
    h = np.maximum(h, 0.0)

    max_h = h.max()
    if max_h < 1e-15:
        return a

    h_norm = h / max_h
    a_new = a * np.exp(-lr * h_norm)
    a_new = np.maximum(a_new, 1e-18 / n)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    return a_new / s_new


def _lp_direction(f, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k=0..2n-2.
    Returns g normalized to sum=1, or None on failure.
    """
    n = len(f)
    if n > 6000:
        return None  # too large

    # Build sparse constraint matrix A_ub of shape (2n-1, n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 60})
        if result.success and result.x is not None:
            g = np.asarray(result.x, dtype=float)
            g = np.maximum(g, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _lp_refine(a, max_iters=10, time_limit_total=300):
    """
    Iteratively apply LP direction to flatten the convolution.
    Returns (best_a, best_score).
    """
    t_start = time.monotonic()
    n = len(a)
    best = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit_total:
            break

        c = _conv_full(a, a)
        M = c.max()

        g = _lp_direction(a, M)
        if g is None:
            break

        # Try different blend factors
        improved = False
        for blend in (0.3, 0.5, 0.7, 1.0):
            a_try = (1.0 - blend) * a + blend * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-13:
                best = a_try.copy()
                best_score = sc
                a = a_try
                improved = True
                break
        if not improved:
            break

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_cosine_mod(n, rng):
    """Cosine-modulated initialization with random phases."""
    p = np.ones(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for k in range(1, min(20, n // 4)):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_two_tone(n, rng):
    """Two-level with random perturbation."""
    ratio = rng.uniform(1.5, 4.0)
    frac = rng.uniform(0.3, 0.7)
    p = np.ones(n, dtype=float)
    p[:int(n * frac)] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.05 * rng.standard_normal(n))
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Multi-phase: spectral shaping -> LP refinement -> gradient polish.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20  # safety buffer

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={time.monotonic()-start_time:.0f}s ***")

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"Parent score: {sc:.6f} (n={len(parent)})")

    # Determine working length and initial sequence
    if parent is not None:
        n_work = len(parent)
        a = parent.copy()
    else:
        n_work = 3072
        a = _init_uniform(n_work)
        _try_update(a, _score(a))

    L_spec = _next_pow2(2 * n_work - 1)
    print(f"Working: n={n_work}, L_spec={L_spec}, initial best={best_score_overall:.6f}")

    # ========== PHASE 1: Spectral shaping with annealing (35% time) ==========
    print("\n=== Phase 1: Spectral Shaping (annealed t) ===")
    t1_end = start_time + 0.30 * budget

    beta = 50.0
    lr = 0.08
    stall_count = 0
    iter_count = 0

    while time.monotonic() < t1_end:
        # Annealing schedule for t: start conservative, increase, then decrease
        progress = (time.monotonic() - start_time) / (t1_end - start_time + 1e-10)
        if progress < 0.5:
            t_shape = 0.1 + 0.4 * (progress / 0.5)  # ramp up 0.1 -> 0.5
        else:
            t_shape = 0.5 - 0.2 * ((progress - 0.5) / 0.5)  # ramp down 0.5 -> 0.3

        # Spectral shaping
        a_new = _spectral_shape(a, t_shape, L_spec)
        sc_new = _score(a_new)

        if sc_new < best_score_overall - 1e-13:
            a = a_new
            _try_update(a, sc_new)
            stall_count = 0
        elif sc_new < _score(a) + 0.001:
            a = a_new
            stall_count += 1
        else:
            stall_count += 1

        # Occasionally interleave gradient steps
        if rng.random() < 0.25:
            a_g = _gradient_update(a, lr, beta)
            sc_g = _score(a_g)
            if sc_g < best_score_overall - 1e-13:
                a = a_g
                _try_update(a, sc_g)
                stall_count = 0
            elif sc_g < _score(a):
                a = a_g

        # Adapt
        if stall_count > 20:
            a = best_overall.copy()
            stall_count = 0
            lr = max(0.01, lr * 0.7)

        if iter_count % 30 == 0:
            beta = min(3000.0, beta * 1.03)

        iter_count += 1

    print(f"Phase 1 done ({iter_count} iters): best={best_score_overall:.6f} at t={time.monotonic()-start_time:.0f}s")

    # ========== PHASE 2: LP-based refinement (20% time) ==========
    print("\n=== Phase 2: LP Refinement ===")
    t2_end = start_time + 0.50 * budget

    if n_work <= 5000:
        a_lp = best_overall.copy()
        lp_time_left = t2_end - time.monotonic()
        if lp_time_left > 30:
            a_lp, sc_lp = _lp_refine(a_lp, max_iters=15, time_limit_total=lp_time_left)
            _try_update(a_lp, sc_lp)
            print(f"Phase 2 LP done: best={best_score_overall:.6f} at t={time.monotonic()-start_time:.0f}s")
    else:
        print(f"Skipping LP (n={n_work} too large)")

    # ========== PHASE 3: Fine-grained gradient + spectral polish (30% time) ==========
    print("\n=== Phase 3: Fine Polish ===")
    t3_end = start_time + 0.82 * budget

    a = best_overall.copy()
    beta = 500.0
    lr = 0.04
    stall_count = 0
    iter_count = 0

    while time.monotonic() < t3_end:
        r = rng.random()

        if r < 0.45:
            # Gradient step
            a_new = _gradient_update(a, lr, beta)
            sc = _score(a_new)
            if sc < best_score_overall - 1e-13:
                a = a_new
                _try_update(a, sc)
                stall_count = 0
            elif sc < _score(a):
                a = a_new
                stall_count += 1
            else:
                stall_count += 1

        elif r < 0.75:
            # Spectral shaping (milder)
            t_sub = rng.uniform(0.1, 0.4)
            a_new = _spectral_shape(a, t_sub, L_spec)
            sc = _score(a_new)
            if sc < best_score_overall - 1e-13:
                a = a_new
                _try_update(a, sc)
                stall_count = 0
            elif sc < _score(a) * 1.002:
                a = a_new
                stall_count += 1
            else:
                stall_count += 1

        elif r < 0.90:
            # Block scaling
            bs = int(rng.choice([4, 8, 16, 32, 64, 128, 256]))
            bs = min(bs, n_work)
            start_idx = int(rng.integers(0, n_work - bs + 1))
            f = float(rng.choice([0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0]))
            a_try = a.copy()
            a_try[start_idx:start_idx + bs] *= f
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc = _score(a_try)
                if sc < best_score_overall - 1e-13:
                    a = a_try
                    _try_update(a, sc)
                    stall_count = 0
                elif sc < _score(a):
                    a = a_try
                else:
                    stall_count += 1
            else:
                stall_count += 1

        else:
            # Small perturbation
            std = 0.005 + 0.002 * min(stall_count, 20)
            a_new = a * np.exp(rng.normal(0.0, std, n_work))
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc = _score(a_new)
                if sc < best_score_overall * 1.005:
                    a = a_new
                if sc < best_score_overall - 1e-13:
                    _try_update(a_new, sc)
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1

        if stall_count > 40:
            a = best_overall.copy()
            stall_count = 0
            lr = max(0.005, lr * 0.6)

        if iter_count % 50 == 0:
            beta = min(8000.0, beta * 1.05)

        iter_count += 1

    print(f"Phase 3 done ({iter_count} iters): best={best_score_overall:.6f} at t={time.monotonic()-start_time:.0f}s")

    # ========== PHASE 4: Diverse restarts + final polish (remaining time) ==========
    print("\n=== Phase 4: Diverse restarts + Final Polish ===")

    # Try a few diverse starts and brief optimization
    t4_end = deadline
    n_best = len(best_overall)

    for mode in range(4):
        if time.monotonic() > t4_end - 60:
            break

        if mode == 0:
            sp = _init_cosine_mod(n_best, rng)
        elif mode == 1:
            sp = _init_two_tone(n_best, rng)
        elif mode == 2:
            # Perturbed best
            sp = best_overall * np.exp(rng.normal(0, 0.1, n_best))
            sp = np.maximum(sp, 0)
            sp = sp / sp.sum()
        else:
            sp = _init_uniform(n_best)

        # Brief spectral + gradient optimization
        t_alloc = min(60.0, (t4_end - time.monotonic()) * 0.3)
        t_alloc = max(10.0, t_alloc)
        t_local_end = time.monotonic() + t_alloc

        a_trial = sp
        for _ in range(200):
            if time.monotonic() > t_local_end:
                break
            r = rng.random()
            if r < 0.5:
                a_trial = _spectral_shape(a_trial, 0.3, L_spec)
            else:
                a_trial = _gradient_update(a_trial, 0.05, 200.0)
            sc = _score(a_trial)
            if sc < best_score_overall - 1e-13:
                _try_update(a_trial, sc)
                a = a_trial.copy()

    # Final high-beta gradient polish on the best
    if best_overall is not None and time.monotonic() < deadline - 10:
        a = best_overall.copy()
        t_final_end = deadline - 5
        beta_f = 2000.0
        lr_f = 0.02
        for _ in range(2000):
            if time.monotonic() > t_final_end:
                break
            a_new = _gradient_update(a, lr_f, beta_f)
            sc = _score(a_new)
            if sc < best_score_overall - 1e-13:
                a = a_new
                _try_update(a, sc)
            elif sc < _score(a):
                a = a_new
            # Also try spectral
            if rng.random() < 0.3:
                a_spec = _spectral_shape(a, 0.2, L_spec)
                sc_spec = _score(a_spec)
                if sc_spec < best_score_overall - 1e-13:
                    a = a_spec
                    _try_update(a, sc_spec)
                elif sc_spec < _score(a):
                    a = a_spec

    # ========== Final output ==========
    print("\n=== Final ===")
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)

    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            print(f"Official eval: {eval_score:.6f}")
            if eval_score < final_score:
                final_score = eval_score
    except Exception:
        pass

    print(f"\nFINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
