import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
       Returns np.inf if the input is invalid."""
    #---Security Checks---
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (since bool is a subclass of int) and
        # any other non-integer/non-float types (like strings, complex numbers).
        if isinstance(x, bool) or type(x) not in (int, float):
            return np.inf
        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Ensure non-negative numbers and protect against too large numbers for potential clipping
    sequence = [max(0.0, min(1000.0, x)) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = np.max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    score = float(2.0 * n * max_b / (sum_a ** 2))

    return score


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2*n-1
    M_fft = 1 << (M.bit_length() - 1)
    A = np.fft.rfft(a, M_fft)
    conv = np.fft.irfft(A * A)[:M]
    return conv


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum(a))^2."""
    n = len(a)
    if n < 1:
        return np.inf

    # Safeguard against small sums (though the input is non-negative, ensuring sum >=0.01)
    conv = fast_conv(a)
    total_sum = np.sum(a)
    if total_sum < 0.01:
        return np.inf

    max_conv_val = np.max(conv)
    score = 2.0 * n * max_conv_val / (total_sum ** 2)
    return score
