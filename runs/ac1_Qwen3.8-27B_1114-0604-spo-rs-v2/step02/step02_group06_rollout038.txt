def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _sanitize(a, normalize=True):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a

def _compute_c(a, L):
    n = len(a)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    return np.maximum(c, 0.0)

def _exact_score(a):
    n = len(a)
    L = _next_pow2(2*n-1)
    c = _compute_c(a, L)
    return float(2.0 * n * c.max())

def _softmax_w(c, beta):
    bc = beta * c
    bc -= bc.max()
    e = np.exp(bc)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(len(c), 1.0/len(c))
    return e / s

def _cross_corr_wa(w, a, L):
    n = len(a)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    g = np.fft.irfft(W * np.conj(A), L)[:n]
    return np.maximum(g, 0.0)

def _obj_grad_x(x, beta, L, n):
    """Objective and gradient with respect to log-parameters x."""
    xm = x - x.max()
    ex = np.exp(xm)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = ex / s
    
    c = _compute_c(a, L)
    w = _softmax_w(c, beta)
    g = _cross_corr_wa(w, a, L)
    
    # soft_max value
    bc = beta * c
    log_sum_exp = bc.max() + np.log(np.exp(bc - bc.max()).sum())
    soft_max_val = log_sum_exp / beta
    obj = 2.0 * n * soft_max_val
    
    # gradient w.r.t. x
    ag = float(np.dot(a, g))
    grad = 4.0 * n * a * (g - ag)
    
    return float(obj), grad

def _lgbfs_optimize(x0, beta_max, time_limit, verbose=False):
    """L-BFGS optimization in log-space with increasing beta."""
    n = len(x0)
    L = _next_pow2(2*n-1)
    
    best_x = x0.copy()
    best_score = _exact_score(np.exp(x0 - x0.max()) / np.exp(x0 - x0.max()).sum())
    
    deadline = time.monotonic() + time_limit
    betas = [10, 30, 100, 300, 1000, 3000]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        beta_time = min(60.0, (deadline - time.monotonic()) / max(1, len(betas)))
        
        def fun(x):
            obj, _ = _obj_grad_x(x, beta, L, n)
            return obj
        
        def jac(x):
            _, g = _obj_grad_x(x, beta, L, n)
            return g
        
        bounds = [(-40, 40)] * n
        
        try:
            res = minimize(
                fun, x0, jac=jac, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxfun': 500, 'iprint': 0}
            )
            if res.x is not None:
                a_new = np.exp(res.x - res.x.max())
                a_new = a_new / a_new.sum()
                sc = _exact_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_x = res.x.copy()
                    x0 = res.x.copy()
                    if verbose:
                        print(f"  L-BFGS beta={beta}: score={sc:.7f}")
        except Exception as e:
            if verbose:
                print(f"  L-BFGS beta={beta} error: {e}")
    
    a_best = np.exp(best_x - best_x.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _projected_grad_desc(a0, time_limit, beta_start=50, beta_end=2000, verbose=False):
    """Custom projected gradient descent in a-space."""
    n = len(a0)
    L = _next_pow2(2*n-1)
    a = a0.copy()
    best_a = a.copy()
    best_sc = _exact_score(a)
    
    deadline = time.monotonic() + time_limit
    beta = float(beta_start)
    lr = 1.0 / (4.0 * n)
    stall = 0
    
    while time.monotonic() < deadline:
        c = _compute_c(a, L)
        max_c = c.max()
        cur_sc = 2.0 * n * max_c
        
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
            if verbose and (int((deadline - time.monotonic()) * 10) % 50 == 0):
                print(f"  PGD: score={best_sc:.7f}, beta={beta:.1f}")
        else:
            stall += 1
        
        w = _softmax_w(c, beta)
        g = _cross_corr_wa(w, a, L)
        ag = float(np.dot(a, g))
        grad = g - ag  # projected gradient direction (to subtract)
        
        # Line search
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * mult
            a_try = a - step * grad
            a_try = np.maximum(a_try, 1e-15)
            s = a_try.sum()
            if s <= 0:
                continue
            a_try = a_try / s
            sc_try = _exact_score(a_try)
            if sc_try < cur_sc * 1.001:
                a = a_try
                break
        else:
            a = best_a.copy()
        
        # Adapt beta and lr
        beta = min(beta_end, beta * 1.05)
        if stall > 20:
            lr *= 0.8
            if stall > 100:
                # Add noise to escape
                noise = np.exp(_RNG.normal(0, 0.1, n))
                a = a * noise
                a = a / a.sum()
                stall = 0
        elif stall > 5:
            lr *= 0.95
        
        # Reset if stuck for too long
        if stall > 200:
            a = best_a.copy()
            stall = 0
            beta = max(beta_start, beta * 0.5)
            lr = 1.0 / (4.0 * n)
    
    return best_a, best_sc

def _resample(a, m):
    """Resample sequence to length m."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-15
    return out / out.sum()

    n = len(a0)

    large = max(mid + 1, min(n, n // 64))

            cand[start:start+w] *= f

            if s <= 0:

                best_a = cand.copy()
                a = cand.copy()

                    a = best_a.copy()
                stagnant = 0
        
        if stagnant > 200:
            a = best_a.copy()
            stagnant = 0
    
    return best_a, best_sc

Optimization of non-negative step function heights minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
1. L-BFGS in log-space with soft-max smoothed objective and progressive beta increase
2. Custom projected gradient descent with adaptive learning rate and beta
3. Multi-scale coarse-to-fine refinement (1024 -> 2048 -> 4096)
4. Block-scaling local search for final polish
5. Exploits optimality condition: at optimum, cross-correlation of soft-max
   weights with sequence should be constant

import numpy as np
import time
import sys
from scipy.optimize import minimize

_RNG = None

def _init_rng(seed=42):
    global _RNG
    _RNG = np.random.default_rng(seed)

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _sanitize(a, normalize=True):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a

def _compute_c(a, L):
    """Auto-convolution via FFT. L must be >= 2*len(a)-1."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return np.maximum(c[:2*len(a)-1], 0.0)

def _exact_score(a):
    n = len(a)
    L = _next_pow2(2*n - 1)
    c = _compute_c(a, L)
    return float(2.0 * n * c.max())

def _softmax_w(c, beta):
    bc = beta * c
    bc -= bc.max()
    e = np.exp(bc)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(len(c), 1.0 / len(c))
    return e / s

def _cross_corr(w, a, L):
    """g[i] = sum_j w[j+i]*a[j] for i=0..n-1 via FFT. L >= len(w)+len(a)-1."""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    g = np.fft.irfft(W * np.conj(A), L)[:len(a)]
    return np.maximum(g, 0.0)

def _obj_grad_x(x, beta, L_conv, L_cc, n):
    """Objective and gradient wrt log-params x. a = softmax(x)."""
    xm = x - x.max()
    ex = np.exp(xm)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = ex / s

    c = _compute_c(a, L_conv)
    w = _softmax_w(c, beta)
    g = _cross_corr(w, a, L_cc)

    bc = beta * c
    log_sum_exp = bc.max() + np.log(np.exp(bc - bc.max()).sum())
    obj = 2.0 * n * (log_sum_exp / beta)

    ag = float(np.dot(a, g))
    grad = 4.0 * n * a * (g - ag)
    return float(obj), grad

def _lgbfs_optimize(a0, time_limit, beta_start=20, beta_end=5000, verbose=False):
    """L-BFGS in log-space with increasing beta."""
    n = len(a0)
    L_conv = _next_pow2(2 * n - 1)
    L_cc = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_score = _exact_score(a)

    deadline = time.monotonic() + time_limit
    x0 = np.log(a + 1e-15)

    # Geometric beta schedule
    n_betas = 8
    betas = [beta_start * (beta_end / beta_start) ** (i / (n_betas - 1)) for i in range(n_betas)]

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def fun(x):
            obj, _ = _obj_grad_x(x, beta, L_conv, L_cc, n)
            return obj

        def jac(x):
            _, g = _obj_grad_x(x, beta, L_conv, L_cc, n)
            return g

        bounds = [(-35, 35)] * n

        try:
            res = minimize(
                fun, x0, jac=jac, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 800, 'maxls': 50, 'iprint': 0}
            )
            if res.x is not None and res.success or (res.x is not None and res.fun < fun(x0)):
                a_new = np.exp(res.x - res.x.max())
                a_new = a_new / a_new.sum()
                sc = _exact_score(a_new)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_new.copy()
                    x0 = res.x.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"    L-BFGS beta={beta:.0f}: score={sc:.7f} (improved)")
                else:
                    x0 = res.x.copy()
                    if verbose:
                        print(f"    L-BFGS beta={beta:.0f}: score={sc:.7f} (no improvement)")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta:.0f} error: {e}")

    return best_a, best_score

def _pgd_optimize(a0, time_limit, beta_start=50, beta_end=5000, verbose=False):
    """Projected gradient descent in a-space with adaptive parameters."""
    n = len(a0)
    L_conv = _next_pow2(2 * n - 1)
    L_cc = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_sc = _exact_score(a)

    deadline = time.monotonic() + time_limit
    beta = float(beta_start)
    lr = 2.0 / (n * np.max(np.ones(n)))
    lr = 1.0 / (2.0 * n)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _compute_c(a, L_conv)
        cur_sc = 2.0 * n * c.max()

        if cur_sc < best_sc - 1e-10:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    PGD it={it}: score={best_sc:.7f} beta={beta:.0f} lr={lr:.2e}")
        else:
            stall += 1

        w = _softmax_w(c, beta)
        g = _cross_corr(w, a, L_cc)
        ag = float(np.dot(a, g))
        grad = g - ag

        # Backtracking line search
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = lr * mult
            a_try = a - step * grad
            a_try = np.maximum(a_try, 1e-15)
            s = a_try.sum()
            if s <= 0 or not np.isfinite(s):
                continue
            a_try = a_try / s
            sc_try = _exact_score(a_try)
            if sc_try < cur_sc * (1.0 + 1e-8):
                a = a_try
                improved = True
                break
        if not improved:
            a = best_a.copy()

        # Adapt parameters
        beta = min(beta_end, beta * 1.03)
        if stall > 15:
            lr *= 0.9
        if stall > 60:
            noise = np.exp(_RNG.normal(0, 0.08, n))
            a = best_a * noise
            a = a / a.sum()
            stall = 0
            beta = max(beta_start, beta * 0.7)
            lr = 1.0 / (2.0 * n)
        if stall > 300:
            a = best_a.copy()
            lr = 1.0 / (2.0 * n)
            beta = beta_start
            stall = 0

        it += 1

    return best_a, best_sc

def _resample(a, m):
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()

        print(f"Phase 4 block search: score={best_sc:.6f}")

    # === Phase 5: Coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t5 = min(rem * 0.3, 200.0)
        bp, scp = _coordinate_polish(best_a, t5, verbose=False)
        if scp < best_sc:
            best_a, best_sc = bp, scp
            print(f"Phase 5 coord polish: score={best_sc:.6f}")

    # === Phase 6: Final L-BFGS polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.5, 150.0)
        a_fin, sc_fin = _lgbfs_optimize(best_a, t6, beta_start=500, beta_end=10000, verbose=True)
        if sc_fin < best_sc:
            best_a, best_sc = a_fin, sc_fin
            print(f"Phase 6 final L-BFGS: score={best_sc:.6f}")

    # === Phase 7: Try different n if time permits ===
    rem = deadline - time.monotonic()
    if rem > 200 and len(best_a) == 4096:
        # Try n=2048 (sometimes smaller is better)
        a_small = _resample(best_a, 2048)
        sc_small = _exact_score(a_small)
        if sc_small < best_sc:
            ts = min(rem * 0.3, 200.0)
            a_small, sc_small = _lgbfs_optimize(a_small, ts, beta_start=50, beta_end=3000, verbose=False)
            if sc_small < best_sc:
                best_a, best_sc = a_small, sc_small
                print(f"Phase 7 smaller n: score={best_sc:.6f}")

    # Verification with evaluate_sequence
    rem = deadline - time.monotonic()
    if rem > 3 and len(best_a) <= 16384:
        try:
            lst = [float(x) for x in best_a]
            official = evaluate_sequence(lst)
            if np.isfinite(official):
                print(f"Official score: {official:.7f} (internal: {best_sc:.7f})")
                if official < best_sc:
                    best_sc = official
        except Exception as e:
            print(f"Eval error: {e}")

    # Final output
    best_a = _sanitize(best_a, True)
    if best_a is None:
        best_a = np.ones(2048) / 2048.0
    out = [float(x) for x in best_a]
    print(f"FINAL: n={len(out)}, score={best_sc:.7f}")
    return out
