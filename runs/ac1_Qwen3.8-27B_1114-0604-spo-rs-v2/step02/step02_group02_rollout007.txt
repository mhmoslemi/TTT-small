import numpy as np
from scipy.optimize import minimize
import time

def _autocorr_fft(f):
    """Compute full linear convolution of f with f via FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2*n-1]

def _compute_grad(f, w):
    """Compute g[i] = sum_k w[k] * (f[k-i] + f[i-k]) via FFT."""
    n = len(f)
    size = 1 << (4*n).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    cross1 = np.fft.irfft(np.conj(F) * W, size)
    cross2 = np.fft.irfft(np.conj(W) * F, size)
    return cross1[:n] + cross2[:n]

def _soft_obj_grad(x, beta):
    """
    Objective and gradient for soft-max of convolution.
    x: log-space parameters, f = softmax(x)
    Returns (objective_value, gradient)
    """
    n = len(x)
    # Compute f from log params
    x_shifted = x - x.max()
    f = np.exp(x_shifted)
    Z = f.sum()
    f = f / Z
    
    # Compute convolution
    c = _autocorr_fft(f)
    
    # Soft-max
    m = c.max()
    bc = beta * (c - m)
    # Numerical stability
    bc = np.clip(bc, -500, 500)
    exp_bc = np.exp(bc)
    S = exp_bc.sum()
    w = exp_bc / S
    obj = m + np.log(S) / beta
    
    # Gradient w.r.t. f
    g = _compute_grad(f, w)
    
    # Chain rule through softmax: df_i/dx_j = f_i*(delta_ij - f_j)
    # d(obj)/dx_j = sum_i (d(obj)/df_i) * df_i/dx_j = sum_i g[i] * f_i * (delta_ij - f_j)
    # = g[j]*f_j - f_j * sum_i g[i]*f_i
    S_gf = np.dot(g, f)
    grad = f * (g - S_gf)
    
    return obj, grad

"""
Step function optimizer using soft-max smoothed L-BFGS with annealing,
Nesterov subgradient polishing, and coordinate descent.
Key innovation: O(n log n) smooth objective via FFT-based gradient of
log-sum-exp of convolution entries, optimized with L-BFGS in log-space.
"""

import numpy as np
from scipy.optimize import minimize
import time

# --- Core computation functions ---

def _convolve_fft(f):
    """Full linear convolution of f with f via FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)

def _grad_via_fft(f, w):
    """
    Compute g[i] = sum_k w[k] * (f[k-i] + f[i-k]) for i in [0, n-1].
    Uses FFT for O(n log n) complexity.
    """
    n = len(f)
    size = 1 << (4*n).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    # (f cross-correlate w)[i] = sum_m f[m]*w[m+i] = ifft(conj(F)*W)[i]
    # (w cross-correlate f)[i] = sum_m w[m]*f[m+i] = ifft(conj(W)*F)[i]
    # g[i] = (f⋆w)[i] + (w⋆f)[i]
    c1 = np.fft.irfft(np.conj(F) * W, size)
    c2 = np.fft.irfft(np.conj(W) * F, size)
    return c1[:n] + c2[:n]

def _softmax_logsumexp(f, beta):
    """Compute soft-max of convolution: (1/beta)*log(sum exp(beta*c_k))."""
    c = _convolve_fft(f)
    m = c.max()
    bc = np.clip(beta * (c - m), -500, 500)
    S = np.exp(bc).sum()
    return m + np.log(S) / beta, c

def _soft_obj_and_grad(x, beta):
    """
    Objective and gradient for: soft_max(beta * conv(softmax(x), softmax(x)))
    Parameters: x in R^n (log-space)
    """
    n = len(x)
    # Compute f = softmax(x)
    x_shift = x - x.max()
    f = np.exp(x_shift)
    Z = f.sum()
    f = f / Z
    
    # Compute convolution and soft-max weights
    c = _convolve_fft(f)
    m = c.max()
    bc = np.clip(beta * (c - m), -500, 500)
    exp_bc = np.exp(bc)
    S = exp_bc.sum()
    w = exp_bc / S
    obj = m + np.log(S) / beta
    
    # Gradient w.r.t. f
    g = _grad_via_fft(f, w)
    
    # Chain rule through softmax
    S_gf = np.dot(g, f)
    grad = f * (g - S_gf)
    
    return obj, grad

def _optimize_lbfgs(f0, deadline, n):
    """
    Optimize using L-BFGS with annealed soft-max.
    Returns the best f found.
    """
    t_start = time.monotonic()
    x = np.log(f0 + 1e-30)
    
    best_peak = _convolve_fft(f0).max()
    best_f = f0.copy()
    
    # Anneal beta from small to large
    # Small beta: smooth, easy to optimize
    # Large beta: close to true max
    betas = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        t_beta_start = time.monotonic()
        
        def obj_func(x_):
            val, _ = _soft_obj_and_grad(x_, beta)
            return val
        
        def grad_func(x_):
            _, grad = _soft_obj_and_grad(x_, beta)
            return grad
        
        try:
            result = minimize(
                obj_func, x, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 100, 'gtol': 1e-10, 'maxls': 30,
                         'ftol': 1e-12}
            )
            x = result.x
        except Exception as e:
            print(f"    L-BFGS error at beta={beta}: {e}")
            break
        
        f_new = np.exp(x - x.max())
        f_new /= f_new.sum()
        
        c = _convolve_fft(f_new)
        peak = c.max()
        score = 2 * n * peak
        
        if peak < best_peak:
            best_peak = peak
            best_f = f_new.copy()
        
        elapsed = time.monotonic() - t_beta_start
        print(f"    beta={beta:6.1f}: peak={peak:.12f}, score={score:.8f}, "
              f"time={elapsed:.1f}s, iters={result.nit}")
    
    return best_f, best_peak

def _nesterov_subgrad(f0, deadline, n, rng):
    """
    Nesterov-accelerated subgradient descent on the true max.
    Works in the simplex (f >= 0, sum f = 1).
    """
    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    # Nesterov momentum
    v = np.zeros(n)
    momentum = 0.9
    eta = 0.1
    T = _convolve_fft(f).max() * 0.01
    
    best_peak = _convolve_fft(f).max()
    best_f = f.copy()
    last_improve = time.monotonic()
    
    iter_count = 0
    while time.monotonic() < deadline:
        c = _convolve_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-12:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()
        
        # Top-k subgradient
        k = min(5, 2*n-1)
        top_k = np.argsort(c)[-k:]
        grad = np.zeros(n)
        for idx in top_k:
            lo = max(0, idx - n + 1)
            hi = min(n, idx + 1)
            grad[lo:hi] += 2.0 * f[idx - np.arange(lo, hi)]
        grad /= k
        
        # Nesterov step
        v = momentum * v + eta * grad
        f_new = f - v
        f_new = np.maximum(f_new, 0)
        if f_new.sum() < 1e-10:
            f_new = np.ones(n) / n
        else:
            f_new /= f_new.sum()
        
        # SA acceptance
        c_new = _convolve_fft(f_new)
        new_peak = c_new.max()
        delta = new_peak - cur_peak
        
        if delta <= 0:
            f = f_new
            eta = min(eta * 1.01, 1.0)
        elif T > 1e-15:
            if np.log(max(rng.uniform(), 1e-300)) < -delta / T:
                f = f_new
            else:
                eta *= 0.95
        else:
            eta *= 0.9
        
        T = max(T * 0.999, _convolve_fft(f0).max() * 1e-15)
        
        # Reheat if stuck
        if time.monotonic() - last_improve > 20:
            T = _convolve_fft(f0).max() * 0.05
            eta = 0.1
            f = f + 0.01/np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f /= f.sum()
            last_improve = time.monotonic()
            v = np.zeros(n)
        
        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"    Nesterov iter {iter_count}: peak={cur_peak:.10f}, "
                  f"best={best_peak:.10f}")
    
    return best_f, best_peak
