        # = 2*a[k-i] (if i = k-i, i.e., 2i=k, and valid)
        # = a[k-i] (if only one term is valid)
        
        # This is getting complex. Let me use a simpler smoothed approach.
        # Use the "energy" method: gradient of sum(w*conv) w.r.t. a
        # = w * a + flip(w) * flip(a) (circular correlation, truncated)
        
        # For now, use a simpler heuristics-based approach:
        # Where the convolution is high, reduce those elements.
        # The convolution at lag k involves pairs (i, i+k) for i=0..n-1-k.
        # So high conv at lag k means we should reduce elements that participate in those pairs.
        
        # Build a "pressure" vector: for each position i, how much it contributes to high conv values
        pressure = np.zeros(n)
        # For each lag k with high weight, add to positions involved
        for k in range(n):
            if w[k] > 1e-10:
                # Positions 0..n-1-k participate in conv[k]
                # Weight each by the partner value
                for i in range(n - k):
                    pressure[i] += w[k] * p[i + k]
                    pressure[i + k] += w[k] * p[i]
        
        # Normalize
        pmax = pressure.max()
        if pmax < 1e-30:
            break
        pressure /= pmax
        
        # Multiplicative update: reduce where pressure is high
        p_try = p * np.exp(-step_size * pressure)
        p_try = np.maximum(p_try, 1e-15)
        s_try = p_try.sum()
        if s_try < 1e-10:
            continue
        p_try = p_try / s_try
        sc_try = _score_fast(p_try)
        
        if sc_try < cur_score - 1e-13:
            p = p_try
            cur_score = sc_try
            stall = 0
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
                if verbose and it % 50 == 0:
                    print(f"      iter={it}, score={best_score:.8f}, step={step_size:.4f}")
        else:
            stall += 1
            if stall % 20 == 0:
                step_size *= 0.8
            if stall > 100:
                p = best.copy()
                cur_score = best_score
                step_size = 0.05
                stall = 0

        it += 1

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    
    Key insight: By Cauchy-Schwarz, the score is always >= 2 for any non-negative
    sequence. The uniform sequence achieves exactly 2. This program verifies this
    and performs optimization to confirm or exploit any edge cases.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.8f} (n={len(p)}) ***")

    print("=" * 60)
    print(f"Starting optimization, budget={budget:.0f}s")
    print(f"Theoretical lower bound: 2.0 (by Cauchy-Schwarz)")

    # Evaluate parent
    if parent is not None:
        sc = _score_fast(parent)
        print(f"Parent: score={sc:.8f} (n={len(parent)})")
        try_update(parent, sc)
        # Also verify with exact
        sc_exact = _score_exact(parent)
        if abs(sc_exact - sc) > 1e-6:
            print(f"  WARNING: FFT vs exact differ: {sc:.8f} vs {sc_exact:.8f}")

    # Try uniform sequences at various lengths
    for n_try in (1024, 2048, 4096, 8192):
        u = np.ones(n_try) / n_try
        sc = _score_fast(u)
        print(f"Uniform n={n_try}: score={sc:.8f}")
        try_update(u, sc)

    if parent is not None:
        n_parent = len(parent)
        if n_parent not in (1024, 2048, 4096, 8192):
            u = np.ones(n_parent) / n_parent
            sc = _score_fast(u)
            print(f"Uniform n={n_parent}: score={sc:.8f}")
            try_update(u, sc)

    # If parent is close to uniform, it's already near optimal
    if best_overall is not None:
        print(f"\nCurrent best: {best_score_overall:.8f} (n={len(best_overall)})")
        if best_score_overall <= 2.0001:
            print("Already at or near the theoretical minimum of 2.0!")
            print("Attempting small optimizations to see if any improvement is possible...")

    # Phase 1: Try local search from parent (if it exists and is good)
    t1_end = start_time + 0.2 * budget
    if parent is not None and time.monotonic() < t1_end:
        print("\nPhase 1: Local search from parent")
        t_alloc = min(t1_end - time.monotonic(), 300.0)
        if t_alloc > 20:
            p_best, sc_best = _local_search(parent, rng, t_alloc, verbose=True)
            if p_best is not None:
                try_update(p_best, sc_best)

    # Phase 2: Gradient descent
    t2_end = start_time + 0.5 * budget
    if best_overall is not None and time.monotonic() < t2_end:
        print("\nPhase 2: Gradient descent")
        t_alloc = min(t2_end - time.monotonic(), 300.0)
        if t_alloc > 20:
            p_best, sc_best = _gradient_descent(best_overall, rng, t_alloc, verbose=True)
            if p_best is not None:
                try_update(p_best, sc_best)

    # Phase 3: Try structured constructions
    t3_end = start_time + 0.7 * budget
    if time.monotonic() < t3_end:
        print("\nPhase 3: Structured constructions")
        n = len(best_overall) if best_overall is not None else 2048
        
        # Two-level construction
        for ratio in (1.5, 2.0, 3.0, 5.0):
            for frac in (0.3, 0.4, 0.5, 0.6, 0.7):
                k = int(n * frac)
                p = np.ones(n)
                idx = rng.permutation(n)
                p[idx[k:]] = ratio
                s = p.sum()
                p = p / s
                sc = _score_fast(p)
                try_update(p, sc)
        
        print(f"  After structured: best={best_score_overall:.8f}")

    # Phase 4: Final polishing with local search
    t4_end = deadline
    if best_overall is not None and time.monotonic() < t4_end:
        print("\nPhase 4: Final polishing")
        t_alloc = min(t4_end - time.monotonic(), 300.0)
        if t_alloc > 10:
            p_best, sc_best = _local_search(best_overall, rng, t_alloc, verbose=True)
            if p_best is not None:
                try_update(p_best, sc_best)

            # One more gradient pass
            t_alloc2 = min(t4_end - time.monotonic(), 100.0)
            if t_alloc2 > 10:
                p_best, sc_best = _gradient_descent(best_overall, rng, t_alloc2, verbose=False)
                if p_best is not None:
                    try_update(p_best, sc_best)

    # Final verification
    if best_overall is None:
        best_overall = np.ones(2048) / 2048

    # Use exact evaluation for final candidate
    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(2048) / 2048

    final_score_fast = _score_fast(p_final)
    final_score_exact = _score_exact(p_final)
    print(f"\nFinal: FFT score={final_score_fast:.8f}, exact score={final_score_exact:.8f}")

    # Try official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        print(f"Official evaluator score: {eval_score:.8f}")
        if np.isfinite(eval_score):
            final_score_exact = min(final_score_exact, eval_score)
    except Exception as e:
        print(f"  Evaluator error: {e}")

    # Ensure output is valid
    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0.01 or not np.isfinite(s):
        out = np.ones(2048) / 2048
        s = 1.0

    print(f"\nResult: n={len(out)}, sum={s:.4f}, score={final_score_exact:.8f}")
    print(f"Time used: {time.monotonic() - start_time:.1f}s / {budget:.0f}s")

    return [float(x) for x in out]
