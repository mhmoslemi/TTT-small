"""
Multi-scale LSE optimization for minimum peak autocorrelation sequences.
Uses coarse-to-fine strategy, top-k robust objectives, and Adam optimizer.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S

            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) at the active lag
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    final_sc = compute_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    return best_sc, best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        p = parent
        if len(p) != n:
            p = upsample(p, n)
        inits.append(np.clip(p, 0.0, 1000.0).copy())
    
    inits.append(np.ones(n) * 0.5)
    inits.append(0.3 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.8 * rng.random(n), 0.01, 1000.0))
    
    # Gaussian bump
    center = n * (0.3 + 0.4 * rng.random())
    width = n * (0.1 + 0.3 * rng.random())
    inits.append(0.1 + 0.8 * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2))
    
    # Two bumps
    c1 = n * rng.random() * 0.4
    c2 = n * (0.6 + 0.3 * rng.random())
    w = n * 0.15
    inits.append(0.1 + 0.5 * np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2)
                + 0.5 * np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2))
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak autocorrelation sequence.
    Phase 1: Coarse-to-fine global optimization
    Phase 2: Multi-objective refinement (LSE + top-k)
    Phase 3: Fine local search and peak reduction
    Phase 4: Subgradient polish and final refinement
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = float(budget_s) - 15.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={len(parent) if parent is not None else 0}")
    if parent is not None:
        print(f"Parent score: {compute_score(parent):.8f}")
        best_score = compute_score(parent)
        best_a = parent.copy()

    # ==================================================================
    # Phase 1: Coarse-to-fine global search (~35% of time)
    # ==================================================================
    phase1_end = t_start + 0.35 * effective_budget
    print(f"\n=== Phase 1: Coarse-to-Fine (deadline t={phase1_end-t_start:.0f}s) ===")

    # Determine target length for final answer
    target_n = 8192 if parent is None else max(4096, min(16384, len(parent)))
    if parent is not None:
        # Try to stay close to parent length or slightly larger
        pn = len(parent)
        target_n = pn  # keep same length initially

    # Coarse level
    coarse_n = max(256, min(1024, target_n // 8))
    print(f"  Coarse n={coarse_n}, target n={target_n}")

    # Optimize at coarse level
    coarse_a = None
    if parent is not None:
        coarse_a = downsample(parent, max(1, len(parent) // coarse_n))
    
    coarse_inits = build_diverse_inits(coarse_n, rng, parent)
    for i, a0 in enumerate(coarse_inits[:6]):
        if time.monotonic() > phase1_end - 30:
            break
        sc, a_opt = run_lbfgs(a0, 30, maxiter=3000, maxfun=50000)
        if coarse_a is None or sc < compute_score(coarse_a):
            coarse_a = a_opt
        sc2, a_opt2 = run_lbfgs(a_opt, 200, maxiter=3000, maxfun=50000)
        if coarse_a is None or sc2 < compute_score(coarse_a):
            coarse_a = a_opt2
        print(f"    coarse init {i}: best={compute_score(coarse_a):.8f}")

    # Medium level
    if time.monotonic() < phase1_end - 60:
        med_n = max(1024, min(4096, target_n // 2))
        print(f"  Medium n={med_n}")
        med_a = upsample(coarse_a, med_n)
        if parent is not None and len(parent) == med_n:
            med_a = parent.copy()
        
        # Try both upsampled and parent (if matches)
        candidates = [med_a]
        if parent is not None:
            p_med = upsample(parent, med_n) if len(parent) != med_n else parent.copy()
            candidates.append(p_med)
        
        for ci, a0 in enumerate(candidates[:3]):
            if time.monotonic() > phase1_end - 30:
                break
            sc, a_opt = adaptive_lbfgs(a0, maxiter_total=15000, maxfun_total=300000)
            if sc < compute_score(med_a):
                med_a = a_opt
            print(f"    med candidate {ci}: {compute_score(med_a):.8f}")

    # Upsample to target
    if time.monotonic() < phase1_end - 10:
        fine_a = upsample(coarse_a, target_n)
        sc = compute_score(fine_a)
        report(sc, fine_a, f"coarse->fine n={target_n}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # ==================================================================
    # Phase 2: Direct optimization at target length (~30% of time)
    # ==================================================================
    phase2_end = t_start + 0.65 * effective_budget
    print(f"\n=== Phase 2: Direct Fine Optimization (deadline t={phase2_end-t_start:.0f}s) ===")

    # Start from best available at target_n
    if len(best_a) != target_n:
        start_fine = upsample(best_a, target_n)
    else:
        start_fine = best_a.copy()
    
    # Also prepare parent at target_n
    if parent is not None:
        start_parent = upsample(parent, target_n) if len(parent) != target_n else parent.copy()
    else:
        start_parent = None

    fine_candidates = [start_fine]
    if start_parent is not None and not np.allclose(start_fine, start_parent):
        fine_candidates.append(start_parent)

    for ci, a0 in enumerate(fine_candidates):
        if time.monotonic() > phase2_end - 30:
            break
        print(f"  Fine candidate {ci}: starting score={compute_score(a0):.8f}")
        # Multi-beta LSE
        a_cur = a0.copy()
        for beta in [50, 150, 400, 1000, 2500]:
            if time.monotonic() > phase2_end - 15:
                break
            it_rem = max(500, int((phase2_end - time.monotonic()) / max(1, (2500 - beta) / 500) * 10))
            sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=min(8000, it_rem), maxfun=150000)
            a_cur = a_opt
            report(sc, a_opt, f"fine c{ci} beta={beta}")
        
        # Try top-k objective
        if time.monotonic() < phase2_end - 30:
            k_val = max(5, target_n // 50)
            sc, a_opt, _ = run_lbfgs_topk(a_cur, k_val, 20, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"fine c{ci} topk={k_val}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # ==================================================================
    # Phase 3: Local search with peak reduction (~20% of time)
    # ==================================================================
    phase3_end = t_start + 0.85 * effective_budget
    print(f"\n=== Phase 3: Local Search & Peak Reduction ===")

    n_best = len(best_a)
    n_trials = 40
    for trial in range(n_trials):
        if time.monotonic() > phase3_end - 5:
            break
        
        strategy = trial % 5
        t0 = time.monotonic()
        
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak reduction
            a_pert = peak_reduction_step(best_a, n_peaks=3 + trial % 5,
                                         strength=0.01 + 0.03 * rng.random())
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 0.5 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Blend with uniform (encourages flatness)
            mix = 0.05 + 0.15 * rng.random()
            a_pert = (1 - mix) * best_a + mix * np.ones(n_best) * (best_a.mean())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Anti-peak: reduce elements at positions correlated with peak lag
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                # Create a "notch" at the peak-contributing region
                center = (j_start + j_end) / 2
                width = (j_end - j_start) / 2 + 1
                a_pert[j_start:j_end+1] *= (1.0 - 0.05 * np.exp(-0.5 * ((np.arange(j_start, j_end+1) - center) / width)**2))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Short refinement from perturbed point
        beta_use = 300 + int(200 * rng.random())
        sc, a_opt, _ = run_lbfgs(a_pert, beta_use, maxiter=4000, maxfun=80000)
        improved = report(sc, a_opt, f"local #{trial} s{strategy}")
        elapsed = time.monotonic() - t0
        if improved:
            print(f"    (took {elapsed:.1f}s)")

    print(f"  Phase 3 best: {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient polish + final L-BFGS (~15% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Polish ===")
    
    # Subgradient polish
    sc, a_pol = subgrad_polish(best_a, total_iters=3000)
    report(sc, a_pol, "subgrad polish")
    
    # Final high-beta LSE rounds
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        report(sc, a_opt, f"final lbfgs beta={beta}")
    
    # One more subgradient pass
    if time.monotonic() < deadline - 20:
        sc, a_pol2 = subgrad_polish(best_a, total_iters=2000)
        report(sc, a_pol2, "subgrad polish 2")
    
    # Final LSE from polished
    if time.monotonic() < deadline - 10:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "very final lbfgs")

    # ==================================================================
    # Try a different length if time permits
    # ==================================================================
    if time.monotonic() < deadline - 60 and parent is not None:
        alt_n = target_n + 1024 if target_n < 16384 else target_n - 1024
        if alt_n >= 1024:
            print(f"\n  Trying alternative length n={alt_n}")
            a_alt = upsample(best_a, alt_n)
            sc, a_opt, _ = run_lbfgs(a_alt, 200, maxiter=5000, maxfun=100000)
            sc2, a_opt2, _ = run_lbfgs(a_opt, 1000, maxiter=5000, maxfun=100000)
            report(sc2, a_opt2, f"alt n={alt_n}")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
