def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear autocorrelation via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    n = f.size
    c = _autocorr(f)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _weighted_gradient(p, f):
    """Gradient of weighted sum of c with weights p, w.r.t. f."""
    n = f.size
    L = p.size
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _interp_to_length(f, n):
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 1:
        return _normalize(np.full(n, 1.0 / m))
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _fourier_seed(n, rng, n_modes=8):
    """Generate a smooth non-negative seed using low-frequency Fourier modes."""
    x = np.linspace(0.0, n - 1, n, dtype=np.float64)
    t = x / max(1, n - 1)
    log_f = np.zeros(n, dtype=np.float64)
    for m in range(1, n_modes + 1):
        amp = float(rng.uniform(-0.8, 0.8)) / m
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        log_f += amp * np.cos(2.0 * np.pi * m * t + phase)
    log_f += float(rng.uniform(-0.3, 0.3))
    f = np.exp(log_f - log_f.max())
    return _normalize(f)


def _beta_seed(n, rng):
    """Beta-distribution shaped seed."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = float(rng.uniform(0.5, 4.0))
    b = float(rng.uniform(0.5, 4.0))
    f = (t + 0.01) ** (a - 1) * (1.0 - t + 0.01) ** (b - 1)
    return _normalize(f)


def _power_seed(n, rng):
    """Power-law shaped seed: a[i] ~ (i+1)^alpha * (n-i)^beta."""
    i = np.arange(n, dtype=np.float64)
    alpha = float(rng.uniform(-1.0, 2.0))
    beta = float(rng.uniform(-1.0, 2.0))
    f = (i + 1.0) ** alpha * (n - i) ** beta
    return _normalize(f)


def _hard_topk_gradient(f, c, K=10):
    """Compute gradient using hard top-K lags (not softmax)."""
    n = f.size
    L = c.size
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - K)[L - K:]
    # Equal weight on top-K
    p = np.zeros(L, dtype=np.float64)
    p[idx] = 1.0 / float(len(idx))
    g = _weighted_gradient(p, f)
    return g, idx

    n = f.size
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    print(f"[{tag}] start score={best_score:.6f}")

    deadline = time.monotonic() + float(budget_s)
    step_size = 0.3
    no_accept = 0
    it = 0

    # Adaptive K for top lags
    if n > 5000:
        K = 15
    elif n > 2000:
        K = 12
    else:
        K = 10

    last_polish = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        maxc = float(c.max())

        # Compute hard top-K gradient
        g, active_idx = _hard_topk_gradient(f, c, K)
        g_centered = g - float(g.mean())
        g_range = float(g_centered.max() - g_centered.min())

        if g_range < 1e-14:
            no_accept += 1
            if no_accept > 50:
                break
            continue

        # Try multiple step sizes
        best_mc = maxc
        best_f_new = None
        best_c_new = None

        # Decreasing step sizes
        for ss in (step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1):
            # Log-domain step (multiplicative)
            arg = -ss * g_centered / g_range
            np.clip(arg, -5.0, 5.0, out=arg)
            f_new = f * np.exp(arg)
            f_new = _normalize(f_new)
            c_new = _autocorr(f_new)
            mc = float(c_new.max())
            if mc < best_mc:
                best_mc = mc
                best_f_new = f_new
                best_c_new = c_new

            # Euclidean projected step
            f_new2 = f - ss * g_centered / g_range
            f_new2 = _project_simplex(f_new2)
            c_new2 = _autocorr(f_new2)
            mc2 = float(c_new2.max())
            if mc2 < best_mc:
                best_mc = mc2
                best_f_new = f_new2
                best_c_new = c_new2

        if best_f_new is not None and best_mc < maxc - 1e-14:
            f = best_f_new
            c = best_c_new
            maxc = best_mc
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] new best: {best_score:.6f} at iter {it}")
            step_size = min(0.5, step_size * 1.02)
        else:
            no_accept += 1
            step_size = max(0.01, step_size * 0.8)

            # Periodic coordinate polish
            if no_accept > 20 or (time.monotonic() - last_polish > 3.0):
                fp, sp, cp = _coordinate_polish(f, c, min(2.0, max(0.5, 0.05 * budget_s)), rng)
                if sp < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    no_accept = 0
                    if sp < best_score:
                        best_score = sp
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] polish improved: {best_score:.6f}")
                else:
                    if no_accept > 60:
                        # Perturb from best
                        r = float(rng.random())
                        if r < 0.5:
                            sigma = float(rng.uniform(0.02, 0.5))
                            f = best_f * np.exp(sigma * rng.standard_normal(n))
                        elif r < 0.8:
                            alpha = float(rng.uniform(0.05, 0.3))
                            noise = rng.random(n)
                            noise /= noise.sum()
                            f = (1 - alpha) * best_f + alpha * noise
                        else:
                            # Add Fourier mode perturbation
                            t = np.linspace(0, 1, n)
                            m = int(rng.integers(1, min(15, n // 10)))
                            amp = float(rng.uniform(0.05, 0.3))
                            phase = float(rng.uniform(0, 2 * np.pi))
                            pert = 1 + amp * np.cos(2 * np.pi * m * t + phase)
                            f = best_f * pert
                        f = _normalize(f)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        step_size = 0.2
                        no_accept = 0
                last_polish = time.monotonic()

        if it % 5000 == 0:
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={2.0*n*maxc:.6f} ss={step_size:.3f}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()
    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _coordinate_polish(f0, c0, budget_s, rng):
    """Local search by exact single mass transfers between coordinates."""
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0.0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max()), c

    L = 2 * n - 1
    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = _autocorr(f)
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)

    maxc = float(np.max(c))
    best_f = f.copy()
    best_mc = maxc
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0
    move_count = 0

    D = min(14, n // 10 + 2)
    R = min(14, n // 10 + 2)
    activeK = max(20, min(L, n // 6))
    fracs = (0.02, 0.06, 0.15, 0.40)

    while time.monotonic() < deadline and no_improve < 15:
        maxc = float(np.max(c))

        # Active top lags
        if activeK >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, L - activeK)[L - activeK:]
        p = np.zeros(L, dtype=np.float64)
        p[active] = 1.0 / float(len(active))
        g = _weighted_gradient(p, f)

        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if len(donors) < 2 or len(receivers) < 2:
            no_improve += 1
            if no_improve >= 6:
                break
            continue

        donor_units = []
        for i in donors:
            u = np.zeros(L, dtype=np.float64)
            u[i:i + n] = -f
            donor_units.append((i, u))

        recv_units = []
        for j in receivers:
            u = np.zeros(L, dtype=np.float64)
            u[j:j + n] = f
            recv_units.append((j, u))

        best_mc_this = maxc
        best_move = None

        for i, ui in donor_units:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            for j, uj in recv_units:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0.0:
                        continue
                    np.add(ui, uj, out=delta)
                    delta *= (2.0 * dv)
                    q = dv * dv
                    delta[2 * j] += q
                    delta[2 * i] += q
                    delta[i + j] -= 2.0 * q
                    np.add(c, delta, out=total)
                    mc = float(np.max(total))
                    if mc < best_mc_this:
                        best_mc_this = mc
                        best_move = (i, j, dv, delta.copy())

        # Random fallback
        if best_move is None:
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                frac = float(rng.uniform(0.01, 0.3))
                dv = float(f[i]) * frac
                delta.fill(0.0)
                delta[i:i + n] -= f
                delta[j:j + n] += f
                delta *= (2.0 * dv)
                q = dv * dv
                delta[2 * j] += q
                delta[2 * i] += q
                delta[i + j] -= 2.0 * q
                np.add(c, delta, out=total)
                mc = float(np.max(total))
                if mc < best_mc_this:
                    best_mc_this = mc
                    best_move = (i, j, dv, delta.copy())

        if best_move is None:
            no_improve += 1
            continue

        i, j, dv, d_saved = best_move
        f[i] -= dv
        f[j] += dv
        c = c + d_saved
        move_count += 1
        no_improve = 0

        cur_score = float(2.0 * n * best_mc_this)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        if move_count % 40 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            maxc = float(c.max())
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    if c.size:
        mc = float(c.max())
        sc = float(2.0 * n * mc)
        if sc <= best_score:
            return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    return bf, float(2.0 * n * float(bc.max())), bc


def _coarse_fourier_optimize(n, budget_s, rng, n_modes=12):
    """
    Coarse optimization using Fourier parameterization.
    Parameterize log(a[i] * n) as a sum of cos/sin modes and optimize with Nelder-Mead.
    """
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # Precompute basis
    basis = []
    for m in range(1, n_modes + 1):
        basis.append(np.cos(2.0 * np.pi * m * t))
        basis.append(np.sin(2.0 * np.pi * m * t))
    basis = np.array(basis)  # shape (2*n_modes, n)
    dim = 2 * n_modes

    def objective(params):
        log_f = params @ basis  # shape (n,)
        f = np.exp(log_f - log_f.max())
        f = _normalize(f)
        c = _autocorr(f)
        return float(2.0 * n * c.max())

    # Initial parameters: small random
    x0 = rng.standard_normal(dim) * 0.1
    try:
        res = minimize(objective, x0, method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8,
                                'adaptive': True})
        x_opt = res.x
    except Exception:
        x_opt = x0

    log_f = x_opt @ basis
    f = np.exp(log_f - log_f.max())
    f = _normalize(f)
    sc = _fast_score(f)
    print(f"  [coarse_fourier] n={n}, n_modes={n_modes}, score={sc:.6f}")
    return f, sc


def _try_lengths(small_length, rng, time_budget, initial_f=None):
    """Try optimizing at a smaller length for quick exploration."""
    f, sc = _coarse_fourier_optimize(small_length, time_budget, rng, n_modes=10)
    return f, sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    total_deadline = total_start + max(120.0, float(budget_s)) - 30.0

    # Get initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.maximum(f0, 0.0, out=f0)
    f0 = _normalize(f0)
    n0 = f0.size
    print(f"Initial sequence length: {n0}, initial score: {_fast_score(f0):.6f}")

    best_f = f0.copy()
    best_score = _fast_score(best_f)

    # ========== Phase 1: Coarse multi-resolution search ==========
    print("\n=== Phase 1: Coarse exploration ===")
    coarse_

    # Try different target lengths
    target_lengths = [6121]
    if n0 > 3000:
        target_lengths.append(8000)
    if time.monotonic() < total_deadline - 800:
        target_lengths.append(10000)

    # Coarse optimization at small resolution
    coarse_results = []
    for n_coarse in [256, 512, 1024]:
        if time.monotonic() > total_deadline - 600:
            break
        t_start = time.monotonic()
        f_coarse, sc = _coarse_fourier_optimize(n_coarse, 15.0, rng, n_modes=12)
        coarse_results.append((sc, f_coarse))
        print(f"  Coarse n={n_coarse}: score={sc:.6f} (time={time.monotonic()-t_start:.1f}s)")

    # Also try from the given sequence downsampled
    for n_coarse in [512, 1024]:
        if time.monotonic() > total_deadline - 600:
            break
        f_d = _interp_to_length(f0, n_coarse)
        # Optimize from this starting point
        f_opt, sc = _refine_gradient(f_d, 10.0, rng, tag=f"coarse_{n_coarse}_from_init")
        coarse_results.append((sc, f_opt))
        print(f"  Coarse n={n_coarse} from init: score={sc:.6f}")

    # ========== Phase 2: Upsample and refine at target length ==========
    print(f"\n=== Phase 2: Full-size refinement ===")

    # Build candidate pool for full-size optimization
    candidates = []
    # Original
    candidates.append(("init", f0.copy(), best_score))
    # From coarse results
    for n_tgt in target_lengths:
        if time.monotonic() > total_deadline - 400:
            break
        for sc_c, f_c in coarse_results[:4]:
            f_up = _interp_to_length(f_c, n_tgt)
            sc_up = _fast_score(f_up)
            candidates.append((f"coarse_{f_c.size}_->_{n_tgt}", f_up, sc_up))
        # From different seeds
        for i in range(3):
            f_s = _fourier_seed(n_tgt, rng, n_modes=15)
            sc_s = _fast_score(f_s)
            candidates.append((f"seed_{i}_{n_tgt}", f_s, sc_s))
            f_b = _beta_seed(n_tgt, rng)
            sc_b = _fast_score(f_b)
            candidates.append((f"beta_{i}_{n_tgt}", f_b, sc_b))

    # Sort by score
    candidates.sort(key=lambda x: x[2])
    print(f"Candidates: {len(candidates)}")
    for name, _, sc in candidates[:8]:
        print(f"  {name}: {sc:.6f}")

    # ========== Phase 3: Full-size gradient refinement ==========
    print(f"\n=== Phase 3: Gradient refinement ===")

    # Time allocation
    remaining = total_deadline - time.monotonic()
    # Reserve time for final polish
    polish_reserve = min(200.0, max(60.0, 0.12 * remaining))
    main_budget = remaining - polish_reserve - 30.0

    # Optimize top candidates
    top_k = min(5, len(candidates))
    weights = [0.45, 0.25, 0.15, 0.10, 0.05][:top_k]

    for idx in range(top_k):
        if time.monotonic() > total_deadline - polish_reserve - 30:
            break
        name, f_c, sc_c = candidates[idx]
        n_c = f_c.size
        alloc = main_budget * weights[idx]
        if idx == 0:
            alloc = min(alloc, main_budget * 0.5)
        alloc = max(20.0, alloc)

        print(f"\n--- Refining candidate '{name}' n={n_c} alloc={alloc:.0f}s current_best={best_score:.6f} ---")
        f_opt, sc_opt = _refine_gradient(f_c, alloc, rng, tag=f"main_{idx}_{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  >>> New best: {best_score:.6f}")

    # ========== Phase 4: Coordinate polish ==========
    print(f"\n=== Phase 4: Coordinate polish ===")
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 30:
        polish_time = min(remaining, polish_reserve)
        print(f"Polish budget: {polish_time:.0f}s")
        bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), polish_time, rng)
        if bs < best_score:
            best_score = bs
            best_f = bf.copy()
            print(f"  >>> After polish: {best_score:.6f}")

    # ========== Phase 5: Final gradient pass if time remains ==========
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 60:
        grad_time = min(remaining, 120.0)
        print(f"\n=== Phase 5: Final gradient pass ({grad_time:.0f}s) ===")
        f_opt, sc_opt = _refine_gradient(best_f, grad_time, rng, tag="final_grad")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  >>> After final grad: {best_score:.6f}")

    # ========== Final validation ==========
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.maximum(best_f, 0.0, out=best_f)
    np.minimum(best_f, 1000.0, out=best_f)
    s = best_f.sum()
    if s < 0.01:
        best_f = np.ones(n0, dtype=np.float64) / n0
    else:
        best_f = best_f / s * math.sqrt(2.0 * best_f.size)  # Scale so score is well-defined

    # Actually, let's keep sum = 1 for the internal score, but return as-is
    # The evaluator handles its own scaling
    best_f = _normalize(best_f) * math.sqrt(2.0 * best_f.size)
    # This makes sum_a = sqrt(2n), so score = 2n * max_b / (2n) = max_b
    # Actually let's just return normalized and let the evaluator handle it.
    best_f = _normalize(best_f)

    out = [float(x) for x in best_f]
    internal_score = _fast_score(best_f)
    print(f"\nFINAL internal score = {internal_score:.6f}")
    print(f"FINAL target = {TARGET_SCORE}")

    # Try official evaluation
    try:
        official = evaluate_sequence(out)
        if np.isfinite(official):
            print(f"Official evaluate_sequence = {official:.6f}")
    except Exception:
        pass

    total_time = time.monotonic() - total_start
    print(f"Total time: {total_time:.1f}s")
    return out
