import time


def fast_self_conv(a):

    """Compute softmax weights and LSE value (numerically stable)."""
    c_shifted = (c - c.max()) / T
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()
    LSE = c.max() + T * np.log(exp_c.sum())
    return w, LSE


def compute_lse_grad(a, w):
    """
    Gradient of LSE(conv(a,a)/T) w.r.t. a.
    g_i = 2 * sum_j a[j] * w[i+j]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    M = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, M)
    A = np.fft.rfft(a, M)
    R = np.fft.irfft(W * np.conj(A), M)[:n]
    g = 2.0 * R
    return g


def full_objective_grad(a, g_lse, LSE):
    """
    Full gradient of 2n*LSE/S^2 w.r.t. a.
    = (2n/S^2)*g_lse - (4n*LSE/S^3)*a
    """
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    n = len(a)
    grad = (2.0 * n) * (g_lse / S2 - 2.0 * LSE / S3 * a)
    return grad


def adam_update(a, g, m, v, t, lr):
    """Adam update in place."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)


def normalize_sum(a, target):
    """Normalize a in-place to have the given sum, clip to [0, 1000]."""
    s = a.sum()
    if s < 1e-15:
        a[:] = target / len(a)
        return
    a *= target / s
    np.clip(a, 0.0, 1000.0, out=a)


def make_init(n, kind, rng):
    """Generate initialization of given type."""
    x = np.linspace(0, 1, n)
    if kind == 0:
        return np.ones(n)
    elif kind == 1:
        return 1.0 + 0.4 * np.cos(2.0 * np.pi * x)
    elif kind == 2:
        return 1.0 + 0.6 * np.cos(4.0 * np.pi * x)
    elif kind == 3:
        return 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x) + 0.2 * np.cos(2.0 * np.pi * 5.0 * x)
    elif kind == 4:
        return np.clip(0.3 + 0.7 * rng.random(n), 0.01, 10.0)
    elif kind == 5:
        a = np.ones(n)
        a[1::2] = 0.4
        return a
    elif kind == 6:
        a = np.ones(n)
        a[1::3] = 0.3
        a[2::3] = 0.7
        return a
    elif kind == 7:
        return 1.0 + 0.5 * (2.0 * x - 1.0)
    elif kind == 8:
        mid = n // 2
        d = np.abs(np.arange(n) - mid)
        return np.clip(1.0 + 0.5 * np.cos(np.pi * d / mid), 0.01, 10.0)
    elif kind == 9:
        # Sine-squared modulated
        return np.clip(0.5 + 0.5 * np.sin(np.pi * x) ** 2, 0.01, 10.0)
    else:
        return np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, 10.0)


def lse_optimize(a0, end_time, rng, init_lr, T_start, T_end, patience_init):
    """
    Main LSE-based optimization loop with temperature annealing and restarts.
    Returns (best_score, best_array).
    """
    n = a0.size
    target_sum = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    normalize_sum(a, target_sum)

    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    patience = patience_init
    total_steps = 0

    while time.monotonic() < end_time:
        # Compute current state
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(best_a, 0.01, 1000.0)
            normalize_sum(a, target_sum)
            c = fast_self_conv(a)
            S = float(a.sum())

        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Restart if stuck
        if no_imp >= patience:
            sc = 0.03 + 0.07 * rng.random()
            a = best_a * (1.0 + sc * rng.standard_normal(n))
            np.clip(a, 0.0, 1000.0, out=a)
            normalize_sum(a, target_sum)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            patience = max(40, patience // 2)
            T_start = max(T_end * 2, T_start * 0.5)
            continue

        # Temperature schedule: geometric annealing based on steps since restart
        step_frac = min(t / max(patience * 3, 1), 1.0)
        T = T_start * (T_end / T_start) ** step_frac
        T = max(T, 1e-8)

        # Compute gradient
        w, LSE = lse_weights(c, T)
        g_lse = compute_lse_grad(a, w)
        grad = full_objective_grad(a, g_lse, LSE)

        t += 1
        total_steps += 1
        adam_update(a, grad, m, v, t, init_lr)
        normalize_sum(a, target_sum)

    return best_score, best_a


def lbfgs_polish(a0, end_time, T):
    """
    Final polish using scipy L-BFGS-B on the LSE approximation.
    Returns (best_score, best_array).
    """
    from scipy.optimize import minimize

    n = a0.size
    target_sum = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    normalize_sum(a, target_sum)

    def objective(x):
        xx = np.clip(x, 0.0, 1000.0)
        ss = xx.sum()
        if ss < 1e-10:
            return 1e10
        cc = fast_self_conv(xx)
        w, LSE = lse_weights(cc, T)
        return 2.0 * n * LSE / (ss * ss)

    def gradient(x):
        xx = np.clip(x, 0.0, 1000.0)
        ss = xx.sum()
        if ss < 1e-10:
            return np.zeros_like(x)
        cc = fast_self_conv(xx)
        w, LSE = lse_weights(cc, T)
        g_lse = compute_lse_grad(xx, w)
        return full_objective_grad(xx, g_lse, LSE)

    x0 = a.copy()
    best_score = compute_score(n, fast_self_conv(a), float(a.sum()))
    best_a = a.copy()

    try:
        result = minimize(objective, x0, method='L-BFGS-B', jac=gradient,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'maxls': 20, 'ftol': 1e-12, 'gtol': 1e-8})
        if result.fun < best_score:
            best_a = np.clip(result.x, 0.0, 1000.0)
            normalize_sum(best_a, target_sum)
            c = fast_self_conv(best_a)
            best_score = compute_score(n, c, float(best_a.sum()))
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    # Second polish with lower T
    T2 = max(T * 0.1, 1e-6)

    def objective2(x):
        xx = np.clip(x, 0.0, 1000.0)
        ss = xx.sum()
        if ss < 1e-10:
            return 1e10
        cc = fast_self_conv(xx)
        w, LSE = lse_weights(cc, T2)
        return 2.0 * n * LSE / (ss * ss)

    def gradient2(x):
        xx = np.clip(x, 0.0, 1000.0)
        ss = xx.sum()
        if ss < 1e-10:
            return np.zeros_like(x)
        cc = fast_self_conv(xx)
        w, LSE = lse_weights(cc, T2)
        g_lse = compute_lse_grad(xx, w)
        return full_objective_grad(xx, g_lse, LSE)

    try:
        result2 = minimize(objective2, best_a, method='L-BFGS-B', jac=gradient2,
                           options={'maxiter': 300, 'maxfun': 1000,
                                    'maxls': 30, 'ftol': 1e-14, 'gtol': 1e-10})
        if result2.fun < best_score:
            best_a = np.clip(result2.x, 0.0, 1000.0)
            normalize_sum(best_a, target_sum)
            c = fast_self_conv(best_a)
            best_score = compute_score(n, c, float(best_a.sum()))
    except Exception as e:
        print(f"  L-BFGS polish2 error: {e}")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses LSE (log-sum-exp) smooth gradient with Adam, temperature annealing,
    multi-scale exploration, and L-BFGS final polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 5.0

    best_score = np.inf
    best_a = None
    best_n = 0

    # Get parent sequence if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                parent = np.clip(parent, 0.0, 1000.0)
                c_parent = fast_self_conv(parent)
                s_parent = compute_score(parent.size, c_parent, float(parent.sum()))
                print(f"Parent: n={parent.size}, score={s_parent:.6f}")
                best_score = s_parent
                best_a = parent.copy()
                best_n = parent.size
    except Exception as e:
        print(f"Parent loading error: {e}")

    # Phase 1: Multi-scale exploration (35% of time)
    p1_end = start + 0.35 * (total - 5.0)
    print(f"\n=== Phase 1: Exploration (until {p1_end - start:.0f}s) ===")

    # n values to try: parent size and powers of 2
    n_candidates = []
    if parent is not None and parent.size > 512:
        n_candidates.append(parent.size)
    for n_try in [1024, 2048, 4096, 8192]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    # Sort by size, try smaller first (faster)
    n_candidates.sort()

    allocations = {}
    # Allocate time: more time to parent n if available
    if parent is not None:
        allocations[parent.size] = 0.35
    remaining = 1.0 - allocations.get(parent.size, 0.0)
    n_other = [n for n in n_candidates if n != (parent.size if parent is not None else -1)]
    n_other.sort()
    if n_other:
        per_other = remaining / len(n_other)
        for n_try in n_other:
            allocations[n_try] = per_other

    inits_per_n = {
        0: [0, 1, 2, 3],
        1: [0, 4, 5],
        2: [0, 1, 4, 6, 7, 8, 9, 10]
    }

    for n_try in n_candidates:
        if time.monotonic() >= p1_end:
            break
        t_alloc = allocations.get(n_try, 0.1) * (p1_end - start)
        p1_n_end = min(start + t_alloc, p1_end)

        # Determine which inits to try for this n
        if n_try == 1024:
            kinds = inits_per_n[0]
        elif n_try == 2048:
            kinds = inits_per_n[1]
        else:
            kinds = inits_per_n[2]

        per_init_time = (p1_n_end - start) / max(len(kinds), 1)

        for kind in kinds:
            if time.monotonic() >= p1_n_end:
                break
            a0 = make_init(n_try, kind, rng)
            t_end = min(start + per_init_time * (kinds.index(kind) + 1), p1_n_end)

            T_start = 0.5 * (a0 ** 2).max() if a0.size > 0 else 1.0
            T_start = max(T_start, 0.1)
            T_end = T_start * 0.01

            sc, aa = lse_optimize(a0, t_end, rng,
                                  init_lr=0.1,
                                  T_start=T_start,
                                  T_end=T_end,
                                  patience_init=200)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
                best_n = n_try
                print(f"  n={n_try}, kind={kind}: {sc:.6f} (NEW BEST)")
            else:
                print(f"  n={n_try}, kind={kind}: {sc:.6f}")

    print(f"\nPhase 1 best: {best_score:.6f} at n={best_n}")

    # Phase 2: Main optimization with LSE (45% of time)
    p2_end = start + 0.80 * (total - 5.0)
    print(f"\n=== Phase 2: LSE Optimization (until {p2_end - start:.0f}s) ===")

    if best_a is not None:
        n_opt = best_n
        # Determine T from current peak structure
        c_cur = fast_self_conv(best_a)
        peak_val = float(c_cur.max())
        mean_val = float(c_cur.mean())
        T_start_2 = max(0.1 * (peak_val - mean_val), 0.01)
        T_end_2 = max(0.01 * (peak_val - mean_val), 0.001)

        sc, aa = lse_optimize(best_a, p2_end, rng,
                              init_lr=0.03,
                              T_start=T_start_2,
                              T_end=T_end_2,
                              patience_init=400)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            best_n = n_opt
        print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Fine polish with very low temperature (10% of time)
    p3_end = deadline - 120.0  # Leave 120s for L-BFGS
    print(f"\n=== Phase 3: Fine Polish (until {p3_end - start:.0f}s) ===")

    if best_a is not None:
        c_cur = fast_self_conv(best_a)
        peak_val = float(c_cur.max())
        mean_val = float(c_cur.mean())
        T_start_3 = max(0.05 * (peak_val - mean_val), 0.001)
        T_end_3 = max(0.005 * (peak_val - mean_val), 0.0001)

        sc, aa = lse_optimize(best_a, p3_end, rng,
                              init_lr=0.005,
                              T_start=T_start_3,
                              T_end=T_end_3,
                              patience_init=600)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            best_n = len(aa)
        print(f"Phase 3 best: {best_score:.6f}")

    # Phase 4: L-BFGS final polish
    print(f"\n=== Phase 4: L-BFGS Polish ===")
    if best_a is not None and time.monotonic() < deadline - 10.0:
        c_cur = fast_self_conv(best_a)
        peak_val = float(c_cur.max())
        mean_val = float(c_cur.mean())
        T_lbfgs = max(0.02 * (peak_val - mean_val), 0.001)

        sc, aa = lbfgs_polish(best_a, deadline - 5.0, T=T_lbfgs)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            best_n = len(aa)
        print(f"Phase 4 (L-BFGS) best: {best_score:.6f}")

    # Final verification and output
    if best_a is None:
        best_a = np.ones(2048) / 2048
        best_n = 2048

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception as e:
        print(f"Verification error: {e}")

    elapsed = time.monotonic() - start
    print(f"\nFINAL: internal={best_score:.6f}, verified={verified:.6f}, "
          f"n={best_n}, time={elapsed:.1f}s")

    return out
