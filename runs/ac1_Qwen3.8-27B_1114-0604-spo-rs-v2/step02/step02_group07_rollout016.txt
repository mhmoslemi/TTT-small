    # This is the cross-correlation of w (length 2n-1) and a (length n).
    # We can compute it via FFT: pad a to length >= 2n-1, compute FFT, multiply by conj(FFT(w)), IFFT.
    
    # Actually, a simpler way: b = correlate(w, a) in numpy terms.
    # np.correlate(w, a, 'full') gives the cross-correlation.
    # But that might be slow. Let's use FFT.
    
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    W = np.fft.rfft(wts, M_fft)  # wts has length k_act (sparse in the full 2n-1 space)
    
    # Wait, wts is only defined on the top-k indices, not on the full convolution.
    # Let me create the full weight vector.
    
    # Actually, let me reconsider. The weights w_j are only non-zero for the top-k indices.
    # So dM/da[t] = 2 * sum_{j in top-k} w_j * a[j-t] * I(0 <= j-t < n)
    
    # This is a sparse sum. For each top-k index j, we add w_j * a[j-t] to position t.
    # j-t ranges from 0 to n-1, so t ranges from j-(n-1) to j.
    # But t must also be in [0, n-1].
    # So for each j: t ranges from max(0, j-n+1) to min(j, n-1).
    
    # This is O(k*n) which for k=32 and n=7000 is about 224K operations per gradient step.
    # That's fine.
    
    dM_da = np.zeros(n)
    for idx in range(k_act):
        wj = float(wts[idx])
        if wj < 1e-15:
            continue
        j = int(part[idx])
        # t ranges from max(0, j-n+1) to min(j, n-1)
        t_start = max(0, j - n + 1)
        t_end = min(j, n - 1)
        # dM_da[t] += 2 * wj * a[j-t]
        # j-t ranges from j-t_end to j-t_start
        # When t = t_start: j-t = j - t_start
        # When t = t_end: j-t = j - t_end
        dM_da[t_start:t_end+1] += 2.0 * wj * a[j - np.arange(t_start, t_end+1)]
    
    # The objective is F = 2n * M / S^2
    # dF/da[t] = 2n * (dM/da[t] * S^2 - M * 2*S * 1) / S^4
    #          = 2n * (dM_da[t] / S^2 - 2*M / S^3)
    
    grad = 2.0 * n * (dM_da / S2 - 2.0 * M / (S2 * S))
    return grad, M

def optimize_phase(a, end_time, rng, lr_init, topk_init, topk_max, patience, temp_init):
    """
    Run optimization phase with annealing.
    """
    n = a.size
    a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    
    m = np.zeros(n)  # Adam moment 1
    v = np.zeros(n)  # Adam moment 2
    t = 0
    lr = lr_init
    temp = temp_init
    topk = topk_init
    no_imp = 0
    steps = 0
    
    # Annealing schedule
    total_steps_est = 5000  # rough estimate
    step_count = 0
    
    while time.monotonic() < end_time:
        S = float(a.sum())
        if S < 1e-10:
            a += 0.1
            S = float(a.sum())
        
        c = fast_self_conv(a)
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= patience:
            # Restart from best with perturbation
            scale = 0.01 + 0.05 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.01
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.0001, lr * 0.3)
            temp *= 0.5
            no_imp = 0
            patience = max(50, patience // 2)
            continue
        
        # Compute gradient
        g, M = soft_max_gradient(a, c, topk, temp)
        
        # Adam update
        t += 1
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.clip(a, 0.0, 1000.0)
        
        step_count += 1
        
        # Annealing
        if step_count % 100 == 0:
            # Gradually increase topk
            topk = min(topk_max, topk + 1)
            # Decrease temperature
            temp *= 0.98
            # Slightly decrease learning rate
            lr *= 0.995
    
    return best_score, best_a

def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Linear ramps (non-palindromic)
    inits.append(1.0 + 0.5 * (2*x - 1))  # increasing ramp
    inits.append(1.0 - 0.5 * (2*x - 1))  # decreasing ramp
    inits.append(1.0 + 0.3 * (2*x - 1))  # mild ramp
    
    # Quadratic (U-shaped, palindromic)
    inits.append(1.0 + 0.5 * (2*x - 1)**2)
    
    # Cosine variations
    inits.append(1.0 + 0.4 * np.cos(2*np.pi*x))
    inits.append(1.0 + 0.4 * np.cos(4*np.pi*x))
    inits.append(1.0 + 0.3 * np.sin(2*np.pi*x))  # non-palindromic!
    
    # Asymmetric bump
    inits.append(1.0 + 0.5 * np.exp(-((x - 0.3)/0.2)**2))
    inits.append(1.0 + 0.5 * np.exp(-((x - 0.7)/0.2)**2))
    
    # Random
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    
    # Gapped patterns
    a_g = np.ones(n)
    a_g[1::2] = 0.5
    inits.append(a_g)
    
    return inits

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses adaptive soft-max subgradient with multi-phase annealing.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 3.0
    
    best_score = np.inf
    best_a = None
    
    # Try to use parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                parent = np.clip(parent, 0.0, 1000.0)
    except:
        pass
    
    if parent is not None:
        n = parent.size
        # Validate parent
        c = fast_self_conv(parent)
        S = float(parent.sum())
        s = compute_score(n, c, S)
        print(f"Parent: n={n}, score={s:.6f}")
        if s < best_score and np.isfinite(s):
            best_score = s
            best_a = parent.copy()
    else:
        n = 4096
        print(f"No parent, using n={n}")
    
    n = min(max(n, 256), 16384)
    
    # Phase 1: Quick refinement of parent (25% of time)
    p1_end = start + 0.25 * (total - 3.0)
    if best_a is not None:
        sc, aa = optimize_phase(best_a, p1_end, rng, 
                                lr_init=0.05, topk_init=4, topk_max=32,
                                patience=200, temp_init=0.05)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1: {best_score:.6f} (t={time.monotonic()-start:.0f}s)")
    else:
        # Explore with structured inits
        inits = generate_structured_inits(n, rng)
        per_time = (p1_end - start) / len(inits)
        for i, a0 in enumerate(inits):
            t_end = min(p1_end, start + per_time * (i + 1))
            if time.monotonic() >= p1_end:
                break
            sc, aa = optimize_phase(a0, t_end, rng,
                                    lr_init=0.1, topk_init=3, topk_max=16,
                                    patience=100, temp_init=0.1)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"  Init {i}: {sc:.6f}")
        print(f"Phase 1 (explore): {best_score:.6f}")
    
    # Phase 2: Main refinement with tighter parameters (40% of time)
    p2_end = start + 0.65 * (total - 3.0)
    if best_a is not None and time.monotonic() < p2_end:
        sc, aa = optimize_phase(best_a, p2_end, rng,
                                lr_init=0.01, topk_init=8, topk_max=64,
                                patience=400, temp_init=0.02)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2: {best_score:.6f} (t={time.monotonic()-start:.0f}s)")
    
    # Phase 3: Fine polish (20% of time)
    p3_end = start + 0.85 * (total - 3.0)
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = optimize_phase(best_a, p3_end, rng,
                                lr_init=0.003, topk_init=16, topk_max=128,
                                patience=600, temp_init=0.01)
        if sc < best_score:
            best_score = sc
