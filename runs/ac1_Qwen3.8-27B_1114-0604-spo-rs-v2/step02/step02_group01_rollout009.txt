    Compute L = log(sum exp(c/T)) and its gradient w.r.t. f,

        g_max = np.abs(grad).max()
        if g_max < 1e-30:
            break
        grad = grad / g_max

        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad
        momentum = momentum / (np.abs(momentum).max() + 1e-30)

        f = f * np.exp(-eta * momentum)
        f = np.maximum(f, 1e-300)
        f_sum = f.sum()

            f = best_f.copy()
            c = _autocorr_fft(f)
            break
        f = f / f_sum
        
        # Update convolution
        c = _autocorr_fft(f)
        
        # Track best
        if it % 100 == 0:
            score = _score_fast(f)
            if score < best_score - 1e-9:

                best_f = f.copy()
                last_improve = it
                stuck = 0
            else:

            if stuck > 50:

                stuck = 0
        
        # Cool temperature
        T = max(T_min, T * cool_rate)

        eta = max(0.01, eta * eta_decay)
    
    # Final check
    score = _score_fast(f)
    if score < best_score:

        best_f = f.copy()

def _polish_active_set(f, n, deadline, rng, K=5):
    """
    Polish by focusing only on the top-K active lags.
    Use a simpler gradient: just minimize the sum of the top-K convolution values.
    """
    f = f.copy()
    best_score = _score_fast(f)
    best_f = f.copy()
    
    momentum = np.zeros(n)
    beta = 0.8
    eta = 0.3
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        
        # Find top K lags
        if K < len(c):
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(len(c))
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # Gradient of sum of top-K lags
        grad = np.zeros(n)
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end+1] += 2.0 * f[idx]
        
        # Normalize
        g_max = np.abs(grad).max()
        if g_max < 1e-30:
            break
        grad = grad / g_max
        
        # Momentum + mirror step
        momentum = beta * momentum + (1 - beta) * grad
        momentum = momentum / (np.abs(momentum).max() + 1e-30)
        
        f = f * np.exp(-eta * momentum)
        f = np.maximum(f, 1e-300)
        f_sum = f.sum()
        if f_sum < 1e-300:
            break
        f = f / f_sum
        
        if it % 200 == 0:
            score = _score_fast(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        
        eta = max(0.01, eta * 0.9999)
    
    score = _score_fast(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score
