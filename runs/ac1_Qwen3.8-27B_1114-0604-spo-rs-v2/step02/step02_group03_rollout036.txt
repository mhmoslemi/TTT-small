    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1]) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.ones(n) / n
    return out / s


def _grad_logsumexp(a, c, alpha):
    """
    Gradient of (1/alpha)*log(sum_k exp(alpha*c[k])) w.r.t. a.
    Returns g[i] = 2 * sum_j a[j] * p[j+i] where p is softmax over lags.
    """
    n = a.size
    L = c.size
    z = alpha * (c - c.max())
    np.clip(z, -60.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p = np.ones(L) / L
    else:
        p /= ps
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[j+i]
    # Compute via FFT: conv(a[::-1], p)[i + n - 1]
    M = _fft_size(n + L - 1)
    A = np.fft.rfft(a[::-1], M)
    P = np.fft.rfft(p, M)
    conv = np.fft.irfft(A * P, M)
    return 2.0 * conv[n - 1:2 * n - 1]


def _transfer_eval(c, f, i, j, dv, delta, total):
    """Evaluate max of c after transferring dv from f[i] to f[j]."""
    n = f.size
    L = c.size
    delta.fill(0.0)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= 2.0 * dv
    q = dv * dv
    if 2 * j < L:
        delta[2 * j] += q
    if 2 * i < L:
        delta[2 * i] += q
    if i + j < L:
        delta[i + j] -= 2.0 * q
    np.add(c, delta, out=total)
    return float(total.max())


def _coord_polish(f0, budget, rng):
    """
    Exact coordinate descent: try mass transfers between pairs.
    Returns (f, score, c) of best found.
    """
    f = _normalize(f0).copy()
    n = f.size
    L = 2 * n - 1
    c = _autocorr(f)
    best_f = f.copy()
    best_mc = float(c.max())
    best_score = float(2.0 * n * best_mc)
    deadline = time.monotonic() + budget
    delta = np.zeros(L)
    total = np.zeros(L)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline and no_improve < 30:
        iter_count += 1
        mc = float(c.max())
        peak_lag = int(np.argmax(c))

        # Compute per-position contribution to peak lag
        contrib = np.zeros(n)
        j_lo = max(0, peak_lag - n + 1)
        j_hi = min(n - 1, peak_lag)
        for j in range(j_lo, j_hi + 1):
            other = peak_lag - j
            if j == other:
                contrib[j] = f[j] ** 2
            else:
                contrib[j] = 2.0 * f[j] * f[other]

        improved = False

        # Phase A: targeted transfers (donor from high contrib, receiver from low f)
        if n > 10:
            # Top contributors to peak
            top_idx = np.argsort(contrib)[::-1]
            bottom_idx = np.argsort(f)

            donors = [int(x) for x in top_idx[:8] if f[x] > 1e-10]
            receivers = [int(x) for x in bottom_idx[:8] if x not in set(donors)]

            for i in donors:
                if improved:
                    break
                for j in receivers:
                    for frac in (0.05, 0.15, 0.3, 0.5):
                        dv = f[i] * frac
                        if dv <= 0:
                            continue
                        new_mc = _transfer_eval(c, f, i, j, dv, delta, total)
                        if new_mc < mc - 1e-15:
                            f[i] -= dv
                            f[j] += dv
                            c = c + delta
                            mc = new_mc
                            improved = True
                            no_improve = 0
                            sc = 2.0 * n * mc
                            if sc < best_score:
                                best_score = sc
                                best_f = f.copy()
                            break

        # Phase B: random transfers
        if not improved:
            for _ in range(40):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] < 1e-12:
                    continue
                frac = float(rng.uniform(0.02, 0.5))
                dv = f[i] * frac
                new_mc = _transfer_eval(c, f, i, j, dv, delta, total)
                if new_mc < mc - 1e-15:
                    f[i] -= dv
                    f[j] += dv
                    c = c + delta
                    mc = new_mc
                    improved = True
                    no_improve = 0
                    sc = 2.0 * n * mc
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    break

        if not improved:
            no_improve += 1

        # Periodic re-normalization to avoid drift
        if iter_count % 100 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            mc = float(c.max())
            sc = 2.0 * n * mc
            if sc < best_score:
                best_score = sc
                best_f = f.copy()

    f = _normalize(best_f)
    c = _autocorr(f)
    sc = float(2.0 * n * c.max())
    return f, sc, c


def _gradient_optimize(f0, budget, rng, tag=""):
    """
    Main gradient optimizer using log-sum-exp smoothing with multiplicative updates.
    """
    f = _normalize(f0).copy()
    n = f.size
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    deadline = time.monotonic() + budget

    print(f"[{tag}] start: score={best_score:.6f}, n={n}")

    # Alpha schedule: start moderate, increase over time
    alpha = 20.0
    alpha_max = 2000.0
    lr = 0.5
    lr_min = 0.01
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    no_accept = 0
    last_polish = time.monotonic()
    polish_interval = 5.0
    polish_budget = 3.0

    while time.monotonic() < deadline:
        it += 1
        c = _autocorr(f)
        maxc = float(c.max())
        cur_score = float(2.0 * n * maxc)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            if best_score < 1.5027:
                print(f"[{tag}] NEW BEST below target: {best_score:.8f} at iter {it}")

        # Compute gradient via log-sum-exp
        g = _grad_logsumexp(f, c, alpha)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # Multiplicative (log-domain) step
        step_arg = -lr * momentum / (np.abs(momentum).max() + 1e-30)
        np.clip(step_arg, -30.0, 30.0, out=step_arg)
        f_new = f * np.exp(step_arg)
        f_new = _normalize(f_new)

        # Evaluate
        c_new = _autocorr(f_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        if score_new < cur_score - 1e-14:
            f = f_new
            c = c_new
            maxc = maxc_new
            no_accept = 0
            # Maybe decrease lr slightly for stability
            lr = max(lr_min, lr * 0.999)
        else:
            no_accept += 1
            lr = max(lr_min, lr * 0.95)

            # If stuck, try a larger exploration step
            if no_accept > 20:
                sigma = float(rng.uniform(0.1, 0.8))
                f = _normalize(f * np.exp(sigma * rng.standard_normal(n)))
                c = _autocorr(f)
                maxc = float(c.max())
                no_accept = 0
                lr = max(lr_min, 0.3)

            # Periodic coordinate polish
            if time.monotonic() - last_polish > polish_interval:
                fp, sp, cp = _coord_polish(f, polish_budget, rng)
                if sp < 2.0 * n * maxc:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    if sp < best_score:
                        best_score = sp
                        best_f = f.copy()
                        print(f"[{tag}] polish improved to {best_score:.8f}")
                last_polish = time.monotonic()

        # Sharpen alpha periodically
        if it % 500 == 0:
            alpha = min(alpha_max, alpha * 1.5)

        # Occasional structural perturbation from best
        if no_accept > 80 and it % 50 == 0:
            mode = rng.integers(0, 4)
            if mode == 0:
                # Gaussian multiplicative noise
                sigma = float(rng.uniform(0.05, 0.4))
                f = _normalize(best_f * np.exp(sigma * rng.standard_normal(n)))
            elif mode == 1:
                # Mix with random
                alpha_mix = float(rng.uniform(0.1, 0.5))
                noise = rng.random(n)
                noise /= noise.sum()
                f = _normalize((1 - alpha_mix) * best_f + alpha_mix * noise)
            elif mode == 2:
                # Smooth then roughen
                k = max(3, n // 500)
                kernel = np.ones(k) / k
                smoothed = np.convolve(f, kernel, mode='same')
                sigma = float(rng.uniform(0.05, 0.3))
                f = _normalize(smoothed * np.exp(sigma * rng.standard_normal(n)))
            else:
                # Segment shuffle
                f = _normalize(best_f)
                nseg = int(rng.integers(4, 16))
                seg_size = n // nseg
                perms = rng.permutation(nseg)
                f_new = np.zeros(n)
                for p in range(nseg):
                    lo = p * seg_size
                    hi = lo + seg_size if p < nseg - 1 else n
                    f_new[perms[p] * seg_size:(perms[p] + 1) * seg_size] = f[lo:hi]
                f = _normalize(f_new)

            c = _autocorr(f)
            maxc = float(c.max())
            no_accept = 0
            lr = 0.3
            alpha = max(20.0, alpha * 0.5)

        if it % 5000 == 0:
            print(f"[{tag}] iter={it} best={best_score:.8f} cur={cur_score:.8f} alpha={alpha:.1f} lr={lr:.4f}")

    # Final evaluation
    f = _normalize(best_f)
    c = _autocorr(f)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
    print(f"[{tag}] end: best={best_score:.8f}")
    return _normalize(best_f), float(best_score)


def _make_seeds(f0, n, rng):
    """Create diverse starting points."""
    seeds = []
    f0 = _normalize(f0)

    def add(name, f):
        f = _normalize(_interp(f, n))
        sc = _score(f)
        if np.isfinite(sc):
            seeds.append((sc, name, f))

    def _interp(v, target_n):
        v = _normalize(v)
        m = v.size
        if m == target_n:
            return v
        if m == 1 or target_n == 1:
            return np.full(target_n, v[0]) if m == 1 else np.array([v.sum()])
        x_old = np.linspace(0, m - 1, m)
        x_new = np.linspace(0, target_n - 1, target_n)
        return np.interp(x_new, x_old, v)

    # Given sequence
    add("given", f0)
    # Reversed
    add("reversed", f0[::-1])
    # Smoothed versions
    for win in (5, 21, 101, 501):
        if win < n:
            kernel = np.ones(min(win, n)) / min(win, n)
            add(f"smooth{win}", np.convolve(f0, kernel, mode='same'))
    # Gamma shaped
    for g in (0.5, 0.75, 1.5, 2.0):
        add(f"gamma{g}", np.power(f0 + 1e-15, g))
    # Uniform
    add("uniform", np.ones(n))
    # Fourier-based seeds
    for i in range(8):
        x = np.linspace(0, 1, n)
        u = np.zeros(n)
        for m in (1, 2, 3, 5, 7, 11):
            u += float(rng.uniform(-1, 1)) * np.cos(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
        add(f"fourier{i}", np.exp(0.5 * u))
    # Beta-shaped seeds
    for i in range(6):
        x = np.linspace(0, 1, n)
        a = float(rng.uniform(0.3, 4.0))
        b = float(rng.uniform(0.3, 4.0))
        add(f"beta{i}", (x + 1e-3) ** a * (1 - x + 1e-3) ** b)
    # Raised cosine
    x = np.linspace(0, 1, n)
    add("raised_cos", (1 + np.cos(2 * np.pi * x)) / 2 + 1e-6)
    # Sine squared
    add("sin2", np.sin(np.pi * x) ** 2 + 1e-6)
    # Two peaks
    add("two_peak", np.exp(-50 * (x - 0.2) ** 2) + np.exp(-50 * (x - 0.8) ** 2))
    # Triangular
    add("triangular", 1 - np.abs(2 * x - 1))

    seeds.sort(key=lambda x: x[0])
    return seeds


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(120.0, float(budget_s)) - 30.0

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121)

    if f0.size == 0:
        f0 = np.ones(6121)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = f0.size
    if n < 10:
        f0 = np.ones(4096)
        n = 4096
    f0 = _normalize(f0)

    print(f"Starting with n={n}")
    best_f = f0.copy()
    best_score = _score(f0)
    print(f"Initial score: {best_score:.8f}")

    # Build seed pool
    seeds = _make_seeds(f0, n, rng)
    print(f"Seed pool: {len(seeds)} candidates, best seed score: {seeds[0][0]:.8f}")

    # Phase 1: Optimize top seeds with gradient method
    now = time.monotonic()
    remaining = total_deadline - now - 60.0  # reserve 60s for final polish
    if remaining < 100:
        remaining = total_deadline - now - 10.0

    # Allocate time to top seeds
    n_seeds = min(5, len(seeds))
    weights = [0.45, 0.25, 0.15, 0.10, 0.05][:n_seeds]
    t_cursor = time.monotonic()

    for idx in range(n_seeds):
        if time.monotonic() >= total_deadline - 60.0:
            break
        sc, name, f = seeds[idx]
        remain = total_deadline - time.monotonic() - 50.0
        if remain < 30:
            break
        alloc = min(remain - 10, remaining * weights[idx] / sum(weights[idx:]))
        alloc = max(20.0, alloc)
        print(f"\n=== Gradient optimize seed '{name}' (score={sc:.6f}) budget={alloc:.0f}s ===")
        f_opt, sc_opt = _gradient_optimize(f, alloc, rng, f"seed{idx}_{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 2: Final coordinate polish
    now = time.monotonic()
    polish_time = max(0.0, total_deadline - now - 20.0)
    if polish_time > 30:
        print(f"\n=== Final coordinate polish budget={polish_time:.0f}s ===")
        f_pol, sc_pol, _ = _coord_polish(best_f, polish_time, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"  Polish NEW BEST: {best_score:.8f}")

    # Phase 3: If time remains, one more gradient pass
    now = time.monotonic()
    extra = max(0.0, total_deadline - now - 15.0)
    if extra > 60:
        print(f"\n=== Extra gradient pass budget={extra:.0f}s ===")
        f2, s2 = _gradient_optimize(best_f, extra, rng, "extra")
        if s2 < best_score:
            best_score = s2
            best_f = f2.copy()
            print(f"  Extra NEW BEST: {best_score:.8f}")

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))
    final_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * final_c.max())

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official):
            print(f"Official score: {official:.8f}")
            if official < best_score:
                best_score = float(official)
    except Exception:
        pass

    out = [float(x) for x in best_f]
    print(f"\nFINAL score = {best_score:.8f}")
    return out
