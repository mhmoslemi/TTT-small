"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses L-BFGS-B in log-space with adaptive softmax temperature,
multiplicative weight updates, random restarts, and coarse-to-fine strategy.
"""

import numpy as np
import time
from scipy.optimize import minimize

RNG = np.random.default_rng(2024)

def _next_pow2(x):
    L = 1
    while L < x:
        L <<= 1
    return L

def _autocorr(a, L=None):
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2*n-1]
    return np.maximum(c, 0.0)

def _exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-30:
        return float('inf')
    a = a / s
    n = len(a)
    L = _next_pow2(2 * n - 1)
    c = _autocorr(a, L)
    return float(2.0 * n * c.max())

def _smooth_obj(theta, beta):
    """Smooth objective: 2*n*soft_max(conv(softmax(theta), softmax(theta)))."""
    t = theta - theta.max()
    e = np.exp(t)
    a = e / e.sum()
    n = len(a)
    L = _next_pow2(2 * n - 1)
    c = _autocorr(a, L)
    t_c = beta * (c - c.max())
    w = np.exp(t_c)
    wsum = w.sum()
    if not np.isfinite(wsum) or wsum <= 0:
        return 2.0 * n * c.max()
    return 2.0 * n * (c.max() + np.log(wsum) / beta)

def _smooth_grad(theta, beta):
    """Gradient of smooth objective in log-space."""
    t = theta - theta.max()
    e = np.exp(t)
    a = e / e.sum()
    n = len(a)
    L = _next_pow2(2 * n - 1)
    c = _autocorr(a, L)
    t_c = beta * (c - c.max())
    w = np.exp(t_c)
    wsum = w.sum()
    if not np.isfinite(wsum) or wsum <= 0:
        return np.zeros(n)
    w = w / wsum

    # Cross-correlation g_j = sum_t a_t * w_{j+t}
    L2 = _next_pow2(3 * n)
    A2 = np.fft.rfft(a, L2)
    W2 = np.fft.rfft(w, L2)
    cross = np.fft.irfft(np.conjugate(A2) * W2, L2)
    g = cross[:n]
    mean_g = np.dot(g, a)
    return a * (g - mean_g)

def _run_lbfgsb(a_init, beta, max_iter=300, gtol=1e-5):
    """Run L-BFGS-B optimization from a_init with given beta."""
    n = len(a_init)
    theta = np.log(np.maximum(a_init, 1e-30))
    theta = theta - theta.mean()  # center for numerical stability

    def fun(th):
        return _smooth_obj(th, beta)

    def jac(th):
        return _smooth_grad(th, beta)

    try:
        result = minimize(
            fun, theta, jac=jac, method='L-BFGS-B',
            args=(), options={'maxiter': max_iter, 'gtol': gtol, 'iprint': 0}
        )
        t = result['x'] - result['x'].max()
        e = np.exp(t)
        a_new = e / e.sum()
        a_new = np.maximum(a_new, 0.0)
        return a_new
    except Exception:
        return a_init.copy()

def _mult_update(a, beta, alphas, num_iter=20):
    """Multiplicative weight update iterations."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n)
    best = a.copy()
    best_sc = _exact_score(a)
    curr = a.copy()

    for it in range(num_iter):
        alpha = alphas[it % len(alphas)]
        c = _autocorr(curr, L)
        t_c = beta * (c - c.max())
        w = np.exp(t_c)
        wsum = w.sum()
        if not np.isfinite(wsum) or wsum <= 0:
            break
        w = w / wsum

        A2 = np.fft.rfft(curr, L2)
        W2 = np.fft.rfft(w, L2)
        cross = np.fft.irfft(np.conjugate(A2) * W2, L2)
        g = cross[:n]
        g_max = g.max()
        if g_max <= 1e-30:
            break
        g = g / g_max

        new_a = curr * np.exp(-alpha * g)
        new_a = np.maximum(new_a, 1e-30)
        new_a = new_a / new_a.sum()

        sc = _exact_score(new_a)
        if sc < best_sc:
            best_sc = sc
            best = new_a.copy()
        curr = new_a

    return best, best_sc

def _resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-10)
    return out / out.sum()

def _initial_seq(n, rng, mode):
    t = (np.arange(n, dtype=float) + 0.5) / n
    if mode == 0:
        p = np.ones(n)
        for k in range(1, 16):
            amp = rng.normal(0, 1.0 / (k * 0.6))
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.maximum(p, 0.01)
    elif mode == 1:
        ap = rng.uniform(0.2, 2.5)
        bp = rng.uniform(0.2, 2.5)
        p = np.power(np.clip(t, 1e-6, 1-1e-6), ap) * np.power(1 - np.clip(t, 1e-6, 1-1e-6), bp)
    elif mode == 2:
        K = int(rng.integers(2, 7))
        p = np.zeros(n)
        for _ in range(K):
            center = rng.uniform(0.05, 0.95)
            width = rng.uniform(0.03, 0.4)
            weight = rng.uniform(0.3, 2.0)
            p += weight * np.exp(-((t - center) / width) ** 2)
    elif mode == 3:
        K = int(rng.integers(3, 12))
        edges = np.sort(rng.uniform(0, 1, K - 1))
        edges = np.concatenate([[0], edges, [1]])
        heights = rng.lognormal(0, 0.8, K)
        p = np.zeros(n)
        for i in range(K):
            s = int(edges[i] * n)
            e2 = max(s + 1, int(edges[i+1] * n))
            p[s:e2] = heights[i]
    elif mode == 4:
        cut = rng.uniform(0.25, 0.75)
        ratio = rng.choice([0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0])
        p = np.ones(n)
        p[int(cut * n):] = ratio
    elif mode == 5:
        # Three-level
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        r1 = rng.choice([0.3, 0.5, 0.8])
        r2 = rng.choice([0.3, 0.5, 0.8])
        p = np.ones(n)
        p[:int(c1*n)] = r1
        p[int(c2*n):] = r2
    else:
        p = rng.exponential(1.0, n)
        p = p + rng.exponential(0.5, n)

    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _optimize_at_size(a_init, time_budget, verbose=False):
    """Full optimization at a fixed size: L-BFGS-B phases + mult updates + restarts."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _exact_score(a_init)
    deadline = time.monotonic() + time_budget

    if verbose:
        print(f"  Optimizing n={n}, initial score={best_sc:.8f}")

    # Beta schedule for L-BFGS-B
    beta_schedule = [8, 15, 30, 60, 120, 250, 500, 1000, 2000, 4000]

    # Determine how many beta levels we can afford
    est_per_level = 2.0  # seconds per L-BFGS-B level (rough)
    max_levels = max(3, int((time_budget * 0.6) / est_per_level))
    beta_schedule = beta_schedule[:min(len(beta_schedule), max_levels)]

    for beta in beta_schedule:
        if time.monotonic() > deadline - 10:
            break

        # L-BFGS-B
        a_new = _run_lbfgsb(best_a, beta, max_iter=200, gtol=1e-5)
        sc = _exact_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            if verbose:
                print(f"    LBFGS beta={beta}: {best_sc:.8f}")

        # Multiplicative updates as polish
        alphas = [0.05, 0.08, 0.12, 0.15, 0.2]
        a_mu, sc_mu = _mult_update(best_a, min(beta, 200), alphas, num_iter=15)
        if sc_mu < best_sc:
            best_sc = sc_mu
            best_a = a_mu.copy()
            if verbose:
                print(f"    MultUpd beta={beta}: {best_sc:.8f}")

    # Final high-beta polish
    rem = deadline - time.monotonic()
    if rem > 10:
        a_new = _run_lbfgsb(best_a, 5000, max_iter=100, gtol=1e-6)
        sc = _exact_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a, best_sc
