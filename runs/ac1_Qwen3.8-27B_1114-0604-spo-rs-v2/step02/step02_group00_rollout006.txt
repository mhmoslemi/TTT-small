def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        log_sum = 0.0
    else:
        p = exp_scaled / exp_sum
        log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    # Gradient of L w.r.t. a: g = 2 * cross_corr(a, p)
    # g[i] = 2 * sum_k p[k] * d conv(k)/da[i] = 2 * sum_k p[k] * 2*a[k-i] (valid)
    # = 4 * sum over valid k of p[k] * a[k-i]
    # This is effectively 2 * conv(a, p) but we need to be careful about the linear convolution.
    # Actually: d conv(k)/da[i] = 2*a[k-i] if valid. So d L/da[i] = sum_k p[k] * 2*a[k-i] (valid k)
    # = 2 * (a * p)[i] where * is the appropriate cross-correlation.
    # Let's compute it as: for each k, the contribution to g[i] is p[k] * 2*a[k-i] for valid i.
    # This is the same as 2 * (circ shifted convolution).
    # More efficiently: g = 2 * ifft(fft(a) * conj(fft(p))) -- no, let's just do it with FFT.
    # g[i] = 2 * sum_{k: 0<=k-i<n} p[k] * a[k-i] = 2 * sum_{m: 0<=m<n} p[i+m] * a[m] (where m=k-i)
    # = 2 * cross_corr(p, a)[i]
    # where cross_corr(p, a)[i] = sum_m p[i+m] * a[m]
    # This is the cross-correlation, which can be computed as:
    # cross_corr(p, a) = ifft(conj(fft(a)) * fft(p)) -- need to be careful with linear vs circular.
    
    # Let's use the approach: g[i] = 2 * sum_j a[j] * p[i+j] (where i+j < 2n-1 and i+j >= 0)
    # This is the cross-correlation between a and p, evaluated at lag i.
    # Using FFT: cross_corr = ifft(fft(a, N) * conj(fft(p, N))) with appropriate N.
    # Actually for linear cross-correlation:
    # (a ⊛ p)[i] = sum_j a[j] p[j+i] for i from -(n-1) to 2n-2-n = n-1... 
    
    # Let me just use the direct formula with FFT.
    # We want: g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[j+i] for i = 0, ..., n-1
    # where p[j+i] = 0 if j+i >= 2n-1.
    # This is the cross-correlation of a and p (with p being the "longer" signal of length 2n-1).
    # cross_corr[i] = sum_j a[j] * p[j+i]
    # = (a * p_flipped)[n-1+i] where p_flipped[j] = p[2n-2-j]
    # Hmm, let me just use the approach from the previous code.
    
    # From previous code: g = 2 * conv(a_rev, p)[n-1 : 2n-1]
    # where a_rev = a[::-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


# ---------------------------------------------------------------------------
# ADAM optimizer with blended gradient
# ---------------------------------------------------------------------------

def adam_optimize(a0, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  max_iters=5000, lse_beta=200, patience=500,
                  time_limit=None, t_start=None, normalize_sum=False):
    """
    ADAM optimizer with blended LSE + exact-max gradient.
    Returns (best_score, best_a).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_score = compute_score(a)
    best_a = a.copy()
    best_iter = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break
        
        t += 1
        # Compute LSE gradient (smooth)
        sc_lse, grad_lse = lse_grad(a, lse_beta)
        
        # Compute exact subgradient at the max
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M = float(c[k_star])
        exact_sc = 2.0 * n * M / (S * S)
        grad_exact = subgrad_score(a, k_star)
        
        # Blend: use exact gradient weighted by how sharp the LSE is
        # At high lse_beta, LSE grad ≈ exact grad. At low beta, LSE is smoother.
        # Use a fixed blend: 0.5 * lse + 0.5 * exact
        grad = 0.5 * grad_lse + 0.5 * grad_exact
        
        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate decay
        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        a -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        
        if normalize_sum and S > 1e-12:
            S_new = float(a.sum())
            if S_new > 1e-12:
                a *= S / S_new
                a = np.clip(a, 0.0, 1000.0)
        
        # Check improvement
        if it % 50 == 0 or it < 100:
            sc = compute_score(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                best_iter = it
                no_improve = 0
            else:
                no_improve += 50
        
        if no_improve >= patience:
            break
    
    # Final check
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


# ---------------------------------------------------------------------------
# Peak shaving: directly reduce top-K lags
# ---------------------------------------------------------------------------

def peak_shave(a, top_k=10, strength=0.01, n_iters=200):
    """
    Iteratively reduce the top-K convolution lags by adjusting a.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    S = float(a.sum())
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        
        # Find top K lags
        top_lags = np.argsort(c)[::-1][:top_k]
        avg_c = float(c.sum()) / len(c)
        
        # For each top lag, compute subgradient and reduce
        cumulative_sg = np.zeros(n)
        for k in top_lags:
            if c[k] > avg_c:
                cumulative_sg += subgrad_max(a, int(k))
        
        # Normalize step
        snorm = np.linalg.norm(cumulative_sg)
        if snorm < 1e-15:
            break
        
        # Step size: reduce mass from peak contributors
        S2 = S * S
        M_max = float(c.max())
        step = strength * M_max / (snorm * S)
        step = min(step, 0.1)
        
        a -= step * cumulative_sg
        a = np.clip(a, 0.0, 1000.0)
        
        # Preserve sum approximately
        S_new = float(a.sum())
        if S_new > 1e-12 and S_new < S * 0.5:
            a *= S / S_new
            a = np.clip(a, 0.0, 1000.0)
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


# ---------------------------------------------------------------------------
# L-BFGS with LSE
# ---------------------------------------------------------------------------

def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize."""
    sc, grad = lse_grad(a, beta)
    return sc, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=None, t_start=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    
    # Time-aware maxiter
    if time_limit is not None and t_start is not None:
        remaining = t_start + time_limit - time.monotonic()
        if remaining < 5:
            return compute_score(a0), a0, None
        # Estimate: each function eval ~ n*20 microseconds for FFT
        est_per_eval = n * 20e-6
        maxfun_t = max(100, int(remaining / est_per_eval / 3))
        maxfun = min(maxfun, maxfun_t)
        maxiter = min(maxiter, max(10, maxfun // 20))
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


# ---------------------------------------------------------------------------
# Coordinate descent on impactful coordinates
# ---------------------------------------------------------------------------

def coordinate_descent(a, n_coords=50, n_rounds=5, time_limit=None, t_start=None):
    """
    For each round: find the top n_coords elements by |subgradient|,
    then do 1D line search on each.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for rnd in range(n_rounds):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break
        
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M = float(c[k_star])
        sc = 2.0 * n * M / (S * S)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        
        grad = subgrad_score(a, k_star)
        
        # Pick top coordinates by |gradient|
        top_idx = np.argsort(np.abs(grad))[::-1][:n_coords]
        
        for i in top_idx:
            if time_limit is not None and t_start is not None:
                if time.monotonic() > t_start + time_limit:
                    return best_score, best_a
            
            # 1D line search: try different values of a[i]
            a_i = a[i]
            best_a_i = a_i
            best_sc_i = sc
            
            # Try geometric grid around current value
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                a_try = a[i] * mult
                a_try = np.clip(a_try, 0.0, 1000.0)
                a[i] = a_try
                sc_i = compute_score(a)
                if sc_i < best_sc_i:
                    best_sc_i = sc_i
                    best_a_i = a_try
            a[i] = best_a_i
        
        # Also try zeroing some coordinates
        for i in top_idx[:10]:
            save = a[i]
            a[i] = 0.0
            sc_i = compute_score(a)
            if sc_i < best_score - 1e-12:
                best_score = sc_i
                best_a = a.copy()
            else:
                a[i] = save
    
    return best_score, best_a


# ---------------------------------------------------------------------------
# Build diverse initializations
# ---------------------------------------------------------------------------

def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Parent (resized if needed)
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            # Resize parent to length n
            p = parent.astype(np.float64)
            # Use interpolation
            x_old = np.linspace(0, 1, len(p))
            x_new = np.linspace(0, 1, n)
            p_resized = np.interp(x_new, x_old, p)
            inits.append(np.clip(p_resized, 0.0, 1000.0).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Cosine/sine modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.clip(0.5 + 0.3 * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * np.cos(4.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.7 * np.abs(np.cos(3.0 * np.pi * x)), 0.01, 1000.0))
    inits.append(np.clip(0.2 + 0.6 * np.cos(2.0 * np.pi * x * 7.0), 0.01, 1000.0))
    
    # Quadratic / parabolic
    inits.append(np.clip(0.1 + 0.8 * (2.0 * x - 1.0) ** 2, 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * (1.0 - (2.0 * x - 1.0) ** 2), 0.01, 1000.0))
    
    # Ramp
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))
    
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Block structure
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    # "Sparse" initialization: mostly zeros with a few peaks
    a = np.zeros(n)
    n_peaks = max(3, n // 100)
    peak_positions = rng.choice(n, size=n_peaks, replace=False)
    a[peak_positions] = 1.0
    # Smooth slightly
    kernel = np.ones(5) / 5.0
    a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() > 1e-6:
        inits.append(a.copy())
    
    return inits


# ---------------------------------------------------------------------------
# Simulated annealing for exploration
# ---------------------------------------------------------------------------

def simulated_annealing(a0, n_iters=500, T_start=0.1, T_end=0.001,
                        time_limit=None, t_start=None):
    """
    Simple simulated annealing with Gaussian perturbations.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    S = float(a.sum())
    
    for it in range(n_iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break
        
        # Annealing schedule
        frac = it / max(1, n_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        
        scale = T * S / n
        a_pert = a + scale * np.random.standard_normal(n)
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        sc = compute_score(a_pert)
        sc_cur = compute_score(a)
        
        if sc < sc_cur or np.random.random() < np.exp(-(sc - sc_cur) / max(T, 1e-10)):
            a = a_pert
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_score, best_a

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid search: ADAM + LSE-LBFGS + peak shaving + coordinate descent.
    Multi-length exploration around parent solution.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    best_len = 1024

    def report(sc, a, label):
        nonlocal best_score, best_a, best_len
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_len = len(a)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ==================================================================
    # Phase 1: Refine parent with ADAM (quick, ~25% time)
    # ==================================================================
    phase1_end = t_start + 0.25 * (float(budget_s) - 15.0)
    print(f"{'='*60}")
    print(f"Phase 1: ADAM refinement of parent")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")
    print(f"{'='*60}")

    if parent is not None:
        n_p = len(parent)
        a_p = parent.copy()
        sc_p = compute_score(a_p)
        print(f"  Parent score: {sc_p:.8f}, n={n_p}")
        report(sc_p, a_p, "parent")

        # ADAM with different learning rates
        for lr in [0.005, 0.01, 0.02]:
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            remain = phase1_end - time.monotonic()
            sc, a_out = adam_optimize(a_p, lr=lr, max_iters=3000,
                                       lse_beta=200, patience=400,
                                       time_limit=remain, t_start=t0)
            report(sc, a_out, f"ADAM lr={lr}")
        
        # Also try L-BFGS from parent
        if time.monotonic() < phase1_end - 30:
            for beta in [100, 300]:
                remain = phase1_end - time.monotonic()
                if remain < 10:
                    break
                sc, a_out, _ = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000,
                                          time_limit=remain, t_start=t_start)
                report(sc, a_out, f"LBFGS parent beta={beta}")

    # ==================================================================
    # Phase 2: Multi-length exploration (~35% time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (float(budget_s) - 15.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-length exploration")
    print(f"{'='*60}")

    # Choose lengths to explore
    lengths_to_try = set()
    if parent is not None:
        np_ = len(parent)
        lengths_to_try.add(np_)
        # Nearby lengths
        for delta in [-100, -50, 0, 50, 100, 200]:
            lengths_to_try.add(np_ + delta)
    # Some round numbers
    for n in [4096, 5000, 6000, 7000, 8000, 10000]:
        lengths_to_try.add(n)
    
    # Filter and sort
    lengths_to_try = sorted([n for n in lengths_to_try if 1024 <= n <= 16000])
    # Limit to at most 5 lengths for time management
    if len(lengths_to_try) > 5:
        # Keep parent length and spread others
        if parent is not None:
            np_ = len(parent)
            others = [n for n in lengths_to_try if n != np_]
            # Pick evenly spaced
            step = max(1, len(others) // 4)
            others = others[::step][:4]
            lengths_to_try = sorted([np_] + others)
        else:
            lengths_to_try = lengths_to_try[::max(1, len(lengths_to_try)//5)][:5]
    
    print(f"  Lengths: {lengths_to_try}")
    time_per_length = (phase2_end - time.monotonic()) / max(1, len(lengths_to_try))
    
    for n in lengths_to_try:
        if time.monotonic() > phase2_end:
            break
        print(f"\n  --- n = {n} (time budget={time_per_length:.0f}s) ---")
        inits = build_inits(n, rng, parent)
        
        # Sort: parent-like first
        # Try top 4-6 inits
        n_inits = min(len(inits), max(4, int(time_per_length / 40)))
        inits = inits[:n_inits]
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase2_end - 10:
                break
            remain = phase2_end - time.monotonic()
            if remain < 10:
                break
            
            # Quick ADAM from this init
            bucket = max(2.0, remain / (n_inits - i))
            sc, a_out = adam_optimize(a0, lr=0.01, max_iters=1500,
                                       lse_beta=100, patience=300,
                                       time_limit=bucket * 0.6, t_start=time.monotonic())
            report(sc, a_out, f"n={n} init={i} ADAM")
            
            # If improved, do a quick LBFGS polish
            if sc < best_score + 0.05 and remain - bucket * 0.6 > 15:
                sc2, a_out2, _ = run_lbfgs(a_out, 200, maxiter=2000, maxfun=30000,
                                            time_limit=remain - bucket * 0.6, t_start=time.monotonic())
                report(sc2, a_out2, f"n={n} init={i} LBFGS")

    print(f"\n  After Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 3: Intensive local search on best candidate (~35% time)
    # ==================================================================
    phase3_end = deadline
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive local search")
    print(f"  Starting score: {best_score:.8f}, n={len(best_a)}")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_cur = len(a_cur)
    
    # Step 3a: ADAM with increasing LSE beta
    for beta in [100, 300, 800, 1500]:
        if time.monotonic() > phase3_end - 60:
            break
        remain = phase3_end - time.monotonic() - 60
        if remain < 10:
            break
        t0 = time.monotonic()
        sc, a_out = adam_optimize(a_cur, lr=0.005, max_iters=5000,
                                   lse_beta=beta, patience=600,
                                   time_limit=remain * 0.7, t_start=t0)
        report(sc, a_out, f"ADAM beta={beta}")
        a_cur = a_out
    
    # Step 3b: Peak shaving
    if time.monotonic() < phase3_end - 30:
        remain = phase3_end - time.monotonic() - 30
        for top_k in [5, 10, 20]:
            if time.monotonic() > phase3_end - 30:
                break
            sc, a_out = peak_shave(a_cur, top_k=top_k, strength=0.005,
                                    n_iters=min(200, int(remain // 2 // max(1, 200 // top_k))))
            report(sc, a_out, f"peak_shave top_k={top_k}")
            if sc < best_score:
                a_cur = a_out
    
    # Step 3c: L-BFGS with high beta
    for beta in [500, 1000, 2000, 5000]:
        if time.monotonic() > phase3_end - 20:
            break
        remain = phase3_end - time.monotonic() - 20
        if remain < 10:
            break
        sc, a_out, _ = run_lbfgs(a_cur, beta, maxiter=5000, maxfun=100000,
                                  time_limit=remain * 0.8, t_start=time.monotonic())
        report(sc, a_out, f"LBFGS refine beta={beta}")
        a_cur = a_out
    
    # Step 3d: Coordinate descent
    if time.monotonic() < phase3_end - 30:
        remain = phase3_end - time.monotonic() - 30
        sc, a_out = coordinate_descent(a_cur, n_coords=30, n_rounds=3,
                                        time_limit=remain, t_start=time.monotonic())
        report(sc, a_out, "coord descent")
        if sc < best_score:
            a_cur = a_out
    
    # Step 3e: Final polish cycle
    if time.monotonic() < phase3_end - 10:
        remain = phase3_end - time.monotonic() - 10
        # Quick ADAM
        sc, a_out = adam_optimize(a_cur, lr=0.003, max_iters=2000,
                                   lse_beta=2000, patience=300,
                                   time_limit=remain * 0.4, t_start=time.monotonic())
        report(sc, a_out, "final ADAM")
        a_cur = a_out
        # Final LBFGS
        if time.monotonic() < phase3_end - 5:
            sc, a_out, _ = run_lbfgs(a_cur, 3000, maxiter=3000, maxfun=50000,
                                      time_limit=phase3_end - time.monotonic() - 5,
                                      t_start=time.monotonic())
            report(sc, a_out, "final LBFGS")

    # ==================================================================
    # Phase 4: Try a few different lengths with the best structure
    # ==================================================================
    if time.monotonic() < deadline - 30:
        print(f"\n--- Phase 4: Length variation ---")
        n_best = len(best_a)
        for n_try in [n_best - 10, n_best + 10, n_best - 50, n_best + 50, n_best + 100]:
            if n_try < 512 or n_try > 16000:
                continue
            if time.monotonic() > deadline - 15:
                break
            # Resize best_a to n_try
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_resized = np.interp(x_new, x_old, best_a)
            a_resized = np.clip(a_resized, 0.0, 1000.0)
            
            remain = min(30, deadline - time.monotonic() - 10)
            if remain < 5:
                break
            sc, a_out = adam_optimize(a_resized, lr=0.005, max_iters=1000,
                                       lse_beta=300, patience=200,
                                       time_limit=remain, t_start=time.monotonic())
            report(sc, a_out, f"len_try={n_try}")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
