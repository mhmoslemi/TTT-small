"""
Smooth Max Mirror Descent with multi-start and parametric initialization
for minimizing 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Key ideas:
- Use softmax-weighted gradient (smooth max) for smooth, FFT-computable updates
- Multi-start from parametric families (power law, exponential ramps)
- Adaptive temperature and step size with line search
- Coarse-to-fine multi-scale optimization
- Local block scaling for final refinement
"""

import numpy as np
import time

_GLOBAL_RNG = np.random.default_rng(42)
_MIN_VAL = 1e-18


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_aa(a, L=None):
    """Compute full convolution a*a (non-negative clipped)."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv(a,a))."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    c = _conv_aa(a, L)
    return 2.0 * n * float(c.max())


def _softmax(x, beta):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=float)
    if beta < 1e-6:
        return np.ones_like(x) / len(x)
    sh = x - x.max()
    e = np.exp(beta * sh)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(x)
        p[np.argmax(x)] = 1.0
        return p
    return e / s


def _grad_smooth(a, beta, L):
    """
    Compute gradient of smooth-max of conv(a,a) w.r.t. a.
    Returns (grad, c) where grad has same length as a.
    grad[i] ≈ 2 * sum_k p[k] * a[k-i] (for valid indices)
    """
    n = len(a)
    c = _conv_aa(a, L)
    p = _softmax(c, beta)
    # blame[i] = sum_k p[k] * a[k-i] = conv(a_rev, p)[n-1 : 2n-1]
    L2 = _next_pow2(3 * n - 2)
    F_ar = np.fft.rfft(a[::-1], L2)
    F_p = np.fft.rfft(p, L2)
    corr = np.fft.irfft(F_ar * F_p, L2)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return np.maximum(grad, 0.0), c


def _gradient_step(a, eta, beta, L):
    """One smooth-max mirror descent step."""
    grad, c = _grad_smooth(a, beta, L)
    gm = grad.max()
    if gm < 1e-30:
        return a.copy()
    a_new = a * np.exp(-eta * grad / (gm + 1e-30))
    a_new = np.maximum(a_new, _MIN_VAL)
    s = a_new.sum()
    if s <= 0 or not np.isfinite(s):
        return a.copy()
    return a_new / s


def _line_search_step(a, beta, L, base_eta=0.5):
    """Try multiple step sizes, pick the best."""
    cur_score = _score_norm(a)
    best_p = a.copy()
    best_sc = cur_score
    for mult in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]:
        eta = base_eta * mult
        p_try = _gradient_step(a, eta, beta, L)
        sc = _score_norm(p_try)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p_try
    return best_p, best_sc


def _parametric_init(n, rng):
    """Generate a parametric initial sequence."""
    t = (np.arange(n) + 0.5) / n
    mode = int(rng.integers(0, 10))
    if mode == 0:
        alpha = float(rng.uniform(0.5, 4.0))
        a = np.power(t, alpha)
    elif mode == 1:
        beta = float(rng.uniform(1.0, 8.0))
        a = np.exp(beta * (t - 0.5))
    elif mode == 2:
        alpha = float(rng.uniform(0.5, 4.0))
        a = np.power(1.0 - t + 1e-10, alpha)
    elif mode == 3:
        alpha = float(rng.uniform(0.3, 2.0))
        beta = float(rng.uniform(0.3, 2.0))
        a = np.power(t + 1e-10, alpha) * np.power(1.0 - t + 1e-10, beta)
    elif mode == 4:
        # Two-level: low for first fraction, high for rest
        frac = float(rng.uniform(0.3, 0.7))
        ratio = float(rng.uniform(1.5, 5.0))
        a = np.ones(n)
        a[int(frac * n):] = ratio
    elif mode == 5:
        # Smooth two-level
        c = float(rng.uniform(0.3, 0.7))
        k = float(rng.uniform(5, 30))
        r = float(rng.uniform(2.0, 5.0))
        a = 1.0 + (r - 1.0) / (1.0 + np.exp(-k * (t - c)))
    elif mode == 6:
        # Cosine taper (high in middle)
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5))
        a = np.power(a + 1e-10, float(rng.uniform(1.0, 3.0)))
    elif mode == 7:
        # Inverse cosine (high at ends)
        a = 1.0 - 0.5 * np.cos(np.pi * (t - 0.5))
        a = np.power(a + 1e-10, float(rng.uniform(1.0, 3.0)))
    elif mode == 8:
        # S-curve with noise
        c = float(rng.uniform(0.35, 0.65))
        k = float(rng.uniform(8, 40))
        a = 1.0 / (1.0 + np.exp(-k * (t - c)))
        a = a * float(rng.uniform(2.0, 5.0)) + 1.0
    else:
        # Cubic spline-like
        a = 1.0 + 2.0 * t + t * t + 0.5 * t**3
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

        p_new, sc_new = _line_search_step(a, beta, L, eta)

                    a = best.copy()
                    cur_sc = best_sc
                stall = 0
                eta = base_eta
                beta = max(beta_start, beta * 0.5)

            if stall > 100:
                a = best.copy()
                cur_sc = best_sc
                eta = base_eta
                beta = beta_start
                stall = 0

        it += 1

    return best, best_sc

def _resample(a, m):
    """Resample sequence a to length m using interpolation."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = np.arange(n) / n
    x_new = np.arange(m) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-10)
    return out / out.sum()

        best_overall = np.power(t, 1.5)
        best_overall = best_overall / best_overall.sum()
        best_overall_sc = _score_norm(best_overall)
        print(f"  Fallback: score={best_overall_sc:.8f}")

    # Phase 2: Refinement with local block search
    rem = deadline - time.monotonic()
    if rem > 30.0:
        t_refine = min(rem - 15.0, 0.25 * total_time)
        if t_refine > 10.0:
            print(f"  Refinement (block search) for {t_refine:.0f}s...")
            a_rf, sc_rf = _local_block_search(best_overall, t_refine, verbose)
            if sc_rf < best_overall_sc:
                best_overall = a_rf
                best_overall_sc = sc_rf
                print(f"  After block search: {best_overall_sc:.8f}")

    # Phase 3: Final fine-tuning with smooth max (higher beta)
    rem = deadline - time.monotonic()
    if rem > 20.0:
        t_final = min(rem - 10.0, 0.15 * total_time)
        if t_final > 5.0:
            print(f"  Final polish for {t_final:.0f}s...")
            a_f, sc_f = _optimize_smooth_max(
                best_overall, t_final,
                beta_start=200.0, beta_end=2000.0,
                base_eta=0.15, verbose=verbose
            )
            if sc_f < best_overall_sc:
                best_overall = a_f
                best_overall_sc = sc_f
                print(f"  After final polish: {best_overall_sc:.8f}")

    # Phase 4: Another round of block search on the polished result
    rem = deadline - time.monotonic()
    if rem > 15.0:
        t_bs = min(rem - 10.0, 0.1 * total_time)
        if t_bs > 5.0:
            a_bs, sc_bs = _local_block_search(best_overall, t_bs, verbose)
            if sc_bs < best_overall_sc:
                best_overall = a_bs
                best_overall_sc = sc_bs
                print(f"  After final block search: {best_overall_sc:.8f}")

    # Ensure sum is 1 and all values are valid
    best_overall = np.asarray(best_overall, dtype=float).ravel()
    best_overall = np.where(np.isfinite(best_overall), best_overall, 0.0)
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    s = best_overall.sum()
    if s <= 0:
        best_overall = np.ones(target_n) / target_n
    else:
        best_overall = best_overall / s

    # Verify with official evaluator if possible
    try:
        lst = [float(x) for x in best_overall]
        official_sc = float(evaluate_sequence(lst))
        if np.isfinite(official_sc):
            print(f"  Official evaluation: {official_sc:.8f}")
            if official_sc < best_overall_sc:
                best_overall_sc = official_sc
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"=== Done in {elapsed:.1f}s. Final score: {best_overall_sc:.8f} ===")

    return [float(x) for x in best_overall]
