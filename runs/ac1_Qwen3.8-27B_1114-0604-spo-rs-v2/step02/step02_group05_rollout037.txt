import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 5000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _lp_direction(f, M):
    """
    Solve the LP:
    maximize sum(b)
    subject to: (f*b)_k <= M for all k
                b >= 0
    
    Returns the optimal b or None if infeasible.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix A
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0, and we need an upper bound to prevent unboundedness
    # Actually, the LP should be bounded because f has positive entries
    # But let's add a generous upper bound just in case
    bnds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(num_constraints, M), bnds=bnds, method='highs')
    
    if res.status != 0:

    return res.x

LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a linear programming approach to find descent directions, combined
with gradient-based refinement and diverse search strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 5000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _build_lp_structure(n):
    """Precompute the sparse structure of the LP constraint matrix."""
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return rows, cols, num_constraints


def _solve_lp(f, M, lp_struct):
    """Solve the LP to find a direction that reduces max cross-convolution."""
    rows, cols, num_constraints = lp_struct
    n = len(f)
    
    # Values are f[k-j] for the (k,j) pair
    # For row k and col j, value is f[k-j]
    vals = []
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):

    bnds = [(0, None)] * n

    if res.status != 0:
        return None
    return res.x


def _golden_section_search(f_eval, a=0.0, b=1.0, tol=1e-8, max_iter=50):
    """Golden section search for minimum of f_eval on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~ 0.618
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_eval(c)
    fd = f_eval(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_eval(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_eval(d)
    
    t_opt = (a + b) / 2
    return t_opt, f_eval(t_opt)


def _lp_optimize(f0, time_budget, verbose=False):
    """LP-based optimization: iteratively solve LP for descent directions."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    
    # Precompute LP structure
    lp_struct = _build_lp_structure(n)
    
    best = f.copy()
    best_score = _score(f)
    cur_score = best_score
    
    deadline = time.monotonic() + time_budget
    stall = 0
    iteration = 0
    
    if verbose:
        print(f"    LP opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")
    
    while time.monotonic() < deadline:
        c = _conv_full(f, f)
        M = float(c.max())
        
        # Solve LP
        b_opt = _solve_lp(f, M, lp_struct)
        if b_opt is None:
            break
        
        b_opt = np.maximum(b_opt, 0.0)
        s_b = b_opt.sum()
        if s_b < 1e-10 or not np.isfinite(s_b):
            break
        
        # Rescale to same sum as f
        s_f = f.sum()
        g = b_opt * (s_f / s_b)
        g = np.maximum(g, 0.0)
        
        # Line search
        def F(t):
            h = (1 - t) * f + t * g
            h = np.maximum(h, 0.0)
            sh = h.sum()
            if sh < 1e-10:
                return float('inf')
            h = h / sh
            ch = _conv_full(h, h)
            return 2.0 * n * ch.max()
        
        t_opt, F_opt = _golden_section_search(F, 0.0, 1.0, tol=1e-6, max_iter=30)
        
        if F_opt < cur_score - 1e-12:
            f = (1 - t_opt) * f + t_opt * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            cur_score = F_opt
            stall = 0
            if cur_score < best_score - 1e-10:
                best = f.copy()
                best_score = cur_score
                if verbose and iteration % 10 == 0:
                    print(f"      LP iter={iteration}, t={t_opt:.4f}, score={best_score:.6f}")
        else:
            stall += 1
            if stall > 10:
                # Try a smaller step
                f_try = f + 0.1 * (g - f)
                f_try = _sanitize(f_try)
                if f_try is not None:
                    sc_try = _score(f_try)
                    if sc_try < cur_score - 1e-12:
                        f = f_try
                        cur_score = sc_try
                        stall = 0
                        if cur_score < best_score - 1e-10:
                            best = f.copy()
                            best_score = cur_score
        
        iteration += 1
    
    return best, best_score


def _gradient_optimize(f0, time_budget, beta_start=50.0, beta_end=5000.0, verbose=False):
    """Multiplicative gradient descent with soft max."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    L = _next_pow2(3 * n + 10)
    
    best = f.copy()
    best_score = _score(f)
    cur_score = best_score
    
    beta = float(beta_start)
    alpha = 0.1
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    
    if verbose:
        print(f"    Grad opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")
    
    while time.monotonic() < deadline:
        c = _conv_full(f, f)
        d = n * c
        exact_score = float(2.0 * d.max())
        
        # Soft weights
        shift = d.max()
        if beta < 1e-8:
            w = np.ones(len(d), dtype=float) / len(d)
        else:
            e = np.exp(beta * (d - shift))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.ones(len(d), dtype=float) / len(d)
            else:
                w = e / s
        
        # Gradient: h[j] = 2 * sum_m w[j+m] * f[m] for j = 0..n-1
        W = np.fft.rfft(w, L)
        F = np.fft.rfft(f, L)
        R = np.fft.irfft(W * np.conj(F), L)
        h = 2.0 * R[:n]
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        max_h = float(h.max())
        
        if max_h < 1e-15:
            stall += 1
            continue
        
        h_norm = h / max_h
        
        # Try step sizes
        best_trial = None
        best_trial_sc = cur_score
        for a_try in (alpha * 0.5, alpha, alpha * 1.5):
            a_try = float(np.clip(a_try, 0.005, 3.0))
            p_try = f * np.exp(-a_try * h_norm)
            p_try = np.maximum(p_try, 1e-18 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-13:
                best_trial = p_try
                best_trial_sc = sc_try
        
        if best_trial is not None:
            f = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-12:
                best = f.copy()
                best_score = cur_score
                if verbose and it % 100 == 0:
                    print(f"      grad iter={it}, score={best_score:.6f}, beta={beta:.0f}")
        else:
            stall += 1
            if stall > 5:
                alpha = max(0.01, alpha * 0.7)
            if stall > 30:
                f = best.copy()
                cur_score = best_score
                alpha = 0.1
                beta = max(beta_start, beta * 0.5)
                stall = 5
        
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.1)
        
        it += 1
    
    return best, best_score


def _block_refine(f0, rng, time_budget):
    """Block-wise multiplicative refinement."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_score = _score(f)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 1.4, 1.7, 2.0, 3.0)
    
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 4))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 128), max(3, n // 16)))
        else:
            bs = int(rng.integers(max(8, n // 16), max(16, n // 2)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))
        
        improved = False
        for fac in factors:
            f_try = f.copy()
            f_try[start:start + bs] *= fac
            f_try = _sanitize(f_try)
            if f_try is None:
                continue
            sc = _score(f_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                f = f_try
                improved = True
                break
        
        if improved:
            if cur_score < best_score - 1e-12:
                best = f.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 30:
                f = best.copy()
                cur_score = best_score
                stall = 0
    
    return best, best_score


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 15):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s

def _build_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    num_constraints = 2 * n - 1
    rows = np.repeat(np.arange(num_constraints), 
                     [min(n, k+1, 2*n-1-k) for k in range(num_constraints)])
    cols = []
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        cols.extend(range(j_min, j_max + 1))
    cols = np.array(cols)
    return rows, cols, num_constraints

def _solve_lp(f, M, lp_struct):
    rows, cols, num_constraints = lp_struct
    n = len(f)
    vals = f[rows - cols]  # Vectorized!
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bnds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bnds=bnds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status != 0:
        return None
    return res.x

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses linear programming to find descent directions by
solving max sum(b) s.t. (f*b)_k <= M, b >= 0, which gives a provably
valid descent direction for the max self-convolution. Combined with
multiplicative gradient descent, block refinement, and diverse search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 5000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _build_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix A where A[k,j]=f[k-j]."""
    num_constraints = 2 * n - 1
    counts = np.array([min(n, k + 1, 2 * n - 1 - k) for k in range(num_constraints)], dtype=int)
    rows = np.repeat(np.arange(num_constraints, dtype=np.int32), counts)
    cols = np.concatenate([
        np.arange(max(0, k - (n - 1)), min(n - 1, k) + 1, dtype=np.int32)
        for k in range(num_constraints)
    ])
    return rows, cols, num_constraints


def _solve_lp(f, M, lp_struct):
    """Solve max sum(b) s.t. conv(f,b)_k <= M, b >= 0."""
    rows, cols, num_constraints = lp_struct
    n = len(f)
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bnds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bnds=bnds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)


def _golden_section_min(f_eval, a=0.0, b=1.0, tol=1e-7, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_eval(c)
    fd = f_eval(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_eval(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_eval(d)
    t_opt = (a + b) / 2.0
    return t_opt, f_eval(t_opt)


def _lp_optimize(f0, time_budget, verbose=False):
    """LP-based optimization: iteratively solve LP for descent directions."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)

    if n > 5000:
        lp_struct = None
    else:
        lp_struct = _build_lp_structure(n)

    best = f.copy()
    best_score = _score(f)
    cur_score = best_score

    deadline = time.monotonic() + time_budget
    stall = 0
    iteration = 0

    if verbose:
        print(f"    LP opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() < deadline:
        c = _conv_full(f, f)
        M = float(c.max())

        if lp_struct is not None:
            b_opt = _solve_lp(f, M, lp_struct)
        else:
            b_opt = None

        if b_opt is not None:
            s_b = b_opt.sum()
            if s_b < 1e-10 or not np.isfinite(s_b):
                stall += 1
            else:
                s_f = f.sum()
                g = b_opt * (s_f / s_b)
                g = np.maximum(g, 0.0)
                sg = g.sum()
                if sg < 1e-10:
                    stall += 1
                else:
                    g = g / sg

                    def F(t):
                        h = (1.0 - t) * f + t * g
                        h = np.maximum(h, 0.0)
                        sh = h.sum()
                        if sh < 1e-10:
                            return float('inf')
                        h = h / sh
                        ch = _conv_full(h, h)
                        return 2.0 * n * ch.max()

                    F0 = F(0.0)
                    F1 = F(1.0)

                    if F1 < F0:
                        t_opt, F_opt = _golden_section_min(F, 0.0, 1.0)
                    else:
                        t_opt, F_opt = _golden_section_min(F, 0.0, 0.5)

                    if F_opt < cur_score - 1e-12:
                        f = (1.0 - t_opt) * f + t_opt * g
                        f = np.maximum(f, 0.0)
                        sf = f.sum()
                        if sf < 1e-10:
                            f = best.copy()
                            cur_score = best_score
                        else:
                            f = f / sf
                            cur_score = F_opt
                        stall = 0
                        if cur_score < best_score - 1e-10:
                            best = f.copy()
                            best_score = cur_score
                            if verbose and iteration % 20 == 0:
                                print(f"      LP iter={iteration}, t={t_opt:.4f}, score={best_score:.6f}")
                    else:
                        stall += 1
        else:
            stall += 1

        if stall > 15:
            f = best.copy()
            cur_score = best_score
            stall = 0

        iteration += 1

    return best, best_score


def _gradient_optimize(f0, time_budget, beta_start=50.0, beta_end=8000.0, verbose=False):
    """Multiplicative gradient descent with soft max."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    L = _next_pow2(3 * n + 10)

    best = f.copy()
    best_score = _score(f)
    cur_score = best_score

    beta = float(beta_start)
    alpha = 0.08
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    if verbose:
        print(f"    Grad opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() < deadline:
        c = _conv_full(f, f)
        d = n * c
        shift = d.max()
        if beta < 1e-8:
            w = np.ones(len(d), dtype=float) / len(d)
        else:
            e = np.exp(beta * (d - shift))
            se = e.sum()
            if se < 1e-300 or not np.isfinite(se):
                w = np.ones(len(d), dtype=float) / len(d)
            else:
                w = e / se

        W = np.fft.rfft(w, L)
        Ff = np.fft.rfft(f, L)
        R = np.fft.irfft(W * np.conj(Ff), L)
        h = 2.0 * R[:n]
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        max_h = float(h.max())

        if max_h < 1e-15:
            stall += 1
            continue

        h_norm = h / max_h

        best_trial = None
        best_trial_sc = cur_score
        for a_try in (alpha * 0.4, alpha, alpha * 2.0):
            a_try = float(np.clip(a_try, 0.003, 5.0))
            p_try = f * np.exp(-a_try * h_norm)
            p_try = np.maximum(p_try, 1e-18 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-13:
                best_trial = p_try
                best_trial_sc = sc_try

        if best_trial is not None:
            f = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-12:
                best = f.copy()
                best_score = cur_score
                if verbose and it % 200 == 0:
                    print(f"      grad iter={it}, score={best_score:.6f}, beta={beta:.0f}")
        else:
            stall += 1
            if stall > 8:
                alpha = max(0.005, alpha * 0.6)
            if stall > 40:
                f = best.copy()
                cur_score = best_score
                alpha = 0.08
                beta = max(beta_start, beta * 0.3)
                stall = 5

        if it % 10 == 0:
            beta = min(beta_end, beta * 1.12)

        it += 1

    return best, best_score


def _freq_flatten(f, alpha_power, L_fft=None):
    """Frequency-domain convolution flattening."""
    n = len(f)
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(f, L_fft)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L_fft)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return f.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(np.maximum(C_norm, 0.0), 1.0 / alpha_power)
    C_freq_new = np.fft.rfft(C_new, L_fft)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L_fft)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.25:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _block_refine(f0, rng, time_budget):
    """Block-wise multiplicative refinement."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_score = _score(f)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.25, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0, 5.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.12:
            bs = int(rng.integers(1, 5))
        elif r < 0.55:
            bs = int(rng.integers(max(2, n // 256), max(3, n // 32)))
        else:
            bs = int(rng.integers(max(8, n // 32), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for fac in factors:
            f_try = f.copy()
            f_try[start:start + bs] *= fac
            f_try = _sanitize(f_try)
            if f_try is None:
                continue
            sc = _score(f_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                f = f_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = f.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 50:
                f = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 15):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s


def _init_block_random(n, rng):
    K = int(rng.integers(4, 20))
    p = np.zeros(n, dtype=float)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1, K)
    for i in range(K):
        s = int(round(blocks[i] * n))
        e = max(s + 1, int(round(blocks[i + 1] * n)))
        e = min(n, e)
        p[s:e] = heights[i]
    p = np.maximum(p, 1e-10)
    return p / p.sum()

            f_try[start:start + bs] *= f_scale
            f_try = _sanitize(f_try)
            if f_try is not None:
                sc_try = _score(f_try)
                if sc_try < cur_score - 1e-13:
                    f = f_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-10:
                        best = f.copy()
                        best_score = sc_try
                else:
                    stall += 1

        if it % 10 == 0:
            beta = min(20000.0, beta * 1.08)

        if cur_score > best_score * 1.2 and best_score < float('inf'):
            if rng.random() < 0.3:
                f = best.copy()
                cur_score = best_score
                stall = 0

        it += 1

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    LP-guided + gradient + frequency flattening optimization.
    Uses linear programming (max sum(b) s.t. f*b <= M) for provable
    descent directions, combined with multiplicative gradient descent,
    frequency-domain flattening, and block refinement.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # ========== PHASE 1: Quick exploration (8%) ==========
    print("=" * 60)
    print(f"Phase 1: Quick exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.08 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    explore_lengths = [3072, 4096]
    if parent is not None:
        pn = len(parent)
        if pn not in explore_lengths:
            explore_lengths.append(pn)
            explore_lengths.sort()

    for n in explore_lengths:
        if time.monotonic() > t1_end:
            break
        inits = [
            np.ones(n, dtype=float) / n,
            _init_two_level(n, rng),
            _init_gaussian_mixture(n, rng),
            _init_cosine(n, rng),
            _init_block_random(n, rng),
        ]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: LP-based optimization (40%) ==========
    print("Phase 2: LP-based optimization (main engine)")
    t2_end = start_time + 0.48 * budget

    if best_overall is None:
        best_overall = _init_two_level(3072, rng)
        best_score_overall = _score(best_overall)

    n_main = len(best_overall)

    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.6 * t_rem, 2000.0)
        p, sc = _lp_optimize(best_overall, t_alloc, verbose=True)
        _try_update(p, sc)

    t_rem = t2_end - time.monotonic()
    if t_rem > 120:
        n_alt = 4096 if n_main != 4096 else (3072 if n_main != 3072 else 2048)
        p_alt = _resample(best_overall, n_alt, rng)
        if p_alt is not None:
            t_alloc = min(0.6 * t_rem, 600.0)
            p, sc = _lp_optimize(p_alt, t_alloc, verbose=True)
            _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Hybrid refinement (25%) ==========
    print("Phase 3: Hybrid refinement (LP + grad + freq)")
    t3_end = start_time + 0.73 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.8 * t_rem, 1200.0)
        p, sc = _hybrid_optimize(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        p_re = _resample(best_overall, len(best_overall), rng)
        if p_re is not None:
            t_alloc = min(0.5 * t_rem, 400.0)
            p, sc = _hybrid_optimize(p_re, rng, t_alloc, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Block + diverse (12%) ==========
    print("Phase 4: Block refinement and diverse restarts")
    t4_end = start_time + 0.88 * budget

    t_rem = t4_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.4 * t_rem, 300.0)
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    t_rem = t4_end - time.monotonic()
    if t_rem > 90:
        n_best = len(best_overall)
        for mode in range(4):
            if time.monotonic() > t4_end - 15:
                break
            if mode == 0:
                sp = _init_two_level(n_best, rng)
            elif mode == 1:
                sp = _init_gaussian_mixture(n_best, rng)
            elif mode == 2:
                sp = _init_cosine(n_best, rng)
            else:
                sp = _resample(best_overall, n_best, rng)
                if sp is None:
                    sp = _init_block_random(n_best, rng)

            t_alloc = min(45.0, (t4_end - time.monotonic()) * 0.5)
            if t_alloc > 10:
                p, sc = _hybrid_optimize(sp, rng, t_alloc, verbose=False)
                _try_update(p, sc)

    print(f"  Phase 4 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 5: Final polishing (12%) ==========
    print("Phase 5: Final polishing")

    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 300.0)
            p, sc = _gradient_optimize(best_overall, t_alloc,
                                       beta_start=500.0, beta_end=30000.0, verbose=True)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 60:
            t_alloc = min(0.35 * t_rem, 300.0)
            p, sc = _lp_optimize(best_overall, t_alloc, verbose=True)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.3 * t_rem, 200.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.4 * t_rem, 200.0)
            p, sc = _gradient_optimize(best_overall, t_alloc,
                                       beta_start=2000.0, beta_end=50000.0, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = np.ones(3072, dtype=float) / 3072

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(3072, dtype=float) / 3072

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float) / 3072
    else:
        out = out / s

    return [float(x) for x in out]
