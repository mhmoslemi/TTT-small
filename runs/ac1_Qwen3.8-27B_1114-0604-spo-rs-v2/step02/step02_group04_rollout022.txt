"""
Low-peak step-function optimization via momentum gradient descent with FFT gradients,
L-BFGS polishing, and active-set refinement.
"""
import numpy as np
import time
import sys

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

def _lse_grad(a, T):
    """Compute the gradient of the LSE-smoothed max of conv(a) with respect to a."""
    c = _fft_convolve(a)
    if c.size == 0:
        return np.zeros(a.size), c
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / exp_c.sum()
    grad = _grad_via_fft(w, a)
    return grad, c

def _make_anti_sine(n, amp, n_modes=1, mode_freqs=None, phases=None, seed=42):
    """Create an anti-palindromic multi-mode sinusoidal sequence."""
    rng = np.random.RandomState(seed)

        theta = 2.0 * np.pi * freq * (i + ph) / n
        a += amp_k * np.sin(theta)
    a = np.clip(a, 0.0, 1000.0)

        # Chain rule: grad w.r.t. x
        # d(ax_i)/dx_j: if i==j, ax_i*(1-ax_i); if i!=j, -ax_i*ax_j
        # grad_x_j = sum_i obj_grad_ax_i * d(ax_i)/dx_j

        # = ax_j * (obj_grad_ax_j - sum_i obj_grad_ax_i * ax_i)
        s = np.dot(obj_grad_ax, ax)
        grad_x = ax * (obj_grad_ax - s)
        return grad_x
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    t0 = time.monotonic()

    result = minimize(
        objective, x0, method='L-BFGS-B', jac=gradient,

    if result.success or result.fun < best_score:

        a_new = np.exp(x_opt - np.max(x_opt))

        sc = _score(a_new)

Wait, I realize there might be an issue with the L-BFGS approach. The log-parameterization x = log(a) means that a = exp(x)/sum(exp(x)), which is a softmax. The gradient computation I wrote above should be correct, but let me double-check.

If a_i = exp(x_i) / S where S = sum_j exp(x_j), then:
- da_i/dx_j = exp(x_i) * delta_{ij} / S - exp(x_i) * exp(x_j) / S^2
  = a_i * delta_{ij} - a_i * a_j
  = a_i * (delta_{ij} - a_j)

So the gradient of f(a(x)) with respect to x_j is:
df/dx_j = sum_i (df/da_i) * a_i * (delta_{ij} - a_j)
= a_j * df/da_j - a_j * sum_i df/da_i * a_i
= a_j * (df/da_j - sum_i df/da_i * a_i)

"""
Low-peak step-function optimization combining momentum gradient descent with
log-sum-exp smoothing, L-BFGS polishing, and active-set refinement.
Uses FFT-based gradient computation in O(n log n) per iteration.
"""
import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """Compute grad[j] = 2 * sum_l a[l]*w[l+j] via FFT cross-correlation."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N] if w.size >= N else w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _lse_weights(c, T):
    """Compute softmax weights for LSE-smoothed max."""
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / s


def _make_anti_sine(n, amp, freqs=None, phases=None):
    """Anti-palindromic multi-mode sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    if freqs is None:
        freqs = [1]
    if phases is None:
        phases = [0.5] * len(freqs)
    for k, (freq, ph) in enumerate(zip(freqs, phases)):
        amp_k = amp / (k + 1)
        theta = 2.0 * np.pi * freq * (i + ph) / n
        a += amp_k * np.sin(theta)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _momentum_optimize(a, deadline, T_start=5.0, T_end=60.0, lr_factor=0.3, momentum=0.85, patience=4000):
    """Nesterov momentum gradient descent on LSE-smoothed objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, 1000)
    a = a / a.sum()
    n = a.size
    lr0 = lr_factor * n

    best_a = a.copy()
    best_score = _score(best_a)
    v = np.zeros(n, dtype=np.float64)
    it = 0
    last_improve = 0
    rng = np.random.RandomState(42)
    eval_interval = 3

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 80000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.0005 * it)

        # Compute LSE-smoothed gradient
        c = _fft_convolve(a)
        if c.size == 0:
            break
        w = _lse_weights(c, T)
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad_proj
        update = np.clip(v, -8.0, 8.0)
        a_new = a * np.exp(update)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 1e-12, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a_new / s_new

        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n)
                a = a + 0.003 * a.mean() * noise
                a = np.clip(a, 1e-10, 1000.0)
                a = a / a.sum()
                v = np.zeros(n)
                patience = max(1500, int(patience * 1.25))
                lr0 = max(0.05 * n, lr0 * 0.75)

    return best_a, best_score


def _lbfgs_optimize(a, deadline, T=12.0):
    """L-BFGS on LSE-smoothed objective with log-parameterization."""
    from scipy.optimize import minimize

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, 1000)
    a = a / a.sum()
    n = a.size
    x0 = np.log(a)

    def objective_and_grad(x):
        xm = x - np.max(x)
        ax = np.exp(xm)
        ax = ax / ax.sum()
        c = _fft_convolve(ax)
        c_max = np.max(c)
        exp_c = np.exp(T * (c - c_max))
        lse = c_max + np.log(exp_c.sum()) / T
        val = 2.0 * n * lse
        w = exp_c / exp_c.sum()
        grad_ax = _grad_via_fft(w, ax)
        obj_grad_ax = 2.0 * n * grad_ax
        s_dot = np.dot(obj_grad_ax, ax)
        grad_x = ax * (obj_grad_ax - s_dot)
        return val, grad_x

    def objective(x):
        v, _ = objective_and_grad(x)
        return v

    def gradient(x):
        _, g = objective_and_grad(x)
        return g

    best_a = a.copy()
    best_score = _score(best_a)
    t_start = time.monotonic()

    try:
        # Wrap to check time
        max_evals = 3000
        result = minimize(
            objective, x0, method='L-BFGS-B', jac=gradient,
            options={'maxiter': 800, 'maxfun': max_evals, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        )
        a_new = np.exp(result.x - np.max(result.x))
        a_new = a_new / a_new.sum()
        a_new = np.clip(a_new, 0, 1000)
        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new
    except Exception as e:
        print(f"  L-BFGS error: {e}", flush=True)

    return best_a, best_score


def _active_set_polish(a, deadline, K=15):
    """Iteratively reduce top-K peaks using backtracking line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, 1000)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = _score(best_a)
    lr = 0.1 * n
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        # Soft weights within top-K
        top_vals = c[top_idx]
        T_local = 5.0
        weights = np.exp(T_local * (top_vals - top_vals.max()))
        weights /= weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        # Backtracking line search
        lr_try = lr
        improved = False
        for _ in range(15):
            update = np.clip(-lr_try * grad_proj, -5.0, 5.0)
            a_try = a * np.exp(update)
            a_try = np.clip(a_try, 1e-12, 1000)
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                a = a_try
                best_score = sc_try
                best_a = a.try_copy() if hasattr(a, 'try_copy') else a.copy()
                improved = True
                break
            lr_try *= 0.6

        if not improved:
            lr *= 0.7

    return best_a, best_score

            best_a = a_try.copy()
            a = a_try.copy()  # Continue from improved point

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak nonnegative step function.
    Phase 1: Initialization scan.
    Phase 2: Momentum gradient descent with LSE smoothing.
    Phase 3: L-BFGS polish.
    Phase 4: Active-set refinement.
    Phase 5: Final perturbation + validation.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 15.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Initialization
    d1 = t0 + 45.0
    print("[Phase 1] Scanning initializations...", flush=True)
    init_a, init_score = _scan_initialization(d1)
    print(f"  Best init: n={init_a.size}, score={init_score:.8f}", flush=True)

    best_overall = init_a.copy()
    best_score_overall = init_score

    # Phase 2: Momentum gradient descent (main workhorse)
    d2 = hard_deadline - 350.0
    remaining = d2 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 2] Momentum optimization ({remaining:.0f}s)...", flush=True)
        a2, s2 = _momentum_optimize(init_a, d2, T_start=5.0, T_end=50.0, lr_factor=0.25)
        print(f"  Momentum best: {s2:.8f}", flush=True)
        if s2 < best_score_overall:
            best_score_overall = s2
            best_overall = a2.copy()

    # Phase 3: L-BFGS polish
    d3 = hard_deadline - 180.0
    remaining = d3 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 3] L-BFGS polish ({remaining:.0f}s)...", flush=True)
        a3, s3 = _lbfgs_optimize(best_overall, d3, T=15.0)
        print(f"  L-BFGS best: {s3:.8f}", flush=True)
        if s3 < best_score_overall:
            best_score_overall = s3
            best_overall = a3.copy()

    # Phase 4: Active set polish
    d4 = hard_deadline - 60.0
    remaining = d4 - time.monotonic()
    if remaining > 30:
        print(f"[Phase 4] Active set polish ({remaining:.0f}s)...", flush=True)
        a4, s4 = _active_set_polish(best_overall, d4, K=15)
        print(f"  Active set best: {s4:.8f}", flush=True)
        if s4 < best_score_overall:
            best_score_overall = s4
            best_overall = a4.copy()

    # Phase 5: Final perturbation
    d5 = hard_deadline - 20.0
    remaining = d5 - time.monotonic()
    if remaining > 10:
        print(f"[Phase 5] Final perturbation ({remaining:.0f}s)...", flush=True)
        a5, s5 = _final_perturbation(best_overall, d5, n_trials=500)
        print(f"  Perturbation best: {s5:.8f}", flush=True)
        if s5 < best_score_overall:
            best_score_overall = s5
            best_overall = a5.copy()

    # Phase 6: Validate
    print("[Phase 6] Validating with evaluate_sequence...", flush=True)
    candidates = [(best_overall, best_score_overall), (init_a, init_score)]

    best_out = None
    best_actual = np.inf
    for arr, sc in candidates:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  n={arr.size}: actual={actual:.8f}", flush=True)

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in init_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = init_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
