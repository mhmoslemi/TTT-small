import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def grad_equalize(a, c, S, topk=16, temp_scale=0.05, eq_weight=0.3):
    """
    Compute gradient with both soft-max peak reduction AND peak equalization.
    The equalization term pushes all significant peaks toward the mean,
    which helps when peaks are close together near the optimum.
    """
    n = len(a)
    S2 = S * S
    S3 = S2 * S

    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]

    top1 = float(vals[0])
    if top1 < 1e-15:
        return np.zeros(n)

    # Soft-max weights over top-k peaks
    temp = max(temp_scale * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts = wts / wts.sum()

    # Also compute mean of top-k for equalization
    mean_val = float(np.mean(vals))

    # Combined gradient of convolution w.r.t. a
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    # Main gradient: reduce weighted max
    M_weighted = float(np.dot(vals, wts))
    grad_main = (2.0 * n) * (grad_c / S2 - 2.0 * M_weighted / S3)

    # Equalization gradient: push each peak toward the mean
    # This encourages all peaks to be similar
    grad_eq_c = np.zeros(n)
    for i in range(k_act):
        deviation = vals[i] - mean_val
        if abs(deviation) < 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_eq_c[js:je + 1] += deviation * 2.0 * a[js:je + 1][::-1]

    # The equalization term doesn't directly affect the objective the same way
    # We use it as a secondary regularization
    grad = grad_main + eq_weight * (2.0 * n / S2) * grad_eq_c / max(top1, 1e-10)

    return grad


def adam_step(a, g, m, v, t, lr):
    """In-place Adam optimizer step."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    m_hat = m / (1.0 - 0.9 ** t)
    v_hat = v / (1.0 - 0.999 ** t)
    a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)


def cosine_lr(init_lr, final_lr, t, total_steps):
    """Cosine learning rate schedule."""
    ratio = min(1.0, t / max(1, total_steps))
    return final_lr + 0.5 * (init_lr - final_lr) * (1.0 + np.cos(np.pi * ratio))


def optimize_run(a0, end_time, rng, init_lr, topk, patience, temp_scale=0.08, eq_weight=0.2):
    """

    t = 0
    no_imp = 0
    cur_pat = patience

        lr = cosine_lr(init_lr, init_lr * 0.01, t, est_steps)
        lr = max(lr, 1e-5)

        g = grad_equalize(a, c, S, topk, temp_scale, eq_weight)
        adam_step(a, g, m, v, t, lr)

def coordinate_polish(a, end_time, rng):
    """
    Simple coordinate descent polish: try small perturbations of individual elements.
    """
    n = a.size
    a = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = float(a.sum())
    best_score = compute_score(n, c, S)
    best_a = a.copy()

    step_sizes = [0.01, 0.005, 0.001]

    while time.monotonic() < end_time:
        improved = False
        for idx in range(0, n, max(1, n // 500)):
            if time.monotonic() >= end_time:
                break
            for delta in step_sizes:
                for sign in [1, -1]:
                    a_new = a.copy()
                    a_new[idx] = max(0.0, a_new[idx] + sign * delta * max(1.0, a_new[idx]))
                    if a_new.sum() < 0.01:
                        continue
                    c_new = fast_self_conv(a_new)
                    s_new = compute_score(n, c_new, float(a_new.sum()))
                    if s_new < best_score:
                        a = a_new
                        best_score = s_new
                        best_a = a.copy()
                        c = c_new
                        S = float(a.sum())
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            # Try a random set of coordinates
            for _ in range(min(200, n // 4)):
                if time.monotonic() >= end_time:
                    break
                idx = rng.integers(0, n)
                for delta in step_sizes:
                    for sign in [1, -1]:
                        a_new = a.copy()
                        a_new[idx] = max(0.0, a_new[idx] + sign * delta * max(1.0, abs(a_new[idx])))
                        if a_new.sum() < 0.01:
                            continue
                        c_new = fast_self_conv(a_new)
                        s_new = compute_score(n, c_new, float(a_new.sum()))
                        if s_new < best_score:
                            a = a_new
                            best_score = s_new
                            best_a = a.copy()
                            c = c_new
                            S = float(a.sum())
                            break

    return best_score, best_a

        pass
    n = min(max(2048, n), 16384)
    print(f"n={n}, budget={budget_s:.0f}s, parent={'yes' if parent is not None else 'no'}")

    # === Phase 0: Quick evaluation of parent ===
    if parent is not None and parent.size == n:
        a0 = np.clip(parent, 0.0, 1000.0)
        c0 = fast_self_conv(a0)
        s0 = compute_score(n, c0, float(a0.sum()))
        print(f"Parent score: {s0:.6f}")
        if s0 < best_score and np.isfinite(s0):
            best_score = s0
            best_a = a0.copy()

    # === Phase 1: Exploration (25% of time) ===
    p1_end = start + 0.25 * (total - 10.0)

    if parent is not None and parent.size == n:
        # Optimize from parent with moderate settings
        a0 = np.clip(parent, 0.0, 1000.0)
        sc, aa = optimize_run(a0, min(p1_end, start + 0.10 * (total - 10.0)),
                             rng, init_lr=0.05, topk=16, patience=200,
                             temp_scale=0.08, eq_weight=0.1)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1a (parent refine): {best_score:.6f}")

        # Try a chirp-based initialization
        if time.monotonic() < p1_end:
            k = np.arange(n)
            chirp_init = np.clip(0.5 + 0.5 * np.abs(np.cos(np.pi * k**2 / n)), 0.01, 1000.0)
            sc, aa = optimize_run(chirp_init, p1_end,
                                 rng, init_lr=0.1, topk=8, patience=150,
                                 temp_scale=0.1, eq_weight=0.1)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"Phase 1b (chirp): {best_score:.6f}")
    else:
        guesses = init_guesses(n, rng)
        per_time = (p1_end - start) / len(guesses)
        for i, a0 in enumerate(guesses):
            t_end = min(p1_end, start + per_time * (i + 1))
            if time.monotonic() >= p1_end:
                break
            sc, aa = optimize_run(a0, t_end, rng, init_lr=0.15, topk=8, patience=100,
                                 temp_scale=0.1, eq_weight=0.1)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"Phase 1 init {i}: {sc:.6f}")

    # === Phase 2: Main refinement (35% of time) ===
    p2_end = start + 0.60 * (total - 10.0)
    if best_a is not None and time.monotonic() < p2_end:
        sc, aa = optimize_run(best_a, p2_end, rng, init_lr=0.015, topk=32, patience=400,
                             temp_scale=0.05, eq_weight=0.2)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2: {best_score:.6f}")

    # === Phase 3: Multi-peak fine tuning (25% of time) ===
    p3_end = start + 0.85 * (total - 10.0)
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = optimize_multi_peak(best_a, p3_end, rng, init_lr=0.003, temp_scale=0.02)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3: {best_score:.6f}")

    # === Phase 4: Coordinate descent polish (10% of time) ===
    if best_a is not None and time.monotonic() < deadline - 3.0:
        sc, aa = coordinate_polish(best_a, deadline - 3.0, rng)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4 (coord descent): {best_score:.6f}")

    # === Final micro-polish ===
    if best_a is not None and time.monotonic() < deadline - 1.0:
        sc, aa = optimize_multi_peak(best_a, deadline - 1.0, rng, init_lr=0.001, temp_scale=0.01)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Final polish: {best_score:.6f}")

    # Final output
    if best_a is None:
        best_a = np.ones(n) / n

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
