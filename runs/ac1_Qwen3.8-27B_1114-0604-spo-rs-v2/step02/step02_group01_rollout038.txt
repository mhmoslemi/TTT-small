def _smooth_perturb(f, rng):
    f=np.asarray(f,dtype=np.float64).copy()
    n=f.size
    x=np.linspace(0.0,1.0,n)
    mode=int(rng.integers(0,5))
    if mode==0:
       alpha=float(rng.uniform(-1.0,1.0))
       f = f * np.exp(alpha*(x-0.5))
    elif mode==1:
       freq=int(rng.integers(1, max(2, min(20, n//10)+1)))
       phase=float(rng.uniform(0,2*np.pi))
       amp=float(rng.uniform(0.05,0.25))
       f = f * (1.0 + amp*np.sin(2*np.pi*freq*x + phase))
    elif mode==2:
       # add low-freq random sine sum
       mod=np.ones(n)
       for _ in range(3):
          freq=int(rng.integers(1, 20))
          amp=float(rng.uniform(0.02,0.12))
          phase=float(rng.uniform(0,2*np.pi))
          mod += amp*np.sin(2*np.pi*freq*x + phase)
       f=f*mod
    elif mode==3:
       c=float(rng.uniform(0.1,0.9)); bw=float(rng.uniform(0.05,0.2)); amp=float(rng.uniform(0.05,0.3))
       f=f*(1.0+amp*np.exp(-((x-c)/bw)**2))
    else:
       # small random multiplicative noise smoothed by averaging
       noise=rng.normal(1.0, 0.05, n)
       # simple moving average to smooth
       w=5
       if n> w:
          kernel=np.ones(w)/w

    f=np.maximum(f,0.0)

```python
"""
Optimizer for non-negative step-function sequences minimizing
    2 * n * max(conv(a, a)) / (sum(a))^2.

The search uses a combination of:
- smooth log-mirror descent on a soft/active maximum convolution lag,
- focused top-lag mirror descent,
- exact coordinate-transfer polishing,
- multi-resolution initialisation from the provided construction.
"""

import numpy as np
import time

_EPS = 1e-300


def _clean_input(seq):
    try:
        arr = np.asarray(seq, dtype=np.float64)
    except Exception:
        arr = None

    if arr is None or arr.size == 0:
        arr = np.ones(8192, dtype=np.float64)

    arr = arr.astype(np.float64, copy=True).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(arr.sum())
    if s < 0.01:
        arr = np.ones(8192, dtype=np.float64)
    else:
        arr = arr / s

    return arr


def _normalize(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s <= 1e-300:
        f = np.ones(f.size, dtype=np.float64)
        return f / f.size
    return f / s


def _fft_size(n):
    return 1 << (2 * n - 1).bit_length()


def _conv_full(f, size=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if size is None:
        size = _fft_size(n)
    L = 2 * n - 1
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)
    c = c[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_max(f, size=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if size is None:
        size = _fft_size(n)
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)
    L = 2 * n - 1
    c = c[:L]
    return float(np.max(c))


def _score_norm(f, size=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    s = float(f.sum())
    if s < 1e-12:
        return np.inf
    if size is None:
        size = _fft_size(n)
    cm = _conv_max(f, size)
    return float(2.0 * n * cm / (s * s))


def _resample(f, new_n):
    f = _normalize(f)
    n = f.size
    if n == new_n:
        return f.copy()
    if n == 1:
        return np.full(new_n, float(f[0]), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, int(new_n), dtype=np.float64)
    y = np.interp(x_new, x_old, f)
    return _normalize(y)


def _generate_shapes(L, rng):
    outs = []
    if L <= 0:
        return outs

    x = np.linspace(0.0, 1.0, int(L), dtype=np.float64)

    raws = [
        ("ramp_up", 0.05 + x),
        ("ramp_down", 0.05 + 1.0 - x),
        ("quad_up", 0.10 + 0.60 * x + 0.30 * x * x),
        ("quad_down", 0.10 + 0.60 * (1.0 - x) + 0.30 * (1.0 - x) * (1.0 - x)),
        ("exp_up", np.exp(2.0 * x) + 0.02),
        ("exp_down", np.exp(-2.0 * x) + 0.02),
        ("sine_phi1", 0.35 + 0.65 * np.sin(2.0 * np.pi * x + 0.25)),
        ("sine_phi2", 0.35 + 0.65 * np.sin(4.0 * np.pi * x + 0.75)),
        ("bell_shift", 0.15 + np.exp(-((x - 0.68) / 0.22) ** 2)),
        ("u_shape", 0.25 + 0.75 * ((x - 0.5) ** 2) / 0.25),
    ]

    for name, arr in raws:
        if arr.size != L:
            continue
        arr = np.asarray(arr, dtype=np.float64)
        if not np.all(np.isfinite(arr)):
            continue
        arr = np.maximum(arr, 0.0)
        if float(arr.sum()) <= 1e-300:
            continue
        outs.append((name, arr / float(arr.sum())))
    return outs


def _smooth_perturb(f, rng):
    f = _normalize(f)
    n = f.size
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    mode = int(rng.integers(0, 6))

    if mode == 0:
        alpha = float(rng.uniform(-1.2, 1.2))
        f = f * np.exp(alpha * (x - 0.5))

    elif mode == 1:
        freq = int(rng.integers(1, max(2, min(25, max(2, n // 10)))))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.04, 0.25))
        f = f * (1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase))

    elif mode == 2:
        mod = np.ones(n, dtype=np.float64)
        for _ in range(3):
            freq = int(rng.integers(1, 25))
            amp = float(rng.uniform(0.02, 0.12))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            mod += amp * np.sin(2.0 * np.pi * freq * x + phase)
        f = f * mod

    elif mode == 3:
        c = float(rng.uniform(0.15, 0.85))
        bw = float(rng.uniform(0.05, 0.22))
        amp = float(rng.uniform(0.05, 0.30))
        f = f * (1.0 + amp * np.exp(-((x - c) / bw) ** 2))

    elif mode == 4:
        # Two gentle bumps.
        c1, c2 = sorted(rng.uniform(0.10, 0.90, size=2))
        bw = float(rng.uniform(0.06, 0.20))
        amp = float(rng.uniform(0.04, 0.22))
        boost = np.exp(-((x - c1) / bw) ** 2) + np.exp(-((x - c2) / bw) ** 2)
        f = f * (1.0 + amp * boost)

    else:
        # Smoothed random multiplicative noise.
        noise = rng.normal(1.0, 0.05, size=n)
        if n > 7:
            k = 5
            kernel = np.ones(k, dtype=np.float64) / float(k)
            noise = np.convolve(noise, kernel, mode="same")
        f = f * noise

    f = np.maximum(f, 0.0)
    return _normalize(f)


def _gradient_top(f, lags, weights, n=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    if n is None:
        n = f.size
    grad = np.zeros(n, dtype=np.float64)

    for k, w in zip(lags, weights):
        w = float(w)
        if w == 0.0:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = k - np.arange(js, je + 1, dtype=np.int64)
            grad[js:je + 1] += 2.0 * w * f[idx]

    return grad


def _optimize_generic(f, deadline, rng, mode=0, beta=1.0, K=16,
                      weight_power=2.0, ls_interval=40, eta=1.0,
                      patience=200, verbose=False, label=""):
    """
    Log-mirror descent.

    mode = 0: soft maximum over all lags (FFT-based weighted gradient).
    mode = 1: hard/focused maximum over top-K lags.
    """
    f = _normalize(f)
    n = f.size
    if n < 2:
        return f, 2.0

    size = _fft_size(n)
    L = 2 * n - 1

    c0 = _conv_full(f, size)
    best_score = float(2.0 * n * np.max(c0))
    best_f = f.copy()

    it = 0
    stagnant = 0
    beta_cur = float(beta)
    eta_cur = float(eta)
    max_eta = 50.0

    while time.monotonic() < deadline:
        F = np.fft.rfft(f, size)
        c = np.fft.irfft(F * F, size)
        c = c[:L]
        np.maximum(c, 0.0, out=c)

        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            stagnant = 0

            if verbose and (it % 10000 == 0):
                print(f"    [{label}] iter {it:7d} best {best_score:.8f}")
        else:
            stagnant += 1

        if stagnant > patience:
            if score > best_score * (1.0 + 0.01):
                f = best_f.copy()
            else:
                f = _smooth_perturb(f, rng)

            stagnant = 0
            eta_cur = min(max_eta, max(0.02, eta_cur * float(rng.choice([0.5, 0.7, 1.0, 1.4, 2.0]))))
            if mode == 0:
                beta_cur = min(50.0, max(0.05, beta_cur * float(rng.choice([0.5, 0.7, 1.0, 1.4, 2.0]))))
            it += 1
            continue

        if mode == 0:
            scale = beta_cur * float(n)
            w = np.exp(scale * (c - cmax))
            w = np.where(np.isfinite(w), w, 0.0)
            ws = float(w.sum())
            if ws <= 1e-300:
                w = np.zeros(L, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= ws

            wr = w[::-1]
            W = np.fft.rfft(wr, size)
            h = np.fft.irfft(F * W, size)

            if n == 1:
                G = h[:1].copy()
            else:
                G = h[2 * n - 2 : n - 2 : -1].copy()

        else:
            Kcur = int(min(max(1, int(K)), L))
            if Kcur >= L:
                top = np.arange(L, dtype=np.int64)
                base = c.copy()
            else:
                top = np.argpartition(c, -Kcur)[-Kcur:]
                top = top[np.argsort(c[top])[::-1]]
                base = c[top]

            if base[0] > 1e-300:
                r = base / base[0]
                w = r ** float(weight_power)
                ws = float(w.sum())
                if ws <= 1e-300 or not np.isfinite(ws):
                    w = np.ones(Kcur, dtype=np.float64) / float(Kcur)
                else:
                    w /= ws
            else:
                w = np.ones(Kcur, dtype=np.float64) / float(Kcur)

            G = _gradient_top(f, top, w, n)

        eff = G - float(np.mean(G))
        denom = float(np.max(np.abs(eff)))
        if denom < 1e-300:
            f = _smooth_perturb(f, rng)
            it += 1
            continue

        # Periodic expensive line search.
        do_ls = (it % max(1, int(ls_interval)) == 0) and it > 0
        if do_ls:
            cands = [
                eta_cur * 0.5,
                eta_cur * 0.75,
                eta_cur,
                eta_cur * 1.5,
                eta_cur * 2.0,
                0.1,
                0.3,
                1.0,
                3.0,
                8.0,
            ]

            uniq = []
            for e in cands:
                e = float(e)
                if e <= 0.0 or e > max_eta:
                    continue
                dup = False
                for u in uniq:
                    if abs(e - u) <= 1e-8 * max(1.0, abs(u)):
                        dup = True
                        break
                if not dup:
                    uniq.append(e)

            logbase = np.log(f + _EPS)
            best_e = 0.0
            best_sc = score
            best_try = None

            for e in uniq:
                lf = logbase - e * eff / denom
                lf -= float(np.max(lf))
                ft = np.exp(lf)
                st = float(ft.sum())
                if st <= 1e-200:
                    continue
                ft /= st
                cm = float(np.max(_conv_full(ft, size)))
                sc = float(2.0 * n * cm)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_e = e
                    best_try = ft

            if best_try is not None and best_e > 0.0:
                eta_cur = float(best_e)
                f = best_try
                if best_sc < best_score - 1e-12:
                    best_score = best_sc
                    best_f = f.copy()
                    stagnant = 0
                    if verbose and (it % 10000 == 0):
                        print(f"    [{label}] line-search best {best_score:.8f}")
            else:
                # Direction is not promising; shrink step and try againt soon.
                eta_cur = max(0.02, min(max_eta, eta_cur * 0.5))

            it += 1
            continue

        lf = np.log(f + _EPS) - eta_cur * eff / denom
        lf -= float(np.max(lf))
        f = np.exp(lf)
        ss = float(f.sum())
        if ss <= 1e-200:
            f = best_f.copy()
        else:
            f /= ss

        it += 1

    final_c = _conv_full(best_f, size)
    final_score = float(2.0 * n * np.max(final_c))
    if final_score < best_score:
        best_score = final_score
    return best_f, best_score


def _polish_coordinate(f, deadline, rng, K=80, D=24, verbose=False, label=""):
    """
    Coordinate-transfer polish.

    For a selected active lag set, finds small transfers p -> q that reduce
    the predicted active maximum, then verifies by full convolution.
    """
    f = _normalize(f)
    n = f.size
    if n < 2:
        return f, 2.0

    size = _fft_size(n)
    L = 2 * n - 1

    c0 = _conv_full(f, size)
    best_score = float(2.0 * n * np.max(c0))
    best_f = f.copy()
    cur_score = best_score

    no_improve = 0
    pass_count = 0

    while time.monotonic() < deadline:
        c = _conv_full(f, size)
        cmax = float(np.max(c))
        cur_score = float(2.0 * n * cmax)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 25:
            if cur_score > best_score * (1.0 + 0.01):
                f = best_f.copy()
                cur_score = best_score
            else:
                f = _smooth_perturb(f, rng)
            no_improve = 0
            continue

        Kcur = int(min(max(1, int(K)), L))
        top = np.argpartition(c, -Kcur)[-Kcur:]
        top = top[np.argsort(c[top])[::-1]]
        base = c[top]

        if base[0] > 1e-300:
            r = base / base[0]
            w = r * r
            ws = float(w.sum())
            if ws <= 1e-300:
                w = np.ones(Kcur, dtype=np.float64) / float(Kcur)
            else:
                w /= ws
        else:
            w = np.ones(Kcur, dtype=np.float64) / float(Kcur)

        v = 0.5 * _gradient_top(f, top, w, n)

        order = np.argsort(v)
        donor_order = order[::-1]

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-18 and len(donors) < D:
                donors.append(idx)

        if len(donors) < D:
            for idx in donor_order:
                idx = int(idx)
                ok = True
                for d in donors:
                    if d == idx:
                        ok = False
                        break
                if ok and len(donors) < D:
                    donors.append(idx)

        receivers = [int(i) for i in order[:D]]

        c_active = c[top]
        M = float(c_active[0])
        moves = []

        # Candidate transfer fractions.
        base_ts = (0.05, 0.10, 0.15, 0.25, 0.40, 0.60, 0.80, 1.00)

        for p in donors:
            fp = float(f[p])
            if fp <= 1e-30:
                continue

            kp = top - p
            validp = (kp >= 0) & (kp < n)
            bp = np.zeros(Kcur, dtype=np.float64)
            bp[validp] = f[kp[validp]]

            is_2p = (top == (2 * p))
            is_p = (top == p)
            is_2q_placeholder = None

            for q in receivers:
                if q == p:
                    continue

                kq = top - q
                validq = (kq >= 0) & (kq < n)
                aq = np.zeros(Kcur, dtype=np.float64)
                aq[validq] = f[kq[validq]]

                delta1 = aq - bp

                Dvec = np.zeros(Kcur, dtype=np.float64)
                Dvec[is_2p] += 1.0
                Dvec[top == (2 * q)] += 1.0
                if p != q:
                    Dvec[top == (p + q)] -= 2.0

                A = 2.0 * fp * delta1
                B = (fp * fp) * Dvec

                t_allow = 1.0
                for kk in range(Kcur):
                    rhs = M - float(c_active[kk])
                    if rhs < -1e-14:
                        continue
                    Ak = float(A[kk])
                    Bk = float(B[kk])

                    if Ak <= 0.0 and Bk <= 0.0:
                        continue

                    if Bk > 1e-30:
                        disc = Ak * Ak + 4.0 * Bk * rhs
                        if disc < 0.0:
                            t_allow = 0.0
                            break
                        troot = (-Ak + float(np.sqrt(disc))) / (2.0 * Bk)
                        if troot < t_allow:
                            t_allow = troot
                    elif Bk < -1e-30:
                        # Concave contribution is handled by explicit trial values.
                        pass
                    else:
                        if Ak > 0.0:
                            troot = rhs / Ak
                            if troot < t_allow:
                                t_allow = troot

                if t_allow <= 0.0:
                    continue
                if t_allow > 1.0:
                    t_allow = 1.0

                trial_ts = list(base_ts)
                trial_ts.extend([0.5 * t_allow, 0.75 * t_allow, 0.95 * t_allow, t_allow])

                for t in trial_ts:
                    t = float(t)
                    if t <= 0.0:
                        continue
                    if t > 1.0:
                        t = 1.0
                    if t > t_allow + 1e-12:
                        continue

                    c_new = c_active + t * A + t * t * B
                    pred = float(np.max(c_new))
                    if pred < M - 1e-14:
                        moves.append((pred, p, q, t))

        if moves:
            moves.sort()
            accepted = False

            for pred, p, q, t in moves[:10]:
                ft = f.copy()
                amt = float(f[p]) * float(t)
                ft[p] -= amt
                ft[q] += amt
                ft = np.maximum(ft, 0.0)
                st = float(ft.sum())
                if st <= 1e-300:
                    continue
                ft /= st

                cm = float(np.max(_conv_full(ft, size)))
                sc = float(2.0 * n * cm)

                if sc < best_score - 1e-12:
                    f = ft
                    best_score = sc
                    best_f = f.copy()
                    cur_score = sc
                    no_improve = 0
                    accepted = True
                    if verbose and (pass_count % 20 == 0):
                        print(f"    [{label}] polish new best {best_score:.8f}")
                    break

                if sc < cur_score - 1e-12:
                    f = ft
                    cur_score = sc
                    accepted = True
                    break

            pass_count += 1
            if accepted:
                continue

        # No reliable coordinate move: try a few tiny random transfers.
        moved = False
        for _ in range(8):
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            if p == q or f[p] <= 1e-18:
                continue
            ft = f.copy()
            t = float(rng.uniform(0.03, 0.12))
            amt = float(f[p]) * t
            ft[p] -= amt
            ft[q] += amt
            ft = np.maximum(ft, 0.0)
            st = float(ft.sum())
            if st <= 1e-300:
                continue
            ft /= st
            cm = float(np.max(_conv_full(ft, size)))
            sc = float(2.0 * n * cm)
            if sc < cur_score - 1e-12:
                f = ft
                cur_score = sc
                moved = True
                break

        pass_count += 1

        if not moved:
            if no_improve > 8:
                if cur_score > best_score * (1.0 + 0.01):
                    f = best_f.copy()
                    cur_score = best_score
                else:
                    f = _smooth_perturb(f, rng)
                no_improve = 0
            else:
                f = _smooth_perturb(f, rng)
                no_improve = 0

    final_c = _conv_full(best_f, size)
    final_score = float(2.0 * n * np.max(final_c))
    if final_score < best_score:
        best_score = final_score
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(int(seed))
    t_start = time.monotonic()
    hard_deadline = t_start + float(budget_s) - 30.0

    f_init = _clean_input(globals().get("height_sequence_1", None))
    n0 = f_init.size

    print(f"Initial construction length: {n0}")

    # Choose practical working lengths. 8192 is often attractive because it
    # keeps the FFT size small while giving a finer discretisation.
    if 300 <= n0 <= 9000:
        base_L = n0
    else:
        base_L = 8192

    length_set = {base_L}
    if base_L < 8192:
        length_set.add(8192)
    elif base_L > 8192:
        length_set.add(8192)

        # If original is very large, avoid using it directly.
        # If it is only moderately larger, also try a reduced length.
        if base_L <= 16000:
            length_set.add(1 << (int(np.ceil(np.log2(base_L)))))

    lengths = [L for L in length_set if 300 <= L <= 9000]
    if not lengths:
        lengths = [8192]

    candidates = []

    # Primary candidates: resampled existing construction.
    for L in lengths:
        f = _resample(f_init, int(L))
        sc = _score_norm(f)
        candidates.append((sc, f, int(L), "resample"))

    # Add analytic shapes at the primary length.
    primary_L = min(candidates, key=lambda z: z[0])[2]
    for name, f in _generate_shapes(primary_L, rng):
        sc = _score_norm(f)
        candidates.append((sc, f, int(primary_L), name))

    candidates.sort(key=lambda z: z[0])

    best_score = float(candidates[0][0])
    best_f = candidates[0][1].copy()
    best_len = int(candidates[0][2])
    best_name = candidates[0][3]

    print(f"Best initial candidate: length={best_len} name={best_name} score={best_score:.8f}")
    for sc, _, L, name in candidates[:6]:
        print(f"  candidate length={L:6d} name={name:12s} score={sc:.8f}")

    # Short trial on the best two distinct candidates, especially different
    # lengths, before committing most of the time budget.
    remain = hard_deadline - time.monotonic()
    if remain > 900:
        quick_budget = min(180.0, 0.08 * remain)
        tried = set()
        for sc, f, L, name in candidates:
            key = (L, name)
            if key in tried:
                continue
            tried.add(key)
            if len(tried) > 2:
                break
            d = time.monotonic() + quick_budget / float(len(tried))
            d = min(hard_deadline - 300.0, d)
            if d <= time.monotonic() + 30:
                break
            f_opt, sc_opt = _optimize_generic(
                f, d, rng,
                mode=0, beta=1.0, K=16, weight_power=2.0,
                ls_interval=30, eta=1.0, patience=120,
                verbose=False, label=f"quick-{name}"
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                best_len = int(f_opt.size)
                best_name = name
                print(f"Quick trial improved: length={best_len} score={best_score:.8f}")

    # Main staged optimisation using the current global best as anchor.
    phases = [
        (0, {"beta": 0.55, "K": 16, "weight_power": 2.0, "ls_interval": 40, "eta": 0.65, "patience": 220}),
        (0, {"beta": 2.20, "K": 16, "weight_power": 2.0, "ls_interval": 40, "eta": 1.10, "patience": 220}),
        (1, {"beta": 1.0, "K": 8,  "weight_power": 1.0, "ls_interval": 30, "eta": 0.85, "patience": 180}),
        (1, {"beta": 1.0, "K": 24, "weight_power": 2.0, "ls_interval": 40, "eta": 0.55, "patience": 180}),
        (2, {}),
        (2, {}),
    ]

    pi = 0
    while time.monotonic() < hard_deadline - 300:
        remain = hard_deadline - time.monotonic()
        if remain < 120:
            break

        mode, params = phases[pi % len(phases)]
        budget = max(70.0, min(420.0, 0.18 * remain))
        budget = min(budget, remain - 100.0)
        if budget <= 20.0:
            break

        d = time.monotonic() + budget

        f_start = best_f.copy()

        if mode == 2:
            f_opt, sc_opt = _polish_coordinate(
                f_start, d, rng,
                K=80, D=24,
                verbose=True,
                label=f"polish{pi}"
            )
        else:
            f_opt, sc_opt = _optimize_generic(
                f_start, d, rng,
                mode=mode,
                beta=params.get("beta", 1.0),
                K=params.get("K", 16),
                weight_power=params.get("weight_power", 2.0),
                ls_interval=params.get("ls_interval", 40),
                eta=params.get("eta", 1.0),
                patience=params.get("patience", 200),
                verbose=True,
                label=f"phase{pi}"
            )

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_len = int(best_f.size)
            print(f"Phase {pi} improved best to {best_score:.8f}")
        else:
            print(f"Phase {pi} best unchanged ({best_score:.8f})")

        pi += 1

    # Final dedicated polishing.
    if time.monotonic() < hard_deadline - 30:
        f_opt, sc_opt = _polish_coordinate(
            best_f,
            hard_deadline - 10.0,
            rng,
            K=120,
            D=30,
            verbose=True,
            label="final-polish"
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_len = int(best_f.size)
            print(f"Final polish improved best to {best_score:.8f}")

    # If there is still a little time, one more short top-lag run can squeeze
    # out tiny gains.
    if time.monotonic() < hard_deadline - 40:
        f_opt, sc_opt = _optimize_generic(
            best_f,
            hard_deadline - 5.0,
            rng,
            mode=1,
            K=12,
            weight_power=1.5,
            ls_interval=25,
            eta=0.4,
            patience=120,
            verbose=False,
            label="last"
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_len = int(best_f.size)
            print(f"Last top-lag improved best to {best_score:.8f}")

    # Final sanitisation.
    best_f = np.nan_to_num(np.asarray(best_f, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    best_f = np.clip(best_f, 0.0, 1000.0)
    s = float(best_f.sum())
    if s < 0.01:
        best_f = np.ones(best_len if best_len > 0 else 8192, dtype=np.float64)
        s = float(best_f.sum())
    else:
        best_f = best_f / s

    # Internal final score.
    internal_score = _score_norm(best_f)
    print(f"Internal final score: {internal_score:.10f} length={best_f.size}")

    # Call the official evaluator if available.
    try:
        official = evaluate_sequence([float(x) for x in best_f])
        official = float(official)
        print(f"Official final score: {official:.10f}")
    except Exception as e:
        official = np.inf
        print(f"Official evaluation failed: {e}")

    return [float(x) for x in best_f]
```
