```python
"""
Low-peak step-function optimization via LP-based active set method with FFT convolution.

Core algorithm:
1. Initialize with various starting points (sinusoidal, random, previous best).
2. Iteratively solve a linear program to find a direction g that maximizes sum(g)
   while keeping (a*g)[k] <= M for active peaks k.
3. Update a via convex combination with line search.
4. Use multiple restarts and different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _make_sine(n, amp, phase=0.5):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_ramp(n, alpha):
    """Ramp initialization: a[i] = (1 + alpha * i/n) / n."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + alpha * i / (n - 1)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    a /= s
    return a


def _make_anti_palindromic(n, rng):
    """Anti-palindromic: a[j] + a[n-1-j] = 1/n (constant)."""
    half = n // 2
    a = np.zeros(n, dtype=np.float64)
    for j in range(half):
        a[j] = rng.dirichlet([1.0, 1.0])[0] / n  # random split
        a[n - 1 - j] = (1.0 / n) - a[j]
    if n % 2 == 1:
        a[half] = 1.0 / (2 * n)
    # Normalize
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return np.clip(a, 0, 1000)


def _solve_lp(a, active_idx, M):
    """
    Solve: maximize sum(g)
    subject to: g >= 0
                (a*g)[k] <= M for k in active_idx
    
    Returns g (normalized to sum 1) or None if infeasible.
    """
    n = a.size
    K = len(active_idx)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ g <= b_ub
    # For each active index k: sum_j a[j] * g[k-j] <= M
    # where valid j: 0 <= j < n and 0 <= k-j < n => j in [max(0, k-n+1), min(n-1, k)]
    
    A_ub = np.zeros((K, n), dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    
    for i, k in enumerate(active_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # (a*g)[k] = sum_{j=j_start}^{j_end} a[j] * g[k-j]
        # Let m = k-j, so m ranges from k-j_end to k-j_start
        # g[m] has coefficient a[j] = a[k-m]
        for j in range(j_start, j_end + 1):
            m = k - j
            if 0 <= m < n:
                A_ub[i, m] += a[j]
    
    # Variable bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    g = result.x
    s = g.sum()
    if s < 1e-15:
        return None
    
    return g / s


def _optimize_lp(a, deadline, K_active=20, max_iter=None):
    """
    LP-based active set optimization.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 200
    rng = np.random.RandomState(42)
    
    if max_iter is None:
        max_iter = 100000
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        
        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        M = float(np.max(c))
        
        # Find active set: top K peaks
        if c.size <= K_active:
            active_idx = np.arange(c.size)
        else:
            # Use argpartition for efficiency
            active_idx = np.argpartition(c, -K_active)[-K_active:]
            active_idx = active_idx[np.argsort(-c[active_idx])]  # sort by value descending
        
        # Solve LP
        g = _solve_lp(a, active_idx, M)
        
        if g is None:
            # LP failed, try with more active constraints
            K_try = min(c.size, K_active * 2)
            active_idx2 = np.argpartition(c, -K_try)[-K_try:]
            g = _solve_lp(a, active_idx2, M)
            if g is None:
                # Reset
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01 * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                continue
        
        # Line search for step size
        best_t = 0.5
        best_t_score = np.inf
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1 - t) * best_a + t * g
            a_trial = np.clip(a_trial, 0, 1000)
            s_trial = a_trial.sum()
            if s_trial < 0.01:
                continue
            a_trial = a_trial / s_trial
            sc = _score(a_trial)
            if sc < best_t_score:
                best_t_score = sc
                best_t = t
        
        if best_t_score < best_score:
            a = (1 - best_t) * best_a + best_t * g
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
            best_a = a.copy()
            best_score = best_t_score
            last_improve = it
        elif it - last_improve > patience:
            # Restart with perturbation
            a = best_a.copy()
            noise_scale = 0.05
            noise = rng.standard_normal(n) * noise_scale * a.mean()
            a = a + noise
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
            patience = max(100, int(patience * 1.5))
        
        # Progress reporting
        if it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}", flush=True)
    
    return best_a, best_score
```

                g >= 0

```python
    # Matrix A_ub has K rows and (n+1) columns

    rows = []
    cols = []
    vals = []

        # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)

            m = k - j  # index of g

                rows.append(k)
                cols.append(m)
                vals.append(a[j])
        # -t term
        rows.append(k)
        cols.append(n)  # t is variable index n
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0 (handled by bounds), t unbounded (or t >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    t_val = result.x[n]
    return g, t_val
```

```python
LP-based minimax optimization for low-peak non-negative step functions.

Algorithm:
1. Initialize with sinusoidal/ramp sequences of various parameters.
2. Iteratively solve a sparse LP to find g that minimizes max(convolve(a, g))
   over the simplex, then update a towards g with line search.
3. Complement with gradient-based polish.
4. Try multiple sequence lengths and restarts.

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _score(a):

    a = 1.0 + alpha * i / max(1, n - 1)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n

def _solve_lp_minimax(a, use_subset=False, subset_frac=0.5):
    """
    Solve minimax LP:
        minimize t
        subject to: (a*g)[k] <= t for (subset of) k
                    sum(g) = 1
                    g >= 0
    Returns (g, t) or (None, None).
    """
    n = a.size
    K = 2 * n - 1
    
    if use_subset:
        # Only use a subset of constraints (every other one)
        idx = np.arange(0, K, max(1, int(1 / subset_frac)))
        idx = idx[idx < K]
    else:
        idx = np.arange(K)
    
    K_use = len(idx)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for ii, k in enumerate(idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(ii)
                cols.append(m)
                vals.append(float(a[j]))
        rows.append(ii)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K_use, n + 1))
    b_ub = np.zeros(K_use)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'time_limit': 10})
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    t_val = result.x[n]
    return g, t_val
```

```python
"""
LP-based minimax optimization for low-peak non-negative step functions.
Uses sparse LP to find optimal update directions, combined with line search
and multiple initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_sine(n, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _build_sparse_constraints(a, idx=None):
    """Build sparse constraint matrix for the minimax LP."""
    n = a.size
    K = 2 * n - 1
    if idx is None:
        idx = np.arange(K)
    K_use = len(idx)
    
    # Pre-allocate
    max_total = K_use * n + K_use
    rows = np.zeros(max_total, dtype=np.int32)
    cols = np.zeros(max_total, dtype=np.int32)
    vals = np.zeros(max_total, dtype=np.float64)
    total = 0
    
    for ii in range(K_use):
        k = idx[ii]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        ns = j_end - j_start + 1
        # m = k - j, so m ranges from k-j_end to k-j_start
        m_start = k - j_end
        m_end = k - j_start
        # coefficients: a[j] for j = j_start..j_end, corresponding to m = k-j
        # a[k-m] for m = m_start..m_end
        coeff = a[k - m_end : k - m_start + 1][::-1]
        
        rows[total:total+ns] = ii
        cols[total:total+ns] = m_start + np.arange(ns)
        vals[total:total+ns] = coeff
        total += ns
        
        rows[total] = ii
        cols[total] = n
        vals[total] = -1.0
        total += 1
    
    return csr_matrix((vals[:total], (rows[:total], cols[:total])), shape=(K_use, n + 1))


def _solve_lp_minimax(a, timeout=5.0):
    """Solve the minimax LP. Returns (g, t) or (None, None)."""
    n = a.size
    K = 2 * n - 1
    
    A_ub = _build_sparse_constraints(a)
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': timeout, 'presolve': True})
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    t_val = float(result.x[n])
    return g, t_val


def _optimize_lp(a, deadline, verbose=True):
    """LP-based minimax optimization with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 100
    rng = np.random.RandomState(2024)
    
    if verbose:
        print(f"    Starting LP opt: n={n}, score={best_score:.8f}", flush=True)
    
    while time.monotonic() < deadline:
        it += 1
        
        # Solve LP
        g, t_val = _solve_lp_minimax(a, timeout=5.0)
        
        if g is None:
            # LP failed, perturb and retry
            a = best_a.copy()
            noise = rng.standard_normal(n) * 0.02 * a.mean()
            a = a + noise
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
            continue
        
        # Line search
        best_t = 0
        best_t_score = best_score
        for step in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            a_trial = (1 - step) * best_a + step * g
            a_trial = np.clip(a_trial, 0, 1000)
            s_trial = a_trial.sum()
            if s_trial < 0.01:
                continue
            a_trial = a_trial / s_trial
            sc = _score(a_trial)
            if sc < best_t_score:
                best_t_score = sc
                best_t = step
        
        if best_t > 0:
            a = (1 - best_t) * best_a + best_t * g
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
            best_a = a.copy()
            best_score = best_t_score
            last_improve = it
        else:
            if it - last_improve > patience:
                # Restart
                a = best_a.copy()
                scale = 0.1
                noise = rng.standard_normal(n) * scale * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(50, int(patience * 1.3))
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - (deadline - 2400)  # approx
            print(f"    LP iter {it}: score={best_score:.8f}", flush=True)
    
    return best_a, best_score


def _gradient_polish(a, deadline):
    """Gradient-based polish using FFT gradients."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    last_improve = 0
    patience = 300
    lr = 0.1 * n
    rng = np.random.RandomState(555)
    
    while time.monotonic() < deadline:
        it += 1
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        M = float(np.max(c))
        
        # Soft-max weights (temperature adapted)
        T = 50.0 + it * 0.5
        c_shift = c - M
        w = np.exp(c_shift / (M / T) if M > 0 else c_shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: grad[j] = 2 * sum_k w[k] * (a * delta_k)[j]
        # where (a * delta_k)[j] = a[k-j] if 0<=k-j<n else 0
        # This is the cross-correlation of a with w, times 2
        N = _next_pow2(2 * n)
        fa = rfft(a, N)
        fw = rfft(w, N)
        grad = irfft(fa * np.conj(fw), N)[:n] * 2.0
        
        # Project to tangent of simplex
        grad = grad - grad.mean()
        
        # Mirror descent update
        a = a * np.exp(-lr / n * grad)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
```

The LP is:
  min t
  s.t. (a*g)[k] <= t, k=0..2n-2
       sum(g) = 1
       g >= 0

The optimal t* tells us the minimum possible peak of conv(a, g) over all g on the simplex.

If t* < max(conv(a, a)), then moving towards g* should help.

But there's a subtlety: the LP finds g* that minimizes the peak of the CROSS-convolution conv(a, g*), not the AUTO-convolution. The auto-convolution of the updated sequence a' = (1-s)a + s*g* is:
conv(a', a') = (1-s)² conv(a,a) + 2s(1-s) conv(a,g*) + s² conv(g*,g*)

The middle term is bounded by t*, but the last term conv(g*,g*) is unbounded (well, bounded by sum(g*²) but potentially large if g* is peaked).

For small s, the s² term is negligible, so the update is effective. This is why line search with small step sizes should work.

```python
"""
LP-based minimax optimization for low-peak non-negative step functions.

Core algorithm:
1. Initialize with sinusoidal/multi-sine sequences and height_sequence_1.
2. Iteratively solve a sparse LP to find g on the simplex that minimizes
   max_k (a*g)[k] (the cross-convolution peak), then update a towards g
   with line search.
3. Use subset-of-constraints LP for speed, full LP when stuck.
4. Complement with gradient-based polish using FFT.
5. Try multiple sequence lengths (1024, 2048) and restarts.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_sine(n, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _build_lp_matrix(a, k_indices):
    """Build sparse constraint matrix for minimax LP given subset of peak indices."""
    n = a.size
    K_use = len(k_indices)

    rows = []
    cols = []
    vals = []

    for ii in range(K_use):
        k = k_indices[ii]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        ns = j_end - j_start + 1
        m_start = k - j_end
        # g-coefficients: a[k-m] for m = m_start..m_start+ns-1
        # which is a[j_end], a[j_end-1], ..., a[j_start] = a[j_start:j_end+1][::-1]
        for jj in range(ns):
            rows.append(ii)
            cols.append(m_start + jj)
            vals.append(float(a[j_start + jj]))
        # -t term
        rows.append(ii)
        cols.append(n)
        vals.append(-1.0)

    return csr_matrix((vals, (rows, cols)), shape=(K_use, n + 1))


def _solve_lp_minimax(a, k_indices=None, timeout=5.0):
    """
    Solve: minimize t
           s.t. (a*g)[k] <= t for k in k_indices (or all k)
                sum(g) = 1
                g >= 0
    Returns (g, t) or (None, None).
    """
    n = a.size
    K = 2 * n - 1

    if k_indices is None:
        k_indices = np.arange(K)
    k_indices = np.asarray(k_indices, dtype=np.int64)
    k_indices = k_indices[k_indices < K]

    if len(k_indices) == 0:
        return None, None

    A_ub = _build_lp_matrix(a, k_indices)
    b_ub = np.zeros(len(k_indices))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None, None

    if result.status != 0:
        return None, None

    g = result.x[:n]
    t_val = float(result.x[n])
    if g.sum() < 1e-15:
        return None, None
    return g, t_val

    while time.monotonic() < deadline:
        it += 1
        if it % 500 == 0:
            patience = max(100, int(patience * 1.3))

        c = _fft_convolve(a)
        if c.size == 0:
            break

        M = float(np.max(c))
        if M < 1e-15:
            break

        # Softmax weights on peaks
        T = 30.0 + 0.05 * it
        c_shift = c - M
        scale = M / T
        if scale < 1e-15:
            scale = 1e-15
        w = np.exp(c_shift / scale)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        # Gradient via FFT cross-correlation
        fa = rfft(a, N_fft)
        fw = rfft(w, N_fft)
        grad = irfft(fa * np.conj(fw), N_fft)[:n] * 2.0

        # Project to simplex tangent
        grad = grad - grad.mean()

        # Mirror descent (multiplicative update)
        a = a * np.exp(-lr / n * grad)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 3 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(0.3 * n, lr * 1.005)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.008 * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                lr = max(0.05 * n, lr * 0.85)
                T = max(20.0, T * 0.9)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    Grad iter {it} ({elapsed:.0f}s): score={best_score:.8f}", flush=True)

    return best_a, best_score


def _scan_init(deadline, n, rng):
    """Scan for best initialization of given length."""
    best_a = None
    best_score = np.inf

    configs = []
    for amp in [0.5, 0.6, 0.7, 0.8, 0.85, 0.9, 0.95, 1.0, 1.1, 1.3]:
        for phase in [0.0, 0.25, 0.4, 0.5]:
            configs.append(('sine', amp, phase))

    for _ in range(8):
        a1 = rng.uniform(0.4, 1.0)
        a2 = rng.uniform(0.1, 0.5)
        p1 = rng.uniform(0.0, 0.5)
        p2 = rng.uniform(0.0, 1.0)
        configs.append(('multi', a1, a2, p1, p2))

    for cfg in configs:
        if time.monotonic() > deadline:
            break
        if cfg[0] == 'sine':
            a = _make_sine(n, cfg[1], cfg[2])
        else:
            a = _make_multi_sine(n, [cfg[1], cfg[3-1]], [cfg[2], cfg[4-1]])
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is None:
        best_a = _make_sine(n, 0.8, 0.5)
        best_score = _score(best_a)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP-based minimax, gradient polish,
    and multiple initializations/lengths.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    print(f"[Start] Budget: {budget_s}s, deadline in {hard_deadline - t0:.0f}s", flush=True)

    all_candidates = []

    # Phase 1: Initialization scan (n=1024, n=2048)
    print("\n[Phase 1] Initialization scan...", flush=True)
    init_deadline = min(t0 + 45.0, hard_deadline - 200.0)

    inits = {}
    for n in [1024, 2048]:
        if init_deadline - time.monotonic() < 5.
            break
        idx = n // 512
        idl = init_deadline if idx == 1 else min(init_deadline, t0 + 30.0)
        a_init, s_init = _scan_init(idl, n, rng)
        inits[n] = (a_init, s_init)
        all_candidates.append(a_init.copy())
        print(f"  n={n}: init score={s_init:.8f}", flush=True)

    # Try height_sequence_1
    try:
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        if hs1.size > 0:
            hs1 = np.clip(hs1, 0, 1000)
            s_hs1 = hs1.sum()
            if s_hs1 > 0.01:
                hs1 = hs1 / s_hs1
                sc_hs1 = _score(hs1)
                print(f"  height_sequence_1 (n={hs1.size}): score={sc_hs1:.8f}", flush=True)
                all_candidates.append(hs1.copy())
                if hs1.size in inits:
                    if sc_hs1 < inits[hs1.size][1]:
                        inits[hs1.size] = (hs1, sc_hs1)
                else:
                    inits[hs1.size] = (hs1, sc_hs1)
    except Exception as e:
        print(f"  (height_sequence_1 not available: {e})", flush=True)

    # Phase 2: LP optimization (primary workhorse)
    # Allocate time based on remaining budget
    remaining = hard_deadline - time.monotonic()
    lp_frac_n1024 = 0.55
    lp_frac_grad = 0.25
    lp_frac_other = 0.15
    val_frac = 0.05

    print(f"\n[Phase 2] LP optimization ({remaining:.0f}s remaining)", flush=True)

    # LP on n=1024 (or whatever best init we have for that size)
    n_primary = 1024
    if n_primary in inits:
        a_start, s_start = inits[n_primary]
        lp_deadline_1 = time.monotonic() + remaining * lp_frac_n1024
        print(f"  LP n={n_primary} ({lp_deadline_1 - time.monotonic():.0f}s)...", flush=True)
        a_lp1, s_lp1 = _optimize_lp(a_start, lp_deadline_1)
        print(f"  LP n={n_primary} done: score={s_lp1:.8f}", flush=True)
        all_candidates.append(a_lp1.copy())

        # Gradient polish from LP result
        grad_deadline = min(hard_deadline - 60.0, lp_deadline_1 + remaining * lp_frac_grad)
        if grad_deadline - time.monotonic() > 30:
            print(f"  Grad polish n={n_primary} ({grad_deadline - time.monotonic():.0f}s)...", flush=True)
            a_gr1, s_gr1 = _gradient_polish(a_lp1, grad_deadline)
            print(f"  Grad n={n_primary} done: score={s_gr1:.8f}", flush=True)
            all_candidates.append(a_gr1.copy())
    elif 2048 in inits:
        n_primary = 2048
        a_start, s_start = inits[n_primary]
        lp_deadline_1 = time.monotonic() + remaining * lp_frac_n1024
        print(f"  LP n={n_primary} ({lp_deadline_1 - time.monotonic():.0f}s)...", flush=True)
        a_lp1, s_lp1 = _optimize_lp(a_start, lp_deadline_1)
        print(f"  LP n={n_primary} done: score={s_lp1:.8f}", flush=True)
        all_candidates.append(a_lp1.copy())

        grad_deadline = min(hard_deadline - 60.0, lp_deadline_1 + remaining * lp_frac_grad)
        if grad_deadline - time.monotonic() > 30:
            print(f"  Grad polish n={n_primary} ({grad_deadline - time.monotonic():.0f}s)...", flush=True)
            a_gr1, s_gr1 = _gradient_polish(a_lp1, grad_deadline)
            print(f"  Grad n={n_primary} done: score={s_gr1:.8f}", flush=True)
            all_candidates.append(a_gr1.copy())

    # LP on n=2048 (if not already done)
    n_secondary = 2048 if n_primary == 1024 else 1024
    if n_secondary in inits and time.monotonic() < hard_deadline - 60:
        a_start2, s_start2 = inits[n_secondary]
        lp_deadline_2 = min(hard_deadline - 40.0, time.monotonic() + remaining * lp_frac_other)
        if lp_deadline_2 - time.monotonic() > 30:
            print(f"  LP n={n_secondary} ({lp_deadline_2 - time.monotonic():.0f}s)...", flush=True)
            a_lp2, s_lp2 = _optimize_lp(a_start2, lp_deadline_2)
            print(f"  LP n={n_secondary} done: score={s_lp2:.8f}", flush=True)
            all_candidates.append(a_lp2.copy())

            # Quick gradient polish
            if time.monotonic() < hard_deadline - 20:
                a_gr2, s_gr2 = _gradient_polish(a_lp2, hard_deadline - 15)
                print(f"  Grad n={n_secondary} done: score={s_gr2:.8f}", flush=True)
                all_candidates.append(a_gr2.copy())

    # Phase 3: Additional LP from gradient-polished result (if time)
    if time.monotonic() < hard_deadline - 30:
        best_local = min(all_candidates, key=lambda x: _score(x))
        sc_best = _score(best_local)
        print(f"\n[Phase 3] Extra LP from best (score={sc_best:.8f}, n={best_local.size})...", flush=True)
        extra_deadline = hard_deadline - 15
        a_extra, s_extra = _optimize_lp(best_local, extra_deadline)
        print(f"  Extra LP done: score={s_extra:.8f}", flush=True)
        all_candidates.append(a_extra.copy())

    # Phase 4: Validation
    print(f"\n[Phase 4] Validating {len(all_candidates)} candidates...", flush=True)

    # Deduplicate by (size, rounded values)
    scored = []
    seen = set()
    for a in all_candidates:
        key = (a.size, tuple(np.round(a, 6).astype(np.int64)[:10]))
        if key in seen:
            continue
        seen.add(key)
        sc = _score(a)
        scored.append((sc, a))
    scored.sort(key=lambda x: x[0])

    best_out = None
    best_actual = np.inf

    for sc, a in scored:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in a]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  n={a.size}: local={sc:.8f}, actual={actual:.8f}", flush=True)
            if best_actual < 1.48:
                break

    if best_out is None:
        a = _make_sine(1024, 0.8)
        best_out = [float(x) for x in a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = _score(a)
        print(f"  Fallback: {best_actual:.8f}")

    total = time.monotonic() - t0
    print(f"\n[Done] Time: {total:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
```
