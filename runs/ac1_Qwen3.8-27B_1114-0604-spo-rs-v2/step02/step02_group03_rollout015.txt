import numpy as np

def _normalize(v):
    """Project to non-negative vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of 2 >= m, at least 2."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Score = 2*n*max(autocorr) for normalized f."""
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _lag_gradient(k, f):
    """
    Gradient of c_k = (f*f)_k with respect to f.
    dc_k/df_i = f_{k-i} + f_{i-k} (with f_j = 0 for j out of [0, n-1]).
    """
    n = f.size
    g = np.zeros(n, dtype=np.float64)
    # dc_k/df_i = f_{k-i} + f_{i-k}
    # For i in [0, n-1]:
    #   f_{k-i} is valid when 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
    #   f_{i-k} is valid when 0 <= i-k <= n-1, i.e., k <= i <= k+n-1
    
    lo1 = max(0, k - (n - 1))
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        # f_{k-i} for i in [lo1, hi1]
        idx = np.arange(lo1, hi1 + 1)
        g[lo1:hi1+1] += f[k - idx]
    
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        # f_{i-k} for i in [lo2, hi2]
        idx = np.arange(lo2, hi2 + 1)
        g[lo2:hi2+1] += f[idx - k]
    
    return g


def _subgradient_top(c, f, K=10, temperature=1.0):
    """
    Compute a weighted subgradient using the top K autocorrelation lags.
    Weights are softmax over the top K values.
    Returns gradient vector of length n.
    """
    n = f.size
    L = c.size
    # Take top K lags
    if K >= L:
        top_idx = np.arange(L)
        top_vals = c.copy()
    else:
        top_idx = np.argpartition(c, L - K)[L - K:]
        top_vals = c[top_idx]
    
    # Softmax weights
    if temperature > 0:
        z = (top_vals - top_vals.max()) / temperature
        np.clip(z, -40, 0, out=z)
        w = np.exp(z)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= w_sum
    else:
        w = np.ones(len(top_idx)) / len(top_idx)
    
    # Accumulate weighted gradient
    g = np.zeros(n, dtype=np.float64)
    for idx_w, idx_k in zip(w, top_idx):
        if idx_w < 1e-10:
            continue
        g += idx_w * _lag_gradient(int(idx_k), f)
    
    return g

def _normalize(v):
    """Project to non-negative vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of 2 >= m, at least 2."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Score = 2*n*max(autocorr) for sum-normalized non-negative f."""
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _subgradient_fft(c, f, K=20, temperature=0.1):
    """
    Compute softmax-weighted subgradient of max(c) over top K lags.
    Uses two FFT convolutions for O(n log n) efficiency.

        return np.zeros(n, dtype=np.float64)

    z = (vals - peak_val) / max(temperature * peak_val, 1e-300)
    np.clip(z, -40.0, 0.0, out=z)

    s = float(w_vals.sum())
    if s < 1e-300:

    else:
        w_vals /= s

    w[idx] = w_vals

def _lag_gradient_single(k, f):
    """Gradient of c_k with respect to f: g_i = f_{k-i} + f_{i-k}."""
    n = f.size
    g = np.zeros(n, dtype=np.float64)
    # f_{k-i} valid for i in [max(0,k-n+1), min(n-1, k)]
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:

    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += f[lo2 - k:hi2 - k + 1]
    return g


def _transfer_max(c, f, i, j, dv):
    """Exact max of (f*f) after f[i]-=dv, f[j]+=dv."""
    n = f.size
    L = c.size
    delta = np.zeros(L, dtype=np.float64)
    # Linear terms
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    # Quadratic terms
    q = dv * dv
    delta[2 * j] += q
    delta[2 * i] += q
    delta[i + j] -= 2.0 * q
    new_c = c + delta
    return float(np.max(new_c))

    else:
        D = 16
        R = 16

    fracs = (0.01, 0.03, 0.08, 0.15, 0.30, 0.50)

    while time.monotonic() < deadline and no_improve < 15:
        maxc = float(np.max(c))
        if maxc <= 0:
            break

        # Identify top lags
        K_active = min(L, max(10, n // 50))
        top_lags = np.argpartition(c, L - K_active)[L - K_active:]
        top_lags = top_lags[c[top_lags] > maxc * 0.8]  # only near-peak lags

        if len(top_lags) == 0:
            top_lags = np.array([int(np.argmax(c))])

        # Compute gradient for the top lag (dominant one)
        k_star = int(np.argmax(c[top_lags]))
        k_star = int(top_lags[k_star])
        g = _lag_gradient_single(k_star, f)

        # Also use subgradient over top lags for diversity
        g_sub = _subgradient_fft(c, f, K=len(top_lags), temperature=0.05)
        g_mix = 0.5 * g + 0.5 * g_sub

        # Select donors (high gradient) and receivers (low gradient)
        donor_order = np.argsort(g_mix)[::-1]
        recv_order = np.argsort(g_mix)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-10:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 8:
                break
            # Random fallback
            donors = [int(rng.integers(0, n)) for _ in range(min(D, n // 2))]
            donors = [d for d in donors if f[d] > 1e-10]
            receivers = [int(rng.integers(0, n)) for _ in range(min(R, n // 2))]
            receivers = [r for r in receivers if r not in set(donors)]
            if not donors or not receivers:
                no_improve += 1
                continue

        best_mc_cur = maxc
        best_move = None

        # Try structured transfers
        for i in donors:
            fi = float(f[i])
            if fi <= 1e-10:
                continue
            for j in receivers:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    mc = _transfer_max(c, f, i, j, dv)
                    if mc < best_mc_cur - 1e-15:
                        best_mc_cur = mc
                        best_move = (i, j, dv)

        # Random fallback if no improvement found
        if best_move is None:
            for _ in range(40):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-10:
                    continue
                frac = float(rng.uniform(0.01, 0.30))
                dv = float(f[i]) * frac
                mc = _transfer_max(c, f, i, j, dv)
                if mc < best_mc_cur - 1e-15:
                    best_mc_cur = mc
                    best_move = (i, j, dv)

        if best_move is None:
            no_improve += 1
            continue

        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        # Recompute autocorrelation periodically to avoid drift
        if move_count % 30 == 0:
            f = _normalize(f)
            c = _autocorr(f)
        else:
            # Incremental update
            L2 = c.size
            c[i:i + n] -= dv * f  # approximate (using updated f)
            c[j:j + n] += dv * f
            # More precise: recompute
            pass  # Just recompute to be safe for correctness
            c = _autocorr(f)

        move_count += 1
        no_improve = 0
        f = _normalize(f)
        c = _autocorr(f)
        maxc = float(np.max(c))
        cur_score = float(2.0 * n * maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bmc = float(bc.max()) if bc.size else 0.0
    return bf, float(2.0 * n * bmc), bc


def _perturb(f, rng, strength=0.3):
    """Multiply by random exp noise and renormalize."""
    n = f.size
    r = float(rng.random())
    if r < 0.5:
        sigma = float(rng.uniform(0.1, strength))
        z = rng.standard_normal(n)
        f_new = f * np.exp(sigma * z)
    elif r < 0.8:
        alpha = float(rng.uniform(0.1, 0.5))
        noise = rng.random(n)
        s = noise.sum()
        if s > 1e-300:
            noise /= s
        f_new = (1.0 - alpha) * f + alpha * noise
    else:
        # Shift the mass distribution
        shift = int(rng.integers(-n // 10, n // 10 + 1))
        f_new = np.roll(f, shift)
        # Fade edges to avoid wrap-around artifacts
        fade = min(n // 20, 50)
        if fade > 1:
            for k in range(fade):
                w = (k + 1) / (fade + 1)
                f_new[k] *= w
                f_new[n - 1 - k] *= w
    return _normalize(f_new)


def _gradient_descent(f0, budget_s, rng, tag=""):
    """
    Main gradient descent loop with subgradient, line search,
    coordinate polish, and perturbation.
    """
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0.copy()
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    eta = 0.5  # step size
    min_eta = 1e-8
    K_lags = min(20, max(5, n // 200))
    temperature = 0.08
    no_accept = 0
    it = 0
    last_polish = time.monotonic()

    if n < 1000:
        polish_interval = 0.3
        polish_budget = 0.2
    elif n < 5000:
        polish_interval = 0.5
        polish_budget = 0.4
    else:
        polish_interval = 0.8
        polish_budget = 0.6

    print(f"[{tag}] start n={n} score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0:
            break

        # Compute subgradient
        g = _subgradient_fft(c, f, K=K_lags, temperature=temperature)
        g_mean = float(g.mean())
        g_centered = g - g_mean
        g_range = float(g_centered.max() - g_centered.min())

        if g_range < 1e-15:
            no_accept += 1
            break

        # Try multiple step sizes (log-domain and Euclidean)
        best_mc_try = maxc
        best_f_try = None

        steps = [eta * 0.1, eta * 0.3, eta, eta * 3.0, eta * 10.0]

        for s in steps:
            if s < min_eta:
                continue
            # Log-domain step
            arg = -s * (g_centered / g_range)
            np.clip(arg, -30.0, 30.0, out=arg)
            ft = f * np.exp(arg)
            ft = _normalize(ft)
            ct = _autocorr(ft)
            mc = float(ct.max())
            if mc < best_mc_try - 1e-15:
                best_mc_try = mc
                best_f_try = ft

            # Euclidean projected step
            v = f - s * (g_centered / g_range)
            ft2 = _project_simplex(v)
            ft2 = _normalize(ft2)
            ct2 = _autocorr(ft2)
            mc2 = float(ct2.max())
            if mc2 < best_mc_try - 1e-15:
                best_mc_try = mc2
                best_f_try = ft2

        if best_f_try is not None and best_mc_try < maxc - 1e-14:
            f = best_f_try
            c = _autocorr(f)
            maxc = float(c.max())
            no_accept = 0
            eta = min(eta * 1.2, 5.0)  # slightly increase step
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] NEW BEST: {best_score:.6f} (iter {it})")
        else:
            no_accept += 1
            eta = max(min_eta, eta * 0.7)
            temperature = min(temperature * 0.98, 0.5)

            # Periodic coordinate polish
            now = time.monotonic()
            if now - last_polish > polish_interval or no_accept > 25:
                fp, sp, cp = _coordinate_polish(f, c, polish_budget, rng)
                if sp < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    if sp < best_score:
                        best_score = sp
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coord NEW BEST: {best_score:.6f}")
                    no_accept = 0
                    eta = max(eta, 0.3)
                else:
                    if no_accept > 50:
                        # Perturb to escape local minimum
                        f = _perturb(best_f, rng, strength=0.4)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        eta = max(eta, 0.8)
                        temperature = min(0.3, temperature * 2.0)
                        no_accept = 0
                last_polish = time.monotonic()

        if it % 5000 == 0:
            cur_score = float(2.0 * n * maxc)
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={cur_score:.6f} eta={eta:.3e} no_acc={no_accept}")

    # Final check
    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] done best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _interp_to_length(f, n):
    """Linearly resample f to length n and normalize."""
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 1:
        return _normalize(np.full(n, 1.0))
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _asymmetric_seed(n, alpha, beta, rng):
    """
    Asymmetric power-law seed: f_j proportional to (j+1)^alpha * (n-j)^beta.
    Asymmetry (alpha != beta) is crucial for reducing center autocorrelation.
    """
    j = np.arange(n, dtype=np.float64)
    f = (j + 1.0) ** alpha * (n - j) ** beta
    f += 1e-10
    return _normalize(f)


def _smooth_seed(n, rng):
    """Smooth random seed with a few harmonics."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    u += float(rng.uniform(-0.5, 0.5)) * (2.0 * x - 1.0)
    for m in (1, 2, 3, 5, 7):
        u += 0.3 * float(rng.uniform(-1, 1)) * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0, 2 * np.pi)))
    f = np.exp(0.8 * u)
    return _normalize(f)


def _bump_seed(n, center, width, rng):
    """Gaussian bump at given center with given width."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    c = float(center) + float(rng.uniform(-0.05, 0.05))
    w = float(width) * float(rng.uniform(0.7, 1.3))
    f = np.exp(-0.5 * ((x - c) / w) ** 2)
    return _normalize(f)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + max(120.0, float(budget_s)) - 30.0

    # Get starting construction
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(2048, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(2048, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n_full = int(f0.size)
    if n_full < 5:
        f0 = np.ones(2048, dtype=np.float64)
        n_full = f0.size
    f0 = _normalize(f0)

    # Evaluate starting point
    score0 = _fast_score(f0)
    print(f"Starting n={n_full}, initial score={score0:.6f}")
    print(f"Target: {TARGET_SCORE}")
    print(f"Budget: {budget_s}s")

    best_f = f0.copy()
    best_score = score0

    # === Phase 1: Coarse exploration at multiple resolutions ===
    print("\n=== Phase 1: Coarse exploration ===")
    coarse_n = max(256, n_full // 8)
    coarse_budget = min(120.0, 0.08 * (total_deadline - time.monotonic()))
    if coarse_budget < 30:
        coarse_budget = 30.0

    coarse_f = _interp_to_length(f0, coarse_n)
    seeds_low = [coarse_f]

    # Add diverse seeds at coarse resolution
    for alpha in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        for beta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
            seeds_low.append(_asymmetric_seed(coarse_n, alpha, beta, rng))
    for _ in range(8):
        seeds_low.append(_smooth_seed(coarse_n, rng))
    for _ in range(6):
        seeds_low.append(_bump_seed(coarse_n, float(rng.uniform(0.2, 0.8)),
                                    float(rng.uniform(0.1, 0.3)), rng))

    # Evaluate and sort seeds
    scored_seeds = []
    for i, s in enumerate(seeds_low):
        sc = _fast_score(s)
        if np.isfinite(sc):
            scored_seeds.append((sc, i, s))
    scored_seeds.sort(key=lambda x: x[0])

    # Optimize top coarse seeds
    n_coarse_opt = min(4, len(scored_seeds))
    per_seed_time = coarse_budget / max(1, n_coarse_opt)

    for idx in range(n_coarse_opt):
        if time.monotonic() > total_deadline - 600:
            break
        sc, _, fs = scored_seeds[idx]
        alloc = min(per_seed_time, total_deadline - time.monotonic() - 600)
        if alloc < 5:
            break
        # Upsample to medium resolution and optimize
        med_n = max(1024, n_full // 4)
        f_med = _interp_to_length(fs, med_n)
        f_opt, sc_opt = _gradient_descent(f_med, alloc, rng, f"coarse{idx}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = _interp_to_length(f_opt, n_full)
            print(f"Coarse seed {idx} improved full-res score: {best_score:.6f}")

    # === Phase 2: Medium resolution optimization ===
    print("\n=== Phase 2: Medium resolution ===")
    med_n = max(2048, n_full // 3)
    med_budget = min(600.0, 0.25 * (total_deadline - time.monotonic()))
    if med_budget < 60:
        med_budget = max(30.0, total_deadline - time.monotonic() - 300)

    # Start from best full-res downsampled, plus the best coarse result
    f_med_start = _interp_to_length(best_f, med_n)
    f_opt, sc_opt = _gradient_descent(f_med_start, med_budget, rng, "medium")
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = _interp_to_length(f_opt, n_full)

    # === Phase 3: Full resolution optimization ===
    print("\n=== Phase 3: Full resolution ===")
    full_budget = max(60.0, total_deadline - time.monotonic() - 200)

    # Multiple restarts at full resolution
    n_restarts = min(3, max(1, int(full_budget / 400)))
    per_restart = full_budget / max(1, n_restarts)

    for ri in range(n_restarts):
        if time.monotonic() > total_deadline - 250:
            break
        if ri == 0:
            f_start = best_f.copy()
        elif ri == 1:
            # Perturbed version
            f_start = _perturb(best_f, rng, strength=0.3)
        else:
            # Fresh seed at full resolution
            f_start = _interp_to_length(seeds_low[0][2], n_full)
            f_start = _perturb(f_start, rng, strength=0.2)

        alloc = min(per_restart, total_deadline - time.monotonic() - 250)
        if alloc < 20:
            break
        f_opt, sc_opt = _gradient_descent(f_start, alloc, rng, f"full_r{ri}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"Full restart {ri}: new best {best_score:.6f}")

    # === Phase 4: Intensive coordinate polish ===
    print("\n=== Phase 4: Coordinate polish ===")
    polish_time = max(30.0, total_deadline - time.monotonic() - 20)
    if polish_time > 20:
        f_pol, sc_pol, _ = _coordinate_polish(best_f, _autocorr(best_f), polish_time, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"Polish improved to: {best_score:.6f}")

    # === Final evaluation ===
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    s = float(best_f.sum())
    if s > 1e-300:
        best_f /= s

    final_c = _autocorr(best_f)
    final_score = float(2.0 * best_f.size * final_c.max())
    if final_score < best_score:
        best_score = final_score

    # Official evaluation
    try:
        out_list = [float(x) for x in best_f]
        official = evaluate_sequence(out_list)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.6f} target={TARGET_SCORE} time={elapsed:.1f}s n={best_f.size}")

    return [float(x) for x in best_f]
