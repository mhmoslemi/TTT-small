"""
Hybrid optimizer combining L-BFGS (LSE-smoothed), LP-based refinement (CVXPY),
and spectral shaping (GM-like) to minimize 2n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. LP-based direction finding: solves max sum(b) s.t. conv(a,b)<=M, b>=0
   to find "anti-aligned" update directions
2. Spectral shaping: GM-like iterations to flatten power spectrum
3. Chirp and M-sequence initializations for diverse starting points
4. Adaptive step-size control for LP mixing
"""

import numpy as np
from scipy.optimize import minimize
import time
import scipy.sparse as sp
import cvxpy as cp
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=60.0):
    """Run L-BFGS-B on LSE-smoothed score with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    res = None
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun,
                     'maxls': 50}
        )
    except Exception:
        pass
    elapsed = time.monotonic() - t0
    if res is not None:
        a_out = np.clip(res.x, 0.0, 1000.0)
        sc = compute_score(a_out)
        return sc, a_out, elapsed
    return compute_score(a0), a0, elapsed


def build_convolution_matrix(a):
    """Build sparse convolution matrix A where (A@b)_k = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M, solver=cp.CLARABEL):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b or None if failed.
    """
    n = len(a)
    if n > 8192:
        return None
    A = build_convolution_matrix(a)
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            b_val = np.clip(b.value, 0.0, 1000.0)
            if b_val.sum() > 1e-6:
                return b_val
    except Exception:
        pass
    return None


def spectral_shaping(a, n_iters=200, target_smooth=0.5):
    """
    GM-like spectral shaping to flatten the power spectrum.
    Alternates between:
    - In time domain: project to non-negative (clip)
    - In frequency domain: push |A| towards uniform
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    # Zero-pad for linear convolution -> circular
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a_padded = np.zeros(M_fft)
    a_padded[:n] = a

    for it in range(n_iters):
        # FFT
        A = np.fft.rfft(a_padded)
        amp = np.abs(A)
        # Target: make amplitude more uniform
        amp_mean = amp.mean()
        # Soft projection: push amplitudes towards mean
        amp_new = amp + target_smooth * (amp_mean - amp)
        amp_new = np.maximum(amp_new, amp * 0.1)  # don't go to zero
        # Reconstruct with new amplitudes, same phases
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        # Back to time domain
        a_padded = np.fft.irfft(A_new, M_fft)
        # Project to non-negative
        a_padded = np.maximum(a_padded, 0.0)
        a = a_padded[:n].copy()

    return a


def chirp_init(n, rng, kind='quadratic'):
    """Generate chirp-based initialization."""
    x = np.arange(n, dtype=np.float64) / n
    if kind == 'quadratic':
        return 1.0 + 0.4 * np.sin(2.0 * np.pi * x * x)
    elif kind == 'cubic':
        return 1.0 + 0.3 * np.sin(2.0 * np.pi * x * x * x)
    elif kind == 'fractal':
        val = 1.0
        amp = 0.3
        for k in range(1, 8):
            val += amp * np.sin(2.0 * np.pi * (2.0 ** k) * x + rng.random() * 2 * np.pi)
            amp *= 0.6
        return np.maximum(val, 0.05)
    else:
        return 1.0 + 0.4 * np.cos(2.0 * np.pi * x * (1.5 + rng.random()))


def msequence_init(n, rng):
    """Generate M-sequence-inspired initialization."""
    # Use a simple LFSR with a known good polynomial
    # For flexibility, use random ±1 and smooth
    mseq = rng.choice([-1.0, 1.0], size=n)
    # Smooth it a bit
    kernel = np.ones(3) / 3.0
    smoothed = np.convolve(mseq, kernel, mode='same')
    # Convert to non-negative
    a = 1.0 + 0.4 * smoothed
    return np.maximum(a, 0.05)


def walsh_init(n, rng):
    """Generate Walsh-Hadamard-inspired initialization (n must be power of 2)."""
    if n & (n - 1) != 0:
        return 1.0 + 0.3 * rng.standard_normal(n)
    # Compute Walsh matrix (row 0 is all ones)
    log_n = int(math.log2(n))
    # Use a few Walsh functions
    a = np.ones(n)
    for order in rng.choice(range(1, log_n + 1), size=min(3, log_n), replace=False):
        # Walsh function of given order
        w = np.zeros(n)
        for i in range(n):
            w[i] = 1.0 if (i >> order) & 1 == 0 else -1.0
        a += 0.15 * w
    return np.maximum(a * 0.8 + 0.4, 0.05)

    inits = []

    inits.append(np.ones(n) * 0.5)
    inits.append(chirp_init(n, rng, 'quadratic'))
    inits.append(chirp_init(n, rng, 'cubic'))
    inits.append(chirp_init(n, rng, 'fractal'))
    inits.append(msequence_init(n, rng))
    inits.append(walsh_init(n, rng))
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(np.clip(0.5 + 0.6 * rng.standard_normal(n), 0.01, 100.0))
    return [np.clip(init, 0.0, 1000.0) for init in inits]


def exact_subgrad_polish(a, total_iters=300):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def lp_refinement_phase(a, budget_s, deadline, rng):
    """
    LP-based refinement: iteratively solve LP, mix, polish with L-BFGS.
    Returns (best_score, best_a).
    """
    n = len(a)
    best_sc = compute_score(a)
    best_a = a.copy()

    if n > 4096:
        print(f"  LP phase: n={n} too large, skipping")
        return best_sc, best_a

    M = float(fast_conv(a).max())
    S = float(a.sum())

    # Build the matrix once (it changes as a changes, so we rebuild each time)
    # But for efficiency, limit iterations
    n_lp_iters = max(5, int(budget_s / 30))  # rough estimate
    t_mixed = 0.15

    for it in range(n_lp_iters):
        if time.monotonic() > deadline - 10:
            break

        M = float(fast_conv(best_a).max())
        S = float(best_a.sum())

        b_opt = solve_lp_direction(best_a, M)
        if b_opt is None or b_opt.sum() < 1e-6:
            print(f"  LP iter {it}: failed or trivial solution")
            break

        B = float(b_opt.sum())
        # Normalize b to same sum as a
        b_norm = b_opt * S / B

        # Mix with adaptive step
        for t in [t_mixed, t_mixed * 0.5, t_mixed * 2.0]:
            a_new = (1 - t) * best_a + t * b_norm
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc_new = compute_score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                print(f"  LP iter {it}: t={t:.3f}, score={sc_new:.8f} (improved)")
                break
        else:
            # No improvement with simple mix, try L-BFGS polish from mixed point
            a_try = (1 - t_mixed) * best_a + t_mixed * b_norm
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc_p, a_p, _ = run_lbfgs(a_try, 300, maxiter=2000, maxfun=40000,
                                      time_limit=30)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()
                print(f"  LP iter {it}: polish score={sc_p:.8f}")

        # Update t_mixed adaptively
        t_mixed *= 0.9

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Load parent solution if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = min(max(parent.size, 1024), 16384)
        lengths.append(pa)
    for n in [2048, 4096, 8192]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    lengths = lengths[:3]  # Limit to 3 lengths

    print(f"Lengths: {lengths}, Parent: {parent is not None}")
    print(f"Time budget: {budget_s}s")

    # ================================================================
    # Phase 1: L-BFGS with continuation (45% of time)
    # ================================================================
    phase1_deadline = t_start + 0.45 * (float(budget_s) - 15)
    print(f"\n=== Phase 1: L-BFGS (deadline at t={phase1_deadline - t_start:.0f}s) ===")

    beta_schedule = [30, 80, 200, 500]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = build_inits(n, rng, parent)

        # Limit initializations based on time
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 40)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            init_sc = compute_score(a_cur)
            print(f"  Init {i}: score={init_sc:.6f}")
            report(init_sc, a_cur, f"n={n}, init={i}")

            for beta in beta_schedule:
                if time.monotonic() > phase1_deadline:
                    break
                sc, a_opt, elapsed = run_lbfgs(a_cur, beta, maxiter=5000, maxfun=100000)
                a_cur = a_opt
                report(sc, a_cur, f"n={n}, init={i}, β={beta}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: LP-based refinement (30% of time)
    # ================================================================
    phase2_deadline = t_start + 0.75 * (float(budget_s) - 15)
    print(f"\n=== Phase 2: LP Refinement (deadline at t={phase2_deadline - t_start:.0f}s) ===")

    # Use the best solution, but if it's too large for LP, use a smaller one
    lp_a = best_a.copy()
    if len(lp_a) > 4096:
        # Try the best at a smaller length, or truncate
        if parent is not None and len(parent) <= 4096:
            lp_a = parent.copy()
        else:
            lp_a = best_a[:4096].copy() if len(best_a) >= 4096 else best_a.copy()

    lp_budget = max(60, phase2_deadline - time.monotonic())
    sc_lp, a_lp = lp_refinement_phase(lp_a, lp_budget, phase2_deadline, rng)
    report(sc_lp, a_lp, "LP phase")

    # Also try LP from a few other starting points
    for n_try in [2048, 4096]:
        if time.monotonic() > phase2_deadline - 30:
            break
        a_start = build_inits(n_try, rng, parent)[0]  # parent or uniform
        a_start = np.clip(a_start[:n_try], 0.0, 1000.0)
        sc_s, a_s, _ = run_lbfgs(a_start, 200, maxiter=2000, maxfun=40000, time_limit=30)
        if time.monotonic() > phase2_deadline - 15:
            break
        lp_budget2 = max(30, phase2_deadline - time.monotonic())
        sc_lp2, a_lp2 = lp_refinement_phase(a_s, lp_budget2, phase2_deadline, rng)
        report(sc_lp2, a_lp2, f"LP from n={n_try}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Spectral shaping (10% of time)
    # ================================================================
    phase3_deadline = t_start + 0.88 * (float(budget_s) - 15)
    print(f"\n=== Phase 3: Spectral Shaping (deadline at t={phase3_deadline - t_start:.0f}s) ===")

    n_spec = min(len(best_a), 4096)
    a_spec_base = best_a[:n_spec] if len(best_a) >= n_spec else best_a

    for n_sp in [n_spec // 2, n_spec]:
        if time.monotonic() > phase3_deadline:
            break
        for smooth in [0.3, 0.5, 0.7]:
            if time.monotonic() > phase3_deadline:
                break
            a_shaped = spectral_shaping(a_spec_base, n_iters=200, target_smooth=smooth)
            # Polish with L-BFGS
            sc_sp, a_sp, _ = run_lbfgs(a_shaped, 300, maxiter=5000, maxfun=100000,
                                       time_limit=60)
            report(sc_sp, a_sp, f"spectral n={n_sp}, smooth={smooth}")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish (remaining time)
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # High-beta LSE
    for beta in [800, 1500, 3000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        report(sc, a_opt, f"final β={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 20:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=500)
        report(sc, a_pol, "subgrad")

    # One more L-BFGS from subgrad result
    if time.monotonic() < deadline - 10:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final lbfgs")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
