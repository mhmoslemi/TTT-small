import numpy as np
import time

def _autocorr(f, fft_size):
    """Compute linear autocorrelation via FFT."""
    F = np.fft.rfft(f, fft_size)
    result = np.fft.irfft(F * F, fft_size)
    return result[:2*len(f)-1]

def _softmax_weights(c, t):
    """Compute softmax weights for smoothed max gradient."""
    c_shifted = c - c.max()
    w = np.exp(c_shifted / t)
    w /= w.sum()
    return w

def _grad_smoothed(f, c, w, m, rng):
    """
    Compute gradient of smoothed max using importance sampling.
    w: softmax weights (length 2n-1)
    m: number of sampled lags
    Returns gradient vector of length n.
    """
    n = len(f)
    # Sample m lags proportional to w
    # Use inverse CDF method
    cw = np.cumsum(w)
    u = rng.uniform(0, 1, m)
    idx = np.searchsorted(cw, u)
    idx = np.clip(idx, 0, 2*n-2)
    
    grad = np.zeros(n)
    for k in idx:
        # d(c[k])/d(f[i]) = 2*f[k-i] for valid i
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        i_range = np.arange(lo, hi)
        grad[lo:hi] += 2.0 * f[k - i_range] * w[k]
    
    return grad / (w[idx].sum() + 1e-30)  # Normalize by total weight sampled

        hi = min(n, k + 1)
        i_range = np.arange(lo, hi)
        grad[lo:hi] += 2.0 * f[k - i_range]
    return grad / m

Let me also think about a completely different angle: **can we exploit the problem structure analytically?**

The objective is 2n * max_k c[k] / (sum a)^2. Let's set sum a = 1 (WLOG by scale invariance).

We want to minimize max_k c[k] where c[k] = sum_{valid j} a[j] * a[k-j].

Note that c has a specific structure: it's the "autocorrelation" (in the signal processing sense) of a, but with full convolution (not circular).

For a "well-behaved" sequence (smooth, no sharp features), c is also smooth. The peak of c is determined by:
- c[0] = a[0]^2 (first element squared)
- c[1] = 2*a[0]*a[1]
- ...
- c[n-1] = sum a[i]*a[n-1-i] (the "anti-diagonal" sum)
- ...
- c[2n-2] = a[n-1]^2

For the peak to be small, we need:
1. a[0] small (to keep c[0] small)
2. a[n-1] small (to keep c[2n-2] small)
3. The "anti-diagonal" sum to be small
4. The "near-diagonal" sums (small lags) to be small

Condition 3 is the key: sum a[i]*a[n-1-i] small means a[i] and a[n-1-i] should be "anti-correlated" - when a[i] is large, a[n-1-i] should be small.

This suggests a sequence that is "lopsided" - large on one side, small on the other. But conditions 1 and 2 require both ends to be small. So the sequence should be like a "skewed bump" - rising from 0 at the left, peaking somewhere off-center (not in the middle), and falling back to 0 at the right.

If the peak is at position p (0 < p < n-1), then:
- a[i] is large near i=p
- a[n-1-i] is large near i=n-1-p
- For the anti-diagonal sum to be small, we want p and n-1-p to be far apart
- This means p should be close to 0 or n-1 (but not at the endpoints)
- So the peak should be in the first half (or second half) but not at the very edge

This suggests a shape like: a[i] = exp(-((i-p)^2)/(2*s^2)) where p = alpha*n for alpha < 0.5 (or > 0.5), and s is the width.

Let me test this hypothesis with a specific example. Let n=100, p=25, s=15:
a[i] = exp(-((i-25)^2)/450)
sum a ≈ s*sqrt(2*pi) ≈ 15*2.5 ≈ 37.5 (rough estimate for the Gaussian)
Let's normalize: a[i] /= 37.5

c[n-1] = sum a[i]*a[n-1-i] = sum a[i]*a[99-i]
a[99-i] = exp(-((99-i-25)^2)/450) = exp(-((74-i)^2)/450)

So c[n-1] involves products of exp(-((i-25)^2)/450) * exp(-((74-i)^2)/450)
= exp(-((i-25)^2 + (74-i)^2)/450)

The exponent: (i-25)^2 + (74-i)^2 = i^2-50i+625 + 5476-148i+i^2 = 2i^2-198i+6101
Minimum at i = 198/4 = 49.5
Value at i=49.5: 2*2450-198*49.5+6101 = 4900-9801+6101 = 1200
So min exponent = 1200/450 = 2.67, giving exp(-2.67) ≈ 0.069

Compare to the peak of a: exp(0) = 1.
The anti-diagonal sum is roughly n * 0.069 * (normalization factor) which is much smaller than the typical c[k] values near the peak.

This confirms the hypothesis: an off-center Gaussian should give a good solution.

"""
Step function optimizer using off-center bump initialization, 
importance-sampled mirror descent with SA, and structured coordinate descent polish.
Key improvements: better gradient estimation via importance sampling, 
asymmetric initialization exploiting the anti-diagonal structure, 
and multi-timescale optimization.
"""

import numpy as np
import time

def _autocorr(f, fft_size):
    n = len(f)
    F = np.fft.rfft(f, fft_size)
    result = np.fft.irfft(F * F, fft_size)
    return result[:2*n-1]

def _score(f, fft_size):
    c = _autocorr(f, fft_size)
    return 2.0 * len(f) * c.max()  # assuming sum(f) = 1

def _mirror_step(f, grad, eta):
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def _grad_from_lags(f, lags, weights):
    """Compute weighted gradient from specific lags."""
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        grad[lo:hi] += w * 2.0 * f[k - idx]
    return grad

def _optimize_mirror(f, deadline, rng, fft_size, T_init, m_samples):
    """
    Mirror descent with importance-sampled gradient, SA acceptance, momentum.
    """
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    c = _autocorr(f, fft_size)
    cur_peak = c.max()
    best_peak = cur_peak
    best_f = f.copy()
    
    eta = 0.5
    T = T_init * cur_peak
    momentum = np.zeros(n)
    beta = 0.9
    iter_count = 0
    last_improve = time.monotonic()
    stuck_count = 0
    
    while time.monotonic() < deadline:
        # Compute softmax weights for sampling
        t_smooth = max(0.01 * cur_peak, 1e-10)
        c_shifted = c - c.max()
        w = np.exp(c_shifted / t_smooth)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= w_sum
        
        # Sample m lags
        cw = np.cumsum(w)
        u = rng.uniform(0, 1, m_samples)
        sampled = np.searchsorted(cw, u)
        sampled = np.clip(sampled, 0, 2*n-2)
        
        # Get unique lags and their total weights
        unique_lags, counts = np.unique(sampled, return_counts=True)
        lag_weights = w[unique_lags] * counts / m_samples
        
        # Compute gradient
        grad = _grad_from_lags(f, unique_lags, lag_weights)
        
        # Momentum
        momentum = beta * momentum + (1-beta) * grad
        grad_eff = momentum
        grad_eff -= grad_eff.mean()  # Center the gradient
        
        # Mirror descent step
        f_new = _mirror_step(f, grad_eff, eta)
        
        c_new = _autocorr(f_new, fft_size)
        new_peak = c_new.max()
        
        # SA acceptance
        delta = new_peak - cur_peak
        if delta <= 0 or (T > 1e-20 and rng.uniform() < np.exp(-delta / T)):
            f = f_new
            cur_peak = new_peak
            eta = min(eta * 1.002, 2.0)
            if cur_peak < best_peak - 1e-14:
                best_peak = cur_peak
                best_f = f.copy()
                last_improve = time.monotonic()
        else:
            eta *= 0.9
        
        # Cooling
        T = max(1e-20, T * 0.999)
        eta *= 0.9999
        
        # Stuck detection and reheat
        stuck_time = time.monotonic() - last_improve
        if stuck_time > 10:
            # Reheat and perturb
            T = T_init * 0.1 * best_peak
            mag = min(0.03, 0.005 + 0.005 * stuck_time / 30.0)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr(f, fft_size)
            cur_peak = c.max()
            momentum *= 0.5
            last_improve = time.monotonic()
        
        c = _autocorr(f, fft_size)
        cur_peak = c.max()
        iter_count += 1
        
        if iter_count % 5000 == 0:
            print(f"    iter {iter_count}: peak={cur_peak:.10f}, best={best_peak:.10f}, eta={eta:.4f}, T={T:.2e}")
    
    # Final check
    c = _autocorr(f, fft_size)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, best_peak

def _coord_polish(f, deadline, rng, fft_size):

        c = _autocorr(f, fft_size)
        cur_peak = c.max()

            best_peak = cur_peak
            best_f = f.copy()
        
        # Get top 5 lags
        top_5 = np.argsort(c)[-5:]

        # Use exponential weights based on how close to max each lag is
        gap = cur_peak - c[top_5]
        w = np.exp(-gap / (0.005 * cur_peak + 1e-10))
        w /= w.sum()
        
        grad = _grad_from_lags(f, top_5, w)
        
        # Find donor (highest grad) and recipients (lowest grad)
        donor = int(np.argmax(grad))
        recv_idx = np.argsort(grad)[:30]

                s = f_try.sum()
                if s < 1e-15:

                f_try /= s
                c_try = _autocorr(f_try, fft_size)

                    f = f_try
                    improved = True
                    break
            if improved:

            k_star = int(np.argmax(c))
