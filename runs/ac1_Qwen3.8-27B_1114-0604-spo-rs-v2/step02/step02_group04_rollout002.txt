Low-peak step-function optimization using L-BFGS with softmax parameterization.
Combines sinusoidal initialization, multi-start, and active-set refinement.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.special import softmax
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_of_conv_max(a, K=10, T=5.0):
    """
    Compute smoothed subgradient of max(conv(a,a)) at a.
    Uses top-K peaks with softmax weighting.
    Returns (max_val, gradient_vector).
    """
    n = a.size
    c = _fft_convolve(a)
    
    if c.size <= K:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -K)[-K:]
    
    top_vals = c[top_idx]
    max_val = float(top_vals.max())
    
    # Softmax weights on top-K
    w = np.exp(T * (top_vals - max_val))
    w /= w.sum()
    
    # Gradient: sum_l w_l * d(conv)_k_l / da
    # d(conv)_k / da_j = 2 * a_{k-j} if 0 <= k-j < n
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-15:
            continue
        # For this peak k, grad_j = 2 * a_{k-j} for j where 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            a_idx = k - idx
            grad[idx] += w[i] * 2.0 * a[a_idx]
    
    return max_val, grad

softmax-smoothed max objective in log-space, multi-length exploration, and active-set refinement.

import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _score_normalized(a):
    """Score assuming sum(a)=1 (faster, no re-clipping)."""
    n = a.size
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))


def _grad_of_conv_max(a, K, T):
    """
    Compute smoothed subgradient of max(conv(a,a)) at a.
    Uses top-K peaks with softmax weighting.
    Returns (max_val, gradient_vector_of_length_n).
    """
    n = a.size
    c = _fft_convolve(a)

    if c.size <= K:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -K)[-K:]

    top_vals = c[top_idx]
    max_val = float(top_vals.max())

    # Softmax weights on top-K
    shifted = top_vals - max_val
    w = np.exp(T * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(shifted)] = 1.0
    else:
        w /= w_sum

    # Gradient: sum_l w_l * d(conv)_k_l / da
    # d(conv)_k / da_j = 2 * a_{k-j} if 0 <= k-j < n
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            a_idx = k - idx
            grad[idx] += w[i] * 2.0 * a[a_idx]

    return max_val, grad

    s = a.sum()

        a /= s
    else:
        a = np.ones(n) / n
    return a
