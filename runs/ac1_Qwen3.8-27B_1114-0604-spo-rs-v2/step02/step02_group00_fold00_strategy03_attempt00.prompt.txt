def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""

def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    LSE(c, beta) = c_max + (1/beta)*log(sum(exp(beta*(c-c_max))))
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: g_j = 2 * sum_m a[m] * p[m+j]  (cross-correlation)
    # g = 2 * conv(a_rev, p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def exact_subgrad_polish(a, total_iters=500):
    """
    Projected subgradient descent on the exact max-conv objective.
    Uses adaptive step size (Polyak-style when possible).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()

    # Polyak step: need a lower bound on optimal value
    # Use 1.0 as a conservative lower bound (score >= 2n/(2n-1) > 1)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        # Full gradient of score = 2n*(sg_M/S^2 - 2*M/(S^3))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Polyak step size

        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        a -= step_decay * 0.001 * grad / gn
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

def targeted_perturb(a, rng, strength=0.05):
    """
    Perturb the elements that contribute most to the peak convolution lag,
    shifting mass to elements that don't.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Find which a[j] contribute to c[k_star]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    # Contribution of a[j] to c[k_star] is a[k_star - j]
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    # Pick the top contributors and reduce them; redistribute to others
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:-top_count] if n > 2 * top_count else np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)
