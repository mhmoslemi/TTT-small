import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, L=None):
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    nout = 2*n - 1
    if L is None:
        L = _next_pow2(nout)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:nout]

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _conv_full(a)
    return 2.0 * n * c.max()

def _score(a):
    """General score: 2*n*max(conv(a,a)) / (sum(a))^2."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    return _score_normalized(a_norm)

def _gradient_soft(a, c, beta):
    """
    Compute soft gradient of max(c) using exp(β*c) weights.
    Returns grad array of length n where grad[j] = 2*sum_m w[j+m]*a[m].
    """
    n = len(a)
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    w = w / w.sum()
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Use FFT: R = IFFT(FFT(w_pad) * conj(FFT(a_pad)))
    L = _next_pow2(3 * n)
    w_pad = np.zeros(L)
    w_pad[:len(c)] = w
    a_pad = np.zeros(L)
    a_pad[:n] = a
    R = np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), L)
    grad = 2.0 * R[:n]
    return grad, w

"""
Optimized search for non-negative sequences minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses tilted-window initialization, FFT-based soft-gradient descent with adaptive sharpening,
frequency-domain flattening, and block coordinate refinement.
"""
import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, L=None):
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    nout = 2 * n - 1
    if L is None:
        L = _next_pow2(nout)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:nout].astype(float)


def _score_normalized(a):
    n = len(a)
    c = _conv_full(a)
    return float(2.0 * n * c.max())


def _score(a):
    s = a.sum()
    if s < 1e-10 or not np.isfinite(s):
        return float('inf')
    return _score_normalized(a / s)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _gradient_soft(a, c, beta, L=None):
    n = len(a)
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        return np.zeros(n), w
    w = w / ws
    if L is None:
        L = _next_pow2(3 * n)
    w_pad = np.zeros(L)
    w_pad[:len(c)] = w
    a_pad = np.zeros(L)
    a_pad[:n] = a
    R = np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), L)
    grad = 2.0 * R[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad, w


def _tilted_window(n, alpha, beta_tilt, rng=None):
    t = (np.arange(n) + 0.5) / n
    base = np.sin(np.pi * t) ** alpha
    tilt = 1.0 + beta_tilt * (2.0 * t - 1.0)
    a = base * tilt
    if rng is not None:
        a = a * np.exp(rng.normal(0, 0.02, n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _freq_flatten(a, power, L=None):
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _conv_full(a, L)
    c_max = c.max()
    if c_max < 1e-15:
        return a.copy()
    c_norm = c / c_max
    c_new = c_max * np.power(np.maximum(c_norm, 0.0), power)
    c_freq = np.fft.rfft(c_new, L)
    a_new = np.fft.irfft(np.sqrt(np.maximum(c_freq, 0.0)), L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.25:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _optimize_main(p0, rng, time_budget, beta_start=30.0, beta_end=5000.0, verbose=False):

    best_score = _score_normalized(p)
    cur_score = best_score
    beta = float(beta_start)
    eta = 0.15
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    if verbose:

    while time.monotonic() < deadline:

        if r < 0.60:

            sc, h, _ = _score_normalized(p), None, None
            grad, w = _gradient_soft(p, c, beta, L_grad)
            hmax = grad.max()
            if hmax < 1e-15:
                stall += 1
                continue
            grad_norm = grad / hmax
            improved = False
            for eta_try in (eta * 0.5, eta, eta * 1.8, eta * 3.0):

                        best_score = sc_new

                else:

                    cur_score = sc_try

                else:

                        cur_score = sc_try

                cur_score = best_score
                stall = 0

        it += 1

    return best, best_score


def _block_refine(p0, rng, time_budget):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_score = _score_normalized(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.35, 1.6, 2.0, 2.8)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 4))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 512), max(4, n // 64)))
        elif r < 0.8:
            bs = int(rng.integers(max(4, n // 64), max(8, n // 16)))
        else:
            bs = int(rng.integers(max(8, n // 16), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score_normalized(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s
