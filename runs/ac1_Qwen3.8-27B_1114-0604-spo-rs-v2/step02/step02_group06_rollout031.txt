    c = _compute_conv(a)

    c_max = c.max()
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    M = c_max + np.log(sum_exp) / beta
    w = exp_shifted / sum_exp

    a = np.exp(x - x.max())  # numerical stability
    n = len(a)
    S = a.sum()

    L = _next_pow2(2*n - 1)
    A = scipy.fft.rfft(a, L)
    c = np.maximum(scipy.fft.irfft(A * A, L)[:2*n-1], 0.0)

"""
Gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2
using log-parameterization, soft-max approximation of max, L-BFGS-B,
multi-scale search, and local refinement.
"""

import numpy as np
import time
import scipy.fft
from scipy.optimize import minimize

_GLOBAL_RNG = np.random.default_rng(42)


def _next_pow2(x):
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


def _compute_conv(a, L=None):
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = scipy.fft.rfft(a, L)
    c = scipy.fft.irfft(A * A, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    c = _compute_conv(a)
    return 2.0 * n * c.max() / (S * S)


def _soft_max(c, beta):
    c_max = c.max()
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    if sum_exp <= 0 or not np.isfinite(sum_exp):
        M = c_max
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return M, w
    M = c_max + np.log(sum_exp) / beta
    w = exp_s / sum_exp
    return M, w


def _obj_grad(x, beta, L_conv, L_grad):
    a = np.exp(x - x.max())
    n = len(a)
    S = a.sum()

    A = scipy.fft.rfft(a, L_conv)
    c = np.maximum(scipy.fft.irfft(A * A, L_conv)[:2 * n - 1], 0.0)

    M, w = _soft_max(c, beta)
    f = 2.0 * n * M / (S * S)

    A_full = scipy.fft.rfft(a, L_grad)
    Wr = scipy.fft.rfft(w[::-1], L_grad)
    D = scipy.fft.irfft(A_full * Wr, L_grad)
    conv_part = D[2 * n - 2 : n - 2 : -1].copy()

    diag_corr = a * w[0::2]
    dM_da = conv_part + diag_corr

    df_da = 2.0 * n / (S * S) * dM_da - 4.0 * n * M / (S ** 3)
    df_dx = a * df_dx_check = None

    # Remove mean to handle scale invariance (gradient should already be orthogonal)
    df_dx = a * df_da

    return f, df_dx


def _optimize_lbfgs(a0, time_limit, beta_start=5.0, beta_end=300.0, max_iter=200):
    """Optimize using L-BFGS-B with continuation on beta."""
    global _GLOBAL_RNG
    n = len(a0)
    x0 = np.log(np.maximum(a0, 1e-15))

    L_conv = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_x = x0.copy()
    best_f = _obj_grad(x0, beta_start, L_conv, L_grad)[0]

    deadline = time.monotonic() + time_limit
    n_beta = 6
    iters_per = max(20, max_iter // n_beta)

    for step in range(n_beta):
        if time.monotonic() > deadline - 5.0:
            break
        frac = step / max(1, n_beta - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        def obj(x):
            return _obj_grad(x, beta, L_conv, L_grad)[0]

        def grad(x):
            return _obj_grad(x, beta, L_conv, L_grad)[1]

        result = minimize(
            obj, x0, jac=grad, method='L-BFGS-B',
            options={'maxiter': iters_per, 'gtol': 1e-8, 'iprint': -1}
        )

        x0 = result.x.copy()

    a_best = a_best / a_best.sum()

        beta = beta_start + (beta_end - beta_start) * frac
        cur_lr = lr * (1.0 - 0.8 * frac) + 0.01

        f, g = _obj_grad(x, beta, L_conv, L_grad)
        if np.isfinite(f) and f < best_f:
            best_f = f
            best_x = x.copy()

        g = np.minimum(np.maximum(g, -10.0), 10.0)
        v = momentum * v + cur_lr * g
        x = x - v
        x = x - x.max()  # keep bounded

    a_best = np.exp(best_x - best_x.max())
    a_best = a_best / a_best.sum()
    return a_best, best_f


def _local_search(a0, time_limit):
    """Coordinate and block local search."""
    global _GLOBAL_RNG
    n = len(a0)
    a = a0.copy()
    sc = _score(a)
    best_a = a.copy()
    best_sc = sc

    L = _next_pow2(2 * n - 1)
    deadline = time.monotonic() + time_limit
    stagnant = 0

    block_sizes = [1, 2, 4, 8, 16, 32, 64, 128]
    factors = [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0]

    while time.monotonic() < deadline:
        r = _GLOBAL_RNG.random()
        if r < 0.4:
            w = 1
        elif r < 0.6:
            w = int(_GLOBAL_RNG.integers(2, 33))
        elif r < 0.8:
            w = int(_GLOBAL_RNG.integers(32, 129))
        else:
            w = int(_GLOBAL_RNG.integers(64, min(512, n)))
        w = min(w, n)
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))

        improved = False
        for fct in factors:
            cand = a.copy()
            cand[start:start + w] *= fct
            cand = np.maximum(cand, 1e-15)
            cand = cand / cand.sum()
            sc2 = _score(cand)
            if sc2 < sc - 1e-12:
                a = cand
                sc = sc2
                improved = True
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                noise = np.exp(_GLOBAL_RNG.normal(0, 0.1, n))
                a = a * noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                sc = _score(a)
                stagnant = 0
            if stagnant > 200:
                a = best_a.copy()
                sc = best_sc
                stagnant = 0

    return best_a, best_sc


def _resample(a, m):
    n = len(a)
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()


def _make_init(n, mode, rng):
    t = (np.arange(n) + 0.5) / n
    if mode == 0:
        # Linear ramp (asymmetric)
        a = 1.0 + 2.0 * t
    elif mode == 1:
        # Reverse ramp
        a = 1.0 + 2.0 * (1.0 - t)
    elif mode == 2:
        # Exponential decay
        rate = rng.uniform(2.0, 8.0)
        a = np.exp(-rate * t)
    elif mode == 3:
        # Sigmoid step
        c = rng.uniform(0.3, 0.7)
        k = rng.uniform(10.0, 50.0)
        a = 1.0 / (1.0 + np.exp(-k * (c - t)))
    elif mode == 4:
        # Two-level
        cut = rng.uniform(0.3, 0.7)
        ratio = rng.uniform(0.2, 3.0)
        idx = int(round(cut * n))
        a = np.ones(n)
        a[idx:] = ratio
    elif mode == 5:
        # Asymmetric Gaussian
        center = rng.uniform(0.2, 0.5)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((t - center) / width) ** 2)
        a *= (1.0 + 0.5 * (1.0 - t))
    elif mode == 6:
        # Power law
        p = rng.uniform(0.5, 3.0)
        tt = np.clip(t, 1e-6, 1.0 - 1e-6)
        a = tt ** p
    elif mode == 7:
        # Sinusoidal asymmetric
        freq = rng.uniform(1.0, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t + phase)
        a *= (1.0 + 0.3 * t)
    else:
        # Random smooth
        n_knots = int(rng.integers(3, 8))
        knots = np.sort(rng.uniform(0, 1, n_knots))
        heights = rng.lognormal(0, 0.8, n_knots)
        a = np.interp(t, knots, heights)
        a *= (0.5 + t)  # add asymmetry

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _GLOBAL_RNG
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0
    total = max(60.0, budget - 30.0)
    deadline = start_time + total

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    _GLOBAL_RNG = np.random.default_rng((iseed % 1000000007) + 1)

    # Try to use previous best as seed
    parent = None
    try:
        parent_raw = height_sequence_1
        if parent_raw is not None and len(parent_raw) > 10:
            parent = np.asarray(parent_raw, dtype=float)
            parent = np.maximum(parent, 0.0)
            s = parent.sum()
            if s > 0:
                parent = parent / s
    except (NameError, Exception):
        parent = None

    best_a = None
    best_sc = float('inf')

    def _update_best(a, sc):
        nonlocal best_a, best_sc
        if a is not None and np.isfinite(sc) and sc < best_sc:
            best_a = a.copy()
            best_sc = sc
            print(f"  New best: {sc:.8f} (n={len(a)})")

    # Phase 1: Small-scale search with multiple initializations
    print("Phase 1: Small-scale search")
    small_sizes = [128, 256, 512]
    n_modes = 8

    for size in small_sizes:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_per = min(rem * 0.25, 60.0)
        print(f"  Size {size}, time budget {t_per:.1f}s")

        candidates = []
        # Add parent if available
        if parent is not None:
            p_resamp = _resample(parent, size)
            candidates.append(p_resamp)

        # Random initializations
        for mode in range(n_modes):
            a_init = _make_init(size, mode, _GLOBAL_RNG)
            candidates.append(a_init)

        # Optimize each candidate
        n_cand = len(candidates)
        t_each = t_per / n_cand
        for i, a_init in enumerate(candidates):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(t_each, rem - 5.0)
            if t_alloc < 3:
                continue

            try:
                a_opt, sc = _optimize_lbfgs(a_init, t_alloc, beta_start=5, beta_end=100, max_iter=100)
                _update_best(a_opt, sc)
            except Exception:
                pass

    # Phase 2: Medium-scale search
    print(f"\nPhase 2: Medium-scale search (best so far: {best_sc:.8f})")
    med_sizes = [1024, 2048]
    for size in med_sizes:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.3, 300.0)
        print(f"  Size {size}, time budget {t_alloc:.1f}s")

        # Build initial candidates
        candidates = []
        if best_a is not None:
            candidates.append(_resample(best_a, size))
        if parent is not None:
            candidates.append(_resample(parent, size))
        for mode in [0, 1, 4, 5, 7]:
            candidates.append(_make_init(size, mode, _GLOBAL_RNG))

        n_cand = len(candidates)
        t_each = t_alloc / n_cand
        for i, a_init in enumerate(candidates):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_e = min(t_each, rem - 10.0)
            if t_e < 5:
                continue
            try:
                a_opt, sc = _optimize_lbfgs(a_init, t_e, beta_start=10, beta_end=200, max_iter=150)
                _update_best(a_opt, sc)
            except Exception:
                pass
            # Also try momentum method
            rem = deadline - time.monotonic()
            if rem > 20:
                try:
                    a_opt2, sc2 = _optimize_momentum(a_init, min(rem - 10, t_e * 0.5), beta_start=10, beta_end=150)
                    _update_best(a_opt2, sc2)
                except Exception:
                    pass

    # Phase 3: Large-scale optimization
    print(f"\nPhase 3: Large-scale optimization (best: {best_sc:.8f})")
    large_sizes = [4096, 8192]
    for size in large_sizes:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.4, 600.0)
        print(f"  Size {size}, time budget {t_alloc:.1f}s")

        candidates = []
        if best_a is not None:
            candidates.append(_resample(best_a, size))
        if parent is not None:
            candidates.append(_resample(parent, size))
        candidates.append(_make_init(size, 0, _GLOBAL_RNG))
        candidates.append(_make_init(size, 4, _GLOBAL_RNG))

        n_cand = len(candidates)
        t_each = t_alloc / n_cand
        for i, a_init in enumerate(candidates):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_e = min(t_each, rem - 15.0)
            if t_e < 10:
                continue
            try:
                a_opt, sc = _optimize_lbfgs(a_init, t_e, beta_start=20, beta_end=500, max_iter=200)

                    a_opt, sc = _optimize_lbfgs(best_a, t3, beta_start=200, beta_end=5000, max_iter=300)
                    _update_best(a_opt, sc)
                except Exception:

        rem = deadline - time.monotonic()
        if rem > 30:
