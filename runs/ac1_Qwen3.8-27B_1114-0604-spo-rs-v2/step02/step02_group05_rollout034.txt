```python
import cvxpy as cp

def solve_qp(p, peaks_indices, target, n_constraints=5):
    n = len(p)
    p_arr = np.array(p)
    c = np.convolve(p_arr, p_arr)
    
    delta = cp.Variable(n)
    constraints = [delta @ ones(n) == 0]  # maintain sum
    constraints += [p_arr + delta >= 0]  # non-negativity
    
    for k_star in peaks_indices[:n_constraints]:
        # Linearized constraint: sum_j delta[j] * p[k_star - j] <= (target - c[k_star]) / 2
        # where the sum is over valid j (0 <= j < n and 0 <= k_star - j < n)
        coeff = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                coeff[j] = p_arr[idx]
        constraints.append(coeff @ delta <= (target - c[k_star]) / 2)
    
    # Objective: minimize ||delta/p||^2 (relative change)
    obj = cp.Minimize(cp.sum_squares(delta / (p_arr + 1e-10)))
    
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=ECOS)
    
    return p_arr + delta.value
```

"""
Multi-strategy optimization for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2
Uses: softmax-parameterized gradient descent with adaptive temperature,
frequency-domain convolution flattening, LP-guided steps, block refinement,
and multi-restart diversification. Exploits FFT structure for O(n log n) 
gradient computation.
"""

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _softmax_weights(d, beta):
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-8:
        return np.ones(n, dtype=float) / n
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=float) / n
    return e / s


def _gradient_info(p, beta, L):
    """Compute (score, gradient h of length n, max_h) for smoothed max."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n + 10)
    c = _conv_full(p, p)
    d = n * c
    exact_score = float(2.0 * d.max())
    w = _softmax_weights(d, beta)
    # h[j] = 2 * sum_m w[m] * p[m-j] for valid (equiv: cross-correlation)
    # conv(w, p)[j] = sum_m w[m] * p[j-m] (not quite what we want)
    # We want: h[j] = 2 * sum_k w[k] * p[k-j] for valid k,j
    # = 2 * (conv(w, p_flip))[j] where p_flip = p[::-1]
    # Actually: h[j] = 2 * sum_{k: 0<=k<n, 0<=k-j<n} w[k] * p[k-j]
    # Let i = k-j: k = j+i, range: 0<=j+i<n, 0<=i<n => i in [max(0,-j), min(n-1-j, n-1)]
    # Simpler: use the fact that h = 2 * irfft(rfft(w,L) * conj(rfft(p,L)))[:n]
    # Wait: irfft(rfft(w)*conj(rfft(p))) gives cross-correlation of w and p
    # xcorr[m] = sum_k w[k+m] * conj(p[k]) -- not quite
    # Let me use: h[j] = 2 * sum_{i} w[j+i] * p[i] (where valid: 0<=j+i<2n-1, 0<=i<n)
    # This is the cross-correlation (w ⊛ p)[j]
    # Using FFT: if L >= 3n-2, then irfft(rfft(w, L) * conj(rfft(p, L)))[j] = sum_i w[j+i] * p[i]
    # (circular cross-correlation, valid for j=0..n-1 when L >= 3n-2)
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 2.0 * R[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    max_h = float(h.max()) if h.size > 0 else 0.0
    return exact_score, h, max_h


def _freq_flatten(p, epsilon, L):
    """
    Frequency-domain: blend current convolution towards flat, take square root.
    epsilon in (0,1): larger = more flattening.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _conv_full(p, p, L)
    c_mean = np.mean(c)
    c_target = (1.0 - epsilon) * c + epsilon * c_mean
    c_target = np.maximum(c_target, 1e-15)
    C_freq = np.fft.rfft(c_target, L)
    # Take complex square root
    # Handle potential negative real parts (shouldn't happen for valid convolution)
    phase = np.angle(C_freq)
    mag = np.sqrt(np.abs(C_freq))
    A_new = mag * np.exp(0.5j * phase)
    # Choose the minimum-phase branch: ensure causality
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.35:
        # Try with flipped phase
        A_new2 = -A_new
        a_new2 = np.fft.irfft(A_new2, L)[:n]
        neg_frac2 = (a_new2 < 0).sum() / n
        if neg_frac2 < neg_frac:
            a_new = a_new2
            neg_frac = neg_frac2
    if neg_frac > 0.5:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _lp_direction(p, top_k=5):
    """
    Compute a descent direction using a small LP on the top-k peaks.
    Returns direction d (length n) such that p + alpha*d reduces the max.
    """
    n = len(p)
    c = _conv_full(p, p)
    # Get top-k peak indices
    peak_indices = np.argsort(c)[::-1][:top_k]
    # Build constraint matrix: A[j, k] = 2 * p[peak_k - j] if valid
    # We minimize t s.t. c[peak_k] + sum_j A[j,k]*d[j] <= t
    # Plus: sum d[j] = 0, p[j] + d[j] >= 0, |d[j]| <= step_bound
    # Simplified: just use the subgradient for the worst peak
    # For the top peak, gradient of c[peak_0] w.r.t. p[j] = 2*p[peak_0 - j]
    k_star = peak_indices[0]
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * p[idx]
    # For secondary peaks, add weighted gradient
    for ki in range(1, min(3, len(peak_indices))):
        wk = c[peak_indices[ki]] / c[peak_indices[0]] * 0.3
        k2 = peak_indices[ki]
        for j in range(n):
            idx = k2 - j
            if 0 <= idx < n:
                grad[j] += wk * 2.0 * p[idx]
    return -grad  # descent direction


def _block_refine(p, rng, time_budget):
    """Block-wise multiplicative refinement."""
    p = _sanitize(p)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 0.9, 1.1, 1.4, 1.8, 2.5, 4.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.25:
            bs = int(rng.integers(1, max(2, n // 512)))
        elif r < 0.6:
            bs = int(rng.integers(max(2, n // 128), max(3, n // 16)))
        else:
            bs = int(rng.integers(max(4, n // 64), max(8, n // 8)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-12:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-11:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                p = best.copy()
                cur_score = best_score
                stall = 0
            elif stall > 30:
                # Random perturbation to escape
                p_try = p * np.exp(rng.normal(0, 0.05, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc = _score(p_try)
                    if sc < best_score * 1.1:
                        p = p_try
                        cur_score = sc
                        stall = 10

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 12):
        amp = rng.uniform(-0.7, 0.7) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_block_random(n, rng):
    K = int(rng.integers(4, 25))
    p = np.zeros(n, dtype=float)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1.2, K)
    for i in range(K):
        s = int(round(blocks[i] * n))
        e = max(s + 1, int(round(blocks[i + 1] * n)))
        e = min(n, e)
        p[s:e] = heights[i]
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_m_sequence_like(n, rng):
    """Initialize based on a pseudo-random pattern with binary structure."""
    # Create a binary sequence with good aperiodic autocorrelation properties
    K = int(rng.integers(8, 32))
    p = np.zeros(n)
    # Place K "bursts" of random width
    for _ in range(K):
        start = int(rng.integers(0, n))
        width = int(rng.integers(max(1, n // (4 * K)), max(2, n // (2 * K))))
        end = min(n, start + width)
        p[start:end] += rng.uniform(0.5, 2.0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == 1:
        return np.ones(n_new, dtype=float)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    result = np.interp(x_new, x_old, p)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

        if n_up != n_main and n_up > n_main:
            t_alloc = min(150.0, t_rem * 0.5)
            t_alloc = max(30.0, t_alloc)
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                p, sc = _optimize_main(p_up, rng, t_alloc, beta_start=50.0, beta_end=4000.0, verbose=True)
                _try_update(p, sc)

    # Run 4: Block refinement
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.6)
        t_alloc = max(20.0, t_alloc)
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    # Run 5: Another gradient pass with different params
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(150.0, t_rem * 0.7)
        t_alloc = max(20.0, t_alloc)
        p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=100.0, beta_end=6000.0, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.75 * budget

    n_best = len(best_overall)
    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        modes = [
            _init_two_level,
            _init_gaussian_mixture,
            _init_cosine,
            _init_block_random,
            _init_m_sequence_like,
        ]
        budgets_per_mode = max(20.0, (t_rem - 10) / len(modes))

        for i, mode in enumerate(modes):
            if time.monotonic() > t3_end - 5:
                break
            sp = mode(n_best, rng)
            # Mix with current best
            mix = (1 - 0.3) * best_overall + 0.3 * sp
            mix = _sanitize(mix)
            if mix is None:
                mix = sp
            t_alloc = min(budgets_per_mode, 80.0)
            p, sc = _optimize_main(mix, rng, t_alloc, beta_start=50.0, beta_end=3000.0, verbose=False)
            _try_update(p, sc)

    # Also try a completely different length
    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        n_alt = 4096 if n_best != 4096 else 3072
        sp = _resample(best_overall, n_alt, rng)
        if sp is None:
            sp = _init_two_level(n_alt, rng)
        t_alloc = min(100.0, t_rem * 0.8)
        p, sc = _optimize_main(sp, rng, t_alloc, beta_start=50.0, beta_end=4000.0, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # High-beta gradient for fine tuning
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 200.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=500.0, beta_end=15000.0, verbose=True)
            _try_update(p, sc)

        # Block refine (final)
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.25 * t_rem, 120.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # One more gradient pass with moderate beta
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 150.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=200.0, beta_end=8000.0, verbose=False)
            _try_update(p, sc)

        # Final tiny perturbations
        t_rem = deadline - time.monotonic()
        if t_rem > 10 and best_overall is not None:
            n = len(best_overall)
            p = best_overall.copy()
            cur_sc = best_score_overall
            while time.monotonic() < deadline - 3:
                # Try very small multiplicative perturbations
                for trial in range(10):
                    delta = rng.normal(0, 0.005, n)
                    p_try = p * np.exp(delta)
                    p_try = _sanitize(p_try)
                    if p_try is None:
                        continue
                    sc_try = _score(p_try)
                    if sc_try < cur_sc - 1e-14:
                        p = p_try
                        cur_sc = sc_try
                        break
            if cur_sc < best_score_overall - 1e-14:
                _try_update(p, cur_sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
