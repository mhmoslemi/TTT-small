def _next_pow2(m):
    n = 1

    return n

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

"""
Improved low-peak step-function optimization using multi-phase adaptive mirror descent.
Key improvements over previous approach:
1. Warm start from known good solution (height_sequence_1)
2. Gradual K-sweep (large K for broad optimization, small K for fine-tuning)
3. Adaptive learning rate with geometric decay and periodic boosts
4. Iterated local search with diverse perturbation strategies
5. Coarse-to-fine multi-scale initialization
6. Extended patience and slower annealing for deeper optimization
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    nlen = a.size
    if nlen == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * nlen - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * nlen - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    nlen = a.size
    N = _next_pow2(2 * nlen)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:nlen]


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_gaussian(n, sigma_frac=0.3):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _normalize_nonneg(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _optimize_phase(a, deadline, mode, K, T, lr_start, lr_end, decay, patience, eval_int):
    """Core optimization loop with geometric LR decay."""
    a = _normalize_nonneg(np.asarray(a, dtype=np.float64).copy())
    nlen = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    rng = np.random.RandomState(777)
    lr = lr_start
    max_lr = lr_start * 5.0

    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Compute weights
        if mode == 'soft':
            c_min = c.min()
            c_max = c.max()
            rng_c = c_max - c_min
            if rng_c < 1e-15:
                w = np.ones(c.size) / c.size
            else:
                c_norm = (c - c_min) / rng_c - 1.0
                exp_c = np.exp(T * c_norm)
                s_exp = exp_c.sum()
                if s_exp < 1e-300:
                    w = np.zeros(c.size)
                    w[np.argmax(c)] = 1.0
                else:
                    w = exp_c / s_exp
        elif mode == 'hard':
            k = min(K, c.size)
            top_idx = np.argpartition(c, -k)[-k:]
            top_vals = c[top_idx]
            T_local = 5.0
            weights = np.exp(T_local * (top_vals - top_vals.max()))
            weights /= weights.sum()
            w = np.zeros(c.size, dtype=np.float64)
            for idx_i in range(k):
                w[top_idx[idx_i]] = weights[idx_i]
        else:
            # 'medium' mode
            k = min(K, c.size)
            top_idx = np.argpartition(c, -k)[-k:]
            top_vals = c[top_idx]
            T_local = 1.5
            weights = np.exp(T_local * (top_vals - top_vals.max()))
            weights /= weights.sum()
            w = np.zeros(c.size, dtype=np.float64)
            for idx_i in range(k):
                w[top_idx[idx_i]] = weights[idx_i]

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -6.0, 6.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        lr *= decay

        if it % eval_int == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                strat = rng.randint(4)
                if strat == 0:
                    noise = rng.standard_normal(nlen) * 0.015
                    a = a + noise * a.mean()
                elif strat == 1:
                    start = rng.randint(0, max(1, nlen // 3))
                    ln = rng.randint(nlen // 8, nlen // 3)
                    region = slice(start, min(nlen, start + ln))
                    a[region] *= np.exp(rng.standard_normal(ln) * 0.08)
                elif strat == 2:
                    a = a * np.exp(rng.standard_normal(nlen) * 0.03)
                else:
                    # Frequency domain perturbation
                    fa = np.fft.rfft(a)
                    fa *= np.exp(rng.standard_normal(fa.size) * 0.02)
                    a = np.fft.irfft(fa, nlen)
                
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(1500, int(patience * 1.15))
                lr = max(lr_end, lr_start * rng.uniform(0.5, 1.0))

    return best_a, best_score


def _iterated_local_search(best_a, best_score, deadline, max_rounds=5):
    """Perturb and re-optimize multiple times."""
    nlen = best_a.size
    rng = np.random.RandomState(31337)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        round_time = max(30, (deadline - time.monotonic()) / (max_rounds - rnd))
        round_deadline = time.monotonic() + round_time
        
        # Perturb
        a_pert = best_a.copy()
        pert_type = rng.randint(3)
        if pert_type == 0:
            a_pert += rng.standard_normal(nlen) * best_a.mean() * 0.05
        elif pert_type == 1:
            # Swap regions
            s1 = rng.randint(0, nlen // 2)
            s2 = rng.randint(nlen // 2, nlen - 10)
            ln = min(nlen // 10, nlen - s2)
            tmp = a_pert[s1:s1+ln].copy()
            a_pert[s1:s1+ln] = a_pert[s2:s2+ln]
            a_pert[s2:s2+ln] = tmp
        else:
            # Local scaling
            center = rng.randint(0, nlen)
            width = rng.randint(nlen // 8, nlen // 4)
            for j in range(nlen):
                dist = min(abs(j - center), nlen - abs(j - center))
                if dist < width:
                    a_pert[j] *= rng.uniform(0.9, 1.1)
        
        a_pert = _normalize_nonneg(a_pert)
        sc_pert = _score(a_pert)
        
        # Short optimization
        a_opt, sc_opt = _optimize_phase(
            a_pert, round_deadline, mode='hard', K=12, T=5.0,
            lr_start=0.3 * nlen, lr_end=0.02 * nlen,
            decay=0.9998, patience=2000, eval_int=3
        )
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"    ILS round {rnd+1}: improved to {best_score:.8f}", flush=True)
    
    return best_a, best_score
