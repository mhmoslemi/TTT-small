"""
Step function optimizer using adaptive mirror descent with population-based
diversification, multi-lag gradient, and analytical coordinate polish.
Key improvements: monotone/polynomial starts, adaptive learning rate with
backtracking, population evolution for escaping local minima.
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _compute_grad(f, c, k_lags):
    """Compute gradient contribution from specified lag positions."""
    n = len(f)
    grad = np.zeros(n)
    for k in k_lags:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # conv[k] = sum_{i=lo}^{hi-1} f[i] * f[k-i]
        # d(conv[k])/d(f[i]) = f[k-i] for i in [lo, hi)
        grad[lo:hi] += f[k - np.arange(lo, hi)]
    return grad


def _mirror_update(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on the simplex."""
    # f_new[i] ∝ f[i] * exp(-eta * grad[i])
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _score(f):
    """Compute the objective value."""
    n = len(f)
    c = _autocorr_fft(f)
    return 2.0 * n * c.max()

def _conv_fft(f):
    """Full convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _objective_value(f):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(f)

    return 2.0 * n * c.max() / (s * s)


def _weighted_gradient(f, c, top_k_indices, temperature=0.5):
    """
    Compute weighted gradient from top-k lag positions.
    Weight each lag by softmax of (c[k] - max_c) / temperature.
    """
    n = len(f)
    max_c = c[top_k_indices].max()
    if max_c < 1e-300:
        return np.zeros(n)
    
    # Softmax weighting
    weights = np.exp((c[top_k_indices] - max_c) / (temperature * max_c + 1e-300))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        grad[lo:hi] += weights[i] * f[k - np.arange(lo, hi)]
    
    return grad


def _mirror_step(f, grad, eta):

    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()

    return f_new / s

def _scipy_polish(f, deadline, rng, n):
    """
    Use scipy.optimize with subgradient to refine the solution.
    Minimize 2*n*max(conv) subject to sum(f)=1, f>=0.
    """
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    best_f = f.copy()
    best_val = _conv_fft(f).max()
    
    start_time = time.monotonic()
    
    # Use L-BFGS-B with subgradient
    # Parameterize as f = exp(x) / sum(exp(x)) to maintain positivity
    
    def _soft_obj(x):
        f_x = np.exp(x - x.max())
        f_x /= f_x.sum()
        c = _conv_fft(f_x)
        # Use soft-max for smoothness
        beta = 500.0 / best_val
        sm = np.log(np.sum(np.exp(beta * c))) / beta
        return 2.0 * n * sm

    def _soft_grad(x):
        f_x = np.exp(x - x.max())
        f_x /= f_x.sum()
        c = _conv_fft(f_x)
        beta = 500.0 / best_val
        # d(softmax)/d(f_i)
        p = np.exp(beta * c) / np.sum(np.exp(beta * c))
        # d(softmax)/d(c_k) = beta * p[k]
        # d(c_k)/d(f_i) = f[k-i] for valid i
        dc_df = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            dc_df[lo:hi] += beta * p[k] * f_x[k - np.arange(lo, hi)]
        
        # Account for normalization f = exp(x)/sum(exp(x))
        # d(f_i)/d(x_j) = f_i*(delta_ij - f_j)
        grad_f = 2.0 * n * dc_df
        # Chain rule: d(obj)/d(x_j) = sum_i grad_f[i] * f_i*(delta_ij - f_j)
        # = f_j * (sum_i grad_f[i]*f_i - sum_i grad_f[i]*f_i*f_j)... 
        # Simpler: grad_x = f * (grad_f - grad_f @ f)
        gf_dot_f = np.dot(grad_f, f_x)
        grad_x = f_x * (grad_f - gf_dot_f)
        return grad_x

    x0 = np.log(f + 1e-300)
    
    try:
        result = scipy_minimize(
            _soft_obj, x0, method='L-BFGS-B',
            jac=_soft_grad,
            options={'maxiter': 200, 'gtol': 1e-10, 'ftol': 1e-12}
        )
        if result.success or result.fun < _conv_fft(best_f).max():
            f_new = np.exp(result.x - result.x.max())
            f_new /= f_new.sum()
            c_new = _conv_fft(f_new)
            if c_new.max() < best_val:
                best_val = c_new.max()
                best_f = f_new.copy()
    except Exception as e:
        print(f"    scipy_polish error: {e}")
    
    return best_f, 2.0 * n * best_val

    for s in starts:
        s = np.clip(np.asarray(s, dtype=np.float64), 0.0, 1000.0)
        if s.sum() > 0.01:
            s = s / s.sum()
            valid_starts.append(s)
    
    return valid_starts


def propose_candidate(height_sequence_1=None, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(f,f))/(sum f)^2.
    Uses diverse initialization, adaptive mirror descent with weighted multi-lag
    gradient, SA acceptance, coordinate polish, and scipy refinement.
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + budget_s - 20.0

    # Get reference sequence
    if height_sequence_1 is not None and len(height_sequence_1) > 0:
        f_ref = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        f_ref = np.ones(6121)
    
    n = len(f_ref)
    s_ref = f_ref.sum()
    if s_ref < 0.01:
        f_ref = np.ones(n)
    else:
        f_ref = f_ref / f_ref.sum()

    print(f"Sequence length: {n}")
    c0 = _conv_fft(f_ref)
    init_score = 2.0 * n * c0.max()
    print(f"Initial score: {init_score:.8f}")

    # Phase 1: Generate and score starts
    t_start = time.monotonic()
    starts = _generate_starts(f_ref, n, rng, count=60)
    
    scored = []
    for s in starts:
        c = _conv_fft(s)
        scored.append((c.max(), s))
    scored.sort(key=lambda x: x[0])
    
    elapsed = time.monotonic() - t_start
    print(f"Generated {len(starts)} starts in {elapsed:.1f}s")
    print(f"Top 5 peaks: {[f'{p:.8f}' for p, _ in scored[:5]]}")
    print(f"Top 5 scores: {[f'{2*n*p:.6f}' for p, _ in scored[:5]]}")

    best_score = 2.0 * n * scored[0][0]
    best_f = scored[0][1].copy()
    print(f"Best initial score: {best_score:.8f}")

    # Phase 2: Optimize top starts
    # Allocate: 60% main optimization, 25% polish, 15% buffer
    opt_frac = 0.60
    polish_frac = 0.25
    buffer_frac = 0.15
    
    opt_budget = budget_s * opt_frac
    polish_budget = budget_s * polish_frac
    buffer_time = budget_s * buffer_frac
    
    num_main = 5
    time_per_main = opt_budget / num_main

    for idx in range(num_main):
        remaining = deadline - time.monotonic() - polish_budget - buffer_time
        if remaining < 30:
            break
        
        alloc = min(time_per_main, remaining)
        peak0, f0 = scored[idx]
        print(f"\n  Main opt {idx+1}/{num_main}: peak={peak0:.8f} (score={2*n*peak0:.6f}), time={alloc:.0f}s")
        
        t0 = time.monotonic()
        f_opt, score = _optimize_main(f0, t0 + alloc, rng, n)
        
        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"    IMPROVED! New best: {best_score:.8f}")
        else:
            print(f"    Best from this run: {score:.8f} (no improvement)")

    # Phase 3: Polish
    remaining = deadline - time.monotonic() - buffer_time
    if remaining > 60:
        print(f"\n  Polish phase ({remaining:.0f}s available)...")
        t0 = time.monotonic()
        pol_time = min(polish_budget, remaining)
        f_pol, pol_score = _polish(best_f, t0 + pol_time * 0.7, rng, n)
        if pol_score < best_score:
            best_score = pol_score
            best_f = f_pol.copy()
            print(f"    Polished to: {best_score:.8f}")
        else:
            print(f"    Polish best: {pol_score:.8f}")
        
        # scipy polish
        remaining2 = deadline - time.monotonic() - buffer_time
        if remaining2 > 30:
            print(f"    scipy polish ({min(remaining2, 120):.0f}s)...")
            t0 = time.monotonic()
            f_sp, sp_score = _scipy_polish(best_f, t0 + min(remaining2, 120), rng, n)
            if sp_score < best_score:
                best_score = sp_score
                best_f = f_sp.copy()
                print(f"    scipy improved to: {best_score:.8f}")

    # Phase 4: Final optimization round from best
    remaining = deadline - time.monotonic() - 10
    if remaining > 60:
        print(f"\n  Final optimization ({remaining:.0f}s)...")
        t0 = time.monotonic()
        f_fin, fin_score = _optimize_main(best_f, t0 + remaining * 0.8, rng, n)
        if fin_score < best_score:
            best_score = fin_score
            best_f = f_fin.copy()
            print(f"    Final improved to: {best_score:.8f}")
        
        # Final polish
        remaining2 = deadline - time.monotonic() - 5
        if remaining2 > 20:
            t0 = time.monotonic()
            f_fp, fp_score = _polish(best_f, t0 + min(remaining2, 60), rng, n)
            if fp_score < best_score:
                best_score = fp_score
                best_f = f_fp.copy()
                print(f"    Final polish: {best_score:.8f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = np.ones(n) / n
        best_score = 2.0 * n
    
    # Verify with official evaluator
    result = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {result:.8f}")
    
    if not np.isfinite(result):
        print("Falling back to uniform!")
        best_f = np.ones(n)
        result = evaluate_sequence(best_f.tolist())
        print(f"Uniform score: {result:.8f}")
    
    if result < best_score:
        best_score = result

    print(f"=== FINAL SCORE: {best_score:.8f} ===")
    return [float(x) for x in best_f]
