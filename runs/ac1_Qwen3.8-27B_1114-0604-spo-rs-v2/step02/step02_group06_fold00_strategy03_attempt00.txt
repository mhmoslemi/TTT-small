def evaluate_sequence(sequence: list[float]) -> float:
    n = len(sequence)
    if n <= 0:
        return np.inf
    try:
        conv_result = np.convolve(sequence, sequence, mode='full')
        max_b = np.max(conv_result)
    except Exception as e:
        print(f"Evaluating convolution: {e}")
        return np.inf
    sum_a = np.sum(sequence)
    if abs(sum_a) < 1e-5:
        return np.inf
    return float(2.0 * n * float(max_b) / (sum_a ** 2))
