This is minimized by putting all mass on j where a[k*-j] is smallest (i.e., j = k* - argmin(a)).

            c = d

            if old_x < 1e-18:
                continue

            # Affected range in convolution: k in [i, i+n-1] capped to [0, 2n-2]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Precompute the "outside max" (peaks not affected by a[i])
            if k_lo > 0:
                max_before = float(np.max(c[:k_lo]))
            else:
                max_before = 0.0
            if k_hi < 2 * n - 2:
                max_after = float(np.max(c[k_hi + 1:]))
            else:
                max_after = 0.0
            max_outside = max(max_before, max_after)

            # Affected slice of c and corresponding a values
            affected_len = k_hi - k_lo + 1
            c_aff = c[k_lo:k_hi + 1]
            a_aff = a[k_lo - i: k_hi - i + 1]  # a[k-i] for k in range

            # The diagonal index (k = 2i) in the affected range
            diag_offset = 2 * i - k_lo  # position in affected slice where k=2i
            has_diag = (0 <= diag_offset < affected_len)

            def f_1d(x):
                """Score as function of a[i] = x."""
                new_S = S - old_x + x
                if new_S < 1e-10:
                    return 1e30
                # Compute new max of affected peaks
                if has_diag:
                    new_c_aff = c_aff + 2.0 * (x - old_x) * a_aff
                    new_c_aff[diag_offset] = c_aff[diag_offset] + x * x - old_x * old_x
                else:
                    new_c_aff = c_aff + 2.0 * (x - old_x) * a_aff
                max_inside = float(np.max(new_c_aff))
                new_max = max(max_outside, max_inside)
                if new_max < 0:
                    new_max = 0.0
                return 2.0 * n * new_max / (new_S * new_S)

            # Golden section search on [0, upper_bound]
            # Good upper bound: old_x * 5 (don't go too far)
            upper = max(old_x * 5.0, 1.0)
            golden = (np.sqrt(5.0) - 1.0) / 2.0

            c1 = upper * (1.0 - golden)
            c2 = upper * golden
            f1 = f_1d(c1)
            f2 = f_1d(c2)

            for _ in range(40):
                if upper - 0.0 < 1e-14:
                    break
                if f1 < f2:
                    upper = c2
                    c2 = c1
                    f2 = f1
                    c1 = upper * (1.0 - golden)
                    f1 = f_1d(c1)
                else:
                    lower_bound = c1
                    c1 = c2
                    f1 = f2
                    c2 = (lower_bound + upper) / 2.0
                    f2 = f_1d(c2)
                    # Actually golden section:
                    # a = c1_old, b = upper
                    # Let me redo this properly
                    break

            # Simpler: just use ternary search
            lo = 0.0
            hi = upper
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = f_1d(m1)
                f2 = f_1d(m2)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = (lo + hi) / 2.0
            f_opt = f_1d(x_opt)

            # Also check x=0 and x=old_x
            f_zero = f_1d(0.0)
            f_old = f_1d(old_x)

            # Pick the best
            if f_zero < f_opt and f_zero < f_old:
                x_pick = 0.0
                f_pick = f_zero
            elif f_old < f_opt:
                x_pick = old_x
                f_pick = f_old
            else:
                x_pick = x_opt
                f_pick = f_opt

            if x_pick != old_x and f_pick < current_score:
                # Update convolution incrementally
                if x_pick == 0.0:
                    # Remove contribution
                    delta = -2.0 * old_x * a_aff
                    if has_diag:
                        delta[diag_offset] = -old_x * old_x
                    c[k_lo:k_hi + 1] += delta
                else:
                    delta = 2.0 * (x_pick - old_x) * a_aff
                    if has_diag:
                        delta[diag_offset] = x_pick * x_pick - old_x * old_x
                    c[k_lo:k_hi + 1] += delta

                a[i] = x_pick
                S = S - old_x + x_pick
                current_score = f_pick

                if current_score < best_c_score:
                    best_c_score = current_score
                    best_a = a.copy()
                    best_S = S
                    last_improve_sweep = sweep

        # Restart if stuck
        if sweep - last_improve_sweep > 50 and sweep > 10:
            a = best_a.copy()
            S = best_S
            c = _fft_convolve(a)
            current_score = best_c_score
            # Add small noise
            noise = np.random.standard_normal(n) * 0.005 * a.mean()
            a = a + noise
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            c = _fft_convolve(a)
            current_score = 2.0 * n * float(np.max(c)) / (S * S)
            last_improve_sweep = sweep

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: score={current_score:.8f}, best={best_c_score:.8f}", flush=True)

    return best_a / best_a.sum(), best_c_score


def _grad_topK_optimize(a, deadline, K_start=1, K_end=30, T_base=100.0):
    """Top-K peak-targeting mirror descent for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    lr0 = 0.3 * n
    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(777)

    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / 100000.0)
        K = max(1, int(K_start + (K_end - K_start) * progress))
        T_eff = T_base * (1.0 + 3.0 * progress)
        lr = lr0 / (1.0 + 0.005 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Top-K with local softmax
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        max_val = top_vals.max()
        if max_val < 1e-15:
            break
        T_local = T_eff / max_val
        shifts = top_vals / max_val - 1.0
        weights = np.exp(T_local * shifts)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(weights)
            weights[np.argmax(top_vals)] = 1.0
        else:
            weights /= wsum

        w = np.zeros(c.size, dtype=np.float64)
        for idx in range(len(top_idx)):
            w[top_idx[idx]] = weights[idx]

        # Gradient via FFT
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[:N], N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        grad = 2.0 * xcorr[:n]

        # Mirror descent update
        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.005 * a.mean()
                a = a + noise
                a = np.clip(a, 0.0, 1000.0)
                a = a / a.sum()
                patience = max(1000, int(patience * 1.4))
                lr0 = max(0.05 * n, lr0 * 0.7)

    return best_a, best_score


def _scan_initializations(deadline):
    """Try various initializations and return the best few."""
    candidates = []
    rng = np.random.RandomState(12345)

    # Try to use height_sequence_1 if available
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0.0, 1000.0)
        if hs.size > 0 and hs.sum() > 0.01:
            sc = _score(hs)
            candidates.append((hs.copy(), sc, "height_sequence_1"))
    except (NameError, UnboundLocalError):
        pass

    # Sinusoidal family
    for n in [1024, 2048, 512]:
        if time.monotonic() > deadline:
            break
        for amp in [0.5, 0.6, 0.7, 0.75, 0.8, 0.85, 0.9, 1.0, 1.2]:
            for phase in [0.0, 0.25, 0.5, 0.75]:
                a = _init_sine(n, amp, phase)
                sc = _score(a)
                candidates.append((a, sc, f"sine(n={n},amp={amp},ph={phase})"))

        # Multi-mode
        for _ in range(8):
            amp1 = rng.uniform(0.4, 1.2)
            amp2 = rng.uniform(0.05, 0.4)
            ph1 = rng.uniform(0.0, 1.0)
            ph2 = rng.uniform(0.0, 1.0)
            a = _init_multi_sine(n, [amp1, amp2], [ph1, ph2])
            sc = _score(a)
            candidates.append((a, sc, f"multi(n={n})"))

    # Sort by score
    candidates.sort(key=lambda x: x[1])
    return candidates


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak nonnegative step function.
    Phase 1: Scan initializations (60s)
    Phase 2: Coordinate descent on top candidates (majority of time)
    Phase 3: Top-K gradient refinement (polish)
    Phase 4: Validate and return best
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 15.0
    np.random.seed(seed)

    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Initializations
    d1 = t0 + min(60.0, budget_s * 0.05)
    print("[Phase 1] Scanning initializations...", flush=True)
    inits = _scan_initializations(d1)
    if not inits:
        inits = [(np.ones(1024, dtype=np.float64) / 1024.0, 2.0, "uniform")]
    print(f"  Best init: {inits[0][2]}, score={inits[0][1]:.8f}", flush=True)
    print(f"  2nd best: {inits[1][2]}, score={inits[1][1]:.8f}", flush=True)

    all_results = []

    # Phase 2: Coordinate descent on top 3 initializations
    d2 = hard_deadline - 300.0
    n_candidates_cd = min(3, len(inits))
    time_per_cd = (d2 - time.monotonic()) / max(1, n_candidates_cd)

    for ci in range(n_candidates_cd):
        if time.monotonic() > d2:
            break
        init_a, init_sc, init_name = inits[ci]
        cd_deadline = time.monotonic() + time_per_cd
        print(f"[Phase 2.{ci}] CD on {init_name} (budget {time_per_cd:.0f}s)...", flush=True)
        cd_a, cd_sc = _coord_descent(init_a, cd_deadline, recompute_interval=100)
        print(f"  CD result: {cd_sc:.8f}", flush=True)
        all_results.append((cd_a, cd_sc, f"CD({init_name})"))

    # Phase 3: Top-K gradient refinement on best CD results
    d3 = hard_deadline - 60.0
    remaining3 = d3 - time.monotonic()
    if remaining3 > 30:
        # Sort results by score
        all_results.sort(key=lambda x: x[1])
        n_grads = min(2, len(all_results))
        time_per_grad = remaining3 / max(1, n_grads)

        for gi in range(n_grads):
            if time.monotonic() > d3:
                break
            grad_deadline = time.monotonic() + time_per_grad
            a_start = all_results[gi][0]
            print(f"[Phase 3.{gi}] Top-K on {all_results[gi][2]} ({time_per_grad:.0f}s)...", flush=True)
            g_a, g_sc = _grad_topK_optimize(a_start, grad_deadline, K_start=1, K_end=40, T_base=200.0)
            print(f"  Top-K result: {g_sc:.8f}", flush=True)
            all_results.append((g_a, g_sc, f"TopK({all_results[gi][2]})"))

    # Also run CD on the best top-K result if time permits
    d3b = hard_deadline - 30.0
    if time.monotonic() < d3b - 60:
        all_results.sort(key=lambda x: x[1])
        cd_deadline2 = d3b
        print(f"[Phase 3b] CD on best result ({d3b - time.monotonic():.0f}s)...", flush=True)
        cd_a2, cd_sc2 = _coord_descent(all_results[0][0], cd_deadline2, recompute_interval=50)
        print(f"  CD2 result: {cd_sc2:.8f}", flush=True)
        all_results.append((cd_a2, cd_sc2, "CD2"))

    # Phase 4: Validate
    print("[Phase 4] Validating with evaluate_sequence...", flush=True)
    all_results.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    for name_arr in all_results[:6]:
        if time.monotonic() > hard_deadline:
            break
        arr, sc, name = name_arr
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)
            if best_actual < 1.48:
                break

    if best_out is None:
        arr = all_results[0][0]
        best_out = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = all_results[0][1]
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
