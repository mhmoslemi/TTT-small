```
def fast_conv(y):
    n=len(y)
    m=2*n-1
    M=1 << (m-1).bit_length()
    Y=np.fft.rfft(y,M)
    return np.fft.irfft(Y*Y,M)[:m]
```

```
def conv_grad(y,p): # p len 2n-1
    n=len(y)
    m2=3*n-2
    M=1 << (m2-1).bit_length()
    yr=np.fft.rfft(y[::-1],M)
    P=np.fft.rfft(p,M)
    return np.fft.irfft(yr*P,M)[n-1:2*n-1]
def lse_grad_y(y,beta):
    c=fast_conv(y)
    cmax=c.max()
    z=beta*(c-cmax)
    pz=np.exp(z)
    s=pz.sum()
    p=pz/s
    g=2*conv_grad(y,p)
    return cmax + log(s)/beta, g
```

```
def lbfgs_smooth(y0, beta, deadline, maxiter=3000):
    n=len(y0)
    y = normalize_y(y0,n).copy()
    mu = 20.0
    # precompute ones
    ones = np.ones(n)
    def fun_grad(y):
        y = np.clip(y,0,None)
        c = fast_conv(y)
        cmax = c.max()
        z = beta*(c-cmax)
        ez = np.exp(z)
        s = ez.sum()
        L = cmax + np.log(s)/beta
        p = ez/s
        g = 2.0 * conv_grad(y,p)
        S = y.sum() - n
        f = L + 0.5*mu*S*S
        gg = g + mu*S*ones
        return f, gg
    res = minimize(fun_grad, y, jac=True, method='L-BFGS-B', bounds=[(0.0, 1000.0)]*n, options={'maxiter':maxiter,'ftol':1e-9,'gtol':1e-6,'maxls':40})
    y = normalize_y(res.x,n)
    sc = compute_score_y(y)
    return sc,y
```

```python
"""
Search for a non-negative sequence minimizing
    2 * n * max(conv(a, a)) / (sum(a)^2).

The implementation works on the scale-y invariant representation y = n*a
(so sum(y) is normally n) and uses several complementary routines:

  * FFT-constrained Adam on an adaptively shifted LSE / violation objective.
  * Active-set LP directions inspired by the SOTA update, but with a
    controllable subset of lags, solved with HiGHS when available.
  * L-BFGS-B polish on an LSE-smoothed objective with a soft sum penalty.
  * Short local-search perturbations and final smoothing candidates.

The goal is to improve on the supplied parent construction while staying
safely inside the time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog


# ---------------------------------------------------------------------------
# Basic numerical helpers
# ---------------------------------------------------------------------------

def interpolate_array(a, m):
    """Linearly resample a 1-D array to length m."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.ones(m, dtype=np.float64)
    if n == m:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    return np.interp(x_new, x_old, a).astype(np.float64)


def normalize_y(y, n=None):
    """
    Clip to non-negative and scale to mean 1 (sum = n).
    If the length differs from n, interpolate first.
    """
    y = np.asarray(y, dtype=np.float64).ravel()
    if n is None:
        n = y.size
    else:
        if y.size != n:
            y = interpolate_array(y, n)
    y = np.clip(y, 0.0, 1.0e6)
    s = float(y.sum())
    if s <= 1.0e-12:
        return np.ones(n, dtype=np.float64)
    y = y * (float(n) / s)
    # If anything is absurdly large, clip and renormalize once. The cap
    # matches evaluate_sequence's 1000.0 clamp.
    if y.size and y.max() > 1000.0:
        y = np.clip(y, 0.0, 1000.0)
        s2 = float(y.sum())
        if s2 > 1.0e-12:
            y = y * (float(n) / s2)
            y = np.clip(y, 0.0, 1000.0)
    return y


def fast_conv(a):
    """Full linear self-convolution by real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    if L <= 1:
        return a * a
    M = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:L]
    return c


def conv_grad(y, p):
    """
    Gradient contribution from weights p on convolution coefficients:

        h_m = sum_r y_r p_{r+m},   m = 0..n-1.

    This is the vector satisfying d <p, conv(y,y)> / dy = 2*h.
    """
    y = np.asarray(y, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = y.size
    L = 3 * n - 2
    M = 1 << (L - 1).bit_length()
    yr = np.fft.rfft(y[::-1], M)
    Pr = np.fft.rfft(p, M)
    res = np.fft.irfft(yr * Pr, M)
    return res[n - 1:2 * n - 1]


def score_y(y):
    """Exact score for y, using the generic scale-invariant formula."""
    y = np.asarray(y, dtype=np.float64).ravel()
    n = y.size
    S = float(y.sum())
    if n == 0 or S < 1.0e-12:
        return np.inf
    c = fast_conv(y)
    return float(2.0 * n * np.max(c) / (S * S))


def score_from_conv(c, n):
    """Score assuming sum(y) = n. Used in inner loops for speed."""
    return float(2.0 * np.max(c) / n)


def load_parent_sequence():
    """Load and normalize the global parent sequence if available."""
    p = globals().get("height_sequence_1", None)
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 32:
        return None
    arr = np.clip(arr, 0.0, 1.0e6)
    if arr.sum() < 1.0e-12:
        return None
    return normalize_y(arr, arr.size)


# ---------------------------------------------------------------------------
# Weight construction and Adam-style descent
# ---------------------------------------------------------------------------

def make_weights(c, T, beta, mix=0.3):
    """
    Build a probability vector p over convolution lags.

    p is a mixture of:
      - an LSE/softmax peak weight around the current maximum
      - a violation weight proportional to squared excess above T
    """
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.ones_like(c, dtype=np.float64) / c.size

    if beta <= 0.0 or not np.isfinite(beta):
        p_s = np.zeros_like(c, dtype=np.float64)
        p_s[int(np.argmax(c))] = 1.0
    else:
        z = beta * (c - cmax)
        z = np.clip(z, -745.0, 0.0)
        ez = np.exp(z)
        s = float(ez.sum())
        if s <= 1.0e-300 or not np.isfinite(s):
            p_s = np.zeros_like(c, dtype=np.float64)
            p_s[int(np.argmax(c))] = 1.0
        else:
            p_s = ez / s

    if T is not None and np.isfinite(T) and T < cmax:
        excess = c - T
        pos = excess > 0.0
        if np.any(pos):
            ev = excess[pos]
            ev = ev * ev
            s2 = float(ev.sum())
            if s2 > 1.0e-300 and np.isfinite(s2):
                p_v = np.zeros_like(c, dtype=np.float64)
                p_v[pos] = ev / s2
                p = (1.0 - mix) * p_v + mix * p_s
                ps = float(p.sum())
                if ps > 1.0e-300 and np.isfinite(ps):
                    return p / ps
    return p_s


def exact_subgradient_from_max(c, y):
    """Subgradient of the exact max coefficient at the current argmax."""
    n = y.size
    k = int(np.argmax(c))
    g = np.zeros(n, dtype=np.float64)
    start = max(0, k - n + 1)
    end = min(k, n - 1)
    if end >= start:
        js = np.arange(start, end + 1)
        g[js] = 2.0 * y[k - js]
    return g


def run_adam(y0, deadline, rng, target_score=1.45, base_lr=0.05, mix=0.3):
    """
    Projectile/normalizing Adam on the y-scale variable.

    The objective is not explicitly returned every iteration; instead we track
    the exact max-conv score and adapt a threshold T for peak violation weights.
    """
    y = normalize_y(y0, None).copy()
    n = y.size
    c = fast_conv(y)
    M = float(np.max(c))
    best = score_from_conv(c, n)
    best_y = y.copy()

    target_M = 0.5 * target_score * n
    if not np.isfinite(target_M) or target_M <= 0.0:
        target_M = 0.0

    M0 = M
    T = 0.98 * M0

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1.0e-8

    start = time.monotonic()
    total = max(5.0, float(deadline - start))
    betas = (1.0, 5.0, 20.0, 100.0, 300.0)
    fracs = np.cumsum((0.15, 0.20, 0.25, 0.25, 0.15))

    last_best = start
    restarts = 0
    max_restarts = 6

    for idx, beta in enumerate(betas):
        stage_end = start + fracs[idx] * total
        lr = base_lr * (1.0 - 0.5 * (idx / max(1, len(betas) - 1)))

        while time.monotonic() < min(stage_end, deadline):
            c = fast_conv(y)
            M = float(np.max(c))
            sc = score_from_conv(c, n)

            if sc < best - 1.0e-12:
                best = sc
                best_y = y.copy()
                last_best = time.monotonic()
                T = max(target_M, M * 0.98)

            now = time.monotonic()

            # If we have stalled badly, perturb and restart Adam state.
            if now - last_best > 45.0 and t > 2000 and restarts < max_restarts:
                noise = 0.03 * rng.standard_normal(n)
                y = np.clip(y + noise, 0.0, None)
                y = normalize_y(y, n)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                c = fast_conv(y)
                M = float(np.max(c))
                T = max(target_M, M * 0.98)
                last_best = now
                restarts += 1
                continue

            # Very slow, safe downward ratchet on the violation threshold.
            if target_M > 0.0 and T > target_M and now - last_best > 30.0:
                T = max(target_M, T - 0.005 * M0)

            p = make_weights(c, T, beta, mix=mix)
            g = 2.0 * conv_grad(y, p)

            if not np.all(np.isfinite(g)):
                g = exact_subgradient_from_max(c, y)
                if not np.all(np.isfinite(g)):
                    break

            # Tangent to the sum constraint, then projection after update.
            g = g - float(np.mean(g))

            t += 1
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * g * g
            mhat = m / (1.0 - beta1 ** t)
            vhat = v / (1.0 - beta2 ** t)

            y_new = y - lr * mhat / (np.sqrt(vhat) + eps)
            y_new = np.clip(y_new, 0.0, None)
            s = float(y_new.sum())
            if s < 1.0e-12:
                y_new = best_y.copy()
            else:
                y_new = y_new * (float(n) / s)
            y = y_new

    return best, best_y


# ---------------------------------------------------------------------------
# Active-set LP direction
# ---------------------------------------------------------------------------

def lp_K_for_length(n):
    """Choose how many active lags to use in the LP step."""
    if n <= 3000:
        return 48
    if n <= 8000:
        return 40
    if n <= 12000:
        return 32
    return 24


def build_active_matrix(f, ks, n):
    """
    Build a dense matrix A such that (A @ b)_r = (f * b)_{ks[r]}.
    Uses a tiny positive floor in f to guarantee finite bounds.
    """
    K = ks.size
    A = np.zeros((K, n), dtype=np.float64)
    fa = np.maximum(f, 1.0e-6)
    for r, k in enumerate(ks):
        k = int(k)
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        if end >= start:
            seg = fa[k - end:k - start + 1]
            A[r, start:end + 1] = seg[::-1]
    return A


def lp_search(y0, deadline, rng, ub_extra=10.0):
    """
    Run a sequence of active-set LP steps inspired by:
        max sum(b) s.t. b >= 0, (f * b)_{selected lags} <= max(f*f).
    Only accepted if the full objective improves.
    """
    f = normalize_y(y0, None).copy()
    n = f.size
    K = lp_K_for_length(n)

    c = fast_conv(f)
    M = float(np.max(c))
    best = score_from_conv(c, n)
    best_f = f.copy()

    c_obj = -np.ones(n, dtype=np.float64)
    fmax = float(np.max(f)) if f.size else 1.0
    ub = max(ub_extra, 2.0 * fmax + 1.0)
    bounds = [(0.0, ub)] * n

    attempts = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        c = fast_conv(f)
        M = float(np.max(c))
        sc_f = score_from_conv(c, n)

        if sc_f < best - 1.0e-12:
            best = sc_f
            best_f = f.copy()
            last_improve = time.monotonic()
        elif sc_f > best * 1.05:
            f = best_f.copy()
            c = fast_conv(f)
            M = float(np.max(c))

        if M <= 0.0 or not np.isfinite(M):
            break

        # Active set: always include some evenly spaced coverage lags plus
        # currently important/large lags.
        top = np.argsort(c)[::-1][:K]
        cover = np.linspace(0, 2 * n - 2, K, dtype=float).astype(int)

        if attempts % 2 == 0:
            ks = np.unique(np.concatenate((top, cover)))
        else:
            level = np.where(c > 0.95 * M)[0]
            if level.size > K:
                level = rng.choice(level, size=K, replace=False)
            ks = np.unique(np.concatenate((level, cover)))

        if ks.size > K:
            ks = ks[:K]

        A = build_active_matrix(f, ks, n)
        b_ub = np.full(K, M, dtype=np.float64)

        now = time.monotonic()
        rem = deadline - now
        if rem < 5.0:
            break
        tl = max(5.0, min(60.0, rem - 5.0))

        res = None
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": tl},
            )
        except Exception:
            try:
                res = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                res = None

        if res is not None:
            x = getattr(res, "x", None)
            if x is not None:
                b = np.asarray(x, dtype=np.float64).ravel()
                if b.size == n and np.all(np.isfinite(b)):
                    b = np.clip(b, 0.0, ub)
                    bs = float(b.sum())
                    if bs > 1.0e-12:
                        b = b * (float(n) / bs)
                        for tv in (0.2, 0.5, 0.8, 1.0):
                            mixed = f + tv * (b - f)
                            mixed = np.clip(mixed, 0.0, None)
                            s = float(mixed.sum())
                            if s < 1.0e-12:
                                continue
                            mixed = mixed * (float(n) / s)
                            sc = score_y(mixed)
                            if sc < best - 1.0e-12:
                                best = sc
                                best_f = mixed.copy()
                                f = mixed.copy()
                                c = fast_conv(f)
                                M = float(np.max(c))
                                last_improve = time.monotonic()

        attempts += 1

        # Occasionally perturb the working point (not the certified best).
        if attempts % 3 == 0:
            f = f + 0.01 * rng.standard_normal(n)
            f = normalize_y(f, n)

        # Safety: if no improvement for a long time and we are far from
        # deadline, reset to best to avoid wandering.
        if time.monotonic() - last_improve > 90.0 and attempts % 2 == 0:
            f = best_f.copy()

    return best, best_f


# ---------------------------------------------------------------------------
# L-BFGS-B polish
# ---------------------------------------------------------------------------

def lbfgs_fun(y, beta, n, mu):
    """
    LSE-smoothed max-conv objective on y plus a soft quadratic penalty
    enforcing sum(y) ~ n.
    """
    y = np.asarray(y, dtype=np.float64)
    c = fast_conv(y)
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 1.0e-300 or not np.isfinite(s):
        L = cmax
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        L = cmax + np.log(s) / beta
        p = ez / s

    g = 2.0 * conv_grad(y, p)
    S = float(np.sum(y)) - float(n)
    fval = L + 0.5 * mu * S * S
    gg = g + mu * S * np.ones(n, dtype=np.float64)
    return fval, gg


def lbfgs_smooth(y0, beta, deadline):
    """Run a bounded L-BFGS-B polish for an LSE-smoothed objective."""
    y0 = normalize_y(y0, None)
    n = y0.size
    if n == 0:
        return np.inf, y0

    now = time.monotonic()
    rem = float(deadline - now)
    if rem < 10.0:
        return score_y(y0), y0

    # Rough per-iteration time budget, conservative.
    if n <= 4096:
        maxiter = int(rem / 0.035)
    elif n <= 8192:
        maxiter = int(rem / 0.06)
    else:
        maxiter = int(rem / 0.10)
    maxiter = max(80, min(1000, maxiter))

    fmax = float(np.max(y0)) if y0.size else 1.0
    ub = max(100.0, 2.0 * fmax + 1.0)
    bounds = [(0.0, ub)] * n
    mu = 50.0

    try:
        res = minimize(
            lbfgs_fun,
            y0,
            args=(beta, n, mu),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1.0e-9,
                "gtol": 1.0e-5,
                "maxiter": maxiter,
                "maxls": 30,
            },
        )
        y = normalize_y(res.x, n)
    except Exception:
        y = y0

    return score_y(y), y


# ---------------------------------------------------------------------------
# Local search / initialization builders

def estimate_adam_time(n):
    """Rough per-invocation time for Adam runs by length."""
    if n <= 4096:
        return 90.0
    if n <= 8192:
        return 180.0
    if n <= 12288:
        return 260.0
    return 380.0


def build_secondary_inits(n, rng):
    """
    A small set of structured/non-random-looking initializations.
    All are normalized to mean 1.
    """
    inits = []
    x = np.linspace(0.0, 1.0, n)

    inits.append(np.ones(n, dtype=np.float64))

    # Decaying / increasing periodic shapes.
    inits.append(1.0 + np.cos(np.pi * x))          # high at left, low at right
    inits.append(1.0 - np.cos(np.pi * x))          # inverse
    inits.append(1.0 - x)                          # linear decay
    inits.append(x)                                # linear growth

    # Rational decay.
    inits.append(2.0 / (1.0 + 3.0 * x))
    inits.append(2.0 / (1.0 + 3.0 * (1.0 - x)))

    # Exponential decay/growth.
    a = 2.0
    base = a / max(1.0e-9, 1.0 - np.exp(-a))
    p = base * np.exp(-a * x)
    inits.append(p)
    inits.append(p[::-1])

    # Smooth random via low-pass Fourier.
    half = n // 2 + 1
    Z = rng.standard_normal(half).astype(np.float64)
    freqs = np.arange(half, dtype=np.float64)
    cutoff = max(2, n // 10)
    Z[freqs >= cutoff] = 0.0
    y = np.fft.irfft(Z, n)
    y = np.abs(y) + 0.5 + 0.2 * rng.random(n)
    inits.append(y)

    # Log-normal-ish random, normalized later.
    inits.append(np.exp(0.35 * rng.standard_normal(n)))

    out = [normalize_y(arr, n) for arr in inits]

    # Deduplicate very similar vectors by a cheap checksum.
    seen = set()
    unique = []
    for arr in out:
        key = (float(arr[0]), float(arr[-1]), float(np.sum(arr * np.linspace(0, 1, n))))
        if key not in seen:
            seen.add(key)
            unique.append(arr)
    return unique


def targeted_perturb_y(y, rng, strength=0.05):
    """
    Move a small amount of mass away from indices contributing most to the
    current peak convolution coefficient, and inject it into low-impact spots.
    """
    y = normalize_y(y, None)
    n = y.size
    c = fast_conv(y)
    k = int(np.argmax(c))

    start = max(0, k - n + 1)
    end = min(k, n - 1)
    contrib = np.zeros(n, dtype=np.float64)
    if end >= start:
        js = np.arange(start, end + 1)
        contrib[js] = y[js] * y[k - js]

    top_count = max(3, min(n, n // 100))
    top_idx = np.argsort(contrib)[::-1][:top_count]

    mask = np.ones(n, dtype=bool)
    mask[top_idx] = False
    bottom_candidates = np.where(mask)[0]
    if bottom_candidates.size == 0:
        return y
    # Prefer low contribution among the remaining indices.
    bottom_idx = bottom_candidates[np.argsort(contrib[bottom_candidates])][:top_count]

    y2 = y.copy()
    total = 0.0
    for j in top_idx:
        red = min(y2[j], y2[j] * strength)
        y2[j] -= red
        total += red

    if bottom_idx.size > 0 and total > 0.0:
        y2[bottom_idx] += total / float(bottom_idx.size)

    y2 = np.clip(y2, 0.0, None)
    s = float(y2.sum())
    if s < 1.0e-12:
        return y
    return y2 * (float(n) / s)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns a list of non-negative heights minimizing the evaluation function
    within the supplied time budget.
    """
    t_start = time.monotonic()
    total_budget = max(120.0, float(budget_s))
    final_deadline = t_start + total_budget - 35.0
    work = total_budget - 35.0

    rng = np.random.default_rng(seed)
    target_score = 1.45  # aggressive internal target; actual best may be higher

    parent = load_parent_sequence()

    if parent is not None:
        best_y = parent.copy()
        best_score = score_y(best_y)
    else:
        best_y = np.ones(8192, dtype=np.float64)
        best_score = 2.0

    print(f"[init] parent={'yes' if parent is not None else 'no'} "
          f"n={best_y.size} score={best_score:.8f}")

    # Candidate lengths to explore.
    cands = []
    if parent is not None:
        if 512 <= parent.size <= 16384:
            cands.append(parent.size)
        elif parent.size > 16384:
            cands.append(16384)
    for L in (8192, 16384):
        if L not in cands:
            cands.append(L)
    cands = sorted(set(cands))
    print(f"[plan] candidate lengths: {cands}")

    # ------------------------------------------------------------------
    # Phase 1: Adam refinement from the parent-derived seeds and a few
    # structured secondary seeds.
    # ------------------------------------------------------------------
    phase1_deadline = t_start + 0.45 * work
    primary_lengths = [best_y.size]
    for L in (8192, 16384):
        if L not in primary_lengths:
            primary_lengths.append(L)

    for L in primary_lengths:
        now = time.monotonic()
        if now > phase1_deadline - 30.0:
            break
        y_start = normalize_y(interpolate_array(best_y, L), L)
        alloc = minimize_call_time = estimate_adam_time(L)
        dl = min(phase1_deadline - 5.0, now + alloc)
        sc, y_out = run_adam(y_start, dl, rng, target_score=target_score,
                             base_lr=0.05, mix=0.3)
        if sc < best_score - 1.0e-12:
            best_score = sc
            best_y = y_out.copy()
            print(f"[adam] n={L} new best={best_score:.8f} t={time.monotonic()-t_start:.0f}s")

    # Secondary short Adam runs from structured inits.
    for L in (8192, 16384):
        if time.monotonic() > phase1_deadline - 45.0:
            break
        inits = build_secondary_inits(L, rng)
        for init in inits:
            now = time.monotonic()
            if now > phase1_deadline - 20.0:
                break
            dl = min(phase1_deadline - 5.0, now + 40.0)
            sc, y_out = run_adam(init, dl, rng, target_score=target_score,
                                 base_lr=0.05, mix=0.3)
            if sc < best_score - 1.0e-12:
                best_score = sc
                best_y = y_out.copy()
                print(f"[adam2] n={L} new best={best_score:.8f} t={time.monotonic()-t_start:.0f}s")

    # ------------------------------------------------------------------
    # Phase 2: Active-set LP directions, at the current best length and,
    # if time remains, at a larger length.
    # ------------------------------------------------------------------
    phase2_deadline = t_start + 0.75 * work
    lp_lengths = [best_y.size]
    if 16384 not in lp_lengths and 16384 > best_y.size:
        lp_lengths.append(16384)

    for L in lp_lengths:
        now = time.monotonic()
        if now > phase2_deadline - 60.0:
            break
        y_start = normalize_y(interpolate_array(best_y, L), L)
        if L <= 8192:
            alloc = min(280.0, phase2_deadline - now - 20.0)
        else:
            alloc = min(380.0, phase2_deadline - now - 20.0)
        alloc = max(45.0, alloc)
        dl = min(phase2_deadline - 5.0, now + alloc)

        sc, y_out = lp_search(y_start, dl, rng)
        if sc < best_score - 1.0e-12:
            best_score = sc
            best_y = y_out.copy()
            print(f"[lp] n={L} new best={best_score:.8f} t={time.monotonic()-t_start:.0f}s")

    # ------------------------------------------------------------------
    # Phase 3: Final polishing with L-BFGS and short targeted Adam/downhill.
    # ------------------------------------------------------------------
    polish_deadline = final_deadline
    polish_lengths = [best_y.size]
    if 16384 not in polish_lengths:
        polish_lengths.append(16384)

    for L in polish_lengths:
        now = time.monotonic()
        if now > polish_deadline - 25.0:
            break
        y_cur = normalize_y(interpolate_array(best_y, L), L)

        # L-BFGS polish at several temperatures.
        for beta in (20.0, 100.0, 400.0):
            now = time.monotonic()
            rem = polish_deadline - now
            if rem < 50.0:
                break
            alloc = min(120.0, 0.45 * rem)
            dl = min(polish_deadline - 5.0, now + alloc)
            sc, y_out = lbfgs_smooth(y_cur, beta, dl)
            if sc < best_score - 1.0e-12:
                best_score = sc
                best_y = y_out.copy()
                print(f"[lbfgs] n={L} beta={beta:.0f} new best={best_score:.8f} "
                      f"t={time.monotonic()-t_start:.0f}s")
            y_cur = out if False else y_out if sc < 2.0 else y_cur  # keep working point stable

        # Short targeted local search.
        now = time.monotonic()
        rem = polish_deadline - now
        if rem > 35.0:
            for trial in range(4):
                now = time.monotonic()
                if now > polish_deadline - 15.0:
                    break
                strength = 0.03 + 0.04 * rng.random()
                y_try = targeted_perturb_y(best_y if best_y.size == L
                                           else normalize_y(interpolate_array(best_y, L), L),
                                           rng, strength)
                alloc = min(45.0, max(20.0, (polish_deadline - now - 10.0) / max(1, 4 - trial)))
                dl = min(polish_deadline - 5.0, now + alloc)
                sc, y_out = run_adam(y_try, dl, rng, target_score=min(target_score, best_score * 0.999),
                                     base_lr=0.035, mix=0.35)
                if sc < best_score - 1.0e-12:
                    best_score = sc
                    best_y = y_out.copy()
                    print(f"[local] n={L} trial={trial} new best={best_score:.8f} "
                          f"t={time.monotonic()-t_start:.0f}s")

    # ------------------------------------------------------------------
    # Phase 4: Cheap final candidate transformations.
    # ------------------------------------------------------------------
    y_cur = best_y
    try:
        n = y_cur.size

        # Reverse symmetry candidate.
        y_rev = 0.5 * (y_cur + y_cur[::-1])
        y_rev = normalize_y(y_rev, n)
        sc = score_y(y_rev)
        if sc < best_score - 1.0e-12:
            best_score = sc
            best_y = y_rev
            print(f"[final-rev] new best={best_score:.8f}")

        # Mild box smoothing candidates.
        for radius in (2, 5, 10):
            kcnt = 2 * radius + 1
            kern = np.ones(kcnt, dtype=np.float64) / float(kcnt)
            ys = np.convolve(y_cur, kern, mode="same")
            for mixf in (0.1, 0.25, 0.5):
                y_cand = (1.0 - mixf) * y_cur + mixf * ys
                y_cand = np.clip(y_cand, 0.0, None)
                s = float(y_cand.sum())
                if s < 1.0e-12:
                    continue
                y_cand = y_cand * (float(n) / s)
                sc = score_y(y_cand)
                if sc < best_score - 1.0e-12:
                    best_score = sc
                    best_y = y_cand
                    y_cur = y_cand
                    print(f"[smooth] r={radius} mix={mixf:.2f} new best={best_score:.8f}")
    except Exception:
        pass

    # Final safety: return a Python list of finite non-negative floats.
    out = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_y, dtype=np.float64).ravel()]
    if not out or any(not np.isfinite(x) for x in out):
        out = [0.5] * 1024

    # Verify once with the provided evaluator (it is assumed already imported).
    try:
        verified = float(evaluate_sequence(out))
        print(f"[final] internal={best_score:.8f} verified={verified:.8f} "
              f"n={len(out)} time={time.monotonic()-t_start:.1f}s")
    except Exception:
        print(f"[final] internal={best_score:.8f} n={len(out)} "
              f"time={time.monotonic()-t_start:.1f}s")

    return out
```
