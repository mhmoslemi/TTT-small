"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed gradient optimization, peak-shaving local search,
and sequence extension to explore longer sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_fraction=1.0):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_fraction < 1.0:
        # Only use top fraction of conv values
        threshold = np.percentile(c, (1 - top_fraction) * 100)
        mask = c >= threshold
        c_sel = c[mask]
    else:
        c_sel = c
    
    c_max = float(c_sel.max())
    scaled = beta * (c_sel - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation with L-BFGS-B, momentum-based exact subgradient polish,
perturbation restarts, and sequence extension to longer lengths.
Key improvements over previous approach: higher beta schedule, momentum polish,
longer sequence exploration, and better time management.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_compute(a, beta):
    """Compute LSE-smoothed score, gradient, and normalized weights p."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n), None
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-30:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        L = c_max
    else:
        p = exp_shifted / exp_sum
        log_sum = np.log(exp_sum)
        L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    # Gradient: dL/da[j] = 2 * sum_m p[j+m] * a[m] (cross-correlation)
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_pa
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad, p


def lse_val_func(x, beta):
    """Objective function for scipy minimize."""
    val, _, _ = lse_compute(x, beta)
    return val


def lse_grad_func(x, beta):
    """Gradient function for scipy minimize."""
    _, grad, _ = lse_compute(x, beta)
    return grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B to minimize LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_val_func, a0, args=(beta,),
            method='L-BFGS-B', jac=lse_grad_func, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return a_out


def momentum_polish(a, n_iters=2000, step_init=0.001, decay=0.999):
    """
    Momentum-based projected subgradient method on the exact max-conv objective.
    Maintains sum(a) approximately constant via renormalization.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = float(a.sum())
    if S0 < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    momentum = np.zeros(n)
    step = step_init
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a:
        # d(conv[k*])/da[j] = 2*a[k*-j] for j in [max(0,k*-n+1), min(k*,n-1)]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        seg = a[j_start:j_end + 1][::-1]
        sg[j_start:j_end + 1] = 2.0 * seg
        # Full gradient of score:
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        # Momentum update
        momentum = 0.9 * momentum - step * grad
        a = a + momentum
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize to maintain sum
        S_new = float(a.sum())
        if S_new > 1e-12:
            a = a * (S0 / S_new)
        step *= decay
        if step < 1e-8:
            break
    return best_a


def target_peak_reduction(a, strength=0.02):
    """Reduce elements contributing to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_new = a.copy()
    contrib = np.zeros(n)
    contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    total_contrib = contrib.sum()
    if total_contrib < 1e-15:
        return a_new
    # Find a threshold: reduce top contributors
    threshold = np.percentile(contrib[contrib > 0], 50) if (contrib > 0).sum() > 4 else 0.0
    reduce_mask = contrib > threshold
    if reduce_mask.sum() < 3:
        reduce_mask = contrib > threshold * 0.5
    if reduce_mask.sum() < 1:
        return a_new
    add_mask = ~reduce_mask
    if add_mask.sum() < 1:
        return a_new
    total_reduced = 0.0
    for j in np.where(reduce_mask)[0]:
        amt = a_new[j] * strength
        a_new[j] -= amt
        if a_new[j] < 0:
            a_new[j] = 0.0
            amt = a_new[j] + amt
        total_reduced += amt
    add_indices = np.where(add_mask)[0]
    per = total_reduced / len(add_indices)
    for j in add_indices:
        a_new[j] += per
    return np.clip(a_new, 0.0, 1000.0)


def extend_sequence(a_parent, n_target, rng):
    """Create a longer sequence by interpolating the parent with added noise."""
    n_p = len(a_parent)
    if n_p >= n_target:
        a_out = a_parent[:n_target].copy()
        return a_out
    # Cubic-like interpolation using FFT
    x_parent = np.linspace(0, 1, n_p + 1)
    y_parent = np.concatenate([a_parent, [a_parent[0]]])
    x_target = np.linspace(0, 1, n_target + 1)
    # Use smooth interpolation via sinusoidal basis (avoids edge artifacts)
    # Fit a smooth function to the parent and evaluate at new points
    from numpy.fft import rfft, irfft
    M = 1 << (max(n_p, n_target) - 1).bit_length()
    Y = rfft(y_parent, M)
    # Evaluate at new points via phase adjustment
    k = np.arange(M // 2 + 1)
    phases_parent = np.exp(-2j * np.pi * k * x_parent[-1] / n_p)
    # Simple approach: use np.interp with cubic fallback
    a_interp = np.interp(x_target[:-1], x_parent, y_parent)
    # Add small noise to help escape local minima
    noise_scale = 0.05 * a_interp.mean()
    a_out = a_interp + noise_scale * rng.standard_normal(n_target)
    return np.clip(a_out, 0.0, 1000.0)


def structured_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None:

    # Uniform

    # Smooth periodic modulations

        inits.append(np.clip(0.5 + amp * np.cos(2.0 * np.pi * freq * x), 0.01, 10.0))
        inits.append(np.clip(0.5 + amp * np.sin(2.0 * np.pi * freq * x), 0.01, 10.0))

    inits.append(np.clip(np.linspace(0.05, 1.0, n), 0.01, 10.0))

        kernel = np.ones(kernel_size) / kernel_size
        a_smooth = np.convolve(a_r, kernel, mode='same')
        inits.append(np.clip(a_smooth, 0.01, 10.0))
