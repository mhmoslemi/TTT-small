import numpy as np

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize as scipy_minimize

_MIN_A = 1e-14
_EPS = 1e-12


def _fft_len(n):
    return next_fast_len(2 * n - 1)


def _convolve_fft(a, L=None):
    n = a.size
    if L is None:
        L = _fft_len(n)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    S = a.sum()
    if S < _EPS:
        return float('inf')
    an = a / S
    n = an.size
    c = _convolve_fft(an)
    return float(2.0 * n * c.max())


def _soft_max_val_and_weights(c, tau):
    cmax = c.max()
    if cmax < 1e-300:
        return 0.0, np.full(c.shape, 1.0 / c.size)
    diff = tau * (c - cmax)
    e = np.exp(diff)
    w = e / e.sum()
    sm = cmax + (1.0 / tau) * np.log(e.sum())
    return sm, w


def _grad_h(a, w, L=None):
    """Compute h_i = sum_k w_k * a_{k-i} for i=0..n-1."""
    n = a.size
    a_rev = a[::-1]
    Lc = next_fast_len(w.size + n - 1)
    fw = rfft(w, Lc)
    far = rfft(a_rev, Lc)
    conv = irfft(fw * far, Lc)
    return conv[n - 1:2 * n - 1]


def _obj_grad(x, tau, L=None, use_exact_max_obj=True):
    """Objective and gradient for softmax-parameterized a = exp(x)/sum(exp(x))."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    a = e / e.sum()
    n = a.size

    c = _convolve_fft(a, L)
    sm, w = _soft_max_val_and_weights(c, tau)

    if use_exact_max_obj:
        obj = 2.0 * n * c.max()
    else:
        obj = 2.0 * n * sm

    h = _grad_h(a, w, L)
    g_a = 2.0 * h
    g_bar = np.dot(a, g_a)
    grad = a * (g_a - g_bar)

    return obj, grad


def _run_lbfgs(x_init, tau, L, deadline, max_iter=300, maxfun=5000):
    """Run L-BFGS-B with timeout."""
    state = {'count': 0}

    def fun(x):
        return _obj_grad(x, tau, L)[0]

    def jac(x):
        return _obj_grad(x, tau, L)[1]

    def callback(xk):
        state['count'] += 1
        if time.monotonic() > deadline:
            raise TimeoutError()
        return False

    try:
        res = scipy_minimize(
            fun=fun,
            x0=x_init,
            jac=jac,
            method='L-BFGS-B',
            callback=callback,
            options={
                'maxiter': max_iter,
                'maxfun': maxfun,
                'factr': 1e8,
                'gtol': 1e-5,
                'disp': False,
            }
        )
        return res.x, res.fun
    except TimeoutError:
        return x_init, float('inf')
    except Exception:
        return x_init, float('inf')


def _multiplicative_update(a, w, alpha, L=None):
    """One step of multiplicative update: a *= exp(-alpha * h), then normalize."""
    n = a.size
    h = _grad_h(a, w, L)
    a_new = a * np.exp(-alpha * h)
    a_new = np.maximum(a_new, _MIN_A)
    s = a_new.sum()
    if s < _EPS:
        return a.copy()
    return a_new / s

def _init_bump(n, rng):
    """Multi-bump initialization."""
    K = rng.integers(2, 6)
    centers = np.sort(rng.uniform(0.1, 0.9, K))
    sigmas = rng.uniform(0.03, 0.2, K)
    weights = rng.uniform(0.5, 2.0, K)
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(K):
        a += weights[i] * np.exp(-((t - centers[i]) / sigmas[i]) ** 2)
    a = np.maximum(a, _MIN_A)
    return a / a.sum()


def _init_modulated(n, rng):
    """Modulated initialization: base * (1 + modulation)."""
    t = np.linspace(0, 1, n)
    base = np.ones(n)
    K = rng.integers(2, 5)
    for _ in range(K):
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.8)
        base *= (1.0 + amp * np.cos(2 * np.pi * freq * t + phase))
    base = np.maximum(base, _MIN_A)
    noise = np.exp(rng.normal(0, 0.3, n))
    a = base * noise
    a = np.maximum(a, _MIN_A)
    return a / a.sum()


def _init_two_scale(n, rng):
    """Two-scale: coarse structure × fine noise."""
    n_coarse = max(8, n // 64)
    coarse = np.ones(n_coarse)
    K = rng.integers(1, 4)
    t_c = np.linspace(0, 1, n_coarse)
    for _ in range(K):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.3)
        w = rng.uniform(0.3, 1.0)

    fine = np.exp(rng.normal(0, 0.5, n))
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse) * fine
    a = np.maximum(a, _MIN_A)
    return a / a.sum()


def _resample(a, m):
    n = a.size
    if n == m:

    out = np.interp(t_new, t_old, a)
    out = np.maximum(out, _MIN_A)
    return out / out.sum()

        L = _fft_len(n_target)
        tau_schedule = [10, 50, 200, 1000, 3000]
        t_per_tau = min(rem / (len(tau_schedule) + 2), 200.0)

        for tau in tau_schedule:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(t_per_tau, rem - 60)
            if t_alloc < 20:
                break

            x_new, f_new = _run_lbfgs(x_init, tau, L, start_time + t_alloc,
                                       max_iter=200, maxfun=3000)
            a_new = np.exp(x_new - x_new.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_score:
                update_best(a_new, sc_new)
                x_init = x_new.copy()
            elif sc_new < _score(a_init) * 1.01:
                x_init = x_new.copy()

        # Stage 3: Final multiplicative polish
        rem = deadline - time.monotonic()
        if rem < 60:
            continue
        t_polish = min(rem * 0.4, 120.0)
        if t_polish > 20:
            a_polish, sc_polish = _multiplicative_loop(
                best_a if best_a.size == n_target else _resample(best_a, n_target),
                start_time + t_polish,
                tau_start=500.0, tau_end=10000.0,
                alpha_start=0.1,
                verbose=False
            )
            update_best(a_polish, sc_polish)

    # Phase 2: LP refinement on the best solution found
    rem = deadline - time.monotonic()
    if rem > 60 and best_a.size <= 4096:
        print(f"Running LP refinement... t={time.monotonic()-start_time:.1f}s")
        a_lp, sc_lp = _lp_refine(best_a, deadline, n_iters=15)
        update_best(a_lp, sc_lp)

    # Phase 3: Try diverse restarts if time remains
    rem = deadline - time.monotonic()
    if rem > 90:
        n_restart = best_a.size
        init_funcs = [_init_bump, _init_modulated, _init_two_scale]
        t_per_restart = min(rem / (len(init_funcs) + 1), 60.0)

        for func in init_funcs:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_r = func(n_restart, rng)
            # Quick multiplicative optimization
            t_alloc = min(t_per_restart, rem - 20)
            if t_alloc > 15:
                a_opt, sc_opt = _multiplicative_loop(
                    a_r, start_time + t_alloc,
                    tau_start=10.0, tau_end=3000.0,
                    alpha_start=0.3, verbose=False
                )
                update_best(a_opt, sc_opt)

    # Phase 4: Final L-BFGS-B polish
    rem = deadline - time.monotonic()
    if rem > 30:
        n_final = best_a.size
        x_init = np.log(best_a + _EPS)
        L = _fft_len(n_final)
        for tau in [2000, 5000, 10000]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            x_new, f_new = _run_lbfgs(x_init, tau, L, start_time + min(rem - 10, 60),
                                       max_iter=100, maxfun=2000)
            a_new = np.exp(x_new - x_new.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_score:
                update_best(a_new, sc_new)
                x_init = x_new.copy()

    # Final output
    final = best_a.astype(float)
    final = np.clip(final, 0, 1000)
    s = final.sum()
    if s < _EPS:
        final = np.ones(2048) / 2048.0
    else:
        final = final / s

    print(f"FINAL: n={len(final)}, score={best_score:.7f}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in final]
