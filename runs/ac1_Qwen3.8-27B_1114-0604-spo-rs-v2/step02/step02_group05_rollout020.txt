import numpy as np

def lp_solve(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) >= 0]  # nonneg already ensures this
    
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # We need (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        constraint = cp.sum([f[j] * g[k - j] for j in range(j_lo, j_hi + 1)]) <= M
        constraints.append(constraint)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve()
    return g.value

    """Build the (2n-1) x n constraint matrix for the LP.
    A[k, i] = 1 if the constraint k involves variable i (with coefficient from f).
    Actually we build the structure and multiply by f later.
    """

    # For column i: A[i:i+n, i] = f[0:n]
    # This is a Toeplitz-like (band) matrix

    for i in range(n):
        A[i:i + n, i] = 1.0  # Unit column, will be scaled by f later
    # But we need A[k,i] = f[k-i], so let's build it properly
    # Clear and rebuild
    A[:] = 0.0
    for i in range(n):
        A[i:i + n, i] = 1.0
    return A

            if s > 1e-10:

    Iteratively: solve LP for optimal g, blend with current a.
    """
    a = _normalize(a0)
    if a is None:
        return None, float('inf')

    n = len(a)
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score

    deadline = time.monotonic() + time_budget
    t_blend = 0.25
    it = 0
    stall = 0
    lp_fail_count = 0

    print(f"    LP opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() < deadline and it < max_iters:
        # Check time for LP
        t_remaining = deadline - time.monotonic()
        if t_remaining < 15:
            break

        # Compute current max
        c = np.convolve(a, a)
        M = c.max()

        # Solve LP
        t0 = time.monotonic()
        g = _solve_lp(a, M, n)
        t_lp = time.monotonic() - t0

        if g is None:
            lp_fail_count += 1
            print(f"    iter={it}: LP failed ({t_lp:.1f}s), using freq smooth")
            # Fallback to frequency smoothing
            g = _freq_smooth(a, 2.0)
            if g is None:
                g = _freq_smooth(a, 3.0)
            if g is None:
                stall += 1
                it += 1
                continue
            is_fallback = True
        else:
            is_fallback = False
            if it % 10 == 0:
                print(f"    iter={it}: M={M:.8f}, score={cur_score:.6f}, lp_time={t_lp:.1f}s")

        # Line search on blending parameter
        best_trial = None
        best_trial_sc = cur_score
        candidates_t = [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1, 0.3, 0.5, 0.7]
        candidates_t = [max(0.01, min(0.99, t)) for t in candidates_t]
        candidates_t = list(set(round(t, 4) for t in candidates_t))

        for t_try in candidates_t:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = _normalize(a_try)
            if a_try is None:
                continue
            sc = _score(a_try)
            if sc < best_trial_sc - 1e-13:
                best_trial = a_try
                best_trial_sc = sc
                best_trial_t = t_try

        if best_trial is not None:
            a = best_trial
            cur_score = best_trial_sc
            t_blend = best_trial_t
            stall = 0
            if cur_score < best_score - 1e-10:
                best = a.copy()
                best_score = cur_score
                print(f"    iter={it}: NEW BEST score={best_score:.6f}, t={t_blend:.3f}")
        else:
            stall += 1
            if stall > 3:
                # Try frequency smoothing directly
                for pwr in (1.5, 2.0, 3.0, 4.0):
                    a_new = _freq_smooth(a, pwr)
                    if a_new is not None:
                        sc = _score(a_new)
                        if sc < cur_score - 1e-12:
                            a = a_new
                            cur_score = sc
                            stall = 0
                            if sc < best_score - 1e-10:
                                best = a.copy()
                                best_score = sc
                            break
                        elif sc < best_score - 1e-10:
                            best = a_new.copy()
                            best_score = sc
            if stall > 10:
                # Reset blending
                t_blend = 0.3
                stall = 5

        it += 1

    # Final: try frequency smoothing on best
    for pwr in (2.0, 3.0, 4.0, 5.0):
        b_new = _freq_smooth(best, pwr)
        if b_new is not None:
            sc = _score(b_new)
            if sc < best_score - 1e-12:
                best = b_new
                best_score = sc

    return best, best_score


def _gradient_refine(a0, rng, time_budget):
    """
    Multiplicative gradient descent refinement (faster per iteration than LP).
    Uses smoothed-max approximation of the max convolution.
    """
    a = _normalize(a0)
    if a is None:
        return None, float('inf')

    n = len(a)
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    alpha = 0.1
    beta = 50.0
    stall = 0
    it = 0

    # Use FFT for speed
    L = 1
    while L < 3 * n:
        L *= 2

    while time.monotonic() < deadline:
        # Compute convolution
        Af = np.fft.rfft(a, L)
        C = np.fft.irfft(Af * Af, L)
        c = C[:2 * n - 1]
        d = n * c  # d[k] = n * (a*a)[k]

        dmax = d.max()

        # Soft weights
        shift = dmax
        e = np.exp(beta * (d - shift))
        es = e.sum()
        if es < 1e-300:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = e / es

        # Gradient: h[j] = 2 * sum_k w[k] * a_contribution to d[k] from a[j]
        # d[k] = n * sum_i a[i] * a[k-i]
        # dd/d a[j] = n * sum_k w[k] * (a[k-j] + (k-j != j)*(a[k-j] ... ))
        # Actually: d[k] = n * (a*a)[k] = n * sum_i a[i]*a[k-i]
        # dd[k]/da[j] = n * (a[k-j] + a[j-(k-j)] ... ) -- need to be careful
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/d a[j] = a[k-j] + a[j-k+n-1]... no
        # Actually for linear convolution: (a*a)[k] = sum_{i:0<=i<n,0<=k-i<n} a[i]*a[k-i]
        # d(a*a)[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j!=k-j... )
        # Wait, there's a subtlety. Let's just use:
        # d(sum_k w[k]*(a*a)[k])/da[j] = sum_k w[k] * (a[k-j] + a[j-k+n-1]... )
        # Hmm, this is getting complicated. Let me use the cross-correlation form.
        #
        # Actually, (a*a)[k] = sum_i a[i] a[k-i].
        # If we differentiate wrt a[j]:
        #   = a[k-j] (from the a[i] term when i=j) + a[j-(k-j)]... no
        #   = a[k-j] (term where i=j, so a[j]*a[k-j], derivative is a[k-j])
        #   + sum over other i where k-i=j, i.e., i=k-j, giving a[k-j]*a[j], derivative a[k-j]
        #   But these are the same term if i=j means k-i=k-j, and i=k-j means the other factor is a[j].
        # So d(a*a)[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
        #                      = a[k-j] = a[j] if j = k-j (i.e., k=2j)
        #
        # For simplicity, let's use the approximation:
        # gradient[j] ≈ 2 * n * (w * a)[j] = 2n * sum_k w[k] * a[k-j]
        # which is 2n * (w cross-correlated with a) -- but with the right indexing
        #
        # More precisely: gradient[j] = n * sum_k w[k] * (a[k-j] + a[j-k+n... ])
        # Let's just compute it as: for each j, sum over k of w[k] * a[k-j] (valid k)
        # This is the cross-correlation of w and a.

        # Use FFT: gradient ≈ 2 * n * irfft(rfft(w,L) * conj(rfft(a,L)))[:n]
        Wf = np.fft.rfft(w, L)
        Af2 = np.fft.rfft(a, L)
        R = np.fft.irfft(Wf * np.conj(Af2), L)
        h = 2.0 * n * R[:n]
        h = np.maximum(h, 0.0)
        h = np.where(np.isfinite(h), h, 0.0)

        hmax = h.max()
        if hmax < 1e-15:
            stall += 1
            if stall > 20:
                break
            continue

        h_norm = h / hmax

        # Try step sizes
        improved = False
        for a_try_val in (alpha, alpha * 0.5, alpha * 2.0):
            a_new = a * np.exp(-a_try_val * h_norm)
            a_new = np.maximum(a_new, 1e-18 / n)
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break

        if improved:
            stall = 0
            if cur_score < best_score - 1e-10:
                best = a.copy()
                best_score = cur_score
                if it % 50 == 0:
                    print(f"    grad iter={it}: score={best_score:.6f}, beta={beta:.0f}")
        else:
            stall += 1
            if stall > 10:
                alpha = max(0.01, alpha * 0.7)
            if stall > 30:
                a = best.copy()
                cur_score = best_score
                alpha = 0.1
                stall = 10

        # Increase beta
        if it % 20 == 0:
            beta = min(5000.0, beta * 1.1)

        it += 1

    return best, best_score


def _init_two_level(n, rng):
    """Two-level initial sequence."""
    frac = rng.uniform(0.3, 0.7)
    ratio = rng.uniform(1.5, 4.0)
    p = np.ones(n)
    k = int(n * frac)
    p[:k] = ratio
    p = p[rng.permutation(n)]
    p = p * (1.0 + 0.05 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return p / s


def _init_cosine(n, rng):
    """Cosine-based initial sequence."""
    t = np.arange(n, dtype=float) / n
    p = np.ones(n)
    for k in range(1, 12):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_gaussian(n, rng):
    """Gaussian mixture initial sequence."""
    K = int(rng.integers(2, 6))
    p = np.zeros(n)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        w = rng.uniform(0.5, 2.0)
        x = np.minimum(np.abs(t - c), 1.0 - np.abs(t - c)) / s
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for minimum-peak-autoconvolution sequences.
    Phase 1: Quick exploration of initial candidates
    Phase 2: LP-based optimization (primary method)
    Phase 3: Gradient refinement
    Phase 4: Frequency-domain polishing
    """
    start_time = time.monotonic()
    total_budget = float(budget_s)
    deadline = start_time + total_budget - 15

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        ps = parent.sum()
        if ps < 1e-10:
            parent = None
        else:
            parent = parent / ps
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            elapsed = time.monotonic() - start_time
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}, t={elapsed:.0f}s) ***")

    # ========== PHASE 1: Quick exploration ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={total_budget:.0f}s)")
    t1_end = start_time + min(30.0, 0.05 * total_budget)

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.6f}, n={len(parent)}")

    # Try a few quick initializations
    for n_try in [2048, 3072, 4096]:
        if time.monotonic() > t1_end:
            break
        inits = [
            np.ones(n_try) / n_try,
            _init_two_level(n_try, rng),
            _init_cosine(n_try, rng),
        ]
        if parent is not None:
            rp = _resample(parent, n_try)
            if rp is not None:
                inits.append(rp)
        for p_init in inits:
            sc = _score(p_init)
            _try_update(p_init, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: LP Optimization ==========
    print("Phase 2: LP-based optimization")
    t2_end = start_time + 0.55 * total_budget

    if best_overall is None:
        best_overall = np.ones(3072) / 3072
        best_score_overall = _score(best_overall)

    # Run LP on the best found so far
    if parent is not None:
        t_rem = t2_end - time.monotonic()
        if t_rem > 60:
            t_alloc = min(0.4 * t_rem, 600.0)
            t_alloc = max(60.0, t_alloc)
            print(f"  Running LP on parent (n={len(parent)}, alloc={t_alloc:.0f}s)")
            p, sc = _lp_optimize(parent, rng, t_alloc, max_iters=100)
            _try_update(p, sc)

    # Also try on best_overall if different from parent
    if best_overall is not None and (parent is None or len(best_overall) != len(parent)):
        t_rem = t2_end - time.monotonic()
        if t_rem > 60:
            t_alloc = min(0.3 * t_rem, 400.0)
            t_alloc = max(60.0, t_alloc)
            print(f"  Running LP on best (n={len(best_overall)}, alloc={t_alloc:.0f}s)")
            p, sc = _lp_optimize(best_overall, rng, t_alloc, max_iters=80)
            _try_update(p, sc)

    # Try a different length
    n_alt = 4096 if len(best_overall) != 4096 else 3072
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        p_init = _resample(best_overall, n_alt)
        if p_init is None:
            p_init = _init_two_level(n_alt, rng)
        t_alloc = min(0.25 * t_rem, 300.0)
        t_alloc = max(45.0, t_alloc)
        print(f"  Running LP on alt length n={n_alt} (alloc={t_alloc:.0f}s)")
        p, sc = _lp_optimize(p_init, rng, t_alloc, max_iters=60)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Gradient Refinement ==========
    print("Phase 3: Gradient refinement")
    t3_end = start_time + 0.80 * total_budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 30 and best_overall is not None:
        t_alloc = min(0.5 * t_rem, 300.0)
        t_alloc = max(30.0, t_alloc)
        print(f"  Running gradient refinement (n={len(best_overall)}, alloc={t_alloc:.0f}s)")
        p, sc = _gradient_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    # Also try from a slightly perturbed version
    t_rem = t3_end - time.monotonic()
    if t_rem > 30 and best_overall is not None:
        a_pert = best_overall * np.exp(rng.normal(0, 0.02, len(best_overall)))
        a_pert = _normalize(a_pert)
        if a_pert is not None:
            t_alloc = min(0.4 * t_rem, 200.0)
            t_alloc = max(20.0, t_alloc)
            p, sc = _gradient_refine(a_pert, rng, t_alloc)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final Polishing ==========
    print("Phase 4: Final polishing")
    t_rem = deadline - time.monotonic()

    if best_overall is not None and t_rem > 20:
        # Multi-power frequency smoothing
        a = best_overall.copy()
        for _ in range(30):
            if time.monotonic() > deadline - 10:
                break
            improved = False
            for pwr in (1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 7.0, 10.0):
                a_new = _freq_smooth(a, pwr)
                if a_new is not None:
                    sc = _score(a_new)
                    if sc < _score(a) - 1e-13:
                        a = a_new
                        improved = True
                        break
                    if sc < best_score_overall - 1e-12:
                        best_overall = a_new.copy()
                        best_score_overall = sc
                        print(f"  Polish: new best {sc:.6f} (pwr={pwr})")
            if not improved:
                break

        _try_update(a, _score(a))

        # One more short gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.5 * t_rem, 120.0)
            t_alloc = max(20.0, t_alloc)
            p, sc = _gradient_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

    # ========== Final Output ==========
    if best_overall is None:
        best_overall = np.ones(2048) / 2048

    p_final = _normalize(best_overall)
    if p_final is None:
        p_final = np.ones(2048) / 2048

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f}, n={len(p_final)}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.clip(p_final, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048) / 2048
    else:
        out = out / s

    return [float(x) for x in out]
