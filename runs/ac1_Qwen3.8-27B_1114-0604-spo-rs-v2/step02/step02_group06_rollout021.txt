"""
Multi-scale multiplicative gradient descent for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Coarse-to-fine: optimize at increasing sequence lengths
- Exponential peak weighting for smooth gradient approximation
- Aggressive line search with multiple step sizes
- Periodic restarts to escape local minima
- Also start from parent sequence (resampled)
- Final polishing phase at the best sequence length found
"""

import numpy as np
import time

RNG = np.random.default_rng(7)


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _sanitize(p, normalize=True):
    a = np.asarray(p, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a


def _score_conv(p, L=None):
    n = int(p.size)
    if L is None:
        L = _next_pow2(2 * n - 1)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    sc = float(2.0 * n * c.max())
    return sc, c


def _resample(p, m):
    m = int(max(1, m))
    p = _sanitize(p)
    if p is None:
        return None
    n = int(p.size)
    if n == m:
        return p.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, p)
    return _sanitize(out)


def _initial_sequence(n, mode):

    elif mode == 4:

        w = float(RNG.uniform(3.0, 12.0))

    else:

        p = np.convolve(p, kern, mode='same')
    return _sanitize(p)


def _compute_force(p, c, L_force, weight='exp', beta=30.0, K=None):
    """Compute the force vector for multiplicative update."""
    n = int(p.size)
    maxc = float(c.max())
    if maxc <= 1e-300:
        return np.ones(n) / n

    if weight == 'exp':
        d = (c - maxc) / maxc
        w_full = np.exp(np.clip(beta * d, -60.0, 0.0))
        w_full = w_full / (w_full.sum() + 1e-300)
    elif weight == 'topk':
        K = int(K) if K else max(3, n // 16)
        w_full = np.zeros(2 * n - 1)
        idx = np.argsort(c)[-K:]
        vals = c[idx]
        w_full[idx] = vals / (vals.sum() + 1e-300)
    else:
        w_full = np.zeros(2 * n - 1)
        w_full[int(np.argmax(c))] = 1.0

    # force[i] = sum_k w_full[k] * p[k-i] (cross-correlation)
    # Use FFT: need length >= (2n-1) + n - 1 = 3n-2
    L = _next_pow2(3 * n - 1)
    Fw = np.fft.rfft(w_full, L)
    Fp = np.fft.rfft(p, L)
    force = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
    force = np.maximum(force, 0.0)
    fm = float(force.max())
    if fm <= 1e-300:
        force = np.ones(n) / n
    else:
        force = force / fm
    return force


def _compute_force_shift(p, c, K=1):
    """Simpler force: use top-K peaks, each contributing a shifted copy of p."""
    n = int(p.size)
    maxc = float(c.max())
    if maxc <= 1e-300:
        return np.ones(n) / n

    force = np.zeros(n)
    # Get top K peak positions
    if K <= 1:
        k_star = int(np.argmax(c))
        weights = [1.0]
        peaks = [k_star]
    else:
        idx = np.argsort(c)[-K:]
        vals = c[idx]
        wsum = vals.sum()
        weights = [float(v) for v in vals / wsum]
        peaks = [int(i) for i in idx]

    for k, w in zip(peaks, weights):
        # force[i] += w * p[k - i] for valid k-i in [0, n-1]
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # p[k - i] for i in [i_lo, i_hi] = p[k - i_hi : k - i_lo + 1][::-1]
            seg_start = k - i_hi
            seg_end = k - i_lo + 1
            force[i_lo:i_hi + 1] += w * p[seg_start:seg_end][::-1]

    fm = float(force.max())
    if fm <= 1e-300:
        return np.ones(n) / n
    return force / fm

def _polish(p, time_limit, verbose=False):
    """Fine polishing with very small steps and coordinate-like moves."""
    p = _sanitize(p)
    if p is None:
        return None, float('inf')

    n = int(p.size)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_sc, _ = _score_conv(p, L)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0
    last_print = -100

    # Use small blocks
    small = max(1, n // 512)
    mid = max(small + 1, n // 64)

    while time.monotonic() < deadline:
        r = RNG.random()
        if r < 0.4:
            w = int(RNG.integers(1, max(2, small * 3)))
        elif r < 0.8:
            w = int(RNG.integers(max(1, small), max(2, mid)))
        else:
            w = int(RNG.integers(mid, min(n, mid * 4 + 1)))
        w = max(1, min(n - 1, w))
        start = int(RNG.integers(0, n - w))

        factors = [0.85, 0.92, 0.96, 1.04, 1.08, 1.15]
        best_cand = None
        best_cand_sc = cur_sc

        for f in factors:
            cand = p.copy()
            cand[start:start + w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            sc_try, _ = _score_conv(cand, L)
            if sc_try < best_cand_sc - 1e-13:
                best_cand_sc = sc_try
                best_cand = cand

        if best_cand is not None:
            p = best_cand
            cur_sc = best_cand_sc
            stagnant = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = p.copy()
                if verbose and (it - last_print >= 200):
                    print(f"    polish n={n} it={it} best={best_sc:.8f}")
                    last_print = it
        else:
            stagnant += 1
            if stagnant >= 50:
                p_try = _sanitize(p * np.exp(RNG.normal(0, 0.03, n)))
                if p_try is not None:
                    sc_try, _ = _score_conv(p_try, L)
                    if sc_try < best_sc * 1.01:
                        p = p_try
                        cur_sc = sc_try
                stagnant = 0
            if stagnant >= 200:
                p = best.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best, best_sc
