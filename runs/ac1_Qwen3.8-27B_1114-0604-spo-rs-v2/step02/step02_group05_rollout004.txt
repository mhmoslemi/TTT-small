def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b, L):
    """Full linear convolution via FFT."""
    la, lb = len(a), len(b)
    nout = la + lb - 1
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout]


def _score_fast(p, conv=None):
    """Compute score. If conv is provided, use it directly."""
    if conv is not None:
        s = p.sum()
        if s < 1e-10:
            return float('inf')
        p_norm = p / s
        n = len(p_norm)
        # conv was computed from p (not p_norm), so scale
        conv_norm = conv / (s * s)
        return float(2.0 * n * conv_norm.max())
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)

def _freq_flatten_v2(p, alpha_power, L_conv):
    """
    Frequency-domain convolution flattening.
    Makes the convolution flatter by modifying its spectrum.
    """
    n = len(p)
    A = np.fft.rfft(p, L_conv)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L_conv)[:2*n-1]
    
    cmax = C.max()
    cmin = C[C > 0].min() if (C > 0).any() else 0
    
    if cmax < 1e-15:
        return p.copy()
    
    # Flatten: new_C = cmax * (C/cmax)^(1/alpha_power)
    # alpha_power > 1 flattens
    C_pos = np.maximum(C, 0)
    C_norm = C_pos / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha_power)
    C_new = np.maximum(C_new, 0)
    
    C_freq_new = np.fft.rfft(C_new, L_conv)
    # Take square root (complex)
    A_new = np.sqrt(C_freq_new.astype(complex))
    a_new = np.fft.irfft(A_new, L_conv)[:n]
    
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.25:
        return None
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

                    cur = best.copy()
                    cur_score = best_score
                    stall = 0
    
    return best, best_score
