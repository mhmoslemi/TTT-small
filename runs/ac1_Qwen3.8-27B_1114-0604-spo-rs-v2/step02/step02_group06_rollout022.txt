import time

def _fft_conv(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L <<= 1
    FA = np.fft.rfft(a, L)
    FB = np.fft.rfft(b, L)
    return np.fft.irfft(FA * FB, L)[:n]

def _score(a):
    """Compute the objective value for a sequence (assumed normalized to sum=1)."""
    n = len(a)
    c = _fft_conv(a, a)

def _smooth_max_grad(a, beta, w=None):

"""
Gradient-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max approximation with FFT-accelerated gradient computation,
adaptive step sizes, and periodic restarts.
"""

import numpy as np
import time

_RNG = np.random.default_rng(42)

def _next_pow2(x):
    if x <= 1:
        return 1
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    FA = np.fft.rfft(a, L)
    FB = np.fft.rfft(b, L)
    return np.fft.irfft(FA * FB, L)[:n]

def _compute_score(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * np.max(c))

def _compute_conv(a):
    return _fft_convolve(a, a)

def _softmax_weights(c, beta):
    x = beta * c
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Fallback: uniform over the top elements
        w = np.zeros_like(c)
        top_k = max(1, int(0.01 * len(c)))
        idx = np.argsort(c)[-top_k:]
        w[idx] = 1.0 / len(idx)
        return w
    return e / s

def _gradient_smooth_max(a, c, w):
    """Compute gradient of smooth max of c with respect to a.
    g[i] = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv = _fft_convolve(a_rev, w)
    # g[i] = 2 * conv[i + n - 1]
    g = 2.0 * conv[n-1 : 2*n-1]
    return g

def _optimize_gradient(a0, time_limit, beta0=50.0, beta1=500.0, alpha0=1.0, 
                       min_alpha=0.01, verbose=False):
    """Main gradient-based optimizer."""
    a = a0.copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, _compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = _compute_score(a)
    current_score = best_score
    
    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0
    alpha = alpha0
    beta = beta0
    
    while time.monotonic() < deadline:
        c = _compute_conv(a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall = 0
            if verbose and it % 100 == 0:
                print(f"  grad: it={it}, score={best_score:.8f}")
        else:
            stall += 1
        
        # Compute softmax weights
        w = _softmax_weights(c, beta)
        
        # Compute gradient
        g = _gradient_smooth_max(a, c, w)
        
        # Center the gradient for the multiplicative update
        g_centered = g - g.mean()
        g_max = float(np.max(np.abs(g_centered)))
        if g_max < 1e-15:
            break
        
        g_normed = g_centered / g_max

        best_trial_score = current_score
        for trial_alpha in [alpha * 0.5, alpha, alpha * 2.0]:

            a_new = np.maximum(a_new, 1e-18)
            s = a_new.sum()
            if s < 1e-15 or not np.isfinite(s):
                continue
            a_new = a_new / s
            sc = _compute_score(a_new)
            if not np.isfinite(sc):
                continue
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_new
        
        if best_trial is not None:
            a = best_trial
            current_score = best_trial_score
            alpha = min(alpha * 1.05, 10.0)

                alpha = alpha0
                stall = 0

    a = a0.copy()

                    a = a_try
                    break
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 50:
            noise = _RNG.normal(0, 0.05, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 1e-18)
            a = a / a.sum()
            current_score = _compute_score(a)
            stall = 0
        
        it += 1
    
    return best_a, best_score

"""
Gradient-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max and threshold-based gradients with FFT-accelerated computation,
adaptive step sizes, multi-phase search, and periodic restarts.
"""

import numpy as np
import time

_RNG = np.random.default_rng(42)

def _next_pow2(x):
    if x <= 1:
        return 1
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    FA = np.fft.rfft(a, L)
    FB = np.fft.rfft(b, L)
    return np.fft.irfft(FA * FB, L)[:n]

def _compute_conv(a):
    n = len(a)
    L = _next_pow2(2 * n - 1)
    FA = np.fft.rfft(a, L)
    c = np.fft.irfft(FA * FA, L)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _compute_score(a):
    n = len(a)
    c = _compute_conv(a)
    return float(2.0 * n * np.max(c))

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15 or not np.isfinite(s):
        return None
    return a / s

def _smooth_max_grad(a, c, beta):
    """Compute gradient of smooth max of c via softmax weights."""
    n = len(a)
    x = beta * c
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        top_k = max(1, int(0.005 * len(c)))
        idx = np.argsort(c)[-top_k:]
        w[idx] = 1.0 / len(idx)
    else:
        w = e / s
    a_rev = a[::-1].copy()
    conv = _fft_convolve(a_rev, w)
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g

def _threshold_grad(a, c, threshold):
    """Compute gradient of sum(max(c[k]-threshold, 0))."""
    n = len(a)
    w = np.maximum(c - threshold, 0.0)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    a_rev = a[::-1].copy()
    conv = _fft_convolve(a_rev, w)
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g

def _gradient_optimize(a0, time_limit, mode="smooth", beta0=50.0, beta1=500.0,
                       alpha0=2.0, min_alpha=0.05, target_score=None, verbose=False):
    """Main gradient-based optimizer with adaptive parameters."""
    a = a0.copy()
    n = len(a)
    a = _normalize(a)
    if a is None:
        return a0, _compute_score(a0)

    best_a = a.copy()
    best_score = _compute_score(a)
    current_score = best_score

    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0
    alpha = alpha0
    beta = beta0

    while time.monotonic() < deadline:
        c = _compute_conv(a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    grad: it={it}, score={best_score:.8f}, alpha={alpha:.3f}, beta={beta:.1f}")
        else:
            stall += 1

        # Choose gradient type
        if mode == "smooth":
            g = _smooth_max_grad(a, c, beta)
        elif mode == "threshold":
            if target_score is not None:
                target_m = target_score / (2.0 * n)
            else:
                target_m = max_c * 0.9
            threshold = 0.5 * (max_c + target_m)
            g = _threshold_grad(a, c, threshold)
        else:
            # Mixed
            if it % 3 < 2:
                g = _smooth_max_grad(a, c, beta)
            else:
                if target_score is not None:
                    target_m = target_score / (2.0 * n)
                else:
                    target_m = max_c * 0.9
                threshold = 0.5 * (max_c + target_m)
                g = _threshold_grad(a, c, threshold)

        g_centered = g - g.mean()
        g_max = float(np.max(np.abs(g_centered)))

        if g_max < 1e-15:
            # Flat gradient, need perturbation
            noise = _RNG.normal(0, 0.1, n)
            a_new = a * np.exp(noise)
            a_new = _normalize(a_new)
            if a_new is not None:
                a = a_new
                current_score = _compute_score(a)
            stall += 1
            it += 1
            continue

        g_normed = g_centered / g_max

        # Try multiple step sizes
        best_trial = None
        best_trial_score = current_score
        for trial_alpha in [alpha * 0.4, alpha * 0.7, alpha, alpha * 1.5]:
            a_new = a * np.exp(-trial_alpha * g_normed)
            a_new = _normalize(a_new)
            if a_new is None:
                continue
            sc = _compute_score(a_new)
            if np.isfinite(sc) and sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_new

        if best_trial is not None:
            a = best_trial
            current_score = best_trial_score
            alpha = min(alpha * 1.03, 15.0)
        else:
            alpha = max(min_alpha, alpha * 0.6)
            stall += 1

                noise = _RNG.normal(0, min(0.3, 0.05 + 0.01 * stall), n)

                a_new = _normalize(a_new)
                if a_new is not None:
                    sc = _compute_score(a_new)
                    if sc < best_score * 1.05:
                        a = a_new
                        current_score = sc
                    else:
                        a = best_a.copy()
                        current_score = best_score
                stall = 0
                alpha = alpha0

            if stall > 80:
                a = best_a.copy()
                current_score = best_score
                alpha = alpha0
                stall = 0

        # Anneal beta

        it += 1

    return best_a, best_score

def _local_refine(a0, time_limit, verbose=False):

    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score(a)
    current_score = best_score

    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        r = _RNG.random()
        improved = False

        if r < 0.4:
            # Single element adjustment
            i = int(_RNG.integers(0, n))
            val = a[i]
            if val < 1e-15:
                continue
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a_try = a.copy()
                a_try[i] = val * factor
                a_try = _normalize(a_try)
                if a_try is None:
                    continue
                sc = _compute_score(a_try)
                if sc < current_score - 1e-10:
                    current_score = sc
                    a = a_try
                    improved = True
                    break
        elif r < 0.7:
            # Small block
            w = int(_RNG.integers(2, min(16, n)))
            i = int(_RNG.integers(0, n - w))
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a_try = a.copy()
                a_try[i:i + w] *= factor
                a_try = _normalize(a_try)
                if a_try is None:

                sc = _compute_score(a_try)
                if sc < current_score - 1e-10:
                    current_score = sc
                    a = a_try
                    improved = True
                    break
        else:
            # Medium block
            w = int(_RNG.integers(max(2, n // 200), max(3, n // 50)))
            i = int(_RNG.integers(0, n - w))
            for factor in [0.8, 1.2]:
                a_try = a.copy()
                a_try[i:i + w] *= factor
                a_try = _normalize(a_try)
                if a_try is None:
                    continue
                sc = _compute_score(a_try)
                if sc < current_score - 1e-10:
                    current_score = sc
                    a = a_try
                    improved = True
                    break

            if current_score < best_score - 1e-10:

                best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 40:
            noise = _RNG.normal(0, 0.05, n)

                    current_score = sc
                else:
                    a = best_a.copy()

            stall = 0

def _gradient_descent_log(a0, time_limit, beta_start=100.0, beta_end=2000.0,
                          lr_start=0.1, lr_end=0.001, verbose=False):
    """Gradient descent in log-space for smoother optimization."""
    a = a0.copy()
    n = len(a)
    a = _normalize(a)
    if a is None:
        return a0, _compute_score(a0)

    x = np.log(a + 1e-30)  # work in log space
    best_a = a.copy()
    best_score = _compute_score(a)

    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        # Compute gradient in a-space
        c = _compute_conv(a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    loggd: it={it}, score={best_score:.8f}")

        beta = beta_start + (beta_end - beta_start) * (1.0 - max(0, (deadline - time.monotonic()) / time_limit))
        lr = lr_start * (lr_end / lr_start) ** ((1.0 - (deadline - time.monotonic()) / time_limit))

        g = _smooth_max_grad(a, c, beta)
        g_centered = g - g.mean()
        g_max = float(np.max(np.abs(g_centered)))

        if g_max < 1e-15:
            stall += 10
        else:
            # Update in log-space: x_new = x - lr * grad_log
            # grad_log[i] = grad_a[i] / a[i]
            grad_log = g_centered / (a + 1e-30)
            grad_log_max = float(np.max(np.abs(grad_log)))
            if grad_log_max > 1e-10:
                x = x - lr * grad_log / grad_log_max

        # Convert back to a-space
        a = np.exp(x - x.max())
        a = _normalize(a)
        if a is None:
            a = best_a.copy()
            x = np.log(a + 1e-30)

        if current_score >= best_score:
            stall += 1
        else:
            stall = 0

        if stall > 50:
            a = best_a.copy()
            x = np.log(a + 1e-30)
            noise = _RNG.normal(0, 0.1, n)
            a = _normalize(a * np.exp(noise))
            if a is not None:
                x = np.log(a + 1e-30)
            stall = 0

        it += 1

    return best_a, best_score

        if parent is not None:
            a = np.array(parent, dtype=float)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            a = _normalize(a)
            if a is not None:
                sc = _compute_score(a)
                candidates.append((sc, a.copy()))
                print(f"Parent (n={len(a)}): score={sc:.8f}")
    except (NameError, Exception):
        pass

    # Generate diverse initial points
    for n_try in [3072, 4096, 2048]:
        # Uniform
        a = np.ones(n_try) / n_try
        sc = _compute_score(a)
        candidates.append((sc, a.copy()))

        # Two-level
        a = np.ones(n_try)
        a[n_try // 2:] = 0.5
        a = _normalize(a)
        sc = _compute_score(a)
        candidates.append((sc, a.copy()))

        # Gaussian bumps
        t = np.linspace(0, 1, n_try)
        a = np.exp(-((t - 0.3) / 0.1) ** 2) + np.exp(-((t - 0.7) / 0.1) ** 2)
        a = _normalize(a)
        sc = _compute_score(a)
        candidates.append((sc, a.copy()))

        # Random spectral
        n_freq = n_try // 2 + 1
        spectrum = np.abs(_RNG.normal(0, 1, n_freq)) + 0.5
        spectrum[0] = 0  # zero DC for centered
        sig = np.fft.irfft(spectrum, n_try)
        sig = np.abs(sig)
        sig = _normalize(sig)
        if sig is not None:
            sc = _compute_score(sig)
            candidates.append((sc, sig.copy()))

    candidates.sort(key=lambda x: x[0])
    best_a, best_score = candidates[0]
    print(f"Best initial: n={len(best_a)}, score={best_score:.8f}")
    print(f"Available time: {total_time:.0f}s")

    # Phase 1: Main gradient optimization on best candidate (40% time)
    rem = deadline - time.monotonic()
    if rem > 60:
        t1 = min(rem * 0.40, 700)
        print(f"\nPhase 1: smooth-max gradient (t={t1:.0f}s)")
        a, sc = _gradient_optimize(best_a, t1, mode="smooth",
                                   beta0=30, beta1=300, alpha0=2.0,
                                   target_score=1.50, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Phase 1 result: {sc:.8f}")

    # Phase 2: Threshold-based gradient (20% time)
    rem = deadline - time.monotonic()
    if rem > 60:
        t2 = min(rem * 0.20, 400)
        print(f"\nPhase 2: threshold gradient (t={t2:.0f}s)")
        a, sc = _gradient_optimize(best_a, t2, mode="threshold",
                                   alpha0=3.0, min_alpha=0.1,
                                   target_score=1.50, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Phase 2 result: {sc:.8f}")

    # Phase 3: Try other starting points (15% time)
    rem = deadline - time.monotonic()
    if rem > 40:
        t3_total = min(rem * 0.15, 300)
        print(f"\nPhase 3: diverse starts (t={t3_total:.0f}s)")
        n_starts = min(3, len(candidates) - 1)
        t_each = t3_total / max(1, n_starts)
        for i in range(1, min(4, len(candidates))):
            rem2 = deadline - time.monotonic()
            if rem2 < 30:
                break
            t = min(t_each, rem2 - 20)
            if t < 15:
                break
            sc0, a0 = candidates[i]
            a, sc = _gradient_optimize(a0, t, mode="mixed",
                                       beta0=20, beta1=400, alpha0=3.0,
                                       target_score=1.50, verbose=False)
            if sc < best_score:
                best_score = sc
                best_a = a
                print(f"  Phase 3 improvement: {sc:.8f}")

    # Phase 4: Log-space gradient descent (15% time)
    rem = deadline - time.monotonic()
    if rem > 40:
        t4 = min(rem * 0.15, 350)
        print(f"\nPhase 4: log-space gradient (t={t4:.0f}s)")
        a, sc = _gradient_descent_log(best_a, t4,
                                      beta_start=50, beta_end=800,
                                      lr_start=0.2, lr_end=0.01,
                                      verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Phase 4 result: {sc:.8f}")

    # Phase 5: Local refinement (10% time)
    rem = deadline - time.monotonic()
    if rem > 30:
        t5 = min(rem * 0.10, 200)
        print(f"\nPhase 5: local refine (t={t5:.0f}s)")
        a, sc = _local_refine(best_a, t5)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Phase 5 result: {sc:.8f}")

    # Phase 6: Final fine gradient (remaining time)
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem - 10, 300)
        if t6 > 10:
            print(f"\nPhase 6: fine gradient (t={t6:.0f}s)")
            a, sc = _gradient_optimize(best_a, t6, mode="smooth",
                                       beta0=200, beta1=2000, alpha0=0.5,
                                       min_alpha=0.01, target_score=1.50,
                                       verbose=True)
            if sc < best_score:
                best_score = sc
                best_a = a
                print(f"  Phase 6 result: {sc:.8f}")

    # Final local refine
    rem = deadline - time.monotonic()
    if rem > 15:
        t7 = min(rem - 5, 60)
        if t7 > 5:
            a, sc = _local_refine(best_a, t7)
            if sc < best_score:
                best_score = sc
                best_a = a

    print(f"\nFinal: n={len(best_a)}, score={best_score:.8f}")
    print(f"Total time used: {time.monotonic() - start:.1f}s")

    result = [float(x) for x in best_a]
    return result
