"""
Step-function optimizer using peak-targeted gradient descent with
adaptive temperature, block mass transfer, and multi-length search.
"""

import numpy as np
import time
import math

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v

def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _score(f):
    n = f.size
    c = _autocorr(f)
    return float(2.0 * n * c.max())

def _peak_grad(f, c, temperature=0.01):
    """
    Compute gradient of soft-max of autocorrelation.
    Returns gradient vector of length n.
    """
    n = f.size
    L = c.size
    max_c = c.max()
    if max_c <= 0:
        return np.zeros(n)
    
    # Soft weights over lags
    z = (c - max_c) / (temperature * max_c)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(L)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    
    # Gradient: g_i = sum_k w_k * 2*f_{k-i} (approximately, ignoring k=2i edge case)
    # For efficiency, compute via direct sum over significant lags
    # Only iterate over lags with significant weight
    g = np.zeros(n, dtype=np.float64)
    
    # Find lags with significant weight
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    
    for k in active:
        k = int(k)
        wi = float(w[k])
        if wi < 1e-10:
            continue
        # dc_k/df_i ≈ 2*f_{k-i} for i where 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        k_minus_i = k - idx  # these are in [0, n-1]
        g[lo:hi+1] += 2.0 * wi * f[k_minus_i]
    
    return g

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.ones(n) / n
    return out / s

def _interp_to(f, n_new):
    f = _normalize(f)
    n_old = f.size
    if n_new == n_old:
        return f
    if n_new == 1:
        return np.array([1.0])
    if n_old == 1:
        return _normalize(np.full(n_new, f[0]))
    x_old = np.linspace(0, n_old - 1, n_old)
    x_new = np.linspace(0, n_new - 1, n_new)
    return _normalize(np.interp(x_new, x_old, f))

def _gradient_step(f, c, cmax, lr, rng):
    """
    Perform one gradient step with line search.
    Returns (new_f, new_c, new_cmax, accepted).
    """
    n = f.size
    
    # Compute peak gradient
    g = _peak_grad(f, c, temperature=lr * 5)
    
    # Center the gradient (to preserve sum)
    g -= g.mean()
    gscale = g.max() - g.min()
    if gscale < 1e-300:
        return f, c, cmax, False
    
    # Try several step sizes
    best_f = f
    best_c = c
    best_mc = cmax
    accepted = False
    
    for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
        # Log-domain step (keeps non-negativity)
        f_new = f * np.exp(-alpha * g / gscale)
        f_new = _normalize(f_new)
        c_new = _autocorr(f_new)
        mc_new = c_new.max()
        if mc_new < best_mc:
            best_mc = mc_new
            best_f = f_new
            best_c = c_new
            accepted = True
        
        # Also try Euclidean step for larger alphas
        if alpha >= lr * 0.5:
            f_new2 = f - alpha * g / gscale
            f_new2 = _project_simplex(f_new2)
            f_new2 = _normalize(f_new2)
            c_new2 = _autocorr(f_new2)
            mc_new2 = c_new2.max()
            if mc_new2 < best_mc:
                best_mc = mc_new2
                best_f = f_new2
                best_c = c_new2
                accepted = True
    
    return best_f, best_c, best_mc, accepted

def _block_transfer(f, c, cmax, n_blocks, rng):
    """
    Transfer mass between blocks to reduce peak autocorrelation.
    """
    n = f.size
    block_size = max(1, n // n_blocks)
    actual_blocks = n // block_size
    
    best_f = f
    best_c = c
    best_mc = cmax
    improved = False
    
    for _ in range(20):
        # Pick two random blocks
        i = int(rng.integers(actual_blocks))
        j = int(rng.integers(actual_blocks))
        if i == j:
            continue
        
        # Transfer a fraction of block i's mass to block j
        total_i = f[i*block_size:(i+1)*block_size].sum()
        if total_i < 1e-12:
            continue
        
        frac = float(rng.uniform(0.05, 0.4))
        dv = total_i * frac
        
        f_new = f.copy()
        f_new[i*block_size:(i+1)*block_size] -= dv / block_size
        f_new[j*block_size:(j+1)*block_size] += dv / block_size
        
        if (f_new < 0).any():
            # Fall back to proportional transfer
            block_i = f[i*block_size:(i+1)*block_size]
            transfer = block_i * frac
            f_new = f.copy()
            f_new[i*block_size:(i+1)*block_size] -= transfer
            f_new[j*block_size:(j+1)*block_size] += transfer
            if (f_new < 0).any():
                continue
        
        f_new = _normalize(f_new)
        c_new = _autocorr(f_new)
        mc_new = c_new.max()
        if mc_new < best_mc:
            best_mc = mc_new
            best_f = f_new
            best_c = c_new
            improved = True
    
    return best_f, best_c, best_mc, improved

def _coordinate_polish(f, c, budget, rng):
    """
    Fast coordinate-level local search.
    """
    n = f.size
    L = c.size
    f = _normalize(f)
    c = _autocorr(f)
    mc = c.max()
    best_f = f.copy()
    best_mc = mc
    best_score = 2.0 * n * mc
    
    deadline = time.monotonic() + budget
    delta = np.zeros(L)
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 30:
        mc = c.max()
        
        # Try transfers from high-gradient to low-gradient positions
        # Compute simple gradient: for each position, how much does it contribute to peak?
        peak_lags = np.where(c >= mc * 0.999)[0]

        # So positions that appear in many peak lags are "hot"
        hotness = np.zeros(n)

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            hotness[lo:hi+1] += f[k - np.arange(lo, hi+1)]

        thresh_hot = np.median(hotness) + 0.5 * (hotness.max() - hotness.min() + 1e-300)
        thresh_cold = np.median(hotness) - 0.5 * (hotness.max() - hotness.min() + 1e-300)
        
        hot_positions = np.where((hotness > thresh_hot) & (f > 1e-12))[0]
        cold_positions = np.where(hotness < thresh_cold)[0]
        
        if len(hot_positions) == 0 or len(cold_positions) == 0:
            no_improve += 1
            break
        
        best_move = None
        best_move_mc = mc
        
        # Try several random hot->cold transfers
        for _ in range(40):
            i = int(hot_positions[rng.integers(len(hot_positions))])
            j = int(cold_positions[rng.integers(len(cold_positions))])
            if i == j:
                continue
            frac = float(rng.uniform(0.01, 0.15))
            dv = f[i] * frac
            
            # Compute new c max

            delta[i:i+n] -= 2.0 * dv * f

            delta[2*i] += dv*dv

            c_new = c + delta
            mc_new = c_new.max()

                best_move_mc = mc_new
                best_move = (i, j, dv, delta.copy())

            continue

        f[i] -= dv
        f[j] += dv
        c = c + d
        no_improve = 0

            best_score = cur_score
            best_f = f.copy()

            f = _normalize(f)
            c = _autocorr(f)

    c = _autocorr(f)
    mc = c.max()
    sc = 2.0 * n * mc

"""
Step-function optimizer using adaptive-temperature peak-targeted gradient descent,
systematic coordinate polish, block mass transfer, and multi-length exploration.
Focuses on reducing the maximum autocorrelation peak by targeting the hottest lags.
"""

import numpy as np
import time
import math


def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _autocorr(f):
    """Full linear autocorrelation via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _score(f):
    """Compute 2*n*max(autocorr) for a normalized sequence."""
    n = f.size
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())


def _peak_gradient(f, c, temperature):
    """
    Compute the gradient of the soft-max of autocorrelation values.
    g_i = sum_k p_k * dc_k/df_i where p_k = softmax(c/T)
    dc_k/df_i ≈ 2*f_{k-i} for most i (ignoring k=2i edge case)
    """
    n = f.size
    L = c.size
    max_c = float(c.max())
    if max_c <= 1e-300:
        return np.zeros(n, dtype=np.float64)

    # Softmax weights over lags
    z = (c - max_c) / (temperature * max_c)
    np.clip(z, -50.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws

    # Compute g_i = 2 * sum_k w_k * f_{k-i} (cross-correlation)
    # Only iterate over lags with significant weight
    g = np.zeros(n, dtype=np.float64)
    thresh = w.max() * 0.05
    active = np.where(w > thresh)[0]

    for k in active:
        k = int(k)
        wi = float(w[k])
        if wi < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        k_minus_i = k - idx
        g[lo:hi + 1] += 2.0 * wi * f[k_minus_i]

    return g


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def _interp_to(f, n_new):
    """Linearly resample f to length n_new and normalize."""
    f = _normalize(f)
    n_old = f.size
    if n_new == n_old:
        return f
    if n_new == 1:
        return np.array([1.0], dtype=np.float64)
    if n_old == 1:
        return _normalize(np.full(n_new, f[0]))
    x_old = np.linspace(0.0, n_old - 1, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, n_new - 1, n_new, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _gradient_optimize(f0, budget, rng):
    """
    Main optimization: adaptive-temperature gradient descent with
    line search, periodic coordinate polish, and block transfer.
    """
    f = _normalize(f0)
    n = f.size
    c = _autocorr(f)
    mc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * mc)

    deadline = time.monotonic() + budget
    temperature = 0.05
    min_temp = 0.002
    lr = 0.03
    min_lr = 1e-7
    no_accept = 0
    it = 0
    last_polish = time.monotonic()

    print(f"    start: n={n}, score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break

        # Compute peak gradient
        g = _peak_gradient(f, c, temperature)
        g_centered = g - float(g.mean())
        g_range = float(g_centered.max() - g_centered.min())

        accepted = False
        if g_range > 1e-300:
            best_mc = mc
            best_f_new = f
            best_c_new = c

            # Try multiple step sizes
            alphas = [lr, lr * 0.5, lr * 0.2, lr * 0.08]
            for alpha in alphas:
                # Log-domain step
                arg = -alpha * g_centered / g_range
                np.clip(arg, -40.0, 40.0, out=arg)
                f_new = _normalize(f * np.exp(arg))
                c_new = _autocorr(f_new)
                mc_new = float(c_new.max())
                if mc_new < best_mc:
                    best_mc = mc_new
                    best_f_new = f_new
                    best_c_new = c_new

                # Euclidean step for larger alphas
                if alpha >= lr * 0.4:
                    f_new2 = _project_simplex(f - alpha * g_centered / g_range)
                    f_new2 = _normalize(f_new2)
                    c_new2 = _autocorr(f_new2)
                    mc_new2 = float(c_new2.max())
                    if mc_new2 < best_mc:
                        best_mc = mc_new2
                        best_f_new = f_new2
                        best_c_new = c_new2

            if best_mc < mc - 1e-16:
                f = best_f_new
                c = best_c_new
                mc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            lr = min(0.08, lr * 1.01)
            temperature = max(min_temp, temperature * 0.997)
            cur_score = 2.0 * n * mc
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if it % 500 == 0:
                    print(f"    iter={it}, score={best_score:.6f}, temp={temperature:.4f}, lr={lr:.4f}, left={remaining:.0f}s")
        else:
            no_accept += 1
            temperature = min(0.15, temperature * 1.02)
            lr = max(min_lr, lr * 0.98)

            # Periodic coordinate polish
            if no_accept > 15 and time.monotonic() - last_polish > 1.0 and remaining > 30:
                polish_budget = min(20.0, remaining * 0.04)
                fp, sp = _quick_polish(f, c, polish_budget, rng)
                if sp < 2.0 * n * mc:
                    f = fp
                    c = _autocorr(f)
                    mc = float(c.max())
                    no_accept = 0
                    lr = min(0.05, lr * 1.3)
                    cur_score = 2.0 * n * mc
                    if cur_score < best_score:
                        best_score = cur_score
                        best_f = f.copy()
                last_polish = time.monotonic()

            # Restart if stuck
            if no_accept > 60:
                sigma = 0.15 + 0.2 * min(1.0, no_accept / 150.0)
                f = _normalize(best_f * np.exp(sigma * rng.standard_normal(n)))
                c = _autocorr(f)
                mc = float(c.max())
                temperature = min(0.12, temperature * 2.0)
                lr = 0.02
                no_accept = 0

        # Heavy periodic polish
        if it % 2000 == 0 and remaining > 60:
            polish_budget = min(40.0, remaining * 0.05)
            fp, sp = _quick_polish(f, c, polish_budget, rng)
            if sp < best_score:
                best_score = sp
                best_f = fp.copy()
                f = best_f
                c = _autocorr(f)
                mc = float(c.max())
                print(f"    iter={it}, polish improved to {best_score:.6f}")

    final_c = _autocorr(best_f)
    final_score = 2.0 * n * float(final_c.max())
    if final_score < best_score:
        best_score = final_score
    print(f"    end: score={best_score:.6f}")
    return best_f, best_score


def _quick_polish(f0, c0, budget, rng):
    """
    Fast coordinate-level local search using exact transfer evaluation.
    """
    f = _normalize(f0)
    n = f.size
    L = 2 * n - 1
    c = _autocorr(f)
    mc = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * mc
    deadline = time.monotonic() + budget

    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)

    # Precompute contribution vectors
    # For position i, the change in c when moving dv from i to j:
    # delta_c = 2*dv*(unit_j - unit_i) + dv^2*(e_{2j} + e_{2i} - 2*e_{i+j})
    # where unit_i[k] = f_{k-i} (shifted f)

    f_units = []
    for i in range(n):
        u = np.zeros(L, dtype=np.float64)
        u[i:i + n] = f
        f_units.append(u)

    no_improve = 0
    move_count = 0

    while time.monotonic() < deadline and no_improve < 25:
        mc = float(c.max())

        # Find hot and cold positions based on peak contribution
        peak_lags = np.where(c >= mc * 0.995)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(c))])
        peak_lags = peak_lags[:30]  # limit

        hotness = np.zeros(n, dtype=np.float64)
        for k in peak_lags:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            hotness[lo:hi + 1] += f[k - np.arange(lo, hi + 1)]

        med = float(np.median(hotness))
        rng_h = float(hotness.max() - hotness.min())
        if rng_h < 1e-300:
            break

        hot_mask = (hotness > med + 0.3 * rng_h) & (f > 1e-10)
        cold_mask = hotness < med - 0.3 * rng_h
        hot_idx = np.where(hot_mask)[0]
        cold_idx = np.where(cold_mask)[0]

        if len(hot_idx) == 0 or len(cold_idx) == 0:
            # Relax thresholds
            hot_idx = np.where(f > np.percentile(f, 80))[0]
            cold_idx = np.where(f < np.percentile(f, 50))[0]
            if len(hot_idx) == 0 or len(cold_idx) == 0:
                no_improve += 1
                break

        best_move_mc = mc
        best_move = None

        # Try transfers
        n_try = min(60, len(hot_idx) * len(cold_idx))
        for t in range(n_try):
            i = int(hot_idx[rng.integers(len(hot_idx))])
            j = int(cold_idx[rng.integers(len(cold_idx))])
            if i == j:
                continue
            frac = float(rng.uniform(0.02, 0.2))
            dv = float(f[i]) * frac
            if dv < 1e-14:
                continue

            # Compute new c
            delta.fill(0.0)
            delta[i:i + n] -= 2.0 * dv * f
            delta[j:j + n] += 2.0 * dv * f
            delta[2 * j] += dv * dv
            delta[2 * i] += dv * dv
            delta[i + j] -= 2.0 * dv * dv
            total = c + delta
            mc_new = float(total.max())

            if mc_new < best_move_mc:
                best_move_mc = mc_new
                best_move = (i, j, dv, delta.copy())

        # Also try some random transfers
        for t in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or f[i] < 1e-10:
                continue
            frac = float(rng.uniform(0.01, 0.1))
            dv = float(f[i]) * frac

            delta.fill(0.0)
            delta[i:i + n] -= 2.0 * dv * f
            delta[j:j + n] += 2.0 * dv * f
            delta[2 * j] += dv * dv
            delta[2 * i] += dv * dv
            delta[i + j] -= 2.0 * dv * dv
            total = c + delta
            mc_new = float(total.max())

            if mc_new < best_move_mc:
                best_move_mc = mc_new
                best_move = (i, j, dv, delta.copy())

        if best_move is None:
            no_improve += 1
            continue

        i, j, dv, d = best_move
        f[i] -= dv
        f[j] += dv
        c = c + d
        move_count += 1
        no_improve = 0

        cur_score = 2.0 * n * best_move_mc
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Rebuild f_units periodically to avoid drift
        if move_count % 100 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            mc = float(c.max())
            for i in range(n):
                f_units[i].fill(0.0)
                f_units[i][i:i + n] = f

    f = _normalize(f)
    c = _autocorr(f)
    sc = 2.0 * n * float(c.max())
    if sc < best_score:
        return f, sc
    return best_f, best_score


def _try_length(f_best, n_new, budget, rng):
    """Try optimizing at a different length."""
    f = _interp_to(f_best, n_new)
    return _gradient_optimize(f, budget, rng)


def _parametric_seed(n, rng):
    """Generate an asymmetric smooth parametric seed."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    mu = float(rng.uniform(0.3, 0.7))
    sigma = float(rng.uniform(0.1, 0.35))
    asym = float(rng.uniform(-1.5, 1.5))
    f = np.exp(-((x - mu) / sigma) ** 2 + asym * (x - mu))
    # Add small noise
    f *= np.exp(0.05 * rng.standard_normal(n))
    return _normalize(f)


def _smoothed_seed(f, n, rng):
    """Create smoothed version with noise."""
    f = _interp_to(f, n)
    sigma = float(rng.uniform(0.02, 0.2))
    f = f * np.exp(sigma * rng.standard_normal(n))
    return _normalize(f)
