Step function optimizer using softmax-annealed mirror descent with LP polishing.
Key innovations:
1. Softmax temperature annealing for smooth max approximation
2. Backtracking line search for guaranteed monotone improvement
3. LP-based direction finding for final polishing
4. Multi-phase approach with adaptive time allocation

import numpy as np
from scipy.optimize import linprog
import time

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_grad(f, c, T):
    """Gradient of softmax objective (1/T)*log(sum(exp(T*c_k))) w.r.t. f."""
    # d/dc_k of (1/T)*log(sum(exp(T*c_j))) = exp(T*c_k) / sum(exp(T*c_j))
    # c_k = conv(f,f)[k], dc_k/df_j = 2*f[k-j] (valid)
    # So grad[j] = 2 * sum_k w_k * f[k-j] where w_k = softmax(T*c)
    
    logits = T * (c - c.max())  # for numerical stability
    weights = np.exp(logits)
    weights /= weights.sum()
    
    n = len(f)
    grad = np.zeros(n)
    # Only use lags with significant weight
    thresh = weights.max() * 1e-6
    active = np.where(weights > thresh)[0]
    
    for k in active:
        w = weights[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    
    return grad

def _mirror_step(f, grad, step):
    """Multiplicative update: f[j] *= exp(-step * grad[j]), then normalize."""
    log_f = np.log(np.maximum(f, 1e-300)) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def _optimize_softmax_mirror(f0, n, deadline, rng, T_init=50, T_final=500, max_iters=100000):
    """
    Softmax-annealed mirror descent with backtracking.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    local_best = _score_fast(f)
    local_best_f = f.copy()
    
    step = 0.5
    patience = 0
    max_patience = 20
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Annealing temperature
        progress = min(1.0, it / max_iters)
        T = T_init + (T_final - T_init) * progress**0.7
        
        # Compute current convolution
        c = _autocorr_fft(f)
        peak_val = c.max()
        
        # Compute softmax gradient
        grad = _softmax_grad(f, c, T)
        
        # Adaptive step: scale by inverse gradient magnitude
        grad_scale = np.abs(grad).max()
        if grad_scale > 1e-30:
            step_eff = step / grad_scale
        else:
            step_eff = step
        
        # Mirror step
        f_new = _mirror_step(f, grad, step_eff)
        
        # Evaluate and backtrack if needed
        score_new = _score_fast(f_new)
        if score_new < local_best:
            f = f_new

                local_best = score_new
                local_best_f = f_new.copy()
                patience = 0

        # Periodic convergence check

def _perturb(f, rng, n, strength=0.1):
    """Random perturbation to escape local minima."""
    f_try = f * (1.0 + strength * rng.standard_normal(n))
    f_try = np.maximum(f_try, 0)
    s = f_try.sum()
    if s < 1e-10:
        return f.copy()
    return f_try / s

def _lp_direction(f, M, active_lags, n):

            b_end = min(n - 1, k)
            # Wait, let me redo this.
            # (f*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * b[k-j]

            # = sum_{i': 0<=i'<n, 0<=k-i'<n} f[k-i'] * b[i']
            # So A_ub[i, i'] = f[k-i'] if 0 <= k-i' < n, else 0

def _polish_subgradient(f, n, deadline, rng, max_iters=20000):
    """
    Subgradient polishing: at each step, find the argmax lag and
    take a small step in the negative subgradient direction.
    """
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    
    local_best = _score_fast(f)
    local_best_f = f.copy()
    
    step = 0.005
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        peak_val = c.max()
        
        # Find all lags within epsilon of peak
        eps = peak_val * 0.001
        near_peak = np.where(c >= peak_val - eps)[0]
        
        # Average subgradient over near-peak lags
        grad = np.zeros(n)
        for k in near_peak:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += f[idx]
        grad /= max(1, len(near_peak))
        grad *= 2.0  # factor of 2 from the quadratic
        
        # Mirror step
        f_new = _mirror_step(f, grad, step)
        
        score_new = _score_fast(f_new)
        if score_new < local_best:
            f = f_new
            if score_new < local_best - 1e-10:
                local_best = score_new
                local_best_f = f_new.copy()
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.95
            if step < 1e-6:
                # Try perturbation
                f = _perturb(f, rng, n, strength=0.02)
                step = 0.005
    
    return local_best_f, local_best

def _two_point_exchange(f, n, deadline, rng, max_iters=100000):

            contrib[j_start:j_end + 1] = f[idx]

Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-temperature mirror descent with FFT-based gradients,
Armijo backtracking, multi-phase strategy, and fine subgradient polishing.

Key innovations over previous approach:
1. FFT-based gradient for full softmax-weighted objective (smoother than top-K)
2. Temperature annealing for progressive peak sharpening
3. Armijo backtracking for guaranteed monotone improvement
4. Multi-phase: coarse softmax -> peak-focused subgradient -> two-point exchange
5. Periodic spectral analysis to guide perturbation direction

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_grad_fft(f, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * f[k-j] for j=0..n-1 using FFT.
    w has length 2n-1 (the convolution weights).
    """
    n = len(f)
    size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(W * np.conj(F), size)[:n]
    return 2.0 * result


def _softmax_weights(c, T):
    """Compute softmax weights over convolution values."""
    shifted = T * (c - c.max())
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        # All equal weight
        return np.ones(len(c)) / len(c)
    return exp_vals / total


def _mirror_update(f, grad, step):
    """Multiplicative (mirror) update: f[j] *= exp(-step * grad[j]), normalize."""
    log_f = np.log(np.maximum(f, 1e-300)) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

        f_try = f * np.maximum(w, 0.1)
    elif mode == 3:
        c1, c2 = sorted(rng.uniform(0.15, 0.85, size=2))
        bw = rng.uniform(0.05, 0.15)
        boost = np.exp(-((j / n - c1) / bw) ** 2) + np.exp(-((j / n - c2) / bw) ** 2)
        f_try = f * (1.0 + 0.2 * boost)
    elif mode == 4:
        f_try = f + rng.normal(0, 0.02 * f.mean(), size=n)
    else:
        freq1 = int(rng.integers(1, max(2, n // 5) + 1))
        freq2 = int(rng.integers(freq1 + 1, max(freq1 + 2, n // 3) + 1))
        p1, p2 = rng.uniform(0, 2 * np.pi, size=2)
        a1, a2 = rng.uniform(0.05, 0.2, size=2)
        f_try = f * (1.0 + a1 * np.sin(2 * np.pi * freq1 * j / n + p1) +
                     a2 * np.sin(2 * np.pi * freq2 * j / n + p2))

    f_try = np.maximum(f_try, 0.0)
    s = f_try.sum()
    if s < 1e-10:
        return f.copy()
    return f_try / s


def _optimize_softmax(f0, n, deadline, rng, T_init=50.0, T_end=800.0,
                      step_init=0.5, max_iters=200000):
    """
    Phase 1: Softmax-temperature mirror descent with backtracking.
    Minimizes the softmax-smoothed max of convolution.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    step = step_init
    best_step = step
    patience = 0
    max_patience = 30
    stagnation_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        # Temperature annealing: start smooth, end sharp
        T = T_init + (T_end - T_init) * (progress ** 0.5)

        # Compute convolution
        c = _autocorr_fft(f)

        # Compute softmax weights
        w = _softmax_weights(c, T)

        # Compute gradient via FFT
        grad = _softmax_grad_fft(f, w)

        # Scale gradient for step size adaptation
        grad_norm = np.sqrt(np.mean(grad ** 2))
        if grad_norm < 1e-30:
            break

        # Adaptive step: normalize gradient, use step as learning rate
        grad_scaled = grad / grad_norm
        step_eff = step * grad_norm * 0.1

        # Mirror update
        f_new = _mirror_update(f, grad, step_eff)

        # Evaluate
        score_new = _score_fast(f_new)

        if score_new < local_best - 1e-12:
            f = f_new
            local_best = score_new
            local_best_f = f_new.copy()
            patience = 0
            stagnation_count = 0
            # Increase step slightly
            step = min(step * 1.02, 5.0)
            best_step = step
        elif score_new < local_best + 1e-10:
            # Accept neutral moves sometimes (for exploration)
            if rng.random() < 0.1:
                f = f_new
            patience += 1
        else:
            patience += 1
            # Backtrack: reduce step
            step *= 0.8

        # Stagnation handling
        if patience >= max_patience:
            stagnation_count += 1
            patience = 0
            step = best_step * 0.5

            if stagnation_count <= 3:
                # Mild perturbation
                f = _perturb_structured(f, rng, mode=stagnation_count - 1)
            elif stagnation_count <= 6:
                # Moderate perturbation
                f = _perturb_structured(f, rng, mode=int(rng.integers(0, 5)))
                step = 0.3
            else:
                # Aggressive reset toward uniform with noise
                f = 0.7 * f + 0.3 * (np.ones(n) / n)
                f = f + rng.normal(0, 0.01, size=n)
                f = np.maximum(f, 0)
                f = f / f.sum()
                step = 0.3
                stagnation_count = 0

    return local_best_f, local_best


def _optimize_peak_focused(f0, n, deadline, rng, max_iters=200000):
    """
    Phase 2: Focused subgradient on peak lags with narrow bandwidth.
    Uses exact max (not softmax) with averaging over near-peak lags.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    step = 0.02
    patience = 0
    max_patience = 50

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute convolution
        c = _autocorr_fft(f)
        peak_val = c.max()

        # Find near-peak lags (within 0.1% of peak)
        threshold = peak_val * (1.0 - 0.001)
        near_peak = np.where(c >= threshold)[0]

        # Weight: emphasize the top few most
        # Use exponential decay from peak value
        c_near = c[near_peak]
        weights = np.exp(-100 * (peak_val - c_near) / peak_val)
        weights = weights / weights.sum()

        # Compute gradient: weighted average of subgradients
        # grad[j] = 2 * sum_k w_k * f[k-j]
        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[near_peak] = weights

        grad = _softmax_grad_fft(f, w_full)

        # Scale and update
        grad_norm = np.sqrt(np.mean(grad ** 2))
        if grad_norm < 1e-30:
            break

        step_eff = step * grad_norm * 0.05
        f_new = _mirror_update(f, grad, step_eff)

        score_new = _score_fast(f_new)

        if score_new < local_best - 1e-13:
            f = f_new
            local_best = score_new
            local_best_f = f_new.copy()
            patience = 0
            step = min(step * 1.005, 0.5)
        else:
            patience += 1
            step *= 0.95
            if patience > max_patience:
                # Small perturbation
                f = f * (1.0 + 0.01 * rng.standard_normal(n))
                f = np.maximum(f, 0)
                f = f / f.sum()
                patience = 0
                step = 0.02

    return local_best_f, local_best


def _two_point_polish(f0, n, deadline, rng, max_iters=500000):
    """
    Phase 3: Two-point exchange polish.
    Find pairs (donor, receiver) that reduce the peak convolution.
    Uses FFT for fast evaluation.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()

    # Contribution vector for the peak lag
    peak_lag = int(c.argmax())

    def get_contrib(k, f_arr):
        """Contribution of each position f[j] to conv(f,f)[k]."""
        n_arr = len(f_arr)
        contrib = np.zeros(n_arr)
        j_start = max(0, k - n_arr + 1)
        j_end = min(n_arr - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            contrib[j_start:j_end + 1] = f_arr[idx]
        return contrib

    contrib = get_contrib(peak_lag, f)

    # Score each position: high f * high contrib = "important for peak"
    importance = f * contrib

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodically recompute
        if it % 200 == 0:
            c = _autocorr_fft(f)
            peak = c.max()
            peak_lag = int(c.argmax())
            contrib = get_contrib(peak_lag, f)
            importance = f * contrib

        # Find top donors (high importance) and bottom receivers (low contrib)
        # Use sorted indices for efficiency
        top_k = min(10, n // 10)
        donor_candidates = np.argsort(importance)[-top_k:][::-1]
        receiver_candidates = np.argsort(contrib)[:top_k:]

        improved = False
        for d in donor_candidates:
            if f[d] < 1e-12:
                continue
            for r in receiver_candidates:
                if d == r:
                    continue
                # Try a few fractions
                for frac in [0.02, 0.05, 0.1]:
                    amt = f[d] * frac
                    if amt < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[d] -= amt
                    f_try[r] += amt
                    f_try = np.maximum(f_try, 0)
                    s_try = f_try.sum()
                    if s_try < 1e-10:
                        continue
                    f_try /= s_try

                    c_try = _autocorr_fft(f_try)
                    peak_try = c_try.max()

                    if peak_try < peak * (1 - 1e-10):
                        f = f_try
                        c = c_try
                        peak = peak_try
                        peak_lag = int(c.argmax())
                        contrib = get_contrib(peak_lag, f)
                        importance = f * contrib
                        score = _score_fast(f)
                        if score < local_best:
                            local_best = score
                            local_best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random moves
            if it % 500 == 0 and it > 100:
                for _ in range(3):
                    d = int(rng.integers(0, n))
                    r = int(rng.integers(0, n))
                    if d == r:
                        continue
                    amt = f[d] * 0.03
                    f_try = f.copy()
                    f_try[d] -= amt
                    f_try[r] += amt
                    f_try = np.maximum(f_try, 0)
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak * (1 - 1e-10):
                        f = f_try
                        c = c_try
                        peak = c.max()
                        peak_lag = int(c.argmax())
                        contrib = get_contrib(peak_lag, f)
                        importance = f * contrib
                        score = _score_fast(f)
                        if score < local_best:
                            local_best = score
                            local_best_f = f.copy()
                        break

                if it % 5000 == 0:
                    # Larger perturbation
                    f = _perturb_structured(f, rng, mode=int(rng.integers(0, 3)))
                    c = _autocorr_fft(f)
                    peak = c.max()
                    peak_lag = int(c.argmax())
                    contrib = get_contrib(peak_lag, f)
                    importance = f * contrib

    return local_best_f, local_best


def _spectral_guided_search(f0, n, deadline, rng, max_iters=100000):
    """
    Alternative phase: Optimize in a way that flattens the power spectrum.
    The idea: a flat power spectrum means the autocorrelation is close to
    a delta function, which minimizes the peak (aside from the lag-0 term).
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    step = 0.1
    patience = 0

    # Target: flat power spectrum (excluding DC and the trivially high low frequencies)
    size = 1 << (2 * n - 1).bit_length()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute power spectrum
        F = np.fft.rfft(f, size)
        power = np.abs(F) ** 2

        # Focus on mid-high frequencies where non-flatness hurts
        # Exclude DC (index 0) and very low frequencies
        start_bin = max(1, n // (4 * n))  # skip a few low bins
        mid_power = power[start_bin:min(n, size // 2)]

        if len(mid_power) == 0:
            break

        # Gradient of variance of power spectrum w.r.t. f
        # d(power[k])/d(f[j]) = 2 * f[j] * cos(2*pi*k*j/size) (for real f)
        # This is expensive to compute directly, so use a proxy:
        # Penalize positions where f has high correlation with its own Fourier basis

        # Simpler approach: compute the "spectral gradient" via FFT
        # The gradient of sum_k (power[k] - mean)^2 w.r.t. f is related to
        # the time-domain "autocorrelation" of the power spectrum deviation

        # Practical approach: use the convolution peak gradient as before
        # but add a spectral regularizer
        c = _autocorr_fft(f)
        peak_val = c.max()
        near_peak = np.where(c >= peak_val * 0.999)[0]

        w_full = np.zeros(2 * n - 1)
        c_near = c[near_peak]
        weights = np.exp(-50 * (peak_val - c_near) / max(peak_val, 1e-30))
        weights = weights / weights.sum()
        w_full[near_peak] = weights

        grad = _softmax_grad_fft(f, w_full)

        # Add spectral regularizer: penalty on power variance
        # Approximate gradient: for each frequency bin, the "excess power"
        # tells us which time-domain patterns to suppress
        mean_p = mid_power.mean()
        if mean_p > 1e-30:
            excess = (mid_power - mean_p) / mean_p
            # Gradient contribution: sum_k excess[k] * 2*f[j]*cos(2*pi*k*j/size)
            # = 2 * Re(ifft(excess_padded * conj(F)))... approximately
            # Simplified: just use the convolution gradient with a small spectral weight
            reg_weight = 0.05
            # Compute a simple spectral penalty gradient
            # For flat spectrum, we want |F[k]|^2 to be constant
            # d/d(f[j]) |F[k]|^2 = 2 * Re(f[j] * conjugate(F[k]) * exp(-2*pi*i*k*j/size))
            # Sum over k with excess weight:
            # = 2 * Re(sum_k excess[k] * f[j] * conjugate(exp(2*pi*i*k*j/size)))
            # = 2 * f[j] * Re(sum_k excess[k] * exp(-2*pi*i*k*j/size))
            # = 2 * f[j] * ifft(excess)[j] (for appropriate indexing)

            excess_full = np.zeros(size // 2 + 1)
            excess_full[start_bin:start_bin + len(excess)] = excess

            # ifft of the excess spectrum gives the "pattern to suppress"
            pattern = np.fft.irfft(excess_full * size, size)[:n]
            grad += reg_weight * 2.0 * f * pattern

        # Mirror update
        grad_norm = np.sqrt(np.mean(grad ** 2))
        if grad_norm < 1e-30:
            break
        step_eff = step * grad_norm * 0.05
        f_new = _mirror_update(f, grad, step_eff)

        score_new = _score_fast(f_new)
        if score_new < local_best - 1e-12:
            f = f_new
            local_best = score_new
            local_best_f = f_new.copy()
            patience = 0
            step = min(step * 1.01, 1.0)
        else:
            patience += 1
            step *= 0.95
            if patience > 40:
                f = _perturb_structured(f, rng, mode=int(rng.integers(0, 4)))
                patience = 0
                step = 0.1

    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Softmax-temperature mirror descent (coarse optimization)
    Phase 2: Peak-focused subgradient (medium optimization)
    Phase 3: Spectral-guided search (alternative exploration)
    Phase 4: Two-point exchange (fine polish)
    Phase 5: Final micro-subgradient
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    # Load and normalize initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    s0 = f_init.sum()
    if s0 > 1e-10:
        f_init = f_init / s0
    else:
        f_init = np.ones(n)

    best_score = _score_fast(f_init)
    best_f = f_init.copy()
    print(f"Initial: n={n}, score={best_score:.8f}")

    # Try diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    start_candidates = {
        'given': f_init.copy(),
        'uniform': np.ones(n),
        'u_shape': 1.0 + 2.0 * ((j - center) / max(center, 1)) ** 2,
        'bimodal': np.exp(-((j / n - 0.3) / 0.12) ** 2) + np.exp(-((j / n - 0.7) / 0.12) ** 2) + 0.01,
        'asym': np.exp(-((j / n - 0.4) / 0.2) ** 2) + 0.01,
        'sine': 1.0 + 0.3 * np.sin(2 * np.pi * 3 * j / n),
        'random_exp': rng.exponential(1.0, size=n),
    }

    start_scores = []
    for name, f_s in start_candidates.items():
        f_s = np.maximum(f_s, 0)
        f_s = f_s / f_s.sum()
        sc = _score_fast(f_s)
        start_scores.append((sc, name, f_s))

    start_scores.sort()
    print("\nStart scores:")
    for sc, name, _ in start_scores[:5]:
        print(f"  {name}: {sc:.8f}")

    if start_scores[0][0] < best_score:
        best_score = start_scores[0][0]
        best_f = start_scores[0][2].copy()
        print(f"  -> Using '{start_scores[0][1]}' as best start")

    # === Phase 1: Softmax mirror descent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Softmax mirror descent (score={best_score:.8f})")
    print(f"{'='*60}")

    t_p1_end = t_start + (budget_s - 300) * 0.35
    f_opt, sc_opt = _optimize_softmax(best_f, n, t_p1_end, rng,
                                       T_init=30.0, T_end=600.0,
                                       step_init=0.3, max_iters=300000)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  Phase 1 improved to: {best_score:.8f}")
    else:
        print(f"  Phase 1 result: {sc_opt:.8f} (no improvement)")

    # === Phase 2: Peak-focused subgradient ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak-focused subgradient (score={best_score:.8f})")
    print(f"{'='*60}")

    t_p2_end = min(t_start + (budget_s - 200) * 0.55, deadline - 200)
    f_opt, sc_opt = _optimize_peak_focused(best_f, n, t_p2_end, rng, max_iters=300000)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  Phase 2 improved to: {best_score:.8f}")
    else:
        print(f"  Phase 2 result: {sc_opt:.8f}")

    # === Phase 3: Spectral-guided search ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral-guided (score={best_score:.8f})")
    print(f"{'='*60}")

    t_p3_end = min(t_start + (budget_s - 150) * 0.7, deadline - 150)
    # Start from a slightly perturbed version
    f_pert = _perturb_structured(best_f, rng, mode=1)
    f_opt, sc_opt = _spectral_guided_search(f_pert, n, t_p3_end, rng, max_iters=150000)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  Phase 3 improved to: {best_score:.8f}")
    else:
        print(f"  Phase 3 result: {sc_opt:.8f}")

    # === Phase 4: Additional softmax run from best ===
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*60}")
        print(f"Phase 4: Refined softmax (score={best_score:.8f})")
        print(f"{'='*60}")
        t_p4_end = min(t_start + (budget_s - 100) * 0.8, deadline - 100)
        f_opt, sc_opt = _optimize_softmax(best_f, n, t_p4_end, rng,
                                           T_init=200.0, T_end=2000.0,
                                           step_init=0.1, max_iters=200000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Phase 4 improved to: {best_score:.8f}")
        else:
            print(f"  Phase 4 result: {sc_opt:.8f}")

    # === Phase 5: Two-point exchange polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Two-point exchange (score={best_score:.8f})")
    print(f"{'='*60}")

    t_p5_end = min(deadline - 60, t_start + (budget_s - 50) * 0.92)
    f_opt, sc_opt = _two_point_polish(best_f, n, t_p5_end, rng, max_iters=1000000)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  Phase 5 improved to: {best_score:.8f}")
    else:
        print(f"  Phase 5 result: {sc_opt:.8f}")

    # === Phase 6: Final micro-subgradient ===
    if time.monotonic() < deadline - 15:
        print(f"\n{'='*60}")
        print(f"Phase 6: Final micro-polish (score={best_score:.8f})")
        print(f"{'='*60}")
        f_opt, sc_opt = _optimize_peak_focused(best_f, n, deadline - 5, rng,
                                                 max_iters=50000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Final improved to: {best_score:.8f}")
        else:
            print(f"  Final result: {sc_opt:.8f}")

    # Prepare final output
    best_f = np.clip(best_f, 0.0, 1000.0)
    s_final = best_f.sum()
    if s_final < 1.0:
        best_f = best_f * (2.0 / s_final)
    elif s_final > 1000.0:
        best_f = best_f * (500.0 / s_final)

    # Official validation
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\n{'='*60}")
    print(f"Official evaluation: {final_score:.8f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    if not np.isfinite(final_score):
        # Fallback to initial
        best_f = f_init * 1.0
        final_score = evaluate_sequence([float(x) for x in best_f])
        print(f"Fallback score: {final_score:.8f}")

    return [float(x) for x in best_f]
