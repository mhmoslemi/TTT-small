    from scipy.optimize import linprog
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):

                    rows.append(k)

                    vals.append(f[i])

Step function optimizer using LP-based active set method with

from scipy.signal import fftconvolve
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _score_fast(f):
    """Fast score: 2*n*max(conv(f,f))/(sum(f))^2"""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf

def _coordinate_polish(f, deadline, rng):
    """Coordinate descent: try transferring mass between pairs."""
    n = len(f)
    f = _normalize(f)
    best_score = _score_fast(f)
    best_f = f.copy()
    
    c = _autocorr_fft(f)
    peak = c.max()
    
    # Focus on the top peaks
    top_idx = np.argpartition(c, -3)[-3:]
    
    passes = 0
    while time.monotonic() < deadline:
        passes += 1
        improved = False
        
        # For each of the top peaks, find the best coordinate transfer
        for k_star in top_idx:
            if time.monotonic() > deadline:
                break
            
            # Contribution of each position to peak at k_star
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(n-1, k_star)
            if j_start <= j_end:
                idx_f = k_star - np.arange(j_start, j_end+1)
                contrib[j_start:j_end+1] = f[j_start:j_end+1] * f[idx_f]
            
            # Also compute total contribution across all top peaks
            total_contrib = np.zeros(n)
            for k in top_idx:
                js = max(0, k - n + 1)
                je = min(n-1, k)
                if js <= je:
                    idx_k = k - np.arange(js, je+1)
                    total_contrib[js:je+1] += f[js:je+1] * f[idx_k]
            
            # Best donor: highest total contribution
            j_don = int(np.argmax(total_contrib))
            # Best receiver: lowest total contribution (and not the donor)
            total_contrib[j_don] = np.inf
            j_rec = int(np.argmin(total_contrib))
            
            if total_contrib[j_don] < 1e-15:
                continue
            
            # Try several transfer fractions
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                amt = f[j_don] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    score = _score_fast(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    improved = True
                    break
        
        if not improved and passes > 100:
            # Random coordinate moves
            for _ in range(10):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                amt = f[i] * 0.1
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    score = _score_fast(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    break
        
        if not improved and passes > 500:
            break
    
    return best_f, best_score

def _structured_perturb(f, rng, mode=0):
    """Apply structured perturbation."""
    n = len(f)
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()
    
    if mode == 0:
        alpha = rng.uniform(-1.0, 1.0)
        tilt = np.exp(alpha * (j/n - 0.5))
        f_try = f * tilt
    elif mode == 1:
        freq = int(rng.integers(1, max(2, n//4) + 1))
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.2, 0.8)
        f_try = f * (1.0 + amp * np.sin(2*np.pi*freq*j/n + phase))
    elif mode == 2:
        # Noise
        f_try = f + rng.normal(0, 0.1 * f.mean(), n)
    elif mode == 3:
        # Bimodal
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        bw = rng.uniform(0.05, 0.2)
        boost = np.exp(-((j/n - c1)/bw)**2) + np.exp(-((j/n - c2)/bw)**2)
        f_try = f * (1.0 + 0.5 * boost)
    elif mode == 4:
        # Frequency domain perturbation
        size = 1 << (2*n-1).bit_length()
        F = np.fft.rfft(f, size)
        # Randomize phases slightly
        phases = np.angle(F)
        mags = np.abs(F)
        phase_noise = rng.normal(0, 0.1, mags.shape)
        F_new = mags * np.exp(1j*(phases + phase_noise))
        f_try = np.fft.irfft(F_new, size)[:n]
    else:
        f_try = f * (1.0 + 0.1 * rng.chisquare(n) / n)
    
    return _normalize(f_try)
