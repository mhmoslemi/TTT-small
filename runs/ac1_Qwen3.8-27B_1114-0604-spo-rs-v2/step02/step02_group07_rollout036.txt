def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    result = np.fft.irfft(A * A, M_fft)[:M]
    return result

def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def gradient_topk(a, c, topk, temp_scale=0.05):
    """Compute gradient using top-k peaks with softmax."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    
    # Softmax weights
    temp = max(temp_scale * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts = wts / wts.sum()
    
    M = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S

            if log_space:
