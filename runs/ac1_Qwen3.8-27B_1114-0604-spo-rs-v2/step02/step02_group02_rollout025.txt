"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2 using soft-max subgradient
with Adam optimization, variance-flattening gradient, diverse restarts, SA exploration,
and coordinate descent polishing. Key improvements over parent: adaptive soft-max beta,
combined peak+variance gradient, Adam with simplex projection, and structured
initialization from binomial-inspired profiles.
"""

import numpy as np
import time


def _convolve_full(f):
    """Compute full self-convolution of f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = (u - cssv / ind) > 0
    i = np.where(cond)[0]
    if len(i) == 0:
        return np.ones(n) / n
    i = i[-1]
    rho = (cssv[i] - 1.0) / (i + 1.0)
    return np.maximum(v - rho, 0.0)


def _grad_soft_peak(f, h, beta, top_k=30):
    """Compute soft-max weighted subgradient for the peak of h."""
    n = len(f)
    if top_k >= len(h):
        idx = np.arange(len(h))
        vals = h.copy()
    else:
        idx = np.argpartition(h, -top_k)[-top_k:]
        vals = h[idx]
    # Soft-max weights
    shifted = vals - vals.max()
    weights = np.exp(beta * shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(len(idx))
        weights[-1] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        if w < 1e-15:
            continue
        k = int(k)
        # First term: f[k-i] for valid i
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if hi1 > lo1:
            j1 = k - np.arange(lo1, hi1)
            grad[lo1:hi1] += w * f[j1]
        # Second term: f[i-k] for valid i
        lo2 = max(0, k)
        hi2 = n
        if hi2 > lo2:
            j2 = np.arange(lo2, hi2) - k
            j2_valid = j2[j2 < n]
            i_valid = np.arange(lo2, hi2)[j2 < n]
            if len(j2_valid) > 0:
                grad[i_valid] += w * f[j2_valid]
    return grad


def _grad_variance(f, h):
    """
    Compute gradient of sum_k h[k]^2 (variance-flattening gradient).
    grad[i] = 2 * sum_k h[k] * (f[k-i] + f[i-k])
    Computed via FFT for efficiency.
    """
    n = len(f)
    # grad1[i] = sum_k h[k] * f[k-i] (cross-correlation of h and f)
    # grad2[i] = sum_k h[k] * f[i-k] (convolution of h and f)
    size = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, size)
    F = np.fft.rfft(f, size)

    # Cross-correlation: IFFT(H * conj(F))
    corr = np.fft.irfft(H * np.conj(F), size)[:n]
    # Convolution: IFFT(H * F)
    conv = np.fft.irfft(H * F, size)[:n]

    return 2.0 * (corr + conv)


def _optimize_adam(f0, deadline, rng, n, verbose=True):
    """
    Adam optimizer on the simplex with soft-max peak gradient + variance gradient.
    Uses adaptive beta, SA-like restarts, and backtracking.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    h = _convolve_full(f)
    cur_peak = h.max()
    local_best = cur_peak
    local_best_f = f.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Soft-max beta (adaptive)
    beta_sm = 5.0
    beta_sm_max = 200.0
    beta_sm_min = 2.0

    # Mixing weight between peak grad and variance grad
    alpha_mix = 0.7

    # SA parameters
    T = cur_peak * 0.02
    T_floor = 1e-16
    last_improve = time.monotonic()
    stall_count = 0

    if verbose:
        print(f"    Start: peak={cur_peak:.10f}, score={2*n*cur_peak:.6f}")

    while time.monotonic() < deadline:
        # Check time
        if time.monotonic() >= deadline:
            break

        h = _convolve_full(f)
        cur_peak = h.max()

        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()
            last_improve = time.monotonic()
            stall_count = 0
            # Slightly increase beta to focus more on the peak
            beta_sm = min(beta_sm * 1.02, beta_sm_max)
        else:
            stall_count += 1

        # Adaptive beta: increase when stuck less, decrease for exploration
        stuck_time = time.monotonic() - last_improve
        if stuck_time > 30:
            # Reheat
            T = cur_peak * 0.05
            beta_sm = max(beta_sm_min, beta_sm * 0.9)
            alpha_mix = 0.4  # More variance gradient for exploration
            # Perturb
            mag = min(0.03, 0.005 + 0.002 * (stuck_time / 60.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f = _project_simplex(f)
            last_improve = time.monotonic()
        else:
            T = max(T_floor, T * 0.9998)
            # Gradually focus more on peak
            beta_sm = min(beta_sm_max, beta_sm * 1.001)
            alpha_mix = min(0.9, alpha_mix + 0.001)

        # Compute gradients
        grad_peak = _grad_soft_peak(f, h, beta_sm, top_k=20)
        grad_var = _grad_variance(f, h)

        # Normalize gradients for balance
        gp_norm = np.sqrt(np.mean(grad_peak ** 2)) + 1e-30
        gv_norm = np.sqrt(np.mean(grad_var ** 2)) + 1e-30
        grad_peak_norm = grad_peak / gp_norm
        grad_var_norm = grad_var / gv_norm

        grad = alpha_mix * grad_peak_norm + (1.0 - alpha_mix) * grad_var_norm

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_try = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_try = np.maximum(f_try, 0.0)
        f_try = _project_simplex(f_try)

        # Evaluate new point
        h_try = _convolve_full(f_try)
        peak_try = h_try.max()

        # Accept/reject with SA
        delta = peak_try - cur_peak
        if delta <= 0 or (T > 1e-30 and np.log(max(rng.uniform(), 1e-300)) < -delta / T):
            f = f_try
            if delta <= 0:
                lr = min(lr * 1.01, 0.1)
            else:
                lr *= 0.95
        else:
            lr *= 0.9

        # Periodic logging
        if t % 50000 == 0 and verbose:
            print(f"    Adam iter {t}: peak={cur_peak:.10f}, best={local_best:.10f}, "
                  f"lr={lr:.5f}, beta={beta_sm:.1f}, alpha={alpha_mix:.2f}, T={T:.2e}")

    # Final check
    h = _convolve_full(f)
    if h.max() < local_best:
        local_best = h.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _polish_coordinate(f, deadline, rng, n):
    """
    Coordinate descent polishing: try moving mass between coordinate pairs
    to reduce the peak convergence.
    """
    f = f.copy()
    f = np.maximum(f, 0.0)
    f = _project_simplex(f)

    h = _convolve_full(f)
    local_best = h.max()
    local_best_f = f.copy()
    passes = 0

    while time.monotonic() < deadline:
        h = _convolve_full(f)
        cur_peak = h.max()
        k_star = int(np.argmax(h))

        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()

        # Contribution of each coordinate to h[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        j_idx = k_star - np.arange(lo, hi)
        valid = (j_idx >= 0) & (j_idx < n)
        contrib[lo:hi][valid] = f[j_idx[valid]]
        # Second term
        lo2 = max(0, k_star)
        if lo2 < n:
            j2 = np.arange(lo2, n) - k_star
            contrib[lo2:] += f[j2]

        # Try moving from high-contrib to low-contrib coordinates
        improved = False
        donor = int(np.argmax(contrib))
        low_indices = np.argsort(contrib)[:20]

        for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
            for recv in low_indices[:10]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try = _project_simplex(f_try)
                h_try = _convolve_full(f_try)
                if h_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break
            if improved:
                break

        # Random pair moves
        if not improved:
            for _ in range(50):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = float(rng.choice([0.01, 0.03, 0.08, 0.15, 0.3]))
                src = i if f[i] >= f[j] else j
                dst = j if src == i else i
                amt = f[src] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[src] -= amt
                f_try[dst] += amt
                f_try = _project_simplex(f_try)
                h_try = _convolve_full(f_try)
                if h_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break

        # Small global perturbation
        if not improved and passes % 100 == 50:
            f_try = f + 0.003 / np.sqrt(n) * rng.standard_normal(n)
            f_try = _project_simplex(f_try)
            h_try = _convolve_full(f_try)
            if h_try.max() < cur_peak - 1e-15:
                f = f_try

        passes += 1

    h = _convolve_full(f)
    if h.max() < local_best:
        local_best = h.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _generate_starts(f_init, n, rng):
    """Generate diverse starting points for optimization."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # U-shaped profiles
    for alpha in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]:
        s = 1.0 + alpha * ((j - center) / center) ** 2
        starts.append(s / s.sum())

    # Inverted U (bell)
    for alpha in [1.0, 3.0, 5.0, 10.0, 20.0]:
        s = np.exp(-alpha * ((j - center) / center) ** 2)
        starts.append(s / s.sum())

    # Bimodal
    for pos in [0.2, 0.25, 0.3, 0.35, 0.4]:
        s = (np.exp(-((j - pos * n) / (0.05 * n)) ** 2) +
             np.exp(-((j - (1 - pos) * n) / (0.05 * n)) ** 2))
        s = np.maximum(s, 1e-10)
        starts.append(s / s.sum())

    # Cosine-based
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            s = 1.0 + amp * np.cos(2 * np.pi * freq * j / n)
            s = np.maximum(s, 1e-10)
            starts.append(s / s.sum())

    # Sine absolute
    for freq in [1, 2, 4, 8]:
        s = 1.0 + 0.5 * np.abs(np.sin(np.pi * freq * j / n))
        starts.append(s / s.sum())

    # Power law edges
    for p in [0.5, 1.0, 2.0, 4.0]:
        dist = np.maximum(np.abs(j - center) / center, 1e-10)
        s = dist ** p
        s = np.maximum(s, 1e-10)
        starts.append(s / s.sum())

    # Random starts
    for _ in range(15):
        r = rng.random()
        if r < 0.25:
            s = rng.exponential(1.0, n)
        elif r < 0.5:
            s = np.abs(rng.standard_normal(n)) + 0.1
        elif r < 0.75:
            s = rng.dirichlet(np.ones(n) * 0.1)
        else:
            # Sparse-ish
            base = np.zeros(n)
            n_nz = max(10, n // 4)
            idx = rng.choice(n, n_nz, replace=False)
            base[idx] = rng.exponential(1.0, n_nz)
            s = base + 1e-10
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss > 1e-10:
            starts.append(s / ss)

    # Perturbed versions of f_init
    for mag in [0.05, 0.1, 0.2]:
        s = f_init + mag / np.sqrt(n) * rng.standard_normal(n)
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss > 1e-10:
            starts.append(s / ss)

    return starts
