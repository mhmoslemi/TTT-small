"""
Low-peak step-function optimization using multi-phase adaptive gradient descent
with softmax-temperature annealing, peak-targeted refinement, and coordinate descent.
Starts from height_sequence_1 if available, applies multiple optimization strategies.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_quadratic(n, width=0.8):
    i = np.arange(n, dtype=np.float64)
    x = (i - n / 2) / (n / 2)
    a = np.exp(-width ** 2 * x ** 2)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _softmax_weights(c, T):
    c_max = float(np.max(c))
    c_shift = c - c_max
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / s


def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _optimize_mirror(a, deadline, T_start=3.0, T_end=200.0, lr_factor=0.25,
                     patience_base=6000, eval_every=8, seed_override=0):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    lr0 = lr_factor * n
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = patience_base
    rng = np.random.RandomState(1000 + seed_override + n)

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 40000.0)
        T = T_start + (T_end - T_start) * t_frac ** 0.7
        lr = lr0 / (1.0 + 0.001 * it)

        c = _fft_convolve(a)
        if c.size == 0:

        w = _softmax_weights(c, T)
        grad = _grad_via_fft(w, a)

        update = -lr * grad_proj
        update = np.clip(update, -4.0, 4.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)

def _optimize_peak_target(a, deadline, K=6, T_local=4.0, lr_factor=0.2,
                          patience_base=4000, eval_every=5, seed_override=0):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    lr0 = lr_factor * n
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = patience_base
    rng = np.random.RandomState(2000 + seed_override + n)

    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.002 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i_idx, k in enumerate(top_idx):
            w[k] = weights[i_idx]

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr * grad_proj
        update = np.clip(update, -4.0, 4.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)

        if it % eval_every == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.008 * a.mean()
                a = a + noise
                a = _normalize(a)
                patience = min(30000, int(patience * 1.3))
                lr0 = max(0.03 * n, lr0 * 0.9)

    return best_a, best_score


def _coordinate_descent(a, deadline, search_radius=0.05, eval_every=20, seed_override=0):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(3000 + seed_override + n)
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1
        i = rng.randint(n)
        base = a[i]
        mean_val = a.mean()

        candidates = [
            base * (1 + search_radius),
            base * (1 - search_radius * 0.5),
            base + search_radius * mean_val,
            base - search_radius * mean_val * 0.5,
            base * (1 + search_radius * 0.3),
            max(0.0, base - search_radius * mean_val),
        ]

        for new_val in candidates:
            if new_val < 0 or new_val > 1000:
                continue
            new_a = a.copy()
            new_a[i] = new_val
            new_a = _normalize(new_a)
            sc = _score(new_a)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                a = new_a
                best_score = sc
                best_a = a.copy()
                last_improve = it
                break

        if it % eval_every == 0 and it - last_improve > 3000:
            break

    return best_a, best_score


def _adaptive_temperature_run(a, deadline, seed_override=0):
    """Run with progressively changing temperature in segments."""
    n = a.size
    segment_time = 120.0
    T_schedule = [2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 400.0]
    lr_schedule = [0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.04]

    best_a = a.copy()
    best_score = _score(best_a)
    current_a = a.copy()

    for seg_idx in range(len(T_schedule)):
        seg_deadline = min(deadline, time.monotonic() + segment_time)
        if time.monotonic() > seg_deadline - 5:
            break
        T = T_schedule[seg_idx]
        lr_f = lr_schedule[seg_idx]
        a_seg, s_seg = _optimize_mirror(current_a, seg_deadline,
                                         T_start=T, T_end=T * 1.5,
                                         lr_factor=lr_f, patience_base=3000,
                                         eval_every=5, seed_override=seed_override + seg_idx)
        if s_seg < best_score:
            best_score = s_seg
            best_a = a_seg.copy()
        current_a = a_seg.copy()
        print(f"    [AdaptT] seg {seg_idx}: T={T}, lr={lr_f}, score={s_seg:.8f}", flush=True)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Multi-phase optimization for low-peak nonnegative step function."""
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s, deadline in {hard_deadline - t0:.0f}s", flush=True)

    best_a = None
    best_score = np.inf

    # Get initial sequence from height_sequence_1
    try:
        hs1 = list(height_sequence_1)
        if hs1 and len(hs1) > 10:
            init_a = np.array([float(max(0.0, min(1000.0, x))) for x in hs1])
            s_init = init_a.sum()
            if s_init > 0.01:
                init_a = init_a / s_init
                sc = _score(init_a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = init_a.copy()
                print(f"[Init] height_sequence_1: n={len(init_a)}, score={sc:.8f}", flush=True)
    except NameError:
        pass
    except Exception as e:
        print(f"[Init] Error reading height_sequence_1: {e}", flush=True)

    # Also try other initializations
    if best_a is None or best_score > 2.0:
        for n_try in [1024, 2048, 512]:
            for amp in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2]:
                a_s = _make_sine(n_try, amp)
                sc = _score(a_s)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_s.copy()
                    print(f"[Init] sine n={n_try} amp={amp}: {sc:.8f}", flush=True)
            for wid in [0.5, 0.7, 1.0, 1.5]:
                a_q = _make_quadratic(n_try, wid)
                sc = _score(a_q)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_q.copy()
                    print(f"[Init] quad n={n_try} w={wid}: {sc:.8f}", flush=True)

    if best_a is None:
        best_a = _make_sine(1024, 0.7)
        best_score = _score(best_a)

    print(f"[Init] Best: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 1: Main mirror descent (most time)
    d1 = hard_deadline - 500.0
    remaining = d1 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 1] Mirror descent ({remaining:.0f}s)...", flush=True)
        a1, s1 = _optimize_mirror(best_a, d1, T_start=2.0, T_end=150.0,
                                   lr_factor=0.3, patience_base=8000,
                                   eval_every=8, seed_override=seed)
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        print(f"  Phase 1 done: {s1:.8f}", flush=True)

    # Phase 2: Peak-targeted refinement
    d2 = hard_deadline - 300.0
    remaining = d2 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 2] Peak-targeted ({remaining:.0f}s)...", flush=True)
        a2, s2 = _optimize_peak_target(best_a, d2, K=6, T_local=4.0,
                                        lr_factor=0.2, seed_override=seed)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        print(f"  Phase 2 done: {s2:.8f}", flush=True)

    # Phase 3: Coordinate descent
    d3 = hard_deadline - 200.0
    remaining = d3 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 3] Coordinate descent ({remaining:.0f}s)...", flush=True)
        a3, s3 = _coordinate_descent(best_a, d3, search_radius=0.04,
                                      eval_every=30, seed_override=seed)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
        print(f"  Phase 3 done: {s3:.8f}", flush=True)

    # Phase 4: Adaptive temperature run
    d4 = hard_deadline - 100.0
    remaining = d4 - time.monotonic()
    if remaining > 120:
        print(f"[Phase 4] Adaptive temperature ({remaining:.0f}s)...", flush=True)
        a4, s4 = _adaptive_temperature_run(best_a, d4, seed_override=seed)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
        print(f"  Phase 4 done: {s4:.8f}", flush=True)

    # Phase 5: Final peak-targeted polish
    d5 = hard_deadline - 30.0
    remaining = d5 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 5] Final polish ({remaining:.0f}s)...", flush=True)
        a5, s5 = _optimize_peak_target(best_a, d5, K=4, T_local=6.0,
                                        lr_factor=0.1, patience_base=3000,
                                        eval_every=4, seed_override=seed + 100)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
        print(f"  Phase 5 done: {s5:.8f}", flush=True)

    # Phase 6: One more mirror descent with very low T
    d6 = hard_deadline - 20.0
    remaining = d6 - time.monotonic()
    if remaining > 60:
        print(f"[Phase 6] Final mirror ({remaining:.0f}s)...", flush=True)
        a6, s6 = _optimize_mirror(best_a, d6, T_start=1.0, T_end=20.0,
                                   lr_factor=0.1, patience_base=3000,
                                   eval_every=5, seed_override=seed + 200)
        if s6 < best_score:
            best_score = s6
            best_a = a6.copy()
        print(f"  Phase 6 done: {s6:.8f}", flush=True)

    # Final validation
    print("[Validating with evaluate_sequence...]", flush=True)
    arr_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        actual = float(evaluate_sequence(arr_list))
    except Exception as e:
        print(f"  Validation error: {e}", flush=True)
        actual = best_score

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Score: {actual:.8f}, n={len(arr_list)}", flush=True)
    return arr_list
