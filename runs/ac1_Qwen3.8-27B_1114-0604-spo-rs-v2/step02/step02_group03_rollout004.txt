"""
Step-function optimizer combining Frank-Wolfe (conditional gradient) method
with gradient descent and exact coordinate polishing.

The key idea: at each FW iteration, solve an LP to find the optimal "vertex"
direction, which is more informative than a gradient step. This is combined
with gradient steps for continuation and coordinate polish for local refinement.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / v.size
    v /= s
    return v


def _fft_conv(f, g=None):
    """Full linear convolution using FFT."""
    if g is None:
        g = f
    n = f.size
    m = g.size
    L = n + m - 1
    size = 1 << (L - 1).bit_length()
    F = np.fft.rfft(f, size)
    G = np.fft.rfft(g, size)
    return np.fft.irfft(F * G, size)[:L]


def _score(f):
    """Compute 2*n*max(f*f)/(sum f)^2."""
    n = f.size
    c = _fft_conv(f)
    s = f.sum()
    if s < 1e-300:
        return np.inf, 0.0
    return float(2.0 * n * c.max() / (s * s)), float(c.max())


def _interp(f, n):
    """Linearly interpolate f to length n."""
    m = f.size
    if m == n:
        return f.copy()
    if n == 1:
        return np.array([f.sum()])
    if m == 1:
        return np.full(n, f[0])
    x0 = np.linspace(0, m - 1, m)
    x1 = np.linspace(0, n - 1, n)
    return np.interp(x1, x0, f)


def _build_conv_matrix(f):
    """Build the convolution constraint matrix A where (A @ g)[k] = (f*g)[k]."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _fw_step(f, M, tp=30):
    """
    Frank-Wolfe LP step. Returns g (sum-normalized) or None if LP fails.
    tp: time parameter (not used directly, for API compatibility)
    """
    n = f.size
    try:
        A = _build_conv_matrix(f)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b, method='highs',
                      options={'time_limit': tp, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _fw_step_reduced(f, M, K=2000, tp=15):
    """
    Reduced-size FW step: downsample, solve LP, upsample.
    """
    n = f.size
    if n <= K:
        return _fw_step(f, M, tp)
    f_k = _interp(f, K)
    f_k = _normalize(f_k)
    c_k = _fft_conv(f_k)
    M_k = float(c_k.max())
    g_k = _fw_step(f_k, M_k, tp)
    if g_k is not None:
        g = _interp(g_k, n)
        g = _normalize(np.maximum(g, 0))
        return g
    return None


def _weighted_grad(f, c, temp=0.05):
    """Compute softmax-weighted gradient direction."""
    n = f.size
    L = c.size
    M = c.max()
    if M < 1e-300:
        return np.zeros(n)
    z = (c - M) / (M * temp + 1e-300)
    np.clip(z, -50, 0, out=z)
    p = np.exp(z)
    ps = p.sum()
    if ps < 1e-300:
        p /= L
    else:
        p /= ps
    # Gradient: g[j] = 2 * sum_k p[k] * f[k-j] (valid indices)
    # This is 2 * (p * f_reversed)[n-1+j] ... let me just use FFT
    size = 1 << (L + n - 2).bit_length()
    P = np.fft.rfft(p, size)
    Fr = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * Fr, size)
    g = 2.0 * conv[n - 1:n + n - 1]
    return g

                if s < best_score:

                        M = mq

                            best_f = f.copy()

    # Phase 3: Final coordinate polish (remaining time)
    remaining = total_deadline - time.monotonic() - 20
    if remaining > 10:
        print(f"\n--- Final coordinate polish: {remaining:.0f}s ---")
        bf, bs, bc = _coordinate_polish(best_f, _fft_conv(best_f), remaining, rng)
        if bs < best_score:

            print(f"  Coordinate polish improved to: {best_score:.6f}")

    best_f = _normalize(best_f)

    try:

        official = evaluate_sequence(out)
        if np.isfinite(official):
            print(f"Official score: {official:.6f}")
            if official < best_score:
                best_score = official
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    print(f"\nFINAL score: {best_score:.6f}")
    return [float(x) for x in best_f]
