def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)

    c = np.fft.irfft(fa * fa, N)

def _next_pow2(m):
    n = 1

    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

LP-based iterative refinement for minimizing peak self-convolution of non-negative sequences.

Algorithm:
1. Initialize from height_sequence_1 or scan sinusoidal family.
2. Main loop: solve LP (max sum(b) s.t. x*b <= M, b>=0), line-search blend parameter, update.

4. Validate and return best candidate.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _fft_cross(a, b):
    n = a.size
    if n == 0 or b.size != n:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _build_A(f):
    n = f.size
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    mask = (k_idx >= j_idx) & (k_idx < j_idx + n)
    return np.where(mask, f[None, :], 0.0)


def _solve_lp(f, M):
    n = f.size
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if not result.success:
        return None
    return result.x


def _find_blend_t(a, b, M):
    # Compute convolutions
    ca = _fft_convolve(a)
    cb = _fft_convolve(b)
    cab = _fft_cross(a, b)
    
    # Bisection for largest t with max(c_new) <= M
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        c_new = (1-mid)**2 * ca + 2*mid*(1-mid) * cab + mid**2 * cb
        if np.max(c_new) <= M:
            lo = mid
        else:
            hi = mid
    return lo


def _make_sine(n, amp=0.78, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + amp * np.sin(2.0 * np.pi * (i + phase) / n)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _lp_optimize(a_init, deadline, verbose=True, seed=42):

    best_score = _score(best_a)

    stall_count = 0
    max_stall = 50
    rng = np.random.RandomState(seed)
    
    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        b_opt = _solve_lp(a, M)
        if b_opt is None:
            stall_count += 1
            if stall_count > max_stall:
                break
            continue
        
        b_opt = np.clip(b_opt, 0, None)
        sum_b = float(b_opt.sum())
        if sum_b < 1e-10:

        t = _find_blend_t(a, b_opt, M)
        if t < 1e-6:
            stall_count += 1
            if verbose and it % 10 == 0:

                a = a + noise
                a = np.clip(a, 0, None)

                stall_count = 0

        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 0.01:

            continue
        a_new /= s_new

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall_count = 0
            a = a_new

                a /= a.sum()
                stall_count = 0
                max_stall = int(max_stall * 1.3)

"""
LP-based iterative refinement for minimizing peak self-convolution of non-negative sequences.

Core algorithm: Iteratively solve the LP (max sum(b) s.t. x*b <= M, b>=0) where
M = max(x*x), then blend with line search. Combined with multi-start, adaptive
restarts, and multiple sequence lengths to escape local minima.

Key insight: The LP finds a direction b that "spreads mass" to positions where
the cross-convolution with x is small, effectively reducing peak density.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _fft_cross(a, b):
    n = a.size
    if n == 0 or b.size != n:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _build_A(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[j] if 0<=k-j<n else 0."""
    n = f.size
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    mask = (k_idx >= j_idx) & (k_idx < j_idx + n)
    return np.where(mask, f[None, :], 0.0)


def _solve_lp(f, M):
    """Solve: max sum(b) s.t. A(f)@b <= M, b>=0. Returns b or None."""
    n = f.size
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0.0, None)] * n, method='highs')
    if not result.success:
        return None
    return result.x


def _find_blend_t(a, b, M):
    """Bisection for largest t in [0,1] with max(((1-t)a + tb)*((1-t)a + tb)) <= M."""
    ca = _fft_convolve(a)
    cb = _fft_convolve(b)
    cab = _fft_cross(a, b)
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        c_new = (1.0 - mid) ** 2 * ca + 2.0 * mid * (1.0 - mid) * cab + mid ** 2 * cb
        if float(np.max(c_new)) <= M * (1.0 + 1e-12):
            lo = mid
        else:
            hi = mid
    return lo


def _make_sine(n, amp=0.78, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + amp * np.sin(2.0 * np.pi * (i + phase) / n)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_bimodal(n, sigma_frac=0.15):
    i = np.arange(n, dtype=np.float64)
    centers = np.array([0.25 * n, 0.75 * n])
    sigma = sigma_frac * n
    a = np.zeros(n, dtype=np.float64)
    for c in centers:
        a += np.exp(-0.5 * ((i - c) / sigma) ** 2)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_flat_spectrum(n, decay=0.3):
    """Sequence designed to approximate flat magnitude spectrum."""
    i = np.arange(n, dtype=np.float64)
    # Raised cosine window (Hanning-like)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _lp_optimize(a_init, deadline, verbose=True, seed=42):
    """
    LP-based iterative optimization.
    Each iteration: solve LP, line-search blend, update.
    On stall: perturb with noise.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
    a /= s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    stall_count = 0
    max_stall = 30
    rng = np.random.RandomState(seed)
    last_print = 0

    while time.monotonic() < deadline:
        it += 1
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(np.max(c))

        b_opt = _solve_lp(a, M)
        if b_opt is None:
            stall_count += 1
            if stall_count > max_stall:
                break
            continue

        b_opt = np.clip(b_opt, 0, None)
        sum_b = float(b_opt.sum())
        if sum_b < 1e-10:
            stall_count += 1
            continue

        t = _find_blend_t(a, b_opt, M)
        if t < 1e-8:
            stall_count += 1
        else:
            a_new = (1.0 - t) * a + t * b_opt
            a_new = np.clip(a_new, 0, None)
            s_new = a_new.sum()
            if s_new < 0.01:
                stall_count += 1
                continue
            a_new /= s_new

            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stall_count = 0
                a = a_new
                if verbose and (it - last_print) >= 50:
                    print(f"    iter {it}: score={sc:.8f}, t={t:.4f}, M={M:.6e}", flush=True)
                    last_print = it
            else:
                # Accept anyway for exploration, but count as non-improving
                a = a_new
                stall_count += 1

        if stall_count >= max_stall:
            # Perturb to escape
            noise_scale = 0.005 * a.mean() * (1.0 + 0.1 * np.log1p(stall_count / max_stall))
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.clip(a, 0, None)
            a /= a.sum()
            stall_count = 0
            max_stall = int(max_stall * 1.4)
            if max_stall > 500:
                max_stall = 500
            if verbose:
                print(f"    iter {it}: restart with noise, patience={max_stall}", flush=True)

    return best_a, best_score


def _gradient_polish(a, deadline, K=8):
    """
    Quick gradient descent polish using top-K peak weights.
    Complements LP by doing local refinement.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return a, _score(a)
    a /= s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(777)
    lr = 0.1
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        weights = np.exp(3.0 * (top_vals - top_vals.max()))
        weights /= weights.sum()

        w = np.zeros(c.size)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        # Gradient via FFT cross-correlation
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[:N], N) if w.size > N else np.fft.rfft(w, N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        grad = 2.0 * xcorr[:n]

        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a /= s

        if it % 5 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                lr = min(lr * 1.002, 0.5)
            elif it % 50 == 0:
                lr = max(lr * 0.9, 0.01)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining LP iterations, gradient polish, and multi-start.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 25.0

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    print(f"[Start] Budget: {budget_s}s, hard deadline in {hard_deadline - t0:.0f}s", flush=True)

    # ========== Phase 1: Collect initial candidates ==========
    print("[Phase 1] Collecting initial candidates...", flush=True)
    inits = []

    # Try height_sequence_1 if available
    if 'height_sequence_1' in globals() and globals()['height_sequence_1'] is not None:
        try:
            h = np.array(height_sequence_1, dtype=np.float64)
            h = np.clip(h, 0, 1000)
            if h.size >= 10 and h.sum() > 0.01:
                h = h / h.sum()
                sc = _score(h)
                inits.append((h, sc, 'parent'))
                print(f"  parent: n={h.size}, score={sc:.8f}", flush=True)
        except Exception as e:
            print(f"  parent error: {e}", flush=True)

    # Sinusoidal family
    for n in [512, 1024, 2048, 4096]:
        for amp in [0.5, 0.7, 0.78, 0.85, 0.9, 1.0, 1.2]:
            a = _make_sine(n, amp)
            sc = _score(a)
            inits.append((a, sc, f'sine_{n}_{amp}'))

    # Bimodal
    for n in [1024, 2048]:
        for sf in [0.1, 0.15, 0.2, 0.3]:
            a = _make_bimodal(n, sf)
            sc = _score(a)
            inits.append((a, sc, f'bimodal_{n}_{sf}'))

    # Raised cosine
    for n in [1024, 2048]:
        a = _make_flat_spectrum(n)
        sc = _score(a)
        inits.append((a, sc, f'cosine_{n}'))

    # Random with anti-palindromic structure
    for _ in range(5):
        n = 1024
        base = rng.standard_normal(n)
        # Make it anti-palindromic: a[i] = 0.5 + base[i] - base[n-1-i]
        anti = 0.5 + (base - base[::-1]) * 0.3
        a = np.clip(anti, 0, None)
        s = a.sum()
        if s > 0.01:
            a /= s
            sc = _score(a)
            inits.append((a, sc, f'random_{_}'))

    inits.sort(key=lambda x: x[1])
    print(f"  Best init: {inits[0][2]}, n={inits[0][0].size}, score={inits[0][1]:.8f}", flush=True)
    print(f"  Top 5: {[(n, f'{s:.6f}') for _, s, n in inits[:5]]}", flush=True)

    # ========== Phase 2: LP optimization on top candidates ==========
    # Allocate time: 75% for LP, 15% for gradient polish, 10% for validation
    lp_deadline = t0 + (hard_deadline - t0) * 0.75
    polish_deadline = t0 + (hard_deadline - t0) * 0.90

    best_overall_a = inits[0][0].copy()
    best_overall_score = inits[0][1]

    # Try top candidates, but skip if same (n, type) as already tried
    tried_keys = set()
    n_topics = 0
    for i, (cand_a, cand_sc, cand_name) in enumerate(inits):
        if time.monotonic() >= lp_deadline - 30:
            break

        key = (cand_a.size, cand_name.split('_')[0])
        if key in tried_keys:
            continue
        tried_keys.add(key)
        n_topics += 1

        # Time budget per topic: at least 60s for first, less for later
        if n_topics == 1:
            topic_deadline = min(lp_deadline, time.monotonic() + (lp_deadline - time.monotonic()) * 0.5)
        else:
            remaining = lp_deadline - time.monotonic()
            topic_deadline = time.monotonic() + remaining / max(1, 4 - n_topics + 1)
            topic_deadline = min(topic_deadline, lp_deadline - 10)

        if topic_deadline - time.monotonic() < 30:
            break

        print(f"\n[Phase 2.{n_topics}] LP on {cand_name} (n={cand_a.size}, init={cand_sc:.6f}, budget={topic_deadline - time.monotonic():.0f}s)", flush=True)
        res_a, res_sc = _lp_optimize(cand_a, topic_deadline, seed=seed + i * 100)
        print(f"  LP result: score={res_sc:.8f}", flush=True)

        if res_sc < best_overall_score:
            best_overall_score = res_sc
            best_overall_a = res_a.copy()
            print(f"  ** New best: {res_sc:.8f} **", flush=True)

    # ========== Phase 3: Gradient polish ==========
    remaining_polish = polish_deadline - time.monotonic()
    if remaining_polish > 30:
        print(f"\n[Phase 3] Gradient polish (budget {remaining_polish:.0f}s)...", flush=True)
        ga, gs = _gradient_polish(best_overall_a, polish_deadline, K=8)
        print(f"  Gradient polish result: {gs:.8f}", flush=True)
        if gs < best_overall_score:
            best_overall_score = gs
            best_overall_a = ga.copy()
            print(f"  ** New best: {gs:.8f} **", flush=True)

    # ========== Phase 4: Try one more LP round on the best ==========
    remaining_lp2 = hard_deadline - 30 - time.monotonic()
    if remaining_lp2 > 60:
        print(f"\n[Phase 4] Final LP round (budget {remaining_lp2:.0f}s)...", flush=True)
        fa, fs = _lp_optimize(best_overall_a, hard_deadline - 30, seed=seed + 9999)
        print(f"  Final LP result: {fs:.8f}", flush=True)
        if fs < best_overall_score:
            best_overall_score = fs
            best_overall_a = fa.copy()
            print(f"  ** New best: {fs:.8f} **", flush=True)

    # ========== Phase 5: Validation ==========
    print(f"\n[Phase 5] Validation... (best internal score: {best_overall_score:.8f})", flush=True)

    # Also validate the initial best
    candidates_to_try = [
        ('best', best_overall_a, best_overall_score),
        ('init', inits[0][0], inits[0][1]),
    ]
    # Add second best if different
    if len(inits) > 1 and inits[1][0].size != best_overall_a.size:
        candidates_to_try.append(('init2', inits[1][0], inits[1][1]))

    candidates_to_try.sort(key=lambda x: x[2])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in candidates_to_try:
        if time.monotonic() > hard_deadline:
            break
        arr_clip = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000)
        arr_list = [float(x) for x in arr_clip]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:

        best_out = [float(x) for x in arr_clip]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_overall_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"\n[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
