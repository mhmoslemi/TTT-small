import time

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S*S)

    """One LP iteration: solve max sum(b) s.t. (a*b)[k] <= M for top-K k, b >= 0"""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    # Top K lags
    k = min(K, len(c))
    top_lags = np.argsort(c)[::-1][:k]
    # Build constraint matrix
    A_mat = np.zeros((k, n))
    for i, lag in enumerate(top_lags):
        lag = int(lag)
        j_start = max(0, lag - n + 1)
        j_end = min(lag, n - 1)
        if j_end >= j_start:
            A_mat[i, j_start:j_end+1] = a[j_start:j_end+1][::-1]

    b_var = cp.Variable(n)
    constraints = [A_mat @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver)

        return a.copy()
    if b_var.value is None:

    b = np.clip(b_var.value, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-12:
        return a.copy()
    # Rescale to match sum
    b = b * (a.sum() / b_sum)
    return b

    """Run LP iteration for num_steps steps"""

    S = float(a.sum())

        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxiter * 20,
                 'maxls': 50}

    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)

def build_convolution_matrix_row(a, k):
    """Return the row of the convolution matrix for lag k."""
    n = len(a)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_end >= j_start:
        row[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    return row


def lp_step(a, K=500, solver='ECOS'):
    """One LP iteration step. Returns new b."""
    if not HAS_CVXPY:
        return a.copy()
    n = len(a)
    c = fast_conv(a)
    M = float(np.max(c))
    K = min(K, len(c))
    top_lags = np.argsort(c)[::-1][:K]
    # Build constraint matrix
    A_mat = np.zeros((K, n))
    for i, lag in enumerate(top_lags):
        A_mat[i, :] = build_convolution_matrix_row(a, int(lag))
    b_var = cp.Variable(n)
    constraints = [A_mat @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return a.copy()
    if b_var.value is None:
        return a.copy()
    b = np.clip(b_var.value, 0, None)
    b_sum = float(b.sum())
    if b_sum < 1e-12:
        return a.copy()
    b = b * (float(a.sum()) / b_sum)
    return b


def lp_iteration(a, num_steps=100, K=500, t=0.15, solver='ECOS', time_limit=60):
    """Run LP iteration for num_steps or until time_limit."""
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    for step in range(num_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = lp_step(a, K=K, solver=solver)
        a_new = (1 - t) * a + t * b
        a_new = np.clip(a_new, 0, None)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if step % 20 == 0:
            print(f"    LP step {step}: score={sc:.8f}, best={best_score:.8f}")
    return best_score, best_a


def multiplicative_weights_step(a, eta=0.1):
    """Approximate LP step using multiplicative weights (no CVXPY needed)."""
    n = len(a)
    c = fast_conv(a)
    M = float(np.max(c))
    # For each j, compute the maximum "pressure": max over k of a[k-j]
    # This is the max element of the "column" of the conv matrix
    # Equivalent to: for each j, max_k a[k-j] where 0<=k-j<n
    # = max(a) (since as k ranges, k-j ranges over all valid indices)
    # Actually it's max over k of a[(k-j) mod ... ] -- let me think again.
    # The constraint for lag k: sum_j b[j] * a[k-j] <= M
    # The "dual variable" y[k] tells us how tight the constraint is.
    # At the optimum, b[j] = exp(-sum_k y[k] * a[k-j])
    # Approximation: y[k] = 1 for all k (uniform dual)
    # Then b[j] = exp(-sum_k a[k-j]) = exp(-sum_m a[m]) = exp(-S) for all j
    # This gives b = uniform, which is not helpful.
    # Better: y[k] proportional to c[k] (weight by how peaky the lag is)
    # y[k] = c[k] / sum(c)
    # b[j] = exp(-sum_k y[k] * a[k-j])
    y = c / c.sum()
    # b[j] = exp(-sum_k y[k] * a[k-j])
    # This is a "soft" convolution
    # sum_k y[k] * a[k-j] = (y * a_rev)[n-1-j] ... let me compute via FFT
    # Actually: sum_k y[k] * a[k-j] for k=0..2n-2
    # = sum_k y[k] * a[k-j] where a is 0 outside [0,n-1]
    # Let m = k-j: = sum_m y[m+j] * a[m] for m=0..n-1 (where y[m+j] is defined for m+j <= 2n-2)
    # This is the correlation of y and a shifted by j.
    # = (y * a_rev)[j + n - 1] ... hmm, let me just compute it directly via FFT.
    # Define: P[j] = sum_k y[k] * a[k-j]
    # This is the cross-correlation of y and a.
    # P = correlate(y, a) or equivalently convolve(y, a_rev)
    y_padded = np.zeros(2 * n)
    y_padded[:len(y)] = y
    a_rev = a[::-1]
    M_fft = 1 << (3 * n - 1).bit_length()
    Y = np.fft.rfft(y_padded, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    P_full = np.fft.irfft(Y * AR, M_fft)
    # P[j] = P_full[j + n - 1] for j = 0..n-1
    P = P_full[n - 1:n - 1 + n]
    P = np.maximum(P, 1e-30)
    b = a * np.exp(-eta * P / P.max())
    b = np.clip(b, 0, None)
    if b.sum() < 1e-12:
        return a.copy()
    b = b * (a.sum() / b.sum())
    return b


def peak_aware_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[top_count:top_count * 2]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0, 1000.0)


def build_inits(n, rng, parent):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            # Interpolate parent to length n
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            f = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
            inits.append(np.clip(f(x_new), 0, 1000))
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4 * np.pi * x))
    inits.append(np.maximum(0.05, 0.5 + 0.45 * np.cos(np.pi * (2 * x - 1))))
    inits.append(np.maximum(0.01, 0.1 + 0.9 * (2 * x - 1) ** 2))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10))
    # Smoothed random
    r = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 10)
    kernel = np.ones(10) / 10
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(np.clip(r_smooth, 0.01, 10))
    return inits

"""
LP-constraint iteration with coarse-to-fine multiscale strategy for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovations over previous approach:
1. LP-based iteration (from the inequality paper) as main engine - handles max exactly
2. Coarse-to-fine multiscale: solve at small n, upsample, solve at larger n
3. Adaptive constraint generation (top-K lags for large n, all for small n)
4. Combined LP + L-BFGS + peak-aware local search
"""

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVEXPY = False


# ---------------------------------------------------------------------------
# Core numerical helpers
# ---------------------------------------------------------------------------

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

# LP-based step

def build_conv_rows(a, lags):
    """Build rows of convolution matrix for given lags. Returns (K, n) matrix."""
    n = len(a)
    K = len(lags)
    A_mat = np.zeros((K, n))
    for i, k in enumerate(lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A_mat[i, j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    return A_mat


def lp_step(a, K=None, solver_list=None):
    """
    One LP iteration: solve max sum(b) s.t. (a*b)[k] <= M for specified lags, b >= 0.
    Returns the optimal b (rescaled to match sum(a)).
    """
    if not HAS_CVXPY:
        return a.copy()

    n = len(a)
    c = fast_conv(a)
    M = float(np.max(c))

    if K is None or K >= len(c):
        lags = np.arange(len(c))
    else:
        lags = np.argsort(c)[::-1][:K]

    A_mat = build_conv_rows(a, lags)

    b_var = cp.Variable(n)
    constraints = [A_mat @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    if solver_list is None:
        solver_list = ['ECOS', 'CLARABEL', 'SCS', 'MOSEK']

    b_val = None
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None:
                b_val = b_var.value
                break
        except Exception:
            continue

    if b_val is None:
        return a.copy()

    b = np.clip(b_val, 0, None)
    b_sum = float(b.sum())
    if b_sum < 1e-12:
        return a.copy()
    b = b * (float(a.sum()) / b_sum)
    return b


def run_lp_iteration(a, num_steps=50, K=None, t=0.15, time_limit=120,
                     solver_list=None, verbose=False):
    """Run LP iteration for num_steps or until time_limit."""
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    for step in range(num_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = lp_step(a, K=K, solver_list=solver_list)
        a_new = (1.0 - t) * a + t * b
        a_new = np.clip(a_new, 0, None)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if verbose and step % 10 == 0:
            print(f"      LP step {step}: score={sc:.8f}, best={best_score:.8f}")
    return best_score, best_a


def multiplicative_weights_step(a, eta=0.1):
    """Fast approximate LP step using multiplicative weights."""
    n = len(a)
    c = fast_conv(a)
    y = c / (c.sum() + 1e-30)
    # P[j] = sum_k y[k] * a[k-j]
    y_padded = np.zeros(2 * n)
    y_padded[:len(y)] = y
    a_rev = a[::-1]
    M_fft = 1 << (3 * n).bit_length()
    Y = np.fft.rfft(y_padded, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    P_full = np.fft.irfft(Y * AR, M_fft)
    P = P_full[n - 1:n - 1 + n]
    P = np.maximum(P, 1e-30)
    b = a * np.exp(-eta * P / (P.max() + 1e-30))
    b = np.clip(b, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-12:
        return a.copy()
    b = b * (a.sum() / b_sum)
    return b


def run_mw_iteration(a, num_steps=200, eta=0.1, time_limit=60):
    """Run multiplicative weights iteration."""
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    for step in range(num_steps):
        if time.monotonic() - t0 > time_limit:
            break
        a_new = multiplicative_weights_step(a, eta=eta)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=3000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxiter * 20,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def upscale_by_replication(a, factor=2):
    """Upscale by repeating each element 'factor' times."""
    return np.repeat(a, factor)


def upscale_by_interp(a, factor=2):
    """Upscale by cubic interpolation."""
    n_old = len(a)
    n_new = n_old * factor
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    # Use cubic spline via numpy (simple: linear then smooth)
    a_interp = np.interp(x_new, x_old, a)
    # Light smoothing
    k = np.array([0.1, 0.3, 0.4, 0.3, 0.1])
    a_smooth = np.convolve(a_interp, k, mode='same')
    return np.clip(a_smooth, 0, None)

def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent))

    x = np.arange(n, dtype=np.float64) / n
    # Structured
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(np.maximum(0.05, 0.5 + 0.4 * np.cos(np.pi * (2 * x - 1))))
    inits.append(np.maximum(0.01, 0.1 + 0.8 * (2.0 * x - 1.0) ** 2))
    inits.append(0.5 + 0.25 * np.cos(4.0 * np.pi * x) + 0.15 * np.cos(6.0 * np.pi * x))
    # Random smooth
    r = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
    kernel = np.ones(7) / 7.0
    inits.append(np.clip(np.convolve(r, kernel, mode='same'), 0.01, 10.0))
    # Two-valued
    inits.append(np.where(rng.random(n) > 0.5, 1.0, 0.1))

    return inits


def peak_aware_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[top_count:top_count + top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - amt, 0.0)
        total_reduced += amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0, 1000.0)


def multi_peak_perturb(a, rng, num_peaks=5, strength=0.04):
    """Reduce elements contributing to the top num_peaks convolution lags."""
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:num_peaks]
    contrib = np.zeros(n)
    for k in top_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 25)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[top_count:top_count + top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - amt, 0.0)
        total_reduced += amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0, 1000.0)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    # Use the verified score for reporting (in case of numerical differences)
    if verified < best_score:
        best_score = verified

    return out
