"""
Step function optimizer using iterative LP (alternating direction) 
with gradient polish.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _build_lp_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def _solve_lp(f, n, rows, cols):
    c_conv = _autocorr_fft(f)
    M = c_conv.max()
    if M < 1e-30:
        return f.copy()
    kij = (rows - cols).astype(np.int64)
    vals = f[kij]
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = M * np.ones(2*n-1)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return f.copy()

def _lp_iteration(f, n, rows, cols, t=1.0):
    g = _solve_lp(f, n, rows, cols)
    s = g.sum()
    if s < 1e-10:
        return f.copy()
    g = g / s
    f_new = (1 - t) * f + t * g
    return np.maximum(f_new, 0)

def _gradient_step(f, n, step=0.1, K=3):
    c = _autocorr_fft(f)
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
    weights = 1.0 / gaps
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    
    # Mirror descent step
    span = grad.max() - grad.min()
    if span < 1e-30:
        return f.copy()
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

"""
Step function optimizer using iterative LP (alternating best-response),
gradient mirror descent, and coordinate descent polish.
Main novelty: uses LP to find optimal "direction" at each step,
then combines with gradient and coordinate methods for convergence.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize_scalar
import time


def _autocorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _build_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _solve_lp_uniform(f, n, rows, cols):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_conv = _autocorr_fft(f)
    M = c_conv.max()
    if M < 1e-30:
        return f.copy()
    kij = (rows - cols).astype(np.int64)
    vals = f[kij]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return f.copy()


def _solve_lp_individual(f, n, rows, cols):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= (f*f)[k] for all k, g >= 0."""
    c_conv = _autocorr_fft(f)
    b = np.maximum(c_conv, 1e-30)
    kij = (rows - cols).astype(np.int64)
    vals = f[kij]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return f.copy()


def _lp_step(f, n, rows, cols, t=1.0, mode='uniform'):
    """Take one LP-based step."""
    if mode == 'uniform':
        g = _solve_lp_uniform(f, n, rows, cols)
    else:
        g = _solve_lp_individual(f, n, rows, cols)
    s = g.sum()
    if s < 1e-10:
        return f.copy()
    g = g / s
    f_new = (1.0 - t) * f + t * g
    return np.maximum(f_new, 0.0)


def _mirror_gradient_step(f, n, step=0.5, K=3):
    """Mirror descent step targeting top K autocorrelation peaks."""
    c = _autocorr_fft(f)
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
    weights = 1.0 / gaps
    weights /= weights.sum()

    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]

    span = grad.max() - grad.min()
    if span < 1e-30:
        return f.copy()
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def _coordinate_descent(f, n, deadline, max_rounds=50):
    """Coordinate descent: optimize each position one at a time."""
    f = f.copy().astype(np.float64)
    best_score = _score_fast(f)
    best_f = f.copy()
    s = f.sum()
    c = _autocorr_fft(f)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_v = f[j]
            if old_v < 1e-15:
                continue

            # Compute delta contributions for each lag k
            # (f*f)[k] changes by:
            #   if k == 2j: v^2 - old_v^2
            #   if k != 2j and 0 <= k-j < n: 2*(v - old_v)*f[k-j]
            #   else: 0
            # sum changes by (v - old_v)

            # Build the function score(v) for golden section
            new_s = s - old_v  # sum without f[j]

            def obj(v):
                if v < -1e-12:
                    return 1e10
                v = max(v, 0.0)
                c_new = c.copy()
                # Update c_new for position j
                k_eq = 2 * j
                if 0 <= k_eq < 2 * n - 1:
                    c_new[k_eq] = c_new[k_eq] - old_v * old_v + v * v
                k_lo = max(0, j)  # k-j >= 0 => k >= j
                k_hi = min(2 * n - 2, j + n - 1)  # k-j <= n-1 => k <= j+n-1
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * j:
                        continue
                    c_new[k] = c_new[k] + 2.0 * (v - old_v) * f[k - j]
                new_sum = new_s + v
                if new_sum < 0.01:
                    return 1e10
                return 2.0 * n * c_new.max() / (new_sum * new_sum)

            # Golden section on [0, 2*old_v + 0.1*f.mean()]
            lo = 0.0
            hi = 2.0 * old_v + 0.05 * f.mean()
            res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 100})
            if res.success and res.fun < best_score - 1e-12:
                f[j] = res.x
                s = new_s + res.x
                c = _autocorr_fft(f)
                best_score = res.fun
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_score


def _gradient_polish(f, n, deadline, initial_step=0.5):
    """Gradient mirror descent polish."""
    f = f.copy().astype(np.float64)
    best_score = _score_fast(f)
    best_f = f.copy()
    step = initial_step
    stuck = 0
    threshold = 1e-9

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        f = _mirror_gradient_step(f, n, step=step, K=3)
        if it % 100 == 0:
            sc = _score_fast(f)
            if sc < best_score - threshold:
                best_score = sc
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 20:
                step *= 0.7
                stuck = 0
                if step < 0.001:
                    break
    sc = _score_fast(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    return best_f, best_score


def _structured_perturb(f, rng, mode=0):
    """Apply structured perturbation for diversity."""
    n = len(f)
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()

    if mode == 0:

        tilt = np.exp(alpha * (j / n - 0.5))
        f_try = f * tilt
    elif mode == 1:
        freq = rng.integers(1, max(2, n // 4) + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        f_try = f * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.3, 0.8)
        w = 1.0 - width * ((j / n - center) / 0.5) ** 2
        f_try = f * np.maximum(w, 0.1)
    elif mode == 3:
        f_try = f + rng.uniform(-0.03, 0.03, size=n) * f.mean()
    else:
        freq1 = rng.integers(1, max(2, n // 5) + 1)
        freq2 = rng.integers(1, max(2, n // 3) + 1)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        f_try = f * (1.0 + 0.15 * np.sin(2 * np.pi * freq1 * j / n + ph1))
        f_try = f_try * (1.0 + 0.15 * np.sin(2 * np.pi * freq2 * j / n + ph2))

    f_try = np.maximum(f_try, 0.0)
    s = f_try.sum()
    if s < 1e-10:

    return f_try / s

        if s < 1e-10:
            break
        f_new = f_new / s
        sc = _score_fast(f_new)

        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            patience = 0
            if it % 50 == 0:
                print(f"    {label} LP iter {it}: {sc:.7f} (t={t_step:.3f})")
        else:
            patience += 1
            if patience >= max_patience:
                # Try reducing step or switching mode
                if t_step > 0.1:
                    t_step *= 0.5
                else:
                    # Switch mode
                    if mode == 'uniform':
                        mode = 'individual'
                    else:
                        mode = 'uniform'
                    t_step = 1.0
                patience = 0
                if t_step < 0.05 and mode == 'individual':
                    break

    return best_f, best_score


def _run_hybrid(f0, n, rows, cols, deadline, rng, label=""):
    """Hybrid: alternate LP steps and gradient steps."""
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    best_score = _score_fast(f)
    best_f = f.copy()
    t_lp = 1.0
    t_grad = 0.5
    phase = 'lp'
    count = 0

    while time.monotonic() < deadline:
        count += 1
        if phase == 'lp':
            f_new = _lp_step(f, n, rows, cols, t=t_lp, mode='uniform')
            s = f_new.sum()
            if s < 1e-10:
                break
            f_new = f_new / s
            # After 5 LP steps, switch to gradient
            if count % 5 == 0:
                phase = 'grad'
        else:
            f_new = _mirror_gradient_step(f, n, step=t_grad, K=3)
            # After 5 grad steps, switch to LP
            if count % 5 == 0:
                phase = 'lp'

        sc = _score_fast(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            if count % 100 == 0:
                print(f"    {label} hybrid iter {count}: {sc:.7f}")
        else:
            # Decay step sizes
            if phase == 'lp':
                t_lp = max(0.1, t_lp * 0.95)
            else:
                t_grad = max(0.01, t_grad * 0.95)
            if t_lp < 0.1 and t_grad < 0.01:
                break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses iterative LP (best-response), gradient mirror descent,
    and coordinate descent in a multi-phase search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    print(f"Sequence length: n={n}")

    # Normalize initial
    f0 = f_init.copy()
    s0 = f0.sum()
    if s0 < 1e-10:
        f0 = np.ones(n)
    else:
        f0 = f0 / s0

    initial_score = _score_fast(f0)
    best_score = initial_score
    best_f = f0.copy()
    print(f"Initial score: {initial_score:.7f}")

    # Build LP structure
    print("Building LP structure...")
    rows, cols = _build_lp_structure(n)
    print(f"LP structure: {len(rows)} nonzeros")

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = {}
    starts['init'] = f0.copy()
    starts['uniform'] = np.ones(n)
    starts['raised_cos'] = 0.5 * (1 + np.cos(2 * np.pi * j / n)) + 0.01
    starts['hamming'] = 0.54 - 0.46 * np.cos(2 * np.pi * j / (n - 1))
    starts['gauss_wide'] = np.exp(-1.5 * ((j - center) / (n * 0.4)) ** 2) + 0.01
    starts['gauss_narrow'] = np.exp(-4.0 * ((j - center) / (n * 0.25)) ** 2) + 0.01
    starts['bimodal'] = (np.exp(-((j / n - 0.3) / 0.12) ** 2) +
                         np.exp(-((j / n - 0.7) / 0.12) ** 2) + 0.01)
    starts['u_shape'] = 1.0 + 2.0 * ((j / n - 0.5) * 2) ** 2
    starts['u_shape_strong'] = 1.0 + 6.0 * ((j / n - 0.5) * 2) ** 2
    starts['sine2'] = 1.0 + 0.3 * np.sin(4 * np.pi * j / n)
    starts['exp_asym'] = np.exp(-1.5 * j / n) + 0.01
    starts['exp_asym2'] = np.exp(-1.5 * (n - 1 - j) / n) + 0.01

    # Normalize and score starts
    start_list = []
    for name, sf in starts.items():
        sf = np.maximum(sf, 0.0)
        ss = sf.sum()
        if ss < 1e-10:
            sf = np.ones(n)
        else:
            sf = sf / ss
        sc = _score_fast(sf)
        start_list.append((sc, name, sf))

    start_list.sort(key=lambda x: x[0])
    print("\nStarting points (sorted by score):")
    for sc, name, _ in start_list[:6]:
        print(f"  {name}: {sc:.7f}")

    # Update best
    if start_list[0][0] < best_score:
        best_score = start_list[0][0]
        best_f = start_list[0][2].copy()

    # Phase 1: LP optimization from top starts (50% of time)
    phase1_deadline = t_start + (budget_s - 60) * 0.50
    n_starts = min(4, len(start_list))
    print(f"\n=== Phase 1: LP optimization from top {n_starts} starts ===")

    for idx in range(n_starts):
        if time.monotonic() > phase1_deadline:
            break
        sc, name, sf = start_list[idx]
        # Allocate time: more for better starts
        remaining = phase1_deadline - time.monotonic()
        frac = 1.0 / (n_starts - idx) * 0.85
        sub_deadline = time.monotonic() + remaining * frac

        print(f"  LP from '{name}' (initial={sc:.6f}, budget={sub_deadline - time.monotonic():.0f}s)")
        f_opt, sc_opt = _run_lp_optimization(sf, n, rows, cols, sub_deadline, rng, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> Best from this start: {sc_opt:.7f}")

    # Phase 2: Hybrid LP+gradient from best (25% of time)
    phase2_deadline = t_start + (budget_s - 60) * 0.75
    if time.monotonic() < phase2_deadline:
        print(f"\n=== Phase 2: Hybrid optimization (score={best_score:.7f}) ===")
        # Try from best and perturbed versions
        n_hybrid = 3
        for i in range(n_hybrid):
            if time.monotonic() > phase2_deadline:
                break
            if i == 0:
                f_start = best_f.copy()
            else:
                f_start = _structured_perturb(best_f, rng, mode=i - 1)
            sub_deadline = min(phase2_deadline, time.monotonic() + 120)
            f_opt, sc_opt = _run_hybrid(f_start, n, rows, cols, sub_deadline, rng,
                                         label=f"hybrid{i}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"    hybrid{i}: New best: {best_score:.7f}")
            else:
                print(f"    hybrid{i}: {sc_opt:.7f}")

    # Phase 3: Gradient polish (15% of time)
    phase3_deadline = t_start + (budget_s - 60) * 0.88
    if time.monotonic() < phase3_deadline:
        print(f"\n=== Phase 3: Gradient polish (score={best_score:.7f}) ===")
        f_opt, sc_opt = _gradient_polish(best_f, n, phase3_deadline, initial_step=0.3)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    Polished to: {best_score:.7f}")
        else:
            print(f"    Gradient result: {sc_opt:.7f}")

    # Phase 4: Coordinate descent polish (7% of time)
    phase4_deadline = t_start + (budget_s - 60) * 0.95
    if time.monotonic() < phase4_deadline:
        print(f"\n=== Phase 4: Coordinate descent (score={best_score:.7f}) ===")
        f_opt, sc_opt = _coordinate_descent(best_f, n, phase4_deadline, max_rounds=30)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    Coord descent: {best_score:.7f}")
        else:
            print(f"    Coord descent: {sc_opt:.7f}")

    # Phase 5: Final LP micro-steps + gradient
    final_deadline = deadline - 10
    if time.monotonic() < final_deadline:
        print(f"\n=== Phase 5: Final polish (score={best_score:.7f}) ===")
        # Small LP steps
        f = best_f.copy()
        for _ in range(50):
            if time.monotonic() > final_deadline:
                break
            f_new = _lp_step(f, n, rows, cols, t=0.3, mode='uniform')
            s = f_new.sum()
            if s < 1e-10:
                break
            f_new = f_new / s
            sc = _score_fast(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
        # Tiny gradient steps
        step = 0.1
        for _ in range(5000):
            if time.monotonic() > final_deadline:
                break
            f = _mirror_gradient_step(best_f, n, step=step, K=2)
            sc = _score_fast(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            step *= 0.998

    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation: {final_score:.7f}")

    if not np.isfinite(final_score):
        # Fallback to initial
        final_score = initial_score
        best_f = f0.copy()

    print(f"\n=== FINAL SCORE: {final_score:.7f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_f]
