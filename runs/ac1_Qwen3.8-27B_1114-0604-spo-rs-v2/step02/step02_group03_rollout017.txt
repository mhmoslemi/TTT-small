import numpy as np

TARGET_SCORE = 1.5027


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / v.size
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (m - 1).bit_length()


def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    n = f.size
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    w = np.arange(1, n + 1, dtype=np.float64)
    cond = u * w > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    out /= s
    return out


def _weighted_gradient(p, f):
    n = f.size
    L = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _focused_gradient(c, f, alpha):
    """Compute gradient using focused softmax on the top lags."""
    n = f.size
    L = c.size
    cmax = float(c.max())
    if cmax <= 0:
        return np.zeros(n, dtype=np.float64)
    T = cmax / alpha
    z = (c - cmax) / T
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p = np.zeros(L)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= ps
    return _weighted_gradient(p, f)


def _topk_gradient(c, f, K):
    """Gradient using uniform weights on top-K lags."""
    n = f.size
    L = c.size
    if K >= L:
        p = np.ones(L) / L
    else:
        top_idx = np.argpartition(c, L - K)[L - K:]
        p = np.zeros(L)
        p[top_idx] = 1.0 / K
    return _weighted_gradient(p, f)


def _armijo_step(f, c, g, alpha, max_bt=15):
    """Armijo backtracking line search. Returns (new_f, new_c, step_taken)."""
    n = f.size
    cmax0 = float(c.max())
    fg = np.dot(g, g)
    if fg < 1e-300:
        return f, c, 0.0
    rho = 0.6
    c1 = 0.01
    # Initial step size heuristic
    gnorm = float(np.sqrt(fg))
    lr0 = 0.5 * cmax0 / gnorm if gnorm > 1e-300 else 0.01
    lr0 = min(max(lr0, 0.001), 5.0)

    for bt in range(max_bt):
        lr = lr0 * (rho ** bt)
        if lr < 1e-10:
            break
        f_new = _normalize(f - lr * g)
        c_new = _autocorr(f_new)
        cmax_new = float(c_new.max())
        # Armijo condition: cmax_new <= cmax0 - c1 * lr * fg * 0.5
        # (factor 0.5 because we're measuring cmax, not the full objective properly)
        if cmax_new <= cmax0 - c1 * lr * fg * 0.1:
            return f_new, c_new, lr
        # Also accept if just improving
        if cmax_new < cmax0 - 1e-12:
            return f_new, c_new, lr
    return f, c, 0.0


def _spectral_perturb(f, rng, strength=0.3):
    """Perturb in frequency domain: multiply spectrum by random filter."""
    n = f.size
    size = _fft_size(2 * n - 1)
    F = np.fft.rfft(f, size)
    freqs = np.fft.rfftfreq(size, 1.0 / n)
    # Random low-pass-ish filter
    cutoff = float(rng.uniform(0.1, 0.5))
    filter_val = np.exp(-freqs ** 2 / (2.0 * cutoff ** 2))
    # Add random phase perturbation
    phase = strength * float(rng.uniform(0.2, 1.0)) * rng.standard_normal(F.size)
    F_perturbed = F * filter_val * np.exp(1j * phase)
    f_new = np.fft.irfft(F_perturbed, size)[:n]
    return _normalize(f_new)


def _random_restart(n, rng, base=None):
    """Generate a random-ish starting point."""
    r = float(rng.random())
    if r < 0.3:
        x = np.linspace(0, 1, n)
        a = float(rng.uniform(0.5, 3.0))
        b = float(rng.uniform(0.5, 3.0))
        f = (x + 0.01) ** a * (1 - x + 0.01) ** b
    elif r < 0.5:
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.5) / float(rng.uniform(0.05, 0.3))) ** 2)
    elif r < 0.7:
        f = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            freq = float(rng.uniform(1, 5))
            phase = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.1, 0.5))
            f = f * (1.0 + amp * np.cos(2 * np.pi * freq * np.linspace(0, 1, n) + phase))
    elif r < 0.85 and base is not None:
        f = base * np.exp(0.3 * rng.standard_normal(n))
    else:
        f = rng.random(n) + 0.01
    return _normalize(f)

            alpha = max(min_alpha, alpha * 0.98)
        elif no_improve > 10:
            alpha = min(max_alpha, alpha * 1.1)

        # Compute focused gradient
        g = _focused_gradient(c, f, alpha)

        # Also compute top-5 gradient for targeted moves
        g_top = _topk_gradient(c, f, K=5)

        # Combine: primary gradient + small top-K component
        g_combined = 0.7 * g + 0.3 * g_top

        # Center the gradient (remove mean to preserve sum approximately)
        g_combined = g_combined - float(g_combined.mean())

        # Try the combined step
        f_new, c_new, lr = _armijo_step(f, c, g_combined, alpha)

        if lr > 0:
            cmax_new = float(c_new.max())
            f = f_new
            c = c_new
            cmax = cmax_new
            no_improve = 0
            if cmax < best_cmax:
                best_cmax = cmax
                best_f = f.copy()
                best_score = float(2.0 * n * cmax)
        else:
            # Try top-K gradient alone
            f_new2, c_new2, lr2 = _armijo_step(f, c, g_top, alpha)
            if lr2 > 0:
                f = f_new2
                c = c_new2
                cmax = float(c_new2.max())
                no_improve = 0
                if cmax < best_cmax:
                    best_cmax = cmax
                    best_f = f.copy()
                    best_score = float(2.0 * n * cmax)
            else:
                no_improve += 1

        # Periodic targeted peak reduction
        if no_improve > 3 and time.monotonic() - last_polish > 0.3:
            last_polish = time.monotonic()
            f_pol, cmax_pol = _targeted_peak_reduction(f, c, n, rng)
            if cmax_pol < cmax:
                f = f_pol
                c = _autocorr(f)
                cmax = cmax_pol
                no_improve = 0
                if cmax < best_cmax:
                    best_cmax = cmax
                    best_f = f.copy()
                    best_score = float(2.0 * n * cmax)

        # Periodic perturbation for exploration
        if no_improve > 25 and time.monotonic() - last_perturb > 2.0:
            last_perturb = time.monotonic()
            f_pert = _spectral_perturb(f, rng, strength=float(rng.uniform(0.2, 0.8)))
            c_pert = _autocorr(f_pert)
            cmax_pert = float(c_pert.max())
            # Also try mixing with best
            f_mix = _normalize(0.7 * best_f + 0.3 * f_pert)
            c_mix = _autocorr(f_mix)
            cmax_mix = float(c_mix.max())

            if cmax_pert < cmax:
                f = f_pert
                c = c_pert
                cmax = cmax_pert
            elif cmax_mix < cmax:
                f = f_mix
                c = c_mix
                cmax = cmax_mix
            else:
                # Keep current but bump temperature
                alpha = min(max_alpha, alpha * 1.5)
            no_improve = 0

        if it % 5000 == 0:
            print(f"[{tag}] it={it} best={best_score:.6f} cur={float(2*n*cmax):.6f} alpha={alpha:.1f} noimp={no_improve}")

    # Final check
    c_final = _autocorr(f)
    cmax_final = float(c_final.max())
    if cmax_final < best_cmax:
        best_cmax = cmax_final
        best_f = f.copy()
        best_score = float(2.0 * n * best_cmax)

    print(f"[{tag}] end: best={best_score:.6f}")
    return _normalize(best_f), best_score


def _targeted_peak_reduction(f, c, n, rng):
    """
    Targeted moves to reduce the dominant convolution peak.
    For the peak lag k*, transfer mass from positions i where a[i]*a[k*-i] is large
    to positions j where a[j]*a[k*-j] is small.
    """
    f = _normalize(f)
    L = c.size
    peak_idx = int(np.argmax(c))
    peak_val = float(c[peak_idx])

    # Compute the "contribution vector" for the peak lag
    # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
    # Contribution of position i to c[peak_idx] is a[i] * a[peak_idx - i]
    contrib = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contrib[i] = f[i] * f[j]

    # Find top contributors (positions with high contribution to the peak)
    top_n = min(20, n)
    top_idx = np.argpartition(-contrib, top_n)[:top_n]

    # Find low contributors (positions with low contribution)
    bottom_idx = np.argpartition(contrib, top_n)[:top_n]

    best_f = f.copy()
    best_cmax = peak_val
    c_work = c.copy()

    for trial in range(30):
        # Pick a donor (high contributor) and receiver (low contributor)
        i = int(top_idx[rng.integers(0, len(top_idx))])
        j = int(bottom_idx[rng.integers(0, len(bottom_idx))])
        if i == j or f[i] < 1e-12:
            continue

        # Try different transfer fractions
        for frac in (0.05, 0.15, 0.3, 0.5):
            dv = f[i] * frac
            if dv <= 0:
                continue
            f_test = f.copy()
            f_test[i] -= dv
            f_test[j] += dv
            f_test = _normalize(f_test)
            c_test = _autocorr(f_test)
            cmax_test = float(c_test.max())
            if cmax_test < best_cmax:
                best_cmax = cmax_test
                best_f = f_test.copy()
                f = f_test.copy()
                c = c_test

    return _normalize(best_f), float(best_cmax)


def _coordinate_polish(f0, budget_s, rng):
    """
    Coordinate descent polish: try single-transfer moves guided by the
    current convolution structure.
    """
    f = _normalize(f0)
    n = f.size
    if n < 2:
        return f, float(2.0 * n * _autocorr(f).max())

    c = _autocorr(f)
    cmax = float(c.max())
    best_f = f.copy()
    best_cmax = cmax
    best_score = float(2.0 * n * cmax)

    deadline = time.monotonic() + budget_s
    L = c.size

    # Precompute: for each position i, what's its "influence" on the top lags?
    # We'll use the top-10 lags to guide moves.
    no_improve = 0

    while time.monotonic() < deadline and no_improve < 20:
        # Get top lags
        K = min(10, L)
        top_lags = np.argpartition(-c, K)[:K]
        top_lags_sorted = top_lags[np.argsort(-c[top_lags])]

        # Build influence vector: for each position, how much does it contribute
        # to the top lags?
        influence = np.zeros(n)
        for kl in top_lags_sorted:
            kl = int(kl)
            for i in range(n):
                j = kl - i
                if 0 <= j < n:
                    # Marginal effect of increasing a[i] on c[kl]
                    influence[i] += f[j]
                    influence[j] += f[i]

        # Donors: positions with high influence (reducing them helps the most)
        # Receivers: positions with low influence (adding mass has less cost)
        D = min(15, n // 2)
        R = min(15, n // 2)

        donor_set = set(np.argsort(-influence)[:D].tolist())
        receiver_candidates = np.argsort(influence)[:R * 2]

        receivers = []
        for idx in receiver_candidates:
            if int(idx) not in donor_set:
                receivers.append(int(idx))
                if len(receivers) >= R:
                    break

        if not donor_set or not receivers:
            no_improve += 1
            break

        best_move = None
        best_move_cmax = cmax

        # Try transfers
        for i in donor_set:
            if f[i] < 1e-12:
                continue
            for j in receivers:
                for frac in (0.03, 0.1, 0.3):
                    dv = f[i] * frac
                    f_test = f.copy()
                    f_test[i] -= dv
                    f_test[j] += dv
                    f_test = _normalize(f_test)
                    c_test = _autocorr(f_test)
                    cmax_test = float(c_test.max())
                    if cmax_test < best_move_cmax:
                        best_move_cmax = cmax_test
                        best_move = (i, j, dv)

        if best_move is not None:
            i, j, dv = best_move
            f[i] -= dv
            f[j] += dv
            f = _normalize(f)
            c = _autocorr(f)
            cmax = float(c.max())
            no_improve = 0
            if cmax < best_cmax:
                best_cmax = cmax
                best_f = f.copy()
                best_score = float(2.0 * n * cmax)
        else:
            no_improve += 1
            # Try random moves
            improved = False
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] < 1e-12:
                    continue
                dv = f[i] * float(rng.uniform(0.02, 0.2))
                f_test = _normalize(f.copy())
                f_test[i] -= dv
                f_test[j] += dv
                f_test = _normalize(f_test)
                c_test = _autocorr(f_test)
                cmax_test = float(c_test.max())
                if cmax_test < cmax:
                    f = f_test
                    c = c_test
                    cmax = cmax_test
                    improved = True
                    if cmax < best_cmax:
                        best_cmax = cmax
                        best_f = f.copy()
                        best_score = float(2.0 * n * cmax)
                    break
            if improved:
                no_improve = 0

    return _normalize(best_f), best_score


def _interp_to_length(f, n_new):
    f = _normalize(f)
    n_old = f.size
    if n_old == n_new:
        return f
    if n_old == 1:
        return _normalize(np.full(n_new, f[0]))
    if n_new == 1:
        return np.array([1.0])
    x_old = np.linspace(0.0, n_old - 1, n_old)
    x_new = np.linspace(0.0, n_new - 1, n_new)
    return _normalize(np.interp(x_new, x_old, f))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(100.0, float(budget_s)) - 30.0

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    f0 = _normalize(f0)
    n = f0.size

    print(f"Initial sequence length: {n}")
    init_score = _fast_score(f0)
    print(f"Initial score: {init_score:.6f}")

    best_f = f0.copy()
    best_score = init_score

    # Build diverse starting points
    starts = []
    starts.append(("given", f0))
    starts.append(("reversed", _normalize(f0[::-1])))

    # Smoothed versions
    for win in (7, 21, 51, 101):
        kernel = np.ones(win) / win
        sm = _normalize(np.convolve(f0, kernel, mode='same'))
        starts.append((f"smooth{win}", sm))

    # Gamma transforms
    for g in (0.5, 0.7, 1.5, 2.0):
        gt = _normalize(np.power(f0 + 1e-10, g))
        starts.append((f"gamma{g}", gt))

    # Random structured starts
    for i in range(6):
        starts.append((f"random{i}", _random_restart(n, rng, base=f0)))

    # Evaluate all starts
    scored_starts = []
    for name, sf in starts:
        sf = _normalize(sf)
        sc = _fast_score(sf)
        if np.isfinite(sc):
            scored_starts.append((sc, name, sf))

    scored_starts.sort()
    print(f"\nTop starting points:")
    for sc, name, _ in scored_starts[:5]:
        print(f"  {name}: {sc:.6f}")

    # Phase 1: Short gradient optimization from top starts
    phase1_deadline = time.monotonic() + 480.0
    print(f"\n=== Phase 1: Initial optimization (deadline at +480s) ===")

    top_starts = scored_starts[:min(4, len(scored_starts))]
    for idx, (sc, name, sf) in enumerate(top_starts):
        if time.monotonic() >= phase1_deadline - 30:
            break
        remaining = phase1_deadline - time.monotonic() - 10
        alloc = min(120.0, remaining * 0.4)
        if alloc < 10:
            break
        print(f"\n--- Phase1 #{idx+1}: {name} (score={sc:.6f}) alloc={alloc:.0f}s ---")
        f_opt, sc_opt = _gradient_optimize(sf, alloc, rng, tag=f"P1_{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 2: Deeper optimization from the best found
    phase2_deadline = time.monotonic() + 900.0
    print(f"\n=== Phase 2: Deep optimization (deadline at +1380s total) ===")

    # Multiple restarts from best with perturbations
    for restart in range(3):
        if time.monotonic() >= phase2_deadline - 60:
            break
        alloc = min(300.0, phase2_deadline - time.monotonic() - 30)
        if alloc < 30:
            break

        # Perturb the best
        if restart == 0:
            start_f = best_f.copy()
        elif restart == 1:
            start_f = _spectral_perturb(best_f, rng, strength=0.3)
        else:
            start_f = _normalize(best_f * np.exp(0.5 * rng.standard_normal(n)))

        tag = f"P2_r{restart}"
        print(f"\n--- Phase2 #{restart+1} alloc={alloc:.0f}s ---")
        f_opt, sc_opt = _gradient_optimize(start_f, alloc, rng, tag=tag)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 3: Coordinate polish
    phase3_deadline = time.monotonic() + 600.0
    print(f"\n=== Phase 3: Coordinate polish (total deadline +1980s) ===")

    for polish_round in range(2):
        if time.monotonic() >= phase3_deadline - 30:
            break
        alloc = min(300.0, phase3_deadline - time.monotonic() - 20)
        if alloc < 20:
            break
        print(f"\n--- Polish round {polish_round+1} alloc={alloc:.0f}s ---")
        f_pol, sc_pol = _coordinate_polish(best_f, alloc, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 4: Final refinement - short gradient + polish
    print(f"\n=== Phase 4: Final refinement ===")
    remaining = max(0.0, total_deadline - time.monotonic() - 20.0)
    if remaining > 60:
        grad_alloc = min(120.0, remaining * 0.5)
        if grad_alloc > 20:
            print(f"\n--- Final gradient alloc={grad_alloc:.0f}s ---")
            f_opt, sc_opt = _gradient_optimize(best_f, grad_alloc, rng, tag="final_grad")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()

        polish_alloc = max(0.0, total_deadline - time.monotonic() - 20.0)
        if polish_alloc > 15:
            print(f"\n--- Final polish alloc={polish_alloc:.0f}s ---")
            f_pol, sc_pol = _coordinate_polish(best_f, polish_alloc, rng)
            if sc_pol < best_score:
                best_score = sc_pol
                best_f = f_pol.copy()

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))

    # Final evaluation
    final_score = _fast_score(best_f)
    if final_score < best_score:
        best_score = final_score

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official):
            print(f"\nOfficial score: {official:.6f}")
            if official < best_score:
                best_score = official
    except Exception as e:
        print(f"Official eval error: {e}")

    out = [float(x) for x in best_f]
    print(f"\nFINAL score: {best_score:.6f}")
    print(f"TARGET: {TARGET_SCORE}")
    return out
