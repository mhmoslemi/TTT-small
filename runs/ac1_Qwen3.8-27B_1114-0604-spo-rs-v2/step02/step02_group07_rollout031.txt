"""
Optimized step function search using normalized Adam with annealed penalty,
parametric initialization, and multi-restart strategy.
"""

import numpy as np
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_from(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def soft_grad(a, c, topk, temp_frac=0.1):
    """Compute soft-max adaptive gradient (similar to previous but improved)."""
    n = len(a)
    S = a.sum()
    S2 = S * S
    S3 = S2 * S
    k = min(topk, c.size)
    
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx].copy()
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    
    top1 = vals[0]
    if top1 < 1e-15:
        return np.zeros(n), top1, S
    
    # Temperature
    temp = max(temp_frac * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts /= wts.sum()
    M_wt = float(np.dot(vals, wts))
    
    # Gradient
    g = np.zeros(n)
    for i in range(k):
        wi = wts[i]
        if wi < 1e-15:
            continue
        ki = int(idx[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        g[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    # Full gradient of 2n * M_wt / S^2
    grad = 2.0 * n * (g / S2 - 2.0 * M_wt / S3)

"""
Optimized step function search combining normalized Adam with annealed soft-max gradient,
frequency-domain peak analysis, parametric initialization, and adaptive multi-restart strategy.
Uses log-sum-exp smooth approximation with adaptive temperature and top-k selection.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from(n, c, S):
    """Compute objective 2n*max(c)/S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def soft_max_gradient(a, c, topk, temp_frac):
    """
    Compute gradient of 2n * M_wt / S^2 where M_wt is soft-max weighted peak.
    Returns (gradient, M_wt, S).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k = min(topk, c.size)

    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx].copy()
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]

    top1 = float(vals[0])
    if top1 < 1e-15:
        return np.zeros(n), top1, S

    temp = max(temp_frac * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts_sum = wts.sum()
    if wts_sum < 1e-30:
        wts = np.zeros(k)
        wts[0] = 1.0
    else:
        wts /= wts_sum
    M_wt = float(np.dot(vals, wts))

    g = np.zeros(n)
    for i in range(k):
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        ki = int(idx[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        g[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    grad = 2.0 * n * (g / S2 - 2.0 * M_wt / S3)
    return grad, M_wt, S


def fft_gradient(a, c, threshold):
    """
    FFT-based gradient for sum_k max(0, c[k]-threshold)^2 objective.
    More efficient when many peaks are active.
    """
    n = len(a)
    w = np.where(c > threshold, 2.0 * (c - threshold), 0.0)
    if w.sum() < 1e-15:
        return np.zeros(n)
    M_fft = 1 << (w.size + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    A = np.fft.rfft(a, M_fft)
    h = np.fft.irfft(W * np.conj(A), M_fft)[:n]
    return 2.0 * h

            a = np.clip(a, 0.0, 1000)
            if a.sum() < 1e-3:
                a = np.ones(n)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            cur_pat = max(40, int(cur_pat * patience_scale))
            continue

        progress = min(1.0, (time.monotonic() - t_start) / dur)
        topk = int(topk0 + (topk_end - topk0) * progress)
        temp = temp0 + (temp_end - temp0) * (progress ** 0.7)

        if use_fft_grad:
            threshold = float(np.max(c)) * (1.0 - 0.05 * progress)
            g = fft_gradient(a, c, threshold)
            S = float(a.sum())
            g = g - g.mean()
        else:
            g, M_wt, S = soft_max_gradient(a, c, topk, temp)

        t += 1
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)

        lr = lr0 * ((temp_end / temp0) ** (2.0 * progress))
        a = a - lr * mhat / (np.sqrt(vhat) + 1e-8)
        a = np.clip(a, 0.0, 1000)

    return best_sc, best_a


def parametric_sequence(n, harmonics, rng):
    """Generate sequence as 1 + sum of cosines with random params."""
    a = np.ones(n)
    x = np.arange(n, dtype=np.float64)
    for i in range(harmonics):
        k = i + 1
        alpha = rng.uniform(-0.6, 0.6)
        phi = rng.uniform(0, 2.0 * np.pi)
        a += alpha * np.cos(2.0 * np.pi * k * x / n + phi)
    return np.clip(a, 0.01, 1000.0)


def diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = [np.ones(n)]
    x = np.linspace(0, 1, n)
    inits.append(np.clip(1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.7), 0.01, None))
    inits.append(np.clip(1.0 + 0.5 * np.cos(2 * np.pi * 3 * x + 1.2) + 0.15 * np.cos(2 * np.pi * 7 * x), 0.01, None))
    inits.append(np.clip(1.0 + rng.standard_normal(n) * 0.4, 0.01, None))
    g = np.ones(n)
    g[1::2] = 0.4
    inits.append(g)
    g2 = np.ones(n)
    g2[2::3] = 0.3
    inits.append(g2)
    for _ in range(4):
        inits.append(parametric_sequence(n, 5, rng))
    # Multi-scale
    a_ms = np.ones(n)
    a_ms += 0.3 * np.sin(2 * np.pi * x) * np.cos(2 * np.pi * 5 * x)
    inits.append(np.clip(a_ms, 0.01, None))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses multi-phase optimization: diverse init, Adam with annealing,
    FFT-gradient refinement, and fine polish.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0
    total_time = budget_s - 10.0

    # Determine n from parent
    n = 4096
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 512:
                n = int(parent.size)
    except Exception:
        pass
    n = min(max(1024, n), 12000)

    print(f"Config: n={n}, budget={budget_s:.0f}s, parent={'Y' if parent is not None else 'N'}")

    best_score = np.inf
    best_a = None

    # --- Phase 1: Diverse initialization evaluation (5% time) ---
    p1_end = t0 + 0.05 * total_time
    candidates = diverse_inits(n, rng)
    if parent is not None and parent.size == n:
        candidates.insert(0, np.clip(parent, 0.0, 1000.0))

    for i, cand in enumerate(candidates):
        if time.monotonic() > p1_end:
            break
        c = fast_self_conv(cand)
        s = score_from(n, c, float(cand.sum()))
        if s < best_score:
            best_score = s
            best_a = cand.copy()
    print(f"Phase 1 (init): {best_score:.6f}")

    # --- Phase 2: Main Adam optimization (45% time) ---
    p2_end = t0 + 0.50 * total_time
    if best_a is not None:
        sc, aa = adam_optimize(best_a, p2_end, rng, lr0=0.12, topk0=6,
                               topk_end=40, temp0=0.25, temp_end=0.04, patience=350)
        if sc < best_score:
            best_score = sc
            best_a = aa
        print(f"Phase 2 (main): {best_score:.6f}")

    # --- Phase 3: FFT-gradient refinement (25% time) ---
    p3_end = t0 + 0.75 * total_time
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = adam_optimize(best_a, p3_end, rng, lr0=0.04, topk0=16,
                               topk_end=64, temp0=0.1, temp_end=0.02,
                               patience=400, use_fft_grad=True)
        if sc < best_score:
            best_score = sc
            best_a = aa
        print(f"Phase 3 (fft-grad): {best_score:.6f}")

    # --- Phase 4: Fine polish (15% time) ---
    p4_end = t0 + 0.90 * total_time
    if best_a is not None and time.monotonic() < p4_end:
        sc, aa = adam_optimize(best_a, p4_end, rng, lr0=0.008, topk0=32,
                               topk_end=48, temp0=0.05, temp_end=0.008,
                               patience=600)
        if sc < best_score:
            best_score = sc
            best_a = aa
        print(f"Phase 4 (polish): {best_score:.6f}")

    # --- Phase 5: Ultra-fine final (5% time) ---
    if best_a is not None and time.monotonic() < deadline:
        sc, aa = adam_optimize(best_a, deadline, rng, lr0=0.002, topk0=48,
                               topk_end=64, temp0=0.03, temp_end=0.003,
                               patience=800)
        if sc < best_score:
            best_score = sc
            best_a = aa
        print(f"Phase 5 (ultra): {best_score:.6f}")

    # Final output
    if best_a is None:
        best_a = np.ones(n)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, time={time.monotonic()-t0:.1f}s")
    return out
