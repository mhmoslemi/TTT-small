```python
def _solve_lp(f, M):
    """Solve the LP: max 1^T b s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    # Build the constraint matrix A (2n-1 constraints, n variables)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Using sparse matrix
    from scipy.sparse import lil_matrix
    
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    
    A_csr = A.tocsr()
    
    # Maximize 1^T b => Minimize -1^T b
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_csr, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None
```

```python
"""
Step function optimizer using multi-phase mirror descent with backtracking,
momentum, adaptive top-k subgradient, and coordinate descent polish.
"""

import numpy as np
import time

def _convolve(f):
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _score_from_conv(n, c):
    return c.max()  # We'll normalize sum to 1, so score = 2*n*max(c)

def _subgrad_topk(f, c, k):
    """Subgradient from top-k lags of c."""
    n = len(f)
    top = np.argpartition(c, -k)[-k:]
    grad = np.zeros(n)
    for idx in top:
        lo = max(0, idx - n + 1)
        hi = min(n, idx + 1)
        # d conv(f,f)[idx] / d f[i] = 2*f[idx-i] for i in [lo, hi)
        # (as shown above)
        grad[lo:hi] += 2.0 * f[idx - np.arange(lo, hi)]
    return grad / k

def _mirror_step(f, grad, eta):
    """Exponential gradient update on the simplex."""
    # f_new ∝ f * exp(-eta * (grad - grad.mean()))
    # Centering helps with numerical stability
    centered_grad = grad - grad.mean()
    log_f = np.log(f + 1e-300) - eta * centered_grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def _optimize(f0, deadline, rng, n, verbose=False):
    """Mirror descent with backtracking, momentum, and adaptive top-k."""
    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    c = _convolve(f)
    best_peak = c.max()
    best_f = f.copy()
    
    eta = 1.0  # Initial step size
    eta_min = 1e-10
    momentum = 0.7
    grad_ema = np.zeros(n)
    
    last_improve = time.monotonic()
    iters = 0
    n_accepted = 0
    n_rejected = 0
    
    while time.monotonic() < deadline:
        c = _convolve(f)
        peak = c.max()
        
        if peak < best_peak - 1e-15:
            best_peak = peak
            best_f = f.copy()
            last_improve = time.monotonic()
        
        # Adaptive top-k
        progress = min(1.0, iters / 50000.0)
        k = 2 + int(6 * progress)
        
        # Compute subgradient
        grad = _subgrad_topk(f, c, k)
        
        # Momentum
        grad_ema = momentum * grad_ema + (1 - momentum) * grad
        
        # Backtracking line search
        step_eta = min(eta, 2.0)  # Cap the step
        improved = False
        for _ in range(15):
            f_new = _mirror_step(f, grad_ema, step_eta)
            c_new = _convolve(f_new)
            new_peak = c_new.max()
            if new_peak < peak:
                f = f_new
                improved = True
                n_accepted += 1
                break
            step_eta *= 0.5
        
        if not improved:
            n_rejected += 1
            # If we can't improve with any step, try a perturbation
            stuck_time = time.monotonic() - last_improve
            if stuck_time > 10:
                # Random perturbation
                mag = 0.01 + 0.02 * min(1.0, stuck_time / 60.0)
                f = f + mag / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                last_improve = time.monotonic()
                # Reset momentum
                grad_ema *= 0.5
                eta = max(eta * 0.5, eta_min)
            else:
                eta = max(eta * 0.9, eta_min)
        else:
            # Slightly increase eta if we're making progress
            if peak < best_peak:
                eta = min(eta * 1.01, 3.0)
        
        iters += 1
        if verbose and iters % 50000 == 0:
            print(f"  iter {iters}: peak={peak:.10f}, best={best_peak:.10f}, "
                  f"eta={eta:.4f}, acc={n_accepted}, rej={n_rejected}")
    
    # Final check
    c = _convolve(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak

def _polish(f0, deadline, rng, n):
    """Coordinate descent polish."""
    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    c = _convolve(f)
    best_peak = c.max()
    best_f = f.copy()
    
    iters = 0
    while time.monotonic() < deadline:
        c = _convolve(f)
        peak = c.max()
        
        if peak < best_peak - 1e-15:
            best_peak = peak
            best_f = f.copy()
        
        # Find peak lag
        k_star = int(np.argmax(c))
        
        # Compute contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        contrib[lo:hi] = 2.0 * f[k_star - np.arange(lo, hi)]
        
        # Strategy: move mass from high-contrib to low-contrib positions
        # Try moving from the top contributor to various recipients
        donor = int(np.argmax(contrib))
        
        # Also consider all near-peak lags
        top_lags = np.argpartition(c, -3)[-3:]
        all_contrib = np.zeros(n)
        for kl in top_lags:
            lo_k = max(0, kl - n + 1)
            hi_k = min(n, kl + 1)
            all_contrib[lo_k:hi_k] += 2.0 * f[kl - np.arange(lo_k, hi_k)]
        
        donor = int(np.argmax(all_contrib))
        low_positions = np.where(all_contrib < np.percentile(all_contrib, 10))[0]
        
        improved = False
        # Try various fractions and recipients
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            for recv in low_positions[:5]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                if amt < 1e-16:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try /= f_try.sum()
                c_try = _convolve(f_try)
                if c_try.max() < peak - 1e-14:
                    f = f_try
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random moves
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = f[i] * frac
                if amt < 1e-16:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _convolve(f_try)
                if c_try.max() < peak - 1e-14:
                    f = f_try
                    break
        
        iters += 1
    
    c = _convolve(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak

def _generate_starts(f_init, n, rng, count=30):
    """Generate diverse starting points."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    norm_j = (j - center) / center  # Range [-1, 1]
    
    # Gaussian family
    for sigma in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5]:
        starts.append(np.exp(-0.5 * (norm_j / sigma) ** 2))
    
    # Polynomial tapers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        starts.append(np.maximum(0, 1 - norm_j**p + 0.01))
    
    # Raised cosine
    for beta in [0.5, 1.0, 2.0, 3.0]:
        starts.append(0.5 * (1 + np.cos(np.pi * norm_j / beta)))
    
    # Sine-based
    for freq in [1, 2, 3, 5, 8]:
        starts.append(1 + 0.5 * np.sin(2 * np.pi * freq * j / n))
        starts.append(1 + 0.8 * np.cos(2 * np.pi * freq * j / n))
    
    # Inverted (U-shaped)
    for alpha in [1, 2, 5]:
        starts.append(1 + alpha * norm_j**2)
    
    # Frequency domain construction
    # Try to make the spectrum flat
    for _ in range(3):
        size = 1 << (n - 1).bit_length()
        phases = rng.standard_normal(size // 2 + 1)
        F = np.exp(1j * phases)
        F[0] = 1.0
        # Ensure conjugate symmetry
        F_full = np.zeros(size, dtype=complex)
        F_full[:size//2+1] = F
        F_full[size//2+1:] = np.conj(F[1:size//2][::-1])
        f_candidate = np.fft.irfft(F_full, size)[:n]
        f_candidate = np.maximum(f_candidate, 0)
        if f_candidate.sum() > 0:
            starts.append(f_candidate)
    
    # Random starts
    while len(starts) < count:
        r = rng.random()
        if r < 0.25:
            starts.append(rng.exponential(1.0, n))
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.75:
            # Bimodal
            pos = rng.uniform(0.2, 0.4)
            starts.append(
                np.exp(-((j - pos*n) / (0.05*n))**2) +
                np.exp(-((j - (1-pos)*n) / (0.05*n))**2) + 0.001
            )
        else:
            starts.append(np.ones(n) + 0.5 * rng.standard_normal(n))
    
    # Normalize and ensure non-negative
    result = []
    for s in starts:
        s = np.maximum(s, 0)
        if s.sum() > 1e-10:
            result.append(s / s.sum())
    return result
```

```python
def _generate_starts(f_init, n, rng, count=35):

        starts.append(np.exp(-0.5 * (norm_j / sigma) ** 2))

    for p in [0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0]:
        val = np.maximum(0.0, 1.0 - np.abs(norm_j) ** p)

    # Raised cosine variants
    for beta in [0.5, 0.7, 1.0, 1.5, 2.0]:
        starts.append(0.5 * (1 + np.cos(np.pi * np.clip(norm_j / beta, -1, 1))))

    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.6, 0.9]:

    # U-shaped (inverted)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        starts.append(1.0 + alpha * norm_j ** 2)

    # Bimodal

        for width in [0.03, 0.05, 0.08]:
            starts.append(
                np.exp(-((j - pos * n) / (width * n)) ** 2) +
                np.exp(-((j - (1 - pos) * n) / (width * n)) ** 2) + 0.001
            )

    # Triangular
    starts.append(1.0 - np.abs(norm_j))

    # Frequency domain: try flat spectrum
    for trial in range(4):
        size_fft = 1 << (n).bit_length()

            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.6:
            # Smooth random: low-pass filtered noise
            raw = rng.standard_normal(n)
            freqs = np.fft.rfftfreq(n)
            Mask = np.exp(-10 * freqs / (n / (2 * n)))
            spec = np.fft.rfft(raw) * Mask
            smooth = np.fft.irfft(spec, n)
            starts.append(np.abs(smooth) + 0.1)
        elif r < 0.8:
            # Piecewise constant (step-like)
            n_steps = rng.integers(3, 15)
            edges = np.sort(rng.uniform(0, 1, n_steps))
            vals = rng.uniform(0.1, 1.0, n_steps)
            steps = np.zeros(n)
            for k in range(n_steps):
                lo_k = int(edges[k] * n) if k > 0 else 0
                hi_k = int(edges[k + 1] * n) if k < n_steps - 1 else n
                steps[lo_k:hi_k] = vals[k]
            starts.append(steps)
        else:
            starts.append(np.ones(n) + 0.5 * rng.standard_normal(n))

    # Normalize
    result = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        s = np.maximum(s, 0)
        if s.sum() > 1e-10:
            result.append(s / s.sum())
    return result


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main optimization function. Searches for non-negative sequence minimizing
    2*n*max(conv(f,f))/(sum(f))^2 using iterated mirror descent with SA,
    diverse initializations, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 12.0

    # Load initial sequence
    f_init = np.asarray(height_sequence_1, dtype=np.float64)
    n = len(f_init)
    f_init = np.clip(f_init, 0.0, 1000.0)
    if f_init.sum() < 1e-10:
        f_init = np.ones(n) / n
    else:
        f_init = f_init / f_init.sum()

    print(f"n = {n}")
    c0 = _convolve(f_init)
    init_score = 2.0 * n * c0.max()
    print(f"Initial score: {init_score:.6f}")

    # Generate and score starts
    t0 = time.monotonic()
    starts = _generate_starts(f_init, n, rng, count=40)
    scored = []
    for s in starts:
        c = _convolve(s)
        scored.append((c.max(), s))
    scored.sort(key=lambda x: x[0])
    gen_time = time.monotonic() - t0
    print(f"Generated {len(starts)} starts in {gen_time:.1f}s")
    print(f"Top 10 peaks: {[f'{p:.8f}' for p, _ in scored[:10]]}")

    best_ever_peak = scored[0][0]
    best_ever_f = scored[0][1].copy()
    best_ever_score = 2.0 * n * best_ever_peak
    print(f"Best initial: {best_ever_score:.6f}")

    # Phase 1: Optimize top starts with ILS
    # Allocate: 55% for ILS, 25% for polish, 20% for final
    ils_budget = budget_s * 0.52
    polish_budget = budget_s * 0.22
    num_starts = 6
    time_per_start = ils_budget / num_starts

    for idx in range(num_starts):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        alloc = min(time_per_start, remaining - polish_budget - 30)
        if alloc < 10:
            break

        peak0, f0 = scored[idx]
        # Skip if this start is much worse than best (save time)
        if peak0 > best_ever_peak * 1.05 and idx > 0:
            print(f"  Skipping start {idx+1} (peak={peak0:.8f} much worse than best)")
            continue

        print(f"\nILS on start {idx+1}/{num_starts} (peak={peak0:.8f}), budget={alloc:.0f}s")
        t_start = time.monotonic()
        f_opt, score = _optimize_ils(f0, t_start + alloc, rng, n)

        if score < best_ever_score:
            best_ever_score = score
            best_ever_peak = score / (2.0 * n)
            best_ever_f = f_opt.copy()
            print(f"  *** IMPROVED to {best_ever_score:.6f} ***")
        else:
            print(f"  Best from this start: {score:.6f}")

    # Phase 2: Coordinate polish
    remaining = deadline - time.monotonic()
    pol_time = min(polish_budget, remaining * 0.4)
    if pol_time > 10:
        print(f"\nPolishing best solution (score={best_ever_score:.6f}, time={pol_time:.0f}s)")
        f_pol, pol_score = _polish(best_ever_f, time.monotonic() + pol_time, rng, n)
        if pol_score < best_ever_score:
            best_ever_score = pol_score
            best_ever_peak = pol_score / (2.0 * n)
            best_ever_f = f_pol.copy()
            print(f"  *** Polished to {best_ever_score:.6f} ***")
        else:
            print(f"  Polish best: {pol_score:.6f}")

    # Phase 3: Final ILS round from best
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\nFinal ILS round (time={remaining:.0f}s)")
        f_fin, fin_score = _optimize_ils(best_ever_f, deadline, rng, n)
        if fin_score < best_ever_score:
            best_ever_score = fin_score
            best_ever_peak = fin_score / (2.0 * n)
            best_ever_f = f_fin.copy()
            print(f"  *** Final ILS: {best_ever_score:.6f} ***")

    # Final polish if time remains
    remaining = deadline - time.monotonic()
    if remaining > 10:
        f_p2, p2_score = _polish(best_ever_f, deadline, rng, n)
        if p2_score < best_ever_score:
            best_ever_score = p2_score
            best_ever_f = f_p2.copy()

    # Verify with official evaluator
    best_ever_f = np.clip(best_ever_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_ever_f.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        print("Falling back to uniform!")
        best_ever_f = np.ones(n)
        final_score = evaluate_sequence(best_ever_f.tolist())

    print(f"=== FINAL SCORE: {final_score:.6f} ===")
    return [float(x) for x in best_ever_f]
```
