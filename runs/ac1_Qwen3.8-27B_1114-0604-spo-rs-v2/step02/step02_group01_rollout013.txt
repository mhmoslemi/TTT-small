import numpy as np
import time

def _convolve_fft(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    z = _convolve_fft(a)
    return float(2.0 * n * z.max() / (s * s))

def _compute_gradient(a, z, T):
    """Compute gradient of T*logsumexp(z/T) w.r.t. a."""
    n = len(a)
    # Softmax weights
    z_scaled = z / T
    z_scaled -= z_scaled.max()
    w = np.exp(z_scaled)
    w /= w.sum()
    # g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    g = np.correlate(w, a, mode='valid')
    return g

def _mirror_step(a, g, eta, velocity=None, beta=0.9):
    """Mirror descent step with momentum."""
    if velocity is None:
        velocity = np.zeros_like(a)
    
    # Update velocity
    velocity = beta * velocity + (1 - beta) * g
    
    # Center and scale
    v_centered = velocity - velocity.mean()
    v_scale = np.abs(v_centered).max()
    if v_scale < 1e-30:
        return a, velocity
    v_norm = v_centered / v_scale
    
    # Mirror step
    a_new = a * np.exp(-eta * v_norm)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-10:
        return a, velocity
    a_new /= s
    return a_new, velocity

def _optimize(a0, deadline, label="", verbose=True):
    """Main optimization loop."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
    else:
        a /= s
    n = len(a)
    
    best_score = _score_fast(a)
    best_a = a.copy()
    
    T = 0.05  # initial temperature (will be tuned)
    eta = 0.3  # initial step size
    velocity = np.zeros(n)
    beta = 0.85
    
    n_iters = 100000
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Progress-based parameter scheduling
        progress = it / n_iters
        T = 0.05 * (0.01 / 0.05) ** (progress ** 0.5)  # anneal T
        T = max(T, 0.001)
        eta = 0.3 * (0.01 / 0.3) ** (progress ** 0.3)  # decrease step
        eta = max(eta, 0.005)
        
        # Compute convolution
        z = _convolve_fft(a)
        z_max = z.max()
        
        # Compute gradient
        g = _compute_gradient(a, z, T)
        
        # Mirror step
        a, velocity = _mirror_step(a, g, eta, velocity, beta)
        
        # Periodic evaluation
        if it % 1000 == 0 or it == n_iters - 1:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    {label} it={it} score={score:.7f} T={T:.4f} eta={eta:.4f} t={elapsed:.0f}s")
    
    # Final
    score = _score_fast(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

    # For D[j]: D[j] = sum_{m=0}^{N-1} a[m] * w[(m+j) mod N]
    # Since a is zero for m >= n: D[j] = sum_{m=0}^{n-1} a[m] * w[m+j] (assuming m+j < N)
    # Since j <= n-1 and m <= n-1: m+j <= 2n-2 < N. So D[j] = sum_{m=0}^{n-1} a[m] * w[m+j] = g[j]!
    
    # So g = D[0:n] where D = IFFT(FFT(a, N) * conj(FFT(w, N)))
    
    D = np.fft.irfft(np.fft.rfft(a, N) * np.conj(np.fft.rfft(w, N)), N)
    g = D[:n]
    return g

```python
"""
Step function optimizer using softmax-approximated mirror descent.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 by:
1. Approximating max with log-sum-exp (temperature annealing)
2. Using mirror descent (exponentiated gradient) on the simplex
3. Multi-start with diverse initializations
4. FFT-accelerated gradient computation
5. Momentum and adaptive step sizing
6. Final coordinate-descent polish
"""

import numpy as np
import time

def _fft_size(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _convolve_fft(a, N=None):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if N is None:
        N = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]

def _score_fast(a):
    """Fast score computation."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    N = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, N)
    z = np.fft.irfft(A * A, N)[:2 * n - 1]
    return float(2.0 * n * z.max() / (s * s))

def _gradient_softmax(a, z, T, N, A_cached=None):
    """
    Compute gradient of T*logsumexp(z_norm/T) w.r.t. a using softmax weights.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Normalize z to [0, 1]
    z_min = z.min()
    z_max = z.max()
    z_range = z_max - z_min
    if z_range < 1e-15:
        return np.full(n, 1.0 / n)  # uniform gradient (flat z)
    
    z_norm = (z - z_min) / z_range
    z_scaled = z_norm / T
    z_scaled -= z_scaled.max()
    w = np.exp(z_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(z)) / len(z)
    else:
        w /= w_sum
    
    # g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # Via FFT: g = IFFT(A * conj(W))[:n]
    if A_cached is None:
        A = np.fft.rfft(a, N)
    else:
        A = A_cached
    W = np.fft.rfft(w, N)
    g = np.fft.irfft(A * np.conj(W), N)[:n]
    return g

def _mirror_descent_step(a, g, eta, velocity, beta):
    """Mirror descent step with momentum."""
    n = len(a)
    # Update velocity
    velocity[:] = beta * velocity + (1.0 - beta) * g
    
    # Center and normalize
    v = velocity - velocity.mean()
    v_max = np.abs(v).max()
    if v_max < 1e-30:
        return a, velocity
    v = v / v_max
    
    # Exponentiated step
    log_ratio = -eta * v
    # Clip for numerical stability
    log_ratio = np.clip(log_ratio, -50, 50)
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-10:
        return a, velocity
    a_new /= s
    return a_new, velocity

def _optimize_softmax(a0, deadline, label="", verbose=True):
    """
    Main optimization: softmax mirror descent with temperature annealing.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
    else:
        a /= s
    n = len(a)
    
    N = _fft_size(3 * n - 2)  # FFT size for gradient computation
    # N should be >= 3n-2 for the cross-correlation to work without aliasing
    
    best_score = _score_fast(a)
    best_a = a.copy()
    
    velocity = np.zeros(n)
    beta = 0.8
    
    # Parameters
    T_start = 0.5
    T_end = 0.01
    eta_start = 0.5
    eta_end = 0.01
    n_planned = 50000
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    patience = 5000
    
    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / n_planned)
        
        # Annealing schedule
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.5)
        eta = max(eta, eta_end)
        
        # Compute convolution
        A = np.fft.rfft(a, N)
        z = np.fft.irfft(A * A, N)[:2 * n - 1]
        z_max = z.max()
        
        # Compute gradient
        g = _gradient_softmax(a, z, T, N, A_cached=A)
        
        # Mirror descent step
        a, velocity = _mirror_descent_step(a, g, eta, velocity, beta)
        
        # Periodic evaluation
        if it % 2000 == 0 or it == 1:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stagnant = 0
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    {label} it={it} score={score:.7f} T={T:.4f} eta={eta:.4f} t={elapsed:.0f}s")
            else:
                stagnant += 1
            
            # If very stagnant, perturb
            if stagnant > patience * 3:
                # Random perturbation
                a_pert = a + 0.01 * np.random.randn(n) * a.mean()
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                a = a_pert
                velocity *= 0.1
                stagnant = 0
    
    # Final evaluation
    score = _score_fast(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _coordinate_polish(a, deadline, label="", verbose=True):
    """
    Coordinate descent: find the peak convolution lag, then try to reduce it
    by moving mass from high-contributing positions to low-contributing ones.
    """
    a = a.copy()
    n = len(a)
    best_score = _score_fast(a)
    best_a = a.copy()
    
    N = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, N)
    z = np.fft.irfft(A * A, N)[:2 * n - 1]
    z_max = z.max()
    k_peak = z.argmax()
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Find top 3 peak lags
        top3 = np.argpartition(z, -3)[-3:]
        top3 = top3[np.argsort(z[top3])[::-1]]
        
        # For each peak lag, compute contribution of each position
        # z[k] = sum_i a[i]*a[k-i], so dz[k]/da[j] = a[k-j] (valid)
        best_improvement = None
        
        for k in top3:
            # Contribution of position j to z[k]: a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            contrib = a[k - np.arange(j_start, j_end + 1)]
            
            # Total contribution across all top lags
            total_contrib = np.zeros(n)
            for k2 in top3:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    total_contrib[js2:je2+1] += a[k2 - np.arange(js2, je2 + 1)]
            
            # Find best position to reduce (high total contrib) and best to increase (low)
            j_don = int(np.argmax(total_contrib))
            j_rec = int(np.argmin(total_contrib))
            
            if j_don == j_rec:
                continue
            
            # Try several transfer amounts
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[j_don] * frac
                if amt < 1e-10:

                a_try = a.copy()
                a_try[j_don] -= amt
                a_try[j_rec] += amt
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:

                a_try /= s_try
                
                # Quick check: compute the peak
                A_try = np.fft.rfft(a_try, N)
                z_try = np.fft.irfft(A_try * A_try, N)[:2 * n - 1]
                if z_try.max() < z_max - 1e-10:
                    if best_improvement is None or z_try.max() < best_improvement[0]:
                        best_improvement = (z_try.max(), a_try.copy(), z_try)
        
        if best_improvement is not None:
            z_max, a, z = best_improvement
            stagnant = 0
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose and it % 100 == 0:

        else:

            for _ in range(3):

                a_try[i] -= amt
