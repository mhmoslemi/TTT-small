"""
Subgradient + spectral shaping optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences of length n.
"""

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_conv(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _fft_conv(p, p)
    return float(2.0 * n * c.max())

def _score_with_conv(p):
    """Return (score, convolution) for a normalized non-negative sequence."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf'), None
    p = p / s
    n = len(p)
    c = _fft_conv(p, p)
    mx = c.max()
    return float(2.0 * n * mx), c

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _subgradient_at_lags(p, lags):
    """
    Compute the average subgradient of max(c) at the given lags.
    sg[k] = (1/len(lags)) * sum_{m in lags} 2*p[m-k] (if 0 <= m-k < n)
    """
    n = len(p)
    sg = np.zeros(n)
    for m in lags:
        k_start = max(0, m - n + 1)
        k_end = min(n, m + 1)
        if k_start < k_end:
            idx = m - np.arange(k_start, k_end)
            sg[k_start:k_end] += 2.0 * p[idx]
    sg /= len(lags)
    return sg

def _find_top_lags(c, threshold_frac=0.0, top_k=5):
    """Find lags where c is within threshold_frac of max, or just top_k."""
    mx = c.max()
    if threshold_frac > 0:
        thresh = mx * (1.0 - threshold_frac)
        lags = np.where(c >= thresh)[0].tolist()
    else:
        sorted_idx = np.argsort(c)[::-1][:top_k]
        lags = sorted_idx.tolist()
    return lags

def _multiplicative_step(p, sg, lr):
    """Take a multiplicative gradient step: p * exp(-lr * sg / max(sg))."""
    sg_max = sg.max()
    if sg_max < 1e-15:
        return p.copy()
    sg_norm = sg / sg_max
    p_new = p * np.exp(-lr * sg_norm)
    p_new = np.maximum(p_new, 1e-15 / len(p))
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s

def _spectral_flatten(p, beta, L=None):
    """
    Flatten the power spectrum of p in frequency domain.
    beta in (0,1): how much to flatten (0=no change, 1=fully flat).
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    power = np.abs(A) ** 2
    mean_power = power.mean()
    if mean_power < 1e-15:
        return p.copy()
    # Flatten: power_new = power * (mean_power/power)^beta
    # = power^(1-beta) * mean_power^beta
    eps = 1e-15
    power_new = power * (mean_power / np.maximum(power, eps)) ** beta
    # Take square root to get new magnitude, keep phase
    mag_new = np.sqrt(power_new)
    phase = np.angle(A)
    A_new = mag_new * np.exp(1j * phase)
    p_new = np.fft.irfft(A_new, L)[:n]
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return None
    return p_new / s

def _optimize_subgradient(p0, rng, time_budget, verbose=False):
    """
    Main optimization loop using subgradient descent with adaptive steps.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    
    lr = 0.1  # initial learning rate
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    
    if verbose:
        print(f"  Subgradient opt: n={n}, initial score={best_score:.6f}, budget={time_budget:.0f}s")
    
    while time.monotonic() < deadline:
        c = _fft_conv(p, p, L)
        lags = _find_top_lags(c, threshold_frac=0.02, top_k=3)
        sg = _subgradient_at_lags(p, lags)
        
        # Try multiple step sizes
        best_trial = None
        best_trial_sc = cur_score
        for lr_try in (lr * 0.5, lr, lr * 1.5, lr * 2.0):
            lr_try = float(np.clip(lr_try, 0.01, 2.0))
            p_try = _multiplicative_step(p, sg, lr_try)
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-12:
                best_trial = p_try
                best_trial_sc = sc_try
        
        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-10:
                best = p.copy()
                best_score = cur_score
                if verbose and it % 100 == 0:
                    print(f"    iter={it}, score={best_score:.6f}, lr={lr:.4f}")
        else:
            stall += 1
            if stall > 5:
                lr = max(0.01, lr * 0.8)
            if stall > 30:
                # Try spectral flattening
                if rng.random() < 0.5:
                    beta = float(rng.uniform(0.1, 0.5))
                    p_new = _spectral_flatten(p, beta, L)
                    if p_new is not None:
                        sc_new = _score(p_new)
                        if sc_new < cur_score:
                            p = p_new
                            cur_score = sc_new
                            stall = 0
                            if sc_new < best_score:
                                best = p.copy()
                                best_score = sc_new
                else:
                    # Random perturbation
                    p_try = p * np.exp(rng.normal(0, 0.05, n))
                    p_try = _sanitize(p_try)
                    if p_try is not None:
                        sc_try = _score(p_try)
                        if sc_try < best_score * 1.05:
                            p = p_try
                            cur_score = sc_try
                            stall = 0
            if stall > 60:
                p = best.copy()
                cur_score = best_score
                lr = 0.1
                stall = 0
        
        it += 1
    
    return best, best_score

    n = len(p)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    
    lr = 0.08
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    
    if verbose:
        print(f"  Hybrid V2: n={n}, initial={best_score:.6f}, budget={time_budget:.0f}s")
    
    while time.monotonic() < deadline:
        r = rng.random()
        
        if r < 0.5:
            # Subgradient step (exact argmax)
            c = _fft_conv(p, p, L)
            lags = _find_top_lags(c, threshold_frac=0.01, top_k=3)
            sg = _subgradient_at_lags(p, lags)
            
            best_trial = None
            best_trial_sc = cur_score
            for lr_try in (lr * 0.5, lr, lr * 1.5):
                lr_try = float(np.clip(lr_try, 0.005, 1.5))
                p_try = _multiplicative_step(p, sg, lr_try)
                sc_try = _score(p_try)
                if sc_try < best_trial_sc - 1e-12:
                    best_trial = p_try
                    best_trial_sc = sc_try
            
            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-10:
                    best = p.copy()
                    best_score = cur_score
            else:
                stall += 1
                lr = max(0.005, lr * 0.9)
                
        elif r < 0.65:
            # Smoothed subgradient (multiple lags with weighting)
            c = _fft_conv(p, p, L)
            mx = c.max()
            # Soft weights over lags
            beta_soft = 50.0 + 50.0 * min(it / 1000, 1.0)
            shift = c.max()
            w = np.exp(beta_soft * (c - shift))
            s_w = w.sum()
            if s_w < 1e-15:
                w = np.zeros_like(c)
                w[_find_top_lags(c, top_k=3)] = 1.0 / 3.0
                s_w = 1.0
            w = w / s_w
            
            # Compute weighted subgradient using FFT
            # sg[k] = sum_m w[m] * 2*p[m-k] (if valid)
            # This is 2 * cross_correlation(w, p) [restricted to valid indices]
            W = np.fft.rfft(w, L_large)
            P = np.fft.rfft(p, L_large)
            # Cross-correlation: ifft(FFT(w) * conj(FFT(p)))
            R = np.fft.irfft(W * np.conj(P), L_large)
            # R[k] = sum_m w[m] * p[m-k] (for circular correlation of length L_large)
            # We need the first n elements (k=0 to n-1)
            sg = 2.0 * R[:n]
            sg = np.maximum(sg, 0)
            sg = np.where(np.isfinite(sg), sg, 0)
            
            best_trial = None
            best_trial_sc = cur_score
            for lr_try in (lr * 0.5, lr, lr * 1.5):
                lr_try = float(np.clip(lr_try, 0.005, 1.5))
                p_try = _multiplicative_step(p, sg, lr_try)
                sc_try = _score(p_try)
                if sc_try < best_trial_sc - 1e-12:
                    best_trial = p_try
                    best_trial_sc = sc_try
            
            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-10:
                    best = p.copy()
                    best_score = cur_score
            else:
                stall += 1
                
        elif r < 0.78:
            # Spectral flattening
            beta = float(rng.uniform(0.05, 0.4))
            p_new = _spectral_flatten(p, beta, L)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-12:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-10:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1
                
        elif r < 0.88:
            # Block scaling
            bs = int(rng.choice([2, 4, 8, 16, 32, 64, 128, 256]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            f = float(rng.choice([0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]))
            p_try = p.copy()
            p_try[start:start + bs] *= f
