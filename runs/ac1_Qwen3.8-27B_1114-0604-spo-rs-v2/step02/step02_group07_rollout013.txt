def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def smooth_obj_grad(a, tau):
    """
    Returns (f, grad) for the log-sum-exp smoothed objective.
    f ≈ 2n * max(c) for large tau.
    The objective is scale-invariant; we normalize internally.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros_like(a)
    
    b = a / S
    c = fast_self_conv(b)
    c_max = float(c.max())
    
    # Log-sum-exp with numerical stability
    scaled = tau * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_sum = float(np.exp(scaled).sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    lse = c_max + np.log(exp_sum) / tau
    f = 2.0 * n * lse

    """Objective and gradient in log-space with softmax normalization."""
    n = len(x)

    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    S_exp = exp_x.sum()
    a = exp_x / S_exp  # a = softmax(x), sum(a) = 1, a > 0

    exp_sum = float(exp_terms.sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
