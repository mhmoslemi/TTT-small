"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with LSE-smoothed max-convolution, momentum subgradient descent,
and targeted local search. Multi-length exploration with aggressive continuation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: g_j = 2 * sum_m a[m] * p[m+j] (cross-correlation of a and p)
    # This is the convolution of reversed a with p, evaluated at positions n-1 to 2n-2
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrap(a, beta):
    return lse_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrap, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def momentum_subgrad(a0, n_iters=5000, momentum=0.9, lr_init=0.01, 
                     lr_decay=0.999, top_k=5, tau=0.1):
    """
    Momentum-based subgradient descent on the exact max-conv objective.
    Uses top-k peaks with softmax weighting for a smoother gradient.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    # Normalize to sum = sqrt(2n) for numerical stability
    target_sum = np.sqrt(2.0 * n)
    a = a / a.sum() * target_sum
    
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k peaks
        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]
        
        # Softmax weights over peak values
        peak_vals = c[peaks]
        pv_shifted = peak_vals - peak_vals.max()
        weights = np.exp(pv_shifted / tau)
        weights /= weights.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, peaks):
            k = int(k)
            # Gradient of c[k] w.r.t. a
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # dc[k]/da[j] = a[k-j] for j in [j_start, j_end]
                # (plus symmetry: if k-j != j, there's also a[j] * a[k-j] from the other side)
                # Actually for self-convolution: c[k] = sum_j a[j]*a[k-j]
                # dc[k]/da[j] = a[k-j] + a[j-k] (if both valid)
                # But since it's self-conv, a[k-j] and a[j-k] are the same element
                # (just different indices). Let me be careful.
                # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
                # dc[k]/da[m] = sum over terms where m appears:
                #   Term 1: a[m]*a[k-m] -> derivative w.r.t. a[m] is a[k-m]
                #   Term 2: a[k-m]*a[m] -> this is the same term, already counted
                # Wait, for self-convolution, c[k] = sum_j a[j]*a[k-j]
                # Each pair (j, k-j) is counted once. If j != k-j, it appears as both
                # a[j]*a[k-j] and a[k-j]*a[j] in the sum (different j values).
                # So dc[k]/da[m] = a[k-m] (if k-m is valid) + a[m-k] (if m-k is valid)
                # But a[m-k] = a[|m-k|] only if we're being careless. Let me just use:
                # dc[k]/da[m] = a[k-m] (when 0 <= k-m < n)
                # because in the sum c[k] = sum_j a[j]*a[k-j], the term involving a[m]
                # is a[m]*a[k-m] (when j=m), and its derivative is a[k-m].
                # But there's also the term where j = k-m, which gives a[k-m]*a[m],
                # and its derivative w.r.t. a[m] is a[k-m].
                # So in total: dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m] (when m != k-m)
                # or a[m] (when m = k-m, i.e., k = 2m)
                # Wait no. In the sum c[k] = sum_j a[j]*a[k-j], each j gives one term.
                # The derivative w.r.t. a[m] gets contributions from:
                #   j = m: derivative of a[m]*a[k-m] w.r.t. a[m] = a[k-m]
                #   j = k-m: derivative of a[k-m]*a[m] w.r.t. a[m] = a[k-m]
                #   (these are the same if m = k-m)
                # So dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m] if m != k-m
                # and = 2*a[m] if m = k-m (since both terms give a[m])
                # Actually when m = k-m (k=2m): the term in the sum is a[m]*a[m] = a[m]^2
                # and it appears only once (j=m=k-m). So derivative is 2*a[m].
                # When m != k-m: the term a[m]*a[k-m] appears at j=m, and the term
                # a[k-m]*a[m] appears at j=k-m. Both involve a[m]. 
                # d/d a[m] of a[m]*a[k-m] = a[k-m]
                # d/d a[m] of a[k-m]*a[m] = a[k-m]
                # Total: 2*a[k-m]
                # So in all cases: dc[k]/da[m] = 2*a[k-m] (if 0 <= k-m < n)
                # Wait, that's not right for the case m = k-m.
                # When k = 2m: c[k] contains the term a[m]*a[m] = a[m]^2 (at j=m).
                # d/d a[m] of a[m]^2 = 2*a[m]. And 2*a[k-m] = 2*a[m]. ✓
                # When k != 2m: c[k] contains a[m]*a[k-m] (at j=m) and a[k-m]*a[m] (at j=k-m).
                # d/d a[m] of a[m]*a[k-m] = a[k-m]
                # d/d a[m] of a[k-m]*a[m] = a[k-m]  (since a[k-m] is treated as constant)
                # Total: 2*a[k-m]. ✓
                # Great, so dc[k]/da[m] = 2*a[k-m] when 0 <= k-m < n, else 0.
                
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end+1] += w * 2.0 * a[idx]
        
        # Scale gradient by the objective's chain rule
        # Objective: f = 2n * max(c) / S^2
        # df/da = 2n * (grad_c / S^2 - 2*max_c * (1/S^3) * sum(1))  -- wait
        # f = 2n * M / S^2 where M = max(c) (or weighted sum of peaks)
        # df/da[m] = 2n * (dM/da[m] / S^2 - M * 2 * S_m / S^3)
        # where S_m = dS/da[m] = 1 for all m
        # So df/da[m] = 2n * (dM/da[m] / S^2 - 2*M / S^3)
        
        # But we're using a weighted sum of peaks as a surrogate for max
        # M_surrogate = sum_i w_i * c[peak_i]
        # dM_surrogate/da[m] = sum_i w_i * 2*a[peak_i - m] (as computed above)
        # But we already computed 'grad' as sum_i w_i * 2*a[peak_i - m]
        # So M_surrogate = sum_i w_i * c[peak_i]
        M_surr = float(np.sum(weights * peak_vals))
        S2 = S * S
        S3 = S2 * S
        full_grad = (2.0 * n) * (grad / S2 - 2.0 * M_surr / S3)
        
        # Gradient step with momentum
        v = momentum * v - lr * full_grad
        a_new = a + v  # Note: v already has the negative sign from the update
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Renormalize
        S_new = a_new.sum()
        if S_new > 1e-6:
            a_new = a_new / S_new * target_sum
        
        a = a_new
        lr *= lr_decay
        
        # Periodic L-BFGS polish (expensive, do infrequently)
        # (handled externally)
    
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Edge-emphasized (parabolic)
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    for alpha in [0.1, 0.3, 0.5, 0.8]:
        inits.append(np.clip(1.0 + alpha * (2.0 * x - 1.0)**2, 0.01, 1000.0))
    # Center-suppressed
    for alpha in [0.1, 0.3, 0.5]:
        inits.append(np.clip(1.0 - alpha * (2.0 * x - 1.0)**2, 0.01, 1000.0))
    # Linear ramp (antisymmetric perturbation)
    for beta in [0.1, 0.3, 0.5]:
        inits.append(np.clip(1.0 + beta * (2.0 * x - 1.0), 0.01, 1000.0))
    # Cosine modulation
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    # Random
    for scale in [0.1, 0.3, 0.5]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in [4, 8, 16, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits
