def _adam_optimize(a0, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                   T_start=5.0, T_end=50.0, K=8, topK_after=5000, 
                   eval_interval=10, patience=5000):
    """
    Adam optimization for minimizing the peak of self-convolution.
    Works with full sequence, projects to anti-palindromic + non-negative.
    """
    a = a0.copy().astype(np.float64)
    n = a.size
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score(a)
    last_improve = 0
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute convolution and scores
        conv = _fft_convolve(a)
        if conv.size == 0:
            break
        
        # Peak weighting: anneal from soft to hard
        t_frac = min(1.0, t / 20000.0)
        T = T_start + (T_end - T_start) * t_frac
        
        if t > topK_after and t % 100 < 50:
            # Use top-K weights
            K_eff = min(K, conv.size)
            top_idx = np.argpartition(conv, -K_eff)[-K_eff:]
            w = np.zeros(conv.size)
            top_vals = conv[top_idx]
            T_local = 5.0
            weights = np.exp(T_local * (top_vals - top_vals.max()))
            weights /= weights.sum()
            for i, idx in enumerate(top_idx):
                w[idx] = weights[i]
        else:
            # Soft-max weights
            c_shift = conv - conv.max()
            exp_c = np.exp(T * c_shift)
            s_exp = exp_c.sum()
            if s_exp < 1e-300:
                w = np.zeros(conv.size)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_c / s_exp
        
        # Gradient
        grad = _grad_weighted(a, w)
        
        # Project gradient to anti-palindromic subspace
        # anti-palindromic direction at index i: grad[i] - grad[n-1-i]
        # We want to move in the anti-palindromic direction
        # Actually, we should project the gradient onto the anti-palindromic subspace
        # The anti-palindromic subspace is spanned by vectors e_i - e_{n-1-i}
        # Projection of g onto this subspace: 
        #   p[i] = (g[i] - g[n-1-i]) / 2 (for i < n/2)
        #   p[n-1-i] = -p[i]
        # But we also need to handle the symmetric part (which is the "center" c/2)
        # The constraint a[i] + a[n-1-i] = c means the sum of pairs is fixed
        # So we should only move in the anti-palindromic direction within each pair
        half = n // 2
        grad_anti = np.zeros(n)
        grad_anti[:half] = (grad[:half] - grad[n-1:half-1:-1]) / 2.0
        grad_anti[half:] = -grad_anti[half-1::-1]
        if n % 2 == 1:
            grad_anti[half] = 0.0
        
        # Also subtract the mean to preserve the sum (scale invariance)
        grad_anti -= grad_anti.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_anti
        v = beta2 * v + (1 - beta2) * grad_anti ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate with decay
        lr_t = lr / (1.0 + 0.001 * t)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to anti-palindromic
        c_val = 2.0 * a.mean()
        d = a - c_val / 2.0
        d_anti = np.zeros(n)
        d_anti[:half] = (d[:half] - d[n-1:half-1:-1]) / 2.0
        d_anti[half:] = -d_anti[half-1::-1]
        if n % 2 == 1:
            d_anti[half] = 0.0
        a = c_val / 2.0 + d_anti
        
        # Enforce non-negativity
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize (preserve sum for scale invariance)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Evaluate periodically
        if t % eval_interval == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = t
            elif t - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.005 * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(2000, int(patience * 1.5))
                lr = max(0.001, lr * 0.7)
        
        # Periodic reset of momentum
        if t % 10000 == 0:
            m *= 0.5
            v *= 0.5
    
    return best_a, best_score
