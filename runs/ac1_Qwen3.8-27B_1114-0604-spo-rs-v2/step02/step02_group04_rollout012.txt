from scipy.optimize import linprog
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n *= 2
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _solve_lp(a, M):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    """
    n = len(a)
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M for each lag k
    # (a*b)[k] = sum_i a[i]*b[k-i] where 0<=i<n and 0<=k-i<n
    # For lag k (0 <= k <= 2n-2):
    #   i ranges from max(0, k-n+1) to min(n-1, k)
    #   (a*b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
    
    # Build constraint matrix A (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            A[k, k - i] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    return -res.fun, res.x if res.success else None

def _lp_refine(a, deadline, max_iters=1000, t_start=0.1):
    """
    LP-based refinement using the state-of-the-art approach.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = float(2 * n * np.max(_fft_convolve(a)))
    
    t = t_start
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = np.max(_fft_convolve(a))
        
        # Solve LP
        lp_val, b = _solve_lp(a, M)
        if b is None:
            break
        
        b = np.clip(b, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b = b / b_sum  # Normalize to sum = 1
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.clip(a_new, 0, 1000)
        a_new = a_new / a_new.sum()
        
        score = float(2 * n * np.max(_fft_convolve(a_new)))
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptively decrease t
        t = max(0.01, t * 0.995)
    
    return best_a, best_score

"""
LP-based optimization for low-peak self-convolution step functions.
Combines the linear programming approach (finding max-L1 directions under
convolution constraints) with FFT-based soft-max gradient descent for polish.
Uses multiple initializations and adaptive line search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_norm1(a):
    n = len(a)
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * np.max(c))


def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_windowed_sine(n, amp, phase=0.5, width=0.35):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - center) / (width * n)) ** 2)
    a = a * w
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_chirp(n, amp):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(np.pi * i * i / n)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_freq(n, freqs, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for f, amp, ph in zip(freqs, amps, phases):
        theta = 2.0 * np.pi * (f * i + ph) / n
        a += amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _build_lp_matrix(a):
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    safe_diff = np.clip(diff, 0, n - 1)
    A = np.where(mask, a[safe_diff], 0.0)
    return A


def _solve_lp(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return -res.fun, res.x
    return None, None


def _lp_optimize(a, deadline, max_iters=20000):
    a = _normalize(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm1(a)
    rng = np.random.RandomState(42)
    no_improve = 0
    patience = 300
    last_print = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(np.max(c))

        lp_val, b = _solve_lp(a, M)
        if b is None:
            found = False
            for factor in [1.01, 1.05, 1.1, 1.5]:
                lp_val, b = _solve_lp(a, M * factor)
                if b is not None:
                    found = True
                    break
            if not found:
                break

        b = np.clip(b, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b / b_sum

        improved = False
        for t_try in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t_try) * a + t_try * b
            a_new = _normalize(a_new)
            sc = _score_norm1(a_new)
            if sc <= best_score + 1e-12:
                a = a_new
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve >= patience:
                noise = rng.standard_normal(n) * 0.008
                a = best_a + noise * best_a.mean()
                a = _normalize(a)
                no_improve = 0
                patience = int(patience * 1.3)
                if patience > 1000:
                    patience = 1000

        if it - last_print > 500:
            print(f"    LP iter {it}: best={best_score:.8f}, M={M:.6f}", flush=True)
            last_print = it

    return best_a, best_score


def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N] if w.size > N else w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _gradient_polish(a, deadline, T_start=5.0, T_end=60.0, lr0=None):
    a = _normalize(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    if lr0 is None:
        lr0 = 0.08 * n

    best_a = a.copy()
    best_score = _score_norm1(a)
    rng = np.random.RandomState(77)
    last_improve = 0
    eval_count = 0

    while time.monotonic() < deadline:
        t_frac = min(1.0, eval_count / 40000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.003 * eval_count)

        c = _fft_convolve(a)
        c_max = c.max()
        if c_max <= 0:
            break
        c_norm = c / c_max
        w = np.exp(T * (c_norm - 1.0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)


        a = a * np.exp(-lr * grad_proj)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)

        eval_count += 1
        if eval_count % 5 == 0:
            sc = _score_norm1(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = eval_count
                lr0 = min(0.25 * n, lr0 * 1.008)

            if eval_count - last_improve > 4000:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.004
                a = a + noise * a.mean()
                a = _normalize(a)
                last_improve = eval_count
                lr0 = max(0.02 * n, lr0 * 0.6)
                eval_count = 0
                T_start = max(2.0, T_start * 0.7)

    return best_a, best_score


def _active_set_polish(a, deadline, K=16):
    """Polish using only the top-K active constraints."""
    a = _normalize(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm1(a)
    rng = np.random.RandomState(123)
    last_improve = 0
    count = 0

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        T_local = 4.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights = weights / weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        lr = 0.05 * n / (1.0 + 0.002 * count)
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        a = a * np.exp(-lr * grad_proj)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)

        count += 1
        if count % 3 == 0:
            sc = _score_norm1(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = count
            if count - last_improve > 3000:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.003
                a = a + noise * a.mean()
                a = _normalize(a)
                last_improve = count

    return best_a, best_score


def _scan_init(deadline):
    """Scan multiple initialization families."""
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(2024)

    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline:
            break

        candidates = []
        # Single sine
        for amp in [0.6, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0]:
            for phase in [0.3, 0.4, 0.5, 0.6]:
                candidates.append(_make_sine(n, amp, phase))
        # Windowed sine
        for amp in [0.7, 0.8, 0.9]:
            candidates.append(_make_windowed_sine(n, amp, 0.5, 0.35))
            candidates.append(_make_windowed_sine(n, amp, 0.5, 0.5))
        # Chirp
        candidates.append(_make_chirp(n, 0.7))
        candidates.append(_make_chirp(n, 0.9))
        # Multi-frequency
        candidates.append(_make_multi_freq(n, [1, 2], [0.7, 0.15], [0.5, 0.0]))
        candidates.append(_make_multi_freq(n, [1, 3], [0.8, 0.1], [0.5, 0.25]))
        # Random on simplex
        for _ in range(3):
            x = rng.dirichlet(np.ones(n))
            candidates.append(x)

        for cand in candidates:
            sc = _score_norm1(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

    return best_a, best_score
