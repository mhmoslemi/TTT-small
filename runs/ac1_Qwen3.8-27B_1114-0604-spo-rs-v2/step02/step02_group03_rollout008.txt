"""
Step-function optimizer minimizing 2*n*max(conv)/sum^2 for non-negative sequences.
Uses multi-phase strategy: diverse initialization, log-sum-exp smoothed gradient descent,
simulated annealing, coordinate descent, and structured frequency-domain perturbations.
"""

import numpy as np
import time
import math


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out

def _weighted_conv_grad(p, f):
    """Compute sum_k p[k] * d(conv[k])/d(f[i]) for all i.
    p has length 2n-1, f has length n.
    Result has length n.
    
    d(conv[k])/d(f[i]) = f[k-i] (if 0<=k-i<n) + f[i-k] (if 0<=i-k<n)
    
    So sum_k p[k] * f[k-i] = (p * f)[i] where * denotes the "anti-convolution"
    Actually: sum_k p[k]*f[k-i] = sum_j p[i+j]*f[j] = (flip(p) * f)[n-1-i]... 
    
    Let me just use the direct formula:
    result[i] = sum_k p[k] * (f[k-i] if 0<=k-i<n else 0) + (f[i-k] if 0<=i-k<n else 0)
    
    = sum_j p[i+j]*f[j] (j from 0 to n-1, i+j from 0 to 2n-2)
    + sum_j p[i-j]*f[j] (j from 0 to n-1, i-j from 0 to 2n-2)
    
    First term: for fixed i, sum_j f[j]*p[i+j] = cross-correlation of f and p at lag i
    Second term: for fixed i, sum_j f[j]*p[i-j] = convolution of f and p at position i
    
    We can compute both via FFT.
    """
    n = f.size
    L = p.size  # 2n-1
    m = n + L - 1  # max length of conv/fold
    size = _fft_size(m)
    
    # First part: sum_j f[j] * p[i+j] for valid i+j
    # This is the cross-correlation: corr[i] = sum_j f[j]*p[i+j]
    # = (f * flip(p))[n-1+i] ... let me think differently
    # corr[i] = sum_j f[j]*p[i+j], i+j ranges from i to i+n-1
    # Let k = i+j, j = k-i: corr[i] = sum_k f[k-i]*p[k] (where 0<=k-i<n, 0<=k<L)
    # This is (f_flipped * p) evaluated at appropriate positions
    # f_flipped[i] = f[n-1-i]
    # (f_flipped * p)[m] = sum_i f_flipped[i]*p[m-i] = sum_i f[n-1-i]*p[m-i]
    # Let j = n-1-i: = sum_j f[j]*p[m-n+1+j]
    # So corr[i] = sum_j f[j]*p[i+j] = (f_flipped * p)[i + n - 1]
    
    F = np.fft.rfft(f, size)
    P = np.fft.rfft(p, size)
    # f_flipped * p via FFT:
    FF = np.fft.rfft(f[::-1], size)
    conv1 = np.fft.irfft(FF * P, size)
    # corr[i] = conv1[i + n - 1], for i = 0..n-1, need i+n-1 < size
    part1 = conv1[n-1:n-1+n]
    
    # Second part: sum_j f[j]*p[i-j] for valid i-j
    # = (f * p_flipped)[i + n - 1] ... 
    # (f * p)[m] = sum_j f[j]*p[m-j]
    # We want sum_j f[j]*p[i-j] = (f * p)[i] (with p extended by zeros)
    # But p has length 2n-1, so (f * p) has length n + 2n - 2 = 3n - 2
    # We want positions i = 0, 1, ..., n-1
    conv2 = np.fft.irfft(F * P, size)
    part2 = conv2[0:n]
    
    result = part1 + part2
    return result


def _logsumexp_smooth_score(f, T):
    """Compute smoothed score and the softmax weights over lags."""
    n = f.size
    c = _autocorr(f)
    maxc = float(c.max())
    z = (c - maxc) / T
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p.fill(1.0 / len(c))
    else:
        p /= ps
    # Smoothed value: maxc + T * log(sum exp(z)) / len... 
    # Actually: (1/T)*log(sum exp(c[k]/T)) = maxc + (1/T)*log(sum exp((c[k]-maxc)/T))
    # = maxc + (1/T)*log(sum p[k]*sum... ) 
    # The smoothed objective is proportional to maxc + (1/T)*log(L_eff) where L_eff = sum p
    # But since p is normalized, the smoothed value is just:
    # (1/T) * log(sum_k exp(c_k / T)) = maxc + (1/T)*log(sum_k p_k * (sum_k exp(...)))... 
    # Let's just use: smooth_val = maxc + T * (mean of log weights) ... 
    # Simpler: we'll just track maxc and use p for the gradient direction.
    return maxc, p, c


def _gradient_step(f, p, T, lr, momentum_buf, momentum_coef):
    """One gradient step in log-space with momentum."""
    n = f.size
    g = _weighted_conv_grad(p, f)
    # Normalize gradient
    g_std = float(np.std(g))
    if g_std < 1e-300:
        g_centered = g - float(g.mean())
    else:
        g_centered = g - float(g.mean())
    
    # Momentum
    momentum_buf = momentum_coef * momentum_buf + (1 - momentum_coef) * g_centered
    
    # Log-space step: f_new = f * exp(-lr * momentum_buf / scale)
    scale = float(np.max(np.abs(momentum_buf))) + 1e-300
    step = -lr * momentum_buf / scale
    np.clip(step, -5.0, 5.0, out=step)
    f_new = f * np.exp(step)
    f_new = _normalize(f_new)
    return f_new, momentum_buf
