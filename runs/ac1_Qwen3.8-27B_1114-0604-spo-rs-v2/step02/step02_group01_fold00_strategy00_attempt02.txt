def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    s = 0.0
    for x in sequence:
        if isinstance(x, bool):
            return np.inf
        if not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
        s += np.clip(x, 0.0, 1000.0)
    if s < 0.01:
        return np.inf
    a = np.clip([float(x) for x in sequence], 0.0, 1000.0)
    s_val = sum(a)
    if s_val < 0.01:
        return np.inf
    conv_result = np.convolve(a, a, mode='full')
    max_conv = np.max(conv_result)
    norm_sum_sq = s_val ** 2
    return float(2.0 * len(a) * max_conv / norm_sum_sq)

# Utility function to replace the previous _autocorr_fft with a standard linear convolution
def autocorr_fft(f):
    """Compute the full linear cross-correlation of f with itself using FFT for efficiency."""
    if len(f) == 0:
        return np.array([])
    # Compute FFT and then multiply to get correlation
    n = len(f)
    height = (2 * n - 1) * (2**(np.ceil(np.log2(2 * n - 1)) / 2).astype(int)
    if height == 0:
        return np.zeros(2 * n - 1)
    F = np.fft.rfft(f, height)
    return np.maximum(0, np.fft.irfft(F * F, 2 * n - 1))

        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]

        # Compute gaps and assign weights inversely proportional to gaps between top values
        gaps = top_vals[:-1] - top_vals[1:] + 0.001
        if np.min(gaps) <= 0:
            # Avoid division by zero
            gaps = np.maximum(gaps, 0.001)
        weights = gaps / (gaps.sum() + 0.001)

        # Initialize gradient direction (will be derived from active lags)
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            for i in range(len(current_seq)):
                if i >= max(0, k - n + 1) and i <= min(n - 1, k):
                    # The gradient is from the second factor for the first lag
                    grad[i] += w * current_seq[k - i] + w  # Two cross terms: from current_seq[i] * current_seq[k-i] and cross
        grad_max = grad.max() + 0.001
        grad_min = grad.min() + 0.001
        grad_span = grad_max - grad_min + 0.001

        # Calculate the effective growth rate eta using gradient active points
        eta = step_size / max(0.001, grad_span) * 0.4 + step_size * 0.6

        # Update the sequence using log-space and exponentiation for smooth growth
        log_seq = np.log(current_seq + 0.001)  # Log transform to handle zeros: using 10^{-3} = 0.001
        log_seq += eta * (-grad + 0.2)  # Invert gradient and add small constant to stabilize
        log_seq -= log_seq.max()  # Normalize log values to avoid overflow

        current_seq = np.exp(log_seq)  # Exponentiate to get the new un-normalized sequence
        s_val = current_seq.sum()

        # Recenter normalization near previous best sequence length
        if abs(s_val) > 1e-7:
            current_seq = current_seq / s_val
        else:
            # Fallback to uniform if sum is too close to zero
            current_seq = np.ones(n) / n

        s_val = sum(current_seq)

        # Prevent normalization from below 0.01 if the sum is too small
        if s_val < 0.01 and s_val > 0.0001:
            current_seq = (1.0) * current_seq
            s_val = 1.0

        # Early termination if we see improvement over the current best
        score = evaluate_sequence(current_seq)
        if score < best_score - 0.001 and best_seq is None:
            best_seq = current_seq.copy()
            best_score = score

    # Fallback mechanism if the polish_coordinate_local could not find a valid improvement
    if best_seq is None:
        best_seq = current_seq.copy()
        best_score = evaluate_sequence(best_seq)
        if best_score == np.inf:
            # If invalid, then use the default fallback
            default_sequence = np.ones(n) / n
            default_score = evaluate_sequence(default_sequence)
            if default_score < best_score:
                best_seq = default_sequence.copy()
                best_score = default_score

    return best_seq, best_score

def propose_candidate(height_sequence_1, budget=2400, seed=None):

    else:

            # But note: this is more efficient for finding local minima than gradient descent
            # However, we'll use a more structured approach: coordinate ascent until there's no improvement

            # Gaps and weights are set to target the top peaks
            gaps = conv_result[top_idx] - conv_result[min(top_idx+1, n-1)]  # Find gap to adjacent lag
            gaps += 0.001
            weights = 1.0 / gaps

            updated_seq = base_sequence.copy()
            grad_around_top = np.zeros(n)
            if top_idx >= 0:
                # Left neighbor contribution (if exists)
                if top_idx - int(0.2 * n) >= 0:
                    grad_around_top[top_idx-1] += weights[top_idx-1] * updated_seq[top_idx-1]
                # Right neighbor contribution (if exists)
                if top_idx + int(0.2 * n) <= 2*n-2:
                    grad_around_top[top_idx+1] += weights[top_idx+1] * updated_seq[top_idx+1]

            # Update using momentum (grad_eff) and ensure exponential growth reflects the problem space
            # Avoid negative values by a shift in the log space
            if top_idx >= 0:

                updated_seq[top_idx] = max(0, updated_seq[top_idx] - 7 * 0.0001)  # Ensure non-negative
