```python
    """
    Gradient of sum_k weights[k] * conv(a,a)[k] with respect to a[i].
    conv(a,a)[k] = sum_j a[j]*a[k-j]

    This is 2 * (weighted sum of shifted a).
    Computed via FFT cross-correlation.
    """
    n = a.size
    # We want: for each i, sum over k of weights[k] * a[k-i] (valid range)

    # This is a correlation!

    # where w_padded is w placed at the right position
    N = _next_pow2(2 * n)

        a_local = np.clip(a_flat, 1e-15, None)
        a_local = a_local / a_local.sum()
        c = _fft_convolve(a_local)
        c_max = c.max()
        c_shift = (c - c_max) / (c_max + 1e-15)
        w = np.exp(T * c_shift)
        w /= w.sum()
        obj = 2.0 * n * float(np.dot(w, c))
        grad = _grad_normalized(a_local, np.arange(len(c)), w)
        # Project gradient to simplex tangent (sum=0)
        grad = grad - grad.mean()
        return obj, grad
    
    # L-BFGS-B doesn't handle the sum=1 constraint well directly
    # Instead, parameterize with a[i] = exp(x[i]) / sum(exp(x[j]))
    # This automatically ensures a >= 0 and sum(a) = 1
    
    x_init = np.log(np.clip(a, 1e-15, None))
    
    def log_param_obj(x):
        if time.monotonic() > deadline:
            return 0.0
        a_soft = np.exp(x - x.max())
        a_soft = a_soft / a_soft.sum()
        c = _fft_convolve(a_soft)
        c_max = c.max()
        c_shift = (c - c_max) / (c_max + 1e-15)
        w = np.exp(T * c_shift)
        w /= w.sum()
        obj = 2.0 * n * float(np.dot(w, c))
        return obj
    
    def log_param_grad(x):
        if time.monotonic() > deadline:
            return np.zeros(n)
        a_soft = np.exp(x - x.max())
        a_soft = a_soft / a_soft.sum()
        c = _fft_convolve(a_soft)
        c_max = c.max()
        c_shift = (c - c_max) / (c_max + 1e-15)
        w = np.exp(T * c_shift)
        w /= w.sum()
        # grad of obj w.r.t. a
        grad_a = _grad_normalized(a_soft, np.arange(len(c)), w)
        # Chain rule: a = softmax(x), so da_i/dx_j = a_i*(delta_ij - a_j)
        # dx_j = sum_i grad_a_i * da_i/dx_j = sum_i grad_a_i * a_i*(delta_ij - a_j)
        # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        grad_x = a_soft * (grad_a - np.dot(grad_a, a_soft))
        return grad_x
    
    # Use scipy minimize
    try:
        result = minimize(
            log_param_obj, x_init, jac=log_param_grad,
            method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 1000, 'gtol': 1e-8}
        )
        if result.success or result.fun < best_score / (2.0 * n):
            a_soft = np.exp(result.x - result.x.max())
            a_soft = a_soft / a_soft.sum()
            sc = _score(a_soft)
            if sc < best_score:
                best_a = a_soft.copy()
                best_score = sc
                print(f"  L-BFGS-B improved: {sc:.8f}", flush=True)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)
    
    return best_a, best_score
```

Actually, I realize I should think about this differently. Let me look at what the theoretical optimum looks like.

For a probability distribution on {0, ..., n-1}, we want to minimize max_k P(X+Y=k) where X,Y are iid with that distribution.

If X is uniform, P(X+Y=k) is a triangle with peak 1/n at k=n-1. Score = 2.

If X is "bimodal" (concentrated at two points), say X takes values 0 and n/2 with probability 1/2 each, then X+Y takes values 0, n/2, n-1 with probabilities 1/4, 1/2, 1/4. Max = 1/2. Score = 2n * 1/2 = n. That's terrible.

If X is concentrated in a small interval, the convolution is also concentrated, giving a high peak.

The optimum seems to be somewhere in between: spread out enough that the convolution is flat, but with some structure that reduces the peak below the uniform case.

The sinusoidal (1 + A*sin(2πi/n)): when A = 1, this is zero at i = 3n/4 (mod n), creating a "notch" that reduces the convolution peak.

What about using higher harmonics? (1 + A1*sin(2πi/n) + A2*sin(4πi/n) + ...)?

Or what about using a "raised cosine" window: a[i] = (1 + cos(2πi/n))/n = 2*cos^2(πi/n)/n? This is always non-negative.

For the raised cosine: the Fourier transform is concentrated at frequencies 0 and ±1, so the convolution should be smoother.

```python
"""
Multi-phase low-peak step-function optimization:
1. Broad initialization scan over n, amplitude, and family
2. Soft-max mirror descent (coarse optimization)
3. Adaptive hard-max refinement (K from large to 1)
4. Coordinate-descent polish
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = _fft_convolve(a_n)
    return 2.0 * a.size * float(np.max(c))


def _grad_conv(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Result[i] = 2 * sum_k w[k] * a[k-i] for valid k-i.
    Computed via FFT.
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    w_pad = np.zeros(N)
    w_len = min(len(w), N)
    w_pad[:w_len] = w[:w_len]
    fw = np.fft.rfft(w_pad, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * h[:n]


def _make_sine(n, amp, phase=0.25):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_cosine(n, amp, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.cos(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_multi(n, components):
    """components: list of (amp, freq, phase, 'sin'/'cos')"""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, freq, phase, kind in components:
        theta = 2.0 * np.pi * freq * (i + phase) / n
        if kind == 'sin':
            a += amp * np.sin(theta) / n
        else:
            a += amp * np.cos(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _scan_init(deadline):
    """Broad scan of initialization family."""
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(777)
    
    configs = []
    for n in [2048, 4096, 8192, 1024]:
        # Pure sine
        for amp in [0.7, 0.8, 0.85, 0.9, 0.95, 1.0]:
            for phase in [0.25, 0.3, 0.4]:
                configs.append((n, 'sine', amp, phase))
        # Pure cosine
        for amp in [0.7, 0.8, 0.9, 1.0]:
            configs.append((n, 'cosine', amp, 0.0))
        # Multi-sine (2 components)
        for _ in range(3):
            a1 = rng.uniform(0.6, 1.0)
            a2 = rng.uniform(0.1, 0.4)
            f2 = rng.randint(2, 4)
            p1 = rng.uniform(0.2, 0.4)
            p2 = rng.uniform(0.0, 1.0)
            configs.append((n, 'multi', [(a1, 1, p1, 'sin'), (a2, f2, p2, 'sin')]))
    
    for cfg in configs:
        if time.monotonic() > deadline:
            break
        n = cfg[0]
        if cfg[1] == 'sine':
            a = _make_sine(n, cfg[2], cfg[3])
        elif cfg[1] == 'cosine':
            a = _make_cosine(n, cfg[2], cfg[3])
        else:
            a = _make_multi(n, cfg[2])
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    New best init: n={n}, score={sc:.8f}", flush=True)
    
    if best_a is None:
        best_a = _make_sine(4096, 0.9, 0.25)
        best_score = _score(best_a)
    
    return best_a, best_score


def _optimize_soft(a, deadline, T_start=5.0, T_end=300.0):
    """Soft-max mirror descent."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    
    lr = 0.3
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 5000
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 80000.0)
        T = T_start + (T_end - T_start) * t_frac
        step = lr / (1.0 + 0.003 * it)
        
        c = _fft_convolve(a)
        c_max = c.max()
        c_shift = (c - c_max) / (c_max + 1e-15)
        w = np.exp(T * c_shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        grad = _grad_conv(a, w)
        # Center the gradient (tangent to simplex)
        grad = grad - grad.mean()
        
        a = a * np.exp(-step * grad)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(2000, int(patience * 1.3))
                lr = max(0.05, lr * 0.85)
                if it % 200 == 0:
                    print(f"    [soft] it={it}, score={best_score:.8f}, lr={lr:.4f}", flush=True)
    
    return best_a, best_score


def _optimize_hard(a, deadline, K_max=256, K_min=1, n_phases=16):
    """Hard-max with adaptive K reduction."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 4000
    rng = np.random.RandomState(99)
    
    # K schedule: exponential reduction
    K_values = []
    for i in range(n_phases):
        frac = i / max(1, n_phases - 1)
        K = max(K_min, int(K_max * (1.0 - frac) ** 2))
        K_values.append(K)
    
    time_per_phase = max(1.0, (deadline - time.monotonic()) / n_phases)
    
    for phase_i, K in enumerate(K_values):
        if time.monotonic() >= deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + time_per_phase)
        
        lr = max(0.02, 0.2 / (1.0 + 0.02 * phase_i))
        
        while time.monotonic() < phase_deadline:
            it += 1
            c = _fft_convolve(a)
            
            if c.size <= K:
                top_idx = np.arange(c.size)
            else:
                top_idx = np.argpartition(c, -K)[-K:]
            
            top_vals = c[top_idx]
            T_local = 3.0
            top_shift = top_vals - top_vals.max()
            weights = np.exp(T_local * top_shift)
            weights /= weights.sum()
            
            w = np.zeros(c.size, dtype=np.float64)
            for i, k in enumerate(top_idx):
                w[k] = weights[i]
            
            grad = _grad_conv(a, w)
            grad = grad - grad.mean()
            
            step = lr / (1.0 + 0.001 * it)
            a = a * np.exp(-step * grad)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
                break
            a = a / s
            
            if it % 2 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    last_improve = it
                elif it - last_improve > patience:
                    a = best_a.copy()
                    noise = rng.standard_normal(n) * 0.005
                    a = a + noise * a.mean()
                    a = np.clip(a, 0, 1000)
                    a = a / a.sum()
                    patience = max(1000, int(patience * 1.2))
                    break
        
        print(f"  [hard] K={K}, score={best_score:.8f}, it={it}", flush=True)
    
    return best_a, best_score


def _coord_descent_polish(a, deadline, n_trials=50000):
    """
    Coordinate descent: try moving small amounts of mass from one coordinate
    to another, accepting if score improves.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(2024)
    
    it = 0
    while time.monotonic() < deadline and it < n_trials:
        it += 1
        
        # Pick two random coordinates
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try moving delta from i to j
        delta = a[i] * rng.uniform(0.01, 0.1)
        
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        
        if a_trial[i] < 0:
            continue
        
        sc = _score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score


def _perturbation_polish(a, deadline):
    """Try small random perturbations and keep improvements."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(31415)
    
    sigma = 0.01
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Apply a structured perturbation
        pert = rng.standard_normal(n) * sigma * a.mean()
        a_trial = a + pert
        a_trial = np.clip(a_trial, 0, 1000)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        
        sc = _score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            sigma = max(0.001, sigma * 1.05)  # increase slightly
        else:
            sigma = max(0.0001, sigma * 0.95)  # decrease
    
    return best_a, best_score
```

```python
"""
Multi-phase low-peak step-function optimization:
1. Broad initialization scan over n, amplitude, harmonic families
2. Soft-max mirror descent with FFT gradients (coarse)
3. Adaptive hard-max refinement with K reduction (fine)
4. Gradient-direction line search polish
5. Multi-start with different n values
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = _fft_convolve(a_n)
    if c.size == 0:
        return np.inf
    return 2.0 * a.size * float(np.max(c))


def _grad_conv(a, w):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    w_pad = np.zeros(N)
    w_len = min(len(w), N)
    w_pad[:w_len] = w[:w_len]
    fw = np.fft.rfft(w_pad, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * h[:n]


def _make_sine(n, amp, phase=0.25):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_cosine(n, amp, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.cos(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_multi(n, components):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, freq, phase, kind in components:
        theta = 2.0 * np.pi * freq * (i + phase) / n
        if kind == 'sin':
            a += amp * np.sin(theta) / n
        else:
            a += amp * np.cos(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_hermite(n, order):
    """Hermite-Gaussian-like profile."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / n - 1.0
    if order == 0:
        a = np.exp(-2.0 * x * x)
    elif order == 1:
        a = (1.0 + 0.5 * x) * np.exp(-2.0 * x * x)
    elif order == 2:
        a = (1.0 + 0.3 * x * x) * np.exp(-2.0 * x * x)
    else:
        a = np.ones(n, dtype=np.float64) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _scan_init(deadline):
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(777)

    n_values = [4096, 8192, 2048, 1024]

    for n in n_values:
        if time.monotonic() > deadline:
            break
        # Sine family
        for amp in [0.8, 0.85, 0.9, 0.95, 1.0]:
            for phase in [0.2, 0.25, 0.3]:
                if time.monotonic() > deadline:
                    break
                a = _make_sine(n, amp, phase)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # Cosine family
        for amp in [0.8, 0.9, 1.0]:
            if time.monotonic() > deadline:
                break
            a = _make_cosine(n, amp, 0.0)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Multi-harmonic
        for _ in range(4):
            if time.monotonic() > deadline:
                break
            a1 = rng.uniform(0.6, 1.0)
            a2 = rng.uniform(0.05, 0.3)
            f2 = rng.randint(2, 5)
            p1 = rng.uniform(0.15, 0.35)
            p2 = rng.uniform(0.0, 1.0)
            a = _make_multi(n, [(a1, 1, p1, 'sin'), (a2, f2, p2, 'sin')])
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Sine + cosine mix
        for _ in range(2):
            if time.monotonic() > deadline:
                break
            a1 = rng.uniform(0.5, 0.9)
            a2 = rng.uniform(0.2, 0.6)
            p1 = rng.uniform(0.2, 0.4)
            a = _make_multi(n, [(a1, 1, p1, 'sin'), (a2, 1, 0.0, 'cos')])
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _make_sine(4096, 0.9, 0.25)
        best_score = _score(best_a)

    return best_a, best_score


def _optimize_soft(a, deadline, T_start=4.0, T_end=250.0):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
    a = a / s
    n = a.size

    lr_base = 0.35
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 6000
    rng = np.random.RandomState(42)

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 100000.0)
        T = T_start + (T_end - T_start) * t_frac
        step = lr_base / (1.0 + 0.004 * it)

        c = _fft_convolve(a)
        c_max_val = float(c.max())
        c_shift = (c - c_max_val) / (c_max_val + 1e-15)
        w = np.exp(T * c_shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        grad = _grad_conv(a, w)
        grad = grad - grad.mean()

        a = a * np.exp(-step * grad)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(2000, int(patience * 1.25))
                lr_base = max(0.05, lr_base * 0.85)

    return best_a, best_score


def _optimize_hard(a, deadline, K_max=128, K_min=1, n_phases=14):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
    a = a / s
    n = a.size

    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 4000
    rng = np.random.RandomState(99)

    K_values = []
    for i in range(n_phases):
        frac = i / max(1, n_phases - 1)
        K = max(K_min, int(K_max * (1.0 - frac) ** 2.5))
        K_values.append(K)

    time_per_phase = max(1.0, (deadline - time.monotonic()) / n_phases)

    for phase_i, K in enumerate(K_values):
        if time.monotonic() >= deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + time_per_phase)
        lr = max(0.03, 0.25 / (1.0 + 0.05 * phase_i))

        while time.monotonic() < phase_deadline:
            it += 1
            c = _fft_convolve(a)

            if c.size <= K:
                top_idx = np.arange(c.size)
            else:
                top_idx = np.argpartition(c, -K)[-K:]

            top_vals = c[top_idx]
            T_local = 3.0
            top_shift = top_vals - top_vals.max()
            weights = np.exp(T_local * top_shift)
            weights /= weights.sum()

            w = np.zeros(c.size, dtype=np.float64)
            for i, k in enumerate(top_idx):
                w[k] = weights[i]

            grad = _grad_conv(a, w)
            grad = grad - grad.mean()

            step = lr / (1.0 + 0.002 * it)
            a = a * np.exp(-step * grad)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
                break
            a = a / s

            if it % 2 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    last_improve = it
                elif it - last_improve > patience:
                    a = best_a.copy()
                    noise = rng.standard_normal(n) * 0.005
                    a = a + noise * a.mean()
                    a = np.clip(a, 0, 1000)
                    a = a / a.sum()
                    patience = max(1500, int(patience * 1.2))
                    break

        if (phase_i + 1) % 3 == 0 or phase_i == n_phases - 1:
            print(f"    [hard] K={K}, iters={it}, score={best_score:.8f}", flush=True)

    return best_a, best_score


def _gradient_polish(a, deadline):
    """
    Gradient-based line search polish: compute gradient at the top-1 lag,
    then do a 1D line search along -gradient projected to simplex.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(555)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        top_k = np.argmax(c)

        w = np.zeros(c.size)
        w[top_k] = 1.0

        grad = _grad_conv(a, w)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        direction = -grad / gnorm

        # Line search: find best alpha in (0, 0.5)
        best_alpha = 0.0
        best_sc = best_score
        for alpha in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
            a_trial = a + alpha * direction * a.mean()
            a_trial = np.clip(a_trial, 0, 1000)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        if best_alpha > 0 and best_sc < best_score:
            a = (a + best_alpha * direction * a.mean())
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
            best_score = best_sc
            best_a = a.copy()
        else:
            # Perturb and continue
            noise = rng.standard_normal(n) * 0.003
            a = a + noise * a.mean()
            a = np.clip(a, 0, 1000)
            a = a / a.sum()

    return best_a, best_score


def _multi_start_optimize(deadline, n_candidates=3):
    """Run optimization from multiple different initializations."""
    rng = np.random.RandomState(2024)
    results = []

    n_options = [4096, 8192, 2048]
    amps = [0.85, 0.9, 0.95, 1.0]
    phases = [0.2, 0.25, 0.3]

    attempts = []
    for n in n_options:
        for amp in amps:
            for phase in phases:
                attempts.append((n, amp, phase))
    rng.shuffle(attempts)

    time_per_attempt = (deadline - time.monotonic()) / max(1, n_candidates)

    for idx in range(min(n_candidates, len(attempts))):
        if time.monotonic() >= deadline:
            break
        n, amp, phase = attempts[idx]
        a_init = _make_sine(n, amp, phase)
        init_sc = _score(a_init)

        sub_deadline = time.monotonic() + time_per_attempt - 5.0

        # Soft phase (60% of sub-budget)
        soft_deadline = time.monotonic() + (sub_deadline - time.monotonic()) * 0.5
        a_soft, s_soft = _optimize_soft(a_init, soft_deadline)

        # Hard phase (remaining)
        a_hard, s_hard = _optimize_hard(a_soft, sub_deadline, K_max=64, K_min=1, n_phases=8)

        results.append((s_hard, a_hard))
        print(f"    [multi] start {idx+1}: n={n}, amp={amp}, phase={phase}, "
              f"init={init_sc:.6f}, final={s_hard:.8f}", flush=True)

    if not results:
        a = _make_sine(4096, 0.9, 0.25)
        return a, _score(a)

    results.sort(key=lambda x: x[0])
    return results[0][1], results[0][0]

        print(f"[Phase 4] Gradient polish ({(d4 - time.monotonic()):.0f}s)...", flush=True)
        a_polish, s_polish = _gradient_polish(a_hard, d4)
        print(f"  Polish best: n={a_polish.size}, score={s_polish:.8f}", flush=True)
        if s_polish < global_best_score:
            global_best = a_polish.copy()
            global_best_score = s_polish
        a_final = a_polish

        a_final = a_hard

    # Phase 5: Multi-start with different n (if time allows)

        ms_deadline = hard_deadline - 30.0

    candidates.append(a_final)
    candidates.append(a_hard)
    candidates.append(a_soft)
    candidates.append(init_a)

    best_out = None
    best_actual = np.inf

    seen = set()
    for arr in candidates:
        if time.monotonic() > hard_deadline:
            break
        key = (arr.size, round(arr[0], 6), round(arr[-1], 6))
        if key in seen:
            continue
        seen.add(key)
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  Validated n={arr.size}: actual={actual:.8f}", flush=True)
            if best_actual < 1.50:
                break

    if best_out is None:
        arr = global_best if global_best is not None else init_a
        best_out = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = global_best_score if global_best is not None else init_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
```
