"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated multiplicative gradient descent, adaptive spectral shaping,

to flatten the autocorrelation peak below the uniform bound of 2.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _score_and_grad(p, beta):
    """Compute exact score and smoothed gradient for multiplicative updates."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf'), None
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    d = n * c
    max_d = float(d.max())
    score = 2.0 * max_d

    if beta < 1e-6:
        w = np.zeros(len(d))
        w[np.argmax(d)] = 1.0
    else:
        shift = max_d
        e = np.exp(beta * (d - shift))
        ss = e.sum()
        if ss < 1e-300 or not np.isfinite(ss):
            w = np.zeros(len(d))
            w[np.argmax(d)] = 1.0
        else:
            w = e / ss

    L = _next_pow2(2 * (2 * n - 1) + n)
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 4.0 * n * R[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    return score, h


def _spectral_flatten(p, alpha, L=None):
    """Frequency-domain: reshape autocorrelation to be flatter."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.15:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _spectral_flatten_v2(p, alpha, smooth=0.3, L=None):
    """Improved spectral flatten with smoothing to reduce negative parts."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    C = np.maximum(C, 0.0)
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    # Mix between original and flattened
    C_target = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = (1.0 - smooth) * C + smooth * C_target
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    # Ensure non-negative real part for sqrt
    C_freq_new = np.maximum(C_freq_new.real, 0.0) + 1j * C_freq_new.imag
    A_new = np.sqrt(C_freq_new.astype(complex))
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _init_u_shape(n, rng, sharpness=2.0):
    """Initialize with U-shaped profile (high at edges, low in center)."""
    t = np.linspace(0, 1, n)
    # U-shape: high at 0 and 1, low at 0.5
    base = 1.0 + sharpness * np.cos(np.pi * t)
    base = np.maximum(base, 0.1)
    # Add some noise
    noise = 1.0 + 0.1 * rng.normal(0, 1, n)
    p = base * noise
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_bimodal(n, rng):
    """Initialize with two peaks."""
    t = np.arange(n, dtype=float) / n
    K = int(rng.integers(2, 5))
    p = np.zeros(n)
    for _ in range(K):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.03, 0.15)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), np.abs(x - 1.0))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine_mod(n, rng):
    """Cosine-modulated initialization."""
    t = np.arange(n, dtype=float) / n
    p = np.ones(n)
    for k in range(1, 6):
        amp = rng.uniform(0.3, 1.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_from_parent(parent, n, rng=None):
    """Resample parent to new length."""
    parent = np.asarray(parent, dtype=float)
    parent = np.maximum(parent, 0)
    s = parent.sum()
    if s < 1e-10:
        return None
    parent = parent / s
    n_old = len(parent)
    if n_old == n:
        result = parent.copy()
    elif n_old == 1:
        result = np.ones(n, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n) + 0.5) / n
        result = np.interp(x_new, x_old, parent)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _optimize_momentum(p0, rng, time_budget, beta_start=20.0, beta_end=2000.0):
    """Nesterov-style multiplicative gradient descent with momentum."""
    p = np.maximum(p0, 1e-15)
    p = p / p.sum()
    n = len(p)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    beta = float(beta_start)
    lr = 0.05
    momentum = 0.7
    v = np.zeros(n)

    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        sc, h = _score_and_grad(p, beta)
        if h is None:
            break
        cur_score = sc

        # Normalize gradient for stable updates
        h_max = h.max()
        if h_max < 1e-15:
            stall += 1
            break
        h_norm = h / h_max

        # Momentum update
        v = momentum * v + (1.0 - momentum) * h_norm

        # Try multiple step sizes
        improved = False
        for a_try in (lr * 0.5, lr, lr * 2.0, lr * 0.25):
            p_try = p * np.exp(-a_try * v)
            p_try = np.maximum(p_try, 1e-18 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < cur_score - 1e-14:
                p = p_try
                cur_score = sc_try
                improved = True
                break

        if improved:
            stall = 0
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
        else:
            stall += 1
            lr = max(0.001, lr * 0.8)
            if stall > 30:
                p = best.copy()
                momentum *= 0.9
                lr = 0.05
                stall = 0
                v = np.zeros(n)
            if stall > 100:
                # Spectral reshaping escape
                p_new = _spectral_flatten_v2(best, float(rng.uniform(1.5, 4.0)), smooth=0.2)
                if p_new is not None:
                    sc_new = _score(p_new)
                    if sc_new < best_score * 1.05:
                        p = p_new
                        cur_score = sc_new
                        v = np.zeros(n)
                        lr = 0.05
                        stall = 0
                else:
                    p = best.copy()
                    cur_score = best_score
                    v = np.zeros(n)
                    lr = 0.05
                    stall = 0

        # Update beta (sharpen)
        if it % 20 == 0:
            beta = min(beta_end, beta * 1.15)

        # Periodic momentum decay
        if it % 100 == 0:
            momentum = min(0.9, momentum * 1.02)
            if cur_score > best_score * 1.1:
                p = best.copy()
                v = np.zeros(n)
                cur_score = best_score

        it += 1

    return best, best_score

            alpha_f = float(rng.uniform(1.3, 5.0))
            smooth = float(rng.uniform(0.1, 0.5))
            p_new = _spectral_flatten_v2(p, alpha_f, smooth=smooth)
            if p_new is None:
                p_new = _spectral_flatten(p, alpha_f)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-14:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.82:
            # Block scaling (target high-convolution regions)
            c = _conv_full(p, p)
            d = n * c
            # Find blocks in the sequence that correspond to high convolution regions
            # The convolution peak at position k is influenced by p[0..n-1] and p[k..k+n-1]
            # Simply: scale random blocks
            n_blocks = int(rng.integers(1, 8))
            block_size = max(1, n // (n_blocks * 4))
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_size)))
                # Decide direction based on local contribution
                # Scale down: reduce contribution to peak
                f = float(rng.choice([0.5, 0.7, 1.4, 1.8]))
                p_try = p.copy()
                p_try[start:start + block_size] *= f
                p_try = np.maximum(p_try, 1e-15)
                s_try = p_try.sum()
                if s_try < 1e-10:
                    continue
                p_try = p_try / s_try
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-14:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_try
                    break
            else:
                stall += 1

        elif r < 0.92:
            # Edge enhancement: increase edges, decrease center
            center = n // 2
            width = max(1, n // 6)
            p_try = p.copy()
            # Boost edges
            edge_f = float(rng.uniform(1.1, 1.5))
            center_f = float(rng.uniform(0.7, 0.95))
            p_try[:width] *= edge_f
            p_try[-width:] *= edge_f
            p_try[center - width:center + width] *= center_f
            p_try = np.maximum(p_try, 1e-15)
            s_try = p_try.sum()
            if s_try > 1e-10:
                p_try = p_try / s_try
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-14:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
        else:
            # Random perturbation (escape)
            if stall > 10:
                std = 0.03 + 0.01 * min(stall, 20)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = np.maximum(p_try, 1e-15)
                s_try = p_try.sum()
                if s_try > 1e-10:
                    p_try = p_try / s_try
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.1:
                        p = p_try
                        cur_score = sc_try
                        v = np.zeros(n)
                        stall = 0
            else:
                stall += 1

        # Update parameters
        if it % 15 == 0:
            beta = min(5000.0, beta * 1.12)
        if it % 50 == 0:
            momentum = min(0.85, momentum + 0.01)

        # Pull back if drifting
        if cur_score > best_score * 1.12:
            if rng.random() < 0.4:
                p = best.copy()
                cur_score = best_score
                v *= 0.5
                stall = 0

        it += 1

    return best, best_score


def _block_refine(p0, rng, time_budget):
    """Fine-grained block coordinate refinement."""
    p = np.maximum(p0, 1e-15)
    p = p / p.sum()
    n = len(p)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0)

    while time.monotonic() < deadline:
        # Choose block size
        r = rng.random()
        if r < 0.3:
            bs = int(rng.integers(1, 4))
        elif r < 0.7:
            bs = int(rng.integers(max(2, n // 512), max(3, n // 64)))
        else:
            bs = int(rng.integers(max(4, n // 64), max(8, n // 16)))
        bs = min(bs, n)
        start = int(rng.integers(0, max(1, n - bs)))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = np.maximum(p_try, 1e-15)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase: exploration -> momentum gradient -> adaptive hybrid -> block refine -> polish.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        parent = np.maximum(parent, 0)
        ps = parent.sum()
        if ps < 1e-10:
            parent = None
        else:
            parent = parent / ps
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = np.array(p, dtype=float)
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={time.monotonic()-start_time:.0f}s ***")

    # ========== PHASE 1: Exploration (15%) ==========
    print("=" * 60)
    print("Phase 1: Exploration")
    t1_end = start_time + 0.15 * budget

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.6f}, n={len(parent)}")

    # Try various initializations at multiple lengths
    explore_lengths = [2048, 3072, 4096]
    if parent is not None:
        pn = len(parent)
        if pn not in explore_lengths:
            explore_lengths.append(pn)
            explore_lengths.sort()

    for n in explore_lengths:
        if time.monotonic() > t1_end:
            break
        inits = [
            _init_u_shape(n, rng, sharpness=1.5),
            _init_u_shape(n, rng, sharpness=3.0),
            _init_bimodal(n, rng),
            _init_cosine_mod(n, rng),
            np.ones(n, dtype=float) / n,
        ]
        if parent is not None:
            sp = _init_from_parent(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for init in inits:
            if time.monotonic() > t1_end:
                break
            sc = _score(init)
            _try_update(init, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main Optimization (40%) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.55 * budget

    if best_overall is None:
        best_overall = np.ones(3072, dtype=float) / 3072
        best_score_overall = _score(best_overall)
        print(f"  Fallback uniform: {best_score_overall:.6f}")

    n_main = len(best_overall)

    # Run 1: Momentum gradient on best
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.35 * t_rem, 400.0)
        t_alloc = max(30.0, t_alloc)
        print(f"  Run 1: momentum gradient, n={n_main}, alloc={t_alloc:.0f}s")
        p, sc = _optimize_momentum(best_overall, rng, t_alloc, beta_start=20.0, beta_end=1500.0)
        _try_update(p, sc)

    # Run 2: Adaptive hybrid with different params
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.30 * t_rem, 300.0)
        t_alloc = max(30.0, t_alloc)
        print(f"  Run 2: adaptive hybrid, alloc={t_alloc:.0f}s")
        p1 = _init_from_parent(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_adaptive(p1, rng, t_alloc)
        _try_update(p, sc)

    # Run 3: Try different length
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        for n_try in [2048, 4096, 6144]:
            if n_try == n_main or time.monotonic() > t2_end - 30:
                continue
            t_alloc = min(120.0, t_rem * 0.3)
            t_alloc = max(20.0, t_alloc)
            init = _init_u_shape(n_try, rng, sharpness=2.0)
            if parent is not None:
                sp = _init_from_parent(parent, n_try, rng)
                if sp is not None:
                    # Blend
                    init = 0.5 * init + 0.5 * sp
                    init = init / init.sum()
            print(f"  Run 3: n={n_try}, alloc={t_alloc:.0f}s")
            p, sc = _optimize_adaptive(init, rng, t_alloc)
            _try_update(p, sc)

    # Run 4: Block refine
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.4 * t_rem, 150.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 4: block refine, alloc={t_alloc:.0f}s")
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts (15%) ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.72 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        n_best = len(best_overall)
        n_restarts = int(t_rem / 80)
        n_restarts = max(3, min(n_restarts, 8))

        for i in range(n_restarts):
            if time.monotonic() > t3_end - 10:
                break
            t_alloc = min(60.0, (t3_end - time.monotonic()) * 0.8)
            t_alloc = max(15.0, t_alloc)

            # Different initialization each time
            if i % 3 == 0:
                sp = _init_u_shape(n_best, rng, sharpness=float(rng.uniform(1.0, 4.0)))
            elif i % 3 == 1:
                sp = _init_bimodal(n_best, rng)
            else:
                sp = _init_from_parent(best_overall, n_best, rng)
                if sp is None:
                    sp = _init_cosine_mod(n_best, rng)

            # Blend with current best
            blend = float(rng.uniform(0.3, 0.7))
            sp = blend * sp + (1 - blend) * best_overall
            sp = np.maximum(sp, 1e-10)
            sp = sp / sp.sum()

            print(f"  Restart {i+1}/{n_restarts}, alloc={t_alloc:.0f}s")
            p, sc = _optimize_adaptive(sp, rng, t_alloc)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (remaining time) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # High-beta gradient polish
        t_rem = deadline - time.monotonic()
        if t_rem > 40:
            t_alloc = min(0.35 * t_rem, 250.0)
            t_alloc = max(20.0, t_alloc)
            print(f"  Polish 1: high-beta gradient, alloc={t_alloc:.0f}s")
            p, sc = _optimize_momentum(best_overall, rng, t_alloc, beta_start=100.0, beta_end=5000.0)
            _try_update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.30 * t_rem, 150.0)
            t_alloc = max(15.0, t_alloc)
            print(f"  Polish 2: block refine, alloc={t_alloc:.0f}s")
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # Final gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 200.0)
            t_alloc = max(15.0, t_alloc)
            print(f"  Polish 3: final gradient, alloc={t_alloc:.0f}s")
            p, sc = _optimize_adaptive(best_overall, rng, t_alloc)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = np.ones(3072, dtype=float) / 3072

    p_final = np.maximum(best_overall, 0.0)
    s = p_final.sum()
    if s < 1e-10 or not np.isfinite(s):
        p_final = np.ones(3072, dtype=float) / 3072
    else:
        p_final = p_final / s

    final_score = _score(p_final)

    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float) / 3072
    else:
        out = out / s

    return [float(x) for x in out]
