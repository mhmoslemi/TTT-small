   def fast_conv(a):
       n = len(a)
       M = 2*n - 1
       M_fft = 1 << (M-1).bit_length()
       A = np.fft.rfft(a, M_fft)
       return np.fft.irfft(A*A, M_fft)[:M]

       n = len(a)
       c = fast_conv(a)
       S = float(a.sum())
       if S < 1e-12:
           return np.inf
       return 2.0 * n * float(np.max(c)) / (S*S)

       Returns (t*, g*)
       """

       # Objective: minimize t -> c = [0, 0, ..., 0, 1]
       c_obj = np.zeros(n + 1)
       c_obj[-1] = 1.0

       A_eq = np.zeros((1, n + 1))

           # Coefficient of t is -1
           A_ub[k, n] = -1.0
           b_ub[k] = 0.0

       bounds = [(0, None)] * n + [(0, None)]
       
       res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

Looking at this problem, I need to minimize `2n * max(conv(a,a)) / (sum(a))^2`. 

Key insight: For a probability vector $b$ (sum=1), the score is simply $2n \cdot \max(\text{conv}(b,b))$. At each iteration, I can solve an LP to find a direction $g$ that minimizes the cross-correlation with the current iterate, then take a convex combination step. The LP is:

$$\min_t \quad \text{s.t.} \quad g \geq 0, \sum g = 1, \; (f * g)[k] \leq t \;\forall k$$

This is a **linear** program in $(g, t)$ since $f$ is fixed. I'll use a cutting-plane approach to keep the number of constraints small, combined with L-BFGS polishing.

LP-direction + L-BFGS hybrid optimizer for min 2n*max(conv(a,a))/(sum(a))^2.
Phase 1: Cutting-plane LP direction finding with adaptive step size.
Phase 2: L-BFGS with LSE-smoothed max for polishing.
Phase 3: Local perturbations and final refinement.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csc_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b, n):
    """Cross-convolution (a*b)[k] = sum_j a[j]*b[k-j] for k=0..2n-2."""
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)

    ff = fast_conv(f)
    # Start with top lags
    top_count = min(30, 2 * n - 1)
    initial_lags = list(np.argsort(ff)[::-1][:top_count])
    
    active_lags = set(initial_lags)
    t_star = None
    g_star = None

    for cp_iter in range(max_cp_iters):
        if len(active_lags) > max_active:
            # Keep only the most violated ones
            if g_star is not None:
                fg = cross_conv(f, g_star, n)
                sorted_lags = np.argsort(fg)[::-1]
                active_lags = set(int(k) for k in sorted_lags[:max_active])
            break

        lags_sorted = sorted(active_lags)
        num_constr = len(lags_sorted)
        num_vars = n + 1

        # Build sparse A_ub
        row_list = []
        col_list = []
        val_list = []

        for idx, k in enumerate(lags_sorted):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                f_idx = k - js
                row_list.extend([idx] * (j_end - j_start + 1))
                col_list.extend(js.tolist())
                val_list.extend(f[f_idx].tolist())
            row_list.append(idx)
            col_list.append(n)
            val_list.append(-1.0)

        A_ub = csc_matrix((val_list, (row_list, col_list)), shape=(num_constr, num_vars))
        b_ub = np.zeros(num_constr)

        A_eq = np.zeros((1, num_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        c_obj = np.zeros(num_vars)
        c_obj[n] = 1.0

        bounds_lp = [(0, None)] * num_vars

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break

        if not res.success:
            break

        t_star = float(res.x[n])
        g_star = res.x[:n].copy()
        g_star = np.maximum(g_star, 0.0)
        s = g_star.sum()
        if s < 1e-12:
            g_star[:] = 1.0 / n
        else:
            g_star /= s

        # Check for violated non-active constraints
        fg = cross_conv(f, g_star, n)
        # Sample non-active lags (for efficiency, check a subset)
        all_lags = np.arange(2 * n - 1)
        non_active_mask = np.array([k not in active_lags for k in all_lags])
        if non_active_mask.any():
            non_active_vals = fg[non_active_mask]
            violations = non_active_vals - t_star
            if violations.max() > 1e-7:
                # Add the worst violated lag(s)
                worst_indices = np.where(non_active_mask)[0][np.argsort(violations)[::-1][:5]]
                for k in worst_indices:
                    active_lags.add(int(k))
                continue
        # No violations: solution is optimal for the full problem
        return t_star, g_star

    return t_star, g_star


def lp_iteration(f, n, current_score, best_score):
    """
    One LP-based iteration: find direction, line search, update.
    Returns (new_score, new_f, improved).
    """
    t_star, g_star = solve_direction_lp_cutting_plane(f, n)
    if g_star is None:
        return current_score, f, False

    # Line search over alpha
    best_alpha_score = current_score
    best_alpha_f = f
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
        f_new = (1.0 - alpha) * f + alpha * g_star
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            continue
        f_new /= s
        sc = compute_score(f_new)
        if sc < best_alpha_score:
            best_alpha_score = sc
            best_alpha_f = f_new.copy()
        if sc < best_score:
            break

    improved = best_alpha_score < current_score - 1e-12
    return best_alpha_score, best_alpha_f, improved


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        p = parent.astype(np.float64).copy()
        s = p.sum()
        if s > 1e-12:
            inits.append(p / s)
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64) / n
    for freq in [1, 2, 3, 5, 8]:
        for phase in [0, np.pi / 2, np.pi]:
            a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
            a = np.maximum(a, 0.01)
            inits.append(a / a.sum())
    inits.append(np.linspace(0.01, 1.0, n))
    last = inits[-1]
    inits.append(last / last.sum())
    inits.append(np.linspace(1.0, 0.01, n) / np.linspace(1.0, 0.01, n).sum())
    for scale in [0.2, 0.5, 1.0, 2.0]:
        a = np.clip(1.0/n + scale * rng.standard_normal(n) / n, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            inits.append(a / s)
    return inits


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    s = a_pert.sum()
    if s > 1e-12:
        a_pert /= s
    return a_pert


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-direction method + L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) / 1024.0

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = min(max(parent.size, 1024), 16384)
        lengths.append(pa)
    for n_try in [4096, 6962, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, size={parent.size if parent is not None else 0}")

    # ============================================================
    # Phase 1: LP-based iterations (55% of time)
    # ============================================================
    phase1_deadline = t_start + 0.55 * (total_budget - 15.0)
    print(f"\n=== Phase 1: LP-direction iterations ===")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- LP phase: n = {n} ---")
        inits = build_inits(n, rng, parent)

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            f = a0.astype(np.float64).copy()
            s = f.sum()
            if s < 1e-12:
                f[:] = 1.0 / n
            else:
                f /= s

            cur_sc = compute_score(f)
            init_best_sc = cur_sc
            stall_count = 0

            report(cur_sc, f, f"n={n}, init={init_idx}, start")

            # LP iterations
            max_lp_iters = 200
            for lp_iter in range(max_lp_iters):
                if time.monotonic() > phase1_deadline:
                    break
                # Solve LP and update
                new_sc, new_f, improved = lp_iteration(f, n, cur_sc, best_score)
                cur_sc = new_sc
                f = new_f
                report(cur_sc, f, f"n={n}, init={init_idx}, lp={lp_iter}")

                if improved:
                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count >= 10:
                        # Perturb and restart LP
                        f = targeted_perturb(f, rng, strength=0.05)
                        cur_sc = compute_score(f)
                        stall_count = 0

            # Quick L-BFGS polish after LP phase
            if time.monotonic() < phase1_deadline:
                for beta in [200, 500]:
                    if time.monotonic() > phase1_deadline:
                        break
                    sc, a_opt = run_lbfgs(f, beta, maxiter=3000, maxfun=50000)
                    report(sc, a_opt, f"n={n}, init={init_idx}, lbfgs b={beta}")
                    f = a_opt

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: High-beta L-BFGS refinement (25% of time)
    # ============================================================
    phase2_deadline = t_start + 0.80 * (total_budget - 15.0)
    print(f"\n=== Phase 2: L-BFGS refinement ===")

    working = best_a.copy()
    s = working.sum()
    if s > 1e-12:
        working /= s

    for beta in [300, 800, 2000, 5000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs(working, beta, maxiter=10000, maxfun=200000)
        report(sc, a_opt, f"refine beta={beta}")
        working = a_opt

    # ============================================================
    # Phase 3: Local search with perturbations (12% of time)
    # ============================================================
    phase3_deadline = t_start + 0.92 * (total_budget - 15.0)
    print(f"\n=== Phase 3: Local search ===")

    n_best = len(best_a)
    working = best_a.copy()
    s = working.sum()
    if s > 1e-12:
        working /= s

    n_local = 40
    for trial in range(n_local):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            scale = 0.02 + 0.05 * rng.random()
            a_pert = working * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0)
        elif strategy == 1:
            a_pert = targeted_perturb(working, rng, strength=0.03 + 0.04 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = working * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.maximum(a_pert, 0)
        elif strategy == 3:
            # Swap two random blocks
            a_pert = working.copy()
            blk = max(10, n_best // 200)
            i = int(rng.integers(0, n_best - 2 * blk))
            j = int(rng.integers(0, n_best - 2 * blk))
            if abs(i - j) > blk:
                a_pert[i:i+blk], a_pert[j:j+blk] = a_pert[j:j+blk].copy(), a_pert[i:i+blk].copy()
        else:
            # LP-based direction from current point
            t_star, g_star = solve_direction_lp_cutting_plane(working, n_best, max_cp_iters=15, max_active=100)
            if g_star is not None:
                a_pert = 0.5 * working + 0.5 * g_star
            else:
                a_pert = working * (1.0 + 0.03 * rng.standard_normal(n_best))
                a_pert = np.maximum(a_pert, 0)

        s_p = a_pert.sum()
        if s_p > 1e-12:
            a_pert /= s_p
        sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"local #{trial} s{strategy}")
        if sc < best_score:
            working = a_opt.copy()
            s_w = working.sum()
            if s_w > 1e-12:
                working /= s_w

    # ============================================================
    # Phase 4: Final polish (5% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final polish ===")

    working = best_a.copy()
    s = working.sum()
    if s > 1e-12:
        working /= s

    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline:
            break
        sc, a_opt = run_lbfgs(working, beta, maxiter=20000, maxfun=500000)
        report(sc, a_opt, f"final beta={beta}")
        working = a_opt

    # Try one more LP-based round from the best
    if time.monotonic() < deadline - 30:
        n_w = len(working)
        s_w = working.sum()
        if s_w > 1e-12:
            working /= s_w
        for _ in range(20):
            if time.monotonic() > deadline - 10:
                break
            new_sc, new_f, improved = lp_iteration(working, n_w, compute_score(working), best_score)
            if improved:
                working = new_f
                report(new_sc, working, "final lp")
            else:
                break

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = evaluate_sequence(out)
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
