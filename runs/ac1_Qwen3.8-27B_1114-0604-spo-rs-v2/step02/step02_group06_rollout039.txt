def _solve_direction_lp(a, n_constraints=None):
    """Solve the direction-finding LP.
    min M s.t. conv(a', a)_k <= M for k in selected lags,
               sum(a') = 1, a' >= 0
    Returns: a' (the optimal direction)
    """
    n = len(a)
    # Build constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For k in selected lags
    
    if n_constraints is None:
        selected = list(range(2*n-1))
    else:
        # Select top n_constraints lags by |a| contribution
        c = _fft_convolve(a, a)
        selected = list(np.argsort(c)[-n_constraints:]).tolist()
    
    m = len(selected)
    # A is m x n
    A = np.zeros((m, n))
    for idx, k in enumerate(selected):
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        for j in range(j_min, j_max):
            A[idx, j] = a[k - j]
    
    # Constraints: A @ a' - M*1 <= 0 → A @ [a'; -M] <= 0
    # Also: sum(a') = 1, a' >= 0, M is free (but M >= 0)
    
    # Variables: [a'_0, ..., a'_{n-1}, M] (n+1 variables)
    # Objective: minimize M → c_obj = [0, 0, ..., 0, 1]
    # Inequality: A @ a' - M <= 0 for each k → [A, -1] @ [a'; M] <= 0
    # Equality: sum(a') = 1 → [1, 1, ..., 1, 0] @ [a'; M] = 1
    # Bounds: a'_j in [0, inf], M in [0, inf]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    return None

        else:
            stall += 1
            if stall > 30:
                noise = a * np.exp(_rng.normal(0, 0.08, n))
                noise = _sanitize(noise, 1e-15)
                if noise is not None:
                    sc2 = _objective(noise)
                    if sc2 < best_score * 1.01:
                        a = noise
                        cur_score = sc2
                else:
                    a = best.copy()
                    cur_score = best_score
                stall = 0
            if stall > 150:
                break

        it += 1
    return best, best_score


def _resample(a, m):
    n = len(a)
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-15
    return _sanitize(out, 1e-15)


def _make_initial(n, seed_offset=0):
    rng = np.random.default_rng(42 + seed_offset)
    t = (np.arange(n) + 0.5) / n
    mode = int(rng.integers(0, 5))

    if mode == 0:
        # U-shape with asymmetry
        a_left = float(rng.uniform(0.5, 2.0))
        a_right = float(rng.uniform(0.5, 2.0))
        p = np.concatenate([
            np.full(n // 2, a_left) * (1.0 - 0.5 * np.cos(np.pi * np.arange(n // 2) / max(1, n // 4))),
            np.full(n - n // 2, a_right) * (1.0 - 0.5 * np.cos(np.pi * np.arange(n - n // 2) / max(1, n // 4)))
        ])
        p = p + 0.3 * (1.0 - 2.0 * t) ** 2
    elif mode == 1:
        # Random smooth
        k = int(rng.integers(2, 8))
        p = np.ones(n)
        for _ in range(k):
            freq = float(rng.uniform(1, 6))
            phase = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.1, 0.4))
            p = p + amp * np.cos(2 * np.pi * freq * t + phase)
    elif mode == 2:
        # Bell with dip
        center = float(rng.uniform(0.3, 0.7))
        width = float(rng.uniform(0.1, 0.4))
        p = np.exp(-((t - center) / width) ** 2)
        p = p + 0.5 * np.exp(-((t - (1 - center)) / width) ** 2)
    elif mode == 3:
        # Two-bump asymmetric
        c1 = float(rng.uniform(0.1, 0.4))
        c2 = float(rng.uniform(0.6, 0.9))
        w1 = float(rng.uniform(0.05, 0.2))
        w2 = float(rng.uniform(0.05, 0.2))
        p = np.exp(-((t - c1) / w1) ** 2) + float(rng.uniform(0.5, 1.5)) * np.exp(-((t - c2) / w2) ** 2)
    else:
        p = 1.0 + 0.3 * np.sin(2 * np.pi * float(rng.uniform(2, 5)) * t + float(rng.uniform(0, 6)))

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.01)
    return _sanitize(p, 1e-15)
