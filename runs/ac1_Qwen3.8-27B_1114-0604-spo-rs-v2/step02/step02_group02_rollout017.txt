"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2 using:
1. Diverse parametric + random initialization
2. Projected gradient descent with Nesterov momentum and BB step size
3. Peak-targeted coordinate descent for local refinement
4. Structured simulated annealing with peak-aware perturbations
5. Multi-start with time-adaptive strategy switching

import numpy as np
import time


def _autocorr(f):
    """Full self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0.0
    if not cond.any():
        return np.ones(n) / n
    r = np.max(np.where(cond)[0]) + 1
    theta = (cssv[r - 1] - 1.0) / r
    return np.maximum(v - theta, 0.0)


def _subgrad_topk(f, c, k):
    """Subgradient: weighted average of gradients at top-k peak positions."""
    n = len(f)
    top = np.argsort(c)[-k:]
    # Weight by exponential of proximity to max
    vals = c[top]
    w = np.exp(10.0 * (vals - vals.max()) / max(vals.max(), 1e-30))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, pos in zip(w, top):
        j1_lo = max(0, pos - n + 1)
        j1_hi = min(n - 1, pos)
        if j1_lo <= j1_hi:
            j1 = np.arange(j1_lo, j1_hi + 1)
            grad[j1] += wi * f[pos - j1]
        j2_lo = max(0, pos)
        j2_hi = min(n - 1, pos + n - 1)
        if j2_lo <= j2_hi:
            j2 = np.arange(j2_lo, j2_hi + 1)
            grad[j2] += wi * f[j2 - pos]
    return 2.0 * grad


def _compute_contribs(f, pos, n):
    """Compute contribution f[j]*f[pos-j] for each j to (f*f)[pos]."""
    contribs = np.zeros(n)
    j1_lo = max(0, pos - n + 1)
    j1_hi = min(n - 1, pos)
    if j1_lo <= j1_hi:
        j1 = np.arange(j1_lo, j1_hi + 1)
        contribs[j1] += f[j1] * f[pos - j1]
    j2_lo = max(0, pos)
    j2_hi = min(n - 1, pos + n - 1)
    if j2_lo <= j2_hi:
        j2 = np.arange(j2_lo, j2_hi + 1)
        contribs[j2] += f[j2] * f[pos - j2]
    return contribs


def _cd_step(f, c, peak, n, rng):
    """One coordinate descent step: move mass from peak contributors to non-contributors."""
    k_star = int(np.argmax(c))
    contribs = _compute_contribs(f, k_star, n)
    total_contrib = contribs.sum()
    if total_contrib < 1e-30:
        return f, False
    
    # Find top contributors (high f[j] * f[k_star-j])
    n_top = max(5, n // 100)
    top_idx = np.argsort(contribs)[-n_top:]
    # Find low contributors
    bottom_idx = np.argsort(contribs)[:n_top]
    
    best_f = f.copy()
    best_peak = peak
    improved = False
    
    for frac in [0.1, 0.05, 0.02, 0.15, 0.2]:
        for t_i in top_idx[:5]:
            for b_i in bottom_idx[:5]:
                if t_i == b_i:
                    continue
                amt = f[t_i] * frac
                if amt < 1e-18 or f[t_i] - amt < -1e-18:
                    continue
                f_try = f.copy()
                f_try[t_i] -= amt
                f_try[b_i] += amt
                f_try = _project_simplex(f_try)
                c_try = _autocorr(f_try)
                p_try = c_try.max()
                if p_try < best_peak - 1e-14:
                    best_peak = p_try
                    best_f = f_try
                    improved = True
    return best_f, improved


def _cd_step_multi(f, c, peak, n, rng, n_steps=5):
    """Multiple coordinate descent steps targeting all near-peak positions."""
    f_curr = f.copy()
    peak_curr = peak
    for _ in range(n_steps):
        f_curr, improved = _cd_step(f_curr, c, peak_curr, n, rng)
        if improved:
            c = _autocorr(f_curr)
            peak_curr = c.max()
        else:
            break
    return f_curr, peak_curr


def _optimize_hybrid(f0, deadline, rng, n):
    """Hybrid optimizer: PGD with momentum + CD + SA restarts."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _autocorr(f)
    peak0 = c.max()
    best_peak = peak0
    best_f = f.copy()

    # PGD state
    momentum_alpha = 0.85
    v = np.zeros(n)
    step = 1.0
    T = peak0 * 0.02
    last_improve = time.monotonic()
    it = 0
    last_cd_time = 0.0
    use_cd = False

    while time.monotonic() < deadline:
        c = _autocorr(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()

        stuck = time.monotonic() - last_improve

        # When stuck, switch strategy
        if stuck > 30:
            # Aggressive restart
            T = best_peak * 0.1
            mag = min(0.05, 0.01 + 0.005 * (stuck - 30) / 30.0)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            v = np.zeros(n)
            step = max(step, 0.5)
            last_improve = time.monotonic()
            use_cd = True
        elif stuck > 10:
            # Moderate perturbation
            T = max(T, best_peak * 0.03)
            mag = 0.01
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            v *= 0.5
            last_improve = time.monotonic()
            use_cd = True
        else:
            use_cd = False

        if use_cd:
            # Coordinate descent phase
            f, new_peak = _cd_step_multi(f, c, cur_peak, n, rng, n_steps=3)
            if new_peak < cur_peak - 1e-14:
                cur_peak = new_peak
                step = max(step * 1.05, 0.1)
            else:
                step *= 0.8
        else:
            # PGD phase with adaptive k
            k = 3 + int(4 * min(1.0, it / 10000.0))
            grad = _subgrad_topk(f, c, k)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-20:
                break

            f_new = _project_simplex(f + v)

            c_new = _autocorr(f_new)
            new_peak = c_new.max()
            delta = new_peak - cur_peak

            if delta <= 0:
                f = f_new
                #Increase step on success

                    f = f_new
                else:
                    step *= 0.6
                    v *= 0.3
            T = max(T * 0.998, best_peak * 1e-16)

        it += 1
        if it % 10000 == 0:
            print(f"    Hybrid it={it}: peak={cur_peak:.12f}, best={best_peak:.12f}, "
                  f"score={2*n*best_peak:.6f}, step={step:.4f}")

    c = _autocorr(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    return best_f, 2.0 * n * best_peak


def _polish_cd(f, deadline, rng, n):
    """Aggressive coordinate descent polish."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    f /= f.sum()

                f_try[i] -= amt
                f_try[j] += amt

                f_try = _project_simplex(f_try)

def _generate_candidates(f_init, n, rng, count=60):
    """Generate diverse candidate sequences."""
    j = np.arange(n, dtype=np.float64)
    candidates = []

        f = np.maximum(0.0, 1.0 - (j / slope) ** 2)
        if f.sum() > 1e-10:
            candidates.append(f / f.sum())

    # Power law: f[i] = (1 - i/n)^p
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        f = (1.0 - j / n) ** p
        f = np.maximum(f, 0.0)
        if f.sum() > 1e-10:
            candidates.append(f / f.sum())

    # Exponential: f[i] = exp(-i/tau)
    for tau in [n * 0.05, n * 0.1, n * 0.2, n * 0.3, n * 0.5, n * 1.0, n * 2.0]:
        f = np.exp(-j / tau)
        candidates.append(f / f.sum())

    # Cosine window: f[i] = cos(pi * i / (2*n))^p
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f = np.power(np.cos(np.pi * j / (2.0 * n) + 1e-10), p)
        f = np.maximum(f, 0.0)
        if f.sum() > 1e-10:
            candidates.append(f / f.sum())

    # Two-sided (peak at position c)
    for c_frac in [0.1, 0.2, 0.3, 0.4]:
        for tau in [n * 0.1, n * 0.2, n * 0.5]:
            c = c_frac * n
            f = np.exp(-np.abs(j - c) / tau)
            candidates.append(f / f.sum())

    # U-shaped (inverted)
    for alpha in [1.0, 3.0, 10.0, 30.0]:
        f = 0.1 + alpha * (j / n) ** 2
        candidates.append(f / f.sum())

    # Random starts
    remaining = count - len(candidates)
    for _ in range(max(0, remaining)):
        r = rng.random()
        if r < 0.25:
            f = rng.exponential(1.0, n)
        elif r < 0.5:
            f = np.abs(rng.standard_normal(n)) + 0.01
        elif r < 0.7:
            # Near-monotone
            base = np.exp(-j / (n * rng.uniform(0.5, 3.0)))
            f = base + 0.05 * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
        elif r < 0.85:
            # Sparse
            f = np.zeros(n)
            nnz = int(n * rng.uniform(0.3, 0.8))
            idx = rng.choice(n, nnz, replace=False)
            f[idx] = rng.exponential(1.0, nnz)
        else:
            # Smoothed random
            f = np.abs(rng.standard_normal(n))
            kernel_size = max(3, n // 100)
            kernel = np.ones(kernel_size) / kernel_size
            f = np.convolve(f, kernel, mode='same')
        if f.sum() > 1e-10:
            candidates.append(f / f.sum())
        else:
            candidates.append(np.ones(n) / n)
