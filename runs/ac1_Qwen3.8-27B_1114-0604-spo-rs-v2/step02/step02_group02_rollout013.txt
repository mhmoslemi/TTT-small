Actually, convolve(f,f)[k] = sum_{i: valid} f[i]*f[k-i]. This is a sum of products of pairs of components. For fixed k, this is a quadratic form f^T A_k f where A_k is a symmetric 0-1 matrix (specifically, A_k[i,j] = 1 if i+j=k, else 0, appropriately bounded).

Is it convex? A quadratic form f^T A f is convex iff A is positive semi-definite. The matrix A_k (which is a "reversal" submatrix) - is it PSD?

- The right endpoint t = a

import numpy as np
import time

def _autocorr_fft(f):
    """Linear autocorrelation (full convolution) of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

def _multipeak_gradient(f, c, m, alpha):
    """Compute weighted gradient from top-m peaks."""
    n = len(f)
    # Find top-m peaks
    top_idx = np.argsort(c)[-m:]
    top_vals = c[top_idx]
    max_val = top_vals.max()
    
    # Weights: exponential decay from the max
    weights = np.exp(-alpha * (max_val - top_vals))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lo = max(0, idx - n + 1)
        hi = min(n, idx + 1)
        # Contribution of f[i] to c[idx] is f[idx - i]
        paired = idx - np.arange(lo, hi)
        grad[lo:hi] += w * f[paired]
    
    return grad

def _mirror_step(f, grad, eta):
    """Mirror descent step on the simplex (with non-negativity)."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _score(f):
    """Compute the objective value."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr_fft(f)
    return 2.0 * n * c.max() / (s * s)

def _optimize(f0, deadline, rng, n, phase=1):
    """
    Main optimization loop with adaptive mirror descent.
    phase=1: exploration (larger steps, more SA)
    phase=2: exploitation (smaller steps, targeted gradient)
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    
    c = _autocorr_fft(f)
    peak0 = c.max()
    best_peak = peak0
    best_f = f.copy()
    
    if phase == 1:
        eta = 1.0
        T = peak0 * 0.1
        m_peaks = 3
        alpha_w = 2.0
    else:
        eta = 0.3
        T = peak0 * 0.01
        m_peaks = 2
        alpha_w = 5.0
    
    T_floor = peak0 * 1e-15
    last_improve = time.monotonic()
    iter_count = 0
    stall_threshold = 20 if phase == 1 else 10
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-15:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()
        
        stuck = time.monotonic() - last_improve
        
        # Reheating / perturbation when stuck
        if stuck > stall_threshold:
            T = peak0 * (0.15 if phase == 1 else 0.05)
            eta = max(eta, 0.8 if phase == 1 else 0.4)
            mag = min(0.08, 0.01 + 0.005 * (stuck / 30.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
        else:

            eta *= 0.99997

def _generate_starts(f_init, n, rng, count=30):

        f[:boundary] = 0

    f = np.zeros(n)

        right_edge = (j >= center + half_w*0.5) & (j <= center + half_w)
        f[inner] = 1.0

        f[right_edge] = 0.5 * (1 - np.sin(np.pi * (j[right_edge] - (center + half_w*0.5)) / edge_w))
        f[:boundary] = 0
        f[n-boundary:] = 0
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

    for (f1, f2, f3) in [(1, 3, 5), (2, 5, 7), (1, 4, 6), (3, 7, 11)]:
        f = 1.0 + 0.5*np.cos(2*np.pi*f1*j/n) + 0.3*np.cos(2*np.pi*f2*j/n) + 0.2*np.cos(2*np.pi*f3*j/n)
        f = np.maximum(f, 0)
        f[:boundary] = 0
        f[n-boundary:] = 0
        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)

        if r < 0.3:
            f = rng.exponential(1.0, n)

        s = f.sum()
        if s > 1e-10:
            starts.append(f / s)
        else:
            starts.append(np.ones(n) / n)
