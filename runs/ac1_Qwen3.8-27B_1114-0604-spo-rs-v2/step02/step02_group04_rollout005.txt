**Key algorithmic insight: LP-based alternating minimization**

The "state of the art" approach in the problem description is essentially:
1. Fix a, compute M = max(a*a)
2. Solve: max sum(b) s.t. (a*b)[k] <= M ∀k, b >= 0
3. Update a toward b

This is aasarkan block coordinate descent. Each LP solve gives the best "response" to the current a.

Let me implement this carefully:

```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    N = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)
    return c[:2*n-1]

def _score(a):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0*a.size*mx/(s*s)

def _build_conv_matrix(n, a):
    """Build the sparse matrix C where (C @ b)[k] = (a*b)[k]."""
    # C[k, i] = a[i] if 0 <= k-i < n, else 0
    # For each i, k ranges from i to i+n-1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        if a[i] > 1e-15:
            k_start = i
            k_end = i + n  # exclusive
            ks = np.arange(k_start, min(k_end, 2*n-1))
            rows.extend(ks.tolist())
            cols.extend([i]*len(ks))
            vals.extend([a[i]]*len(ks))
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _solve_lp(a, M):
    """Solve: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    C = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return a.copy()  # fallback

def _lp_optimize(a_init, deadline, max_iter=5000):
    a = a_init.copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size)/a.size
        s = 1.0
    a = a/s
    
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(42)
    
    t = 1.0  # initial mixing parameter
    stall_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = c.max()
        
        # Solve LP
        b = _solve_lp(a, M)
        if b is None or not b.any():
            break
        b = b / b.sum()
        
        # Update
        a_new = (1-t)*a + t*b
        a_new = np.clip(a_new, 0, 1000)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Optionally increase t
            t = min(1.0, t * 1.05)
        else:
            stall_count += 1
            if stall_count > 50:
                # Reset with perturbation
                t = max(0.1, t * 0.8)
                a = best_a + rng.standard_normal(n) * 0.01 * best_a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                stall_count = 0
    
    return best_a, best_score

"""
Low-peak step-function optimization via LP-based alternating minimization
with gradient polish.

Algorithm:
1. Initialize with anti-palindromic sinusoidal sequence or height_sequence_1.
2. LP-based alternating minimization: fix a, solve LP for optimal b, update a.
3. Gradient-based polish with mirror descent.
4. Validate and return best.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * a.size * mx / (s * s))


def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _precompute_conv_structure(n):
    """Precompute row/col indices for the convolution matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for k in range(i, min(i + n, 2 * n - 1)):
            rows_list.append(k)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def _build_conv_matrix(n, a, rows, cols):
    """Build sparse convolution matrix from precomputed structure."""
    vals = a[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(a, M, rows, cols):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    C = _build_conv_matrix(n, a, rows, cols)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_optimize(a_init, deadline, max_iter=10000):
    """LP-based alternating minimization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    rows, cols = _precompute_conv_structure(n)

    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(42)

    t_mix = 1.0
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = _fft_convolve(a)
        M = float(c.max())

        b = _solve_lp(a, M, rows, cols)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b / b_sum

        a_new = (1.0 - t_mix) * a + t_mix * b
        a_new = np.clip(a_new, 0, 1000)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
            t_mix = min(1.0, t_mix * 1.02)
        else:
            stall += 1
            if stall > 20:
                t_mix = max(0.1, t_mix * 0.7)
                if stall > 100:
                    a = best_a + rng.standard_normal(n) * 0.005 * best_a.mean()
                    a = np.clip(a, 0, 1000)
                    a = a / a.sum()
                    stall = 0
                    t_mix = 0.5

        if it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={t_mix:.3f}", flush=True)

    return best_a, best_score


def _gradient_polish(a_init, deadline, T_start=5.0, T_end=50.0):

    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.RandomState(99)

    lr = 0.1 * n
    it = 0
    last_improve = 0
    patience = 2000

    while time.monotonic() < deadline and it < 50000:
        it += 1
        t_frac = min(1.0, it / 20000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr_curr = lr / (1.0 + 0.01 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Softmax weights
        c_min = c.min()
        c_max = c.max()
        rng_ = c_max - c_min
        if rng_ < 1e-15:
            w = np.ones(c.size) / c.size
        else:
            c_shift = (c - c_max) / rng_ * T  # in [-T, 0]
            exp_c = np.exp(c_shift)
            s_exp = exp_c.sum()
            if s_exp < 1e-300:
                w = np.zeros(c.size)
                w[c.argmax()] = 1.0
            else:
                w = exp_c / s_exp

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        update = -lr_curr * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:

                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(500, int(patience * 1.5))
                lr = max(0.02 * n, lr * 0.8)

```python
"""
Low-peak step-function optimization via LP-based alternating minimization
with gradient polish.

The key insight is that fixing one sequence 'a' and optimizing the 'response'
sequence 'b' under the constraint (a*b)[k] <= max(a*a) is a linear program.
Alternating between a and b via this LP provides a principled descent direction.

Algorithm:
1. Initialize with anti-palindromic sinusoid or height_sequence_1.
2. LP-based alternating minimization (main optimization).
3. Mirror descent gradient polish.
4. Validate and return best.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (matches evaluate_sequence for valid input)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * a.size * mx / (s * s))


def _grad_via_fft(w, a):
    """Gradient via FFT cross-correlation."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size > N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]
