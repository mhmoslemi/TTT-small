```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy import sparse

def autocorr_peak(f):
    """Compute peak of linear self-convolution."""
    n = len(f)
    c = np.convolve(f, f)
    return c.max(), c

def solve_lp_direction(f, M, n):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k.
    Returns the optimal g (normalized) or None if infeasible/unbounded.
    """
    # Constraint matrix A: shape (2n-1, n)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M (vector of M's)
    b = np.full(2*n - 1, M)
    
    # g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    g = result.x
    if g.sum() <= 0:
        return None
    
    return g / g.sum()  # Normalize to sum 1
```

**Approach: "Push down the peak" via targeted coordinate updates**

1. Find the peak lag k*
2. The peak is $(f*f)(k*) = \sum_i f[i] f[k*-i]$
3. To reduce this, we can:

   b. But this increases other lags (due to the sum constraint)

1. Find the peak lag k*
2. Find the "worst" index i (largest f[i] * f[k*-i])
3. Find the "best" index j (smallest contribution to all near-peak lags)
4. Move mass from i to j
5. Repeat

This is the coordinate descent / "pair move" approach in the polishing phase. If done well, it might be simple and effective.

```python
"""
Step function optimizer using LP-based direction finding with line search,
combined with mirror descent refinement and coordinate polishing.
Key innovation: uses the LP relaxation (max sum(g) s.t. g>=0, (f*g)<=M)
to find globally-best descent directions, with adaptive line search and
structured perturbations for escaping local minima.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _convolve(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score(f):
    """Compute 2n * peak(f*f) / (sum f)^2."""
    n = len(f)
    c = _convolve(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _build_lp_matrix(f, n, rows, cols):
    """Build the sparse LP constraint matrix using precomputed structure."""
    # vals[i * n + j] corresponds to the entry at (rows[i*n+j], cols[i*n+j])
    # For diagonal s (offset -s), all entries have value f[s]
    # Rows for diagonal s: s, s+1, ..., s+n-1
    # Cols for diagonal s: 0, 1, ..., n-1
    # So vals is: [f[0]]*n, [f[1]]*n, ..., [f[n-1]]*n
    vals = np.repeat(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _precompute_lp_structure(n):
    """Precompute row and column indices for the LP constraint matrix."""
    # For each s in 0..n-1 (diagonal offset -s):
    #   rows: s, s+1, ..., s+n-1
    #   cols: 0, 1, ..., n-1
    rows = np.concatenate([np.arange(s, s + n) for s in range(n)])
    cols = np.tile(np.arange(n), n)
    return rows, cols


def _solve_lp_direction(f, M, n, rows, cols):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for k=0..2n-2.
    Returns normalized g (sum=1) or None.
    """
    A = _build_lp_matrix(f, n, rows, cols)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 30})
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    g = res.x
    g_sum = g.sum()
    if g_sum <= 1e-15:
        return None

    return g / g_sum


def _line_search(f, g, t_values, n):
    """Try multiple step sizes, return best (t, f_new, peak)."""
    best_t = 0.0
    best_f = f.copy()
    best_peak = _convolve(f).max()

    for t in t_values:
        f_try = (1.0 - t) * f + t * g
        f_try = np.maximum(f_try, 0.0)
        s = f_try.sum()
        if s < 1e-10:
            continue
        f_try /= s
        c = _convolve(f_try)
        peak = c.max()
        if peak < best_peak:
            best_peak = peak
            best_f = f_try.copy()
            best_t = t
    return best_t, best_f, best_peak


def _lp_optimize(f0, deadline, rng, n, rows, cols, max_iterations=200):
    """
    LP-based optimization loop.
    Each iteration: solve LP for direction, line search, update.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _convolve(f)
    local_best = c.max()
    local_best_f = f.copy()
    last_improve = time.monotonic()
    iter_count = 0
    stuck_count = 0

    print(f"  LP start: peak={local_best:.10f}, score={2*n*local_best:.6f}")

    while time.monotonic() < deadline and iter_count < max_iterations:
        c = _convolve(f)
        cur_peak = c.max()
        M = cur_peak

        # Solve LP
        g = _solve_lp_direction(f, M, n, rows, cols)

        if g is None:
            # LP infeasible or failed; perturb and increase M
            stuck_count += 1
            mag = 0.02 * min(1.0, stuck_count * 0.3)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            iter_count += 1
            continue

        iter_count += 1
        g = np.maximum(g, 0.0)
        g /= g.sum()

        # Check if g is meaningfully different from f
        diff = np.abs(g - f).sum()
        if diff < 1e-8:
            stuck_count += 1
            if stuck_count > 5:
                # Perturb significantly
                f = f + 0.05 / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
            continue

        # Line search
        t_values = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t, f_new, new_peak = _line_search(f, g, t_values, n)

        if new_peak < cur_peak - 1e-14:
            f = f_new
            if new_peak < local_best:
                local_best = new_peak
                local_best_f = f.copy()
                last_improve = time.monotonic()
            stuck_count = 0
            if iter_count % 5 == 0:
                print(f"  LP iter {iter_count}: peak={new_peak:.10f}, "
                      f"best={local_best:.10f}, t={best_t:.2f}")
        else:
            # No improvement from LP direction; try smaller steps or perturb
            stuck_count += 1
            small_t = [0.01, 0.02, 0.03, 0.05]
            best_t2, f_new2, new_peak2 = _line_search(f, g, small_t, n)
            if new_peak2 < cur_peak - 1e-14:
                f = f_new2
                if new_peak2 < local_best:
                    local_best = new_peak2
                    local_best_f = f.copy()
                    last_improve = time.monotonic()
                stuck_count = 0
            else:
                # Perturb
                mag = 0.01 + 0.01 * min(stuck_count, 5)
                f = f + mag / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()

        # Check time
        if time.monotonic() > deadline - 30:
            break

    c = _convolve(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _mirror_descent(f0, deadline, rng, n, k_max=8):
    """
    Mirror descent (exponential gradient) with adaptive top-k and SA.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _convolve(f)
    local_best = c.max()
    local_best_f = f.copy()
    last_improve = time.monotonic()
    iter_count = 0
    eta = 1.0
    T = local_best * 0.03

    while time.monotonic() < deadline:
        c = _convolve(f)
        cur_peak = c.max()

        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()
            last_improve = time.monotonic()

        stuck = time.monotonic() - last_improve

        # Adaptive k: start small, grow over time
        progress = min(1.0, iter_count / 50000.0)
        k = 2 + int(k_max * progress)

        # Top-k subgradient
        top_k = np.argsort(c)[-k:]
        grad = np.zeros(n)
        for idx in top_k:
            lo = max(0, idx - n + 1)
            hi = min(n, idx + 1)
            grad[lo:hi] += 2.0 * f[idx - np.arange(lo, hi)]
        grad /= k

        # Heuristic: stuck -> perturb and reheat
        if stuck > 20:
            T = local_best * 0.05
            mag = min(0.05, 0.005 + 0.002 * stuck / 30.0)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
            eta = 1.0
        else:
            T = max(local_best * 1e-12, T * 0.9997)
            eta *= 0.99997

        # Mirror descent step
        scale = grad.max() - grad.min()
        if scale < 1e-18:
            iter_count += 1
            continue
        log_f = np.log(f + 1e-300) - (eta / scale) * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-300:
            iter_count += 1
            continue
        f_new /= s_new

        c_new = _convolve(f_new)
        new_peak = c_new.max()
        delta = new_peak - cur_peak

        if delta <= -1e-14:
            f = f_new
            eta = min(eta * 1.01, 5.0)
        elif T > 1e-30:
            if np.log(max(rng.uniform(), 1e-300)) < -delta / T:
                f = f_new
            else:
                eta *= 0.9
        else:
            eta *= 0.85

        iter_count += 1
        if iter_count % 100000 == 0:
            print(f"    MD iter {iter_count}: peak={cur_peak:.10f}, "
                  f"best={local_best:.10f}, eta={eta:.4f}")

    c = _convolve(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _polish(f0, deadline, rng, n):
    """
    Coordinate descent polish: targeted pair moves to reduce peak.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    c = _convolve(f)
    local_best = c.max()
    local_best_f = f.copy()
    pass_count = 0
    no_improve = 0

    while time.monotonic() < deadline:
        c = _convolve(f)
        cur_peak = c.max()

        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            # Perturb to escape
            f = f + 0.003 / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            no_improve = 0
            pass_count += 1
            continue

        # Find peak lag
        k_star = int(np.argmax(c))

        # Compute contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        contrib[lo:hi] = 2.0 * f[k_star - np.arange(lo, hi)]

        # Also consider near-peak lags
        near_lags = np.argsort(c)[-3:]
        contrib_all = np.zeros(n)
        for kl in near_lags:
            lo2 = max(0, kl - n + 1)
            hi2 = min(n, kl + 1)
            contrib_all[lo2:hi2] += f[kl - np.arange(lo2, hi2)]
        contrib_all *= 2.0

        # Donor: highest contribution to peak
        donor = int(np.argmax(contrib_all))

        # Recipient: low contribution
        sorted_idx = np.argsort(contrib_all)
        candidates = sorted_idx[:min(20, n // 20)]

        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            for recv in candidates[:10]:
                if recv == donor or f[donor] * frac < 1e-15:
                    continue
                f_try = f.copy()
                amt = f[donor] * frac
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try /= s_try
                c_try = _convolve(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break
            if improved:
                break

        # Random pair moves if stuck
        if not improved:
            for _ in range(50):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.02, 0.05, 0.1, 0.2])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0.0)
                f_try /= f_try.sum()
                c_try = _convolve(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    break

        pass_count += 1

    c = _convolve(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _generate_starts(f_init, n, rng, count=30):
    """Generate diverse starting points."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Various structured starts
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        starts.append(np.exp(-alpha * ((j - center) / center) ** 2))

    for alpha in [0.3, 1.0, 3.0]:
        starts.append(1.0 + alpha * np.abs((j - center) / center))

    for freq in [1, 2, 3, 4, 6, 8]:
        starts.append(1.0 + 0.5 * np.sin(2 * np.pi * freq * j / n))
        starts.append(1.0 + 0.8 * np.abs(np.sin(2 * np.pi * freq * j / n)))

    # U-shape
    starts.append(1.0 + 2.0 * ((j - center) / center) ** 2)
    starts.append(1.0 + 0.5 * ((j - center) / center) ** 4)

    # Bimodal
    for pos in [0.2, 0.25, 0.3, 0.35, 0.4]:
        starts.append(
            np.exp(-((j - pos * n) / (0.05 * n)) ** 2) +
            np.exp(-((j - (1 - pos) * n) / (0.05 * n)) ** 2) + 0.01
        )

    # Random
    while len(starts) < count:
        r = rng.random()
        if r < 0.25:
            starts.append(rng.exponential(1.0, n))
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.75:
            starts.append(np.ones(n) + 0.3 * rng.standard_normal(n))
        else:
            # Block structure
            block_size = rng.integers(5, 50)
            n_blocks = (n + block_size - 1) // block_size
            block_vals = rng.exponential(1.0, n_blocks)
            starts.append(np.repeat(block_vals, block_size)[:n])

    return starts

    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nFinal polish ({remaining:.0f}s)...")
        f_pol2, pol2_score = _polish(best_f, time.monotonic() + min(remaining - 10, 45),
                                      rng, n)
        if pol2_score < best_score:
            best_score = pol2_score
            best_f = f_pol2.copy()
            print(f"  Final polish: {best_score:.6f}")

    # Verify with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        best_f = np.ones(n)
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Fallback uniform: {final_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
```
