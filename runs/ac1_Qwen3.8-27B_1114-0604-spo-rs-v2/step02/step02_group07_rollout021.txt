Minimizes 2n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
Phase 1: Nesterov gradient descent with adaptive softmax peak selection

import numpy as np
import time
from scipy.optimize import linprog

def fast_self_conv(a):
    """Compute self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def grad_softmax(a, c, topk, beta):
    """Compute gradient using softmax approximation of max."""
    n = len(a)
    S = float(a.sum())
    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    
    # Softmax weights
    max_val = vals.max()
    weights = np.exp(beta * (vals - max_val))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(k_act)
        weights[0] = 1.0
    else:
        weights /= wsum
    M = float(np.dot(weights, vals))
    
    # Gradient of M w.r.t. a
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = weights[i]
        if wi < 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    
    # Full gradient: d/da [2n*M/S^2] = 2n*(grad_c/S^2 - 2M/(S^3))
    grad = (2.0 * n) * (grad_c / (S * S) - 2.0 * M / (S * S * S))

def solve_lp_direction(a, M):
    """Solve LP to find direction g that maximizes sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: (a*g)[k] = sum_j a[j]*g[k-j] <= M
    # Variable: g (length n)
    # Constraint k: sum_{j: valid} a[k-j] * g[j] <= M  for k = 0..2n-2
    
    # For efficiency, use the structure: row k has nonzeros in columns max(0,k-n+1)..min(k,n-1)
    # For scipy linprog, we need the dense matrix or use sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            for j in range(js, je + 1):
                val = a[k - j]
                if val > 1e-15:
                    rows.append(k)
                    cols.append(j)
                    data.append(val)
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    return None

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov momentum gradient descent with adaptive softmax peak selection,
LP-guided direction updates, and multi-resolution refinement.
Key improvements: adaptive temperature scheduling, LP corrections,
larger sequence lengths, and careful phase management.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_self_conv(a):
    """Compute self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def grad_softmax(a, c, topk, beta):
    """Compute gradient using softmax approximation of max.
    Returns (gradient, smooth_max, sum_a)."""
    n = len(a)
    S = float(a.sum())
    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()

    max_val = float(vals.max())
    exp_vals = np.exp(beta * (vals - max_val))
    wsum = float(exp_vals.sum())
    if wsum < 1e-30:
        weights = np.zeros(k_act)
        weights[0] = 1.0
    else:
        weights = exp_vals / wsum

    M = float(np.dot(weights, vals))

    # Gradient of M w.r.t. a
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(weights[i])
        if wi < 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    # Full gradient of f = 2n*M/S^2
    grad = (2.0 * n) * (grad_c / (S * S) - 2.0 * M / (S * S * S))
    return grad, M, S


def solve_lp_direction(a, M, max_time=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g or None if infeasible/too slow."""
    n = len(a)
    start_t = time.monotonic()

    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    data_list = []

    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            idx = np.arange(js, je + 1)
            vals = a[k - idx]
            mask = vals > 1e-12
            if np.any(mask):
                rows_list.extend(np.full(np.sum(mask), k))
                cols_list.extend(idx[mask])
                data_list.extend(vals[mask])

    if not data_list:
        return None

    A_ub = csr_matrix((data_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None

    if time.monotonic() - start_t > max_time + 5:
        return None

    if res.success and res.x is not None:
        g = np.maximum(0, res.x)
        if g.sum() > 1e-10:
            return g
    return None


def optimize_gradient(a0, end_time, rng, init_lr, topk_init, patience,
                      lr_decay=0.9995, lr_min_frac=0.01):
    """
    Nesterov momentum gradient descent with adaptive softmax.
    Returns (best_score, best_array).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    v = np.zeros(n)

    c = fast_self_conv(a)
    score = compute_score(a)
    best_score = score
    best_a = a.copy()
    no_imp = 0
    t = 0
    beta_mom = 0.9

    while time.monotonic() < end_time:
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(a0 + 0.1 * rng.standard_normal(n), 0.01, 1000.0)
            v[:] = 0.0
            S = float(a.sum())

        c = fast_self_conv(a)
        M = float(c.max())
        score = 2.0 * n * M / (S * S)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= patience:
            sc = 0.03 + 0.07 * rng.random()
            a = best_a * (1.0 + sc * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.05
            v[:] = 0.0
            t = 0
            no_imp = 0
            continue

        # Adaptive softmax temperature
        sorted_c = np.sort(c)[::-1]
        topk_actual = min(topk_init, c.size)
        if topk_actual > 1 and sorted_c[0] > 1e-15:
            gap = sorted_c[0] - sorted_c[min(topk_actual - 1, c.size - 1)]
            rel_gap = gap / sorted_c[0]
        else:
            rel_gap = 1.0
        beta_smooth = float(np.clip(3.0 / max(rel_gap, 1e-6), 5.0, 500.0))

        # Compute gradient
        g, M_sm, S = grad_softmax(a, c, topk_init, beta_smooth)

        # Nesterov step with decaying learning rate
        t += 1
        cur_lr = init_lr * (lr_decay ** t)
        cur_lr = max(cur_lr, init_lr * lr_min_frac)

        v = beta_mom * v + g
        a = a - cur_lr * v
        a = np.clip(a, 0.0, 1000.0)

    return best_score, best_a


def optimize_lp_guided(a0, end_time, blend_lr=0.15, lp_interval=15,
                       grad_lr=0.01, max_lp_time=20.0):
    """
    LP-guided optimization: alternate between LP direction and gradient steps.
    Returns (best_score, best_array).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S_target = float(a.sum())

    c = fast_self_conv(a)
    score = compute_score(a)
    best_score = score
    best_a = a.copy()

    step_count = 0
    v = np.zeros(n)
    no_imp = 0

    while time.monotonic() < end_time:
        step_count += 1

        if step_count % lp_interval == 1:
            # Solve LP every lp_interval steps
            c = fast_self_conv(a)
            M = float(c.max())
            g = solve_lp_direction(a, M, max_time=max_lp_time)
            if g is not None:
                g_sum = float(g.sum())
                if g_sum > 1e-10:
                    g_scaled = g * (S_target / g_sum)
                    a_new = (1.0 - blend_lr) * a + blend_lr * g_scaled
                    a_new = np.clip(a_new, 0.0, 1000.0)
                    sc_new = compute_score(a_new)
                    if sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                        a = a_new
                        no_imp = 0
                    else:
                        # Try smaller step
                        a_new2 = (1.0 - blend_lr * 0.3) * a + blend_lr * 0.3 * g_scaled
                        a_new2 = np.clip(a_new2, 0.0, 1000.0)
                        sc_new2 = compute_score(a_new2)
                        if sc_new2 < best_score:
                            best_score = sc_new2
                            best_a = a_new2.copy()
                            a = a_new2
                            no_imp = 0
                        else:
                            no_imp += 1
                    continue

        # Gradient step between LP solves
        S = float(a.sum())
        c = fast_self_conv(a)
        g, M_sm, S = grad_softmax(a, c, 16, 20.0)
        v = 0.85 * v + g
        a = a - grad_lr * v
        a = np.clip(a, 0.0, 1000.0)

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= 500:
            # Perturb and restart
            a = best_a * (1.0 + 0.05 * np.random.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            v[:] = 0.0
            no_imp = 0

    return best_score, best_a


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None and parent.size == n:
        inits.append(np.clip(parent, 0.01, 1000.0).copy())

    x = np.linspace(0, 1, n)
    inits.append(np.ones(n))
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * x))
    inits.append(np.clip(0.1 + 0.9 * rng.random(n), 0.01, 1000.0))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    inits.append(np.clip(tri, 0.01, 1000.0))

    # Gapped (various ratios)
    for gap_val in [0.2, 0.4, 0.6]:
        g = np.ones(n)
        g[1::2] = gap_val
        inits.append(g)

    # Quadratic
    quad = 1.0 - (2.0 * x - 1.0) ** 2
    inits.append(np.clip(quad, 0.01, 1000.0))

    # Sinc-like (windowed)
    t = np.arange(n) - n / 2.0
    sinc_seq = np.sinc(t / (n * 0.3))
    inits.append(np.clip(sinc_seq, 0.01, 1000.0))

    return inits
