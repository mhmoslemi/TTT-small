"""
Optimized search for non-negative sequences minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based gradient descent with adaptive step sizes, peak-peeling operations,
and multi-length exploration. Starts from the parent sequence and refines.

import numpy as np

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _score_normalized(p):
    """Compute score assuming p is normalized (sum=1)."""
    n = len(p)
    L = _next_pow2(2 * n - 1)
    P = np.fft.rfft(p, L)
    C = np.fft.irfft(P * P, L)
    max_c = C[:2*n-1].max()
    return float(2.0 * n * max_c)

def _score_full(p):
    """Compute score for arbitrary non-negative p."""
    p = np.asarray(p, dtype=float)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return float('inf')
    p = p / s
    return _score_normalized(p)

def _gradient_peak(p, beta, L=None):
    """
    Compute gradient of smooth max of convolution.
    Returns (score, gradient_vector).
    p should be normalized (sum=1).
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    
    P = np.fft.rfft(p, L)
    C = np.fft.irfft(P * P, L)[:2*n-1]
    
    # Soft max weights
    cmax = C.max()
    if cmax < 1e-15:
        return 2.0 * n * cmax, np.zeros(2*n-1)
    
    # Soft-max weights
    e = np.exp(beta * (C - cmax))
    w = e / e.sum()
    
    # Gradient: grad_k = 2 * sum_j w_k * (conv of w with p)
    # Actually, d/dp_j [C_k] = 2 * p_{k-j} (for valid k-j)
    # So grad_j = 2 * sum_k w_k * p_{k-j} = 2 * (w * p)_{crosscorr}[j]
    
    # (w * p) cross-correlation: h[j] = sum_k w[k] * p[k-j] (with bounds)
    # This is the same as the convolution w * p reversed... 
    # Actually, h[j] = sum_k w[k] * p[k-j] for k-j in [0,n-1]
    # = sum_i w[i+j] * p[i] for i in [0, n-1] and i+j in [0, 2n-2]
    # This is a cross-correlation of w (length 2n-1) with p (length n)
    
    # Using FFT: cross-correlation of w and p
    # corr[j] = sum_i w[i+j] * p[i]
    # = ifft(conj(fft(w)) * fft(p))[j] ... need to be careful with lengths
    
    # Let's use the direct approach via FFT
    # Define: h = ifft(fft(w, L2) * conj(fft(p, L2)))[:n]
    # where L2 >= (2n-1) + n - 1 = 3n-2
    
    L2 = _next_pow2(3 * n)
    W = np.fft.rfft(w, L2)
    P2 = np.fft.rfft(p, L2)
    H = np.fft.irfft(W * np.conj(P2), L2)
    grad = 2.0 * H[:n]
    
    score = float(2.0 * n * cmax)
    return score, grad

def _peel_step(p, k_peak, frac=0.1):
    """
    Reduce the elements that most contribute to the peak at position k_peak.
    p is normalized. Returns new normalized p.
    """
    n = len(p)
    p_new = p.copy()
    
    # Find which elements contribute most to R(k_peak)
    # R(k_peak) = sum_i p[i] * p[k_peak - i]
    # The contribution of p[i] is p[i] * p[k_peak - i]
    
    contributions = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            contributions[i] = p[i] * p[j]
    
    # Reduce the top contributors
    top_frac = max(1, int(frac * n))
    top_idx = np.argsort(contributions)[-top_frac:]
    
    # Scale down the top contributors
    reduction = 0.3  # reduce by 30%
    p_new[top_idx] *= (1.0 - reduction)
    
    # Renormalize
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p
    return p_new / s

def _convolve_fft(a, L):

    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:len(a) + len(a) - 1]

"""
Optimized search for non-negative sequences minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based gradient descent with adaptive step sizes, frequency-domain flattening,
peak-peeling operations, and multi-length exploration. Builds on parent sequence.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _score_norm(p):
    """Score for normalized sequence (sum=1). Uses FFT."""
    n = len(p)
    if n == 0:
        return float('inf')
    L = _next_pow2(2 * n - 1)
    P = np.fft.rfft(p, L)
    C = np.fft.irfft(P * P, L)
    return float(2.0 * n * C[:2 * n - 1].max())


def _score_any(p):
    """Score for arbitrary non-negative sequence."""
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return float('inf')
    return _score_norm(p / s)


def _conv_full_fft(a, L):
    """Full self-convolution via FFT."""
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2 * len(a) - 1]


def _grad_softmax(p, beta, L):
    """
    Compute gradient of smooth-max of convolution.
    p normalized, L >= 3n. Returns (score, grad_of_length_n).
    """
    n = len(p)
    C = _conv_full_fft(p, L)
    cmax = C.max()
    if cmax < 1e-15:
        return 2.0 * n * cmax, np.zeros(n)

    # Soft-max weights over convolution positions
    e = np.exp(beta * (C - cmax))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
    else:
        w = e / s_e

    # Gradient: d/dp_j [sum_k w_k R(k)] = 2 * sum_k w_k * p_{k-j} (valid range)
    # This is a cross-correlation: h[j] = sum_k w[k] * p[k-j]
    # h = ifft(fft(w) * conj(fft(p)))[:n]
    P2 = np.fft.rfft(p, L)
    W = np.fft.rfft(w, L)
    H = np.fft.irfft(W * np.conj(P2), L)
    grad = 2.0 * H[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad = np.maximum(grad, 0.0)

    score = float(2.0 * n * cmax)
    return score, grad


def _freq_flatten(p, alpha, L):
    """
    Frequency-domain convolution flattening.
    Raises the "root" of the convolution to flatten it.
    Returns normalized non-negative sequence or None.
    """
    n = len(p)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()

    # Raise normalized convolution to 1/alpha power (alpha > 1 flattens)
    C_norm = C / cmax
    C_new = cmax * np.power(np.maximum(C_norm, 0.0), 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)

    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]

    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _peel_at_peak(p, L, frac=0.08, strength=0.4):
    """
    Peel: reduce elements most contributing to the peak autocorrelation.
    Returns new normalized sequence.
    """
    n = len(p)
    C = _conv_full_fft(p, L)
    k_peak = int(np.argmax(C[:2 * n - 1]))

    # Contribution of p[i] to R(k_peak) is p[i] * p[k_peak - i]
    contrib = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            contrib[i] = p[i] * p[j]

    # Sort and reduce top contributors
    top_k = max(1, int(frac * n))
    top_idx = np.argsort(contrib)[-top_k:]

    p_new = p.copy()
    p_new[top_idx] *= (1.0 - strength)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p
    return p_new / s


def _shift_mass(p, L, from_peak=True, frac=0.05):
    """
    Shift mass from high-autocorrelation-contribution regions to low ones.
    """
    n = len(p)
    C = _conv_full_fft(p, L)
    k_peak = int(np.argmax(C[:2 * n - 1]))

    # Contribution of each element to peak
    contrib = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            contrib[i] = p[i] * p[j]

    # Move mass from top to bottom
    top_k = max(1, int(frac * n))
    bottom_k = max(1, int(frac * n))
    top_idx = np.argsort(contrib)[-top_k:]
    bot_idx = np.argsort(contrib)[:-bottom_k] if bottom_k < n else np.zeros(0, dtype=int)

    p_new = p.copy()
    mass = p_new[top_idx].sum() * 0.3
    p_new[top_idx] *= 0.7
    if len(bot_idx) > 0:
        p_new[bot_idx] += mass / len(bot_idx)

    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p
    return p_new / s


def _opt_gradient(p0, budget, beta_start, beta_end, alpha_init, use_peel=True, verbose=False):
    """
    Main gradient optimizer with FFT acceleration.
    Returns (best_p, best_score).
    """
    p = np.asarray(p0, dtype=float).copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return p0, float('inf')
    p = p / s

    n = len(p)
    L = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score_norm(p)
    cur_score = best_score
    beta = float(beta_start)
    alpha = float(alpha_init)
    stall = 0
    it = 0
    deadline = time.monotonic() + budget

    if verbose:
        print(f"      [grad-opt] n={n}, init={best_score:.6f}, beta={beta_start:.0f}->{beta_end:.0f}, alpha={alpha_init}, budget={budget:.0f}s")

    while time.monotonic() < deadline:
        # Compute gradient
        _, grad = _grad_softmax(p, beta, L)
        gmax = grad.max()
        if gmax < 1e-15:
            stall += 1
            if stall > 10:
                break
            continue
        grad_n = grad / gmax

        # Backtracking line search
        best_trial = None
        best_trial_sc = cur_score
        for a_try in (alpha * 0.4, alpha * 0.7, alpha, alpha * 1.5, alpha * 2.5):
            p_try = p * np.exp(-a_try * grad_n)
            p_try = np.maximum(p_try, 1e-15 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score_norm(p_try)
            if sc_try < best_trial_sc - 1e-13:
                best_trial = p_try
                best_trial_sc = sc_try

        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-11:
                best = p.copy()
                best_score = cur_score
                if verbose and it % 200 == 0:
                    print(f"        iter={it}, score={best_score:.6f}, beta={beta:.0f}, alpha={alpha:.4f}")
        else:
            stall += 1
            if stall > 5:
                alpha = max(0.005, alpha * 0.75)
            if stall > 20 and use_peel:
                # Try peel
                p_peel = _peel_at_peak(p, L, frac=0.05, strength=0.3)
                sc_peel = _score_norm(p_peel)
                if sc_peel < cur_score - 1e-12:
                    p = p_peel
                    cur_score = sc_peel
                    stall = 0
                    if cur_score < best_score - 1e-11:
                        best = p.copy()
                        best_score = cur_score
                elif stall > 40:
                    # Try freq flatten

                    if p_fl is not None:
                        sc_fl = _score_norm(p_fl)
                        if sc_fl < cur_score:
                            p = p_fl
                            cur_score = sc_fl
                            stall = 0
                            if cur_score < best_score - 1e-11:
                                best = p.copy()
                                best_score = cur_score
                    if stall > 40:
                        p = best.copy()
                        cur_score = best_score
                        alpha = alpha_init
                        stall = 10

        # Update beta (increase for sharper focus)
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.06)

        it += 1

    if verbose:
        print(f"      [grad-opt done] iters={it}, final={best_score:.6f}")
    return best, best_score

def _resample_to(p, n_new):
    """Resample sequence to new length."""
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s
