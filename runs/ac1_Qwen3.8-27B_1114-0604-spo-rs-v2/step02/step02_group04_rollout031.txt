    n = len(a)
    # A matrix: (2n-1) x n
    # A[k,i] = a[k-i] if 0 <= k-i <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, i] = a[k-i]
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    
    # g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = -res.x  # negate because we minimized -sum
        return g
    return None

"""
Low-peak step-function optimization via Alternating Linear Programming (ALP).

Key idea: Alternately fix one copy of the sequence and solve a linear program
for the other. The LP: max sum(g) s.t. (a*g)_k <= M, g >= 0, provides a
principled descent direction. Blending the LP solution with the current
sequence ensures monotone (near) convergence to a local optimum.

This differs from the previous gradient-based approach by using exact LP
solves at each step, which provides stronger convergence guarantees and
avoids the sensitivity to learning rate / temperature scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_two_lobe(n):
    """Two-lobe anti-palindromic: high in middle of first half, high in middle of second half."""
    i = np.arange(n, dtype=np.float64)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.15
    a = np.exp(-0.5 * ((i - center1) / width) ** 2) + np.exp(-0.5 * ((i - center2) / width) ** 2)
    a = a / a.sum()
    return np.clip(a, 0, 1000)


def _make_four_lobe(n):
    """Four-lobe pattern."""
    i = np.arange(n, dtype=np.float64)
    centers = [n * 0.125, n * 0.375, n * 0.625, n * 0.875]
    width = n * 0.08
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((i - c) / width) ** 2)
    a = a / a.sum()
    return np.clip(a, 0, 1000)


def _build_lp_matrix(a):
    """Build (2n-1) x n matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = _build_lp_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(0.0, -res.x)
            return g
    except Exception:
        pass
    return None


def _alp_optimize(a, deadline, t_init=0.35, eval_every=5):
    """Alternating LP optimization with adaptive blending."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    it = 0
    last_improve = 0
    patience = 500
    rng = np.random.RandomState(42 + n)
    consecutive_failures = 0

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(a)
        M = float(np.max(c))

        g = _solve_lp(a, M)
        if g is None:
            consecutive_failures += 1
            if consecutive_failures > 50:
                break
            continue
        consecutive_failures = 0

        g = np.clip(g, 0, 1000)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            continue
        g = g / g_sum

        # Adaptive blending: start large, decay slowly
        t = t_init / (1.0 + 0.002 * it)
        t = max(0.02, t)

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000)
        s_new = float(a_new.sum())
        if s_new < 1e-12:
            a = best_a.copy()
            break
        a_new = a_new / s_new
        a = a_new

        # Periodic evaluation
        if it % eval_every == 0 or it <= 10:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                if it % (eval_every * 20) == 0 or it <= 10:
                    print(f"    [ALP] it={it}, score={sc:.8f}, M={M:.6f}, t={t:.4f}", flush=True)

        # Patience-based restart
        if it - last_improve > patience:
            a = best_a.copy()
            noise_scale = 0.03 * a.mean()
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.clip(a, 0, 1000)
            s = a.sum()
            if s < 1e-12:
                break
            a = a / s
            patience = max(300, int(patience * 1.3))
            last_improve = it
            print(f"    [ALP] Restart at it={it}, patience={patience}", flush=True)

    # Final check
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _soft_polish(a, deadline):
    """Soft-max mirror descent polish, similar to previous approach but as a finishing step."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    it = 0
    lr = 0.15 * n
    rng = np.random.RandomState(777)

    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        if c.size == 0:
            break
        c_max = float(np.max(c))
        c_min = float(np.min(c))
        rng_c = c_max - c_min
        if rng_c < 1e-15:
            break
        c_norm = (c - c_min) / rng_c - 1.0  # in [-1, 0]
        T = 20.0 + 0.5 * it
        w = np.exp(T * c_norm)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum

        # Gradient via FFT cross-correlation
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[:N] if w.size >= N else w, N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        grad = 2.0 * xcorr[:n]

        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a / s

        lr = lr / (1.0 + 0.001 * it)

        if it % 10 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            elif it % 200 == 0:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01 * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                a = a / a.sum()

    return best_a, best_score

        print(f"[Phase 3] ALP from perturbation ({remain3:.0f}s)...", flush=True)
        a2, s2 = _alp_optimize(perturbed, d3, t_init=0.25)
        print(f"  ALP2 result: {s2:.8f}", flush=True)
        candidates.append((a2, s2))

    # ===== Phase 4: ALP from different init =====
    d4 = d3 + 150
    remain4 = d4 - time.monotonic()
    if remain4 > 100:
        alt_init = _make_sine(n, 1.1, 0.4)
        print(f"[Phase 4] ALP from alt init ({remain4:.0f}s)...", flush=True)
        a3, s3 = _alp_optimize(alt_init, d4, t_init=0.3)
        print(f"  ALP3 result: {s3:.8f}", flush=True)
        candidates.append((a3, s3))

    # ===== Phase 5: Soft-max polish on best candidate =====
    d5 = hard_deadline - 30
    remain5 = d5 - time.monotonic()
    if remain5 > 60:
        candidates.sort(key=lambda x: x[1])
        best_so_far = candidates[0][0]
        print(f"[Phase 5] Soft polish ({remain5:.0f}s)...", flush=True)
        a_pol, s_pol = _soft_polish(best_so_far, d5)
        print(f"  Polish result: {s_pol:.8f}", flush=True)
        candidates.append((a_pol, s_pol))

    # ===== Phase 6: Validation =====
    print("[Phase 6] Validation...", flush=True)
    candidates.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    seen = set()
    for arr, sc in candidates:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  Validated (n={arr.size}): actual={actual:.8f}", flush=True)

    if best_out is None:
        arr_list = [float(max(0.0, min(1000.0, x))) for x in candidates[0][0]]
        best_out = arr_list
        try:
            best_actual = float(evaluate_sequence(arr_list))
        except Exception:
            best_actual = np.inf
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
