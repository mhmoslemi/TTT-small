def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute corr[k] = sum_l w[l] * a[l-k] for k=0..n-1 (a[j]=0 outside [0,n-1]).
    Returns array of length n. The subgradient of max(conv(a,a)) is 2*corr_pair(a, w_active).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    total_len = n + m - 1
    N = next_power_of_two(total_len)
    Ar = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(Ar * W, N)[:total_len]
    result = full[n - 1:2 * n - 1]
    return np.nan_to_num(result, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    """Return normalized self-convolution d = n*conv(a,a)/S^2 and S."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_and_grad(a, beta):
    """Compute LSE-approximated score and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta < 1.0:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -100.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    LSE_c = dmax + np.log(sum_e) / beta
    score = 2.0 * LSE_c

    # Gradient: grad[k] = 4n*corr[k]/S^2 - 4n*LSE_c/S^3
    corr = corr_pair(a, w)
    grad = (4.0 * float(n) / (S * S)) * corr
    grad -= (4.0 * float(n) * LSE_c) / (S * S * S)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * (grad * grad)
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new


def topk_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def exact_subgrad(a, k):
    """Compute exact subgradient of max(conv(a,a))/S^2 * 2n using top-k lags."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n, dtype=np.float64)
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.zeros(n, dtype=np.float64)
    w = topk_weights(d, k)
    corr = corr_pair(a, w)
    grad = (4.0 * float(n) / (S * S)) * corr
    LSE_c = float(np.dot(w, d))
    grad -= (4.0 * float(n) * LSE_c) / (S * S * S)
    return np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)


def smooth_noise(n, rng, smoothness=50):
    """Generate smooth Gaussian noise of length n."""
    n_noise = n
    noise = rng.standard_normal(n_noise)
    # Apply FFT-based smoothing
    N = next_power_of_two(n_noise)
    spec = np.fft.rfft(noise, N)
    freqs = np.fft.rfftfreq(N) * N
    # Low-pass filter
    cutoff = max(1.0, float(smoothness))
    spec *= np.exp(-0.5 * (np.abs(freqs) / cutoff) ** 2)
    noise_smooth = np.fft.irfft(spec, N)[:n_noise]
    return noise_smooth


def perturb_smooth(a, rng, strength=0.1):
    """Perturb with smooth noise in log-space."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel()
    log_a = np.log(np.maximum(a, 1e-300))
    noise = smooth_noise(n, rng, smoothness=max(10, n // 100))
    noise = noise - float(noise.mean())
    log_new = log_a + strength * noise
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def perturb_vshape(a, rng, strength=0.1):
    """Perturb by adding a V-shape component."""
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    v = 1.0 + strength * np.abs(2.0 * x - 1.0)
    return normalize_a(a * v)


def perturb_gaussian(a, rng, strength=0.1):
    """Perturb with a Gaussian bump at random location."""
    n = len(a)
    center = rng.uniform(0.0, 1.0)
    width = rng.uniform(0.02, 0.15)
    x = np.linspace(0.0, 1.0, n)
    bump = np.exp(-((x - center) / width) ** 2)
    sign = 1.0 if rng.random() < 0.5 else -1.0
    a_new = a * (1.0 + sign * strength * 0.5 * bump)
    return normalize_a(np.maximum(a_new, 1e-10))


def perturb_random(a, rng, strength=0.1):
    """Perturb with random (less smooth) noise."""
    n = len(a)
    noise = rng.standard_normal(n)
    noise = noise - float(noise.mean())
    a_new = a * np.exp(strength * 0.5 * noise)
    return normalize_a(a_new)


def adam_lse_phase(a_start, deadline, seed=42):
    """Adam optimization with LSE smoothing and adaptive beta schedule."""
    rng = np.random.default_rng(seed)
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    state = adam_state_init(n)

    # Beta schedule: gradually increase
    beta_start = 50.0
    beta_end = 500000.0
    total_time = max(1.0, float(deadline - time.monotonic()))

    # Learning rate schedule
    lr_start = 1.0 / n
    lr_end = 1.0 / (n * 100)

    t0 = time.monotonic()
    while time.monotonic() < deadline - 2.0:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / total_time)

        # Exponential beta ramp
        beta = beta_start * (beta_end / beta_start) ** frac
        # Linear LR decay
        lr = lr_start * (lr_end / lr_start) ** frac

        # Do a batch of gradient steps
        for _ in range(3):
            if time.monotonic() > deadline - 2.0:
                break
            score, grad = lse_score_and_grad(a, beta)
            if not np.isfinite(score):
                break
            a_raw = adam_step(a, grad, state, lr, 0.9, 0.999, 1e-8)
            a = normalize_a(np.maximum(a_raw, 0.0))

        sc = exact_score_a(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Occasionally restart momentum to escape saddle points
        if rng.random() < 0.01:
            state['m'][:] = 0.0
            state['v'][:] = 0.0
            state['t'] = 0

        # Occasionally add small perturbation to escape plateaus
        if rng.random() < 0.02:
            a = perturb_smooth(a, rng, strength=0.05)

    return float(best_sc), best_a


def nesterov_subgrad_phase(a_start, deadline, seed=42):
    """Nesterov accelerated subgradient method with adaptive step size."""
    rng = np.random.default_rng(seed)
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    a_prev = a.copy()
    momentum = 0.0
    step_size = 1.0 / n

    for it in range(10000):
        if time.monotonic() > deadline - 2.0:
            break

        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)

        # Choose k adaptively
        k = 1 + (it % 5)  # cycle through k=1,2,3,4,5

        # Compute subgradient at y
        n_y = normalize_a(y)
        grad = exact_subgrad(n_y, k)

        # Step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_a(a_new)

        # Update momentum (Nesterov schedule)
        t_next = it + 2
        momentum_new = (t_next - 1) / (t_next + 1)
        momentum = momentum_new

        # Adaptive step size
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            step_size *= 1.05  # slightly increase
        elif np.isfinite(sc):
            step_size *= 0.8  # decrease if worse

        step_size = max(step_size, 1e-10)
        step_size = min(step_size, 5.0 / n)

        a_prev = a
        a = a_new

        # Periodic perturbation
        if it % 50 == 0 and rng.random() < 0.3:
            a = perturb_smooth(a, rng, strength=0.03)
            a_prev = a.copy()

    return float(best_sc), best_a


def simulated_annealing_phase(a_start, deadline, seed=42):
    """Simulated annealing on the exact objective."""
    rng = np.random.default_rng(seed)
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    T_start = 0.01
    T_end = 0.0001
    total_time = max(1.0, float(deadline - time.monotonic()))
    t0 = time.monotonic()

    accept_count = 0
    total_count = 0

    while time.monotonic() < deadline - 2.0:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / total_time)
        # Exponential cooling
        T = T_start * (T_end / T_start) ** frac

        # Generate perturbation (mix of smooth and rough)
        mode = rng.integers(0, 4)
        strength = float(rng.uniform(0.02, 0.15)) * (1.0 + 0.5 * (1.0 - frac))
        if mode == 0:
            a_new = perturb_smooth(a, rng, strength)
        elif mode == 1:
            a_new = perturb_vshape(a, rng, strength)
        elif mode == 2:
            a_new = perturb_gaussian(a, rng, strength)
        else:
            a_new = perturb_random(a, rng, strength)

        sc_new = exact_score_a(a_new)
        total_count += 1

        if np.isfinite(sc_new):
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            if sc_new <= exact_score_a(a) or rng.random() < math.exp(-(sc_new - exact_score_a(a)) / max(T, 1e-10)):
                a = a_new
                accept_count += 1

    return float(best_sc), best_a


def lp_active_set_phase(a_start, deadline, seed=42):
    """LP-based active set: solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Use at a reduced length, then resample back."""
    rng = np.random.default_rng(seed)
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    # Work at reduced length for LP feasibility
    n_lp = min(2048, len(a))
    a_lp = normalize_a(resample_sequence(a, n_lp))

    for iteration in range(30):
        if time.monotonic() > deadline - 5.0:
            break

        c = fast_self_conv(a_lp)
        M = float(np.max(c))
        n = n_lp

        # Build constraint matrix A: A[l, j] = a_lp[l-j] if 0 <= l-j < n
        # This is a (2n-1) x n matrix
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            # A[l, j] = a_lp[l-j] for l = j, j+1, ..., j+n-1
            l_start = j
            l_end = min(j + n, 2 * n - 1)
            if l_end > l_start:
                A[l_start:l_end, j] = a_lp[:l_end - l_start]

        # Solve: maximize sum(g) s.t. A@g <= M, g >= 0
        try:
            c_obj = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                bounds=[(0, None)] * n,
                method='highs',
                options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-7}
            )
            if res.success and res.x is not None:
                g = res.x
                g_sum = float(g.sum())
                if g_sum > 1e-10:
                    g = g / g_sum  # normalize to sum=1
                    # Evaluate
                    sc_g = exact_score_a(g)
                    if np.isfinite(sc_g) and sc_g < best_sc:
                        best_sc = sc_g
                        # Resample back and keep
                        a_full = normalize_a(resample_sequence(g, len(a_start)))
                        sc_full = exact_score_a(a_full)
                        if sc_full < best_sc:
                            best_sc = sc_full
                            best_a = a_full.copy()
                    # Use g as new starting point for next LP iteration
                    a_lp = normalize_a(g)
                    # Slight perturbation to avoid cycling
                    if rng.random() < 0.5:
                        a_lp = perturb_smooth(a_lp, rng, 0.02)
                else:
                    break
            else:
                break
        except Exception:
            break

    # Also try to polish the best found at full length
    if time.monotonic() < deadline - 10.0:
        a_polished = normalize_a(best_a)
        s
        # Quick Adam polish
        state = adam_state_init(len(a_polished))
        for _ in range(100):
            if time.monotonic() > deadline - 5.0:
                break
            score, grad = lse_score_and_grad(a_polished, 100000.0)
            if not np.isfinite(score):
                break
            lr = 1.0 / (len(a_polished) * 10)
            a_raw = adam_step(a_polished, grad, state, lr)
            a_polished = normalize_a(np.maximum(a_raw, 0.0))
        sc = exact_score_a(a_polished)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_polished.copy()

    return float(best_sc), best_a


def population_phase(a_start, deadline, pop_size=15, seed=42):
    """Population-based search with local optimization of each member."""
    rng = np.random.default_rng(seed)
    n = len(a_start)

    # Initialize population
    population = []
    scores = []

    # Add the starting point
    a0 = normalize_a(a_start)
    population.append(a0)
    scores.append(exact_score_a(a0))

    # Add diverse initializations
    x = np.linspace(0.0, 1.0, n)
    inits = [a0.copy()]
    inits.append(normalize_a(1.0 + 0.5 * np.abs(2.0 * x - 1.0)))
    inits.append(normalize_a(1.0 + 0.8 * np.cos(np.pi * x)))
    inits.append(normalize_a(np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x))))
    inits.append(normalize_a(1.0 + 1.5 * np.abs(2.0 * x - 1.0) ** 2))
    inits.append(normalize_a(np.exp(-(((x - 0.3) / 0.1) ** 2)) + np.exp(-(((x - 0.7) / 0.1) ** 2))))

    for idx in range(1, min(pop_size, len(inits))):
        p = perturb_smooth(inits[idx], rng, 0.1)
        population.append(p)
        scores.append(exact_score_a(p))

    # Fill remaining with perturbations of best
    while len(population) < pop_size:
        best_idx = int(np.argmin(scores))
        p = perturb_smooth(population[best_idx], rng, float(rng.uniform(0.05, 0.2)))
        population.append(p)
        scores.append(exact_score_a(p))

    best_overall = min(scores)
    best_seq = population[int(np.argmin(scores))].copy()

    # Evolution loop
    gen = 0
    while time.monotonic() < deadline - 10.0:
        gen += 1
        # Sort by score
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        # Keep top half, replace bottom half with offspring
        keep = max(3, pop_size // 2)
        for i in range(keep, pop_size):
            if time.monotonic() > deadline - 10.0:
                break
            # Select parent from top
            parent_idx = int(rng.integers(0, keep))
            parent = population[parent_idx]

            # Mutation
            mode = rng.integers(0, 5)
            strength = float(rng.uniform(0.03, 0.15))
            if mode == 0:
                child = perturb_smooth(parent, rng, strength)
            elif mode == 1:
                child = perturb_gaussian(parent, rng, strength)
            elif mode == 2:
                child = perturb_vshape(parent, rng, strength)
            elif mode == 3:
                # Crossover with another good individual
                other_idx = int(rng.integers(0, keep))
                if other_idx != parent_idx:
                    t = float(rng.uniform(0.3, 0.7))
                    child = normalize_a(t * parent + (1.0 - t) * population[other_idx])
                else:
                    child = perturb_smooth(parent, rng, strength)
            else:
                child = perturb_random(parent, rng, strength)

            # Quick local optimization (few Adam steps)
            state = adam_state_init(n)
            for _ in range(20):
                if time.monotonic() > deadline - 5.0:
                    break
                score, grad = lse_score_and_grad(child, 5000.0)
                if not np.isfinite(score):
                    break
                lr = 1.0 / (n * 20)
                a_raw = adam_step(child, grad, state, lr)
                child = normalize_a(np.maximum(a_raw, 0.0))

            sc_child = exact_score_a(child)
            if not np.isfinite(sc_child):
                sc_child = 100.0
            population[i] = child
            scores[i] = sc_child

        # Update best
        current_best = min(scores)
        if current_best < best_overall:
            best_overall = current_best
            best_idx = int(np.argmin(scores))
            best_seq = population[best_idx].copy()

    return float(best_overall), best_seq


def final_polish(a_start, deadline, seed=42):
    """Final polishing with multiple techniques."""
    rng = np.random.default_rng(seed)
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    # Phase A: High-beta Adam
    state = adam_state_init(n)
    for beta in (50000.0, 200000.0, 1000000.0):
        if time.monotonic() > deadline - 5.0:
            break
        steps = 50
        for s in range(steps):
            if time.monotonic() > deadline - 3.0:
                break
            score, grad = lse_score_and_grad(a, beta)
            if not np.isfinite(score):
                break
            lr = 1.0 / (n * 50)
            a_raw = adam_step(a, grad, state, lr)
            a = normalize_a(np.maximum(a_raw, 0.0))
        sc = exact_score_a(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Phase B: Exact subgradient with small steps
    for _ in range(200):
        if time.monotonic() > deadline - 2.0:
            break
        improved = False
        for k in (1, 2, 3, 5):
            for eta in (0.001, 0.005, 0.01, 0.02):
                if time.monotonic() > deadline - 2.0:
                    break
                grad = exact_subgrad(a, k)
                q = normalize_a(np.maximum(a - eta * grad, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q
                    improved = True
        if not improved:
            break

    # Phase C: Tiny perturbations
    for _ in range(50):
        if time.monotonic() > deadline - 1.0:
            break
        for strength in (0.005, 0.01, 0.02):
            q = perturb_smooth(a, rng, strength)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q

    return float(best_sc), best_a


def load_parent():
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(p.sum()) > 0.01:
                return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search: Adam+LSE, Nesterov subgradient, simulated annealing,
    LP active set, population-based search, and final polish.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = load_parent()
    print("Parent available: {}".format(parent is not None))

    # Choose primary length
    n_primary = 10240
    if parent is not None:
        n_primary = parent.size
        if n_primary > 16384:
            n_primary = 16384
        if n_primary < 2048:
            n_primary = 2048

    # Initialize from parent
    if parent is not None:
        a_start = normalize_a(resample_sequence(parent, n_primary))
    else:
        x = np.linspace(0.0, 1.0, n_primary)
        a_start = normalize_a(1.0 + 0.8 * np.abs(2.0 * x - 1.0))

    best_sc = exact_score_a(a_start)
    best_seq = a_start.copy()
    print("Initial score: {:.8f} (length {})".format(best_sc, len(best_seq)))

    # === Phase 1: Adam + LSE (35% of time) ===
    p1_end = t_start + 0.35 * (deadline - t_start)
    print("Phase 1: Adam+LSE (until t={:.0f}s)".format(p1_end - t_start))
    sc, seq = adam_lse_phase(a_start, p1_end, seed=seed)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 1 best: {:.8f}".format(best_sc))
    else:
        print("Phase 1 no improvement (best={:.8f})".format(best_sc))

    # === Phase 2: Nesterov subgradient (15% of time) ===
    p2_end = t_start + 0.50 * (deadline - t_start)
    print("Phase 2: Nesterov subgradient (until t={:.0f}s)".format(p2_end - t_start))
    sc, seq = nesterov_subgrad_phase(best_seq, p2_end, seed=seed + 1)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 2 best: {:.8f}".format(best_sc))
    else:
        print("Phase 2 no improvement (best={:.8f})".format(best_sc))

    # === Phase 3: Simulated annealing (20% of time) ===
    p3_end = t_start + 0.70 * (deadline - t_start)
    print("Phase 3: Simulated annealing (until t={:.0f}s)".format(p3_end - t_start))
    sc, seq = simulated_annealing_phase(best_seq, p3_end, seed=seed + 2)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 3 best: {:.8f}".format(best_sc))
    else:
        print("Phase 3 no improvement (best={:.8f})".format(best_sc))

    # === Phase 4: LP active set (15% of time) ===
    p4_end = t_start + 0.85 * (deadline - t_start)
    print("Phase 4: LP active set (until t={:.0f}s)".format(p4_end - t_start))
    sc, seq = lp_active_set_phase(best_seq, p4_end, seed=seed + 3)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 4 best: {:.8f}".format(best_sc))
    else:
        print("Phase 4 no improvement (best={:.8f})".format(best_sc))

    # === Phase 5: Population-based search (if time remains) ===
    if time.monotonic() < deadline - 120.0:
        p5_end = deadline - 60.0
        print("Phase 5: Population search (until t={:.0f}s)".format(p5_end - t_start))
        sc, seq = population_phase(best_seq, p5_end, pop_size=12, seed=seed + 4)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 5 best: {:.8f}".format(best_sc))
        else:
            print("Phase 5 no improvement (best={:.8f})".format(best_sc))

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 5.0:
        print("Phase 6: Final polish")
        sc, seq = final_polish(best_seq, deadline - 3.0, seed=seed + 5)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 6 best: {:.8f}".format(best_sc))

    # Final verification and cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
