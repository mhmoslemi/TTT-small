"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set subgradient optimization with exact line search,
combined with LSE smoothing, coordinate refinement, and diverse restarts.
"""

import numpy as np
import time

PEN_SUM = 1e-3

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute the correlation of a with w, returned as a vector of length n.
    result[i] = sum_j w[i+j] * a[j] (approximately, via FFT)
    Actually: result = the part of conv(a, w) that corresponds to lags 0..n-1
    from the perspective of a.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))

def d_from_a(a):
    """Compute normalized self-convolution d = n*conv(a,a)/sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def lse_score_grad(a, beta):
    n = int(len(a))
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def run_lbfgs(a0, beta, deadline, maxfun):
    from scipy.optimize import minimize
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    def fun(x):
        return lse_score_grad(x, beta)
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out

def estimate_eval_time(n):

    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def active_weight_soft(d, temperature):
    """Soft active set: weights proportional to exp(d/temp)."""
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.ones(d.size) / d.size
    z = (d - dmax) / max(temperature, 1e-10)
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    wsum = float(w.sum())
    if wsum < 1e-30:
        return np.ones(d.size) / d.size
    return w / wsum

def subgrad_step_exact(a, w, step_size):
    """One subgradient step using given active weights w."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    d, _ = d_from_a(a)
    h = corr_pair(a, w)
    grad = (2.0 * float(n) * h) / S  # gradient of n*max_b w.r.t. a (times 2 for the 2n factor)
    # Actually the score is 2n*max_b/S^2, gradient w.r.t. a:
    # d(score)/da_i = 2n * (2 * sum_j w[j] * a[i+j] or appropriate) / S^2 - 2 * 2n * max_b * a_i / S^3
    # Let me use the LSE gradient formula but with the soft weights
    grad = (4.0 * float(n) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize_a(a_new)

def exact_backtracking(a, w, max_steps=30):
    """Backtracking line search on the exact score, moving in direction -grad."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a, exact_score(a)
    d, _ = d_from_a(a)
    h = corr_pair(a, w)
    grad = (4.0 * float(n) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, exact_score(a)
    grad_dir = -grad / gnorm
    f0 = exact_score(a)
    step = 1.0 / (gnorm + 1e-10)
    best_a = a.copy()
    best_f = f0
    for _ in range(max_steps):
        a_trial = np.maximum(a + step * grad_dir, 0.0)
        a_trial = normalize_a(a_trial)
        f_trial = exact_score(a_trial)
        if np.isfinite(f_trial) and f_trial < best_f:
            best_f = f_trial
            best_a = a_trial.copy()
        step *= 0.5
        if step < 1e-12:
            break
    return best_a, best_f

def adaptive_subgrad_optimize(a_start, deadline, max_rounds=200):
    """Adaptive active-set subgradient optimization with exact line search."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    
    # Temperature for soft active set (start hot, cool down)
    temp_start = 0.5
    temp_end = 0.01
    n_rounds = max_rounds
    
    for it in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Adaptive temperature
        frac = float(it) / max(1, n_rounds - 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        
        # Compute soft active set
        d, _ = d_from_a(a)
        w = active_weight_soft(d, temp)
        
        # Backtracking line search
        a_new, f_new = exact_backtracking(a, w, max_steps=25)
        if f_new < best_sc:
            best_sc = f_new
            best_a = a_new.copy()
        if f_new < exact_score(a) - 1e-12:
            a = a_new
        else:
            # Try smaller step
            a = normalize_a(np.maximum(a + 0.01 * (-corr_pair(a, w)), 0.0))
        
        # Occasionally try a different active set size
        if it % 10 == 5:
            d2, _ = d_from_a(a)
            dmax = float(np.max(d2))
            if dmax > 0:
                # Hard top-k active set
                for k in (1, 3, 5):
                    if d2.size > k:
                        idx = np.argpartition(d2, -k)[-k:]
                        w_k = np.zeros(d2.size)
                        w_k[idx] = 1.0 / k
                        a_k, f_k = exact_backtracking(a, w_k, max_steps=15)
                        if f_k < best_sc:
                            best_sc = f_k
                            best_a = a_k.copy()
                        if f_k < exact_score(a):
                            a = a_k
                            break
    
    return float(best_sc), best_a

def lse_optimize_trajectory(a_start, deadline, betas=(5000, 20000, 50000, 100000, 200000, 500000)):

        frac = (i + 1) / n_betas

def power_iteration(a_start, deadline, max_iters=100):
    """Power iteration on the active set."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        d, S = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 0:
            break
        
        # Try different active set sizes
        improved = False
        for k in (1, 2, 3, 5):
            if k >= d.size:
                continue
            idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(d.size)
            w[idx] = 1.0 / k
            h = corr_pair(a, w)
            h = np.maximum(h, 1e-15)
            for blend in (0.3, 0.5, 0.7):

                    best_a = q.copy()
                    a = q.copy()
                    improved = True

            break
    
    return float(best_sc), best_a

The sum is n/3. The convolution at lag 0: sum of a[i]^2 = n/3. At lag 3: sum of a[i]*a[i+3] = n/9. The peak is still at lag 0 (or near it), with value n/3. Score = 2n * (n/3) / (n/3)^2 = 2n * (n/3) * 9/n^2 = 6. That's worse!

What about a smoother "comb"? Like `a_i = 1 + epsilon * cos(2*pi*k*i/n)` for some k? The convolution would have peaks at lag 0 and lag n-k (and k). The peak at lag 0 is roughly n + epsilon^2 * n/2, and the peak at lag k is roughly (epsilon/2) * n. For small epsilon, the peak at lag 0 dominates. So this doesn't help.

The key insight is that for any non-negative sequence, the "energy" is concentrated where the sequence has large values, and the self-convolution peak tends to be near where the sequence is most concentrated. To flatten the convolution, we need the sequence to be as "uniform" as possible, but with some structure that helps spread the convolution peak.

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set subgradient optimization with exact backtracking line search,
LSE smoothing with aggressive beta continuation, power iteration, coordinate refinement,
and diverse multi-basin restarts. Key novelty: exact line search on the true max objective
combined with temperature-annealed soft active sets for escaping local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    return np.maximum(d, 0.0), S


def lse_score_grad(a, beta):
    n = int(len(a))
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z = z - z.max()
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), np.clip(grad, -1e6, 1e6)


def fun_lse(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None and float(deadline - time.monotonic()) < 5.0:
        return exact_score(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-15, "gtol": 1e-12, "maxiter": int(maxiter),
            "maxfun": int(maxfun), "maxls": 50}
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a
    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        return exact_score(a), a
    return float(sc), a_out


def estimate_eval_time(n):
    return 0.02 * max(0.5, float(n) / 8192.0) * max(1.0, np.log2(max(16, n)) / np.log2(8192.0))


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / max(1, n)
    if m == int(n):
        return a.copy()
    return np.interp(np.linspace(0.0, 1.0, int(n)),
                     np.linspace(0.0, 1.0, m), a).astype(np.float64)


def soft_active_weights(d, temp):
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.ones(d.size) / d.size
    z = (d - dmax) / max(temp, 1e-10)
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    s = float(w.sum())
    if s < 1e-30:
        return np.ones(d.size) / d.size
    return w / s


def hard_active_weights(d, k):
    if k >= d.size:
        return np.ones(d.size) / d.size
    idx = np.argpartition(d, d.size - k)[d.size - k:]
    w = np.zeros(d.size)
    w[idx] = 1.0 / k
    return w


def compute_grad(a, w):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(n), 0.0, 0.0
    d, _ = d_from_a(a)
    h = corr_pair(a, w)
    grad = (4.0 * float(n) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    return grad, LDS, S


def backtracking_exact(a, w, max_depth=30):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a, exact_score(a)
    grad, _, _ = compute_grad(a, w)
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-15:
        return a, exact_score(a)
    direction = -grad / gnorm
    f0 = exact_score(a)
    best_a = a.copy()
    best_f = f0
    step = 2.0 / (gnorm + 1e-10)
    for _ in range(max_depth):
        a_trial = normalize_a(np.maximum(a + step * direction, 0.0))
        f_trial = exact_score(a_trial)
        if np.isfinite(f_trial) and f_trial < best_f:
            best_f = f_trial
            best_a = a_trial.copy()
        step *= 0.5
        if step < 1e-14:
            break
    return best_a, best_f


def adaptive_subgrad(a_start, deadline, max_rounds=300):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    temp_start, temp_end = 0.4, 0.005
    stagnation = 0

    for it in range(max_rounds):
        if time.monotonic() > deadline:
            break
        frac = float(it) / max(1, max_rounds - 1)
        temp = temp_start * (temp_end / temp_start) ** frac

        d, _ = d_from_a(a)
        w = soft_active_weights(d, temp)
        a_new, f_new = backtracking_exact(a, w, max_depth=25)

        if f_new < best_sc - 1e-10:
            best_sc = f_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        if f_new < exact_score(a) - 1e-12:
            a = a_new
        else:
            # Stall: try hard active sets
            improved = False
            for k in (1, 2, 3, 5, 8):
                w_h = hard_active_weights(d, k)
                a_h, f_h = backtracking_exact(a, w_h, max_depth=15)
                if f_h < best_sc:
                    best_sc = f_h
                    best_a = a_h.copy()
                    a = a_h
                    improved = True
                    break
                elif f_h < exact_score(a):
                    a = a_h
            if not improved:
                a = normalize_a(np.maximum(a + 0.005 * np.random.default_rng(it).standard_normal(len(a)), 0.0))

    return float(best_sc), best_a


def power_iter(a_start, deadline, max_iters=150):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        d, _ = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 0:
            break
        improved = False
        for k in (1, 2, 3, 5):
            w = hard_active_weights(d, k)
            h = corr_pair(a, w)
            h = np.maximum(h, 1e-15)
            for blend in (0.3, 0.5, 0.7, 0.9):
                q = normalize_a(blend * h + (1.0 - blend) * a)
                sc = exact_score(q)
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
        if not improved and it > 30:
            break
    return float(best_sc), best_a


def cooperative_descent(a_start, deadline, max_rounds=100):
    """Alternating between power iteration and subgradient with adaptive k."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    k_adaptive = 3

    for it in range(max_rounds):
        if time.monotonic() > deadline:
            break
        d, _ = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 0:
            break

        # Adapt k based on how "peaked" the active set is
        d_sorted = np.sort(d)[::-1]
        if d_sorted.size > 5 and d_sorted[4] > 0.9 * d_sorted[0]:
            k_adaptive = max(3, k_adaptive - 1)
        elif d_sorted.size > 1 and d_sorted[1] < 0.7 * d_sorted[0]:
            k_adaptive = min(10, k_adaptive + 1)
        k_adaptive = max(1, min(10, k_adaptive))

        # Power iteration step
        w = hard_active_weights(d, k_adaptive)
        h = corr_pair(a, w)
        h = np.maximum(h, 1e-15)
        a_pi = normalize_a(0.5 * h + 0.5 * a)
        sc_pi = exact_score(a_pi)

        # Subgradient step
        a_sg, sc_sg = backtracking_exact(a, soft_active_weights(d, 0.05), max_depth=20)

        # Pick the better
        if sc_pi <= sc_sg:
            a_new, sc_new = a_pi, sc_pi
        else:
            a_new, sc_new = a_sg, sc_sg

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        if sc_new < exact_score(a):
            a = a_new

    return float(best_sc), best_a


def coordinate_refine(a_start, deadline, max_sweeps=10):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Focus on coordinates with highest values (most impactful)
        order = np.argsort(a)[::-1][:min(n, 2048)]
        for i in order:
            if time.monotonic() > deadline:
                break
            best_local = best_sc
            best_local_a = a
            for factor in (0.92, 0.97, 1.03, 1.08):
                a_t = a.copy()
                a_t[i] *= factor
                a_t = normalize_a(a_t)
                sc = exact_score(a_t)
                if sc < best_local:
                    best_local = sc
                    best_local_a = a_t
            if best_local < best_sc:
                best_sc = best_local
                best_a = best_local_a.copy()
                a = best_local_a.copy()
    return float(best_sc), best_a


def diverse_inits(n, rng, parent):
    inits = []
    if parent is not None:
        inits.append(normalize_a(resample_sequence(parent, n)))
        inits.append(normalize_a(resample_sequence(parent, n)[::-1]))
    x = np.linspace(0.0, 1.0, n)
    inits.append(normalize_a(np.ones(n)))
    inits.append(normalize_a(1.0 + 0.7 * np.abs(2.0 * x - 1.0)))
    inits.append(normalize_a(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * x)))
    inits.append(normalize_a(np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x))))
    Nf = next_power_of_two(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** rng.uniform(1.5, 3.5))
        power[0] = 0.0
        spec = power * np.exp(1j * rng.uniform(-np.pi, np.pi, len(power)))
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        inits.append(normalize_a(1.0 + 0.5 * noise))
    return inits


def perturb(a, rng, strength=0.25):
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.05, 0.2)
        q = (1.0 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        Nf = next_power_of_two(n)
        A = np.fft.rfft(a, Nf)
        A = A * np.exp(1j * rng.uniform(-0.05, 0.05, len(A)))
        q = np.fft.irfft(A, Nf)[:n]
    elif mode == 3:
        # Local reduction at peak-contributing area
        d, _ = d_from_a(a)
        peak_lag = int(np.argmax(d))
        center = peak_lag // 2
        w = max(10, n // 50)
        center = max(w, min(n - w, center))
        q = a.copy()
        q[center - w:center + w] *= (1.0 - 0.2 * np.exp(
            -((np.arange(center - w, center + w) - center) ** 2) / (2.0 * (w // 2) ** 2)))
    elif mode == 4:
        k = max(5, min(31, n // 200))
        if k % 2 == 0:
            k += 1
        sig = k / 4.0
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2.0 * sig ** 2))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    else:
        # Random coordinate swap (mixing)
        i = int(rng.integers(0, n // 2))
        j = int(rng.integers(n // 2, n))
        q = a.copy()
        t = rng.uniform(0.1, 0.3)
        tmp = t * q[i]
        q[i] -= tmp
        q[j] += tmp
    return normalize_a(np.clip(q, 0.0, None))

def _lse_traj(a_start, deadline, betas):
    a = normalize_a(a_start)
    best_a, best_sc = a.copy(), exact_score(a)
    t0 = time.monotonic()
    total = max(5.0, float(deadline - t0 - 5.0))
    for i, beta in enumerate(betas):
        sub_dl = t0 + (i + 1) * total / len(betas)
        sub_dl = min(sub_dl, deadline - 5.0)
        if time.monotonic() > sub_dl - 3:
            continue
        rem = max(3.0, sub_dl - time.monotonic())
        mf = max(64, int(min(2000, rem * 0.8 / max(1e-4, estimate_eval_time(len(a))))))
        sc, a2 = run_lbfgs(a, float(beta), sub_dl, mf)
        if sc < best_sc:
            best_sc, best_a = sc, a2.copy()
        a = normalize_a(a2)
    return float(best_sc), best_a
