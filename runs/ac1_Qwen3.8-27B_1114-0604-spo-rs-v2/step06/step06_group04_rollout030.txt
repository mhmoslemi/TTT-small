"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive LSE smoothing with L-BFGS-B
2. Frank-Wolfe conditional gradient with exact subgradient

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _normalized_score_and_conv(a):
    """Return (score, normalized_autocorr, active_indices) for a unit-sum sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, None, None
    c = _fast_self_conv(a)
    # Normalized: multiply by n/S^2
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    return score, d, dmax

def _subgradient(a, k_active=3, tol=1e-8):
    """
    Compute subgradient of 2*n*max(c)/S^2 at point a (assumed unit sum).
    Returns (score, gradient).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    
    # Find active peaks
    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(d))])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    
    # R[i] = sum_k w_active[k] * a[k-i] (correlation)
    # Using FFT: R = conv(a_rev, w_active) centered
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

def _frank_wolfe(a0, n_iters=5000, deadline=None, k_active=3, tol=1e-8):
    """
    Frank-Wolfe (conditional gradient) method for minimizing 
    2*n*max(c)/S^2 over the simplex.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    # FW parameters
    t = 1
    score0, g0 = _subgradient(a, k_active, tol)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        
        t = it + 1
        gamma = 2.0 / (t + 2.0)
        
        score, grad = _subgradient(a, k_active, tol)
        if not np.isfinite(score):
            break
        
        # FW step: move towards e_{argmin grad}
        i_min = int(np.argmin(grad))
        # Standard FW step
        v = np.zeros(n)
        v[i_min] = 1.0
        a_new = (1.0 - gamma) * a + gamma * v
        
        # Also try Polyak step
        if np.isfinite(score) and score > best_score:
            gap = score - best_score
            # Polyak step: gamma_poly = gap / (grad_dot_diff)
            diff = a - v
            gdot = float(grad.dot(diff))
            if gdot < -1e-12:
                gamma_p = gap / (-gdot)
                gamma_p = min(gamma_p, 1.0)
                a_poly = (1.0 - gamma_p) * a + gamma_p * v
                sc_poly = _exact_score(a_poly)
                if sc_poly < _exact_score(a_new):
                    a_new = a_poly
        
        a = a_new
        
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Convergence check
        if it > 100 and it % 100 == 0:
            if best_score < 1.503:
                # Tighten tolerance
                tol = max(tol * 0.5, 1e-12)
    
    return best_score, best_a

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S

def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient for unit-sum a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    
    # Gradient computation
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

        best_a = a.copy()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive LSE smoothing with L-BFGS-B (aggressive beta schedule)
2. Frank-Wolfe conditional gradient with exact subgradient
3. Peak-shaving coordinate descent with direct evaluation
4. Multi-length search and intensive final polishing
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _score_and_peaks(a, tol_frac=1e-8, max_peaks=10):
    """Return (score, autocorr_data, peak_indices)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, None, None
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax * (1.0 - tol_frac)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(d))])
    if len(active) > max_peaks:
        top_idx = np.argsort(d[active])[-max_peaks:]
        active = active[top_idx]
    return score, d, active


def _subgradient(a, k_active=5, tol=1e-8):
    """Compute subgradient of 2*n*max(c)/S^2 at point a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(d))])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _lse_value_grad(a, beta):
    """LSE-smoothed 2*max(n*c/S^2) and its gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, maxfun=500, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        val, grad = _lse_value_grad(x, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": max(maxfun // 2, 30),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = _normalize(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _frank_wolfe(a0, n_iters=10000, deadline=None, k_active=5, tol=1e-8):
    """
    Frank-Wolfe (conditional gradient) with Polyak step size.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        score, grad = _subgradient(a, k_active, tol)
        if not np.isfinite(score):
            break

        # FW direction
        i_min = int(np.argmin(grad))
        v_fw = np.zeros(n)
        v_fw[i_min] = 1.0

        # Compute objective gap
        gap = score - best_score
        if gap < 1e-14:
            break

        # Try FW step with Polyak and standard step sizes
        diff_fw = a - v_fw
        gdot_fw = float(grad.dot(diff_fw))

        candidates = []
        # Standard FW step
        gamma_std = 2.0 / (it + 2)
        candidates.append(gamma_std)
        # Polyak step
        if gdot_fw < -1e-12:
            gamma_poly = gap / (-gdot_fw)
            gamma_poly = min(gamma_poly, 1.0)
            candidates.append(gamma_poly)

        best_step = None
        best_step_score = score
        for gamma in candidates:
            gamma = max(gamma, 1e-10)
            a_trial = (1.0 - gamma) * a + gamma * v_fw
            sc_trial = _exact_score(a_trial)
            if np.isfinite(sc_trial) and sc_trial < best_step_score:
                best_step_score = sc_trial
                best_step = gamma

        if best_step is not None:
            a = (1.0 - best_step) * a + best_step * v_fw
            sc = _exact_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # No improvement possible from FW step, try a small perturbation
            # Move slightly in the negative gradient direction (projected)
            g_proj = grad - np.mean(grad)
            gn = np.linalg.norm(g_proj)
            if gn > 1e-12:
                step = 0.001 / gn
                a_trial = np.clip(a - step * g_proj, 0, None)
                S_trial = a_trial.sum()
                if S_trial > 1e-12:
                    a_trial = a_trial / S_trial
                    sc_trial = _exact_score(a_trial)
                    if np.isfinite(sc_trial) and sc_trial < best_score:
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        a = a_trial
                    else:
                        break
            else:
                break

        # Tighten tolerance periodically
        if it > 0 and it % 2000 == 0:
            tol = max(tol * 0.5, 1e-12)

    return best_score, best_a


def _nesterov_opt(a0, n_iters=5000, deadline=None, k_active=5, tol=1e-8):
    """Nesterov accelerated subgradient with adaptive step."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t_val = 1.0
    y = a.copy()
    v = np.zeros(n)

    score0, g0 = _subgradient(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a

    eta = 0.03 / g_norm
    no_improve = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta_mom = (t_val - 1.0) / t_new

        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy

        score_y, grad_y = _subgradient(y, k_active, tol)
        if not np.isfinite(score_y):
            break

        step = eta / (1.0 + 0.05 * it)
        x_new = y - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t_val = t_new

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            eta *= 0.5
            no_improve = 0
            if eta < 1e-10:
                break

    return best_score, best_a


def _peak_shaving_cd(a, n_steps=50000, deadline=None, rng=None):
    """
    Peak-shaving coordinate descent: identify the peak, then move mass
    from coordinates that contribute most to the peak to those that contribute
    least.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    if rng is None:
        rng = np.random.default_rng(99999)

    c = _fast_self_conv(a)
    S = 1.0  # a is normalized
    d = n * c / S
    dmax = float(d.max())
    k_star = int(np.argmax(d))

    step = 0
    while step < n_steps:
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break
        step += 1

        if step % 200 == 0:
            sc = _exact_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            c = _fast_self_conv(a)
            d = n * c
            dmax = float(d.max())
            k_star = int(np.argmax(d))

        # Contribution of coordinate i to peak at k_star:
        # (a*a)[k_star] includes a[i]*a[i+k_star] and a[i]*a[i-k_star]
        # (with index wrapping handled by bounds)
        contrib = np.zeros(n)
        for i in range(n):
            contrib[i] = 0.0
        # Vectorized: contrib[i] = a[i+k_star] (if valid) + a[max(0, i-k_star)] (if valid)
        # More precisely: (a*a)[k] = sum_{i=0}^{n-1-k} a[i]*a[i+k]
        # So d/d(a[i]) of (a*a)[k] = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
        idx_plus = np.arange(n) + k_star
        valid_plus = idx_plus < n
        idx_minus = np.arange(n) - k_star
        valid_minus = idx_minus >= 0
        contrib[valid_plus] += a[idx_plus[valid_plus]]
        if k_star > 0:
            contrib[valid_minus] += a[idx_minus[valid_minus]]
        # Also (a*a)[n-1-k] involves the same terms (by symmetry of linear conv)
        # Actually for linear conv, (a*a)[k] = sum_{i} a_i a_{i+k} where both indices in [0,n-1]
        # And (a*a)[n-1-k] = sum_i a_i a_{i+n-1-k}... this is a different peak
        # Let's just focus on k_star

        # Find top contributors (to remove mass from) and bottom contributors (to add mass to)
        # Add random noise to break ties
        noise = rng.standard_normal(n) * 1e-6
        contrib_noisy = contrib + noise

        # Source: high contribution, low value (easy to remove)
        # Destination: low contribution, and we want to add mass
        # Score for source: contrib[i] / max(a[i], eps) - higher is better to remove
        # Score for dest: contrib[j] - lower is better to add to
        eps = 1e-12
        source_score = contrib_noisy / (a + eps)
        dest_score = contrib_noisy

        # Randomize among top/bottom
        n_cand = max(3, n // 100)
        top_sources = np.argsort(source_score)[-n_cand:]
        top_dests = np.argsort(dest_score)[:n_cand]

        i_src = top_sources[int(rng.integers(0, n_cand))]
        i_dst = top_dests[int(rng.integers(0, n_cand))]
        if i_src == i_dst:
            continue

        # Try transferring a fraction of a[i_src] to a[i_dst]
        for frac in [0.05, 0.02, 0.01]:
            delta = a[i_src] * frac
            if delta < 1e-15:
                break
            a[i_src] -= delta
            a[i_dst] += delta
            sc = _exact_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                break
            else:
                a[i_src] += delta
                a[i_dst] -= delta
                if sc >= best_score + 1e-6:
                    break  # Larger won't help either if smaller didn't

    sc = _exact_score(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def _resample(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0 or n_new == m:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _perturb(a, rng, strength=0.3):
    """Generate a perturbation of a."""
    a = _normalize(a)
    n = a.size
    mode = int(rng.integers(0, 4))
    if mode == 0:
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Smooth perturbation
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.2 * noise)
    else:
        q = a + 0.01 * rng.standard_normal(n)

    return _normalize(np.clip(q, 0.0, None))


def _multi_stage_optimize(a0, deadline, label=""):
    """
    Multi-stage optimization pipeline:
    1. LSE + L-BFGS (increasing beta)
    2. Frank-Wolfe
    3. LSE + L-BFGS (very high beta)
    4. Nesterov
    5. Peak-shaving CD
    """
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline) - t0)

    print("[{}] start: score={:.8f} n={}".format(label, best_score, a.size))

    # Stage 1: LSE + L-BFGS with increasing beta (40% of time)
    betas = [100, 500, 2000, 10000, 50000]
    stage1_end = t0 + 0.40 * total
    for beta in betas:
        rem = stage1_end - time.monotonic()
        if rem < 10.0:
            break
        n = len(a)
        maxfun = int(max(50, min(1000, rem / max(0.02, 0.005 * np.log2(max(16, n) / 1024)))))
        sc, a2 = _lbfgs_lse(a, beta, maxfun=maxfun, deadline=None)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Stage 2: Frank-Wolfe (20% of time)
    stage2_end = t0 + 0.60 * total
    rem = stage2_end - time.monotonic()
    if rem > 15.0:
        n_it = int(max(1000, rem * 30))
        for k_act in [3, 5, 8]:
            rem = stage2_end - time.monotonic()
            if rem < 10.0:
                break
            sc, a2 = _frank_wolfe(a, n_iters=n_it, deadline=stage2_end,
                                   k_active=k_act, tol=1e-8)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] FW k={} tol=1e-8: {:.8f}".format(label, k_act, best_score))
            a = _normalize(a2)

    # Stage 3: Very high beta LSE + L-BFGS (20% of time)
    stage3_end = t0 + 0.80 * total
    for beta in [200000, 1000000, 5000000]:
        rem = stage3_end - time.monotonic()
        if rem < 10.0:
            break
        maxfun = int(max(30, min(500, rem / 0.05)))
        sc, a2 = _lbfgs_lse(a, beta, maxfun=maxfun, deadline=None)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] LSE-high beta={:.0f}: {:.8f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Stage 4: Nesterov (10% of time)
    stage4_end = t0 + 0.90 * total
    rem = stage4_end - time.monotonic()
    if rem > 10.0:
        n_it = int(max(500, rem * 25))
        for k_act in [2, 3, 5]:
            rem = stage4_end - time.monotonic()
            if rem < 5.0:
                break
            sc, a2 = _nesterov_opt(a, n_iters=n_it, deadline=stage4_end,
                                    k_active=k_act, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Nesterov k={} tol=1e-9: {:.8f}".format(label, k_act, best_score))
            a = _normalize(a2)

    # Stage 5: Peak-shaving CD (10% of time)
    rem = float(deadline) - time.monotonic()
    if rem > 10.0:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
        n_steps = int(max(5000, rem * 20))
        sc, a2 = _peak_shaving_cd(a, n_steps=n_steps, deadline=float(deadline), rng=rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] PeakCD: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns list of non-negative floats."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths to try (around parent length)
    if parent is not None:
        plen = min(parent.size, 16384)
        lengths = [plen]
        for L in [plen - 512, plen + 512, plen - 256, plen + 256, 10000, 10500]:
            if 2048 <= L <= 16384 and L not in lengths:
                lengths.append(L)
        lengths = lengths[:4]
    else:
        lengths = [10240, 8192, 12288]

    print("Lengths to try: {}".format(lengths))

    # Initialize best
    best_score = float(np.inf)
    best_seq = None

    # Try each length
    n_lengths = len(lengths)
    # Allocate time: first (parent) length gets most time
    time_fracs = [0.55] + [0.45 / (n_lengths - 1)] * (n_lengths - 1) if n_lengths > 1 else [1.0]

    for li, n in enumerate(lengths):
        rem_total = deadline - time.monotonic()
        if rem_total < 60.0:
            break

        alloc = time_fracs[li] * rem_total
        sub_deadline = time.monotonic() + alloc

        # Generate initial candidates
        inits = []
        if parent is not None and li == 0:
            # Start from parent
            p_resampled = _resample(parent, n)
            inits.append(p_resampled)
            inits.append(p_resampled[::-1])

        # Add structured inits
        x = np.linspace(0.0, 1.0, n)
        inits.append(np.ones(n) / n)
        for alpha in [0.5, 1.0, 2.0]:
            inits.append(np.clip(1.0 + alpha * np.abs(2.0 * x - 1.0), 0.01, None))
        for alpha in [0.5, 1.0]:
            inits.append(np.clip(1.0 + alpha * np.cos(2.0 * np.pi * x), 0.01, None))
        # Perturbed parent (if available)
        if parent is not None:
            for _ in range(2):
                p_resampled = _resample(parent, n)
                inits.append(_perturb(p_resampled, rng))

        # Evaluate inits
        scored_inits = []
        for arr in inits:
            arr = _normalize(arr)
            sc = _exact_score(arr)
            if np.isfinite(sc):
                scored_inits.append((sc, arr))

        scored_inits.sort(key=lambda x: x[0])
        print("Length {}: best init = {:.8f}".format(n, scored_inits[0][0] if scored_inits else 'inf'))

        if not scored_inits:
            continue

        # Pick top 2 inits (different shapes)
        starts = []
        for sc, arr in scored_inits[:3]:
            starts.append(arr)
        if not starts:
            starts.append(np.ones(n) / n)

        # Optimize each start
        for si, start in enumerate(starts[:2]):
            rem = sub_deadline - time.monotonic()
            if rem < 30.0:
                break

            # Allocate time between starts
            if si == 0:
                sub_dead = time.monotonic() + rem * 0.7
            else:
                sub_dead = sub_deadline

            # For the second start, blend with best
            if si > 0 and best_seq is not None:
                best_r = _resample(best_seq, n)
                start = _normalize(0.6 * start + 0.4 * best_r)

            sc, seq = _multi_stage_optimize(start, sub_dead, "len{}_s{}".format(n, si))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("  -> New best: {:.8f} (n={}, start={})".format(sc, n, si))

    # Final intensive polish on the best sequence
    if best_seq is not None:
        rem = deadline - time.monotonic()
        print("Final polish: {:.1f}s remaining, current score={:.8f}".format(rem, best_score))

        if rem > 60.0:
            # Intensive LSE with very high beta
            for beta in [1000000, 5000000, 20000000]:
                rem = deadline - time.monotonic()
                if rem < 15.0:
                    break
                maxfun = int(max(30, min(300, rem / 0.1)))
                sc, a2 = _lbfgs_lse(best_seq, beta, maxfun=maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("  Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
                best_seq = _normalize(a2)

        if rem > 30.0:
            # Frank-Wolfe with tight tolerance
            remaining = deadline - 20.0 - time.monotonic()
            if remaining > 10.0:
                n_it = int(max(2000, remaining * 25))
                sc, a2 = _frank_wolfe(best_seq, n_iters=n_it, deadline=deadline - 15.0,
                                       k_active=5, tol=1e-10)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("  Final FW: {:.8f}".format(best_score))
                best_seq = _normalize(a2)

        if rem > 20.0:
            # Peak shaving
            remaining = deadline - 10.0 - time.monotonic()
            if remaining > 5.0:
                rng2 = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                n_steps = int(max(5000, remaining * 30))
                sc, a2 = _peak_shaving_cd(best_seq, n_steps=n_steps,
                                           deadline=deadline - 5.0, rng=rng2)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("  Final PeakCD: {:.8f}".format(best_score))
                best_seq = _normalize(a2)

        # Ultimate Nesterov polish
        rem = deadline - 5.0 - time.monotonic()
        if rem > 5.0:
            n_it = int(max(500, rem * 30))
            sc, a2 = _nesterov_opt(best_seq, n_iters=n_it, deadline=deadline - 3.0,
                                    k_active=3, tol=1e-10)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("  Final Nesterov: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Final output
    if best_seq is None:
        best_seq = np.ones(1024, dtype=np.float64) * 0.5
        best_score = 2.0

    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
