Optimized step function sequence search using:

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:2*n-1]

    """Score: 2n * max(convolve(a,a)) / (sum(a))^2"""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S*S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:

    exp_s = np.exp(shifted - shift)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S

"""
Step function optimization using:
1. Adaptive GS-Fienup spectral shaping
2. LP-inspired peak reweighting (solve small QPs on top peaks)
3. High-beta LSE-LBFGS for near-exact max minimization
4. Stochastic local search for final polish
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:2*n-1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S*S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shift = float(shifted.max())
    exp_s = np.exp(shifted - shift)
    log_sum = shift + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S*S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 3*n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

    bounds = [(0.0, 1000.0)] * n
    def obj(x):
        return lse_score_grad(x, beta)
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-15, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    return np.clip(res.x, 0.0, 1000.0)

def gs_adaptive(a, n_iters=400):
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n)/n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        t = it / n_iters
        # Peak-aware target
        target = np.where(c > c_mean, c * 0.2 + c_mean * 0.8, c)
        target = np.maximum(target, 1e-10)
        T = np.fft.fft(target[:2*n-1], M)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        alpha = 0.3 + 0.1 * t
        new_mag = A_mag ** (1-alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1-alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
