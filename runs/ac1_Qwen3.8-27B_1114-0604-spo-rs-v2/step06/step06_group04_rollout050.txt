from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

    x_old = np.linspace(0, 1, m)

    return np.interp(x_new, x_old, a).astype(np.float64)

    a = np.asarray(a, dtype=np.float64).ravel()

        return np.inf, np.zeros(n)

    n = a.size

    v = np.zeros(n)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline - 2.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
        beta_mom = (t_count - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())

            Sy = 1.0

        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.1 * it)
        x_new = y_norm - step * grad_y

            best_a = a.copy()
        if it > 0 and it % 100 == 0:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 with progressive LSE + Nesterov + mirror descent.
Strategy: Start from parent, aggressive multi-phase optimization with all active peaks.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _np2(m):
    """Next power of 2, minimum 1."""
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _selfconv(a):
    """Fast full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _np2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _score(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _selfconv(a)
    return float(2.0 * n * c.max() / (S * S))


def _norm(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(a.size, 1)
    return a / S


def _resample(a, n):
    """Resample array to length n via linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    xo = np.linspace(0.0, 1.0, m)
    xn = np.linspace(0.0, 1.0, n)
    return np.interp(xn, xo, a).astype(np.float64)


def _lse_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient.
    Objective: 2*n*(1/beta)*log(sum(exp(beta * d_k))) where d_k = n*c_k/S^2
    Returns (value, gradient) for the unnormalized variable a.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _selfconv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400.0, 400.0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se
    LSE = dmax + np.log(se) / beta
    val = 2.0 * LSE
    # Gradient of LSE w.r.t. a
    # d LSE / da_i = sum_k w_k * D c_k / da_i where D c_k / da_i = 2*a[k-i] (valid range)
    # = 2 * R[i] where R[i] = sum_k w_k * a[k-i]
    # But we also have the 1/S^2 factor:
    # d = n * c / S^2
    # dd_i / da_i = n * (D c_i / da_i) / S^2 - 2n * c_i / S^3
    # d LSE / da_i = sum_k w_k * dd_k / da_i
    # = n/S^2 * sum_k w_k * D c_k / da_i - 2n/S^3 * sum_k w_k * c_k
    # = n/S^2 * 2*R[i] - 2n*T/S^3
    # where T = sum_k w_k * c_k
    T = float(np.dot(w, c))
    N2 = _np2(3 * n - 1)
    ar = a[::-1]
    AR = np.fft.rfft(ar, N2)

    R = cf[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)

def _subgrad_full(a, tol_frac=1e-6):
    """
    Exact subgradient of 2*n*max(c)/S^2.
    Uses ALL peaks within tol_frac of the maximum.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _selfconv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    thr = dmax * (1.0 - tol_frac)
    active = np.where(d >= thr)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(d))])
    if len(active) > 100:
        idx = np.argsort(d[active])[-100:]
        active = active[idx]
    wa = np.zeros(2 * n - 1)
    wa[active] = 1.0 / len(active)
    T = float(np.dot(wa, c))
    N2 = _np2(3 * n - 1)
    ar = a[::-1]
    AR = np.fft.rfft(ar, N2)
    WR = np.fft.rfft(wa, N2)
    cf = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = cf[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs(a0, beta, deadline, maxfun=300):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = _norm(a0)
    n = a.size
    bounds = [(0.0, None)] * n
    opts = {"ftol": 1e-15, "gtol": 1e-11, "maxiter": max(50, maxfun // 2),
            "maxfun": int(maxfun), "maxls": 30}
    try:
        res = minimize(_lse_wrap, a, method="L-BFGS-B", jac=True,
                       args=(beta,), bounds=bounds, options=opts)
    except Exception:
        return _score(a), a
    a_out = _norm(res.x) if hasattr(res, 'x') and res.x is not None else a
    sc = _score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        return _score(a), a
    return float(sc), a_out


def _lse_wrap(x, beta):

    x = np.clip(x, 0.0, None)
    S = float(x.sum())
    if S < 1e-12:

    val, grad = _lse_grad(x, beta)
    return val, grad
