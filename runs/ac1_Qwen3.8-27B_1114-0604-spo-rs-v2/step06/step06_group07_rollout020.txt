def fienup_geometric(a, n_iters=200, alpha_start=0.05, alpha_end=0.7):
    """Fienup with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S = float(a_new.sum())

            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a

            best_a = a.copy()

        # Find top peaks
        top_idx = np.argpartition(c, -min(n_top, len(c)))[-min(n_top, len(c)):]

        # For each peak at lag k, the frequency content that contributes is
        # where |A(f)|^2 has high values at phase related to k.
        # Simply: reduce amplitude at bins where constructive interference
        # at the peak lags is strongest.
        amp = np.sqrt(A.real ** 2 + A.imag ** 2) + 1e-20
        phase = np.angle(A)

        # Compute "pressure" from peaks: for each bin, how much does it
        # contribute to the peaks?
        # c[k] = (1/M) * sum_f |A(f)|^2 * exp(i*f*k) ... (for circular)
        # For linear conv, approximate: reduce bins with high |A(f)|
        # that align with peak phases.
        pressure = np.zeros(n_bins)
        for k in top_idx:
            k = int(k)
            # Phase of contribution at bin f: exp(i * 2*pi*f*k/M_fft)
            freqs = np.arange(n_bins) * (2.0 * np.pi / M_fft)
            contribution = np.cos(freqs * k)
            pressure += np.maximum(contribution, 0) * amp

        # Reduce amplitude at high-pressure bins
        p_max = float(pressure.max())
        if p_max > 1e-15:
            p_norm = pressure / p_max
            reduction = 0.05 * (it / n_iters + 0.1)
            amp_new = amp * (1.0 - reduction * p_norm)
            amp_new = np.maximum(amp_new, 1e-20)
        else:
            amp_new = amp

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

    return best_a


def run_lbfgs(a0, beta, maxiter, maxfun, M_fft, M2_fft):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_for_lbfgs, a0, args=(beta, M_fft, M2_fft),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def structured_perturb(a, rng, scale):
    """Apply structured perturbation: smooth noise + random phase."""
    n = len(a)
    x = np.arange(n) / n
    # Smooth perturbation
    freqs = rng.integers(1, 8, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    noise = np.zeros(n)
    for f, ph in zip(freqs, phases):
        noise += np.cos(2 * np.pi * f * x + ph)
    noise /= (np.abs(noise).max() + 1e-15)
    # Add small white noise
    noise += 0.3 * rng.standard_normal(n)
    a_new = a * (1.0 + scale * noise)
    a_new = np.maximum(a_new, 0.0)
    S = float(a_new.sum())
    if S < 1e-12:
        return a.copy()
    return a_new / S


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function using multi-phase optimization:
    Phase 1: L2 warmup + Proximal gradient with LSE (adaptive beta)
    Phase 2: Soft peak-directed descent
    Phase 3: L-BFGS polish with high beta
    Phase 4: Final iterations and verification
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}")
    print(f"Parent available: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # Precompute FFT sizes
    M_fft = 1 << (2 * n_work - 1).bit_length()
    M2 = n_work + 2 * n_work - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    # ============================================================
    # Phase 1: L2 warmup + Proximal gradient (40% time)
    # ============================================================
    phase1_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 1: L2 Warmup + Proximal Gradient (deadline t={phase1_end - t_start:.0f}s) ===")

    # L2 warmup
    print("  L2 warmup...")
    a_l2 = l2_warmup(best_a, n_iters=200)
    sc = compute_score(a_l2)
    update_best(sc, a_l2, "L2")

    # Proximal gradient with adaptive beta
    remaining = phase1_end - time.monotonic()
    if remaining > 60:
        n_iters_pg = int(remaining * 8)  # ~8 iterations per second target
        n_iters_pg = max(500, min(n_iters_pg, 20000))
        print(f"  Proximal gradient: {n_iters_pg} iters, beta 10->5000")
        a_pg = prox_gradient_lse(best_a, beta_start=10.0, beta_end=5000.0,
                                 n_iters=n_iters_pg, verbose=True)
        sc = compute_score(a_pg)
        update_best(sc, a_pg, "PROX-GRAD")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Soft peak-directed descent (25% time)
    # ============================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Soft Peak Descent (deadline t={phase2_end - t_start:.0f}s) ===")

    remaining = phase2_end - time.monotonic()
    if remaining > 60:
        n_iters_spd = int(remaining * 5)
        n_iters_spd = max(500, min(n_iters_spd, 50000))
        print(f"  Soft peak descent: {n_iters_spd} iters, temp=0.05")
        a_spd = soft_peak_descent(best_a, n_iters=n_iters_spd,
                                  temperature=0.05, step_init=0.015)
        sc = compute_score(a_spd)
        update_best(sc, a_spd, "SOFT-PEAK")

    # Fienup targeted (frequency-domain peak reduction)
    remaining = phase2_end - time.monotonic()
    if remaining > 60:
        n_iters_ft = int(remaining * 3)
        n_iters_ft = max(200, min(n_iters_ft, 10000))
        print(f"  Fienup targeted: {n_iters_ft} iters")
        a_ft = fienup_targeted(best_a, n_iters=n_iters_ft, n_top=10)
        sc = compute_score(a_ft)
        update_best(sc, a_ft, "F-TARGETED")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS polish (20% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS Polish (deadline t={phase3_end - t_start:.0f}s) ===")

    betas = [500, 2000, 8000, 20000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(5000, int(remaining * 20))
        maxfun = maxiter * 20
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun,
                               M_fft=M_fft, M2_fft=M2_fft)
        update_best(sc, a_opt, f"LBFGS b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish with restarts (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # More proximal gradient with very high beta
    if time.monotonic() < deadline - 120:
        remaining = deadline - 120 - time.monotonic()
        n_iters = int(remaining * 8)
        n_iters = max(200, min(n_iters, 10000))
        print(f"  Final proximal gradient: {n_iters} iters, beta 5000->50000")
        a_final = prox_gradient_lse(best_a, beta_start=5000.0, beta_end=50000.0,
                                    n_iters=n_iters)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-PROX")

    # Soft peak with lower temperature
    if time.monotonic() < deadline - 90:
        remaining = deadline - 90 - time.monotonic()
        n_iters = int(remaining * 5)
        n_iters = max(100, min(n_iters, 5000))
        print(f"  Final soft peak: {n_iters} iters, temp=0.02")
        a_final2 = soft_peak_descent(best_a, n_iters=n_iters,
                                     temperature=0.02, step_init=0.008)
        sc = compute_score(a_final2)
        update_best(sc, a_final2, "FINAL-SOFT")

    # L-BFGS with very high beta
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = min(3000, int(remaining * 25))
        maxfun = maxiter * 20
        print(f"  Final L-BFGS: beta=50000, maxiter={maxiter}")
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxfun,
                               M_fft=M_fft, M2_fft=M2_fft)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # Structured perturbations + short optimization
    if time.monotonic() < deadline - 60:
        for trial in range(4):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.01 * (1 + trial * 0.5)
            a_p = structured_perturb(best_a, rng, scale)
            # Quick proximal gradient
            a_r = prox_gradient_lse(a_p, beta_start=2000.0, beta_end=10000.0,
                                    n_iters=300)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    # Fienup geometric as final touch
    if time.monotonic() < deadline - 30:
        a_fg = fienup_geometric(best_a, n_iters=100, alpha_start=0.02, alpha_end=0.4)
        sc = compute_score(a_fg)
        update_best(sc, a_fg, "FINAL-GEO")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
