"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/S^2 on non-negative sequences.
Uses: high-beta LSE annealing, adaptive RMSProp peak shaving, spectral perturbation
restarts, top-K order-statistic smoothing, and block coordinate descent.
Targets sub-1.503 score from a 1.5058 starting point.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad

                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        a_out = normalize(a_out)
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def rmsprop_peak_shave(a, n_iters=5000, top_k=10, lr=0.002, decay=0.9):
    """
    RMSProp-style adaptive peak shaving.
    Maintains a running average of squared gradients for per-component step sizes.
    Targets top-K autocorrelation peaks simultaneously.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    sq_avg = np.full(n, lr * lr)
    t = 0.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Build subgradient
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M_val * top_k / (top_k * S3))
        # Simplify: grad = 2n * (sg/S^2 - 2*M_val/(K*S^3) * K) = 2n*(sg/S^2 - 2*M_val/S^3)
        # Actually the subgradient of 2n*M/S^2 w.r.t. a:
        # d(2n*M)/da = 2n * dM/da (subgradient)
        # d(S^2)/da = 2S*e
        # d(M/S^2)/da = dM/S^2 - M*2S*e/S^4 = dM/S^2 - 2M*e/S^3

        # Recompute properly
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
        # grad now = dM/da (subgradient, averaged over top-K)
        grad /= float(top_k)
        grad = (2.0 * n) * (grad / S2 - 2.0 * M_val * np.ones(n) / (S * S2))

        # RMSProp update
        t += 1.0
        sq_avg = decay * sq_avg + (1.0 - decay) * grad * grad
        step_size = lr / (np.sqrt(sq_avg) + 1e-8)

        a = a - step_size * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        # Periodically try a slightly larger step
        if it % 1000 == 500:
            a_trial = a - 2.0 * step_size * grad
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st > 1e-6:
                a_trial = a_trial / st
                sc_trial = compute_score(a_trial)
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a


def fienup_relaxed(a, n_iters=200, relax=0.5):
    """
    Relaxed Fienup: blend between current spectrum and flat spectrum.
    Uses a challenging profile that gradually increases flattening.
    """
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Relaxed: interpolate toward flat with increasing strength
        alpha = relax * (0.3 + 0.7 * progress)
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-20)

        ratio = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_notch(a, n_iters=200):
    """Fienup that specifically notches out spectral peaks."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Notch: reduce top 30% of spectral peaks
        thresh = np.quantile(P, 0.70 + 0.25 * progress)
        P_new = P.copy()
        P_new[P_new > thresh] = thresh

        # Also lift the bottom 20%
        low_thresh = np.quantile(P, 0.20)
        P_new[P_new < low_thresh] = low_thresh * 0.8

        P_new = np.maximum(P_new, 1e-20)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-20))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def block_coord_descent(a, n_sweeps=20, block_size=200):
    """
    Block coordinate descent: optimize consecutive blocks of variables.
    Within each block, use L-BFGS with frozen outside.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        # Shuffle block order
        order = np.random.permutation(n_blocks)
        for bi in order:
            start = bi * block_size
            end = min(start + block_size, n)
            if end - start < 10:
                continue

            a_block = a[start:end].copy()
            a_frozen = a.copy()

            def block_obj(x):
                a_trial = a_frozen.copy()
                a_trial[start:end] = x
                return compute_score(a_trial)

            # Simple Nelder-Mead on the block (works for small blocks)
            if end - start <= 50:
                try:
                    res = minimize(block_obj, a_block.copy(), method='Nelder-Mead',
                                   options={'fatol': 1e-20, 'maxiter': 200, 'xatol': 1e-15, 'fatol': 1e-15})
                    a_trial = a_frozen.copy()
                    a_trial[start:end] = np.maximum(res.x, 0.0)
                    a_trial = normalize(a_trial)
                    sc = compute_score(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        a_frozen = a.copy()
                except Exception:
                    pass

        # Laplacian: occasionally swap adjacent blocks for exploration
        if sweep % 5 == 4:
            idx1 = np.random.randint(n // 2)
            idx2 = idx1 + np.random.randint(1, block_size)
            if idx2 < n:
                a[idx1], a[idx2] = a[idx2], a[idx1]
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[idx1], a[idx2] = a[idx2], a[idx1]
                    a = normalize(a)

    return best_a


def spectral_perturb(a, rng, mode=0):
    """
    Apply perturbations in the frequency domain for diverse restarts.
    mode: 0=phase jitter, 1=amplitude jitter, 2=additive spectral noise
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    if mode == 0:
        # Phase perturbation
        phase = np.angle(A)
        mag = np.abs(A)
        phase_noise = 0.05 * rng.standard_normal(len(phase))
        A_new = mag * np.exp(1j * (phase + phase_noise))
    elif mode == 1:
        # Amplitude perturbation
        phase = np.angle(A)
        mag = np.abs(A) * (1.0 + 0.03 * rng.standard_normal(len(mag)))
        mag = np.maximum(mag, 0.0)
        A_new = mag * np.exp(1j * phase)
    else:
        # Additive low-frequency spectral noise
        noise = np.zeros_like(A)
        for m in range(2):
            freq = rng.integers(1, max(2, len(A) // 4))
            amp = 0.01 * rng.uniform(0.5, 1.0)
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            noise[freq] += amp * np.exp(1j * phase)
        A_new = A + noise

    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def lp_direction(a, M_target):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M_target, b >= 0.
    Returns the optimal b (normalized) or None if infeasible.
    """
    n = len(a)
    # Constraints: for each k=0..2n-2, sum_j a[j]*b[k-j] <= M_target
    # This is a Toeplitz-like system.
    # For efficiency, use scipy linprog with sparse-like structure.
    # Actually for n=10000, this LP has 19999 constraints and 10000 vars - too slow.
    # Use a projected subgradient approximation instead.
    
    # Approximate: use a few iterations of projected subgradient
    b = a.copy()
    b = normalize(b)
    lr = 0.001
    for it in range(200):
        c = fast_conv_cross(a, b)
        # Find violated constraints
        violations = c - M_target
        if np.max(violations) < 1e-10:
            break
        # Subgradient of max(0, c-M) w.r.t. b
        top_viol = np.argsort(violations)[::-1][:50]
        grad_b = np.zeros(n)
        for k in top_viol:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad_b[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
        lr *= 0.995
        b -= lr * (grad_b / 50.0)
        b = np.maximum(b, 0.0)
        sb = b.sum()
        if sb < 1e-12:
            return None
        b = b / sb
    
    return b


def overshoot_retry(a, rng):
    """
    Try larger perturbations and see if they lead to a better basin.
    """
    n = len(a)
    results = []
    for scale in [0.02, 0.04, 0.06, 0.08, 0.10]:
        a_pert = a + scale * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-6:
            continue
        a_pert = a_pert / S
        results.append(a_pert)
    return results


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage optimizer targeting sub-1.503 score.
    Phases: Fienup init, high-beta LSE annealing, RMSProp peak shaving,
            spectral restarts, block coordination, ultra-fine polish.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
        a_init = normalize(parent.copy())
        n_seq = len(a_init)
    else:
        n_seq = 10000
        a_init = np.ones(n_seq) / n_seq
        print("No parent. Starting uniform n=10000.")

    if best_a is None:
        best_a = a_init.copy()

    print(f"Initial best: {best_score:.8f}, n={n_seq}")

    # ==================================================================
    # Phase 1: Spectral initialization (5% time)
    # ==================================================================
    p1_end = t_start + 0.05 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Spectral Init (budget {p1_end - t_start:.0f}s) ===")

    a_f1 = fienup_relaxed(a_init, n_iters=150, relax=0.6)
    update(compute_score(a_f1), a_f1, "FIENUP-relaxed")

    if time.monotonic() < p1_end - 5:
        a_f2 = fienup_notch(a_init, n_iters=150)
        update(compute_score(a_f2), a_f2, "FIENUP-notch")

    # Try a random-phase spectral init
    n = n_seq
    M_fft = 1 << (2 * n - 1).bit_length()
    for trial in range(3):
        if time.monotonic() > p1_end - 2:
            break
        phase = 2 * np.pi * rng.random(M_fft // 2 + 1)
        A = np.full(M_fft // 2 + 1, np.sqrt(n / M_fft)) * np.exp(1j * phase)
        a_rand = np.fft.irfft(A, M_fft)[:n]
        a_rand = np.maximum(a_rand, 0.0)
        S = a_rand.sum()
        if S > 1e-6:
            a_rand = a_rand / S
            sc = compute_score(a_rand)
            if sc < best_score:
                update(sc, a_rand, f"SPEC-init-{trial}")

    # ==================================================================
    # Phase 2: High-beta LSE Annealing (30% time)
    # ==================================================================
    p2_end = t_start + 0.35 * (budget_s - 30.0)
    print(f"\n=== Phase 2: LSE Annealing High-Beta (budget {p2_end - t_start:.0f}s) ===")

    a_work = normalize(best_a.copy())
    
    # Coarse-to-fine, going much higher than before
    beta_schedule = [50, 200, 1000, 5000, 20000, 80000, 200000, 500000]
    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > p2_end - 30:
            break
        remaining = p2_end - 30 - time.monotonic()
        if remaining < 20:
            break
        # More iterations for higher betas (where the real optimization happens)
        if beta <= 1000:
            maxiter = max(500, int(remaining * 15))
        else:
            maxiter = max(500, int(remaining * 10))
        
        print(f"  LSE beta={beta:>6}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 30,
                              tol=1e-18)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE b={beta}")
        a_work = normalize(a_opt.copy())

    # Oscillating beta: go up and down to escape plateaus
    if time.monotonic() < p2_end - 30:
        print("  Oscillating beta pass...")
        for beta in [100000, 300000, 100000, 500000, 200000, 1000000]:
            if time.monotonic() > p2_end - 15:
                break
            remaining = p2_end - 15 - time.monotonic()
            if remaining < 15:
                break
            maxiter = max(300, int(remaining * 8))
            sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 20,
                                  tol=1e-18)
            print(f"    osc beta={beta}: score={sc:.8f}")
            update(sc, a_opt, f"OSC b={beta}")
            a_work = normalize(a_opt.copy())

    # ==================================================================
    # Phase 3: RMSProp Peak Shaving (18% time)
    # ==================================================================
    p3_end = t_start + 0.53 * (budget_s - 30.0)
    print(f"\n=== Phase 3: RMSProp Peak Shaving (budget {p3_end - t_start:.0f}s) ===")

    for top_k, lr in [(5, 0.003), (10, 0.002), (20, 0.0015), (3, 0.005)]:
        if time.monotonic() > p3_end - 20:
            break
        remaining_iters = int((p3_end - 20 - time.monotonic()) * 1500)
        remaining_iters = max(1000, min(remaining_iters, 50000))
        a_ps = rmsprop_peak_shave(best_a, n_iters=remaining_iters, top_k=top_k, lr=lr)
        sc = compute_score(a_ps)
        update(sc, a_ps, f"RMSPROP k={top_k}")
        
        # Follow-up LSE on shaved result
        if sc < best_score and time.monotonic() < p3_end - 15:
            sc2, a2 = run_lbfgs(a_ps, 300000, maxiter=2000, maxfun=50000, tol=1e-18)
            update(sc2, a2, f"RMSPROP+LSE k={top_k}")

    # ==================================================================
    # Phase 4: Spectral Multi-Restart (18% time)
    # ==================================================================
    p4_end = t_start + 0.71 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Spectral Multi-Restart (budget {p4_end - t_start:.0f}s) ===")

    n_restarts = 20
    for r in range(n_restarts):
        if time.monotonic() > p4_end - 25:
            break
        remaining = p4_end - 25 - time.monotonic()
        if remaining < 25:
            break

        mode = r % 3
        a_pert = spectral_perturb(best_a, rng, mode=mode)
        
        # Also try additive perturbation for variety
        if r % 4 == 2:
            a_pert = best_a + 0.005 * rng.standard_normal(n_seq)
            a_pert = normalize(a_pert)
        elif r % 4 == 3:
            a_perts = overshoot_retry(best_a, rng)
            a_pert = a_perts[r % len(a_perts)]

        # Quick Fienup on perturbed
        if time.monotonic() < p4_end - 40:
            a_pert = fienup_relaxed(a_pert, n_iters=50, relax=0.4)

        # LSE optimization
        beta = 50000 + r * 30000
        maxiter = max(500, int(remaining * 8))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 20,
                              tol=1e-17)
        update(sc, a_opt, f"RESTART-{r} b={beta}")

        # If improved, do a second pass
        if sc < best_score and time.monotonic() < p4_end - 20:
            sc2, a2 = run_lbfgs(a_opt, beta * 3, maxiter=1500, maxfun=30000, tol=1e-18)
            update(sc2, a2, f"RESTART-{r}-2pass")

    # ==================================================================
    # Phase 5: Block Coordinate Descent (8% time)
    # ==================================================================
    p5_end = t_start + 0.79 * (budget_s - 30.0)
    print(f"\n=== Phase 5: Block Coord Descent ===")

    a_cd = block_coord_descent(best_a, n_sweeps=10, block_size=100)
    sc = compute_score(a_cd)
    update(sc, a_cd, "BLOCK-CD")

    if time.monotonic() < p5_end - 10:
        sc2, a2 = run_lbfgs(a_cd, 200000, maxiter=2000, maxfun=40000, tol=1e-18)
        update(sc2, a2, "BLOCK+LSE")

    # ==================================================================
    # Phase 6: Ultra-Fine Polish (19% time)
    # ==================================================================
    print(f"\n=== Phase 6: Ultra-Fine Polish ===")

    # Very high beta LSE, multiple passes
    for beta in [500000, 1000000, 2000000]:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - 20 - time.monotonic()
        if remaining < 15:
            break
        maxiter = max(500, int(remaining * 6))
        sc, a_final = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 15,
                                tol=1e-19)
        update(sc, a_final, f"ULTRA b={beta}")

    # Final RMSProp
    if time.monotonic() < deadline - 15:
        remaining_iters = max(2000, int((deadline - 15 - time.monotonic()) * 2000))
        a_ps = rmsprop_peak_shave(best_a, n_iters=min(remaining_iters, 30000),
                                   top_k=5, lr=0.001, decay=0.95)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-RMSPROP")

    # Final tiny LSE
    if time.monotonic() < deadline - 8:
        sc, a_final = run_lbfgs(best_a, 5000000, maxiter=1000, maxfun=20000, tol=1e-19)
        update(sc, a_final, "FINAL-LSE-5M")

    # ==================================================================
    # LP-based direction (if time remains)
    # ==================================================================
    if time.monotonic() < deadline - 10:
        c_best = fast_conv(normalize(best_a))
        M_target = float(c_best.max()) * 1.001
        a_new_dir = lp_direction(normalize(best_a), M_target)
        if a_new_dir is not None:
            # Blend
            for t_blend in [0.3, 0.5, 0.7]:
                a_blend = (1 - t_blend) * normalize(best_a) + t_blend * a_new_dir
                a_blend = normalize(a_blend)
                sc = compute_score(a_blend)
                update(sc, a_blend, f"LP-blend {t_blend}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
