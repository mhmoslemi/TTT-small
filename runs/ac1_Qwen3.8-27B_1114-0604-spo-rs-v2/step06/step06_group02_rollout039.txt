```python
"""
Alternating minimization + LSE polish for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses the insight that min_a max_k c_k = min_a min_q max... leads to an
alternating scheme between peak distribution and a QP with Hankel structure.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def hankel_mv(q, v):
    """Compute M(q) @ v where M(i,j) = q_{i+j}. q has length 2n-1, v has length n."""
    n = len(v)
    v_r = v[::-1]
    # Use FFT for speed
    M_fft = 1 << (len(q) + n - 2).bit_length()
    Q = np.fft.rfft(q, M_fft)
    V = np.fft.rfft(v_r, M_fft)
    full = np.fft.irfft(Q * V, M_fft)
    return full[n-1:2*n-1]


def simplex_project(z):
    """Project z onto {a >= 0, sum a = 1} in O(n log n)."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1)
    rho = np.nonzero(cond > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - tau, 0)


def solve_qp_hankel(q, n, max_iters=500, step=0.49, tol=1e-10, init=None):
    """Solve min a^T M(q) a s.t. a >= 0, sum a = 1 using FISTA.
    M(q)(i,j) = q[i+j]."""
    if init is not None:
        a = init.copy()
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = float(best_a @ hankel_mv(q, best_a))
    
    for it in range(max_iters):
        # FISTA momentum
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + (t - 1.0) / t_new * (a - a_prev)
            t = t_new
        else:
            y = a
        
        # Gradient step
        grad = 2.0 * hankel_mv(q, y)
        a_candidate = y - step * grad
        
        # Project onto simplex
        a_new = simplex_project(a_candidate)
        
        # Track best
        val = float(a_new @ hankel_mv(q, a_new))
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        # Convergence check
        if np.max(np.abs(a_new - a)) < tol:
            break
        
        a_prev = a
        a = a_new
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000):
    """Run L-BFGS on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxiter * 25, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def alternating_minimization(a_init, deadline, max_outer=200, qp_iters=300):
    """
    Alternating minimization:
    1. Compute c = a*a
    2. Update q from c (peak distribution)
    3. Solve QP: min a'^T M(q) a' s.t. a' >= 0, sum a' = 1
    4. Set a = a'
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    print(f"    AM start: score={best_sc:.8f}")
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Update q: use softmax with adaptive temperature
        # Start with moderate beta, increase over iterations
        progress = outer / max(1, max_outer)
        q_beta = 50.0 + 200.0 * progress  # Increase sharpness over time
        q_scaled = q_beta * c / (c_max + 1e-30)
        q_scaled = np.clip(q_scaled, -500, 500)
        q_exp = np.exp(q_scaled - q_scaled.max())
        q = q_exp / q_exp.sum()
        
        # Solve QP
        a_new = solve_qp_hankel(q, n, max_iters=qp_iters, init=a)
        
        # Evaluate
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if outer % 20 == 0:
            print(f"    AM iter {outer}: score={sc:.8f}, best={best_sc:.8f}")
    
    return best_a, best_sc
```

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Novel: alternating minimization with Hankel-QP (exploiting M(i,j)=q_{i+j} structure),
exact subgradient polishing, aggressive LSE-beta L-BFGS, and multi-length search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear self-convolution of a using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def hankel_mv(q, v):
    """Compute M(q) @ v where M(i,j) = q_{i+j}. q length 2n-1, v length n. O(n log n)."""
    n = len(v)
    v_r = v[::-1]
    M_fft = 1 << (2 * n - 1 + n - 1).bit_length()
    Q = np.fft.rfft(q, M_fft)
    V = np.fft.rfft(v_r, M_fft)
    full = np.fft.irfft(Q * V, M_fft)
    return full[n - 1:2 * n - 1]


def simplex_project(z):
    """Project onto {a >= 0, sum a = 1} in O(n log n)."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1)
    rho_arr = np.nonzero(cond > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = int(rho_arr[-1])
    tau = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - tau, 0)


def solve_qp_hankel(q, n, max_iters=400, step=0.49, tol=1e-11, init=None):
    """FISTA for min a^T M(q) a s.t. a>=0, sum a=1, where M(i,j)=q[i+j]."""
    if init is not None:
        a = init.copy()
        S = a.sum()
        if S > 1e-12:
            a = a / S
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n

    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = float(best_a @ hankel_mv(q, best_a))

    for it in range(max_iters):
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + (t - 1.0) / t_new * (a - a_prev)
            t = t_new
        else:
            y = a

        grad = 2.0 * hankel_mv(q, y)
        a_candidate = y - step * grad
        a_new = simplex_project(a_candidate)

        val = float(a_new @ hankel_mv(q, a_new))
        if val < best_val:
            best_val = val
            best_a = a_new.copy()

        if np.max(np.abs(a_new - a)) < tol:
            break
        a_prev = a
        a = a_new

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000):
    """L-BFGS on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def alternating_minimization(a_init, deadline, max_outer=300, qp_iters=300, blend=True):
    """

    a = a_init.copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    print(f"    AM start: score={best_sc:.8f}")

    stall_count = 0
    stall_threshold = 30

        c = fast_conv(a)
        c_max = float(c.max())

        q_beta = 20.0 + 500.0 * progress ** 1.5

        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:

            best_a = a_new.copy()
            stall_count = 0

        a = a_new

            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall_count = 0

def subgradient_polish(a, n_steps=5000, deadline=None):
    """Polish using exact subgradient of max(self-conv). O(n log n) per step."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n, compute_score(np.ones(n) / n)
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.08

    for t in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Subgradient: 2 * a[k_star - j] for valid j
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        sg[j_start:j_end + 1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        a_new = a - eta * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            eta = min(eta * 1.02, 0.15)
        else:
            eta *= 0.85

        a = a_new

    return best_a, best_sc


def fienup_flat(a, n_iters=300, alpha_init=0.6, alpha_end=0.05):
    """Fienup iterative with annealed flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_end + (alpha_init - alpha_end) * (1.0 - progress) ** 2

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        ratio = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * (ratio ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
