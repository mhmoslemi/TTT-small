def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_objective_and_grad(a, beta):
    """Compute LSE objective and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    # Normalize: we want to minimize 2n * max(c) / S^2
    # LSE approximation: 2n * (max(c) + log(sum(exp(beta*(c-cmax))))/beta) / S^2
    # But since we're on the simplex (S=1), this is just 2n * LSE_beta(c)
    # Actually, since the objective is scale-invariant, let's just work with the raw LSE.
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    LSE_val = cmax + np.log(sum_e) / beta
    # Objective: 2n * LSE_val / S^2
    obj = 2.0 * n * LSE_val / (S * S)
    # Gradient: d(obj)/da = 2n/S^2 * d(LSE)/da - 4n*LSE_val/S^3 * d(S)/da
    # d(LSE)/da_i = sum_k w_k * db_k/da_i = 2 * g_i (from our derivation)
    g = fast_cross_corr(a, w)
    dLSE_da = 2.0 * g
    dS_da = np.ones(n)
    grad = (2.0 * n / (S * S)) * dLSE_da - (4.0 * n * LSE_val / (S * S * S)) * dS_da
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(obj), grad


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)
