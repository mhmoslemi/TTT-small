def lse_grad(a, beta):

    c = fast_conv(a)
    c_max = float(c.max())

    max_s = float(scaled.max())

    exp_s = np.exp(shifted)
    log_sum = max_s + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    
    # Gradient of L w.r.t. a_i: sum_k p_k * 2*a_{k-i} (with proper indexing)
    # L = sum_k b_k (in the smoothed sense)

def blended_loss_grad(a, beta, quantile_level=0.95):
    """Blended loss: LSE for peaks + quantile penalty for tail."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    
    # LSE part
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())

    log_sum = max_s + float(np.log(exp_s.sum()))
    LSE = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    
    # Quantile part: penalty for values above the quantile
    c_sorted = np.sort(c)
    q_idx = int(quantile_level * len(c))
    q_val = c_sorted[q_idx]
    above = c[c > q_val]
    if len(above) > 0:

    else:
        q_penalty = 0.0
    
    # Combined
    w_lse = 0.7
    w_q = 0.3
    loss = w_lse * LSE + w_q * q_penalty
    
    # Gradient
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    
    # LSE gradient
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_lse = 2.0 * conv_res[n-1:2*n-1]
    
    # Quantile gradient (subgradient)
    above_mask = c > q_val
    grad_q = np.zeros(n)
    # For each k where c_k > q_val, add 2*a_{k-i} to grad_i

    for k in above_indices[:100]:  # Limit to top 100 for speed

    grad_q /= len(c)
    
    grad = w_lse * grad_lse + w_q * grad_q

    a = np.maximum(a, 0.0)
    a = a / max(a.sum(), 1e-12)
    
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_loss = loss_grad_func(a)[0]
    
    for t in range(1, max_iters + 1):
        loss, grad = loss_grad_func(a)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)

                best_a = a.copy()

def active_set_equalize(a0, n_peaks=50, n_iters=500):
    """Equalize the top peaks by solving a linear system."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Build linear system: for each peak k, sum_j a_j * a_{k-j} = target
        # This is nonlinear in a, but we can linearize:
        # Let a = a0 + delta, then b_k(a0+delta) ≈ b_k(a0) + 2*sum_j a0_{k-j}*delta_j
        # We want b_k(a0+delta) = target for all k in top_idx
        # So: 2*sum_j a0_{k-j}*delta_j = target - b_k(a0)
        
        # Set target to the mean of the top peaks
        target = float(c[top_idx].mean())
        
        # Build the linear system
        m = len(top_idx)
        A_mat = np.zeros((m, n))
        b_vec = np.zeros(m)
        
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:

                b_vec[i] = target - c[k]
        
        # Solve least squares
        try:
            delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except:
            break
        
        # Limit step size
        delta_norm = np.linalg.norm(delta)
        if delta_norm > 0:
            delta = delta * min(1.0, 0.1 / delta_norm)
        
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def coordinate_descent_peaks(a0, n_rounds=100, n_peaks=20):
    """Coordinate descent: for each peak, find the coordinate that most reduces it."""
    n = len(a0)
    a = a0.astype(np.float64).copy()

    S = a.sum()

        # For the global max peak, find which coordinate to reduce
        k_max = int(np.argmax(c))
        j_start = max(0, k_max - n + 1)
        j_end = min(k_max, n - 1)
        
        if j_end >= j_start:
            # Contribution of a_j to c_{k_max} is a_j * a_{k_max - j}
            seg = a[j_start:j_end+1]

        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c))


def power_mean_loss_grad(a, m):
    """Power mean of convolution and gradient. Assumes sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    K = len(c)
    
    if m == np.inf:
        c_max = float(c.max())
        idx = int(np.argmax(c))
        p = np.zeros(K)
        p[idx] = 1.0
        return c_max, _conv_with_weights(p, a) * 2.0
    
    c_safe = np.maximum(c, 1e-30)
    w = c_safe ** (m - 1)
    w_sum = float(w.sum())
    if w_sum < 1e-30:
        return 0.0, np.zeros(n)
    
    pm = (w_sum / K) ** (1.0 / m)
    w_norm = w / K
    
    grad = 2.0 * (m / K) * (pm ** (1.0 - m)) * _conv_with_weights(w_norm, a)
    return pm, grad


def _conv_with_weights(w, a):
    """Compute sum_k w_k * a_{k-i} for all i via FFT."""
    n = len(a)
    rev_a = a[::-1]
    M2 = len(w) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[:M2]
    return conv_res[n - 1:2 * n - 1]


def lse_loss_grad(a, beta):
    """LSE smoothed max and gradient. Assumes sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    shifted = scaled - max_s
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    L = c_max + float(np.log(exp_s.sum())) / beta
    
    grad = 2.0 * _conv_with_weights(p, a)
    return L, grad


def adam_optimize(a0, loss_grad_func, max_iters, lr_init, lr_final, momentum_reset_interval=5000):
    """Adam optimizer with cosine annealing and momentum resets."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for t in range(1, max_iters + 1):
        # Cosine annealing
        progress = (t - 1) / max(1, max_iters - 1)
        lr = lr_final + 0.5 * (lr_init - lr_final) * (1.0 + np.cos(np.pi * progress))
        
        loss, grad = loss_grad_func(a)
        
        # Momentum reset
        if (t - 1) % momentum_reset_interval == 0 and t > 1:
            m = np.zeros(n)
            v = np.zeros(n)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def remez_exchange(a0, n_extremes=30, n_iters=200, step_scale=0.05):
    """Remez exchange: equalize top peaks via linearized subproblem."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        # Find extreme points (alternating high/low)
        error = c - c_mean
        sorted_by_abs = np.argsort(np.abs(error))[::-1][:n_extremes]
        sorted_by_abs = np.sort(sorted_by_abs)
        
        # Build linearized system
        m_pts = len(sorted_by_abs)
        A_mat = np.zeros((m_pts, n))
        b_vec = np.zeros(m_pts)
        
        target = float(c[sorted_by_abs].mean())
        for i, k in enumerate(sorted_by_abs):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                A_mat[i, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
                b_vec[i] = target - c[k]
        
        # Solve
        try:
            delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except np.linalg.LinAlgError:
            break
        
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-15:
            break
        delta = delta * min(1.0, step_scale / delta_norm)
        
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


defcoarse_to_fine(a_fine, n_coarse_factor=2):
    """Downsample by averaging pairs."""
    n = len(a_fine)
    n_c = n // n_coarse_factor
    a_coarse = a_fine[:n_c * n_coarse_factor].reshape(n_c, n_coarse_factor).mean(axis=1)
    a_coarse = np.maximum(a_coarse, 0.0)
    S = a_coarse.sum()
    if S < 1e-12:
        return a_coarse
    return a_coarse / S


def upsample(a_coarse, n_fine):
    """Upsample via linear interpolation."""
    n_c = len(a_coarse)
    x_c = np.linspace(0, 1, n_c)
    x_f = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_f, x_c, a_coarse)
    a_fine = np.maximum(a_fine, 0.0)
    S = a_fine.sum()
    if S < 1e-12:
        return a_fine
    return a_fine / S


def multi_resolution_optimize(a0, n_levels=3, iters_per_level=20000):
    """Coarse-to-fine optimization."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / max(a.sum(), 1e-12)
    
    # Coarsen
    a_coarse = a.copy()
    for _ in range(n_levels):
        a_coarse = coarse_to_fine(a_coarse, 2)
    
    # Optimize at coarse level
    for level in range(n_levels, -1, -1):
        target_len = n_levels and len(a0) // (2 ** (n_levels - level)) or len(a0)
        if level == 0:
            target_len = len(a0)
        else:
            target_len = len(a0) // (2 ** (n_levels - level))
        
        if level < n_levels:
            a_coarse = upsample(a_coarse, target_len)
        
        # Optimize
        def pm_grad(x, m=30):
            return power_mean_loss_grad(x, m)
        
        a_coarse = adam_optimize(
            a_coarse, lambda x: pm_grad(x),
            max_iters=iters_per_level, lr_init=0.01, lr_final=0.0001
        )
    
    return a_coarse


def fienup_flat_conv(a0, n_iters=500, alpha=0.3):
    """GS-Fienup targeting flat convolution."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    # Target: flat convolution
    b_target = np.ones(M) * (S * S / (2 * n - 1))
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        # Interpolate magnitude toward target
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

def peak_reduction(a0, n_rounds=200, n_peaks=50, reduction=0.9):
    """Directly reduce elements contributing to peaks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            n_reduce = min(5, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            for li in top_local:
                j = j_start + li
                a[j] *= reduction
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0) / n)
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)
    
    # Gaussian-like
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2) / np.exp(-2.0 * (x - 0.5) ** 2).sum())
    # Bimodal
    inits.append(np.maximum(np.exp(-8*(x-0.3)**2) + np.exp(-8*(x-0.7)**2), 0.01))
    
    for init in inits:
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-12:
            init = init / s
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    Uses power-mean Adam, Remez exchange, multi-resolution, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return False
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Determine n values to try
    if parent is not None:
        n_parent = len(parent)
        n_values = [n_parent]
        # Add nearby values
        for dn in [-1000, -500, 500, 1000, 2000, -2000]:
            nv = n_parent + dn
            if 3000 <= nv <= 15000:
                n_values.append(nv)
    else:
        n_values = [6962, 8000, 10000]
    
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")
    print(f"Will try n values: {n_values}")
    
    # Phase 1: Quick scan over n values with Adam (20% of time)
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Adam scan over n values ( until {phase1_end - t_start:.0f}s) ===")
    
    for n in n_values[:4]:
        if time.monotonic() > phase1_end:
            break
        
        inits = diverse_inits(n, rng, parent)
        for a0 in inits[:3]:
            if time.monotonic() > phase1_end:
                break
            a_opt = adam_optimize(
                a0, lambda x: power_mean_loss_grad(x, 50),
                max_iters=30000, lr_init=0.005, lr_final=0.0001
            )
            sc = compute_score(a_opt)
            update_best(sc, a_opt)
    
    print(f"After Phase 1: best = {best_score:.8f}")
    
    if best_a is None:
        n = n_values[0]
        best_a = np.ones(n) / n
        best_score = compute_score(best_a)
    
    # Phase 2: Refined Adam with multiple m values on best n (30% of time)
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Refined Adam (until {phase2_end - t_start:.0f}s) ===")
    
    n_best = len(best_a)
    for m_val in [20, 50, 100, 200, 500]:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        a_opt = adam_optimize(
            best_a, lambda x: power_mean_loss_grad(x, m_val),
            max_iters=50000, lr_init=0.003, lr_final=0.00005,
            momentum_reset_interval=10000
        )
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Adam m={m_val}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # LSE refinement
    for beta in [500, 2000, 10000, 50000]:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        a_opt = adam_optimize(
            best_a, lambda x: lse_loss_grad(x, beta),
            max_iters=30000, lr_init=0.002, lr_final=0.00005
        )
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Adam LSE beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    print(f"After Phase 2: best = {best_score:.8f}")
    
    # Phase 3: Remez exchange + peak reduction (20% of time)
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Remez + Peak reduction (until {phase3_end - t_start:.0f}s) ===")
    
    for n_ext in [20, 50, 100]:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        a_rem = remez_exchange(best_a, n_extremes=n_ext, n_iters=500, step_scale=0.03)
        sc = compute_score(a_rem)
        update_best(sc, a_rem)
        print(f"  Remez n_ext={n_ext}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    if time.monotonic() < phase3_end:
        t0 = time.monotonic()
        a_pr = peak_reduction(best_a, n_rounds=300, n_peaks=50, reduction=0.92)
        sc = compute_score(a_pr)
        update_best(sc, a_pr)
        print(f"  Peak reduction: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # Phase 4: Fienup + multi-resolution (15% of time)
    phase4_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Fienup + Multi-res (until {phase4_end - t_start:.0f}s) ===")
    
    if time.monotonic() < phase4_end:
        t0 = time.monotonic()
        a_fien = fienup_flat_conv(best_a, n_iters=500, alpha=0.2)
        sc = compute_score(a_fien)
        update_best(sc, a_fien)
        print(f"  Fienup: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    if time.monotonic() < phase4_end:
        t0 = time.monotonic()
        a_mr = multi_resolution_optimize(best_a, n_levels=2, iters_per_level=20000)
        sc = compute_score(a_mr)
        update_best(sc, a_mr)
        print(f"  Multi-res: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    print(f"After Phase 4: best = {best_score:.8f}")
    
    # Phase 5: Final polish - interleaved methods (15% of time)
    print(f"\n=== Phase 5: Final polish (until {deadline - t_start:.0f}s) ===")
    
    polish_rounds = 8
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break
        
        # Adam with increasing m
        m_val = 50 + 50 * pr
        a_opt = adam_optimize(
            best_a, lambda x: power_mean_loss_grad(x, m_val),
            max_iters=20000, lr_init=0.001, lr_final=0.00001
        )
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        
        if time.monotonic() > deadline:
            break
        
        # Quick Remez
        a_rem = remez_exchange(best_a, n_extremes=50, n_iters=100, step_scale=0.02)
        sc = compute_score(a_rem)
        update_best(sc, a_rem)
        
        if time.monotonic() > deadline:
            break
        
        # Small perturbation + reoptimize
        a_pert = best_a + rng.standard_normal(len(best_a)) * 0.01
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / max(a_pert.sum(), 1e-12)
        a_opt = adam_optimize(
            a_pert, lambda x: power_mean_loss_grad(x, 100),
            max_iters=10000, lr_init=0.001, lr_final=0.00005
        )
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        
        print(f"  Polish {pr}: best = {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")
    
    # Final: L-BFGS on the best found
    if time.monotonic() < deadline:
        print("\n=== Final L-BFGS polish ===")
        n_fin = len(best_a)
        best_a_fin = np.clip(best_a, 0.0, 1000.0).copy()
        S_fin = best_a_fin.sum()
        if S_fin > 1e-6:
            best_a_fin = best_a_fin / S_fin
        
        bounds = [(0.0, 1000.0)] * n_fin
        
        def lbfgs_obj(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-6:
                return 1e10, np.zeros(n_fin)
            x = x / sx
            sc, g = lse_loss_grad(x, 10000)
            return sc, g
        
        try:
            res = minimize(lbfgs_obj, best_a_fin, method='L-BFGS-B', jac=True,
                          bounds=bounds, options={'maxiter': 5000, 'maxfun': 20000,
                                                  'ftol': 1e-18, 'gtol': 1e-15})
            a_fin = np.maximum(res.x, 0.0)
            S_fin = a_fin.sum()
            if S_fin > 1e-6:
                a_fin = a_fin / S_fin
                sc = compute_score(a_fin)
                update_best(sc, a_fin)
        except Exception as e:
            print(f"  L-BFGS error: {e}")
    
    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        n_out = len(best_a)
        out = [0.5] * n_out
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
