"""
Optimized LSE-LBFGS with peak shaving for minimizing self-convolution max ratio.
Key improvements:
1. Very high beta LSE (up to 500000) for tight max approximation
2. Peak shaving: direct coordinate descent on top-contributing elements
3. Progressive beta annealing with restarts
4. Directed line search targeting specific peaks
5. Multiple diverse initializations and perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    shifted = scaled - max_s
    exp_s = np.exp(shifted)
    log_sum = max_s + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=100000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    return np.clip(res.x, 0.0, 1000.0)


def peak_shave(a, n_rounds=200, top_k=10, decay=0.95):
    """Directly shave peaks by reducing top-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for ki in peak_idx:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[ki - j_end:ki - j_start + 1][::-1]
            contrib = seg * partner
            # Reduce top contributor
            top_local = np.argmax(contrib)
            j = j_start + top_local
            old_val = a[j]
            a[j] *= decay
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                a[j] = old_val
                continue
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
                improved = True
            else:
                a[j] = old_val
                a = a / S
        
        if not improved:
            break
    
    return best


def directed_line_search(a, beta, n_steps=100):
    """Line search in the direction that reduces the main peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

        a = a_new

    n = len(a)
    c = fast_conv(a)

    if S < 1e-6:

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L * S - 2.0 * L) / S3
    return score, grad


def lbfgs_func(x):
    """Wrapper for scipy minimize using module-level beta."""
    return lse_obj_grad(x, _current_beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=50000):
    """Run L-BFGS-B on LSE objective with given beta."""
    global _current_beta
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    _current_beta = float(beta)
    res = minimize(
        lbfgs_func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    out = np.clip(res.x, 0.0, 1000.0)
    if out.sum() < 1e-6:
        out = a0.copy()
    return out


def peak_shave(a, n_rounds=150, top_k=8, decay=0.95):
    """Directly shave peaks by reducing top-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:top_k]
        improved = False

        for ki in peak_idx:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[ki - j_end:ki - j_start + 1][::-1]
            contrib = seg * partner
            top_local = int(np.argmax(contrib))
            j = j_start + top_local
            old_val = a[j]
            a[j] *= decay
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                a[j] = old_val
                continue
            a = a / S
            sc = compute_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = a.copy()
                improved = True
            else:
                a[j] = old_val
                a = a / S

        if not improved:
            break

    return best


def topk_subgrad_descent(a, k, n_iters=2000, step_init=0.02, decay_rate=0.9998):
    """Subgradient descent on sum of top-k convolution values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:k]
        S = float(a.sum())

        # Subgradient of sum of top-k c values
        sg = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[ki - j_end:ki - j_start + 1][::-1]

        top_sum = float(c[top_idx].sum())
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg * S - 2.0 * top_sum) / S3

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        a -= step * grad / gn
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay_rate

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best = a.copy()
    return best


def directed_line_search(a, n_steps=100):
    """Line search in direction that reduces the global peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)

    c = fast_conv(a)
    k_star = int(np.argmax(c))

    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    direction = np.zeros(n)
    direction[j_start:j_end + 1] = -2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]

    # Also add a small uniform positive component to counteract sum decrease
    direction += 0.1 * np.ones(n)

    d_norm = float(np.linalg.norm(direction))
    if d_norm < 1e-15:
        return a
    direction /= d_norm

    step = 0.05
    current = a.copy()
    current_sc = best_sc

    for _ in range(n_steps):
        trial = current + step * direction
        trial = np.maximum(trial, 0.0)
        S_trial = float(trial.sum())
        if S_trial < 1e-12:
            step *= 0.5
            continue
        trial = trial / S_trial
        trial_sc = compute_score(trial)
        if trial_sc < current_sc - 1e-14:
            current = trial
            current_sc = trial_sc
            if current_sc < best_sc:
                best_sc = current_sc
                best = current.copy()
            step *= 1.05
        else:
            step *= 0.5
        if step < 1e-12:
            break

    return best


def random_perturb(a, rng, strength=0.01):
    """Add Gaussian perturbation and renormalize."""
    a = a.copy()
    a += rng.standard_normal(len(a)) * strength
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    return a / S


def gs_flat_target(a, n_iters=200, alpha_init=0.4, alpha_final=0.05):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)

    b_target = np.zeros(M)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** alpha
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    return best


def coordinate_descent_peak(a, n_sweeps=5, top_k=5):
    """For top peaks, do 1D optimization of each contributing element."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_sweeps):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:top_k]

        for ki in peak_idx:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end < j_start:
                continue

            # For each element contributing to this peak, try reducing it
 seg = a[j_start:j_end + 1]
            partner = a[ki - j_end:ki - j_start + 1][::-1]
            contrib = seg * partner
            # Try reducing each element by 2%
            for li in range(len(contrib)):
                if contrib[li] < 0.01 * np.max(contrib):
                    continue
                j = j_start + li
                old = a[j]
                a[j] *= 0.98
                S = a.sum()
                if S < 1e-12:
                    a[j] = old
                    continue
                a = a / S
                sc = compute_score(a)
                if sc >= best_sc:
                    a[j] = old
                    a = a / S
                else:
                    best_sc = sc
                    best = a.copy()

    return best


def propose_candidate(budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/(sum(a))^2.
    Uses progressive LSE-LBFGS, peak shaving, top-k subgradient, and line search.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_a = parent.copy()
    else:
        n = 7000
        best_a = np.ones(n) * 0.5

    best_sc = compute_score(best_a)
    print(f"Initial: n={n}, score={best_sc:.10f}")
    print(f"Budget: {total}s, deadline at t+{deadline - t_start:.0f}s")

    # Phase 1: Progressive LSE (35% of time)
    t_p1 = t_start + 0.35 * (total - 30.0)
    print(f"\n=== Phase 1: Progressive LSE-LBFGS ===")

    betas_p1 = [500, 1000, 2000, 5000, 10000, 20000, 40000, 80000]
    for beta in betas_p1:
        if time.monotonic() > t_p1:
            break
        t0 = time.monotonic()
        a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=40000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  beta={beta:>6d}: NEW BEST {sc:.10f} ({time.monotonic()-t0:.1f}s)")
        else:
            print(f"  beta={beta:>6d}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Peak shave
    a_sh = peak_shave(best_a, n_rounds=100, top_k=10, decay=0.95)
    sc = compute_score(a_sh)
    if sc < best_sc:
        best_sc = sc
        best_a = a_sh.copy()
        print(f"  Peak shave: NEW BEST {sc:.10f}")

    print(f"  Phase 1 end: best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Alternating LSE + peak shave + subgradient (30%)
    t_p2 = t_start + 0.65 * (total - 30.0)
    print(f"\n=== Phase 2: Alternating refinement ===")

    for rnd in range(30):
        if time.monotonic() > t_p2:
            break

        # High-beta LSE
        beta = 80000 + 20000 * rnd
        a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=25000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

        # Peak shave
        a_sh = peak_shave(best_a, n_rounds=50, top_k=10, decay=0.94)
        sc = compute_score(a_sh)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sh.copy()

        # Top-k subgradient
        a_sg = topk_subgrad_descent(best_a, k=5, n_iters=500, step_init=0.01)
        sc = compute_score(a_sg)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"  Round {rnd}: subgrad NEW BEST {sc:.10f}")

        if rnd % 5 == 4:
            print(f"  Round {rnd}: best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    print(f"  Phase 2 end: best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: Diversification (20%)
    t_p3 = t_start + 0.85 * (total - 30.0)
    print(f"\n=== Phase 3: Diversification ===")

    for trial in range(40):
        if time.monotonic() > t_p3:
            break
        # Various perturbation strategies
        if trial % 4 == 0:
            a_p = random_perturb(best_a, rng, 0.005)
        elif trial % 4 == 1:
            a_p = random_perturb(best_a, rng, 0.01)
        elif trial % 4 == 2:
            # Structured perturbation: smooth noise
            a_p = best_a.copy()
            noise = rng.standard_normal(n)
            # Low-pass filter the noise
            N = np.fft.rfft(noise)
            N[n // 4:] = 0
            noise = np.fft.irfft(N, n)
            a_p = best_a + 0.01 * noise
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
        else:
            #Peak-targeted perturbation: increase elements that are NOT at peaks
            c = fast_conv(best_a)
            c_max = float(c.max())
            a_p = best_a.copy()
            # Increase elements where they contribute less to peaks
            low_contrib = np.random.default_rng(trial).uniform(0, 0.005, n)
            a_p += low_contrib
            a_p = a_p / a_p.sum()

        # Quick LSE
        a_opt = run_lbfgs(a_p, 100000, maxiter=3000, maxfun=15000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  Trial {trial}: NEW BEST {sc:.10f} (pert={trial%4})")

        # Quick peak shave on result
        a_sh = peak_shave(a_opt, n_rounds=30, top_k=5, decay=0.96)
        sc = compute_score(a_sh)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sh.copy()
            print(f"  Trial {trial}: shave NEW BEST {sc:.10f}")

    print(f"  Phase 3 end: best={best_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Final polish (10%)
    print(f"\n=== Phase 4: Final Polish ===")

    for beta in [200000, 300000, 500000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=50000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  beta={beta:>6d}: NEW BEST {sc:.10f} ({time.monotonic()-t0:.1f}s)")
        else:
            print(f"  beta={beta:>6d}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Final peak shave (aggressive)
    if time.monotonic() < deadline - 10:
        a_sh = peak_shave(best_a, n_rounds=300, top_k=20, decay=0.93)
        sc = compute_score(a_sh)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sh.copy()
            print(f"  Final shave: NEW BEST {sc:.10f}")

    # Final directed line search
    if time.monotonic() < deadline - 5:
        a_dls = directed_line_search(best_a, n_steps=200)
        sc = compute_score(a_dls)
        if sc < best_sc:
            best_sc = sc
            best_a = a_dls.copy()
            print(f"  Final DLS: NEW BEST {sc:.10f}")

    # One more LSE after all refinements
    if time.monotonic() < deadline - 5:
        a_opt = run_lbfgs(best_a, 200000, maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  Ultra final LSE: NEW BEST {sc:.10f}")

    # Build output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_sc:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
