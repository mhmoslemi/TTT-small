def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def mirror_descent(a, n_iters=5000, beta_soft=50.0, n_peaks=20,

    Multiplicative weight mirror descent with softmax-weighted subgradient.
    Uses exp(-eta * g) updates which naturally preserve non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 1e-15)
    S = float(a.sum())
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    eta = eta_init

    for it in range(n_iters):
        # Compute convolution
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())

            best_a = a.copy()

        w_sum = float(weights.sum())
        if w_sum < 1e-30:

            max_idx = np.argmax(c)
            p[max_idx] = 1.0

"""

L-BFGS, and multi-scale spectral shaping.

import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_target(a, n_iters=300, alpha=0.5, target_flat=True):
    """
    Fienup with explicit target: push |FFT(a)|^2 toward a flat target.
    target = S^2 / (2n-1) for all frequencies (in the appropriate sense).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)

        # Target: flat amplitude = S / sqrt(M_fft) (roughly)
        # For flat autocorrelation, we want |A|^2 to be proportional to a sinc-like pattern
        # But as an approximation, target = mean(amp^2)
        target = float(np.mean(amp[1:])) if len(amp) > 1 else amp[0]
        target = max(target, 1e-20)

        # Geometric interpolation toward target
        amp_safe = np.maximum(amp, 1e-20)
        ratio = target / amp_safe
        amp_new = amp_safe * ratio**alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def log_mirror_descent(a, n_iters=5000, beta=100.0, 
                       eta_init=0.01, eta_end=0.001):
    """
    Mirror descent in log-space: x_i = log(a_i + eps).
    Update: x_i -= eta * grad_i, then a_i = exp(x_i).
    This is equivalent to multiplicative updates: a_i *= exp(-eta * grad_i).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    S = float(a.sum())
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        eta = eta_init * (eta_end / eta_init)**progress

        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Softmax subgradient
        scaled = beta * (c - M_val)
        scaled = np.clip(scaled, -40, 0)
        w = np.exp(scaled)
        ws = float(w.sum())
        if ws < 1e-30:
            p = np.zeros(len(c))
            p[int(np.argmax(c))] = 1.0
        else:
            p = w / ws

        # Gradient of max(c) w.r.t. a:
        # d c_k / d a_i involves a[k-i] terms
        # Approximate: g_i = 2 * sum_k p_k * a[k-i] (for 0 <= k-i < n)
        rev_a = a[::-1]
        M2 = len(p) + n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        PP = np.fft.rfft(p, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_res = np.fft.irfft(PP * RA, M2_fft)[n-1:2*n-1]
        
        g = 2.0 * conv_res
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (g / S2 - 2.0 * M_val / S3)

        # Multiplicative update
        step = np.clip(-eta * grad, -2.0, 2.0)
        a = a * np.exp(step)
        a = np.maximum(a, 1e-10)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def quadratic_subgradient(a, n_iters=3000, n_peaks=10, eta_init=0.01):
    """
    Subgradient method targeting top peaks with quadratic step size decay.
    Uses hard subgradient (only top-n peaks).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    S = float(a.sum())
    a = a / S

    M_fft = 1 << (2*n-1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A*A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S*S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[js:je+1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)

        # Step size: eta_t = C / sqrt(t+1)
        eta = eta_init / np.sqrt(it + 1)
        eta = min(eta, max(sc - sc_lb, 0.001) / (float(np.linalg.norm(grad))**2 + 1e-30))
        eta = min(eta, 0.05)

        a -= eta * grad
        a = np.maximum(a, 1e-10)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def lse_lbfgs(a0, beta, maxiter=10000):
    """L-BFGS on LSE-smoothed objective."""
    n = len(a0)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    
    def obj_and_grad(x):
        A = np.fft.rfft(x, M_fft)
        c = np.fft.irfft(A*A, M_fft)[:M]
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -50, 100)
        exp_s = np.exp(scaled)
        exp_sum = float(exp_s.sum())
        if exp_sum < 1e-30:
            return 1e10, np.zeros(n)
        L = c_max + float(np.log(exp_sum)) / beta
        score = 2.0 * n * L / (S*S)
        
        p = exp_s / exp_sum
        rev_x = x[::-1]
        M2 = M + n - 1
        M2_fft = 1 << (M2-1).bit_length()
        PP = np.fft.rfft(p, M2_fft)
        RX = np.fft.rfft(rev_x, M2_fft)
        conv_res = np.fft.irfft(PP * RX, M2_fft)[n-1:2*n-1]
        g_L = 2.0 * conv_res
        
        S2 = S*S
        S3 = S2*S
        grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
        return score, grad

    a0 = np.clip(a0.astype(np.float64).copy(), 1e-10, 1000.0)
    a0 = a0 / a0.sum()
    bounds = [(1e-10, 1000.0)] * n
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'ftol': 1e-20, 'gtol': 1e-16,
                                           'maxiter': maxiter, 'maxfun': maxiter*20})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out

"""
Optimized step function construction using momentum gradient descent on LSE-smoothed
objective, Fienup spectral shaping with adaptive targets, and L-BFGS polish.
Key improvements over previous: (1) momentum-based optimization for faster convergence,
(2) adaptive LSE temperature that increases over time, (3) active-set peak targeting,

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_adaptive(a, n_iters=400, alpha_start=0.1, alpha_end=0.9):
    """
    Fienup iteration with adaptive alpha that increases over time.
    Uses geometric shaping toward the median amplitude.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_top_clip(a, n_iters=400, frac=0.3, gamma=0.6):
    """
    Fienup that only clips the top fraction of amplitudes.
    More conservative, preserves low-frequency structure.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)

        nclip = max(1, int(frac * len(amp)))
        top_indices = np.argsort(amp)[::-1][:nclip]
        target = float(np.median(amp))
        target = max(target, 1e-20)

        amp_new = amp.copy()
        for idx in top_indices:
            if amp[idx] > target:
                amp_new[idx] = target + (amp[idx] - target) * (1 - gamma)

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def momentum_lse(a0, beta_start=50.0, beta_end=5000.0, n_iters=5000,
                 lr_start=0.01, lr_end=0.001, momentum=0.9):
    """
    Momentum gradient descent on LSE-smoothed objective with adaptive beta.
    Key: beta increases over time, making the approximation sharper.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 1e-10)
    S = float(a.sum())
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # momentum velocity

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Adaptive beta: exponential increase
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        mu = momentum * (0.999 ** it)  # slight decay of momentum

        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        if S < 1e-12:
            break

        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -40, 50)
        exp_s = np.exp(scaled)
        exp_sum = float(exp_s.sum())
        if exp_sum < 1e-30:
            break
        L = c_max + float(np.log(exp_sum)) / beta
        sc = 2.0 * n * L / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Gradient
        p = exp_s / exp_sum
        rev_a = a[::-1]
        PP = np.fft.rfft(p, M2_fft)
        RA = np.fft.rfft(rev_a, M2_fft)
        conv_res = np.fft.irfft(PP * RA, M2_fft)[n - 1:2 * n - 1]
        g_L = 2.0 * conv_res

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)

        # Momentum update
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 1e-10)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a, best_sc


def peak_reactive_descent(a, n_iters=3000, n_peaks=8, lr=0.005):
    """
    Subgradient descent that reacts to the current peak configuration.
    Uses top-n peaks with uniform weights.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    S = float(a.sum())
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Adaptive peak count: more peaks early, fewer later
        adaptive_peaks = max(3, int(n_peaks * (1.0 - 0.5 * it / n_iters)))
        if adaptive_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -adaptive_peaks)[-adaptive_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Polyak step size
        gap = max(sc - sc_lb, 1e-6)
        gn2 = float(np.dot(grad, grad))
        eta = min(lr, gap / (gn2 + 1e-30) * 0.5)
        eta = max(eta, 1e-7)

        a -= eta * grad
        a = np.maximum(a, 1e-10)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def lse_lbfgs(a0, beta, maxiter=5000):
    """L-BFGS on LSE-smoothed objective with high beta."""
    n = len(a0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    def obj_grad(x):
        A = np.fft.rfft(x, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -50, 80)
        exp_s = np.exp(scaled)
        exp_sum = float(exp_s.sum())
        if exp_sum < 1e-30:
            return 1e10, np.zeros(n)
        L = c_max + float(np.log(exp_sum)) / beta
        score = 2.0 * n * L / (S * S)

        p = exp_s / exp_sum
        rev_x = x[::-1]
        PP = np.fft.rfft(p, M2_fft)
        RX = np.fft.rfft(rev_x, M2_fft)
        conv_res = np.fft.irfft(PP * RX, M2_fft)[n - 1:2 * n - 1]
        g_L = 2.0 * conv_res

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
        return score, grad

    a0 = np.clip(a0.astype(np.float64).copy(), 1e-10, 1000.0)
    a0 = a0 / a0.sum()
    bounds = [(1e-10, 1000.0)] * n

    res = minimize(obj_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16,
                            'maxiter': maxiter, 'maxfun': maxiter * 20})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    return compute_score(a_out), a_out


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2) + 0.05)
    inits.append(np.maximum(0.2 + 0.6 * np.cos(np.pi * (2 * x - 1)), 0.01))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))

    # Low-frequency random
    freqs = rng.integers(1, 5, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    sig = np.ones(n) * 0.5
    for f, ph in zip(freqs, phases):
        sig += 0.25 * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    # Sinc-like (concentrated energy at DC)
    inits.append(np.sinc(3.0 * (x - 0.5)) ** 2 + 0.02)

    if parent is not None and len(parent) == n:
        inits.append(np.maximum(parent.astype(np.float64).copy(), 0.0))

    return inits
