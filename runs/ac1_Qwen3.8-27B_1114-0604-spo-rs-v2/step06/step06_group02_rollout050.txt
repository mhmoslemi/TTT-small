    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    top_k = min(n_subgrads, len(c))
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Compute subgradient as sum of ∇c[k] for top peaks
    # ∇c[k][j] = a[k-j] for valid j (simplified, ignoring self-terms)
    # This is a "sliding sum" of a
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            g[j_start:j_end+1] += a[k - (np.arange(j_start, j_end+1))]

    # Normalized gradient step
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return a
    
    a_new = a - eta * g / g_norm * c[top_idx[0]]  # Scale by peak value
    a_new = np.maximum(a_new, 0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a
    a_new = a_new / S_new
    return a_new

"""
Optimized search minimizing 2*n*max(conv(a,a))/sum(a)^2 using large-n scaling,
LSE-smoothed L-BFGS, subgradient descent for peak targeting, and coordinate
descent polishing. Exploits the fact that larger n gives lower scores while
using aggressive iterative refinement with adaptive learning rates.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fconv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def fscore(a):
    n = len(a)
    c = fconv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_og(a, beta):
    n = len(a)
    c = fconv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    cm = float(c.max())
    sc = beta * (c - cm)
    sc = np.clip(sc, -500, 500)
    es = np.exp(sc)
    L = cm + float(np.log(es.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = es / es.sum()
    ar = a[::-1]
    Mf = 1 << (2 * n - 1).bit_length()
    PA = np.fft.rfft(p, Mf)
    AA = np.fft.rfft(ar, Mf)
    h = np.fft.irfft(AA * PA, Mf)[:2*n-1][n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def rlbfgs(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_og, a0, args=(beta,), method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'ftol': 1e-20, 'gtol': 1e-14,
                                           'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40})
    ao = np.clip(res.x, 0.0, 1000.0)
    S = ao.sum()
    if S < 1e-6:
        ao = a0
    else:
        ao = ao / S
    return fscore(ao), ao


def sg_desc(a, n_iters, eta0=0.3, deadline=None):
    """Subgradient descent targeting the peak directly."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    ba = a.copy()
    bs = fscore(a)
    eta = eta0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fconv(a)
        S = float(a.sum())
        cm = float(c.max())
        sc = 2.0 * n * cm / (S * S)
        if sc < bs:
            bs = sc
            ba = a.copy()

        k = int(np.argmax(c))
        js0 = max(0, k - n + 1)
        js1 = min(k, n - 1)
        if js1 < js0:
            break
        js = np.arange(js0, js1 + 1)
        g = np.zeros(n)
        g[js] = a[k - js]
        sm = (2 * js == k)
        if sm.any():
            g[js[sm]] += a[js[sm]]
        gn = float(np.sqrt(np.sum(g * g)))
        if gn < 1e-15:
            break
        a_new = a - eta * g / gn * (cm / S)
        a_new = np.maximum(a_new, 0.0)
        Sn = a_new.sum()
        if Sn < 1e-12:
            break
        a_new = a_new / Sn
        scn = fscore(a_new)
        if scn < sc:
            a = a_new
            eta = min(eta * 1.03, 3.0)
        else:
            eta *= 0.75
    return ba


def cd_peak(a, nsw=8, deadline=None):
    """Coordinate descent on top-contributing elements of the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    ba = a.copy()
    bs = fscore(a)
    factors = [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2]

    for sw in range(nsw):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fconv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < bs:
            bs = sc
            ba = a.copy()
        k = int(np.argmax(c))
        js0 = max(0, k - n + 1)
        js1 = min(k, n - 1)
        if js1 < js0:
            break
        js = np.arange(js0, js1 + 1)
        ct = a[js] * a[k - js]
        ord_ = np.argsort(ct)[::-1]
        for idx in ord_[:80]:
            j = int(js[idx])
            ao = a[j]
            if ao < 1e-14:
                continue
            bv = ao
            bl = sc
            for f in factors:
                a[j] = ao * f
                St = a.sum()
                if St < 1e-12:
                    a[j] = ao
                    continue
                st = 2.0 * n * float(fconv(a / St).max()) / (St * St) * (St / a.sum()) ** 2
                # Simpler: just recompute
                an = a / St
                st = fscore(an)
                if st < bl:
                    bl = st
                    bv = ao * f
            a[j] = bv
        S = a.sum()
        a = a / S
    return ba


def interp_len(a, tn):
    n = len(a)
    if n == tn:
        return a.copy()
    if n == 1:
        return np.ones(tn) / tn
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, tn)
    an = np.interp(x1, x0, a)
    an = np.maximum(an, 0.0)
    S = an.sum()
    if S > 1e-12:
        an = an / S
    return an
