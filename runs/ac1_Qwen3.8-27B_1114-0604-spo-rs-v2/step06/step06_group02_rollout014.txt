"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a)^2).

This implementation uses:
  - power-mean (L^q-norm) smooth approximations of the maximum convolution
    coefficient, which are exactly scale-invariant for any q > 1
  - corrected gradients for the linear convolution a*a
  - L-BFGS-B refinement in the nonnegative orthant
  - multiplicative on-line updates for final polishing
  - optional length resampling / perturbation exploration

The final sequence is normalized to sum 1, which preserves the score.
"""

import math
import time
import numpy as np
from scipy.optimize import minimize


def _clean_nonnegative(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    return x


def _normalize(a):
    a = _clean_nonnegative(a)
    s = float(a.sum())
    if s > 1.0e-300:
        return a / s
    n = a.size
    if n <= 0:
        return np.array([0.5], dtype=np.float64)
    return np.ones(n, dtype=np.float64) / float(n)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    M = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, M)
    b = np.fft.irfft(A * A, M)[:L]
    b = np.maximum(b, 0.0)
    return b


def _score_and_b(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf"), np.array([], dtype=np.float64), 0.0
    s = float(a.sum())
    if s <= 1.0e-300:
        return float("inf"), np.array([], dtype=np.float64), 0.0
    b = _fft_convolve(np.maximum(a, 0.0))
    m = float(b.max()) if b.size else 0.0
    sc = float(2.0 * n * m / (s * s))
    return sc, b, m


def _cross_corr(a, w):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    M = 1 << (L - 1).bit_length()
    ar = np.fft.rfft(a[::-1], M)
    wr = np.fft.rfft(w[:L], M)
    out = np.fft.irfft(ar * wr, M)[:L]
    out = np.maximum(out, 0.0)
    return out[n - 1:2 * n - 1]


def _qnorm_B_weights(b, q):
    """
    Given nonnegative b, compute B = ||b||_q and weights
    w_k = b_k^{q-1} / B^{q-1} (with q=inf limit handled by large q).
    The gradient of ||b||_q is sum_k w_k * db_k.
    """
    L = b.size
    if L == 0:
        return 0.0, np.array([], dtype=np.float64)

    maxb = float(b.max())
    if maxb <= 0.0:
        return 0.0, np.zeros(L, dtype=np.float64)

    qf = float(q)
    if qf <= 1.0:
        w = b / maxb
        return maxb, w

    u = b / maxb
    qi = int(qf)

    with np.errstate(over="ignore", under="ignore", invalid="ignore", divide="ignore"):
        uq = np.power(u, qi)

    s = float(np.sum(uq))
    if not np.isfinite(s) or s < 1.0:
        s = 1.0

    logQ = math.log(s) / qf
    B = maxb * math.exp(logQ)

    w = np.zeros(L, dtype=np.float64)
    mask = u > 0.0
    if np.any(mask):
        logu = np.log(u[mask])
        logw = (qf - 1.0) * (logu - logQ)
        logw = np.clip(logw, -745.0, 700.0)
        w[mask] = np.exp(logw)

    return B, w


def _qnorm_score_grad(x, q, anchor):
    """
    Scale-invariant smooth objective:
        F(x) = 2n * ||x*x||_q / (sum(x))^2
    plus a tiny anchor penalty keeping sum(x) near 1.
    Returns (value, gradient).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    n = x.size
    if n == 0:
        return 1.0e100, np.array([], dtype=np.float64)

    s = float(x.sum())
    if s <= 1.0e-300:
        val = 1.0e100
        grad = np.zeros(n, dtype=np.float64)
        if anchor > 0.0:
            d = s - 1.0
            val += anchor * d * d
            grad += 2.0 * anchor * d
        return val, grad

    b = _fft_convolve(x)
    B, w = _qnorm_B_weights(b, int(q))
    if B <= 0.0:
        B = 0.0
        w = np.zeros_like(b)

    H = _cross_corr(x, w)

    ns = float(n)
    inv_s = 1.0 / s
    inv_s2 = inv_s * inv_s
    inv_s3 = inv_s2 * inv_s

    val = 2.0 * ns * B * inv_s2
    grad = 4.0 * ns * (H * inv_s2 - B * inv_s3)

    # Remove the negligible scale direction numerically.
    denom = float(np.dot(x, x))
    if denom > 1.0e-300:
        grad -= (float(np.dot(x, grad)) / denom) * x

    if anchor > 0.0:
        d = s - 1.0
        val += anchor * d * d
        grad += (2.0 * anchor * d) * np.ones(n, dtype=np.float64)

    grad = np.where(np.isfinite(grad), grad, 0.0)
    if not np.isfinite(val):
        val = 1.0e100
    return val, grad


def _run_lbgf_qnorm(a_start, q, maxfun, anchor=1.0e-12):
    a = _normalize(a_start)
    n = a.size
    if n == 0:
        return float("inf"), a

    bounds = [(0.0, 1000.0)] * n
    maxfun = max(1, int(maxfun))
    opts = {
        "maxiter": maxfun,
        "maxfun": maxfun,
        "ftol": 1.0e-12,
        "gtol": 1.0e-10,
        "maxline": 50,
    }

    try:
        res = minimize(
            _qnorm_score_grad,
            a,
            args=(int(q), float(anchor)),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=opts,
        )
        out = np.asarray(res.x, dtype=np.float64)
    except Exception as ex:
        print("  L-BFGS error:", ex)
        out = a

    out = _normalize(out)
    sc, _, _ = _score_and_b(out)
    return sc, out


def _multiplicative_qnorm(a_start, q, max_iter, deadline, eta_init=0.08):
    """
    Multiplicative polishing using gradient of ||b||_q on the simplex.
    """
    a = _normalize(a_start)
    sc, b, m = _score_and_b(a)
    if not np.isfinite(sc):
        return a, float("inf")

    best_a = a.copy()
    best_sc = sc
    eta = float(eta_init)
    n = a.size

    for _ in range(max(0, int(max_iter))):
        if time.monotonic() > deadline:
            break
        if m <= 0.0:
            break

        _, w = _qnorm_B_weights(b, int(q))
        H = _cross_corr(a, w)
        phi = H - float(np.mean(H))
        scale = float(np.max(np.abs(phi)))
        if scale < 1.0e-30:
            break

        improved = False
        for _attempt in range(2):
            step = eta * phi / scale
            step = np.clip(step, -30.0, 30.0)
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 0.0)
            sn = float(a_new.sum())
            if sn <= 1.0e-300:
                eta *= 0.5
                continue
            a_new = a_new / sn

            sc_new, b_new, m_new = _score_and_b(a_new)
            if np.isfinite(sc_new) and sc_new < sc:
                a = a_new
                b = b_new
                m = m_new
                sc = sc_new
                improved = True
                eta = min(eta * 1.1, 0.5)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            else:
                eta *= 0.5

        if not improved:
            eta *= 0.7
            if eta < 1.0e-4:
                break

    return best_a, best_sc


def _resample_sequence(a, target_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    target_n = int(target_n)
    if target_n <= 0:
        target_n = n
    if target_n == n:
        return _normalize(a)
    if n == 1:
        return np.full(target_n, a[0] / float(target_n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    a2 = np.interp(x_new, x_old, np.maximum(a, 0.0))
    return _normalize(a2)


def _get_parent_sequence():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is not None:
            a = _clean_nonnegative(seq)
            if a.size >= 8 and a.sum() > 0.01:
                return a
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a nonnegative coefficient sequence minimizing
    2*n*max(conv(a,a))/sum(a)^2, using scale-invariant smooth objectives,
    L-BFGS, and multiplicative polishing.
    """
    t0 = time.monotonic()
    total = float(budget_s)
    reserved = min(90.0, max(10.0, total * 0.05))
    search_deadline = t0 + total - reserved

    parent = _get_parent_sequence()
    rng = np.random.default_rng(seed)

    if parent is None:
        n0 = 4096
        a0 = np.ones(n0, dtype=np.float64) / float(n0)
    else:
        a0 = parent.copy()
        if a0.size > 12000:
            a0 = _resample_sequence(a0, 12000)

    best_a = _normalize(a0)
    best_sc, _, _ = _score_and_b(best_a)
    print(f"Initial: n={best_a.size}, score={best_sc:.8f}")

    search_time = max(1.0, search_deadline - t0)

    # ------------------------------------------------------------
    # Phase 1: quick multiplicative warm-up from the parent.
    # ------------------------------------------------------------
    p1_end = t0 + 0.08 * search_time
    if time.monotonic() < p1_end:
        a_opt, sc_opt = _multiplicative_qnorm(
            best_a, 64, 1000000, p1_end, eta_init=0.10
        )
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt
            print(f"[mult-64] NEW BEST: {best_sc:.8f}")

    # ------------------------------------------------------------
    # Phase 2: main L-BFGS refinement with increasing q.
    # ------------------------------------------------------------
    p2_end = t0 + 0.55 * search_time
    q_list = [64, 256, 1024, 4096, 16384, 65536]
    weights = [1, 2, 3, 4, 3, 2]
    weight_sum = float(sum(weights))

    a_current = best_a.copy()

    if time.monotonic() < p2_end - 30:
        # Estimate iteration cost for maxfun budgeting.
        t_est = time.monotonic()
        _qnorm_score_grad(a_current, q_list[0], 1.0e-12)
        eval_time = time.monotonic() - t_est
        if eval_time < 5.0e-4:
            eval_time = 5.0e-4
        rate = 0.65 / eval_time
        lbgf_budget_s = max(0.0, p2_end - time.monotonic() - 30.0)
        total_maxfun = int(lbgf_budget_s * rate)
        total_maxfun = max(3000, min(total_maxfun, 320000))

        for idx, q in enumerate(q_list):
            if time.monotonic() >= p2_end - 30:
                break
            base_mf = int(total_maxfun * (weights[idx] / weight_sum))
            rem_mf = int((p2_end - time.monotonic() - 15.0) * rate)
            mf = max(500, min(base_mf, max(500, rem_mf)))
            if mf < 800:
                break

            print(f"  L-BFGS q={q}, maxfun={mf}")
            sc_opt, a_opt = _run_lbgf_qnorm(a_current, q, mf, anchor=1.0e-12)
            if np.isfinite(sc_opt):
                if sc_opt < best_sc:
                    best_sc = sc_opt
                    best_a = a_opt.copy()
                    a_current = a_opt.copy()
                    print(f"  L-BFGS q={q} NEW BEST: {best_sc:.8f}")
                elif sc_opt <= best_sc + 0.02:
                    # Continue from a nearby surrogate basin, but keep best.
                    a_current = a_opt.copy()
                else:
                    a_current = best_a.copy()

    # ------------------------------------------------------------
    # Phase 3: high-q multiplicative polishing.
    # ------------------------------------------------------------
    p3_end = t0 + 0.80 * search_time
    if time.monotonic() < p3_end:
        dl = min(p3_end, time.monotonic() + 0.10 * search_time)
        a_opt, sc_opt = _multiplicative_qnorm(
            best_a, 4096, 1000000, dl, eta_init=0.05
        )
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt
            print(f"[mult-4096] NEW BEST: {best_sc:.8f}")

    if time.monotonic() < p3_end:
        dl = min(p3_end, time.monotonic() + 0.06 * search_time)
        a_opt, sc_opt = _multiplicative_qnorm(
            best_a, 16384, 1000000, dl, eta_init=0.04
        )
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt
            print(f"[mult-16384] NEW BEST: {best_sc:.8f}")

    # ------------------------------------------------------------
    # Phase 4: light exploration: resampling to other lengths and
    # low-frequency perturbations.
    # ------------------------------------------------------------
    p4_end = t0 + 0.93 * search_time
    if time.monotonic() < p4_end - 30 and best_a.size < 12000:
        base_n = int(best_a.size)
        targets = []
        for cand in (
            int(base_n * 1.25),
            int(base_n * 1.5),
            8192,
            10000,
        ):
            cand = int(cand)
            if cand > base_n and cand <= 12000 and cand not in targets:
                targets.append(cand)
        targets = sorted(targets)[:2]

        if targets:

            for n2 in targets:
                if time.monotonic() > p4_end - 20:
                    break
                a2 = _resample_sequence(best_a, n2)
                dl = min(p4_end - 20, time.monotonic() + per_length)
                a_opt, sc_opt = _multiplicative_qnorm(
                    a2, 1024, 1000000, dl, eta_init=0.10
                )
                if np.isfinite(sc_opt) and sc_opt < best_sc:
                    best_sc = sc_opt
                    best_a = a_opt

    if time.monotonic() < p4_end - 20:

        z = np.log(np.maximum(best_a, 1.0e-12))
        z -= float(z.mean())

            ap = np.exp(zp)

            dl = min(p4_end - 15, time.monotonic() + 45.0)
            a_opt, sc_opt = _multiplicative_qnorm(

            )

                print(f"[perturb] NEW BEST: {best_sc:.8f}")

    # ------------------------------------------------------------
    # Final very sharp multiplicative polish.

    final_polish_deadline = search_deadline - 5.0

    # ------------------------------------------------------------

    return out
