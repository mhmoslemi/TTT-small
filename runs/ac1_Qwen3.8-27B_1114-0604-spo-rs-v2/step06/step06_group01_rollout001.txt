"""
Multi-phase optimizer with spectral flattening, Adam+LSE, LP alternating,
and peak-specific refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.
Improvements over previous: spectral domain phase, Adam optimizer (faster

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def adam_lse(a0, beta_schedule, n_iters_per_beta=2000, lr=0.001, time_limit=None):
    """Adam optimizer on LSE-smoothed objective with progressive beta."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for beta in beta_schedule:
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        t_count = 0
        for it in range(n_iters_per_beta):
            if time_limit is not None and time.monotonic() > time_limit:
                break
            sc, grad = lse_score_and_grad(a, beta)
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            # Periodic exact score check
            if it % 100 == 0:
                exact_sc = compute_score(a)
                if exact_sc < best_sc:
                    best_sc = exact_sc
                    best_a = a.copy()
                # Adaptive lr
                if exact_sc > best_sc * 1.01:
                    lr *= 0.8
                    a = best_a.copy()  # Revert
                    m[:] = 0
                    v[:] = 0
                    t_count = 0
                elif exact_sc < best_sc * 0.99:
                    lr *= 1.1
                    lr = min(lr, 0.01)
    
    return best_a, best_sc

def lp_fista(a, M_val, n_iters, beta, x_init=None):
    """FISTA for the LP inner problem."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a * a))
    L = beta * n * sum_a2 * 2.0 + 1e-12
    step = 1.0 / L
    
    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    
    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M_val)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum
        
        FA = np.fft.rfft(a, M_fft)
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr  # gradient of LSE w.r.t. x (without the -1 since we're maximizing)
        
        # Actually for max sum(x) s.t. conv(a,x) <= M, the LSE penalty is
        # (1/beta)*log(sum exp(beta*(conv(a,x) - M)))
        # FISTA minimizes: (1/beta)*log(sum exp(beta*(Bx - M))) + indicator(x>=0)
        # Gradient of LSE: (1/beta) * B^T * (exp(beta*(Bx-M)) / sum exp(beta*(Bx-M)))
        # = B^T * p / 1 (since p is normalized)
        # = corr (as computed)
        # But we want to MINIMIZE this, so gradient descent: x -= step * grad
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x
