"""
Multi-scale optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS with progressive beta, multi-scale refinement,
peak-targeted subgradient steps, and stochastic escape.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum())
    log_sum = z.max() - z_shift.max() + np.log(exp_sum)
    L = c_max + log_sum / beta  # LSE approximation of max(c)
    score = 2.0 * n * L / (S * S)
    
    p = exp_z / exp_sum  # softmax weights over convolution lags
    
    # Gradient of L w.r.t. a: 
    # dL/da[m] = sum_k p[k] * dc[k]/da[m] = sum_k p[k] * 2*a[k-m] (with a[i]=0 outside)
    # = 2 * cross_correlation(a, p) at appropriate lags
    # Specifically: sum_k p[k] * a[k-m] = (p * a)[m] in the appropriate sense
    
    # Let's compute: g[m] = sum_k p[k] * a[k-m] (with a[i]=0 for i<0 or i>=n)
    # This is the cross-correlation of a and p, evaluated at lag m
    # Using FFT: g = IFFT(conj(FFT(a)) * FFT(p)) ... no that's not right either.
    
    # Actually: sum_k p[k] * a[k-m] for k=0,...,2n-2
    # Let's define: for fixed m, sum over k where 0 <= k-m < n, i.e., m <= k < m+n
    # g[m] = sum_{k=m}^{m+n-1} p[k] * a[k-m] (with p[k]=0 for k outside [0,2n-2])
    
    # This is the cross-correlation of p (length 2n-1) and a (length n) at lag m
    # = (p cross-correlate a)[m]
    # = sum_j p[j+m] * a[j]
    
    # Using FFT: pad both to length >= 2n-1 + n - 1 = 3n-2
    # Actually, let's use the convolution approach:
    # g[m] = sum_j a[j] * p[j+m]
    # This is the cross-correlation, which can be computed as:
    # g = IFFT(FFT(p_full) * conj(FFT(a_full))) ... 
    
    # Simpler: just use the fact that g[m] = (p * a_flipped)[m + n - 1] where * is convolution
    # Or even simpler: use np.convolve
    
    # Actually the cleanest way: 
    # g[m] = sum_{j=0}^{n-1} a[j] * p[j+m]
    # where p is defined on [0, 2n-2]
    
    # This is equivalent to: take the cross-correlation of a and p
    # corr[m] = sum_j a[j] * p[j+m]
    # Using FFT: corr = IFFT(conj(FFT(a, M)) * FFT(p, M))[:n]
    # where M >= 2n-1
    
    M_fft = 1 << (2 * n - 1).bit_length()
    FA = np.fft.rfft(a, M_fft)
    FP = np.fft.rfft(p, M_fft)
    # cross-correlation: sum_j a[j] * conj(p[j+m]) but p is real so:
    # sum_j a[j] * p[j+m] = IFFT(conj(FA) * FP)[m] ... 
    # Wait, let me think about this more carefully.
    # 
    # We want g[m] = sum_j a[j] * p[j+m] for m = 0, ..., n-1
    # where p has length 2n-1.
    #
    # Let's pad a to length M and p to length M (M >= 3n-2 to be safe, or M >= 2n-1 + n - 1)
    # Actually M >= 2n-1 is enough if we're careful.
    #
    # The cross-correlation (a ⊛ p)[m] = sum_j a[j] * p[j+m]
    # In terms of convolution: (a ⊛ p) = conv(a, flip(p)) shifted
    # 
    # Using FFT: 
    # Let A = FFT(a_padded), P = FFT(p_padded)
    # Then FFT(a ⊛ p) = conj(A) * P (for circular cross-correlation)
    # 
    # For linear cross-correlation, we need M >= n + (2n-1) - 1 = 3n - 2
    # But if we just need m = 0, ..., n-1, and p has support [0, 2n-2], a has support [0, n-1]:
    # g[m] = sum_{j=0}^{min(n-1, 2n-2-m)} a[j] * p[j+m]
    # For m < n, j+m < n + m < 2n, so we need p up to index 2n-2. ✓
    # 
    # So M >= 2n-1 should work for the circular version to give us the correct linear result
    # for m = 0, ..., n-1 (as long as there's no wrap-around).
    # Wrap-around would occur if j+m >= M, but j+m <= (n-1) + (n-1) = 2n-2 < M. ✓
    
    corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
    # corr[m] = sum_j a[j] * p[j+m] = g[m] ✓ (approximately, for M >= 2n-1)
    
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, ftol=1e-15, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_obj_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxiter * 40, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return a_out

def upsample(a, n_new):
    """Upsample sequence to new length using interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

def peak_subgradient_step(a, n_steps=200, rng=None):
    """Subgradient descent targeting the convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        
        # Subgradient of max at k*: d(max)/da[m] = 2*a[k_star - m] (if valid)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            # direction[m] = 2*a[k_star - m] for m in [j_start, j_end]
            # a[k_star - m] for m = j_start,...,j_end
            # k_star - j_start to k_star - j_end
            direction[j_start:j_end+1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        base_step = min(0.02, max(0.001, 0.5 * (best_sc - 1.0) / (gn * gn + 1e-30)))
        improved = False
        for mult in [0.5, 1.0, 2.0, 4.0, 0.25]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved and it % 20 == 0:
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    
    return best_a

def gerchberg_saxton_init(n, n_iters=100):
    """Initialize using GS-style iteration to get flat PSD."""
    M = 1 << (2 * n - 1).bit_length()
    a = np.ones(n) / np.sqrt(n)
    for _ in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        # Make |A| as flat as possible
        amp = np.abs(A)
        amp_mean = amp.mean()
        # Blend: keep some of the original amplitude
        A_new = A * (amp_mean / (amp + 1e-30)) ** 0.5
        # Or: A_new = amp_mean * A / (amp + 1e-30)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a = a_new / a_new.sum() * np.sqrt(n)
    return a

We have:
- c[k] = sum_j a[j] * a[k-j] (full linear convolution, k = 0, ..., 2n-2)
- L = LSE(c, beta) = max(c) + (1/beta) * log(sum_k exp(beta*(c[k] - max(c))))
- Objective: f(a) = 2n * L / S^2 where S = sum(a)

Gradient of L w.r.t. a[m]:
dL/da[m] = sum_k p[k] * dc[k]/da[m] where p[k] = exp(beta*(c[k]-max(c))) / sum exp(...)

And dc[k]/da[m] = 2 * a[k-m] (with a[i] = 0 for i outside [0,n-1])

So dL/da[m] = 2 * sum_k p[k] * a[k-m]

Now, sum_k p[k] * a[k-m]:
- k ranges from 0 to 2n-2
- a[k-m] is non-zero when 0 <= k-m <= n-1, i.e., m <= k <= m+n-1
- So sum_k p[k] * a[k-m] = sum_{k=m}^{m+n-1} p[k] * a[k-m] (with bounds)
  = sum_{j=0}^{n-1} a[j] * p[j+m] (where j = k-m)

This is the cross-correlation of a and p at lag m: (a ⊛ p)[m] = sum_j a[j] * p[j+m]

Using FFT: if we pad both a and p to length M >= 2n-1:

            g = g + mu * (x - center)
            return sc, g

        res = minimize(f_prox, a_cur, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': gtol, 'maxiter': chunk,
                                'maxfun': chunk * 40, 'maxls': 50})
        a_new = np.clip(res.x, 0.0, 1000.0)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a_cur = a_new
        center = a_cur.copy()
        remaining -= chunk

    return best_a


def peak_shaving(a, n_steps, rng=None):
    """Targeted subgradient descent on the convolution peak with adaptive steps."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k = int(np.argmax(c))
        Mv = float(c[k])
        # Subgradient: d(max)/da[m] = 2*a[k-m] (if 0<=k-m<n)
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        d = np.zeros(n)
        if j1 >= j0:
            d[j0:j1 + 1] = 2.0 * a[k - j1:k - j0 + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        g = 2.0 * n * (d / S2 - 2.0 * Mv / S3)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        base = min(0.03, max(0.0005, 0.4 * (best_sc - 1.0) / (gn * gn + 1e-30)))
        improved = False
        for mult in [1.0, 0.5, 2.0, 0.25, 4.0, 0.1]:
            step = base * mult
            a_try = np.clip(a - step * g, 0.0, 1000.0)
            St = float(np.sum(a_try))
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        if not improved and it % 15 == 0:
            a = np.maximum(a * (1.0 + 0.008 * rng.standard_normal(n)), 0.0)
            a = a / a.sum()
    return best_a


def gs_init(n, n_iters=100):
    """Gerchberg-Saxton style initialization for flat PSD."""
    M = 1 << (2 * n - 1).bit_length()
    a = np.ones(n) / np.sqrt(n)
    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        amp = np.abs(A)
        amp_mean = amp.mean()
        # Gentle flattening
        A_flat = A * (amp_mean / (amp + 1e-30)) ** 0.4
        a_new = np.fft.irfft(A_flat, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a = a_new / a_new.sum() * np.sqrt(n)
    return a


def upsample(a, n_new):
    """Interpolate sequence to new length."""
    xo = np.linspace(0, 1, len(a))
    xn = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(xn, xo, a), 0.0)


def frequency_escape(a, rng, n_tries=10):
    """Structured perturbation in frequency domain to explore new basins."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_tries):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        # Random phase perturbation (preserving magnitude structure)
        phase_noise = rng.standard_normal(A.size) * 0.1
        # Only perturb the mid-frequency components (avoid DC and Nyquist)
        phase_noise[0] = 0
        if A.size > 1:
            phase_noise[-1] = 0
        A_mod = A * np.exp(1j * phase_noise)
        a_new = np.fft.irfft(A_mod, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(np.sum(a_new))
        if S < 1e-12:
            continue
        a_new = a_new / S
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
    return best_a, best_sc


def multi_peak_shaving(a, n_rounds=5, rng=None):
    """Shave multiple peaks iteratively, focusing on top-k peaks each round."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(np.sum(a))
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(n_rounds):
        c = self_conv(a)
        # Find top 3 peaks
        top_k = np.argsort(c)[-3:][::-1]
        for k in top_k:
            Mv = float(c[k])
            j0 = max(0, k - n + 1)
            j1 = min(k, n - 1)
            d = np.zeros(n)
            if j1 >= j0:
                d[j0:j1 + 1] = 2.0 * a[k - j1:k - j0 + 1][::-1]
            S2 = S * S
            S3 = S2 * S
            g = 2.0 * n * (d / S2 - 2.0 * Mv / S3)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                continue
            base = min(0.02, max(0.001, 0.3 * (best_sc - 1.0) / (gn * gn + 1e-30)))
            for mult in [1.0, 0.5, 2.0]:
                step = base * mult
                a_try = np.clip(a - step * g, 0.0, 1000.0)
                St = float(np.sum(a_try))
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc = compute_score(a_try)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
    return best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(upsample(parent, n))
    inits.append(gs_init(n, 60))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    # Smoothed random
    noise = rng.standard_normal(n)
    k = max(3, n // 50)
    smooth = np.convolve(noise, np.ones(k) / k, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 10.0))
    # Triangular-ish
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 10.0))
    # Multi-frequency
    inits.append(0.3 + 0.2*np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x+0.5)
                + 0.15*np.cos(6*np.pi*x+1.0) + 0.1*np.cos(8*np.pi*x+2.0))
    return [np.maximum(a, 0.0) for a in inits]


def propose_candidate():
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: diverse multi-scale init, progressive LSE-LBFGS with adaptive
    restart, multi-peak shaving, frequency-domain escape, and final polish.
    """
    rng = np.random.default_rng(7)
    t_start = time.monotonic()
    budget = 2400.0
    deadline = t_start + budget - 25.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_sc = np.inf
    best_a = None

    def rep(sc, a, lb):
        nonlocal best_sc, best_a
        if sc < best_sc:
            best_sc = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            el = time.monotonic() - t_start
            print(f"  [{lb}] NEW BEST: {sc:.8f} (n={len(a)}, t={el:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        rep(ps, parent, "parent")

    target_n = len(parent) if parent is not None else 6962

    # ================================================================
    # Phase 1: Multi-scale warm start (8% time)
    # ================================================================
    t_p1 = t_start + 0.08 * budget
    print(f"\n=== Phase 1: Multi-scale (target n={target_n}) ===")

    a_work = None
    scales = [256, 512, 1024, 2048, 4096]
    if target_n > 4096:
        scales.append(target_n)
    elif target_n not in scales:
        scales.append(target_n)
    scales = sorted(set(s for s in scales if s <= target_n))
    if target_n not in scales:
        scales.append(target_n)

    for ns in scales:
        if time.monotonic() > t_p1:
            break
        if a_work is None:
            a_work = gs_init(ns, 50)
        elif len(a_work) < ns:
            a_work = upsample(a_work, ns)
        else:
            continue
        for b in [30, 100, 400]:
            if time.monotonic() > t_p1 - 5:
                break
            ao = run_lbfgs(a_work, b, maxiter=2000, gtol=1e-12)
            S = float(np.sum(ao))
            if S > 1e-12:
                ao = ao / S
            sc = compute_score(ao)
            if ns == target_n:
                rep(sc, ao, f"MS n={ns} b={b}")
            a_work = ao

    # ================================================================
    # Phase 2: Diverse initialization at target length (7% time)
    # ================================================================
    t_p2 = t_start + 0.15 * budget
    print(f"\n=== Phase 2: Diverse init at n={target_n} ===")

    inits = build_diverse_inits(target_n, rng, parent)
    # If we have a work sequence from multi-scale, include it
    if a_work is not None and len(a_work) == target_n:
        inits.append(a_work.copy())

    a_main = None
    for i, a0 in enumerate(inits):
        if time.monotonic() > t_p2:
            break
        S = float(np.sum(a0))
        if S < 1e-12:
            continue
        a0 = a0 / S
        # Quick LSE to see potential
        ao = run_lbfgs(a0, 200, maxiter=3000, gtol=1e-11)
        S = float(np.sum(ao))
        if S > 1e-12:
            ao = ao / S
        sc = compute_score(ao)
        print(f"  init {i}: {sc:.8f}")
        rep(sc, ao, f"init{i}")
        if a_main is None or sc < compute_score(a_main):
            a_main = ao

    if a_main is None:
        a_main = inits[0] / np.sum(inits[0])

    # ================================================================
    # Phase 3: Main progressive LSE optimization (45% time)
    # ================================================================
    t_p3 = t_start + 0.60 * budget
    print(f"\n=== Phase 3: Main LSE optimization ===")

    a_cur = a_main.copy()
    S = float(np.sum(a_cur))
    if S > 0:
        a_cur = a_cur / S

    # Progressive beta: start smooth, get sharp
    beta_schedule = [20, 50, 100, 200, 400, 800, 1600, 3200, 6400, 10000, 16000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > t_p3 - 40:
            break
        elapsed_in_phase = time.monotonic() - (t_start + 0.15 * budget)
        remaining = (t_p3 - 40) - time.monotonic()
        frac = max(0.2, min(1.0, remaining / (0.45 * budget)))
        maxit = int(4000 * frac) + 1000
        print(f"  beta={beta}, maxiter={maxit}... ", end="", flush=True)
        t0 = time.monotonic()

        # Use adaptive restart for higher betas
        if beta >= 1600:
            ao = run_lbfgs_adaptive(a_cur, beta, maxiter=maxit, gtol=1e-14)
        else:
            ao = run_lbfgs(a_cur, beta, maxiter=maxit, gtol=1e-13)

        S = float(np.sum(ao))
        if S > 1e-12:
            ao = ao / S
        sc = compute_score(ao)
        print(f"{sc:.8f} ({time.monotonic()-t0:.1f}s)")
        rep(sc, ao, f"LSE b={beta}")
        a_cur = ao

    # ================================================================
    # Phase 4: Multi-peak shaving + LSE alternation (15% time)
    # ================================================================
    t_p4 = t_start + 0.75 * budget
    print(f"\n=== Phase 4: Multi-peak shaving ===")

    a_peak = multi_peak_shaving(best_a, n_rounds=8, rng=rng)
    rep(compute_score(a_peak), a_peak, "multi-peak")

    # Alternate: peak shave -> LSE polish
    a_alt = best_a.copy()
    for r in range(6):
        if time.monotonic() > t_p4 - 15:
            break
        a_alt = multi_peak_shaving(a_alt, n_rounds=4, rng=rng)
        ao = run_lbfgs(a_alt, 8000, maxiter=2500, gtol=1e-14)
        S = float(np.sum(ao))
        if S > 1e-12:
            ao = ao / S
        a_alt = ao
        sc = compute_score(a_alt)
        rep(sc, a_alt, f"peak-lse r{r}")

    # Single-peak focused shaving (more aggressive)
    a_sp = peak_shaving(best_a, n_steps=500, rng=rng)
    rep(compute_score(a_sp), a_sp, "single-peak")

    # ================================================================
    # Phase 5: Frequency-domain escape + re-optimization (12% time)
    # ================================================================
    t_p5 = t_start + 0.87 * budget
    print(f"\n=== Phase 5: Frequency escape ===")

    for i in range(6):
        if time.monotonic() > t_p5 - 30:
            break
        # Frequency-domain escape
        a_esc, sc_esc = frequency_escape(best_a, rng, n_tries=15)
        print(f"  freq-esc {i}: {sc_esc:.8f}")
        rep(sc_esc, a_esc, f"freq-esc{i}")

        if sc_esc < best_sc + 0.003:
            # Re-optimize from escaped point
            ao = run_lbfgs(a_esc, 3000, maxiter=3000, gtol=1e-14)
            S = float(np.sum(ao))
            if S > 1e-12:
                ao = ao / S
            sc = compute_score(ao)
            rep(sc, ao, f"freq-esc{i}-opt")
            ap = peak_shaving(ao, n_steps=150, rng=rng)
            rep(compute_score(ap), ap, f"freq-esc{i}-pk")

        # Also try smooth random perturbation
        scale = rng.uniform(0.01, 0.08)
        noise = rng.standard_normal(len(best_a))
        k = max(5, len(noise) // 200)
        noise = np.convolve(noise, np.ones(k) / k, mode='same')
        a_try = best_a * (1.0 + scale * noise / (np.std(noise) + 1e-30))
        a_try = np.maximum(a_try, 0.0)
        S = float(np.sum(a_try))
        if S < 1e-12:
            continue
        a_try = a_try / S
        sc_try = compute_score(a_try)
        if sc_try < best_sc + 0.002:
            ao = run_lbfgs(a_try, 2000, maxiter=3000, gtol=1e-14)
            S = float(np.sum(ao))
            if S > 1e-12:
                ao = ao / S
            rep(compute_score(ao), ao, f"rand-esc{i}")

    # ================================================================
    # Phase 6: Try nearby lengths (5% time)
    # ================================================================
    t_p6 = t_start + 0.92 * budget
    print(f"\n=== Phase 6: Length exploration ===")

    for dn in [-32, -16, 16, 32, 64]:
        if time.monotonic() > t_p6:
            break
        n_try = target_n + dn
        if n_try < 128:
            continue
        a_try = upsample(best_a, n_try)
        S = float(np.sum(a_try))
        if S < 1e-12:
            continue
        a_try = a_try / S
        ao = run_lbfgs(a_try, 5000, maxiter=2000, gtol=1e-13)
        S = float(np.sum(ao))
        if S > 1e-12:
            ao = ao / S
        sc = compute_score(ao)
        print(f"  n={n_try}: {sc:.8f}")
        rep(sc, ao, f"len{n_try}")

    # ================================================================
    # Phase 7: Final polish (remaining time)
    # ================================================================
    print(f"\n=== Phase 7: Final polish ===")
    if time.monotonic() < deadline - 80:
        ao = run_lbfgs_adaptive(best_a, 20000, maxiter=6000, gtol=1e-14)
        S = float(np.sum(ao))
        if S > 1e-12:
            ao = ao / S
        rep(compute_score(ao), ao, "final-20k")

    if time.monotonic() < deadline - 40:
        ap = multi_peak_shaving(best_a, n_rounds=5, rng=rng)
        rep(compute_score(ap), ap, "final-multi")

    if time.monotonic() < deadline - 15:
        ap = peak_shaving(best_a, n_steps=300, rng=rng)
        rep(compute_score(ap), ap, "final-peak")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_sc:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={elapsed:.1f}s/{budget:.0f}s")
    print(f"{'=' * 60}")

    return out
