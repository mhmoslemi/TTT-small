"""
Coordinate descent + L-BFGS hybrid optimizer for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Key improvements: exact 1D coordinate optimization with incremental
convolution updates, multi-length exploration, structured initializations,
and adaptive L-BFGS with very high beta for final polish.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    S2 = S * S
    S3 = S2 * S
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    a_rev = a[::-1]
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(PA * AA_rev, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    dL_da = 2.0 * h
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return 2.0 * n * L / S2, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_obj_grad, a0, args=(beta,),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_sweeps=50, deadline=None, verbose=False):
    """
    Efficient coordinate descent with incremental convolution updates.
    For each coordinate i, optimizes the 1D subproblem exactly.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    # Don't normalize - score is scale invariant
    a = a / S  # normalize for numerical stability
    
    c = fast_conv(a)
    c = c.copy()  # mutable copy for incremental updates
    best_a = a.copy()
    best_sc = compute_score(a)
    c_max = float(c.max())
    
    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        
        # Randomize coordinate order for better mixing
        order = np.random.permutation(n)
        
        for i in order:
            i = int(i)
            x_old = a[i]
            if x_old < 1e-15:
                # Try small positive values
                for x_try in [1e-4, 1e-3, 0.01]:
                    # Compute affected c values
                    k_start = max(0, i)
                    k_end = min(2 * n - 2, i + n - 1)
                    max_new = c_max
                    for k in range(k_start, k_end + 1):
                        ki = k - i
                        if ki < 0 or ki >= n:
                            continue
                        if k == 2 * i:
                            val = c[k] - x_old * x_old + x_try * x_try
                        else:
                            val = c[k] - 2.0 * a[ki] * x_old + 2.0 * a[ki] * x_try
                        if val > max_new and k != i and not (k >= k_start and k <= k_end):
                            # This k is affected, don't skip
                            pass
                    # Simpler: just compute the score directly
                    a[i] = x_try
                    sc = compute_score(a)
                    a[i] = x_old
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        a[i] = x_try
                        c = fast_conv(a)
                        c_max = float(c.max())
                continue
            
            # Normal case: x_old > 0
            # Determine affected range
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            
            # Find k's where the crossing with global max happens
            # c_unaffected_max = max of c outside [k_start, k_end]
            if k_start > 0:
                c_unaff_left = c[:k_start].max()
            else:
                c_unaff_left = -1e30
            if k_end < 2 * n - 2:
                c_unaff_right = c[k_end + 1:].max()
            else:
                c_unaff_right = -1e30
            c_unaff_max = max(c_unaff_left, c_unaff_right)
            
            # For each affected k, compute: c[k](x) = base + slope*x + quad*x^2
            # where base = c[k] - contrib_old, slope and quad depend on k
            # contrib_old = x_old^2 if k==2i, else 2*a[k-i]*x_old
            # contrib_new = x^2 if k==2i, else 2*a[k-i]*x
            
            # Identify top k's in affected range
            affected_ks = []
            for k in range(k_start, k_end + 1):
                ki = k - i
                if ki < 0 or ki >= n:
                    continue
                if k == 2 * i:
                    slope_k = 0.0
                    quad_k = 1.0
                else:
                    slope_k = 2.0 * a[ki]
                    quad_k = 0.0
                base_k = c[k] - (slope_k * x_old + quad_k * x_old * x_old)
                affected_ks.append((k, base_k, slope_k, quad_k))
            
            if len(affected_ks) == 0:
                continue
            
            # Find candidate x values
            candidates = [0.0, x_old]
            
            # x that makes top affected c[k] equal c_unaff_max
            if len(affected_ks) >= 1:
                # Sort by current value descending
                sorted_ks = sorted(affected_ks, key=lambda t: t[1] + t[2] * x_old + t[3] * x_old * x_old, reverse=True)
                top_k, top_base, top_slope, top_quad = sorted_ks[0]
                
                # Solve: top_base + top_slope*x + top_quad*x^2 = c_unaff_max
                # top_quad*x^2 + top_slope*x + (top_base - c_unaff_max) = 0
                if top_quad > 0:
                    A_q = top_quad
                    B_q = top_slope
                    C_q = top_base - c_unaff_max
                    disc = B_q * B_q - 4 * A_q * C_q
                    if disc >= 0:
                        x1 = (-B_q + np.sqrt(disc)) / (2 * A_q)
                        x2 = (-B_q - np.sqrt(disc)) / (2 * A_q)
                        if x1 > 0:
                            candidates.append(x1)
                        if x2 > 0:
                            candidates.append(x2)
                elif top_slope > 0:
                    x_cross = (c_unaff_max - top_base) / top_slope
                    if x_cross > 0:
                        candidates.append(x_cross)
                
                # x where top 2 affected c[k] cross
                if len(sorted_ks) >= 2:
                    k2, b2, s2, q2 = sorted_ks[1]
                    # (top_base - b2) + (top_slope - s2)*x + (top_quad - q2)*x^2 = 0
                    A_q = top_quad - q2
                    B_q = top_slope - s2
                    C_q = top_base - b2
                    if abs(A_q) < 1e-15:
                        if abs(B_q) > 1e-15:
                            x_cross = -C_q / B_q
                            if x_cross > 0:
                                candidates.append(x_cross)
                    else:
                        disc = B_q * B_q - 4 * A_q * C_q
                        if disc >= 0:
                            x1 = (-B_q + np.sqrt(disc)) / (2 * A_q)
                            x2 = (-B_q - np.sqrt(disc)) / (2 * A_q)
                            if x1 > 0:
                                candidates.append(x1)
                            if x2 > 0:
                                candidates.append(x2)
            
            # Also try x values that minimize the top c[k]
            # For a quadratic: x_min = -slope/(2*quad) if quad>0
            if len(affected_ks) >= 1:
                for (k_, b_, s_, q_) in affected_ks[:3]:
                    if q_ > 0:
                        x_min = -s_ / (2.0 * q_)
                        if x_min > 0:
                            candidates.append(x_min)
            
            # Evaluate at candidates and pick best
            best_x = x_old
            best_f = np.inf
            S_prime = S - x_old
            
            for x_try in candidates:
                x_try = max(0.0, min(x_try, 10.0 * S / n))  # reasonable bound
                # Compute new c_max
                new_max = c_unaff_max
                for (k_, b_, s_, q_) in affected_ks:
                    val = b_ + s_ * x_try + q_ * x_try * x_try
                    if val > new_max:
                        new_max = val
                S_new = S_prime + x_try
                if S_new < 1e-12:
                    continue
                f_val = 2.0 * n * new_max / (S_new * S_new)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x_try
            
            # Update
            if best_x != x_old:
                a[i] = best_x
                # Update c incrementally
                delta = best_x - x_old
                for k in range(k_start, k_end + 1):
                    ki = k - i
                    if ki < 0 or ki >= n:
                        continue
                    if k == 2 * i:
                        c[k] += best_x * best_x - x_old * x_old
                    else:
                        c[k] += 2.0 * a[ki] * delta
                S += delta
                c_max = max(c_max, float(c.max()))
                
                sc = 2.0 * n * c_max / (S * S)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        
        # Periodic full recompute to avoid drift
        if (sweep + 1) % 10 == 0:
            c = fast_conv(a)
            c_max = float(c.max())
            S = float(a.sum())
            sc = 2.0 * n * c_max / (S * S)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, best_sc


def fienup_flat(a, n_iters=300):
    """Fienup error reduction targeting flat spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.5 * (1.0 - progress) ** 2 + 0.02
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def make_structured_init(n, rng, kind=0):
    """Create structured initial sequences."""
    if kind == 0:
        # Triangular
        a = np.abs(np.arange(n) - n / 2) / (n / 2)
        a = 1.0 - a  # peak in middle
    elif kind == 1:
        # Alternating high/low
        a = np.ones(n)
        a[1::2] = 0.3
    elif kind == 2:
        # Two peaks
        a = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n * 0.1)) ** 2)
    elif kind == 3:
        # Geometric (favoring one end)
        a = np.geomspace(0.1, 1.0, n) if n > 1 else np.array([1.0])
    elif kind == 4:
        # Random with block structure
        a = np.zeros(n)
        n_blocks = rng.integers(3, 8)
        for _ in range(n_blocks):
            start = rng.integers(0, n)
            length = rng.integers(n // 10, n // 3)
            a[start:start + length] += rng.uniform(0.5, 1.0)
    else:
        # Pure random
        a = rng.uniform(0.1, 1.0, n)
    
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    return a

        candidates.append(parent.copy())
        update(sc, parent, "parent")
    
    # Try different lengths and structured inits
    parent_n = len(parent) if parent is not None else 2048
    lengths_to_try = [parent_n]
    # Add some nearby lengths
    for factor in [0.5, 1.5, 2.0, 0.25]:
        new_n = int(parent_n * factor)
        new_n = max(64, min(16384, new_n))
        # Round to power of 2 for FFT efficiency
        new_n = 1 << (new_n - 1).bit_length()
        if new_n not in lengths_to_try:
            lengths_to_try.append(new_n)
    
    for n in lengths_to_try[:5]:
        if time.monotonic() > p1_end:
            break
        for kind in range(5):
            if time.monotonic() > p1_end:
                break
            a = make_structured_init(n, rng, kind)
            sc = compute_score(a)
            if sc < best_score + 0.5:  # only keep promising ones
                candidates.append(a)
                if sc < best_score:
                    update(sc, a, f"init-n{n}-k{kind}")
    
    # Quick optimization of top candidates
    print(f"  Quick optimizing {len(candidates)} candidates...")
    for idx, a in enumerate(candidates[:8]):
        if time.monotonic() > p1_end:
            break
        a_opt = fienup_flat(a, n_iters=100)
        sc = compute_score(a_opt)
        update(sc, a_opt, f"quick-{idx}")
    
    print(f"  After Phase 1: best={best_score:.8f}")
    
    # ============ Phase 2: Coordinate Descent (35% time) ============
    p2_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Coordinate Descent (until t={p2_end - t_start:.0f}s) ===")
    
    a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=200, deadline=p2_end)
    update(sc_cd, a_cd, "coord-descent")
    
    # ============ Phase 3: L-BFGS with increasing beta (30% time) ============
    p3_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (until t={p3_end - t_start:.0f}s) ===")
    
    betas = [200, 500, 1000, 3000, 8000, 20000, 50000, 100000]
    for i, beta in enumerate(betas):
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 20:
            break
        # Allocate more time to later betas
        maxiter = int(min(15000, max(2000, remaining * 3)))
        print(f"  L-BFGS beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s)")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 25)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"lbfgs-b{beta}")
    
    # ============ Phase 4: More coordinate descent + L-BFGS (15% time) ============
    p4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Refinement (until t={p4_end - t_start:.0f}s) ===")
    
    # Coordinate descent again
    if time.monotonic() < p4_end - 60:
        a_cd2, sc_cd2 = coordinate_descent(best_a, n_sweeps=100, deadline=p4_end - 60)
        update(sc_cd2, a_cd2, "coord-descent-2")
    
    # L-BFGS polish
    if time.monotonic() < p4_end - 30:
        remaining = p4_end - 30 - time.monotonic()
        maxiter = int(min(15000, max(3000, remaining * 5)))
        sc, a_opt = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, "lbfgs-polish-100k")
    
    # ============ Phase 5: Final polish (10% time) ============
    print(f"\n=== Phase 5: Final Polish ===")
    
    # Try a few perturbations
    for pert in range(5):
        if time.monotonic() > deadline:
            break
        a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(len(best_a)))
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        
        # Quick coordinate descent
        a_opt, sc_opt = coordinate_descent(a_pert, n_sweeps=20,
                                           deadline=min(deadline, time.monotonic() + 30))
        update(sc_opt, a_opt, f"perturb-{pert}")
    
    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        maxiter = int(min(10000, max(2000, remaining * 4)))
        sc, a_opt = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_opt, "final-lbfgs-200k")
    
    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd_final, sc_cd_final = coordinate_descent(
            best_a, n_sweeps=30, deadline=deadline - 2)
        update(sc_cd_final, a_cd_final, "final-cd")
    
    # ============ Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
