"""

gradient with natural active sets, dual-gap-based convergence, multi-scale
refinement via upsampling, and LSE-LBFGS polishing. Key differences from prior:
(1) natural growing/shrinking active set, (2) dual gap monitoring for adaptive
stopping, (3) multi-resolution upsampling to escape local minima, (4) Fourier
structured perturbations, (5) sweep over sequence lengths.

import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute h[j] = sum_i a[i] * w[i+j] (cross-correlation of a with w at lag j)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def natural_active_set(conv, rel_tol=0.01, min_count=3, max_frac=0.3):
    """Find natural active set: all positions within rel_tol of max."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - rel_tol)
    active = conv >= threshold
    n_active = int(active.sum())
    if n_active < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    max_count = max(min_count, int(max_frac * conv.size))
    if n_active > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_update(a, w, eta):
    """One exponentiated gradient step with given active weights."""
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)

        best_sc_iter = sc

                best_a_iter = new_a
                best_eta = eta_try

            best_a = best_a_iter
            a = best_a_iter
            eta = best_eta
            stagnant = 0

            if stagnant >= max_stagnant:

def resample_sequence(a, n):
    """Resample sequence a to length n via linear interpolation."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def fourier_perturb(a, rng, strength):
    """Perturb in Fourier domain for structured changes."""
    n = len(a)
    N = next_power_of_two(n)
    A = np.fft.rfft(a, N)
    # Random phase perturbation at low and mid frequencies
    freqs = np.arange(len(A))
    for _ in range(3):
        band_width = max(1, n // 10)
        lo = int(rng.integers(1, max(2, n // 4)))
        hi = min(len(A) - 1, lo + band_width)
        indices = np.arange(lo, hi + 1)
        A[indices] *= (1.0 + strength * rng.standard_normal(len(indices)))
        A[indices] = np.maximum(A[indices], 0.01 * np.abs(A[indices]))  # keep positive-ish
    out = np.fft.irfft(A, N)[:n]
    return normalize_a(np.clip(out, 0.0, None))


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-10:
            inits.append(arr / arr.sum())
    
    # Uniform
    add(np.ones(n))
    
    # V-shapes (high at center)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(np.maximum(0.01, 1.0 - alpha * (2.0 * x - 1.0) ** 2))
    
    # U-shapes (high at ends)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    
    # Cosine modulations
    for f in (1, 2, 3, 4, 5, 7, 11):
        for alpha in (0.2, 0.5, 0.8, 1.0, 1.5):
            add(np.maximum(0.01, 1.0 + alpha * np.cos(2.0 * np.pi * f * x)))
    
    # Power-law tapers
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        add(np.maximum(0.01, (1.0 - np.abs(2.0 * x - 1.0)) ** p))
    
    # Gaussian
    for sigma in (0.1, 0.15, 0.2, 0.3, 0.4, 0.5):
        add(np.exp(-((x - 0.5) ** 2) / (2 * sigma * sigma)))
    
    # Bimodal
    for width in (0.05, 0.08, 0.1, 0.15, 0.2):
        for center in (0.2, 0.25, 0.3, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    
    # Random smooth (Fourier random)
    Nf = next_power_of_two(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        decay = float(rng.uniform(1.5, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** decay)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(np.maximum(0.01, 1.0 + float(rng.uniform(0.3, 1.5)) * noise))
    
    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits
