"""
Optimized step function search using progressive L-BFGS with very high LSE beta,
multi-peak shaving with exact line search, and improved LP-alternating method.
Key changes from previous: 50% time on L-BFGS (beta up to 20000), new multi-peak
shaving, backtracking FISTA for LP, and spectral initialization.

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=8000, maxfun=300000, gtol=1e-14):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 0.5
    else:
        a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fista_lp(a, M, n_iters=20000, beta=300.0, x_init=None):
    """FISTA for: min (1/beta)*log(sum exp(beta*(Bx - M))) s.t. x >= 0"""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    if x.sum() < 1e-12:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)  # Precompute

    # Lipschitz constant estimate
    sum_a2 = float(np.dot(a, a))
    L = beta * n * sum_a2 * 2.0 + 1e-12
    step = 1.0 / L
    step_max = 10.0 / L  # Allow step to grow

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr - x.mean()  # Center the gradient (since we're on the simplex)

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        # Periodic step size update (backtracking)
        if (it + 1) % 500 == 0:
            gnorm = float(np.linalg.norm(grad))
            f_change = float(np.dot(grad, x - x_new))
            if f_change < 0 and gnorm > 1e-15:
                # Try increasing step
                step_new = min(step * 1.5, step_max)
                # Verify with a simple check
                v_test = u - step_new * grad
                x_test = np.maximum(v_test, 0.0)
                if float(np.linalg.norm(x_test - x)) < float(np.linalg.norm(x_new - x)) * 2:
                    step = step_new
            else:
                step = max(step / 1.5, 1.0 / (L * 2))

    return x


def multi_peak_shave(a, n_iters=200, top_k=5):
    """Reduce multiple peaks simultaneously with exact line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        if c_max - c_mean < 1e-15:
            break

        # Find top-k peaks
        thresh = c_mean + 0.3 * (c_max - c_mean)
        peaks = np.where(c >= thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(c))])
        
        # Compute combined gradient to reduce all peaks
        grad = np.zeros(n)
        for k in peaks[:top_k]:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                other = k - j
                if 0 <= other < n:
                    grad[j] += 2.0 * a[other]
        
        # Score gradient (accounting for normalization)
        S2 = S * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * c_max / (S2 * S))
        
        # Line search: minimize score(a - t * grad_score)
        gn = float(np.linalg.norm(grad_score))
        if gn < 1e-15:
            break
        
        # Golden section search
        t_lo, t_hi = 0.0, min(0.5, 2.0 * (best_sc - 1.0) / (gn * gn + 1e-30))
        if t_hi < 1e-10:
            t_hi = 0.1
        
        def f_at_t(t):
            a_try = a - t * grad_score
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                return np.inf
            a_try = a_try / St
            return compute_score(a_try)
        
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        t1 = t_lo + (1 - gr) * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)
        f1 = f_at_t(t1)
        f2 = f_at_t(t2)
        for _ in range(20):
            if f1 < f2:
                t_hi = t2
                t2 = t1
                f2 = f1
                t1 = t_lo + (1 - gr) * (t_hi - t_lo)
                f1 = f_at_t(t1)
            else:
                t_lo = t1
                t1 = t2
                f1 = f2
                t2 = t_lo + gr * (t_hi - t_lo)
                f2 = f_at_t(t2)
        
        t_opt = (t_lo + t_hi) / 2
        a_try = a - t_opt * grad_score
        a_try = np.clip(a_try, 0.0, 1000.0)
        St = float(a_try.sum())
        if St < 1e-12:
            break
        a_try = a_try / St
        sc_try = compute_score(a_try)
        
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            a_try = a - 0.1 * t_opt * grad_score
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St > 1e-12:
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a

"""
Optimized step function search using progressive high-beta L-BFGS, LP-alternating
with backtracking FISTA, multi-peak shaving with golden-section line search,
and coordinate-descent polishing. Key improvements: 45% time on L-BFGS (beta up to
20000), improved FISTA with adaptive step, spectral initialization, and dual-weighted
peak shaving. Targets sub-1.503 score for n~6962.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(x, beta):
    a = np.maximum(x, 0.0)
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = float(z.max() - z_shifted.max() + np.log(exp_z.sum()))
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, gtol):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 0.5
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0,
        method='L-BFGS-B', jac=True, args=(beta,), bounds=bounds,
        options={'ftol': 1e-18, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fista_lp(a, M, n_iters, beta, x_init):
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    if x.sum() < 1e-12:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)
    sum_a2 = float(np.dot(a, a))
    L = beta * n * sum_a2 * 2.0 + 1e-12
    step = 1.0 / L
    step_base = step
    step_max = 20.0 * step_base
    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        x_mean = float(x.mean())
        grad = corr - x_mean
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
        if (it + 1) % 1000 == 0:
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-12:
                step = step_base
            else:
                f_approx = float(np.dot(grad, x_new - x))
                if f_approx < -1e-15:
                    step = min(step * 1.3, step_max)
                elif f_approx > 1e-8:
                    step = max(step * 0.7, step_base * 0.5)
    return x


def lp_alternating(a, max_outer_iters, fista_iters, time_limit, beta):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    x_warm = None
    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        c = self_conv(a)
        M = float(c.max())
        x = fista_lp(a, M, n_iters=fista_iters, beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        best_t = 0.3
        best_t_sc = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t
        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                break
        a = a_new
    return best_a, best_sc, x_warm


def multi_peak_shave(a, n_iters, top_k, time_limit=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        c = self_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        if c_max - c_mean < 1e-14:
            break
        frac = 0.3 + 0.5 * (it / max(1, n_iters))
        thresh = c_mean + frac * (c_max - c_mean)
        peaks = np.where(c >= thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(c))])
        peaks = peaks[np.argsort(c[peaks])[::-1]][:top_k]
        grad = np.zeros(n)
        for k in peaks:
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            for j in range(j_start, j_end + 1):
                other = int(k) - j
                grad[j] += 2.0 * a[other]
        S2 = S * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * c_max / (S2 * S))
        gn = float(np.linalg.norm(grad_score))
        if gn < 1e-15:
            break
        t_lo = 0.0
        t_hi = min(1.0, 5.0 / (gn * gn + 1e-30))
        if t_hi < 1e-8:
            t_hi = 0.5
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t1 = t_lo + (1.0 - gr) * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)
        a_t1 = np.clip(a - t1 * grad_score, 0.0, 1000.0)
        St1 = float(a_t1.sum())
        f1 = compute_score(a_t1 / St1) if St1 > 1e-12 else np.inf
        a_t2 = np.clip(a - t2 * grad_score, 0.0, 1000.0)
        St2 = float(a_t2.sum())
        f2 = compute_score(a_t2 / St2) if St2 > 1e-12 else np.inf
        for _ in range(25):
            if f1 < f2:
                t_hi = t2
                t2 = t1
                f2 = f1
                t1 = t_lo + (1.0 - gr) * (t_hi - t_lo)
                a_t1 = np.clip(a - t1 * grad_score, 0.0, 1000.0)
                St1 = float(a_t1.sum())
                f1 = compute_score(a_t1 / St1) if St1 > 1e-12 else np.inf
            else:
                t_lo = t1
                t1 = t2
                f1 = f2
                t2 = t_lo + gr * (t_hi - t_lo)
                a_t2 = np.clip(a - t2 * grad_score, 0.0, 1000.0)
                St2 = float(a_t2.sum())
                f2 = compute_score(a_t2 / St2) if St2 > 1e-12 else np.inf
        t_opt = (t_lo + t_hi) / 2.0
        a_try = np.clip(a - t_opt * grad_score, 0.0, 1000.0)
        St = float(a_try.sum())
        if St < 1e-12:
            break
        a_try = a_try / St
        sc_try = compute_score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            a_try2 = np.clip(a - 0.2 * t_opt * grad_score, 0.0, 1000.0)
            St2v = float(a_try2.sum())
            if St2v > 1e-12:
                a_try2 = a_try2 / St2v
                sc2 = compute_score(a_try2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a_try2.copy()
                    a = a_try2
    return best_a


def spectral_init(n, n_modes, rng):
    freqs = np.fft.rfftfreq(n)
    target_power = np.zeros(len(freqs))
    target_power[0] = 1.0 / n
    for k in range(1, n_modes):
        target_power[k] = 1.0 / (n * (1.0 + 0.3 * np.sin(2.0 * np.pi * k / n)))
    A = np.sqrt(target_power) * np.exp(1j * rng.standard_normal(len(freqs)))
    a_seq = np.fft.irfft(A * n, n)
    a_seq = np.maximum(a_seq, 0.01)
    a_seq = a_seq / a_seq.sum()
    return a_seq


def error_reduction(a, n_iters, alpha, time_limit=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        mask = np.abs(A) < 1e-10
        d[mask] = 0.0
        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def coord_descent_polish(a, n_sweeps, time_limit=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweps):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        improved_any = False
        for j in range(0, n, max(1, n // 2000)):
            if time_limit is not None and time.monotonic() > time_limit:
                break
            orig = a[j]
            for trial_frac in [0.8, 0.9, 0.95, 1.05, 1.1, 1.3, 0.7, 0.5]:
                trial = orig * trial_frac
                if trial < 0 or trial > 1000:
                    continue
                a[j] = trial
                St = float(a.sum())
                if St < 1e-12:
                    a[j] = orig
                    continue
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved_any = True
                else:
                    a[j] = orig
                    break
            a[j] = orig
            a[j] = best_a[j]
        if not improved_any:
            break
    return best_a

    # Phase 4: L-BFGS progressive (42% time) - MAIN WORKHORSE
    p4_deadline = t_start + 0.84 * total_budget
    print(f"\n--- Phase 4: L-BFGS Progressive (t limit={p4_deadline - t_start:.0f}s) ---")
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [500, 1000, 2000, 4000, 8000, 12000, 16000, 20000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > p4_deadline - 60:
            break
        remaining = p4_deadline - time.monotonic()
        if remaining < 30:
            break
        maxit = min(int(remaining * 3), 30000)
        maxfun = min(int(remaining * 10), 200000)
        gtol = 1e-13 if beta < 10000 else 1e-14
        print(f"  L-BFGS beta={beta}, maxiter={maxit}, remaining={remaining:.0f}s...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=maxfun, gtol=gtol)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s, nit={res.nit}")
        check_report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        # Intermediate peak shaving between L-BFGS runs
        if time.monotonic() < p4_deadline - 60 and bi < len(beta_schedule) - 1:
            a_mid = multi_peak_shave(a_norm, n_iters=100, top_k=3,
                                     time_limit=min(time.monotonic() + 30, p4_deadline - 60))
            sc_mid = compute_score(a_mid)
            check_report(sc_mid, a_mid, f"MidPeak b={beta}")
            a_norm = best_a.copy()
            S = float(a_norm.sum())
            if S > 0:
                a_norm = a_norm / S
    print(f"After Phase 4: {best_score:.8f}")

    # Phase 5: Final polish - coordinate descent + last L-BFGS (12% time)
    p5_deadline = t_start + 0.96 * total_budget
    print(f"\n--- Phase 5: Final Polish (t limit={p5_deadline - t_start:.0f}s) ---")
    a_cd = coord_descent_polish(best_a, n_sweeps=2, time_limit=min(time.monotonic() + 120, p5_deadline - 60))
    sc_cd = compute_score(a_cd)
    check_report(sc_cd, a_cd, "CoordDesc")

    if time.monotonic() < p5_deadline - 45:
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 0:
            a_final_in = a_final_in / S
        remaining = p5_deadline - time.monotonic() - 20
        sc_fin, a_fin, _ = run_lbfgs(a_final_in, 20000,
                                      maxiter=min(int(remaining * 3), 15000),
                                      maxfun=min(int(remaining * 8), 100000),
                                      gtol=1e-14)
        check_report(sc_fin, a_fin, "FinalLBFGS b=20000")

    if time.monotonic() < p5_deadline - 10:
        a_shf = multi_peak_shave(best_a, n_iters=100, top_k=2,
                                  time_limit=p5_deadline - 5)
        sc_shf = compute_score(a_shf)
        check_report(sc_shf, a_shf, "FinalShave")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
