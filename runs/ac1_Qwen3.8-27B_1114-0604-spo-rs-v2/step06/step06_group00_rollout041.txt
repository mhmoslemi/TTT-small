def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):

    if S < 1e-15:

    return a / S

def lse_score_and_grad(a, beta):

        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        lp_time = time.monotonic() - t0
        
        if not res.success:
            # Try removing the tightest cut and retry
            if len(cuts) > 5:
                # Remove the latest cut
                cuts.pop()
                continue
            else:
                break
        
        a_new = res.x[:n]
        t_lp = res.x[-1]
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        # Evaluate true objective
        c = fast_conv(a)
        M_true = float(c.max())
        score = 2.0 * n * M_true
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    OA iter {it}: score={score:.8f}, t_lp={t_lp:.6f}, gap={M_true-t_lp:.2e}, lp_time={lp_time:.2f}s, cuts={len(cuts)}")
        else:
            if it % 5 == 0:
                print(f"    OA iter {it}: score={score:.8f} (best={best_score:.8f}), t_lp={t_lp:.6f}, cuts={len(cuts)}")
        
        # Check for violated cuts
        violated = c > t_lp + 1e-10
        if violated.any():
            viol_lags = np.where(violated)[0]
            # Add the most violated ones
            viol_sorted = viol_lags[np.argsort(c[viol_lags])[::-1][:10]]
            for k in viol_lags:
                if k not in [cut[0] for cut in cuts]:  # Avoid duplicates
                    cuts.append((int(k), a.copy(), float(c[k])))
                    if len(cuts) >= max_cuts:
                        # Remove the least active cut
                        break
        
        # Also update active set for next iteration
        # (We keep all cuts, so this is implicit)
        
        # Check convergence: if M_true ≈ t_lp, we're at a local optimum
        if M_true - t_lp < 1e-8:
            print(f"    OA converged at iter {it}: gap < 1e-8")
            break
    
    return best_a, best_score

def lse_anneal_single(a0, betas, maxiter_per_beta=3000, time_budget=None, t_start=None):
    """LSE annealing."""
    a = normalize(a0.copy())
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for beta in betas:
        if time_budget is not None and time.monotonic() - t_start > time_budget:
            break
        bounds = [(0.0, 1000.0)] * len(a)
        try:
            res = minimize(
                lse_score_and_grad, a, args=(beta,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter_per_beta,
                         'maxfun': maxiter_per_beta * 20}
            )
            a_out = np.clip(res.x, 0.0, 1000.0)
            if a_out.sum() < 1e-6:
                a_out = a
            sc = compute_score(a_out)
            if sc < best_sc:
                best_sc = sc
                best_a = a_out.copy()
            a = normalize(a_out)
        except Exception:
            pass
    
    return best_sc, best_a

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid OA + LSE optimizer for minimizing 2n*max(conv(a,a))/S^2.
    Uses outer approximation (cutting plane LP) as the primary method,
    combined with LSE-smoothed L-BFGS for local refinement.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    
    best_score = np.inf
    best_a = None
    
    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
        a_init = parent.copy()
    else:
        n0 = 10000
        a_init = np.ones(n0) / n0
        # Add some random structure
        a_init = a_init + 0.1 * rng.standard_normal(n0)
        a_init = normalize(a_init)
        sc_p = compute_score(a_init)
        update(sc_p, a_init, "init")
    
    if best_a is None:
        best_a = normalize(a_init)
    
    n = len(best_a)
    print(f"Starting optimization: n={n}, initial score={best_score:.8f}")
    
    # ==============================================================
    # Phase 1: Quick LSE warmup (8% time)
    # ==============================================================
    phase1_end = t_start + 0.08 * (budget_s - 30)
    print(f"\n=== Phase 1: LSE Warmup (until {phase1_end - t_start:.0f}s) ===")
    
    a_work = best_a.copy()
    for beta in [50, 200, 1000]:
        if time.monotonic() > phase1_end - 10:
            break
        remaining = phase1_end - 10 - time.monotonic()
        if remaining < 10:
            break
        maxiter = max(200, int(remaining * 10))
        a_work, sc = lse_anneal_single(a_work, [beta], maxiter_per_beta=maxiter)
        update(sc, a_work, f"LSE-warmup b={beta}")
    
    # ==============================================================
    # Phase 2: Outer Approximation (35% time)
    # ==============================================================
    phase2_end = t_start + 0.43 * (budget_s - 30)
    print(f"\n=== Phase 2: Outer Approximation (until {phase2_end - t_start:.0f}s) ===")
    
    oa_budget = phase2_end - time.monotonic()
    if oa_budget > 30:
        a_oa, sc_oa = oa_method(best_a, time_budget=oa_budget * 0.7, max_cuts=n)
        update(sc_oa, a_oa, "OA")
        
        # Refine OA result with LSE
        if time.monotonic() < phase2_end - 20:
            a_ref, sc_ref = lse_anneal_single(a_oa, [5000, 20000], maxiter_per_beta=2000)
            update(sc_ref, a_ref, "OA+LSE")
    
    # ==============================================================
    # Phase 3: Multi-restart OA with perturbations (25% time)
    # ==============================================================
    phase3_end = t_start + 0.68 * (budget_s - 30)
    print(f"\n=== Phase 3: Multi-Restart OA ===")
    
    n_restarts = 8
    for r in range(n_restarts):
        if time.monotonic() > phase3_end - 30:
            break
        remaining = phase3_end - 30 - time.monotonic()
        if remaining < 30:
            break
        
        # Create perturbed starting point
        if r % 3 == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.005 * r
            a_pert = normalize(best_a + scale * rng.standard_normal(n))
        elif r % 3 == 1:
            # Multiplicative perturbation
            pert = np.exp(0.02 * rng.standard_normal(n))
            a_pert = normalize(best_a * pert)
        else:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            a_pert = best_a.copy()
            # Reduce peak contributors slightly
            a_pert[j_start:j_end+1] *= 0.95
            # Add to non-contributors
            non_peak = np.array([i for i in range(n) if i < j_start or i > j_end])
            if len(non_peak) > 0:
                add_idx = rng.choice(non_peak, min(100, len(non_peak)), replace=False)
                a_pert[add_idx] += 0.001
            a_pert = normalize(a_pert)
        
        # Run OA with limited time
        oa_t = min(remaining * 0.8 / (n_restarts - r), remaining * 0.5)
        a_oa, sc_oa = oa_method(a_pert, time_budget=oa_t, max_cuts=min(n, 500))
        update(sc_oa, a_oa, f"OA-restart {r}")
        
        # Quick LSE follow-up
        if sc_oa < best_score and time.monotonic() < phase3_end - 15:
            a_ref, sc_ref = lse_anneal_single(a_oa, [10000], maxiter_per_beta=1000)
            update(sc_ref, a_ref, f"OA-restart+LSE {r}")
    
    # ==============================================================
    # Phase 4: Aggressive LSE with high beta (17% time)
    # ==============================================================
    phase4_end = t_start + 0.85 * (budget_s - 30)
    print(f"\n=== Phase 4: High-Beta LSE Refinement ===")
    
    a_work = best_a.copy()
    for beta in [5000, 15000, 50000, 100000]:
        if time.monotonic() > phase4_end - 15:
            break
        remaining = phase4_end - 15 - time.monotonic()
        if remaining < 10:
            break
        maxiter = max(500, int(remaining * 8))
        a_work, sc = lse_anneal_single(a_work, [beta], maxiter_per_beta=maxiter)
        update(sc, a_work, f"HIGH-LSE b={beta}")
    
    # ==============================================================
    # Phase 5: Final polish (15% time)
    # ==============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    
    # Last LSE
    if time.monotonic() < deadline - 20:
        remaining = deadline
