def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def cross_corr_weights(w, x):
    """Compute h_j = sum_k w_k * x_{k-j} for j=0..n-1.
    w has length 2n-1 (convolution lags), x has length n."""
    n = len(x)
    if n == 0:
        return np.empty(0)
    N = next_power_of_two(2 * n - 1)
    xr = np.zeros(N)
    xr[:n] = x[::-1]
    wp = np.zeros(N)
    wp[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(xr, N) * np.fft.rfft(wp, N), N)
    return conv[n - 1:2 * n - 1]


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def lse_score_grad(a, beta):
    """Compute LSE-smoothed objective and gradient.
    Objective: F = 2 * (dmax + log(sum exp(beta*(d-dmax)))/beta)
    where d_k = n*c_k/S^2, c = a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    z = np.clip(z, -60.0, 60.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        sum_e = float(e.sum())
        if sum_e <= 0:
            return 1e10, np.zeros(n)

    p = e / sum_e  # LSE weights
    L = dmax + np.log(sum_e) / beta

    # Gradient: dF/da_j = (4n/S^2)*h_j - (4n/S^3)*a_j*dot(p,c)
    h = cross_corr_weights(p, a)
    dot_pc = float(np.dot(p, c))
    grad = (4.0 * float(n) / (S * S)) * h - (4.0 * float(n) / (S ** 3)) * a * dot_pc
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    score = 2.0 * L
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a_safe(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a_safe(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize(res.x)
    else:
        a_out = a
    sc = exact_score_a_safe(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a_safe(a)
    return float(sc), a_out


def exact_score_a_safe(a):
    try:
        return exact_score(a)
    except Exception:
        return np.inf


def estimate_eval_time(n):
    base = 0.025
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, np.log2(max(16, n)) / np.log2(8192.0))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 5.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    return max(64, min(4000, mf))


def adaptive_mirror_descent(a_start, max_iters, deadline, eta_range=(0.1, 1.0),
                            k_range=(1, 2, 3, 5, 8, 15), support_thresh=1e-7):
    """Adaptive multiplicative mirror descent.
    At optimum: h_j = const on support(a).
    Update: a_new = a * exp(-eta * (h_j - h_mean_support))
    with adaptive eta and dynamic active set."""
    a = normalize(a_start)
    best_a = a.copy()
    best_sc = exact_score_a_safe(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        c = fast_self_conv(a)
        S = float(np.sum(a))
        d = (n * c) / (S * S)
        dmax = float(np.max(d))
        if dmax <= 0:
            break

        # Dynamic active set: all lags within 1% of max, capped
        threshold = dmax * 0.995
        active_mask = d >= threshold
        active_count = int(np.sum(active_mask))
        if active_count == 0:
            active_mask = np.zeros(len(d), dtype=bool)
            active_mask[np.argmax(d)] = True
            active_count = 1
        # Cap active set size
        if active_count > 30:
            idx = np.argpartition(d, -30)[-30:]
            active_mask = np.zeros(len(d), dtype=bool)
            active_mask[idx] = True
            active_count = 30

        w = np.zeros(len(d))
        w[active_mask] = 1.0 / active_count

        # Compute cross-correlation h
        h = cross_corr_weights(w, a)
        h = np.maximum(h, 0.0)

        # Determine support (where a is significant)
        support = a > support_thresh * a.max()
        if np.sum(support) < 10:
            support = a > 1e-10

        h_support = h[support]
        if len(h_support) > 0:
            h_mean = float(np.mean(h_support))
        else:
            h_mean = float(np.max(h)) + 1e-10

        if h_mean < 1e-15:
            break

        # Try multiple step sizes and active set sizes
        improved_this_iter = False
        for eta in eta_range:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            # Multiplicative update
            h_norm = h / max(h_mean, 1e-15)
            log_a = np.log(np.maximum(a, 1e-300))
            log_new = log_a - eta * (h_norm - 1.0)
            a_new = np.exp(log_new - np.max(log_new))
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)

            sc = exact_score_a_safe(a_new)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved_this_iter = True

        # Also try smaller active sets for more focused updates
        for k in k_range:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            if k >= len(d):
                continue
            idx_k = np.argpartition(d, -k)[-k:]
            w_k = np.zeros(len(d))
            w_k[idx_k] = 1.0 / k

            h_k = cross_corr_weights(w_k, a)
            h_k = np.maximum(h_k, 0.0)
            support = a > support_thresh * a.max()
            if np.sum(support) < 10:
                support = a > 1e-10
            h_s = h_k[support]
            h_mean_k = float(np.mean(h_s)) if len(h_s) > 0 else float(np.max(h_k)) + 1e-10
            if h_mean_k < 1e-15:
                continue

            for eta in (0.05, 0.1, 0.2, 0.5):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                h_norm_k = h_k / max(h_mean_k, 1e-15)
                log_a = np.log(np.maximum(a, 1e-300))
                log_new = log_a - eta * (h_norm_k - 1.0)
                a_new = np.exp(log_new - np.max(log_new))
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new)
                sc = exact_score_a_safe(a_new)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved_this_iter = True

        if not improved_this_iter and it > 20:
            # Try support pruning: zero out very small entries
            if it > 50:
                thresh = a.max() * (1e-8 + 1e-6 * (it % 5))
                a_pruned = np.where(a > thresh, a, 0.0)
                if a_pruned.sum() > 1e-6:
                    a_pruned = normalize(a_pruned)
                    sc = exact_score_a_safe(a_pruned)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_pruned.copy()
                        a = a_pruned.copy()

    return float(best_sc), best_a


def subgrad_polish(a_start, max_iters, deadline, k_list=(1, 2, 3, 5, 8)):
    """Subgradient polish using exact max subgradient."""
    a = normalize(a_start)
    best_a = a.copy()
    best_sc = exact_score_a_safe(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        c = fast_self_conv(a)
        S = float(np.sum(a))
        d = (n * c) / (S * S)
        dmax = float(np.max(d))
        if dmax <= 0:
            break

        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            if k >= len(d):
                continue
            idx = np.argpartition(d, -k)[-k:]
            w = np.zeros(len(d))
            w[idx] = 1.0 / k

            h = cross_corr_weights(w, a)
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-15:
                continue

            grad = (2.0 * float(n) * h) / (S * S)
            dot_wc = float(np.dot(w, c))
            grad -= (2.0 * float(n) * dot_wc) / (S ** 2) * a
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

            for eta in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                a_new = a - eta * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new)
                sc = exact_score_a_safe(a_new)
                if np.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True

        if not improved and it > 10:
            break
    return float(best_sc), best_a


def eigen_refine_single_lag(a_start, lag, n_iters=100, deadline=None):
    """Refine using power iteration for a single active lag.
    At optimum with single active lag k: the matrix B_k (Hankel) has a
    satisfying the eigenvalue problem B_k * a = lambda * a on support."""
    a = normalize(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a_safe(a)

    # For lag k: (Ba)_j = sum_i B_{ji} a_i where B_{ji} = 1 if j+i=k
    # This is: (Ba)_j = a_{k-j} (if 0 <= k-j < n)
    # So h_j = a_{k-j}, and the eigenvalue condition is a_{k-j} = lambda * a_j
    # This means a_j = a_{k-2j} = a_{k-4j} = ... (geometric in indices)
    # For k=0: a_j = a_{-j} = 0 for j>0, so only a_0 can be nonzero.
    # For k>0: this creates a specific pattern.

    # Actually for the constrained problem (sum=1, a>=0), the solution is more complex.
    # Let's use the multiplicative update directly.
    w = np.zeros(2 * n - 1)
    if 0 <= lag <= 2 * n - 2:
        w[lag] = 1.0

    for it in range(int(n_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        h = cross_corr_weights(w, a)
        h = np.maximum(h, 1e-15)
        support = a > 1e-8
        if np.sum(support) == 0:
            break
        h_mean = float(np.mean(h[support]))
        if h_mean < 1e-15:
            break

        for blend in (0.2, 0.5, 0.8):
            a_new = a * (h / max(h_mean, 1e-15)) ** blend
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            sc = exact_score_a_safe(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()

    return float(best_sc), best_a


def perturb_sequence(a, rng, strength=0.2):
    """Generate a perturbed version of the sequence."""
    p = normalize(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = p * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        # Frequency perturbation
        Nf = next_power_of_two(n)
        A = np.fft.rfft(p, Nf)
        freqs = np.fft.rfftfreq(Nf, d=1.0 / Nf)
        mask = np.abs(freqs) < float(rng.uniform(0.1, 3.0))
        A[mask] *= (1.0 + strength * 0.3 * rng.standard_normal(np.sum(mask)))
        q = np.fft.irfft(A, Nf)[:n]
    elif mode == 3:
        # Local smoothing/denoising
        k = int(rng.integers(5, max(7, n // 500)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 4:
        # Targeted: reduce peak contributors
        c = fast_self_conv(p)
        S = float(np.sum(p))
        d = (n * c) / (S * S)
        idx = np.argsort(d)[-3:]
        w = np.zeros(len(d))
        w[idx] = 1.0 / len(idx)
        h = cross_corr_weights(w, p)
        hmax = float(np.max(h) + 1e-15)
        q = p * np.exp(-strength * 0.5 * h / hmax)
    else:
        # V-shape modulation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * 0.3 * np.abs(2.0 * x - 1.0)
        q = p * vshape

    return normalize(np.clip(np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None))


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    n = int(n)
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        if arr.sum() > 1e-12:
            inits.append(normalize(arr))

    if parent is not None and len(parent) > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    # Uniform
    add(np.ones(n))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Inverted V (peak in center)
    for alpha in (0.3, 0.5, 1.0, 2.0):
        add(1.0 + alpha * (1.0 - (2.0 * x - 1.0) ** 2))

    # Cosine
    for f in (1, 2, 3):
        for alpha in (0.2, 0.5, 1.0, 1.5, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Bimodal (peaks at edges)
    for width in (0.05, 0.1, 0.15, 0.2, 0.3):
        add(np.exp(-(((x - 0.0) / width) ** 2)) + np.exp(-(((x - 1.0) / width) ** 2)) + 0.01)
        add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)) + 0.01)

    # Smooth random (low-pass filtered noise)
    Nf = next_power_of_two(n)
    for pwr in (1.5, 2.0, 3.0, 4.0, 6.0):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** pwr)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + 0.8 * noise)

    # Power law
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(x ** p + (1.0 - x) ** p + 0.01)

    # Triangle waves
    for f in (1, 2, 3):
        add(1.0 + 0.5 * np.abs(2.0 * ((f * x) % 1.0) - 1.0))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: length={}".format(parent.size if parent is not None else 0))

    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = min(parent.size, 16384)
        n_primary = max(n_primary, 4096)
        # Round to even for symmetry
        if n_primary % 2 != 0:
            n_primary += 1

    # Starting point
    if parent is not None:
        a_start = normalize(resample_sequence(parent, n_primary))
    else:
        x = np.linspace(0.0, 1.0, n_primary)
        a_start = normalize(1.0 + 0.7 * np.abs(2.0 * x - 1.0))

    best_sc = exact_score_a_safe(a_start)
    best_seq = a_start.copy()
    print("Initial: score={:.8f}, n={}".format(best_sc, n_primary))

    # === Phase 1: LSE L-BFGS trajectory (30% of time) ===
    t1_end = t_start + 0.30 * total_budget
    print("Phase 1: LSE L-BFGS until t={:.0f}s".format(t1_end - t_start))

    a = normalize(a_start)
    beta_list = [500.0, 2000.0, 5000.0, 15000.0, 50000.0, 150000.0, 500000.0, 1500000.0, 5000000.0]
    time_per_beta = (t1_end - t_start) / len(beta_list)

    for beta in beta_list:
        if time.monotonic() > t1_end - 5.0:
            break
        beta_dead = min(time.monotonic() + time_per_beta, t1_end - 3.0)
        while time.monotonic() < beta_dead - 2.0:
            rem = float(beta_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, len(a))
            if mf < 50:
                break
            sc, a_new = run_lbfgs(a, beta, beta_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(a_new)
                print("  LSE beta={:.0f}: {:.8f}".format(beta, best_sc))
            a = normalize(a_new)

    print("After Phase 1: {:.8f}".format(best_sc))

    # === Phase 2: Adaptive Mirror Descent (30% of time) ===
    t2_end = t_start + 0.60 * total_budget
    print("Phase 2: Mirror Descent until t={:.0f}s".format(t2_end - t_start))

    a = normalize(best_seq)
    # Break into sub-iterations with decreasing eta
    sub_iters = 2000
    for sub in range(sub_iters):
        if time.monotonic() > t2_end - 3.0:
            break
        # Decay eta over time
        frac = float(sub) / sub_iters
        eta_max = 1.5 * (1.0 - 0.7 * frac)
        eta_min = 0.05 * (1.0 - 0.5 * frac)

        c = fast_self_conv(a)
        S = float(np.sum(a))
        d = (n_primary * c) / (S * S)
        dmax = float(np.max(d))
        if dmax <= 0:
            break

        # Adaptive active set
        threshold = dmax * (0.99 - 0.03 * frac)
        active_mask = d >= threshold
        if not np.any(active_mask):
            active_mask = np.zeros(len(d), dtype=bool)
            active_mask[np.argmax(d)] = True
        active_count = int(np.sum(active_mask))
        if active_count > 20:
            idx = np.argpartition(d, -20)[-20:]
            active_mask = np.zeros(len(d), dtype=bool)
            active_mask[idx] = True
            active_count = 20

        w = np.zeros(len(d))
        w[active_mask] = 1.0 / active_count
        h = cross_corr_weights(w, a)
        h = np.maximum(h, 0.0)

        support = a > 1e-7 * a.max()
        if np.sum(support) < 20:
            support = a > 1e-12
        h_supp = h[support]
        h_mean = float(np.mean(h_supp)) if len(h_supp) > 0 else float(np.max(h)) + 1e-10
        if h_mean < 1e-15:
            break

        h_norm = h / max(h_mean, 1e-15)
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta_max * (h_norm - 1.0)
        a_new = np.exp(log_new - np.max(log_new))
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        sc = exact_score_a_safe(a_new)
        if np.isfinite(sc):
            a = a_new
            if sc < best_sc:
                best_sc = sc
                best_seq = a_new.copy()
                if sub % 100 == 0:
                    print("  MD iter {}: {:.8f} (active={}, eta={:.3f})".format(
                        sub, best_sc, active_count, eta_max))

    print("After Phase 2: {:.8f}".format(best_sc))

    # === Phase 3: Subgradient polish + eigenvalue refine (15% of time) ===
    t3_end = t_start + 0.75 * total_budget
    print("Phase 3: Subgrad + eigen until t={:.0f}s".format(t3_end - t_start))

    a = normalize(best_seq)
    # Subgradient polish
    sc, a_new = subgrad_polish(a, max_iters=200, deadline=t3_end,
                               k_list=(1, 2, 3, 5, 8, 12, 20))
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = normalize(a_new)
        print("  Subgrad: {:.8f}".format(best_sc))
    a = normalize(best_seq)

    # Single-lag eigenvalue refinement
    for lag in (0, 1, 2, n_primary // 2, 2 * n_primary // 3):
        if time.monotonic() > t3_end - 5.0:
            break
        sc, a_new = eigen_refine_single_lag(a, lag, n_iters=50, deadline=t3_end)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = normalize(a_new)
            a = normalize(best_seq)
            print("  Eigen lag={}: {:.8f}".format(lag, best_sc))

    print("After Phase 3: {:.8f}".format(best_sc))

    # === Phase 4: More LSE at very high beta (10% of time) ===
    t4_end = t_start + 0.85 * total_budget
    print("Phase 4: High-beta LSE until t={:.0f}s".format(t4_end - t_start))

    a = normalize(best_seq)
    for beta in (5000000.0, 10000000.0, 20000000.0, 50000000.0):
        if time.monotonic() > t4_end - 5.0:
            break
        rem = float(t4_end - time.monotonic())
        if rem < 10.0:
            break
        mf = estimate_maxfun(rem * 0.75, len(a))
        if mf < 50:
            break
        sc, a_new = run_lbfgs(a, beta, t4_end, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = normalize(a_new)
            print("  LSE beta={:.0f}: {:.8f}".format(beta, best_sc))
        a = normalize(a_new)

    # === Phase 5: Perturbation restarts (8% of time) ===
    t5_end = t_start + 0.93 * total_budget
    print("Phase 5: Perturbations until t={:.0f}s".format(t5_end - t_start))

    n_perturb = 5
    chunk_time = (t5_end - t_start) / n_perturb
    for pi in range(n_perturb):
        if time.monotonic() > t5_end - 5.0:
            break
        chunk_dead = min(t_start + (pi + 1) * chunk_time, t5_end - 3.0)

        p = perturb_sequence(best_seq, rng, strength=float(rng.uniform(0.05, 0.20)))

        # Quick optimization: LSE + mirror descent
        for beta in (50000.0, 200000.0, 1000000.0):
            if time.monotonic() > chunk_dead - 3.0:
                break
            rem = float(chunk_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = int(min(200, max(50, (rem * 0.6) / estimate_eval_time(len(p)))))
            sc, p_new = run_lbfgs(p, beta, chunk_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(p_new)
                print("  Perturb {} LSE b={:.0f}: {:.8f}".format(pi, beta, best_sc))
            p = normalize(p_new)

        # Mirror descent on perturbed
        if time.monotonic() < chunk_dead - 3.0:
            sc, p_new = adaptive_mirror_descent(p, max_iters=100, deadline=chunk_dead,
                                                eta_range=(0.1, 0.3, 0.5, 0.8),
                                                k_range=(1, 2, 3, 5))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(p_new)
                print("  Perturb {} MD: {:.8f}".format(pi, best_sc))

    # === Phase 6: Multi-length exploration (4% of time) ===
    t6_end = t_start + 0.97 * total_budget
    print("Phase 6: Multi-length until t={:.0f}s".format(t6_end - t_start))

    other_lengths = []
    for nl in (4096, 8192, 12288, 16384, 20480):
        if nl != n_primary and nl not in other_lengths:
            other_lengths.append(nl)

    for nl in other_lengths[:2]:
        if time.monotonic() > t6_end - 10.0:
            break
        a_other = normalize(resample_sequence(best_seq, nl))
        # Quick LSE
        for beta in (5000.0, 50000.0, 500000.0):
            if time.monotonic() > t6_end - 8.0:
                break
            rem = float(t6_end - time.monotonic())
            if rem < 5.0:
                break
            mf = int(min(150, max(40, (rem * 0.5) / estimate_eval_time(nl))))
            sc, a_new = run_lbfgs(a_other, beta, t6_end, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(a_new)
                print("  Multi-length n={}: {:.8f}".format(nl, best_sc))
            a_other = normalize(a_new)
        # Quick MD
        if time.monotonic() < t6_end - 5.0:
            sc, a_new = adaptive_mirror_descent(a_other, max_iters=50, deadline=t6_end,
                                                eta_range=(0.2, 0.5, 1.0),
                                                k_range=(1, 2, 3, 5))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(a_new)
                print("  Multi-length MD n={}: {:.8f}".format(nl, best_sc))

    # === Phase 7: Final polish ===
    if time.monotonic() < deadline - 5.0:
        a = normalize(best_seq)
        sc, a_new = subgrad_polish(a, max_iters=50, deadline=deadline - 5.0,
                                   k_list=(1, 2, 3, 5))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = normalize(a_new)

        # Final mirror descent
        if time.monotonic() < deadline - 3.0:
            sc, a_new = adaptive_mirror_descent(best_seq, max_iters=50, deadline=deadline - 3.0,
                                                eta_range=(0.05, 0.1, 0.2),
                                                k_range=(1, 2, 3))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = normalize(a_new)

    # Final verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
