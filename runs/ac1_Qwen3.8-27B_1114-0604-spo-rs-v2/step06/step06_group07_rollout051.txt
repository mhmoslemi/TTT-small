"""
Step function optimization using LSE-smoothed gradient descent with
progressive smoothing, spectral equalization, and peak-directed refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def fienup_iter(a, n_iters=300, alpha_start=0.05, alpha_end=0.8):
    """Fienup iteration with geometric spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters))
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # LSE with stability
    shifted = beta * (c - c_max)
    max_shifted = float(shifted.max())
    e = np.exp(shifted - max_shifted)
    sum_e = float(e.sum())
    L_val = c_max + max_shifted / beta + float(np.log(sum_e)) / beta
    
    # Softmax weights
    p = e / sum_e
    
    # Gradient: grad_L[i] = 2 * sum_k p[k] * a[k-i]
    # Using cross-correlation via FFT
    M2 = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, M2)
    Afull = np.fft.rfft(a, M2)
    cross = np.fft.irfft(P * np.conj(Afull), M2)
    
    g = np.empty(n)
    g[0] = cross[0]
    if n > 1:
        g[1:] = cross[M2 - n + 1:M2][::-1]
    
    grad_L = 2.0 * g
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L_val / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shave(a, n_iters=2000, n_peaks=10, step_init=0.01, decay=0.99998):
    """Directly shave peaks by reducing contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]

"""
Step function optimization using LSE-smoothed gradient descent with progressive
smoothing, spectral equalization via Fienup iterations, peak-directed shaving,
and multi-stage L-BFGS refinement. Key innovations: (1) verified LSE gradient
computation via FFT cross-correlation, (2) progressive beta schedule from smooth
to sharp, (3) alternating spectral/time-domain refinement, (4) perturbation
restarts to escape local minima.

import numpy as np

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2*n*max(conv)/S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.8):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters - 1))
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_ellip(a, n_iters=300, beta_shape=2.0):

    a = np.maximum(a.astype(np.float64), 0.0).copy()

def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    f(a) = 2n * L(a) / S(a)^2
    where L(a) = cmax + (1/beta)*log(sum(exp(beta*(c_k - cmax))))
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    max_shifted = float(shifted.max())
    e = np.exp(shifted - max_shifted)
    sum_e = float(e.sum())
    if sum_e < 1e-300:
        return 1e10, np.zeros(n)

    L_val = c_max + max_shifted / beta + float(np.log(sum_e)) / beta
    p = e / sum_e

    # Gradient of L: dL/da_i = sum_k p[k] * dc_k/da_i
    # dc_k/da_i ≈ 2*a[k-i] (for 0<=k-i<n)
    # So dL/da_i ≈ 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlation(p, a) evaluated at appropriate lags.

    M2 = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, M2)
    Afull = np.fft.rfft(a, M2)
    cross = np.fft.irfft(P * np.conj(Afull), M2)

    # cross[d] = sum_m p[m] * a[m+d mod M2]
    # I want g[i] = sum_k p[k] * a[k-i] = cross at lag -i
    # For i=0: g[0] = cross[0]
    # For i>0: g[i] = cross[M2-i]
    g = np.empty(n)
    g[0] = cross[0]
    if n > 1:
        g[1:] = cross[M2 - n + 1:M2][::-1]

    grad_L = 2.0 * g

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L_val / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shave(a, n_iters=2000, n_peaks=5, step_init=0.01, decay=0.99998):
    """Peak-directed time-domain descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - 1.0) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.02)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_hybrid(a, n_iters=200, alpha_start=0.03, alpha_end=0.5,
                  n_peaks=5, step_peak=0.008):
    """Hybrid: alternate spectral shaping and peak-directed descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters))

        # Spectral shaping
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Peak-directed step every 2nd iteration
        if it % 2 == 0:
            A2 = np.fft.rfft(a, M_fft)
            c = np.fft.irfft(A2 * A2, M_fft)[:2 * n - 1]
            S2 = float(a.sum())
            M_val = float(c.max())

            if n_peaks >= len(c):
                top_idx = np.arange(len(c))
            else:
                top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

            sg = np.zeros(n)
            for k in top_idx:
                k = int(k)
                js = max(0, k - n + 1)
                je = min(k, n - 1)
                if je >= js:
                    sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

            S2v = S2 * S2
            S3v = S2v * S2
            grad = (2.0 * n) * (sg / S2v - 2.0 * M_val / S3v)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                step = min(step_peak, max((2.0 * n * M_val / S2v - 1.0) / (gn * gn + 1e-30), 1e-8))
                step = min(step, 0.015)
                a -= step * grad
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S < 1e-12:
                    break
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def conv_variance_obj(a):
    """Compute variance of convolution as secondary metric."""
    c = fast_conv(a)
    S = float(a.sum())
    return float(np.var(c)) / (S * S)


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(0.3 + 0.3 * np.cos(2.0 * np.pi * (x - 0.2)) + 0.3 * np.cos(2.0 * np.pi * (x - 0.8)))
    inits[-1] = np.maximum(inits[-1], 0.01)

    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2.0 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    inits.append(np.maximum(0.3 + 0.3 * x ** 0.5 + 0.3 * (1 - x) ** 0.5, 0.01))
    inits.append(np.maximum(0.3 * (1 + np.cos(2.0 * np.pi * x)), 0.01))

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.01))

    return inits


def resample_to(a, target_n):
    """Resample sequence to target length via interpolation."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Fienup spectral equalization from diverse starts
    Phase 2: Peak shaving + hybrid refinement
    Phase 3: Progressive L-BFGS with increasing beta
    Phase 4: Final polish and perturbation restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine working length: use power of 2 for FFT efficiency
    n_work = 8192

    # Initialize from parent
    a0 = None
    if parent is not None:
        # Evaluate parent at native length
        sc_parent = compute_score(parent)
        if sc_parent < best_score:
            best_score = sc_parent
            best_a = parent.copy()
            print(f"  [PARENT-NATIVE] score={sc_parent:.8f} (n={len(parent)})")

        # Resample to n_work
        a0 = resample_to(parent, n_work)
        S = a0.sum()
        if S > 0:
            a0 = a0 / S
        sc = compute_score(a0)
        print(f"  [PARENT-RESAMPLED] score={sc:.8f} (n={n_work})")
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    else:
        a0 = np.ones(n_work) * 0.5
        sc = compute_score(a0)
        print(f"  [UNIFORM] score={sc:.8f} (n={n_work})")

    def _try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Fienup spectral equalization (20% time)
    # ============================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (budget {phase1_end - t_start:.0f}s) ===")

    inits = build_diverse_inits(n_work, rng, parent)
    print(f"  {len(inits)} initializations")

    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a_init = np.maximum(a_init, 0.0)
        S = a_init.sum()
        if S < 1e-12:
            continue
        a_init = a_init / S
        sc_init = compute_score(a_init)
        print(f"  Init {i}: {sc_init:.6f}", end="")

        # Geometric Fienup
        a_g = fienup_geometric(a_init, n_iters=250, alpha_start=0.05, alpha_end=0.7)
        sc = compute_score(a_g)
        _try_update(sc, a_g, f"GEO i={i}")
        print(f" -> GEO {sc:.6f}", end="")

        # Elliptical (if time)
        if time.monotonic() < phase1_end:
            a_e = fienup_ellip(a_g, n_iters=150, beta_shape=2.5)
            sc_e = compute_score(a_e)
            _try_update(sc_e, a_e, f"ELLI i={i}")
            print(f" -> ELLI {sc_e:.6f}", end="")

        print()

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Peak shaving + hybrid (20% time)
    # ============================================================
    phase2_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Peak Shave + Hybrid (budget {phase2_end - t_start:.0f}s) ===")

    for n_peaks in [3, 5, 10, 20]:
        if time.monotonic() > phase2_end:
            break
        a_ps = peak_shave(best_a, n_iters=1500, n_peaks=n_peaks,
                          step_init=0.01, decay=0.99998)
        sc = compute_score(a_ps)
        _try_update(sc, a_ps, f"PK m={n_peaks}")

    if time.monotonic() < phase2_end:
        a_h = fienup_hybrid(best_a, n_iters=300, alpha_start=0.03, alpha_end=0.5,
                            n_peaks=5, step_peak=0.008)
        sc = compute_score(a_h)
        _try_update(sc, a_h, "HYBRID")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS with progressive beta (40% time)
    # ============================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS Progressive (budget {phase3_end - t_start:.0f}s) ===")

    betas = [30, 50, 100, 200, 500, 1000, 3000, 10000, 30000, 100000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  L-BFGS beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(8000, int(remaining * 15))
        maxfun = min(150000, maxiter * 20)
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        _try_update(sc, a_opt, f"LBFGS b={beta}")
        print(f"    -> score={sc:.8f}")

    # Run L-BFGS from parent (native length) if available
    if parent is not None and len(parent) != n_work:
        if time.monotonic() < phase3_end - 60:
            print(f"  L-BFGS from parent native length...")
            for beta in [100, 1000]:
                if time.monotonic() > phase3_end - 30:
                    break
                remaining = phase3_end - time.monotonic()
                if remaining < 30:
                    break
                maxiter = min(5000, int(remaining * 10))
                sc, a_opt = run_lbfgs(parent, beta, maxiter=maxiter, maxfun=maxiter * 15)
                _try_update(sc, a_opt, f"LBFGS-P b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (10% time)
    # ============================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Final Polish (budget {phase4_end - t_start:.0f}s) ===")

    # Short Fienup
    if time.monotonic() < phase4_end - 120:
        a_f = fienup_geometric(best_a, n_iters=100, alpha_start=0.02, alpha_end=0.3)
        sc = compute_score(a_f)
        _try_update(sc, a_f, "Final-GEO")

    # Short peak shave
    if time.monotonic() < phase4_end - 120:
        a_p = peak_shave(best_a, n_iters=800, n_peaks=3, step_init=0.005, decay=0.99999)
        sc = compute_score(a_p)
        _try_update(sc, a_p, "Final-PEAK")

    # High-beta L-BFGS
    if time.monotonic() < phase4_end - 60:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=40000)
        _try_update(sc, a_opt, "Final-LBFGS-HI")

    # Perturbation restarts
    for trial in range(4):
        if time.monotonic() > phase4_end - 20:
            break
        noise_scale = 0.005 * (1 + 0.5 * trial)
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_pert = a_pert.sum()
        if S_pert < 1e-12:
            continue
        a_pert = a_pert / S_pert

        # Quick Fienup + L-BFGS
        a_r = fienup_geometric(a_pert, n_iters=60, alpha_start=0.05, alpha_end=0.4)
        sc = compute_score(a_r)
        _try_update(sc, a_r, f"RESTART {trial}")

        if time.monotonic() < phase4_end - 30 and sc < best_score + 0.01:
            sc2, a_r2 = run_lbfgs(a_r, 10000, maxiter=2000, maxfun=25000)
            _try_update(sc2, a_r2, f"RESTART-{trial}-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    if best_a is None:
        best_a = np.ones(64) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
