"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Exponentiated gradient (EG) with adaptive full-set subgradient
2. LSE-smoothed L-BFGS-B for smooth phases
3. Active-set Nesterov with full peak tracking
4. Coordinate descent polishing
5. Multi-length exploration (up to 32768)
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    """Full self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    """Compute 2*n*max(c)/S^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _subgrad_full(a, tol=1e-7):
    """
    Compute subgradient of 2*n*max(c)/S^2 using ALL peaks within tol of max.
    Returns (score, grad).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)  # d = n*c/S^2, score = 2*max(d)
    dmax = float(d.max())
    score = 2.0 * dmax

    # Active set: all peaks within tol of max
    threshold = dmax * (1.0 - tol)
    active_mask = d >= threshold
    n_active = int(active_mask.sum())
    if n_active == 0:
        active_mask[np.argmax(d)] = True
        n_active = 1

    # Weights: uniform over active set
    w = np.zeros(2 * n - 1)
    w[active_mask] = 1.0 / n_active

    # Grad computation:
    # ∂f/∂a_i = (4n/S^2) * sum_l w[l] * a[l-i] - (4n/S^3) * sum_l w[l] * c[l]
    # where a is zero-extended
    
    # R[i] = sum_l w[l] * a[l-i] (zero-extended)
    # This is the cross-correlation of w and a, which we compute via FFT
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]  # R[i] = sum_l w[l] * a[l-i]

    T = float(w.dot(c))  # sum_l w[l] * c[l]
    grad = (4.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    # Gradient
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w.dot(c))
    grad = (4.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _eg_optimize(a0, deadline, n_max_iter=None, tol=1e-7):
    """
    Exponentiated gradient method with adaptive step size.
    Minimizes 2*n*max(c)/S^2 over a >= 0.
    """
    a = _normalize(a0)
    n = a.size
    if n_max_iter is None:
        n_max_iter = 50000

    best_a = a.copy()
    best_score = _exact_score(a)
    
    # Initial step size
    sc, g = _subgrad_full(a, tol)
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-14:
        return best_score, best_a
    
    eta = 2.0 / (g_norm * np.sqrt(n))
    
    no_improve = 0
    max_no_improve = 200
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Compute subgradient
        sc, g = _subgrad_full(a, tol)
        if not np.isfinite(sc):
            break
        
        # EG update: a_i *= exp(-eta * g_i)
        # Center the gradient to make the update more balanced
        g_centered = g - g.mean()
        step = np.exp(-eta * g_centered)
        a_new = a * step
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Evaluate
        sc_new = _exact_score(a_new)
        
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive step size
        if no_improve > max_no_improve:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-16:
                break
        
        # Progress logging
        if (it + 1) % 5000 == 0:
            print("  EG iter {}: score={:.10f} eta={:.2e}".format(it + 1, best_score, eta))
    
    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=max(100, 50)):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": max(maxfun, 100),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(np.clip(res.x, 0.0, None))
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out

def _coord_descend(a0, deadline, n_sweeps=50):
    """
    Coordinate descent: for each coordinate, find the best value.
    Since the objective is not separable, we do a 1D line search per coordinate.
    """
    a = _normalize(a0)
    n = a.size
    best_score = _exact_score(a)
    best_a = a.copy()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 1.0:
            break
        
        improved = False
        for i in range(n):
            # Try small perturbations of a[i]
            base = a[i]
            if base < 1e-12:
                # Try setting it to small values
                candidates = [0.0, 1e-4 / n, 1e-3 / n, 1e-2 / n]
            else:
                candidates = [0.0, base * 0.5, base * 0.75, base, base * 1.25, base * 1.5,
                              base * 2.0, 0.0]
            
            best_val = base
            best_sc = best_score
            
            for cv in candidates:
                a[i] = cv
                S = float(a.sum())
                if S < 1e-12:
                    a[i] = base
                    continue
                a[i] = cv  # Keep the value, we'll re-normalize later
                sc = _exact_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
            
            if best_val != base:
                a[i] = best_val
                best_score = best_sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_score, _normalize(best_a)


def _perturb(a, rng, strength=0.3):
    """Generate a perturbed version of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    mode = int(rng.integers(0, 5))

    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Smooth (low-freq) perturbation
        K = min(255, max(11, n // 50))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        sig = max(1.0, c / 3.0)
        kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 3:
        # Frequency-domain perturbation
        A = np.fft.rfft(a)
        A *= np.exp(0.1 * strength * rng.standard_normal(A.size))
        q = np.fft.irfft(A, n * 2)[:n]
    else:
        # Blend with a different structured sequence
        x = np.linspace(0, 1, n)
        f = int(rng.integers(1, 10))
        structured = 1.0 + 0.5 * np.cos(2 * np.pi * f * x)
        structured = np.clip(structured, 0.1, None)
        t = float(rng.uniform(0.2, 0.6))
        q = (1 - t) * a + t * structured / structured.sum() * a.sum()

    return _normalize(np.clip(q, 0.0, None))


def _generate_inits(n, rng, parent, cap=20):
    """Generate diverse initial sequences of length n."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, arr.size), arr)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    x = np.linspace(0, 1, n)

    # Parent-derived
    if parent is not None:
        pr = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        _add(pr)
        _add(pr[::-1])

    # Uniform
    _add(np.ones(n))

    # Windowed oscillations (key for low autocorrelation peak)
    for f in [3, 5, 7, 11, 13, 17, 23, 31]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            _add(1.0 + amp * np.cos(2 * np.pi * f * x))
            _add(1.0 + amp * np.sin(2 * np.pi * f * x + 0.5))
            _add(1.0 + amp * np.cos(2 * np.pi * f * x) * np.sin(np.pi * x))

    # V-shapes and smooth windows
    for alpha in [0.2, 0.5, 0.8, 1.5]:
        _add(1.0 + alpha * np.abs(2 * x - 1))
        _add(1.0 + alpha * (1 - np.abs(2 * x - 1)))
        _add(np.exp(-alpha * (x - 0.5) ** 2))

    # Bimodal / multimodal
    for w in [0.1, 0.15, 0.2, 0.3]:
        for ctr in [0.2, 0.25, 0.3, 0.35]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n / 10 + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)

    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    # Chirp-like
    for rate in [5, 10, 20, 50]:
        _add(1.0 + 0.5 * np.cos(np.pi * rate * x ** 2))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: minimize 2*n*max(conv(a,a))/sum(a)^2.
    Uses EG + LSE-LBFGS + Nesterov + coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Determine lengths
    lengths = []
    if parent is not None:
        pn = parent.size
        if pn >= 8192:
            lengths = [pn, min(pn * 2, 32768), 16384]
        else:
            lengths = [16384, 32768, max(pn, 8192)]
    else:
        lengths = [16384, 32768, 8192]

    # Deduplicate and filter
    lengths = sorted(set(L for L in lengths if 4096 <= L <= 65536), reverse=True)
    # Ensure we have at least 2
    while len(lengths) < 2:
        lengths.append(16384)
    lengths = lengths[:4]
    print("Lengths to try: {}".format(lengths))

    # Phase 1: Generate and evaluate initializations
    all_scores = []
    all_seqs = []

    for n in lengths:
        inits = _generate_inits(n, rng, parent, cap=15)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)

    if not all_scores:
        all_seqs.append(np.ones(16384) / 16384)
        all_scores.append(2.0)

    order = np.argsort(all_scores)
    print("Best init: {:.8f}".format(all_scores[order[0]]))

    # Select diverse starts (different lengths)
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 4:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main optimization (60% of time)
    main_end = t_start + 0.60 * (budget_s - 60.0)
    main_end = min(main_end, deadline - 200.0)

    # Sort starts by score
    starts_scores = [_exact_score(s) for s in starts]
    starts_sorted = sorted(range(len(starts)), key=lambda i: starts_scores[i])

    # Time allocation for each start
    n_starts = len(starts_sorted)
    fracs = []
    for i in range(n_starts):
        fracs.append(1.0 / (i + 1) - (1.0 / (i + 2)) if i < n_starts - 1 else 1.0)
    # Simpler: first gets 50%, rest split
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.6, 0.4]
    elif n_starts == 3:
        fracs = [0.5, 0.3, 0.2]
    else:
        fracs = [0.45, 0.25, 0.15, 0.15]

    cum = 0.0
    for rank, si in enumerate(starts_sorted):
        if time.monotonic() > main_end - 5.0:
            break
        cum += fracs[rank]
        sub_dead = min(t_start + (cum / sum(fracs)) * (main_end - t_start), main_end)
        sub_dead = min(sub_dead, deadline - 100.0)

        if time.monotonic() > sub_dead - 10.0:
            continue

        sq = starts[si].copy()

        # For secondary starts, blend with best
        if rank > 0 and len(best_seq) != len(sq):
            best_resampled = np.interp(
                np.linspace(0, 1, len(sq)),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            sq = _normalize(0.6 * _normalize(sq) + 0.4 * _normalize(best_resampled))

        print("\n[Start {}] n={} initial={:.8f}".format(rank, len(sq), _exact_score(sq)))

        sc, a2 = _main_optimize(sq, sub_dead, rank)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("[Start {}] improved to {:.8f}".format(rank, best_score))

    print("\nAfter main optimization: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts (20% of time)
    perturb_end = t_start + 0.80 * (budget_s - 60.0)
    if time.monotonic() < perturb_end - 30.0:
        n_perturb = 4
        chunk = (perturb_end - time.monotonic()) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = time.monotonic() + chunk
            if chunk_dead > perturb_end:
                chunk_dead = perturb_end
            if time.monotonic() > chunk_dead - 5.0:
                break

            p = _perturb(best_seq, rng, strength=0.2 + 0.1 * pi)
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 4: Final intensive polish (remaining time)
    final_dead = deadline - 10.0
    if time.monotonic() < final_dead - 20.0:
        print("\nFinal polish phase...")
        sc, a2 = _final_polish(best_seq, final_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("After polish: {:.8f}".format(best_score))

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
