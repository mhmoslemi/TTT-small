Multi-scale optimizer with direct equalization, LSE-LBFGS, and active-set
exponentiated gradient for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
import math


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_conv(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    M = n + m - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_vec(a, w):
    """Compute correlation: result[i] = sum_j a[j] * w[i-j] (valid range)

    conv[k] = sum_i a[i]*a[k-i]

    This is a shifted/reversed version of a.
    
    For multiple k values, we need: for each i, sum over active k of a[k-i] * weight_k
    = sum_k weight_k * a[k-i]
    
    This is a cross-correlation of a with the weight vector (placed at positions k).
    """

    w = np.asarray(w, dtype=np.float64)

    m = w.size

    # But we need the full range where a is defined

    # So grad[i] = a[k-i] for the specific k.

    # = sum_k w_k * a[k-i]

    # = (w * reverse(a))[i + n - 1] ... 
    
    # Simplest: use FFT
    # corr[i] = sum_k w[k] * a[k-i]
    # Let's define: for each i, corr[i] = sum_k w[k] * a[k-i]
    # This is the cross-correlation of w and a: corr[i] = <w, shift_i(a)>
    # = <shift_{-i}(w), a> = (a * reverse(w))[n-1-i + offset]...

def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-15:
        return np.ones(a.size) / a.size
    return a / S


def score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def active_set_grad(a, k_active):
    """Compute gradient of max(conv) using active set k_active.
    Returns gradient vector of length n.
    grad[i] = mean of a[k-i] over k in k_active (where valid)
    """
    n = a.size
    m = len(k_active)
    # For each active k, the gradient is: g[i] = a[k-i] if 0<=k-i<=n-1
    # Average over active set:
    grad = np.zeros(n, dtype=np.float64)
    for k in k_active:
        # a[k-i] is valid for i in max(0,k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1) + 1
        if i_start < i_end:
            # a[k-i] for i in i_start..i_end-1 is a[k-i_end+1..k-i_start]
            vals = a[k - (i_end - 1): k - i_start + 1][::-1]
            grad[i_start:i_end] += vals
    grad /= m
    return grad


def direct_equalization(a, top_m=5, eta=0.1):
    """Directly equalize the top-m convolution values.
    Find δ that minimizes |Gδ - target| subject to sum(δ)=0.
    """
    n = a.size
    conv = fast_conv(a)
    # Get top m positions
    idx = np.argsort(conv)[::-1][:top_m]
    c_top = conv[idx]
    mean_c = c_top.mean()
    
    # Build G matrix: (top_m + 1) x n
    # Row j: g_j[i] = a[idx[j] - i] if valid, else 0
    # Last row: ones (sum constraint)
    G = np.zeros((top_m + 1, n), dtype=np.float64)
    for j, k in enumerate(idx):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1) + 1
        if i_start < i_end:
            G[j, i_start:i_end] = a[k - (i_end - 1): k - i_start + 1][::-1]
    G[top_m, :] = 1.0
    
    # Target: for j < top_m: mean_c - conv[k_j], for j = top_m: 0
    target = np.zeros(top_m + 1)
    for j in range(top_m):
        target[j] = mean_c - conv[idx[j]]
    
    # Solve G G^T y = target, then δ = G^T y
    GtG = G @ G.T  # (m+1) x (m+1)
    try:
        y = np.linalg.solve(GtG + 1e-10 * np.eye(top_m + 1), target)
    except np.linalg.LinAlgError:
        y = np.linalg.lstsq(GtG, target, rcond=None)[0]
    delta = G.T @ y
    
    # Apply
    a_new = a + eta * delta
    a_new = np.clip(a_new, 0.0, None)
    return normalize(a_new)


def lse_objective_grad(a, beta):
    """LSE-smoothed objective: (n/beta)*log(sum exp(beta*conv[k]*(n/S^2)))
    Actually, let's work with normalized a (sum=1), objective = 2n * max(conv)
    LSE approx: (2n/beta) * log(sum_k exp(beta * conv[k]))
    """
    n = a.size
    conv = fast_conv(a)
    z = beta * conv
    z = z - z.max()  # for numerical stability
    exp_z = np.exp(z)
    s = exp_z.sum()
    if s < 1e-300:
        return 2.0 * n * float(conv.max()), np.zeros(n)
    # Objective value (approximation to 2n*max(conv)):
    val = 2.0 * n * (float(conv.max()) + float(np.log(s)) / beta)
    
    # Gradient: (2n/beta) * sum_k exp_z[k]/s * grad(conv[k])
    w = exp_z / s  # (2n-1,)
    # grad(conv[k])/da[i] = a[k-i] if valid
    # grad_objective[i] = (2n/beta) * sum_k w[k] * a[k-i]
    # = (2n/beta) * corr(a, w)[i]
    
    # Compute: result[i] = sum_k w[k] * a[k-i]
    # This is the "odd correlation" - let's compute via FFT
    # corr[i] = sum_k w[k] * a[k-i]
    
    n_conv = 2 * n - 1
    N = next_pow2(n_conv)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    full = np.fft.irfft(W * A_rev, N)[:n_conv + n - 1]
    # corr[i] = full[n-1+i] for i=0..n-1
    corr = full[n-1:n-1+n]
    
    grad = (2.0 * n / beta) * corr
    return val, grad


def lse_lbfgs(a, deadline, betas=[100, 500, 2000, 10000, 50000]):
    """Run LSE-LBFGS with increasing beta."""
    from scipy.optimize import minimize
    a = normalize(a)
    best_a = a.copy()
    for beta in betas:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        a = normalize(best_a)
        n = a.size
        maxfun = max(50, int(remaining * 0.6 / 0.1))
        maxfun = min(maxfun, 3000)
        
        try:
            res = minimize(
                lse_objective_grad, a, method='L-BFGS-B', jac=True,
                args=(float(beta),),
                bounds=[(0.0, 1e6)] * n,
                options={'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': maxfun // 2,
                         'maxfun': maxfun, 'maxls': 30}
            )
            a_new = normalize(res.x) if res.x is not None else a
        except Exception:
            a_new = a
        
        sc = score(a_new)
        if np.isfinite(sc) and sc < score(best_a):
            best_a = a_new
    
    return best_a


def exp_guided_step(a, conv, top_k, eta):
    """Exponentiated gradient step focusing on top-k conv positions."""
    n = a.size
    idx = np.argsort(conv)[::-1][:top_k]
    grad = active_set_grad(a, idx)
    # a_new[i] = a[i] * exp(-eta * grad[i] / max(grad))
    gmax = grad.max()
    if gmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300) - eta * grad / gmax
    log_a -= log_a.max()
    return np.exp(log_a)


def resample(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    new_a = np.interp(x_new, x_old, a)
    return np.clip(new_a, 0, None)

                if time.monotonic() > deadline:
                    break
                obj, grad = lse_objective_grad(a, beta)
                # Simple gradient descent step
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    continue
                for eta in [0.01, 0.05, 0.1, 0.5]:
                    a_try = a - eta * grad / gnorm
                    a_try = np.clip(a_try, 0, None)
                    a_try = normalize(a_try)
                    s_try = score(a_try)
                    if s_try < best_sc - 1e-12:
                        best_sc = s_try
