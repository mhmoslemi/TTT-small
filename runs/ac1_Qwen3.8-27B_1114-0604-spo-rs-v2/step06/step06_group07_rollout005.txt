def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]

    # Correct gradient: dL/dx_i = (2/beta) * sum_k p_k * x_{k-i}
    # conv_res[i] = sum_k p_k * x_{k-i}
    # So dL/dx_i = (2/beta) * conv_res[i]
    g_L = (2.0 / beta) * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_fn_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_subgrad(a, n_iters=5000, momentum_init=0.5, momentum_end=0.9,
                     step_init=0.01, step_decay=0.9999):
    """Nesterov-accelerated subgradient method for min max(conv(a,a)) s.t. a>=0."""
    n = len(a)
    y = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(y.sum())
    if S < 1e-12:
        y = np.ones(n)
        S = 1.0
    y = y / S

    v = np.zeros(n)
    momentum = momentum_init
    step = step_init
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = y.copy()
    best_sc = compute_score(y)
    sc_lb = 2.0 * n / M

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        momentum = momentum_init + (momentum_end - momentum_init) * progress
        step = step_init * (step_decay ** it)

        c = fast_conv(y)
        S = float(y.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Subgradient of max(c) w.r.t. y
        # For each peak at position k: dc_k/dy_i = 2*y_{k-i}
        top_k = np.where(c >= 0.95 * M_val)[0]
        if len(top_k) > 200:
            top_k = np.argpartition(c, -200)[-200:]

        sg = np.zeros(n)
        for k in top_k:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * y[js:je + 1][::-1]

        # Normalize subgradient to have comparable scale
        # The objective is 2n*max/S^2, gradient includes the 2n/S^2 factor
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Nesterov update
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        S_new = float(x_new.sum())
        if S_new < 1e-12:
            break
        x_new = x_new / S_new

        v = momentum * v + (1.0 - momentum) * (x_new - y)
        y = x_new + momentum * v
        y = np.maximum(y, 0.0)
        S = float(y.sum())
        if S < 1e-12:
            break
        y = y / S

    return best_a


def peak_coordinate_descent(a, n_iters=5000, n_peaks=10, step=0.01, decay=0.99995):
    """Coordinate descent targeting top peaks of convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - sc_lb < 1e-10:
            break

        # Adaptive number of peaks
        effective_peaks = min(n_peaks, max(3, int(n_peaks * (1 + 0.5 * np.sin(it * 0.01)))))

        top_idx = np.argpartition(c, -effective_peaks)[-effective_peaks:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        adaptive_step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        adaptive_step = min(adaptive_step, 0.03)

        a -= adaptive_step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_shaped(a, n_iters=300, alpha_start=0.05, alpha_end=0.9,
                  shape_type='geometric', shape_param=1.0):
    """Fienup iteration with configurable spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)

        if shape_type == 'geometric':
            amp_new = amp_safe * (C / amp_safe) ** alpha
        elif shape_type == 'power_law':
            amp_new = amp.copy()
            above = amp > C * 1.001
            if above.any():
                amp_new[above] = amp[above] * (C / amp[above]) ** (shape_param * alpha)
        elif shape_type == 'elliptical':
            ratio = amp / C
            amp_new = amp / np.sqrt(1.0 + shape_param * (ratio - 1.0) ** 2)
        elif shape_type == 'soft_threshold':
            threshold = C * (1.0 - alpha * 0.5)
            amp_new = np.maximum(amp - threshold * 0.1, 0.0)
            amp_new = np.maximum(amp_new, amp * 0.3)
        else:
            amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent=None):
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)

    freqs = rng.integers(1, 10, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.5, size=5)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))

    # Random smooth noise
    noise = rng.standard_normal(n)
    # Smooth it
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    smooth_noise = np.convolve(noise, kernel, mode='same')
    inits.append(np.maximum(0.5 + 0.3 * smooth_noise / (np.abs(smooth_noise).max() + 1e-10), 0.01))

    # Triangle wave
    inits.append(np.maximum(0.3 + 0.5 * np.abs(2.0 * (x % 0.5) - 0.5), 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits


def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    n_work = len(parent) if parent is not None else 8192
    print(f"Working length: {n_work}, parent: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ============================================================
    # Phase 1: Quick spectral shaping diverse starts (15% time)
    # ============================================================
    phase1_end = t_start + 0.15 * (total_budget - 40.0)
    print(f"\n=== Phase 1: Spectral Shaping (deadline {phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n_work, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        sc0 = compute_score(a0)
        print(f"  Init {i}: score={sc0:.6f}")

        a_g = fienup_shaped(a0, n_iters=200, shape_type='geometric',
                            alpha_start=0.05, alpha_end=0.85)
        sc = compute_score(a_g)
        update_best(sc, a_g, f"GEO i={i}")

        if time.monotonic() < phase1_end:
            a_e = fienup_shaped(a0, n_iters=150, shape_type='elliptical',
                                shape_param=2.0, alpha_start=0.1, alpha_end=0.7)
            sc = compute_score(a_e)
            update_best(sc, a_e, f"ELLI i={i}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Nesterov subgradient on non-smooth max (20% time)
    # ============================================================
    phase2_end = t_start + 0.35 * (total_budget - 40.0)
    print(f"\n=== Phase 2: Nesterov Subgradient (deadline {phase2_end - t_start:.0f}s) ===")

    a_ns = nesterov_subgrad(best_a, n_iters=3000, momentum_init=0.3,
                            momentum_end=0.85, step_init=0.008, step_decay=0.99998)
    sc = compute_score(a_ns)
    update_best(sc, a_ns, "NESTEROV")

    if time.monotonic() < phase2_end:
        a_ns2 = nesterov_subgrad(best_a + 0.01 * rng.standard_normal(n_work),
                                  n_iters=2000, momentum_init=0.5,
                                  momentum_end=0.9, step_init=0.005, step_decay=0.99995)
        sc = compute_score(a_ns2)
        update_best(sc, a_ns2, "NESTEROV-2")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Peak-directed coordinate descent (15% time)
    # ============================================================
    phase3_end = t_start + 0.50 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Peak Coordinate Descent (deadline {phase3_end - t_start:.0f}s) ===")

    for n_peaks in [5, 10, 20, 50]:
        if time.monotonic() > phase3_end:
            break
        print(f"  n_peaks={n_peaks}")
        a_pd = peak_coordinate_descent(best_a, n_iters=3000, n_peaks=n_peaks,
                                       step=0.01, decay=0.99997)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS with beta annealing (35% time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 40.0)
    print(f"\n=== Phase 4: L-BFGS Beta Annealing (deadline {phase4_end - t_start:.0f}s) ===")

    # Anneal beta from low to high
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 90:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(5000, int(remaining * 20))
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if not improved:
            # Try from parent too if we haven't improved
            if parent is not None and len(parent) == len(best_a) and beta <= 500:
                if time.monotonic() < phase4_end - 60:
                    sc2, a_opt2 = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
                    update_best(sc2, a_opt2, f"LBFGS-P b={beta}")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish with mixed methods (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Spectral shaping on best
    if time.monotonic() < deadline - 90:
        a_f = fienup_shaped(best_a, n_iters=200, shape_type='geometric',
                            alpha_start=0.02, alpha_end=0.5)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FINAL-GEO")

    # More Nesterov
    if time.monotonic() < deadline - 90:
        a_f2 = nesterov_subgrad(best_a, n_iters=2000, momentum_init=0.7,
                                momentum_end=0.95, step_init=0.003, step_decay=0.99999)
        sc = compute_score(a_f2)
        update_best(sc, a_f2, "FINAL-NEST")

    # Peak descent with small steps
    if time.monotonic() < deadline - 60:
        a_f3 = peak_coordinate_descent(best_a, n_iters=3000, n_peaks=15,
                                       step=0.003, decay=0.99999)
        sc = compute_score(a_f3)
        update_best(sc, a_f3, "FINAL-PEAK")

    # Final L-BFGS at high beta
    if time.monotonic() < deadline - 60:
        sc, a_opt = run_lbfgs(best_a, 30000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # Perturbation restarts
    for trial in range(4):
        if time.monotonic() > deadline - 30:
            break
        noise_scale = 0.01 * (1 + trial * 0.5)
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = a_pert.sum()
        if S < 1e-12:
            break
        a_pert = a_pert / S

        a_r = peak_coordinate_descent(a_pert, n_iters=1000, n_peaks=10,
                                      step=0.005, decay=0.99998)
        sc = compute_score(a_r)
        update_best(sc, a_r, f"RESTART {trial}")

        if time.monotonic() < deadline - 30:
            a_r2 = fienup_shaped(a_pert, n_iters=100, shape_type='geometric',
                                 alpha_start=0.05, alpha_end=0.6)
            sc = compute_score(a_r2)
            update_best(sc, a_r2, f"RESTART-GEO {trial}")

    # Final verification pass
    if time.monotonic() < deadline - 20:
        sc_final = compute_score(best_a)
        if sc_final < best_score:
            best_score = sc_final
            print(f"  Verification improved: {sc_final:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
