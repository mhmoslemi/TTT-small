    n = len(a)
    if x_init is not None and len(x_init) == n:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n)
    u = x.copy()
    t_mom = 1.0

    # Estimate Lipschitz constant
    # Hessian of (1/beta)*log(sum exp(beta*(Bx-M))) has eigenvalues bounded by
    # beta * ||B^T diag(p) B|| / ... 
    # Simple bound: L = beta * n * max(a)^2 * 2 (rough)
    # Better: L = beta * ||B||^2 where ||B|| is spectral norm ~ n * max(a)
    # So L ~ beta * (n * max(a))^2
    # But we want to be conservative for the first few iters
    max_a = float(np.max(a))
    L_est = beta * 4.0 * n * max_a * max_a
    step = 1.0 / (L_est + 1e-15)

    n_c = 2 * n - 1
    M_cross = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        # Gradient = B^T p
        FA = np.fft.rfft(a, M_cross)
        FP = np.fft.rfft(p, M_cross)
        corr = np.fft.irfft(np.conj(FA) * FP, M_cross)[:n]

        # FISTA update
        v = u - step * corr
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def golden_section_mix(a, x, best_sc, n_evals=20):
    """Golden section search on t in [0,1] for a_new = (1-t)*a + t*x."""
    phi = (np.sqrt(5) - 1) / 2  # ~0.618
    lo, hi = 0.0, 1.0
    a1 = hi - phi * (hi - lo)
    a2 = lo + phi * (hi - lo)
    
    def score_at(t):
        mixed = (1.0 - t) * a + t * x
        mixed = np.maximum(mixed, 0.0)
        S = float(mixed.sum())
        if S < 1e-12:
            return np.inf
        mixed = mixed / S
        return compute_score(mixed)
    
    sc_a1 = score_at(a1)
    sc_a2 = score_at(a2)
    
    for _ in range(n_evals):
        if sc_a1 < sc_a2:
            hi = a2
            a2 = a1
            sc_a2 = sc_a1
            a1 = hi - phi * (hi - lo)
            sc_a1 = score_at(a1)
        else:
            lo = a1
            a1 = a2
            sc_a1 = sc_a2
            a2 = lo + phi * (hi - lo)
            sc_a2 = score_at(a2)
    
    t_opt = (lo + hi) / 2
    mixed = (1.0 - t_opt) * a + t_opt * x
    mixed = np.maximum(mixed, 0.0)
    S = float(mixed.sum())
    if S < 1e-12:
        return best_sc, a.copy(), t_opt
    mixed = mixed / S
    return compute_score(mixed), mixed, t_opt


def lp_alternating_improved(a, max_outer, fista_iters, beta, time_limit, x_warm=None):
    """LP alternating with golden section mixing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        sc_new, a_new, t_opt = golden_section_mix(a, x, best_sc, n_evals=15)

            best_sc = sc_new
            best_a = a_new.copy()

        else:

            # Try direct replacement
            sc_direct = compute_score(x)
            if sc_direct < best_sc:
                best_sc = sc_direct

def error_reduction(a, n_iters=100, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0
        
        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def simulated_annealing_lp(a, rng, n_iters=50, T_start=0.1, T_end=0.001,
                            fista_iters=8000, beta=300.0, time_limit=None):
    """Simulated annealing where each "move" is an LP step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S > 0:
        a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        
        c = self_conv(a)
        M = float(c.max())
        x = fista_lp_solve(a, M, n_iters=fista_iters, beta=beta, x_init=a.copy())
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Mix with random t
        t = rng.uniform(0.2, 0.8)
        a_try = (1.0 - t) * a + t * x
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            continue
        a_try = a_try / St
        sc_try = compute_score(a_try)
        
        if sc_try < cur_sc or rng.random() < np.exp(-(sc_try - cur_sc) / max(T, 1e-10)):
            a = a_try
            cur_sc = sc_try
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_try.copy()
    
    return best_a, best_sc


def build_initializations(n, rng, parent=None):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 * np.ones(n))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2, 0.01))
    inits.append(np.maximum(0.5 + 0.3 * rng.standard_normal(n), 0.01))
    smooth = uniform_filter1d(rng.standard_normal(n), size=max(3, n // 50))
    inits.append(np.maximum(0.5 + 0.3 * smooth, 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.sin(np.pi * x) ** 2, 0.01))
    # U-shaped (high at edges)
    inits.append(0.3 + 0.7 * (2.0 * x - 1.0) ** 2)
    # Peak in middle
    inits.append(np.clip(1.0 - 1.5 * np.abs(2.0 * x - 1.0), 0.01, 2.0))
    return inits

Key insight: the FISTA step size should be:

- For B to be (2n-1) x n: B[k,j] = a[k-j] if 0 <= k-j < n, else 0.
- For a fixed j, k ranges from j to j+n-1, so there are n values of k for which B[k,j] != 0.
