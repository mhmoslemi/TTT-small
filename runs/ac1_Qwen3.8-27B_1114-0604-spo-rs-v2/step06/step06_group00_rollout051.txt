"""
LP-guided + spectral + LSE hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses LP-based direction finding (cutting plane), Fienup spectral shaping,
and LSE annealing for final polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def build_conv_constraint_matrix(a, n, k_values):
    """Build sparse matrix A where A[i,j] = a[k_i - j] if 0 <= k_i - j < n else 0."""
    rows, cols, vals = [], [], []
    a_ext = np.zeros(2*n-1)
    a_ext[:n] = a  # a[k-j] where k-j in [0, n-1]
    for i, k in enumerate(k_values):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                idx = k - j
                if 0 <= idx < n:
                    rows.append(i)
                    cols.append(j)
                    vals.append(a[idx])
    return coo_matrix((vals, (rows, cols)), shape=(len(k_values), n))


def solve_direction_lp(a, M, top_k=200, extra_constraints=0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0.
    Returns the optimal g and its sum.
    """
    n = len(a)
    c = fast_conv(a)
    
    # Select top k_values by convolution value
    sorted_idx = np.argsort(c)[::-1]
    k_values = sorted_idx[:top_k]
    
    # Optionally add extra constraints (random spread)
    if extra_constraints > 0:
        remaining = np.setdiff1d(np.arange(len(c)), k_values)
        extra = np.random.choice(remaining, min(extra_constraints, len(remaining)), replace=False)
        k_values = np.concatenate([k_values, extra])
    
    A_mat = build_conv_constraint_matrix(a, n, k_values)
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(len(k_values), M)
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True, "time_limit": 60})
        if res.status == 0:
            return res.x, -res.fun
        elif res.status == 3:  # time limit
            return res.x, -res.fun if res.fun != 0 else None
        else:
            return None, None
    except Exception as e:
        print(f"  LP error: {e}")
        return None, None


def lp_direction_update(a, n_iters=50, top_k=200, t_initial=0.1, t_decay=0.98):
    """Iteratively apply LP direction updates."""
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    t = t_initial
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        S = float(a.sum())
        sc = 2.0 * len(a) * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        g, g_sum = solve_direction_lp(a, M, top_k=top_k)
        if g is None or g_sum is None:
            break
        
        g = np.maximum(g, 0)
        g_sum_actual = g.sum()
        if g_sum_actual < 1e-10:
            break
        
        # Normalize g to have same sum as a
        g_norm = g / g_sum_actual
        
        # Update
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.01:
            # Regression, reduce step
            t *= 0.5
            continue
        
        a = a_new
        t = max(0.01, t * t_decay)
    
    return best_a


def fienup_flat(a, n_iters=200):
    n = len(a)
    a = normalize(a.copy())

"""
LP-guided direction finding + Fienup + LSE hybrid optimizer.
Key innovation: uses linear programming to find optimal descent directions
that simultaneously respect all peak constraints, avoiding the "whack-a-mole"
problem of naive peak shaving. Combined with spectral shaping and LSE refinement.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def build_conv_matrix_sparse(a, n, k_values):
    """Build sparse (len(k_values) x n) matrix A where A[i,j] = a[k_i - j] if valid."""
    k_arr = np.asarray(k_values, dtype=int)
    rows_list, cols_list, vals_list = [], [], []
    for i in range(len(k_arr)):
        k = k_arr[i]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            idxs = k - js
            valid = (idxs >= 0) & (idxs < n)
            if valid.any():
                rows_list.extend((i,) * int(valid.sum()))
                cols_list.extend(js[valid].tolist())
                vals_list.extend(a[idxs[valid]].tolist())
    return coo_matrix((vals_list, (rows_list, cols_list)), shape=(len(k_values), n))


def solve_lp_direction(a, M, top_k=200, extra_k=0, time_limit=45):
    """
    Solve primal LP: max sum(g) s.t. (a*g)[k_idx] <= M, g >= 0.
    Returns (g, objective_value) or (None, None).
    """
    n = len(a)
    c = fast_conv(a)
    sorted_idx = np.argsort(c)[::-1]
    k_values = sorted_idx[:top_k]
    if extra_k > 0:
        remaining = np.setdiff1d(np.arange(len(c)), k_values)
        if len(remaining) >= extra_k:
            extra = np.random.choice(remaining, extra_k, replace=False)
            k_values = np.concatenate([k_values, extra])

    A_mat = build_conv_matrix_sparse(a, n, k_values)
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_values), M)

    try:
        res = linprog(c_obj, A_ub=A_mat.tocsc(), b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={"presolve": True, "time_limit": time_limit,
                               "dual_feasibility_tolerance": 1e-8})
        if res.status == 0 or res.status == 3:
            g = np.maximum(res.x, 0.0)
            return g, float(g.sum())
    except Exception:
        pass
    return None, None


def lp_direction_search(a, n_iters=80, top_k=200, t_init=0.15):
    """Iteratively apply LP-guided direction updates."""
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    t = t_init
    stagnation = 0

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        S = float(a.sum())
        sc = 2.0 * len(a) * M / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 15:
            break

        # Solve LP for direction
        g, g_sum = solve_lp_direction(a, M, top_k=top_k, extra_k=min(50, top_k // 4))
        if g is None or g_sum < 1e-8:
            break

        g_norm = g / g_sum

        # Try multiple step sizes
        best_t = t
        best_a_new = a
        best_sc_new = sc
        for trial_t in [t, t * 0.5, t * 0.25]:
            a_candidate = (1.0 - trial_t) * a + trial_t * g_norm
            a_candidate = np.maximum(a_candidate, 0.0)
            S_cand = a_candidate.sum()
            if S_cand < 1e-12:

            a_candidate = a_candidate / S_cand
            sc_cand = compute_score(a_candidate)
            if sc_cand < best_sc_new:

                best_a_new = a_candidate
                best_t = trial_t

        a = best_a_new
        if best_sc_new >= sc:

        else:

def fienup_spectral(a, n_iters=300, mode='flat'):
    """Fienup-type spectral shaping with non-negativity enforcement."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if mode == 'flat':
            alpha = 0.1 + 0.7 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean
        elif mode == 'quantile':
            q = 0.65 + 0.3 * progress
            thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - progress))
            P_new = np.minimum(P, thresh)
        else:  # 'hybrid'
            if it % 3 == 0:
                alpha = 0.15 + 0.5 * progress
                P_new = (1.0 - alpha) * P + alpha * P_mean
            elif it % 3 == 1:
                q = 0.70 + 0.2 * progress
                thresh = np.quantile(P, q)
                P_new = np.minimum(P, thresh)
            else:
                P_new = P.copy()
                low = P < P_mean * 0.4
                P_new[low] = P[low] / (1.0 + 0.3 * progress)

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        # Multiplicative: reduce high-contributors, increase low
        r = np.ones(n)
        for j in js:
            c_j = a[j] * a[k_peak - j]
            if c_j > 0.5 * max_contrib:
                r[j] = 1.0 - rate * (c_j / max_contrib)

        a_new = a * r
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer combining LP direction finding, spectral shaping,
    LSE annealing, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        best_a = np.ones(8192) / 8192.0
        sc_init = compute_score(best_a)
        update(sc_init, best_a, "init-uniform")
        print(f"Starting from uniform n=8192, score={sc_init:.8f}")

    # =========================================================================
    # Phase 1: Fienup spectral initialization (5% time)
    # =========================================================================
    phase1_end = t_start + 0.05 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Fienup Init ({phase1_end - t_start:.0f}s) ===")

    a_f1 = fienup_spectral(best_a, n_iters=200, mode='flat')
    update(compute_score(a_f1), a_f1, "FIENUP-flat")

    a_f2 = fienup_spectral(best_a, n_iters=200, mode='quantile')
    update(compute_score(a_f2), a_f2, "FIENUP-quant")

    a_f3 = fienup_spectral(best_a, n_iters=200, mode='hybrid')
    update(compute_score(a_f3), a_f3, "FIENUP-hybrid")

    # =========================================================================
    # Phase 2: LP Direction Finding (30% time) - KEY NOVEL CONTRIBUTION
    # =========================================================================
    phase2_end = t_start + 0.35 * (budget_s - 30.0)
    print(f"\n=== Phase 2: LP Direction Search ({phase2_end - t_start:.0f}s) ===")

    a_work = normalize(best_a.copy())

    # Multiple passes with increasing top_k
    lp_configs = [
        (100, 60, "LP-k100"),
        (200, 60, "LP-k200"),
        (400, 50, "LP-k400"),
        (200, 80, "LP-k200b"),
        (300, 60, "LP-k300"),
    ]

    for top_k, n_iters, label in lp_configs:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - 30 - time.monotonic()
        if remaining < 60:
            break

        # Determine max iterations based on time
        max_iters = min(n_iters, int(remaining / 90) + 10)
        max_iters = max(max_iters, 10)

        print(f"  [{label}] top_k={top_k}, max_iters={max_iters}")
        a_lp = lp_direction_search(a_work, n_iters=max_iters, top_k=top_k, t_init=0.12)
        sc = compute_score(a_lp)
        improved = update(sc, a_lp, label)

        if improved:
            a_work = normalize(a_lp.copy())

        # Follow-up LSE on LP result
        if time.monotonic() < phase2_end - 40:
            sc2, a2 = run_lbfgs(a_work, 2000, maxiter=2000, maxfun=40000)
            update(sc2, a2, label + "+LSE")
            if sc2 < best_score:
                a_work = normalize(a2.copy())

    # =========================================================================
    # Phase 3: LSE Annealing (20% time)
    # =========================================================================
    phase3_end = t_start + 0.55 * (budget_s - 30.0)
    print(f"\n=== Phase 3: LSE Annealing ({phase3_end - t_start:.0f}s) ===")

    a_work = normalize(best_a.copy())
    betas = [50, 150, 400, 1000, 3000, 8000, 20000, 50000]
    for i, beta in enumerate(betas):
        if time.monotonic() > phase3_end - 20:
            break
        remaining = phase3_end - 20 - time.monotonic()
        if remaining < 20:
            break
        maxiter = max(500, int(remaining * 8))
        maxiter = min(maxiter, 8000)
        print(f"  LSE beta={beta:>6}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 20)
        dt = time.monotonic() - t0
        print(f"    score={sc:.10f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE-{beta}")
        a_work = normalize(a_opt.copy())

    # =========================================================================
    # Phase 4: Peak Shaving + Multiplicative (10% time)
    # =========================================================================
    phase4_end = t_start + 0.65 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Peak Shaving ({phase4_end - t_start:.0f}s) ===")

    for top_k in [5, 10, 20]:
        if time.monotonic() > phase4_end - 20:
            break
        a_ps = peak_shave_subgrad(best_a, n_iters=2000, top_k=top_k)
        sc = compute_score(a_ps)
        update(sc, a_ps, f"PSAVE-k{top_k}")

        if time.monotonic() < phase4_end - 15:
            a_mu = multiplicative_update(best_a, best_a, rate=0.08)
            sc_mu = compute_score(a_mu)
            update(sc_mu, a_mu, f"MULT-k{top_k}")

    # =========================================================================
    # Phase 5: Multi-restart with LP + LSE (15% time)
    # =========================================================================
    phase5_end = t_start + 0.80 * (budget_s - 30.0)
    print(f"\n=== Phase 5: Multi-Restart ({phase5_end - t_start:.0f}s) ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 30:
            break
        remaining = phase5_end - 30 - time.monotonic()
        if remaining < 30:
            break

        if r % 4 == 0:
            # Small perturbation + LP
            a_pert = normalize(best_a + 0.005 * rng.standard_normal(len(best_a)))
        elif r % 4 == 1:
            # Focused perturbation at non-peak areas
            a_pert = best_a.copy()
            c = fast_conv(a_pert)
            k_peak = int(np.argmax(c))
            j_s = max(0, k_peak - len(a_pert) + 1)
            j_e = min(k_peak, len(a_pert) - 1)
            mask = np.ones(len(a_pert), dtype=bool)
            mask[j_s:j_e + 1] = False
            add_idx = np.where(mask)[0]
            n_add = max(20, len(a_pert) // 30)
            if len(add_idx) > n_add:
                add_idx = rng.choice(add_idx, n_add, replace=False)
            delta = 0.001 / len(a_pert)
            a_pert[add_idx] += delta
            a_pert = normalize(a_pert)
        elif r % 4 == 2:
            # Multiplicative perturbation
            freq = rng.integers(1, 30)
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            x = np.arange(len(a_pert), dtype=np.float64) / len(a_pert)
            pert = 0.02 * np.cos(2.0 * np.pi * freq * x + phase)
            a_pert = normalize(a_pert * (1.0 + pert))
        else:
            # Different beta LSE from perturbed start
            a_pert = normalize(best_a + 0.003 * rng.standard_normal(len(best_a)))

        # LP direction (quick)
        if time.monotonic() < phase5_end - 40:
            a_lp = lp_direction_search(a_pert, n_iters=15, top_k=150, t_init=0.1)
            sc_lp = compute_score(a_lp)
            update(sc_lp, a_lp, f"R{r}-LP")
            if sc_lp < best_score:
                a_pert = normalize(a_lp.copy())

        # LSE refinement
        if time.monotonic() < phase5_end - 15:
            maxiter = max(300, int((phase5_end - 15 - time.monotonic()) * 5))
            beta = 1000 + r * 500
            sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 15)
            update(sc, a_opt, f"R{r}-LSE")

    # =========================================================================
    # Phase 6: Size exploration (5% time)
    # =========================================================================
    phase6_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 6: Size Exploration ===")

    current_n = len(best_a)
    target_ns = [n for n in [12000, 15000, 20000] if n > current_n]

    for new_n in target_ns:
        if time.monotonic() > phase6_end - 30:
            break
        a_large = interpolate_sequence(best_a, new_n)
        sc = compute_score(a_large)
        print(f"  n={new_n}: score={sc:.10f}")
        update(sc, a_large, f"INTERP-{new_n}")

        if time.monotonic() < phase6_end - 20:
            sc2, a2 = run_lbfgs(a_large, 5000, maxiter=2000, maxfun=30000)
            update(sc2, a2, f"INTERP+LSE-{new_n}")
            if sc2 < best_score:
                best_a = normalize(a2.copy())

    # =========================================================================
    # Phase 7: Final Polish (12% time)
    # =========================================================================
    print(f"\n=== Phase 7: Final Polish ===")

    # High-beta LSE
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = max(1000, int(remaining * 6))
        sc, a_final = run_lbfgs(best_a, 30000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-30k")

    # One more LP pass on the best
    if time.monotonic() < deadline - 60:
        a_lp_final = lp_direction_search(best_a, n_iters=30, top_k=300, t_init=0.08)
        sc = compute_score(a_lp_final)
        update(sc, a_lp_final, "FINAL-LP")

    # Final LSE
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        maxiter = max(500, int(remaining * 5))
        sc, a_final = run_lbfgs(best_a, 80000, maxiter=maxiter, maxfun=maxiter * 12)
        update(sc, a_final, "FINAL-LSE-80k")

    # Final peak shave
    if time.monotonic() < deadline - 20:
        a_ps = peak_shave_subgrad(best_a, n_iters=1500, top_k=5)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    # =========================================================================
    # Output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if len(out) < 1 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
