"""
Optimized step function construction using geometric-beta-annealed LSE optimization,
Nesterov-accelerated subgradient method, and multi-scale refinement.
Key innovations: (1) geometric beta annealing for LSE smoothing, 
(2) Nesterov acceleration for exact objective, (3) coordinate descent polish.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value (assumes a is normalized to sum=1, but works generally)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """
    LSE-smoothed objective with gradient.
    Objective: 2n * LSE_beta(c(x)) / (sum(x))^2
    where LSE_beta(c) = (1/beta) * log(sum(exp(beta * c)))
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Weights: p_k = exp(beta*(c_k - c_max)) / sum(exp(...))
    p = exp_scaled / exp_sum

    # Gradient of L w.r.t. x:
    # dL/dx_i = 2 * sum_k p_k * x_{k-i} (approx, ignoring diagonal correction)
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    
    # Project gradient to account for normalization (we optimize on the simplex)
    # Actually, since we use L-BFGS-B with bounds [0, inf], we don't need to project.
    # The normalization is handled implicitly.
    
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=5000, time_limit=60.0):
    """Run L-BFGS-B on the LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S
    
    bounds = [(0.0, None)] * n
    t_start = time.monotonic()
    
    # Use a wrapper that stops based on time
    state = {'count': 0}
    
    def fun_and_grad(x):
        state['count'] += 1
        if time.monotonic() - t_start > time_limit:
            raise StopIteration()
        return lse_objective_and_grad(x, beta)
    
    try:
        res = minimize(
            fun_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-10,

        )
    except StopIteration:
        res = None
    except Exception:
        res = None
    
    if res is not None and hasattr(res, 'x'):
        a_out = np.clip(res.x, 0.0, 1000.0)
        S_out = a_out.sum()
        if S_out < 1e-6:
            a_out = a0.copy()
        else:

        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    else:
        return compute_score(a0), a0


def nesterov_subgradient(a0, n_iters=5000, n_peaks=10, step_init=0.01, 
                          step_decay=0.9999, time_limit=120.0):
    """
    Nesterov-accelerated subgradient method for the exact objective.
    Targets the top n_peaks of the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    # Nesterov momentum
    y = a.copy()
    moment = 0.0
    step = step_init
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Evaluate at y (Nesterov lookahead point)
        A = np.fft.rfft(y, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_y = float(y.sum())
        M_val = float(c.max())
        sc_y = 2.0 * n * M_val / (S_y * S_y)
        
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()
        
        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Subgradient at y
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * y[js:je + 1][::-1]
        
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Nesterov momentum update
        moment_new = (1.0 + moment) / 2.0
        momentum = moment_new
        t_nest = 1.0 + moment
        
        # Gradient step
        step = min(step, max((sc_y - sc_lb) / (float(np.dot(grad, grad)) + 1e-30), 1e-12))
        step = min(step, 0.02)
        
        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        S_new = float(y_new.sum())
        if S_new < 1e-12:
            break
        y_new = y_new / S_new
        
        # Nesterov update: a_new = y_new + momentum * (y_new - y)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        S_a = float(a_new.sum())
        if S_a < 1e-12:
            a = y_new
            y = y_new
            continue
        a_new = a_new / S_a
        
        a = a_new
        y = y_new
        step *= step_decay
    
    return best_a


def coordinate_descent_peaks(a0, n_iters=2000, n_peaks=5, time_limit=60.0):
    """
    Coordinate descent targeting the peaks of the autocorrelation.
    For each peak, find the element that contributes most and reduce it.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the single peak
        k_peak = int(np.argmax(c))
        
        # Elements contributing to this peak: a[j] * a[k_peak - j] for valid j
        js = max(0, k_peak - n + 1)
        je = min(k_peak, n - 1)
        
        if je < js:
            break
        
        # Contribution of each pair
        pairs = a[js:je+1] * a[k_peak - je:k_peak - js + 1][::-1]
        # Find the pair with maximum contribution
        max_pair_idx = int(np.argmax(pairs))
        j1 = js + max_pair_idx
        j2 = k_peak - j1
        
        # Try reducing the larger of the two
        if j1 == j2:
            # Single element
            a[j1] *= 0.95
        else:
            # Reduce the one that appears in more peaks
            # Simple heuristic: reduce the larger one
            if a[j1] >= a[j2]:
                a[j1] *= 0.95
            else:
                a[j2] *= 0.95
        
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def fienup_geometric_v2(a, n_iters=300, alpha_start=0.05, alpha_end=0.7,
                         time_limit=120.0):
    """Improved geometric Fienup with time limit."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

"""
Optimized step function construction using geometric-beta-annealed LSE optimization,
Nesterov-accelerated subgradient method, and multi-scale refinement.
Key innovations: (1) geometric beta annealing for LSE smoothing with scale penalty,
(2) Nesterov acceleration for exact objective optimization,
(3) coordinate descent polish on peak-contributing elements,
(4) multi-start exploration with diverse perturbations.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta, scale_pen):
    """
    LSE-smoothed objective with gradient plus scale penalty.
    Objective: 2n * LSE_beta(c(x)) / (sum(x))^2 + scale_pen * (sum(x) - 1)^2
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S) + scale_pen * (S - 1.0) ** 2

    # Weights
    p = exp_scaled / exp_sum

    # Gradient of L w.r.t. x via convolution
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad_main = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    grad_pen = 2.0 * scale_pen * (S - 1.0) * np.ones(n)
    grad = grad_main + grad_pen

    return score, grad


def scipy_wrapper(x, beta, scale_pen):
    """Wrapper for scipy minimize (top-level to avoid closures)."""
    return lse_obj_grad(x, beta, scale_pen)


def run_lbfgs_lse(a0, beta, scale_pen, maxiter, time_limit):
    """Run L-BFGS-B on the LSE-smoothed objective with time limit via maxiter."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = n * 0.5
    a0 = a0 / S

    bounds = [(0.0, None)] * n
    t_start = time.monotonic()

    try:
        res = minimize(
            scipy_wrapper, a0, args=(beta, scale_pen),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': maxiter * 10, 'maxls': 50}
        )
    except Exception:
        res = None

    elapsed = time.monotonic() - t_start
    if res is not None and hasattr(res, 'x'):
        a_out = np.clip(res.x, 0.0, 1000.0)
        S_out = a_out.sum()
        if S_out < 1e-6:
            a_out = a0.copy()
        else:
            a_out = a_out / S_out
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    else:
        return compute_score(a0), a0


def nesterov_subgradient(a0, n_iters, n_peaks, step_init, step_decay, time_limit):
    """
    Nesterov-accelerated subgradient method for the exact objective.
    Targets the top n_peaks of the autocorrelation.
    """
    n = len(a0)
    y = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(y.sum())
    if S < 1e-12:
        return np.ones(n) / n
    y = y / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = y.copy()
    best_sc = compute_score(y)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    x = y.copy()
    t_nest = 1.0
    step = step_init
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        # Evaluate at y
        A = np.fft.rfft(y, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_y = float(y.sum())
        M_val = float(c.max())
        sc_y = 2.0 * n * M_val / (S_y * S_y)

        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()

        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Subgradient at y
        sg = np.zeros(n)
        for k in top_idx:
            k_i = int(k)
            js = max(0, k_i - n + 1)
            je = min(k_i, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * y[js:je + 1][::-1]

        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Adaptive step
        gnorm_sq = float(np.dot(grad, grad))
        step = min(step, max((sc_y - sc_lb) / (gnorm_sq + 1e-30), 1e-12))
        step = min(step, 0.015)

        # Gradient step from y
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        S_new = float(x_new.sum())
        if S_new < 1e-12:
            break
        x_new = x_new / S_new

        # Nesterov momentum update
        t_nest_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = (t_nest - 1.0) / t_nest_new
        t_nest = t_nest_new

        y_new = x_new + momentum * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        S_ynew = float(y_new.sum())
        if S_ynew < 1e-12:
            y = x_new
        else:
            y = y_new / S_ynew

        x = x_new
        step *= step_decay

    return best_a


def coord_descent_peaks(a0, n_iters, time_limit):
    """
    Coordinate descent: for the dominant peak, reduce the largest contributing element.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_peak = int(np.argmax(c))
        js = max(0, k_peak - n + 1)
        je = min(k_peak, n - 1)
        if je < js:
            break

        # Find the pair with max contribution to this peak
        a_seg = a[js:je + 1]
        a_seg_rev = a[k_peak - je:k_peak - js + 1][::-1]
        products = a_seg * a_seg_rev
        max_idx = int(np.argmax(products))
        j1 = js + max_idx
        j2 = k_peak - j1

        # Reduce the larger element (or the one at j1 if equal)
        reduction = 0.97 + 0.02 * (it / max(1, n_iters))  # gradually reduce less
        if j1 == j2:
            a[j1] *= reduction
        else:
            if a[j1] >= a[j2]:
                a[j1] *= reduction
            else:
                a[j2] *= reduction

        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def fienup_geometric(a, n_iters, alpha_start, alpha_end, time_limit):
    """Geometric Fienup iteration with time limit."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_elliptical(a, n_iters, beta_shape, time_limit):
    """Elliptical spectral shaping with time limit."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        progress = it / max(1, n_iters)
        beta = beta_shape * (0.5 + progress)

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        ratio = amp / C
        amp_new = amp / np.sqrt(1.0 + beta * (ratio - 1.0) ** 2)

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_directed_descent(a0, n_iters, n_peaks, step_init, step_decay, time_limit):
    """Peak-directed subgradient descent (non-accelerated)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    step = step_init
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break

        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k_i = int(k)
            js = max(0, k_i - n + 1)
            je = min(k_i, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gnorm = float(np.linalg.norm(grad))

        step *= step_decay

    return best_a

            best_a = a_cd.copy()
            print(f"  [COORD] NEW BEST: {sc:.8f}")

    # Very high beta LSE
    for beta in [20000, 50000, 100000]:
        if time.monotonic() > phase4_end - 15:
            break
        remaining = phase4_end - time.monotonic() - 5
        if remaining < 10:
            break
        maxiter = min(3000, max(50, int(remaining * 10)))
        time_limit = min(remaining, 60.0)
        sc, a_opt = run_lbfgs_lse(best_a, beta, 500.0, maxiter=maxiter,
                                   time_limit=time_limit)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [POLISH b={beta}] NEW BEST: {sc:.8f}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Multi-start exploration (10% time)
    # ============================================================
    print(f"\n=== Phase 5: Multi-Start Exploration ===")

    for trial in range(6):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic() - 5
        if remaining < 10:
            break

        # Different perturbation strategies
        if trial < 3:
            # Gaussian noise scaling
            noise_scale = 0.005 * (1 + trial)
            a_p = best_a * (1.0 + noise_scale * rng.standard_normal(len(best_a)))
        else:
            # Spectral perturbation
            a_p = best_a.copy()
            M_fft = 1 << (2 * len(a_p) - 1).bit_length()
            A = np.fft.rfft(a_p, M_fft)
            noise_amp = 0.01 * (trial - 2) * rng.random(len(A))
            A += noise_amp * np.exp(1j * rng.uniform(0, 2 * np.pi, len(A)))
            a_p = np.fft.irfft(A, M_fft)[:len(best_a)]
            a_p = np.maximum(a_p, 0.0)

        a_p = np.clip(a_p, 0.0, 1000.0)
        S_p = a_p.sum()
        if S_p < 1e-12:
            continue
        a_p = a_p / S_p

        # Quick LSE polish from perturbed start
        time_limit = min(remaining, 50.0)
        maxiter = min(2000, max(50, int(time_limit * 10)))
        sc, a_opt = run_lbfgs_lse(a_p, 2000, 200.0, maxiter=maxiter,
                                   time_limit=time_limit)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [RESTART {trial}] NEW BEST: {sc:.8f}")

    # Final polish
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            sc, a_opt = run_lbfgs_lse(best_a, 50000, 1000.0,
                                       maxiter=min(2000, int(remaining * 8)),
                                       time_limit=remaining)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  [FINAL] NEW BEST: {sc:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
