```python
def solve_dual_lp(f, M, n):
    """Solve: max Σb s.t. (f*b)[k] ≤ M, b ≥ 0."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    # Build constraint matrix A (2n-1) x n
    # A[k,j] = f[j] if j ≤ k ≤ j+n-1, else 0
    total_nz = n * n
    rows = np.zeros(total_nz, dtype=np.int32)
    cols = np.zeros(total_nz, dtype=np.int32)
    vals = np.zeros(total_nz, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        cnt = k_end - k_start + 1  # = n
        rows[idx:idx+cnt] = np.arange(k_start, k_end+1)
        cols[idx:idx+cnt] = j
        vals[idx:idx+cnt] = f[j]
        idx += cnt
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -Σb
    c_obj = -np.ones(n)
    
    # Constraints: A @ b ≤ M, b ≥ 0
    A_ub = A.tocsr()
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        b_opt = res.x
        return b_opt, -res.fun  # -fun = Σb
    else:
        return None, None
```
