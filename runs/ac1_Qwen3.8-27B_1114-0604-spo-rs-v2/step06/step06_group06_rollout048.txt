"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with adaptive soft
active set, periodic LSE-LBFGS refinement, cross-correlation gradient computation,
multi-scale resampling, and diversification restarts.
"""

import numpy as np
import time
import math


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fft_self_conv(a):
    n = len(a)
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _fft_cross(a, b):
    M = len(a) + len(b) - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:M]


def _xcorr(a, w):
    """grad[i] = 2*sum_j a[j]*w[j+i] for i=0..n-1, w length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    full = _fft_cross(a_rev, w)
    return 2.0 * full[n - 1:2 * n - 1]


def _normalize(a):
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = float(a.sum())
    if S < 1e-14:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-14:
        return np.inf
    c = _fft_self_conv(a)
    return float(2.0 * n * np.max(c) / (S * S))


def _soft_active_weights(conv, beta, top_k=20):
    """Soft active set via softmax on top positions."""
    cmax = float(np.max(conv))
    if cmax <= 0:
        return np.ones(conv.size) / conv.size
    z = beta * (conv - cmax)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-30:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(conv.size)
        w[idx] = 1.0 / top_k
        return w
    w = e / s
    # Floor: ensure at least top_k positions have non-negligible weight
    floor = 1.0 / (conv.size * 10)
    w = np.maximum(w, floor)
    w /= w.sum()
    return w


def _mirror_descent_step(a, conv, beta, eta, top_k=20):
    """One mirror descent (exponentiated gradient) step."""
    n = len(a)
    w = _soft_active_weights(conv, beta, top_k)
    grad = _xcorr(a, w)
    gmax = float(np.max(grad))
    if gmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * grad / gmax
    log_a -= float(np.max(log_a))
    new_a = np.exp(log_a)
    return _normalize(new_a)


def _topm_equalize_step(a, conv, top_m, eta):
    """Push down top-m convolution peaks by distributing pressure."""
    n = len(a)
    d = np.sort(conv)[::-1][:top_m]
    mean_top = float(np.mean(d))
    excess = np.maximum(conv - mean_top, 0.0)
    total = float(excess.sum())
    if total < 1e-15:
        return a.copy()
    w = excess / total
    grad = _xcorr(a, w)
    gmax = float(np.max(grad))
    if gmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * grad / gmax
    log_a -= float(np.max(log_a))
    return _normalize(np.exp(log_a))


def _lse_smoothed_obj_and_grad(a, beta):
    """LSE-smoothed proxy objective and gradient."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _fft_self_conv(a)
    scale = float(n) / (S * S)
    d = scale * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -400.0, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se
    L = dmax + np.log(se) / beta
    # gradient of 2n*L w.r.t. a
    # L = dmax + (1/beta)*log(sum exp(beta*(d_k - dmax)))
    # dL/d a_i = sum_k w_k * d d_k / d a_i
    # d_k = (n/S^2) * conv_k, d conv_k / d a_i = 2*a_{k-i}
    h = _xcorr(a, w)  # this gives 2*sum w_k * a_{k-i}
    # d d_k / d a_i = (n/S^2) * 2 * a_{k-i}
    # Also d(dmax)/d(a_i) matters if dmax is achieved at the argmax... 
    # For LSE, the gradient is: sum_k w_k * d d_k / d a_i - (2n * (sum a) / S^2) * (sum_k w_k * d_k)
    # Simplified: grad_score[i] = (2n/S^2) * sum_k w_k * 2*a_{k-i} - (4n*wdotd/S^3)
    wdotd = float(np.dot(w, d))
    grad = (float(n) / (S * S)) * h - (4.0 * float(n) * wdotd / (S * S)) * np.ones(n)
    # Actually the score is 2n * dmax (when sum=1), so with general sum:
    # score = 2n * max(d_k) where d_k = (n/S^2)*conv_k
    # LSE approx: score ≈ 2n*(dmax + log(sum exp(beta(d-dmax)))/beta)
    # d(score)/d(a_i) = 2n * sum_k w_k * d(d_k)/d(a_i) - 2n*(2/S)*sum_k w_k*contribution
    # This is getting complex. Let me just use the subgradient direction.
    grad = h - (float(np.mean(h)))  # center to handle the sum constraint
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * float(n) * L), grad


def _run_lbfgs(a, beta, deadline, maxiter=200):
    from scipy.optimize import minimize
    a0 = _normalize(a)
    n = len(a0)
    if time.monotonic() > deadline - 3.0:
        return _score(a0), a0

    def fun_and_grad(x):
        x = np.clip(x, 0.0, None)
        if x.sum() < 1e-14:
            return 1e10, np.zeros_like(x)
        x = x / x.sum()
        val, g = _lse_smoothed_obj_and_grad(x, beta)
        # Project gradient: remove component along (1,1,...,1) for simplex
        g = g - np.mean(g)
        return val, g

    bounds = [(0.0, 10.0)] * n
    opts = {"maxiter": maxiter, "maxfun": maxiter * 2, "ftol": 1e-14, "gtol": 1e-10}
    try:
        res = minimize(fun_and_grad, a0, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        out = _normalize(res.x)
    except Exception:
        out = a0
    sc = _score(out)
    if not np.isfinite(sc) or sc > 1e6:
        out = a0
        sc = _score(a0)
    return float(sc), out


def _resample(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n_new) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, np.clip(a, 0, None)).astype(np.float64)


def _perturb(a, rng, strength=None):
    p = _normalize(a)
    n = len(p)
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = p * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.03 * rng.standard_normal(n))
    elif mode == 3:
        # Active-set guided
        conv = _fft_self_conv(p)
        w = _soft_active_weights(conv, 200, top_k=15)
        grad = _xcorr(p, w)
        gmax = max(float(np.max(grad)), 1e-15)
        q = p * np.exp(-strength * grad / gmax)
    elif mode == 4:
        # Smooth noise in log space
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        q = p * (1.0 - strength * 0.4 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        # Block perturbation
        nb = int(rng.integers(6, 30))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    else:
        q = p * (0.85 + 0.3 * rng.random(n))
    return _normalize(np.clip(q, 0.0, None))


def _active_set_optimize(a, deadline, verbose=True, tag=""):
    """Main mirror descent optimizer with adaptive parameters."""
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()

    beta = 100.0
    eta = 0.2
    stagnant = 0
    max_stagnant = 40
    top_k = 12

    for phase in range(30):
        if time.monotonic() > deadline - 2.0:
            break
        phase_dead = min(deadline, t0 + (deadline - t0) * (phase + 1) / 30.0 + 3.0)

        for it in range(300):
            if time.monotonic() > phase_dead - 1.0:
                break

            conv = _fft_self_conv(a)
            sc = float(2.0 * n * np.max(conv))

            # Try multiple (beta, eta) combos
            best_a_iter = a
            best_sc_iter = sc
            best_params = (beta, eta)

            for b_try in [beta * 0.5, beta, beta * 2.0, beta * 4.0]:
                for e_try in [eta * 0.5, eta, eta * 1.5, eta * 2.5]:
                    if e_try < 0.005 or b_try < 5:
                        continue
                    new_a = _mirror_descent_step(a, conv, b_try, e_try, top_k)
                    new_sc = float(2.0 * n * np.max(_fft_self_conv(new_a)))
                    if new_sc < best_sc_iter:
                        best_sc_iter = new_sc
                        best_a_iter = new_a
                        best_params = (b_try, e_try)

            # Also try top-m equalization
            for tm in [5, 10, 20, 40]:
                if tm > n:
                    continue
                for e_try in [eta * 0.3, eta * 0.5, eta]:
                    if e_try < 0.005:
                        continue
                    vr_a = _topm_equalize_step(a, conv, tm, e_try)
                    vr_sc = float(2.0 * n * np.max(_fft_self_conv(vr_a)))
                    if vr_sc < best_sc_iter:

                        best_a_iter = vr_a

            if best_sc_iter < best_sc - 1e-12:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter
                beta, eta = best_params
                stagnant = 0
                if verbose and it % 30 == 0:
                    print("  [{} p{} i{}] sc={:.10f} beta={:.0f} eta={:.3f} tk={}".format(
                        tag, phase, it, best_sc, beta, eta, top_k))
            else:
                stagnant += 1
                eta *= 0.88
                if stagnant >= max_stagnant:
                    break

        # Adapt parameters
        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.4) + 2, 80)
            beta = max(20.0, beta * 0.5)
            eta = max(0.05, eta * 0.6)
            stagnant = 0
            if verbose:

        else:
            top_k = max(5, int(top_k * 0.97))
            beta = min(20000.0, beta * 1.05)

    return float(best_sc), best_a


def _generate_inits(n, rng, parent, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = _resample(arr, n)
        arr = np.clip(np.nan_to_num(arr), 0.0, None)
        if arr.sum() < 1e-10:
            arr = np.ones(n) * 0.1
        inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    _add(np.ones(n))
    for alpha in (0.3, 0.7, 1.2, 2.0, 3.0):
        _add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
        _add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 5):
        for alpha in (0.3, 0.8, 1.5):
            _add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    for rate in (1.0, 3.0, 6.0, 10.0):
        _add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        _add(np.exp(-rate * np.abs(x - 0.5)) + 0.01)
    for p_exp in (0.3, 0.5, 1.0, 2.0, 4.0):
        _add((1.0 - (2.0 * x - 1.0) ** 2) ** p_exp + 0.01)
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        _add(1.0 + 0.7 * noise)
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))
    return inits

    best_sc = _score(a)
    print("Initial: n={}, sc={:.10f} t={:.1f}s".format(n, best_sc, time.monotonic() - t_start))

    # Phase 1: Active-set mirror descent (45% of time)
    p1_dead = t_start + 0.45 * total_budget
    sc, a = _active_set_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After P1: {:.10f} t={:.1f}s".format(best_sc, time.monotonic() - t_start))

    # Phase 2: LSE-LBFGS refinement (20% of time)
    p2_dead = t_start + 0.65 * total_budget
    a = _normalize(best_a)
    for beta in [200, 500, 1000, 3000, 10000, 30000]:
        if time.monotonic() > p2_dead - 5:
            break
        rem = p2_dead - time.monotonic()
        if rem < 10:
            break
        mi = max(40, int(rem * 0.8 / 0.1))
        mi = min(mi, 500)
        sc, a = _run_lbfgs(a, beta, p2_dead, mi)
        if sc < best_sc:
            best_sc = sc
            best_a = _normalize(a).copy()
            print("  P2 beta={}: {:.10f}".format(beta, best_sc))

    # Phase 3: Active-set again on improved (15% of time)
    p3_dead = t_start + 0.80 * total_budget
    a = _normalize(best_a)
    sc, a = _active_set_optimize(a, deadline=p3_dead, verbose=True, tag="p3")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After P3: {:.10f} t={:.1f}s".format(best_sc, time.monotonic() - t_start))

    # Phase 4: Diversification restarts (10% of time)
    p4_dead = t_start + 0.90 * total_budget
    restarts = 0
    while time.monotonic() < p4_dead - 20 and restarts < 5:
        rem = p4_dead - 20.0 - time.monotonic()
        if rem < 30:
            break
        restarts += 1
        p = _perturb(best_a, rng, strength=float(rng.uniform(0.03, 0.2)))
        chunk_dead = time.monotonic() + rem * 0.75
        sc, seq = _active_set_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(restarts))
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print("Restart {}: {:.10f}".format(restarts, best_sc))

    # Phase 5: Final LSE-LBFGS polish (5% of time)
    p5_dead = deadline - 5.0
    if time.monotonic() < p5_dead - 10:
        a = _normalize(best_a)
        for beta in [5000, 20000, 50000]:
            if time.monotonic() > p5_dead - 3:
                break
            sc, a = _run_lbfgs(a, beta, p5_dead, 100)
            if sc < best_sc:
                best_sc = sc
                best_a = _normalize(a).copy()
                print("  P5 beta={}: {:.10f}".format(beta, best_sc))

    # Phase 6: Final exact gradient polish
    if time.monotonic() < deadline - 3:
        a = _normalize(best_a)
        eta = 0.1
        for it in range(200):
            if time.monotonic() > deadline - 2:
                break
            conv = _fft_self_conv(a)
            sc = float(2.0 * n * np.max(conv))
            improved = False
            for e_try in [eta * 0.5, eta, eta * 1.5]:
                for b_try in [50, 200, 500]:
                    new_a = _mirror_descent_step(a, conv, b_try, e_try, 10)
                    new_sc = float(2.0 * n * np.max(_fft_self_conv(new_a)))
                    if new_sc < sc - 1e-13:
                        a = new_a
                        sc = new_sc
                        improved = True
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
            if not improved:
                eta *= 0.8

    # Final: try a few more mirror steps with very small eta
    if time.monotonic() < deadline - 1:
        a = _normalize(best_a)
        eta = 0.02
        for it in range(500):
            if time.monotonic() > deadline - 1:
                break
            conv = _fft_self_conv(a)
            new_a = _mirror_descent_step(a, conv, 500.0, eta, 8)
            new_sc = float(2.0 * n * np.max(_fft_self_conv(new_a)))
            cur_sc = float(2.0 * n * np.max(conv))
            if new_sc < cur_sc - 1e-14:
                a = new_a
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_a = a.copy()
            else:
                eta *= 0.9

    print("Best internal: {:.10f}".format(best_sc))

    # Cleanup and return
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
