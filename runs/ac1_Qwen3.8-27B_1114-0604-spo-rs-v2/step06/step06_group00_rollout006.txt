coordinated peak redistribution, and multi-scale spectral optimization
for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements over previous: faster per-iteration PGD (vs L-BFGS),
Nesterov momentum, coordinated peak-off mass redistribution,
adaptive Polyak step sizes, and multi-scale n exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.zeros_like(a)
    return a / S


def lse_score_grad(a, beta, lower_bound):
    """
    Compute LSE-smoothed score and gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a via conv of p with reversed a
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def nesterov_pgd(a0, beta, n_iters, lower_bound, lr_init=0.02, momentum_init=0.9):
    """
    Nesterov accelerated projected gradient descent on LSE-smoothed objective.
    Projects onto {a >= 0, sum(a) = 1} after each step.
    """
    n = len(a0)
    a = normalize(a0.copy()).astype(np.float64)
    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    best_a = a.copy()
    best_sc = compute_score(a)
    prev_y = a.copy()

    for it in range(n_iters):
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-15:
            y = a
        else:
            y = y / S_y

        sc, grad = lse_score_grad(y, beta, lower_bound)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Adaptive step size (Polyak-like with safety)
        gap = max(sc - lower_bound, 1e-8)
        lr_polyak = gap / (gn * gn + 1e-30)
        lr = np.clip(lr, 1e-8, min(0.1, lr_polyak * 0.5))

        # Update
        v_new = momentum * v - lr * grad
        a_new = a + v_new
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-15:
            a_new = prev_y
        else:
            a_new = a_new / S_new

        # Check improvement with exact score (every 20 iters)
        if it % 20 == 0 or it == n_iters - 1:
            sc_exact = compute_score(a_new)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a_new.copy()

        v = v_new
        a = a_new
        prev_y = a

        # Slowly increase momentum
        momentum = min(0.98, momentum + 0.00001)

    return best_sc, best_a


def subgrad_pgd(a0, n_iters, lower_bound, top_k=3, lr_init=0.01):
    """
    Projected subgradient descent targeting top-k peaks simultaneously.
    Uses exact max (not LSE) for the subgradient computation.
    """
    n = len(a0)
    a = normalize(a0.copy()).astype(np.float64)
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    momentum = 0.85
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-9:
            break

        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Build weighted subgradient
        a_rev = a[::-1]
        sg_full = np.zeros(2 * n - 1)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            sg_full[k] += w

        # conv(sg_full, a_rev) gives d(conv)/da at each position
        M3_fft = 1 << (((2 * n - 1) + n - 1) - 1).bit_length()
        PA = np.fft.rfft(sg_full, M3_fft)
        RA = np.fft.rfft(a_rev, M3_fft)
        conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3_fft]
        grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

        S2 = S * S
        S3 = S2 * S
        avg_w = float(np.mean([c[k] / (M_val + 1e-30) for k in top_idx]))
        grad = (2.0 * n) * (grad_L / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Polyak step
        gap = max(sc - lower_bound, 1e-8)
        lr_p = gap / (gn * gn + 1e-30)
        lr = max(1e-10, min(lr, 0.05 * lr_p, 0.03))

        v = momentum * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-15:
            break
        a = a / S
        momentum = min(0.95, momentum + 0.00005)

    return best_sc, best_a


def coord_descent_redist(a, n_sweeps=5, n_top=30):
    """
    Coordinate descent with redistribution: reduce top peak contributors,
    simultaneously add to non-contributor positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(123)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        # Non-peak positions
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[j_start:j_end + 1] = True
        non_peak_idx = np.where(~peak_mask)[0]
        if len(non_peak_idx) < 5:
            break
        n_add = min(len(non_peak_idx), top_count * 2)
        add_idx = rng.choice(non_peak_idx, n_add, replace=False)

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-12:
                continue
            # Try reducing and redistributing
            best_reduction = 1.0
            best_local_sc = best_sc
            for frac in [0.5, 0.6, 0.7, 0.8, 0.85, 0.9]:
                a[j] = a_old * frac
                delta = (a_old - a[j]) * 0.5
                for k_add in add_idx:
                    a[k_add] += delta / len(add_idx)
                a_temp = normalize(a)
                sc = compute_score(a_temp)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_reduction = frac
                a[j] = a_old
                for k_add in add_idx:
                    a[k_add] -= delta / len(add_idx)

            a[j] = a_old * best_reduction
            delta = (a_old - a[j]) * 0.5
            for k_add in add_idx:
                a[k_add] += delta / len(add_idx)
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def spectral_step(a, n_iters=50, mode="adapt"):
    """
    Modified Fienup: adaptively shape the spectrum to reduce autocorrelation peak.
    Instead of just flattening, boost frequencies that correspond to the peak lag.
    """
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_sqrt = np.sqrt(P)

        if mode == "adapt":
            # Target: shape P so that IFFT(P) has minimal peak
            # Boost low amplitude bins, damp high ones (adaptive level)
            P_median = np.median(P)
            boost = 1.0 + 0.3 * progress
            damp = 1.0 - 0.15 * progress
            P_new = P.copy()
            P_new[P_new < P_median] *= boost
            P_new[P_new > P_median * 3] *= damp
        elif mode == "flat":
            P_mean = float(P.mean())
            alpha = 0.2 + 0.5 * progress
            P_new = (1 - alpha) * P + alpha * P_mean
        else:  # peak_target
            # Target a specific spectral shape that minimizes peak autocorr
            # Use raised-cosine window in frequency domain
            freqs = np.arange(len(P)) / len(P)
            target = 1.0 + 0.5 * np.cos(2 * np.pi * freqs)
            target = target ** 2
            target = target / target.sum() * P.sum()
            alpha = 0.1 + 0.3 * progress
            P_new = (1 - alpha) * P + alpha * target

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        if it % 5 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a


def peak_off_redistribute(a, n_steps=2000):
    """
    Direct peak reduction: at each step, move mass from peak contributors
    to positions that don't contribute to ANY of the top peaks.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(456)
    step_size = 0.005 / n

    for it in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Identify peak contributor set
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        peak_set = set(range(j_start, j_end + 1))

        # Find top contributors within peak set
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        n_reduce = max(3, n // 200)
        top_local = np.argsort(contribs)[::-1][:n_reduce]
        reduce_idx = j_start + top_local

        # Non-peak positions (don't contribute to this lag)
        non_peak = np.array([i for i in range(n) if i not in peak_set])
        if len(non_peak) < 10:
            return best_a
        n_add = min(len(non_peak), n_reduce * 3)
        add_idx = rng.choice(non_peak, n_add, replace=False)

        # Move mass
        total_moved = 0.0
        for j in reduce_idx:
            move = min(a[j] * 0.1, step_size)
            if move > 1e-15:
                a[j] -= move
                total_moved += move

        if total_moved > 0:
            for j in add_idx:
                a[j] += total_moved / len(add_idx)

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer using Nesterov PGD, subgradient descent,
    coordinated redistribution, and adaptive spectral shaping.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_score = np.inf
    best_a = None
    n_target = 10000

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        best_a = np.ones(n_target) / n_target
        print("No parent. Starting uniform n=10000.")

    if best_a is None:
        best_a = np.ones(n_target) / n_target

    # Resize to target n if needed
    if len(best_a) != n_target:
        best_a = interpolate_sequence(best_a, n_target)
        best_a = normalize(best_a)
        sc = compute_score(best_a)
        update(sc, best_a, "resized")

    n = n_target
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    print(f"Lower bound: {lower_bound:.6f}")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Spectral initialization (5% time ~ 120s)
    # ==================================================================
    phase1_end = t_start + 0.05 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Spectral Init (budget {phase1_end - t_start:.0f}s) ===")

    a_spec = spectral_step(best_a, n_iters=200, mode="adapt")
    update(compute_score(a_spec), a_spec, "SPEC-adapt")

    a_spec2 = spectral_step(best_a, n_iters=150, mode="flat")
    update(compute_score(a_spec2), a_spec2, "SPEC-flat")

    a_work = best_a.copy()

    # ==================================================================
    # Phase 2: Nesterov PGD - low/medium beta (20% time ~ 480s)
    # ==================================================================
    phase2_end = t_start + 0.25 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Nesterov PGD low-beta (budget {phase2_end - t_start:.0f}s) ===")

    for beta in [10, 30, 100, 300]:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - 30 - time.monotonic()
        if remaining < 30:
            break
        n_iters = int(remaining * 15)
        n_iters = max(2000, min(n_iters, 20000))
        print(f"  Nesterov beta={beta}, iters={n_iters}")
        t0 = time.monotonic()
        sc, a_opt = nesterov_pgd(a_work, beta, n_iters, lower_bound,
                                  lr_init=0.01, momentum_init=0.9)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"PGD b={beta}")
        a_work = a_opt.copy()

        # Spectral step between phases
        if time.monotonic() < phase2_end - 20:
            a_spec = spectral_step(a_work, n_iters=50, mode="adapt")
            sc_spec = compute_score(a_spec)
            if sc_spec < compute_score(a_work):
                a_work = a_spec
                update(sc_spec, a_spec, "PGD+SPEC")

    # ==================================================================
    # Phase 3: Subgradient peak targeting (25% time ~ 600s)
    # ==================================================================
    phase3_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Subgradient Peak Targeting (budget {phase3_end - t_start:.0f}s) ===")

    for top_k in [1, 2, 3, 5]:
        if time.monotonic() > phase3_end - 30:
            break
        remaining = phase3_end - 30 - time.monotonic()
        if remaining < 30:
            break
        n_iters = int(remaining * 10)
        n_iters = max(1000, min(n_iters, 15000))
        print(f"  Subgrad top_k={top_k}, iters={n_iters}")
        t0 = time.monotonic()
        sc, a_opt = subgrad_pgd(best_a, n_iters, lower_bound, top_k=top_k)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"SUBG k={top_k}")

        # Follow with Nesterov at high beta
        if sc < best_score and time.monotonic() < phase3_end - 20:
            n_it2 = int((phase3_end - 20 - time.monotonic()) * 8)
            n_it2 = max(500, min(n_it2, 5000))
            sc2, a2 = nesterov_pgd(a_opt, 2000, n_it2, lower_bound, lr_init=0.005)
            update(sc2, a2, f"SUBG+PGD k={top_k}")

    # ==================================================================
    # Phase 4: Nesterov PGD - high beta (20% time ~ 480s)
    # ==================================================================
    phase4_end = t_start + 0.70 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Nesterov PGD high-beta (budget {phase4_end - t_start:.0f}s) ===")

    for beta in [1000, 5000, 20000, 100000]:
        if time.monotonic() > phase4_end - 30:
            break
        remaining = phase4_end - 30 - time.monotonic()
        if remaining < 20:
            break
        n_iters = int(remaining * 12)
        n_iters = max(1000, min(n_iters, 15000))
        print(f"  Nesterov beta={beta}, iters={n_iters}")
        t0 = time.monotonic()
        sc, a_opt = nesterov_pgd(best_a, beta, n_iters, lower_bound,
                                  lr_init=0.003, momentum_init=0.95)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"PGD-hi b={beta}")

    # ==================================================================
    # Phase 5: Peak redistribution + coordinate descent (15% time ~ 360s)
    # ==================================================================
    phase5_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 5: Redistribution + CoordDesc (budget {phase5_end - t_start:.0f}s) ===")

    a_pr = peak_off_redistribute(best_a, n_steps=3000)
    sc = compute_score(a_pr)
    update(sc, a_pr, "PEAK-REDIST")

    a_cd = coord_descent_redist(best_a, n_sweeps=8, n_top=40)
    sc = compute_score(a_cd)
    update(sc, a_cd, "COORD")

    # LSE polish after coord descent
    if time.monotonic() < phase5_end - 20:
        remaining = phase5_end - 20 - time.monotonic()
        n_iters = max(500, int(remaining * 8))
        sc2, a2 = nesterov_pgd(a_cd, 50000, n_iters, lower_bound, lr_init=0.002)
        update(sc2, a2, "COORD+PGD")

    # ==================================================================
    # Phase 6: Multi-restart (10% time ~ 240s)
    # ==================================================================
    phase6_end = deadline - 30.0
    print(f"\n=== Phase 6: Multi-Restart (budget {phase6_end - t_start:.0f}s) ===")

    n_restarts = 10
    for r in range(n_restarts):
        if time.monotonic() > phase6_end - 20:
            break
        remaining = phase6_end - 20 - time.monotonic()
        if remaining < 20:
            break

        # Different perturbation strategies
        if r % 4 == 0:
            # Small Gaussian noise
            a_pert = best_a + 0.005 * rng.standard_normal(n)
            a_pert = normalize(a_pert)
        elif r % 4 == 1:
            # Multiplicative low-freq modulation
            x = np.arange(n, dtype=np.float64) / n
            freq = rng.integers(2, 15)
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            a_pert = best_a * (1.0 + 0.02 * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = normalize(a_pert)
        elif r % 4 == 2:
            # Peak-targeted: boost non-peak areas
            c = fast_conv(best_a)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            peak_mask = np.ones(n, dtype=bool)
            peak_mask[j_start:j_end + 1] = False
            non_peak = np.where(peak_mask)[0]
            n_boost = max(100, n // 50)
            boost_idx = rng.choice(non_peak, n_boost, replace=False)
            a_pert = best_a.copy()
            delta = 0.003 / n
            a_pert[boost_idx] += delta
            a_pert = normalize(a_pert)
        else:
            # Block swap: exchange small blocks
            a_pert = best_a.copy()
            b1 = rng.integers(0, n - 200)
            b2 = rng.integers(0, n - 200)
            blen = rng.integers(50, 200)
            block = a_pert[b1:b1+blen].copy()
            a_pert[b1:b1+blen] = a_pert[b2:b2+blen]
            a_pert[b2:b2+blen] = block
            a_pert = normalize(a_pert)

        # Optimize the perturbation
        n_iters = max(500, int(remaining * 6))
        beta = 1000 + r * 2000
        sc, a_opt = nesterov_pgd(a_pert, beta, n_iters, lower_bound,
                                  lr_init=0.005, momentum_init=0.9)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 7: Final polish (last few seconds)
    # ==================================================================
    print(f"\n=== Phase 7: Final Polish ===")

    if time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        n_iters = max(200, int(remaining * 10))
        sc, a_final = nesterov_pgd(best_a, 50000, n_iters, lower_bound,
                                    lr_init=0.002, momentum_init=0.97)
        update(sc, a_final, "FINAL-PGD")

    if time.monotonic() < deadline - 5:
        a_cd = coord_descent_redist(best_a, n_sweeps=3, n_top=50)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 10000

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
