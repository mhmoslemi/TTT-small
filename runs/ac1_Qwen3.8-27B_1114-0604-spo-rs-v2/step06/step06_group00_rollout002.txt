import time

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
    if n == 0:
        return np.inf
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -600, 600)

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    return best_a
