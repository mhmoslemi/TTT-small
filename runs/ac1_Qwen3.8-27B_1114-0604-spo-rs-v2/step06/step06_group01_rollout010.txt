def self_conv(a):
    n = len(a)
    M = 2 * n - 1

    B = np.fft.rfft(b, M_fft)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_lp(a, M, n_iters, beta, x_init=None):
    """Solve LP via FISTA on LSE-smoothed objective over simplex."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    S = float(x.sum())
    if S < 1e-12:
        x = np.ones(n) / n
    else:
        x = x / S
    u = x.copy()
    t_mom = 1.0
    
    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft).copy()
    
    # Lipschitz constant
    L = beta * 2.0 * float(np.sum(a * a)) * n_c + 1e-12
    step = 1.0 / L
    
    for it in range(n_iters):
        # Adaptive step: occasional recompute
        if it > 0 and it % 500 == 0:
            L = beta * 2.0 * float(np.sum(a * a)) * n_c + 1e-12
            step = 1.0 / L
        
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum
        
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr  # gradient of LSE part; simplex handling via projection
        
        # FISTA update
        v = u - step * grad
        x_new = project_simplex(v)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        x = x_new
        t_mom = t_new
    
    return x

def lse_score_and_grad(x, beta):
    """LSE-smoothed score and gradient (unnormalized)."""
    n = len(x)
    c = self_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    
    p = exp_z / exp_z.sum()
    n_c = 2 * n - 1
    M_fft = 1 << (n_c - 1).bit_length()
    RA = np.fft.rfft(x, M_fft)
    FP = np.fft.rfft(p, M_fft)
    corr = np.fft.irfft(np.conj(RA) * FP, M_fft)[:n]
    grad_L = 2.0 * corr
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft).copy()

    # Lipschitz constant (conservative)
    L = beta * 2.0 * float(np.sum(a * a)) * n_c + 1e-12
    step = 1.0 / L

    for it in range(n_iters):
        if it > 0 and it % 1000 == 0:
            L = beta * 2.0 * float(np.sum(a * a)) * n_c + 1e-12
            step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr

        v = u - step * grad
        x_new = project_simplex(v)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        x = x_new
        t_mom = t_new

    return x


def fista_lp_two_stage(a, M, n_iters_total, x_init=None):
    """Two-stage FISTA: low beta then high beta for better LP approximation."""
    n = len(a)
    # Stage 1: low beta, fewer iterations
    beta1 = 50.0
    n1 = max(2000, n_iters_total // 4)
    x = fista_lp(a, M, n1, beta1, x_init=x_init)
    # Stage 2: high beta, more iterations
    beta2 = min(500.0, 50.0 * 10.0)
    n2 = n_iters_total - n1
    x = fista_lp(a, M, n2, beta2, x_init=x)
    return x


def lp_alternating(a, time_limit, max_outer=50, base_fista=20000):
    """LP alternating with adaptive beta and warm starts and line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    x_warm = None
    no_improve = 0

    for outer in range(max_outer):
        if time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())

        # Adaptive: increase beta, reduce fista iters as we converge
        beta = min(30.0 * (outer + 2), 1500.0)
        n_fista = max(5000, base_fista - outer * 400)

        x = fista_lp(a, M, n_fista, beta, x_init=x_warm)
        x_warm = x.copy()

        # Broader line search
        best_t = 0.2
        best_t_sc = best_sc
        t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
        a = a_new

    return best_a, best_sc, x_warm


def lse_score_and_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = self_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + float(np.log(exp_z.sum()))
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)

    p = exp_z / float(exp_z.sum())
    n_c = 2 * n - 1
    M_fft = 1 << (n_c - 1).bit_length()
    RA = np.fft.rfft(x, M_fft)
    FP = np.fft.rfft(p, M_fft)
    corr = np.fft.irfft(np.conj(RA) * FP, M_fft)[:n]
    grad_L = 2.0 * corr

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=150000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-13,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def exact_subgrad_polish(a, n_steps=300, rng=None):
    """Exact subgradient of max(conv) with backtracking line search."""
    if rng is None:
        rng = np.random.default_rng(99)
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        M = float(c.max())
        tol = 1e-9 * M + 1e-20
        k_max_list = np.where(c >= M - tol)[0]

        # Subgradient: weighted average over argmax lags
        grad = np.zeros(n)
        w = np.ones(len(k_max_list)) / len(k_max_list)
        for wi, ki in zip(w, k_max_list):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1]

        S2 = S * S
        S3 = S2 * S
        g = 2.0 * n * (grad / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break

        base_step = max(0.002, (best_sc - 1.001) / (gn * gn + 1e-30) * 0.15)
        improved = False
        for mult in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]:
            step = base_step * mult
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 15 == 0:
            noise_scale = 0.003 * (1.0 + 0.5 * np.random.random())
            a = a * (1.0 + noise_scale * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def error_reduction(a, n_iters=50, alpha=0.65):
    """Frequency-domain Newton-like error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.15 + 0.55 * (it / max(1, n_iters // 2))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 * np.ones(n))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x) ** 2)
    inits.append(np.clip(0.5 + 0.25 * rng.standard_normal(n), 0.01, 1.0))
    inits.append(0.4 + 0.25 * np.cos(2.0 * np.pi * 3 * x) + 0.15 * np.cos(2.0 * np.pi * 7 * x))
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 25))
    inits.append(np.clip(0.5 + 0.2 * smooth, 0.01, 1.0))
    inits.append(np.clip(0.35 + 0.35 * np.sin(np.pi * x) ** 2 + 0.1 * np.cos(4 * np.pi * x), 0.01, 1.0))
    inits.append(0.4 * np.ones(n) + 0.2 * np.cos(2.0 * np.pi * x) + 0.15 * np.cos(4.0 * np.pi * x + 0.3) + 0.08 * np.cos(8.0 * np.pi * x + 0.7))
    return inits


def plateau_distribute(a, n_rounds=20):
    """Iteratively redistribute mass to flatten the convolution plateau."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = self_conv(a)
        M = float(c.max())
        # Target: reduce peaks above threshold
        thresh = M * 0.999
        peaks = np.where(c >= thresh)[0]
        if len(peaks) == 0:
            break
        # Compute gradient directions at peak lags
        grad = np.zeros(n)
        for ki in peaks:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += a[j_start:j_end + 1]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        # Small step in negative direction
        step_size = 0.01 * (1.0 + 0.5 * rnd / n_rounds)
        a_try = a - step_size * grad
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            break
        a_try = a_try / St
        sc = compute_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
        a = a_try

    return best_a


def propose_candidate(budget_s=2400):
    """
    Main search: multi-phase optimization of non-negative step function.
    Phase 1: Quick diverse exploration with error reduction
    Phase 2: LP alternating (core driver) with adaptive FISTA
    Phase 3: L-BFGS progressive beta polish
    Phase 4: Exact subgradient + plateau distribution + final checks
    """
    rng = np.random.default_rng(777)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(6962) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent: score={ps:.10f}, len={len(parent)}")
        report(ps, parent, "parent")
    else:
        print("No parent available.")

    n_main = len(parent) if parent is not None else 6962
    print(f"Main length: {n_main}")

    # ============ Phase 1: Quick exploration (7%) ============
    p1_deadline = t_start + 0.07 * (float(budget_s) - 40.0)
    print(f"\n=== Phase 1: Quick Exploration ===")

    inits = build_inits(n_main, rng, parent)
    for i, a0 in enumerate(inits[:5]):
        if time.monotonic() > p1_deadline:
            break
        a_er = error_reduction(a0, n_iters=35, alpha=0.6)
        sc = compute_score(a_er)
        report(sc, a_er, f"ER{i}")

    # Try one alternate length
    alt_n = n_main + 256 if n_main < 10000 else n_main - 256
    if alt_n >= 1024:
        if time.monotonic() < p1_deadline:
            a0 = np.clip(0.5 + 0.2 * rng.standard_normal(alt_n), 0.01, 1.0)
            a_er = error_reduction(a0, n_iters=25, alpha=0.6)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={alt_n}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============ Phase 2: LP Alternating (55%) ============
    p2_deadline = t_start + 0.62 * (float(budget_s) - 40.0)
    print(f"\n=== Phase 2: LP Alternating (main) ===")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Split Phase 2 into two sub-phases with increasing precision
    p2a_deadline = t_start + 0.35 * (float(budget_s) - 40.0)
    p2b_deadline = p2_deadline

    # Sub-phase 2a: moderate precision
    print(f"  Phase 2a: moderate FISTA...")
    a_out, sc_out, x_warm = lp_alternating(a_work, time_limit=p2a_deadline,
                                            max_outer=25, base_fista=15000)
    sc = compute_score(a_out)
    report(sc, a_out, "LP-2a")
    a_work = a_out.copy()

    # Sub-phase 2b: higher precision
    print(f"  Phase 2b: high-precision FISTA...")
    a_out, sc_out, x_warm2 = lp_alternating(a_work, time_limit=p2b_deadline - 10,
                                             max_outer=30, base_fista=25000)
    sc = compute_score(a_out)
    report(sc, a_out, "LP-2b")

    # Plateau distribution
    a_plateau = plateau_distribute(a_out, n_rounds=15)
    sc = compute_score(a_plateau)
    report(sc, a_plateau, "plateau")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============ Phase 3: L-BFGS Progressive Beta (28%) ============
    p3_deadline = t_start + 0.90 * (float(budget_s) - 40.0)
    print(f"\n=== Phase 3: L-BFGS Progressive Beta ===")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > p3_deadline - 20:
            break
        remaining = p3_deadline - time.monotonic()
        if remaining < 30:
            break
        maxit = min(6000, int(remaining * 150))
        maxfun = min(150000, int(remaining * 3000))
        if maxit < 100:
            break
        print(f"  L-BFGS beta={beta}, maxiter={maxit}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=maxfun)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.10f}, {elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS-{beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============ Phase 4: Exact Subgradient + Final Polish (9%) ============
    p4_deadline = deadline
    print(f"\n=== Phase 4: Exact Subgradient + Final ===")

    # Exact subgradient polish
    a_sg = exact_subgrad_polish(best_a, n_steps=200, rng=rng)
    sc = compute_score(a_sg)
    report(sc, a_sg, "subgrad")

    # Plateau + subgradient combo
    a_pl = plateau_distribute(a_sg, n_rounds=10)
    sc = compute_score(a_pl)
    report(sc, a_pl, "plateau2")

    # One more LP round if time
    if time.monotonic() < p4_deadline - 120:
        print("  Final LP round...")
        a_final = best_a.copy()
        S = float(a_final.sum())
        if S > 0:
            a_final = a_final / S
        a_out, sc_out, _ = lp_alternating(a_final, time_limit=p4_deadline - 60,
                                           max_outer=10, base_fista=20000)
        sc = compute_score(a_out)
        report(sc, a_out, "final-LP")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        remaining = deadline - time.monotonic()
        maxit = min(5000, int(remaining * 80))
        sc, a_opt = run_lbfgs(a_norm, 10000, maxiter=maxit, maxfun=maxit * 30)
        report(sc, a_opt, "final-LBFGS")

    # Final subgradient
    if time.monotonic() < deadline - 10:
        a_final = exact_subgrad_polish(best_a, n_steps=100, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final-sg")

    # ============ Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
